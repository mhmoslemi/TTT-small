    order = np.argsort(-h)
    h_bin = np.zeros(h.size, dtype=float)
    h_bin[order[:int(round(target))]] = 1.0
    return h_bin

def upsample_h(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    if n0 == n:
        return p.copy()
    if n0 <= 0:
        return np.full(n, 0.5)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.interp(x1, x0, p, left=p[0], right=p[-1])
    return project_to_sum(h, 0.5 * n)

def binary_local_search(h_bin, deadline, max_iter=80000, top_frac=0.08, min_k=8):
    n = h_bin.size
    target = int(np.sum(h_bin))
    best = h_bin.copy().astype(float)
    best_corr = np.correlate(best, 1.0 - best, mode="full").astype(float)
    best_c = float(np.max(best_corr)) * (2.0 / n)
    center = n - 1
    dx = 2.0 / n
    n_corr = len(best_corr)
    
    rng = np.random.default_rng(12345)
    stall = 0
    max_stall = 60
    
    for it in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        
        # Identify top-K worst lags
        k_count = max(min_k, int(np.ceil(n_corr * top_frac)))
        worst_indices = np.argpartition(best_corr, -k_count)[-k_count:]
        worst_vals = best_corr[worst_indices]
        threshold = worst_vals.min()
        
        # Score each position for flipping (1->0 or 0->1)
        # Effect on active lags
        pos_score_1to0 = np.zeros(n)
        pos_score_0to1 = np.zeros(n)
        
        for wi in worst_indices:
            v = int(wi) - center
            if v > 0:
                # corr[center+v] = sum_{i=0}^{n-1-v} h[i]*(1-h[i+v])
                # Flipping h[i] 1->0: removes (1-h[i+v]) from this corr
                # Flipping h[j] 0->1: adds (1-h[j+v]) to this corr
                for i in range(n - v):
                    if best[i] == 1 and best[i+v] == 0:
                        pos_score_1to0[i] += 1
                for j in range(n - v):
                    if best[j] == 0:
                        # Turning on h[j] adds (1-h[j+v]) if j+v < n
                        if j + v < n:
                            pos_score_0to1[j] += -1.0 * (1.0 - best[j+v])
                    # Also: turning on h[j] removes h[j-v] contribution at lag -v
            elif v < 0:
                v2 = -v
                # corr[center-v2] = sum_{i=v2}^{n-1} h[i-v2]*(1-h[i])
                for i in range(v2, n):
                    if best[i] == 1:
                        pos_score_0to1[i] += -1.0 * (best[i-v2])
                    if best[i-v2] == 1 and best[i] == 0:
                        pos_score_1to0[i-v2] += 1
            else:
                pass
        
        # Find good flip candidates: 1->0 that reduces worst lags, 0->1 that doesn't increase
        ones_mask = best == 1
        zeros_mask = best == 0
        good_ones = np.where(ones_mask & (pos_score_1to0 > 0))[0]
        good_zeros = np.where(zeros_mask & (pos_score_0to1 >= -0.5))[0]
        
        if len(good_ones) == 0:
            good_ones = np.where(ones_mask)[0]
        if len(good_zeros) == 0:
            good_zeros = np.where(zeros_mask)[0]
        
        # Try flips
        found = False
        n_try = 25
        for _ in range(n_try):
            if len(good_ones) > 0:
                a = good_ones[rng.integers(len(good_ones))]
            else:
                a = int(rng.integers(n))
                if best[a] != 1:
                    continue
            if len(good_zeros) > 0:
                b = good_zeros[rng.integers(len(good_zeros))]
            else:
                b = int(rng.integers(n))
                if best[b] != 0:
                    continue
            if a == b:
                continue
            
            best[a] = 0.0
            best[b] = 1.0
            new_corr = np.correlate(best, 1.0 - best, mode="full")
            new_max = float(np.max(new_corr))
            
            if new_max < float(np.max(best_corr)) - 1e-12:
                best_corr = new_corr
                best_c = new_max * dx
                found = True
                break
            else:
                best[b] = 0.0
                best[a] = 1.0
        
        if not found:
            # Try random pairs
            for _ in range(50):
                a = int(rng.integers(n))
                b = int(rng.integers(n))
                if a == b or best[a] == best[b]:
                    continue
                best[a], best[b] = best[b], best[a]
                new_corr = np.correlate(best, 1.0 - best, mode="full")
                if np.max(new_corr) < np.max(best_corr) - 1e-12:
                    best_corr = new_corr
                    best_c = float(np.max(best_corr)) * dx
                    found = True
                    break
                best[b], best[a] = best[a], best[b]
        
        if found:
            stall = 0
        else:
            stall += 1
            if stall >= max_stall:
                break
    
    return best, best_c

def continuous_polish(h, deadline, max_iter=2000):
    n = h.size
    target = 0.5 * n
    best = project_to_sum(h, target)
    best_c = evaluate_erdos(best)
    center = n - 1
    dx = 2.0 / n
    
    rng = np.random.default_rng(54321)
    stall = 0
    
    for it in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        
        corr = np.correlate(best, 1.0 - best, mode="full")
        k_count = max(8, int(np.ceil(len(corr) * 0.06)))
        top_idx = np.argsort(corr)[-k_count:]
        top_vals = corr[top_idx]
        dv = top_vals - top_vals.min()
        w = np.exp(dv / max(1e-10, dv.max() + 1e-10))
        w /= w.sum()
        
        g = np.zeros(n)
        for i, t in enumerate(top_idx):
            k = int(t) - center
            if k == 0:
                g += w[i] * (1.0 - 2.0 * best)
            else:
                lag = abs(k)
                if k > 0:
                    g[lag:] += w[i]
                    g[:n-lag] -= w[i] * best[lag:]
                    g[lag:] -= w[i] * best[:n-lag]
                else:
                    g[:n-lag] += w[i]
                    g[:n-lag] -= w[i] * best[lag:]
                    g[lag:] -= w[i] * best[:n-lag]
        
        d = -g
        d -= np.mean(d)
        norm = np.linalg.norm(d)
        if norm < 1e-14:
            break
        d /= norm
        
        improved = False
        for step in (0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.4):
            cand = project_to_sum(best + step * d, target)
            c = evaluate_erdos(cand)
            if c < best_c - 1e-13:
                best = cand
                best_c = c
                improved = True
                break
        
        if not improved:
            # Try random coordinate moves
            for _ in range(10):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    continue
                for d_val in (0.01, 0.05, 0.1, 0.2, -0.01, -0.05, -0.1, -0.2):
                    cand = best.copy()
                    cand[i] += d_val
                    cand[j] -= d_val
                    if cand[i] < -1e-10 or cand[i] > 1+1e-10 or cand[j] < -1e-10 or cand[j] > 1+1e-10:
                        continue
                    cand = project_to_sum(cand, target)
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-13:
                        best = cand
                        best_c = c
                        improved = True
                        break
                if improved:
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 30:
                break
    
    return project_to_sum(best, target), evaluate_erdos(best)

def pairwise_swap_continuous(h, deadline, max_iter=5000, top_frac=0.06):
    n = h.size
    target = 0.5 * n
    best = project_to_sum(h, target)
    best_c = evaluate_erdos(best)
    center = n - 1
    dx = 2.0 / n
    rng = np.random.default_rng(9999)
    stall = 0
    
    for it in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        
        corr = np.correlate(best, 1.0 - best, mode="full")
        k_count = max(6, int(np.ceil(len(corr) * top_frac)))
        top_idx = np.argsort(corr)[-k_count:]
        top_vals = corr[top_idx]
        dv = top_vals - top_vals.min()
        w = np.exp(dv / max(1e-10, dv.max() + 1e-10))
        w /= w.sum()
        
        g = np.zeros(n)
        for i, t in enumerate(top_idx):
            k = int(t) - center
            if k == 0:
                g += w[i] * (1.0 - 2.0 * best)
            else:
                lag = abs(k)
                if k > 0:
                    g[lag:] += w[i]
                    g[:n-lag] -= w[i] * best[lag:]
                    g[lag:] -= w[i] * best[:n-lag]
                else:
                    g[:n-lag] += w[i]
                    g[:n-lag] -= w[i] * best[lag:]
                    g[lag:] -= w[i] * best[:n-lag]
        
        neg_cand = np.argsort(g)[:12]
        pos_cand = np.argsort(-g)[:12]
        found = False
        
        for ii in neg_cand:
            ii = int(ii)
            for jj in pos_cand:
                jj = int(jj)
                if ii == jj:
                    continue
                cap = min(1.0 - best[ii], best[jj])
                if cap < 1e-12:
                    continue
                for d in (0.2*cap, 0.5*cap, 0.8*cap, cap):
                    cand = best.copy()
                    cand[ii] += d
                    cand[jj] -= d
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-13:
                        best = cand
                        best_c = c
                        found = True
                        break
                if found:
                    break
            if found:
                break
        
        if not found:
            stall += 1
            if stall > 40:
                # Random restart
                for _ in range(30):
                    i1, i2 = int(rng.integers(n)), int(rng.integers(n))
                    if i1 == i2:
                        continue
                    cap = min(1.0 - best[i1], best[i2])
                    if cap < 1e-12:
                        continue
                    d = rng.random() * cap
                    cand = best.copy()
                    cand[i1] += d
                    cand[i2] -= d
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-13:
                        best = cand
                        best_c = c
                        found = True
                        stall = 0
                        break
            if stall > 100:
                break
        else:
            stall = 0
    
    return project_to_sum(best, target), evaluate_erdos(best)

def make_m_sequence(n, m_val):
    """Generate an m-sequence pattern of length n"""
    # Simple LFSR-based construction
    poly = {4: 0b1101, 5: 0b11011, 7: 0b11000001, 8: 0b110001110, 9: 0b110010001, 10: 0b1100001001}
    period = (1 << m_val) - 1
    if period not in [63, 127, 255, 511]:
        m_val = 8
        period = 255
    # Generate m-sequence
    reg = (1 << m_val) - 1
    seq = []
    for _ in range(period):
        seq.append(reg & 1)
        fb = (reg & 1) ^ (reg >> 4 & 1) if m_val == 8 else (reg & 1) ^ (reg >> 2 & 1)
        reg = (reg >> 1) | (fb << (m_val - 1))
    # Tile to length n
    h = np.zeros(n)
    for i in range(n):
        h[i] = seq[i % period]
    return h

def make_golden_pattern(n, rng):
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    positions = set()
    n_ones = n // 2
    for k in range(1, n_ones + 1):
        pos = int((k * golden) % n)
        while pos in positions:
            pos = (pos + 1) % n
        positions.add(pos)
    h = np.zeros(n)
    for p in positions:
        h[p] = 1.0
    return h

def make_diverse_start(n, rng, mode):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    if mode == 0:
        h = 0.5 + 0.35 * np.sin(2 * np.pi * 3 * x / 2 + rng.uniform(0, 6.28))
        h += 0.12 * np.sin(2 * np.pi * 7 * x / 2 + rng.uniform(0, 6.28))
    elif mode == 1:
        h = np.zeros(n)
        m = 8 if n % 8 == 0 else 6
        period = n // m
        for i in range(m):
            h[i*period:(i+1)*period] = float(rng.uniform(0.2, 0.8))
    elif mode == 2:
        h = make_m_sequence(n, 8).astype(float)
        h += rng.normal(0, 0.08, n)
    else:
        h = make_golden_pattern(n, rng)
        h += rng.normal(0, 0.1, n)
    return project_to_sum(h, 0.5 * n)

def run(seed=42, budget_s=1300, **kwargs):
    start = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = start + max(1.0, budget - 15.0)
    rng = np.random.default_rng(seed)
    
    best_h = None
    best_c = float("inf")
    best_n = 0
    
    # Get parent
    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.clip(np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    except Exception:
        parent = None
    
    if parent is not None:
        np_ = parent.size
        pp = project_to_sum(parent, 0.5 * np_)
        cp = evaluate_erdos(pp)
        best_h = pp.copy()
        best_c = cp
        best_n = np_
    else:
        n0 = 450
        h0 = np.zeros(n0)
        for k in range(n0 // 2):
            h0[int((k * 1.618) % n0)] = 1.0
        best_h = project_to_sum(h0, 0.5 * n0)
        best_c = evaluate_erdos(best_h)
        best_n = n0
    
    # Strategy 1: Binary local search at parent resolution
    rem = deadline - time.perf_counter()
    n_parent = best_n
    h_bin = to_binary(best_h, 0.5 * n_parent)
    t1_deadline = start + max(10, (budget - 15) * 0.35)
    hb, cb = binary_local_search(h_bin, min(t1_deadline, deadline))
    if cb < best_c:
        best_h = hb.copy()
        best_c = cb
        best_n = n_parent
    
    # Strategy 2: Continuous polish from best binary
    rem = deadline - time.perf_counter()
    if rem > 20:
        t2_deadline = start + max(10, (budget - 15) * 0.55)
        h2, c2 = continuous_polish(best_h, min(t2_deadline, deadline))
        if c2 < best_c:
            best_h = h2.copy()
            best_c = c2
    
    # Strategy 3: Pairwise swap continuous
    rem = deadline - time.perf_counter()
    if rem > 20:
        t3_deadline = start + max(10, (budget - 15) * 0.75)
        h3, c3 = pairwise_swap_continuous(best_h, min(t3_deadline, deadline))
        if c3 < best_c:
            best_h = h3.copy()
            best_c = c3
    
    # Strategy 4: Try at a different resolution (n=300)
    rem = deadline - time.perf_counter()
    if rem > 60:
        n_alt = 300
        h_alt = upsample_h(best_h, n_alt)
        h_alt_bin = to_binary(h_alt, 0.5 * n_alt)
        t4_deadline = min(deadline, time.perf_counter() + rem * 0.4)
        ha, ca = binary_local_search(h_alt_bin, t4_deadline, max_iter=40000)
        if ca < best_c:
            best_h = ha.copy()
            best_c = ca
            best_n = n_alt
        rem = deadline - time.perf_counter()
        if rem > 15:
            h4, c4 = continuous_polish(best_h, deadline, max_iter=1000)
            if c4 < best_c:
                best_h = h4.copy()
                best_c = c4
    
    # Strategy 5: Try m-sequence based start at n=450
    rem = deadline - time.perf_counter()
    if rem > 40:
        n_ms = 450
        h_ms = make_m_sequence(n_ms, 8)
        h_ms = project_to_sum(h_ms, 0.5 * n_ms)
        h_ms_bin = to_binary(h_ms, 0.5 * n_ms)
        t5_deadline = min(deadline, time.perf_counter() + rem * 0.5)
        hm, cm = binary_local_search(h_ms_bin, t5_deadline, max_iter=30000)
        if cm < best_c:
            best_h = hm.copy()
            best_c = cm
            best_n = n_ms
    
    # Strategy 6: Try golden ratio pattern
    rem = deadline - time.perf_counter()
    if rem > 30:
        n_gr = 450
        h_gr = make_golden_pattern(n_gr, rng)
        h_gr = project_to_sum(h_gr, 0.5 * n_gr)
        h_gr_bin = to_binary(h_gr, 0.5 * n_gr)
        t6_deadline = min(deadline, time.perf_counter() + rem * 0.6)
        hg, cg = binary_local_search(h_gr_bin, t6_deadline, max_iter=25000)
        if cg < best_c:
            best_h = hg.copy()
            best_c = cg
            best_n = n_gr
    
    # Final intensive polish on the best solution
    rem = deadline - time.perf_counter()
    if best_h is not None and rem > 15:
        nb = int(best_h.size)
        hb_best = project_to_sum(best_h, 0.5 * nb)
        
        # Pairwise swap
        if rem > 20:
            h_ps, c_ps = pairwise_swap_continuous(hb_best, deadline, max_iter=3000, top_frac=0.05)
            if c_ps < best_c:
                best_h = h_ps
                best_c = c_ps
                best_n = nb
        
        # Continuous polish
        rem2 = deadline - time.perf_counter()
        if rem2 > 10:
            h_cp, c_cp = continuous_polish(best_h, deadline, max_iter=1500)
            if c_cp < best_c:
                best_h = h_cp
                best_c = c_cp
                best_n = nb
    
    # Final projection and evaluation
    if best_h is None:
        fallback = np.full(450, 0.5)
        return fallback, evaluate_erdos(fallback), 450
    
    final_h = project_to_sum(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    return final_h, float(final_c), int(final_h.size)
