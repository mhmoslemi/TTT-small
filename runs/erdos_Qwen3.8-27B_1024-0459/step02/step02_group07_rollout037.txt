The objective is f(h) = max_m (c_m(h)) where c_m(h) = ∑_n h[n+ell_m](1-h[n]).

Then project onto sum(h) = n/2.

With appropriate step size decay, this converges to a local optimum.

The key advantage: we only need to compute the subgradient from the ACTIVE constraints (the m values where c_m is close to the max). There are typically only a handful of active shifts.

```python
def subgradient_step(h, n, alpha, top_k=10):
    corr = get_corr(h)
    dx = 2.0 / n
    # Find active constraints
    vals = np.sort(corr)[::-1][:top_k]
    thresh = vals[-1]
    active = np.where(corr >= thresh - 1e-10)[0]
    
    # Compute subgradient (average of active gradients)
    grad = np.zeros(n)
    for m in active:
        ell = m - (n-1)
        # grad contribution for this ell:
        # d(corr[m])/d(h[j]) = (1-h[j-ell])*[0<=j-ell<n] - h[j+ell]*[0<=j+ell<n]
        # For all j: 
        # Term 1: (1-h[j-ell]) for j in [ell, ell+n-1] intersect [0,n-1]
        #   i.e., j from max(0,ell) to min(n-1, ell+n-1)
        #   grad[j] += (1-h[j-ell])
        # Term 2: -h[j+ell] for j in [-ell, n-1-ell] intersect [0,n-1]
        #   i.e., j from max(0,-ell) to min(n-1, n-1-ell)
        #   grad[j] += -h[j+ell]
        
        j1 = max(0, ell)
        j2 = min(n, n + ell)  # exclusive
        if j1 < j2:
            grad[j1:j2] += (1.0 - h[j1-ell:j2-ell])
        
        j3 = max(0, -ell)
        j4 = min(n, n - ell)  # exclusive
        if j3 < j4:
            grad[j3:j4] += -h[j3+ell:j4+ell]
    
    grad /= len(active)
    
    h_new = h - alpha * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = make_feasible(h_new, n)
    return h_new
```

This is O(k*n) per step where k is the number of active constraints. With k=10 and n=160, that's very fast.

The step size α is critical. A good choice is α_k = 1/sqrt(k) or Polyak step size.

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_src)
    return make_feasible(h_interp, n_dst)


def coord_descent(h, n, max_passes, time_limit):
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    corr = get_corr(h)
    t_start = time.time()
    ell_arr = np.arange(-(n - 1), n)
    
    for p in range(max_passes):
        if time.time() - t_start > time_limit:
            break
        for j in range(n):
            old_val = h[j]
            d = np.zeros(2 * n - 1)
            valid1 = (j - ell_arr >= 0) & (j - ell_arr < n)
            if np.any(valid1):
                d[valid1] += (1.0 - h[j - ell_arr[valid1]])
            valid2 = (j + ell_arr >= 0) & (j + ell_arr < n)
            if np.any(valid2):
                d[valid2] += -h[j + ell_arr[valid2]]
            
            s = corr - d * old_val
            lo_t = 0.0
            hi_t = 1.0
            for _ in range(60):
                mid = 0.5 * (lo_t + hi_t)
                vals = s + d * mid
                mx = np.max(vals)
                active = vals >= mx - 1e-14
                cnt_a = np.sum(active)
                if cnt_a == 0:
                    break
                sg = np.sum(d[active]) / cnt_a
                if sg > 1e-13:
                    hi_t = mid
                elif sg < -1e-13:
                    lo_t = mid
                else:
                    break
            t_opt = 0.5 * (lo_t + hi_t)
            h[j] = t_opt
            corr += d * (t_opt - old_val)
        
        h = make_feasible(h, n)
        corr = get_corr(h)
        c5 = np.max(corr) * dx
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5


def subgradient_optimize(h, n, time_limit, n_top=8):
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t_start = time.time()
    alpha = 0.15
    step = 0
    
    while time.time() - t_start < time_limit:
        corr = get_corr(h)
        thresh = np.sort(corr)[::-1][n_top - 1]
        active = np.where(corr >= thresh - 1e-12)[0]
        
        grad = np.zeros(n)
        for m in active:
            ell = int(m - (n - 1))
            j1 = max(0, ell)
            j2 = min(n, n + ell)
            if j1 < j2:
                grad[j1:j2] += (1.0 - h[j1 - ell:j2 - ell])
            j3 = max(0, -ell)
            j4 = min(n, n - ell)
            if j3 < j4:
                grad[j3:j4] += -h[j3 + ell:j4 + ell]
        
        grad /= max(len(active), 1)
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = make_feasible(h_new, n)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
        step += 1
        alpha = 0.15 / (1.0 + 0.01 * step)
        
        if step % 50 == 0 and time.time() - t_start < time_limit - 15:
            h_cd, c5_cd = coord_descent(h, n, 20, 10)
            if c5_cd < best_c5 - 1e-14:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                h = h_cd.copy()
    
    return best_h, best_c5


def binary_local_search(h_init, n, time_limit, seed_val=42):
    rng = np.random.RandomState(seed_val)
    h_bin = (h_init >= 0.5).astype(float)
    s = np.sum(h_bin)
    deficit = int(round(n / 2.0 - s))
    if deficit > 0:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h_init[zeros])[:deficit]
        h_bin[zeros[order]] = 1.0
    elif deficit < 0:
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h_init[ones])[:(-deficit)]
        h_bin[ones[order]] = 0.0
    
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    t_start = time.time()
    T = 0.05
    cooling = 0.9997
    no_improve = 0
    
    while time.time() - t_start < time_limit:
        if no_improve > 5000:
            T = max(T * 0.5, 0.001)
            no_improve = 0
        
        move_type = rng.randint(0, 3)
        if move_type == 0:
            # Single swap: flip a 1 to 0 and a 0 to 1
            ones = np.where(best_h == 1)[0]
            zeros = np.where(best_h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                continue
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h_try = best_h.copy()
            h_try[i] = 0.0
            h_try[j] = 1.0
        elif move_type == 1:
            # Double swap: two 1s and two 0s
            ones = np.where(best_h == 1)[0]
            zeros = np.where(best_h == 0)[0]
            if len(ones) < 2 or len(zeros) < 2:
                continue
            i1, i2 = rng.choice(ones, 2, replace=False)
            j1, j2 = rng.choice(zeros, 2, replace=False)
            h_try = best_h.copy()
            h_try[i1] = 0.0
            h_try[i2] = 0.0
            h_try[j1] = 1.0
            h_try[j2] = 1.0
        else:
            # Block move: shift a block of consecutive same values
            i = rng.randint(0, n)
            h_try = best_h.copy()
            # Find a run of same value starting at i
            val = best_h[i]
            run_start = i
            while run_start > 0 and best_h[run_start - 1] == val:
                run_start -= 1
            run_end = i
            while run_end < n - 1 and best_h[run_end + 1] == val:
                run_end += 1
            run_len = run_end - run_start + 1
            if run_len == 0 or run_len >= n:
                continue
            shift = rng.choice([1, 2, -1, -2])
            new_start = run_start + shift
            new_end = run_end + shift
            if new_start < 0 or new_end >= n:
                continue
            # Check if target positions have opposite value
            ok = True
            for k in range(new_start, new_end + 1):
                if best_h[k] == val:
                    ok = False
                    break
            if not ok:
                continue
            h_try[run_start:run_end + 1] = 1.0 - val
            h_try[new_start:new_end + 1] = val
        
        if abs(np.sum(h_try) - n / 2.0) < 1e-10:
            c5_try = compute_c5(h_try)
            delta = c5_try - best_c5
            if delta < -1e-14:
                best_c5 = c5_try
                best_h = h_try.copy()
                no_improve = 0
            else:
                if T > 1e-6 and rng.random() < np.exp(-delta / T):
                    best_h = h_try.copy()
                no_improve += 1
                T *= cooling
    
    return best_h, best_c5


def slsqp_polish(h, n, max_iter=5000):
    def obj_func(x):
        return np.max(np.correlate(x, 1.0 - x, mode='full'))
    
    def con_fun(x):
        return np.sum(x) - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun}]
    try:
        res = minimize(obj_func, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-16})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None


def block_coord_descent(h, n, block_size, max_passes, time_limit):
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    corr = get_corr(h)
    t_start = time.time()
    
    for p in range(max_passes):
        if time.time() - t_start > time_limit:
            break
        for j0 in range(0, n, block_size):
            j1 = min(j0 + block_size, n)
            blk = h[j0:j1].copy()
            # For each shift, compute the linear coefficient on the block
            # corr[m] = sum_j h[j]*(1-h[j-ell]) 
            # For j in block: contribution is h[j]*(1-h[j-ell]) + h[j+ell]*(1-h[j])
            # The block variables affect:
            # - As h[j] (value): coeff is (1-h[j-ell])
            # - As (1-h[j]): coeff is -h[j+ell]
            # Net linear coeff for h[j0+i]: (1-h[j0+i-ell]) - h[j0+i+ell]
            # Plus the constant part from other j's
            
            # For each block variable, compute its "constant" contribution to corr
            # and its linear coefficient
            # This is complex for blocks > 1 due to cross terms within block
            # For block_size=2, handle the cross term explicitly
            
            if block_size == 2:
                j_a, j_b = j0, j0 + 1
                # corr[m] = S_m + d_a[m]*t_a + d_b[m]*t_b + cross*m * t_a*t_b
                # cross only for ell where j_b - j_a = ell or j_a - j_b = ell
                # i.e., ell = 1 or ell = -1
                
                d_a = np.zeros(2 * n - 1)
                d_b = np.zeros(2 * n - 1)
                cross = np.zeros(2 * n - 1)
                S = corr.copy()
                
                for m in range(2 * n - 1):
                    ell = m - (n - 1)
                    # Remove j_a and j_b contributions from S
                    if 0 <= j_a - ell < n:
                        S[m] -= h[j_a] * (1.0 - h[j_a - ell])
                    if 0 <= j_a + ell < n:
                        S[m] -= h[j_a + ell] * (1.0 - h[j_a])
                    if 0 <= j_b - ell < n:
                        S[m] -= h[j_b] * (1.0 - h[j_b - ell])
                    if 0 <= j_b + ell < n:
                        S[m] -= h[j_b + ell] * (1.0 - h[j_b])
                    
                    # d_a[m] = coefficient of t_a in S_a + d_a*t_a + d_b*t_b
                    if 0 <= j_a - ell < n:
                        d_a[m] += (1.0 - h[j_a - ell])
                    if 0 <= j_a + ell < n and j_a + ell != j_b:
                        d_a[m] += -h[j_a + ell]
                    # Cross term: if j_b = j_a + ell or j_a = j_b + ell
                    if ell == 1 and j_b - ell == j_a:
                        d_a[m] += -1.0
                        cross[m] = 1.0
                    if ell == -1 and j_a - ell == j_b:
                        d_a[m] += -1.0
                        cross[m] = 1.0
                    
                    if 0 <= j_b - ell < n:
                        d_b[m] += (1.0 - h[j_b - ell])
                    if 0 <= j_b + ell < n and j_b + ell != j_a:
                        d_b[m] += -h[j_b + ell]
                    if ell == -1 and j_a - ell == j_b:
                        pass  # already handled
                    if ell == 1 and j_b - ell == j_a:
                        d_b[m] += -1.0
                        pass
                
                # Now minimize max_m (S[m] + d_a[m]*t_a + d_b[m]*t_b + cross[m]*t_a*t_b)
                # over 0<=t_a<=1, 0<=t_b<=1
                # Grid search (coarse then fine)
                best_t = (h[j_a], h[j_b])
                best_val = np.inf
                for _ in range(40):
                    ta_grid = np.linspace(0, 1, 9)
                    tb_grid = np.linspace(0, 1, 9)
                    found = False
                    for ta in ta_grid:
                        for tb in tb_grid:
                            vals = S + d_a * ta + d_b * tb + cross * ta * tb
                            mx = np.max(vals)
                            if mx < best_val - 1e-12:
                                best_val = mx
                                best_t = (ta, tb)
                                found = True
                    if found:
                        # Refine around best
                        ta_c, tb_c = best_t
                        ta_grid = np.linspace(max(0, ta_c - 0.125), min(1, ta_c + 0.125), 5)
                        tb_grid = np.linspace(max(0, tb_c - 0.125), min(1, tb_c + 0.125), 5)
                    else:
                        break
                
                h[j_a] = best_t[0]
                h[j_b] = best_t[1]
                corr = get_corr(h)
                
                c5 = np.max(corr) * dx
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    best_h = h.copy()
            else:
                # For larger blocks, just do sequential single updates
                for j in range(j0, j1):
                    old_val = h[j]
                    d = np.zeros(2 * n - 1)
                    ell_arr = np.arange(-(n - 1), n)
                    valid1 = (j - ell_arr >= 0) & (j - ell_arr < n)
                    if np.any(valid1):
                        d[valid1] += (1.0 - h[j - ell_arr[valid1]])
                    valid2 = (j + ell_arr >= 0) & (j + ell_arr < n)
                    if np.any(valid2):

                    s_vec = corr - d * old_val

                        mx = np.max(vals)
                        active = vals >= mx - 1e-14
                        seg = np.sum(d[active]) / max(np.sum(active), 1)
                        if seg > 1e-13:
                            hi_t = mid
                        elif seg < -1e-13:
                            lo_t = mid
                        else:
                            break
                    t_opt = 0.5 * (lo_t + hi_t)
                    h[j] = t_opt
                    corr += d * (t_opt - old_val)
        
        h = make_feasible(h, n)
        corr = get_corr(h)
        c5 = np.max(corr) * dx
        if c5 < best_c5 - 1e-14:

    return best_h, best_c5

            best_c5 = c5_slq
            best_h = h_slq.copy()
    
    # Phase 7: Final coordinate descent
    if remaining() > 30:
        tl = max(10, remaining() - 15)
        best_h, best_c5 = coord_descent(best_h, best_n, 50000, tl)
    
    # Phase 8: One more SLSQP
    if remaining() > 30:
        h_slq, c5_slq = slsqp_polish(best_h, best_n, max_iter=5000)
        if h_slq is not None and c5_slq < best_c5 - 1e-14:
            best_c5 = c5_slq
            best_h = h_slq.copy()
    
    # Phase 9: Quick binary search on final
    if remaining() > 20:
        bl_tl = min(30, remaining() - 10)
        h_bin, c5_bin = binary_local_search(best_h, best_n, bl_tl, seed_val=seed + 100)
        if c5_bin < best_c5 - 1e-14:
            best_c5 = c5_bin
            best_h = h_bin.copy()
    
    # Final feasibility check
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, best_n)
