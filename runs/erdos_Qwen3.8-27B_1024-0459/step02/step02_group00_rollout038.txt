```python
def smooth_obj_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500.0, 0.0)
    exps = np.exp(shifted)
    total = np.sum(exps)
    f = (m + np.log(total) / beta) * dx
    
    w = exps / total  # length 2n-1
    g = 1.0 - 2.0 * h  # length n
    conv = np.convolve(w, g, mode='full')  # length 3n-2
    grad = dx * conv[n-1:n-1+n]
    
    return f, grad
```
