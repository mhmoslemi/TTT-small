import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    total = np.sum(np.exp(shifted))
    return m + np.log(total) / beta

def smooth_max_weights(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    exps = np.exp(shifted)
    return exps / np.sum(exps)

def obj_and_grad(x, beta, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    sm = smooth_max(corr, beta)
    weights = smooth_max_weights(corr, beta)
    
    # Sum constraint penalty
    s = np.sum(x) - n / 2.0
    obj = sm * dx + penalty * s * s
    
    # Gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:
            continue
        L = k - (n - 1)
        if L == 0:
            grad += w * (1.0 - 2.0 * x)
        elif L > 0:
            grad[:n-L] += w * (1.0 - x[L:])
            grad[L:] -= w * x[:n-L]
        else:
            Lp = -L
            grad[:n-Lp] += w * (1.0 - x[Lp:])
            grad[Lp:] -= w * x[:n-Lp]
    grad *= dx
    # Add penalty gradient
    grad += 2.0 * penalty * s * np.ones(n)
    
    return obj, grad

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmin(np.maximum(h, 1.0 - h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def smooth_max(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    exps = np.exp(shifted)
    return exps / np.sum(exps)

    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')

def grad_func(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    weights = smooth_max_weights(corr, beta)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        L = k - (n - 1)
        if L == 0:
            grad += w * (1.0 - 2.0 * x)
        elif L > 0:
            grad[:n-L] += w * (1.0 - x[L:])
            grad[L:] -= w * x[:n-L]
        else:
            Lp = -L
            grad[:n-Lp] += w * (1.0 - x[Lp:])
            grad[Lp:] -= w * x[:n-Lp]
    return dx * grad

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmin(np.maximum(h, 1.0 - h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def run_slsqp(h0, n, beta, max_iter):

        return obj_func(x, beta)
    def jac(x):
        return grad_func(x, beta)
    def con(x):
        return np.sum(x) - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con}]
    try:
        res = minimize(fun, h0, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-15, 'iprint': 0})
        h_out = project_feasible(res.x, n)
        return h_out, compute_c5(h_out)
    except Exception:
        return h0, compute_c5(h0)

def run_lbfgs(h0, n, beta, max_iter, penalty):

        f = rng.randint(1, 8)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.05, 0.25)
        h += a * np.sin(2 * np.pi * f * x + p)
    h = np.clip(h, 0, 1)
    return project_feasible(h, n)

def run_slsqp(h0, n, beta, max_iter):
    def fun(x):
        return obj_func(x, beta)
    def jac(x):
        return grad_func(x, beta)
    def con(x):
        return np.sum(x) - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con}]
    try:
        res = minimize(fun, h0, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-16, 'iprint': 0})
        h_out = project_feasible(res.x, n)
        return h_out, compute_c5(h_out)
    except Exception:
        return h0, compute_c5(h0)

def run_lbfgs(h0, n, beta, max_iter, penalty):
    def fun(x):
        c5_v = obj_func(x, beta)
        s = np.sum(x) - n / 2.0
        return c5_v + penalty * s * s
    def jac(x):
        g = grad_func(x, beta)
        s = np.sum(x) - n / 2.0
        return g + 2.0 * penalty * s
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h0, jac=jac, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-16,
                                'gtol': 1e-14, 'maxfun': max_iter * 15, 'iprint': 0})
        h_out = project_feasible(res.x, n)
        return h_out, compute_c5(h_out)
    except Exception:
        return h0, compute_c5(h0)

def binary_round(h, n):
    h_b = np.zeros(n)
    sorted_idx = np.argsort(-h)
    h_b[sorted_idx[:n // 2]] = 1.0
    return h_b

def binary_local_search(h, n, max_rounds, rng, target_c5):
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    if best_c5 < target_c5:
        return h, best_c5

    no_improve = 0
    for rnd in range(max_rounds):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = float(np.max(corr))

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        improved = False
        n_tries = 100
        for _ in range(n_tries):
            i1 = ones[rng.randint(len(ones))]
            i0 = zeros[rng.randint(len(zeros))]
            h[i1] = 0.0
            h[i0] = 1.0
            new_corr = np.correlate(h, 1.0 - h, mode='full')
            new_max = float(np.max(new_corr))
            if new_max < max_corr - 1e-12:
                max_corr = new_max
                best_c5 = new_max * dx
                corr = new_corr
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                improved = True
                no_improve = 0
            else:
                h[i1] = 1.0
                h[i0] = 0.0

        if not improved:
            no_improve += 1
            if no_improve >= 5:
                # Try triple flips
                triple_improved = False
                for _ in range(30):
                    try:
                        i1s = ones[rng.randint(len(ones), size=2)]
                        i0s = zeros[rng.randint(len(zeros), size=2)]
                    except Exception:
                        break
                    h[i1s[0]] = 0.0
                    h[i1s[1]] = 0.0
                    h[i0s[0]] = 1.0
                    h[i0s[1]] = 1.0
                    new_corr = np.correlate(h, 1.0 - h, mode='full')
                    new_max = float(np.max(new_corr))
                    if new_max < max_corr - 1e-12:
                        max_corr = new_max
                        best_c5 = new_max * dx
                        corr = new_corr
                        ones = np.where(h == 1.0)[0]
                        zeros = np.where(h == 0.0)[0]
                        improved = True
                        triple_improved = True
                        no_improve = 0
                    else:
                        h[i1s[0]] = 1.0
                        h[i1s[1]] = 1.0
                        h[i0s[0]] = 0.0
                        h[i0s[1]] = 0.0
                if not triple_improved:
                    no_improve += 3

        if best_c5 <= target_c5 * (1 + 1e-8):
            break

    return h, best_c5

def targeted_binary_search(h, n, max_rounds, rng, target_c5):
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    if best_c5 < target_c5:
        return h, best_c5

    for rnd in range(max_rounds):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = float(np.max(corr))
        max_k = int(np.argmax(corr))
        L_star = max_k - (n - 1)

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        improved = False
        alphas = np.zeros(n)
        betas = np.zeros(n)

        if L_star == 0:
            for i in range(n):
                alphas[i] = 1.0 - 2.0 * h[i]
        elif L_star > 0:
            L = L_star
            for i in range(n - L):
                alphas[i] += (1.0 - h[i + L])
                betas[i + L] += h[i]
            for i in range(L, n):
                alphas[i] -= h[i - L]
        else:
            L = -L_star
            for i in range(n - L):
                alphas[i + L] += (1.0 - h[i])
                betas[i] += h[i + L]
            for i in range(L, n):
                alphas[i] -= h[i - L]

        # Find good 1->0 flips (reduce max_corr lag)
        good_ones = []
        for i in ones:
            if L_star > 0:
                effect = (1.0 - h[i + L_star]) if (i < n - L_star) else 0.0
                effect += h[i - L_star] if (i >= L_star) else 0.0
            elif L_star < 0:
                Lp = -L_star
                effect = (1.0 - h[i - Lp]) if (i >= Lp) else 0.0
                effect += h[i + Lp] if (i < n - Lp) else 0.0
            else:
                effect = 2.0 * h[i] - 1.0
            if effect > 0.01:
                good_ones.append((i, effect))

        if len(good_ones) > 100:
            good_ones = good_ones[rng.randint(0, len(good_ones), size=100)]

        rng.shuffle(good_ones)
        for i, eff in good_ones[:30]:
            for _ in range(5):
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0.0
                h[j] = 1.0
                new_corr = np.correlate(h, 1.0 - h, mode='full')
                new_max = float(np.max(new_corr))
                if new_max < max_corr - 1e-12:
                    max_corr = new_max
                    best_c5 = new_max * dx
                    ones = np.where(h == 1.0)[0]
                    zeros = np.where(h == 0.0)[0]
                    improved = True
                    break
                else:
                    h[i] = 1.0
                    h[j] = 0.0
            if improved:
                break

        if not improved:
            # Perturb and try again
            n_pert = rng.randint(2, 5)
            pert_ones = ones[rng.randint(0, len(ones), size=min(n_pert, len(ones)))]
            pert_zeros = zeros[rng.randint(0, len(zeros), size=min(n_pert, len(zeros)))]
            h[pert_ones] = 0.0
            h[pert_zeros] = 1.0
            new_c5 = compute_c5(h)
            if new_c5 < best_c5:
                best_c5 = new_c5
            else:
                h[pert_ones] = 1.0
                h[pert_zeros] = 0.0
                break

    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - start_time

    def remaining():
        return max(0.0, budget_s - elapsed() - 5.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        if parent_n > 0:
            parent_h = project_feasible(parent_h, parent_n)
            parent_c5 = compute_c5(parent_h)
            best_h = parent_h.copy()
            best_c5 = parent_c5
            best_n = parent_n
            print("Parent: n={}, C5={:.6f}".format(parent_n, parent_c5))
    except Exception:
        pass

    if best_h is None:
        best_n = 200
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_c5 = compute_c5(best_h)
        print("Default: n={}, C5={:.6f}".format(best_n, best_c5))

    # Phase 1: Continuous optimization at multiple resolutions
    n_list = sorted(set([best_n, 200, 256, 320, 400]))
    n_list = [nn for nn in n_list if 80 <= nn <= 400 and nn % 2 == 0]

    for n in n_list:
        if remaining() < 50:
            break
        print("=== Continuous opt: n={}, remaining {:.0f}s ===".format(n, remaining()))

        inits = []
        if parent_h is not None:
            inits.append(interp_h(parent_h, n))
        inits.append(interp_h(best_h, n))
        inits.append(fourier_init(n, rng, n_modes=5))
        inits.append(fourier_init(n, rng, n_modes=8))

        for h0 in inits[:3]:
            if remaining() < 30:
                break
            h0 = project_feasible(h0, n)
            cur = h0.copy()

            if n <= 256:
                beta_schedule = [15, 40, 100, 250, 600, 1200]
                iters_schedule = [300, 300, 250, 200, 150, 150]
            else:
                beta_schedule = [15, 40, 100, 250, 600]
                iters_schedule = [200, 200, 180, 150, 120]

            for bi, bv in enumerate(beta_schedule):
                if remaining() < 25:
                    break
                mi = iters_schedule[bi] if bi < len(iters_schedule) else 150
                if n <= 256:
                    h_new, c5_new = run_slsqp(cur, n, bv, mi)
                else:
                    h_new, c5_new = run_lbfgs(cur, n, bv, mi, penalty=1e4)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n
                    print("  n={}, beta={}, C5={:.6f}".format(n, bv, best_c5))
                cur = h_new

    # Phase 2: Try a few more continuous optimizations from diverse starts
    for n in [best_n, 200, 256]:
        if remaining() < 40:
            break
        h_f = fourier_init(n, rng, n_modes=10)
        for bv in [50, 200, 500]:
            if remaining() < 25:
                break
            h_new, c5_new = run_slsqp(h_f, n, bv, 250)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n
                print("  Fourier n={}, beta={}, C5={:.6f}".format(n, bv, best_c5))

    # Phase 3: Binary construction and local search
    print("=== Binary search phase, remaining {:.0f}s ===".format(remaining()))

    # Try multiple n values for binary
    binary_ns = sorted(set([best_n, 200, 256, 320]))
    binary_ns = [nn for nn in binary_ns if 80 <= nn <= 320 and nn % 2 == 0]

    for n_b in binary_ns:
        if remaining() < 60:
            break

        h_cont = interp_h(best_h, n_b)
        h_bin = binary_round(h_cont, n_b)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            best_n = n_b
            print("  Binary n={}, C5={:.6f}".format(n_b, best_c5))

        # Local search
        h_ls, c5_ls = binary_local_search(h_bin, n_b, max_rounds=500, rng=rng, target_c5=best_c5)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
            best_n = n_b
            print("  LocalSearch n={}, C5={:.6f}".format(n_b, best_c5))

        # Targeted search
        if remaining() > 40:
            h_ts, c5_ts = targeted_binary_search(h_ls, n_b, max_rounds=200, rng=rng, target_c5=best_c5)
            if c5_ts < best_c5:
                best_c5 = c5_ts
                best_h = h_ts.copy()
                best_n = n_b
                print("  Targeted n={}, C5={:.6f}".format(n_b, best_c5))

    # Phase 4: Extended binary search on best solution
    if remaining() > 30:
        n_final = best_n
        h_final_bin = binary_round(best_h, n_final)
        c5_fb = compute_c5(h_final_bin)
        if c5_fb < best_c5:
            best_c5 = c5_fb
            best_h = h_final_bin.copy()

        # Long local search
        rounds = min(2000, int(remaining() * 50))
        h_long, c5_long = binary_local_search(best_h, n_final, max_rounds=rounds, rng=rng, target_c5=best_c5)
        if c5_long < best_c5:
            best_c5 = c5_long
            best_h = h_long.copy()
            print("  Long local search: C5={:.6f}".format(best_c5))

    # Phase 5: Final polish - try small continuous perturbations of binary
    if remaining() > 30 and np.all((best_h == 0) | (best_h == 1)):
        n_p = best_n
        for bv in [500, 1000]:
            if remaining() < 20:
                break
            h_c = best_h.astype(float)
            # Small random perturbation
            h_c += rng.uniform(-0.01, 0.01, n_p)
            h_c = project_feasible(h_c, n_p)
            h_new, c5_new = run_slsqp(h_c, n_p, bv, 200)
            # Round back
            h_round = binary_round(h_new, n_p)
            c5_round = compute_c5(h_round)
            if c5_round < best_c5:
                best_c5 = c5_round
                best_h = h_round.copy()
                print("  Polish round: C5={:.6f}".format(best_c5))
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                print("  Polish cont: C5={:.6f}".format(best_c5))

    # Final: ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
