import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def thue_morse_init(n):
    tm = np.zeros(n)
    for i in range(n):
        tm[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    return make_feasible(tm, n)

def quadratic_residue_init(n, p=None):
    """Use quadratic residues mod p for a structured binary sequence."""
    if p is None:
        p = n  # should be prime for interesting structure
    h = np.zeros(n)
    for i in range(n):
        # Check if i is a quadratic residue mod p
        # For simplicity, use the Legendre symbol pattern
        if i == 0:
            h[i] = 0.0
        else:
            # Check if i^((p-1)/2) ≡ 1 mod p (for prime p)
            try:
                if p > 1 and p % 2 == 1:
                    val = pow(i, (p-1)//2, p)
                    h[i] = 1.0 if val == 1 else 0.0
                else:
                    h[i] = 1.0 if (i * i) % p < p // 2 else 0.0
            except:
                h[i] = 0.5
    return make_feasible(h, n)

def pattern_init(n, period, pattern):
    """Create a periodic pattern."""
    h = np.zeros(n)
    for i in range(n):
        h[i] = pattern[i % period]
    return make_feasible(h, n)

def simulated_annealing(h_init, n, rng, timeout=100.0, t0=1.0, t_end=0.01):
    h = np.round(h_init).astype(float)
    h = make_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    ones = set(np.where(h == 1.0)[0].tolist())
    zeros = set(np.where(h == 0.0)[0].tolist())
    
    start_t = time.time()
    n_steps = 0
    no_improve = 0
    max_no_improve = 20000
    
    while time.time() - start_t < timeout:
        # Cool down
        elapsed = time.time() - start_t
        progress = min(1.0, elapsed / timeout)
        T = t0 * (t_end / t0) ** progress
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Random swap
        i = list(ones)[rng.randint(len(ones))]
        j = list(zeros)[rng.randint(len(zeros))]
        if i == j:
            continue
        
        h[i] = 0.0
        h[j] = 1.0
        c5_try = compute_c5(h)
        delta = c5_try - best_c5
        
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            # Accept
            ones.discard(i)
            zeros.add(i)
            zeros.discard(j)
            ones.add(j)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            n_steps += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
        
        if no_improve > max_no_improve:
            break
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break

def random_binary(n, rng):

    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]

        excess = len(ones) - target

        h[to_flip] = 0.0
    elif len(ones) < target:
        deficit = target - len(ones)
        to_flip = rng.choice(zeros, deficit, replace=False)
        h[to_flip] = 1.0
    return h

def sa_search(h_init, n, rng, timeout=100.0, t0=2.0, t_min=0.001):
    """Simulated annealing with 1-swap moves."""
    h = np.round(h_init).astype(float).copy()
    h = make_feasible(h, n)
    dx = 2.0 / n
    
    corr_max_best = int(np.max(np.correlate(h, 1.0 - h, mode='full')))
    best_c5 = corr_max_best * dx
    best_h = h.copy()
    
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    pos2oi = {p: i for i, p in enumerate(ones)}
    pos2zi = {p: i for i, p in enumerate(zeros)}
    
    start = time.time()
    T = t0
    # Cooling: T from t0 to t_min over timeout
    # Number of expected iterations: ~timeout / 20e-6 = timeout * 50000
    n_expected = int(timeout * 40000)
    cool_rate = (t_min / t0) ** (1.0 / max(1, n_expected))
    
    n_total = 0
    no_improve = 0
    
    while time.time() - start < timeout:
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        
        new_max = int(np.max(np.correlate(h, 1.0 - h, mode='full')))
        
        delta = new_max - corr_max_best
        if delta <= 0 or rng.rand() < np.exp(-delta / T):
            # Accept
            oi = pos2oi[i]
            zi = pos2zi[j]
            ones[oi] = j
            pos2oi[j] = oi
            del pos2oi[i]
            pos2zi[i] = zi
            zeros[zi] = i
            del pos2zi[j]
            
            if new_max < corr_max_best:
                corr_max_best = new_max
                best_c5 = new_max * dx
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            n_total += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
            n_total += 1
        
        T = max(t_min, T * cool_rate)
        
        # Early stop if very stuck
        if no_improve > 30000:
            break
    
    return best_h, best_c5
