import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    # Clip and adjust sum
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-9
        else:
            mask = h > 1e-9
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_sa(h_init, n, rng, iterations, t_start, t_end, timeout):

    h = make_feasible(h, n)
    c5 = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()
    ones = set(np.where(h == 1.0)[0].tolist())
    zeros = set(np.where(h == 0.0)[0].tolist())

        if it % 10000 == 0 and len(ones) == 0 or len(zeros) == 0:
            break
        
        T = t_start * (t_end / t_start) ** (it / max(iterations - 1, 1))

        j = rng.choice(tuple(zeros)) if zeros else None
        if i is None or j is None or i == j:

        h[i], h[j] = 0.0, 1.0

    best_h = h.copy()

    ones_arr = np.where(h == 1.0)[0]
    zeros_arr = np.where(h == 0.0)[0]
    start_t = time.time()

    for it in range(n_iter):
        if time.time() - start_t > timeout:
            break
        if len(ones_arr) < 2 or len(zeros_arr) < 2:
            break

        T = t_start * (t_end / t_start) ** (it / max(n_iter - 1, 1))

        i1 = int(ones_arr[rng.randint(len(ones_arr))])
        i2 = int(ones_arr[rng.randint(len(ones_arr))])
        j1 = int(zeros_arr[rng.randint(len(zeros_arr))])
        j2 = int(zeros_arr[rng.randint(len(zeros_arr))])
        if len({i1, i2, j1, j2}) < 4:
            continue

        h[i1] = 0.0
        h[i2] = 0.0
        h[j1] = 1.0
        h[j2] = 1.0
        c5_new = compute_c5(h)
        delta = c5_new - c5

        if delta < 0 or (T > 1e-14 and rng.rand() < np.exp(-delta / T)):
            c5 = c5_new
            ones_arr = np.sort(np.concatenate([
                ones_arr[ones_arr != i1], ones_arr[ones_arr != i2], [j1, j2]
            ]))
            zeros_arr = np.sort(np.concatenate([
                zeros_arr[zeros_arr != j1], zeros_arr[zeros_arr != j2], [i1, i2]
            ]))
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i1] = 1.0
            h[i2] = 1.0
            h[j1] = 0.0
            h[j2] = 0.0

    return best_h, best_c5

def _sm_obj(x, beta):
    n = len(x)
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    shifted = np.clip(shifted, -500, 50)
    val = cmax + (1.0 / beta) * np.log(np.sum(np.exp(shifted)))
    return val

def slsqp_smooth_max(h_init, n, timeout, betas=(20, 50, 120, 300, 600)):
    global _N_G
    _N_G = n
    best_h = make_feasible(h_init.copy(), n)
    best_c5 = compute_c5(best_h)
    bounds = [(0.0, 1.0)] * n
    start_t = time.time()

    for beta in betas:
        if time.time() - start_t > timeout - 10:
            break
        max_iter = 500
        con_fun = lambda x: np.sum(x) - n / 2.0
        con_jac = lambda x: np.ones(n)
        con = {'type': 'eq', 'fun': con_fun, 'jac': con_jac}
        try:
            res = minimize(_sm_obj, best_h.copy(), args=(beta,), method='SLSQP',
                          bounds=bounds, constraints=con,
                          options={'maxiter': max_iter, 'ftol': 1e-12})
            if res.fun < 1e10:
                cand = make_feasible(res.x, n)
                c5v = compute_c5(cand)
                if c5v < best_c5 - 1e-14:
                    best_c5 = c5v
                    best_h = cand
        except Exception:
            pass

    return best_h, best_c5

def greedy_hill_climb(h_init, n, rng, max_iters, timeout):
    h = np.round(h_init).astype(float)
    h = make_feasible(h, n)
    c5 = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()
    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])
    start_t = time.time()
    stall = 0
    stall_limit = 20000

    for it in range(max_iters):
        if time.time() - start_t > timeout:
            break
        if stall > stall_limit:
            break
        if not ones_list or not zeros_list:
            break

        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]
        if i == j:
            continue

        h[i] = 0.0
        h[j] = 1.0
        c5_new = compute_c5(h)
        if c5_new < c5 - 1e-14:
            c5 = c5_new
            pos_i = ones_list.index(i)
            pos_j = zeros_list.index(j)
            ones_list[pos_i] = j
            zeros_list[pos_j] = i
            stall = 0
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
            stall += 1

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _START_T
    _START_T = time.time()
    np.random.seed(seed)
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - _START_T

    def remaining():
        return budget_s - elapsed()

    n = 256

    h_parent = None
    try:
        h_parent_arr = np.array(initial_h_values, dtype=float)
        if h_parent_arr is not None and len(h_parent_arr) > 0:
            h_parent = h_parent_arr.copy()
    except Exception:
        h_parent = None

    # Generate diverse initializations
    inits = []
    if h_parent is not None:
        hp_n = make_feasible(interp_to_n(h_parent, n), n)
        inits.append(hp_n)
    inits.append(thue_morse_init(n))
    for _ in range(3):
        inits.append(random_binary_init(n, rng))
    for _ in range(2):
        inits.append(block_init(n, rng))
    inits.append(m_seq_init(n, rng))
    # Phase-shifted Thue-Morse
    for sh in [1, 3, 7]:
        tm = np.zeros(n)
        for i in range(n):
            tm[i] = 1.0 if bin((i + sh) % n).count('1') % 2 == 0 else 0.0
        inits.append(make_feasible(tm, n))

    best_h = inits[0]
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick greedy from each init to get good starting points
    for h0 in inits:
        if remaining() < 30:
            break
        h_g, c5_g = greedy_hill_climb(h0, n, rng, max_iters=50000, timeout=remaining() * 0.05)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()

    # Phase 2: SA from best (long run)
    if remaining() > 120:
        t0 = remaining()
        h_sa, c5_sa = sa_single_swap(best_h, n, rng,
                                     n_iter=8000000,
                                     t_start=0.025, t_end=0.00005,
                                     timeout=min(t0 * 0.45, 400),
                                     guided_period=5000, guided_frac=0.3)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 3: Multi-swap SA from best
    if remaining() > 60:
        t0 = remaining()
        h_ms, c5_ms = sa_multi_swap(best_h, n, rng,
                                    n_iter=3000000,
                                    t_start=0.02, t_end=0.0001,
                                    timeout=min(t0 * 0.2, 150))
        if c5_ms < best_c5:
            best_c5 = c5_ms
            best_h = h_ms.copy()

    # Phase 4: SA from diverse inits (shorter runs)
    for h0 in inits[1:min(len(inits), 6)]:
        if remaining() < 60:
            break
        h_sa2, c5_sa2 = sa_single_swap(h0, n, rng,
                                       n_iter=2000000,
                                       t_start=0.025, t_end=0.0001,
                                       timeout=min(remaining() * 0.15, 80),
                                       guided_period=4000, guided_frac=0.25)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()

    # Phase 5: SLSQP smooth-max polish
    if remaining() > 60:
        h_sm, c5_sm = slsqp_smooth_max(best_h, n, timeout=min(remaining() * 0.25, 120),
                                       betas=(20, 60, 150, 400))
        if c5_sm < best_c5:
            best_c5 = c5_sm
            best_h = h_sm.copy()

    # Phase 6: Final SA from polished (lower temp)
    if remaining() > 60:
        h_f, c5_f = sa_single_swap(best_h, n, rng,
                                   n_iter=5000000,
                                   t_start=0.01, t_end=0.00002,
                                   timeout=min(remaining() * 0.4, 300),
                                   guided_period=3000, guided_frac=0.2)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 7: Greedy hill climb (exploitation)
    if remaining() > 30:
        h_gf, c5_gf = greedy_hill_climb(best_h, n, rng,
                                        max_iters=200000, timeout=remaining() - 10)
        if c5_gf < best_c5:
            best_c5 = c5_gf
            best_h = h_gf.copy()

    # Phase 8: Try n=512 if time remains
    if remaining() > 180:
        n2 = 512
        h_big = make_feasible(interp_to_n(best_h, n2), n2)
        c5_big = compute_c5(h_big)
        h_big_sa, c5_big_sa = sa_single_swap(h_big, n2, rng,
                                            n_iter=2000000,
                                            t_start=0.015, t_end=0.00005,
                                            timeout=min(remaining() * 0.5, 300),
                                            guided_period=5000, guided_frac=0.25)
        if c5_big_sa < best_c5:
            best_c5 = c5_big_sa
            best_h = h_big_sa
            n = n2

    # Phase 9: Final SLSQP polish at whatever n we're at
    if remaining() > 30:
        n_final = len(best_h)
        h_fin, c5_fin = slsqp_smooth_max(best_h, n_final,
                                         timeout=min(remaining() - 10, 60),
                                         betas=(50, 150, 400))
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    # Final feasibility check
    n_final = len(best_h)
    best_h = make_feasible(best_h, n_final)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_final)
