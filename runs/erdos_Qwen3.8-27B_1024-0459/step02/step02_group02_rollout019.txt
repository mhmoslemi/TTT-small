def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(h_src, n_dst):
    if len(h_src) < 2:

    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def simulated_annealing(h_init, n, rng, t_start=0.1, t_end=0.001, max_iters=200000, timeout=100.0):
    """Simulated annealing on binary sequences."""
    h = np.round(np.array(h_init, dtype=float)).astype(float)
    h = make_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    
    start_t = time.time()
    current_c5 = best_c5
    
    for it in range(max_iters):
        if time.time() - start_t > timeout:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Temperature schedule (geometric cooling)
        progress = it / max(max_iters, 1)
        T = t_start * (t_end / t_start) ** progress

        j = zeros[rng.randint(len(zeros))]

        h[j] = 1.0

            current_c5 = new_c5
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0

        t_opt = min(50, remaining() - 30)
        if t_opt < 15:
            break
        h_opt, c5_opt = multi_precise_lbfgs(h0, n, timeout=t_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 3: Binary LS from best (rounded) and from top results
    h_bin_best = np.round(best_h).astype(float)
    h_bin_best = make_feasible(h_bin_best, n)
    c5_bin = compute_c5(h_bin_best)

    for h0 in [h_bin_best] + [r[1] for r in quick_results[:3]]:
        if remaining() < 35:
            break
        t_bs = min(35, remaining() - 20)
        if t_bs < 10:
            break
        h_b, c5_b = binary_local_search(h0, n, rng, max_trials=150000,
                                        stall_limit=15000, timeout=t_bs)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 4: Simulated annealing
    if remaining() > 60:
        h_sa, c5_sa = simulated_annealing(best_h, n, rng,
                                          t_start=0.06, t_end=0.0003,
                                          max_iters=200000, timeout=min(55, remaining() - 25))
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 5: Targeted search on worst shift
    if remaining() > 40:
        h_t, c5_t = targeted_search(best_h, n, rng, max_trials=100000,
                                    timeout=min(35, remaining() - 20))
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()

    # Phase 6: Another SA round at lower temperature
    if remaining() > 40:
        h_sa2, c5_sa2 = simulated_annealing(best_h, n, rng,
                                            t_start=0.02, t_end=0.0001,
                                            max_iters=150000, timeout=min(35, remaining() - 20))
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()

    # Phase 7: Continuous perturbation search
    if remaining() > 30:
        h_c, c5_c = continuous_perturb_search(best_h, n, rng,
                                              max_trials=15000, timeout=min(25, remaining() - 15))
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c.copy()

    # Phase 8: Final L-BFGS-B polish with very high beta
    if remaining() > 25:
        h_f, c5_f = multi_precise_lbfgs(best_h, n, timeout=min(20, remaining() - 10))
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 9: Final binary LS
    if remaining() > 20:
        h_bf, c5_bf = binary_local_search(best_h, n, rng, max_trials=80000,
                                          stall_limit=8000, timeout=min(15, remaining() - 8))
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()

    # Final: one more targeted search if time
    if remaining() > 15:
        h_tf, c5_tf = targeted_search(best_h, n, rng, max_trials=50000,
                                      timeout=min(10, remaining() - 5))
        if c5_tf < best_c5:
            best_c5 = c5_tf
            best_h = h_tf.copy()

    # Final feasibility check
    best_h = make_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n)
