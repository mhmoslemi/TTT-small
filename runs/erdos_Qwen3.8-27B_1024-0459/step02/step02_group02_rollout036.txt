            h = h_try
            best_c5 = c5_try
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
    
    return best_h, best_c5

def multi_target_sa(h_init, n, rng, T_start=0.03, T_end=0.0002, n_iters=150000, timeout=180.0):
    """SA that biases moves toward reducing the worst shifts."""
    h = np.round(h_init).astype(float)
    h = make_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    curr_c5 = best_c5
    corr_curr = get_corr(h).copy()
    dx = 2.0 / n
    
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    
    start_t = time.time()
    no_improve = 0
    stall_limit = 60000
    
    for it in range(n_iters):
        if time.time() - start_t > timeout:
            break
        if no_improve > stall_limit:
            break
        T = T_start * (T_end / T_start) ** (it / n_iters)
        
        # 70% of the time, target worst shifts; 30% random
        if rng.rand() < 0.7:
            top_k = np.argsort(corr_curr)[-3:]
            worst_idx = top_k[rng.randint(3)]
            worst_lag = worst_idx - (n - 1)
            bad_pos = []
            if worst_lag >= 0:
                for pos in range(max(worst_lag, 0), n):
                    if h[pos] == 1.0 and (pos - worst_lag >= 0) and h[pos - worst_lag] == 0.0:
                        bad_pos.append(pos)
            else:
                for pos in range(0, min(n, n + worst_lag)):
                    if h[pos] == 1.0 and (pos - worst_lag < n) and h[pos - worst_lag] == 0.0:
                        bad_pos.append(pos)
            if bad_pos:
                i = bad_pos[rng.randint(len(bad_pos))]
            else:
                i = ones[rng.randint(len(ones))]
        else:
            i = ones[rng.randint(len(ones))]
        
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        
        old_hi = h[i]
        old_hj = h[j]
        h[i] = 1.0 - old_hi
        h[j] = 1.0 - old_hj
        
        new_corr = get_corr(h)
        new_max = np.max(new_corr)
        new_c5 = new_max * dx
        delta = new_c5 - curr_c5
        
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-12)):
            curr_c5 = new_c5
            corr_curr = new_corr
            no_improve += 1
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
        else:
            h[i] = old_hi
            h[j] = old_hj
    
    return best_h, best_c5

def final_slsqp_polish(h_init, n, max_iter=2000, timeout=60.0):
    global _GN
    _GN = [n]
    h0 = make_feasible(h_init.copy(), n)
    bounds = [(0.0, 1.0)] * n
    con = {'type': 'eq', 'fun': _sum_con_fun, 'jac': _sum_con_jac}
    start_t = time.time()
    try:
        res = minimize(direct_obj, h0, method='SLSQP',
                      bounds=bounds, constraints=con,
                      options={'maxiter': max_iter, 'ftol': 1e-15})
        if time.time() - start_t < timeout and res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return h_init, compute_c5(h_init)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    
    def remaining():
        return budget_s - (time.time() - start_time)
    
    # Get parent
    h_parent = None
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if h_parent is not None and len(h_parent) > 0:
            h_parent = h_parent.copy()
        else:
            h_parent = None
    except Exception:
        h_parent = None
    
    best_h = None
    best_c5 = float('inf')
    best_n = 256
    
    # ===== Phase 1: n=256 optimization =====
    n1 = 256
    
    inits_256 = []
    if h_parent is not None:
        inits_256.append(make_feasible(h_parent, n1))
    inits_256.append(thue_morse_binary(n1))
    for _ in range(3):
        inits_256.append(random_binary(n1, rng))
    
    for h0 in inits_256:
        if remaining() < 20:
            break
        h_opt, c5_val = slsqp_optimize(h0, n1, beta_start=15.0, beta_end=300.0,
                                        n_steps=4, max_iter=400, timeout=30)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt.copy()
            best_n = n1
    
    # Binary SA at n=256
    if best_h is not None and remaining() > 60:
        h_sa, c5_sa = binary_sa(best_h, n1, rng, T_start=0.04, T_end=0.0005,
                                n_iters=150000, timeout=40)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            best_n = n1
    
    # Targeted search at n=256
    if best_h is not None and remaining() > 40:
        h_t, c5_t = targeted_binary_search(best_h, n1, rng, n_trials=80000, timeout=30)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            best_n = n1
    
    # ===== Phase 2: n=512 optimization =====
    n2 = 512
    
    inits_512 = []
    if best_h is not None:
        inits_512.append(upsample(best_h, n2))
    inits_512.append(thue_morse_binary(n2))
    for _ in range(3):
        inits_512.append(random_binary(n2, rng))
    
    # Evaluate inits
    best_512_h = None
    best_512_c5 = float('inf')
    for h0 in inits_512:
        c5_init = compute_c5(h0)
        if c5_init < best_512_c5:
            best_512_c5 = c5_init
            best_512_h = h0.copy()
    
    # Simulated annealing at n=512 from best init
    if best_512_h is not None and remaining() > 100:
        h_sa512, c5_sa512 = binary_sa(best_512_h, n2, rng, T_start=0.04, T_end=0.0003,
                                       n_iters=300000, timeout=min(180, remaining() - 60))
        if c5_sa512 < best_512_c5:
            best_512_c5 = c5_sa512
            best_512_h = h_sa512.copy()
    
    # Multi-target SA
    if best_512_h is not None and remaining() > 80:
        h_mt, c5_mt = multi_target_sa(best_512_h, n2, rng, T_start=0.03, T_end=0.0002,
                                       n_iters=200000, timeout=min(150, remaining() - 50))
        if c5_mt < best_512_c5:
            best_512_c5 = c5_mt
            best_512_h = h_mt.copy()
    
    # Targeted binary search at n=512
    if best_512_h is not None and remaining() > 50:
        h_t512, c5_t512 = targeted_binary_search(best_512_h, n2, rng, n_trials=150000,
                                                  timeout=min(80, remaining() - 30))
        if c5_t512 < best_512_c5:
            best_512_c5 = c5_t512
            best_512_h = h_t512.copy()
    
    # Second round of SA from current best
    if best_512_h is not None and remaining() > 60:
        h_sa2, c5_sa2 = binary_sa(best_512_h, n2, rng, T_start=0.02, T_end=0.0001,
                                   n_iters=200000, timeout=min(100, remaining() - 40))
        if c5_sa2 < best_512_c5:
            best_512_c5 = c5_sa2
            best_512_h = h_sa2.copy()
    
    # Try SA from Thue-Morse at n=512 (different basin)
    h_tm512 = thue_morse_binary(n2)
    if compute_c5(h_tm512) < best_512_c5 - 0.001 and remaining() > 80:
        h_tm_sa, c5_tm_sa = binary_sa(h_tm512, n2, rng, T_start=0.04, T_end=0.0003,
                                       n_iters=200000, timeout=min(100, remaining() - 40))
        if c5_tm_sa < best_512_c5:
            best_512_c5 = c5_tm_sa
            best_512_h = h_tm_sa.copy()
    
    # Continuous local search at n=512
    if best_512_h is not None and remaining() > 40:
        h_cl, c5_cl = continuous_local_search(best_512_h, n2, rng, n_trials=30000,
                                               timeout=min(50, remaining() - 20),
                                               step_sizes=[0.01, 0.03, 0.08])
        if c5_cl < best_512_c5:
            best_512_c5 = c5_cl
            best_512_h = h_cl.copy()
    
    # Final SLSQP polish at n=512
    if best_512_h is not None and remaining() > 30:
        h_pol, c5_pol = final_slsqp_polish(best_512_h, n2, max_iter=1500,
                                            timeout=min(40, remaining() - 10))
        if c5_pol < best_512_c5:
            best_512_c5 = c5_pol
            best_512_h = h_pol.copy()
    
    # One more targeted search after polishing
    if best_512_h is not None and remaining() > 20:
        h_final_t, c5_final_t = targeted_binary_search(best_512_h, n2, rng, n_trials=80000,
                                                        timeout=min(25, remaining() - 5))
        if c5_final_t < best_512_c5:
            best_512_c5 = c5_final_t
            best_512_h = h_final_t.copy()
    
    # Compare n=256 and n=512 results
    if best_512_c5 < best_c5:
        best_c5 = best_512_c5
        best_h = best_512_h.copy()
        best_n = n2
    
    # Final feasibility check
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, best_n)
