    pref = np.zeros(n+1)
    np.cumsum(h, out=pref[1:])

    p = exps / np.sum(exps)

        g[:m] += -w * h[L:]
    
    return f, g

    Z = np.sum(exps)
    p = exps / Z
    obj = max_v + np.log(Z) / beta
    return obj, p, vals

    n = len(h)
