import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def soft_max(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def soft_max_grad(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    exps = np.exp(shifted)
    return exps / np.sum(exps)

def compute_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    wt = soft_max_grad(corr, beta)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = wt[k]
        if w < 1e-14:
            continue
        L = k - (n - 1)
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += w * (1.0 - h[:n-L])
            grad[:n-L] -= w * h[L:]
        else:
            Lp = -L
            grad[:n-Lp] += w * (1.0 - h[Lp:])
            grad[Lp:] -= w * h[:n-Lp]
    return dx * grad

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(np.maximum(h, 1.0 - h))
        h[idx] += diff
    return h

def projected_gradient(h, beta, alpha0, n_steps, tol=1e-10):
    n = len(h)
    h = project_feasible(h.copy(), n)
    prev_obj = soft_max(np.correlate(h, 1.0-h, mode='full'), beta)
    for it in range(n_steps):

        alpha = alpha0
        h_new = h - alpha * grad
        h_new = project_feasible(h_new, n)
        obj_new = soft_max(np.correlate(h_new, 1.0-h_new, mode='full'), beta)

        h = h_new
        if abs(prev_obj - obj_new) < tol:

        prev_obj = obj_new
    return h

    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)

            else:
                h[p1] = 1.0
                h[p0] = 0.0
        if not improved:
            break
    return h, c5

def block_swap_search(h, n, n_rounds, rng):
    c5 = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        for _ in range(20):
            blen = rng.randint(2, 5)
            pos = rng.randint(0, n - blen)
            h_new = h.copy()
            h_new[pos:pos+blen], h_new[n-blen:n] = h_new[n-blen:n].copy(), h_new[pos:pos+blen].copy()
            h_new = project_feasible(h_new, n)
            nc5 = compute_c5(h_new)
            if nc5 < c5 - 1e-14:
                h = h_new
                c5 = nc5
                improved = True
                break
        if not improved:
            break
    return h, c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - start_time

    def remaining():
        return max(0.0, budget_s - elapsed() - 10.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        if len(parent_h) > 0:
            parent_n = len(parent_h)
            parent_c5 = compute_c5(parent_h)
            best_h = parent_h.copy()
            best_c5 = parent_c5
            best_n = parent_n
            print("Parent: n={}, C5={:.8f}".format(parent_n, parent_c5))
    except Exception:
        pass

    if best_h is None:
        best_n = 200
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_c5 = compute_c5(best_h)
        print("Default binary: n={}, C5={:.8f}".format(best_n, best_c5))

    # Try multiple resolutions
    n_list = [best_n, 256, 320, 400, 512]
    n_list = sorted(set([nn for nn in n_list if 64 <= nn <= 600]))

    for n in n_list:
        if remaining() < 60:
            break
        print("=== n={} (remain {:.0f}s) ===".format(n, remaining()))

        # Initialize from parent + structured
        inits = [interp_h(best_h, n)]
        inits.append(make_anti_phase(n, rng))
        inits.append(make_block_pattern(n, rng))
        # Binary init from parent
        hb = interp_h(best_h, n)
        hb_bin = (hb > 0.5).astype(float)
        n_ones = int(np.sum(hb_bin))
        n_want = n // 2
        if n_ones != n_want:
            ones_idx = np.where(hb_bin == 1)[0]
            zeros_idx = np.where(hb_bin == 0)[0]
            if n_ones > n_want:
                order = np.argsort(hb[ones_idx])[:n_ones - n_want]
                hb_bin[ones_idx[order]] = 0
            else:
                order = np.argsort(-hb[zeros_idx])[:n_want - n_ones]
                hb_bin[zeros_idx[order]] = 1
        inits.append(hb_bin)

        for h0 in inits:
            if remaining() < 30:
                break
            h0 = project_feasible(h0, n)
            c5_0 = compute_c5(h0)
            if c5_0 < best_c5:
                best_c5 = c5_0
                best_h = h0.copy()
                best_n = n

            # Phase 1: L-BFGS-B with penalty
            h_cur = h0.copy()
            for beta_val, pen in [(15, 10.0), (40, 50.0), (100, 200.0)]:
                if remaining() < 15:
                    break
                max_it = 200 if beta_val <= 40 else 150
                try:
                    h_new = lbfgsb_penalty(h_cur, n, beta_val, pen, max_iter=max_it)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        best_n = n
                        print("  n={}, beta={}, C5={:.8f}".format(n, beta_val, best_c5))
                    h_cur = h_new
                except Exception:
                    pass

            # Phase 2: Projected gradient with increasing beta
            if remaining() > 20 and best_n == n:
                try:
                    beta_sched = [50, 150, 400, 800]
                    h_pg = projected_gradient(h_cur, n, beta_sched, max_iter=800)
                    c5_pg = compute_c5(h_pg)
                    if c5_pg < best_c5:
                        best_c5 = c5_pg
                        best_h = h_pg.copy()
                        print("  PG: n={}, C5={:.8f}".format(n, best_c5))
                except Exception:
                    pass

        # Binary rounding + local search at this n
        if remaining() > 40 and best_n == n:
            h_round = (best_h > 0.5).astype(float)
            n_ones = int(np.sum(h_round))
            n_want = n // 2
            if n_ones > n_want:
                ones_idx = np.where(h_round == 1)[0]
                vals = best_h[ones_idx]
                order = np.argsort(vals)[:n_ones - n_want]
                h_round[ones_idx[order]] = 0.0
            elif n_ones < n_want:
                zeros_idx = np.where(h_round == 0)[0]
                vals = best_h[zeros_idx]
                order = np.argsort(-vals)[:n_want - n_ones]
                h_round[zeros_idx[order]] = 1.0
            c5_round = compute_c5(h_round)
            if c5_round < best_c5:
                best_c5 = c5_round
                best_h = h_round.copy()
                print("  Binary round: C5={:.8f}".format(best_c5))

            # Local search on binary
            h_bin = (best_h > 0.5).astype(float)
            if np.all((h_bin == 0) | (h_bin == 1)):
                h_bin, c5_lbs = binary_local_search(h_bin, n, 5000, rng)
                if c5_lbs < best_c5:
                    best_c5 = c5_lbs
                    best_h = h_bin.copy()
                    print("  Binary LS: C5={:.8f}".format(best_c5))

                # Block swap
                h_bin, c5_bs = block_swap_search(h_bin, n, 1000, rng)
                if c5_bs < best_c5:
                    best_c5 = c5_bs
                    best_h = h_bin.copy()
                    print("  Block swap: C5={:.8f}".format(best_c5))

    # Final continuous polish on best solution
    if remaining() > 30:
        n_fin = best_n
        h_fin = best_h.copy()
        try:
            h_fin = lbfgsb_penalty(h_fin, n_fin, 200, 500.0, max_iter=200)
            c5_fin = compute_c5(h_fin)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()
                print("  Final L-BFGS-B: C5={:.8f}".format(best_c5))
        except Exception:
            pass

        try:
            h_fin = projected_gradient(h_fin, n_fin, [300, 600, 1000], max_iter=600)
            c5_fin = compute_c5(h_fin)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()
                print("  Final PG: C5={:.8f}".format(best_c5))
        except Exception:
            pass

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
