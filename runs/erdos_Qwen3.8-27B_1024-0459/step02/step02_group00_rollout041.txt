c[idx] = sum over valid i of h[i+k] * (1-h[i])

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0

def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_sum(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        c = int(mask.sum())
        if c > 0:
            h[mask] += diff / c
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(h_src, n_src, n_dst):
    if n_src == n_dst:
        return h_src.copy()
    xs = (np.arange(n_src) + 0.5) * 2.0 / n_src
    xd = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    return np.interp(xd, xs, h_src)

def _smooth_corr_obj(x, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = float(corr.max())
    sh = corr - m
    e = np.exp(beta * sh)
    s = float(e.sum())
    if s < 1e-300:
        return m * (2.0 / len(x))
    dx = 2.0 / len(x)
    return (m + np.log(s) / beta) * dx

def _smooth_corr_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = float(corr.max())
    sh = corr - m
    e = np.exp(beta * sh)
    s = float(e.sum())
    w = e / s
    grad = np.zeros(n)
    for idx in range(corr.size):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        # First term: (1-x[j-k]) for j in [max(0,k), min(n, n+k))
        ja = max(0, k)
        jb = min(n, n + k)
        if ja < jb:
            grad[ja:jb] += wi * (1.0 - x[ja-k:jb-k])
        # Second term: -x[j+k] for j in [max(0,-k), min(n, n-k))
        jc = max(0, -k)
        jd = min(n, n - k)
        if jc < jd:
            grad[jc:jd] -= wi * x[jc+k:jd+k]
    return grad * dx

def _sum_con(x, n):
    return x.sum() - n / 2.0

def _sum_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, beta, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: _sum_con(x, n), 'jac': lambda x: _sum_jac(x, n)}]
    try:
        res = minimize(_smooth_corr_obj, h, args=(beta,),
                       jac=lambda x: _smooth_corr_grad(x, beta),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return project_sum(res.x, n)
    except Exception:
        return h

                if c5 < best_val - 1e-14:

            if best_v != old:
