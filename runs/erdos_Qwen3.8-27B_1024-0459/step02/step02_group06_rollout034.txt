                if free2.size > 0:
                    h[free2[0]] += res2
                    h = np.clip(h, 0.0, 1.0)
    return h


def upsample_h(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    if n0 == n:
        return p.copy()
    if n0 <= 0:
        return project_density(np.full(max(1, n), 0.5), 0.5 * n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return project_density(h, 0.5 * n)


def coord_descent(h, n, passes=50, deadline=None, rng=None, project_every=5):
    center = n - 1
    shifts = np.arange(2 * n - 1) - center
    target = 0.5 * n
    h = project_density(h.copy(), target)
    best_c = evaluate_erdos(h)
    best_h = h.copy()
    stall = 0

    for _p in range(passes):
        if deadline is not None and time.perf_counter() >= deadline:
            break

        if rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode="full")

        for j in order:
            j = int(j)
            js = j + shifts
            js2 = j - shifts

            a = np.zeros(2 * n - 1)
            vf = (js >= 0) & (js < n)
            vs = (js2 >= 0) & (js2 < n)

            if np.any(vf):
                a[vf] = 1.0 - h[js[vf]]
            if np.any(vs):
                a[vs] -= h[js2[vs]]

            base = corr - h[j] * a

            lo_t, hi_t = 0.0, 1.0
            for _t in range(30):
                m1 = lo_t + (hi_t - lo_t) / 3.0
                m2 = hi_t - (hi_t - lo_t) / 3.0
                f1 = float(np.max(base + m1 * a))
                f2 = float(np.max(base + m2 * a))
                if f1 <= f2:
                    hi_t = m2
                else:
                    lo_t = m1
            t_opt = (lo_t + hi_t) / 2.0

            old_j = h[j]
            if abs(t_opt - old_j) > 1e-14:
                h[j] = t_opt
                corr += (t_opt - old_j) * a

        if (_p + 1) % project_every == 0 or _p == passes - 1:
            h = project_density(h, target)
            c = evaluate_erdos(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            if stall > 8:
                break

    return best_h, best_c


def full_cd_pipeline(h, n, deadline, rng):
    target = 0.5 * n
    best_h = project_density(h.copy(), target)
    best_c = evaluate_erdos(best_h)

    # Phase 1: Cyclic CD
    rem = deadline - time.perf_counter()
    if rem > 5:
        h1, c1 = coord_descent(best_h, n, passes=min(200, int(rem * 8)), deadline=time.perf_counter() + rem * 0.4, rng=None, project_every=5)
        if c1 < best_c:
            best_h = h1
            best_c = c1

    # Phase 2: Random order CD
    rem = deadline - time.perf_counter()
    if rem > 5:
        h2, c2 = coord_descent(best_h, n, passes=min(200, int(rem * 5)), deadline=time.perf_counter() + rem * 0.4, rng=rng, project_every=10)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    # Phase 3: Binarize push + CD
    if n <= 800 and deadline - time.perf_counter() > 5:
        h3 = best_h.copy()
        changed = False
        for i in range(n):
            if h3[i] < 0.03:
                h3[i] = 0.0
                changed = True
            elif h3[i] > 0.97:
                h3[i] = 1.0
                changed = True
        if changed:
            h3 = project_density(h3, target)
            c3 = evaluate_erdos(h3)
            if c3 < best_c:
                best_h = h3
                best_c = c3
            h3b, c3b = coord_descent(best_h, n, passes=min(80, int((deadline - time.perf_counter()) * 5)), deadline=deadline, rng=None, project_every=10)
            if c3b < best_c:
                best_h = h3b
                best_c = c3b

    # Phase 4: More random CD
    rem = deadline - time.perf_counter()
    if rem > 5:
        h4, c4 = coord_descent(best_h, n, passes=min(100, int(rem * 4)), deadline=deadline, rng=rng, project_every=10)
        if c4 < best_c:
            best_h = h4
            best_c = c4

    return project_density(best_h, target), best_c


def pair_refine(h, n, deadline, rng, max_iter=3000):
    target = 0.5 * n
    best = project_density(h.copy(), target)
    best_c = evaluate_erdos(best)
    center = n - 1
    top_tracks = 0

    for _ in range(max_iter):
        if time.perf_counter() >= deadline:
            break

        corr = np.correlate(best, 1.0 - best, mode="full")
        k_max = int(np.argmax(corr))
        shift = k_max - center

        js = np.arange(n) + shift
        js2 = np.arange(n) - shift
        grad = np.zeros(n)
        vf = (js >= 0) & (js < n)
        vs = (js2 >= 0) & (js2 < n)
        if np.any(vf):
            grad[vf] = 1.0 - best[js[vf]]
        if np.any(vs):
            grad[vs] -= best[js2[vs]]

        inc_cands = np.argsort(grad)[-6:]
        dec_cands = np.argsort(grad)[:6]

        found = False
        for ii in inc_cands:
            ii = int(ii)
            for jj in dec_cands:
                jj = int(jj)
                if ii == jj:
                    continue
                cap = min(1.0 - best[ii], best[jj])
                if cap < 1e-12:
                    continue
                for frac in (1.0, 0.7, 0.4, 0.2):
                    d = frac * cap
                    cand = best.copy()
                    cand[ii] += d
                    cand[jj] -= d
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-13:
                        best = cand
                        best_c = c
                        found = True
                        break
                if found:
                    break
            if found:
                break

        if not found:
            for _r in range(8):
                ii = int(rng.integers(n))
                jj = int(rng.integers(n))
                if ii == jj:
                    continue
                cap = min(1.0 - best[ii], best[jj])
                if cap < 1e-12:
                    continue
                d = 0.3 * cap
                cand = best.copy()
                cand[ii] += d
                cand[jj] -= d
                c = evaluate_erdos(cand)
                if c < best_c - 1e-13:
                    best = cand
                    best_c = c
                    found = True
                    break
        if found:
            top_tracks = 0
        else:
            top_tracks += 1
            if top_tracks > 50:
                break

    return project_density(best, target), best_c


def multi_lag_pair_refine(h, n, deadline, rng, max_iter=2000):
    """Use multiple active lags for pair decisions"""
    target = 0.5 * n
    best = project_density(h.copy(), target)
    best_c = evaluate_erdos(best)
    center = n - 1
    stall = 0

    for _ in range(max_iter):
        if time.perf_counter() >= deadline:
            break

        corr = np.correlate(best, 1.0 - best, mode="full")
        max_val = float(np.max(corr))
        tol = 0.05 * max_val + 1e-10
        active = np.where(corr >= max_val - tol)[0]
        if active.size == 0:
            active = np.array([int(np.argmax(corr))])

        grad = np.zeros(n)
        for k_idx in active[:12]:
            k_idx = int(k_idx)
            shift = k_idx - center
            js = np.arange(n) + shift
            js2 = np.arange(n) - shift
            vf = (js >= 0) & (js < n)
            vs = (js2 >= 0) & (js2 < n)
            if np.any(vf):
                grad[vf] += 1.0 - best[js[vf]]
            if np.any(vs):
                grad[vs] -= best[js2[vs]]

        inc_cands = np.argsort(grad)[-8:]
        dec_cands = np.argsort(grad)[:8]

        found = False
        for ii in inc_cands:
            ii = int(ii)
            for jj in dec_cands:
                jj = int(jj)
                if ii == jj:
                    continue
                cap = min(1.0 - best[ii], best[jj])
                if cap < 1e-12:
                    continue
                for frac in (1.0, 0.6, 0.3):
                    d = frac * cap
                    cand = best.copy()
                    cand[ii] += d
                    cand[jj] -= d
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-13:
                        best = cand
                        best_c = c
                        found = True
                        break
                if found:
                    break
            if found:
                break

        if not found:
            ii = int(rng.integers(n))
            jj = int(rng.integers(n))
            if ii != jj:
                cap = min(1.0 - best[ii], best[jj])
                if cap > 1e-12:
                    d = 0.3 * cap
                    cand = best.copy()
                    cand[ii] += d
                    cand[jj] -= d
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-13:
                        best = cand
                        best_c = c
                        found = True
        if found:
            stall = 0
        else:
            stall += 1
            if stall > 40:
                break

    return project_density(best, target), best_c


def make_random_start(n, rng, mode):
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    if mode == 0:
        h = 0.5 + 0.35 * np.sin(2 * np.pi * 3 * x / 2 + rng.uniform(0, 6.28))
        h += 0.12 * np.sin(2 * np.pi * 7 * x / 2 + rng.uniform(0, 6.28))
    elif mode == 1:
        u = rng.random(n)
        h = np.where(u > 0.5, 0.75, 0.25).astype(float)
        h += rng.normal(0, 0.03, n)
    elif mode == 2:
        seg = max(1, n // 10)
        h = np.empty(n)
        for i in range(0, n, seg):
            end = min(i + seg, n)
            h[i:end] = float(rng.uniform(0.1, 0.9))
    else:
        h = 0.5 + 0.4 * np.sin(2 * np.pi * int(rng.integers(2, 8)) * x / 2 + rng.uniform(0, 6.28))
    return project_density(h, 0.5 * n)


def run(seed=42, budget_s=1300, **kwargs):
    start = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = start + max(1.0, budget - 20.0)
    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float("inf")
    best_n = 0

    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0).astype(float, copy=True)
            parent = np.clip(parent, 0.0, 1.0)
    except Exception:
        parent = None

    if parent is not None:
        np_ = int(parent.size)
        best_h = project_density(parent.copy(), 0.5 * np_)
        best_c = evaluate_erdos(best_h)
        best_n = np_
    else:
        n0 = 90
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        h_init = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x / 2) + 0.15 * np.sin(2 * np.pi * 5 * x / 2 + 1.0)
        best_h = project_density(h_init, 0.5 * n0)
        best_c = evaluate_erdos(best_h)
        best_n = n0

    # Define resolutions to try
    resolutions = []
    base = best_n
    resolutions.append(base)
    for mult in [2, 3, 4]:
        cand = base * mult
        if cand <= 1200 and cand not in resolutions:
            resolutions.append(cand)
    for r in [90, 180, 270, 450, 600, 900]:
        if r not in resolutions and 2 <= r <= 1200:
            resolutions.append(r)
    resolutions = sorted(set(resolutions))

    # Time allocation: favor parent resolution and one step up
    total_time = deadline - time.perf_counter()
    n_res = len(resolutions)

    # Process each resolution
    for ni in resolutions:
        rem = deadline - time.perf_counter()
        if rem < 30:
            break

        # Allocate time based on resolution
        if ni == base:
            alloc_frac = 0.6
        elif ni <= 2 * base:
            alloc_frac = 0.5
        else:
            alloc_frac = 0.3
        if n_res > 3:
            alloc_frac *= (3.0 / n_res)
        alloc = min(rem * max(0.15, alloc_frac), rem - 10.0)
        alloc = max(25.0, alloc)
        sub_deadline = time.perf_counter() + alloc

        # Starting points
        starts = [upsample_h(best_h, ni)]
        if parent is not None and ni != parent.size:
            starts.append(upsample_h(parent, ni))

        n_random = 2 if ni <= 200 else (1 if ni <= 500 else 1)
        for mm in range(n_random):
            starts.append(make_random_start(ni, rng, mm % 4))

        # Deduplicate
        uniq = []
        for s in starts:
            s = project_density(s, 0.5 * ni)
            dup = False
            for u in uniq:
                if np.allclose(s, u, atol=1e-5):
                    dup = True
                    break
            if not dup:
                uniq.append(s)

        per_start = max(10.0, sub_deadline / len(uniq))

        for s in uniq:
            rem2 = deadline - time.perf_counter()
            if rem2 < 15:
                break
            sub_dl = min(sub_deadline, time.perf_counter() + per_start)
            h_out, c_out = full_cd_pipeline(s, ni, sub_dl, rng)
            if c_out < best_c:
                best_h = h_out.copy()
                best_c = c_out
                best_n = ni

    # Final intensive optimization
    rem = deadline - time.perf_counter()
    if best_h is not None and rem > 30:
        nbest = int(best_h.size)

        # Try one more resolution up
        if nbest < 900 and rem > 80:
            n_high = min(2 * nbest, 900)
            h_high = upsample_h(best_h, n_high)
            h_high, c_high = full_cd_pipeline(h_high, n_high, time.perf_counter() + rem * 0.3, rng)
            if c_high < best_c:
                best_h = h_high.copy()
                best_c = c_high
                best_n = n_high

        # Intensive CD on best
        rem2 = deadline - time.perf_counter()
        nbest = int(best_h.size)
        if rem2 > 20:
            h_cd, c_cd = coord_descent(best_h, nbest, passes=min(500, int(rem2 * 6)),
                                       deadline=time.perf_counter() + rem2 * 0.5, rng=None, project_every=5)
            if c_cd < best_c:
                best_h = h_cd
                best_c = c_cd

        # Multi-lag pair refine
        rem3 = deadline - time.perf_counter()
        if rem3 > 20 and nbest <= 900:
            h_mp, c_mp = multi_lag_pair_refine(best_h, nbest, time.perf_counter() + rem3 * 0.5, rng, max_iter=3000)
            if c_mp < best_c:
                best_h = h_mp
                best_c = c_mp
                nbest = int(best_h.size)

        # More CD after pair refine
        rem4 = deadline - time.perf_counter()
        if rem4 > 15:
            nbest = int(best_h.size)
            h_cd2, c_cd2 = coord_descent(best_h, nbest, passes=min(200, int(rem4 * 5)),
                                         deadline=time.perf_counter() + rem4 * 0.6, rng=rng, project_every=5)
            if c_cd2 < best_c:
                best_h = h_cd2
                best_c = c_cd2

        # Final pair refine
        rem5 = deadline - time.perf_counter()
        if rem5 > 10 and best_n <= 600:
            nbest = int(best_h.size)
            h_pf, c_pf = pair_refine(best_h, nbest, time.perf_counter() + rem5 * 0.7, rng, max_iter=2000)
            if c_pf < best_c:
                best_h = h_pf
                best_c = c_pf

        # Very final CD
        rem6 = deadline - time.perf_counter()
        if rem6 > 8:
            nbest = int(best_h.size)
            h_final, c_final = coord_descent(best_h, nbest, passes=min(100, int(rem6 * 4)),
                                             deadline=deadline, rng=None, project_every=5)
            if c_final < best_c:
                best_h = h_final
                best_c = c_final

    if best_h is None:
        fallback = project_density(np.full(45, 0.5), 22.5)
        return fallback, evaluate_erdos(fallback), 45

    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    return final_h, float(final_c), int(final_h.size)
