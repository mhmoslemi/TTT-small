                for d in [1, -1, 2, -2, 3, -3, 4, -4, 6, -6]:
                    j = p1 + d
                    if 0 <= j < n and h[j] < 0.5:
                        h_try = h.copy()
                        h_try[p1] = 0.0
                        h_try[j] = 1.0
                        h_try = project_feasible(h_try, n)
                        c5_try = compute_c5(h_try)
                        if c5_try < best_c5 - 1e-13:
                            h = h_try
                            best_c5 = c5_try
                            moved = True
                            break
                if moved:
                    break
        if not moved:
            _BETA = 500
            try:
                res = minimize(smooth_obj, h, method='L-BFGS-B',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 200, 'ftol': 1e-14,
                                        'maxfun': 1000, 'iprint': 0})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-13:
                    h = h_new
                    best_c5 = c5_new
            except Exception:
                pass
    return h, best_c5


def make_periodic(n, period, duty):
    h = np.zeros(n)
    for i in range(n):
        if (i % period) < int(period * duty):
            h[i] = 1.0
    return h
