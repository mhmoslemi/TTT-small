import numpy as np
import time
import math
from scipy.optimize import minimize

_SLQP = {}
_LB = {}


def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n <= 1:
        return 0.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / float(n)))


def c_raw(h):
    h = np.asarray(h, dtype=float)
    return np.correlate(h, 1.0 - h, mode="full")


def project_box_sum(z, target, iters=45):
    z = np.asarray(z, dtype=float)
    n = z.size
    if n == 0:
        return z
    if not np.all(np.isfinite(z)):
        z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    z = np.clip(z, 0.0, 1.0)
    if target <= 0.0:
        return np.zeros(n, dtype=float)
    if target >= float(n):
        return np.ones(n, dtype=float)

    lo = -float(np.max(z)) - 1.0
    hi = 1.0 - float(np.min(z)) + 1.0

    for _ in range(iters):
        mid = 0.5 * (lo + hi)
        s = np.clip(z + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid

    out = np.clip(z + 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(6):
        diff = target - out.sum()
        if abs(diff) < 1e-14:
            break
        mask = (out > 1e-13) & (out < 1.0 - 1e-13)
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            break
        out[mask] += diff / float(cnt)
        out = np.clip(out, 0.0, 1.0)
    return out


def make_feasible(h, n):
    n = int(n)
    h = np.asarray(h, dtype=float)
    if h.size != n:
        h = interpolate_to_n(h, n)
    out = project_box_sum(h, 0.5 * n, iters=50)
    diff = 0.5 * n - out.sum()
    if abs(diff) > 1e-13:
        mask = (out > 1e-14) & (out < 1.0 - 1e-14)
        cnt = int(np.count_nonzero(mask))
        if cnt:
            out[mask] += diff / float(cnt)
    return np.clip(out, 0.0, 1.0)


def interpolate_to_n(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float)
    n_src = h_src.size
    n_dst = int(n_dst)
    if n_src == n_dst:
        return h_src.copy()
    if n_src == 1:
        return np.full(n_dst, float(h_src[0]), dtype=float)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    out = np.interp(x_new, x_old, h_src)
    return make_feasible(out, n_dst)


def random_binary(n, rng):
    n = int(n)
    h = np.zeros(n, dtype=float)
    if n > 0:
        k = n // 2
        if k > 0:
            idx = rng.choice(n, k, replace=False)
            h[idx] = 1.0
        if 2 * k < n:
            h[rng.randint(n)] = 0.5
    return make_feasible(h, n)


def threshold_quantile(h, n):
    n = int(n)
    h = np.asarray(h, dtype=float)
    if h.size != n:
        h = interpolate_to_n(h, n)
    out = np.zeros(n, dtype=float)
    k = n // 2
    if k > 0:
        idx = np.argsort(h, kind="mergesort")[::-1][:k]
        out[idx] = 1.0
    if 2 * k < n and n > 0:
        nxt = np.argsort(h, kind="mergesort")[::-1][k]
        out[nxt] = 0.5
    return make_feasible(out, n)


def structured_start(n, rng):
    n = int(n)
    best_h = None
    best_c = math.inf
    candidates = []

    if n % 4 == 0:
        candidates.append(np.tile(np.array([1.0, 1.0, 0.0, 0.0], dtype=float), n // 4))

    x = np.arange(n, dtype=float)
    candidates.append(0.5 + 0.4 * np.sin(2.0 * math.pi * x / float(max(2, n))))
    candidates.append(0.5 + 0.4 * np.cos(2.0 * math.pi * x / float(max(2, n // 2 or n))))
    candidates.append(0.5 + 0.35 * np.sin(2.0 * math.pi * x * 3.0 / float(max(6, n))))
    candidates.append(rng.rand(n))

        best_h = make_feasible(np.full(n, 0.5, dtype=float), n)

def top_indices(c, K, keep_center=True):
    n_h = (len(c) + 1) // 2
    K = int(max(1, min(K, len(c))))
    order = np.argsort(c, kind="mergesort")[::-1][:K]
    idxs = []
    seen = set()
    for ii in order:
        ii = int(ii)
        if ii not in seen:
            seen.add(ii)
            idxs.append(ii)
    if keep_center:
        center = int(n_h - 1)
        if center not in seen:
            idxs.append(center)
    return idxs


def smooth_components(h, beta, K):
    n = len(h)
    c = c_raw(h)
    idxs = top_indices(c, K, keep_center=True)
    vals = c[idxs]
    if vals.size == 0:
        return 0.0, [], c

    mx = float(vals.max())
    z = np.exp(float(beta) * (vals - mx))
    s = float(z.sum())
    if not math.isfinite(s) or s <= 0.0:
        w = np.full(vals.size, 1.0 / float(vals.size), dtype=float)
        val = mx
    else:
        w = z / s
        val = mx + (1.0 / float(beta)) * math.log(s)

    center = int(n - 1)
    meta = []
    for ii, ww in zip(idxs, w):
        ww = float(ww)
        if ww <= 1e-16:
            continue
        k = int(ii) - center
        if k == 0:
            meta.append((0, 0, ww))
        elif k > 0:
            meta.append((k, 1, ww))
        else:
            meta.append((-k, -1, ww))
    return float(val), meta, c


def grad_weighted(h, meta, center=True):
    g = np.zeros_like(h, dtype=float)
    for d, sgn, w in meta:
        if w <= 1e-16:
            continue
        if sgn == 0:
            g += w * (1.0 - 2.0 * h)
        elif d > 0 and d < h.size:
            if sgn > 0:
                g[:-d] -= w * h[d:]
                g[d:] += w * (1.0 - h[:-d])
            else:
                g[:-d] += w * (1.0 - h[d:])
                g[d:] -= w * h[:-d]
    if center and g.size:
        g -= g.mean()
    return g


def q_weighted(h, meta):
    q = 0.0
    one = 1.0 - h
    for d, sgn, w in meta:
        if w <= 1e-16:
            continue
        if sgn == 0:
            q += w * float(np.dot(h, one))
        elif d > 0 and d < h.size:
            if sgn > 0:
                q += w * float(np.dot(h[d:], one[:-d]))
            else:
                q += w * float(np.dot(h[:-d], one[d:]))
    return float(q)


def slqp_obj(x):
    p = _SLQP
    if p.get("active") and time.time() > p.get("deadline", 0.0):
        raise RuntimeError("slqp-timeout")
    if not np.all(np.isfinite(x)):
        return 1e100
    val, _, c = smooth_components(x, p["beta"], p["K"])
    cur = float(np.max(c) * (2.0 / float(p["n"])))
    if cur < p.get("best_c", math.inf) - 1e-15:
        p["best_c"] = cur
        p["best_h"] = np.array(x, dtype=float, copy=True)
    return float(val)


def slqp_jac(x):
    p = _SLQP
    if p.get("active") and time.time() > p.get("deadline", 0.0):
        raise RuntimeError("slqp-timeout")
    _, meta, _ = smooth_components(x, p["beta"], p["K"])
    return grad_weighted(x, meta, center=True)


def slqp_eq(x):
    p = _SLQP
    return float(np.sum(x) - p["target"])


def slqp_eq_jac(x):
    p = _SLQP
    return np.ones(p["n"], dtype=float)


def run_slqp_smooth(h0, n, beta, K, max_iter, deadline):
    h0 = make_feasible(h0, int(n))
    n = int(n)
    if deadline <= time.time():
        return None, None

    _SLQP.clear()
    _SLQP["active"] = True
    _SLQP["n"] = n
    _SLQP["beta"] = float(beta)
    _SLQP["K"] = int(K)
    _SLQP["target"] = 0.5 * float(n)
    _SLQP["deadline"] = float(deadline)
    _SLQP["best_h"] = h0.copy()
    _SLQP["best_c"] = compute_c5(h0)

    res = None
    try:
        bounds = [(0.0, 1.0)] * n
        cons = [{"type": "eq", "fun": slqp_eq, "jac": slqp_eq_jac}]
        res = minimize(
            slqp_obj,
            h0,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            jac=slqp_jac,
            options={"maxiter": int(max_iter), "ftol": 1e-12, "iprint": 0},
        )
    except Exception:
        res = None
    finally:
        _SLQP["active"] = False

    best_h = np.asarray(_SLQP.get("best_h", h0), dtype=float)
    best_h = make_feasible(best_h, n)
    best_c = compute_c5(best_h)
    return best_h, best_c


def lb_obj(x):
    p = _LB
    if p.get("active") and time.time() > p.get("deadline", 0.0):
        raise RuntimeError("lbfgs-timeout")
    if not np.all(np.isfinite(x)):
        return 1e100
    val, _, _ = smooth_components(x, p["beta"], p["K"])
    s = float(np.sum(x) - p["target"])
    return float(val + p["mu"] * s * s)


def lb_jac(x):
    p = _LB
    if p.get("active") and time.time() > p.get("deadline", 0.0):
        raise RuntimeError("lbfgs-timeout")
    _, meta, _ = smooth_components(x, p["beta"], p["K"])
    g = grad_weighted(x, meta, center=False)
    s = float(np.sum(x) - p["target"])
    g += 2.0 * p["mu"] * s
    return g


def run_lbfgs_smooth(h0, n, beta, K, mu, max_iter, deadline):
    h0 = make_feasible(h0, int(n))
    n = int(n)
    if deadline <= time.time():
        return None, None

    _LB.clear()
    _LB["active"] = True
    _LB["n"] = n
    _LB["beta"] = float(beta)
    _LB["K"] = int(K)
    _LB["mu"] = float(mu)
    _LB["target"] = 0.5 * float(n)
    _LB["deadline"] = float(deadline)

    res = None
    try:
        bounds = [(0.0, 1.0)] * n
        res = minimize(
            lb_obj,
            h0,
            method="L-BFGS-B",
            bounds=bounds,
            jac=lb_jac,
            options={
                "maxiter": int(max_iter),
                "ftol": 1e-12,
                "gtol": 1e-7,
                "maxfun": int(max(1500, 25 * max_iter)),
                "maxcor": 20,
                "maxls": 20,
                "disp": 0,
            },
        )
    except Exception:
        res = None
    finally:
        _LB["active"] = False

    if res is not None and hasattr(res, "x"):
        h = make_feasible(res.x, n)
        c = compute_c5(h)
        return h, c
    return None, None


def smooth_stage_fixed_meta(h, meta, n, deadline, alpha=0.10, max_iter=1000, exact_check=10):
    n = int(n)
    h = make_feasible(h, n)
    target = 0.5 * float(n)
    hb = h.copy()
    cb = compute_c5(hb)
    q = q_weighted(h, meta)
    alpha_cur = float(max(1e-5, alpha))
    stall = 0
    it = 0

    if max_iter <= 0 or len(meta) == 0:
        return hb, cb

    while it < int(max_iter) and time.time() < deadline:
        it += 1
        g = grad_weighted(h, meta, center=True)
        if not np.all(np.isfinite(g)):
            break

        q_old = q
        lam = alpha_cur
        accepted = False
        accepted_lam = 0.0

        for _ls in range(8):
            ht = project_box_sum(h - lam * g, target, iters=40)
            mov = float(np.max(np.abs(ht - h)))
            if mov < 1e-15:
                lam *= 0.5
                continue
            qt = q_weighted(ht, meta)
            if qt <= q_old + 1e-12:
                h = ht
                q = qt
                accepted_lam = lam
                accepted = True
                break
            lam *= 0.5

        if not accepted:
            alpha_cur *= 0.5
            if alpha_cur < 1e-8:
                break
            continue

        if q < q_old - 1e-10 and accepted_lam > 0.5 * alpha_cur:
            alpha_cur = min(2.0, alpha_cur * 1.10)
        elif accepted_lam < 0.25 * alpha_cur:
            alpha_cur = max(1e-7, accepted_lam)

        if (it % int(max(1, exact_check))) == 0 or it == int(max_iter):
            ct = compute_c5(h)
            if ct < cb - 1e-14:
                hb = h.copy()
                cb = ct
                stall = 0
            else:
                stall += int(max(1, exact_check))

            if ct > cb + 3e-4:
                h = hb.copy()
                q = q_weighted(h, meta)
                alpha_cur = max(1e-6, 0.7 * alpha_cur)
                stall = 0
            elif stall > 500:
                rp = np.random.RandomState(987654 + it)
                h = make_feasible(h + rp.normal(0.0, 0.02, n), n)
                q = q_weighted(h, meta)
                stall = 0

    ct = compute_c5(h)
    if ct < cb - 1e-14:
        hb = h.copy()
        cb = ct
    return hb, cb


def active_pair_local(h, n, deadline, max_candidates=500, max_passes=5):
    n = int(n)
    h = make_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h)

    for _pass in range(max(1, int(max_passes))):
        if time.time() >= deadline:
            break

        c = c_raw(h)
        cur_c = float(np.max(c) * (2.0 / float(n)))
        if cur_c < best_c - 1e-15:
            best_c = cur_c
            best_h = h.copy()

        order = np.argsort(c, kind="mergesort")[::-1]
        active = order[: min(16, len(order))]
        cand = {}

        for t in active:
            k = int(t) - (n - 1)
            if k == 0:
                continue
            d = abs(k)
            if d >= n:
                continue
            if k > 0:
                src = h[d:]
                dst = h[:-d]
            else:
                src = h[:-d]
                dst = h[d:]
            pot = src - dst
            mask = pot > 0.03
            if not np.any(mask):
                continue
            pos = np.nonzero(mask)[0]
            vals = pot[pos]
            L = max(8, max_candidates // 6)
            if vals.size > L:
                part = np.argpartition(-vals, L)[:L]
                top = pos[part]
            else:
                top = pos
            for ii in top:
                pp = float(pot[ii])
                if pp < 0.03:
                    continue
                if k > 0:
                    s = int(ii + d)
                    ds = int(ii)
                else:
                    s = int(ii)
                    ds = int(ii + d)
                key = (s, ds)
                old = cand.get(key)
                if old is None or pp > old:
                    cand[key] = pp

        _, meta, _ = smooth_components(h, 16.0, 64)
        g = grad_weighted(h, meta, center=True)
        gs = float(np.max(np.abs(g)))
        if gs > 1e-12:
            g = g / gs
            mg = min(50, n // 2)
            if mg > 0:
                og = np.argsort(g, kind="mergesort")
                lows = og[:mg]
                highs = og[::-1][:mg]
                for ii in highs:
                    gi = float(g[ii])
                    for jj in lows:
                        if ii == jj:
                            continue
                        gj = float(g[jj])
                        score = gi - gj
                        if score > 0.05:
                            key = (int(ii), int(jj))
                            old = cand.get(key)
                            if old is None or score > old:
                                cand[key] = score

        if not cand:
            break

        cand_list = []
        for key, val in cand.items():
            cand_list.append((float(val), int(key[0]), int(key[1])))
        cand_list.sort(reverse=True)

        improved = False
        tested = 0
        for _score, s, ds in cand_list[: int(max_candidates)]:
            if time.time() >= deadline:
                break
            if s == ds:
                continue
            maxd = min(float(h[s]), 1.0 - float(h[ds]))
            if maxd <= 1e-6:
                continue

            old_s = h[s]
            old_d = h[ds]
            local_best = best_c
            local_delta = 0.0

            for f in (0.01, 0.02, 0.05, 0.10, 0.20, 0.35, 0.50, 0.75, 0.95):
                delta = f * maxd
                if delta <= 1e-7 or delta > maxd + 1e-12:
                    continue
                h[s] = old_s - delta
                h[ds] = old_d + delta
                ct = compute_c5(h)
                if ct < local_best - 1e-14:
                    local_best = ct
                    local_delta = delta
                h[s] = old_s
                h[ds] = old_d

            if local_delta > 0.0:
                h[s] = old_s - local_delta
                h[ds] = old_d + local_delta
                best_c = local_best
                best_h = h.copy()
                improved = True

            tested += 1

        if not improved:
            break

    return best_h, best_c


def random_pair_local(h, n, deadline, seed=0, base_step=0.05):
    n = int(n)
    h = make_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h)
    rng = np.random.RandomState(int(seed) + n)
    step = float(base_step)
    no_improve = 0

    while time.time() < deadline:
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        delta = float(rng.uniform(-step, step))
        old_i = h[i]
        old_j = h[j]
        ni = old_i + delta
        nj = old_j - delta
        if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
            continue
        h[i] = ni
        h[j] = nj
        ct = compute_c5(h)
        if ct < best_c - 1e-14:
            best_c = ct
            best_h = h.copy()
            no_improve = 0
        else:
            h[i] = old_i
            h[j] = old_j
            no_improve += 1
            if no_improve >= 120:
                no_improve = 0
                step *= 0.88
                if step < 1e-5:
                    break

    return best_h, best_c


def continuous_pipeline(h0, n, deadline, seed, allow_slqp=True):
    n = int(n)
    h0 = make_feasible(h0, n)
    hb = h0.copy()
    cb = compute_c5(hb)

    if time.time() < deadline - 30 and allow_slqp and n <= 256:
        rem = deadline - time.time()
        slqp_end = min(time.time() + max(60.0, 0.30 * rem), deadline - 30)
        cycles = [
            (1.0, min(2 * n - 1, 256), 80),
            (6.0, min(2 * n - 1, 160), 100),
            (32.0, min(2 * n - 1, 100), 120),
            (160.0, min(2 * n - 1, 64), 120),
        ]
        for beta, K, mx in cycles:
            if time.time() >= slqp_end:
                break
            stage_dl = min(time.time() + 35.0, slqp_end)
            try:
                h2, c2 = run_slqp_smooth(hb, n, beta, K, mx, stage_dl)
            except Exception:
                h2, c2 = None, None
            if h2 is not None and c2 < cb - 1e-15:
                hb = h2
                cb = c2

    if time.time() < deadline - 40 and n <= 1024:
        rem = deadline - time.time()
        lb_end = min(time.time() + max(50.0, 0.20 * rem), deadline - 40)
        if n <= 384:
            cycles = [
                (2.0, min(2 * n - 1, 160), 80.0, 180),
                (20.0, min(2 * n - 1, 96), 250.0, 160),
                (120.0, min(2 * n - 1, 64), 700.0, 130),
            ]
        else:
            cycles = [
                (10.0, 96, 120.0, 140),
                (80.0, 64, 300.0, 100),
            ]
        for beta, K, mu, mx in cycles:
            if time.time() >= lb_end:
                break
            stage_dl = min(time.time() + 30.0, lb_end)
            try:
                h2, c2 = run_lbfgs_smooth(hb, n, beta, K, mu, mx, stage_dl)
            except Exception:
                h2, c2 = None, None
            if h2 is not None and c2 < cb - 1e-15:
                hb = h2
                cb = c2

    if time.time() < deadline - 35:
        if n <= 300:
            cycles = [
                (1.0, min(2 * n - 1, 160), 0.15, 800),
                (8.0, min(2 * n - 1, 120), 0.10, 650),
                (40.0, min(2 * n - 1, 96), 0.06, 520),
                (160.0, min(2 * n - 1, 64), 0.04, 420),
                (650.0, min(2 * n - 1, 36), 0.02, 320),
            ]
        elif n <= 700:
            cycles = [
                (8.0, min(2 * n - 1, 112), 0.10, 520),
                (40.0, min(2 * n - 1, 80), 0.06, 420),
                (180.0, min(2 * n - 1, 56), 0.035, 320),
                (700.0, min(2 * n - 1, 32), 0.018, 240),
            ]
        else:
            cycles = [
                (40.0, 72, 0.08, 360),
                (220.0, 48, 0.045, 260),
                (900.0, 28, 0.02, 180),
            ]

        for cycle_idx in range(2):
            for ci, cyc in enumerate(cycles):
                if time.time() >= deadline - 35:
                    break
                beta, K, alpha, mx = cyc
                stage_dl = min(
                    time.time() + min(120.0, max(20.0, (deadline - time.time() - 35) / 3.0)),
                    deadline - 35,
                )
                start_h = hb.copy()
                if cycle_idx == 1:
                    rp = np.random.RandomState(int(seed) + 777 + n + 17 * ci)
                    if ci == len(cycles) - 1:
                        start_h = threshold_quantile(hb, n)
                    else:
                        start_h = make_feasible(hb + rp.normal(0.0, 0.02, n), n)

                _, meta, _ = smooth_components(start_h, beta, K)
                h2, c2 = smooth_stage_fixed_meta(start_h, meta, n, stage_dl, alpha, mx)
                if c2 < cb - 1e-15:
                    hb = h2
                    cb = c2

    if time.time() < deadline - 20:
        if n <= 256:
            hpol, cpol = active_pair_local(
                hb,
                n,
                min(time.time() + 90.0, deadline - 12.0),
                max_candidates=520,
                max_passes=4,
            )
        else:
            hpol, cpol = active_pair_local(
                hb,
                n,
                min(time.time() + 55.0, deadline - 12.0),
                max_candidates=180,
                max_passes=2,
            )
        if cpol < cb - 1e-15:
            hb = hpol
            cb = cpol

    if time.time() < deadline - 8:
        hr, cr = random_pair_local(hb, n, min(time.time() + 25.0, deadline - 5.0), int(seed) + 1234)
        if cr < cb - 1e-15:
            hb = hr
            cb = cr

    hb = make_feasible(hb, n)
    cb = compute_c5(hb)
    return hb, cb


def run(seed=42, budget_s=1300.0, **kwargs):
    t_start = time.time()
    budget_s = float(budget_s)
    total_deadline = t_start + max(60.0, budget_s - 5.0)
    rng = np.random.RandomState(int(seed))

    h_parent = None
    try:
        hp = np.asarray(initial_h_values, dtype=float)
        if hp is not None and hp.size > 0:
            h_parent = hp
    except Exception:
        h_parent = None

    if h_parent is not None:
        best_n = int(len(h_parent))
        best_h = make_feasible(h_parent, best_n)
        best_c = compute_c5(best_h)
        if best_n < 32:
            best_n = 160
            best_h = make_feasible(np.full(best_n, 0.5, dtype=float), best_n)
            best_c = compute_c5(best_h)
    else:
        best_n = 160
        best_h, best_c = structured_start(best_n, rng)

    # Quick polish at initial resolution.
    quick_deadline = min(time.time() + 80.0, total_deadline - 320.0)
    if quick_deadline > time.time() + 10.0:
        try:
            hq, cq = active_pair_local(
                best_h,
                best_n,
                quick_deadline,
                max_candidates=260 if best_n <= 300 else 120,
                max_passes=3,
            )
            if cq < best_c - 1e-15:
                best_h, best_c = hq, cq
        except Exception:
            pass

    # Main work at the current resolution.
    main_deadline = min(t_start + max(360.0, 0.55 * (budget_s - 20.0)), total_deadline - 260.0)
    try:
        h_main, c_main = continuous_pipeline(best_h, best_n, main_deadline, int(seed) + 1, allow_slqp=True)
        if c_main < best_c - 1e-15:
            best_h, best_c, best_n = h_main, c_main, best_n
    except Exception:
        pass

    # Resolution probes.
    side_deadline = min(t_start + max(600.0, 0.73 * (budget_s - 20.0)), total_deadline - 130.0)
    base = best_n if best_n % 2 == 0 else best_n + 1
    if base <= 192:
        try_ns = [base + 32, base + 64, 256, 320, 384, 512]
    elif base <= 320:
        try_ns = [384, 448, 512, 640, 800]
    else:
        try_ns = [max(192, base - 64), base + 128, base + 256]

    seen_ns = set()
    for n_try in try_ns:
        if time.time() >= side_deadline:
            break
        n_try = int(n_try)
        if n_try % 2:
            n_try += 1
        if n_try < 96 or n_try > 800 or n_try in seen_ns or n_try == best_n:
            continue
        seen_ns.add(n_try)
        stage_deadline = min(time.time() + 110.0, side_deadline)
        if stage_deadline <= time.time() + 15.0:
            continue
        try:
            init_h = interpolate_to_n(best_h, n_try)
            h_r, c_r = continuous_pipeline(init_h, n_try, stage_deadline, int(seed) + n_try, allow_slqp=True)
            if c_r < best_c - 1e-15:
                best_h, best_c, best_n = h_r, c_r, n_try
        except Exception:
            pass

    # Long final polish.
    if time.time() < total_deadline - 30:
        if best_n <= 256:
            h_final, c_final = active_pair_local(
                best_h,
                best_n,
                total_deadline - 18.0,
                max_candidates=720,
                max_passes=8,
            )
        else:
            h_final, c_final = active_pair_local(
                best_h,
                best_n,
                total_deadline - 18.0,
                max_candidates=220,
                max_passes=3,
            )
        if c_final < best_c - 1e-15:
            best_h, best_c = h_final, c_final

    if time.time() < total_deadline - 10:
        try:
            h_rand, c_rand = random_pair_local(
                best_h,
                best_n,
                total_deadline - 8.0,
                int(seed) + 31337,
                base_step=0.03,
            )
            if c_rand < best_c - 1e-15:
                best_h, best_c = h_rand, c_rand
        except Exception:
            pass

    best_h = make_feasible(best_h, best_n)
    best_c = compute_c5(best_h)
    return best_h, float(best_c), int(best_n)
