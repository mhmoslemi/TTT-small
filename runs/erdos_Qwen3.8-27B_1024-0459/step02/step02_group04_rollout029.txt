        ac = float(np.dot(h[:m], h[L:]))
        vals[n - 1 + L] = ((total - pref[L]) - ac) * dx
        vals[n - 1 - L] = (pref[m] - ac) * dx

    vmax = float(np.max(vals))
    shifted = beta * (vals - vmax)
    shifted = np.maximum(shifted, -700.0)
    exps = np.exp(shifted)
    s = float(np.sum(exps))
    logsum = vmax + np.log(s) / beta
    w = exps / s

    g = np.zeros(n, dtype=float)
    w0 = w[n - 1]
    if w0 > 1e-18:
        g += w0 * (-2.0 * h)

    for L in range(1, n):
        m = n - L
        wp = w[n - 1 + L]
        wm = w[n - 1 - L]
        if wp > 1e-18 or wm > 1e-18:
            if wm > 1e-18:
                g[:m] += wm * (1.0 - h[L:])
                g[L:] += -wm * h[:m]
            if wp > 1e-18:
                g[L:] += wp * (1.0 - h[:m])
                g[:m] += -wp * h[L:]

    return logsum, g


def adam_optimize(h0, n, time_limit, seed, lr0=0.002, phases=None, beta_start=5.0, beta_end=500.0):
    rng = np.random.RandomState(int(seed))
    h = project_feasible(h0, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c

    if phases is None:
        phases = [(3.0, 15.0, 0.003), (15.0, 60.0, 0.0015), (60.0, 200.0, 0.0008), (200.0, 800.0, 0.0004)]

    nph = len(phases)
    for pi in range(nph):
        if time.time() >= time_limit:
            break
        b_lo, b_hi, lr = phases[pi]
        alpha1, alpha2 = 0.9, 0.999
        eps = 1e-8
        m_buf = np.zeros(n, dtype=float)
        v_buf = np.zeros(n, dtype=float)
        t = 0

        phase_start = time.time()
        phase_dur = max(1.0, time_limit - time.time()) / float(nph - pi)
        phase_end = phase_start + phase_dur

        while time.time() < phase_end:
            t += 1
            beta = b_lo + (b_hi - b_lo) * min(1.0, float(t) / max(100.0, phase_dur / 0.001))
            beta = min(beta, float(beta_end))

            _, g = smooth_max_grad(h, beta)
            lr_t = lr / (1.0 + 0.001 * float(t))

            m_buf = alpha1 * m_buf + (1.0 - alpha1) * g
            v_buf = alpha2 * v_buf + (1.0 - alpha2) * g * g
            m_hat = m_buf / (1.0 - alpha1 ** t)
            v_hat = v_buf / (1.0 - alpha2 ** t)
            step = lr_t * m_hat / (np.sqrt(v_hat) + eps)

            h_try = h - step
            h_try = project_feasible(h_try, n)

            if t % 500 == 0:
                c_try = compute_c5(h_try)
            else:
                c_try = c

            if c_try < best_c - 1e-15:
                best_h = h_try.copy()
                best_c = c_try
                c = c_try
            h = h_try

            if t % 5000 == 0 and t > 100:
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h)
    return best_h, best_c


def interpolate_to_n(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float)
    n_dst = int(n_dst)
    n_src = h_src.size
    if n_src == n_dst:
        return h_src.copy()
    if n_src <= 1:
        val = float(h_src[0]) if n_src == 1 else 0.5
        return np.full(n_dst, val, dtype=float)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)


def block_perturb(h, n, time_limit, seed, max_trials=20000):
    rng = np.random.RandomState(int(seed))
    h = project_feasible(h, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    cur_h = h
    cur_c = c
    temp = 0.001
    min_temp = 1e-7
    max_block = min(12, n)

    for _ in range(max_trials):
        if time.time() > time_limit:
            break
        L = int(rng.randint(1, max_block + 1))
        st = int(rng.randint(0, n - L + 1))
        idx = np.arange(st, st + L)
        delta = rng.randn(L) * 0.015
        delta -= delta.mean()
        h_try = cur_h.copy()
        h_try[idx] += delta
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try)

        if c_try < best_c - 1e-15:
            best_h = h_try.copy()
            best_c = c_try
            cur_h = h_try
            cur_c = c_try
            temp = min(0.015, max(temp * 1.1, min_temp))
        elif c_try < cur_c:
            cur_h = h_try
            cur_c = c_try
        else:
            d = c_try - cur_c
            if 0.0 < d < temp:
                if rng.rand() < np.exp(-d / temp):
                    cur_h = h_try
                    cur_c = c_try
        temp = max(temp * 0.999, min_temp)

    return best_h, float(best_c)


def coordinate_refine(h, n, time_limit, seed):
    rng = np.random.RandomState(int(seed))
    h = project_feasible(h, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    order = rng.permutation(n)

    for _ in range(5):
        if time.time() > time_limit:
            break
        for _ in range(2):
            if time.time() > time_limit:
                break
            improved = False
            for j in order:
                orig = h[j]
                for trial_val in (orig + 0.1, orig - 0.1, orig + 0.05, orig - 0.05, orig + 0.01, orig - 0.01):
                    if trial_val < 0.0 or trial_val > 1.0:
                        continue
                    h[j] = trial_val
                    h = project_feasible(h, n)
                    c_try = compute_c5(h)
                    if c_try < c - 1e-15:
                        c = c_try
                        improved = True
                        break
                    else:
                        h[j] = orig
                if not improved:
                    h[j] = orig
            h = project_feasible(h, n)
            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()

    return best_h, float(best_c)
