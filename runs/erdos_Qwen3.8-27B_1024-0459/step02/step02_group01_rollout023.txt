    target = n / 2.0

        else:
            idx = np.argmax(h)

        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmax(h)
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_to(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return project_feasible(h, new_n)
    x_old = (np.arange(old_n) + 0.5) * 2.0 / old_n
    x_new = (np.arange(new_n) + 0.5) * 2.0 / new_n
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, new_n)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smoothmax_weights(corr, beta):
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    shifted = np.clip(shifted, -500.0, 0.0)
    weights = np.exp(shifted)
    sw = np.sum(weights)
    if sw < 1e-300:
        weights = np.zeros_like(corr)
        weights[np.argmax(corr)] = 1.0
        sw = 1.0
    else:
        weights /= sw
    smax = cmax + np.log(sw) / beta
    return smax, weights

def smoothmax_grad(h, n, weights):
    dx = 2.0 / n
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for m in active:
        k = m - (n - 1)
        w = weights[m]
        if k >= 0:
            if k == 0:
                grad += w * (1.0 - 2.0 * h)
            else:
                grad[k:] += w * (1.0 - h[:n - k])
                grad[:n - k] -= w * h[k:]
        else:
            o = -k
            grad[:n - o] += w * (1.0 - h[o:])
            grad[o:] -= w * h[:n - o]
    return grad * dx

def sgd_optimize(h0, n, time_limit_s, seed=42):
    t_start = time.time()
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n

    stages = [
        (5.0, 0.4, 0.12),
        (20.0, 0.3, 0.10),
        (100.0, 0.2, 0.10),
        (500.0, 0.14, 0.10),
        (2000.0, 0.10, 0.10),
        (8000.0, 0.07, 0.10),
        (30000.0, 0.05, 0.10),
        (100000.0, 0.04, 0.12),
        (500000.0, 0.03, 0.16),
    ]

    for stage_i, (beta, base_lr, time_frac) in enumerate(stages):
        if time.time() - t_start > time_limit_s - 2:
            break
        stage_time = time_frac * time_limit_s
        lr = base_lr
        v = np.zeros(n)
        momentum = 0.92
        max_iters = 5000

        for it in range(max_iters):
            if time.time() - t_start > time_limit_s - 2:
                break
            if time.time() - t_start > t_start + stage_time:
                break

            corr = compute_corr(h)
            smax, weights = smoothmax_weights(corr, beta)
            grad = smoothmax_grad(h, n, weights)

            v = momentum * v - lr * grad
            h_new = h + v
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_feasible(h_new, n)
            c5_new = compute_c5(h_new)

            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

            h = h_new
            if it % 200 == 199:
                lr *= 0.9
            if it % 1000 == 999:
                v *= 0.5

    return best_h, best_c5

def toptk_obj(x, n, top_idx):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return np.mean(corr[top_idx]) * (2.0 / n)

def toptk_slsqp(h0, n, time_limit_s):
    t_start = time.time()
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    corr = compute_corr(h)

    for k_top in [5, 3, 2]:
        if time.time() - t_start > time_limit_s - 5:
            break
        top_idx = np.argsort(corr)[-k_top:]
        top_idx = list(top_idx)
        bounds = [(0.0, 1.0)] * n

        def obj_fn(x, idx=top_idx):
            c = np.correlate(x, 1.0 - x, mode='full')
            return np.mean(c[idx]) * (2.0 / n)

        def eq_fun(x):
            return np.sum(x) - n / 2.0

        cons = [{'type': 'eq', 'fun': eq_fun}]
        try:
            res = minimize(obj_fn, h, method='SLSQP', bounds=bounds,
                         constraints=cons, options={'maxiter': 400, 'ftol': 1e-14, 'iprint': 0})
            h = project_feasible(res.x, n)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        except Exception:
            pass

    return best_h, best_c5

def slsqp_smooth(h0, n, time_limit_s):
    t_start = time.time()
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    global _slq_n, _slq_beta

    def obj_fn(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        sh = _slq_beta * (c - m)
        sh = np.clip(sh, -500.0, 0.0)
        smax = m + np.log(np.sum(np.exp(sh))) / _slq_beta
        return smax * (2.0 / _slq_n)

    def eq_fun(x):
        return np.sum(x) - _slq_n / 2.0

    cons = [{'type': 'eq', 'fun': eq_fun}]

    for beta in [1000.0, 5000.0, 20000.0, 80000.0]:
        if time.time() - t_start > time_limit_s - 3:
            break
        _slq_n = n
        _slq_beta = beta
        try:
            res = minimize(obj_fn, h, method='SLSQP', bounds=bounds,
                         constraints=cons, options={'maxiter': 300, 'ftol': 1e-14, 'iprint': 0})
            h = project_feasible(res.x, n)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        except Exception:
            pass

    return best_h, best_c5

def coordinate_descent(h0, n, time_limit_s):
    t_start = time.time()
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng_cd = np.random.RandomState(12345)
    dx = 2.0 / n

    for sweep in range(100000):
        if time.time() - t_start > time_limit_s - 2:
            break
        idx = rng_cd.randint(n)
        for delta in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02]:
            h[idx] += delta
            h[idx] = np.clip(h[idx], 0.0, 1.0)
            h[idx2] = h[idx]
            # Adjust one other element to maintain sum
            diff = np.sum(h) - n / 2.0
            if abs(diff) > 1e-12:
                # Spread correction
                others = np.arange(n)
                others = others[others != idx]
                h[others] -= diff / (n - 1)
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, n)
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h.copy()
            else:
                h[idx] -= delta
                break
        h = best_h.copy()

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        ph = np.array(initial_h_values, dtype=float)
        pn = len(ph)
        if pn > 0:
            best_h = ph.copy()
            best_c5 = compute_c5(ph)
            best_n = pn
    except Exception:
        pass

    if best_h is None:
        best_n = 192
        best_h = np.ones(best_n) * 0.5
        best_c5 = compute_c5(best_h)

    def tl():
        return max(0.0, budget_s - (time.time() - t0) - 5.0)

    # Phase 1: SGD at higher resolutions (main workhorse)
    target_res = [384, 480, 576]
    # Include parent's resolution
    if best_n not in target_res and best_n >= 128:
        target_res.insert(0, best_n)
    target_res = sorted(set([rr for rr in target_res if 96 <= rr <= 720]))

    # Allocate time: 60% to phase 1, 15% to phase 2 (topk+slsqp), 20% to phase 3 (high-res), 5% reserve
    phase1_time = 0.55 * budget_s
    for n in target_res:
        if tl() < phase1_time / len(target_res) + 30:
            if tl() < 40:
                break
        time_alloc = min(120.0, max(20.0, tl() * 0.35))

        # Start from upsampled best
        h_start = upsample_to(best_h, n)
        h_opt, c5_opt = sgd_optimize(h_start, n, time_alloc, seed=seed + n)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n

        # Second start: perturbed
        if tl() > 30:
            hp = h_start + 0.05 * rng.randn(n)
            hp = project_feasible(hp, n)
            time_alloc2 = min(60.0, tl() * 0.15)
            h_opt2, c5_opt2 = sgd_optimize(hp, n, time_alloc2, seed=seed + n + 777)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2.copy()
                best_n = n

    # Phase 2: Top-k SLSQP + smooth SLSQP polish at best resolution
    if tl() > 40:
        time_p2 = min(200.0, tl() * 0.35)
        h_tc, c5_tc = toptk_slsqp(best_h, best_n, time_p2 * 0.5)
        if c5_tc < best_c5:
            best_c5 = c5_tc
            best_h = h_tc.copy()

        if tl() > 20:
            h_sm, c5_sm = slsqp_smooth(best_h, best_n, min(100.0, tl() * 0.25))
            if c5_sm < best_c5:
                best_c5 = c5_sm
                best_h = h_sm.copy()

    # Phase 3: Try one more higher resolution
    if tl() > 60:
        n_high = best_n + 128
        n_high = min(n_high, 768)
        if n_high > best_n:
            h_high = upsample_to(best_h, n_high)
            time_p3 = min(150.0, tl() * 0.35)
            h_opt, c5_opt = sgd_optimize(h_high, n_high, time_p3 * 0.6, seed=seed + 9999)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_high

            if tl() > 30:
                h_tc, c5_tc = toptk_slsqp(best_h, best_n, min(60.0, tl() * 0.2))
                if c5_tc < best_c5:
                    best_c5 = c5_tc
                    best_h = h_tc.copy()

    # Phase 4: Final SLSQP polish
    if tl() > 15:
        h_final, c5_final = slsqp_smooth(best_h, best_n, min(60.0, tl() * 0.5))
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Phase 5: Binary rounding
    if tl() > 3:
        h_bin = (best_h > 0.5).astype(float)
        n_ones = int(np.round(np.sum(h_bin)))
        n_want = best_n // 2
        if n_ones > n_want:
            ones_idx = np.where(h_bin == 1)[0]
            order = np.argsort(best_h[ones_idx])[:n_ones - n_want]
            h_bin[ones_idx[order]] = 0.0
        elif n_ones < n_want:
            zeros_idx = np.where(h_bin == 0)[0]
            order = np.argsort(-best_h[zeros_idx])[:n_want - n_ones]
            h_bin[zeros_idx[order]] = 1.0
        c5b = compute_c5(h_bin)
        if c5b < best_c5:
            best_c5 = c5b
            best_h = h_bin.copy()

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
