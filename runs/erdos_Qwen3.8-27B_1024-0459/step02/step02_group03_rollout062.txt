    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    vals = np.exp(beta * (corr - m))
    sm = m + np.log(np.sum(vals)) / beta
    penalty = mu * (np.sum(x) - n / 2.0) ** 2
    return sm * dx + penalty

def smooth_max_grad(x, n, beta, mu):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    vals = np.exp(beta * (corr - m))
    weights = vals / np.sum(vals)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        L = k - (n - 1)
        if L == 0:
            grad += w * (1.0 - 2.0 * x)
        elif L > 0:
            grad[L:] += w * (1.0 - x[:n - L])
            grad[:n - L] -= w * x[L:]
        else:
            Lp = -L
            grad[:n - Lp] += w * (1.0 - x[Lp:])
            grad[Lp:] -= w * x[:n - Lp]
    
    grad *= dx
    grad += 2.0 * mu * (np.sum(x) - n / 2.0)
    return grad

def lbfgs_optimize(h0, n, beta, mu, max_iter=500):
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        smooth_max_obj, h0.copy(),
        jac=lambda x: smooth_max_grad(x, n, beta, mu),
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-10}
    )
    h_out = project_feasible(res.x, n)
    return h_out, compute_c5(h_out)

def coordinate_descent(h, n, max_sweeps=50, mu=100.0):
    h = h.copy()
    for sweep in range(max_sweeps):
        h = project_feasible(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full')
        
        for i in range(n):
            old_val = h[i]
            if old_val <= 0 or old_val >= 1:
                # Check if we can move
                pass
            
            # Build 1D function
            # Affected corr indices and their (a, b) coefficients
            # corr_with_t[k] = base_k + a_k * t for affected k, corr[k] otherwise
            
            # I'll compute the 1D function by evaluating corr at a few t values
            # Actually, let me build (a, b) for each affected k
            
            affected_a = {}  # k -> a_k
            affected_b = {}  # k -> b_k (base value)
            
            # corr[n-1+L] for L = 0, ..., n-1-i: h[i] as first factor
            for L in range(n - i):
                k = n - 1 + L
                coeff = 1.0 - h[i + L]
                # corr[k] currently has old_val * coeff as part of its value
                # New value: corr[k] - old_val*coeff + t*coeff = (corr[k] - old_val*coeff) + t*coeff
                if k not in affected_a:
                    affected_a[k] = 0.0
                    affected_b[k] = corr[k]
                affected_a[k] += coeff
                affected_b[k] -= 0  # already accounted for in the subtraction below
            
            # Wait, I need to be more careful. Let me redo.
            # Current corr[k] includes the term old_val * (1-h[i+L]) (if i <= n-1-L)
            # New corr[k] replaces old_val with t: corr[k] + (t - old_val) * (1-h[i+L])
            # So a_k += (1-h[i+L]), and the base is corr[k] (with old_val still in it)
            # corr_new[k] = corr[k] + (t - old_val) * a_k_total
            # = (corr[k] - old_val * a_k_total) + t * a_k_total
            
            # Let me accumulate delta_a[k] = sum of coefficients of t at index k
            # Then corr_new[k] = corr[k] + (t - old_val) * delta_a[k]
            # = (corr[k] - old_val * delta_a[k]) + t * delta_a[k]
            
            delta_a = np.zeros(2 * n - 1)
            
            # h[i] as first factor in corr[n-1+L], L = 0 to n-1-i
            for L in range(n - i):
                k = n - 1 + L
                delta_a[k] += 1.0 - h[i + L]
            
            # h[i] as second factor in corr[n-1+L], L = 1 to i
            for L in range(1, i + 1):
                k = n - 1 + L
                delta_a[k] -= h[i - L]
            
            # h[i] as first factor (h[j+M]) in corr[n-1-M], M = 1 to min(i, n-1)
            for M in range(1, min(i, n - 1) + 1):
                k = n - 1 - M
                delta_a[k] += 1.0 - h[i - M]
            
            # h[i] as second factor (h[j]) in corr[n-1-M], M = 1 to min(n-1-i, n-1)
            for M in range(1, min(n - 1 - i, n - 1) + 1):
                k = n - 1 - M
                delta_a[k] -= h[i + M]
            
            # Now corr_new[k] = corr[k] + (t - old_val) * delta_a[k] for all k
            # f(t) = max_k (corr[k] + (t - old_val) * delta_a[k])
            # = max_k (corr[k] - old_val * delta_a[k] + t * delta_a[k])
            
            # Precompute base values
            base = corr - old_val * delta_a  # base[k] = corr[k] - old_val * delta_a[k]
            
            # f(t) = max(base + t * delta_a)
            # This is max of affine functions of t
            
            def f_of_t(t):
                vals = base + t * delta_a
                sm = np.max(vals)
                # Penalty for sum
                s = np.sum(h) - old_val + t
                pen = mu * (s - n / 2.0) ** 2
                return sm + pen
            
            # Minimize
            res = minimize_scalar(f_of_t, bounds=(0.0, 1.0), method='bounded',
                                  options={'maxiter': 30, 'xatol': 1e-8})
            new_val = float(np.clip(res.x, 0.0, 1.0))
            
            h[i] = new_val
            # Update corr for next coordinate
            corr += (new_val - old_val) * delta_a
            old_val = new_val
        
        h = project_feasible(h, n)
    
    return h

def binary_local_search(h_bin, n, max_rounds=5000, rng=None):
    """Greedy binary local search: try swapping a 1 and a 0, accept if C5 improves."""
    if rng is None:
        rng = np.random.RandomState(0)
    h = h_bin.copy()
    c5 = compute_c5(h)
    
    for round in range(max_rounds):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(ones) == n:
            break
        
        improved = False
        for _ in range(50):
            i1 = rng.choice(ones)
            i0 = rng.choice(zeros)
            h[i1] = 0.0
            h[i0] = 1.0
            new_c5 = compute_c5(h)
            if new_c5 < c5 - 1e-12:
                c5 = new_c5
                improved = True
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
            else:
                h[i1] = 1.0
                h[i0] = 0.0
            if improved:
                break
        
        if not improved:
            break
    
    return h, c5

def simulated_annealing(h_bin, n, rng, t_start=0.5, t_end=0.01, n_iters=5000):
    """Simulated annealing on binary patterns."""
    h = h_bin.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    for it in range(n_iters):
        T = t_start * (t_end / t_start) ** (it / n_iters)
        
        i1 = rng.choice(ones)
        i0 = rng.choice(zeros)
        
        h[i1] = 0.0
        h[i0] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - c5
        
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            c5 = new_c5
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i1] = 1.0
            h[i0] = 0.0
    
    return best_h, best_c5

def make_sinusoidal_init(n, rng):
    """Create a sinusoidal-like initialization."""
    freq = rng.uniform(1, 8)
    phase = rng.uniform(0, 2 * np.pi)
    h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    return project_feasible(h, n)

def make_block_init(n, rng):
    """Create a block initialization."""
    h = np.zeros(n)
    n_ones = n // 2
    # Random blocks
    n_blocks = rng.randint(2, 7)
    sizes = rng.multinomial(n_ones, [1/n_blocks] * n_blocks)
    pos = 0
    for s in sizes:
        if s > 0:
            start = rng.randint(0, max(1, n - pos - s + 1))
            # Can't easily do non-overlapping random blocks
            pass
    # Simpler: split into alternating blocks
    h[:n_ones] = 1.0
    # Shuffle within constraint
    perm = rng.permutation(n)
    h2 = np.zeros(n)
    h2[perm[:n_ones]] = 1.0
    return h2
