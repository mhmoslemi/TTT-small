    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break

        mask = (h > 1e-10) & (h < 1.0 - 1e-10)

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(12):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    else:
        m = np.max(corr)
        shifted = corr - m
        return (m + np.log(np.sum(np.exp(beta * shifted))) / beta) * dx


def subgradient_c(h, beta, top_k=None):

    dx = 2.0 / n

    m = np.max(corr)
    if top_k is not None and top_k < len(corr):
        idx_all = np.argsort(corr)[-top_k:]
        work_idx = idx_all
        corr_work = corr[work_idx]
        mw = corr_work.max()
        sw = corr_work - mw
        w = np.exp(beta * sw)
        w /= np.sum(w)
    else:
        work_idx = np.arange(len(corr))
        sw = corr - m
        w = np.exp(beta * sw)
        w /= np.sum(w)
    
    grad = np.zeros(n)

        if w[t] < 1e-12:
            continue
        k = k_idx - (n - 1)
        if k >= 0:
            i_lo = k
            i_hi = n - 1
        else:
            i_lo = 0
            i_hi = n - 1 + k
        if i_lo > i_hi:
            continue
        m_vals = np.arange(i_lo, i_hi + 1)
        j_vals = m_vals - k
        grad[m_vals] += w[t] * (1.0 - h[j_vals])
        grad[j_vals] += w[t] * (-h[m_vals])
    return grad * dx
