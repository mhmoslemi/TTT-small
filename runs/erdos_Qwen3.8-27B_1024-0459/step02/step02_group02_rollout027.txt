def compute_c5_fast(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible_binary(h, n, rng):
    h = np.round(h).astype(float).copy()
    target = n // 2
    n_ones = int(np.sum(h))
    if n_ones > target:
        excess = n_ones - target
        ones_idx = np.where(h == 1.0)[0]
        to_flip = rng.choice(ones_idx, excess, replace=False)
        h[to_flip] = 0.0
    elif n_ones < target:
        deficit = target - n_ones
        zeros_idx = np.where(h == 0.0)[0]
        to_flip = rng.choice(zeros_idx, deficit, replace=False)
        h[to_flip] = 1.0

def upsample(h_src, n_dst, rng):

def compute_c5_fast(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible_binary(h, n, rng):
    h = np.round(h).astype(float).copy()
    target = n // 2
    n_ones = int(np.sum(h))
    if n_ones > target:
        excess = n_ones - target
        ones_idx = np.where(h == 1.0)[0]
        if excess > 0:
            to_flip = rng.choice(ones_idx, excess, replace=False)
            h[to_flip] = 0.0
    elif n_ones < target:
        deficit = target - n_ones
        zeros_idx = np.where(h == 0.0)[0]
        if deficit > 0:
            to_flip = rng.choice(zeros_idx, deficit, replace=False)
            h[to_flip] = 1.0
    return h

def upsample_bin(h_src, n_dst, rng):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_old = np.linspace(0, 2, n_src, endpoint=False)
    x_new = np.linspace(0, 2, n_dst, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_src)
    h_bin = np.round(h_interp)
    return make_feasible_binary(h_bin, n_dst, rng)

def gen_thue_morse(n, seed_idx=0):
    t = np.zeros(n)
    for i in range(n):
        t[i] = 1.0 if (bin(i + seed_idx * 7).count('1') % 2 == 0) else 0.0
    rng_tmp = np.random.RandomState(12345)
    return make_feasible_binary(t, n, rng_tmp)

def gen_random_bin(n, rng):
    h = (rng.rand(n) >= 0.5).astype(float)
    return make_feasible_binary(h, n, rng)

def gen_antisymmetric(n, rng):
    h = np.zeros(n)
    half = n // 2
    t = np.zeros(half)
    for i in range(half):
        t[i] = 1.0 if (bin(i * 3 + 11).count('1') % 2 == 0) else 0.0
    h[:half] = t
    h[half:] = 1.0 - t
    return make_feasible_binary(h, n, rng)

def local_search_binary(h, n, rng, max_iters=100000, timeout=40.0, stall_limit=8000):
    h = h.copy()
    best_c5 = compute_c5_fast(h)
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    no_improve = 0
    start_t = time.time()
    for it in range(max_iters):
        if time.time() - start_t > timeout:
            break
        if no_improve > stall_limit:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        c5 = compute_c5_fast(h)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            no_improve = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
    return h, best_c5

def sa_binary(h, n, rng, t_high=0.02, t_low=0.0005, max_iters=400000, timeout=100.0):
    h = h.copy()
    curr_c5 = compute_c5_fast(h)
    best_h = h.copy()
    best_c5 = curr_c5
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    start_t = time.time()
    for it in range(max_iters):
        if time.time() - start_t > timeout:
            break
        frac = it / max_iters
        temp = t_high * (t_low / t_high) ** frac
        if temp < 1e-8:
            temp = 1e-8
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5_fast(h)
        delta = new_c5 - curr_c5
        if delta <= 0 or rng.random() < np.exp(-delta / temp):
            curr_c5 = new_c5
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c5

def iterative_ls(h, n, rng, n_restarts=20, pert_size_range=(5, 15), timeout=120.0):
    h_best = h.copy()
    best_c5 = compute_c5_fast(h_best)
    h_curr = h.copy()
    start_t = time.time()
    for r in range(n_restarts):
        if time.time() - start_t > timeout:
            break
        h_curr, c5_curr = local_search_binary(h_curr, n, rng, max_iters=50000,
                                              timeout=min(15.0, timeout - (time.time() - start_t)),
                                              stall_limit=6000)
        if c5_curr < best_c5 - 1e-15:
            best_c5 = c5_curr
            h_best = h_curr.copy()
        else:
            h_curr = h_best.copy()
        if time.time() - start_t > timeout:
            break
        n_pert = rng.randint(pert_size_range[0], pert_size_range[1] + 1)
        ones = list(np.where(h_curr == 1.0)[0])
        zeros = list(np.where(h_curr == 0.0)[0])
        for _ in range(n_pert):
            if not ones or not zeros:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_curr[i] = 0.0
            h_curr[j] = 1.0
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
    return h_best, best_c5

def targeted_ls(h, n, rng, max_iters=80000, timeout=40.0, stall_limit=6000):
    h = h.copy()
    best_c5 = compute_c5_fast(h)
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    no_improve = 0
    start_t = time.time()
    for it in range(max_iters):
        if time.time() - start_t > timeout:
            break
        if no_improve > stall_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        worst_k = np.argmax(corr)
        worst_s = worst_k - (n - 1)
        bad_i = []
        lo = max(0, -worst_s)
        hi = min(n, n - worst_s)
        for i in range(lo, hi):
            if h[i] == 1.0 and h[i + worst_s] == 0.0:
                bad_i.append(i)
        if bad_i and len(bad_i) > 0:
            i = bad_i[rng.randint(len(bad_i))]
        else:
            i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        h[i] = 0.0
        h[j] = 1.0
        c5_try = compute_c5_fast(h)
        if c5_try < best_c5 - 1e-15:
            best_c5 = c5_try
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            no_improve = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
    return h, best_c5

def continuous_micro_polish(h, n, rng, max_iters=20000, timeout=20.0):
    h = h.copy()
    best_c5 = compute_c5_fast(h)
    target = n / 2.0
    start_t = time.time()
    for it in range(max_iters):
        if time.time() - start_t > timeout:
            break
        h_try = h.copy()
        n_mod = rng.randint(1, 4)
        idxs = rng.choice(n, n_mod, replace=False)
        for idx in idxs:
            if h_try[idx] > 0.5:
                h_try[idx] = max(0.5, h_try[idx] - rng.uniform(0.005, 0.05))
            else:
                h_try[idx] = min(0.5, h_try[idx] + rng.uniform(0.005, 0.05))
        diff = target - np.sum(h_try)
        if abs(diff) > 1e-8:
            mask = (h_try > 0.01) & (h_try < 0.99)
            cnt = np.count_nonzero(mask)
            if cnt > 0:
                h_try[mask] += diff / cnt
                h_try = np.clip(h_try, 0.0, 1.0)
            else:
                h_try = h.copy()
                continue
        c5_try = compute_c5_fast(h_try)
        if c5_try < best_c5 - 1e-13:
            h = h_try
            best_c5 = c5_try
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    np.random.seed(seed)
    start_time = time.time()

    def remaining():
        return budget_s - (time.time() - start_time)

    h_parent = None
    try:
        h_pf = np.array(initial_h_values, dtype=float)
        if h_pf is not None and len(h_pf) > 0:
            h_parent = h_pf.copy()
    except Exception:
        pass

    n = 512
    inits = []

    if h_parent is not None:
        inits.append(upsample_bin(h_parent, n, rng))

    inits.append(gen_thue_morse(n, 0))
    inits.append(gen_thue_morse(n, 1))
    inits.append(gen_thue_morse(n, 3))
    inits.append(gen_thue_morse(n, 7))
    inits.append(gen_antisymmetric(n, rng))
    for _ in range(4):
        inits.append(gen_random_bin(n, rng))

    best_h = inits[0].copy()
    best_c5 = compute_c5_fast(best_h)
    for h0 in inits[1:]:
        c5 = compute_c5_fast(h0)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h0.copy()

    # Phase 1: Local search from top inits
    for idx in range(min(7, len(inits))):
        if remaining() < 50:
            break
        h0 = inits[idx] if idx < len(inits) else best_h
        h_ls, c5_ls = local_search_binary(h0, n, rng, max_iters=40000,
                                          timeout=min(30.0, remaining() - 10), stall_limit=6000)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Phase 2: SA from best
    if remaining() > 80:
        h_sa, c5_sa = sa_binary(best_h, n, rng, t_high=0.025, t_low=0.0005,
                                max_iters=350000, timeout=min(100.0, remaining() - 20))
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 3: Local search after SA
    if remaining() > 50:
        h_ls2, c5_ls2 = local_search_binary(best_h, n, rng, max_iters=60000,
                                            timeout=min(35.0, remaining() - 10), stall_limit=8000)
        if c5_ls2 < best_c5:
            best_c5 = c5_ls2
            best_h = h_ls2.copy()

    # Phase 4: Targeted local search
    if remaining() > 50:
        h_t, c5_t = targeted_ls(best_h, n, rng, max_iters=60000,
                                timeout=min(40.0, remaining() - 10), stall_limit=5000)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()

    # Phase 5: Iterated local search with perturbation
    if remaining() > 100:
        h_ils, c5_ils = iterative_ls(best_h, n, rng, n_restarts=25,
                                     pert_size_range=(8, 20), timeout=min(90.0, remaining() - 20))
        if c5_ils < best_c5:
            best_c5 = c5_ils
            best_h = h_ils.copy()

    # Phase 6: Another SA with different temp schedule
    if remaining() > 80:
        h_sa2, c5_sa2 = sa_binary(best_h, n, rng, t_high=0.04, t_low=0.001,
                                  max_iters=300000, timeout=min(70.0, remaining() - 15))
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()

    # Phase 7: Targeted + local search
    if remaining() > 50:
        h_t2, c5_t2 = targeted_ls(best_h, n, rng, max_iters=50000,
                                  timeout=min(30.0, remaining() - 10), stall_limit=5000)
        if c5_t2 < best_c5:
            best_c5 = c5_t2
            best_h = h_t2.copy()
        if remaining() > 30:
            h_ls3, c5_ls3 = local_search_binary(best_h, n, rng, max_iters=40000,
                                                timeout=min(20.0, remaining() - 5), stall_limit=5000)
            if c5_ls3 < best_c5:
                best_c5 = c5_ls3
                best_h = h_ls3.copy()

    # Phase 8: Iterated local search again
    if remaining() > 80:
        h_ils2, c5_ils2 = iterative_ls(best_h, n, rng, n_restarts=30,
                                       pert_size_range=(5, 15), timeout=min(70.0, remaining() - 15))
        if c5_ils2 < best_c5:
            best_c5 = c5_ils2
            best_h = h_ils2.copy()

    # Phase 9: Continuous micro-polish
    if remaining() > 25:
        h_cp, c5_cp = continuous_micro_polish(best_h, n, rng, max_iters=15000,
                                              timeout=min(20.0, remaining() - 5))
        if c5_cp < best_c5:
            best_c5 = c5_cp
            best_h = h_cp.copy()

    # Phase 10: Final local search
    if remaining() > 15:
        h_final, c5_final = local_search_binary(best_h, n, rng, max_iters=30000,
                                                timeout=min(10.0, remaining() - 3), stall_limit=4000)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Ensure feasibility (binary, sum = n/2)
    best_h = make_feasible_binary(best_h, n, rng)
    best_c5 = compute_c5_fast(best_h)

    return (best_h, best_c5, n)
