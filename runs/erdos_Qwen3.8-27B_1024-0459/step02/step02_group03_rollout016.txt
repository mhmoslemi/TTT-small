def smooth_max_value(corr, beta):
    m = np.max(corr)

def compute_grad(h, beta):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max_value(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    total = np.sum(np.exp(shifted))
    return (m + np.log(total) / beta)

def smooth_max_weights(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    exps = np.exp(shifted)
    return exps / np.sum(exps)

def compute_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    weights = smooth_max_weights(corr, beta)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:
            continue
        L = k - (n - 1)
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += w * (1.0 - h[:n - L])
            grad[:n - L] -= w * h[L:]
        else:
            Lp = -L
            grad[:n - Lp] += w * (1.0 - h[Lp:])
            grad[Lp:] -= w * h[:n - Lp]
    return dx * grad

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        else:
            idx = int(np.argmin(np.maximum(h, 1.0 - h)))
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def interp_to(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def obj_and_jac(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    val = smooth_max_value(corr, beta) * dx
    grad = compute_grad(x, beta)
    return val, grad

def run_slsqp(h0, n, beta, max_iter, ftol=1e-16):
    def obj(x):
        v, _ = obj_and_jac(x, n, beta)
        return v
    def jac(x):
        _, g = obj_and_jac(x, n, beta)
        return g
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n) * (2.0 / n)}]
    h0 = project_feasible(h0.copy(), n)
    res = minimize(obj, h0, jac=jac, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': ftol, 'iprint': 0})
    h_out = project_feasible(res.x, n)
    return h_out, compute_c5(h_out)

def active_set_refine(h, n, n_iter, time_limit, start_t):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iter):
        if time.time() - start_t > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Top 8 shifts
        top_idx = np.argsort(corr)[::-1][:8]
        g = np.zeros(n)
        for rank, idx in enumerate(top_idx):
            weight = 1.0 / (rank + 1)
            L = idx - (n - 1)
            if L == 0:
                g += weight * (1.0 - 2.0 * h)
            elif L > 0:
                g[L:] += weight * (1.0 - h[:n - L])
                g[:n - L] -= weight * h[L:]
            else:
                Lp = -L
                g[:n - Lp] += weight * (1.0 - h[Lp:])
                g[Lp:] -= weight * h[:n - Lp]

                h_new, c5_new = run_slsqp(cur, n1, beta, iters)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n1
                    print("  SLSQP beta={}: C5={:.8f}".format(beta, best_c5))
                cur = h_new
            except Exception:
                break
        # Active set
        if remaining() > 30:
            h_as, c5_as = active_set_refine(best_h, best_n, 400, remaining() - 30, start_time)
            if c5_as < best_c5:
                best_c5 = c5_as
                best_h = h_as.copy()
                print("  ActiveSet: C5={:.8f}".format(best_c5))

    # Phase 2: Higher resolution optimization
    n_targets = []
    for nn in [300, 400, 500, 600]:
        if nn != best_n:
            n_targets.append(nn)

    for n2 in n_targets:
        if remaining() < 90:
            break
        print("Phase 2: n={}".format(n2))
        h0 = interp_to(best_h, n2)

        for beta in [50, 200, 600]:
            if remaining() < 30:
                break
            iters = 400 if beta <= 200 else 300
            try:
                h_new, c5_new = run_slsqp(h0, n2, beta, iters)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n2
                    print("  n={}, beta={}: C5={:.8f}".format(n2, beta, best_c5))
                h0 = h_new
            except Exception:
                break

        if remaining() > 40 and best_n == n2:
            h_as, c5_as = active_set_refine(best_h, best_n, 500, remaining() - 40, start_time)
            if c5_as < best_c5:
                best_c5 = c5_as
                best_h = h_as.copy()
                print("  ActiveSet: C5={:.8f}".format(best_c5))

        if remaining() > 25 and best_n == n2:
            try:
                h_new, c5_new = run_slsqp(best_h, best_n, 1500, 500)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    print("  Final polish: C5={:.8f}".format(best_c5))
            except Exception:
                pass

    # Phase 3: Multi-start exploration at best resolution
    if remaining() > 80:
        print("Phase 3: Multi-start at n={}".format(best_n))
        n3 = best_n
        inits = []
        # Sinusoidal inits with different freqs
        for freq in [1.5, 2.0, 2.5, 3.0]:
            x = np.arange(n3)
            h_try = 0.5 + 0.38 * np.sin(2 * np.pi * freq * x / n3 + rng.uniform(0, 6.28))
            h_try = project_feasible(h_try, n3)
            inits.append(h_try)
        # From best with perturbations
        for _ in range(3):
            h_pert = best_h + rng.normal(0, 0.05, n3)
            h_pert = project_feasible(h_pert, n3)
            inits.append(h_pert)

        for h_init in inits:
            if remaining() < 30:
                break
            try:
                h_new, c5_new = run_slsqp(h_init, n3, 200, 350)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n3
                    print("  Multi-start: C5={:.8f}".format(best_c5))
            except Exception:
                pass

    # Phase 4: Coordinate descent fine-tuning
    if remaining() > 40 and not (np.all((best_h == 0) | (best_h == 1))):
        print("Phase 4: Coordinate descent")
        h_cd, c5_cd = fine_coordinate_descent(best_h, best_n, remaining() - 40, start_time, rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            print("  CoordDesc: C5={:.8f}".format(best_c5))

    # Phase 5: Binary rounding and local search
    if remaining() > 25:
        print("Phase 5: Binary search at n={}".format(best_n))
        h_bin = round_to_binary(best_h, best_n)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            print("  Rounded: C5={:.8f}".format(best_c5))
        else:
            # Try at different resolutions
            for n_try in [best_n, best_n - 4, best_n + 4]:
                if n_try < 50:
                    continue
                if remaining() < 15:
                    break
                h_interp = interp_to(best_h, n_try)
                h_bin_try = round_to_binary(h_interp, n_try)
                c5_try = compute_c5(h_bin_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_bin_try.copy()
                    best_n = n_try
                    print("  Binary n={}: C5={:.8f}".format(n_try, best_c5))
                    break

        # Local search on best
        if remaining() > 30:
            if np.all((best_h == 0) | (best_h == 1)):
                h_ls, c5_ls = binary_local_search(best_h, best_n, remaining() - 30, start_time, rng)
                if c5_ls < best_c5:
                    best_c5 = c5_ls
                    best_h = h_ls.copy()
                    print("  LocalSearch: C5={:.8f}".format(best_c5))
            else:
                # Try rounding and then local search
                h_bin2 = round_to_binary(best_h, best_n)
                c5_b2 = compute_c5(h_bin2)
                h_ls2, c5_ls2 = binary_local_search(h_bin2, best_n, remaining() - 35, start_time, rng)
                if c5_ls2 < best_c5:
                    best_c5 = c5_ls2
                    best_h = h_ls2.copy()
                    print("  Round+LS: C5={:.8f}".format(best_c5))

    # Phase 6: Final SLSQP polish if continuous
    if remaining() > 30 and not (np.all((best_h == 0) | (best_h == 1))):
        try:
            h_pol, c5_pol = run_slsqp(best_h, best_n, 2000, 600)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
                print("  Final SLSQP: C5={:.8f}".format(best_c5))
        except Exception:
            pass

    # Final
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
