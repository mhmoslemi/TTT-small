from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr, dx

def project_binary(h, n):
    """Round h to binary with exactly n/2 ones."""
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    n_want = n // 2
    if n_ones > n_want:
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])[:n_ones - n_want]
        h_bin[ones[order]] = 0.0
    elif n_ones < n_want:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])[:n_want - n_ones]
        h_bin[zeros[order]] = 1.0
    return h_bin

def project_feasible(h, n):
    """Project h to [0,1]^n with sum = n/2."""
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-15:
            break
        free = (h > 0) & (h < 1)
        if np.sum(free) == 0:
            free = np.ones(n, dtype=bool)
        h[free] += diff / np.sum(free)
        h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

            if offset == 0:

def project_feasible(h, n):

import numpy as np
from scipy.optimize import minimize
import time

_GN = 192

def _c5val(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _prj(h, n):
    h = np.array(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    t = n / 2.0
    for _ in range(400):
        d = t - h.sum()
        if abs(d) < 1e-16:
            break
        m = (h > 1e-14) & (h < 1.0 - 1e-14)
        if m.any():
            h[m] += d / m.sum()
        else:
            i = int(np.argmax(1.0 - h)) if d > 0 else int(np.argmax(h))
            h[i] = np.clip(h[i] + d, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-14:
        i = int(np.argmax(1.0 - h)) if d > 0 else int(np.argmax(h))
        h[i] = np.clip(h[i] + d, 0.0, 1.0)
    return h

def _tobin(h, n):
    b = np.where(h >= 0.5, 1.0, 0.0).copy()
    no = int(b.sum())
    nw = n // 2
    if no > nw:
        idx = np.where(b == 1)[0]
        sel = idx[np.argsort(h[idx])[:no - nw]]
        b[sel] = 0.0
    elif no < nw:
        idx = np.where(b == 0)[0]
        sel = idx[np.argsort(-h[idx])[:nw - no]]
        b[sel] = 1.0
    return b

def _sgg(h, n, act):
    g = np.zeros(n, dtype=np.float64)
    na = len(act)
    if na == 0:
        return g
    for m in act:
        k = m - (n - 1)
        if k == 0:
            g += 1.0 - 2.0 * h
        elif k > 0:
            g[k:] += 1.0 - h[:n - k]
            g[:n - k] -= h[k:]
        else:
            o = -k
            g[:n - o] += 1.0 - h[o:]
            g[o:] -= h[:n - o]
    return g / na

def _sgopt(h0, n, tl, lr0=0.3):
    t0 = time.time()
    h = h0.copy()
    bh = h0.copy()
    bc = _c5val(h0)
    mom = np.zeros(n)
    lr = lr0
    it = 0
    while time.time() - t0 < tl:
        corr = np.correlate(h, 1.0 - h, mode='full')
        mv = float(corr.max())
        cv = mv * 2.0 / n
        if cv < bc:
            bc = cv
            bh = h.copy()
        act = np.where(corr >= mv - 1e-10)[0]
        g = _sgg(h, n, act)
        gn = np.linalg.norm(g)
        if gn > 1e-15:
            g /= gn
        step = lr / (1.0 + 0.0005 * it)
        mom = 0.9 * mom - step * g
        h = _prj(h + mom, n)
        it += 1
    return bh, bc

def _hill(h0, n, rng, tl):
    t0 = time.time()
    h = h0.copy()
    c = _c5val(h)
    bh, bc = h.copy(), c
    dx = 2.0 / n
    while time.time() - t0 < tl:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        imp = False
        for _ in range(3000):
            if time.time() - t0 > tl:
                break
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])
            h[i] = 0.0; h[j] = 1.0
            cn = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
            if cn < c - 1e-14:
                c = cn
                imp = True
            else:
                h[j] = 0.0; h[i] = 1.0
        if c < bc:
            bc = c
            bh = h.copy()
        if not imp:
            np_ = max(2, min(10, n // 4))
            for _ in range(np_):
                i = int(ones[rng.randint(len(ones))])
                j = int(zeros[rng.randint(len(zeros))])
                h[i] = 0.0; h[j] = 1.0
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            c = _c5val(h)
    return bh, bc

def _fnb(h0, n, tl):
    t0 = time.time()
    h = h0.copy()
    c = _c5val(h)
    bh, bc = h.copy(), c
    dx = 2.0 / n
    while time.time() - t0 < tl:
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        imp = False
        best_i, best_j = -1, -1
        best_c = c
        for i in ones:
            if time.time() - t0 > tl:
                return bh, bc
            for j in zeros:
                h[i] = 0.0; h[j] = 1.0
                cn = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
                h[j] = 0.0; h[i] = 1.0
                if cn < best_c:
                    best_c = cn
                    best_i, best_j = i, j
        if best_i >= 0:
            h[best_i] = 0.0; h[best_j] = 1.0
            c = best_c
            imp = True
            if c < bc:
                bc = c
                bh = h.copy()
        if not imp:
            break
    return bh, bc

def _slsqp_f(x):
    n = _GN
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    beta = 50000.0
    s = beta * (c - m)
    s = np.clip(s, -500.0, 0.0)
    return (m + np.log(np.exp(s).sum()) / beta) * (2.0 / n)

def _slsqp_eq(x):
    return np.sum(x) - _GN / 2.0

def _slsqp_eqj(x):
    return np.ones(_GN)

def _slsqp(h0, n, tl):
    global _GN
    _GN = n
    t0 = time.time()
    h = _prj(h0, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eqj}]
    while time.time() - t0 < tl:
        rem = tl - (time.time() - t0)
        if rem < 15:
            break
        mx = min(3000, max(200, int(rem * 50)))
        try:
            r = minimize(_slsqp_f, h, method='SLSQP', bounds=bounds,
                         constraints=cons, options={'maxiter': mx, 'ftol': 1e-18, '
