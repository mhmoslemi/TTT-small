    h = np.clip(h + shift, 0.0, 1.0)
    # Fix any tiny residual
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_grad(corr, beta):
    """Compute smooth max and its gradient weights (softmax)."""
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)
    softmax = exps / exps.sum()
    fval = m + np.log(exps.sum()) / beta
    return fval, softmax

def compute_grad_pnorm(h, p=16):
    """Compute gradient of (sum_k (corr_k)^p)^(1/p) * dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # f_p = (sum_k (corr_k * dx)^p)^(1/p)
    # grad = (sum_k (corr_k*dx)^p)^(1/p - 1) * sum_k (corr_k*dx)^(p-1) * dx * dcorr_k/dh[j]
    
    W = corr * dx
    Wp = W ** p
    Sp = Wp.sum()
    prefactor = Sp ** (1.0/p - 1.0)
    
    # grad[j] = prefactor * sum_k W_k^(p-1) * dx * dcorr_k/dh[j]
    # = prefactor * dx * sum_k (corr_k*dx)^(p-1) * dcorr_k/dh[j]
    
    g = np.zeros(n)
    weights = W ** (p-1)  # length 2n-1
    
    # For each shift k (from -(n-1) to n-1), the correlation element is at index n-1+k
    # dcorr_k/dh[j]:
    #   For k >= 0: (1-h[j+k]) if j+k < n, minus h[j-k] if j >= k
    #   For k < 0: (1-h[j+|k|]) if j+|k| < n, minus h[j-|k|]) if j >= |k|
    #   For k = 0: 1 - 2*h[j]
    
    # This is O(n^2). For n up to 512, that's fine.
    # Let's vectorize.
    
    for k_idx in range(-(n-1), n):
        w = weights[n-1+k_idx]
        if w < 1e-30:
            continue
        if k_idx == 0:
            g += w * (1.0 - 2.0 * h)
        elif k_idx > 0:
            # corr_k = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
            # d/dh[j]: (1-h[j+k]) if j+k<n, -h[j-k] if j>=k
            n_pairs = n - k_idx
            # First part: j in [0, n_pairs), contributes (1-h[j+k])
            g[:n_pairs] += w * (1.0 - h[k_idx:k_idx+n_pairs])
            # Second part: j in [k, n), contributes -h[j-k]
            g[k_idx:] -= w * h[:n_pairs]
        else:
            # k_idx < 0, let j_abs = -k_idx
            j_abs = -k_idx
            # corr = sum_{i=j_abs}^{n-1} h[i]*(1-h[i-j_abs])
            # = sum_{i'=0}^{n-1-j_abs} h[i'+j_abs]*(1-h[i'])
            n_pairs = n - j_abs
            # d/dh[j]: (1-h[j-j_abs]) if j>=j_abs, -h[j+j_abs] if j+j_abs<n
            g[j_abs:] += w * (1.0 - h[:n_pairs])
            g[:n_pairs] -= w * h[j_abs:j_abs+n_pairs]
    
    return prefactor * dx * g

def projected_gradient_step(h, grad, step, n):
    """Take gradient step and project."""
    h_new = h - step * grad
    return project_feasible(h_new, n)
