import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    """Convert continuous h to binary with exactly n/2 ones"""
    h_bin = (h >= 0.5).astype(float)
    s = np.sum(h_bin)
    deficit = n / 2.0 - s
    if deficit > 0:
        # Need to add more 1s: flip the 0s with highest h values
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])[:int(deficit)]
        h_bin[zeros[order]] = 1.0
    elif deficit < 0:
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])[:int(-deficit)]
        h_bin[ones[order]] = 0.0
    return h_bin

def binary_sa(h_init, n, time_limit, seed=42):
    """Simulated annealing on binary sequences with 2-swap moves"""
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    
    # Ensure binary
    h = (h >= 0.5).astype(float)
    s = np.sum(h)
    deficit = int(round(n / 2.0 - s))
    if deficit > 0:
        zeros = np.where(h == 0)[0]
        rng.shuffle(zeros)
        h[zeros[:deficit]] = 1.0
    elif deficit < 0:
        ones = np.where(h == 1)[0]
        rng.shuffle(ones)
        h[ones[:deficit]] = 0.0
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    curr_c5 = best_c5
    
    T0 = 0.1
    Tend = 0.001
    start = time.time()
    step = 0
    no_improve = 0
    max_no_improve = 50000
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    while time.time() - start < time_limit:
        # Adaptive temperature
        frac = min(1.0, (time.time() - start) / time_limit)
        T = T0 * (Tend / T0) ** frac
        
        # Pick random 1-position and 0-position
        p = ones[rng.randint(len(ones))]
        q = zeros[rng.randint(len(zeros))]
        
        # Swap
        h[p] = 0.0
        h[q] = 1.0
        new_c5 = compute_c5(h)
        
        delta = new_c5 - curr_c5
        
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-15)):
            curr_c5 = new_c5
            # Update tracking
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if curr_c5 < best_c5 - 1e-12:
                best_c5 = curr_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            step += 1
        else:
            # Reject
            h[p] = 1.0
            h[q] = 0.0
            no_improve += 1
            step += 1
        
        # Restart if stuck
        if no_improve > max_no_improve:
            T = T0 * 0.5
            no_improve = 0
            frac = 0.3
            T = T0 * (Tend / T0) ** frac
    
    return best_h, best_c5

    elif deficit < 0:
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])[:abs(deficit)]
        h_bin[ones[order]] = 0.0
    return h_bin


def binary_sa(h_init, n, time_limit, seed=42):
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    h = (h >= 0.5).astype(float)
    s = int(np.sum(h) + 0.5)
    target = n // 2
    deficit = target - s
    if deficit > 0:
        zeros = np.where(h == 0)[0]
        idx = rng.choice(len(zeros), min(deficit, len(zeros)), replace=False)
        h[zeros[idx]] = 1.0
    elif deficit < 0:
        ones = np.where(h == 1)[0]
        idx = rng.choice(len(ones), min(abs(deficit), len(ones)), replace=False)
        h[ones[idx]] = 0.0

    best_h = h.copy()
    best_c5 = compute_c5(h)
    curr_c5 = best_c5

    T0 = 0.05
    Tend = 0.0005
    start = time.time()
    no_improve = 0
    total_steps = 0
    max_no_improve = 100000

    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]

    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        T = T0 * (Tend / T0) ** frac

        if len(ones) == 0 or len(zeros) == 0:
            break

        p = ones[rng.randint(len(ones))]
        q = zeros[rng.randint(len(zeros))]

        h[p] = 0.0
        h[q] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - curr_c5

        if delta <= 0 or (T > 1e-12 and rng.rand() < np.exp(-delta / T)):
            curr_c5 = new_c5
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if curr_c5 < best_c5 - 1e-13:
                best_c5 = curr_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            total_steps += 1
        else:
            h[p] = 1.0
            h[q] = 0.0
            no_improve += 1
            total_steps += 1

        if no_improve > max_no_improve:
            no_improve = 0
            T = T0 * 0.3
            frac = min(1.0, elapsed / max(time_limit, 1e-6)) * 0.5

    return best_h, best_c5


def coord_descent_binary(h, n, time_limit, n_passes_max=200):
    h = h.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    for p in range(n_passes_max):
        if time.time() - start > time_limit:
            break
        improved = False
        for j in range(n):
            if h[j] == 1.0:
                h[j] = 0.0
                c = compute_c5(h)
                if c < best_c5 - 1e-13:
                    best_c5 = c
                    best_h = h.copy()
                    s = np.sum(h)
                    if s < n / 2.0 - 1e-10:
                        h[j] = 1.0
                        break
                else:
                    h[j] = 1.0
            else:
                h[j] = 1.0
                c = compute_c5(h)
                if c < best_c5 - 1e-13:
                    best_c5 = c
                    best_h = h.copy()
                    s = np.sum(h)
                    if s > n / 2.0 + 1e-10:
                        h[j] = 0.0
                        break
                else:
                    h[j] = 0.0
        h = best_h.copy()
    return best_h, best_c5


def slsqp_optimize(h, n, max_iter=2000):
    def c5_obj(x):
        return np.max(np.correlate(x, 1.0 - x, mode='full'))

    def sum_con(x):
        return np.sum(x) - n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        res = minimize(c5_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12})
        if res.fun < 1e10 and res.x is not None:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None


def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_src)
    return make_feasible(h_interp, n_dst)


def generate_structured_binary(n, seed=0):
    rng = np.random.RandomState(seed)
    inits = []

    # Thue-Morse
    tm = np.zeros(n)
    for i in range(n):
        tm[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    inits.append(to_binary_from_cont(tm, n))

    # Block pattern 1100
    if n % 4 == 0:
        bp = np.tile([1.0, 1.0, 0.0, 0.0], n // 4)
        inits.append(bp.copy())

    # Checkerboard
    if n % 2 == 0:
        cb = np.tile([1.0, 0.0], n // 2)
        inits.append(cb.copy())

    # Random
    for k in range(3):
        r = (rng.rand(n) > 0.5).astype(float)
        inits.append(to_binary_from_cont(r, n))

    # Structured: first half alternating
    sh = np.zeros(n)
    if n % 2 == 0:
        sh[:n//4] = np.tile([1.0, 0.0], n//8) if n % 4 == 0 else (np.arange(n//4) % 2).astype(float)
        sh[n//2:3*n//4] = np.tile([1.0, 0.0], n//8) if n % 4 == 0 else (np.arange(n//4) % 2).astype(float)
        sh[n//4:n//2] = np.tile([0.0, 1.0], n//8) if n % 4 == 0 else ((np.arange(n//4) + 1) % 2).astype(float)
        sh[3*n//4:] = np.tile([0.0, 1.0], n//8) if n % 4 == 0 else ((np.arange(n//4) + 1) % 2).astype(float)
    inits.append(to_binary_from_cont(sh, n))

    return inits


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def remaining():
        return budget_s - (time.time() - start_time)

    # Get parent
    h_parent = None
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if h_parent is not None and len(h_parent) == 0:
            h_parent = None
    except (NameError, Exception):
        h_parent = None

    if h_parent is not None:
        best_h = make_feasible(h_parent.copy(), len(h_parent))
        best_n = len(best_h)
        best_c5 = compute_c5(best_h)
    else:
        best_n = 160
        best_h = np.ones(best_n) * 0.5
        best_c5 = compute_c5(best_h)

    # Phase 1: SA on parent resolution
    if remaining() > 100:
        h_bin = to_binary_from_cont(best_h, best_n)
        tl = min(120, int(remaining() * 0.15))
        h_sa, c5_sa = binary_sa(h_bin, best_n, tl, seed=seed)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            best_n = best_n

        # SLSQP on continuous
        if remaining() > 80:
            h_slq, c5_slq = slsqp_optimize(best_h, best_n, max_iter=3000)
            if h_slq is not None and c5_slq < best_c5:
                best_c5 = c5_slq
                best_h = h_slq.copy()

    # Phase 2: Multi-resolution SA + SLSQP
    # Try various n values
    res_list = [200, 240, 280, 320, 400, 480, 560]
    # Filter to even values and add parent n
    res_list = [r for r in res_list if r % 2 == 0]
    if best_n not in res_list:
        res_list.insert(0, best_n)
    res_list = sorted(set(res_list))

    # Time allocation: spread remaining time across resolutions
    n_res = len(res_list)
    time_per_res = max(40, int((remaining() - 200) / max(n_res, 1)))

    for ri, n in enumerate(res_list):
        if remaining() < 80:
            break
        if n == 0:
            continue

        # Initialize
        if n == best_n and len(best_h) == n:
            h_init = best_h.copy()
        else:
            h_init = interpolate_to_n(best_h, n)

        # Try multiple initializations
        best_local_c5 = compute_c5(h_init)
        best_local_h = h_init.copy()

        # Structured inits
        structs = generate_structured_binary(n, seed=seed + ri)
        for si, s in enumerate(structs[:3]):
            c_s = compute_c5(s)
            if c_s < best_local_c5:
                best_local_c5 = c_s
                best_local_h = s.copy()

        # SA on best init
        tl_sa = min(time_per_res, int(remaining() * 0.25))
        if tl_sa < 20:
            tl_sa = max(15, int(remaining() * 0.1))
        h_sa, c5_sa = binary_sa(best_local_h, n, tl_sa, seed=seed + 1000 * ri)
        if c5_sa < best_local_c5:
            best_local_c5 = c5_sa
            best_local_h = h_sa.copy()

        # Second SA from different seed
        if remaining() > 100:
            tl_sa2 = min(int(tl_sa * 0.5), int(remaining() * 0.1))
            if tl_sa2 > 10:
                h_sa2, c5_sa2 = binary_sa(best_local_h, n, tl_sa2, seed=seed + 50000 + ri)
                if c5_sa2 < best_local_c5:
                    best_local_c5 = c5_sa2
                    best_local_h = h_sa2.copy()

        # SLSQP polish
        if remaining() > 80:
            h_slq, c5_slq = slsqp_optimize(best_local_h, n, max_iter=2000)
            if h_slq is not None and c5_slq < best_local_c5:
                best_local_c5 = c5_slq
                best_local_h = h_slq.copy()

        # Binary coordinate descent
        if remaining() > 60:
            h_cd, c5_cd = coord_descent_binary(best_local_h, n, time_limit=min(30, int(remaining() * 0.1)))
            if c5_cd < best_local_c5:
                best_local_c5 = c5_cd
                best_local_h = h_cd.copy()

        # Update global best
        if best_local_c5 < best_c5 - 1e-12:
            best_c5 = best_local_c5
            best_h = best_local_h.copy()
            best_n = n

        # If SLQOP improved, try one more SA from the continuous solution
        if remaining() > 80 and best_n == n:
            h_b2 = to_binary_from_cont(best_h, n)
            tl_final = min(20, int(remaining() * 0.08))
            if tl_final > 5:
                h_s3, c5_s3 = binary_sa(h_b2, n, tl_final, seed=seed + 99999 + ri)
                if c5_s3 < best_c5:
                    best_c5 = c5_s3
                    best_h = h_s3.copy()

    # Phase 3: Extended SA on best resolution with multiple restarts
    if remaining() > 150:
        n_best = best_n
        h_bin_best = to_binary_from_cont(best_h, n_best)
        total_sa_time = int(remaining() * 0.5)
        n_restarts = min(4, max(1, total_sa_time // 30))
        time_per_restart = max(15, total_sa_time // n_restarts)

        for r in range(n_restarts):
            if remaining() < 40:
                break
            s = seed + 777777 + 12345 * r
            h_r, c5_r = binary_sa(h_bin_best, n_best, time_per_restart, seed=s)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
            # SLSQP after each SA
            if remaining() > 60:
                h_slq, c5_slq = slsqp_optimize(best_h, n_best, max_iter=1500)
                if h_slq is not None and c5_slq < best_c5:
                    best_c5 = c5_slq
                    best_h = h_slq.copy()
                    h_bin_best = to_binary_from_cont(best_h, n_best)

    # Phase 4: Try slightly different resolutions around best
    if remaining() > 100:
        for dn in [-16, 16, -32, 32, -64, 64]:
            n_try = best_n + dn
            if n_try < 64 or n_try > 800 or n_try % 2 != 0:
                continue
            if remaining() < 50:
                break
            h_try = interpolate_to_n(best_h, n_try)
            h_bin_try = to_binary_from_cont(h_try, n_try)
            tl = min(25, int(remaining() * 0.1))
            if tl < 10:
                continue
            h_s, c5_s = binary_sa(h_bin_try, n_try, tl, seed=seed + n_try)
            if c5_s < best_c5 - 1e-12:
                best_c5 = c5_s
                best_h = h_s.copy()
                best_n = n_try

    # Phase 5: Final continuous SLSQP polish
    if remaining() > 40:
        h_fin, c5_fin = slsqp_optimize(best_h, best_n, max_iter=5000)
        if h_fin is not None and c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Phase 6: One more binary SA from the polished solution
    if remaining() > 30:
        h_bfin = to_binary_from_cont(best_h, best_n)
        tl_f = min(20, int(remaining() * 0.4))
        if tl_f > 5:
            h_sf, c5_sf = binary_sa(h_bfin, best_n, tl_f, seed=seed + 1337)
            if c5_sf < best_c5:
                best_c5 = c5_sf
                best_h = h_sf.copy()

    # Final: ensure feasibility and compute exact C5
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
