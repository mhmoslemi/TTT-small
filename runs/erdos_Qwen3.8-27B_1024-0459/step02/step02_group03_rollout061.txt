def project_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmin(np.maximum(h, 1.0 - h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_round(h, n):
    """Round h to binary with exactly n/2 ones"""
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    n_want = n // 2
    if n_ones > n_want:
        ones_idx = np.where(h_bin == 1.0)[0]
        # Remove the ones with smallest h values
        order = np.argsort(h[ones_idx])[:n_ones - n_want]
        h_bin[ones_idx[order]] = 0.0
    elif n_ones < n_want:
        zeros_idx = np.where(h_bin == 0.0)[0]
        # Add the zeros with largest h values
        order = np.argsort(-h[zeros_idx])[:n_want - n_ones]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def greedy_local_search(h, n, max_iters, rng, time_limit, start_time):
    """Greedy local search: at each step, try all pair swaps and keep the best improvement"""
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_max = np.max(corr)
    best_c5 = best_max * 2.0 / n
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]

        best_swap = None

                h[i0] = 1.0
                new_corr = np.correlate(h, 1.0 - h, mode='full')
                new_max = np.max(new_corr)
                h[i1] = 1.0
                h[i0] = 0.0
                
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_swap = (i1, i0)
                    if best_new_max <= best_max - 1:  # Can't do better than this for this round

                        pass
        
        if best_swap is not None:

            h[i1] = 0.0
            h[i0] = 1.0
            best_max = best_new_max
            best_c5 = best_max * 2.0 / n
        else:
            # No improving swap found
            break
    
    return h, best_c5
