```python
def compute_delta(h, i, j, n):
    """Compute change in corr when h[i]:1→0 and h[j]:0→1"""
    delta = np.zeros(2*n-1)
    
    # i: 1→0
    # Contributes -(1-h[i+k]) for valid k where i+k in [0,n)
    # and +h[i-k] for valid k where i-k in [0,n)
    
    # k range for i+k in [0,n): k in [-i, n-1-i], intersected with [-(n-1), n-1]
    ks1 = np.arange(max(-i, -(n-1)), min(n-i, n) + 1)
    ms1 = ks1 + n - 1
    delta[ms1] -= (1.0 - h[i + ks1])
    
    # k range for i-k in [0,n): k in [i-n+1, i], intersected with [-(n-1), n-1]
    ks2 = np.arange(max(i-n+1, -(n-1)), min(i, n-1) + 1)
    ms2 = ks2 + n - 1
    delta[ms2] += h[i - ks2]
    
    # j: 0→1
    ks3 = np.arange(max(-j, -(n-1)), min(n-j, n) + 1)
    ms3 = ks3 + n - 1
    delta[ms3] += (1.0 - h[j + ks3])
    
    ks4 = np.arange(max(j-n+1, -(n-1)), min(j, n-1) + 1)
    ms4 = ks4 + n - 1
    delta[ms4] -= h[j - ks4]
    
    return delta
```

def coord_opt_j(h, j, n, corr, ells):
    a = np.zeros(2 * n - 1)
    idx1 = j - ells
    idx2 = j + ells
    v1 = (idx1 >= 0) & (idx1 < n)
    v2 = (idx2 >= 0) & (idx2 < n)
    if np.any(v1):
        a[v1] += 1.0 - h[idx1[v1]]
    if np.any(v2):
        a[v2] += -h[idx2[v2]]
    b = corr - h[j] * a
    lo = 0.0
    hi = 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        vals = a * mid + b
        mx = float(np.max(vals))
        active = vals >= mx - 1e-14
        sg = float(np.mean(a[active]))
        if sg > 1e-13:
            hi = mid
        elif sg < -1e-13:
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)
