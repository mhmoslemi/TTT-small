            j = zeros[rng.integers(len(zeros))]
            h[i], h[j] = h[j], h[i]
            c = evaluate_erdos(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            else:
                h[i], h[j] = h[j], h[i]
            continue
        
        # Find the worst pair at lag k_shift
        lag = abs(k_shift)
        if k_shift > 0:
            # c_k = Σ h[i+k](1-h[i]) for i=0..n-1-k
            # Worst pair: maximize h[i+k](1-h[i])
            contrib = h[lag:] * (1.0 - h[:n-lag])
            i_best = int(np.argmax(contrib)) + 0  # i index
            j_best = i_best + lag  # j = i + k index
        else:
            # k_shift < 0, effective lag = -k_shift
            # c_k for k<0: c_k = Σ h[i-k'](1-h[i]) where k'=-k
            # = Σ h[i+k'](1-h[i]) ... 
            # Actually from the correlation definition:
            # corr[n-1+k] = Σ h[i+k](1-h[i])
            # For k < 0: Σ h[i+k](1-h[i]) where i >= -k
            # Let i' = i + k (so i' >= 0): Σ h[i'](1-h[i'+k])
            # = Σ h[i'](1-h[i'-k']) where k' = -k > 0
            # Worst pair: maximize h[i'](1-h[i'-k'])
            contrib = h[:n-lag] * (1.0 - h[lag:])
            i_best = int(np.argmax(contrib))  # i' index
            j_best = i_best + lag  # i' + k' index
        
        # Try swapping i_best and j_best
        # h[i_best] should be 0 (or low) and h[j_best] should be 1 (or high)
        # After swap: h[i_best] += 1, h[j_best] -= 1 (if binary)
        h_swap = h.copy()
        h_swap[i_best], h_swap[j_best] = h_swap[j_best], h_swap[i_best]
        
        c_new = evaluate_erdos(h_swap)
        if c_new < best_c - 1e-12:
            h = h_swap
            best_h = h.copy()
            best_c = c_new
        else:
            # Try a few random swaps as backup
            found = False
            for _ in range(20):
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                ii = ones[rng.integers(len(ones))]
                jj = zeros[rng.integers(len(zeros))]
                h_t = h.copy()
                h_t[ii], h_t[jj] = h_t[jj], h_t[ii]
                c_t = evaluate_erdos(h_t)
                if c_t < best_c - 1e-12:
                    h = h_t
                    best_h = h.copy()
                    best_c = c_t
                    found = True
                    break
            if not found:
                break  # Local optimum reached
    
    return best_h, best_c
