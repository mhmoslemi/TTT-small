    if temperature > 1e-12:
        dv = (top_vals - top_vals.min()) / temperature
        w = np.exp(dv)
    else:
        spread = top_vals.max() - top_vals.min()
        if spread < 1e-12:
            w = np.ones(k_count) / k_count
        else:
            dv = (top_vals - top_vals.min()) / spread
            w = (1.0 + dv) ** 4.0
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.ones(k_count) / k_count
    else:
        w /= w_sum
    g = np.zeros(n, dtype=float)
    for i in range(k_count):
        t = int(top_idx[i])
        k = t - center
        g += w[i] * corr_grad_k(h, k)
    return g, corr, top_idx


def smooth_max_obj(h, n, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -700.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0:
        return 1e100
    return (2.0 / n) * (m + np.log(z) / alpha)


def smooth_max_grad(h, n, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -700.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0:
        return np.zeros(n)
    w = e / z
    center = n - 1
    g = np.zeros(n, dtype=float)
    idx = np.where(w > 1e-8)[0]
    for t in idx:
        t = int(t)
        p = float(w[t])
        k = t - center
        g += p * corr_grad_k(h, k)
    return (2.0 / n) * g


def upsample_h(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    if n0 <= 0:
        return project_density(np.full(n, 0.5), 0.5 * n)
    if n0 == n:
        return p.copy()
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.interp(x1, x0, p, left=p[0], right=p[-1])
    return project_density(h, 0.5 * n)


def subgradient_line_search(h, g, n, best_c, max_backtrack=30):
    """Backtracking line search along -g direction, projected to feasible set"""
    target = 0.5 * n
    d = -g
    d -= np.mean(d)
    norm_d = np.linalg.norm(d)
    if norm_d < 1e-14:
        return h, best_c, 0.0
    d = d / norm_d
    step = 0.05
    for _ in range(max_backtrack):
        cand = project_density(h + step * d, target)
        c = evaluate_erdos(cand)
        if c < best_c - 1e-14:
            return cand, c, step
        step *= 0.6
        if step < 1e-8:
            break
    return h, best_c, 0.0


def active_subgradient_phase(h, n, deadline, top_k_start=5, top_k_end=12, max_iter=2000):
    """Iterative active-set subgradient with adaptive parameters"""
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    it = 0
    stall = 0
    top_k = top_k_start
    while it < max_iter:
        if time.perf_counter() >= deadline:
            break
        g, corr, top_idx = active_set_gradient(best, top_k=top_k, temperature=0.0)
        cand, c, step_used = subgradient_line_search(best, g, n, best_c)
        if step_used > 0:
            best = cand
            best_c = c
            stall = 0
            # Gradually increase top_k
            top_k = min(top_k_end, top_k + (1 if (it + 1) % 50 == 0 else 0))
        else:
            stall += 1
            if stall > 60:
                # Try with larger top_k
                top_k = min(top_k_end, top_k + 2)
                stall = 0
                if stall > 100:
                    break
        it += 1
    return project_density(best, target), evaluate_erdos(best)


def coordinate_descent_phase(h, n, deadline, max_iter=5000):
    """Randomized coordinate descent on the active top lags"""
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    rng = np.random.default_rng(12345)
    stall = 0
    for it in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        center = n - 1
        top_k = 8
        top_idx = np.argsort(corr)[-top_k:]
        top_vals = corr[top_idx]
        # Gradient from top-k
        spread = top_vals.max() - top_vals.min() + 1e-15
        w = ((top_vals - top_vals.min()) / spread) ** 3.0 + 1e-8
        w /= w.sum()
        g = np.zeros(n, dtype=float)
        for i in range(top_k):
            k = int(top_idx[i]) - center
            g += w[i] * corr_grad_k(best, k)
        # Pick a coordinate to adjust
        # Weight by |g| to focus on sensitive coordinates
        abs_g = np.abs(g)
        total_w = abs_g.sum()
        if total_w < 1e-14:
            break
        probs = abs_g / total_w
        j = int(rng.choice(n, p=probs))
        # Determine step direction and size
        direction = -np.sign(g[j])
        if np.abs(g[j]) < 1e-10:
            direction = 1.0 if best[j] < 0.5 else -1.0
        # Try several step sizes
        found = False
        for frac in (0.3, 0.5, 0.8, 1.0):
            max_step = frac * min(1.0 - best[j], best[j]) if direction < 0 else frac * (1.0 - best[j])
            if direction > 0:
                max_step = frac * (1.0 - best[j])
            else:
                max_step = frac * best[j]
            if max_step < 1e-8:
                continue
            # Adjust one coordinate, fix sum by adjusting a random other coordinate
            k2 = int(rng.integers(n))
            if k2 == j:
                k2 = (j + 1) % n
            d = direction * max_step
            cand = best.copy()
            cand[j] += d
            cand[k2] -= d
            if cand[j] < -1e-12 or cand[j] > 1 + 1e-12 or cand[k2] < -1e-12 or cand[k2] > 1 + 1e-12:
                continue
            c = evaluate_erdos(cand)
            if c < best_c - 1e-13:
                best = cand
                best_c = c
                found = True
                break
        if found:
            stall = 0
        else:
            stall += 1
            if stall > 500:
                break
    return project_density(best, target), evaluate_erdos(best)


def pair_swap_focus(h, n, deadline, max_iter=3000):
    """Targeted pair swaps focused on reducing the maximum lag"""
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    rng = np.random.default_rng(54321)
    stall = 0
    for it in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        center = n - 1
        # Focus on top 5 lags
        top_k = 5
        top_idx = np.argsort(corr)[-top_k:]
        top_vals = corr[top_idx]
        spread = top_vals.max() - top_vals.min() + 1e-15
        w = ((top_vals - top_vals.min()) / spread) ** 2.0 + 1e-6
        w /= w.sum()
        g = np.zeros(n, dtype=float)
        for i in range(top_k):
            k = int(top_idx[i]) - center
            g += w[i] * corr_grad_k(best, k)
        # Find best swap
        neg_pool = np.argsort(g)[:8]
        pos_pool = np.argsort(-g)[:8]
        found = False
        for ii in neg_pool:
            ii = int(ii)
            for jj in pos_pool:
                jj = int(jj)
                if ii == jj:
                    continue
                cap = min(1.0 - best[ii], best[jj])
                if cap < 1e-12:
                    continue
                for d in (0.5 * cap, 0.8 * cap, cap):
                    cand = best.copy()
                    cand[ii] += d
                    cand[jj] -= d
                    if cand[ii] < -1e-12 or cand[ii] > 1 + 1e-12 or cand[jj] < -1e-12 or cand[jj] > 1 + 1e-12:
                        continue
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-13:
                        best = cand
                        best_c = c
                        found = True
                        break
                if found:
                    break
            if found:
                break
        if not found:
            # Random pair
            for _r in range(8):
                ii = int(rng.integers(n))
                jj = int(rng.integers(n))
                if ii == jj:
                    continue
                cap = min(1.0 - best[ii], best[jj])
                if cap < 1e-10:
                    continue
                d = float(rng.random()) * 0.7 * cap
                cand = best.copy()
                cand[ii] += d
                cand[jj] -= d
                c = evaluate_erdos(cand)
                if c < best_c - 1e-13:
                    best = cand
                    best_c = c
                    found = True
                    break
        if found:
            stall = 0
        else:
            stall += 1
            if stall > 80:
                break
    return project_density(best, target), evaluate_erdos(best)


def slsqp_phase(h, n, deadline, alphas=(20, 60, 200), maxiter_list=None):
    """SLSQP with smooth max objective"""
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    if maxiter_list is None:
        if n <= 500:
            maxiter_list = [100, 80, 60]
        else:
            maxiter_list = [70, 50, 40]
    bounds = [(0.0, 1.0)] * n
    # Use closure-free constraint
    sum_vec = np.ones(n)
    for i, alpha in enumerate(alphas):
        if time.perf_counter() >= deadline - 5.0:
            break
        maxiter = maxiter_list[i] if i < len(maxiter_list) else 50
        try:
            res = minimize(
                lambda x: smooth_max_obj(x, n, alpha),
                best.copy(),
                jac=lambda x: smooth_max_grad(x, n, alpha),
                method="SLSQP",
                bounds=bounds,
                constraints=[{"type": "eq", "fun": lambda x: float(np.sum(x) - target),
                              "jac": lambda x: sum_vec}],
                options={"maxiter": maxiter, "ftol": 1e-11, "disp": False}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                cand = project_density(res.x, target)
                c = evaluate_erdos(cand)
                if c < best_c:
                    best = cand
                    best_c = c
        except Exception:
            pass
    return project_density(best, target), evaluate_erdos(best)


def make_start(n, rng, mode):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    if mode == 0:
        h = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x / 2 + rng.uniform(0, 6.28)) \
            + 0.2 * np.sin(2 * np.pi * 5 * x / 2 + rng.uniform(0, 6.28))
        h += 0.08 * np.sin(2 * np.pi * 8 * x / 2 + rng.uniform(0, 6.28))
    elif mode == 1:
        seg = max(1, int(n // 12))
        h = np.empty(n, dtype=float)
        for i in range(0, n, seg):
            end = min(i + seg, n)
            h[i:end] = float(rng.uniform(0.15, 0.85))
    elif mode == 2:
        u = rng.random(n)
        med = float(np.median(u))
        h = np.where(u > med, 0.7, 0.3).astype(float)
        h += rng.normal(0, 0.06, n)
    elif mode == 3:
        freq = int(rng.integers(2, 8))
        h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x / 2 + rng.uniform(0, 6.28))
    else:
        h = 0.5 + 0.35 * np.cos(2 * np.pi * 4 * x / 2 + rng.uniform(0, 6.28))
        h += 0.1 * np.sin(2 * np.pi * 7 * x / 2)
    return project_density(h, 0.5 * n)


def full_optimize(n, h0, deadline, seed=42):
    """Complete optimization pipeline at fixed resolution n"""
    target = 0.5 * n
    h = project_density(h0, target)
    best_h = h.copy()
    best_c = evaluate_erdos(best_h)

    # Phase 1: SLSQP
    t_rem = deadline - time.perf_counter()
    if t_rem > 20.0:
        h, c = slsqp_phase(h, n, time.perf_counter() + t_rem * 0.3)
        if c < best_c:
            best_h = h
            best_c = c

    # Phase 2: Active subgradient
    t_rem = deadline - time.perf_counter()
    if t_rem > 15.0:
        h, c = active_subgradient_phase(best_h, n, time.perf_counter() + t_rem * 0.35,
                                        top_k_start=4, top_k_end=10, max_iter=3000)
        if c < best_c:
            best_h = h
            best_c = c

    # Phase 3: Coordinate descent
    t_rem = deadline - time.perf_counter()
    if t_rem > 10.0:
        h, c = coordinate_descent_phase(best_h, n, time.perf_counter() + t_rem * 0.3,
                                        max_iter=8000)
        if c < best_c:
            best_h = h
            best_c = c

    # Phase 4: Pair swaps
    t_rem = deadline - time.perf_counter()
    if t_rem > 10.0:
        h, c = pair_swap_focus(best_h, n, time.perf_counter() + t_rem * 0.4,
                               max_iter=5000)
        if c < best_c:
            best_h = h
            best_c = c

    # Phase 5: Another subgradient pass with wider top_k
    t_rem = deadline - time.perf_counter()
    if t_rem > 8.0:
        h, c = active_subgradient_phase(best_h, n, time.perf_counter() + t_rem * 0.5,
                                        top_k_start=8, top_k_end=15, max_iter=2000)
        if c < best_c:
            best_h = h
            best_c = c

    return project_density(best_h, target), evaluate_erdos(best_h)


def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = t_start + max(1.0, budget - 15.0)
    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float("inf")
    best_n = 0

    # Try to use parent
    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 10:
            parent = np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0).astype(float, copy=True)
            parent = np.clip(parent, 0.0, 1.0)
    except Exception:
        parent = None

    # Initial evaluation of parent
    if parent is not None:
        pn = int(parent.size)
        pp = project_density(parent, 0.5 * pn)
        cp = evaluate_erdos(pp)
        best_h = pp.copy()
        best_c = cp
        best_n = pn
    else:
        n0 = 450
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        h_init = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x / 2) + 0.15 * np.sin(2 * np.pi * 5 * x / 2 + 1.0)
        best_h = project_density(h_init, 0.5 * n0)
        best_c = evaluate_erdos(best_h)
        best_n = n0

    # Define resolutions to try
    base_n = best_n if best_n > 10 else 450
    resolutions = []
    for r in [base_n, int(base_n * 1.2), int(base_n * 1.33), base_n + 50, base_n + 100]:
        if 20 <= r <= 700 and r not in resolutions:
            resolutions.append(r)
    # Ensure we include some standard ones
    for r in [360, 450, 500, 540, 600, 640]:
        if 20 <= r <= 700 and r not in resolutions:
            resolutions.append(r)
    resolutions = sorted(set(resolutions))
    # Prioritize: parent resolution, then nearby, then higher
    resolutions = [best_n] + [r for r in resolutions if r != best_n]

    # Time allocation
    total_time = deadline - time.perf_counter()
    # Weight by priority: parent gets most time, higher res gets some
    weights = []
    mid_res = median_res = (min(resolutions) + max(resolutions)) / 2
    for r in resolutions:
        if r == best_n:
            weights.append(4.0)
        elif abs(r - mid_res) < 50:
            weights.append(2.5)
        else:
            weights.append(1.5)
    wsum = sum(weights)
    time_alloc = {}
    for r, w in zip(resolutions, weights):
        time_alloc[r] = max(30.0, total_time * w / wsum)

    # Main loop over resolutions
    for ni in resolutions:
        rem = deadline - time.perf_counter()
        if rem < 35.0:
            break
        alloc = min(time_alloc.get(ni, 60.0), rem - 10.0)
        alloc = max(30.0, alloc)

        # Build starting points
        starts = []
        # Best solution upsampled
        starts.append(upsample_h(best_h, ni))
        # Parent upsampled
        if parent is not None:
            starts.append(upsample_h(parent, ni))
        # Random starts (fewer for large n)
        n_rand = 2 if ni <= 500 else 1
        for mm in range(n_rand):
            starts.append(make_start(ni, rng, mm % 5))

        # Deduplicate
        uniq = []
        for s in starts:
            s = project_density(s, 0.5 * ni)
            dup = False
            for u in uniq:
                if np.allclose(s, u, atol=1e-5):
                    dup = True
                    break
            if not dup:
                uniq.append(s)

        per_start = alloc / len(uniq)
        per_start = max(20.0, per_start)

        for s in uniq:
            rem2 = deadline - time.perf_counter()
            if rem2 < 25.0:
                break
            sub_dl = time.perf_counter() + min(per_start, rem2 - 5.0)
            h_out, c_out = full_optimize(ni, s, sub_dl, seed)
            if h_out is not None:
                h_out = project_density(h_out, 0.5 * ni)
                c_out = evaluate_erdos(h_out)
                if c_out < best_c:
                    best_h = h_out.copy()
                    best_c = c_out
                    best_n = ni

    # Final intensive polish on best solution
    rem = deadline - time.perf_counter()
    if best_h is not None and rem > 20.0:
        nbest = int(best_h.size)
        target = 0.5 * nbest
        h_fin = project_density(best_h, target)

        # Extended subgradient
        if rem > 25.0:
            h2, c2 = active_subgradient_phase(h_fin, nbest, time.perf_counter() + rem * 0.25,
                                              top_k_start=3, top_k_end=8, max_iter=5000)
            if c2 < best_c:
                h_fin = h2
                best_c = c2
                best_h = h2.copy()

        # Coordinate descent
        rem2 = deadline - time.perf_counter()
        if rem2 > 15.0:
            h3, c3 = coordinate_descent_phase(h_fin, nbest, time.perf_counter() + rem2 * 0.35,
                                              max_iter=15000)
            if c3 < best_c:
                h_fin = h3
                best_c = c3
                best_h = h3.copy()

        # Pair swaps
        rem3 = deadline - time.perf_counter()
        if rem3 > 12.0:
            h4, c4 = pair_swap_focus(h_fin, nbest, time.perf_counter() + rem3 * 0.35,
                                     max_iter=8000)
            if c4 < best_c:
                h_fin = h4
                best_c = c4
                best_h = h4.copy()

        # Final subgradient
        rem4 = deadline - time.perf_counter()
        if rem4 > 8.0:
            h5, c5 = active_subgradient_phase(h_fin, nbest, time.perf_counter() + rem4 * 0.5,
                                              top_k_start=6, top_k_end=12, max_iter=3000)
            if c5 < best_c:
                h_fin = h5
                best_c = c5
                best_h = h5.copy()

        # One more coordinate pass
        rem5 = deadline - time.perf_counter()
        if rem5 > 8.0:
            h6, c6 = coordinate_descent_phase(h_fin, nbest, time.perf_counter() + rem5 * 0.6,
                                              max_iter=10000)
            if c6 < best_c:
                h_fin = h6
                best_c = c6
                best_h = h6.copy()

    # Fallback
    if best_h is None:
        fallback = project_density(np.full(450, 0.5), 225.0)
        return fallback, evaluate_erdos(fallback), 450

    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    return final_h, float(final_c), int(final_h.size)
