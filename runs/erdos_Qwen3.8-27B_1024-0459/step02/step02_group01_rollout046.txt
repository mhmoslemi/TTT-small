import numpy as np
from scipy.optimize import minimize
import time

_g_n = 192
_g_beta = 10.0
_g_k = 5

def _obj_smooth(x):
    n = _g_n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    sh = _g_beta * (c - m)
    sh = np.clip(sh, -500.0, 0.0)
    smax = m + np.log(np.sum(np.exp(sh))) / _g_beta
    return smax * (2.0 / n)

def _obj_topk(x):
    n = _g_n
    c = np.correlate(x, 1.0 - x, mode='full')
    k = min(_g_k, len(c))
    top = np.sort(c)[-k:]
    return np.mean(top) * (2.0 / n)

def _obj_raw(x):
    n = _g_n
    c = np.correlate(x, 1.0 - x, mode='full')
    return np.max(c) * (2.0 / n)

def _eq_con(x):
    return np.sum(x) - _g_n / 2.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nfree = np.sum(free)
        if nfree > 0:
            h[free] += diff / nfree
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def corr_single_lag_grad(h, n, lag_idx):
    g = np.zeros(n)
    shift = lag_idx - (n - 1)
    if shift >= 0:
        for i in range(n - shift):
            g[i] += (1.0 - h[i + shift])
            g[i + shift] += (-h[i])
    else:
        s = -shift
        for i in range(s, n):
            g[i] += (1.0 - h[i - s])
            g[i - s] += (-h[i])
    return g

def projected_gradient(h, n, step_size, n_steps):
    dx = 2.0 / n
    h_cur = h.copy()
    best_c5 = compute_c5(h_cur)
    stall = 0
    for it in range(n_steps):
        c = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        max_val = np.max(c)
        tol = max_val * 0.03 + 1e-12
        active = c >= max_val - tol
        n_active = np.sum(active)
        if n_active == 0:
            active = np.zeros(len(c), dtype=bool)
            active[np.argmax(c)] = True
            n_active = 1
        grad = np.zeros(n)
        for lag_idx in np.where(active)[0]:
            grad += corr_single_lag_grad(h_cur, n, lag_idx)
        grad /= n_active
        h_new = h_cur - step_size * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-16:
            h_cur = h_new
            best_c5 = c5_new
            stall = 0
        else:
            stall += 1
        step_size *= 0.97
        if stall > 50 and step_size < 1e-6:
            break
    return h_cur, best_c5

def local_search(h, n, n_iters, rng_seed):
    h_cur = h.copy()
    best_c5 = compute_c5(h_cur)
    rng = np.random.RandomState(rng_seed)
    for it in range(n_iters):
        r = rng.random()
        if r < 0.4:
            idx = rng.randint(n)
            delta = rng.randn() * 0.03
            h_new = h_cur.copy()
            h_new[idx] += delta
            h_new = project_feasible(h_new, n)
        elif r < 0.8:
            i, j = rng.choice(n, 2, replace=False)
            delta = rng.randn() * 0.04
            h_new = h_cur.copy()
            h_new[i] += delta
            h_new[j] -= delta
            h_new = project_feasible(h_new, n)
        else:
            i, j, k = rng.choice(n, 3, replace=False)
            deltas = rng.randn(3) * 0.03
            h_new = h_cur.copy()
            h_new[i] += deltas[0]
            h_new[j] += deltas[1]
            h_new[k] += deltas[2]
            h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            h_cur = h_new
            best_c5 = c5_new
    return h_cur, best_c5

def optimize_at_resolution(h_init, n, time_limit):
    global _g_n, _g_beta, _g_k
    _g_n = n
    h_cur = project_feasible(h_init, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con}]
    t_start = time.time()

    stages = [
        (3.0, 60), (8.0, 100), (30.0, 150), (150.0, 200),
        (800.0, 250), (4000.0, 300), (20000.0, 400), (100000.0, 500)
    ]
    for beta, maxit in stages:
        if time.time() - t_start > time_limit:
            break
        _g_beta = beta
        try:
            res = minimize(_obj_smooth, h_cur, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxit, 'ftol': 1e-15, 'iprint': 0})
            h_cur = project_feasible(res.x, n)
        except Exception:
            pass

    if time.time() - t_start < time_limit:
        for k in [3, 5, 8, 12]:
            if time.time() - t_start > time_limit * 0.85:
                break
            _g_k = k
            try:
                res = minimize(_obj_topk, h_cur, method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 200, 'ftol': 1e-15, 'iprint': 0})
                h_cur = project_feasible(res.x, n)
            except Exception:
                pass

    if time.time() - t_start < time_limit:
        try:
            res = minimize(_obj_raw, h_cur, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 600, 'ftol': 1e-16, 'iprint': 0})
            h_cur = project_feasible(res.x, n)
        except Exception:
            pass

    return h_cur
