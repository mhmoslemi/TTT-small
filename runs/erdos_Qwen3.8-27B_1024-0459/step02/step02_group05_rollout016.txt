import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-10
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)

def interpolate_to_n(h_src, n_dst):
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def compute_grad_smooth(h, n, beta):
    """Compute gradient of soft-C5 and the C5 value."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    g = corr * dx
    c5 = float(np.max(g))
    g_max = g.max()
    # Clip for numerical stability
    exp_g = np.exp(np.clip(beta * (g - g_max), -700, 700))
    w = exp_g / exp_g.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        j = k - (n - 1)
        if j > 0:
            grad[:n-j] += wk * dx * (1.0 - h[j:])
            grad[j:] += wk * dx * (-h[:n-j])
        elif j < 0:
            jp = -j
            grad[jp:] += wk * dx * (1.0 - h[:n-jp])
            grad[:n-jp] += wk * dx * (-h[jp:])
        else:
            grad += wk * dx * (1.0 - 2.0 * h)
    
    return grad, c5

def adam_smooth_descent(h_init, n, beta, n_iters, lr, lam_init=0.0):
    """Adam-based descent on smooth C5 with equality constraint via Lagrange."""
    h = h_init.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    m_hat = 0.0
    v_hat = 0.0
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    lam = lam_init
    lr_lam = 0.1
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for t in range(1, n_iters + 1):
        grad, c5 = compute_grad_smooth(h, n, beta)
        # Add constraint gradient
        grad_cons = lam * np.ones(n)
        grad_total = grad + grad_cons
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_total
        v = beta2 * v + (1 - beta2) * (grad_total ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)

def make_feasible(h, n):

def adam_descent(h_init, n, beta, n_iters, lr, time_limit=None, start_time=None):

    lam = 0.0

                break

        grad += lam * np.ones(n)

            # subject to sum constraint (handled by adjusting after)
            
            # For each shift j, the coefficient of h_i in g_j:
            # From h_i * (1-h[i+j]) term: (1-h[i+j]) if i+j < n and i in valid range
            # From h[i-j] * (1-h_i) term: -h[i-j] if i-j >= 0 and valid
            # Also the constant part: h_i appears in the first sum as just h_i * dx
            
            # Actually, let me compute g_j as a function of h_i directly
            # by evaluating at h_i = 0 and h_i = 1, then
            # g_j(h_i) = g_j(0) + h_i * (g_j(1) - g_j(0))
            
            h_save = h[i]
            h[i] = 0.0
            g0 = compute_c5_components(h, n)  # all g_j at h_i=0
            h[i] = 1.0
            g1 = compute_c5_components(h, n)  # all g_j at h_i=1
            h[i] = h_save
            
            # g_j(h_i) = g0_j + h_i * (g1_j - g0_j)
            # minimize max_j (g0_j + h_i * (g1_j - g0_j)) over h_i in [0,1]
            a = (g1 - g0) * dx  # this is wrong, dx already included
            # Actually g0 and g1 already include dx
            a = g1 - g0  # slope per unit h_i
            b = g0  # intercept
            
            # Find h_i that minimizes max_j (a_j * h_i + b_j)
            # The minimum of the max is at the intersection of the upper envelope
            # Try h_i = 0, h_i = 1, and intersections
            best_val = np.max(b)  # at h_i = 0
            best_hi = 0.0
            
            val_at_1 = np.max(a + b)  # at h_i = 1
            if val_at_1 < best_val:
                best_val = val_at_1
                best_hi = 1.0
            
            # Check intersections of pairs on the upper envelope
            # For efficiency, only check the top few
            # The upper envelope at h_i=0 is determined by argmax(b)
            # The upper envelope at h_i=1 is determined by argmax(a+b)
            # The crossing point is where the two are equal
            
            top_at_0 = b.argmax()
            top_at_1 = (a + b).argmax()
            if top_at_0 != top_at_1 and a[top_at_1] != a[top_at_0]:
                hi_cross = (b[top_at_0] - b[top_at_1]) / (a[top_at_1] - a[top_at_0])
                if 0.0 <= hi_cross <= 1.0:
                    val_cross = b[top_at_0] + hi_cross * a[top_at_0]
                    # Also check if any other line is above at this point
                    actual_max = np.max(b + hi_cross * a)
                    if actual_max < best_val:
                        best_val = actual_max
                        best_hi = hi_cross
            
            if abs(best_hi - h_save) > 1e-10:
                h[i] = best_hi
                improved = True
        
        h = project_constraint(h, n)
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
    
    return h, best_c5

def compute_c5_components(h, n):
    """Return all correlation values (not max, the full array)."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

# Global for SLSQP constraint
_con_n = 160

def _con_func(x):
    return np.sum(x) - _con_n / 2.0

def _con_jac(x):
    return np.ones(_con_n)

def _slsqp_obj(x):
    return np.max(np.correlate(x, 1.0 - x, mode='full'))

def slsqp_polish(h, n, max_iter=3000):
    global _con_n
    _con_n = n
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_func, 'jac': _con_jac}]
    try:
        from scipy.optimize import minimize
        res = minimize(_slsqp_obj, h.copy(), method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-15, 'iprint': 0})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None
