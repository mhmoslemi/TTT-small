def project_onto_constraint(y, n):
    """Project y onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(y - mid, 0.0, 1.0)
        s = np.sum(h)
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2, 0.0, 1.0)
    # Final adjustment
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def lse_val(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -600.0, 600.0)
    return m + (1.0 / beta) * np.log(np.sum(np.exp(shifted)))

def lse_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    corr = corr_raw * dx
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -600.0, 600.0)
    w = np.exp(shifted)
    wsum = np.sum(w)
    if wsum < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= wsum

    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for j in range(n):
        g = 0.0
        for r in range(2 * n - 1):
            sh = r - (n - 1)
            wr = w[r]
            if wr < 1e-12:
                continue
            t1 = 0.0
            t2 = 0.0
            js = j - sh
            if 0 <= js < n:
                t1 = one_minus_h[js]
            js2 = j + sh
            if 0 <= js2 < n:
                t2 = h[js2]
            g += wr * (t1 - t2)
        grad[j] = dx * g
    return grad

def lbfgs_lse(h_init, n, betas, max_iter_each=3000, lam=200.0):
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    bounds = [(0.0, 1.0)] * n

    for beta in betas:
        def obj(x):
            return lse_val(x, beta) + lam * (np.sum(x) - n / 2.0) ** 2

        def grad(x):
            return lse_grad(x, beta) + 2.0 * lam * (np.sum(x) - n / 2.0) * np.ones(n)

        try:
            res = minimize(obj, best_h, jac=grad, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iter_each, 'gtol': 1e-14,
                                    'factr': 1e16, 'disp': False})
            h_opt = make_feasible(res.x, n)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
        except Exception:
            pass
    return best_h, best_c5

def proj_grad_lse(h_init, n, betas, iters_per_beta=500):
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    for beta in betas:
        h_cur = best_h.copy()
        step = 0.5
        for it in range(iters_per_beta):
            obj = lse_val(h_cur, beta)
            grad = lse_grad(h_cur, beta)
            h_new = h_cur - step * grad
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = make_feasible(h_new, n)
            obj_new = lse_val(h_new, beta)
            if obj_new > obj + 1e-8:
                step *= 0.6
            else:
                h_cur = h_new
                step = min(step * 1.05, 2.0)
            c5 = compute_c5(h_cur)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_cur.copy()
        best_h = make_feasible(best_h, n)
        best_c5 = compute_c5(best_h)
    return best_h, best_c5

def continuous_sa(h_init, n, n_steps, t_start, t_end, rng):
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    h_cur = best_h.copy()
    for step in range(n_steps):
        T = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        n_mod = rng.randint(1, 5)
        idxs = rng.choice(n, n_mod, replace=False)
        h_try = h_cur.copy()
        for idx in idxs:

        h_try = np.clip(h_try, 0.0, 1.0)

        mask = (h_try > 0.01) & (h_try < 0.99)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h_try[mask] += diff / cnt
        h_try = np.clip(h_try, 0.0, 1.0)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_h = h_try.copy()
            best_c5 = c5_try
            h_cur = h_try
        elif rng.random() < np.exp(-max(0.0, c5_try - best_c5) / max(T, 1e-15)):
            h_cur = h_try
    return best_h, best_c5

def slsqp_run(h_init, n, max_iter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]

    def obj(x):
        return float(np.max(np.correlate(x, 1.0 - x, mode='full')))

    try:
        res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-15, 'iprint': 0})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None

def fine_tune(h, n, n_steps, base_step, rng):
    h = h.copy()
    best_c5 = compute_c5(h)
    step = base_step
    for _ in range(n_steps):
        j1 = rng.randint(n)
        j2 = rng.randint(n)
        if j1 == j2:
            j2 = (j1 + 1) % n
        d1 = rng.uniform(-step, step)
        d2 = -d1
        h_try = h.copy()
        h_try[j1] += d1
        h_try[j2] += d2
        h_try = np.clip(h_try, 0.0, 1.0)
        diff = n / 2.0 - np.sum(h_try)
        if abs(diff) > 1e-10:
            mask = (h_try > 0.005) & (h_try < 0.995)
            cnt = np.count_nonzero(mask)
            if cnt > 0:
                h_try[mask] += diff / cnt
        h_try = np.clip(h_try, 0.0, 1.0)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            h = h_try
            best_c5 = c5_try
        else:
            if rng.random() < 0.05:
                step *= 0.92
    return h, best_c5
