```python
            pass  # Accept worse (SA)
        else:
            current_h = old_h  # Reject
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-13
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def _sm_fun_global(x):
    return np.sum(x) - _g_target

def _sm_jac_global(x):
    return np.ones_like(x)

_g_target = 0.0

def slsqp_softmax(n, h_init, beta, max_iter=3000):
    global _g_target
    _g_target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sm_fun_global, 'jac': _sm_jac_global}]

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        cmax = np.max(corr)
        scaled = beta * (corr - cmax)
        return cmax + (1.0 / beta) * np.log(np.sum(np.exp(scaled)))

    try:
        res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-15, 'iprint': 0})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None

def guided_sa_search(h, n, max_iter, rng, time_limit, t_start=0.012, t_end=0.0005,
                     target_frac=0.5, rand_frac=0.35):
    current_h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for it in range(max_iter):
        if time.time() > time_limit:
            break

        T = t_start * (t_end / t_start) ** (it / max(max_iter - 1, 1))
        old_h = current_h.copy()

        corr = np.correlate(current_h, 1.0 - current_h, mode='full')
        worst_idx = np.argmax(corr)
        shift = worst_idx - (n - 1)

        moved = False
        if shift != 0:
            if shift > 0:
                valid_i = np.arange(n - shift)
            else:
                valid_i = np.arange(-shift, n)
            j_valid = valid_i + shift
            contribs = current_h[valid_i] * (1.0 - current_h[j_valid])

            if len(contribs) > 0 and contribs.max() > 1e-8:
                thresh = contribs.max() * target_frac
                candidates = np.where(contribs > thresh)[0]
                if len(candidates) > 0:
                    local_idx = candidates[rng.randint(len(candidates))]
                    i = valid_i[local_idx]
                    j = j_valid[local_idx]
                    delta = rng.choice([0.01, 0.02, 0.03, 0.05, 0.07])
                    if current_h[i] >= delta and current_h[j] <= 1.0 - delta:
                        current_h[i] -= delta
                        current_h[j] += delta
                        moved = True

        if not moved or rng.random() < rand_frac:
            k_mod = rng.randint(1, 4)
            idxs = rng.choice(n, k_mod, replace=False)
            for idx in idxs:
                current_h[idx] += rng.uniform(-0.025, 0.025)

        current_h = np.clip(current_h, 0.0, 1.0)

        diff = n / 2.0 - np.sum(current_h)
        if abs(diff) > 1e-12:
            mask = (current_h > 0.003) & (current_h < 0.997)
            cnt = np.count_nonzero(mask)
            if cnt > 0:
                current_h[mask] += diff / cnt
            current_h = np.clip(current_h, 0.0, 1.0)

        new_c5 = compute_c5(current_h)

        if new_c5 <= best_c5:
            best_h = current_h.copy()
            best_c5 = new_c5
        elif rng.random() < np.exp(-(new_c5 - best_c5) / max(T, 1e-15)):
            pass
        else:
            current_h = old_h

    return best_h, best_c5

def greedy_worst_shift(h, n, max_iter, rng, time_limit):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    for it in range(max_iter):
        if time.time() > time_limit:
            break

        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        worst_vals = np.sort(corr)[::-1][:3]
        worst_idx_list = np.argsort(corr)[::-1][:3]

        improved = False
        for worst_idx in worst_idx_list:
            shift = worst_idx - (n - 1)
            if shift == 0:
                continue
            if shift > 0:
                valid_i = np.arange(n - shift)
            else:
                valid_i = np.arange(-shift, n)
            j_valid = valid_i + shift
            contribs = best_h[valid_i] * (1.0 - best_h[j_valid])

            if len(contribs) == 0 or contribs.max() < 1e-8:
                continue

            top_local = np.argmax(contribs)
            i = valid_i[top_local]
            j = j_valid[top_local]

            for delta in [0.06, 0.03, 0.015, 0.008, 0.004]:
                if best_h[i] < delta or best_h[j] > 1.0 - delta:
                    continue
                h_try = best_h.copy()
                h_try[i] -= delta
                h_try[j] += delta
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best_h = h_try
                    best_c5 = c5_try
                    improved = True
                    break
            if best_h is not h:
                pass
            if improved:
                break

        if not improved:
            no_improve += 1
            if no_improve > 2000:
                break
        else:
            no_improve = 0

    return best_h, best_c5

def binary_sa(n, rng, n_steps=50000, t_start=0.5, t_end=0.01, time_limit=None):
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for step in range(n_steps):
        if time_limit is not None and time.time() > time_limit:
            break
        T = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_h = h.copy()
            best_c5 = c5_new
        elif rng.random() < np.exp(-(c5_new - best_c5) / max(T, 1e-15)):
            pass
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_c5
```
