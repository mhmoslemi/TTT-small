    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

    shifted = np.clip(beta * (corr - m), -500, 0)
    exps = np.exp(shifted)
    return exps / np.sum(exps)

    val = smooth_max_val(corr, beta) * dx

    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmin(np.maximum(h, 1.0 - h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)

def gradient_optimize(h0, n, beta_schedule, max_time):
    """Multi-stage gradient optimization."""
    x = h0.copy()
    start = time.time()
    for beta in beta_schedule:
        if time.time() - start > max_time:
            break

            return obj_and_grad(x, beta)[0]

        x = project_feasible(res.x, n)
    return x

            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()
                best_n = n
                print("  [{}{}] round: C5={:.6f}".format(init_idx, '', c5_bin))

            if remaining() > 20:
                t_ls = min(remaining() * 0.35, 80)
                h_ls, c5_ls = binary_local_search_1flip(h_bin, rng, t_ls)
                if c5_ls < best_c5:
                    best_c5 = c5_ls
                    best_h = h_ls.copy()
                    best_n = n
                    print("  [{}{}] 1flip: C5={:.6f}".format(init_idx, '', c5_ls))

            if remaining() > 20:
                t_ls2 = min(remaining() * 0.25, 50)
                h_2f, c5_2f = binary_local_search_2flip(h_ls, rng, t_ls2)
                if c5_2f < best_c5:
                    best_c5 = c5_2f
                    best_h = h_2f.copy()
                    best_n = n
                    print("  [{}{}] 2flip: C5={:.6f}".format(init_idx, '', c5_2f))

    # Final intensive local search on best solution
    if remaining() > 30:
        if len(np.unique(best_h)) <= 2:
            h_final, c5_final = binary_local_search_1flip(best_h, rng, min(remaining() * 0.5, 120))
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final.copy()
            h_final2, c5_final2 = binary_local_search_2flip(best_h, rng, min(remaining() * 0.3, 60))
            if c5_final2 < best_c5:
                best_c5 = c5_final2
                best_h = h_final2.copy()
            print("Final LS: C5={:.6f}".format(best_c5))
        else:
            # Do one more gradient pass then round
            x = best_h.copy()
            if remaining() > 40:
                try:
                    x = run_gradient_opt(x, best_n, start_time, budget_s)
                except Exception:
                    pass
            h_bin = binary_round(x, best_n)
            c5_b = compute_c5(h_bin)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_bin.copy()
            if remaining() > 20:
                h_ls, c5_ls = binary_local_search_1flip(best_h, rng, min(remaining() * 0.5, 60))
                if c5_ls < best_c5:
                    best_c5 = c5_ls
                    best_h = h_ls.copy()

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
