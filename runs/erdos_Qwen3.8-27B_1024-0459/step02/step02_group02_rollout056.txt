def compute_c5(h):

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-13
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

        h[pos_0] = 1.0
        new_c5 = compute_c5(h)
        
        delta = new_c5 - cur_c5
        if delta < 0 or (T > 1e-30 and rng.rand() < np.exp(-delta / T)):
            cur_c5 = new_c5
            ones_arr[ai] = pos_0
            zeros_arr[bi] = pos_1
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[pos_1] = 1.0
            h[pos_0] = 0.0
    
    return best_h, best_c5

def greedy_binary(h_init, n, rng, max_iterations, max_seconds):
    """Greedy local search: only accept improving moves."""
    h = round_to_binary(h_init.copy(), n, rng)
    cur_c5 = compute_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    
    ones_arr = np.where(h == 1.0)[0].copy()
    zeros_arr = np.where(h == 0.0)[0].copy()
    n_half = len(ones_arr)
    
    start_t = time.time()
    no_improve = 0
    stall_limit = 20000
    
    for step in range(max_iterations):
        if time.time() - start_t > max_seconds:
            break
        if no_improve > stall_limit:
            break
        
        # Try focused moves first, then random
        if step % 3 != 2:
            # Focused
            corr = np.correlate(h, 1.0 - h, mode='full')
            worst_idx = int(np.argmax(corr))
            s = worst_idx - (n - 1)
            candidates = []
            if s >= 0:
                for ii in range(n - s):
                    if h[ii] == 1.0 and h[ii + s] == 0.0:
                        candidates.append(ii)
            else:
                for ii in range(-s, n):
                    if h[ii] == 1.0 and h[ii + s] == 0.0:
                        candidates.append(ii)
            
            if len(candidates) > 0:
                pos_1 = candidates[rng.randint(len(candidates))]
                ai = int(np.where(ones_arr == pos_1)[0][0])
                bi = rng.randint(n_half)
            else:
                ai = rng.randint(n_half)
                bi = rng.randint(n_half)
        else:
            ai = rng.randint(n_half)
            bi = rng.randint(n_half)
        
        pos_1 = ones_arr[ai]
        pos_0 = zeros_arr[bi]
        if pos_1 == pos_0:
            continue
        
        h[pos_1] = 0.0
        h[pos_0] = 1.0
        new_c5 = compute_c5(h)
        
        if new_c5 < cur_c5 - 1e-15:
            cur_c5 = new_c5
            ones_arr[ai] = pos_0
            zeros_arr[bi] = pos_1
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            no_improve = 0
        else:
            h[pos_1] = 1.0
            h[pos_0] = 0.0
            no_improve += 1
    
    return best_h, best_c5

def _sum_con_fun(x):
    return float(np.sum(x)) - _N_GLOBAL / 2.0

def _sum_con_jac(x):
    return np.ones(_N_GLOBAL)

def _smooth_max_obj(x, beta):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    shifted = np.minimum(shifted, 50.0)
    val = cmax + (1.0 / beta) * np.log(np.sum(np.exp(shifted)))
    return val

def _direct_max_obj(x):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr))

def continuous_polish(h_init, n, max_iter=2000):
    global _N_GLOBAL
    _N_GLOBAL = n
    h0 = make_feasible(h_init.copy(), n)
    bounds = [(0.0, 1.0)] * n
    con = {'type': 'eq', 'fun': _sum_con_fun, 'jac': _sum_con_jac}
    
    # Coarse-to-fine on beta
    current = h0.copy()
    best_c5 = compute_c5(current)
    
    for beta in [10.0, 30.0, 100.0, 300.0]:
        try:
            res = minimize(_smooth_max_obj, current, args=(beta,), method='SLSQP',
                          bounds=bounds, constraints=con,
                          options={'maxiter': max_iter // 4, 'ftol': 1e-14})
            if res.fun < 1e10:
                candidate = make_feasible(res.x, n)
                c5_val = compute_c5(candidate)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    current = candidate
        except Exception:
            pass
    
    # Final polish with direct objective
    try:
        res = minimize(_direct_max_obj, current, method='SLSQP',
                      bounds=bounds, constraints=con,
                      options={'maxiter': max_iter // 2, 'ftol': 1e-15})
        if res.fun < 1e10:
            candidate = make_feasible(res.x, n)
            c5_val = compute_c5(candidate)
            if c5_val < best_c5:
                best_c5 = c5_val
                current = candidate
    except Exception:
        pass
    
    return current, best_c5
