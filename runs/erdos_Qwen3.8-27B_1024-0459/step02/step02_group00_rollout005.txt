        for sweep in range(40):
            if remaining_time() < 3:
                break
            any_imp = False
            order = rng.permutation(n)
            for i in order:
                if remaining_time() < 2:
                    break
                old_val = h[i]
                candidates = [0.0, 1.0, 0.2, 0.8, 0.33, 0.67,
                              max(0.0, old_val - 0.05), min(1.0, old_val + 0.05)]
                found = False
                for new_val in candidates:
                    if abs(new_val - old_val) < 1e-7:
                        continue
                    h[i] = new_val
                    h = project_feasible(h, n)
                    c5_test = compute_c5(h)
                    if c5_test < best_c5:
                        best_c5 = c5_test
                        best_h = h.copy()
                        any_imp = True
                        found = True
                        break
                    else:
                        h[i] = old_val
                if not found:
                    # Try tiny perturbations
                    for delta in [0.02, -0.02, 0.01, -0.01]:
                        nv = old_val + delta
                        if nv < 0 or nv > 1:
                            continue
                        h[i] = nv
                        h = project_feasible(h, n)
                        c5_test = compute_c5(h)
                        if c5_test < best_c5:
                            best_c5 = c5_test
                            best_h = h.copy()
                            any_imp = True
                            found = True
                            break
                        else:
                            h[i] = old_val
                if found:
                    h = best_h.copy()
            if not any_imp:
                break

    # Phase 4: Final high-precision polish
    if remaining_time() > 12:
        n = best_n
        _G_N = n
        _G_TGT = float(n) / 2.0
        h = best_h.copy()
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': constrain_sum}]

        print("\n=== Final polish (n={}) ===".format(n))

        for beta_val in [500, 1000, 2000, 5000]:
            if remaining_time() < 5:
                break
            _G_BETA = beta_val
            try:
                res = minimize(smooth_obj_func, h, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 400, 'ftol': 1e-16, 'iprint': 0})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new
            except Exception:
                pass

        # One more raw pass
        if remaining_time() > 5:
            try:
                res = minimize(raw_obj_func, h, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 300, 'ftol': 1e-16, 'iprint': 0})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            except Exception:
                pass

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.7f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
