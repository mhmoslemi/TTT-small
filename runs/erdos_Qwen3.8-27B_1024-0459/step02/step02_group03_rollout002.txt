    out = np.empty_like(x)
    pos = x >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))

import numpy as np

def sigmoid_safe(x):
    x = np.asarray(x, dtype=float)
    out = np.empty_like(x)
    pos = x >= 0.0
    neg = ~pos
    if np.any(pos):
        out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    if np.any(neg):
        e = np.exp(x[neg])
        out[neg] = e / (1.0 + e)
    return out


def zeta_for_z(z, target):
    lo = float(np.min(z)) - 40.0
    hi = float(np.max(z)) + 40.0
    if float(np.sum(sigmoid_safe(z - lo))) < target:
        lo -= 50.0
    if float(np.sum(sigmoid_safe(z - hi))) > target:
        hi += 50.0
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(sigmoid_safe(z - mid)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return 0.5 * (lo + hi)


def h_from_z_vars(zv, n):
    n = int(n)
    if n <= 1:
        return np.array([0.5])
    z = np.empty(n, dtype=float)
    z[0] = 0.0
    z[1:] = np.asarray(zv, dtype=float)
    zeta = zeta_for_z(z, 0.5 * n)
    return sigmoid_safe(z - zeta)


def autocorr_lin(h):
    n = int(h.size)
    if n == 0:
        return np.zeros(0)
    if n <= 300:
        c = np.correlate(h, h, mode='full')
        return c[n - 1:].astype(float, copy=False)
    N = 1
    while N < 2 * n:
        N <<= 1
    F = np.fft.rfft(h, N)
    return np.fft.irfft(F * np.conj(F), N)[:n]


def values_from_h(h, dx):
    n = int(h.size)
    if n == 0:
        return np.zeros(0)
    c = autocorr_lin(h)
    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    pref[1:] = np.cumsum(h)
    total = pref[-1]
    vals = np.empty(2 * n - 1, dtype=float)
    vals[0] = (total - c[0]) * dx
    for L in range(1, n):
        vals[L] = (pref[n - L] - c[L]) * dx
        vals[n - 1 + L] = (total - pref[L] - c[L]) * dx
    return vals


def smooth_max_value(vals, beta):
    if beta <= 1e-8:
        return float(np.max(vals))
    m = float(np.max(vals))
    shifted = np.clip(beta * (vals - m), -700.0, 0.0)
    return m + float(np.log(np.sum(np.exp(shifted)))) / beta


def smooth_max_weights(vals, beta):
    m = float(np.max(vals))
    w = np.zeros_like(vals, dtype=float)
    if beta <= 1e-8:
        idx = int(np.argmax(vals))
        w[idx] = 1.0
        return w
    shifted = np.clip(beta * (vals - m), -700.0, 0.0)
    exps = np.exp(shifted)
    s = float(np.sum(exps))
    if s <= 0.0:
        idx = int(np.argmax(vals))
        w[idx] = 1.0
        return w
    return exps / s


def product_grad_from_v(h, v):
    n = int(h.size)
    if n == 1:
        return np.zeros(1, dtype=float)
    if v.size == 0 or not np.any(v):
        return np.zeros(n, dtype=float)
    N = 1
    while N < 2 * n:
        N <<= 1
    k = np.zeros(N, dtype=float)
    idx = np.arange(1, n)
    k[idx] = v
    k[N - idx] = v
    Fh = np.fft.rfft(h, N)
    Fk = np.fft.rfft(k, N)
    return np.fft.irfft(Fh * Fk, N)[:n]


def grad_h_from_weights(h, w, dx):
    n = int(h.size)
    if n == 1:
        return np.array([dx * w[0] * (1.0 - 2.0 * h[0])])
    w0 = w[0]
    wP = w[1:n]
    wN = w[n:2 * n - 1]
    v = wP + wN
    prod = product_grad_from_v(h, v)
    cumP = np.cumsum(wP)
    cumN = np.cumsum(wN)
    cntP = np.arange(n - 1, -1, -1)
    cntN = np.arange(n)
    linear = np.zeros(n, dtype=float)
    posP = cntP > 0
    if np.any(posP):
        linear[posP] = cumP[cntP[posP] - 1]
    posN = cntN > 0
    if np.any(posN):
        linear[posN] += cumN[cntN[posN] - 1]
    return dx * (w0 * (1.0 - 2.0 * h) - prod + linear)


def z_objective(zvars):
    n = int(Z_N)
    if n == 0:
        return 0.0
    h = h_from_z_vars(zvars, n)
    vals = values_from_h(h, Z_DX)
    return smooth_max_value(vals, Z_BETA)


def z_gradient(zvars):
    n = int(Z_N)
    if n <= 1:
        return np.zeros(0, dtype=float)
    h = h_from_z_vars(zvars, n)
    vals = values_from_h(h, Z_DX)
    w = smooth_max_weights(vals, Z_BETA)
    g = grad_h_from_weights(h, w, Z_DX)
    pw = h * (1.0 - h)
    D = float(np.sum(pw))
    if D < 1e-14:
        return np.zeros(n - 1, dtype=float)
    E = float(np.dot(g, pw))
    gz = pw * (g - E / D)
    gz[0] = 0.0
    if not np.all(np.isfinite(gz)):
        return np.zeros(n - 1, dtype=float)
    return gz[1:]


def exact_c5(h):
    n = int(h.size)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx


def fast_c5(h):
    n = int(h.size)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    if n <= 300:
        return exact_c5(h)
    return float(np.max(values_from_h(h, dx)))


def project_box_sum(y, target):
    y = np.asarray(y, dtype=float).copy()
    n = int(y.size)
    if n == 0:
        return y
    if target <= 0.0:
        return np.clip(y, 0.0, 1.0)
    if target >= float(n):
        return np.clip(y, 0.0, 1.0)
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(8):
        diff = target - float(np.sum(x))
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            slack = 1.0 - x
            idx = int(np.argmax(slack))
            add = min(slack[idx], diff)
            if add <= 0.0:
                break
            x[idx] += add
        else:
            slack = x
            idx = int(np.argmax(slack))
            add = min(slack[idx], -diff)
            if add <= 0.0:
                break
            x[idx] -= add
        x = np.clip(x, 0.0, 1.0)
    return x


def interp_h(h_src, n_dst):
    n_src = int(len(h_src))
    n_dst = int(n_dst)
    if n_src == n_dst:
        return project_box_sum(h_src.copy(), 0.5 * n_dst)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_box_sum(h_dst, 0.5 * n_dst)


def h_to_z_vars(h, zmax=35.0):
    h = np.asarray(h, dtype=float)
    if h.size <= 1:
        return np.zeros(0, dtype=float)
    eps = 1e-12
    hc = np.clip(h, eps, 1.0 - eps)
    z = np.log(hc) - np.log1p(-hc)
    zref = z[0]
    zv = z[1:] - zref
    zv = np.nan_to_num(zv, nan=0.0, posinf=zmax, neginf=-zmax)
    return np.clip(zv, -zmax, zmax)

    return exact_c5(h)


def coordinate_polish_z(n, h0, c0, max_time_s, seed):
    n = int(n)
    if n <= 1 or max_time_s <= 5.0:
        h_best = project_box_sum(np.asarray(h0, dtype=float), 0.5 * n)
        return h_best, exact_c5(h_best)
    start = time.time()
    target = 0.5 * n
    h_best = project_box_sum(np.asarray(h0, dtype=float), target)
    c_best = float(c0)
    c_probe = exact_c5(h_best)
    if c_probe < c_best:
        c_best = c_probe
    z_current = h_to_z_vars(h_best)
    zmax = 35.0
    deltas = (0.5, -0.5, 0.25, -0.25, 0.125, -0.125, 0.0625, -0.0625, 0.03125, -0.03125)
    rng = np.random.RandomState(seed)
    for _ in range(4):
        order = rng.permutation(n - 1)
        for idx in order:
            if time.time() - start > max_time_s - 4.0:
                break
            base = float(z_current[idx])
            best_d = 0.0
            best_fast = c_best
            zv = z_current.copy()
            for d in deltas:
                nv = base + d
                if nv < -zmax or nv > zmax:
                    continue
                zv[idx] = nv
                try:
                    h_t = h_from_z_vars(zv, n)
                    c_t = fast_c5(h_t)
                except Exception:
                    continue
                if c_t < best_fast - 1e-14:
                    best_fast = c_t
                    best_d = d
            if best_d != 0.0:
                zv = z_current.copy()
                zv[idx] = z_current[idx] + best_d
                h_t = project_box_sum(h_from_z_vars(zv, n), target)
                c_exact = exact_c5(h_t)
                if c_exact < c_best - 1e-14:
                    z_current = zv.copy()
                    h_best = h_t
                    c_best = c_exact
    return h_best, c_best


def mass_pair_search(h0, c0, max_seconds, seed):
    n = int(h0.size)
    if n <= 1 or max_seconds <= 5.0:
        h_best = project_box_sum(np.asarray(h0, dtype=float), 0.5 * n)
        return h_best, exact_c5(h_best)
    target = 0.5 * n
    h_best = project_box_sum(np.asarray(h0, dtype=float), target)
    c_best = exact_c5(h_best)
    if c0 < c_best:
        h_best = project_box_sum(np.asarray(h0, dtype=float), target)
        c_best = float(c0)
    hcur = h_best.copy()
    ccur = c_best
    start = time.time()
    rng = np.random.RandomState(seed)
    if n <= 180:
        rate = 400
    elif n <= 280:
        rate = 160
    else:
        rate = 60
    max_trials = int(max(250, max_seconds * rate))
    no_improve = 0
    scale = 0.08
    for _ in range(max_trials):
        if time.time() - start > max_seconds - 2.0:
            break
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        low = -hcur[j]
        high = 1.0 - hcur[i]
        if high <= low:
            continue
        d = scale * (high - low) * rng.uniform(-1.0, 1.0)
        hv = hcur.copy()
        hv[i] += d
        hv[j] -= d
        if hv[i] < -1e-14 or hv[i] > 1.0 + 1e-14 or hv[j] < -1e-14 or hv[j] > 1.0 + 1e-14:
            hv = np.clip(hv, 0.0, 1.0)
            if abs(float(np.sum(hv)) - target) > 1e-9:
                continue
        cnew = fast_c5(hv)
        if cnew < ccur - 1e-14:
            hcur = hv
            ccur = cnew
            no_improve = 0
            if ccur < c_best - 1e-14:
                c_best = ccur
                h_best = hcur.copy()
                if c_best <= TARGET_C5 + 1e-12:
                    return h_best, c_best
        else:
            no_improve += 1
            if no_improve > 1200:
                hcur = h_best.copy()
                ccur = c_best
                no_improve = 0
                scale = min(0.08, max(0.01, scale * 0.5))
    return h_best, c_best


def binary_local_search(start_h, start_c, max_seconds, seed):
    n = int(start_h.size)
    if n <= 1 or n % 2 != 0 or max_seconds <= 3.0:
        py = project_box_sum(np.asarray(start_h, dtype=float), 0.5 * n)
        return py, exact_c5(py)

    hcur = np.asarray(start_h, dtype=float).copy()
    if not (np.all(hcur <= 1e-12) or True):
        pass
    if not (np.all((hcur <= 1e-10) | (hcur >= 1.0 - 1e-10))):
        hcur = project_box_sum(hcur, 0.5 * n)
        if not (np.all((hcur <= 1e-10) | (hcur >= 1.0 - 1e-10))):
            hcur = round_to_sum(hcur, n)

    best_h_global = np.asarray(start_h, dtype=float).copy()
    best_c_global = float(start_c)
    cur_c = fast_c5(hcur)
    if cur_c < best_c_global - 1e-14:
        best_c_global = cur_c
        best_h_global = hcur.copy()
        if best_c_global <= TARGET_C5 + 1e-12:
            return best_h_global, best_c_global

    rng = np.random.RandomState(seed)
    start = time.time()
    if n <= 120:
        rate = 2000
    elif n <= 200:
        rate = 800
    elif n <= 320:
        rate = 300
    else:
        rate = 120
    max_trials = int(max(400, max_seconds * rate))
    no_improve = 0
    for _ in range(max_trials):
        if time.time() - start > max_seconds - 2.0:
            break
        while True:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i != j and hcur[i] == 1.0 and hcur[j] == 0.0:
                break
        hcur[i] = 0.0
        hcur[j] = 1.0
        cnew = fast_c5(hcur)
        if cnew < cur_c - 1e-15:
            cur_c = cnew
            no_improve = 0
            if cnew < best_c_global - 1e-14:
                best_c_global = cnew
                best_h_global = hcur.copy()
                if best_c_global <= TARGET_C5 + 1e-12:
                    return best_h_global, best_c_global
        else:
            hcur[i] = 1.0
            hcur[j] = 0.0
            no_improve += 1
            if no_improve > 2000:
                for _flip in range(3):
                    while True:
                        i = int(rng.randint(n))
                        j = int(rng.randint(n))
                        if i != j and hcur[i] == 1.0 and hcur[j] == 0.0:
                            break
                    hcur[i] = 0.0
                    hcur[j] = 1.0
                cur_c = fast_c5(hcur)
                if cur_c > best_c_global + 0.05:
                    hcur = round_to_sum(best_h_global, n)
                    cur_c = fast_c5(hcur)
                no_improve = 0
    if best_c_global > float(start_c):
        h0 = project_box_sum(np.asarray(start_h, dtype=float), 0.5 * n)
        return h0, exact_c5(h0)
    return best_h_global, best_c_global


def optimize_z_n(n, h0, max_time_s, seed, full=True):
    n = int(n)
    start = time.time()
    target = 0.5 * n
    dx = 2.0 / n
    h_best = project_box_sum(np.asarray(h0, dtype=float), target)
    c_best = exact_c5(h_best)
    if n <= 1 or max_time_s < 15.0:
        return h_best, c_best
    if c_best <= TARGET_C5 + 1e-12:
        return h_best, c_best

    global Z_N, Z_DX, Z_TARGET, Z_BETA
    Z_N = n
    Z_DX = dx
    Z_TARGET = target

    zmax = 35.0
    z_cont = h_to_z_vars(h_best, zmax)
    z_pert_base = z_cont.copy()

    if full:
        if n <= 180:
            betas = (2.0, 8.0, 25.0, 80.0, 250.0, 800.0, 2500.0, 8000.0)
            base_iter = 400
        elif n <= 260:
            betas = (3.0, 10.0, 30.0, 100.0, 300.0, 1000.0, 3000.0)
            base_iter = 300
        elif n <= 400:
            betas = (4.0, 15.0, 50.0, 150.0, 500.0, 2000.0, 6000.0)
            base_iter = 200
        else:
            betas = (5.0, 20.0, 70.0, 250.0, 1000.0, 4000.0)
            base_iter = 120
        beta_limit = max_time_s * 0.62
        do_perturb = True
    else:
        betas = (500.0, 1500.0, 5000.0)
        base_iter = 150 if n <= 220 else 90
        beta_limit = max_time_s * 0.55
        do_perturb = False

    bounds = [(-zmax, zmax)] * (n - 1)

    for beta in betas:
        if time.time() - start > beta_limit:
            break
        if c_best <= TARGET_C5 + 1e-12:
            break
        Z_BETA = float(beta)
        maxiter = base_iter
        if beta > 1000.0:
            maxiter = max(60, int(base_iter * 0.7))
        if beta > 5000.0:
            maxiter = max(40, int(base_iter * 0.5))
        try:
            res = minimize(
                z_objective,
                z_cont,
                jac=z_gradient,
                method='L-BFGS-B',
                bounds=bounds,
                options={
                    'maxiter': maxiter,
                    'gtol': 1e-12,
                    'ftol': 1e-15,
                    'maxls': 40,
                }
            )
            if res is not None and np.all(np.isfinite(res.x)):
                z_try = np.asarray(res.x, dtype=float)
                h_raw = h_from_z_vars(z_try, n)
                h_try = project_box_sum(h_raw, target)
                c_try = exact_c5(h_try)
                if c_try < c_best - 1e-14:
                    c_best = c_try
                    h_best = h_try.copy()
                    z_pert_base = h_to_z_vars(h_best, zmax)
                z_cont = np.clip(z_try, -zmax, zmax)
        except Exception:
            pass

    if full and time.time() - start < max_time_s * 0.82 and c_best > TARGET_C5 + 1e-12:
        rng = np.random.RandomState(seed)
        for sigma in (0.15, 0.45, 0.9):
            for _trial in range(2):
                if time.time() - start > max_time_s * 0.85:
                    break
                if c_best <= TARGET_C5 + 1e-12:
                    break
                z_trial = np.clip(z_pert_base + rng.normal(0.0, sigma, n - 1), -zmax, zmax)
                Z_BETA = 500.0
                maxiter = 70 if n <= 220 else 45
                try:
                    res = minimize(
                        z_objective,
                        z_trial,
                        jac=z_gradient,
                        method='L-BFGS-B',
                        bounds=bounds,
                        options={
                            'maxiter': maxiter,
                            'gtol': 1e-11,
                            'ftol': 1e-14,
                            'maxls': 30,
                        }
                    )
                    if res is not None and np.all(np.isfinite(res.x)):
                        z_try = np.asarray(res.x, dtype=float)
                        h_try = project_box_sum(h_from_z_vars(z_try, n), target)
                        c_try = exact_c5(h_try)
                        if c_try < c_best - 1e-14:
                            c_best = c_try
                            h_best = h_try.copy()
                            z_pert_base = h_to_z_vars(h_best, zmax)
                except Exception:
                    pass

    coord_limit = max_time_s * (0.93 if full else 0.90)
    if time.time() - start < coord_limit and c_best > TARGET_C5 + 1e-12:
        remain = max(0.0, coord_limit - (time.time() - start))
        h_best, c_best = coordinate_polish_z(n, h_best, c_best, remain, seed + 17)

    if time.time() - start < max_time_s * 0.97 and c_best > TARGET_C5 + 1e-12:
        rem = max(0.0, max_time_s * 0.97 - (time.time() - start))
        if rem > 12.0:
            limit = min(25.0, rem * 0.6)
            h_best, c_best = mass_pair_search(h_best, c_best, limit, seed + 91)

    if n % 2 == 0 and time.time() - start < max_time_s - 6.0 and c_best > TARGET_C5 + 1e-12:
        h_bin = round_to_sum(h_best, n)
        c_bin = fast_c5(h_bin)
        if c_bin < c_best - 1e-14:
            h_best = h_bin.copy()
            c_best = c_bin
        rem = max_time_s - (time.time() - start) - 5.0
        if rem > 12.0 and c_best > TARGET_C5 + 1e-12:
            limit = min(30.0, rem * 0.55)
            hb, cb = binary_local_search(h_best, c_best, limit, seed + 1000 + n)
            if cb < c_best - 1e-14:
                h_best = hb
                c_best = cb

    h_best = project_box_sum(h_best, target)
    c_best = exact_c5(h_best)
    return h_best, c_best


def round_to_sum(h, n):
    n = int(n)
    h = np.asarray(h, dtype=float)
    order = np.argsort(h)[::-1]
    out = np.zeros(n, dtype=float)
    target = 0.5 * n
    if abs(target - round(target)) < 1e-12:
        k = int(round(target))
        if k > 0:
            out[order[:k]] = 1.0
    else:
        k = int(np.floor(target))
        val = target - float(k)
        if k > 0:
            out[order[:k]] = 1.0
        if k < n and val > 1e-15:
            out[order[k]] = val
    return out


def run(seed=42, budget_s=1300, **kwargs):
    start = time.time()
    best_h = None
    best_c = float('inf')
    best_n = 0

    try:
        if 'initial_h_values' in globals():
            par = np.array(initial_h_values, dtype=float)
            if par.size > 0:
                par = project_box_sum(par, 0.5 * par.size)
                best_h = par
                best_c = exact_c5(par)
                best_n = int(par.size)
    except Exception:
        pass

    if best_h is None:
        best_n = 200
        best_h = make_initial_smooth(best_n, seed)
        best_c = exact_c5(best_h)

    base_n = int(best_n)
    cands = []

    def add_cand(nn):
        nn = int(nn)
        if 48 <= nn <= 600 and nn not in cands:
            cands.append(nn)

    add_cand(base_n)
    add_cand(base_n // 2)
    add_cand(int(round(1.25 * base_n)))
    add_cand(int(round(1.5 * base_n)))

    if len(cands) == 0:
        cands.append(200)

    fracs = [0.50, 0.20, 0.15, 0.15]
    while len(fracs) < len(cands):
        fracs.append(0.05)
    fracs = fracs[:len(cands)]
    sfrac = float(sum(fracs))
    fracs = [f / sfrac for f in fracs]

    total_time = max(120.0, float(budget_s) - 80.0)
    if total_time > float(budget_s) - 60.0:
        total_time = max(60.0, float(budget_s) - 60.0)

    for i, nn in enumerate(cands):
        elapsed = time.time() - start
        if float(budget_s) - elapsed < 80.0:
            break
        alloc = total_time * fracs[i]
        tl = min(alloc, float(budget_s) - 80.0 - elapsed)
        if tl < 20.0:
            break
        if nn == int(best_n):
            h_start = best_h
        else:
            h_start = interp_h(best_h, nn)
        h_opt, c_opt = optimize_z_n(nn, h_start, tl, int(seed) * 1000 + i + 1, full=True)
        if c_opt < best_c:
            best_h = h_opt
            best_c = c_opt
            best_n = int(nn)
            if best_c <= TARGET_C5 + 1e-12:
                break

    if int(best_n) % 2 == 0 and time.time() < float(budget_s) - 60.0 and best_c > TARGET_C5 + 1e-12:
        limit = min(45.0, (float(budget_s) - 60.0 - (time.time() - start)) * 0.5)
        if limit > 10.0:
            hb, cb = binary_local_search(best_h, best_c, limit, int(seed) + 7777)
            if cb < best_c:
                best_h = hb
                best_c = cb

    best_h = project_box_sum(best_h, 0.5 * int(best_n))
    final_c = exact_c5(best_h)

    try:
        ev = evaluate_erdos_solution(best_h)
        if ev is not None:
            ev = float(ev)
            if np.isfinite(ev) and ev < final_c:
                final_c = ev
    except Exception:
        pass

    return (best_h, float(final_c), int(best_n))
