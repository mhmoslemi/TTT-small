def make_fourier_round(n, base_h, num_bands):
    h = base_h.copy()
    n_base = len(h)
    if n_base != n:
        h = interp_h(h, n)
    N = len(h)
    freqs = np.fft.rfftfreq(N) * N
    coeffs = np.fft.rfft(h)
    for i in range(1, len(coeffs)):
        f = freqs[i]
        if f > num_bands:
            coeffs[i] = 0.0
            if i < N - 1:
                coeffs[N - i] = 0.0
    h_new = np.fft.irfft(coeffs, N)
    return project_feasible(h_new, N)

def slsqp_optimize(h0, n, beta, max_iter=300):
    def f(x):
        return obj_func(x, beta)
    def g(x):
        return grad_func(x, beta)
    def c(x):
        return np.sum(x) - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': c}]
    try:
        res = minimize(f, h0, jac=g, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
        h_out = project_feasible(res.x, n)
        return h_out, compute_c5(h_out)
    except Exception:
        return project_feasible(h0, n), compute_c5(h0)

def projected_gradient_opt(h0, n, beta, num_steps, lr_init):
    h = h0.copy()
    f_prev = obj_func(h, beta)
    lr = lr_init
    for i in range(num_steps):
        grad = grad_func(h, beta)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        acceptable = False
        lr_try = lr
        for bt in range(40):
            h_try = project_feasible(h - lr_try * grad, n)
            f_try = obj_func(h_try, beta)
            direction = np.dot(grad, h_try - h)
            if f_try <= f_prev + 1e-4 * lr_try * direction:
                acceptable = True
                break
            lr_try *= 0.5
        if acceptable:
            h = h_try
            f_prev = f_try
            lr = min(lr * 1.05, 0.5)
        else:
            lr *= 0.5
            if lr < 1e-10:
                break
    return h, compute_c5(h)

def coordinate_descent(h0, n, beta, max_rounds=5):
    h = h0.copy()
    f_cur = obj_func(h, beta)
    for rd in range(max_rounds):
        improved = False
        for i in range(n):
            for delta in [-0.1, 0.1, -0.05, 0.05, -0.2, 0.2]:
                h_trial = h.copy()
                h_trial[i] = np.clip(h[i] + delta, 0.0, 1.0)
                h_trial = project_feasible(h_trial, n)
                f_trial = obj_func(h_trial, beta)
                if f_trial < f_cur - 1e-12:
                    h = h_trial
                    f_cur = f_trial
                    improved = True
                    break
        if not improved:
            break
    return h, compute_c5(h)

def binary_local_search(h0, n, rng, max_rounds=2000):
    h = (h0 > 0.5).astype(float)
    ones_count = int(np.sum(h))
    target = n // 2
    if ones_count > target:
        ones = np.where(h == 1.0)[0]
        order = np.argsort(h0[ones])[:ones_count - target]
        h[ones[order]] = 0.0
    elif ones_count < target:
        zeros = np.where(h == 0.0)[0]
        order = np.argsort(-h0[zeros])[:target - ones_count]
        h[zeros[order]] = 1.0
    best_c5 = compute_c5(h)
    for rd in range(max_rounds):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(corr))
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        for _ in range(30):
            i1 = ones[rng.randint(len(ones))]
            i0 = zeros[rng.randint(len(zeros))]
            h[i1] = 0.0
            h[i0] = 1.0
            new_c5 = compute_c5(h)
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                improved = True
            else:
                h[i1] = 1.0
                h[i0] = 0.0
        if not improved:
            break
    return h, best_c5

def subgradient_step(h, n, beta, lr):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_k = int(np.argmax(corr))
    L = max_k - (n - 1)
    grad = np.zeros(n)
    if L == 0:
        grad = (1.0 - 2.0 * h)
    elif L > 0:
        grad[L:] += (1.0 - h[:n - L])
        grad[:n - L] -= h[L:]
    else:
        Lp = -L
        grad[:n - Lp] += (1.0 - h[Lp:])
        grad[Lp:] -= h[:n - Lp]
    dx = 2.0 / n
    h_new = project_feasible(h - lr * dx * grad, n)
    return h_new

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - start_time

    def remaining():
        return max(0.0, budget_s - elapsed() - 10.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        if parent_n > 0:
            parent_c5 = compute_c5(parent_h)
            best_h = parent_h.copy()
            best_c5 = parent_c5
            best_n = parent_n
            print("Parent: n={}, C5={:.6f}".format(parent_n, parent_c5))
    except Exception:
        pass

    if best_h is None:
        best_n = 200
        best_h = make_beatty(best_n, (np.sqrt(5.0) - 1.0) / 2.0)
        best_c5 = compute_c5(best_h)
        print("Beatty init: n={}, C5={:.6f}".format(best_n, best_c5))

    # Phase 1: Optimize at parent resolution with SLSQP
    n_parent = best_n
    h_cur = best_h.copy()
    for beta_val in [100, 300, 800]:
        if remaining() < 60:
            break
        iters = 400 if beta_val <= 300 else 500
        h_cur, c5_cur = slsqp_optimize(h_cur, n_parent, beta_val, iters)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            best_n = n_parent
            print("  SLSQP n={}, beta={}: C5={:.6f}".format(n_parent, beta_val, best_c5))

    # Phase 2: Try Beatty and other structured initializations at parent resolution
    alphas = [(np.sqrt(5.0) - 1.0) / 2.0, 0.7, 0.6, 1.0 / np.sqrt(2.0)]
    for alpha in alphas:
        if remaining() < 40:
            break
        h_b = make_beatty(n_parent, alpha)
        h_opt, c5_opt = slsqp_optimize(h_b, n_parent, 300, 300)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_parent
            print("  Beatty alpha={:.4f}: C5={:.6f}".format(alpha, best_c5))

    # Phase 3: Higher resolution optimization
    for n_try in [300, 400, 500, 640]:
        if remaining() < 120:
            break
        print("=== n={} (remaining {:.0f}s) ===".format(n_try, remaining()))

        h0 = interp_h(best_h, n_try)
        h4 = interp_h(make_beatty(n_try, (np.sqrt(5.0) - 1.0) / 2.0), n_try)

        inits = [h0, h4]

        for h_init in inits:
            if remaining() < 60:
                break
            h_cur = h_init.copy()
            for beta_val in [80, 250, 700]:
                if remaining() < 40:
                    break
                iters = 300 if beta_val < 500 else 400
                h_cur, c5_cur = slsqp_optimize(h_cur, n_try, beta_val, iters)
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
                    best_n = n_try
                    print("  n={}, beta={}: C5={:.6f}".format(n_try, beta_val, best_c5))

            # Projected gradient polish
            if remaining() > 50:
                h_pg, c5_pg = projected_gradient_opt(h_cur, n_try, 500, 800, 0.05)
                if c5_pg < best_c5:
                    best_c5 = c5_pg
                    best_h = h_pg.copy()
                    best_n = n_try
                    print("  PG n={}: C5={:.6f}".format(n_try, best_c5))

    # Phase 4: Subgradient + projected gradient at high resolution
    if remaining() > 100:
        n_hi = 400
        h_hi = interp_h(best_h, n_hi)
        lr = 0.02
        for i in range(min(2000, int(remaining() / 0.05))):
            if remaining() < 30:
                break
            h_hi = subgradient_step(h_hi, n_hi, 500, lr)
            lr = max(0.001, lr * 0.999)
        h_hi, c5_hi = slsqp_optimize(h_hi, n_hi, 800, 300)
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi.copy()
            best_n = n_hi
            print("  Subgrad n={}: C5={:.6f}".format(n_hi, best_c5))

    # Phase 5: Coordinate descent polish
    if remaining() > 30:
        beta_cd = 400
        h_cd, c5_cd = coordinate_descent(best_h, best_n, beta_cd, max_rounds=3)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            print("  CoordDesc: C5={:.6f}".format(best_c5))

    # Phase 6: Binary local search
    if remaining() > 60:
        h_bin, c5_bin = binary_local_search(best_h, best_n, rng, max_rounds=int(min(5000, remaining() / 0.02)))
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            best_n = best_n
            print("  Binary: C5={:.6f}".format(best_c5))

    # Phase 7: If binary solution found, try to improve with continuous relaxation
    if remaining() > 40 and np.all((best_h == 0) | (best_h == 1)):
        h_cont = best_h + 0.1 * rng.uniform(-0.5, 0.5, best_n)
        h_cont = project_feasible(h_cont, best_n)
        h_opt, c5_opt = slsqp_optimize(h_cont, best_n, 500, 400)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            print("  Continuous relax: C5={:.6f}".format(best_c5))

    # Final: multi-restart perturbation
    if remaining() > 50:
        for trial in range(5):
            if remaining() < 25:
                break
            h_pert = best_h.copy()
            n_pert = max(3, best_n // 10)
            pert_idx = rng.choice(best_n, size=n_pert, replace=False)
            h_pert[pert_idx] = np.clip(h_pert[pert_idx] + rng.uniform(-0.4, 0.4, n_pert), 0, 1)
            h_pert = project_feasible(h_pert, best_n)
            h_opt, c5_opt = slsqp_optimize(h_pert, best_n, 500, 300)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                print("  Perturb {}: C5={:.6f}".format(trial, best_c5))

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
