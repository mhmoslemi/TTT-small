import time

def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float('inf')
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_density(x, target):
    x = np.asarray(x, dtype=float).copy()
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    target = float(target)
    if n == 0:
        return x
    s = float(np.sum(x))
    if abs(s - target) < 1e-12:
        return x
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-10:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size > 0:
            h[free] += res / free.size
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    if n0 == n:
        return p.copy()
    if n0 <= 0:
        return project_density(np.full(max(1, n), 0.5), 0.5 * n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return project_density(h, 0.5 * n)

def smooth_obj(x, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -745.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    return dx * (m + np.log(z) / alpha)

def smooth_grad(x, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -745.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    w = e / z
    center = n - 1
    g = np.zeros(n)
    for t in np.where(w > 1e-12)[0]:
        k = int(t) - center
        p = w[t]
        if k == 0:
            g += p * (1.0 - 2.0 * x)
        else:
            lag = abs(k)
            if k > 0:
                g[lag:] += p
            else:
                g[:n-lag] += p
            g[:n-lag] -= p * x[lag:]
            g[lag:] -= p * x[:n-lag]
    return dx * g

def sum_fun(x, target):
    return float(np.sum(x) - target)

def sum_jac(x):
    return np.ones_like(x)

def coord_descent_sweeps(h, n, deadline, max_sweeps=50, seed=42):
    target = 0.5 * n
    dx = 2.0 / n
    best_h = h.copy()
    best_c = evaluate_erdos(best_h)
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    rng = np.random.default_rng(seed)
    prev_c = best_c
    
    for sweep in range(max_sweeps):
        if time.perf_counter() > deadline:
            break
        order = rng.permutation(n)
        improved_this_sweep = 0
        
        for j in order:
            if time.perf_counter() > deadline:
                break
            j = int(j)
            old = best_h[j]
            if old < 1e-12 or old > 1.0 - 1e-12:
                pass  # Still try, boundary can improve
            # Compute g_j: gradient of each corr[k] w.r.t. h[j]
            # corr[lag] = sum_i h[i]*(1-h[i-lag]) for valid i
            # d/dh[j] corr[lag]:
            #   (1-h[j-lag]) if 0 <= j-lag < n  (j as first factor)
            #   -h[j+lag] if 0 <= j+lag < n      (j as second factor)
            
            # First factor: j-lag in [0, n-1] → lag in [j-n+1, j]
            l_lo1 = max(j - n + 1, -(n-1))
            l_hi1 = min(j, n-1)
            # Second factor: j+lag in [0, n-1] → lag in [-j, n-1-j]
            l_lo2 = max(-j, -(n-1))
            l_hi2 = min(n-1-j, n-1)
            
            # Build g_j vector (size 2n-1, index = lag + n - 1)
            # Only nonzero where l in [l_lo1, l_hi1] or [l_lo2, l_hi2]
            g_j = np.zeros(2*n-1)
            for l in range(l_lo1, l_hi1 + 1):
                g_j[l + n - 1] += (1.0 - best_h[j - l])
            for l in range(l_lo2, l_hi2 + 1):
                g_j[l + n - 1] -= best_h[j + l]
            
            # 1D problem: min_{t in [0,1]} max_k (corr[k] + (t - old) * g_j[k])
            a = g_j
            b = corr - old * g_j
            
            # Evaluate at boundaries and current
            f0 = float(np.max(b))  # t=0
            f1 = float(np.max(a + b))  # t=1
            fc = float(np.max(corr))  # t=old (current)
            
            best_t = old
            best_f = fc
            if f0 < best_f:
                best_f = f0
                best_t = 0.0
            if f1 < best_f:
                best_f = f1
                best_t = 1.0
            
            # Ternary search in the interval that might contain the min
            # Since f is convex, min is where subgradient changes sign
            lo = 0.0
            hi = 1.0
            for _ in range(35):
                m1 = lo + (hi - lo) * (3.0 - math.sqrt(3.0)) / 6.0
                m2 = hi - (hi - lo) * (3.0 - math.sqrt(3.0)) / 6.0
                fm1 = float(np.max(a * m1 + b))
                fm2 = float(np.max(a * m2 + b))
                if fm1 <= fm2:
                    hi = m2
                else:
                    lo = m1
            t_mid = 0.5 * (lo + hi)
            f_mid = float(np.max(a * t_mid + b))
            if f_mid < best_f:
                best_f = f_mid
                best_t = t_mid
            
            if abs(best_t - old) > 1e-9:
                delta = best_t - old
                new_corr = corr + delta * g_j
                new_mc = float(np.max(new_corr)) * dx
                if new_mc < best_c - 1e-13:
                    best_h[j] = best_t
                    corr = new_corr
                    best_c = new_mc
                    improved_this_sweep += 1
                # If not better, don't change (keep old)
        
        # Project after sweep
        best_h = project_density(best_h, target)
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        c_now = evaluate_erdos(best_h)
        if c_now < best_c:
            best_c = c_now
        
        if improved_this_sweep == 0 and sweep > 5:
            # Try a perturbation to escape
            perturb = 0.005 * rng.standard_normal(n)
            perturbed = project_density(best_h + perturb, target)
            c_pert = evaluate_erdos(perturbed)
            if c_pert < best_c + 0.002:
                best_h = perturbed
                corr = np.correlate(best_h, 1.0 - best_h, mode='full')
                # Reset stall counter by allowing more sweeps
            elif sweep > 15:
                break
    
    best_h = project_density(best_h, 0.5 * n)
    best_c = evaluate_erdos(best_h)
    return best_h, best_c

def active_gradient_polish(h, n, deadline, max_iter=500, top_frac=0.05, min_k=8):
    target = 0.5 * n
    dx = 2.0 / n
    best_h = h.copy()
    best_c = evaluate_erdos(best_h)
    stall = 0
    for _ in range(max_iter):
        if time.perf_counter() > deadline:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        center = n - 1
        k_count = max(min_k, int(np.ceil(len(corr) * top_frac)))
        top_idx = np.argsort(corr)[-k_count:]
        top_vals = corr[top_idx]
        dv = top_vals - top_vals.min()
        w = np.exp(dv / max(1e-10, dv.max() + 1e-10))
        w /= w.sum()
        g = np.zeros(n)
        for i, t in enumerate(top_idx):
            k = int(t) - center
            p = w[i]
            if k == 0:
                g += p * (1.0 - 2.0 * best_h)
            else:
                lag = abs(k)
                if k > 0:
                    g[lag:] += p
                else:
                    g[:n-lag] += p
                g[:n-lag] -= p * best_h[lag:]
                g[lag:] -= p * best_h[:n-lag]
        d = -g
        d -= np.mean(d)
        norm = np.linalg.norm(d)
        if norm < 1e-14:
            break
        d /= norm
        improved = False
        for step in (0.002, 0.005, 0.01, 0.03, 0.08, 0.15, 0.3, 0.5):
            cand = project_density(best_h + step * d, target)
            c = evaluate_erdos(cand)
            if c < best_c - 1e-12:
                best_h = cand
                best_c = c
                improved = True
                break
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 30:
                break
    best_h = project_density(best_h, target)
    return best_h, evaluate_erdos(best_h)

def slsqp_run(h, n, deadline, alphas=(20, 60, 200), maxiter=150):
    target = 0.5 * n
    best_h = h.copy()
    best_c = evaluate_erdos(best_h)
    bounds = [(0.0, 1.0)] * n
    constr = ({'type': 'eq', 'fun': lambda x: sum_fun(x, target),
               'jac': lambda x: sum_jac(x)},)
    
    for alpha in alphas:
        if time.perf_counter() >= deadline - 3.0:
            break
        try:
            res = minimize(
                lambda x: smooth_obj(x, alpha, n),
                best_h.copy(),
                method='SLSQP',
                jac=lambda x: smooth_grad(x, alpha, n),
                bounds=bounds,
                constraints=constr,
                options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                cand = project_density(res.x, target)
                c = evaluate_erdos(cand)
                if c < best_c:
                    best_h = cand
                    best_c = c
                else:
                    # Don't degrade
                    pass
        except Exception:
            pass
    best_h = project_density(best_h, target)
    return best_h, evaluate_erdos(best_h)

            best_h, best_c, best_n = h1.copy(), c1, N1
        print(f"After pair swap n=900: c={c1:.6f}")
    
    # Phase 7: Try n=1800
    rem = deadline - time.perf_counter()
    if rem > 180:
        N2 = 1800
        h2 = upsample_h(h1, N2)
        h2, c2 = slsqp_run(h2, N2, time.perf_counter() + min(60, rem*0.2), alphas=(20, 60), maxiter=60)
        rem2 = deadline - time.perf_counter()
        if rem2 > 30:
            h2, c2 = coord_descent_sweeps(h2, N2, time.perf_counter() + min(100, rem2*0.4), max_sweeps=40, seed=seed+3)
        rem2 = deadline - time.perf_counter()
        if rem2 > 20:
            h2, c2 = active_gradient_polish(h2, N2, time.perf_counter() + min(30, rem2*0.3), max_iter=300, top_frac=0.04, min_k=12)
        if c2 < best_c:
            best_h, best_c, best_n = h2.copy(), c2, N2
        print(f"After n=1800: c={c2:.6f}")
    
    # Phase 8: Final intensive coordinate descent on best
    rem = deadline - time.perf_counter()
    if rem > 15 and best_h is not None:
        nbest = int(best_h.size)
        hb = project_density(best_h, 0.5 * nbest)
        hb, cb = coord_descent_sweeps(hb, nbest, time.perf_counter() + min(120, rem*0.5), max_sweeps=100, seed=seed+10)
        if cb < best_c:
            best_h, best_c, best_n = hb.copy(), cb, nbest
        print(f"After final coord descent n={nbest}: c={cb:.6f}")
    
    # Phase 9: Final active gradient + pair swap
    rem = deadline - time.perf_counter()
    if rem > 10 and best_h is not None:
        nbest = int(best_h.size)
        hb, cb = active_gradient_polish(best_h, nbest, time.perf_counter() + min(20, rem*0.3), max_iter=300, top_frac=0.03, min_k=15)
        if cb < best_c:
            best_h, best_c, best_n = hb.copy(), cb, nbest
        rem = deadline - time.perf_counter()
        if rem > 5:
            hb, cb = pair_swap(best_h, nbest, time.perf_counter() + min(20, rem*0.5), max_iter=800, n_pos=6, n_neg=6)
            if cb < best_c:
                best_h, best_c, best_n = hb.copy(), cb, nbest
    
    # Phase 10: One more coordinate descent round if time
    rem = deadline - time.perf_counter()
    if rem > 10 and best_h is not None:
        nbest = int(best_h.size)
        hb, cb = coord_descent_sweeps(best_h, nbest, time.perf_counter() + min(60, rem*0.7), max_sweeps=50, seed=seed+20)
        if cb < best_c:
            best_h, best_c, best_n = hb.copy(), cb, nbest
    
    if best_h is None:
        fallback = project_density(np.full(90, 0.5), 45.0)
        return fallback, evaluate_erdos(fallback), 90
    
    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    print(f"FINAL: n={final_h.size}, C5={final_c:.6f}")
    return final_h, float(final_c), int(final_h.size)
