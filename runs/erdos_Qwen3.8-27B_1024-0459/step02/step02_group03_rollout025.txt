    # h is binary, n is length
    dx = 2.0 / n
    curr = h.copy()
    curr_c5 = compute_c5(curr)
    
    if curr_c5 > best_c5:
        return curr, best_c5
    
    best_local = curr.copy()
    best_local_c5 = curr_c5
    
    ones = np.where(curr == 1)[0]
    zeros = np.where(curr == 0)[0]
    
    t_start = time.time()
    
    while time.time() - t_start < time_limit:
        corr = np.correlate(curr, 1.0 - curr, mode='full')
        max_k = np.argmax(corr)
        max_overlap = corr[max_k]
        
        improved = False
        n_ones = len(ones)
        n_zeros = len(zeros)
        
        # Try random swaps
        for attempt in range(100):
            if time.time() - t_start > time_limit:
                break
            i1 = ones[rng.randint(n_ones)]
            i0 = zeros[rng.randint(n_zeros)]
            
            # Swap
            curr[i1] = 0.0
            curr[i0] = 1.0
            
            new_corr = np.correlate(curr, 1.0 - curr, mode='full')
            new_c5 = np.max(new_corr) * dx
            
            if new_c5 < curr_c5:
                # Accept
                curr_c5 = new_c5
                ones = np.where(curr == 1)[0]
                zeros = np.where(curr == 0)[0]
                improved = True
                if curr_c5 < best_local_c5:
                    best_local_c5 = curr_c5
                    best_local = curr.copy()
                if curr_c5 < best_c5:
                    best_c5 = curr_c5
            else:
                # Revert
                curr[i1] = 1.0
                curr[i0] = 0.0
        
        if not improved:
            # Try a more aggressive search or break
            break
    
    return best_local, best_local_c5

                best_h = h_res2.copy()
                best_n = n_try
                print("  Higher res n={}: C5={:.8f}".format(n_try, c5_res2))

    # Phase 4: Final continuous polish at best_n
    if time_left() > 40:
        h_init_final = interp_to_n(best_h, best_n)
        for beta_val in [500, 1000]:
            if time_left() < 25:
                break
            try:
                h_fin, c5_fin = continuous_opt(h_init_final, best_n, beta_val, 300)
                if c5_fin < best_c5:
                    best_c5 = c5_fin
                    best_h = h_fin.copy()
                    print("  Final polish beta={}: C5={:.8f}".format(beta_val, c5_fin))
                h_init_final = h_fin
            except Exception:
                pass

    # Final evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    if final_c5 > best_c5:
        final_c5 = best_c5

    print("FINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
