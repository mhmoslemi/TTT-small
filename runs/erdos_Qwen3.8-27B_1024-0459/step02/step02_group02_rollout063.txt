For a swap of positions i (1→0) and j (0→1):
The correlation at lag s changes only for lags involving i or j.

  - At lag (i-s to i): term was h[i-s]*(1-h[i]) = h[i-s]*0 = 0, now h[i-s]*1. Increase by h[i-s]

- If s = j - i (and i+s = j): complex interaction

def gen_thue_morse(n):

def gen_random_bin(n, rng):
    h = (rng.rand(n) > 0.5).astype(float)
    target = n // 2
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    if len(ones) > target:
        flip = rng.choice(ones, len(ones) - target, replace=False)
        h[flip] = 0.0
    elif len(ones) < target:

def gen_antisym(n, rng):
    half = n // 2

    h = np.zeros(n)
    h[:half] = left
    h[half:] = 1.0 - left

    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)

    target = n_dst // 2

    h_curr = h.copy()

            break
        elapsed = time.time() - start_t

        temp = t_start * (t_end / t_start) ** progress
        temp_eff = temp * dx

        i_idx = rng.randint(len(ones_list))
        j_idx = rng.randint(len(zeros_list))
        i = ones_list[i_idx]
        j = zeros_list[j_idx]

        h_curr[j] = 1.0

def _smooth_obj(x, beta, n):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    shifted = np.clip(shifted, -50, 50)
    val = cmax + (1.0 / beta) * np.log(np.sum(np.exp(shifted)))
    return val * (2.0 / n)

def _sum_eq(x, n):
    return np.sum(x) - n / 2.0

def _sum_jac(x, n):
    return np.ones_like(x)

def continuous_polish(h_init, n, budget_s=30):

    bounds = [(0.0, 1.0)] * n

    start_t = time.time()
    betas = [30.0, 100.0, 300.0]

            if res.fun < 1e10:

            pass

                if c5_512s < c5_512:
                    h512 = h512s
                    c5_512 = c5_512s

            if c5_512 < best_c5:
                best_c5 = c5_512
                best_h = h512.copy()

    # Continuous polish on best binary solution
    if best_h is not None and remaining() > 25:
        t_pol = min(40.0, remaining() - 5)
        if t_pol > 10:
            h_pol, c5_pol = continuous_polish(best_h, len(best_h), budget_s=t_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol

    # Final directed search on polished solution
    if best_h is not None and remaining() > 20:
        n_best = len(best_h)
        t_final = min(20.0, remaining() - 5)
        if t_final > 5:
            h_fin = np.round(best_h).astype(float)
            h_fin = make_feasible(h_fin, n_best)
            h_fin, c5_fin = directed_swap_search(h_fin, n_best, rng, timeout=t_final, stall_limit=3000)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin
            else:
                # Keep continuous version if it was better
                pass

    # Ensure feasibility
    if best_h is None:
        best_h = gen_thue_morse(256)
        best_c5 = compute_c5(best_h)

    n_best = len(best_h)
    best_h = make_feasible(best_h, n_best)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_best)
