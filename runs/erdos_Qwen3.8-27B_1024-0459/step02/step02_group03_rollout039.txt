Let me also think about a completely different approach: **using the structure of the correlation**.

For a binary sequence h of length n with n/2 ones, the correlation at lag L (positive) is:
c[L] = Σ_{i=0}^{n-1-L} h[i+L] · (1-h[i])

This is the number of "0→1" transitions at distance L (where position i has 0 and position i+L has 1).

Similarly, c[-L] = number of "1→0" transitions at distance L.

The total number of transitions at distance L is c[L] + c[-L].

We want to minimize max(L) c[L] and c[-L].

A key observation: for a random binary sequence, c[L] ~ Binomial(n-L, p(1-p)) where p = 1/2, so c[L] ~ (n-L)/4 on average with std ~ sqrt((n-L)/16). The maximum over all L would be about (n/4) + a few standard deviations.

For n=400: average c[L] ≈ 100, std ≈ 5. Max would be around 100 + 4*5 = 120. C5 = 120 * 2/400 = 0.6. That's much worse than 0.38.

So the optimal sequence is very far from random. It must be highly structured to bring the max down from ~0.6 to ~0.38.

This suggests that the optimization landscape is quite "peaked" - good solutions are rare and require careful construction. Simulated annealing might struggle unless it runs for a very long time.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmin(np.maximum(h, 1.0 - h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to_binary(h_src, n_dst, rng):
    n_src = len(h_src)
    if n_src == n_dst:
        # If already binary, just return
        if np.all((h_src == 0) | (h_src == 1)):
            return h_src.copy()
        # Threshold with random tie-breaking
        h_bin = (h_src > 0.5).astype(float)
        n_ones = int(np.sum(h_bin))
        n_want = n_dst // 2
        if n_ones > n_want:
            ones = np.where(h_bin == 1.0)[0]
            vals = h_src[ones]
            order = np.argsort(vals)[:n_ones - n_want]
            h_bin[ones[order]] = 0.0
        elif n_ones < n_want:
            zeros = np.where(h_bin == 0.0)[0]
            vals = h_src[zeros]
            order = np.argsort(-vals)[:n_want - n_ones]
            h_bin[zeros[order]] = 1.0
        return h_bin
    
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    # Threshold
    h_bin = (h_dst > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    n_want = n_dst // 2
    if n_ones > n_want:
        ones = np.where(h_bin == 1.0)[0]
        vals = h_dst[ones]
        order = np.argsort(vals)[:n_ones - n_want]
        h_bin[ones[order]] = 0.0
    elif n_ones < n_want:
        zeros = np.where(h_bin == 0.0)[0]
        vals = h_dst[zeros]
        order = np.argsort(-vals)[:n_want - n_ones]
        h_bin[zeros[order]] = 1.0
    return h_bin

def sa_optimize(h_init, rng, time_limit, informed_values=None):
    """Simulated annealing on binary sequence.
    
    Args:
        h_init: initial binary array
        rng: numpy RandomState
        time_limit: seconds to run
        informed_values: optional continuous values to guide moves (higher = more likely to be 1)
    """
    n = len(h_init)
    h_cur = h_init.copy()
    c5_cur = compute_c5(h_cur)
    h_best = h_cur.copy()
    c5_best = c5_cur
    
    # Temperature
    t = 0.02
    t_min = 1e-6
    t_max = 0.05
    cooling_rate = 0.9995  # per iteration
    
    start = time.time()
    no_improve_count = 0
    improve_threshold = 2000  # iterations without improvement before reheat
    
    # Precompute indices
    ones_idx = np.where(h_cur == 1.0)[0].copy()
    zeros_idx = np.where(h_cur == 0.0)[0].copy()
    
    while time.time() - start < time_limit:
        # Check for reheat
        if no_improve_count > improve_threshold:
            t = min(t * 3.0, t_max)
            no_improve_count = 0
            improve_threshold = int(improve_threshold * 1.5)
        
        t = max(t * cooling_rate, t_min)
        
        # Choose move
        r = rng.random()
        
        if r < 0.55:
            # Single pair swap with optional informed bias
            if informed_values is not None:
                # Bias toward flipping 1s with low informed values to 0
                # and 0s with high informed values to 1
                one_weights = np.maximum(1.0 - informed_values[ones_idx], 0.01)
                zero_weights = np.maximum(informed_values[zeros_idx], 0.01)
                # Normalize
                one_weights /= one_weights.sum()
                zero_weights /= zero_weights.sum()
                i1 = rng.choice(ones_idx, p=one_weights)
                i0 = rng.choice(zeros_idx, p=zero_weights)
            else:
                i1 = ones_idx[rng.randint(len(ones_idx))]
                i0 = zeros_idx[rng.randint(len(zeros_idx))]
            
            h_cur[i1] = 0.0
            h_cur[i0] = 1.0
            new_c5 = compute_c5(h_cur)
            
            if new_c5 < c5_cur or (t > t_min and rng.random() < np.exp(-(new_c5 - c5_cur) / t)):
                # Update indices
                pos1 = np.where(ones_idx == i1)[0][0]
                pos0 = np.where(zeros_idx == i0)[0][0]
                ones_idx[pos1] = i0
                zeros_idx[pos0] = i1
                c5_cur = new_c5
                if new_c5 < c5_best:
                    c5_best = new_c5
                    h_best = h_cur.copy()
                    no_improve_count = 0
                else:
                    no_improve_count += 1
            else:
                h_cur[i1] = 1.0
                h_cur[i0] = 0.0
                no_improve_count += 1
        
        elif r < 0.85:
            # Block swap
            blen = rng.randint(2, 6)
            for _ in range(20):
                i = rng.randint(0, n - blen)
                j = rng.randint(0, n - blen)
                if abs(i - j) < blen:
                    continue
                bi = h_cur[i:i+blen]
                bj = h_cur[j:j+blen]
                if np.all(bi == 1.0) and np.all(bj == 0.0):
                    h_cur[i:i+blen] = 0.0
                    h_cur[j:j+blen] = 1.0
                    new_c5 = compute_c5(h_cur)
                    if new_c5 < c5_cur or (t > t_min and rng.random() < np.exp(-(new_c5 - c5_cur) / t)):
                        # Update indices
                        ones_idx = np.where(h_cur == 1.0)[0]
                        zeros_idx = np.where(h_cur == 0.0)[0]
                        c5_cur = new_c5
                        if new_c5 < c5_best:
                            c5_best = new_c5
                            h_best = h_cur.copy()
                            no_improve_count = 0
                        else:
                            no_improve_count += 1
                    else:
                        h_cur[i:i+blen] = 1.0
                        h_cur[j:j+blen] = 0.0
                        no_improve_count += 1
                    break
                elif np.all(bi == 0.0) and np.all(bj == 1.0):
                    h_cur[i:i+blen] = 1.0
                    h_cur[j:j+blen] = 0.0
                    new_c5 = compute_c5(h_cur)
                    if new_c5 < c5_cur or (t > t_min and rng.random() < np.exp(-(new_c5 - c5_cur) / t)):
                        ones_idx = np.where(h_cur == 1.0)[0]
                        zeros_idx = np.where(h_cur == 0.0)[0]
                        c5_cur = new_c5
                        if new_c5 < c5_best:
                            c5_best = new_c5
                            h_best = h_cur.copy()
                            no_improve_count = 0
                        else:
                            no_improve_count += 1
                    else:
                        h_cur[i:i+blen] = 0.0
                        h_cur[j:j+blen] = 1.0
                        no_improve_count += 1
                    break
        
        else:
            # Multi-flip
            k = rng.randint(3, 9)
            if len(ones_idx) >= k and len(zeros_idx) >= k:
                fi = rng.choice(len(ones_idx), size=k, replace=False)
                fj = rng.choice(len(zeros_idx), size=k, replace=False)
```

```python
def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)

        return h
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)

            h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)

    h_cur = h_init.copy()
    c5_cur = compute_c5(h_cur)
    h_best = h_cur.copy()
    c5_best = c5_cur

    start = time.time()
    stuck = 0

    while time.time() - start < time_limit:
        if stuck > 20000:
            break

        ones = np.where(h_cur == 1.0)[0]
        zeros = np.where(h_cur == 0.0)[0]

        best_delta = 0.0
        best_i1 = -1
        best_i0 = -1
        n_trials = 30

        for _ in range(n_trials):
            i1 = ones[rng.randint(len(ones))]
            i0 = zeros[rng.randint(len(zeros))]
            h_cur[i1] = 0.0
            h_cur[i0] = 1.0
            new_c5 = compute_c5(h_cur)
            delta = new_c5 - c5_cur
            if delta < best_delta:
                best_delta = delta
                best_i1 = i1
                best_i0 = i0
            h_cur[i1] = 1.0
            h_cur[i0] = 0.0

        if best_i1 >= 0:
            h_cur[best_i1] = 0.0
            h_cur[best_i0] = 1.0
            c5_cur = c5_cur + best_delta
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h_cur.copy()
            stuck = 0
        else:
            stuck += 1

    return h_best, c5_best

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - start_time

    def remaining():
        return max(0.0, budget_s - elapsed() - 5.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0
    best_continuous = None

    # Get parent
    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        if parent_n > 0:
            parent_c5 = compute_c5(parent_h)
            best_h = parent_h.copy()
            best_c5 = parent_c5
            best_n = parent_n
            best_continuous = parent_h.copy()
            print("Parent: n={}, C5={:.6f}".format(parent_n, parent_c5))
    except Exception:
        pass

    if best_h is None:
        best_n = 200
        best_h = random_binary(best_n, rng)
        best_c5 = compute_c5(best_h)
        print("Random init: n={}, C5={:.6f}".format(best_n, best_c5))

    # Ensure we have a binary version for SA
    def get_binary(h, n):
        if np.all((h == 0) | (h == 1)):
            return h.copy()
        h_bin = (h > 0.5).astype(float)
        n_ones = int(np.sum(h_bin))
        n_want = n // 2
        if n_ones > n_want:
            ones = np.where(h_bin == 1.0)[0]
            order = np.argsort(h[ones])[:n_ones - n_want]
            h_bin[ones[order]] = 0.0
        elif n_ones < n_want:
            zeros = np.where(h_bin == 0.0)[0]
            order = np.argsort(-h[zeros])[:n_want - n_ones]
            h_bin[zeros[order]] = 1.0
        return h_bin

    best_binary = get_binary(best_h, best_n)

    # Phase 1: SA at parent resolution with informed values
    if remaining() > 120:
        t1 = min(100, remaining() - 20)
        print("Phase 1: SA at n={}, t={}s (informed)".format(best_n, t1))
        if best_continuous is not None and len(best_continuous) == best_n:
            h1, c1 = sa_optimize(best_binary, rng, t1, informed_values=best_continuous)
        else:
            h1, c1 = sa_optimize(best_binary, rng, t1)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1
            best_binary = h1
            print("  Phase 1: C5={:.6f}".format(c1))

    # Phase 2: Try higher resolutions
    n_targets = [300, 400, 500, 600]
    # Filter: skip resolutions already tried or below current
    n_targets = [n for n in n_targets if n != best_n and n % 2 == 0]
    # Sort by proximity to current best (interpolation quality)
    n_targets.sort(key=lambda x: abs(x - best_n))

    for n in n_targets:
        if remaining() < 180:
            break

        alloc = min(300, remaining() - 40)
        print("Phase 2: n={}, alloc={}s".format(n, alloc))

        # Interpolate to get initial binary
        h_interp = interp_to_binary(best_h, n, rng)

        # Run SA (main)
        t_sa = alloc * 0.6
        h2, c2 = sa_optimize(h_interp, rng, t_sa)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2
            best_n = n
            best_binary = h2
            print("  SA n={}: C5={:.6f}".format(n, c2))

        # Targeted SA
        if remaining() > 80:
            t_ta = min(alloc * 0.25, remaining() - 40)
            h3, c3 = targeted_sa(h2 if c2 < best_c5 else h_interp, rng, t_ta)
            if c3 < best_c5:
                best_c5 = c3
                best_h = h3
                best_n = n
                best_binary = h3
                print("  TargetedSA n={}: C5={:.6f}".format(n, c3))

        # Polish
        if remaining() > 50:
            t_pol = min(alloc * 0.15, remaining() - 20)
            cur_h = h3 if c3 < best_c5 else (h2 if c2 < best_c5 else h_interp)
            h4, c4 = polish_local(cur_h, rng, t_pol)
            if c4 < best_c5:
                best_c5 = c4
                best_h = h4
                best_n = n
                best_binary = h4
                print("  Polish n={}: C5={:.6f}".format(n, c4))

    # Phase 3: Extended optimization on best resolution
    if remaining() > 80:
        t3 = remaining() - 30
        print("Phase 3: Extended SA at n={}, t={}s".format(best_n, t3))
        h5, c5ext = sa_optimize(best_binary, rng, t3 * 0.6)
        if c5ext < best_c5:
            best_c5 = c5ext
            best_h = h5
            best_binary = h5
            print("  Ext SA: C5={:.6f}".format(c5ext))

    # Phase 4: Final polish
    if remaining() > 30:
        h6, c6 = polish_local(best_h, rng, remaining() - 10)
        if c6 < best_c5:
            best_c5 = c6
            best_h = h6
            print("  Final polish: C5={:.6f}".format(c6))

    # Final evaluation
    final_c5 = compute_c5(best_h)
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.6f} (budget used {:.0f}s)".format(best_n, final_c5, elapsed()))
    return (best_h, final_c5, best_n)
```
