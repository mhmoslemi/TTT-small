def grad_topk_fast(h, k):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-k:]
    w = 1.0 / k
    grad = np.zeros(n)
    for l_idx in idx:
        lag = l_idx - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * (1.0 - h[:n-lag])
                grad[:n-lag] += w * (-h[lag:])
        else:
            o = -lag
            if o < n:
                grad[o:] += w * (-h[:n-o])
                grad[:n-o] += w * (1.0 - h[o:])
    return grad

def project_feasible(h, n):

    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-14:

            mask = h < 1.0 - 1e-12

                h = np.clip(h, 0, 1)
            else:

            if cnt == 0:
                h += diff / n
                h = np.clip(h, 0, 1)
            else:

        h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
import time
from scipy.optimize import minimize, minimize_scalar

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(np.array(h, dtype=float).copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-14
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argsort(1.0 - h)[::-1][:max(1, int(np.ceil(abs(diff))))]
            h[idx] += diff / len(idx)
        else:
            idx = np.argsort(h)[::-1][:max(1, int(np.ceil(abs(diff))))]
            h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h, n, sweeps=15, time_limit=None, t0=None):
    h = project_feasible(h, n)
    best = h.copy()
    best_c5 = compute_c5(h)
    for sw in range(sweeps):
        if time_limit and t0 and time.time() - t0 > time_limit:
            break
        for j in range(n):
            if time_limit and t0 and time.time() - t0 > time_limit:
                break
            old = h[j]
            # Determine affected lags and precompute coefficients
            # c[lag] for lag in [-(n-1), n-1], full index = lag + (n-1)
            a_vals = {}
            b_vals = {}
            rest_vals = {}
            # Current correlation
            corr = np.correlate(h, 1.0 - h, mode='full')
            max_unaffected = 0.0
            for lag in range(-(n - 1), n):
                fidx = lag + (n - 1)
                # Does h[j] appear in c[lag]?
                affects = False
                coeff = 0.0
                const = 0.0
                if lag >= 0:
                    # h[j] as h[i+lag]: i = j-lag, need 0<=j-lag<=n-1-lag
                    if lag <= j and j <= n - 1 - lag + lag:  # j-lag >= 0 always if j>=lag
                        if j >= lag:
                            coeff += (1.0 - h[j - lag])
                            affects = True
                    # h[j] in (1-h[i]): i=j, need 0<=j<=n-1-lag
                    if j <= n - 1 - lag:
                        coeff += (-h[j + lag])
                        const += h[j + lag]
                        affects = True
                else:
                    s = -lag
                    # h[j] as h[i-s]: i=j+s, need 0<=j+s<=n-1, s<=j+s<=n-1 → 0<=j<=n-1-s
                    if 0 <= j <= n - 1 - s:
                        coeff += (1.0 - h[j + s])
                        affects = True
                    # h[j] in (1-h[i]): i=j, need s<=j<=n-1
                    if s <= j <= n - 1:
                        coeff += (-h[j - s])
                        const += h[j - s]
                        affects = True
                if affects:
                    # Special case: lag=0 has quadratic term h[j]*(1-h[j]) = x - x^2
                    if lag == 0:
                        # c[0] includes h[j] - h[j]^2
                        # Linear part: coeff already has (1-h[j]) - h[j] = 1-2h[j]
                        # Quadratic: -x^2 + x already accounted
                        # Actually for lag=0: term is h[j]*(1-h[j]) = x(1-x)
                        # So coeff of x is (1-2x) but since we're evaluating at x, it's 1-x for linear, -x for quad
                        # Let me handle: c[0] contribution from j: x*(1-x) = x - x^2
                        # So a=1, b=0, and we add -x^2 separately
                        a_vals[fidx] = 1.0
                        b_vals[fidx] = 0.0
                        # rest = corr[fidx] - old*(1-old)
                        rest_vals[fidx] = corr[fidx] - old * (1.0 - old)
                    else:
                        a_vals[fidx] = coeff
                        b_vals[fidx] = const
                        rest_vals[fidx] = corr[fidx] - coeff * old - const
                else:
                    if corr[fidx] > max_unaffected:
                        max_unaffected = corr[fidx]
            
            # 1D search: minimize max(max_unaffected, max_{affected}(rest + a*x + b), if lag0: rest0 + x - x^2)
            # Use grid + refine
            xs = np.linspace(0.0, 1.0, 100)
            best_x = old
            best_val = float('inf')
            for x in xs:
                val = max_unaffected
                for fidx in a_vals:
                    if fidx == n - 1:  # lag 0
                        v = rest_vals[fidx] + x - x * x
                    else:
                        v = rest_vals[fidx] + a_vals[fidx] * x + b_vals[fidx]
                    if v > val:
                        val = v
                if val < best_val:
                    best_val = val
                    best_x = x
            # Refine around best_x
            lo = max(0.0, best_x - 0.01)
            hi = min(1.0, best_x + 0.01)
            for _ in range(40):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                v1 = max_unaffected
                for fidx in a_vals:
                    if fidx == n - 1:
                        v = rest_vals[fidx] + m1 - m1 * m1
                    else:
                        v = rest_vals[fidx] + a_vals[fidx] * m1 + b_vals[fidx]
                    if v > v1:
                        v1 = v
                v2 = max_unaffected
                for fidx in a_vals:
                    if fidx == n - 1:
                        v = rest_vals[fidx] + m2 - m2 * m2
                    else:
                        v = rest_vals[fidx] + a_vals[fidx] * m2 + b_vals[fidx]
                    if v > v2:
                        v2 = v
                if v1 < v2:
                    hi = m2
                else:
                    lo = m1
            best_x = (lo + hi) / 2.0
            h[j] = best_x
        # Project after sweep
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
    return best

def smooth_max_obj(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -500.0, 0.0)
    return (m + np.log(np.sum(np.exp(sh))) / beta) * (2.0 / n)

def topk_obj(x, n, k):
    corr = np.correlate(x, 1.0 - x, mode='full')
    k = min(k, len(corr))
    top = np.sort(corr)[-k:]
    return np.mean(top) * (2.0 / n)

def eq_con(x, n):
    return np.sum(x) - n / 2.0

def grad_descent(h, n, iters=1000, time_limit=None, t0=None):
    h = project_feasible(h, n)
    best = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    # Start with small k (target max), then increase
    phases = [(3, 0.02, iters // 3), (10, 0.015, iters // 3), (3, 0.005, iters // 3)]
    for k, lr0, n_it in phases:
        lr = lr0
        stagn = 0
        for it in range(n_it):
            if time_limit and t0 and time.time() - t0 > time_limit:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            # Top-k gradient
            idx = np.argsort(corr)[-k:]
            grad = np.zeros(n)
            for lidx in idx:
                lag = lidx - (n - 1)
                w = 1.0 / k
                if lag >= 0:
                    if lag < n:
                        grad[lag:] += w * (1.0 - h[:n - lag])
                        grad[:n - lag] += w * (-h[lag:])
                else:
                    o = -lag
                    if o < n:
                        grad[o:] += w * (-h[:n - o])
                        grad[:n - o] += w * (1.0 - h[o:])
            h_new = h - lr * grad
            h_new = project_feasible(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best = h_new.copy()
                h = h_new
                stagn = 0
            else:
                lr *= 0.85
                stagn += 1
                if stagn > 30:
                    # Perturbation
                    h = project_feasible(h + 0.05 * np.random.randn(n), n)
                    lr = lr0
                    stagn = 0
    return best

def slsqp_optimize(h, n, time_limit=None, t0=None):
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: eq_con(x, n)}]
    best = h.copy()
    best_c5 = compute_c5(h)
    for beta in [50.0, 200.0, 1000.0, 5000.0, 20000.0]:
        if time_limit and t0 and time.time() - t0 > time_limit - 5:
            break
        def obj(x):
            return smooth_max_obj(x, n, beta)
        try:
            res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 300, 'ftol': 1e-14, 'iprint': 0})
            h = project_feasible(res.x, n)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best = h.copy()
        except Exception:
            pass
    # Top-k polish
    if time_limit and t0:
        if time.time() - t0 > time_limit - 10:
            return best
    for k in [3, 5, 10]:
        if time_limit and t0 and time.time() - t0 > time_limit - 5:
            break
        def obj_topk(x, kk=k):
            return topk_obj(x, n, kk)
        try:
            res = minimize(obj_topk, h, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-14, 'iprint': 0})
            h = project_feasible(res.x, n)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best = h.copy()
        except Exception:
            pass
    return best

def interpolate_h(h_src, n_src, n_dst):
    xp = (np.arange(n_src) + 0.5) * 2.0 / n_src
    xn = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    h_new = np.interp(xn, xp, h_src)
    return project_feasible(h_new, n_dst)
