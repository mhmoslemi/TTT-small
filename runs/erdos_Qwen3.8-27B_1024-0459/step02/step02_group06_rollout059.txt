def project_density(x, target):

def corr_grad_lag(h, k):
    n = h.size
    g = np.zeros(n, dtype=float)

    else:
        lag = abs(k)

            g[lag:] += 1.0
        else:
            g[:n - lag] += 1.0
        g[:n - lag] -= h[lag:]
        g[lag:] -= h[:n - lag]
    return g

        if h2[i] < thresh:
            h2[i] = 0.0
            changed = True
        elif h2[i] > 1.0 - thresh:
            h2[i] = 1.0
            changed = True
    if not changed:
        return h
    h2 = project_density(h2, target)
    return h2


def binary_local_search(h, deadline, max_iter=10000):
    """For binary or near-binary solutions, search the binary space via swaps"""
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    # Push to binary
    binary = (best > 0.5).astype(float)
    if abs(np.sum(binary) - target) > 0.5:
        # Need to fix count
        s = np.sum(binary)
        if s > target:
            excess = int(s - target)
            zero_pos = np.where(binary == 0)[0]
            if zero_pos.size >= excess:
                to_flip = zero_pos[np.argsort(np.random.default_rng(1).random(zero_pos.size))[:excess]]
                binary[to_flip] = 0.0
            # Actually we need to flip 1->0
            one_pos = np.where(binary == 1)[0]
            binary[one_pos[np.random.default_rng(1).permutation(one_pos.size)[:excess]]] = 0.0
        elif s < target:
            deficit = int(target - s)
            zero_pos = np.where(binary == 0)[0]
            binary[zero_pos[np.random.default_rng(1).permutation(zero_pos.size)[:deficit]]] = 1.0
    best_c = evaluate_c5(binary)
    
    best_h = best.copy()
    if best_c < evaluate_c5(best):
        best = binary.copy()
    else:
        best = best_h.copy()
        best_c = evaluate_c5(best)
    
    rng = np.random.default_rng(200)
    stall = 0
    for _ in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        corr = np.correlate(best, 1.0 - best, mode='full')
        center = n - 1
        top_idx = np.argsort(corr)[-5:]
        top_vals = corr[top_idx]
        spread = max(top_vals.max() - top_vals.min(), 1e-10)
        w = (top_vals - top_vals.min()) / spread + 0.2
        w /= w.sum()
        
        g = np.zeros(n, dtype=float)
        for i, t in enumerate(top_idx):
            k = int(t) - center
            g += w[i] * corr_grad_lag(best, k)
        
        # For binary: try moving a 1 from high-g to low-g position
        one_mask = best > 0.5
        zero_mask = best < 0.5
        
        one_idx = np.where(one_mask)[0]
        zero_idx = np.where(zero_mask)[0]
        if one_idx.size == 0 or zero_idx.size == 0:
            break
        
        # Sort by gradient
        ones_sorted = one_idx[np.argsort(g[one_idx])]  # lowest g first (source)
        zeros_sorted = zero_idx[np.argsort(-g[zero_idx])]  # highest g first (target)
        
        found = False
        for src in ones_sorted[:8]:
            for tgt in zeros_sorted[:8]:
                cand = best.copy()
                cand[src] = 0.0
                cand[tgt] = 1.0
                c = evaluate_c5(cand)
                if c < best_c - 1e-13:
                    best = cand
                    best_c = c
                    found = True
                    break
            if found:
                break
        
        if not found and rng.random() < 0.3:
            # Random binary swap
            si = int(rng.integers(one_idx.size))
            ti = int(rng.integers(zero_idx.size))
            src, tgt = one_idx[si], zero_idx[ti]
            cand = best.copy()
            cand[src] = 0.0
            cand[tgt] = 1.0
            c = evaluate_c5(cand)
            if c < best_c - 1e-13:
                best = cand
                best_c = c
                found = True
        
        if found:
            stall = 0
        else:
            stall += 1
            if stall > 100:
                break
    
    return project_density(best, target), evaluate_c5(best)


def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_ALPHA, _G_TARGET
    t0 = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = t0 + max(1.0, budget - 15.0)
    
    best_h = None
    best_c = float('inf')
    best_n = 0
    
    # Get parent
    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0).copy()
            parent = np.clip(parent, 0.0, 1.0)
    except Exception:
        parent = None
    
    if parent is not None:
        np_ = int(parent.size)
        pp = project_density(parent, 0.5 * np_)
        cp = evaluate_c5(pp)
        best_h = pp.copy()
        best_c = cp
        best_n = np_
    else:
        n0 = 90
        best_h = project_density(np.full(n0, 0.5), 0.5 * n0)
        best_c = evaluate_c5(best_h)
        best_n = n0
    
    rng = np.random.default_rng(seed)
    
    # Phase 1: DE at low resolutions for diverse exploration
    de_ns = [60, 90, 120]
    if best_n <= 90:
        de_ns = [best_n, 90, 120]
    
    for dn in de_ns:
        rem = deadline - time.perf_counter()
        if rem < 40:
            break
        alloc = min(rem * 0.25, 180.0)
        _G_N = dn
        _G_TARGET = 0.5 * dn
        _G_ALPHA = 30.0
        
        # Generate diverse starts
        starts = [upsample_h(best_h, dn)]
        for m in range(3):
            starts.append(make_diverse_start(dn, rng, m))
        
        h_de, c_de = de_global_search(dn, starts, time.perf_counter() + alloc,
                                       pop_size=20, max_evals=2000)
        if c_de < best_c:
            best_c = c_de
            best_n = dn
        # Keep track of good low-res solutions for upscaling
        if c_de < best_c + 0.01:
            # Upsample to current best resolution and try
            pass
    
    # Phase 2: Progressive resolution increase with optimization
    # Start from best_n, double each time
    n_cur = max(best_n, 60)
    n_max = min(720, max(360, n_cur * 2))
    
    resolution_chain = []
    nr = n_cur
    while nr <= n_max:
        resolution_chain.append(nr)
        nr = min(nr * 2, n_max)
    # Add some intermediate
    if len(resolution_chain) < 3:
        for extra in [n_cur * 3 // 2, n_cur * 3]:
            if extra not in resolution_chain and extra <= n_max:
                resolution_chain.append(extra)
    resolution_chain = sorted(set(resolution_chain))
    
    # Allocate time: more to higher resolutions
    total_rem = deadline - time.perf_counter()
    n_stages = len(resolution_chain)
    time_per = total_rem / n_stages
    
    for stage_idx, ni in enumerate(resolution_chain):
        rem = deadline - time.perf_counter()
        if rem < 15:
            break
        alloc = min(time_per * (1 + stage_idx * 0.3), rem * 0.7)
        alloc = max(alloc, 15.0)
        
        # Upsample current best to this resolution
        h_start = upsample_h(best_h, ni)
        target = 0.5 * ni
        
        # Also try from a fresh diverse start
        h_div = make_diverse_start(ni, rng, stage_idx % 5)
        candidates = [h_start, h_div]
        
        # SLSQP
        if ni <= 400:
            slsqp_deadline = time.perf_counter() + alloc * 0.35
            for hc in candidates[:1]:
                rem2 = slsqp_deadline - time.perf_counter()
                if rem2 < 10:
                    break
                h_slq, c_slq = slsqp_refine(hc, ni, slsqp_deadline,
                                            alpha_start=20.0, alpha_end=200.0)
                if c_slq < best_c:
                    best_c = c_slq
                    best_n = ni
                    h_start = h_slq
                    best_h = h_slq.copy()
                h_start = h_slq
        
        # Lag-targeted polish
        rem3 = deadline - time.perf_counter()
        if rem3 > 8:
            pol_dl = time.perf_counter() + min(rem3 * 0.2, 45.0)
            h_pol, c_pol = lag_polish(h_start, pol_dl, max_iter=400, n_top=5)
            if c_pol < best_c:
                best_c = c_pol
                best_n = ni
                best_h = h_pol.copy()
            h_start = h_pol
        
        # Pair swap
        rem4 = deadline - time.perf_counter()
        if rem4 > 8 and ni <= 500:
            swap_dl = time.perf_counter() + min(rem4 * 0.25, 60.0)
            h_swap, c_swap = pair_swap(h_start, swap_dl, max_iter=3000, n_top=5)
            if c_swap < best_c:
                best_c = c_swap
                best_n = ni
                best_h = h_swap.copy()
            h_start = h_swap
        
        # Binarize and refine
        rem5 = deadline - time.perf_counter()
        if rem5 > 5 and ni <= 400:
            h_bin = binarize_and_refine(h_start, target, thresh=0.15)
            c_bin = evaluate_c5(h_bin)
            if c_bin < best_c:
                best_c = c_bin
                h_start = h_bin
                best_h = h_bin.copy()
                best_n = ni
        
        # Binary local search
        rem6 = deadline - time.perf_counter()
        if rem6 > 5 and ni <= 300:
            bls_dl = time.perf_counter() + min(rem6 * 0.3, 30.0)
            h_bls, c_bls = binary_local_search(h_start, bls_dl, max_iter=3000)
            if c_bls < best_c:
                best_c = c_bls
                best_n = ni
                best_h = h_bls.copy()
                h_start = h_bls
        
        # Re-polish after all steps
        rem7 = deadline - time.perf_counter()
        if rem7 > 5:
            pol_dl2 = time.perf_counter() + min(rem7 * 0.3, 30.0)
            h_pol2, c_pol2 = lag_polish(h_start, pol_dl2, max_iter=300, n_top=4)
            if c_pol2 < best_c:
                best_c = c_pol2
                best_n = ni
                best_h = h_pol2.copy()
    
    # Phase 3: Final intensive polish on best solution
    rem = deadline - time.perf_counter()
    if best_h is not None and rem > 10:
        nbest = int(best_h.size)
        target = 0.5 * nbest
        h_fin = project_density(best_h, target)
        
        # Multiple rounds of polish
        for round_num in range(3):
            rem = deadline - time.perf_counter()
            if rem < 8:
                break
            
            # Lag polish
            pol_dl = time.perf_counter() + min(rem * 0.3, 60.0)
            h_p, c_p = lag_polish(h_fin, pol_dl, max_iter=500, n_top=4)
            if c_p < best_c:
                best_c = c_p
                best_n = nbest
            h_fin = h_p
            
            # Pair swap
            rem = deadline - time.perf_counter()
            if rem > 5 and nbest <= 500:
                sw_dl = time.perf_counter() + min(rem * 0.3, 60.0)
                h_s, c_s = pair_swap(h_fin, sw_dl, max_iter=4000, n_top=4)
                if c_s < best_c:
                    best_c = c_s
                    best_n = nbest
                h_fin = h_s
            
            # Try binarization with smaller threshold
            rem = deadline - time.perf_counter()
            if rem > 3 and nbest <= 400:
                h_b = binarize_and_refine(h_fin, target, thresh=0.1 + round_num * 0.05)
                c_b = evaluate_c5(h_b)
                if c_b < best_c:
                    best_c = c_b
                    h_fin = h_b
                    best_h = h_b.copy()
                else:
                    h_fin = project_density(h_fin, target)
        
        # Final binary search if solution is mostly binary
        rem = deadline - time.perf_counter()
        if rem > 5 and nbest <= 300:
            frac_binary = np.mean((h_fin < 0.1) | (h_fin > 0.9))
            if frac_binary > 0.7:
                bls_dl = time.perf_counter() + rem * 0.8
                h_bls, c_bls = binary_local_search(h_fin, bls_dl, max_iter=5000)
                if c_bls < best_c:
                    best_c = c_bls
                    best_h = h_bls.copy()
                    best_n = nbest
                    h_fin = h_bls
    
    # Final verification
    if best_h is None:
        fallback = project_density(np.full(90, 0.5), 45.0)
        return fallback, evaluate_c5(fallback), 90
    
    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_c5(final_h)
    return final_h, float(final_c), int(final_h.size)
