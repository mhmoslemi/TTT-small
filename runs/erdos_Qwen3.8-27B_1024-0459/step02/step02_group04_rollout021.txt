    h_ext[:n] = h

    ac_full = np.fft.ifft(H * np.conj(H)).real[:2*n]

            if float(np.max(tv)) < exact_c + 1e-14:
                break
            step *= 0.5
        h_new = project_feasible(h - step * g, n)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        h_y = h_new + ((t - 1.0) / t_new) * (h_new - h)
        h = h_new
        t = t_new
        if stall > 500:
            h = h_y
            t = 1.0
            stall = 0
    final_c = compute_c5(best_h)
    if final_c < best_c:
        best_c = final_c
    return best_h, best_c

def active_pgd(h0, n, K_list, max_iter, time_limit):
    h = project_feasible(h0, n).copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    dx = 2.0 / float(n)
    for K in K_list:
        for it in range(max_iter):
            if time.time() > time_limit:
                break
            vals = all_lag_vals(h)
            cur_c = float(np.max(vals))
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
            top_idx = np.argsort(-vals)[:K]
            top_vals = vals[top_idx]
            mv = float(top_vals[0])
            d = 30.0 * (top_vals - mv)
            d = np.clip(d, -50, 0)
            ww = np.exp(d)
            ww = ww / float(np.sum(ww))
            g = np.zeros(n)
            for j in range(len(ww)):
                if ww[j] < 1e-8:
                    continue
                idx = int(top_idx[j])
                if idx == 0:
                    L = 0
                elif idx % 2 == 1:
                    L = (idx + 1) // 2
                else:
                    L = -(idx // 2)
                g += ww[j] * lag_grad(h, L, dx)
            step = 0.08
            for _ls in range(10):
                h_trial = project_feasible(h - step * g, n)
                if float(np.max(all_lag_vals(h_trial))) < cur_c:
                    break
                step *= 0.5
            h = project_feasible(h - step * g, n)
    return best_h, best_c

def coord_polish(h0, n, time_limit, max_rounds=200):
    h = project_feasible(h0, n).copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    dx = 2.0 / float(n)
    for rnd in range(max_rounds):
        if time.time() > time_limit:
            break
        improved = False
        for i in range(n):
            if time.time() > time_limit:
                break
            cur_c = compute_c5(h)
            best_local = h[i]
            best_val = cur_c
            for frac in (0.0, 0.25, 0.5, 0.75, 1.0):
                h[i] = frac
                h2 = project_feasible(h, n)
                cv = compute_c5(h2)
                if cv < best_val:
                    best_val = cv
                    best_local = h2.copy()
            if best_val < cur_c - 1e-14:
                h = best_local
                improved = True
        if not improved:
            break
        final_c = compute_c5(h)
        if final_c < best_c:
            best_c = final_c
            best_h = h.copy()
    return best_h, best_c

def block_perturb(h0, n, time_limit, seed, max_trials=5000):
    rng = np.random.RandomState(int(seed))
    h = project_feasible(h0, n).copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_h = h.copy()
    cur_c = best_c
    temp = 0.003
    for _ in range(max_trials):
        if time.time() > time_limit:
            break
        L = int(rng.randint(2, min(10, n)))
        st = int(rng.randint(0, n - L + 1))
        idx = np.arange(st, st + L)
        delta = rng.randn(L) * 0.04
        delta -= float(np.mean(delta))
        h_try = cur_h.copy()
        h_try[idx] += delta
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try)
        if c_try < cur_c:
            cur_h = h_try
            cur_c = c_try
            if c_try < best_c:
                best_h = h_try.copy()
                best_c = c_try
                temp = min(0.03, temp * 1.1)
        else:
            d = c_try - cur_c
            if d < temp:
                if rng.rand() < np.exp(-d / temp):
                    cur_h = h_try
                    cur_c = c_try
        temp = max(temp * 0.997, 1e-6)
    return best_h, best_c

def interpolate_to_n(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float)
    n_dst = int(n_dst)
    n_src = h_src.size
    if n_src == n_dst:
        return h_src.copy()
    if n_src <= 1:
        return np.full(n_dst, 0.5, dtype=float)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def binary_quantize_and_project(h, n):
    h = project_feasible(h, n)
    h2 = (h > 0.5).astype(float)
    h2 = project_feasible(h2, n)
    return h2

def make_variants(h, n, rng, count=5):
    base = project_feasible(h, n)
    out = [base.copy()]
    flip = base[::-1]
    out.append(project_feasible(0.5 * (base + flip), n))
    h_smooth = np.convolve(base, np.array([0.15, 0.35, 0.35, 0.15]), mode='same')
    out.append(project_feasible(h_smooth, n))
    pert = base + rng.randn(n) * 0.04
    out.append(project_feasible(pert, n))
    h_b = binary_quantize_and_project(base, n)
    out.append(h_b)
    if count is not None and len(out) > count:
        out = out[:count]
    return out

def full_optimize(h0, n, time_limit, seed):
    h = project_feasible(h0, n).copy()
    best_h = h.copy()
    best_c = compute_c5(h)

    if time.time() < time_limit - 30:
        h2, c2 = fista_smooth(h, n, 30, 300, time.time() + 25.0)
        if c2 < best_c:
            best_h, best_c = h2, c2
        h = best_h

    if time.time() < time_limit - 25:
        h2, c2 = fista_smooth(h, n, 80, 400, time.time() + 30.0)
        if c2 < best_c:
            best_h, best_c = h2, c2
        h = best_h

    if time.time() < time_limit - 20:
        h2, c2 = fista_smooth(h, n, 200, 500, time.time() + 25.0)
        if c2 < best_c:
            best_h, best_c = h2, c2
        h = best_h

    if time.time() < time_limit - 20:
        K_list = (8, 16, 32, 64) if n <= 200 else (8, 16, 32)
        h2, c2 = active_pgd(h, n, K_list, 200, time.time() + 25.0)
        if c2 < best_c:
            best_h, best_c = h2, c2
        h = best_h

    if time.time() < time_limit - 15:
        h2, c2 = block_perturb(h, n, time.time() + 15.0, seed, max_trials=3000)
        if c2 < best_c:
            best_h, best_c = h2, c2
        h = best_h

    if time.time() < time_limit - 5:
        h2, c2 = fista_smooth(h, n, 300, 300, time.time() + 5.0)
        if c2 < best_c:
            best_h, best_c = h2, c2

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h)
    return best_h, float(best_c)

def run(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    budget_s = float(budget_s)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    hard_end = start_time + max(60.0, budget_s - 45.0)

    parent = get_parent_values()
    if parent is not None:
        pn = int(parent.size)
        best_h = project_feasible(parent, pn)
        best_c = compute_c5(best_h)
        best_n = pn
    else:
        best_n = 160
        best_h = project_feasible(np.full(best_n, 0.5), best_n)
        best_c = compute_c5(best_h)

    # Phase 1: Optimize at parent resolution aggressively
    if best_n >= 8 and time.time() < hard_end - 400:
        slot = min(200.0, hard_end - time.time() - 350)
        h1, c1 = full_optimize(best_h, best_n, time.time() + slot, seed + 1)
        if c1 < best_c:
            best_h, best_c, best_n = h1, c1, best_n

    # Phase 2: Multi-start at parent resolution
    if time.time() < hard_end - 300:
        variants = make_variants(best_h, best_n, rng, count=4)
        for j, v in enumerate(variants[1:]):
            if time.time() > hard_end - 250:
                break
            slot = min(60.0, hard_end - time.time() - 200)
            if slot < 20:
                break
            h2, c2 = full_optimize(v, best_n, time.time() + slot, seed + 100 + j)
            if c2 < best_c:
                best_h, best_c = h2, c2

    # Phase 3: Try nearby resolutions
    res_list = []
    for d in (-16, -8, 8, 16, 32, -32, 48, -48):
        nn = best_n + d
        if 96 <= nn <= 320:
            res_list.append(nn)
    for nn in (128, 192, 224, 256):
        if nn not in res_list:
            res_list.append(nn)
    res_list = sorted(set(res_list))

    for nn in res_list[:5]:
        if time.time() > hard_end - 120:
            break
        nn = int(nn)
        if < 96 or nn > 320:
            continue
        h_interp = interpolate_to_n(best_h, nn)
        h_interp = project_feasible(h_interp, nn)
        variants_nn = make_variants(h_interp, nn, rng, count=3)
        slot = min(100.0, hard_end - time.time() - 80)
        if slot < 30:
            break
        for j, v in enumerate(variants_nn[:3]):
            if time.time() > hard_end - 80:
                break
            s2 = slot / float(3 - j)
            h2, c2 = full_optimize(v, nn, time.time() + s2, seed + 5000 + nn * 7 + j)
            if c2 < best_c:
                best_h, best_c, best_n = h2, c2, nn

    # Phase 4: Final polishing at best resolution
    if time.time() < hard_end - 30 and 8 <= best_n <= 320:
        h2, c2 = fista_smooth(best_h, best_n, 500, 500, time.time() + 20.0)
        if c2 < best_c:
            best_h, best_c = h2, c2

    if time.time() < hard_end - 10 and 8 <= best_n <= 240:
        h2, c2 = active_pgd(best_h, best_n, (16, 32, 64), 150, time.time() + 8.0)
        if c2 < best_c:
            best_h, best_c = h2, c2

    if time.time() < hard_end - 3 and 8 <= best_n <= 240:
        h2, c2 = block_perturb(best_h, best_n, hard_end - 2.0, seed + 99999, max_trials=2000)
        if c2 < best_c:
            best_h, best_c = h2, c2

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)
    return (best_h, float(best_c), int(best_n))
