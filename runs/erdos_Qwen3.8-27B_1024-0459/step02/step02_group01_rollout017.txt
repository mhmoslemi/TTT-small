import numpy as np

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        # Adjust non-saturated elements
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
        else:
            # All saturated, find closest to boundary
            if diff > 0:
                idx = np.argmax(h)  # closest to 1
                h[idx] += min(diff, 1.0 - h[idx])
            else:
                idx = np.argmin(h)
                h[idx] += max(diff, -h[idx])
            h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Distribute among all elements proportionally
        if diff > 0:
            room = 1.0 - h
            total_room = room.sum()
            if total_room > 0:
                h += room * (diff / total_room)
            else:
                h += diff / n
        else:
            room = h
            total_room = room.sum()
            if total_room > 0:
                h -= room * (-diff / total_room)
            else:
                h += diff / n
        h = np.clip(h, 0.0, 1.0)
    # Ensure exact sum
    h = np.clip(h, 0.0, 1.0)
    return h

def compute_all_corrs(h):
    """Compute all cross-correlations h with 1-h, full mode."""
    return np.correlate(h, 1.0 - h, mode='full')

def grad_corr_lag(h, L):
    """Gradient of corr at lag L w.r.t. h. L can be negative."""
    n = len(h)
    g = np.zeros(n)
    if L >= 0:
        if L < n:
            g[:n-L] += (1.0 - h[L:])
            g[L:] -= h[:n-L]
    else:
        Lp = -L
        if Lp < n:
            g[Lp:] += (1.0 - h[:n-Lp])
            g[:n-Lp] -= h[Lp:]
    return g

def projected_gradient_optimize(h0, n, time_limit_s, t0, top_k=1, momentum=0.9):
    """Projected gradient with momentum and backtracking."""
    h = project_feasible(h0.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_start = time.time()
    
    step_size = 0.01
    v = np.zeros(n)  # momentum
    
    while time.time() - t_start < time_limit_s and time.time() - t0 < 1300 - 5:
        corr = compute_all_corrs(h)
        # Find top-k lags
        k = min(top_k, len(corr))
        # Use softmax for smooth weighting
        top_idx = np.argsort(corr)[-k:]
        # Softmax weights
        top_vals = corr[top_idx]
        beta_w = 50.0  # sharpness of weight
        w = np.exp(beta_w * (top_vals - top_vals.max()))
        w /= w.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        for idx in top_idx:
            L = idx - (n - 1)
            grad += w[np.where(top_idx == idx)[0][0]] * grad_corr_lag(h, L)
        
        # Momentum
        v = momentum * v - step_size * grad
        h_new = h + v
        
        # Backtracking
        c5_new = compute_c5(h_new)
        if c5_new > best_c5 + 1e-10:
            # Backtrack
            for _ in range(20):
                step_size *= 0.5
                v = momentum * v - step_size * grad  # hmm, this isn't quite right
                h_new = h + v
                c5_new = compute_c5(h_new)
                if c5_new <= best_c5:
                    break
            if c5_new > best_c5:
                # Accept anyway but reduce step
                step_size *= 0.5
                h_new = h - step_size * grad / (np.linalg.norm(grad) + 1e-10)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        
        h = project_feasible(h_new, n)
        c5_cur = compute_c5(h)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
        
        # Adaptive top_k
        # (could increase over time)
    
    return best_h, best_c5

The element at index i is: sum_j a[j] * v[j + i - (len(v)-1)]

So for a=h, v=1-h, element at index i:
sum_j h[j] * (1-h[j + i - (n-1)])

Let L = i - (n-1). Then:
sum_j h[j] * (1-h[j+L]) for valid j (0≤j<n and 0≤j+L<n)

For the C₅ objective, we take the max over ALL i from 0 to 2n-2.

Note: at i = n-1 (L=0), the value is sum_j h[j]*(1-h[j]) = 0 for binary h.

def grad_corr_lag(h, L):
    n = len(h)
