def bound_step(h, dir):
   pos=dir>1e-12; neg=dir<-1e-12
   maxs=1e9
   if any pos: maxs=min(maxs, np.min((1-h[pos])/dir[pos]))
   if any neg: maxs=min(maxs, np.min(h[neg]/(-dir[neg])))
   return max(maxs,1e-12)

def project_feasible(h, n=None):
    h = np.asarray(h, dtype=float).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)

    if n is None:
        n = h.size
    else:
        n = int(n)

    if n <= 0:
        return np.empty(0, dtype=float)

    if h.size < n:
        pad = np.zeros(n - h.size, dtype=float)
        h = np.concatenate((h, pad))
    elif h.size > n:
        h = h[:n]

    target = 0.5 * float(n)
    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h)) + 1.0

    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid

    tau = 0.5 * (lo + hi)
    out = np.clip(h - tau, 0.0, 1.0)

    diff = target - float(np.sum(out))
    for _ in range(25):
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            mask = out < 1.0 - 1e-12
        else:
            mask = out > 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            if diff > 0.0 and np.any(out < 1.0):
                idx = int(np.argmax(out < 1.0))
                out[idx] += diff
            elif diff < 0.0 and np.any(out > 0.0):
                idx = int(np.argmax(out > 0.0))
                out[idx] += diff
            break
        out[mask] += diff / float(cnt)
        out = np.clip(out, 0.0, 1.0)
        diff = target - float(np.sum(out))

    return out

def gaussian_smooth(h, sigma=1.5):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n <= 1:
        return h.copy()
    length = min(7, n)
    if length % 2 == 0:
        length -= 1
    length = max(1, length)
    half = length // 2
    k = np.arange(-half, half + 1, dtype=float)
    w = np.exp(-(k * k) / (2.0 * float(sigma) * float(sigma)))
    w /= float(np.sum(w))
    return np.convolve(h, w, mode="same")

def linear_acorr(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    ac = np.empty(max(1, n), dtype=float)
    if n == 0:
        ac[0] = 0.0
        return ac

    if n <= 128:
        for L in range(n):
            m = n - L
            ac[L] = float(np.dot(h[:m], h[L:]))
        return ac

    N = 1 << (2 * n - 1).bit_length()
    f = np.fft.rfft(h, N)
    ac = np.fft.irfft(f * np.conj(f), N)[:n]
    ac = np.asarray(ac, dtype=float)
    ac[0] = float(np.dot(h, h))

    if float(np.min(ac)) < -1e-7:
        for L in range(n):
            m = n - L
            ac[L] = float(np.dot(h[:m], h[L:]))
    else:
        ac[ac < 0.0] = 0.0
    return ac


def all_lag_values_raw(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return np.array([0.0])

    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = float(pref[-1])

    ac = linear_acorr(h)
    vals = np.empty(2 * n - 1, dtype=float)
    off = n - 1
    vals[off] = total - ac[0]

    for L in range(1, n):
        m = n - L
        a = ac[L]
        vals[off + L] = pref[m] - a
        vals[off - L] = total - pref[L] - a

    return np.maximum(vals, 0.0)


def eval_fast(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n <= 0:
        return 0.0
    vals = all_lag_values_raw(h)
    return float(np.max(vals)) * (2.0 / float(n))


def exact_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n <= 0:
        return 0.0
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def lag_value_raw(h, L):
    h = np.asarray(h, dtype=float)
    n = h.size
    L = int(L)
    if n == 0:
        return 0.0
    if L == 0:
        return float(np.sum(h) - np.dot(h, h))
    a = abs(L)
    m = n - a
    ac = float(np.dot(h[:m], h[a:]))
    if L > 0:
        return float(np.sum(h[:m]) - ac)
    total = float(np.sum(h))
    return total - float(np.sum(h[:a])) - ac


def lag_gradient_raw(h, L):
    h = np.asarray(h, dtype=float)
    n = h.size
    g = np.zeros(n, dtype=float)
    L = int(L)
    if n == 0:
        return g
    if L == 0:
        return 1.0 - 2.0 * h
    a = abs(L)
    m = n - a
    if L > 0:
        g[:m] += 1.0 - h[a:]
        g[a:] -= h[:m]
    else:
        g[a:] += 1.0 - h[:m]
        g[:m] -= h[a:]
    return g


def add_raw_lag_gradient_to(g, h, L, coef):
    if coef == 0.0:
        return
    h = np.asarray(h, dtype=float)
    n = h.size
    L = int(L)
    if n == 0:
        return
    if L == 0:
        g += coef * (1.0 - 2.0 * h)
        return
    a = abs(L)
    m = n - a
    if L > 0:
        g[:m] += coef * (1.0 - h[a:])
        g[a:] -= coef * h[:m]
    else:
        g[a:] += coef * (1.0 - h[:m])
        g[:m] -= coef * h[a:]


def smooth_value_raw(h, beta):
    vals = all_lag_values_raw(h)
    if vals.size == 0:
        return 0.0
    beta = max(1e-6, float(beta))
    maxv = float(np.max(vals))
    d = np.clip(beta * (vals - maxv), -745.0, 700.0)
    w = np.exp(d)
    s = float(np.sum(w))
    if s <= 0.0:
        return maxv
    return float(maxv + np.log(s) / beta)


def smooth_grad_raw(h, beta):
    h = np.asarray(h, dtype=float)
    n = h.size
    g = np.zeros(n, dtype=float)
    if n == 0:
        return g

    vals = all_lag_values_raw(h)
    beta = max(1e-6, float(beta))
    maxv = float(np.max(vals))
    d = np.clip(beta * (vals - maxv), -745.0, 700.0)
    w = np.exp(d)
    tot = float(np.sum(w))
    if tot <= 0.0:
        mask = vals >= (maxv - 1e-12)
        w = mask.astype(float)
        tot = float(np.sum(w))
    if tot > 0.0:
        w = w / tot
    else:
        return g

    off = n - 1
    c0 = float(w[off])
    if c0 > 1e-15:
        g += c0 * (1.0 - 2.0 * h)

    posw = w[off + 1:]
    negw = w[:off][::-1]

    for L in range(1, n):
        m = n - L
        cp = float(posw[L - 1])
        if cp > 1e-15:
            g[:m] += cp * (1.0 - h[L:])
            g[L:] -= cp * h[:m]
        cm = float(negw[L - 1])
        if cm > 1e-15:
            g[L:] += cm * (1.0 - h[:m])
            g[:m] -= cm * h[L:]
    return g


def sanitize_direction(h, dirv):
    h = np.asarray(h, dtype=float)
    dirv = np.asarray(dirv, dtype=float).copy()
    n = h.size
    if n == 0:
        return dirv
    dirv -= float(np.mean(dirv))
    for _ in range(3):
        bad_pos = (h > 1.0 - 1e-9) & (dirv > 0.0)
        bad_neg = (h < 1e-9) & (dirv < 0.0)
        if not (np.any(bad_pos) or np.any(bad_neg)):
            break
        dirv[bad_pos] = 0.0
        dirv[bad_neg] = 0.0
        s = float(np.sum(dirv))
        if abs(s) > 1e-14:
            mask = (h > 1e-9) & (h < 1.0 - 1e-9)
            cnt = int(np.count_nonzero(mask))
            if cnt > 0:
                dirv[mask] -= s / float(cnt)
            else:
                break
    return dirv


def bound_lambda(h, dirv):
    h = np.asarray(h, dtype=float)
    dirv = np.asarray(dirv, dtype=float)
    lam = 1.0e300
    pos = dirv > 1e-12
    neg = dirv < -1e-12
    if np.any(pos):
        vals = (1.0 - h[pos]) / dirv[pos]
        if vals.size:
            lam = min(lam, float(np.min(vals)))
    if np.any(neg):
        vals = h[neg] / (-dirv[neg])
        if vals.size:
            lam = min(lam, float(np.min(vals)))
    if not np.isfinite(lam) or lam < 0.0:
        lam = 0.0
    return lam


def make_feasible_step(h, dirv, s, n):
    h = np.asarray(h, dtype=float)
    dirv = np.asarray(dirv, dtype=float)
    n = int(n)
    cand = h + float(s) * dirv
    if cand.min() < 0.0 or cand.max() > 1.0:
        cand = np.clip(cand, 0.0, 1.0)

    target = 0.5 * float(n)
    for _ in range(3):
        diff = target - float(np.sum(cand))
        if abs(diff) < 1e-12:
            break
        mask = (cand > 1e-10) & (cand < 1.0 - 1e-10)
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            cand = project_feasible(cand, n)
            break
        cand[mask] += diff / float(cnt)
        cand = np.clip(cand, 0.0, 1.0)

    if abs(0.5 * float(n) - float(np.sum(cand))) > 1e-10:
        cand = project_feasible(cand, n)
    return cand


def trial_direction(h, dirv, step, evals=3):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return None

    dirv = sanitize_direction(h, dirv)
    nrm = float(np.linalg.norm(dirv))
    if nrm < 1e-10:
        return None
    dirv = dirv / nrm

    lam = bound_lambda(h, dirv)
    if lam <= 1e-10:
        return None

    s = min(float(step), 0.5 * lam)
    if s <= 1e-10:
        return None

    best_c = None
    best_cand = None
    best_s = None
    trials = int(max(1, evals))

    for _ in range(trials):
        cand = make_feasible_step(h, dirv, s, n)
        ctry = eval_fast(cand)
        if best_c is None or ctry < best_c:
            best_c = ctry
            best_cand = cand
            best_s = s
        s *= 0.5
        if s < 1e-8:
            break

    if best_c is None:
        return None
    return best_cand, best_c, best_s


def smooth_linesearch_impl(h0, n, beta, max_iter, time_limit, step):
    n = int(n)
    h = project_feasible(h0, n)
    c = eval_fast(h)
    best_h = h.copy()
    best_c = c

    s = float(step)
    no = 0
    max_iter = int(max_iter)

    for _ in range(max_iter):
        if time.time() > float(time_limit):
            break
        g = smooth_grad_raw(h, beta)
        g -= float(np.mean(g))
        nrm = float(np.linalg.norm(g))
        if nrm < 1e-9:
            break
        dirv = -g / nrm

        res = trial_direction(h, dirv, s, evals=6)
        if res is not None and res[1] + 1e-12 < c:
            cand, cnew, snew = res
            h = cand
            c = cnew
            s = min(0.12, max(s, float(snew) * 1.05))
            no = 0
            if c < best_c:
                best_h = cand.copy()
                best_c = c
        else:
            s = max(1e-5, s * 0.6)
            no += 1
            if no > 50 or s < 1e-5:
                break

    return best_h, best_c


def get_active_info(h, k=24):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return [], np.array([])
    vals = all_lag_values_raw(h)
    m = min(int(k), vals.size)
    idx = np.argsort(-vals)[:m]
    off = n - 1
    lags = []
    for ix in idx:
        lags.append(int(ix) - off)
    return lags, vals[idx].copy()


def local_active_search(h0, n, time_limit, seed):
    n = int(n)
    h = project_feasible(h0, n)
    c = eval_fast(h)
    best_h = h.copy()
    best_c = c

    rng = np.random.RandomState(int(seed))
    cur_step = 0.035
    temp = 2.0e-4
    min_temp = 1.0e-9
    no = 0
    need_recompute = True
    active_lags = []
    active_weights = np.array([], dtype=float)

    while True:
        if time.time() > float(time_limit):
            break

        if need_recompute or len(active_lags) == 0:
            active_lags, active_vals = get_active_info(h, k=24)
            if len(active_vals) == 0:
                active_weights = np.array([], dtype=float)
            else:
                mv = float(np.max(active_vals))
                raws = active_vals - mv
                wexp = np.exp(10.0 * raws)
                wexp = np.where(np.isfinite(wexp) & (wexp > 0.0), wexp, 0.0)
                sw = float(np.sum(wexp))
                if sw <= 0.0:
                    wexp = np.ones_like(active_vals, dtype=float)
                    sw = float(wexp.size)
                active_weights = wexp / sw
            need_recompute = False

        r = float(rng.rand())
        g = None

        if len(active_lags) > 0 and r < 0.60:
            m = len(active_lags)
            k = int(rng.randint(1, min(6, m) + 1))
            g = np.zeros(n, dtype=float)
            if m == k:
                idxs = np.arange(m)
            else:
                p = active_weights.copy()
                p = np.where(np.isfinite(p) & (p > 0.0), p, 1.0e-12)
                ps = float(np.sum(p))
                if ps <= 0.0:
                    idxs = rng.choice(m, size=k, replace=False)
                else:
                    p = p / ps
                    idxs = rng.choice(m, size=k, replace=False, p=p)
            for val in idxs:
                ix = int(val)
                coef = 0.25 + 0.75 * float(rng.random())
                add_raw_lag_gradient_to(g, h, active_lags[ix], coef)

        elif r < 0.80:
            beta = (5.0, 20.0, 80.0, 300.0)[int(rng.randint(4))]
            g = smooth_grad_raw(h, beta)

        elif n > 1:
            max_len = min(12, n)
            if max_len < 2:
                L = 1
            else:
                L = int(rng.randint(2, max_len + 1))
            st = int(rng.randint(0, n - L + 1))
            delta = rng.randn(L)
            delta = delta - float(np.mean(delta))
            g = np.zeros(n, dtype=float)
            g[st:st + L] = delta

        if g is None:
            g = rng.randn(n)

        g -= float(np.mean(g))
        nrm = float(np.linalg.norm(g))
        if nrm < 1e-10:
            g = rng.randn(n)
            g -= float(np.mean(g))
            nrm = float(np.linalg.norm(g))
            if nrm < 1e-10:
                continue
        dirv = -g / nrm

        test_step = cur_step
        if no > 20:
            test_step = min(0.12, cur_step * 2.0 + 0.01)

        res = trial_direction(h, dirv, test_step, evals=3)
        accepted = False

        if res is not None:
            cand, cnew, snew = res
            if cnew + 1e-12 < c:
                h = cand
                c = cnew
                cur_step = min(0.12, max(cur_step, float(snew) * 1.05))
                no = 0
                need_recompute = True
                accepted = True
                if c < best_c:
                    best_h = cand.copy()
                    best_c = c
            elif no > 15 and cnew < c + temp and float(rng.rand()) < 0.06:
                h = cand
                c = cnew
                no = 0
                need_recompute = True
                accepted = True

        if not accepted:
            cur_step = max(1e-5, cur_step * 0.85)
            no += 1

        temp = max(min_temp, temp * 0.99)

        if no > 220:
            h = best_h.copy()
            c = best_c
            cur_step = 0.05
            no = 0
            temp = 2.0e-4
            need_recompute = True

    return best_h, best_c


def active_t_obj(x):
    return float(x[-1])


def active_t_obj_jac(x):
    j = np.zeros(len(x), dtype=float)
    j[-1] = 1.0
    return j


def active_sum_eq(x):
    return float(np.sum(x[:-1]) - 0.5 * float(len(x) - 1))


def active_sum_eq_jac(x):
    j = np.zeros(len(x), dtype=float)
    j[:-1] = 1.0
    return j


def active_ineq(x):
    n = len(x) - 1
    dx = 2.0 / float(n)
    h = x[:-1]
    out = np.empty(len(AC_LAGS), dtype=float)
    for i, ll in enumerate(AC_LAGS):
        out[i] = float(x[-1]) - lag_value_raw(h, int(ll)) * dx
    return out


def active_ineq_jac(x):
    n = len(x) - 1
    dx = 2.0 / float(n)
    h = x[:-1]
    m = len(AC_LAGS)
    J = np.zeros((m, n + 1), dtype=float)
    if m == 0:
        return J
    for i, ll in enumerate(AC_LAGS):
        g = lag_gradient_raw(h, int(ll))
        J[i, :n] = -dx * g
        J[i, n] = 1.0
    return J


def optimize_active_t(h0, active_lags, t0, maxiter=50, ftol=1e-13):
    global AC_LAGS
    h = project_feasible(np.asarray(h0, dtype=float))
    n = h.size
    AC_LAGS = [int(l) for l in active_lags]

    x0 = np.empty(n + 1, dtype=float)
    x0[:n] = h
    x0[n] = float(t0)

    bounds = [(0.0, 1.0) for _ in range(n)]
    bounds.append((0.0, 1.0))

    constraints = [{"type": "eq", "fun": active_sum_eq, "jac": active_sum_eq_jac}]
    if len(AC_LAGS) > 0:
        constraints.append({"type": "ineq", "fun": active_ineq, "jac": active_ineq_jac})

    try:
        res = minimize(
            active_t_obj,
            x0,
            jac=active_t_obj_jac,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options={"maxiter": int(maxiter), "ftol": float(ftol), "disp": False},
        )
        return res
    except Exception:
        return None


def slsqp_active_polish(h0, n, time_limit, active_counts=(8, 16), margins=(0.001, 0.0003)):
    n = int(n)
    h = project_feasible(h0, n)
    c = eval_fast(h)
    best_h = h.copy()
    best_c = c
    dx = 2.0 / float(n)

    for acnt in active_counts:
        for margin_c in margins:
            if time.time() > float(time_limit) - 5.0:
                break
            vals = all_lag_values_raw(h)
            maxv = float(np.max(vals))
            cmax = maxv * dx

            threshold = maxv - float(margin_c) / dx
            idx_near = np.where(vals >= threshold)[0]
            idx_top = np.argsort(-vals)[:min(int(acnt), vals.size)]
            idx = np.unique(np.concatenate((idx_near, idx_top)))
            idx = idx[np.argsort(-vals[idx])]
            idx = idx[:max(1, int(acnt))]

            lags = []
            off = n - 1
            for ix in idx:
                lags.append(int(ix) - off)

            if len(lags) == 0:
                continue

            maxit = 50 if acnt <= 8 else 32
            res = optimize_active_t(h, lags, cmax + 1e-8, maxiter=maxit, ftol=1e-13)
            if res is not None and res.x is not None and len(res.x) == n + 1:
                if np.all(np.isfinite(res.x)):
                    h2 = project_feasible(res.x[:-1], n)
                    c2 = eval_fast(h2)
                    if c2 < best_c:
                        best_h = h2.copy()
                        best_c = c2
                    if c2 <= cmax + 1e-8:
                        h = h2

    return best_h, best_c


def lbfgs_objective(x):
    n = len(x)
    if n == 0:
        return 0.0
    beta = max(1.0, float(LBFGS_BETA))
    obj = smooth_value_raw(x, beta)
    diff = float(np.sum(x)) - 0.5 * float(n)
    obj += float(LBFGS_PEN) * diff * diff
    return float(obj)


def lbfgs_gradient(x):
    n = len(x)
    if n == 0:
        return np.empty(0, dtype=float)
    beta = max(1.0, float(LBFGS_BETA))
    g = smooth_grad_raw(x, beta)
    diff = float(np.sum(x)) - 0.5 * float(n)
    if abs(diff) > 0.0:
        g += (2.0 * float(LBFGS_PEN) * diff)
    return g


def run_lbfgs_smooth(h0, n, beta, pen, maxiter=70):
    global LBFGS_BETA, LBFGS_PEN
    n = int(n)
    h = project_feasible(h0, n)
    LBFGS_BETA = float(beta)
    LBFGS_PEN = float(pen)

    bounds = [(0.0, 1.0) for _ in range(n)]
    c0 = eval_fast(h)
    try:
        res = minimize(
            lbfgs_objective,
            h,
            jac=lbfgs_gradient,
            method="L-BFGS-B",
            bounds=bounds,
            options={"maxiter": int(maxiter), "ftol": 1e-12, "gtol": 1e-10, "maxls": 20, "disp": False},
        )
        if res is not None and res.x is not None and np.all(np.isfinite(res.x)):
            h2 = project_feasible(res.x, n)
            c2 = eval_fast(h2)
            if c2 < c0:
                return h2, c2
    except Exception:
        pass
    return h, c0


def lbfgs_smooth_polish(h0, n, time_limit):
    n = int(n)
    h = project_feasible(h0, n)
    c = eval_fast(h)
    best_h = h.copy()
    best_c = c

    betas = (5.0, 20.0, 80.0, 300.0) if n <= 220 else (5.0, 20.0, 80.0)
    for beta in betas:
        if time.time() > float(time_limit) - 5.0:
            break
        maxit = 60 if n <= 200 else 30
        h2, c2 = run_lbfgs_smooth(h, n, beta, 1.0e5, maxiter=maxit)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = c2
        if c2 <= c + 1e-8:
            h = h2
    return best_h, best_c


def improve_candidate(start_h, n, time_limit, seed, smooth=True, use_slsqp=True, use_lbfgs=False):
    n = int(n)
    h = project_feasible(start_h, n)
    c = eval_fast(h)
    best_h = h.copy()
    best_c = c

    now = time.time()
    total_rem = float(time_limit) - now
    if total_rem < 8.0:
        return best_h, best_c

    if smooth and 2 <= n <= 380:
        if use_lbfgs and total_rem > 30:
            lbfgs_end = now + min(45.0, total_rem * 0.22)
            h2, c2 = lbfgs_smooth_polish(best_h, n, lbfgs_end)
            if c2 < best_c:
                best_h = h2.copy()
                best_c = c2

        if n <= 200:
            betas = (2.0, 10.0, 40.0, 160.0, 500.0)
            maxit = 70
        else:
            betas = (2.0, 8.0, 30.0, 120.0)
            maxit = 32

        smooth_end = now + min(70.0, total_rem * 0.30)
        for beta in betas:
            if time.time() > smooth_end:
                break
            h2, c2 = smooth_linesearch_impl(best_h, n, beta, maxiter=maxit, time_limit=smooth_end, step=0.035)
            if c2 < best_c:
                best_h = h2.copy()
                best_c = c2

        local_end = float(time_limit) - max(18.0, total_rem * 0.20)
        local_end = min(local_end, now + total_rem * 0.55)
        if time.time() < local_end - 2.0 and n <= 380:
            h2, c2 = local_active_search(best_h, n, local_end, int(seed) + 17)
            if c2 < best_c:
                best_h = h2.copy()
                best_c = c2

        if use_slsqp and n <= 240 and time.time() < float(time_limit) - 18.0:
            h2, c2 = slsqp_active_polish(best_h, n, float(time_limit) - 2, active_counts=(8, 16), margins=(0.0012, 0.00035))
            if c2 < best_c:
                best_h = h2.copy()
                best_c = c2

    elif time.time() < float(time_limit) - 5.0 and n >= 2:
        h2, c2 = local_active_search(best_h, n, float(time_limit) - 2, int(seed) + 31)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = c2

    best_h = project_feasible(best_h, n)
    best_c = eval_fast(best_h)
    return best_h, float(best_c)


def process_initial_set(inits, n, slot_end, seed, smooth=True, use_slsqp=True, use_lbfgs=False):
    n = int(n)
    best_h = None
    best_c = np.inf
    m = len(inits)
    cap = 75.0 if n <= 200 else 50.0

    for j in range(m):
        if time.time() >= float(slot_end):
            break
        h0 = inits[j]
        rem = float(slot_end) - time.time()
        if rem <= 0.0:
            break
        per = rem / float(max(1, m - j))
        per = min(cap, per)
        per = max(6.0, per)
        per = min(per, rem)
        hh, cc = improve_candidate(h0, n, time.time() + per, int(seed) + 101 * j,
                                   smooth=smooth, use_slsqp=use_slsqp, use_lbfgs=use_lbfgs)
        if cc < best_c:
            best_h = hh
            best_c = cc

    if best_h is None:
        if len(inits) > 0:
            best_h = project_feasible(inits[0], n)
            best_c = eval_fast(best_h)
        else:
            best_h = project_feasible(np.full(int(n), 0.5, dtype=float), n)
            best_c = eval_fast(best_h)

    return best_h, float(best_c)


def build_candidate_order(base_n):
    base = int(base_n)
    raw = []

    deltas = (4, 8, 12, 16, 20, 24, 32, 40, 48, 64, 80, 96, 128)
    for d in deltas:
        for nn in (base - d, base + d):
            nn = int(nn)
            if 64 <= nn <= 384:
                if nn % 2 != 0:
                    nn += 1
                if 64 <= nn <= 384:
                    raw.append(nn)

    extra = (96, 112, 128, 144, 160, 176, 192, 208, 224, 240, 256, 288, 320, 352)
    for nn in extra:
        raw.append(nn)

    seen = set()
    uniq = []
    for nn in raw:
        nn = int(nn)
        if nn not in seen:
            seen.add(nn)
            uniq.append(nn)

    keys = []
    for nn in uniq:
        keys.append((abs(nn - base), nn))
    keys.sort()

    ordered = []
    for key in keys:
        ordered.append(key[1])
    return ordered


def run(seed=42, budget_s=1300, **kwargs):

            raw_h = np.clip(p0, 0.0, 1.0)

        else:

        else:
            best_h = proj_h
            best_c = proj_c
        best_n = pn
    else:
        best_n = 160
        best_h = project_feasible(np.full(best_n, 0.5, dtype=float), best_n)

    if best_n > 320:

        other_slot = 55.0
    else:
        base_slot = min(110.0, remaining_total * 0.45)
        other_slot = max(20.0, remaining_total * 0.10)

    main_end = hard_end - final_reserve
    if main_end < start_time + 30.0:
        main_end = hard_end - 5.0

    base_end = start_time + base_slot
    if base_end > main_end - 25.0:
        base_end = max(start_time + 10.0, main_end - 25.0)

    inits0 = build_inits_for_n(base_n, parent, best_h, best_n, rng,
                               include_structured=(parent is None), include_variants=True)
    if len(inits0) == 0:
        inits0 = [project_feasible(best_h, base_n)]

    hh, cc = process_initial_set(inits0, base_n, base_end, seed + 1,
                                 smooth=(2 <= base_n <= 380), use_slsqp=(base_n <= 240), use_lbfgs=True)
    if hh is not None and cc < best_c:
        best_h = hh
        best_c = cc
        best_n = int(base_n)

    cand_order = build_candidate_order(best_n)
    processed = {int(best_n)}

    for nn in cand_order:
        nn = int(nn)
        if time.time() > main_end:
            break
        if nn in processed:
            continue
        if nn < 48 or nn > 384:
            continue
        processed.add(nn)

        rem = main_end - time.time()
        if rem < 35.0:
            break
        slot = min(other_slot, rem)

        inits = build_inits_for_n(nn, parent, best_h, best_n, rng,
                                  include_structured=False, include_variants=True)
        if len(inits) == 0:
            inits = [project_feasible(np.full(nn, 0.5, dtype=float), nn)]

        hh, cc = process_initial_set(inits, nn, time.time() + slot, seed + 5000 + 13 * nn,
                                     smooth=(2 <= nn <= 380), use_slsqp=(nn <= 240), use_lbfgs=False)
        if hh is not None and cc < best_c:
            best_h = hh
            best_c = cc
            best_n = nn

    if time.time() < hard_end - 15.0 and 2 <= int(best_n) <= 384:
        hh, cc = improve_candidate(best_h, int(best_n), hard_end - 3.0, seed + 99991,
                                   smooth=True, use_slsqp=(int(best_n) <= 240), use_lbfgs=False)
        if hh is not None and cc < best_c:
            best_h = hh
            best_c = cc
            best_n = int(best_n)

    best_h = project_feasible(best_h, int(best_n))
    best_c = exact_c5(best_h)

    return (best_h, float(best_c), int(best_n))
