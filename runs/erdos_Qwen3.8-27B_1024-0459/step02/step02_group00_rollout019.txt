                                         maxfun=50000, iprint=0)
                    h_new = project_feasible(res, n)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_cur:
                        h0 = h_new.copy()
                        c5_cur = c5_new
                except Exception:
                    pass
            
            # Projected gradient with momentum
            if remaining_time() > 10:
                h0 = project_feasible(h0, n)
                c5_cur = compute_c5(h0)
                momentum = np.zeros(n)
                lr = 0.005
                decay = 0.85
                best_pg = c5_cur
                
                for it in range(3000):
                    if remaining_time() < 4:
                        break
                    grad = grad_smooth_max(h0, 200.0)
                    momentum = decay * momentum + (1.0 - decay) * grad
                    gnorm = np.linalg.norm(momentum)
                    if gnorm < 1e-14:
                        break
                    
                    improved = False
                    for sf in [1.0, 0.6, 0.3, 0.15, 0.07, 0.03]:
                        step = lr * sf / max(gnorm, 1e-10)
                        h_trial = h0 - step * momentum
                        h_trial = project_feasible(h_trial, n)
                        c5_trial = compute_c5(h_trial)
                        if c5_trial < c5_cur:
                            h0 = h_trial.copy()
                            c5_cur = c5_trial
                            improved = True
                            break
                    
                    if improved:
                        if c5_cur < best_pg:
                            best_pg = c5_cur
                        lr = min(lr * 1.05, 0.03)
                    else:
                        lr *= 0.6
                        if lr < 1e-7:
                            break
            
            # SLSQP polish
            if remaining_time() > 12:
                h0 = project_feasible(h0, n)
                _g_beta = 800.0
                _g_pen = 1.0
                bounds = [(0.0, 1.0)] * n
                
                def slq_obj(x):
                    dx_v = 2.0 / len(x)
                    corr_v = np.correlate(x, 1.0 - x, mode='full')
                    return smooth_max_val(corr_v, 800.0) * dx_v
                
                def slq_con(x):
                    return np.sum(x) - n / 2.0
                
                cons = [{'type': 'eq', 'fun': slq_con, 'jac': lambda x: np.ones(n)}]
                try:
                    res = minimize(slq_obj, h0, method='SLSQP',
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': 200, 'ftol': 1e-15, 'iprint': 0})
                    h_new = project_feasible(res.x, n)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_cur:
                        h0 = h_new.copy()
                        c5_cur = c5_new
                except Exception:
                    pass
            
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h0.copy()
                best_n = n
                print("  Updated best: C5={:.8f}".format(c5_cur))
        
        # Update parent for next resolution
        if compute_c5(h0) < compute_c5(interp_to(best_h, best_n, n)):
            # h0 is better, use it as new reference
            pass
    
    # Final refinement on best solution
    n = best_n
    _g_n = n
    h = best_h.copy()
    print("\n=== Final refinement n={} ===".format(n))
    
    # High-beta L-BFGS-B
    for beta_val in [2000, 5000]:
        if remaining_time() < 8:
            break
        _g_beta = float(beta_val)
        _g_pen = 10000.0
        try:
            res = fmin_l_bfgs_b(obj_only, h, fprime=None,
                                 pgtol=1e-15, maxiter=200,
                                 maxfun=30000, iprint=0)
            h_new = project_feasible(res, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
                print("  High-beta L-BFGS: C5={:.8f}".format(c5_new))
        except Exception:
            pass
    
    # Projected gradient fine-tuning
    if remaining_time() > 15:
        h = best_h.copy()
        momentum = np.zeros(n)
        lr = 0.001
        decay = 0.9
        for it in range(5000):
            if remaining_time() < 4:
                break
            grad = grad_smooth_max(h, 500.0)
            momentum = decay * momentum + (1.0 - decay) * grad
            gnorm = np.linalg.norm(momentum)
            if gnorm < 1e-15:
                break
            improved = False
            for sf in [1.0, 0.5, 0.25, 0.1, 0.05]:
                step = lr * sf / max(gnorm, 1e-10)
                h_trial = h - step * momentum
                h_trial = project_feasible(h_trial, n)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5:
                    h = h_trial.copy()
                    best_c5 = c5_trial
                    best_h = h.copy()
                    improved = True
                    break
            if improved:
                lr = min(lr * 1.03, 0.01)
            else:
                lr *= 0.5
                if lr < 1e-9:
                    break
    
    # Coordinate descent / bit-flipping
    if remaining_time() > 10:
        h = best_h.copy()
        for sweep in range(30):
            if remaining_time() < 4:
                break
            any_imp = False
            order = rng.permutation(n)
            for i in order:
                if remaining_time() < 2:
                    break
                old_val = h[i]
                for new_val in [0.0, 1.0]:
                    if abs(new_val - old_val) < 1e-7:
                        continue
                    h[i] = new_val
                    h = project_feasible(h, n)
                    c5_test = compute_c5(h)
                    if c5_test < best_c5 - 1e-14:
                        best_c5 = c5_test
                        best_h = h.copy()
                        any_imp = True
                    else:
                        h[i] = old_val
                        h = project_feasible(h, n)
            if not any_imp and sweep > 3:
                break
    
    # Final verification
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    print("\nFINAL: n={}, C5={:.8f}".format(best_n, best_c5))
    return (best_h, best_c5, best_n)
