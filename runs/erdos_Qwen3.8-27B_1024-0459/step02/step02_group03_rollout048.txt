import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_weights(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    return e / e.sum()

def grad_corr(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    weights = smooth_max_weights(corr, beta)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:
            continue
        L = k - (n - 1)
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:n] += w * (1.0 - h[:n-L])
            grad[0:n-L] -= w * h[L:n]
        else:
            Lp = -L
            grad[0:n-Lp] += w * (1.0 - h[Lp:n])
            grad[Lp:n] -= w * h[0:n-Lp]
    return grad

def project_feasible(h, n):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    for _ in range(3):
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt == 0:
            idx = int(np.argmin(np.maximum(h, 1.0 - h)))
            h[idx] += diff
            diff = 0.0
        else:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
            diff = target - np.sum(h)

    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src, n_dst)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def pgd_optimize(h0, n, beta, max_iter, lr=None, momentum=0.85):
    """Projected gradient descent with momentum."""
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    
    if lr is None:
        lr = 0.5 / (beta + 2.0)
    
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -500, 0)
        weights = np.exp(shifted)
        sw = weights.sum()
        if sw < 1e-300:
            weights = np.zeros(len(corr)); weights[np.argmax(corr)] = 1.0
        else:
            weights /= sw
        
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            w = weights[k]
            if w < 1e-15:
                continue
            L = k - (n - 1)

def slsqp_optimize(h0, n, beta, max_iter=200):
    """SLSQP with smooth max."""
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return smooth_max(corr, beta) * (2.0 / n)
    
    def jac(x):
        g = grad_corr(x, beta)
        return g * (2.0 / n)
    
    def con_fun(x):
        return np.sum(x) - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun}]
    res = minimize(obj, h0, jac=jac, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
    h_out = project_feasible(res.x, n)
    return h_out, compute_c5(h_out)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def smooth_max(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_weights(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300:
        idx = int(np.argmax(corr))
        e = np.zeros(len(corr))
        e[idx] = 1.0
        return e
    return e / s

        return project_feasible(h_src, n_dst)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def pgd_optimize(h0, n, beta, max_iter, lr=None, momentum=0.85):
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    if lr is None:
        lr = 1.0 / (2.0 * beta + 4.0)

    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -500, 0)
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300:
            w = np.zeros(len(corr))
            w[int(np.argmax(corr))] = 1.0
        else:
            w = e / s

        grad = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-16:
                continue
            L = k - (n - 1)
            if L == 0:
                grad += wk * (1.0 - 2.0 * h)
            elif L > 0:
                grad[:n-L] += wk * (1.0 - h[L:])
                grad[L:]   -= wk * h[:n-L]
            else:
                Lp = -L
                grad[Lp:]   += wk * (1.0 - h[:n-Lp])
                grad[:n-Lp] -= wk * h[Lp:]
        grad *= 2.0 / n

        v = momentum * v - lr * grad
        h_new = project_feasible(h + v, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        if it % 200 == 199:
            lr *= 0.9
    return best_h, best_c5

def slsqp_optimize(h0, n, beta, max_iter=200):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return smooth_max(corr, beta) * dx
    def jac(x):
        return grad_obj(x, beta)
    def con_fun(x):
        return np.sum(x) - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun}]
    res = minimize(obj, h0, jac=jac, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-15, 'iprint': 0})
    h_out = project_feasible(res.x, n)
    return h_out, compute_c5(h_out)

def make_binary_random(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def make_fourier_init(n, rng, base=None):
    if base is not None:
        h = interp_to_n(base, n)
    else:
        h = np.full(n, 0.5)
    t = np.arange(n) / n
    for k in range(1, 6):
        amp = rng.uniform(0.03, 0.2) / np.sqrt(k)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * k * t + phase)
        h += amp * 0.7 * np.sin(2 * np.pi * k * t + rng.uniform(0, 2 * np.pi))
    return project_feasible(h, n)

def binary_sa(h_start, n, max_iters, rng, T0=0.006, T1=0.0005):
    h = (h_start > 0.5).astype(float)
    n_ones = int(np.sum(h))
    if n_ones > n // 2:
        ones = np.where(h == 1)[0]
        h[rng.choice(ones, size=n_ones - n//2, replace=False)] = 0
    elif n_ones < n // 2:
        zeros = np.where(h == 0)[0]
        h[rng.choice(zeros, size=n//2 - n_ones, replace=False)] = 1

    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    dx = 2.0 / n

    for it in range(max_iters):
        T = T0 * (T1 / T0) ** (it / max(max_iters, 1))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i1 = int(rng.randint(len(ones)))
        i0 = int(rng.randint(len(zeros)))
        p1 = ones[i1]
        p0 = zeros[i0]
        h[p1] = 0.0
        h[p0] = 1.0
        new_c5 = np.max(get_corr(h)) * dx
        delta = new_c5 - cur_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[p1] = 1.0
            h[p0] = 0.0
    return best_h, best_c5

def binary_local_search(h, n, max_rounds, rng):
    h = h.copy()
    cur_c5 = compute_c5(h)
    corr = get_corr(h)
    dx = 2.0 / n
    stall = 0

    for rnd in range(max_rounds):
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for attempt in range(60):
            i1 = int(rng.randint(len(ones)))
            i0 = int(rng.randint(len(zeros)))
            p1 = ones[i1]
            p0 = zeros[i0]
            h[p1] = 0.0
            h[p0] = 1.0
            nc = get_corr(h)
            new_c5 = np.max(nc) * dx
            if new_c5 < cur_c5 - 1e-15:
                cur_c5 = new_c5
                corr = nc
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                improved = True
            else:
                h[p1] = 1.0
                h[p0] = 0.0
        if not improved:
            stall += 1
            if stall >= 8:
                break
        else:
            stall = 0
    return h, cur_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, budget_s - (time.time() - start_time))

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        if parent_n > 0:
            parent_c5 = compute_c5(parent_h)
            best_h = parent_h.copy()
            best_c5 = parent_c5
            best_n = parent_n
            print("Parent: n={}, C5={:.7f}".format(parent_n, parent_c5))
    except Exception:
        pass

    if best_h is None:
        best_n = 200
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_c5 = compute_c5(best_h)
        print("Default: n={}, C5={:.7f}".format(best_n, best_c5))

    # Phase 1: Multi-resolution continuous optimization with correct gradient
    n_list = sorted(set([200, 240, 256, 300, 320, 400, 480, 512]))
    n_list = [n for n in n_list if 64 <= n <= 512]

    for n in n_list:
        if remaining() < 100:
            break
        print("\n=== Phase1 n={} (rem {:.0f}s) ===".format(n, remaining()))

        inits = []
        if parent_h is not None:
            inits.append(interp_to_n(parent_h, n))
        inits.append(interp_to_n(best_h, n))
        inits.append(make_fourier_init(n, rng, base=parent_h))

        for h0 in inits:
            if remaining() < 50:
                break
            h0 = project_feasible(h0, n)
            cur = h0.copy()

            for beta_val in [8, 25, 80, 250, 600]:
                if remaining() < 25:
                    break
                n_it = 400 if beta_val <= 80 else 200
                lr_val = 1.0 / (2.0 * beta_val + 4.0)
                try:
                    h2, c5_2 = pgd_optimize(cur, n, beta_val, n_it, lr=lr_val, momentum=0.88)
                    if c5_2 < best_c5:
                        best_c5 = c5_2
                        best_h = h2.copy()
                        best_n = n
                        print("  PGD b={}, C5={:.7f}".format(beta_val, best_c5))
                    cur = h2
                except Exception:
                    break

            if remaining() > 40 and best_n == n:
                try:
                    h3, c5_3 = slsqp_optimize(cur, n, 150, max_iter=200)
                    if c5_3 < best_c5:
                        best_c5 = c5_3
                        best_h = h3.copy()
                        print("  SLSQP C5={:.7f}".format(best_c5))
                    cur = h3
                except Exception:
                    pass

                if remaining() > 30:
                    try:
                        h4, c5_4 = pgd_optimize(cur, n, 500, 250, lr=0.0008, momentum=0.92)
                        if c5_4 < best_c5:
                            best_c5 = c5_4
                            best_h = h4.copy()
                            print("  Deep C5={:.7f}".format(best_c5))
                    except Exception:
                        pass

        # Perturbation restarts
        if remaining() > 50 and best_n == n:
            for trial in range(3):
                if remaining() < 25:
                    break
                h_p = best_h.copy()
                np_ = int(rng.randint(4, max(5, n // 8)))
                ip = rng.choice(n, size=np_, replace=False)
                h_p[ip] = np.clip(h_p[ip] + rng.uniform(-0.25, 0.25, np_), 0, 1)
                h_p = project_feasible(h_p, n)
                try:
                    h5, c5_5 = pgd_optimize(h_p, n, 80, 250, lr=0.02, momentum=0.85)
                    if c5_5 < best_c5:
                        best_c5 = c5_5
                        best_h = h5.copy()
                        best_n = n
                        print("  Pert C5={:.7f}".format(best_c5))
                except Exception:
                    pass

    # Phase 2: Binary optimization
    print("\n=== Phase2 Binary (rem {:.0f}s) ===".format(remaining()))

    bin_ns = sorted(set([best_n, max(128, best_n - 40), min(600, best_n + 60)]))
    bin_ns = [n for n in bin_ns if n % 2 == 0 and 100 <= n <= 600]

    for n in bin_ns:
        if remaining() < 50:
            break

        h_bin = (interp_to_n(best_h, n) > 0.5).astype(float)
        n_ones = int(np.sum(h_bin))
        if n_ones > n // 2:
            ones = np.where(h_bin == 1)[0]
            h_bin[rng.choice(ones, n_ones - n//2, replace=False)] = 0
        elif n_ones < n // 2:
            zeros = np.where(h_bin == 0)[0]
            h_bin[rng.choice(zeros, n//2 - n_ones, replace=False)] = 1

        n_sa = int(min(60000, max(5000, remaining() * 20 / n)))
        h_sa, c5_sa = binary_sa(h_bin, n, n_sa, rng, T0=0.006, T1=0.0004)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa
            best_n = n
            print("  SA n={}, C5={:.7f}".format(n, best_c5))

        if remaining() > 30:
            h_ls, c5_ls = binary_local_search(h_sa, n, int(min(3000, remaining() * 10 / n)), rng)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls
                best_n = n
                print("  LS n={}, C5={:.7f}".format(n, best_c5))

        # Random binary restart
        if remaining() > 40:
            h_rand = make_binary_random(n, rng)
            n_sa2 = int(min(40000, remaining() * 15 / n))
            h_r2, c5_r2 = binary_sa(h_rand, n, n_sa2, rng, T0=0.01, T1=0.0004)
            if c5_r2 < best_c5:
                best_c5 = c5_r2
                best_h = h_r2
                best_n = n
                print("  SA rand n={}, C5={:.7f}".format(n, best_c5))
            if remaining() > 20:
                h_ls2, c5_ls2 = binary_local_search(h_r2, n, int(min(2000, remaining() * 8 / n)), rng)
                if c5_ls2 < best_c5:
                    best_c5 = c5_ls2
                    best_h = h_ls2
                    best_n = n
                    print("  LS rand C5={:.7f}".format(best_c5))

    # Phase 3: Final polish
    print("\n=== Phase3 Final (rem {:.0f}s) ===".format(remaining()))

    # Binary rounding
    h_round = (best_h > 0.5).astype(float)
    nr = len(h_round)
    n_ones = int(np.sum(h_round))
    n_want = nr // 2
    if n_ones > n_want:
        ones = np.where(h_round == 1)[0]
        order = np.argsort(best_h[ones])[:n_ones - n_want]
        h_round[ones[order]] = 0
    elif n_ones < n_want:
        zeros = np.where(h_round == 0)[0]
        order = np.argsort(-best_h[zeros])[:n_want - n_ones]
        h_round[zeros[order]] = 1
    c5_r = compute_c5(h_round)
    if c5_r < best_c5:
        best_c5 = c5_r
        best_h = h_round
        best_n = nr
        print("  Round C5={:.7f}".format(best_c5))

    # More local search if binary
    if np.all((best_h == 0) | (best_h == 1)):
        if remaining() > 20:
            h_f, c5_f = binary_local_search(best_h, best_n, int(min(8000, remaining() * 20 / best_n)), rng)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f
                print("  Final LS C5={:.7f}".format(best_c5))

    # Continuous re-optimization at best resolution
    if remaining() > 30:
        h_c = interp_to_n(best_h, best_n)
        for bv in [100, 400]:
            if remaining() < 20:
                break
            try:
                h_f2, c5_f2 = pgd_optimize(h_c, best_n, bv, 200, lr=1.0/(2*bv+4), momentum=0.9)
                if c5_f2 < best_c5:
                    best_c5 = c5_f2
                    best_h = h_f2
                    print("  Cont reopt C5={:.7f}".format(best_c5))
                h_c = h_f2
            except Exception:
                pass

    # Final check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.7f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
