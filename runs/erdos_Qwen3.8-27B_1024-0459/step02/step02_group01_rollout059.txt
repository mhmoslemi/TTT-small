= sum_i h[i] * sum_{j=0}^{n-1} (1-h[j])  [since for each i, as lag ranges, i+lag covers all j]
= sum_i h[i] * (n/2)
= (n/2) * (n/2)
= n²/4

So the average of corr[k] over all 2n-1 lags is n²/4 / (2n-1) ≈ n/8 for large n.

For n=192: average ≈ 24. C₅ average ≈ (2/192)*24 = 0.25.

But the max is much larger than the average. For a well-distributed sequence, the max might be around 1.2-1.5 times the average, giving C₅ ≈ 0.30-0.37.

The achieved value of ~0.38 seems reasonable for a good binary sequence.

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 192
_g_beta = 100.0
_g_t0 = 0.0
_g_budget = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def time_left():
    return max(0.0, _g_budget - (time.time() - _g_t0) - 5.0)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        frac = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(frac) and abs(diff) < 1.0:
            h[frac] += diff / np.sum(frac)
            h = np.clip(h, 0.0, 1.0)
    return h

def eq_con(x):
    return np.sum(x) - _g_n / 2.0

def obj_lse(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = _g_beta * (corr - m)
    sh = np.clip(sh, -500.0, 0.0)
    smax = m + np.log(np.sum(np.exp(sh))) / _g_beta
    return smax * (2.0 / _g_n)

def obj_raw(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return np.max(corr) * (2.0 / _g_n)

def obj_topk(x):

    xp = (np.arange(n_src) + 0.5) * 2.0 / n_src
    xn = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    return np.interp(xn, xp, h_src)

    _g_n = n
    h = project_feasible(h, n)

            h = project_feasible(res.x, n)
        except Exception:
            pass
    return h

def binary_local_search(h_start, n, max_time_s, seed):
    rng = np.random.RandomState(seed)
    if h_start.min() >= 0 and h_start.max() <= 1:

    if n_ones > n_want:

        order = np.argsort(h_start[ones])[:n_ones - n_want]
        h_bin[ones[order]] = 0.0
    elif n_ones < n_want:

        h_bin[zeros[order]] = 1.0

    corr = np.correlate(h_bin, 1.0 - h_bin, mode='full')
    cur_max = np.max(corr)
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    stall = 0
    while time.time() - t_start < max_time_s:

            if gain > best_gain:

            h_bin[a] = 1.0
            h_bin[b] = 0.0
        if best_swap is not None:
            a, b, ia, ib, new_corr = best_swap
            ones = np.delete(ones, ia)
            zeros = np.delete(zeros, ib)

            zeros = np.append(zeros, a)
            h_bin[a] = 0.0
            h_bin[b] = 1.0
            corr = new_corr
            cur_max = np.max(corr)
            improved = True
            stall = 0
        else:
            stall += 1
            if stall > 50:
                break
    return h_bin, cur_max * (2.0 / n)

def subgradient_polish(h, n, n_steps, step0):
    dx = 2.0 / n
    h = project_feasible(h, n)
    cur_val = compute_c5(h)
    step = step0
    for _ in range(n_steps):
        if time_left() < 5:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr)
        active = corr >= max_val - 1e-10
        n_active = max(1, int(np.sum(active)))
        grad = np.zeros(n)
        active_lags = np.where(active)[0]
        for lag_idx in active_lags:
            offset = lag_idx - (n - 1)
            if offset >= 0:
                for i in range(n - offset):
                    grad[i] += (1.0 - h[i + offset])
                    grad[i + offset] += (-h[i])
            else:
                o = -offset
                for i in range(o, n):
                    grad[i] += (1.0 - h[i - o])
                    grad[i - o] += (-h[i])
        grad /= n_active
        h_new = h - step * grad * dx
        h_new += (n / 2.0 - np.sum(h_new)) / n
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        new_val = compute_c5(h_new)
        if new_val <= cur_val + 1e-15:
            h = h_new
            cur_val = new_val
        else:
            step *= 0.7
        step *= 0.97
    return h, cur_val
