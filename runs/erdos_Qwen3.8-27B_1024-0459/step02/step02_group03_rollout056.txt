def projected_gradient(h, grad, step, momentum_buf, mu):
    # Heavy-ball momentum
    momentum_buf = mu * momentum_buf - step * grad
    h_new = h + momentum_buf
    h_new = project_feasible(h_new, len(h))
    return h_new, momentum_buf

def project_feasible(h, n):
    # Project onto {0≤h≤1, sum(h)=n/2}
    # Use bisection on the Lagrange multiplier for the sum constraint
    target = n / 2.0
    # Binary search for lambda such that sum(clip(h - lambda, 0, 1)) = target
    lo, hi = h.min() - 2, h.max() + 2
    for _ in range(60):
        mid = (lo + hi) / 2
        val = np.clip(h - mid, 0, 1)
        s = val.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)

        step_t = step / (1.0 + 0.005 * it)
        mob = mu * mob - step_t * grad
        h_new = project_feasible(h + mob, n)
        c5_new = compute_c5(h_new)
        if c5_new <= best_c5 + 1e-14:
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            mob = 0.5 * mob
            if it % 100 == 99:
                step *= 0.8
    return best_h, best_c5

def binary_local_search(h_bin, n, rng, max_rounds):
    h = h_bin.copy()
    c5 = compute_c5(h)
    t_start = time.time()
    for r in range(max_rounds):
        if time.time() - t_start > 40:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_k = int(np.argmax(corr))
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        n_tries = min(40, len(ones) * len(zeros))
        for _ in range(n_tries):
            i1 = rng.randint(len(ones))
            i0 = rng.randint(len(zeros))
            p1, p0 = int(ones[i1]), int(zeros[i0])
            h[p1] = 0.0
            h[p0] = 1.0
            new_c5 = compute_c5(h)
            if new_c5 < c5 - 1e-15:
                c5 = new_c5
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                improved = True
            else:
                h[p1] = 1.0
                h[p0] = 0.0
        if not improved:
            break
    return h, c5

def round_to_binary(h, n):
    h_b = (h > 0.5).astype(float)
    n_ones = int(h_b.sum())
    n_want = n // 2
    if n_ones > n_want:
        ones_idx = np.where(h_b == 1.0)[0]
        order = np.argsort(h[ones_idx])[:n_ones - n_want]
        h_b[ones_idx[order]] = 0.0
    elif n_ones < n_want:
        zeros_idx = np.where(h_b == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:n_want - n_ones]
        h_b[zeros_idx[order]] = 1.0
    return h_b

def make_fourier_init(n, rng, n_harmonics):
    h = np.full(n, 0.5)
    for k in range(1, n_harmonics + 1):
        amp = rng.uniform(0.05, 0.3) / k
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * k * np.arange(n) / n + phase)
    h = np.clip(h, 0, 1)
    return project_feasible(h, n)

def make_wave_init(n, rng):
    breakpoints = np.sort(rng.choice(1, n - 2, size=rng.randint(4, 12)))
    h = np.zeros(n)
    heights = rng.uniform(0.3, 0.7, len(breakpoints) + 1)
    arr = np.concatenate([[0], breakpoints, [n]])
    heights = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(arr)), heights)
    return project_feasible(np.clip(heights, 0, 1), n)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - start_time

    def remaining():
        return max(0.0, budget_s - elapsed() - 5.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        if parent_n > 0:
            parent_c5 = compute_c5(parent_h)
            best_h = parent_h.copy()
            best_c5 = parent_c5
            best_n = parent_n
            print("Parent: n={}, C5={:.6f}".format(parent_n, parent_c5))
    except Exception:
        pass

    if best_h is None:
        best_n = 200
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_c5 = compute_c5(best_h)
        print("Default: C5={:.6f}".format(best_c5))

    # Determine resolutions to try
    base_n = best_n if best_n > 0 else 200
    n_list = sorted(set([base_n, 300, 400, 500, 600]))
    n_list = [n for n in n_list if 64 <= n <= 700]

    for n in n_list:
        if remaining() < 50:
            break
        print("=== n={} (remaining {:.0f}s) ===".format(n, remaining()))

        # Generate initializations
        inits = []
        if parent_h is not None and parent_h is not best_h:
            inits.append(interp_to_n(parent_h, n))
        inits.append(interp_to_n(best_h, n))
        inits.append(make_fourier_init(n, rng, 6))
        inits.append(make_wave_init(n, rng))
        h_rand = np.zeros(n)
        h_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        inits.append(h_rand)

        for init_idx, h0 in enumerate(inits):
            if remaining() < 35:
                break
            print("  Init {} (n={}): C5={:.6f}".format(init_idx, n, compute_c5(h0)))

            # Phase 1: Projected gradient with increasing beta
            if remaining() > 80:
                time_alloc = min(60, remaining() * 0.4)
                h_pg, c5_pg = projected_gradient_opt(
                    h0, n,
                    betas=[30, 100, 300],
                    iters_list=[4000, 3000, 2000],
                    mu=0.82,
                    step0=2.0 / n,
                    time_limit=time_alloc
                )
                if c5_pg < best_c5:
                    best_c5 = c5_pg
                    best_h = h_pg.copy()
                    best_n = n
                    print("    PG: C5={:.6f}".format(c5_pg))
                h0 = h_pg

            # Phase 2: Top-K focused
            if remaining() > 50:
                time_alloc = min(40, remaining() * 0.3)
                h_tk, c5_tk = topk_focused_opt(h0, n, K=6, iters=5000, time_limit=time_alloc)
                if c5_tk < best_c5:
                    best_c5 = c5_tk
                    best_h = h_tk.copy()
                    best_n = n
                    print("    TopK: C5={:.6f}".format(c5_tk))
                h0 = h_tk

            # Phase 3: SLSQP (only for moderate n)
            if remaining() > 40 and n <= 350:
                try:
                    h_sl, c5_sl = slsqp_opt(h0, n, beta=300, maxiter=200)
                    if c5_sl < best_c5:
                        best_c5 = c5_sl
                        best_h = h_sl.copy()
                        best_n = n
                        print("    SLSQP: C5={:.6f}".format(c5_sl))
                        h0 = h_sl
                except Exception:
                    pass

            # Phase 4: Binary rounding + local search
            if remaining() > 25:
                h_b = round_to_binary(best_h, n)
                h_bl, c5_bl = binary_local_search(h_b, n, rng, max_rounds=8000)
                if c5_bl < best_c5:
                    best_c5 = c5_bl
                    best_h = h_bl.copy()
                    best_n = n
                    print("    Binary: C5={:.6f}".format(c5_bl))

    # Final refinement on best solution
    if remaining() > 30:
        n_f = best_n
        if n_f < 400:
            n_f = 400
        h_f = interp_to_n(best_h, n_f)
        h_f, c5_f = projected_gradient_opt(
            h_f, n_f, betas=[200, 500],
            iters_list=[3000, 2000],
            mu=0.85, step0=2.0 / n_f,
            time_limit=min(40, remaining() * 0.5)
        )
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
            best_n = n_f
            print("  Final refine: C5={:.6f}".format(c5_f))

    if remaining() > 20:
        h_b = round_to_binary(best_h, best_n)
        h_bl, c5_bl = binary_local_search(h_b, best_n, rng, max_rounds=10000)
        if c5_bl < best_c5:
            best_c5 = c5_bl
            best_h = h_bl
            best_n = best_n
            print("  Final binary: C5={:.6f}".format(c5_bl))

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
