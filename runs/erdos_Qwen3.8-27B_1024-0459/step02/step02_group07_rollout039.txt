import numpy as np
from scipy.optimize import linprog, minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def solve_lp_fix_g(g, n):
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    n_rows = 2 * n - 1
    A_ub = np.zeros((n_rows, n + 1))
    b_ub = np.zeros(n_rows)
    for idx, k in enumerate(range(-(n - 1), n)):
        if k >= 0:
            for j in range(k, n):
                A_ub[idx, j] = g[j - k]
        else:
            m = -k
            for i in range(0, n - m):
                A_ub[idx, i] = g[i + m]
        A_ub[idx, -1] = -1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'maxiter': 200000, 'dual_feasibility_tolerance': 1e-10})
        if res.success:
            return np.clip(res.x[:n], 0.0, 1.0), res.x[-1]

        pass
    return None, None

def alternating_lp(h_init, n, max_iter=5000, time_limit=300):
    h = make_feasible(h_init.copy(), n)
    g = 1.0 - h
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    stall = 0
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        if h_new is None:

    h_interp = np.interp(x_new, x_old, h_src)
    return make_feasible(h_interp, n_dst)

def coord_opt_j(h, j, n, corr, ells):
    idx1 = j - ells
    idx2 = j + ells
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    a = np.zeros(2 * n - 1)
    if np.any(valid1):
        a[valid1] += 1.0 - h[idx1[valid1]]
    if np.any(valid2):
        a[valid2] += -h[idx2[valid2]]
    b = corr - h[j] * a
    lo = 0.0

        vals = a * mid + b

        if sg > 1e-13:
            hi = mid
        elif sg < -1e-13:
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)

    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    corr = get_corr(h)

    start = time.time()

            break
        for j in range(n):
            t_opt = coord_opt_j(h, j, n, corr, ells)
            h[j] = t_opt
        h = make_feasible(h, n)
        corr = get_corr(h)
        c5 = np.max(corr) * (2.0 / n)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5


def slsqp_polish(h, n, max_iter=3000):
    def c5_obj(x):
        return np.max(np.correlate(x, 1.0 - x, mode='full'))
    def sum_con(x):
        return np.sum(x) - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        res = minimize(c5_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-18})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None


def subgradient_polish(h, n, time_limit):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    alpha = 0.05
    start = time.time()
    ells_all = np.arange(-(n - 1), n)
    while time.time() - start < time_limit:
        corr = get_corr(h)
        top_k = max(3, int(0.05 * len(corr)))
        thresh = np.sort(corr)[-top_k]
        if len(thresh) > 1:
            active_thresh = thresh[0] - 1e-12
        else:
            active_thresh = thresh[0] - 1e-12
        active_idx = np.where(corr >= active_thresh)[0]
        active_ells = ells_all[active_idx]
        grad = np.zeros(n)
        for ell in active_ells:
            e = int(ell)
            js1 = max(0, e)
            je1 = min(n, n + e)
            if js1 < je1:
                grad[js1:je1] += (1.0 - h[js1 - e:je1 - e])
            js2 = max(0, -e)
            je2 = min(n, n - e)
            if js2 < je2:
                grad[js2:je2] += -h[js2 + e:je2 + e]
        grad /= max(len(active_ells), 1)
        h = h - alpha * grad
        h = np.clip(h, 0.0, 1.0)
        h = make_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
        alpha *= 0.999
        if alpha < 0.001:
            break
    return best_h, best_c5
