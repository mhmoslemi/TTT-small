def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_corr(h):
    """Returns the full correlation vector and the C5 value."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr, np.max(corr) * dx

def make_feasible(h, n):

            if cnt == 0:

            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)

            cnt = np.count_nonzero(mask)
            if cnt == 0:

def compute_B(h, i, n):
    """Compute B_k for all k, where corr[k] = A_k + h[i] * B_k."""
    B = np.zeros(2*n-1)
    # offset_k = k - (n-1), ranges from -(n-1) to (n-1)
    offsets = np.arange(-(n-1), n)  # length 2n-1
    
    # Term 1: (1-h[i+offset]) valid if 0 <= i+offset < n
    # i+offset in [0, n-1] => offset in [-i, n-1-i]
    valid1 = (offsets >= -i) & (offsets <= n-1-i)
    # h[i+offset] for valid offsets
    idx1 = i + offsets[valid1]
    B[valid1] += (1.0 - h[idx1])
    
    # Term 2: -h[i-offset] valid if 0 <= i-offset < n
    # i-offset in [0, n-1] => offset in [i-(n-1), i]
    valid2 = (offsets >= i-(n-1)) & (offsets <= i)
    idx2 = i - offsets[valid2]
    B[valid2] -= h[idx2]
    
    return B

2. When j+k-(n-1) = i, i.e., j = i-k+(n-1): h[i-k+(n-1)] * (1-h[i]) [valid if 0 ≤ i-k+(n-1) < n]

- Term 1 exists if 0 ≤ i+offset < n, contributes h[i]*(1-h[i+offset])

import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(3000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def coord_descent_fast(h, n, n_sweeps, budget, rng, repair_every=5):

    dx = 2.0 / n

    for sweep in range(n_sweeps):
        if time.time() - start > budget:
            break
        
        order = rng.permutation(n)
        improved = 0
        
        for i in order:
            B = np.zeros(2 * n - 1)
            
            s1 = max(0, (n-1) - i)
            e1 = min(2*n-2, (n-1) + (n-1-i))
            if s1 <= e1:
                off_slice = offsets[s1:e1+1]
                idx = i + off_slice
                B[s1:e1+1] += (1.0 - h[idx])
            
            s2 = max(0, (n-1) + (i-(n-1)))

def pair_coord_descent(h, n, n_iterations, budget, rng):
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best_c5 = np.max(corr) * dx

    for it in range(n_iterations):
        if time.time() - start > budget:
            break
        
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        Bi = np.zeros(2 * n - 1)
        s1 = max(0, (n-1) - i)
        e1 = min(2*n-2, (n-1) + (n-1-i))
        if s1 <= e1:
            off_s = offsets[s1:e1+1]
            Bi[s1:e1+1] += (1.0 - h[i + off_s])
        s2 = max(0, (n-1) + (i-(n-1)))
        e2 = min(2*n-2, (n-1) + i)
        if s2 <= e2:
            off_s2 = offsets[s2:e2+1]
            Bi[s2:e2+1] -= h[i - off_s2]
        
        Bj = np.zeros(2 * n - 1)
        s1j = max(0, (n-1) - j)
        e1j = min(2*n-2, (n-1) + (n-1-j))
        if s1j <= e1j:
            off_sj = offsets[s1j:e1j+1]
            Bj[s1j:e1j+1] += (1.0 - h[j + off_sj])
        s2j = max(0, (n-1) + (j-(n-1)))
        e2j = min(2*n-2, (n-1) + j)
        if s2j <= e2j:
            off_s2j = offsets[s2j:e2j+1]
            Bj[s2j:e2j+1] -= h[j - off_s2j]
        
        # h[i] -= delta, h[j] += delta (or vice versa)
        # corr += -delta * Bi + delta * Bj = delta * (Bj - Bi)
        D = Bj - Bi
        
        # f(delta) = max_k (corr[k] + delta * D[k])
        # delta is constrained by 0 <= h[i]-delta <= 1 and 0 <= h[j]+delta <= 1
        # i.e., h[i]-1 <= delta <= h[i] and -h[j] <= delta <= 1-h[j]
        lo = max(h[i] - 1.0, -h[j])
        hi = min(h[i], 1.0 - h[j])
        
        if hi <= lo + 1e-14:
            continue
        
        # Golden section on f(delta) = max(corr + delta * D) over [lo, hi]
        lo_g, hi_g = lo, hi
        golden = 0.618
        for _ in range(25):
            m1 = lo_g + (1.0 - golden) * (hi_g - lo_g)
            m2 = lo_g + golden * (hi_g - lo_g)
            if np.max(corr + m1 * D) <= np.max(corr + m2 * D):
                hi_g = m2
            else:
                lo_g = m1
        
        best_delta = (lo_g + hi_g) * 0.5
        best_val = np.max(corr + best_delta * D)
        
        # Check endpoints too
        for d_check in [lo, hi, 0.0]:
            v = np.max(corr + d_check * D)
            if v < best_val:
                best_val = v
                best_delta = d_check
        
        if abs(best_delta) > 1e-12:
            h[i] -= best_delta
            h[j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            corr += best_delta * D
            cur_c5 = np.max(corr) * dx
            if cur_c5 < best_c5:

def slsqp_refine(h, n, max_iter=3000):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        cmax = np.max(corr)
        # Smooth max for better gradient
        beta = 500.0
        shifted = beta * (corr - cmax)
        shifted = np.clip(shifted, -400, 0)
        return cmax + np.log(np.sum(np.exp(shifted))) / beta
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-15, 'iprint': 0})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None

def structured_init(n, rng):
    inits = []
    
    # Triangle wave
    h_tri = np.zeros(n)
    for i in range(n):
        t = (2.0 * i / n) % 1.0
        h_tri[i] = 1.0 - abs(2.0 * t - 1.0)
    inits.append(make_feasible(h_tri, n))
    
    # Sawtooth
    h_saw = np.zeros(n)
    for i in range(n):
        h_saw[i] = (2.0 * i / n) % 1.0
    inits.append(make_feasible(h_saw, n))
    
    # Quadratic residue pattern
    if n % 2 == 0:
        h_qr = np.zeros(n)
        for i in range(n):
            h_qr[i] = 0.5 + 0.5 * np.cos(2.0 * np.pi * i * i / n)
        inits.append(make_feasible(h_qr, n))
    
    # Block pattern [1,1,0,0]
    if n % 4 == 0:
        h_blk = np.tile([1.0, 1.0, 0.0, 0.0], n // 4)
        inits.append(h_blk)
    
    # Random
    h_rand = rng.rand(n)
    inits.append(make_feasible(h_rand, n))
    
    return inits

            best_c5 = c5_opt
            best_h = h_opt
    
    # Phase 8: Final SLSQP polish
    if remaining() > 20:
        h_opt, c5_opt = slsqp_refine(best_h, best_n, max_iter=2000)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Phase 9: Final pair CD
    if remaining() > 15:
        n_iters2 = min(100000, int(remaining() * 60))
        h_opt, c5_opt = pair_coord_descent(best_h, best_n, n_iters2, remaining() * 0.6, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Final feasibility
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    
    assert np.all(best_h >= -1e-10) and np.all(best_h <= 1.0 + 1e-10)
    assert abs(np.sum(best_h) - best_n / 2.0) < 1e-6
    
    return (best_h, best_c5, best_n)
