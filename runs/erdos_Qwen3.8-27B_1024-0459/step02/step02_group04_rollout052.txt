    high = h[L:]
    ac = float(np.dot(low, high))
    if l > 0:
        return float(np.sum(low) - ac)
    return float(np.sum(high) - ac)

    else:

        delta = rng.randn(L) * step
        delta -= delta.mean()
        h_try = cur_h.copy()
        h_try[idx] += delta
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try)

        if c_try < best_c - 1e-15:
            best_h = h_try.copy()
            best_c = c_try
            cur_h = h_try
            cur_c = c_try
            temp = min(0.015, max(temp * 1.2, min_temp))
        elif c_try < cur_c - 1e-15:
            cur_h = h_try
            cur_c = c_try
        else:
            d = c_try - cur_c
            if d < temp:
                prob = np.exp(-d / temp)
                if rng.rand() < prob:
                    cur_h = h_try
                    cur_c = c_try
        temp = max(temp * 0.998, min_temp)

    return best_h, float(best_c)


def targeted_local_search(h0, n, time_limit, seed):
    n = int(n)
    rng = np.random.RandomState(int(seed))
    h = project_feasible(h0, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c

    vals, lags = all_lag_values_lags(h)
    top_lags = lags[np.argsort(-vals)[:8]]

    max_rounds = 500
    step_sizes = [0.04, 0.02, 0.01, 0.005, 0.002]

    for rnd in range(max_rounds):
        if time.time() > time_limit:
            break
        step = step_sizes[min(rnd // 100, len(step_sizes) - 1)]
        improved = False
        for ll in top_lags:
            if time.time() > time_limit:
                break
            g = lag_gradient_only(h, int(ll))
            dn = -g * step
            h_try = h.copy()
            h_try += dn
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try)
            if c_try < best_c - 1e-15:
                best_h = h_try.copy()
                best_c = c_try
                h = h_try
                improved = True
        if improved:
            vals, lags = all_lag_values_lags(h)
            top_lags = lags[np.argsort(-vals)[:8]]
        elif rnd > 100:
            break

    return best_h, float(best_c)


def multi_scale_refine(h0, n, time_limit, seed):
    n = int(n)
    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h)

    betas = [30, 100, 300, 800]
    for beta in betas:
        if time.time() > time_limit - 10.0:
            break
        maxit = 50 if n <= 200 else 40
        h2, c2 = optimize_smooth_h(best_h, n, beta, maxiter=maxit, ftol=1e-12)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = c2

    if time.time() < time_limit - 15.0:
        if n <= 200:
            seq = (24, 48, 96, 192)
            npass = 6
        elif n <= 280:
            seq = (16, 32, 64, 128)
            npass = 5
        else:
            seq = (12, 24, 48, 96)
            npass = 4
        h2, c2 = refine_active(best_h, n, time_limit, active_seq=seq, maxiter=70, n_passes=npass)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    if time.time() < time_limit - 10.0:
        h2, c2 = targeted_local_search(best_h, n, time_limit, seed)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    if time.time() < time_limit - 5.0:
        h2, c2 = local_block_perturb(best_h, n, time_limit, seed + 777, step=0.02)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h)
    return best_h, float(best_c)


def try_resolution(h_parent, n_parent, n_target, time_limit, seed):
    n_t = int(n_target)
    if n_t == n_parent:
        h = project_feasible(h_parent, n_t)
    else:
        h = interpolate_to_n(h_parent, n_t)
        h = project_feasible(h, n_t)
    c = compute_c5(h)
    h, c = multi_scale_refine(h, n_t, time_limit, seed)
    return h, c, n_t


def small_n_search(n_val, time_limit, seed):
    n_val = int(n_val)
    if n_val < 8:
        return None, np.inf, n_val
    rng = np.random.RandomState(seed)
    best_h = None
    best_c = np.inf

    starters = []
    h = np.zeros(n_val)
    q = max(1, n_val // 4)
    h[:q] = 1.0
    h[n_val // 2:n_val // 2 + q] = 1.0
    starters.append(project_feasible(h, n_val))

    h = np.zeros(n_val)
    if n_val >= 4:
        pat = np.array([1.0, 1.0, 0.0, 0.0])
        h = np.tile(pat, n_val // 4)[:n_val]
    starters.append(project_feasible(h, n_val))

    x = np.linspace(0.0, 2.0, n_val, endpoint=False)
    starters.append(project_feasible(0.5 + 0.3 * np.sin(2 * np.pi * x), n_val))
    starters.append(project_feasible(rng.rand(n_val) * 0.8 + 0.1, n_val))

    for i, hs in enumerate(starters):
        if time.time() > time_limit:
            break
        per = (time_limit - time.time()) / max(1, len(starters) - i)
        per = min(25.0, per)
        h2, c2 = multi_scale_refine(hs, n_val, time.time() + per, seed + 100 * n_val + i)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    if best_h is None:
        best_h = project_feasible(np.full(n_val, 0.5), n_val)
        best_c = compute_c5(best_h)
    return best_h, best_c, n_val


def run(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    budget_s = float(budget_s)

    start_time = time.time()
    hard_end = start_time + max(60.0, budget_s - 30.0)

    parent = get_parent_values()
    if parent is not None:
        parent_n = int(parent.size)
        raw_h = parent.copy()
        if (float(np.min(raw_h)) >= -1e-12 and float(np.max(raw_h)) <= 1.0 + 1e-12
                and abs(float(np.sum(raw_h)) - 0.5 * parent_n) <= 1e-8):
            best_h = raw_h.astype(float).copy()
            best_c = compute_c5(best_h)
        else:
            best_h = project_feasible(parent, parent_n)
            best_c = compute_c5(best_h)
        best_n = parent_n
    else:
        best_n = 160
        best_h = project_feasible(np.full(best_n, 0.5), best_n)
        best_c = compute_c5(best_h)

    # Phase 1: Small-n searches for good starting patterns (quick)
    small_end = min(hard_end - 300.0, time.time() + 40.0)
    low_sols = []
    for ln in (16, 24, 32):
        if time.time() > small_end:
            break
        hh, cc, nn = small_n_search(ln, time.time() + 12.0, seed + ln * 31)
        if cc < 0.6:
            low_sols.append((hh.copy(), cc, nn))

    # Phase 2: Primary optimization at parent resolution
    if time.time() < hard_end - 200.0:
        slot = min(250.0, hard_end - 200.0 - time.time())
        h2, c2, n2 = try_resolution(best_h, best_n, best_n, time.time() + slot, seed + 1)
        if c2 < best_c:
            best_h = h2
            best_c = c2
            best_n = n2

    # Phase 3: Try higher resolutions by interpolation
    higher_ns = []
    for cand_n in (best_n + 24, best_n + 32, best_n + 48, 192, 224, 240, 288):
        if 18 <= cand_n <= 320 and cand_n != best_n and cand_n not in higher_ns:
            higher_ns.append(cand_n)
    higher_ns = higher_ns[:3]

    for nn in higher_ns:
        if time.time() > hard_end - 150.0:
            break
        slot = min(180.0, hard_end - 150.0 - time.time())
        if slot < 40.0:
            break
        h2, c2, n2 = try_resolution(best_h, best_n, nn, time.time() + slot, seed + 1000 + nn)
        if c2 < best_c:
            best_h = h2
            best_c = c2
            best_n = n2

    # Phase 4: Re-optimize at best resolution with more aggressive settings
    if time.time() < hard_end - 120.0:
        slot = min(200.0, hard_end - 120.0 - time.time())
        h2, c2, n2 = try_resolution(best_h, best_n, best_n, time.time() + slot, seed + 5000)
        if c2 < best_c:
            best_h = h2
            best_c = c2
            best_n = n2

    # Phase 5: Try a few more resolutions
    for nn in (best_n - 16, best_n + 16, best_n - 8, best_n + 8):
        if time.time() > hard_end - 80.0:
            break
        if nn < 20 or nn > 320 or nn in (best_n,):
            continue
        slot = min(100.0, hard_end - 80.0 - time.time())
        if slot < 30.0:
            break
        h2, c2, n2 = try_resolution(best_h, best_n, nn, time.time() + slot, seed + 9000 + nn)
        if c2 < best_c:
            best_h = h2
            best_c = c2
            best_n = n2

    # Phase 6: Final intensive refinement at best resolution
    if time.time() < hard_end - 30.0:
        remaining = hard_end - 30.0 - time.time()
        h2, c2, n2 = try_resolution(best_h, best_n, best_n, time.time() + remaining, seed + 77777)
        if c2 < best_c:
            best_h = h2
            best_c = c2
            best_n = n2

    # Final projection and verification
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)

    if time.time() > hard_end:
        pass

    return (best_h, float(best_c), int(best_n))
