def _ga_binary(h_init, n, time_limit, seed, pop_size=60, generations=None):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    if generations is None:
        generations = 300

    def fitness(hv):
        return _compute_c5(hv)

    # Initialize population
    pop = np.zeros((pop_size, n))
    pop[0] = h_init
    for i in range(1, pop_size):
        pert = h_init + 0.15 * (rng.rand(n) * 2 - 1)
        pop[i] = _to_binary(pert, n)

    best_h = h_init.copy()
    best_fit = fitness(h_init)
    for i in range(pop_size):
        f = fitness(pop[i])
        if f < best_fit:
            best_fit = f
            best_h = pop[i].copy()

    ts = 4  # tournament size
    cx_prob = 0.8
    mut_prob = 0.15

    for gen in range(generations):
        if time.time() - t0 > time_limit - 2:
            break

        new_pop = np.zeros((pop_size, n))
        # Elitism: copy best
        new_pop[0] = best_h

        for i in range(1, pop_size):
            # Tournament selection
            candidates = rng.randint(0, pop_size, ts)
            fits = np.array([fitness(pop[c]) for c in candidates])
            winner = pop[candidates[np.argmin(fits)]]

            if rng.rand() < cx_prob and i + 1 < pop_size:
                # Second parent
                candidates2 = rng.randint(0, pop_size, ts)
                fits2 = np.array([fitness(pop[c]) for c in candidates2])
                winner2 = pop[candidates2[np.argmin(fits2)]]
                # Uniform crossover
                mask = rng.rand(n) > 0.5
                child = np.where(mask, winner, winner2)
            else:
                child = winner.copy()

            # Mutation: flip some bits

def _continuous_refine(h_init, n, time_limit, seed):
    """SLSQP-based continuous refinement with smooth max."""
    globals()['_g_n_ref'] = n
    globals()['_g_beta_ref'] = 500.0

    bounds = [(0.0, 1.0)] * n

    def _eq_h(x):
        return np.sum(x) - n / 2.0

    def _obj_smooth(x):
        c = _compute_corr(x)
        m = np.max(c)
        beta = globals()['_g_beta_ref']
        sh = beta * (c - m)
        sh = np.clip(sh, -500.0, 0.0)
        smax = m + np.log(np.sum(np.exp(sh))) / beta
        return smax * (2.0 / n)

    def _obj_raw(x):
        return _compute_c5(x)

    h_cur = _project_feasible(h_init, n)
    cons = [{'type': 'eq', 'fun': _eq_h}]
    t0 = time.time()

    for beta in [50.0, 200.0, 1000.0, 5000.0]:
        if time.time() - t0 > time_limit - 3:
            break
        globals()['_g_beta_ref'] = beta
        maxit = int(max(50, (time_limit - (time.time() - t0)) * 5))
        try:

                           bounds=bounds, constraints=cons,

def _subgradient_polish(h, n, time_limit, seed):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _compute_c5(h)
    t0 = time.time()
    step = 0.01

    while time.time() - t0 < time_limit - 1:
        corr = _compute_corr(h)
        max_val = np.max(corr)
        active = corr >= max_val - 1e-8
        n_active = max(1, np.sum(active))

        grad = np.zeros(n)
        active_idx = np.where(active)[0]
        for lag_idx in active_idx:
            offset = int(lag_idx) - (n - 1)
            if offset >= 0:
                for i in range(n - offset):
                    grad[i] += (1.0 - h[i + offset])
                    grad[i + offset] += (-h[i])
            else:
                o = -offset
                for i in range(o, n):
                    grad[i] += (1.0 - h[i - o])
                    grad[i - o] += (-h[i])

        grad /= n_active
        h_new = h - step * grad
        h_new = h_new + (n / 2.0 - np.sum(h_new)) / n
        h_new = np.clip(h_new, 0.0, 1.0)

            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

            best_h = h_sg.copy()

    # Phase 5: Final SLSQP polish
    if _time_left() > 20:
        globals()['_g_n_ref'] = best_n
        globals()['_g_beta_ref'] = 10000.0
        bounds = [(0.0, 1.0)] * best_n

        def _eq_final(x):
            return np.sum(x) - best_n / 2.0

        def _obj_fin(x):
            c = _compute_corr(x)
            m = np.max(c)
            beta = 10000.0
            sh = beta * (c - m)
            sh = np.clip(sh, -500.0, 0.0)
            smax = m + np.log(np.sum(np.exp(sh))) / beta
            return smax * (2.0 / best_n)

        h_fin = _project_feasible(best_h, best_n)
        cons = [{'type': 'eq', 'fun': _eq_final}]
        try:
            res = minimize(_obj_fin, h_fin, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-16, 'iprint': 0})
            h_fin = _project_feasible(res.x, best_n)
            c5_fin = _compute_c5(h_fin)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()
        except Exception:
            pass

    # Phase 6: Binary rounding check
    if _time_quick := _time_left() > 5:
        h_b = _to_binary(best_h, best_n)
        c5_b = _compute_c5(h_b)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Final validation
    best_h = _project_feasible(best_h, best_n)
    final_c5 = _compute_c5(best_h)

    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
