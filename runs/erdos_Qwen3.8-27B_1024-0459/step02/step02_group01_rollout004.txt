def ensure_feasible(h, target):
    h = np.array(h, dtype=float, copy=True)
    if h.size == 0:
        return h
    np.clip(h, 0.0, 1.0, out=h)
    s = float(np.sum(h))
    diff = target - s
    if diff > 1e-15:
        idx = np.argsort(1.0 - h)
        rem = diff
        for i in idx:
            if rem <= 1e-15:
                break
            cap = 1.0 - h[i]
            if cap <= 0.0:
                continue
            add = cap if cap < rem else rem
            h[i] += add
            rem -= add
        if rem > 1e-12:
            # distribute to all (shouldn't happen)
            h += rem / h.size
            np.clip(h,0.0,1.0,out=h)
    elif diff < -1e-15:
        idx = np.argsort(h)
        rem = -diff
        for i in idx:
            if rem <= 1e-15:
                break
            sub = h[i]
            if sub <= 0.0:
                continue
            take = sub if sub < rem else rem
            h[i] -= take
            rem -= take
        if rem > 1e-12:
            h -= rem / h.size
            np.clip(h,0.0,1.0,out=h)

    s = float(np.sum(h))

import numpy as np
import time

Perturb_Scales = (0.03, 0.06, 0.10, 0.15, 0.20, 0.30, 0.35, 0.40, 0.50, 0.60)


def exact_c5(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n == 0:
        return 1.0
    b = 1.0 - h
    c = np.correlate(h, b, mode="full")
    return float(np.max(c) * (2.0 / float(n)))


def solution_c5(h):
    c = exact_c5(h)
    try:
        val = evaluate_erdos_solution(h)
    except Exception:
        return c
    if isinstance(val, (int, float, np.floating)):
        vf = float(val)
        if vf < c:
            c = vf
    return c


def full_corr_rev(h):
    h = np.asarray(h, dtype=float)
    c = np.correlate(h, 1.0 - h, mode="full")
    return c[::-1].astype(float, copy=True)


def valid_stage_n(n, allow_odd=False):
    if n < 64 or n > 1200:
        return False
    if not allow_odd and (n % 2 != 0):
        return False
    return True


def ensure_feasible(h, target):
    h = np.array(h, dtype=float, copy=True)
    if h.size == 0:
        return h

    np.clip(h, 0.0, 1.0, out=h)
    s = float(np.sum(h))
    diff = target - s

    if diff > 1e-15:
        idx = np.argsort(1.0 - h)
        rem = diff
        for i in idx:
            if rem <= 1e-15:
                break
            cap = 1.0 - h[i]
            if cap <= 0.0:
                continue
            add = cap if cap < rem else rem
            h[i] += add
            rem -= add
        if rem > 1e-12:
            h += rem / h.size
            np.clip(h, 0.0, 1.0, out=h)
    elif diff < -1e-15:
        idx = np.argsort(h)
        rem = -diff
        for i in idx:
            if rem <= 1e-15:
                break
            sub = h[i]
            if sub <= 0.0:
                continue
            take = sub if sub < rem else rem
            h[i] -= take
            rem -= take
        if rem > 1e-12:
            h -= rem / h.size
            np.clip(h, 0.0, 1.0, out=h)

    s = float(np.sum(h))
    diff = target - s
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            room = 1.0 - h[idx]
            add = room if room < diff else diff
            h[idx] += add
        else:
            idx = int(np.argmax(h))
            room = h[idx]
            take = room if room < -diff else -diff
            h[idx] -= take

    s = float(np.sum(h))
    diff = target - s
    if abs(diff) > 1e-14 and h.size > 1:
        if diff > 0:
            idx = 0
            for i in range(h.size):
                if h[i] < 1.0 - 1e-15:
                    idx = i
                    break
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = 0
            for i in range(h.size):
                if h[i] > 1e-15:
                    idx = i
                    break
            h[idx] = max(0.0, h[idx] + diff)

    np.clip(h, 0.0, 1.0, out=h)
    return h


def interp_hint(old, new_n):
    old = np.asarray(old, dtype=float)
    n_old = len(old)
    target = 0.5 * new_n
    if n_old == 0:
        return ensure_feasible(np.zeros(new_n, dtype=float), target)
    if n_old == new_n:
        return ensure_feasible(old.copy(), target)

    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    hv = np.interp(x_new, x_old, old)
    return ensure_feasible(hv, target)


def random_binary(n, rng):
    b = np.zeros(int(n), dtype=float)
    m = int(n) // 2 if int(n) % 2 == 0 else int(int(n) / 2)
    if m <= 0:
        return b
    if m >= int(n):
        b.fill(1.0)
        return b
    idx = rng.choice(int(n), m, replace=False)
    b[idx] = 1.0
    return b


def random_smooth(n, rng):
    n = int(n)
    h = 0.5 + 0.35 * rng.standard_normal(n)
    kern = np.ones(5, dtype=float) / 5.0
    h = np.convolve(h, kern, mode="same")
    h = np.convolve(h, kern, mode="same")
    return ensure_feasible(h, 0.5 * n)


def binary_from_values(h, target_count):
    h = np.asarray(h, dtype=float)
    n = len(h)
    b = np.zeros(n, dtype=float)
    m = int(target_count)
    if m <= 0:
        return b
    if m >= n:
        b.fill(1.0)
        return b
    order = np.argsort(-h)
    b[order[:m]] = 1.0
    return b


def perturb_h(h, scale, rng):
    h = np.asarray(h, dtype=float)
    n = len(h)
    h2 = h + float(scale) * rng.standard_normal(n)
    return ensure_feasible(h2, 0.5 * n)


def perturb_state(h, n, scale, rng):
    h = perturb_h(h, scale, rng)
    L = 2 * int(n) - 1
    c = full_corr_rev(h)
    scratch = np.empty(L, dtype=float)
    return h, c, scratch


def pair_coeff_into(h, j, k, n, d, bb):
    d.fill(0.0)
    np.subtract(1.0, h, out=bb)
    hr = h[::-1]
    d[n - 1 - j:2 * n - 1 - j] += bb
    d[j:j + n] -= hr
    d[n - 1 - k:2 * n - 1 - k] -= bb
    d[k:k + n] += hr


def eval_delta_tmp(c, d, delta, tmp):
    np.multiply(d, delta, out=tmp)
    np.add(c, tmp, out=tmp)
    return float(np.max(tmp))


def pair_active_status(c, d, raw, act):
    tol = 1e-11 + 1e-12 * abs(raw)
    np.greater_equal(c, raw - tol, out=act)
    if not np.any(act):
        act[int(np.argmax(c))] = True

    ds = d[act]
    min_s = float(np.min(ds))
    max_s = float(np.max(ds))
    eps = 1e-10

    if min_s <= eps and max_s >= -eps:
        return 0
    if max_s < -eps:
        return 1
    if min_s > eps:
        return -1
    return 0


def pair_best_delta(c, d, a, b, raw, tmp):
    best_delta = 0.0
    best_val = raw

    if b <= a + 1e-15:
        return 0.0, raw

    gr = 0.6180339887498949
    aa = float(a)
    bb = float(b)

    c1 = bb - (bb - aa) * gr
    c2 = aa + (bb - aa) * gr

    f1 = eval_delta_tmp(c, d, c1, tmp)
    if f1 < best_val:
        best_val = f1
        best_delta = c1

    f2 = eval_delta_tmp(c, d, c2, tmp)
    if f2 < best_val:
        best_val = f2
        best_delta = c2

    for _ in range(40):
        if bb - aa < 1e-13:
            break
        if f1 < f2:
            bb = c2
            c2 = c1
            f2 = f1
            c1 = bb - (bb - aa) * gr
            f1 = eval_delta_tmp(c, d, c1, tmp)
            if f1 < best_val:
                best_val = f1
                best_delta = c1
        else:
            aa = c1
            c1 = c2
            f1 = f2
            c2 = aa + (bb - aa) * gr
            f2 = eval_delta_tmp(c, d, c2, tmp)
            if f2 < best_val:
                best_val = f2
                best_delta = c2

    cands = (a, b, 0.0, aa, bb, 0.5 * (aa + bb), c1, c2)
    for delta in cands:
        delta = float(delta)
        if delta < a:
            delta = a
        elif delta > b:
            delta = b
        if abs(delta) < 1e-16:
            f = raw
        else:
            f = eval_delta_tmp(c, d, delta, tmp)
        if f < best_val - 1e-15:
            best_val = f
            best_delta = delta

    return best_delta, best_val


def local_pair_descent(h0, n, rng, stop_time):
    n = int(n)
    target = 0.5 * n

    h = ensure_feasible(np.array(h0, dtype=float, copy=True), target)
    L = 2 * n - 1

    c = full_corr_rev(h)
    scratch = np.empty(L, dtype=float)
    d = np.empty(L, dtype=float)
    bb = np.empty(n, dtype=float)
    act = np.empty(L, dtype=bool)
    raw = float(np.max(c))

    best_h = h.copy()
    best_c = c.copy()
    best_raw = raw

    accepted_since = 0
    stall = 0
    stall_limit = max(4000, 15 * n)
    perturb_left = 10
    perturb_count = 0
    recompute_every = 2048

    while time.time() < stop_time:
        improved = False
        feasible = True

        mode = rng.random_sample()

        if mode < 0.55:
            j = rng.randint(0, n)
            k = rng.randint(0, n)
            for _ in range(5):
                k = rng.randint(0, n)
                if k != j and abs(h[k] - h[j]) > 0.15:
                    break
            if k == j:
                k = (k + 1) % n

        elif mode < 0.80:
            p = int(np.argmax(c))
            o = p - (n - 1)
            if o == 0:
                j = rng.randint(0, n)
                k = (j + 1) % n
            elif o > 0:
                count = n - o
                if count <= 0:
                    j = rng.randint(0, n)
                    k = (j + 1) % n
                else:
                    vals = h[0:count] * (1.0 - h[o:o + count])
                    i = int(np.argmax(vals))
                    j = i
                    k = i + o
            else:
                u = -o
                count = n - u
                if count <= 0:
                    j = rng.randint(0, n)
                    k = (j + 1) % n
                else:
                    vals = h[u:n] * (1.0 - h[0:count])
                    i = int(np.argmax(vals)) + u
                    j = i
                    k = i + o
            if k == j:
                k = (k + 1) % n

        else:
            j = rng.randint(0, n)
            k = rng.randint(0, n)
            for _ in range(5):
                if 0.05 < h[j] < 0.95:
                    break
                j = rng.randint(0, n)
            for _ in range(5):
                if k != j and 0.05 < h[k] < 0.95:
                    break
                k = rng.randint(0, n)
            if k == j:
                k = (k + 1) % n

        hj = h[j]
        hk = h[k]

        lo = -hj
        alt = hk - 1.0
        if alt > lo:
            lo = alt

        hi = 1.0 - hj
        if hk < hi:
            hi = hk

        if hi <= lo + 1e-14:
            feasible = False

        if feasible:
            pair_coeff_into(h, j, k, n, d, bb)
            status = pair_active_status(c, d, raw, act)
            if status == 0:
                feasible = False
            else:
                if status > 0:
                    a0 = 0.0
                    b0 = hi
                else:
                    a0 = lo
                    b0 = 0.0

                if b0 <= a0 + 1e-14:
                    feasible = False
                else:
                    delta, fval = pair_best_delta(c, d, a0, b0, raw, scratch)
                    if abs(delta) < 1e-14 or fval >= raw - 1e-12:
                        feasible = False
                    else:
                        np.multiply(d, delta, out=scratch)
                        np.add(c, scratch, out=scratch)
                        c, scratch = scratch, c
                        h[j] += delta
                        h[k] -= delta
                        raw = fval
                        improved = True

        if improved:
            stall = 0
            accepted_since += 1

            if raw < best_raw - 1e-12:
                if raw < best_raw - 1e-7 or accepted_since % 256 == 0:
                    best_raw = raw
                    best_h = h.copy()
                    best_c = c.copy()

            if accepted_since >= recompute_every:
                accepted_since = 0
                h = ensure_feasible(h, target)
                c = full_corr_rev(h)
                scratch = np.empty(L, dtype=float)
                raw = float(np.max(c))
                if raw < best_raw - 1e-12:
                    best_raw = raw
                    best_h = h.copy()
                    best_c = c.copy()
        else:
            stall += 1

        if stall >= stall_limit and perturb_left > 0:
            perturb_left -= 1
            idx = perturb_count
            if idx < 0:
                idx = 0
            if idx >= len(Perturb_Scales):
                idx = len(Perturb_Scales) - 1
            scale = Perturb_Scales[idx]
            perturb_count += 1

            h, c, scratch = perturb_state(h, n, scale, rng)
            raw = float(np.max(c))
            if raw < best_raw - 1e-12:
                best_raw = raw
                best_h = h.copy()
                best_c = c.copy()
            stall = 0

    c = full_corr_rev(h)
    raw = float(np.max(c))
    if raw < best_raw - 1e-12:
        best_raw = raw
        best_h = h.copy()
        best_c = c.copy()

    best_h = ensure_feasible(best_h, target)
    best_c = full_corr_rev(best_h)
    best_raw = float(np.max(best_c))

    return best_h, best_c, best_raw, h, raw


def optimize_candidate(h0, seconds, rng):
    n = len(h0)
    start_time = time.time()
    if n <= 0:
        return h0, float("inf")
    if seconds <= 2.0:
        h0 = ensure_feasible(np.array(h0, dtype=float, copy=True), 0.5 * n)
        return h0, exact_c5(h0)

    best_h = ensure_feasible(np.array(h0, dtype=float, copy=True), 0.5 * n)
    best_c5 = exact_c5(best_h)

    main_alloc = 0.60 * float(seconds)
    if main_alloc < 5.0:
        main_alloc = float(seconds)
    t_stop = time.time() + main_alloc

    h1, _, _, cur1, _ = local_pair_descent(best_h, n, rng, t_stop)

    c5 = exact_c5(h1)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h1

    c5 = exact_c5(cur1)
    if c5 < best_c5:
        best_c5 = c5
        best_h = cur1

    remaining = float(seconds) - (time.time() - start_time)

    if (n % 2) == 0 and remaining > 20.0:
        hb = binary_from_values(best_h, n // 2)
        hb = ensure_feasible(hb, 0.5 * n)
        c5b = exact_c5(hb)
        if c5b < best_c5:
            best_c5 = c5b
            best_h = hb

        alloc = min(0.35 * float(seconds), max(10.0, remaining * 0.5))
        if alloc > 5.0:
            t_stop = time.time() + alloc
            h2, _, _, cur2, _ = local_pair_descent(hb, n, rng, t_stop)

            c5 = exact_c5(h2)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h2

            c5 = exact_c5(cur2)
            if c5 < best_c5:
                best_c5 = c5
                best_h = cur2

        remaining = float(seconds) - (time.time() - start_time)

    if remaining > 25.0:
        hp = perturb_h(best_h, 0.1, rng)
        c5p = exact_c5(hp)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp

        alloc = min(0.30 * float(seconds), max(10.0, remaining * 0.6))
        if alloc > 5.0:
            t_stop = time.time() + alloc
            h3, _, _, cur3, _ = local_pair_descent(hp, n, rng, t_stop)

            c5 = exact_c5(h3)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h3

            c5 = exact_c5(cur3)
            if c5 < best_c5:
                best_c5 = c5
                best_h = cur3

    best_h = ensure_feasible(best_h, 0.5 * n)
    best_c5 = exact_c5(best_h)
    return best_h, best_c5


def consider_candidate(h, best_h, best_c5, best_n):
    if h is None:
        return best_h, best_c5, best_n
    n = len(h)
    if n <= 0:
        return best_h, best_c5, best_n
    try:
        c5 = exact_c5(h)
    except Exception:
        return best_h, best_c5, best_n
    if c5 < best_c5 - 1e-15:
        h2 = ensure_feasible(np.array(h, dtype=float, copy=True), 0.5 * n)
        c5 = exact_c5(h2)
        return h2, c5, n
    return best_h, best_c5, best_n


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    budget = float(budget_s)
    global_end = t0 + budget - max(5.0, 0.02 * budget)
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float("inf")
    best_n = 0
    parent_h = None
    parent_n = 0

    try:
        parent_arr = initial_h_values
        if parent_arr is not None and len(parent_arr) > 0:
            parent_n = int(len(parent_arr))
            parent_h = ensure_feasible(np.array(parent_arr, dtype=float, copy=True), 0.5 * parent_n)
            best_h, best_c5, best_n = consider_candidate(parent_h, best_h, best_c5, best_n)
    except Exception:
        parent_h = None
        parent_n = 0

    if parent_h is None:
        parent_n = 192
        parent_h = random_binary(parent_n, rng)
        parent_h = ensure_feasible(parent_h, 0.5 * parent_n)
        best_h, best_c5, best_n = consider_candidate(parent_h, best_h, best_c5, best_n)

    if valid_stage_n(parent_n, True):
        parent_used = parent_n
    else:
        parent_used = 192

    if parent_used == len(parent_h):
        h0 = parent_h.copy()
    else:
        h0 = interp_hint(parent_h, parent_used)

    if parent_used >= 384:
        frac_parent = 0.30
    else:
        frac_parent = 0.20

    remaining = global_end - time.time()
    if remaining > 25.0:
        seconds = min(frac_parent * budget, remaining)
        if seconds > 25.0:
            hopt, _ = optimize_candidate(h0, seconds, rng)
            best_h, best_c5, best_n = consider_candidate(hopt, best_h, best_c5, best_n)

    resolutions = (
        (384, 0.32),
        (512, 0.18),
        (768, 0.12),
        (1024, 0.14),
        (192, 0.06),
        (256, 0.06),
        (320, 0.05),
    )

    for nn, frac in resolutions:
        if not valid_stage_n(nn, False):
            continue
        if nn == parent_used:
            continue

        remaining = global_end - time.time()
        if remaining <= 30.0:
            break
        seconds = min(frac * budget, remaining)
        if seconds <= 30.0:
            break

        if best_h is None:
            h0 = random_smooth(nn, rng)
        else:
            h0 = interp_hint(best_h, nn)

        hopt, _ = optimize_candidate(h0, seconds, rng)
        best_h, best_c5, best_n = consider_candidate(hopt, best_h, best_c5, best_n)

    remaining = global_end - time.time()
    if remaining > 45.0:
        n_rand = 384 if valid_stage_n(384, False) else 192
        hrand = random_binary(n_rand, rng)
        seconds = min(0.06 * budget, remaining)
        if seconds > 30.0:
            hopt, _ = optimize_candidate(hrand, seconds, rng)
            best_h, best_c5, best_n = consider_candidate(hopt, best_h, best_c5, best_n)

    remaining = global_end - time.time()
    if remaining > 35.0:
        n_rand = 512 if valid_stage_n(512, False) else 384
        hrand = random_smooth(n_rand, rng)
        seconds = min(0.05 * budget, remaining)
        if seconds > 25.0:
            hopt, _ = optimize_candidate(hrand, seconds, rng)
            best_h, best_c5, best_n = consider_candidate(hopt, best_h, best_c5, best_n)

    remaining = global_end - time.time()
    if remaining > 30.0 and best_h is not None:
        n_final = best_n if valid_stage_n(best_n, True) else 192
        if len(best_h) == n_final:
            h0 = best_h.copy()
        else:
            h0 = interp_hint(best_h, n_final)
        seconds = min(0.08 * budget, remaining)
        if seconds > 25.0:
            hopt, _ = optimize_candidate(h0, seconds, rng)
            best_h, best_c5, best_n = consider_candidate(hopt, best_h, best_c5, best_n)

    if best_h is None:
        best_n = 192
        best_h = random_binary(best_n, rng)
        best_h = ensure_feasible(best_h, 0.5 * best_n)
        best_c5 = exact_c5(best_h)
    else:
        best_n = int(len(best_h))
        best_h = ensure_feasible(best_h, 0.5 * best_n)
        best_c5 = exact_c5(best_h)

    final_c5 = solution_c5(best_h)
    return (best_h, final_c5, best_n)
