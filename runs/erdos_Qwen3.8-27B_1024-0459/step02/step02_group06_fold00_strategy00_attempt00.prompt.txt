def active_gradient(h, top_frac, min_k):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    k_count = max(min_k, int(np.ceil(len(corr) * top_frac)))
    top_idx = np.argsort(corr)[-k_count:]
    top_vals = corr[top_idx]
    dv = top_vals - top_vals.min()
    w = np.exp(dv / max(1e-10, dv.max() + 1e-10))
    w /= w.sum()
    g = np.zeros(n, dtype=float)
    for i, t in enumerate(top_idx):
        k = int(t) - center
        g += w[i] * corr_grad_lag(h, k)
    return g

def binarize_push(h, target, thresh=0.08):
    """Push values near 0 or 1 to exactly 0 or 1, maintaining sum"""
    n = h.size
    h2 = h.copy()
    changed = False
    for i in range(n):
        if h2[i] < thresh:
            target -= h2[i]
            h2[i] = 0.0
            changed = True
        elif h2[i] > 1.0 - thresh:
            target -= (1.0 - h2[i])
            h2[i] = 1.0
            changed = True
    if not changed:
        return h
    h2 = project_density(h2, 0.5 * n)
    return h2

def optimize_resolution(n, h0, deadline, seed=42):

    target = 0.5 * n

    best_c = evaluate_erdos(best_h)

    global _NC, _ALPHA, _TARGET
    _NC = n
    _TARGET = target
    bounds = [(0.0, 1.0)] * n
    constraints = ({"type": "eq", "fun": sum_constraint, "jac": sum_constraint_jac},)

    if n <= 120:
        alphas = (15.0, 50.0, 150.0, 400.0)
        maxiter = 200
    elif n <= 250:
        alphas = (15.0, 50.0, 150.0)
        maxiter = 120
    else:
        alphas = (15.0, 50.0, 120.0)
        maxiter = 70

    for alpha in alphas:

            break
        _ALPHA = alpha
        try:

                smooth_objective, h,
                method="SLSQP", jac=smooth_jacobian,
                bounds=bounds, constraints=constraints,
                options={"maxiter": maxiter, "ftol": 1e-11, "disp": False}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                cand = project_density(res.x, target)
                c = evaluate_erdos(cand)
                if c < best_c:
                    best_h = cand
                    best_c = c
                h = cand if c < best_c + 0.03 else best_h

    rem = deadline - time.perf_counter()
    if rem > 8.0:
        h2, c2 = pair_swap_aggressive(best_h, time.perf_counter() + rem * 0.35, max_iter=600, top_frac=0.05, min_k=6, n_pos=8, n_neg=8)

def make_diverse_start(n, rng, mode):
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    if mode == 0:
        h = 0.5 + 0.35 * np.sin(2 * np.pi * 3 * x / 2 + rng.uniform(0, 6.28))
        h += 0.15 * np.sin(2 * np.pi * 5 * x / 2 + rng.uniform(0, 6.28))
    elif mode == 1:
        h = np.empty(n, dtype=float)
        seg = int(n // 8) + 1
        for i in range(0, n, seg):
            end = min(i + seg, n)
            h[i:end] = float(rng.uniform(0.1, 0.9))
    elif mode == 2:
        u = rng.random(n)
        med = float(np.median(u))
        h = np.where(u > med, 0.75, 0.25).astype(float)
        h += rng.normal(0, 0.05, n)
    else:
        h = 0.5 + 0.4 * np.sin(2 * np.pi * int(rng.integers(2, 7)) * x / 2 + rng.uniform(0, 6.28))
    return project_density(h, 0.5 * n)
