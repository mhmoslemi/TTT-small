def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:

            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:

            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-12:

        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

        # First factor: h[i] where i = max(0,k-n+1) to min(n-1,k)
        lo1 = max(0, k - n + 1)
        hi1 = min(n - 1, k)
        for i in range(lo1, hi1 + 1):
            m = i - k + n - 1
            if 0 <= m < n:
                grad[i] += w * (1.0 - h[m])
        
        # Second factor: h[m] where m = i-k+n-1, i in valid range
        # m ranges over the same valid range as i shifted
        for i in range(lo1, hi1 + 1):
            m = i - k + n - 1
            if 0 <= m < n:
                grad[m] -= w * h[i]
    
    return grad

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr_values(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
            h = np.minimum(h, 1.0)
        else:
            mask = h > 1e-10
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
            h = np.maximum(h, 0.0)
    h = np.clip(h, 0.0, 1.0)
    # Exact fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        order = np.argsort(np.abs(h - 0.5))
        for idx in order:
            room = (1.0 - h[idx]) if diff > 0 else h[idx]
            adj = min(abs(diff), room)
            h[idx] += adj * (1.0 if diff > 0 else -1.0)
            diff -= adj * (1.0 if diff > 0 else -1.0)
            if abs(diff) < 1e-14:
                break
    return h

    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

    cmax = np.max(corr)
    return cmax + (1.0/beta) * np.log(np.sum(np.exp(beta * (corr - cmax))))

    if dx_steps is None:
        dx_steps = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]
    target = n / 2.0
    best_c5 = compute_c5(h)
    
    for _ in range(n_passes):
        improved = False
        for j in range(n):
            for step in dx_steps:
                for sign in [1, -1]:
                    h_try = h.copy()
                    h_try[j] += sign * step
                    if h_try[j] < 0 or h_try[j] > 1:
                        continue
                    # Fix sum
                    diff = target - np.sum(h_try)
                    if abs(diff) > 1e-10:

                        others = [k for k in range(n) if k != j]
                        # Add to coordinate closest to allowing the change
                        if diff > 0:
                            mask = h_try < 1.0 - 1e-10
                            mask[j] = False

                            mask[j] = False
                        cnt = np.count_nonzero(mask)
                        if cnt == 0:
                            continue

                        h_try = np.clip(h_try, 0, 1)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        h = h_try
                        best_c5 = c5_try
                        improved = True
        if not improved:
            break

def slsqp_smoothed(h, n, beta, max_iter=3000):
    """SLSQP with smoothed max objective."""
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
             'jac': lambda x: np.ones(n)}]
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return smoothed_max(corr, beta)
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-15, 'iprint': 0})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None

    target = n / 2.0

        # corr[k_star] = sum_i h[i]*(1-h[i-k_star+n-1])

            step = rng.uniform(0.01, 0.1)

            if h[i] > h[m]:
                h_try[i] -= step
                h_try[m] += step
            else:
                h_try[i] += step
                h_try[m] -= step
            h_try = np.clip(h_try, 0, 1)
            diff = target - np.sum(h_try)
            if abs(diff) > 1e-10:

                cnt = np.count_nonzero(mask)
                if cnt > 0:

                h_try = np.clip(h_try, 0, 1)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                h = h_try
                best_c5 = c5_try
                improved = True
        
        if not improved:

            idxs = rng.choice(n, 3, replace=False)

                h_or[idx] += rng.uniform(-0.05, 0.05)

            mask = (h_or > 0.01) & (h_or < 0.99)
            cnt = np.count_nonzero(mask)
            if cnt > 0:
                h_or[mask] += diff / cnt
            h_or = np.clip(h_or, 0, 1)
            h_or = make_feasible(h_or, n)
            c5_try = compute_c5(h_or)

                best_c5 = c5_try

                h = h_or
                best_c5 = max(compute_c5(h), best_c5)  # keep track of true best
    
    return h, best_c5

def projected_gradient_step(h, n, beta, lr, max_steps=100):
    """Simple projected gradient descent with line search."""
    target = n / 2.0
    best_c5 = compute_c5(h)
    
    for step in range(max_steps):

        cmax = np.max(corr)
        weights = np.exp(beta * (corr - cmax))
        sw = weights.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        for k in range(len(corr)):
            w = weights[k] / sw
            if w < 1e-12:
                continue
            lo1 = max(0, k - n + 1)
            hi1 = min(n - 1, k)
            for i in range(lo1, hi1 + 1):
                m = i - k + n - 1
                if 0 <= m < n:
                    grad[i] += w * (1.0 - h[m])
                    grad[m] -= w * h[i]
        
        # Gradient step
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Project to sum constraint
        diff = target - np.sum(h_new)
        h_new += diff / n
        h_new = np.clip(h_new, 0, 1)
        h_new = make_feasible(h_new, n)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            h = h_new
            best_c5 = c5_new
        else:
            lr *= 0.5
            if lr < 1e-8:
                break
    
    return h, best_c5

        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
            h = np.minimum(h, 1.0)
        else:
            mask = h > 1e-10
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
            h = np.maximum(h, 0.0)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        order = np.argsort(np.abs(h - 0.5))
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                room = 1.0 - h[idx]
                adj = min(diff, room)
                h[idx] += adj
                diff -= adj
            else:
                room = h[idx]
                adj = min(-diff, room)
                h[idx] -= adj
                diff += adj
    return h

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def smoothed_max_val(corr, beta):
    cmax = np.max(corr)
    return cmax + (1.0 / beta) * np.log(np.sum(np.exp(beta * (corr - cmax))))

def slsqp_smoothed(h, n, beta, max_iter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return smoothed_max_val(corr, beta)

    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-15, 'iprint': 0})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None

def gradient_smoothed_max(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    weights = np.exp(beta * (corr - cmax))
    sw = weights.sum()

    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k] / sw
        if w < 1e-12:
            continue
        lo1 = max(0, k - n + 1)
        hi1 = min(n - 1, k)
        for i in range(lo1, hi1 + 1):
            m = i - k + n - 1
            if 0 <= m < n:
                grad[i] += w * (1.0 - h[m])
                grad[m] -= w * h[i]
    return grad

def projected_subgradient(h, n, beta, max_steps=200, lr_init=0.05):
    target = n / 2.0
    best_c5 = compute_c5(h)
    lr = lr_init
    for step in range(max_steps):
        grad = gradient_smoothed_max(h, beta)
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        diff = target - np.sum(h_new)
        h_new += diff / n
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = make_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            h = h_new
            best_c5 = c5_new
            lr = min(lr * 1.05, 0.5)
        else:
            lr *= 0.7
            if lr < 1e-9:
                break
    return h, best_c5

def coordinate_descent(h, n, n_passes=30, steps=None):
    if steps is None:
        steps = [0.002, 0.005, 0.01, 0.02, 0.05, 0.1]
    target = n / 2.0
    best_c5 = compute_c5(h)
    for _ in range(n_passes):
        improved = False
        for j in range(n):
            for step in steps:
                for sign in (1, -1):
                    h_try = h.copy()
                    h_try[j] += sign * step
                    if h_try[j] < -1e-10 or h_try[j] > 1.0 + 1e-10:
                        continue
                    h_try = np.clip(h_try, 0.0, 1.0)
                    diff = target - np.sum(h_try)
                    if abs(diff) > 1e-10:
                        others = np.ones(n, dtype=bool)
                        others[j] = False
                        if diff > 0:
                            mask = others & (h_try < 1.0 - 1e-8)
                        else:
                            mask = others & (h_try > 1e-8)
                        cnt = np.count_nonzero(mask)
                        if cnt == 0:
                            continue
                        h_try[mask] += diff / cnt
                        h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        h = h_try
                        best_c5 = c5_try
                        improved = True
        if not improved:
            break
    return h, best_c5

def pair_local_search(h, n, rng, n_trials=10000):
    target = n / 2.0
    best_c5 = compute_c5(h)
    for _ in range(n_trials):
        i, j = rng.choice(n, 2, replace=False)
        step = rng.uniform(0.005, 0.12)
        h_try = h.copy()
        h_try[i] += step
        h_try[j] -= step
        if h_try[i] > 1.0 + 1e-10 or h_try[j] < -1e-10:
            continue
        h_try = np.clip(h_try, 0.0, 1.0)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            h = h_try
            best_c5 = c5_try
    return h, best_c5

def targeted_lag_optimize(h, n, rng, n_rounds=100):
    target = n / 2.0
    best_c5 = compute_c5(h)
    for _ in range(n_rounds):
        corr = compute_corr_values(h)
        k_star = np.argmax(corr)
        top_k = np.argsort(corr)[::-1][:5]
        improved = False
        for _ in range(30):
            k_t = top_k[rng.randint(len(top_k))]
            i = rng.randint(n)
            m = i - k_t + n - 1
            if m < 0 or m >= n:
                m = (i + k_t) % n
            if m == i:
                continue
            step = rng.uniform(0.005, 0.08)
            h_try = h.copy()
            if h[i] > h[m]:
                h_try[i] -= step
                h_try[m] += step
            else:
                h_try[i] += step
                h_try[m] -= step
            h_try = np.clip(h_try, 0.0, 1.0)
            diff = target - np.sum(h_try)
            if abs(diff) > 1e-10:
                mask = (h_try > 1e-6) & (h_try < 1.0 - 1e-6)
                cnt = np.count_nonzero(mask)
                if cnt > 0:
                    h_try[mask] += diff / cnt
                h_try = np.clip(h_try, 0.0, 1.0)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                h = h_try
                best_c5 = c5_try
                improved = True
        if not improved and rng.random() < 0.3:
            h_or = h.copy()
            idxs = rng.choice(n, 4, replace=False)
            for idx in idxs:
                h_or[idx] += rng.uniform(-0.04, 0.04)
            h_or = np.clip(h_or, 0.0, 1.0)
            h_or = make_feasible(h_or, n)
            c5_try = compute_c5(h_or)
            if c5_try < best_c5:
                h = h_or
                best_c5 = c5_try
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    def remaining():
        return budget_s - (time.time() - start_time)

    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if h_parent is None or len(h_parent) == 0:
            h_parent = None
    except (NameError, Exception):
        h_parent = None

    if h_parent is not None:
        best_h = h_parent.copy()
        best_n = len(best_h)
    else:
        best_n = 160
        best_h = np.ones(best_n) * 0.5

    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP smoothed annealing (coarse to fine)
    for beta in [50, 150, 400, 1000, 2500]:
        if remaining() < 40:
            break
        h_opt, c5_opt = slsqp_smoothed(best_h, best_n, beta, max_iter=4000)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 2: Projected subgradient (fast, handles the max structure)
    if remaining() > 60:
        best_h, best_c5 = projected_subgradient(best_h, best_n, beta=500,
                                                max_steps=300, lr_init=0.03)

    # Phase 3: Fine coordinate descent
    if remaining() > 50:
        best_h, best_c5 = coordinate_descent(best_h, best_n, n_passes=20,
                                             steps=[0.002, 0.005, 0.01, 0.02, 0.05])

    # Phase 4: Pair local search
    if remaining() > 40:
        n_trials = min(15000, int(remaining() / 0.012))
        best_h, best_c5 = pair_local_search(best_h, best_n, rng, n_trials)

    # Phase 5: Targeted lag optimization
    if remaining() > 40:
        best_h, best_c5 = targeted_lag_optimize(best_h, best_n, rng, n_rounds=150)

    # Phase 6: Second round of SLSQP at high beta
    for beta in [800, 3000]:
        if remaining() < 30:
            break
        h_opt, c5_opt = slsqp_smoothed(best_h, best_n, beta, max_iter=3000)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 7: Resolution sweep
    resolutions = [152, 156, 160, 164, 168, 172, 176]
    for n_try in resolutions:
        if remaining() < 35:
            break
        if n_try == best_n:
            continue
        if n_try < 40 or n_try % 2 != 0:
            continue
        h_interp = make_feasible(interpolate_to_n(best_h, n_try), n_try)
        c5_interp = compute_c5(h_interp)
        if remaining() < 15:
            if c5_interp < best_c5:
                best_c5 = c5_interp
                best_h = h_interp
                best_n = n_try
            continue
        h_opt, c5_opt = slsqp_smoothed(h_interp, n_try, 200, max_iter=2000)
        if h_opt is not None and c5_opt < c5_interp:
            c5_interp = c5_opt
            h_interp = h_opt
        # Quick local search at this resolution
        if remaining() > 15:
            h_interp, c5_interp = coordinate_descent(h_interp, n_try, n_passes=8,
                                                     steps=[0.005, 0.01, 0.02])
        if c5_interp < best_c5:
            best_c5 = c5_interp
            best_h = h_interp
            best_n = n_try

    # Phase 8: Multi-start with perturbations
    for trial in range(5):
        if remaining() < 25:
            break
        noise = 0.015 * (1.0 - 0.15 * trial)
        h_try = best_h + rng.randn(best_n) * noise
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = make_feasible(h_try, best_n)
        h_opt, c5_opt = slsqp_smoothed(h_try, best_n, 100, max_iter=2000)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 9: Final intensive local search
    if remaining() > 30:
        best_h, best_c5 = coordinate_descent(best_h, best_n, n_passes=15,
                                             steps=[0.001, 0.002, 0.005, 0.01, 0.02])

    if remaining() > 20:
        best_h, best_c5 = pair_local_search(best_h, best_n, rng,
                                            n_trials=min(10000, int(remaining() / 0.01)))

    if remaining() > 20:
        best_h, best_c5 = projected_subgradient(best_h, best_n, beta=2000,
                                                max_steps=150, lr_init=0.01)

    # Phase 10: Final SLSQP polish
    if remaining() > 15:
        h_opt, c5_opt = slsqp_smoothed(best_h, best_n, 5000, max_iter=4000)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Final micro-polish
    if remaining() > 10:
        best_h, best_c5 = coordinate_descent(best_h, best_n, n_passes=5,
                                             steps=[0.001, 0.002, 0.005])

    # Final feasibility and verification
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    if np.any(best_h < -1e-12) or np.any(best_h > 1.0 + 1e-12):
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = make_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)

    assert abs(np.sum(best_h) - best_n / 2.0) < 1e-8, \
        f"Sum constraint violated: {np.sum(best_h)} vs {best_n/2.0}"

    return (best_h, best_c5, best_n)
