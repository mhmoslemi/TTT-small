```python
def subgradient_psolve(h, n, n_iter, beta_init, beta_final, lr_init, lr_final):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for t in range(n_iter):
        # Smooth parameter
        beta = beta_init * (beta_final/beta_init) ** (t/n_iter)
        
        # Compute correlation
        corr = np.correlate(h, 1.0-h, mode='full')
        
        # Softmax weights
        corr_max = np.max(corr)
        weights = np.exp(beta * (corr - corr_max))
        weights /= np.sum(weights)
        
        # Gradient: weighted average of individual gradients
        grad = np.zeros(n)
        for k in range(len(corr)):
            if weights[k] < 1e-10:
                continue
            shift = k - (n-1)
            # ∂corr[k]/∂h[j] = (1-h[j+shift])*[0≤j+shift<n] - h[j-shift]*[0≤j-shift<n]
            # First term: (1-h[j+shift]) for j where 0≤j+shift<n, i.e., -shift≤j<n-shift
            lo1 = max(0, -shift)
            hi1 = min(n, n-shift)  # j goes from lo1 to hi1-1
            if lo1 < hi1:
                grad[lo1:hi1] += weights[k] * (1.0 - h[lo1+shift:hi1+shift])
            # Second term: -h[j-shift] for j where 0≤j-shift<n, i.e., shift≤j<n+shift
            lo2 = max(0, shift)
            hi2 = min(n, n+shift)
            if lo2 < hi2:
                grad[lo2:hi2] -= weights[k] * h[lo2-shift:hi2-shift]
        
        # Step
        lr = lr_init * (lr_final/lr_init) ** (t/n_iter)
        h = h - lr * grad
        
        # Project to feasible set
        h = np.clip(h, 0.0, 1.0)
        # Adjust sum
        diff = n/2.0 - np.sum(h)
        if diff > 1e-12:
            mask = h < 1.0 - 1e-12
            if np.any(mask):
                h[mask] = np.minimum(h[mask] + diff/np.count_nonzero(mask), 1.0)
        elif diff < -1e-12:
            mask = h > 1e-12
            if np.any(mask):
                h[mask] = np.maximum(h[mask] + diff/np.count_nonzero(mask), 0.0)
        h = np.clip(h, 0.0, 1.0)
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5
```

```python
        corr = np.correlate(h, 1.0-h, mode='full')
        k_max = np.argmax(corr)
        shift = k_max - (n-1)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def interp_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def corr_and_grad(h, n, top_m):
    """Compute correlation and gradient of top-m average."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_m:]
    g = np.zeros(n)
    for k in top_idx:
        shift = k - (n - 1)
        if shift >= 0:
            g[:n - shift] += (1.0 - h[shift:])
            if shift > 0:
                g[shift:] -= h[:-shift]
        else:
            s = -shift
            g[s:] += (1.0 - h[:-s])
            g[:n - s] -= h[s:]
    g /= float(top_m)
    return corr, g

def subgrad_momentum(h_init, n, n_iter, top_m=5, alpha0=0.08, momentum=0.85):
    h = h_init.copy()
    v = np.zeros(n)  # momentum
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stale = 0
    
    for t in range(n_iter):
        corr, g = corr_and_grad(h, n, top_m)
        v = momentum * v + (1.0 - momentum) * g
        a = alpha0 / (1.0 + t / 1000.0)
        h_new = h - a * v
        h_new = np.clip(h_new, 0.0, 1.0)
        # Project sum
        for _ in range(3):
            diff = n / 2.0 - np.sum(h_new)
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                mask = h_new < 1.0 - 1e-10
            else:
                mask = h_new > 1e-10
            cnt = np.count_nonzero(mask)
            if cnt > 0:
                h_new[mask] += diff / cnt
            h_new = np.clip(h_new, 0.0, 1.0)
        h = h_new
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
        if stale > 5000:
            h = best_h.copy()
            v = np.zeros(n)
            stale = 0
    
    return best_h, best_c5

def fw_solve(h_init, n, n_iter, top_m=5):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for t in range(n_iter):
        corr, g = corr_and_grad(h, n, top_m)
        # FW step: vertex minimizing g·v
        order = np.argsort(g)
        v = np.zeros(n)
        v[order[:n // 2]] = 1.0
        gamma = 2.0 / (t + 2.0)
        h = (1.0 - gamma) * h + gamma * v
        h = np.clip(h, 0.0, 1.0)
        # Fix sum
        diff = n / 2.0 - np.sum(h)
        if abs(diff) > 1e-10:
            order_fix = np.argsort(h)
            if diff > 0:
                for i in range(n - 1, -1, -1):
                    room = 1.0 - h[order_fix[i]]
                    adj = min(room, diff)
                    h[order_fix[i]] += adj
                    diff -= adj
                    if abs(diff) < 1e-10:
                        break
            else:
                for i in range(n):
                    room = h[order_fix[i]]
                    adj = min(room, -diff)
                    h[order_fix[i]] -= adj
                    diff += adj
                    if abs(diff) < 1e-10:
                        break
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def smooth_anneal(h_init, n, betas=(20, 100, 500, 2000)):
    current = h_init.copy()
    best_c5 = compute_c5(current)
    best_h = current.copy()
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    for beta in betas:
        def obj_s(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            cmax = np.max(corr)
            return cmax + (1.0 / beta) * np.log(np.sum(np.exp(beta * (corr - cmax))))
        try:
            res = minimize(obj_s, current, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 2000, 'ftol': 1e-14, 'iprint': 0})
            if res.fun < 1e10:
                current = make_feasible(res.x, n)
                c5 = compute_c5(current)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = current.copy()
        except Exception:
            pass
    return best_h, best_c5

def local_polish(h, n, n_rounds, rng):
    h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(n_rounds):
        improved = False
        for i in range(n - 1):
            if h[i] <= h[i + 1]:
                continue
            for step in [0.01, 0.03, 0.08]:
                h_try = h.copy()
                amt = min(step, (h[i] - h[i + 1]) / 2.0)
                if amt < 1e-6:
                    continue
                h_try[i] -= amt
                h_try[i + 1] += amt
                h_try = np.clip(h_try, 0.0, 1.0)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    h = h_try
                    best_c5 = c5_try
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def random_search(h, n, n_trials, base_step, rng):
    h = h.copy()
    best_c5 = compute_c5(h)
    step = base_step
    for _ in range(n_trials):

        diff = n / 2.0 - np.sum(h_try)
        mask = (h_try > 0.005) & (h_try < 0.995)
        cnt = np.count_nonzero(mask)

        h_try = np.clip(h_try, 0.0, 1.0)

        if c5_try < best_c5 - 1e-13:
            h = h_try
            best_c5 = c5_try
        elif rng.random() < 0.05:
            step *= 0.92
    return h, best_c5
```

```python
def make_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)

    s = np.sum(h)
    diff = target - s

    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_m:]

            g[s:] += (1.0 - h[:-s])
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-10:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def interp_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.full(n_dst, 0.5)
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def corr_grad_top_m(h, n, top_m):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_m:]
    g = np.zeros(n)
    for k in top_idx:
        shift = k - (n - 1)
        if shift >= 0:
            g[:n - shift] += (1.0 - h[shift:])
            if shift > 0:
                g[shift:] -= h[:-shift]
        else:
            s = -shift
            g[s:] += (1.0 - h[:-s])
            g[:n - s] -= h[s:]
    g /= float(top_m)
    return corr, g

def subgrad_momentum(h_init, n, n_iter, top_m=5, alpha0=0.08, mu=0.85, t_scale=1000):
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stale = 0
    for t in range(n_iter):
        corr, g = corr_grad_top_m(h, n, top_m)
        v = mu * v + (1.0 - mu) * g
        a = alpha0 / (1.0 + t / float(t_scale))
        h_new = h - a * v
        h_new = np.clip(h_new, 0.0, 1.0)
        for _ in range(5):
            diff = n / 2.0 - np.sum(h_new)
            if abs(diff) < 1e-13:
                break
            if diff > 0:
                mask = h_new < 1.0 - 1e-10
            else:
                mask = h_new > 1e-10
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h_new[mask] += diff / cnt
            h_new = np.clip(h_new, 0.0, 1.0)
        if not np.all(np.isfinite(h_new)):
            h_new = h.copy()
        else:
            h = h_new
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                best_h = h.copy()
                stale = 0
            else:
                stale += 1
        if stale > 4000:
            h = best_h.copy()
            v = np.zeros(n)
            stale = 0
    return best_h, best_c5

def subgrad_nesterov(h_init, n, n_iter, top_m=5, alpha0=0.12, t_scale=500):
    h = h_init.copy()
    y = h.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stale = 0
    for t in range(n_iter):
        t1 = t + 1
        corr, g = corr_grad_top_m(y, n, top_m)
        a = alpha0 / (1.0 + t1 / float(t_scale))
        h_new = y - a * g
        h_new = np.clip(h_new, 0.0, 1.0)
        for _ in range(5):
            diff = n / 2.0 - np.sum(h_new)

                mask = h_new < 1.0 - 1e-10
            else:
                mask = h_new > 1e-10
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h_new[mask] += diff / cnt
            h_new = np.clip(h_new, 0.0, 1.0)
        beta_t = t / (t1 + 1.0 + np.sqrt(1.0 + 4.0 * t * t1))
        y = h_new + beta_t * (h_new - h_prev)
        y = np.clip(y, 0.0, 1.0)
        h_prev = h_new
        h = h_new
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
        if stale > 4000:
            h = best_h.copy()
            y = h.copy()
            h_prev = h.copy()
            stale = 0
    return best_h, best_c5

def fw_solve(h_init, n, n_iter, top_m=5):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for t in range(n_iter):
        corr, g = corr_grad_top_m(h, n, top_m)
        order = np.argsort(g)
        v = np.zeros(n)
        v[order[:n // 2]] = 1.0
        gamma = 2.0 / (t + 2.0)
        h = (1.0 - gamma) * h + gamma * v
        h = np.clip(h, 0.0, 1.0)
        diff = n / 2.0 - np.sum(h)
        if abs(diff) > 1e-10:
            idx_s = np.argsort(h)
            if diff > 0:
                for i in range(n - 1, -1, -1):
                    room = 1.0 - h[idx_s[i]]
                    adj = min(room, diff)
                    if adj <= 1e-15:
                        continue
                    h[idx_s[i]] += adj
                    diff -= adj
                    if abs(diff) < 1e-12:
                        break
            else:
                for i in range(n):
                    room = h[idx_s[i]]
                    adj = min(room, -diff)
                    if adj <= 1e-15:
                        continue
                    h[idx_s[i]] -= adj
                    diff += adj
                    if abs(diff) < 1e-12:
                        break
        h = np.clip(h, 0.0, 1.0)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def fw_away(h_init, n, n_iter, top_m=5):
    h = h_init.copy()
    v_prev = np.zeros(n)
    v_prev[0:n // 2] = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for t in range(n_iter):
        corr, g = corr_grad_top_m(h, n, top_m)
        order = np.argsort(g)
        v = np.zeros(n)
        v[order[:n // 2]] = 1.0
        gamma = 2.0 / (t + 2.0)
        away_coeff = (h - v_prev) @ g
        if away_coeff > 0:
            h = h + gamma * (v - v_prev)
            mask = (h < 0.0) | (h > 1.0)
            if np.any(mask):
                h = (1.0 - gamma) * h_init + gamma * v
        else:
            h = (1.0 - gamma) * h + gamma * v
        h = np.clip(h, 0.0, 1.0)
        diff = n / 2.0 - np.sum(h)
        if abs(diff) > 1e-10:

            if diff > 0:

                    diff -= adj

            else:

                    adj = min(h[idx_s[i]], -diff)

                    h[idx_s[i]] -= adj
                    diff += adj
                    if abs(diff) < 1e-12:

        h = np.clip(h, 0.0, 1.0)
        v_prev = v.copy()
        c5 = compute_c5(h)

def local_polish(h, n, n_rounds, rng):
    h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(n_rounds):
        improved = False
        for i in range(n - 1):
            if h[i] <= h[i + 1]:
                continue
            for sv in [0.005, 0.02, 0.06]:
                amt = min(sv, (h[i] - h[i + 1]) * 0.45)
                if amt < 1e-8:
                    continue
                h_try = h.copy()
                h_try[i] -= amt
                h_try[i + 1] += amt
                h_try = np.clip(h_try, 0.0, 1.0)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    h = h_try
                    best_c5 = c5_try
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def random_search(h_init, n, n_trials, base_step, rng):
    h = h_init.copy()
    best_c5 = compute_c5(h)
    step = base_step
    for _ in range(n_trials):
        h_try = h.copy()
        n_mod = rng.randint(1, 9)
        idxs = rng.choice(n, n_mod, replace=False)
        for idx in idxs:
            h_try[idx] += rng.uniform(-step, step)
        h_try = np.clip(h_try, 0.0, 1.0)
        diff = n / 2.0 - np.sum(h_try)
        mask = (h_try > 0.005) & (h_try < 0.995)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h_try[mask] += diff / cnt
        h_try = np.clip(h_try, 0.0, 1.0)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-13:
            h = h_try
            best_c5 = c5_try
        elif rng.random() < 0.04:
            step = max(step * 0.94, 0.001)
    return h, best_c5

def smooth_anneal_slsqp(h_init, n, betas):
    current = h_init.copy()
    best_c5 = compute_c5(current)
    best_h = current.copy()
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    cons_s = [{'type': 'eq',
               'fun': lambda x: np.sum(x) - target,
               'jac': lambda x: np.ones(n)}]
    for beta in betas:
        def obj_s(x, b=beta):
            corr = np.correlate(x, 1.0 - x, mode='full')
            cmax = np.max(corr)
            scaled = b * (corr - cmax)
            scaled = np.clip(scaled, -500, 500)
            return cmax + np.log(np.sum(np.exp(scaled))) / b
        try:
            res = minimize(obj_s, current, method='SLSQP', bounds=bounds,
                           constraints=cons_s,
                           options={'maxiter': 1500, 'ftol': 1e-14, 'iprint': 0})
            if res.fun < 1e10:
                current = make_feasible(res.x, n)
                c5 = compute_c5(current)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = current.copy()
        except Exception:
            pass
    return best_h, best_c5
```
