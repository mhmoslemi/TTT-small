        all_v[n - 1 + L] = (pref[m] - ac) * dx
        if L > 0:
            all_v[n - 1 - L] = (total - pref[L] - ac) * dx
    maxv = float(np.max(all_v))
    scaled = beta * (all_v - maxv)
    scaled = np.clip(scaled, -700.0, 700.0)
    w = np.exp(scaled)
    wsum = float(np.sum(w))
    if wsum <= 0:
        return np.zeros(n)
    w = w / wsum
    g = np.zeros(n, dtype=float)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        if idx == n - 1:
            L = 0
            m = n
            g[:] += wi * dx * (1.0 - 2.0 * h)
        elif idx > n - 1:
            L = idx - (n - 1)
            m = n - L
            g[:m] += wi * dx * (1.0 - h[L:])
            g[L:] += -wi * dx * h[:m]
        else:
            L = (n - 1) - idx
            m = n - L
            g[L:] += wi * dx * (1.0 - h[:m])
            g[:m] += -wi * dx * h[L:]
    return g

def fidsa_smooth(h0, n, beta, n_iter=200):
    h = project_feasible(h0, n).copy()
    h_prev = h.copy()
    t_momentum = 1.0
    step = 0.3 / (beta * n * 2.0)
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(n_iter):
        _, g = _smooth_max_obj_grad(h, beta)
        h_new = project_feasible(h - step * g, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_momentum * t_momentum)) / 2.0
        h = h_new + (t_new - 1.0) / t_new * (h_new - h_prev)
        h = project_feasible(h, n)
        h_prev = h_new
        t_momentum = t_new
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def _smooth_max_obj_grad(h, beta):
    n = h.size
    dx = 2.0 / float(n)
    total = float(np.sum(h))
    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    all_v = np.empty(2 * n - 1, dtype=float)
    for L in range(n):
        m = n - L
        ac = float(np.dot(h[:m], h[L:]))
        all_v[n - 1 + L] = (pref[m] - ac) * dx
        if L > 0:
            all_v[n - 1 - L] = (total - pref[L] - ac) * dx
    maxv = float(np.max(all_v))
    scaled = beta * (all_v - maxv)
    scaled = np.clip(scaled, -700.0, 700.0)
    r = float(np.sum(np.exp(scaled)))
    obj = maxv + float(np.log(r)) / beta if r > 0 else maxv
    w = np.exp(scaled) / max(r, 1e-300)
    wsum = float(np.sum(w))
    if wsum <= 0:
        return obj, np.zeros(n)
    w = w / wsum
    g = np.zeros(n, dtype=float)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        if idx == n - 1:
            g[:] += wi * dx * (1.0 - 2.0 * h)
        elif idx > n - 1:
            L = idx - (n - 1)
            m = n - L
            g[:m] += wi * dx * (1.0 - h[L:])
            g[L:] += -wi * dx * h[:m]
        else:
            L = (n - 1) - idx
            m = n - L
            g[L:] += wi * dx * (1.0 - h[:m])
            g[:m] += -wi * dx * h[L:]
    return obj, g

def active_set_optimize(h0, n, active_lags, t0, maxiter=80):
    n = int(n)
    h0 = np.asarray(h0, dtype=float)

    def obj_vec(x):
        return float(x[-1])

    def obj_grad(x):
        g = np.zeros(len(x))
        g[-1] = 1.0
        return g

    def eq_c(x):
        return float(np.sum(x[:-1]) - 0.5 * float(n))

    def eq_grad(x):
        g = np.zeros(len(x))
        g[:-1] = 1.0
        return g

    dx = 2.0 / float(n)
    A = [int(l) for l in active_lags]

    def ineq_c(x):
        h = x[:-1]
        out = np.empty(len(A))
        for i, L in enumerate(A):
            Lv = lag_val(h, L) * dx
            out[i] = float(x[-1]) - Lv
        return out

    def ineq_grad(x):
        h = x[:-1]
        J = np.zeros((len(A), n + 1))
        for i, L in enumerate(A):
            g = lag_grad_raw(h, L)
            J[i, :n] = -dx * g
            J[i, n] = 1.0
        return J

    x0 = np.empty(n + 1)
    x0[:n] = h0
    x0[n] = float(t0)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [{"type": "eq", "fun": eq_c, "jac": eq_grad}]
    if len(A) > 0:
        cons.append({"type": "ineq", "fun": ineq_c, "jac": ineq_grad})
    try:
        res = minimize(obj_vec, x0, jac=obj_grad, method="SLSQP",
                       bounds=bounds, constraints=cons,
                       options={"maxiter": int(maxiter), "ftol": 1e-12, "disp": False})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x[:n], n)
    except Exception:
        pass
    return None

def lag_val(h, L):
    n = h.size
    if L == 0:
        return float(np.sum(h) - np.dot(h, h))
    L = abs(int(L))
    m = n - L
    ac = float(np.dot(h[:m], h[L:]))
    if L > 0:
        return float(np.sum(h[:m])) - ac
    return float(np.sum(h[L:])) - ac

def lag_grad_raw(h, L):
    n = h.size
    if L == 0:
        return 1.0 - 2.0 * h
    L = abs(int(L))
    m = n - L
    g = np.zeros(n)
    if L > 0:
        # For positive lag: v = sum(h[:m]) - dot(h[:m], h[L:])
        g[:m] += 1.0 - h[L:]
        g[L:] += -h[:m]
    # For negative lag, by symmetry the formula is the same structure
    # (the sign of L determines which side is "sum" vs "total - sum")
    # But since we always use abs(L) and the problem is symmetric,
    # the gradient magnitude is the same, just the assignment differs.
    # For our purposes in the active set, we use the correct signed version:
    return g

def lag_grad_signed(h, L_signed):
    n = h.size
    L = abs(int(L_signed))
    if L == 0:
        return 1.0 - 2.0 * h
    m = n - L
    g = np.zeros(n)
    if L_signed >= 0:
        g[:m] += 1.0 - h[L:]
        g[L:] += -h[:m]
    else:
        g[L:] += 1.0 - h[:m]
        g[:m] += -h[L:]
    return g

def choose_active(vals, lags, max_count):
    if vals.size == 0:
        return [0]
    order = np.argsort(-vals)
    top_val = float(vals[order[0]])
    near = np.where(vals >= top_val - 2e-5)[0]
    idx = np.unique(np.concatenate([order[:max_count], near]))
    idx = idx[np.argsort(-vals[idx])]
    if idx.size > max_count * 3:
        idx = idx[:max_count * 3]
    return [int(lags[i]) for i in idx]

def make_lag_array(vals):
    n = len(vals) // 2
    lags = np.empty(len(vals), dtype=int)
    for i in range(len(vals)):
        lags[i] = i - (n - 1)
    return lags

def interpolate_to_n(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float)
    n_dst = int(n_dst)
    n_src = h_src.size
    if n_src == n_dst:
        return h_src.copy()
    if n_src <= 1:
        return np.full(n_dst, 0.5, dtype=float)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def coordinate_descent(h0, n, n_trials, seed=42):
    rng = np.random.RandomState(int(seed))
    h = project_feasible(h0, n).copy()
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    step = 0.02
    for _ in range(n_trials):
        i = int(rng.randint(0, n))
        j = int(rng.randint(0, n))
        if i == j:
            continue
        delta = rng.choice([-1.0, 1.0]) * step * (1 - abs(i - j) / max(n, 1) + 0.3)
        h[i] += delta
        h[j] -= delta
        h2 = project_feasible(h, n)
        c2 = compute_c5(h2)
        if c2 < best_c - 1e-15:
            best_c = c2
            best_h = h2.copy()
            h = h2
        elif c2 < c + 0.001:
            h = h2
            c = c2
        else:
            h = best_h if rng.rand() < 0.3 else h
            c = compute_c5(h)
        step = max(0.003, step * 0.999)
    return best_h, best_c

def block_perturb(h0, n, n_trials, seed=42):
    rng = np.random.RandomState(int(seed))
    h = project_feasible(h0, n).copy()
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    for it in range(n_trials):
        L = int(rng.randint(2, min(12, n)))
        st = int(rng.randint(0, max(1, n - L)))
        delta = rng.randn(L) * 0.015
        delta -= delta.mean()
        h_try = h.copy()
        h_try[st:st + L] += delta
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try)
        if c_try < best_c - 1e-15:
            best_c = c_try
            best_h = h_try.copy()
            h = h_try
        elif c_try < c and rng.rand() < 0.8:
            h = h_try
            c = c_try
    return best_h, best_c

def smooth_optimize_slsqp(h0, n, beta, maxiter=60):
    h = project_feasible(h0, n)

    def obj_fn(x):
        return smooth_max_obj(x, beta)

    def jac_fn(x):
        return smooth_max_grad(x, beta)

    def eq_fn(x):
        return float(np.sum(x) - 0.5 * float(n))

    def eq_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": eq_fn, "jac": eq_jac}]
    try:
        res = minimize(obj_fn, h, jac=jac_fn, method="L-BFGS-B",
                       bounds=bounds, constraints=cons,
                       options={"maxiter": maxiter, "ftol": 1e-12, "maxls": 50})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def run(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    budget_s = float(budget_s)
    start_time = time.time()
    hard_end = start_time + max(60.0, budget_s - 30.0)

    # Get parent
    parent = None
    try:
        vals = initial_h_values
        if vals is not None:
            arr = np.asarray(vals, dtype=float).copy()
            if arr.size > 0:
                parent = arr
    except Exception:
        parent = None

    if parent is not None:
        best_n = int(parent.size)
        best_h = project_feasible(parent, best_n)
        best_c = compute_c5(best_h)
    else:
        best_n = 160
        best_h = project_feasible(np.full(best_n, 0.5), best_n)
        best_c = compute_c5(best_h)

    # Phase 1: Smooth optimization at parent resolution
    phase1_end = min(hard_end, start_time + budget_s * 0.15)
    for beta in [30, 80, 200, 500]:
        if time.time() > phase1_end - 5:
            break
        h_new = smooth_optimize_slsqp(best_h, best_n, beta, maxiter=50)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_h = h_new
            best_c = c_new
        # Also try FISTA
        h_f, c_f = fidsa_smooth(best_h, best_n, beta, n_iter=100)
        if c_f < best_c:
            best_h = h_f
            best_c = c_f

    # Phase 2: Multi-resolution active set optimization
    resolutions = [best_n]
    for cand_n in [120, 160, 200, 240, 280, 320, 360, 400]:
        if cand_n not in resolutions and 80 <= cand_n <= 450:
            resolutions.append(cand_n)
    resolutions = sorted(set(resolutions))
    # Put best_n first, then nearby
    resolutions = [best_n] + [r for r in resolutions if r != best_n]

    phase2_end = min(hard_end, start_time + budget_s * 0.75)
    for nn in resolutions:
        if time.time() > phase2_end - 30:
            break
        rem = phase2_end - time.time()
        if rem < 40:
            break
        # Interpolate to this resolution
        if nn == best_n:
            h_cur = best_h.copy()
        else:
            h_cur = interpolate_to_n(best_h, nn)
            h_cur = project_feasible(h_cur, nn)
        c_cur = compute_c5(h_cur)

        # Smooth optimization
        for beta in [40, 120, 300]:
            if time.time() > phase2_end - 30:
                break
            h_s = smooth_optimize_slsqp(h_cur, nn, beta, maxiter=40)
            c_s = compute_c5(h_s)
            if c_s < c_cur:
                h_cur = h_s
                c_cur = c_s

        # Active set passes
        for acnt in [16, 32, 64]:
            if time.time() > phase2_end - 20:
                break
            for _ in range(2):
                if time.time() > phase2_end - 10:
                    break
                vals = get_all_lag_values(h_cur)
                lags = make_lag_array(vals)
                current_c = float(np.max(vals))
                active = choose_active(vals, lags, acnt)
                t0 = current_c + 1e-6
                h_act = active_set_optimize(h_cur, nn, active, t0, maxiter=70)
                if h_act is not None:
                    c_act = compute_c5(h_act)
                    if c_act < c_cur:
                        h_cur = h_act
                        c_cur = c_act

        if c_cur < best_c:
            best_h = h_cur.copy()
            best_c = c_cur
            best_n = nn

    # Phase 3: Coordinate descent and block perturb at best resolution
    phase3_end = min(hard_end, start_time + budget_s * 0.9)
    if time.time() < phase3_end:
        h_cd, c_cd = coordinate_descent(best_h, best_n, int(min(3000, (phase3_end - time.time()) * 10)), seed=seed)
        if c_cd < best_c:
            best_h = h_cd
            best_c = c_cd

    if time.time() < phase3_end:
        h_bl, c_bl = block_perturb(best_h, best_n, int(min(2000, (phase3_end - time.time()) * 10)), seed=seed + 1)
        if c_bl < best_c:
            best_h = h_bl
            best_c = c_bl

    # Phase 4: Final active set polish
    if time.time() < hard_end - 15:
        vals = get_all_lag_values(best_h)
        lags = make_lag_array(vals)
        current_c = float(np.max(vals))
        for acnt in [48, 96]:
            if time.time() > hard_end - 10:
                break
            active = choose_active(vals, lags, acnt)
            h_act = active_set_optimize(best_h, best_n, active, current_c + 1e-7, maxiter=100)
            if h_act is not None:
                c_act = compute_c5(h_act)
                if c_act < best_c:
                    best_h = h_act
                    best_c = c_act
                    current_c = c_act
                vals = get_all_lag_values(best_h)

    # Final: one more smooth pass if time
    if time.time() < hard_end - 8 and best_n <= 320:
        h_s = smooth_optimize_slsqp(best_h, best_n, 400, maxiter=40)
        c_s = compute_c5(h_s)
        if c_s < best_c:
            best_h = h_s
            best_c = c_s

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)
    return (best_h, float(best_c), int(best_n))
