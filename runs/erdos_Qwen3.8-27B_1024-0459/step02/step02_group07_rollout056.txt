```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = int(np.count_nonzero(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-15
            cnt = int(np.count_nonzero(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def coord_opt_j(h, j, n, corr):
    d_range = np.arange(-(n - 1), n)
    A = np.zeros(2 * n - 1)
    for idx, d in enumerate(d_range):
        d = int(d)
        coeff = 0.0
        jd = j - d
        if 0 <= jd < n:
            coeff += 1.0 - h[jd]
        jdd = j + d
        if 0 <= jdd < n:
            coeff -= h[jdd]
        A[idx] = coeff
    B = corr - A * h[j]
    lo, hi = 0.0, 1.0
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        vals = B + A * mid
        mx = np.max(vals)
        active = vals >= mx - 1e-15
        if not np.any(active):
            break
        sg = np.mean(A[active])
        if sg > 1e-14:
            hi = mid
        elif sg < -1e-14:
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)


def coord_descent(h, n, n_passes, time_limit):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    corr = get_corr(h)
    start = time.time()
    for p in range(n_passes):
        if time.time() - start > time_limit:
            break
        for j in range(n):
            t_opt = coord_opt_j(h, j, n, corr)
            h[j] = t_opt
        h = make_feasible(h, n)
        corr = get_corr(h)
        c5 = np.max(corr) * (2.0 / n)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5


def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        h = np.ones(n_dst) * 0.5
        return make_feasible(h, n_dst)
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_src)
    return make_feasible(h_interp, n_dst)


def subgradient_cd(h, n, n_steps, step_size, time_limit):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    for s in range(n_steps):
        if time.time() - start > time_limit:
            break
        corr = get_corr(h)
        top_k = np.argsort(corr)[-8:]
        d_vals = (top_k - (n - 1)).astype(int)
        grad = np.zeros(n)
        for d in d_vals:
            d = int(d)
            for j in range(n):
                jd = j - d
                jdd = j + d
                c = 0.0
                if 0 <= jd < n:
                    c += 1.0 - h[jd]
                if 0 <= jdd < n:
                    c -= h[jdd]
                grad[j] += c
        grad /= len(d_vals)
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = make_feasible(h_new, n)
        c5_new = np.max(get_corr(h_new)) * (2.0 / n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            step_size *= 0.9
            h = h_new
    return best_h, best_c5


def slsqp_polish(h, n, max_iter=2000):
    def obj(x):
        return np.max(np.correlate(x, 1.0 - x, mode='full'))

    def constraint(x):
        return np.sum(x) - n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None


def binary_search_locally(h, n, time_limit, rng):
    h_bin = (h > 0.5).astype(float)
    s = np.sum(h_bin)
    deficit = int(round(n / 2.0 - s))
    if deficit > 0:
        zeros = np.where(h_bin == 0)[0]
        idx_sorted = zeros[np.argsort(-h[zeros])][:deficit]
        h_bin[idx_sorted] = 1.0
    elif deficit < 0:
        ones = np.where(h_bin == 1)[0]
        idx_sorted = ones[np.argsort(h[ones])][:-deficit]
        h_bin[idx_sorted] = 0.0
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    start = time.time()
    max_trials = int(200000 / max(1, n / 100))
    for _ in range(max_trials):
        if time.time() - start > time_limit:
            break
        n_flip = rng.randint(1, 4)
        idxs = rng.choice(n, n_flip, replace=False)
        h_try = best_h.copy()
        for idx in idxs:
            h_try[idx] = 1.0 - h_try[idx]
        if abs(np.sum(h_try) - n / 2.0) < 1e-10:
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()
    return best_h, best_c5


def generate_structured(n, rng):
    inits = []
    # Two blocks
    h1 = np.zeros(n)
    h1[:n // 4] = 1.0
    h1[n // 2:n // 2 + n // 4] = 1.0
    inits.append(h1)
    # Alternating pairs
    if n % 4 == 0:
        h2 = np.tile([1.0, 1.0, 0.0, 0.0], n // 4)
        inits.append(h2)
    # Three blocks
    if n % 3 == 0:
        h3 = np.zeros(n)
        h3[:n // 6] = 1.0
        h3[n // 3:2 * n // 3] = 1.0
        h3[2 * n // 3 + n // 6:] = 1.0
        inits.append(make_feasible(h3, n))
    # Random
    h4 = make_feasible(rng.rand(n) * 2.0, n)
    inits.append(h4)
    # Number-theoretic
    h5 = np.zeros(n)
    positions = sorted(range(n), key=lambda i: (i * 7919) % n)
    for j in range(n // 2):
        h5[positions[j]] = 1.0
    inits.append(make_feasible(h5, n))
    return inits


def logsumexp_gradient_step(h, n, temperature, step_size):
    corr = get_corr(h)
    scaled = corr / temperature
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    weights = exp_scaled / np.sum(exp_scaled)
    d_range = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    for idx, d in enumerate(d_range):
        if weights[idx] < 1e-10:
            continue
        d = int(d)
        w = weights[idx]
        for j in range(n):
            c = 0.0
            jd = j - d
            if 0 <= jd < n:
                c += 1.0 - h[jd]
            jdd = j + d
            if 0 <= jdd < n:
                c -= h[jdd]
            grad[j] += w * c
    return h - step_size * grad
```
