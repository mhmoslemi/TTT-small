```python
def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmin(np.maximum(h, 1.0 - h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    total = np.sum(np.exp(shifted))

            grad[Lp:] -= w * x[:n - Lp]
```

```python
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            # corr = sum_{i=L}^{n-1} h[i](1-h[i-L])
            grad[L:] += w * (1.0 - h[:n-L])  # h[i] term, i from L to n-1
            grad[:n-L] -= w * h[L:]  # (1-h[i-L]) term, j from 0 to n-1-L
        else:
            Lp = -L
            # corr = sum_{i=0}^{n-1-Lp} h[i](1-h[i+Lp])
            grad[:n-Lp] += w * (1.0 - h[Lp:])  # h[i] term
            grad[Lp:] -= w * h[:n-Lp]  # (1-h[i+Lp]) term, j from Lp to n-1
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) > 1e-13:
            mask = (h > 1e-10) & (h < 1.0 - 1e-10)
            cnt = np.sum(mask)
            if cnt > 0:
                h[mask] += diff / cnt
                h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmin(np.maximum(h, 1.0 - h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def smooth_max_weights(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    exps = np.exp(shifted)
    return exps / np.sum(exps)


def overlap_gradient(h, weights, n):
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        L = k - (n - 1)
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += w * (1.0 - h[:n - L])
            grad[:n - L] -= w * h[L:]
        else:
            Lp = -L
            grad[:n - Lp] += w * (1.0 - h[Lp:])
            grad[Lp:] -= w * h[:n - Lp]
    return grad


def smooth_obj_val(h, beta):
    corr = compute_corr(h)
    return smooth_max(corr, beta) * (2.0 / len(h))


def projected_gradient_optimize(h, n, beta, max_iters, step_init, time_limit):
    h_work = h.copy()
    t_start = time.time()
    step = step_init
    best_h = h_work.copy()
    best_c5 = compute_c5(h_work)
    prev_val = smooth_obj_val(h_work, beta)

    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        corr = compute_corr(h_work)
        weights = smooth_max_weights(corr, beta)
        grad = overlap_gradient(h_work, weights, n)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        val = smooth_max(corr, beta) * (2.0 / n)
        accepted = False
        for ls in range(30):
            h_try = project_feasible(h_work - step * grad, n)
            val_try = smooth_obj_val(h_try, beta)
            if val_try < val * (1.0 - 1e-5):
                h_work = h_try
                prev_val = val_try
                accepted = True
                break
            step *= 0.5

        if not accepted:
            step *= 0.5
            if step < 1e-12:
                break

        c5_cur = compute_c5(h_work)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_work.copy()

        step = min(step * 1.5, step_init)

    return best_h, best_c5


def interp_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)


def round_to_binary(h, n):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    n_want = n // 2
    if n_ones > n_want:
        ones = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones])[:n_ones - n_want]
        h_bin[ones[order]] = 0.0
    elif n_ones < n_want:
        zeros = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros])[:n_want - n_ones]
        h_bin[zeros[order]] = 1.0
    return h_bin


def run_simulated_annealing(h_bin, n, time_budget, rng):
    h = h_bin.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_half = n // 2

    T = 0.008
    T_min = 5e-6
    cool_per_iter = 0.99985
    t_start = time.time()

    ones_cache = np.where(h == 1.0)[0].copy()
    zeros_cache = np.where(h == 0.0)[0].copy()
    cache_counter = 0
    no_improve = 0
    stall_limit = 8000

    while time.time() - t_start < time_budget:
        for _ in range(2000):
            if T < T_min:
                T = T_min

            i = rng.randint(len(ones_cache))
            j = rng.randint(len(zeros_cache))
            pos1 = ones_cache[i]
            pos0 = zeros_cache[j]

            h[pos1] = 0.0
            h[pos0] = 1.0
            new_c5 = compute_c5(h)

            if new_c5 <= best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            elif T > T_min and rng.random() < np.exp(-(new_c5 - best_c5) / T):
                no_improve += 1
            else:
                h[pos1] = 1.0
                h[pos0] = 0.0
                no_improve += 1

            ones_cache[i] = pos0
            zeros_cache[j] = pos1
            cache_counter += 1

        if cache_counter > 20000:
            ones_cache = np.where(h == 1.0)[0].copy()
            zeros_cache = np.where(h == 0.0)[0].copy()
            cache_counter = 0

        T *= cool_per_iter ** 2000

        if no_improve > stall_limit:
            n_kick = rng.randint(3, min(10, n // 10))
            kick_pos = rng.choice(n, size=n_kick, replace=False)
            for kp in kick_pos:
                if h[kp] == 1.0:
                    zi = rng.randint(len(zeros_cache))
                    zpos = zeros_cache[zi]
                    h[kp] = 0.0
                    h[zpos] = 1.0
            ones_cache = np.where(h == 1.0)[0].copy()
            zeros_cache = np.where(h == 0.0)[0].copy()
            no_improve = 0

    return best_h, best_c5

def make_m_sequence_init(n, rng):
    """Create initialization based on m-sequence pattern."""
    m = int(np.ceil(np.log2(n + 1)))
    seq_len = (1 << m) - 1
    # Simple LFSR m-sequence
    reg = 1
    taps = {m - 1: m, m - 2: m - 3} if m >= 3 else {1: 0}
    bits = []
    for _ in range(seq_len):
        bits.append(reg & 1)
        feedback = 0
        for tap in range(m):
            if (m - 1 - tap) in taps:
                feedback ^= (reg >> tap) & 1
        reg = (reg >> 1) | (feedback << (m - 1))
    seq = np.array(bits, dtype=float)
    # Repeat/truncate to n
    h = np.tile(seq, n // seq_len + 1)[:n]
    # Adjust to have exactly n/2 ones
    h = (h > 0.5).astype(float)
    n_ones = int(np.sum(h))
    if n_ones > n // 2:
        idx = np.where(h == 1)[0]
        order = rng.permutation(len(idx))[:n_ones - n // 2]
        h[idx[order]] = 0
    elif n_ones < n // 2:
        idx = np.where(h == 0)[0]
        order = rng.permutation(len(idx))[:n // 2 - n_ones]
        h[idx[order]] = 1
    return h


def make_quadratic_residue_init(n, rng):
    """Initialization based on quadratic residues."""
    p = n if n % 4 == 3 else n + 1
    if p % 4 == 3:
        qr = set()
        for x in range(1, p):
            qr.add((x * x) % p)
        h = np.zeros(n)
        for i in range(n):
            if i in qr:
                h[i] = 1.0
    else:
        h = np.zeros(n)
        h[::2] = 1.0
    n_ones = int(np.sum(h))
    if n_ones > n // 2:
        idx = np.where(h == 1)[0]
        order = rng.permutation(len(idx))[:n_ones - n // 2]
        h[idx[order]] = 0
    elif n_ones < n // 2:
        idx = np.where(h == 0)[0]
        order = rng.permutation(len(idx))[:n // 2 - n_ones]
        h[idx[order]] = 1
    return h

        n_f = len(best_h)
        # If continuous, try to push lower with high-beta gradient
        if not np.all((best_h == 0.0) | (best_h == 1.0)):
            h_f, c5_f = projected_gradient_optimize(best_h, n_f, 2000, 200, 0.01, min(40, remaining() * 0.5))
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()
            # Also try binary round
            h_rb = round_to_binary(best_h, n_f)
            c5_rb = compute_c5(h_rb)
            if c5_rb < best_c5:
                best_c5 = c5_rb
                best_h = h_rb.copy()
        # If binary, more SA
        else:
            if remaining() > 40:
                h_f, c5_f = run_simulated_annealing(best_h, n_f, min(50, remaining() * 0.5), rng)
                if c5_f < best_c5:
                    best_c5 = c5_f
                    best_h = h_f.copy()

    # Final projection and evaluation
    best_h = project_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.6f}".format(len(best_h), final_c5))
    return (best_h, final_c5, len(best_h))
```
