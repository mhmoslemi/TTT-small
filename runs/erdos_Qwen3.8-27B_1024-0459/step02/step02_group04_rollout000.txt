def project_feasible(h, n=None):
    h = np.asarray(h, dtype=float).ravel()
    if n is None: n = h.size
    n = int(n)
    if n <= 0: return np.empty(0,dtype=float)
    h = np.where(np.isfinite(h), h, 0.5)
    if h.size < n:
       h = np.concatenate((h, np.zeros(n-h.size)))
    else: h=h[:n]
    target=0.5*float(n)
    h=np.clip(h,0.0,1.0)
    s=float(np.sum(h))
    if abs(s-target) > 1e-12:
       lo=float(np.min(h)) - 1.0
       hi=float(np.max(h)) + 1.0
       for _ in range(80):
           mid=0.5*(lo+hi)
           out=np.clip(h-mid,0.0,1.0)
           ss=float(np.sum(out))
           if ss > target: lo=mid
           else: hi=mid
       h=np.clip(h-0.5*(lo+hi),0.0,1.0)
    # adjust
    for _ in range(50):
       diff=target-float(np.sum(h))
       if abs(diff) < 1e-13: break
       if diff > 0:
          mask = h < 1.0 - 1e-13
       else:
          mask = h > 1e-13
       cnt=int(np.count_nonzero(mask))
       if cnt == 0: break
       h[mask] += diff/float(cnt)
       h=np.clip(h,0.0,1.0)
    # final single adjust
    diff=target-float(np.sum(h))
    if abs(diff) > 0.0:
       if diff > 0:
          idx=int(np.argmin(1.0-h)); room=1.0-h[idx]
          if room > 0: h[idx] += min(diff, room)
       else:
          idx=int(np.argmax(h)); room=h[idx]
          if room >0: h[idx] += max(diff, -room)
       h=np.clip(h,0.0,1.0)
    return h

def project_feasible(h, n=None):
    h = np.asarray(h, dtype=float).ravel()
    if n is None:
        n = h.size
    n = int(n)
    if n <= 0:
        return np.empty(0, dtype=float)
    h = np.where(np.isfinite(h), h, 0.5)
    if h.size < n:
        h = np.concatenate((h, np.zeros(n - h.size, dtype=float)))
    else:
        h = h[:n].copy()
    target = 0.5 * float(n)
    h = np.clip(h, 0.0, 1.0)
    if abs(float(np.sum(h)) - target) > 1e-12:
        lo = float(np.min(h)) - 1.0
        hi = float(np.max(h)) + 1.0
        for _ in range(80):
            mid = 0.5 * (lo + hi)
            ss = float(np.sum(np.clip(h - mid, 0.0, 1.0)))

            else:
                hi = mid

    for _ in range(40):

        if abs(diff) < 1e-13:
            break
        if diff > 0.0:

        if cnt == 0:
            break

def compute_c5(h):

def gaussian_smooth(h, sigma=1.5):
    h = np.asarray(h, dtype=float).ravel()
    n = int(h.size)
    if n <= 1:
        return h.copy()
    length = min(7, n)
    if length % 2 == 0:

    length = max(1, length)
    half = length // 2
    k = np.arange(-half, half + 1, dtype=float)
    w = np.exp(-(k * k) / (2.0 * float(sigma) * float(sigma)))
    w /= w.sum()
    return np.convolve(h, w, mode="same")

def half_masks(p):
    p = int(p)
    if p < 0:
        return
    k = p // 2
    if k == 0:
        yield 0
        return
    if k == p:
        yield (1 << p) - 1
        return
    mask = (1 << k) - 1
    limit = 1 << p
    while mask < limit:
        yield mask
        c = mask & -mask
        r = mask + c
        mask = (((r ^ mask) >> 2) // c) | r


def mask_to_array(p, mask):
    p = int(p)
    out = np.empty(p, dtype=float)
    for i in range(p):
        out[i] = 1.0 if ((int(mask) >> i) & 1) else 0.0
    return out

    return rep

        hs, _cs, _ns = low_sols[0]
        out.append(interpolate_to_n(hs, nn))
    if extra_starts is not None:
        for hh in extra_starts[:1]:
            out.append(np.asarray(hh, dtype=float).ravel())
    if len(out) == 0:
        out.append(np.full(nn, 0.5, dtype=float))
    src = project_feasible(out[0], nn)
    out.append(project_feasible(gaussian_smooth(src, 1.3), nn))
    out.append(project_feasible(rng.permutation(src), nn))
    out.append(make_binary_from_values(src, nn))
    out.append(random_binary(nn, rng))
    st = structured_starts(nn, rng, count=1)
    if st:
        out.extend(st)
    if len(out) > count:
        out = out[:count]
    return out

def lag_value(h, l):
    h = np.asarray(h, dtype=float)
    n = len(h)
    L = abs(int(l))
    if L == 0:
        return float(np.sum(h) - np.dot(h, h))
    m = n - L
    low = h[:m]
    high = h[L:]
    ac = float(np.dot(low, high))
    if int(l) > 0:
        return float(np.sum(low) - ac)
    return float(np.sum(high) - ac)


def lag_gradient(h, l):
    h = np.asarray(h, dtype=float)
    n = len(h)
    L = abs(int(l))
    g = np.zeros(n, dtype=float)
    if L == 0:
        return 1.0 - 2.0 * h
    m = n - L
    low = h[:m]
    high = h[L:]
    if int(l) > 0:
        g[:m] += 1.0 - high
        g[L:] += -low
    else:
        g[L:] += 1.0 - low
        g[:m] += -high
    return g


def all_lag_values_lags(h):
    h = np.asarray(h, dtype=float).ravel()
    n = int(h.size)
    dx = 2.0 / float(n)
    if n == 1:
        vals = np.array([(float(np.sum(h)) - float(np.dot(h, h))) * dx])
        lags = np.array([0])
        return vals, lags
    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = float(pref[-1])
    ac = np.correlate(h, h, mode="full")
    vals = np.empty(2 * n - 1, dtype=float)
    lags = np.empty(2 * n - 1, dtype=int)
    vals[0] = (total - float(ac[n - 1])) * dx
    lags[0] = 0
    idx = 1
    for L in range(1, n):
        a = float(ac[n - 1 + L])
        vals[idx] = (float(pref[n - L]) - a) * dx
        lags[idx] = L
        idx += 1
        vals[idx] = (total - float(pref[L]) - a) * dx
        lags[idx] = -L
        idx += 1
    return vals, lags


def choose_active_lags(vals, lags, max_count, margin=1e-5):
    if vals.size == 0:
        return []
    order = np.argsort(-vals)
    take = min(int(max_count), int(vals.size))
    idx = order[:take]
    if margin > 0.0:
        top = float(vals[order[0]])
        near = np.where(vals >= top - margin)[0]
        if near.size > 0:
            idx = np.unique(np.concatenate((idx, near)))
            idx = idx[np.argsort(-vals[idx])]
            cap = max(int(max_count) * 3, 64)
            if idx.size > cap:
                idx = idx[:cap]
    return [int(x) for x in lags[idx]]

def h_sum_eq(x):
    return float(np.sum(x) - 0.5 * float(len(x)))


def h_sum_eq_jac(x):
    return np.ones(len(x), dtype=float)


def smooth_active_obj(x):
    h = np.asarray(x, dtype=float).ravel()
    n = int(h.size)
    m = len(GLOBAL_S_LAGS)
    if m == 0 or n <= 0:
        return 0.0
    dx = 2.0 / float(n)
    vals = np.empty(m, dtype=float)
    for i, ll in enumerate(GLOBAL_S_LAGS):
        vals[i] = lag_value(h, ll) * dx
    mx = float(np.max(vals))
    beta = max(1.0, float(GLOBAL_S_BETA))
    r = 0.0
    for v in vals:
        d = beta * (float(v) - mx)
        if d > -745.0:
            r += math.exp(d)
    if r <= 0.0:
        return mx
    return mx + math.log(r) / beta


def smooth_active_jac(x):
    h = np.asarray(x, dtype=float).ravel()
    n = int(h.size)
    m = len(GLOBAL_S_LAGS)
    if m == 0 or n <= 0:
        return np.zeros(0, dtype=float)
    dx = 2.0 / float(n)
    vals = np.empty(m, dtype=float)
    for i, ll in enumerate(GLOBAL_S_LAGS):
        vals[i] = lag_value(h, ll) * dx
    mx = float(np.max(vals))
    beta = max(1.0, float(GLOBAL_S_BETA))
    r = 0.0
    for v in vals:
        d = beta * (float(v) - mx)
        if d > -745.0:
            r += math.exp(d)
    if r <= 0.0:
        return np.zeros(n, dtype=float)
    g = np.zeros(n, dtype=float)
    for i, ll in enumerate(GLOBAL_S_LAGS):
        d = beta * (float(vals[i]) - mx)
        if d <= -745.0:
            continue
        w = math.exp(d) / r
        if w > 0.0:
            g += w * lag_gradient(h, ll)
    return dx * g


def smooth_active_sum_eq(x):
    return h_sum_eq(x)


def smooth_active_sum_eq_jac(x):
    return h_sum_eq_jac(x)


def optimize_smooth_active(h0, lags, beta, maxiter=30, ftol=1e-11):
    global GLOBAL_S_LAGS, GLOBAL_S_BETA
    h = project_feasible(h0)
    if len(lags) == 0:
        return h
    GLOBAL_S_LAGS = [int(ll) for ll in lags]
    GLOBAL_S_BETA = max(1.0, float(beta))
    n = h.size
    bounds = [(0.0, 1.0) for _ in range(n)]
    constraints = [{"type": "eq", "fun": smooth_active_sum_eq, "jac": smooth_active_sum_eq_jac}]
    try:
        res = minimize(
            smooth_active_obj,
            h,
            jac=smooth_active_jac,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options={"maxiter": int(maxiter), "ftol": float(ftol), "disp": False},
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def active_t_obj(x):
    return float(x[-1])


def active_t_obj_jac(x):
    j = np.zeros(len(x), dtype=float)
    j[-1] = 1.0
    return j


def active_sum_eq(x):
    n = len(x) - 1
    return float(np.sum(x[:-1]) - 0.5 * float(n))


def active_sum_eq_jac(x):
    j = np.zeros(len(x), dtype=float)
    j[:-1] = 1.0
    return j


def active_ineq(x):
    n = len(x) - 1
    dx = 2.0 / float(n)
    h = x[:-1]
    m = len(GLOBAL_ACT_LAGS)
    out = np.empty(m, dtype=float)
    for i, ll in enumerate(GLOBAL_ACT_LAGS):
        out[i] = float(x[-1]) - lag_value(h, ll) * dx
    return out


def active_ineq_jac(x):
    n = len(x) - 1
    dx = 2.0 / float(n)
    h = x[:-1]
    m = len(GLOBAL_ACT_LAGS)
    J = np.zeros((m, n + 1), dtype=float)
    for i, ll in enumerate(GLOBAL_ACT_LAGS):
        g = lag_gradient(h, ll)
        J[i, :n] = -dx * g
        J[i, n] = 1.0
    return J


def optimize_active_minimax(h0, lags, t0, maxiter=40, ftol=1e-11):
    global GLOBAL_ACT_LAGS
    h = project_feasible(h0)
    if len(lags) == 0:
        return h
    GLOBAL_ACT_LAGS = [int(ll) for ll in lags]
    n = h.size
    x0 = np.empty(n + 1, dtype=float)
    x0[:n] = h
    x0[n] = float(np.clip(t0, 0.0, 1.0))
    bounds = [(0.0, 1.0) for _ in range(n)]
    bounds.append((0.0, 1.0))
    constraints = [
        {"type": "eq", "fun": active_sum_eq, "jac": active_sum_eq_jac},
        {"type": "ineq", "fun": active_ineq, "jac": active_ineq_jac},
    ]
    try:
        res = minimize(
            active_t_obj,
            x0,
            jac=active_t_obj_jac,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options={"maxiter": int(maxiter), "ftol": float(ftol), "disp": False},
        )
        if res.x is not None and res.x.size == n + 1 and np.all(np.isfinite(res.x)):
            return project_feasible(res.x[:-1], n)
    except Exception:
        pass
    return h


def refine_active(h0, n, time_limit, seq=None, seed=1):
    n = int(n)
    h = project_feasible(h0, n)
    c = compute_c5(h)
    if not math.isfinite(c):
        h = project_feasible(np.full(n, 0.0), n)
        c = compute_c5(h)
    best_h = h.copy()
    best_c = float(c)
    cur_c = float(c)
    if seq is None:
        seq = active_seq_for_n(n)
    maxit = maxiter_for_n(n)
    maxit_smooth = max(12, maxit - 12)
    for acnt in seq:
        if time.time() > time_limit - 5.0:
            break
        passes = 2 if int(acnt) <= 16 else 1
        for _ in range(passes):
            if time.time() > time_limit - 5.0:
                break
            vals, lags_arr = all_lag_values_lags(h)
            cur_c = float(np.max(vals))
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
            active = choose_active_lags(vals, lags_arr, int(acnt), margin=1e-5)
            if len(active) == 0:
                active = [int(lags_arr[int(np.argmax(vals))])]
            if time.time() < time_limit - 12.0 and n <= 320:
                betas = (20.0, 60.0) if n <= 200 else (30.0,)
                for beta in betas:
                    if time.time() > time_limit - 8.0:
                        break
                    h2 = optimize_smooth_active(h, active, beta, maxiter=maxit_smooth)
                    c2 = compute_c5(h2)
                    if c2 < best_c:
                        best_c = c2
                        best_h = h2.copy()
                    if c2 <= cur_c + 1e-11:
                        h = h2.copy()
                        cur_c = c2
            if time.time() < time_limit - 8.0 and n <= 320:
                t0 = max(cur_c + 1e-7, 1e-8)
                h2 = optimize_active_minimax(h, active, t0, maxiter=maxit)
                c2 = compute_c5(h2)
                if c2 < best_c:
                    best_c = c2
                    best_h = h2.copy()
                if c2 <= cur_c + 1e-11:
                    h = h2.copy()
                    cur_c = c2
            if time.time() >= time_limit - 5.0:
                break
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h)
    return best_h, float(best_c)


def gradient_polish(h0, n, time_limit, seed=1, max_iters=50):
    n = int(n)
    h = project_feasible(h0, n)
    c = compute_c5(h)
    if not math.isfinite(c):
        return h, c
    best_h = h.copy()
    best_c = float(c)
    cur_c = float(c)
    stagnant = 0
    step_bases = (0.10, 0.05, 0.025, 0.0125, 0.00625, 0.003, 0.0015, 0.0008)
    for _ in range(int(max_iters)):
        if time.time() > time_limit - 2.0:
            break
        vals, lags_arr = all_lag_values_lags(h)
        cur_c = float(np.max(vals))
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        active = choose_active_lags(vals, lags_arr, 24, margin=1e-4)
        vs = []
        for ll in active:
            found = False
            for j in range(len(lags_arr)):
                if int(lags_arr[j]) == int(ll):
                    vs.append(float(vals[j]))
                    found = True
                    break
            if not found:
                vs.append(float(np.max(vals)))
        v = np.asarray(vs, dtype=float)
        vmax = float(np.max(v))
        vmin = float(np.min(v))
        gap = vmax - vmin
        if gap < 1e-8:
            temp = 0.02
        else:
            temp = max(1e-4, 0.25 * gap)
        weights = np.exp((v - vmax) / temp)
        if float(weights.sum()) <= 0.0:
            weights = np.ones_like(weights)
        weights = weights / float(weights.sum())
        dx = 2.0 / float(n)
        g = np.zeros(n, dtype=float)
        for ll, w in zip(active, weights):
            if float(w) > 0.0:
                g += float(w) * lag_gradient(h, ll)
        g *= dx
        g -= float(np.mean(g))
        gn = float(np.linalg.norm(g))
        if gn < 1e-12:
            break
        moved = False
        for base in step_bases:
            if time.time() > time_limit - 2.0:
                break
            d = -base * g / gn
            h_try = project_feasible(h + d, n)
            c_try = compute_c5(h_try)
            if c_try < best_c - 1e-13:
                best_c = c_try
                best_h = h_try.copy()
                h = h_try
                cur_c = c_try
                moved = True
                break
            if c_try < cur_c:
                h = h_try
                cur_c = c_try
                moved = True
                break
        if moved:
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 6:
                break
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h)
    return best_h, float(best_c)


def improve_candidate(start_h, n, time_limit, seed=1):
    n = int(n)
    h = project_feasible(start_h, n)
    c = compute_c5(h)
    if not math.isfinite(c):
        h = project_feasible(np.full(n, 0.0), n)
        c = compute_c5(h)
    best_h = h.copy()
    best_c = float(c)
    if 2 <= n <= 320 and time.time() < time_limit - 20.0:
        h2, c2 = refine_active(h, n, time_limit, seed=seed)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = c2
    if 2 <= n <= 320 and time.time() < time_limit - 8.0:
        h2, c2 = gradient_polish(best_h, n, time_limit, seed + 7, max_iters=35)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = c2
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h)
    return best_h, float(best_c)


def reorder_search(h0, n, time_limit, seed=1):
    n = int(n)
    if n <= 1:
        return h0, compute_c5(h0)
    rng = np.random.RandomState(int(seed))
    cur = project_feasible(h0, n)
    best = cur.copy()
    bc = compute_c5(cur)
    cc = bc
    temp = 0.0035
    min_temp = 1e-7
    no_improve = 0
    trial = 0
    while time.time() < time_limit - 1.0:
        trial += 1
        h_try = cur.copy()
        changed = False
        op = rng.rand()
        if op < 0.45:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i != j:
                h_try[i], h_try[j] = h_try[j], h_try[i]
                changed = True
        elif op < 0.65:
            maxL = min(5, n // 2)
            if maxL >= 1:
                L = 1 + int(rng.randint(maxL))
                if 2 * L <= n and n - 2 * L >= 0:
                    i = int(rng.randint(n - 2 * L + 1))
                    j = i + L + int(rng.randint(max(1, n - 2 * L + 1)))
                    if j + L <= n:
                        tmp1 = h_try[i:i + L].copy()
                        tmp2 = h_try[j:j + L].copy()
                        h_try[i:i + L] = tmp2
                        h_try[j:j + L] = tmp1
                        changed = True
        elif op < 0.80:
            maxL = min(16, n - 1)
            if maxL >= 2:
                L = 2 + int(rng.randint(maxL - 1))
                if L < n:
                    st = int(rng.randint(max(1, n - L)))
                    h_try[st:st + L] = h_try[st:st + L][::-1]
                    changed = True
        elif op < 0.90:
            maxL = min(12, n - 1)
            if maxL >= 1:
                L = 1 + int(rng.randint(maxL))
                if L < n:
                    st = int(rng.randint(max(1, n - L)))
                    seg = h_try[st:st + L].copy()
                    rest = np.concatenate((h_try[:st], h_try[st + L:]))
                    ins = int(rng.randint(len(rest) + 1))
                    h_try = np.insert(rest, ins, seg)
                    changed = True
        else:
            if no_improve > 2000:
                h_try = best.copy()
                for _ in range(min(n, 30)):
                    i = int(rng.randint(n))
                    j = int(rng.randint(n))
                    if i != j:
                        h_try[i], h_try[j] = h_try[j], h_try[i]
            else:
                sh = 1 + int(rng.randint(max(1, n // 2)))
                h_try = np.roll(cur, sh)
            changed = True
        if not changed:
            continue
        ct = compute_c5(h_try)
        if not math.isfinite(ct):
            continue
        if ct < bc - 1e-14:
            best = h_try.copy()
            bc = float(ct)
            cur = h_try.copy()
            cc = float(ct)
            no_improve = 0
        elif ct < cc:
            cur = h_try.copy()
            cc = float(ct)
            no_improve += 1
        else:
            diff = float(ct) - cc
            if diff < temp:
                prob = math.exp(-diff / temp)
            else:
                prob = 0.0
            if rng.rand() < prob:
                cur = h_try.copy()
                cc = float(ct)
            no_improve += 1
        if trial % 200 == 0:
            temp = max(min_temp, temp * 0.985)
        if no_improve > 3000:
            cur = best.copy()
            for _ in range(min(n, 40)):
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i != j:
                    cur[i], cur[j] = cur[j], cur[i]
            cc = compute_c5(cur)
            no_improve = 0
            temp = 0.0045
    best = project_feasible(best, n)
    bc = compute_c5(best)
    return best, float(bc)


def cont_perturb(h0, n, time_limit, seed=1, step=0.02):
    n = int(n)
    if n <= 1:
        return h0, compute_c5(h0)
    rng = np.random.RandomState(int(seed))
    cur = project_feasible(h0, n)
    best = cur.copy()
    bc = compute_c5(cur)
    cc = bc
    temp = 0.002
    min_temp = 1e-7
    no_improve = 0
    trials = 0
    while time.time() < time_limit - 1.0 and trials < 8000:
        trials += 1
        h_try = cur.copy()
        if n > 1:
            L = 1 + int(rng.randint(min(8, n)))
            L = min(L, n)
            st = int(rng.randint(max(1, n - L + 1)))
            delta = rng.randn(L) * float(step)
            delta -= float(delta.mean())
            h_try[st:st + L] += delta
        if rng.rand() < 0.02:
            h_try = gaussian_smooth(cur, 1.0)
        h_try = project_feasible(h_try, n)
        ct = compute_c5(h_try)
        if not math.isfinite(ct):
            continue
        if ct < bc - 1e-14:
            best = h_try.copy()
            bc = float(ct)
            cur = h_try.copy()
            cc = float(ct)
            no_improve = 0
        elif ct < cc:
            cur = h_try.copy()
            cc = float(ct)
            no_improve += 1
        else:
            diff = ct - cc
            prob = math.exp(-diff / temp) if diff < temp else 0.0
            if rng.rand() < prob:
                cur = h_try.copy()
                cc = float(ct)
            no_improve += 1
        temp = max(min_temp, temp * 0.997)
        if no_improve > 2000:
            cur = best.copy()
            cur += rng.randn(n) * 0.03
            cur = project_feasible(cur, n)
            cc = compute_c5(cur)
            no_improve = 0
            temp = 0.003
    best = project_feasible(best, n)
    bc = compute_c5(best)
    return best, float(bc)

        best_n = 160
        best_h = project_feasible(np.full(best_n, 0.5, dtype=float), best_n)
        best_c = compute_c5(best_h)

    low_sols = []
    keep_low_solution(low_sols, best_h, best_c, best_n, 4)

    reserve = min(360.0, max(40.0, budget_s * 0.28))
    if hard_end - start_time > 120.0:
        main_end = hard_end - reserve
    else:
        main_end = start_time + max(20.0, (hard_end - start_time) * 0.5)

    per_ns = []
    for nn in (int(best_n), 96, 128, 144, 160, 192, 256):
        if 24 <= nn <= 320 and nn not in per_ns:
            per_ns.append(nn)
    per_ns = per_ns[:4]
    if per_ns and time.time() < main_end - 60.0:
        per_window = min(60.0, max(8.0, (main_end - time.time()) * 0.20))
        per_slot = max(2.0, per_window / float(len(per_ns)))
        per_time = 16 if budget_s > 700 else 12
        for pn2 in per_ns:
            if time.time() > main_end - 45.0:
                break
            limit = min(time.time() + per_slot, main_end - 30.0)
            plist = periodic_starts(pn2, max_p=per_time, count=3, time_limit=limit)
            if plist:
                for ph in plist:
                    hp = project_feasible(ph, pn2)
                    pc = compute_c5(hp)
                    if pc < best_c:
                        best_h = hp.copy()
                        best_c = pc
                        best_n = pn2
                    keep_low_solution(low_sols, hp, pc, pn2, 4)

    raw_cands = candidate_ns(best_n)
    for pn2 in per_ns:
        raw_cands.append(pn2)
    unique_cands = []
    seen_cands = set()
    for nn in raw_cands:
        nn = int(nn)
        if 24 <= nn <= 320 and nn not in seen_cands:
            seen_cands.add(nn)
            unique_cands.append(nn)

    jobs = []
    jid = 0
    build_end = min(main_end, time.time() + min(70.0, max(25.0, (main_end - time.time()) * 0.15)))
    for nn in unique_cands:
        if time.time() > build_end:
            break
        extra = []
        starts = build_starts_for_n(nn, parent, best_h, best_n, low_sols, rng, extra_starts=extra, count=6)
        evals = []
        for st in starts:
            stp = project_feasible(st, nn)
            c = compute_c5(stp)
            if c < best_c:
                best_h = stp.copy()
                best_c = c
                best_n = nn
            keep_low_solution(low_sols, stp, c, nn, 4)
            evals.append((c, len(evals) + 1, stp))
        if not evals:
            stp = project_feasible(np.full(nn, 0.5, dtype=float), nn)
            c = compute_c5(stp)
            evals.append((c, 1, stp))
        evals.sort()
        st1 = evals[0][2]
        st2 = evals[1][2] if len(evals) > 1 else st1
        score = evals[0][0] + 0.008 * abs(float(nn) - float(best_n)) / max(1.0, float(best_n))
        jobs.append((score, jid, nn, st1, st2, len(evals)))
        jid += 1
    jobs.sort()

    processed_job_ids = set()
    for job in jobs:
        if time.time() >= main_end - 25.0:
            break
        score, jid2, nn, st1, st2, nstarts = job
        if jid2 in processed_job_ids:
            continue
        processed_job_ids.add(jid2)
        rem = main_end - time.time()
        if rem < 45.0:
            break
        slot = min(110.0, rem)
        if slot >= 80.0 and nstarts > 1 and st2 is not None:
            per = slot / 2.0
            starts = (st1, st2)
        else:
            per = slot
            starts = (st1,)
        for k, hs in enumerate(starts):
            if time.time() >= main_end - 20.0:
                break
            limit = min(time.time() + per, main_end - 5.0)
            hh, cc = improve_candidate(hs, nn, limit, seed + 1000 + jid2 * 7 + k)
            if math.isfinite(cc) and cc < best_c:
                best_h = np.asarray(hh, dtype=float).copy()
                best_c = float(cc)
                best_n = int(nn)
            keep_low_solution(low_sols, hh, cc, int(nn), 4)

    if time.time() < hard_end - 30.0 and 2 <= best_n <= 320:
        rem = hard_end - time.time()
        lim = time.time() + min(35.0, rem - 30.0)
        if lim > time.time() + 4.0:
            hh, cc = gradient_polish(best_h, best_n, lim, seed + 9001, max_iters=45)
            if math.isfinite(cc) and cc < best_c:
                best_h = hh.copy()
                best_c = cc
                best_n = int(best_n)

    if time.time() < hard_end - 30.0 and 2 <= best_n <= 320:
        rem = hard_end - time.time()
        lim = time.time() + min(150.0, rem - 30.0)
        if lim > time.time() + 10.0:
            hh, cc = reorder_search(best_h, best_n, lim, seed + 9002)
            if math.isfinite(cc) and cc < best_c:
                best_h = hh.copy()
                best_c = cc
                best_n = int(best_n)

    if time.time() < hard_end - 35.0 and 2 <= best_n <= 320:
        rem = hard_end - time.time()
        lim = time.time() + min(80.0, rem - 35.0)
        if lim > time.time() + 15.0:
            hh, cc = improve_candidate(best_h, best_n, lim, seed + 9003)
            if math.isfinite(cc) and cc < best_c:
                best_h = hh.copy()
                best_c = cc
                best_n = int(best_n)

    if time.time() < hard_end - 15.0 and 2 <= best_n <= 320:
        rem = hard_end - time.time()
        lim = time.time() + min(60.0, rem - 15.0)
        if lim > time.time() + 8.0:
            hh, cc = reorder_search(best_h, best_n, lim, seed + 9004)
            if math.isfinite(cc) and cc < best_c:
                best_h = hh.copy()
                best_c = cc
                best_n = int(best_n)

    if time.time() < hard_end - 10.0 and 2 <= best_n <= 320:
        rem = hard_end - time.time()
        lim = time.time() + min(30.0, rem - 10.0)
        if lim > time.time() + 5.0:
            hh, cc = cont_perturb(best_h, best_n, lim, seed + 9005, step=0.018)
            if math.isfinite(cc) and cc < best_c:
                best_h = hh.copy()
                best_c = cc
                best_n = int(best_n)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)
    if not math.isfinite(best_c):
        best_h = project_feasible(np.full(160, 0.5, dtype=float), 160)
        best_c = compute_c5(best_h)
        best_n = 160
    return (best_h, float(best_c), int(best_n))
