def softmax_max_and_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    pref = np.concatenate([[0.0], np.cumsum(h)])
    total = pref[-1]
    
    # Collect all lag values
    all_vals = []
    all_vals.append((total - np.dot(h, h)) * dx)  # lag 0
    
    for k in range(1, n):
        m = n - k
        ac = np.dot(h[:m], h[k:])
        all_vals.append((pref[m] - ac) * dx)  # lag +k
        all_vals.append((total - pref[k] - ac) * dx)  # lag -k
    
    all_vals = np.array(all_vals)
    max_v = np.max(all_vals)
    
    # Softmax
    scaled = beta * (all_vals - max_v)
    exps = np.exp(np.where(scaled > -700, scaled, -700))
    sum_exp = np.sum(exps)
    sm_max = max_v + np.log(sum_exp) / beta
    weights = exps / sum_exp
    
    # Gradient
    grad = np.zeros(n)
    # Lag 0: dx * (1 - 2h)
    grad += weights[0] * dx * (1 - 2*h)
    
    for k in range(1, n):

        idx_neg = idx_pos + 1  # index for lag -k
        m = n - k

        w_pos = weights[idx_pos] * dx
        if w_pos > 1e-12:
            grad[:m] += w_pos * (1 - h[k:])
            grad[k:] += -w_pos * h[:m]
        
        # Gradient of lag -k: dx * [(-h[k:]) in first k, (1-h[:m]) in last m]
        w_neg = weights[idx_neg] * dx
        if w_neg > 1e-12:

            # d/dh[i] for i < k: -h[i+k]
            grad[k:] += w_neg * (1 - h[:m])
            grad[:k] += -w_neg * h[k:]
    
    return sm_max, grad

    h = np.asarray(x, dtype=float)
    n = len(h)
    dx = 2.0 / float(n)
    vals = fast_lag_values(h)
    max_v = float(np.max(vals))
    scaled = beta * (vals - max_v)
    exps = np.exp(np.clip(scaled, -700.0, 0.0))
    sum_exp = float(np.sum(exps))
    if sum_exp < 1e-300:
        sum_exp = 1e-300
    obj = max_v + np.log(sum_exp) / beta
    sm = float(np.sum(h))
    obj += mu * (sm - 0.5 * n) ** 2
    weights = exps / sum_exp
    grad = np.zeros(n, dtype=float)
    grad += 2.0 * mu * (sm - 0.5 * n)
    for lag_idx in np.where(weights > 1e-10)[0]:
        w = float(weights[lag_idx]) * dx
        if w < 1e-14:
            continue
        lag = int(lag_idx) - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            m = n - lag
            grad[:m] += w * (1.0 - h[lag:])
            grad[lag:] -= w * h[:m]
        else:
            L = -lag
            m = n - L
            grad[L:] += w * (1.0 - h[:m])
            grad[:L] -= w * h[L:]
    return obj, grad

def _sm_obj(x, beta, mu):
    return softmax_obj_and_grad(x, beta, mu)[0]

def _sm_grad(x, beta, mu):
    return softmax_obj_and_grad(x, beta, mu)[1]

def optimize_smooth_lbfgs(h0, n, betas=(8, 25, 60, 150, 300), maxiter=60):
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    for beta in betas:
        def obj_func(x):
            return softmax_obj_and_grad(x, float(beta), 500.0)[0]
        def grad_func(x):
            return softmax_obj_and_grad(x, float(beta), 500.0)[1]
        try:
            res = minimize(obj_func, h, jac=grad_func, method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': int(maxiter), 'ftol': 1e-15, 'gtol': 1e-11, 'maxls': 40})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h2 = project_feasible(res.x, n)
                c2 = compute_c5(h2)
                if c2 < best_c:
                    best_h = h2.copy()
                    best_c = c2
                h = h2
        except Exception:
            pass
    return best_h, best_c

def lag_value_single(h, k):
    n = len(h)
    dx = 2.0 / float(n)
    if k == 0:
        return dx * (float(np.sum(h)) - float(np.dot(h, h)))
    L = abs(int(k))
    m = n - L
    ac = float(np.dot(h[:m], h[L:]))
    if k > 0:
        return dx * (float(np.sum(h[:m])) - ac)
    else:
        return dx * (float(np.sum(h[L:])) - ac)

def lag_grad_single(h, k):
    n = len(h)
    dx = 2.0 / float(n)
    g = np.zeros(n, dtype=float)
    if k == 0:
        return dx * (1.0 - 2.0 * h)
    L = abs(int(k))
    m = n - L
    if k > 0:
        g[:m] = dx * (1.0 - h[L:])
        g[L:] = -dx * h[:m]
    else:
        g[L:] = dx * (1.0 - h[:m])
        g[:L] = -dx * h[L:]
    return g

def active_set_optimize(h0, active_lags, t_init, n, maxiter=50):
    g_lags = [int(l) for l in active_lags]
    m_ac = len(g_lags)
    if m_ac == 0:
        return h0, compute_c5(h0)
    x0 = np.concatenate([h0, [t_init]])
    bounds = [(0.0, 1.0)] * (n + 1)
    def t_obj(x):
        return float(x[-1])
    def t_jac(x):
        j = np.zeros(n + 1, dtype=float)
        j[-1] = 1.0
        return j
    def sum_eq(x):
        return float(np.sum(x[:-1]) - 0.5 * n)
    def sum_eq_jac(x):
        j = np.zeros(n + 1, dtype=float)
        j[:-1] = 1.0
        return j
    def ineq(x):
        h = x[:-1]
        t = float(x[-1])
        out = np.empty(m_ac, dtype=float)
        for i, ll in enumerate(g_lags):
            out[i] = t - lag_value_single(h, ll)
        return out
    def ineq_jac(x):
        h = x[:-1]
        J = np.zeros((m_ac, n + 1), dtype=float)
        for i, ll in enumerate(g_lags):
            J[i, :n] = -lag_grad_single(h, ll)
            J[i, n] = 1.0
        return J
    constraints = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac},
                   {'type': 'ineq', 'fun': ineq, 'jac': ineq_jac}]
    try:
        res = minimize(t_obj, x0, jac=t_jac, method='SLSQP', bounds=bounds,
                       constraints=constraints, options={'maxiter': int(maxiter), 'ftol': 1e-13, 'disp': False})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h2 = project_feasible(res.x[:-1], n)
            c2 = compute_c5(h2)
            return h2, c2
    except Exception:
        pass
    return h0, compute_c5(h0)

def refine_active_set(h0, n, time_limit, active_counts=(12, 24, 48), maxiter=40):
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c = compute_c5(h)
    for acnt in active_counts:
        if time.time() > time_limit - 2.0:
            break
        for _ in range(2):
            if time.time() > time_limit - 2.0:
                break
            vals = fast_lag_values(h)
            current_c = float(np.max(vals))
            if current_c < best_c:
                best_c = current_c
                best_h = h.copy()
            order = np.argsort(-vals)
            take = min(int(acnt), len(vals))
            top_val = vals[order[0]]
            near = np.where(vals >= top_val - 1e-7)[0]
            active_idx = np.unique(np.concatenate([order[:take], near]))
            active_lags = [int(i - (n - 1)) for i in active_idx]
            t0 = current_c + 1e-6
            h2, c2 = active_set_optimize(h, active_lags, t0, n, maxiter=maxiter)
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()
            if c2 <= current_c + 1e-10:
                h = h2
    return best_h, best_c

def interpolate_to_n(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float)
    n_dst = int(n_dst)
    n_src = h_src.size
    if n_src == n_dst:
        return h_src.copy()
    if n_src <= 1:
        return np.full(n_dst, float(h_src[0]) if n_src > 0 else 0.5)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def generate_diverse_inits(n, parent, rng, count=6):
    inits = []
    if parent is not None and len(parent) > 0:
        inits.append(interpolate_to_n(parent, n))
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    # Period-1 pattern
    h = np.zeros(n)
    h[:n // 2] = 1.0
    inits.append(project_feasible(h, n))
    # Two-block
    h = np.zeros(n)
    q = n // 4
    h[:q] = 1.0
    h[n // 2:n // 2 + q] = 1.0
    inits.append(project_feasible(h, n))
    # Cosine-based
    h = 0.5 + 0.4 * np.cos(2 * np.pi * x)
    inits.append(project_feasible(h, n))
    # Random
    inits.append(project_feasible(rng.rand(n) * 1.2, n))
    # Interpolated parent + noise
    if parent is not None:
        base = interpolate_to_n(parent, n)
        inits.append(project_feasible(base + rng.randn(n) * 0.03, n))
    # Gaussian smoothed parent
    if parent is not None:
        base = interpolate_to_n(parent, n)
        k = np.array([np.exp(-0.5 * (i / 2.0) ** 2) for i in range(-3, 4)])
        k /= k.sum()
        inits.append(project_feasible(np.convolve(base, k, mode='same'), n))
    out = []
    for h in inits:
        out.append(project_feasible(h, n))
    return out[:count]

def local_perturb(h0, n, time_limit, seed, step=0.02, max_trials=3000):
    rng = np.random.RandomState(int(seed))
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_h = h
    cur_c = best_c
    temp = 0.002
    end_t = time.time() + time_limit
    for _ in range(max_trials):
        if time.time() > end_t:
            break
        L = int(rng.randint(1, min(6, n) + 1))
        st = int(rng.randint(0, n - L + 1))
        delta = rng.randn(L) * step
        delta -= delta.mean()
        h_try = cur_h.copy()
        h_try[st:st + L] += delta
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try)
        if c_try < best_c - 1e-15:
            best_h = h_try.copy()
            best_c = c_try
            cur_h = h_try
            cur_c = c_try
            temp = min(0.015, temp * 1.1)
        elif c_try < cur_c:
            cur_h = h_try
            cur_c = c_try
        else:
            d = c_try - cur_c
            if 0.0 < d < temp:
                if rng.rand() < np.exp(-d / temp):
                    cur_h = h_try
                    cur_c = c_try
        temp = max(temp * 0.997, 1e-7)
    return best_h, float(best_c)

def block_swap_perturb(h0, n, time_limit, seed, max_trials=2000):
    rng = np.random.RandomState(int(seed))
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_h = h
    cur_c = best_c
    temp = 0.002
    end_t = time.time() + time_limit
    for _ in range(max_trials):
        if time.time() > end_t:
            break
        L = int(rng.randint(2, min(10, n // 4) + 1))
        st1 = int(rng.randint(0, n - 2 * L + 1))
        st2 = st1 + L
        swap = h[st1:st1 + L].copy()
        h_try = cur_h.copy()
        h_try[st1:st1 + L] = cur_h[st2:st2 + L]
        h_try[st2:st2 + L] = swap
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try)
        if c_try < best_c - 1e-15:
            best_h = h_try.copy()
            best_c = c_try
            cur_h = h_try
            cur_c = c_try
        elif c_try < cur_c:
            cur_h = h_try
            cur_c = c_try
        else:
            d = c_try - cur_c
            if 0.0 < d < temp:
                if rng.rand() < np.exp(-d / temp):
                    cur_h = h_try
                    cur_c = c_try
        temp = max(temp * 0.998, 1e-7)
    return best_h, float(best_c)

def run(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    budget_s = float(budget_s)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    hard_end = start_time + max(120.0, budget_s - 30.0)

    # Get parent
    parent = None
    try:
        pv = initial_h_values
        if pv is not None and np.asarray(pv).size > 0:
            parent = np.asarray(pv, dtype=float).copy()
    except Exception:
        pass

    if parent is not None:
        parent_n = int(parent.size)
        raw_c = compute_c5(parent)
        proj = project_feasible(parent, parent_n)
        proj_c = compute_c5(proj)
        if raw_c <= proj_c + 1e-12:
            best_h = parent.astype(float).copy()
            best_c = raw_c
        else:
            best_h = proj
            best_c = proj_c
        best_n = parent_n
    else:
        best_n = 160
        best_h = project_feasible(np.full(best_n, 0.5), best_n)
        best_c = compute_c5(best_h)

    # Resolution candidates
    res_list = sorted(set([best_n, 160, 192, 224, 256, 288, 320, 384, 448, 512]))
    # Prioritize: parent n first, then nearby, then larger
    res_list = [r for r in res_list if 80 <= r <= 512]

    # Phase 1: Quick optimization at parent resolution
    if time.time() < hard_end - 200.0:
        inits = generate_diverse_inits(best_n, parent, rng, count=4)
        slot = min(80.0, hard_end - time.time() - 200.0)
        per_init = max(15.0, slot / len(inits))
        for j, h0 in enumerate(inits):
            if time.time() > hard_end - 200.0:
                break
            remain = hard_end - 200.0 - time.time()
            if remain < 20.0:
                break
            h2, c2 = optimize_smooth_lbfgs(h0, best_n, betas=(10, 30, 80, 200), maxiter=50)
            if c2 < best_c:
                best_h = h2.copy()
                best_c = c2
                best_n = int(best_n)

    # Phase 2: Try other resolutions
    for nn in res_list:
        nn = int(nn)
        if time.time() > hard_end - 120.0:
            break
        if nn == best_n:
            continue
        rem = hard_end - 120.0 - time.time()
        if rem < 40.0:
            break
        inits = generate_diverse_inits(nn, parent if parent is not None else best_h, rng, count=3)
        # Also include interpolated best
        best_interp = interpolate_to_n(best_h, nn)
        inits.append(project_feasible(best_interp, nn))
        slot = min(90.0, rem * 0.6)
        per_init = max(12.0, slot / len(inits))
        local_best_h = None
        local_best_c = best_c
        for j, h0 in enumerate(inits):
            if time.time() > time.time() + per_init + 5.0:
                break
            h2, c2 = optimize_smooth_lbfgs(h0, nn, betas=(10, 40, 100), maxiter=45)
            if c2 < local_best_c:
                local_best_h = h2
                local_best_c = c2
        if local_best_h is not None:
            # Refine with active set
            if time.time() < hard_end - 100.0:
                h3, c3 = refine_active_set(local_best_h, nn, hard_end - 100.0,
                                           active_counts=(10, 20, 40), maxiter=35)
                if c3 < local_best_c:
                    local_best_h = h3
                    local_best_c = c3
            if local_best_c < best_c:
                best_h = local_best_h
                best_c = local_best_c
                best_n = nn

    # Phase 3: Final refinement at best resolution
    if time.time() < hard_end - 20.0 and 2 <= best_n <= 512:
        # Active set refinement
        ac_counts = (16, 32, 64) if best_n <= 320 else (8, 16, 32)
        h2, c2 = refine_active_set(best_h, best_n, hard_end - 15.0,
                                   active_counts=ac_counts, maxiter=50)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    # Phase 4: Local perturbation
    if time.time() < hard_end - 10.0 and 2 <= best_n <= 320:
        h2, c2 = local_perturb(best_h, best_n, min(30.0, hard_end - time.time() - 5.0),
                               seed + 7777, step=0.015, max_trials=2000)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    # Phase 5: Block swap
    if time.time() < hard_end - 5.0 and 2 <= best_n <= 256:
        h2, c2 = block_swap_perturb(best_h, best_n, min(15.0, hard_end - time.time() - 2.0),
                                    seed + 8888, max_trials=1500)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    # Final projection and verification
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)
    return (best_h, float(best_c), int(best_n))
