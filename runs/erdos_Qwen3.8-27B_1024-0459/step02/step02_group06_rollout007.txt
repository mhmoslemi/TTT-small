import numpy as np
from scipy.optimize import minimize

def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float('inf')
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    if corr.size == 0:
        return float('inf')

        best_n = n0

    resolutions = [900, 1800, 450]
    if best_n > 0:
        resolutions.insert(0, best_n)

    time_per_res = max(60.0, (deadline - start_time - 200.0) / 3.0)

    for ni in resolutions:
        rem = deadline - time.perf_counter()
        if rem < 40.0:
            break
        alloc = min(time_per_res, rem - 30.0)
        if alloc < 30.0:
            break

        starts = []
        if parent is not None:
            starts.append(upsample_h(parent, ni))
        starts.append(upsample_h(best_h, ni))
        for _ in range(2):
            starts.append(make_start(ni, rng, int(rng.integers(5))))

        uniq = []
        for s in starts:
            s = project_density(s, 0.5 * ni)
            dup = False
            for u in uniq:
                if np.linalg.norm(s - u) < 1e-4:
                    dup = True
                    break
            if not dup:
                uniq.append(s)

        per_start = alloc / max(1, len(uniq))
        per_start = max(15.0, per_start)

        for si, s in enumerate(uniq):
            rem2 = deadline - time.perf_counter()
            if rem2 < 15.0:
                break
            sub_alloc = min(per_start, rem2 - 10.0)
            if sub_alloc < 10.0:
                break

            # Phase A: Projected gradient
            pg_deadline = time.perf_counter() + sub_alloc * 0.45
            h_a, c_a = projected_gradient(s, ni, pg_deadline, alpha_init=15.0, alpha_final=150.0, k_top=200)
            if c_a < best_c:
                best_h = h_a.copy()
                best_c = c_a
                best_n = ni

            # Phase B: L-BFGS/SLSQP (only for smaller n)
            rem3 = deadline - time.perf_counter()
            if ni <= 500 and rem3 > 20.0:
                sl_deadline = time.perf_counter() + min(60.0, rem3 * 0.3)
                h_b, c_b = lbfgs_phase(h_a, ni, sl_deadline, alpha=80.0)
                if c_b < best_c:
                    best_h = h_b.copy()
                    best_c = c_b
                    best_n = ni
                h_work = h_b if c_b < c_a else h_a
            else:
                h_work = h_a

            # Phase C: Local polish (gradient-based)
            rem4 = deadline - time.perf_counter()
            if rem4 > 15.0:
                lp_deadline = time.perf_counter() + min(sub_alloc * 0.35, rem4 - 10.0)
                h_c, c_c = local_polish(h_work, ni, lp_deadline, max_iter=3000, top_k=60)
                if c_c < best_c:
                    best_h = h_c.copy()
                    best_c = c_c
                    best_n = ni
                h_work = h_c if c_c < evaluate_erdos(h_work) else h_work

            # Phase D: Pair swap
            rem5 = deadline - time.perf_counter()
            if rem5 > 10.0:
                ps_deadline = time.perf_counter() + min(sub_alloc * 0.25, rem5 - 8.0)
                h_d, c_d = pair_swap_polish(h_work, ni, ps_deadline, max_iter=4000, top_frac=0.04, min_k=10, n_pos=8, n_neg=8)
                if c_d < best_c:
                    best_h = h_d.copy()
                    best_c = c_d
                    best_n = ni

    # Final intensive polish on the best solution
    rem = deadline - time.perf_counter()
    if best_h is not None and rem > 30.0:
        nbest = int(best_h.size)
        target = 0.5 * nbest
        h_fin = project_density(best_h, target)
        best_c = evaluate_erdos(h_fin)

        # Extended gradient polish
        if rem > 20.0:
            dl1 = time.perf_counter() + rem * 0.3
            h1, c1 = local_polish(h_fin, nbest, dl1, max_iter=8000, top_k=40)
            if c1 < best_c:
                h_fin = h1
                best_c = c1

        # Pair swap
        rem2 = deadline - time.perf_counter()
        if rem2 > 15.0:
            dl2 = time.perf_counter() + rem2 * 0.35
            h2, c2 = pair_swap_polish(h_fin, nbest, dl2, max_iter=10000, top_frac=0.03, min_k=12, n_pos=12, n_neg=12)
            if c2 < best_c:
                h_fin = h2
                best_c = c2

        # More gradient polish
        rem3 = deadline - time.perf_counter()
        if rem3 > 15.0:
            dl3 = time.perf_counter() + rem3 * 0.4
            h3, c3 = local_polish(h_fin, nbest, dl3, max_iter=10000, top_k=30)
            if c3 < best_c:
                h_fin = h3
                best_c = c3

        # Final micro-swaps
        rem4 = deadline - time.perf_counter()
        if rem4 > 10.0:
            dl4 = time.perf_counter() + rem4 * 0.7
            h4, c4 = pair_swap_polish(h_fin, nbest, dl4, max_iter=15000, top_frac=0.02, min_k=15, n_pos=6, n_neg=6)
            if c4 < best_c:
                h_fin = h4
                best_c = c4

        best_h = h_fin
        best_n = nbest

    # Final verification and projection
    if best_h is None:
        fallback = project_density(np.full(90, 0.5), 45.0)
        return fallback, evaluate_erdos(fallback), 90

    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    return final_h, float(final_c), int(final_h.size)
