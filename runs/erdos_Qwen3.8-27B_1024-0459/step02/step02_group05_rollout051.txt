import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            add = diff / cnt
            h[mask] = np.minimum(h[mask] + add, 1.0)
        else:
            mask = h > 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            sub = (-diff) / cnt
            h[mask] = np.maximum(h[mask] - sub, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def c5_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return np.max(corr)

def soft_c5_obj(x, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    shifted = np.clip(beta * (corr - cmax), -500, 0)
    return cmax + (1.0 / beta) * np.log(np.sum(np.exp(shifted)))

def top_k_smooth_obj(x, k=5, lam=0.5):
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    sort_idx = np.argsort(corr)[::-1][:k]
    top_mean = np.mean(corr[sort_idx])
    return cmax + lam * (cmax - top_mean)

def slsqp_solve(n, h0, max_iter=5000, ftol=1e-16, obj_type='max'):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    if obj_type == 'max':
        obj = c5_obj
    elif obj_type == 'soft':
        obj = lambda x: soft_c5_obj(x, 10000)
    elif obj_type == 'topk':
        obj = lambda x: top_k_smooth_obj(x, k=8, lam=0.3)
    else:
        obj = c5_obj
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': ftol, 'iprint': 0})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None

def coordinate_descent(h, n, n_sweeps=20, rng=None):
    h = h.copy()
    best_c5 = compute_c5(h)
    dx_n = 2.0 / n
    for sweep in range(n_sweeps):
        improved = False
        for j in range(n):
            old = h[j]
            # 1D scan: try several values

            # Add points near current value
            for d in [0.005, 0.01, 0.02, 0.05]:
                grid = np.append(grid, [np.clip(old + d, 0, 1), np.clip(old - d, 0, 1)])
            best_t = old
            best_val = best_c5
            for t in grid:
                if abs(t - old) < 1e-8:
                    continue
                h[j] = t
                # Need sum constraint: adjust a "reservoir" coordinate
                diff = n / 2.0 - np.sum(h)
                # Compensate by adjusting the average of others
                others_mask = np.ones(n, dtype=bool)
                others_mask[j] = False
                others_mean = np.sum(h[others_mask])
                # Just check the objective without fixing sum (SLSQP will fix later)
                val = compute_c5(h)
                if val < best_val - 1e-15:
                    best_val = val
                    best_t = t
            h[j] = best_t

        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            improved = True
        if not improved and sweep >= 3:
            break
    return h, best_c5

    current = h0.copy()
    best_c5 = compute_c5(current)
    best_h = current.copy()

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]

                c5 = compute_c5(current)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = current.copy()
        except Exception:
            pass
    return best_h, best_c5

            h = h_try
            best_c5 = c5_try
        step *= 0.998
    return h, best_c5

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:

    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_src)
    return make_feasible(h_interp, n_dst)

def binary_sa(n, rng, n_steps=20000, t_start=0.5, t_end=0.002):
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)

        i = rng.randint(n)

        if i == j or h[i] == h[j]:

        h[i], h[j] = h[j], h[i]
        c5_new = compute_c5(h)

            best_c5 = c5_new
        elif rng.random() < np.exp(-(c5_new - best_c5) / max(T, 1e-15)):

        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_c5
