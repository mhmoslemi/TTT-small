import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max_val(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_grad(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    exps = np.exp(shifted)
    return exps / np.sum(exps)

def corr_grad(corr_weights, x, n):
    """Compute gradient of sum_k w_k * corr[k] w.r.t. x"""
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = corr_weights[k]
        if w < 1e-14:
            continue
        L = k - (n - 1)
        if L == 0:
            grad += w * (1.0 - 2.0 * x)
        elif L > 0:
            grad[L:] += w * (1.0 - x[:n - L])
            grad[:n - L] -= w * x[L:]
        else:
            Lp = -L
            grad[:n - Lp] += w * (1.0 - x[Lp:])
            grad[Lp:] -= w * x[:n - Lp]
    return grad

def project_feasible(h, n):
    """Project h onto {0<=x<=1, sum(x)=n/2}"""
    h = h.copy()
    target = n / 2.0
    # Bisection on lambda for projection
    # We want: sum(clip(h - lambda, 0, 1)) = target
    lo, hi = -2.0, 2.0 + np.max(h) - np.min(h)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Tiny correction
    diff = target - np.sum(proj)
    if abs(diff) > 1e-15:
        # Distribute among elements not at boundary
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            proj[mask] += diff / np.sum(mask)
            proj = np.clip(proj, 0.0, 1.0)
        else:
            idx = np.argmin(np.maximum(proj, 1.0 - proj))
            proj[idx] += diff
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def interp_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def proj_grad_optimize(h0, n, beta_schedule, max_iter, lr_init=0.01, tol=1e-10):
    """Projected gradient with adaptive step size"""
    h = h0.copy()
    dx = 2.0 / n
    best_obj = smooth_max_val(np.correlate(h, 1-h, mode='full'), beta_schedule[0]) * dx
    
    for it in range(max_iter):
        beta = beta_schedule[min(it // 100, len(beta_schedule) - 1)]
        corr = np.correlate(h, 1.0 - h, mode='full')
        obj = smooth_max_val(corr, beta) * dx
        if obj < best_obj:
            best_obj = obj
        
        weights = smooth_max_grad(corr, beta)
        grad = corr_grad(weights, h, n) * dx
        
        # Adaptive step size (backtracking)
        lr = lr_init
        improved = False
        for _ in range(20):
            h_new = project_feasible(h - lr * grad, n)
            obj_new = smooth_max_val(np.correlate(h_new, 1.0 - h_new, mode='full'), beta) * dx
            if obj_new < obj:
                h = h_new
                improved = True
                break
            lr *= 0.5
        
        if not improved:
            break
        
        if it % 50 == 0:
            actual_c5 = compute_c5(h)
            if actual_c5 < best_obj:
                best_obj = actual_c5
    
    return h, compute_c5(h)

def slsqp_optimize(h0, n, beta, max_iter=500):
    dx = 2.0 / n
    def obj(x):
        return smooth_max_val(np.correlate(x, 1.0-x, mode='full'), beta) * dx
    def jac(x):
        corr = np.correlate(x, 1.0-x, mode='full')
        w = smooth_max_grad(corr, beta)
        return corr_grad(w, x, n) * dx
    def constraint(x):
        return np.sum(x) - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint}]
    
    try:
        res = minimize(obj, h0, jac=jac, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
        h_out = project_feasible(res.x, n)
        return h_out
    except Exception:
        return h0

def make_fourier_init(n, rng, K=5):
    h = np.full(n, 0.5)
    x = (np.arange(n) + 0.5) * (2.0 / n)

def local_search_binary(h, n, rng, max_iters=5000):
    """Simulated annealing on binary swaps"""
    h = (h > 0.5).astype(float)
    # Ensure exactly n/2 ones
    n_ones = int(np.sum(h))
    n_want = n // 2
    if n_ones > n_want:
        ones = np.where(h == 1.0)[0]
        order = rng.permutation(len(ones))
        for i in range(n_ones - n_want):
            h[ones[order[i]]] = 0.0
    elif n_ones < n_want:
        zeros = np.where(h == 0.0)[0]
        order = rng.permutation(len(zeros))
        for i in range(n_want - n_ones):
            h[zeros[order[i]]] = 1.0
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    curr_c5 = best_c5
    T = 0.005  # initial temperature
    cooling = 0.9995
    
    ones_list = np.where(h == 1.0)[0]
    zeros_list = np.where(h == 0.0)[0]
    
    for it in range(max_iters):
        i1 = ones_list[rng.randint(len(ones_list))]
        i0 = zeros_list[rng.randint(len(zeros_list))]
        
        h[i1] = 0.0
        h[i0] = 1.0
        new_c5 = compute_c5(h)
        
        delta = new_c5 - curr_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            # Update lists
            ones_list = np.where(h == 1.0)[0]
            zeros_list = np.where(h == 0.0)[0]
        else:
            h[i1] = 1.0
            h[i0] = 0.0
        
        T *= cooling
        if T < 1e-8:
            break
    
    return best_h, best_c5

Let me also add one more optimization idea: **alternating projection**.

The problem can be reformulated: minimize t subject to corr[k]*dx <= t for all k, 0<=h<=1, sum(h)=n/2.

We can use ADMM or alternating direction:
1. Fix t, find h that minimizes max_k corr[k]*dx - t (or satisfies corr[k]*dx <= t)
2. Fix h, update t = max_k corr[k]*dx

Step 2 is trivial. Step 1 is the hard part.

For step 1, we can use the smooth max as a surrogate for the max. This is essentially what the projected gradient does.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max_val(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_weights(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    exps = np.exp(shifted)
    return exps / np.sum(exps)

def corr_gradient(weights, x, n):
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        L = k - (n - 1)
        if L == 0:
            grad += w * (1.0 - 2.0 * x)
        elif L > 0:
            grad[L:] += w * (1.0 - x[:n - L])
            grad[:n - L] -= w * x[L:]
        else:
            Lp = -L
            grad[:n - Lp] += w * (1.0 - x[Lp:])
            grad[Lp:] -= w * x[:n - Lp]
    return grad

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    if abs(np.sum(h) - target) < 1e-13:
        return h
    lo = -1.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(proj)
    if abs(diff) > 1e-14:
        mask = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            proj[mask] += diff / cnt
        else:
            idx = np.argmin(np.maximum(proj, 1.0 - proj))
            proj[idx] += diff
        proj = np.clip(proj, 0.0, 1.0)
    return proj

def interp_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def proj_grad_step(h, n, beta, lr):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    obj = smooth_max_val(corr, beta) * dx
    weights = smooth_max_weights(corr, beta)
    grad = corr_gradient(weights, h, n) * dx
    # Backtracking line search
    for i in range(25):
        h_new = project_feasible(h - lr * grad, n)
        obj_new = smooth_max_val(np.correlate(h_new, 1.0 - h_new, mode='full'), beta) * dx
        if obj_new < obj:
            return h_new, obj_new
        lr *= 0.5
    return h, obj

def proj_grad_optimize(h0, n, betas, max_iter, lr_init):
    h = h0.copy()
    lr = lr_init
    best_c5 = compute_c5(h)
    for it in range(max_iter):
        beta = betas[min(it // 80, len(betas) - 1)]
        h, _ = proj_grad_step(h, n, beta, lr)
        # Occasionally check true C5
        if it % 40 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
    final_c5 = compute_c5(h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return h, best_c5

def slsqp_opt(h0, n, beta, max_iter):
    dx = 2.0 / n
    def obj(x):
        return smooth_max_val(np.correlate(x, 1.0 - x, mode='full'), beta) * dx
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        w = smooth_max_weights(corr, beta)
        return corr_gradient(w, x, n) * dx
    def con(x):
        return np.sum(x) - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con}]
    try:
        res = minimize(obj, h0, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h0

def make_random_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def make_fourier(n, rng, K):
    h = np.full(n, 0.5)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    for k in range(1, K + 1):
        amp = rng.uniform(0.03, 0.12)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * k * x / 2.0 + phase)
        h += amp * rng.uniform(0.5, 1.0) * np.sin(2 * np.pi * k * x / 2.0 + rng.uniform(0, 2 * np.pi))
    h = np.clip(h, 0, 1)
    return project_feasible(h, n)

def binary_sa(h, n, rng, max_iters, T_init=0.004, cooling=0.9997):
    h = (h > 0.5).astype(float)
    n_ones = int(np.sum(h))
    n_want = n // 2
    if n_ones > n_want:
        ones = np.where(h == 1.0)[0]
        order = rng.permutation(len(ones))
        for i in range(n_ones - n_want):
            h[ones[order[i]]] = 0.0
    elif n_ones < n_want:
        zeros = np.where(h == 0.0)[0]
        order = rng.permutation(len(zeros))
        for i in range(n_want - n_ones):
            h[zeros[order[i]]] = 1.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    curr_c5 = best_c5
    T = T_init
    ones_list = np.where(h == 1.0)[0]
    zeros_list = np.where(h == 0.0)[0]
    for it in range(max_iters):
        i1 = ones_list[rng.randint(len(ones_list))]
        i0 = zeros_list[rng.randint(len(zeros_list))]
        h[i1] = 0.0
        h[i0] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - curr_c5
        if delta < 0 or (T > 1e-10 and rng.random() < np.exp(-delta / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            ones_list = np.where(h == 1.0)[0]
            zeros_list = np.where(h == 0.0)[0]
        else:
            h[i1] = 1.0
            h[i0] = 0.0
        T *= cooling
    return best_h, best_c5

def targeted_lm(h, n, rng, max_trials):
    """Local modifications targeting the worst lags"""
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    dx = 2.0 / n
    for trial in range(max_trials):

        # Get top-3 worst lags

        for k_idx in top_idx:
            L = k_idx - (n - 1)
            if abs(L) >= n:
                continue
            Lp = abs(L)
            if Lp == 0:
                continue
            # Try nudging a few positions
            for _ in range(5):
                h_try = h.copy()
                if sign_L := (1 if L > 0 else -1):
                    if L > 0:
                        i = rng.randint(0, n - Lp)

                    else:

                h_try = project_feasible(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
