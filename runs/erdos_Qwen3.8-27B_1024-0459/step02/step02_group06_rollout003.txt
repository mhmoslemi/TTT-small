                    break
            if improved:
                break
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 50:
                break
    return project_density(best, target), evaluate_erdos(best)

def binary_swap_search(h, deadline, max_iter=10000, window=3):
    """Work with the binary rounding: find optimal 0/1 pattern via swaps"""
    n = h.size
    target = int(round(0.5 * n))
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    rng = np.random.default_rng(31415)
    stall = 0
    it = 0
    while it < max_iter:
        if time.perf_counter() >= deadline:
            break
        it += 1
        improved = False
        # Try swapping a 1-index with a 0-index
        ones = np.where(best > 0.5)[0]
        zeros = np.where(best <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        # Gradient-guided selection
        g = active_gradient(best, top_frac=0.05, min_k=10)
        swap_ones = ones[np.argsort(g[ones])[:window * 2]]
        swap_zeros = zeros[np.argsort(-g[zeros])[:window * 2]]
        found = False
        for oi in swap_ones:
            for zi in swap_zeros:
                oi = int(oi)
                zi = int(zi)
                cand = best.copy()
                cand[oi] = 0.0
                cand[zi] = 1.0
                cand = project_density(cand, 0.5 * n)
                c = evaluate_erdos(cand)
                if c < best_c - 1e-13:
                    best = cand
                    best_c = c
                    found = True
                    break
            if found:
                break
        if not found:
            # Random swaps
            for _ in range(20):
                oi = int(ones[rng.integers(ones.size)])
                zi = int(zeros[rng.integers(zeros.size)])
                cand = best.copy()
                cand[oi] = 0.0
                cand[zi] = 1.0
                cand = project_density(cand, 0.5 * n)
                c = evaluate_erdos(cand)
                if c < best_c - 1e-13:
                    best = cand
                    best_c = c
                    found = True
                    break
        if found:
            improved = True
            stall = 0
        else:
            stall += 1
            if stall > 30:
                break
    return project_density(best, target), evaluate_erdos(best)

def neighborhood_search(h, deadline, max_iter=3000, radius=2):
    """Try small neighborhood modifications"""
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    rng = np.random.default_rng(271828)
    stall = 0
    it = 0
    while it < max_iter:
        if time.perf_counter() >= deadline:
            break
        it += 1
        improved = False
        idx = int(rng.integers(0, n))
        lo = max(0, idx - radius)
        hi = min(n, idx + radius + 1)
        seg = best[lo:hi].copy()
        for _ in range(8):
            seg2 = seg + (rng.random(seg.size) - 0.5) * 0.2
            seg2 = np.clip(seg2, 0.0, 1.0)
            cand = best.copy()
            cand[lo:hi] = seg2
            cand = project_density(cand, target)
            c = evaluate_erdos(cand)
            if c < best_c - 1e-13:
                best = cand
                best_c = c
                improved = True
                break
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 80:
                break
    return project_density(best, target), evaluate_erdos(best)

def combined_polish(h, deadline):
    """Multiple rounds of different search methods"""
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    
    phases = [
        ("grad", lambda x, dl: gradient_local_search(x, dl, max_iter=20000, top_frac=0.03, min_k=10)),
        ("rand", lambda x, dl: random_perturb_search(x, dl, max_iter=10000, n_perturb=8)),
        ("binary", lambda x, dl: binary_swap_search(x, dl, max_iter=5000, window=4)),
        ("neigh", lambda x, dl: neighborhood_search(x, dl, max_iter=2000, radius=3)),
    ]
    
    for round_num in range(3):
        for name, func in phases:
            rem = deadline - time.perf_counter()
            if rem < 5.0:
                return project_density(best, target), evaluate_erdos(best)
            alloc = min(rem * 0.25, 60.0)
            sub_dl = time.perf_counter() + alloc
            h2, c2 = func(best, sub_dl)
            if c2 < best_c - 1e-14:
                best = h2
                best_c = c2
    
    return project_density(best, target), evaluate_erdos(best)

def smooth_soft_max_objective(x, n, alpha):
    h = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(h)):
        return 1e100
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -700.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0 or not np.isfinite(z):
        return 1e100
    return (2.0 / n) * (m + np.log(z) / alpha)
