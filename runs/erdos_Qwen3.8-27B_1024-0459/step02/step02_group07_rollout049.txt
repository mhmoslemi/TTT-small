import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_opt_j(h, j, n, corr, ells):
    idx1 = j - ells
    idx2 = j + ells
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    a = np.zeros(2 * n - 1)
    if np.any(valid1):
        a[valid1] += 1.0 - h[idx1[valid1]]
    if np.any(valid2):
        a[valid2] += -h[idx2[valid2]]
    b = corr - h[j] * a
    lo = 0.0
    hi = 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        vals = a * mid + b
        mx = np.max(vals)
        active = vals >= mx - 1e-14
        sg = np.mean(a[active])
        if sg > 1e-13:
            hi = mid
        elif sg < -1e-13:
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)

def coord_descent(h, n, n_passes, time_limit):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    corr = get_corr(h)
    ells = np.arange(-(n - 1), n)
    start = time.time()
    for p in range(n_passes):
        if time.time() - start > time_limit:
            break
        improved = False
        for j in range(n):
            t_opt = coord_opt_j(h, j, n, corr, ells)
            if abs(t_opt - h[j]) > 1e-12:
                improved = True
            h[j] = t_opt
        h = make_feasible(h, n)
        corr = get_corr(h)
        c5 = np.max(corr) * (2.0 / n)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
        if not improved and c5 >= best_c5 - 1e-14:
            break
    return best_h, best_c5

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_src)
    return make_feasible(h_interp, n_dst)

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    return make_feasible(h, n)

def binary_local_search(h, n, max_trials, time_limit, rng, block_sizes=(1,2,3,4,5,8,12)):
    h_bin = (h > 0.5).astype(float)
    s = np.sum(h_bin)
    deficit = int(round(n / 2.0 - s))
    if deficit > 0:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])[:deficit]
        h_bin[zeros[order]] = 1.0
    elif deficit < 0:
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])[:abs(deficit)]
        h_bin[ones[order]] = 0.0
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    start = time.time()
    trial = 0
    while trial < max_trials:
        if time.time() - start > time_limit:
            break
        trial += 1
        h_try = best_h.copy()
        bs = block_sizes[rng.randint(len(block_sizes))]
        if bs == 1:
            idx = rng.randint(n)
            h_try[idx] = 1.0 - h_try[idx]
        else:
            start_idx = rng.randint(n - bs + 1)
            h_try[start_idx:start_idx+bs] = 1.0 - h_try[start_idx:start_idx+bs]
        if abs(np.sum(h_try) - n / 2.0) < 1e-10:
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try.copy()
    return best_h, best_c5

    h_bin = (h > 0.5).astype(float)
    s = np.sum(h_bin)

    curr_h = h_bin.copy()

    T_min = 0.0001

    start = time.time()
    while time.time() - start < time_limit:
        T = max(T * cool, T_min)
        
        # Generate move
        move_type = rng.rand()
        h_try = curr_h.copy()
        if move_type < 0.3:
            # Single flip
            idx = rng.randint(n)
            h_try[idx] = 1.0 - h_try[idx]
        elif move_type < 0.6:
            # Block flip
            bs = rng.randint(2, min(8, n//4))
            si = rng.randint(n - bs + 1)
            h_try[si:si+bs] = 1.0 - h_try[si:si+bs]
        elif move_type < 0.8:
            # Swap two blocks
            bs = rng.randint(1, min(5, n//8))
            si = rng.randint(n - bs)
            si2 = si + rng.randint(1, n - si)
            if si2 + bs <= n:
                h_try[si:si+bs], h_try[si2:si2+bs] = h_try[si2:si2+bs].copy(), h_try[si:si+bs].copy()
        else:

            curr_h = h_try.copy()
            curr_c5 = c5_try

                best_c5 = c5_try
                best_h = h_try.copy()

def slsqp_polish(h, n, max_iter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x, n=n: np.sum(x) - n/2.0}]
    try:
        res = minimize(lambda x: np.max(np.correlate(x, 1.0-x, mode='full')),
                       h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-15})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None

    best_c5 = compute_c5(h)
    best_h = h.copy()
    corr = get_corr(h)
    ells = np.arange(-(n-1), n)
    order = np.arange(n)
    start = time.time()
    T = 0.1  # initial "temperature"
    T_min = 0.001
    cool = 0.998
    pass_count = 0
    no_improve = 0
    
    while time.time() - start < time_limit:
        pass_count += 1
        # Shuffle order for diversity
        order = rng.permutation(n)
        h, corr = cd_pass(h, n, corr, ells, order)
        
        # Apply perturbation based on temperature
        if T > T_min:
            # Perturb a random subset
            n_pert = max(1, int(n * T * 2))
            pert_idx = rng.choice(n, n_pert, replace=False)
            h[pert_idx] += rng.randn(n_pert) * T
            h = np.clip(h, 0.0, 1.0)
            h = make_feasible(h, n)
            corr = get_corr(h)
            T *= cool
        
        c5 = float(np.max(corr) * (2.0/n))
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
            T = 0.1  # reset temperature on improvement
            T *= 0.5
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                # Force a larger perturbation
                n_pert = max(5, n // 10)
                pert_idx = rng.choice(n, n_pert, replace=False)
                h[pert_idx] += rng.randn(n_pert) * 0.1
                h = np.clip(h, 0.0, 1.0)
                h = make_feasible(h, n)
                corr = get_corr(h)
                T = max(T, 0.05)
                no_improve = 0
    
    return best_h, best_c5

def c5_slsq_obj(x):
    return float(np.max(np.correlate(x, 1.0-x, mode='full')))

def sum_constraint_fn(x, n):
    return float(np.sum(x) - n / 2.0)

def slsqp_polish(h, n, max_iter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint_fn, 'args': (n,)}]
    try:
        res = minimize(c5_slsq_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-15})
        if res.fun < 1e10:
            h_out = make_feasible(np.array(res.x), n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_src)
    return make_feasible(h_interp, n_dst)
