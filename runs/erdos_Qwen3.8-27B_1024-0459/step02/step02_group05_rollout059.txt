```python
def binary_sa(n, rng, n_steps, t_start, t_end):
    # Initialize
    h = np.zeros(n)
    ones = rng.choice(n, n//2, replace=False)
    h[ones] = 1.0
    corr = np.correlate(h, 1-h, mode='full')
    best_c5 = np.max(corr) * 2.0/n
    best_h = h.copy()
    
    for step in range(n_steps):
        T = t_start * (t_end/t_start) ** (step/max(n_steps-1,1))
        # Pick random 1 and 0 positions
        ones_idx = np.where(h==1)[0]
        zeros_idx = np.where(h==0)[0]
        if len(ones_idx)==0 or len(zeros_idx)==0:
            break
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        
        h[i], h[j] = h[j], h[i]  # swap
        corr = np.correlate(h, 1-h, mode='full')
        c5 = np.max(corr) * 2.0/n
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        elif rng.random() >= np.exp(-(c5-best_c5)/max(T,1e-15)):
            h[i], h[j] = h[j], h[i]  # undo
    return best_h, best_c5
```

```python
def smooth_max_and_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = np.max(corr)
    shifted = beta * (corr - c)
    # Numerical stability
    shifted -= np.max(shifted)
    exp_shifted = np.exp(shifted)
    log_sum = np.log(np.sum(exp_shifted))
    smooth_max = c + log_sum / beta
    weights = exp_shifted / np.sum(exp_shifted)
    
    # Gradient w.r.t. h
    grad = np.zeros(n)
    # corr_k = sum_i h[i] * (1-h[i+s]) where s = k-(n-1)
    # d(corr_k)/d(h[i]) = (1-h[i+s]) - h[i-s] (with bounds)
    # 
    # So grad[i] = sum_k weights[k] * ((1-h[i+s_k]) - h[i-s_k])
    # where s_k = k - (n-1)
    
    for k in range(len(corr)):
        s = k - (n-1)
        # First term: sum over i where i+s is valid: h[i] * (1-h[i+s])
        # d/d(h[i]) = (1-h[i+s]) if 0 <= i+s < n
        # Second term: sum over i where i-s is valid: h[i-s] * (1-h[i])
        # d/d(h[i]) = -h[i-s] if 0 <= i-s < n
        
        if weights[k] < 1e-15:
            continue
            
        if 0 <= s <= n-1:
            # i+s < n means i < n-s
            # (1-h[i+s]) for i = 0, ..., n-s-1
            grad[:n-s] += weights[k] * (1 - h[s:n])
        if s < 0:
            # i+s >= 0 means i >= -s
            # (1-h[i+s]) for i = -s, ..., n-1
            grad[-s:] += weights[k] * (1 - h[:n+s])
        
        if 0 <= -s <= n-1:
            # i-s < n means i < n+s, i-s >= 0 means i >= s
            # -h[i-s] for i = s, ..., n-1 (when s >= 0)
            if s >= 0:
                grad[s:] -= weights[k] * h[:n-s]
        if s < 0:
            # i-s >= 0 means i >= s (always true since s < 0)
            # i-s < n means i < n+s
            # -h[i-s] for i = 0, ..., n+s-1
            grad[:n+s] -= weights[k] * h[-s:]
    
    return smooth_max, grad
```

```python
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def de_solve(n, h_init, budget_s, rng):
    """Differential evolution with penalty for constraint."""
    penalty = 50.0
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        x = np.clip(x, 0.0, 1.0)
        corr = np.correlate(x, 1.0 - x, mode='full')
        c5_raw = np.max(corr)
        sum_pen = (np.sum(x) - n / 2.0) ** 2
        return c5_raw + penalty * sum_pen
    
    # Set maxiter based on budget
    # Each generation: pop_size evaluations, each O(n) for correlate
    pop_size = max(20, n * 8)
    # Estimate time per generation
    # ~pop_size * (10μs for correlate + 5μs overhead) = pop_size * 15μs
    t_per_gen = pop_size * 15e-6
    maxiter = max(10, int(budget_s / (t_per_gen * 1.5)))
    maxiter = min(maxiter, 300)
    
    x0 = h_init + rng.randn(n) * 0.1
    x0 = np.clip(x0, 0.0, 1.0)
    
    try:
        res = differential_evolution(
            obj, bounds, seed=rng.randint(2**31),
            maxiter=maxiter, popsize=max(15, n//2),
            tol=1e-10, mutation=(0.5, 1.5), recombination=0.9,
            polish=False, init='soften'
        )
        h_out = make_feasible(res.x, n)
        c5 = compute_c5(h_out)
        return h_out, c5
    except Exception:
        return None, None

def slsqp_solve(n, h_init, max_iter=3000, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
             'jac': lambda x: np.ones(n)}]
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return np.max(corr)
    
    try:
        res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': ftol, 'iprint': 0})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None

def smooth_max_slsqp(n, h_init, betas=(100, 500, 2000, 8000), max_iter=2000):
    """SLSQP with smooth max (log-sum-exp) for differentiability."""
    current = h_init.copy()
    best_c5 = compute_c5(current)
    best_h = current.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
             'jac': lambda x: np.ones(n)}]
    
    for beta in betas:
        def obj_smooth(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            c = np.max(corr)
            shifted = beta * (corr - c)
            shifted -= np.max(shifted)
            return c + np.log(np.sum(np.exp(shifted))) / beta
        
        try:
            res = minimize(obj_smooth, current, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
            if res.fun < 1e10:
                current = make_feasible(res.x, n)
                c5 = compute_c5(current)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = current.copy()
        except Exception:
            pass
    
    return best_h, best_c5

def binary_sa(n, rng, n_steps=50000, t_start=0.5, t_end=0.01):
    """Simulated annealing on binary assignments."""
    h = np.zeros(n)
    h[rng.choice(n, n//2, replace=False)] = 1.0
    
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = np.max(corr) * dx
    best_h = h.copy()
    cur_c5 = best_c5
    
    one_pos = np.where(h == 1)[0].copy()
    zero_pos = np.where(h == 0)[0].copy()
    
    for step in range(n_steps):
        T = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        i = one_pos[rng.randint(len(one_pos))]
        j = zero_pos[rng.randint(len(zero_pos))]
        
        # Swap
        h[i] = 0.0
        h[j] = 1.0
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            cur_c5 = c5
        elif rng.random() < np.exp(-(c5 - cur_c5) / max(T, 1e-15)):
            cur_c5 = c5
        else:
            # Undo
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_c5

def binary_local_search(h, n, n_rounds=100, rng=None):
    """Local search: try all single swaps, keep the best improvement."""
    if rng is None:
        rng = np.random.RandomState(0)
    h = h.copy().astype(float)
    # Ensure binary
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    # Fix count
    ones = np.where(h == 1)[0]
    if len(ones) != n // 2:
        if len(ones) > n // 2:
            h[ones[n//2:]] = 0
        else:
            zeros = np.where(h == 0)[0]
            h[zeros[:n//2-len(ones)]] = 1
    
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = np.max(corr) * dx
    
    for _ in range(n_rounds):
        improved = False
        one_pos = np.where(h == 1)[0]
        zero_pos = np.where(h == 0)[0]
        for i in one_pos:
            for j in zero_pos:
                if i == j:
                    continue
                h[i] = 0.0
                h[j] = 1.0
                corr = np.correlate(h, 1.0 - h, mode='full')
                c5 = np.max(corr) * dx
                h[j] = 0.0
                h[i] = 1.0
                if c5 < best_c5 - 1e-15:
                    h[i] = 0.0
                    h[j] = 1.0
                    best_c5 = c5
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    
    return h, best_c5
```

What about h = 1 on [0, 0.5] ∪ [1, 1.5]? Then for shift 1: h(x) on [0,0.5] → h(x+1) on [1,1.5] = 1, overlap = 0.5. h(x) on [0.5,1] = 0 → no overlap. h(x) on [1,1.5] → h(x+1) on [2,2.5] = 0 (out of range or 0). So corr[shift 1] = 0.5. C₅ includes this: 0.5 * 1 = 0.5. Still bad.

The key is to spread the 1s and 0s as evenly as possible at all scales.
