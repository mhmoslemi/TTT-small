def compute_c5(h):

def make_feasible(h, n):

def active_lag_gradient(h, k_star, n):
    """Gradient of c_{k*} = sum_i h[i]*(1-h[(i+k*)%n]) w.r.t. h[i]"""
    h_shift_fwd = np.roll(h, -k_star)  # h[(i+k*)%n]
    h_shift_bwd = np.roll(h, k_star)   # h[(i-k*)%n]
    return (1.0 - h_shift_fwd) - h_shift_bwd

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(n * 3):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.maximum(h[mask] + step, 0.0)
    h = np.clip(h, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-14:
                    break
        else:
            order = np.argsort(h)[::-1]
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0:
                    break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-14:
                    break
    return h

def project_onto_feasible(h, n):

        else:

def circular_corr(h):
    n = len(h)
    h1m = 1.0 - h
    corr = np.zeros(n)
    for k in range(n):
        corr[k] = np.sum(h * np.roll(h1m, k))
    return corr

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_lag_circular(h, k, n):
    kf = k % n
    h_fwd = np.roll(h, -kf)
    h_bwd = np.roll(h, kf)
    return (1.0 - h_fwd) - h_bwd

def smooth_max_objective(h, n, top_k=5, beta=50.0):
    corr = corr_full(h)
    cmax = np.max(corr)
    # Use top-k smoothed max for better gradient
    sorted_idx = np.argsort(corr)[::-1]
    top = corr[sorted_idx[:top_k]]
    tmax = np.max(top)
    shifted = beta * (top - tmax)
    shifted = np.clip(shifted, -40, 40)
    return tmax + (1.0 / beta) * np.log(np.sum(np.exp(shifted)))

def smooth_max_grad(h, n, top_k=5, beta=50.0):
    corr = corr_full(h)
    sorted_idx = np.argsort(corr)[::-1]
    top_idx = sorted_idx[:top_k]
    top_corr = corr[top_idx]
    tmax = np.max(top_corr)

    shifted = np.clip(shifted, -40, 40)
    exps = np.exp(shifted)
    weights = exps / np.sum(exps)
    
    cmax = np.max(corr)

        if k == 0:
            grad += w * (1.0 - 2.0 * h)

    return grad

def slsqp_optimize(h, n, max_iter=5000):
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return np.max(np.correlate(x, 1.0 - x, mode='full'))
    
    def obj_jac(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        cmax = np.max(corr)
        active = np.where(np.abs(corr - cmax) < 1e-7)[0]
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k == 0:
                grad += (1.0 - 2.0 * x)
            else:
                grad += grad_lag_circular(x, k, n)
        grad /= len(active)
        return grad
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, method='SLSQP', jac=obj_jac, bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-16, 'iprint': 0})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None

        h_hi, c5_hi = pgd_optimize(h_interp, n_hi, max_iter=30000, step0=0.003, patience=3000, top_k=5, beta=50.0)
        if c5_hi < c5_interp:
            c5_interp = c5_hi
            h_interp = h_hi
            
        # SLSQP at higher res
        if remaining() > 150:
            h_opt, c5_opt = slsqp_optimize(h_interp, n_hi, max_iter=3000)
            if h_opt is not None and c5_opt < c5_interp:
                h_interp = h_opt
                c5_interp = c5_opt
        
        # Final PGD
        if remaining() > 80:
            h_hi, c5_hi = pgd_optimize(h_interp, n_hi, max_iter=30000, step0=0.001, patience=5000, top_k=3, beta=80.0)
            if c5_hi < c5_interp:
                h_interp = h_hi
                c5_interp = c5_hi
        
        if c5_interp < best_c5:
            best_c5 = c5_interp
            best_h = h_interp
            best_n = n_hi
    
    # Phase 6: Binary SA at current best resolution
    if remaining() > 120 and best_n <= 256:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = make_feasible(h_bin, best_n)
        n_steps = min(50000, int(remaining() * 40))
        h_sa, c5_sa = binary_sa(h_bin, best_n, n_steps=n_steps, t_start=0.6, t_end=0.002, rng=rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa
        # PGD from SA result
        if remaining() > 50:
            h_pg, c5_pg = pgd_optimize(best_h, best_n, max_iter=20000, step0=0.002, patience=2000, top_k=3, beta=50.0)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg
    
    # Phase 7: Multi-start search
    if remaining() > 100:
        best_h, best_c5 = multi_start_search(best_h, best_n, remaining(), rng)
    
    # Phase 8: Final SLSQP polish
    if remaining() > 60:
        h_opt, c5_opt = slsqp_optimize(best_h, best_n, max_iter=8000)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Phase 9: Final PGD with very tight params
    if remaining() > 30:
        h_pg, c5_pg = pgd_optimize(best_h, best_n, max_iter=50000, step0=0.001, patience=5000, top_k=2, beta=150.0, use_smooth=False)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg
    
    # Final feasibility check
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, best_n)
