        return x
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-10:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size > 0:
            h[free] += res / free.size
            h = np.clip(h, 0.0, 1.0)
            res2 = target - float(np.sum(h))
            if abs(res2) > 1e-10:
                free2 = np.where((h > 0.0) & (h < 1.0))[0]
                if free2.size > 0:
                    h[free2[0]] += res2
                    h = np.clip(h, 0.0, 1.0)
    return h

def binarize_smart(h, target, thresh_lo=0.35, thresh_hi=0.65):
    n = h.size
    h2 = h.copy()
    binary_mask = (h2 < thresh_lo) | (h2 > thresh_hi)
    h2[binary_mask] = np.where(h2[binary_mask] < thresh_lo, 0.0, 1.0)
    sum_bin = float(np.sum(h2))
    diff = target - sum_bin
    if abs(diff) < 1e-12:
        return h2
    # Adjust intermediate values
    inter = np.where(~binary_mask)[0]
    if inter.size > 0:
        h2[inter] += diff / inter.size
        h2 = np.clip(h2, 0.0, 1.0)
        h2 = project_density(h2, target)
    else:
        # Need to flip some binaries
        if diff > 0:
            zeros = np.where(h2 == 0.0)[0]
            for z in zeros[:int(diff)]:
                h2[z] = 1.0
        else:
            ones = np.where(h2 == 1.0)[0]
            for o in ones[:int(-diff)]:
                h2[o] = 0.0
        h2 = project_density(h2, target)
    return h2

def compute_active_pressure(h, top_frac=0.08, min_k=10):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    k_count = max(min_k, int(np.ceil(len(corr) * top_frac)))
    top_idx = np.argsort(corr)[-k_count:]
    top_vals = corr[top_idx]
    dv = top_vals - top_vals.min()
    spread = dv.max()
    if spread < 1e-15:
        w = np.ones(k_count) / k_count
    else:
        w = np.exp(dv / spread)
        w /= w.sum()
    
    S = np.zeros(n)
    for ii in range(k_count):
        t = int(top_idx[ii])
        k = t - center
        wk = w[ii]
        if k == 0:
            S[:] += wk * (1.0 - 2.0 * h)
        else:
            lag = abs(k)
            if k > 0:
                # h[i] in c[center+k] as h[i]*(1-h[i-k]) for i>=k: deriv = (1-h[i-k])
                # h[i] in c[center-k] as h[i+k]*(1-h[i]) for i+k<n: deriv = -h[i+k]
                valid_fwd = slice(lag, n)
                S[lag:] += wk * (1.0 - h[:n-lag])
                valid_bwd = slice(0, n-lag)
                S[:n-lag] -= wk * h[lag:]
            else:
                # k < 0, lag = -k
                # h[i] in c[center-lag] as h[i]*(1-h[i-lag]) for i>=lag: deriv = (1-h[i-lag])
                # h[i] in c[center+lag] as h[i+lag]*(1-h[i]) for i+lag<n: deriv = -h[i+lag]
                S[lag:] += wk * (1.0 - h[:n-lag])
                S[:n-lag] -= wk * h[lag:]
    return S, corr, top_idx

def local_search_binary(h, deadline, max_iters=50000, top_frac=0.06, min_k=8,
                        n_cand=15, anneal_temp=0.0003, anneal_rate=0.9997):
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    temp = anneal_temp
    stall = 0
    
    for it in range(max_iters):
        if time.perf_counter() >= deadline:
            break
        temp *= anneal_rate
        if temp < 1e-8:
            temp = 1e-8
        
        S, corr, top_idx = compute_active_pressure(best, top_frac, min_k)
        
        # Find candidates: positive S (want 0) and negative S (want 1)
        pos_idx = np.where(S > 0)[0]
        neg_idx = np.where(S < 0)[0]
        
        if pos_idx.size == 0 or neg_idx.size == 0:
            # Fallback: random pairs
            pos_idx = np.where(best < 0.99)[0]
            neg_idx = np.where(best > 0.01)[0]
            if pos_idx.size < 2 or neg_idx.size < 2:
                break
        
        # Sort by magnitude of pressure
        pos_sorted = pos_idx[np.argsort(-S[pos_idx])[:n_cand]]
        neg_sorted = neg_idx[np.argsort(S[neg_idx])[:n_cand]]
        
        best_move_c = best_c
        best_move_pair = None
        
        # Try structured pairs
        for ii in neg_sorted:
            ii = int(ii)
            if best[ii] < 0.05:
                continue
            for jj in pos_sorted:
                jj = int(jj)
                if ii == jj:
                    continue
                if best[jj] > 0.95:
                    continue
                # Try full swap
                cand = best.copy()
                cand[ii] = max(0.0, best[ii] - 1.0)
                cand[jj] = min(1.0, best[jj] + 1.0)
                # Hmm, that's not right. Let's do: move mass from i to j
                # If best[ii] > 0.5 and best[jj] < 0.5, swap them
                d = min(best[ii], 1.0 - best[jj])
                if d < 0.05:
                    continue
                for frac in (0.5, 0.75, 1.0):
                    dd = frac * d
                    cand = best.copy()
                    cand[ii] -= dd
                    cand[jj] += dd
                    if cand[ii] < -1e-12 or cand[jj] > 1 + 1e-12:
                        continue
                    c = evaluate_erdos(cand)
                    if c < best_move_c:
                        best_move_c = c
                        best_move_pair = (ii, jj, dd)
                        break
        
        # Also try some random pairs
        for _r in range(5):
            ii = int(np.random.default_rng(it * 7 + 1).integers(0, n))
            jj = int(np.random.default_rng(it * 7 + 2).integers(0, n))
            if ii == jj:
                continue
            d = min(best[ii], 1.0 - best[jj])
            if d < 0.1:
                continue
            dd = 0.8 * d
            cand = best.copy()
            cand[ii] -= dd
            cand[jj] += dd
            c = evaluate_erdos(cand)
            if c < best_move_c:
                best_move_c = c
                best_move_pair = (ii, jj, dd)
        
        if best_move_pair is not None:
            ii, jj, dd = best_move_pair
            best[ii] -= dd
            best[jj] += dd
            best = np.clip(best, 0.0, 1.0)
            best = project_density(best, target)
            if best_move_c < best_c:
                best_c = best_move_c
                stall = 0
            else:
                # Rejected by annealing
                if np.random.default_rng(it * 13).random() > np.exp(-(best_move_c - best_c) / temp):
                    best = project_density(h, target)
                    stall += 1
                else:
                    stall += 1
        else:
            # Try an annealing move
            ii = int(np.random.default_rng(it * 31).integers(0, n))
            jj = int(np.random.default_rng(it * 31 + 1).integers(0, n))
            if ii != jj:
                d = min(best[ii], 1.0 - best[jj])
                if d > 0.05:
                    dd = np.random.default_rng(it * 31 + 2).random() * d
                    cand = best.copy()
                    cand[ii] -= dd
                    cand[jj] += dd
                    c = evaluate_erdos(cand)
                    if c < best_c or np.random.default_rng(it * 31 + 3).random() < np.exp(-(c - best_c) / max(temp, 1e-10)):
                        best = project_density(cand, target)
                        if c < best_c:
                            best_c = c
                            stall = 0
                        else:
                            stall += 1
                    else:
                        stall += 1
        
        if stall > 200:
            break
    
    return project_density(best, target), evaluate_erdos(best)

def multi_pair_search(h, deadline, max_iters=5000, top_frac=0.06, min_k=8, n_cand=10):
    """Try swapping 2 pairs simultaneously (4 coordinates)"""
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    rng = np.random.default_rng(12345)
    stall = 0
    
    for it in range(max_iters):
        if time.perf_counter() >= deadline:
            break
        S, corr, top_idx = compute_active_pressure(best, top_frac, min_k)
        pos_sorted = np.where(S > 0)[0]
        neg_sorted = np.where(S < 0)[0]
        if pos_sorted.size < 2 or neg_sorted.size < 2:
            break
        
        pos_top = pos_sorted[np.argsort(-S[pos_sorted])[:n_cand]]
        neg_top = neg_sorted[np.argsort(S[neg_sorted])[:n_cand]]
        
        found = False
        # Try 2-pair swap
        for i1 in neg_top[:5]:
            i1 = int(i1)
            if best[i1] < 0.1:
                continue
            for i2 in pos_top[:5]:
                i2 = int(i2)
                if i2 == i1 or best[i2] > 0.9:
                    continue
            for j1 in neg_top[:5]:
                j1 = int(j1)
                if j1 == i1 or j1 == i2 or best[j1] < 0.1:
                    continue
                for j2 in pos_top[:5]:
                    j2 = int(j2)
                    if j2 == i1 or j2 == i2 or j2 == j1 or best[j2] > 0.9:
                        continue
                    d = min(best[i1], best[j1], 1.0 - best[i2], 1.0 - best[j2])
                    if d < 0.05:
                        continue
                    dd = 0.7 * d
                    cand = best.copy()
                    cand[i1] -= dd
                    cand[j1] -= dd
                    cand[i2] += dd
                    cand[j2] += dd
                    if np.any(cand < -1e-12) or np.any(cand > 1 + 1e-12):
                        continue
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-12:
                        best = cand
                        best_c = c
                        found = True
                        break
                if found:
                    break
            if found:
                break
        if found:
            stall = 0
        else:
            stall += 1
            if stall > 100:
                break
    
    return project_density(best, target), evaluate_erdos(best)

def rotation_search(h, deadline, max_shifts=5):
    """Try cyclic rotations of the pattern"""
    n = h.size
    best = h.copy()
    best_c = evaluate_erdos(best)
    for shift in range(1, max_shifts + 1):
        if time.perf_counter() >= deadline:
            break
        for s in (shift, -shift):
            cand = np.roll(best, s)
            c = evaluate_erdos(cand)
            if c < best_c:
                best = cand
                best_c = c
    return best, best_c

def continuous_polish(h, deadline, alpha=50.0, max_iter=100):
    """SLSQP polish with log-sum-exp"""
    n = h.size
    target = 0.5 * n
    h0 = project_density(h, target)
    best_c = evaluate_erdos(h0)
    
    def obj(x):
        hx = np.clip(x, 0.0, 1.0)
        corr = np.correlate(hx, 1.0 - hx, mode="full")
        m = float(np.max(corr))
        y = alpha * (corr - m)
        y = np.clip(y, -700.0, 0.0)
        e = np.exp(y)
        z = float(np.sum(e))
        if z <= 0 or not np.isfinite(z):
            return 1e100
        return float((2.0 / n) * (m + np.log(z) / alpha))
    
    def obj_grad(x):
        hx = np.clip(x, 0.0, 1.0)
        corr = np.correlate(hx, 1.0 - hx, mode="full")
        m = float(np.max(corr))
        y = alpha * (corr - m)
        y = np.clip(y, -700.0, 0.0)
        e = np.exp(y)
        z = float(np.sum(e))
        if z <= 0 or not np.isfinite(z):
            return np.zeros(n)
        w = e / z
        idx = np.where(w > 1e-8)[0]
        g = np.zeros(n)
        center = n - 1
        for t in idx:
            t = int(t)
            p = w[t]
            k = t - center
            if k == 0:
                g[:] += p * (1.0 - 2.0 * hx)
            else:
                lag = abs(k)
                g[lag:] += p
                g[:n-lag] -= p * hx[lag:]
                g[lag:] -= p * hx[:n-lag]
        return (2.0 / n) * g
    
    bounds = [(0.0, 1.0)] * n
    cons = ({"type": "eq", "fun": lambda x: float(np.sum(x) - target),
             "jac": lambda x: np.ones(n)},)
    
    try:
        res = minimize(obj, h0, jac=obj_grad, method="SLSQP",
                       bounds=bounds, constraints=cons,
                       options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
        if res.x is not None and np.all(np.isfinite(res.x)):
            cand = project_density(res.x, target)
            c = evaluate_erdos(cand)
            if c < best_c:
                return cand, c
    except Exception:
        pass
    return h0, best_c

def upsample_h(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    if n0 == n:
        return p.copy()
    if n0 <= 0:
        return project_density(np.full(max(1, n), 0.5), 0.5 * n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return project_density(h, 0.5 * n)

def downsolve_and_resample(h, n_target, deadline_fix):
    """Downsample, optimize briefly, upsample back"""
    n = h.size
    # Downsample
    x_fine = (np.arange(n) + 0.5) * (2.0 / float(n))
    x_coarse = (np.arange(n_target) + 0.5) * (2.0 / float(n_target))
    h_coarse = np.interp(x_coarse, x_fine, h, left=float(h[0]), right=float(h[-1]))
    h_coarse = project_density(h_coarse, 0.5 * n_target)
    # Brief polish at coarse
    h_coarse, c_coarse = local_search_binary(h_coarse, deadline_fix, max_iters=2000, top_frac=0.08, min_k=6, n_cand=10)
    # Upsample back
    h_back = upsample_h(h_coarse, n)
    return h_back

def structured_start(n, mode, rng):
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    if mode == 0:
        h = 0.5 + 0.4 * np.sin(2 * np.pi * 3 * x / 2.0 + rng.uniform(0, 6.28))
        h += 0.15 * np.sin(2 * np.pi * 5 * x / 2.0 + rng.uniform(0, 6.28))
    elif mode == 1:
        h = np.zeros(n)
        block = max(1, n // 16)
        for i in range(0, n, block * 2):
            end = min(i + block, n)
            h[i:end] = 1.0
        h = np.clip(h + rng.normal(0, 0.05, n), 0, 1)
    elif mode == 2:
        h = (np.sin(2 * np.pi * 7 * x / 2.0) > 0).astype(float)
        h = np.clip(h + rng.normal(0, 0.1, n), 0, 1)
    elif mode == 3:
        h = (np.sin(2 * np.pi * 2 * x / 2.0) > 0.3).astype(float) * 0.8
        h += 0.2 * (np.sin(2 * np.pi * 4 * x / 2.0) > 0.5).astype(float)
        h = np.clip(h + rng.normal(0, 0.05, n), 0, 1)
    else:
        h = np.clip(0.5 + 0.5 * rng.random(n), 0, 1)
    return project_density(h, 0.5 * n)

def run(seed=42, budget_s=1300, **kwargs):
    start = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = start + max(10.0, budget - 15.0)
    rng = np.random.default_rng(seed)
    
    best_h = None
    best_c = float("inf")
    best_n = 0
    
    # Get parent
    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0).astype(float, copy=True)
            parent = np.clip(parent, 0.0, 1.0)
    except Exception:
        parent = None
    
    if parent is not None:
        pn = parent.size
        pt = 0.5 * pn
        pp = project_density(parent, pt)
        cp = evaluate_erdos(pp)
        best_h = pp.copy()
        best_c = cp
        best_n = pn
    else:
        n0 = 180
        best_h = structured_start(n0, 0, rng)
        best_c = evaluate_erdos(best_h)
        best_n = n0
    
    n_use = best_n
    
    # Phase 1: Binarize and evaluate (quick check)
    h_bin = binarize_smart(best_h, 0.5 * n_use, thresh_lo=0.3, thresh_hi=0.7)
    c_bin = evaluate_erdos(h_bin)
    if c_bin < best_c:
        best_h = h_bin.copy()
        best_c = c_bin
    
    # Also try different thresholds
    for tl, th in [(0.4, 0.6), (0.25, 0.75), (0.45, 0.55)]:
        h_t = binarize_smart(best_h, 0.5 * n_use, thresh_lo=tl, thresh_hi=th)
        c_t = evaluate_erdos(h_t)
        if c_t < best_c:
            best_h = h_t.copy()
            best_c = c_t
    
    # Phase 2: Local search on current best (structured + annealing)
    rem = deadline - time.perf_counter()
    if rem > 30.0:
        sub_dl = time.perf_counter() + min(rem * 0.30, 300.0)
        h1, c1 = local_search_binary(best_h, sub_dl, max_iters=30000,
                                      top_frac=0.06, min_k=8, n_cand=12,
                                      anneal_temp=0.0004, anneal_rate=0.9998)
        if c1 < best_c:
            best_h = h1.copy()
            best_c = c1
    
    # Phase 3: Multi-pair search
    rem = deadline - time.perf_counter()
    if rem > 20.0:
        sub_dl = time.perf_counter() + min(rem * 0.20, 200.0)
        h2, c2 = multi_pair_search(best_h, sub_dl, max_iters=3000,
                                    top_frac=0.06, min_k=8, n_cand=8)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = c2
    
    # Phase 4: Rotation search
    rem = deadline - time.perf_counter()
    if rem > 10.0:
        h3, c3 = rotation_search(best_h, time.perf_counter() + 5.0, max_shifts=8)
        if c3 < best_c:
            best_h = h3.copy()
            best_c = c3
    
    # Phase 5: Try different resolutions
    rem = deadline - time.perf_counter()
    if rem > 60.0:
        # Try at n_use * 0.8 and n_use * 1.2
        alt_ns = []
        for factor in (0.7, 0.8, 1.2, 1.33, 1.5):
            alt_n = int(round(n_use * factor))
            alt_n = max(60, min(600, alt_n))
            if alt_n != n_use and alt_n not in alt_ns:
                alt_ns.append(alt_n)
        
        for alt_n in alt_ns[:3]:
            rem = deadline - time.perf_counter()
            if rem < 30.0:
                break
            sub_dl = time.perf_counter() + min(rem * 0.25, 120.0)
            h_alt = upsample_h(best_h, alt_n)
            # Quick local search
            h_alt, c_alt = local_search_binary(h_alt, sub_dl, max_iters=10000,
                                                top_frac=0.06, min_k=8, n_cand=12,
                                                anneal_temp=0.0003, anneal_rate=0.9999)
            if c_alt < best_c:
                best_h = h_alt.copy()
                best_c = c_alt
                best_n = alt_n
    
    # Phase 6: Structured diverse starts at best resolution
    rem = deadline - time.perf_counter()
    if rem > 40.0:
        n_best = int(best_h.size)
        for mode in range(4):
            rem = deadline - time.perf_counter()
            if rem < 25.0:
                break
            sub_dl = time.perf_counter() + min(rem * 0.2, 80.0)
            h_struct = structured_start(n_best, mode, rng)
            h_struct, c_struct = local_search_binary(h_struct, sub_dl, max_iters=15000,
                                                      top_frac=0.06, min_k=8, n_cand=12,
                                                      anneal_temp=0.0005, anneal_rate=0.9997)
            if c_struct < best_c:
                best_h = h_struct.copy()
                best_c = c_struct
                best_n = n_best
    
    # Phase 7: Extended local search on best (more iterations, tighter params)
    rem = deadline - time.perf_counter()
    if rem > 30.0:
        sub_dl = time.perf_counter() + min(rem * 0.35, 350.0)
        h4, c4 = local_search_binary(best_h, sub_dl, max_iters=50000,
                                      top_frac=0.05, min_k=10, n_cand=15,
                                      anneal_temp=0.0002, anneal_rate=0.9999)
        if c4 < best_c:
            best_h = h4.copy()
            best_c = c4
    
    # Phase 8: Binarize again after optimization
    rem = deadline - time.perf_counter()
    if rem > 10.0:
        h_bin2 = binarize_smart(best_h, 0.5 * best_n, thresh_lo=0.4, thresh_hi=0.6)
        c_bin2 = evaluate_erdos(h_bin2)
        if c_bin2 < best_c:
            best_h = h_bin2.copy()
            best_c = c_bin2
        # Quick polish after binarization
        if rem > 20.0:
            sub_dl = time.perf_counter() + min(rem * 0.4, 60.0)
            h5, c5 = local_search_binary(best_h, sub_dl, max_iters=20000,
                                          top_frac=0.05, min_k=10, n_cand=15,
                                          anneal_temp=0.0002, anneal_rate=0.9999)
            if c5 < best_c:
                best_h = h5.copy()
                best_c = c5
    
    # Phase 9: Multi-pair on final
    rem = deadline - time.perf_counter()
    if rem > 20.0:
        sub_dl = time.perf_counter() + min(rem * 0.3, 80.0)
        h6, c6 = multi_pair_search(best_h, sub_dl, max_iters=3000,
                                    top_frac=0.05, min_k=10, n_cand=10)
        if c6 < best_c:
            best_h = h6.copy()
            best_c = c6
    
    # Phase 10: Continuous polish (SLSQP) on the best
    rem = deadline - time.perf_counter()
    if rem > 20.0:
        n_best = int(best_h.size)
        if n_best <= 400:
            sub_dl = time.perf_counter() + min(rem * 0.3, 60.0)
            h7, c7 = continuous_polish(best_h, sub_dl, alpha=80.0, max_iter=80)
            if c7 < best_c:
                best_h = h7.copy()
                best_c = c7
    
    # Phase 11: Final intensive local search
    rem = deadline - time.perf_counter()
    if rem > 15.0:
        sub_dl = time.perf_counter() + min(rem * 0.85, 200.0)
        h8, c8 = local_search_binary(best_h, sub_dl, max_iters=40000,
                                      top_frac=0.04, min_k=12, n_cand=20,
                                      anneal_temp=0.0001, anneal_rate=0.99995)
        if c8 < best_c:
            best_h = h8.copy()
            best_c = c8
    
    # Final: ensure valid output
    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    
    return final_h, float(final_c), int(final_h.size)
