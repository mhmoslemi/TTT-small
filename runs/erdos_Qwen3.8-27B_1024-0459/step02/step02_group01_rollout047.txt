def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 0.0) & (h < 1.0)
        if not np.any(mask):
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        else:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(1.0 - h)[:n - n//2] if False else np.argsort(1.0 - h)
            for i in idx:
                amt = min(1.0 - h[i], diff)
                h[i] += amt
                diff -= amt
                if diff <= 1e-15:
                    break
        else:
            idx = np.argsort(h)
            for i in idx:
                amt = min(h[i], -diff)
                h[i] -= amt
                diff += amt
                if diff >= -1e-15:
                    break
    return h

def soft_max_obj(h, n, beta):
    dx = 2.0 / n
    c = compute_corr(h)
    m = np.max(c)
    shifted = beta * (c - m)
    shifted = np.clip(shifted, -700.0, 0.0)
    logsum = m + np.log(np.sum(np.exp(shifted))) / beta
    return logsum * dx

def soft_max_grad(h, n, beta):
    dx = 2.0 / n
    c = compute_corr(h)
    m = np.max(c)
    shifted = beta * (c - m)
    shifted = np.clip(shifted, -700.0, 0.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    
    grad = np.zeros(n)
    nonzero_w = w > 1e-15
    for k in np.where(nonzero_w)[0]:
        wi = w[k]
        offset = k - (n - 1)
        if offset >= 0:
            if n - offset > 0:
                grad[0:n-offset] += wi * (1.0 - h[offset:n])
                grad[offset:n] += wi * (-h[0:n-offset])

            o = -offset
            if n - o > 0:
                grad[o:n] += wi * (1.0 - h[0:n-o])
                grad[0:n-o] += wi * (-h[o:n])
    return dx * grad

def proj_grad_step(h, n, alpha, n_sub=4):
    """Projected gradient step with active-set tracking."""
    dx = 2.0 / n
    for _ in range(n_sub):

        active = c >= max_val - 1e-12
        n_act = max(1, np.sum(active))
        
        offset_list = np.where(active)[0]
        grad = np.zeros(n)

            offset = k - (n - 1)

def coordinate_descent(h, n, max_iter=200, delta=None):
    """Local search via coordinate descent with projection."""
    if delta is None:
        delta = 0.02
    best_c5 = compute_c5(h)
    h = h.copy()
    
    for iteration in range(max_iter):
        improved = False
        for i in range(n):
            for new_val in [h[i] + delta, h[i] - delta, 0.0, 1.0]:
                if new_val < 0 or new_val > 1 or abs(new_val - h[i]) < 1e-10:
                    continue
                h[i] = new_val
                h = project_feasible(h, n)
                c5 = compute_c5(h)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    improved = True
                else:
                    h[i] = np.clip(h[i] + (h[i] - new_val), 0.0, 1.0)
                    h = project_feasible(h, n)
        if not improved:
            if delta > 0.005:
                delta *= 0.5
            else:
                break
    return h, best_c5

    h = h.copy()
