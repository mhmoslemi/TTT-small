def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)

    M = np.zeros((n, 2*n-1))

            val = 0.0
            if 0 <= j - s <= n-1:
                val += (1.0 - h[j - s])
            if 0 <= j + s <= n-1:
                val -= h[j + s]
            M[j, k] = val
    
    grad = dx * (M @ p)
    return grad

def projected_gradient_step(h, grad, lr, n):
    h_new = h - lr * grad
    return make_feasible(h_new, n)

def projected_gradient_opt(h, n, betas, steps_per_beta, nlr, rng):

    dx = 2.0 / n

                noise = rng.randn(n) * 0.005
                h = make_feasible(np.clip(h + noise, 0, 1), n)
    
    return best_h, best_c5

                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def compute_gradient(h, p_full, n):
    """Compute gradient of log-sum-exp smoothed objective.
    p_full: softmax weights over corr values, shape (2n-1,)
    """
    dx = 2.0 / n
    j_idx = np.arange(n)[:, None]
    i_idx = np.arange(n)[None, :]
    idx1 = j_idx - i_idx + n - 1
    idx2 = i_idx - j_idx + n - 1
    term1 = (p_full[idx1] * (1.0 - h)[None, :]).sum(axis=1)
    term2 = (p_full[idx2] * h[None, :]).sum(axis=1)
    return dx * (term1 - term2)

def get_p_full(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_shifted = corr - np.max(corr)
    p = np.exp(beta * corr_shifted)
    s = np.sum(p)
    if s < 1e-300:
        p = np.ones(len(corr)) / len(corr)
    else:
        p /= s
    return p

def projected_gradient_opt(h, n, betas, steps_per_beta, lr_init, rng):
    """Polish with projected gradient using log-sum-exp smoothing."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for beta in betas:
        lr = lr_init / (1.0 + 0.01 * beta)
        for step in range(steps_per_beta):
            p_full = get_p_full(h, beta)
            grad = compute_gradient(h, p_full, n)
            h_new = h - lr * grad
            h_new = make_feasible(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
        # Snap near-binary values
        h[np.abs(h - 0.0) < 0.02] = 0.0
        h[np.abs(h - 1.0) < 0.02] = 1.0
        h = make_feasible(h, n)
        c5_snap = compute_c5(h)
        if c5_snap < best_c5:
            best_c5 = c5_snap
            best_h = h.copy()
    
    return best_h, best_c5

def binary_sa(n, rng, n_steps, t_start, t_end):
    """Simulated annealing on binary strings."""
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for step in range(n_steps):
        T = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_h = h.copy()
            best_c5 = c5_new
        elif rng.random() < np.exp(-(c5_new - best_c5) / max(T, 1e-15)):
            pass
        else:
            h[i], h[j] = h[j], h[i]
    
    return best_h, best_c5

def binary_local_search(h_bin, n, n_trials, rng):
    """Local search on binary string using random swaps."""
    h = h_bin.copy()
    best_c5 = compute_c5(h)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    for t in range(n_trials):
        if len(ones) == 0 or len(zeros) == 0:
            break
        oi = rng.randint(len(ones))
        zi = rng.randint(len(zeros))
        i, j = ones[oi], zeros[zi]
        h[i], h[j] = 0.0, 1.0
        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
        else:
            h[i], h[j] = 1.0, 0.0
    
    return h, best_c5

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)
