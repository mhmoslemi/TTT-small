        pop_cvals.append(evaluate_erdos(h))

    best_idx = np.argmin(pop_cvals)
    best_c = pop_cvals[best_idx]
    best_h = pop_vecs[best_idx].copy()
    
    F = 0.8  # mutation factor
    CR = 0.7  # crossover rate
    
    no_improve = 0
    
    for gen in range(max_gen):
        if time.perf_counter() >= deadline:
            break
        
        for i in range(pop):
            if time.perf_counter() >= deadline:
                    break
            
            # Select 3 distinct individuals (different from i)
            candidates = list(range(pop))
            candidates.remove(i)
            r1, r2, r3 = rng.choice(candidates, size=3, replace=False)
            
            # Mutation
            v = pop_vecs[best_idx] + F * (pop_vecs[r1] - pop_vecs[r2])
            # Crossover
            j_rand = rng.integers(n)
            for j in range(n):
                if j != j_rand and rng.random() > CR:
                    v[j] = pop_vecs[i][j]
            
            # Project
            v = project_density(v, target)
            cv = evaluate_erdos(v)
            
            # Selection
            if cv < pop_cvals[i]:
                pop_vecs[i] = v
                pop_cvals[i] = cv
                if cv < best_c:
                    best_c = cv
                    best_h = v.copy()
                    best_idx = i
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        
        if no_improve >= pop * 5:
            break
    
    return best_h, best_c

        pop_cvals.append(evaluate_erdos(h))

    pop_cvals = np.array(pop_cvals)
    best_idx = int(np.argmin(pop_cvals))
    best_c = float(pop_cvals[best_idx])
    best_h = pop_vecs[best_idx].copy()

    no_improve = 0
    F = (F_range[0] + F_range[1]) / 2.0
    CR = 0.7

    for gen in range(max_gen):
        if time.perf_counter() >= deadline:
            break
        if gen % 30 == 29:
            F = float(rng.uniform(F_range[0], F_range[1]))
            CR = float(rng.uniform(0.5, 0.9))

        gen_best = best_c
        for i in range(pop_size):
            if time.perf_counter() >= deadline:
                break
            cands = list(range(pop_size))
            cands.remove(i)
            r1 = int(cands[int(rng.integers(len(cands)))])
            c2 = list(cands)
            c2.remove(r1)
            r2 = int(c2[int(rng.integers(len(c2)))])
            r3 = int(c2[int(rng.integers(len(c2)))])

            if r3 == i:
                r3 = (r3 + 1) % pop_size

            v = pop_vecs[best_idx].copy()
            v += F * (pop_vecs[r1] - pop_vecs[r2])
            j_rand = int(rng.integers(n))
            for j in range(n):
                if j != j_rand and rng.random() > CR:
                    v[j] = pop_vecs[i][j]
            v = project_density(v, target)
            cv = evaluate_erdos(v)

            if cv < pop_cvals[i]:
                pop_vecs[i] = v
                pop_cvals[i] = cv
                if cv < best_c:
                    best_c = cv
                    best_h = v.copy()
                    best_idx = i
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1

        if no_improve >= pop_size * 8:
            break

    return best_h, best_c


def pair_swap_guided(h, deadline, max_iter=50000, n_top=8, n_pos=10, n_neg=10):
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    if n < 2:
        return best, best_c
    rng = np.random.default_rng(12345)
    stall = 0
    max_stall = 300

    for it in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        if it % 200 == 0:
            lags, vals, corr = critical_lags(best, n_top)
            dv = vals - vals[0]
            w = np.exp(np.clip(dv, -10, 10) / max(1e-10, abs(dv).max() + 1e-10))
            w /= w.sum()
            g = np.zeros(n)
            for ii, lag in enumerate(lags):
                g += w[ii] * corr_gradient_at_lag(best, int(lag))

        neg_cand = np.argsort(g)[:n_neg]
        pos_cand = np.argsort(-g)[:n_pos]
        found = False
        for ii in neg_cand:
            ii = int(ii)
            for jj in pos_cand:
                jj = int(jj)
                if ii == jj:
                    continue
                cap = min(1.0 - best[ii], best[jj])
                if cap < 1e-12:
                    continue
                for frac in (0.3, 0.6, 1.0):
                    d = frac * cap
                    cand = best.copy()
                    cand[ii] += d
                    cand[jj] -= d
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-13:
                        best = cand
                        best_c = c
                        found = True
                        break
                if found:
                    break
            if found:
                break

        if not found:
            for _r in range(8):
                ii = int(rng.integers(n))
                jj = int(rng.integers(n))
                if ii == jj:
                    continue
                cap = min(1.0 - best[ii], best[jj])
                if cap < 1e-12:
                    continue
                d = float(rng.random()) * 0.7 * cap
                cand = best.copy()
                cand[ii] += d
                cand[jj] -= d
                c = evaluate_erdos(cand)
                if c < best_c - 1e-13:
                    best = cand
                    best_c = c
                    found = True
                    break
            if not found:
                stall += 1
                if stall > max_stall:
                    break
        else:
            stall = 0

    best = project_density(best, target)
    return best, evaluate_erdos(best)


def gradient_descent_active(h, deadline, max_iter=5000, top_frac=0.08, min_k=8):
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    if n < 2:
        return best, best_c
    stall = 0

    for _ in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        lags, vals, corr = critical_lags(best, max(min_k, int(np.ceil(len(corr) * top_frac))))
        dv = vals - vals.min()
        w = np.exp(np.clip(dv, -10, 10) / max(1e-10, dv.max() + 1e-10))
        w /= w.sum()
        g = np.zeros(n)
        for ii, lag in enumerate(lags):
            g += w[ii] * corr_gradient_at_lag(best, int(lag))

        d = -g
        d -= np.mean(d)
        norm = np.linalg.norm(d)
        if norm < 1e-14:
            break
        d /= norm
        improved = False
        for step in (0.002, 0.005, 0.01, 0.03, 0.08, 0.15, 0.3):
            cand = project_density(best + step * d, target)
            c = evaluate_erdos(cand)
            if c < best_c - 1e-13:
                best = cand
                best_c = c
                improved = True
                break
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 50:
                break

    best = project_density(best, target)
    return best, evaluate_erdos(best)


def binary_flip_search(h, deadline, max_iter=20000):
    n = h.size
    target = 0.5 * n
    bg = np.round(h)
    bg = np.clip(bg, 0, 1)
    s = np.sum(bg)
    if s > target:
        excess = int(s - target)
        frac = bg[np.argsort(1 - bg)[:excess]]
        bg[np.argsort(1 - bg)[:excess]] = 0
    elif s < target:
        deficit = int(target - s)
        bg[np.argsort(bg)[:deficit]] = 1
    bg = project_density(bg, target)
    best_c = evaluate_erdos(bg)
    best = bg.copy()
    rng = np.random.default_rng(777)
    stall = 0

    one_idx = np.where(bg > 0.5)[0]
    zero_idx = np.where(bg < 0.5)[0]

    for _ in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        if one_idx.size < 2 or zero_idx.size < 2:
            break
        i1 = int(one_idx[int(rng.integers(len(one_idx)))])
        i2 = int(zero_idx[int(rng.integers(len(zero_idx)))])
        cand = best.copy()
        cand[i1] = 0.0
        cand[i2] = 1.0
        c = evaluate_erdos(cand)
        if c < best_c - 1e-13:
            best = cand
            best_c = c
            bg = cand.copy()
            one_idx = np.where(bg > 0.5)[0]
            zero_idx = np.where(bg < 0.5)[0]
            stall = 0
        else:
            stall += 1
            if stall > 500:
                break

    return best, best_c


def block_swap_search(h, deadline, max_iter=5000, max_block=8):
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    rng = np.random.default_rng(55555)
    stall = 0

    for _ in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        blen = int(rng.integers(2, max_block + 1))
        i = int(rng.integers(0, n - blen - 1))
        j = int(rng.integers(0, n - blen - 1))
        if abs(i - j) <= blen:
            stall += 1
            if stall > 200:
                break
            continue
        cand = best.copy()
        cand[i:i+blen], cand[j:j+blen] = best[j:j+blen].copy(), best[i:i+blen].copy()
        c = evaluate_erdos(cand)
        if c < best_c - 1e-13:
            best = cand
            best_c = c
            stall = 0
        else:
            stall += 1
            if stall > 200:
                break

    best = project_density(best, target)
    return best, evaluate_erdos(best)


def smooth_logsumexp(h, n, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -745.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not np.isfinite(z):
        return 1e100
    return float((2.0 / float(n)) * (m + np.log(z) / alpha))


def slsqp_refine(h, n, deadline, max_iter=100):
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        if not np.all(np.isfinite(x)):
            return 1e100
        return smooth_logsumexp(x, n, 80.0)

    def constraint_fun(x):
        return float(np.sum(x) - target)

    def constraint_jac(x):
        return np.ones(n, dtype=float)

    constraints = ({"type": "eq", "fun": constraint_fun, "jac": constraint_jac},)
    try:
        res = minimize(
            obj, best.copy(),
            method="SLSQP",
            bounds=bounds, constraints=constraints,
            options={"maxiter": max_iter, "ftol": 1e-12, "disp": False}
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            cand = project_density(res.x, target)
            c = evaluate_erdos(cand)
            if c < best_c:
                best = cand
                best_c = c
    except Exception:
        pass
    return best, best_c


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = start_time + max(1.0, budget - 20.0)
    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float("inf")
    best_n = 0

    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.clip(np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0).copy()
            best_n = int(parent.size)
            best_h = project_density(parent, 0.5 * best_n)
            best_c = evaluate_erdos(best_h)
    except Exception:
        best_n = 450
        x = (np.arange(best_n) + 0.5) * (2.0 / best_n)
        init = 0.5 + 0.3 * np.sin(2.0 * np.pi * 3 * x / 2.0) + 0.15 * np.sin(2.0 * np.pi * 5 * x / 2.0 + 1.0)
        best_h = project_density(init, 0.5 * best_n)
        best_c = evaluate_erdos(best_h)

    n_base = best_n
    target = 0.5 * n_base

    # Phase 1: DE at base resolution (multiple diverse populations)
    rem = deadline - time.perf_counter()
    if rem > 60:
        init_pool = [best_h.copy()]
        for mm in range(5):
            mode = mm % 4
            if mode == 0:
                x = np.arange(n_base) / n_base * 2.0
                h = 0.5 + 0.35 * np.sin(2*np.pi*int(rng.integers(2,7))*x/2.0 + rng.uniform(0,6.28))
                h = np.clip(h, 0, 1)
            elif mode == 1:
                nblocks = int(rng.integers(3, 10))
                h = np.zeros(n_base)
                for b in range(nblocks):
                    if b % 2 == 0:
                        a, bnd = b*n_base//nblocks, (b+1)*n_base//nblocks
                        h[a:bnd] = 1.0
                h += rng.normal(0, 0.1, n_base)
                h = np.clip(h, 0, 1)
            elif mode == 2:
                h = np.clip(0.5 + rng.normal(0, 0.4, n_base), 0, 1)
            else:
                seg = max(1, n_base // int(rng.integers(6, 14)))
                h = np.zeros(n_base)
                for i in range(0, n_base, seg):
                    h[i:min(i+seg, n_base)] = rng.uniform(0.1, 0.9)
                h = np.clip(h, 0, 1)
            init_pool.append(project_density(h, target))

        de_alloc = min(rem * 0.25, 200.0)
        h_de, c_de = de_search(n_base, 30, 150, time.perf_counter() + de_alloc, rng, init_pool, target, (0.3, 1.0))
        if c_de < best_c:
            best_h = h_de.copy()
            best_c = c_de

    # Phase 2: DE with smaller mutation (exploitative)
    rem = deadline - time.perf_counter()
    if rem > 30:
        init_pool2 = [best_h.copy(), best_h + rng.normal(0, 0.05, n_base)]
        init_pool2[1] = project_density(init_pool2[1], target)
        de_alloc = min(rem * 0.2, 150.0)
        h_de2, c_de2 = de_search(n_base, 25, 100, time.perf_counter() + de_alloc, rng, init_pool2, target, (0.2, 0.6))
        if c_de2 < best_c:
            best_h = h_de2.copy()
            best_c = c_de2

    # Phase 3: SLSQP refinement
    rem = deadline - time.perf_counter()
    if rem > 30 and n_base <= 500:
        slsqp_dl = time.perf_counter() + min(rem * 0.15, 60.0)
        h_sq, c_sq = slsqp_refine(best_h, n_base, slsqp_dl, max_iter=150)
        if c_sq < best_c:
            best_h = h_sq.copy()
            best_c = c_sq

    # Phase 4: Intensive pair swap guided by critical lags
    rem = deadline - time.perf_counter()
    if rem > 10:
        ps_dl = time.perf_counter() + min(rem * 0.35, 300.0)
        h_ps, c_ps = pair_swap_guided(best_h, ps_dl, max_iter=80000, n_top=10, n_pos=12, n_neg=12)
        if c_ps < best_c:
            best_h = h_ps.copy()
            best_c = c_ps

    # Phase 5: Gradient descent on active set
    rem = deadline - time.perf_counter()
    if rem > 10:
        gd_dl = time.perf_counter() + min(rem * 0.25, 200.0)
        h_gd, c_gd = gradient_descent_active(best_h, gd_dl, max_iter=8000, top_frac=0.06, min_k=10)
        if c_gd < best_c:
            best_h = h_gd.copy()
            best_c = c_gd

    # Phase 6: Binary flip search
    rem = deadline - time.perf_counter()
    if rem > 10:
        bf_h = np.clip(best_h + rng.normal(0, 0.15, n_base), 0, 1)
        bf_h = project_density(bf_h, target)
        bf_dl = time.perf_counter() + min(rem * 0.15, 100.0)
        h_bf, c_bf = binary_flip_search(bf_h, bf_dl, max_iter=30000)
        if c_bf < best_c:
            best_h = h_bf.copy()
            best_c = c_bf

    # Phase 7: Block swap
    rem = deadline - time.perf_counter()
    if rem > 10:
        bs_dl = time.perf_counter() + min(rem * 0.1, 60.0)
        h_bs, c_bs = block_swap_search(best_h, bs_dl, max_iter=10000, max_block=10)
        if c_bs < best_c:
            best_h = h_bs.copy()
            best_c = c_bs

    # Phase 8: Try a different resolution (upsample best)
    rem = deadline - time.perf_counter()
    if rem > 60 and n_base <= 360:
        n_up = min(n_base * 2, 720)
        h_up = upsample_h(best_h, n_up)
        target_up = 0.5 * n_up
        init_up = [h_up.copy()]
        for mm in range(3):
            h_r = upsample_h(best_h + rng.normal(0, 0.1, n_base), n_up)
            init_up.append(h_r)
        de_alloc_up = min(rem * 0.25, 120.0)
        h_u, c_u = de_search(n_up, 20, 80, time.perf_counter() + de_alloc_up, rng, init_up, target_up, (0.3, 0.8))
        if c_u < best_c:
            best_h = h_u.copy()
            best_c = c_u
            best_n = n_up
            # Polish at new resolution
            rem2 = deadline - time.perf_counter()
            if rem2 > 20:
                h_pol, c_pol = pair_swap_guided(best_h, time.perf_counter() + min(rem2*0.5, 100.0), max_iter=30000, n_top=8, n_pos=10, n_neg=10)
                if c_pol < best_c:
                    best_h = h_pol.copy()
                    best_c = c_pol

    # Phase 9: More pair swaps (second round, different random seed)
    rem = deadline - time.perf_counter()
    if rem > 15:
        ps2_dl = time.perf_counter() + min(rem * 0.3, 200.0)
        h_ps2, c_ps2 = pair_swap_guided(best_h, ps2_dl, max_iter=60000, n_top=8, n_pos=8, n_neg=8)
        if c_ps2 < best_c:
            best_h = h_ps2.copy()
            best_c = c_ps2

    # Phase 10: Final gradient + pair swap polish
    rem = deadline - time.perf_counter()
    if rem > 10:
        h_fin, c_fin = gradient_descent_active(best_h, time.perf_counter() + min(rem * 0.4, 100.0), max_iter=5000, top_frac=0.04, min_k=12)
        if c_fin < best_c:
            best_h = h_fin.copy()
            best_c = c_fin
        rem2 = deadline - time.perf_counter()
        if rem2 > 5:
            h_ps3, c_ps3 = pair_swap_guided(best_h, time.perf_counter() + rem2 * 0.8, max_iter=40000, n_top=12, n_pos=14, n_neg=14)
            if c_ps3 < best_c:
                best_h = h_ps3.copy()
                best_c = c_ps3

    # Final projection and evaluation
    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    final_n = int(final_h.size)

    # One last check: push near-binary
    thresh = 0.05
    h_push = final_h.copy()
    for i in range(final_n):
        if h_push[i] < thresh:
            h_push[i] = 0.0
        elif h_push[i] > 1.0 - thresh:
            h_push[i] = 1.0
    h_push = project_density(h_push, 0.5 * final_n)
    c_push = evaluate_erdos(h_push)
    if c_push < final_c:
        final_h = h_push
        final_c = c_push

    return final_h, float(final_c), int(final_h.size)
