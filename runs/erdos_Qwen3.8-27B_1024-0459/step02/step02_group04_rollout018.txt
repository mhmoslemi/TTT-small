        vals[n-1+k] = (total - pref[k] - ac) * dx  # positive lag

    # Gradient

    # Lag 0 (center): f_0 = (total - sum h^2) * dx
    # df_0/dh[i] = -2*h[i] * dx

    temp = 0.003
    min_temp = 1e-7
    max_len = min(6, n)
    for _ in range(int(n_trials)):
        if time.time() > time_limit:
            break
        L = int(rng.randint(1, max_len + 1))
        st = int(rng.randint(0, n - L + 1))
        idx = np.arange(st, st + L)
        delta = rng.randn(L) * 0.025
        h_try = cur_h.copy()
        h_try[idx] += delta
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try)
        if c_try < best_c:
            best_h = h_try.copy()
            best_c = c_try
            cur_h = h_try
            cur_c = c_try
        elif c_try < cur_c:
            cur_h = h_try
            cur_c = c_try
        else:
            d = c_try - cur_c
            if 0.0 < d < temp:
                prob = np.exp(-d / temp * 50.0)
                if rng.rand() < prob:
                    cur_h = h_try
                    cur_c = c_try
        temp = max(temp * 0.998, min_temp)
    return best_h, float(best_c)


def targeted_perturb(h0, n, time_limit, seed):
    rng = np.random.RandomState(int(seed))
    n = int(n)
    h = project_feasible(h0, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    top_k = 20
    for _ in range(2000):
        if time.time() > time_limit:
            break
        active_lags = choose_active_lags(h, top_k, margin=1e-4)
        lag = int(active_lags[int(rng.randint(0, len(active_lags)))])
        L = abs(lag)
        if L == 0:
            delta = rng.randn(n) * 0.01
        else:
            m = n - L
            delta = np.zeros(n)
            if lag > 0:
                perturb = rng.randn(m) * 0.02
                delta[:m] += perturb
                delta[L:] -= perturb
            else:
                perturb = rng.randn(m) * 0.02
                delta[L:] += perturb
                delta[:m] -= perturb
        h_try = h.copy()
        h_try += delta
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try)
        if c_try < best_c:
            best_h = h_try.copy()
            best_c = c_try
            h = h_try
        elif c_try < c + 0.5e-5:
            if rng.rand() < 0.3:
                h = h_try
                c = c_try
    return best_h, float(best_c)


def run(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    budget_s = float(budget_s)
    rng = np.random.RandomState(seed)
    t_start = time.time()
    hard_end = t_start + max(120.0, budget_s - 30.0)

    parent = get_parent_values()
    if parent is not None:
        parent_n = int(parent.size)
        proj = project_feasible(parent, parent_n)
        best_h = proj
        best_c = compute_c5(proj)
        best_n = parent_n
    else:
        best_n = 160
        best_h = project_feasible(np.full(best_n, 0.5), best_n)
        best_c = compute_c5(best_h)

    # Phase 1: Nesterov smooth gradient at primary resolution (n=best_n)
    phase1_end = min(hard_end - 400.0, t_start + 250.0)
    n_work = best_n
    h_cur = best_h.copy()
    for beta in (10.0, 30.0, 60.0, 120.0, 250.0):
        if time.time() > phase1_end - 10.0:
            break
        lr = 0.8 / (2.0 * beta * 0.6 + 1.0)
        lr = min(lr, 0.5)
        max_it = 400 if beta < 50 else 200
        t_lim = min(phase1_end - 5.0, time.time() + 45.0)
        h2, c2 = nesterov_smooth_opt(h_cur, n_work, beta, max_it, lr, t_lim)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = c2
        h_cur = h2 if c2 < best_c + 1e-4 else best_h

    # L-BFGS-B at moderate beta
    if time.time() < phase1_end - 30.0:
        h2, c2 = lfbgbs_smooth(best_h, best_n, 40.0, 100, min(phase1_end - 10.0, time.time() + 60.0))
        if c2 < best_c:
            best_h = h2
            best_c = c2

    # Phase 2: Active set refinement at primary n
    phase2_end = min(hard_end - 250.0, t_start + 700.0)
    if time.time() < phase2_end - 10.0:
        h2, c2 = refine_active_loop(best_h, best_n, phase2_end, max_passes=8)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    # Phase 3: Try other resolutions
    phase3_end = min(hard_end - 150.0, t_start + 1050.0)
    other_ns = [192, 224, 128, 256]
    for nn in other_ns:
        if time.time() > phase3_end - 40.0:
            break
        if nn in (best_n,):
            continue
        slot_end = min(phase3_end - 10.0, time.time() + 90.0)
        if slot_end - time.time() < 30.0:
            continue
        inits = generate_diverse_starts(nn, rng, parent_h=best_h)
        for h0 in inits[:3]:
            if time.time() > slot_end - 10.0:
                break
            sub_end = min(slot_end - 5.0, time.time() + 50.0)
            if sub_end - time.time() < 15.0:
                break
            # Quick gradient
            h_g, c_g = nesterov_smooth_opt(h0, nn, 60.0, 150, 0.3, sub_end)
            # Active refine
            if time.time() < slot_end - 15.0:
                h_a, c_a = refine_active_loop(h_g, nn, slot_end, max_passes=3)
                if c_a < c_g:
                    h_g, c_g = h_a, c_a
            if c_g < best_c:
                best_h = h_g
                best_c = c_g
                best_n = nn

    # Phase 4: Final polish on best solution
    phase4_end = hard_end - 10.0
    if time.time() < phase4_end - 30.0:
        h2, c2 = refine_active_loop(best_h, best_n, min(phase4_end - 50.0, time.time() + 120.0), max_passes=6)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    if time.time() < phase4_end - 25.0:
        h2, c2 = targeted_perturb(best_h, best_n, min(phase4_end - 20.0, time.time() + 60.0), seed + 777)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    if time.time() < phase4_end - 10.0:
        h2, c2 = local_block_perturb(best_h, best_n, phase4_end, seed + 999, n_trials=5000)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)
    return (best_h, float(best_c), int(best_n))
