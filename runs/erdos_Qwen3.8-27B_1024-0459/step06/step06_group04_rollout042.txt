from scipy.optimize import minimize
import time
import cvxpy as cp

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if interior.any():
            h[interior] += diff / interior.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_fk(h, off, n):
    """Gradient of f_k with respect to h, where off = k_idx - (n-1)."""
    dx = 2.0 / n
    g = np.zeros(n)
    if off >= 0:
        np_ = n - off
        g[off:] += (1.0 - h[:np_])
        g[:np_] -= h[off:]
    else:
        ao = -off
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:])
        g[ao:] -= h[:np_]
    return g * dx


def solve_lp(h, n, active_idx, trust_radius=0.5):
    """Solve the LP linearization around h with active set active_idx."""
    dx = 2.0 / n
    k_max = n - 1
    corr = compute_corr(h)
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n / 2.0,
    ]
    
    # Trust region
    diff = h_var - h
    constraints.append(cp.norm(diff, 2) <= trust_radius)
    
    for k_idx in active_idx:
        off = k_idx - k_max
        f_k = corr[k_idx] * dx
        g = grad_fk(h, off, n)
        # f_k^lin(h_new) = f_k + g·(h_new - h) <= t
        # g·h_new - t <= g·h - f_k
        constraints.append(cp.sum(g * h_var) - t_var <= g @ h - f_k)
    
    problem = cp.Problem(cp.Minimize(t_var), constraints)
    
    # Try solvers in order
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            problem.solve(solver=solver)
            break
        except Exception:
            continue
    
    if h_var.value is not None:
        return project_feasible(h_var.value, n)
    return h.copy()

    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()

        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 10:
            break
        
        corr = compute_corr(h)
        t_idx = np.argsort(-corr)[:K]
        
        # Solve LP
        h_new = solve_lp(h, n, t_idx, trust_radius)
        
        # Check improvement
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
            h = h_new
        else:

    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if interior.any():
            h[interior] += diff / interior.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_fk(h, off, n):
    dx = 2.0 / n
    g = np.zeros(n)
    if off >= 0:
        m = n - off
        g[off:] += (1.0 - h[:m])
        g[:m] -= h[off:]
    else:
        ao = -off
        m = n - ao
        g[:m] += (1.0 - h[ao:])
        g[ao:] -= h[:m]
    return g * dx


def solve_lp_step(h, n, active_k_indices, trust_r):
    """Solve LP linearization with L-infinity trust region."""
    dx = 2.0 / n
    k_max = n - 1
    corr = compute_corr(h)
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n / 2.0,
        h_var >= h - trust_r,
        h_var <= h + trust_r,
    ]
    
    for k_idx in active_k_indices:
        off = k_idx - k_max
        f_k = corr[k_idx] * dx
        g = grad_fk(h, off, n)
        constraints.append(cp.sum(g * h_var) - t_var <= g @ h - f_k)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    for solver in [cp.ECOS, cp.CBC, cp.GLOP]:
        try:
            prob.solve(solver=solver)
            if h_var.value is not None:
                return project_feasible(h_var.value, n)
            break
        except Exception:
            continue
    
    return h.copy()

```python
from scipy.optimize import linprog

def solve_lp_scipy(h, n, active_k_indices, trust_r):
    """Solve LP linearization using scipy's HiGHS solver."""
    dx = 2.0 / n
    k_max = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Variables: x = [h_0, ..., h_{n-1}, t]
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active k: g_k^T h - t <= g_k^T h_old - f_k_old
    A_rows = []
    b_rows = []
    for k_idx in active_k_indices:
        off = k_idx - k_max
        f_k = corr[k_idx] * dx
        g = np.zeros(n)
        if off >= 0:
            m = n - off
            g[off:] += (1.0 - h[:m])
            g[:m] -= h[off:]
        else:
            ao = -off
            m = n - ao
            g[:m] += (1.0 - h[ao:])
            g[ao:] -= h[:m]
        g *= dx
        row = np.concatenate([g, [-1.0]])
        A_rows.append(row)
        b_rows.append(g @ h - f_k)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Equality constraint: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds on h_i: max(h_i - r, 0) <= h_i <= min(h_i + r, 1)
    lb = np.maximum(h - trust_r, 0.0)
    ub = np.minimum(h + trust_r, 1.0)
    bounds_h = list(zip(lb, ub))
    
    # t is unbounded below (well, it should be >= 0 but let's not constrain it)
    bounds = bounds_h + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        h_new = res.x[:n]
        return project_feasible(h_new, n)
    
    return h.copy()
```
