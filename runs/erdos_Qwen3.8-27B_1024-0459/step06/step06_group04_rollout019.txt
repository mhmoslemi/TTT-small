    e = np.exp(s)
    return (m + np.log(e.sum()) / beta) * dx


def smax_jac(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    s = np.clip(beta * (corr - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx] * dx
        if wk < 1e-14:
            continue
        off = k_idx - (n - 1)
        if off >= 0:
            m2 = n - off
            g[off:] += wk * (1.0 - x[:m2])
            g[:m2] -= wk * x[off:]
        else:
            ao = -off
            m2 = n - ao
            g[:m2] += wk * (1.0 - x[ao:])
            g[ao:] -= wk * x[:m2]
    return g


def c_eq_fun(x, n):
    return x.sum() - n / 2.0


def c_eq_jac(x, n):
    return np.ones(n)


def do_slsqp(h, n, beta, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': c_eq_fun, 'jac': c_eq_jac, 'args': (n,)}]
    try:
        res = minimize(smax_func, h, args=(beta,),
                       jac=smax_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def topk_gd(h, n, time_limit, K=20, beta_w=80.0, max_steps=20000):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    for it in range(max_steps):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 5:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(corr)[-K:]
        c_top = corr[idx]
        w = np.exp(beta_w * (c_top - c_top.max()))
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(K) / K
        else:
            w /= ws
        g = np.zeros(n)
        for ii, k_idx in enumerate(idx):
            wk = w[ii]
            if wk < 1e-12:
                continue
            off = k_idx - (n - 1)
            if off >= 0:
                m2 = n - off
                g[off:] += wk * (1.0 - h[:m2])
                g[:m2] -= wk * h[off:]
            else:
                ao = -off
                m2 = n - ao
                g[:m2] += wk * (1.0 - h[ao:])
                g[ao:] -= wk * h[:m2]
        g *= (2.0 / n)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        for step in [0.02, 0.01, 0.05, 0.005, 0.03, 0.015, 0.04, 0.008]:
            ht = project_feasible(h - step * g / gn, n)
            ct = compute_c5(ht)
            if ct < best_c5:
                h = ht
                best_c5 = ct
                best_h = h.copy()
                break
    return best_h, best_c5


def sim_anneal(h_init, n, time_limit, rng, T0=0.01, cool=0.99992, reheat_every=30000, reheat_amt=0.01):
    h = (h_init > 0.5).astype(float)
    target = n // 2
    oc = int(h.sum())
    if oc > target:
        io = np.where(h == 1)[0]
        rng.shuffle(io)
        h[io[:oc - target]] = 0
    elif oc < target:
        iz = np.where(h == 0)[0]
        rng.shuffle(iz)
        h[iz[:target - oc]] = 1
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    T = T0
    T_min = 1e-8
    no_improve = 0
    for it in range(5000000):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 5:
            break
        io = np.where(h == 1)[0]
        iz = np.where(h == 0)[0]
        if len(io) == 0 or len(iz) == 0:
            break
        i = io[rng.randint(len(io))]
        j = iz[rng.randint(len(iz))]
        h[i] = 0
        h[j] = 1
        c_new = compute_c5(h)
        if c_new <= best_c5:
            if c_new < best_c5 - 1e-14:
                best_c5 = c_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            delta = c_new - best_c5
            if rng.random() < np.exp(-delta / T):
                no_improve += 1
            else:
                h[i] = 1
                h[j] = 0
        T = max(T_min, T * cool)
        if no_improve > reheat_every:
            T = max(T, reheat_amt * 0.5)
            no_improve = 0
    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src_h)


def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = bin(i).count('1') % 2
    return h


def binary_greedy_swap(h, n, time_limit, rng, max_rounds=5000):
    h = (h > 0.5).astype(float)
    target = n // 2
    oc = int(h.sum())
    if oc > target:
        io = np.where(h == 1)[0]
        rng.shuffle(io)
        h[io[:oc - target]] = 0
    elif oc < target:
        iz = np.where(h == 0)[0]
        rng.shuffle(iz)
        h[iz[:target - oc]] = 1
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 5:
            break
        improved = False
        io = np.where(h == 1)[0]
        iz = np.where(h == 0)[0]
        n_tries = min(500, len(io) * len(iz))
        for _ in range(n_tries):
            i = io[rng.randint(len(io))]
            j = iz[rng.randint(len(iz))]
            h[i] = 0
            h[j] = 1
            ct = compute_c5(h)
            if ct < best_c5 - 1e-14:
                best_c5 = ct
                best_h = h.copy()
                improved = True
                io = np.where(h == 1)[0]
                iz = np.where(h == 0)[0]
                break
            else:
                h[i] = 1
                h[j] = 0
        if not improved:
            break
    return best_h, best_c5


def small_perturbation(h, n, rng, n_perturb=20):
    h2 = h.copy()
    idx = rng.choice(n, size=n_perturb, replace=False)
    h2[idx] = clip_val = np.clip(h2[idx] + rng.randn(n_perturb) * 0.1, 0, 1)
    h2 = project_feasible(h2, n)
    return h2


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def rem():
        return max(0.0, _BUDGET - (time.time() - _T0))

    # Initialize from parent or fallback
    try:
        best_h = np.array(initial_h_values, dtype=float).copy()
        best_n = len(best_h)
        best_c5 = compute_c5(best_h)
    except Exception:
        best_n = 512
        best_h = thue_morse(best_n)
        best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP polish at current resolution
    if rem() > 100:
        t_phase1 = min(180.0, rem() * 0.12)
        t_start = time.time()
        for beta in [500, 3000, 15000, 60000, 200000]:
            if time.time() - t_start > t_phase1:
                break
            if rem() < 30:
                break
            h_new = do_slsqp(best_h, best_n, beta, maxiter=300)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()

    # Phase 2: top-K gradient
    if rem() > 80:
        t2 = min(100.0, rem() * 0.08)
        h_new, c_new = topk_gd(best_h, best_n, t2, K=25, beta_w=100.0, max_steps=15000)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Phase 3: Long simulated annealing (main differentiator)
    if rem() > 200:
        t3 = min(450.0, rem() * 0.35)
        h_new, c_new = sim_anneal(best_h, best_n, t3, rng,
                                   T0=0.012, cool=0.99993, reheat_every=40000, reheat_amt=0.008)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Phase 4: SLSQP after SA
    if rem() > 60:
        t_start = time.time()
        t4 = min(100.0, rem() * 0.08)
        for beta in [50000, 200000, 800000]:
            if time.time() - t_start > t4:
                break
            if rem() < 20:
                break
            h_new = do_slsqp(best_h, best_n, beta, maxiter=400)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()

    # Phase 5: Try alternative starts at current resolution
    if rem() > 150:
        alt_starts = []
        # Thue-Morse
        tm = thue_morse(best_n)
        tm = project_feasible(tm, best_n)
        alt_starts.append(tm)
        # Random binary
        rb = np.zeros(best_n)
        ones_idx = rng.choice(best_n, size=best_n // 2, replace=False)
        rb[ones_idx] = 1.0
        alt_starts.append(rb)
        # Perturbed current
        for pseed in [7, 13, 29]:
            alt_starts.append(small_perturbation(best_h, best_n, rng, 15))

        for alt in alt_starts[:5]:
            if rem() < 40:
                break
            c_alt = compute_c5(alt)
            if c_alt < best_c5 + 0.005:
                t_alt = min(40.0, rem() * 0.06)
                h_new, c_new = sim_anneal(alt, best_n, t_alt, rng,
                                           T0=0.005, cool=0.99995, reheat_every=20000, reheat_amt=0.003)
                if c_new < best_c5:
                    best_c5 = c_new
                    best_h = h_new.copy()
                h_new = do_slsqp(best_h, best_n, 100000, maxiter=200)
                c_new = compute_c5(h_new)
                if c_new < best_c5:
                    best_c5 = c_new
                    best_h = h_new.copy()

    if rn_candidate > best_n and rn_candidate <= 2048 and rem() > 120:

        c_interp = compute_c5(h_interp)

            t_start = time.time()
            t6 = min(120.0, rem() * 0.1)
            for beta in [1000, 10000, 50000, 200000]:
                if time.time() - t_start > t6:
                    break
                if rem() < 20:
                    break
                h_new = do_slsqp(h_interp, rn_candidate, beta, maxiter=200)
                c_new = compute_c5(h_new)
                if c_new < best_c5:
                    best_c5 = c_new
                    best_h = h_new.copy()
                    best_n = rn_candidate
                    h_interp = h_new.copy()
        if rem() > 80:
            t6b = min(120.0, rem() * 0.1)
            h_new, c_new = sim_anneal(h_interp, best_n, t6b, rng,
                                       T0=0
