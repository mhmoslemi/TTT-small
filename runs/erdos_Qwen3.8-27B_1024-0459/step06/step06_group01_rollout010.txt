def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def _sm_obj_lbfgs(x):
    n = _G_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _G_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    val = (m + np.log(np.sum(ez)) / beta) * dx
    if _G_PEN > 0:
        val += _G_PEN * (np.sum(x) - n / 2.0) ** 2
    return val


def _sm_grad_lbfgs(x):
    n = _G_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _G_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    if _G_PEN > 0:

    return g

def topk_gradient_direction(h, n, top_k):
    """Compute the gradient direction to reduce the top-k correlations."""
    dx = 2.0 / n
    corr = compute_corr(h)
    idx_top = np.argsort(-corr)[:top_k]
    g_dir = np.zeros(n)
    for pos, k in enumerate(idx_top):
        w_k = (1.0 - pos) / (top_k * (top_k + 1) / 2.0)
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g_dir[off:] += w_k * (1.0 - h[:np_])
            g_dir[:np_] -= w_k * h[off:]
        else:
            a = -off
            np_ = n - a
            g_dir[:np_] += w_k * (1.0 - h[a:])
            g_dir[a:] -= w_k * h[:np_]
    g_dir *= -dx  # Negative gradient direction
    return g_dir, idx_top, corr[idx_top[0]] * dx
