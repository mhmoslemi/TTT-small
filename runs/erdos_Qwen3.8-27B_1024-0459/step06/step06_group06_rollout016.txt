            hi = min(n - 1, n - 1 - s)
            corr_top[i] = np.sum(h[lo:hi + 1] * (1.0 - h[lo + s:hi + s + 1])) * (2.0 / n)

        w = np.exp(60.0 * (corr_top - corr_top.max()))
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(top_k) / top_k
        else:
            w /= wsum

        g = np.zeros(n)
        for i, ci in enumerate(top_idx):
            if w[i] < 1e-10:
                continue
            s = ci - (n - 1)
            g += w[i] * grad_shift(h, s, n)

        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break

        step = step_base / (1.0 + 0.005 * it)
        improved = False
        for _ in range(25):
            h_new = project_feasible(h - step * g, n)
            c5_n = c5_val(h_new)
            if c5_n < best_c5 - 1e-15:
                h = h_new
                best_c5 = c5_n
                best_h = h.copy()
                patience = 0
                improved = True
                break
            step *= 0.45

        if not improved:
            patience += 1
            if patience > 150:
                esc = False
                for _ in range(15):
                    pert = rng.randn(n) * 0.015
                    h_try = project_feasible(h + pert, n)
                    c5_try = c5_val(h_try)
                    if c5_try < best_c5:
                        h = h_try
                        best_c5 = c5_try
                        best_h = h.copy()
                        patience = 0
                        esc = True
                        break
                if not esc:
                    break

    return best_h, best_c5


def cutting_plane_lp(h0, n, K=50, delta=0.05):
    dx = 2.0 / n
    corr_full = corr_dx(h0)
    top_idx = np.argsort(-corr_full)[:K]
    c0 = np.zeros(K)
    A_rows = []
    for i, ci in enumerate(top_idx):
        s = ci - (n - 1)
        lo = max(0, -s)
        hi = min(n - 1, n - 1 - s)
        c0[i] = np.sum(h0[lo:hi + 1] * (1.0 - h0[lo + s:hi + s + 1])) * dx
        g = grad_shift(h0, s, n)
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_rows.append(row)

    A_ub = np.array(A_rows)
    b_ub = np.array([-c0 + np.array([np.dot(A_rows[j, :n], h0) for j in range(K)])])

    I = np.eye(n)
    A_trust = np.vstack([np.hstack([I, np.zeros((n, 1))]),
                         np.hstack([-I, np.zeros((n, 1))])])
    b_trust = np.concatenate([h0 + delta, -h0 + delta])

    A_ub_all = np.vstack([A_ub, A_trust])
    b_ub_all = np.concatenate([b_ub, b_trust])

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])

    bounds = [(0.0, 1.0)] * n + [(None, None)]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    try:
        res = linprog(c_obj, A_ub=A_ub_all, b_ub=b_ub_all, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success and res.x is not None:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return h0.copy(), float(np.max(corr_full))


def lp_cutting_plane_loop(h, n, time_limit):
    h = h.copy()
    best_h = h.copy()
    best_c5 = c5_val(h)
    start = time.time()
    delta = 0.05
    K = 40

    for it in range(500):
        if time.time() - start > time_limit or remain() < 5:
            break
        if delta < 0.002:
            break

        h_lp, t_lp = cutting_plane_lp(h, n, K=K, delta=delta)
        cur_c5 = c5_val(h)
        if t_lp < cur_c5 - 1e-10:
            best_alpha = 0.0
            best_trial_c5 = cur_c5
            for alpha in [1.0, 0.8, 0.6, 0.4, 0.25, 0.15]:
                h_try = project_feasible(h + alpha * (h_lp - h), n)
                c_try = c5_val(h_try)
                if c_try < best_trial_c5:
                    best_trial_c5 = c_try
                    best_alpha = alpha
            if best_alpha > 0:
                h = project_feasible(h + best_alpha * (h_lp - h), n)
                c_now = c5_val(h)
                if c_now < best_c5:
                    best_c5 = c_now
                    best_h = h.copy()
                    delta = min(delta, 0.03)
                break
        else:
            delta *= 0.7

    return best_h, best_c5


def coordinate_descent(h, n, time_limit):
    h = h.copy()
    best_c5 = c5_val(h)
    start = time.time()
    no_improve = 0

    while time.time() - start < time_limit and remain() > 5:
        if no_improve > 5:
            break
        improved_sweep = False
        for i in range(n):
            if time.time() - start > time_limit:
                return h, best_c5
            old = h[i]
            candidates = np.array([0.0, 1.0, old])
            if 0.0 < old < 1.0:
                candidates = np.array([0.0, 0.25, 0.5, 0.75, 1.0, old])
            best_v = old
            best_c = best_c5
            for cv in candidates:
                if abs(cv - old) < 1e-9:
                    continue
                h[i] = cv
                h = project_feasible(h, n)
                c = c5_val(h)
                if c < best_c:
                    best_c = c
                    best_v = cv
            h[i] = best_v
            h = project_feasible(h, n)
            c_now = c5_val(h)
            if c_now < best_c5 - 1e-14:
                best_c5 = c_now
                improved_sweep = True
                no_improve = 0
            else:
                no_improve += 1

        if not improved_sweep:
            no_improve += 1

    return h, best_c5


def fine_subgradient(h, n, K, time_limit, rng):
    h = h.copy()
    best_h = h.copy()
    best_c5 = c5_val(h)
    start = time.time()

    corr_full = corr_dx(h)
    top_idx = np.argsort(-corr_full)[:K]

    for it in range(3000):
        if time.time() - start > time_limit or remain() < 3:
            break

        if it % 100 == 1:
            corr_full = corr_dx(h)
            top_idx = np.argsort(-corr_full)[:K]

        corr_top = np.zeros(K)
        for i, ci in enumerate(top_idx):
            s = ci - (n - 1)
            lo = max(0, -s)
            hi = min(n - 1, n - 1 - s)
            corr_top[i] = np.sum(h[lo:hi + 1] * (1.0 - h[lo + s:hi + s + 1])) * (2.0 / n)

        w = np.exp(80.0 * (corr_top - corr_top.max()))
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(K) / K
        else:
            w /= ws

        g = np.zeros(n)
        for i in range(K):
            if w[i] < 1e-12:
                continue
            s = top_idx[i] - (n - 1)
            g += w[i] * grad_shift(h, s, n)

        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break

        best_step = 0
        for step in [0.01, 0.005, 0.002, 0.02, 0.008, 0.001, 0.015]:
            h_try = project_feasible(h - step * g / max(gn, 1e-10), n)
            c_try = c5_val(h_try)
            if c_try < best_c5 - 1e-15:
                h = h_try
                best_c5 = c_try
                best_h = h.copy()
                best_step = step
                break

        if best_step == 0 and it % 200 == 0:
            for _ in range(20):
                pert = rng.randn(n) * 0.005
                h_try = project_feasible(h + pert, n)
                if c5_val(h_try) < best_c5:
                    h = h_try
                    best_c5 = c5_val(h)
                    best_h = h.copy()
                    break

    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def smooth_max_obj_grad(x, n, beta):
    corr = corr_dx(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(beta * sh, -40.0, 40.0)
    ex = np.exp(cl)
    es = ex.sum()
    obj = m + np.log(es) / beta
    w = ex / es
    g = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        g += wk * grad_shift(x, s, n)
    return obj, g


def slsqp_polish(h, n, beta, maxiter=400):
    def obj_fun(x):
        return smooth_max_obj_grad(x, n, beta)[0]

    def grad_fun(x):
        return smooth_max_obj_grad(x, n, beta)[1]

    def sc_fun(x):
        return np.sum(x) - n / 2.0

    def sc_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sc_fun, 'jac': sc_jac}]
    try:
        res = minimize(obj_fun, h, jac=grad_fun, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def augmented_lbfgs(h, n, beta, penalty, maxiter=2000):
    def obj_fun(x):
        corr = corr_dx(x)
        m = np.max(corr)
        sh = np.clip(beta * (corr - m), -40.0, 40.0)
        ex = np.exp(sh)
        es = ex.sum()
        sm = m + np.log(es) / beta
        s = np.sum(x) - n / 2.0
        return sm + 0.5 * penalty * s * s

    def grad_fun(x):
        corr = corr_dx(x)
        m = np.max(corr)
        sh = np.clip(beta * (corr - m), -40.0, 40.0)
        ex = np.exp(sh)
        es = ex.sum()
        w = ex / es
        g = np.zeros(len(x))
        for k in range(len(corr)):
            wk = w[k]
            if wk < 1e-14:
                continue
            s = k - (n - 1)
            g += wk * grad_shift(x, s, len(x))
        s_val = np.sum(x) - n / 2.0
        g += penalty * s_val
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fun, h, jac=grad_fun, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-12, 'maxfun': 30000})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = c5_val(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 512
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = c5_val(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = 1024
    h = interp_to(best_h, best_n, n)
    h = project_feasible(h, n)
    c5_h = c5_val(h)
    if c5_h < best_c5:
        best_c5 = c5_h
        best_h = h.copy()
        best_n = n

    # Also try n=2048 initial
    n2 = 2048
    h2 = interp_to(best_h, best_n, n2)
    h2 = project_feasible(h2, n2)
    c5_h2 = c5_val(h2)

    # Phase 1: Subgradient descent on n=1024
    if remain() > 250:
        h, c5_h = subgradient_minimax(h, n, top_k=35,
                                      time_limit=min(200.0, remain() * 0.25), rng=rng)
        if c5_h < best_c5:
            best_c5 = c5_h
            best_h = h.copy()
            best_n = n

    # Phase 2: LP cutting plane
    if remain() > 100:
        h_lp, c5_lp = lp_cutting_plane_loop(h, n, time_limit=min(150.0, remain() * 0.2))
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp.copy()
            best_n = n
            h = h_lp.copy()

    # Phase 3: L-BFGS-B with smooth max
    for beta in [1000, 5000, 15000]:
        if remain() < 30:
            break
        h_new = augmented_lbfgs(h, n, beta, 2000, maxiter=1500)
        c5_new = c5_val(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 4: Coordinate descent
    if remain() > 80:
        h_cd, c5_cd = coordinate_descent(h, n, time_limit=min(100.0, remain() * 0.25))
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            best_n = n
            h = h_cd.copy()

    # Phase 5: More subgradient with fine steps
    if remain() > 50:
        h_sg2, c5_sg2 = fine_subgradient(h, n, K=25,
                                         time_limit=min(60.0, remain() * 0.25), rng=rng)
        if c5_sg2 < best_c5:
            best_c5 = c5_sg2
            best_h = h_sg2.copy()
            best_n = n
            h = h_sg2.copy()

    # Phase 6: LP cutting plane again
    if remain() > 60:
        h_lp2, c5_lp2 = lp_cutting_plane_loop(h, n, time_limit=min(80.0, remain() * 0.2))
        if c5_lp2 < best_c5:
            best_c5 = c5_lp2
            best_h = h_lp2.copy()
            best_n = n
            h = h_lp2.copy()

    # Phase 7: SLSQP polish
    for beta in [5000, 20000, 50000]:
        if remain() < 20:
            break
        h_new = slsqp_polish(h, n, beta, maxiter=300)
        c5_new = c5_val(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 8: Try n=2048 if time permits and current is close to target
    if remain() > 150 and best_n < 2048:
        h_vh = interp_to(best_h, best_n, n2)
        h_vh = project_feasible(h_vh, n2)
        c5_vh = c5_val(h_vh)
        if remain() > 100:
            h_vh, c5_vh = subgradient_minimax(h_vh, n2, top_k=40,
                                              time_limit=min(80.0, remain() * 0.3), rng=rng)
        if remain() > 60:
            h_vh, c5_vh = lp_cutting_plane_loop(h_vh, n2, time_limit=min(50.0, remain() * 0.3))
        if remain() > 30:
            h_vh = slsqp_polish(h_vh, n2, 10000, maxiter=200)
            c5_vh = c5_val(h_vh)
        if c5_vh < best_c5:
            best_c5 = c5_vh
            best_h = h_vh.copy()
            best_n = n2

    # Phase 9: Final fine subgradient
    if remain() > 20:
        h_fin, c5_fin = fine_subgradient(best_h, best_n, K=20,
                                         time_limit=min(30.0, remain() * 0.4), rng=rng)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Phase 10: Final coordinate descent
    if remain() > 30:
        h_cd2, c5_cd2 = coordinate_descent(best_h, best_n, time_limit=min(40.0, remain() * 0.5))
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()

    # Final feasibility and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = c5_val(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
