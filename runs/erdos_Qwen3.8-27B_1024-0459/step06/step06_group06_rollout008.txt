def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)

def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        g[offset:n] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:n]
    else:
        a = -offset
        npairs = n - a
        g[:npairs] += (1.0 - h[a:n])
        g[a:n] -= h[:npairs]
    return g

    if topK is not None and topK < len(corr):
        idx = np.argpartition(-corr, topK)[:topK]
        idx = idx[np.argsort(-corr[idx])]
        c_sub = corr[idx]
    else:
        c_sub = corr
        idx = np.arange(len(corr))
    m = np.max(c_sub)
    sh = c_sub - m
    cl = np.clip(beta * sh, -50.0, 50.0)
    ex = np.exp(cl)
    sm = m + np.log(np.sum(ex)) / beta
    w = ex / np.sum(ex)
    return sm, w, idx

        s = np.sum(x) - n / 2.0

                                'gtol': 1e-16, 'maxfun': maxiter * 3})
        if res.x is not None:

    except Exception:
        pass
    return h

        sm, _, _ = smoothmax_obj(x, n, beta, topK)
        return sm * (2.0 / n)

        g = np.zeros(n)
        for i in range(len(idx)):
            if w[i] > 1e-14:
                g += w[i] * grad_corr_k(x, idx[i], n)
        g *= (2.0 / n)
        return g

        return np.sum(x) - n / 2.0

    def con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n

def equalize(h, n, K, max_iters):

    for _ in range(max_iters):

def coord_descent(h, n, tlimit):
    h = h.copy()
    bc5 = compute_c5(h)
    t0 = time.time()
    pat = 30
    while time.time() - t0 < tlimit and rem_time() > 5:
        imp = False
        for i in range(n):
            if time.time() - t0 > tlimit or rem_time() < 5:
                break
            old = h[i]
            cands = [0.0, 1.0]
            if 0.05 < old < 0.95:
                cands.append(0.5)
            bc = bc5
            bv = old
            for cv in cands:
                if abs(cv - old) < 1e-10:
                    continue
                h[i] = cv
                h = project_feasible(h, n)
                c = compute_c5(h)
                if c < bc:
                    bc = c
                    bv = cv
            h[i] = bv
            h = project_feasible(h, n)
            cn = compute_c5(h)
            if cn < bc5 - 1e-14:
                bc5 = cn
                imp = True
        if not imp:
            pat -= 1
            if pat <= 0:
                break
        else:
            pat = 30
    return h, bc5

        h2, c2 = equalize(h2, n2, K=20, max_iters=300)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()

    # High-beta L-BFGS-B
    for beta in [30000, 80000, 150000]:
        if rem_time() < 30:
            break
        h2 = lbfgsb_opt(h2, n2, beta, 2000, 2000, 15)
        c2 = compute_c5(h2)
