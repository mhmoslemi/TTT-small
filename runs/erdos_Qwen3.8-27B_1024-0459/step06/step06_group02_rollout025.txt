import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr)) * dx


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.count_nonzero(mask) > 0:
            h[mask] += diff / np.count_nonzero(mask)
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_obj(h, n, beta, active_lags, lam=0.0):
    corr = compute_corr(h)
    dx = 2.0 / n
    c_active = corr[active_lags]
    m = np.max(c_active)
    shifted = beta * (c_active - m)
    shifted = np.clip(shifted, -50.0, 50.0)
    sm = np.sum(np.exp(shifted))
    val = (m + np.log(sm) / beta) * dx
    if lam > 0:
        sum_h = np.sum(h)
        val += lam * (sum_h - n / 2.0) ** 2
    return val


def smooth_max_grad(h, n, beta, active_lags, lam=0.0):
    corr = compute_corr(h)
    dx = 2.0 / n
    c_active = corr[active_lags]
    m = np.max(c_active)
    shifted = beta * (c_active - m)
    shifted = np.clip(shifted, -50.0, 50.0)
    ex = np.exp(shifted)
    s = np.sum(ex)
    w = ex / s  # weights for each active lag
    
    g = np.zeros(n)
    for idx, k in enumerate(active_lags):
        if w[idx] < 1e-15:
            continue
        wk = w[idx] * dx
        d = k - (n - 1)  # offset
        if d >= 0:
            npairs = n - d
            if npairs > 0:
                g[d:] += wk * (1.0 - h[:npairs])
                g[:npairs] -= wk * h[d:]
        else:
            a = -d
            npairs = n - a
            if npairs > 0:
                g[:npairs] += wk * (1.0 - h[a:])
                g[a:] -= wk * h[:npairs]
    
    if lam > 0:
        g += 2.0 * lam * (np.sum(h) - n / 2.0) * np.ones(n)
    
    return g


def find_active_lags(corr, K):
    """Find indices of top K correlation values."""
    if K >= len(corr):
        return np.arange(len(corr))
    return np.argpartition(-corr, K)[:K]

    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

    obj_val = smooth_max_obj(h, n, beta, active, lam)

    best_obj = obj_val

        return smooth_max_obj(x, n, beta, active, lam)
    
    def grad_fun(x):
        return smooth_max_grad(x, n, beta, active, lam)

                                               'gtol': 1e-8, 'maxls': 50})
        h_new = res.x

        h_new = h
    
    h_new = project_feasible(h_new, n)
    c5_new = compute_c5(h_new)

    obj_new = smooth_max_obj(h_new, n, beta, active_new, lam)

        best_obj = obj_new

Actually, I just realized there might be an issue with my FFT correlation. Let me be very careful.

`np.correlate(a, v, 'full')` gives output of length n_a + n_v - 1.
The k-th element (0-indexed) is: sum_{n} a[n + k - (n_v - 1)] * v[n]
where the sum is over all valid n.

Equivalent formulation: the output[k] = sum_{n: valid} a[n + k - (n_v-1)] * v[n]

For a = h (length n), v = 1-h (length n):
output[k] = sum_{n: valid} h[n + k - (n-1)] * (1-h[n])
         = sum_{n: valid} h[n + k - n + 1] * (1-h[n])

Let d = k - (n-1). Then:
output[k] = sum_{n: valid} h[n + d] * (1-h[n])

This is what I had before. Good.

Now, for the FFT computation:
correlate(a, v, 'full') = convolve(a, v[::-1], 'full')

convolve(a, b, 'full')[k] = sum_i a[i] * b[k-i]

So convolve(h, (1-h)[::-1], 'full')[k] = sum_i h[i] * (1-h)[::-1][k-i]
= sum_i h[i] * (1-h)[n-1-(k-i)]
= sum_i h[i] * (1-h)[n-1-k+i]

Let j = n-1-k+i, so i = j - n + 1 + k:
= sum_j h[j-n+1+k] * (1-h)[j]

Let d = k - (n-1):
= sum_j h[j+d] * (1-h)[j]

This matches! So the FFT approach is correct.

For the FFT computation:
N = next power of 2 >= 2n-1
H = rfft(h, N)
R = rfft((1-h)[::-1], N)
conv = irfft(H * R, N)[:2n-1]

This gives the full correlation.

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_context = {}


def compute_corr(h):
    n = len(h)
    if n <= 1024:
        return np.correlate(h, 1.0 - h, mode='full')
    else:
        fsize = 1
        while fsize < 2 * n - 1:
            fsize *= 2
        H = np.fft.rfft(h, fsize)
        V = np.fft.rfft((1.0 - h)[::-1], fsize)
        return np.fft.irfft(H * V, fsize)[:2*n-1]


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr)) * dx


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def find_active_lags(corr, K):
    if K >= len(corr):
        return np.arange(len(corr))
    return np.argpartition(-corr, K)[:K]


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def smooth_max_value(h, n, beta, active, lam):
    corr = compute_corr(h)
    dx = 2.0 / n
    c_a = corr[active]
    m = np.max(c_a)
    shifted = beta * (c_a - m)
    shifted = np.clip(shifted, -50.0, 50.0)
    val = (m + np.log(np.sum(np.exp(shifted))) / beta) * dx
    if lam > 0:
        s = np.sum(h)
        val += lam * (s - n / 2.0) ** 2
    return val


def smooth_max_grad(x, n, beta, active, lam):
    corr = compute_corr(x)
    dx = 2.0 / n
    c_a = corr[active]
    m = np.max(c_a)
    shifted = beta * (c_a - m)
    shifted = np.clip(shifted, -50.0, 50.0)
    ex = np.exp(shifted)
    s = np.sum(ex)
    w = ex / s
    g = np.zeros(n)
    for idx in range(len(active)):
        if w[idx] < 1e-14:
            continue
        wk = w[idx] * dx
        k = active[idx]
        d = k - (n - 1)
        if d >= 0:
            npairs = n - d
            if npairs > 0:
                g[d:] += wk * (1.0 - x[:npairs])
                g[:npairs] -= wk * x[d:]
        else:
            a = -d
            npairs = n - a
            if npairs > 0:
                g[:npairs] += wk * (1.0 - x[a:])
                g[a:] -= wk * x[:npairs]
    if lam > 0:
        g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g


def _lfbgbs_obj(x):
    return smooth_max_value(x, _context['n'], _context['beta'], _context['active'], _context['lam'])


def _lfbgbs_grad(x):
    return smooth_max_grad(x, _context['n'], _context['beta'], _context['active'], _context['lam'])


def lfbgbs_optimize(h, n, beta, K, lam, maxiter=200):
    h = np.array(h, dtype=float).copy()
    h = project_feasible(h, n)
    corr = compute_corr(h)
    active = find_active_lags(corr, K)
    bounds = [(0.0, 1.0)] * n
    _context['n'] = n
    _context['beta'] = beta
    _context['active'] = active
    _context['lam'] = lam
    best_h = h.copy()
    best_c5 = compute_c5(h)
    try:
        res = minimize(_lfbgbs_obj, h, jac=_lfbgbs_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 40})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
    except Exception:
        pass

def coordinate_polish(h, n, max_sweeps=5, time_limit=30.0):

    start = time.time()
    target = n / 2.0

        improved = False
        corr = compute_corr(h)

        for i in range(n):
            if time.time() - start > time_limit:

            old_val = h[i]
            candidates = [0.0, 1.0, old_val - 0.05, old_val - 0.02, old_val - 0.01,
                          old_val + 0.01, old_val + 0.02, old_val + 0.05]
            candidates = [max(0.0, min(1.0, c)) for c in candidates]
            best_cand = old_val
            best_c = best_c5
            for c in candidates:
                if abs(c - old_val) < 1e-10:
                    continue
                h[i] = c
                diff = target - np.sum(h)
                if abs(diff) > 1e-6:

                    if np.count_nonzero(mask) > 0:

                    h = np.clip(h, 0.0, 1.0)
                c_cur = compute_c5(h)

                h = project_feasible(h, n)

            break
    return h, best_c5


def subgradient_step(h, n, K, beta_g, step):
    corr = compute_corr(h)
    idx = np.argpartition(-corr, min(K, len(corr)-1))[:K]
    c_vals = corr[idx]
    weights = np.exp(beta_g * (c_vals - c_vals.max()))
    s = weights.sum()
    if s < 1e-30:
        weights = np.ones(K) / K
    else:
        weights /= s
    g = np.zeros(n)
    dx = 2.0 / n
    for i in range(K):
        if weights[i] < 1e-12:
            continue
        wk = weights[i] * dx
        k = idx[i]
        d = k - (n - 1)
        if d >= 0:
            npairs = n - d
            if npairs > 0:
                g[d:] += wk * (1.0 - h[:npairs])
                g[:npairs] -= wk * h[d:]
        else:
            a = -d
            npairs = n - a
            if npairs > 0:
                g[:npairs] += wk * (1.0 - h[a:])
                g[a:] -= wk * h[:npairs]
    gn = np.linalg.norm(g)
    if gn < 1e-14:
        return h
    h_new = project_feasible(h - step * g / gn, n)
    return h_new
