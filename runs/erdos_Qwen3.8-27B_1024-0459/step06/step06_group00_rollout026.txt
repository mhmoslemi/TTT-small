            if k_prev is None:
                k_prev = k_curr
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-14:
            g = g / g_norm
        improved = False
        base = max(0.0005, 0.01 * (1.0 - 0.5 * it / max_iters))
        for step in [base * 0.3, base * 0.7, base, base * 2.0, base * 4.0, 0.001, 0.005, 0.01, 0.03, 0.05]:
            ht = project_feasible(h - step * g, n)
            c5_new = compute_c5(ht)
            if c5_new < best_c5 - 1e-16:
                h = ht
                best_c5 = c5_new
                best_h = h.copy()
                improved = True
                stale = 0
                break
        if not improved:
            stale += 1
            if stale > 300:
                break
            if stale % 80 == 0:
                rng = np.random.RandomState(stale * 7 + 3)
                pert = rng.randn(n) * 0.003
                ht = project_feasible(h + pert, n)
                c5_new = compute_c5(ht)
                if c5_new < best_c5:
                    h = ht
                    best_c5 = c5_new
                    best_h = h.copy()
                    stale = 0
    return best_h, best_c5


def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)


def binary_2swap_search(h, n, time_limit=90, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h_bin = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        sort_idx = np.argsort(-h_bin)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0

    dx = 2.0 / n
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    curr_c5 = best_c5
    start = time.time()

    T_start = 2.0 / dx
    T_end = 0.0005 / dx
    total_iters = 30000
    no_improve = 0

    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]

    for it in range(total_iters):
        if time.time() - start > time_limit:
            break
        if remaining_time() < 5:
            break

        T = T_start * (T_end / T_start) ** (it / max(1, total_iters))

        i_idx = rng.randint(len(ones))
        j_idx = rng.randint(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        if i == j:
            continue

        h_bin[i] = 0.0
        h_bin[j] = 1.0
        new_c5 = compute_c5(h_bin)
        delta = new_c5 - curr_c5

        if delta < 0 or (T > 1e-12 and rng.random() < np.exp(-delta / max(T, 1e-12))):
            ones[i_idx] = j
            zeros[j_idx] = i
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_bin.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            no_improve += 1

        if no_improve > 8000:
            if T > T_end * 10:
                T = T_end * 10
            else:
                break

    return best_h, best_c5


def targeted_binary_search(h, n, time_limit=60, rng=None):
    if rng is None:
        rng = np.random.RandomState(123)
    h_bin = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        sort_idx = np.argsort(-h_bin)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0

    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    start = time.time()
    no_improve = 0

    while time.time() - start < time_limit and remaining_time() > 5:
        corr = compute_corr(h_bin)
        top_idx = np.argsort(-corr)[:10]
        c_max = corr[top_idx[0]]

        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]

        best_delta = 1e-10
        best_pair = None
        n_trials = min(500, len(ones) * len(zeros))

        for _ in range(n_trials):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            new_c5 = compute_c5(h_bin)
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            delta = new_c5 - best_c5
            if delta < best_delta:
                best_delta = delta
                best_pair = (i, j)

        if best_pair is not None:
            i, j = best_pair
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            best_c5 = compute_c5(h_bin)
            best_h = h_bin.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_bin[i] = 0.0
                h_bin[j] = 1.0
                best_c5 = compute_c5(h_bin)
                best_h = h_bin.copy()
                no_improve = 0

    return best_h, best_c5


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def coordinate_refine(h, n, max_sweeps=15):
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        improved = False
        for i in range(n):
            old = h[i]
            cands = [0.0, 1.0]
            if 0.1 < old < 0.9:
                cands.extend([old + 0.2, old - 0.2, 0.3, 0.7])
            else:
                cands.extend([old + 0.05, old - 0.05])
            cands = [c for c in cands if 0.0 <= c <= 1.0 and abs(c - old) > 1e-5]
            best_val = old
            best_c = best_c5
            for c in cands:
                h[i] = c
                h = project_feasible(h, n)
                c5_v = compute_c5(h)
                if c5_v < best_c - 1e-14:
                    best_c = c5_v
                    best_val = c
                    improved = True
            h[i] = best_val
            h = project_feasible(h, n)
        if not improved:
            break
    return h, best_c5
