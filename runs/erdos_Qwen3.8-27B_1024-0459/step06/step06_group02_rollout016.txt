def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def smax_obj(x, beta, n):
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = beta * (corr - m)
    clipped = np.clip(shifted, -700.0, 700.0)
    ex = np.exp(clipped)
    val = (m + np.log(np.sum(ex)) / beta) * (2.0 / n)
    return val


def smax_grad(x, beta, n, K=300):
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = beta * (corr - m)
    clipped = np.clip(shifted, -700.0, 700.0)
    ex = np.exp(clipped)
    s = np.sum(ex)
    if s < 1e-300:
        w = np.ones(len(corr)) / len(corr)
    else:
        w = ex / s
    dx_val = 2.0 / n
    K_eff = min(K, len(w))
    idx = np.argsort(-w)[:K_eff]
    g = np.zeros(n)
    for k in idx:
        wk = w[k] * dx_val
        if wk < 1e-18:
            continue
        d = k - (n - 1)
        if d == 0:
            g += wk * (1.0 - 2.0 * x)
        elif d > 0:
            g[:n - d] += wk * (1.0 - x[d:])
            g[d:] -= wk * x[:n - d]
        else:
            a = -d
            g[a:] += wk * (1.0 - x[:n - a])
            g[:n - a] -= wk * x[a:]
    return g


def eq_fun(x, n):
    return np.sum(x) - n / 2.0


def eq_jac(x, n):
    return np.ones(len(x))


def slsqp_optimize(h0, n, beta, maxiter=300, K=250):
    def obj(x):
        return smax_obj(x, beta, n)

    def grad(x):
        return smax_grad(x, beta, n, K)

    def cfun(x):
        return np.sum(x) - n / 2.0

    def cjac(x):
        return np.ones(len(x))

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return h0.copy()


def tc_optimize(h0, n, beta, maxiter=150, K=200):
    def obj(x):
        return smax_obj(x, beta, n)

    def grad(x):
        return smax_grad(x, beta, n, K)

    bounds = Bounds(np.zeros(n), np.ones(n))
    A = np.ones((1, n))
    b_val = np.array([n / 2.0])
    lin_con = LinearConstraint(A, b_val, b_val)
    try:
        res = minimize(obj, h0, jac=grad, method='trust-constr',
                       bounds=bounds, constraints=[lin_con],
                       options={'maxiter': maxiter, 'gtol': 1e-7, 'xtol': 1e-9})
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return h0.copy()


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def golden_init(n, seed=0):
    rng = np.random.RandomState(seed)
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    x = np.arange(n)
    for trial in range(20):
        offset = rng.uniform(0, 1)
        vals = np.mod(x * phi / n + offset, 1.0)
        h = (vals < 0.5).astype(float)
        h += rng.uniform(-0.02, 0.02, n)
        h = project_feasible(h, n)
        return h
    return project_feasible(h, n)


def freq_init(n, seed=0):
    rng = np.random.RandomState(seed)
    x = np.arange(n)
    n_freqs = rng.randint(3, 7)
    freqs = rng.choice([1, 2, 3, 5, 7, 11, 13, 17, 19], size=min(n_freqs, 9), replace=False)
    phases = rng.uniform(0, 2 * np.pi, len(freqs))
    amps = rng.uniform(0.5, 1.0, len(freqs))
    vals = np.zeros(n)
    for f, p, a in zip(freqs, phases, amps):
        vals += a * np.sin(2 * np.pi * f * x / n + p)
    h = (vals > 0).astype(float)
    h += rng.uniform(-0.03, 0.03, n)
    return project_feasible(h, n)
