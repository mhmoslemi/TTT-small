def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

    return h

def delta_corr_noncirc(h, i, j, n):
    """Compute change in all correlations when swapping h[i]:1->0, h[j]:0->1. Non-circular."""
    delta = np.zeros(2 * n - 1)
    # For shift L >= 0: C[L] = sum_{a=0}^{n-1-L} h[a+L]*(1-h[a])
    # i (1->0): as h[a] factor: a=i, valid if i <= n-1-L. Change: +h[i+L]
    #            as h[a+L] factor: a+L=i, a=i-L, valid if i>=L and i-L <= n-1-L i.e. i<=n-1. Change: -(1-h[i-L])
    # j (0->1): as h[a] factor: a=j, valid if j <= n-1-L. Change: -h[j+L]
    #            as h[a+L] factor: a+L=j, a=j-L, valid if j>=L and j<=n-1. Change: +(1-h[j-L])
    for L in range(1, n):
        # i as (1-h[a]) at a=i: valid if i <= n-1-L
        if i <= n - 1 - L:
            delta[n - 1 + L] += h[i + L]
        # i as h[a+L] at a=i-L: valid if i >= L
        if i >= L:
            delta[n - 1 + L] -= (1.0 - h[i - L])
        # j as (1-h[a]) at a=j: valid if j <= n-1-L
        if j <= n - 1 - L:
            delta[n - 1 + L] -= h[j + L]
        # j as h[a+L] at a=j-L: valid if j >= L
        if j >= L:
            delta[n - 1 + L] += (1.0 - h[j - L])
    # For shift L < 0, L = -a: C[-a] = sum_{a_start=a}^{n-1} h[a_start]*(1-h[a_start-a])
    # i (1->0): as h[a_start] at a_start=i: valid if i >= a. Change: -(1-h[i-a])
    #            as (1-h[a_start-a]) at a_start-a=i, a_start=i+a: valid if i+a < n. Change: +h[i+a]
    # j (0->1): as h[a_start] at a_start=j: valid if j >= a. Change: +(1-h[j-a])
    #            as (1-h[a_start-a]) at a_start=j+a: valid if j+a < n. Change: -h[j+a]
    for a in range(1, n):
        if i >= a:
            delta[n - 1 - a] -= (1.0 - h[i - a])
        if i + a < n:
            delta[n - 1 - a] += h[i + a]
        if j >= a:
            delta[n - 1 - a] += (1.0 - h[j - a])
        if j + a < n:
            delta[n - 1 - a] -= h[j + a]
    return delta

def slsqp_smooth(h, n, K, beta, maxiter=400):
    """SLSQP with smooth max objective."""
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        idx = np.argsort(-corr)[:K]
        c_top = corr[idx]
        m = np.max(c_top)
        sh = np.clip(beta * (c_top - m), -50.0, 50.0)
        ex = np.exp(sh)
        return (m + np.log(ex.sum()) / beta) * dx
    
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        idx = np.argsort(-corr)[:K]
        c_top = corr[idx]
        m = np.max(c_top)
        sh = np.clip(beta * (c_top - m), -50.0, 50.0)
        ex = np.exp(sh)
        w = ex / ex.sum()
        g = np.zeros(n)
        for i in range(K):
            if w[i] < 1e-14:
                continue
            wk = w[i] * dx
            k = idx[i]
            offset = k - (n - 1)
            if offset >= 0:
                npairs = n - offset
                g[:npairs] += wk * (1.0 - x[offset:offset + npairs])
                g[offset:offset + npairs] -= wk * x[:npairs]
            else:
                a = -offset
                npairs = n - a
                g[a:a + npairs] += wk * (1.0 - x[:npairs])
                g[:npairs] -= wk * x[a:a + npairs]
        return g
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            c5_old = compute_c5(h)
            if c5_new <= c5_old + 1e-12:
                return h_new
    except Exception:
        pass
    return h

def binary_search_slsqp(h, n, time_limit, target_delta=0.0001):
    """Binary search on target T with SLSQP feasibility."""
    start = time.time()
    h_curr = h.copy()
    dx = 2.0 / n
    
    corr = compute_corr(h_curr)
    c_lo = np.min(corr) * dx
    c_hi = np.max(corr) * dx
    
    for _ in range(15):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 10:
            break
        
        T = (c_lo + c_hi) / 2.0
        T_count = T / dx  # in correlation units
        
        # Solve: min sum((corr_k - T_count)^2) for corr_k > T_count
        # subject to 0<=h<=1, sum(h)=n/2
        def obj_pen(x):
            corr_x = np.correlate(x, 1.0 - x, mode='full')
            excess = np.maximum(corr_x - T_count, 0)
            return np.sum(excess ** 2) + 100.0 * (np.sum(x) - n / 2.0) ** 2
        
        def grad_pen(x):
            corr_x = np.correlate(x, 1.0 - x, mode='full')
            excess = np.maximum(corr_x - T_count, 0)
            g = np.zeros(n)
            active = excess > 0
            for k in np.where(active)[0]:
                offset = k - (n - 1)
                if offset >= 0:
                    npairs = n - offset
                    g[:npairs] += 2 * excess[k] * (1.0 - x[offset:offset + npairs])
                    g[offset:offset + npairs] -= 2 * excess[k] * x[:npairs]
                else:
                    a = -offset
                    npairs = n - a
                    g[a:a + npairs] += 2 * excess[k] * (1.0 - x[:npairs])
                    g[:npairs] -= 2 * excess[k] * x[a:a + npairs]
            g += 200.0 * (np.sum(x) - n / 2.0)
            return g
        
        try:
            res = minimize(obj_pen, h_curr, jac=grad_pen, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 200, 'ftol': 1e-16})
            h_try = project_feasible(res.x, n)
            c5_try = compute_c5(h_try)
            max_corr_try = np.max(compute_corr(h_try)) * dx
            if max_corr_try <= T + 1e-6:
                c_lo = T
                h_curr = h_try
            else:
                c_hi = max(c_hi, max_corr_try)
        except Exception:
            c_hi = c_lo + target_delta

        for K in [10, 15, 25]:
            for beta in [500, 2000, 10000]:
                if remaining() < 10:
                    break
                h2 = slsqp_smooth(h2, n2, K, beta, maxiter=250)
                c5_2 = compute_c5(h2)
        
        # Active set optimization
        if remaining() > 40:
            h2, c5_2 = active_set_optimize(h2, n2, min(60.0, remaining() * 0.3))
        
        # Final polish at n=1024
        if remaining() > 20:
            h2, c5_2 = fine_polish(h2, n2, min(40.0, remaining() * 0.3))
        
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
            best_n = n2
    
    # Phase 4: Try n=1536
    n3 = 1536
    if remaining() > 100 and best_n < n3:
        h3 = upscale(best_h, best_n, n3)
        c5_3 = compute_c5(h3)
        
        for K in [10, 20]:
            for beta in [2000, 10000]:
                if remaining() < 10:
                    break
                h3 = slsqp_smooth(h3, n3, K, beta, maxiter=200)
                c5_3 = compute_c5(h3)
        
        if remaining() > 30:
            h3, c5_3 = fine_polish(h3, n3, min(40.0, remaining() * 0.3))
        
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
            best_n = n3
    
    # Phase 5: Multi-start perturbation at best resolution
    n = best_n
    h = best_h.copy()
    if remaining() > 50:
        h_p, c5_p = multi_start_perturb(h, n, min(80.0, remaining() * 0.3), rng, n_starts=4)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()
            h = h_p.copy()
    
    # Phase 6: Binary search on target (new approach)
    if remaining() > 60:
        h_bs, c5_bs = binary_search_slsqp(best_h, best_n, min(80.0, remaining() * 0.3))
        if c5_bs < best_c5:
            best_c5 = c5_bs
            best_h = h_bs.copy()
    
    # Phase 7: Intensive final polish
    n = best_n
    h = best_h.copy()
    for K in [3, 5, 8, 12, 20]:
        for beta in [10000, 50000, 200000, 500000]:
            if remaining() < 5:
                break
            h_new = slsqp_smooth(h, n, K, beta, maxiter=500)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
    
    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    # Cross-verify with evaluate_erdos_solution
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    # Verify constraints
    assert np.all(best_h >= -1e-10) and np.all(best_h <= 1.0 + 1e-10)
    assert abs(np.sum(best_h) * (2.0 / best_n) - 1.0) < 1e-6
    
    return (best_h, best_c5, best_n)
