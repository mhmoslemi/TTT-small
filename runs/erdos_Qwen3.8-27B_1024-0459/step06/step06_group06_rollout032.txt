    def obj_fun(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -60.0, 60.0)
        ex = np.exp(cl)
        sm = (m + np.log(np.sum(ex)) / beta) * dx
        s = np.sum(x) - n / 2.0
        return sm + 0.5 * penalty * s * s

    def grad_fun(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -60.0, 60.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        thr = 1e-14
        for k in range(len(corr)):
            wk = w[k] * dx
            if wk < thr:
                continue
            g += wk * _grad_shift(x, k, n)
        s = np.sum(x) - n / 2.0
        g += penalty * s
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fun, h, jac=grad_fun, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': ftol,
                                'gtol': 1e-18, 'maxfun': 50000, 'maxls': 50,
                                'factr': 1e12})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def slp_step(h, n, K=30):
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:K]
    dx = 2.0 / n

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    A_ub_list = []
    b_ub_list = []

    for k in idx:
        g = _grad_shift(h, k, n)
        fk = corr[k] * dx
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_ub_list.append(row)
        b_ub_list.append(fk - np.dot(g, h))

    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])

    bounds = [(0, 1)] * n + [(None, None)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return h.copy(), float(np.max(corr)) * dx


def slp_loop(h, n, time_limit, K=30):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    stall = 0
    while time.time() - start < time_limit and _remaining() > 5:
        h_lp, t_lp = slp_step(h, n, K=K)
        cur_c5 = compute_c5(h)
        if t_lp < cur_c5 - 1e-12:
            best_alpha = 0.0
            best_val = cur_c5
            for alpha in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05]:
                h_try = project_feasible(h + alpha * (h_lp - h), n)
                c_try = compute_c5(h_try)
                if c_try < best_val:
                    best_val = c_try
                    best_alpha = alpha
            if best_alpha > 0:
                h = project_feasible(h + best_alpha * (h_lp - h), n)
                c_now = compute_c5(h)
                if c_now < best_c5 - 1e-15:
                    best_c5 = c_now
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            stall += 1
        if stall > 30:
            break
    return best_h, best_c5


def equalize_top(h, n, K=15, max_iters=300):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    for it in range(max_iters):
        if time.time() - start > 60 or _remaining() < 5:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:K]
        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        if c_max - c_min < 1e-10:
            break
        g_avg = np.zeros(n)
        for k in idx[:K // 2]:
            g_avg += _grad_shift(h, k, n)
        g_avg /= max(1, K // 2)
        gn = np.linalg.norm(g_avg)
        if gn < 1e-14:
            break
        g_dir = -g_avg / gn
        found = False
        for step in [0.04, 0.02, 0.01, 0.06, 0.005, 0.03, 0.08]:
            ht = project_feasible(h + step * g_dir, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-15:
                h = ht
                best_c5 = cn
                best_h = ht.copy()
                found = True
                break
        if not found:
            for step in [0.02, 0.01, 0.04, 0.005]:
                ht = project_feasible(h - step * g_dir, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-15:
                    h = ht
                    best_c5 = cn
                    best_h = ht.copy()
                    found = True
                    break
        if not found:
            break
    return best_h, best_c5


def subgradient_refine(h, n, time_limit, rng):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    it = 0
    while time.time() - start < time_limit and _remaining() > 5:
        it += 1
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:20]
        c_max = corr[idx[0]]
        beta_g = 80.0 + 40.0 * min(1.0, it / 200.0)
        weights = np.exp(beta_g * (corr[idx] - c_max))
        sw = weights.sum()
        if sw < 1e-30:
            weights = np.ones(len(idx)) / len(idx)
        else:
            weights /= sw
        g = np.zeros(n)
        for i, k in enumerate(idx):
            if weights[i] > 1e-12:
                g += weights[i] * _grad_shift(h, k, n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g = g / gn
        step = 0.03 / (1.0 + 0.05 * it)
        found = False
        for s in [step, step * 0.5, step * 0.25, step * 2.0, step * 0.1]:
            ht = project_feasible(h - s * g, n)
            ct = compute_c5(ht)
            if ct < best_c5 - 1e-15:
                h = ht
                best_c5 = ct
                best_h = ht.copy()
                found = True
                break
        if not found and it % 50 == 0:
            h = best_h.copy()
    return best_h, best_c5


def binary_snap(h, n):
    h_bin = (h > 0.5).astype(float)
    h_bin = project_feasible(h_bin, n)
    return h_bin


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def slsqp_optimize(h, beta, maxiter=500):
    n = len(h)

    def obj_fun(x):
        return smooth_max_obj(x, beta)

    def grad_fun(x):
        return smooth_max_grad(x, beta)

    def sc_fun(x):
        return np.sum(x) - n / 2.0

    def sc_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sc_fun, 'jac': sc_jac}]
    try:
        res = minimize(obj_fun, h, jac=grad_fun, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _t0, _budget
    _t0 = time.time()
    _budget = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
    except Exception:
        parent_n = 512
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0

    # Start at n=2048
    n = 2048
    h = interp_to(parent_h, parent_n, n)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_n = n

    # Phase 1: L-BFGS-B with moderate beta
    for beta in [1000, 3000]:
        if _remaining() < 60:
            break
        h_new = lbfgsb_optimize(h, beta, 1000, maxiter=4000)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: SLP with moderate active set
    if _remaining() > 100:
        h_slp, c5_slp = slp_loop(h, n, time_limit=min(90.0, _remaining() * 0.3), K=30)
        if c5_slp < best_c5:
            best_c5 = c5_slp
            best_h = h_slp.copy()
            h = h_slp.copy()

    # Phase 3: L-BFGS-B with higher beta
    for beta in [8000, 15000]:
        if _remaining() < 50:
            break
        h_new = lbfgsb_optimize(h, beta, 5000, maxiter=4000)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 4: Equalize top shifts
    if _remaining() > 60:
        h_eq, c5_eq = equalize_top(h, n, K=12, max_iters=200)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 5: SLP with larger active set
    if _remaining() > 80:
        h_slp, c5_slp = slp_loop(h, n, time_limit=min(70.0, _remaining() * 0.3), K=50)
        if c5_slp < best_c5:
            best_c5 = c5_slp
            best_h = h_slp.copy()
            h = h_slp.copy()

    # Phase 6: Subgradient refinement
    if _remaining() > 40:
        h_sg, c5_sg = subgradient_refine(h, n, time_limit=min(35.0, _remaining() * 0.3), rng=rng)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 7: L-BFGS-B very high beta
    if _remaining() > 50:
        h_new = lbfgsb_optimize(h, 30000, 10000, maxiter=3000)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 8: SLSQP polish
    if _remaining() > 40:
        h_new = slsqp_optimize(h, 20000, maxiter=400)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 9: Binary snap + re-optimize (diversification)
    if _remaining() > 120:
        h_bin = binary_snap(h, n)
        h_bin = lbfgsb_optimize(h_bin, 5000, 2000, maxiter=3000)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            h = h_bin.copy()
            # Continue from binary solution
            if _remaining() > 60:
                h_new = lbfgsb_optimize(h, 15000, 5000, maxiter=3000)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
                    h = h_new.copy()

    # Phase 10: Final SLP
    if _remaining() > 40:
        h_slp, c5_slp = slp_loop(h, n, time_limit=min(35.0, _remaining() * 0.4), K=60)
        if c5_slp < best_c5:
            best_c5 = c5_slp
            best_h = h_slp.copy()
            h = h_slp.copy()

    # Phase 11: Final fine-tuning subgradient
    if _remaining() > 20:
        h_sg, c5_sg = subgradient_refine(h, n, time_limit=min(15.0, _remaining() * 0.4), rng=rng)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 12: Final L-BFGS-B
    if _remaining() > 20:
        h_new = lbfgsb_optimize(best_h, 50000, 10000, maxiter=2000)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Final feasibility and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
