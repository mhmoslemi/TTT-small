def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):

        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(x, beta, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -60.0, 60.0)
    val = m + np.log(np.sum(np.exp(shifted))) / beta
    return val * (2.0 / n)

def smooth_max_grad(x, beta, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -60.0, 60.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(len(corr)) / len(corr)
    else:
        w = w / ws
    dx = 2.0 / n
    nfull = 2 * n - 1
    g = np.zeros(n)
    for k in range(nfull):
        wk = w[k] * dx
        if abs(wk) < 1e-18:
            continue
        d = k - (n - 1)
        if d >= 0:
            m = n - d
            g[d:d+m] += wk * (1.0 - x[:m])
            g[:m] -= wk * x[d:d+m]
        else:

        return np.sum(x) - n / 2.0

        return np.ones(n)
