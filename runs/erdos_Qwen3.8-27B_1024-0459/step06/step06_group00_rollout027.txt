def grad_single_shift(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)

    else:
        a = -off

    return g

def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def coordinate_refine(h, n, max_sweeps=15):
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        improved_sweep = False
        for i in range(n):
            old = h[i]
            if 0.05 < old < 0.95:
                cands = [old + 0.05, old - 0.05, old + 0.15, old - 0.15, 0.0, 1.0]
            else:
                cands = [0.0, 1.0]
            cands = [c for c in cands if 0.0 <= c <= 1.0 and abs(c - old) > 1e-6]
            best_cand_val = old
            best_cand_c5 = best_c5
            for c in cands:
                h[i] = c
                h = project_feasible(h, n)
                c5_v = compute_c5(h)
                if c5_v < best_cand_c5 - 1e-13:
                    best_cand_c5 = c5_v
                    best_cand_val = c
                    improved_sweep = True
            h[i] = best_cand_val
            h = project_feasible(h, n)
        if not improved_sweep:
            break
    return h, best_c5

        h_lp, c5_lp = active_set_lp(h, n, top_k=15, max_iters=80,
                                     time_limit=min(30, int(remaining_time() * 0.3)))
        if c5_lp < best_c5 - 1e-15:
            best_c5 = c5_lp
            best_h = h_lp.copy()

    if remaining_time() > 10:
        h_eq, c5_eq = top_k_equalize(best_h, n, top_k=12, max_iters=300,
                                      time_limit=min(20, int(remaining_time() * 0.3)))
        if c5_eq < best_c5 - 1e-15:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    if remaining_time() > 10:
        h_sg, c5_sg = subgradient_zigzag(best_h, n, max_iters=3000,
                                          time_limit=min(30, int(remaining_time() * 0.3)))
        if c5_sg < best_c5 - 1e-15:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Coordinate refinement
    if remaining_time() > 10:
        h_cr, c5_cr = coordinate_refine(best_h, n, max_sweeps=10)
        if c5_cr < best_c5 - 1e-15:
            best_c5 = c5_cr
            best_h = h_cr.copy()

    # Final SLSQP
    if remaining_time() > 8:
        h_final = slsqp_opt(best_h, n, 100000000, maxiter=1000)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-15:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Ensure feasibility
    n = best_n
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Validate with evaluator
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, n)
