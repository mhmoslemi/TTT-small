import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_corr_at_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        g[offset:] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:]
    else:
        a = -offset
        npairs = n - a
        g[:npairs] += (1.0 - h[a:])
        g[a:] -= h[:npairs]
    return g


def topk_smooth(corr, K, beta):
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    m = np.max(c_top)
    sh = np.clip(beta * (c_top - m), -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / ex.sum()
    val = m + np.log(ex.sum()) / beta
    return val, idx, w


def gradient_topk(h, n, K, beta):
    corr = compute_corr(h)
    val, idx, w = topk_smooth(corr, K, beta)
    dx = 2.0 / n
    g = np.zeros(n)
    for i in range(K):
        wk = w[i] * dx
        if wk < 1e-15:
            continue
        g += wk * grad_corr_at_k(h, idx[i], n)
    return g, val * dx, corr


def slsqp_optimize(h, n, K, beta, maxiter=300):
    def obj(x):
        corr = compute_corr(x)
        val, _, _ = topk_smooth(corr, K, beta)
        return val * (2.0 / n)

    def grad(x):
        g, _, _ = gradient_topk(x, n, K, beta)
        return g

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            new_h = project_feasible(res.x, n)
            return new_h
    except Exception:
        pass
    return h


def cvxpy_cutting_plane(h_init, n, num_constraints=80, max_iters=20, trust_radius=None):
    dx = 2.0 / n
    h = h_init.copy()
    if trust_radius is None:
        trust_radius = 0.15

    for it in range(max_iters):
        corr = compute_corr(h)
        top_k = np.argsort(-corr)[:num_constraints]

        C = cp.Variable()
        h_var = cp.Variable(n)

        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n / 2.0,
        ]

        for k in top_k:
            offset = k - (n - 1)
            if offset >= 0:
                npairs = n - offset
                a_idx = np.arange(npairs)
                b_idx = a_idx + offset
                # Linearize h[i]*h[i+off] at current h:
                # ≈ h0[b]*h_var[a] + h0[a]*h_var[b] - h0[a]*h0[b]
                lin_prod = h[b_idx] * h_var[a_idx] + h[a_idx] * h_var[b_idx] - h[a_idx] * h[b_idx]
                lhs = cp.sum(h_var[a_idx]) - cp.sum(lin_prod)
                constraints.append(lhs * dx <= C)
            else:
                a_size = -offset
                npairs = n - a_size
                a_idx = np.arange(npairs) + a_size
                b_idx = np.arange(npairs)
                lin_prod = h[b_idx] * h_var[a_idx] + h[a_idx] * h_var[b_idx] - h[a_idx] * h[b_idx]
                lhs = cp.sum(h_var[a_idx]) - cp.sum(lin_prod)
                constraints.append(lhs * dx <= C)

        # Trust region
        constraints.append(cp.norm(h_var - h, 2) <= trust_radius)

        prob = cp.Problem(cp.Minimize(C), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=2000, tol_abs=1e-8, tol_rel=1e-8)
            if h_var.value is not None:
                new_h = np.array(h_var.value).flatten()
                new_h = project_feasible(new_h, n)
                c5_new = compute_c5(new_h)
                c5_old = compute_c5(h)
                if c5_new <= c5_old + 1e-10:
                    h = new_h
                else:
                    # Try a damped step
                    step_h = new_h - h
                    for damp in [0.5, 0.25, 0.1]:
                        damped_h = project_feasible(h + damp * step_h, n)
                        if compute_c5(damped_h) <= c5_old + 1e-10:
                            h = damped_h
                            break
        except Exception:
            pass

        # Shrink trust region slowly
        trust_radius *= 0.95
        if trust_radius < 0.01:
            break

    return h

    c5_best = compute_c5(h)

        if _BUDGET - (time.time() - _T0) < 5:
            break

        if it == 0:
            y = h_curr.copy()
        else:
            gamma = (it - 1 + np.sqrt(1.0 + 4.0 * (it - 1) ** 2)) / (2.0 * it)
            y = h_curr + gamma * (h_curr - h_prev)
            y = np.clip(y, 0.0, 1.0)

            h_t = slsqp_optimize(h_t, target_n, 25, 20000, maxiter=250)
            c5_t = compute_c5(h_t)

        # Nesterov at higher res
        if remaining() > 30:
            h_t, c5_t = nesterov_refine(h_t, target_n, min(40.0, remaining() * 0.2), K=30, beta=80)
            h_t = slsqp_optimize(h_t, target_n, 20, 30000, maxiter=300)
            c5_t = compute_c5(h_t)

        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            best_n = target_n

    # Phase 5: Local moves search
    n = best_n
    if remaining() > 40:
        ls_time = min(80.0, remaining() * 0.25)
        h_ls, c5_ls = local_moves_search(best_h, n, rng, ls_time, K=15, beta=100)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Phase 6: Another round of CVXPY cutting plane
    n = best_n
    if remaining() > 25:
        h_cp2 = cvxpy_cutting_plane(best_h, n, num_constraints=100, max_iters=18, trust_radius=0.08)
        c5_cp2 = compute_c5(h_cp2)
        h_cp2 = slsqp_optimize(h_cp2, n, 15, 30000, maxiter=300)
        c5_cp2 = compute_c5(h_cp2)
        if c5_cp2 < best_c5 - 1e-13:
            best_c5 = c5_cp2
            best_h = h_cp2.copy()

    # Phase 7: Multi-start SLSQP
    n = best_n
    if remaining() > 30:
        h_ms, c5_ms = multi_start_slsqp(best_h, n, rng, 3, 15, 10000, 200)
        if c5_ms < best_c5:
            best_c5 = c5_ms
            best_h = h_ms.copy()

    # Phase 8: Final intensive SLSQP with varying parameters
    n = best_n
    h = best_h.copy()
    for K in [6, 10, 15, 20, 30]:
        for beta in [5000, 20000, 80000, 200000]:
            if remaining() < 4:
                break
            h_new = slsqp_optimize(h, n, K, beta, maxiter=400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new

    # Phase 9: Final Nesterov + local moves
    n = best_n
    if remaining() > 20:
        h_fn, c5_fn = nesterov_refine(best_h, n, min(25.0, remaining() * 0.4), K=10, beta=200)
        if c5_fn < best_c5:
            best_c5 = c5_fn
            best_h = h_fn.copy()

    if remaining() > 15:
        h_lm, c5_lm = local_moves_search(best_h, best_n, rng, min(15.0, remaining() * 0.5), K=8, beta=200)
        if c5_lm < best_c5:
            best_c5 = c5_lm
            best_h = h_lm.copy()

    # Final SLSQP polish
    if remaining() > 5:
        h = slsqp_optimize(best_h, best_n, 10, 100000, maxiter=500)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    # Final feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        best_c5 = min(best_c5, eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
