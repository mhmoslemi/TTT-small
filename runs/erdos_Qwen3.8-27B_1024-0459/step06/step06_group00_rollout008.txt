        cnt = np.count_nonzero(free)
        if cnt > 0:
            adj = diff / cnt
            h[free] = np.clip(h[free] + adj, 0.0, 1.0)
        else:
            h[:] += diff / n
            h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        ones = np.where(h_bin == 1)[0]
        remove = ones[np.argsort(h[ones])[:n_ones - target]]
        h_bin[remove] = 0.0
    elif n_ones < target:
        zeros = np.where(h_bin == 0)[0]
        add = zeros[np.argsort(-h[zeros])[:target - n_ones]]
        h_bin[add] = 1.0
    return h_bin

def grad_shift(h, s, n):
    """Gradient of sum_i h[i]*(1-h[i+s]) with respect to h.
    s is the shift index in the full correlation (0 to 2n-2).
    Returns gradient vector of length n."""
    g = np.zeros(n)
    off = s - (n - 1)  # actual shift
    if off >= 0:
        np_ = n - off
        # Terms: h[i]*(1-h[i+off]) for i=0..np_-1
        # d/dh[i] for i < np_: (1-h[i+off])
        # d/dh[i+off] for i < np_: -h[i]
        g[:np_] += (1.0 - h[off:])
        g[off:] -= h[:np_]
    else:
        a = -off
        np_ = n - a
        # Terms: h[i+a]*(1-h[i]) for i=0..np_-1
        g[a:] += (1.0 - h[:np_])
        g[:np_] -= h[a:]
    return g
