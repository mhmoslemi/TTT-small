def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)

        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        cnt = np.count_nonzero(mask)

        else:

def smooth_max_value(x, n, K, beta):

def penalty_homotopy(h, n, t_start, t_end, time_limit, n_steps=50):
    h_curr = h.copy()
    h_best = h.copy()
    c5_best = compute_c5(h)
    start = time.time()
    dx = 2.0 / n
    ts = np.linspace(t_start, t_end, n_steps)
    
    for t in ts:
        if time.time() - start > time_limit:
            break
        rem = _BUDGET - (time.time() - _T0)
        if rem < 10:
            break
        
        for it in range(300):
            if time.time() - start > time_limit:
                break
            rem = _BUDGET - (time.time() - _T0)
            if rem < 5:
                break
            
            corr = compute_corr(h_curr)
            overlaps = corr * dx
            # Find shifts exceeding threshold
            active = np.where(overlaps > t)[0]
            if len(active) == 0:
                break
            
            # Compute gradient only for active shifts
            g = np.zeros(n)
            for k in active:
                excess = overlaps[k] - t
                offset = k - (n - 1)
                if offset >= 0:
                    npairs = n - offset
                    g[offset:] += 2.0 * excess * dx * (1.0 - h_curr[:npairs])
                    g[:npairs] += -2.0 * excess * dx * h_curr[offset:]
                else:
                    a = -offset
                    npairs = n - a
                    g[:npairs] += 2.0 * excess * dx * (1.0 - h_curr[a:])
                    g[a:] += -2.0 * excess * dx * h_curr[:npairs]
            
            gn = np.linalg.norm(g)
            if gn < 1e-15:
                break
            
            step = 0.05 / gn
            h_curr = project_feasible(h_curr - step * g, n)
        
        c5_curr = compute_c5(h_curr)
        if c5_curr < c5_best:
            c5_best = c5_curr
            h_best = h_curr.copy()
    
    return h_best, c5_best

    h_curr = h.copy()
    h_best = h.copy()
    c5_best = compute_c5(h)
    lam = 0.0
    start = time.time()
    
    for lam_it in range(max_lam_iters):
        if time.time() - start > time_limit:
            break
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        
        def obj(x):
            v = smooth_max_value(x, n, K, beta)
            violation = np.sum(x) - n / 2.0
            return v + lam * violation + (rho / 2.0) * violation ** 2

        def grad(x):
            g = smooth_max_grad(x, n, K, beta)
            violation = np.sum(x) - n / 2.0
            g = g + lam * np.ones(n) + rho * violation * np.ones(n)
            return g

        bounds = [(0.0, 1.0)] * n
        
        try:
            res = minimize(obj, h_curr, jac=grad, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 200, 'ftol': 1e-16, 'iprint': 0})
            h_new = np.clip(res.x, 0.0, 1.0)
        except Exception:
            h_new = h_curr
        
        # Update multiplier
        violation = np.sum(h_new) - n / 2.0
        lam = lam + rho * violation
        
        # Project to feasible
        h_curr = project_feasible(h_new, n)
        c5_curr = compute_c5(h_curr)
        if c5_curr < c5_best:
            c5_best = c5_curr
            h_best = h_curr.copy()
        
        if abs(violation) < 1e-10:
            break
    
    return h_best, c5_best

```python
import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_value(x, n, K, beta):
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    m = np.max(c_top)
    sh = np.clip(beta * (c_top - m), -50.0, 50.0)
    ex = np.exp(sh)
    return (m + np.log(ex.sum()) / beta) * (2.0 / n)

def smooth_max_grad(x, n, K, beta):
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    m = np.max(c_top)
    sh = np.clip(beta * (c_top - m), -50.0, 50.0)
    ex = np.exp(sh)
    s = ex.sum()
    if s < 1e-30:
        w = np.ones(K) / K
    else:
        w = ex / s
    dx = 2.0 / n
    g = np.zeros(n)
    for i in range(K):
        wk = w[i] * dx
        if wk < 1e-15:
            continue
        k = idx[i]
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            g[offset:] += wk * (1.0 - x[:npairs])
            g[:npairs] -= wk * x[offset:]
        else:
            a = -offset
            npairs = n - a
            g[:npairs] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:npairs]
    return g

def slsqp_optimize(h, n, K, beta, maxiter=400):
    def obj(x):
        return smooth_max_value(x, n, K, beta)
    def grad(x):
        return smooth_max_grad(x, n, K, beta)
    def cfun(x):
        return np.sum(x) - n / 2.0
    def cjac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            if compute_c5(h_new) <= compute_c5(h) + 1e-12:
                return h_new
    except Exception:
        pass
    return h

def augmented_lbfgsb(h, n, K, beta, time_limit, rho=10.0, max_lam_iters=30):
    h_curr = h.copy()
    h_best = h.copy()
    c5_best = compute_c5(h)
    lam = 0.0
    start = time.time()
    
    for lam_it in range(max_lam_iters):
        if time.time() - start > time_limit:
            break
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        
        def obj(x):
            v = smooth_max_value(x, n, K, beta)
            viol = np.sum(x) - n / 2.0
            return v + lam * viol + 0.5 * rho * viol * viol
        
        def grad(x):
            g = smooth_max_grad(x, n, K, beta)
            viol = np.sum(x) - n / 2.0
            return g + (lam + rho * viol) * np.ones(n)
        
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj, h_curr, jac=grad, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 300, 'ftol': 1e-16, 'iprint': 0})
            h_new = np.clip(res.x, 0.0, 1.0)
        except Exception:
            h_new = h_curr
        
        viol = np.sum(h_new) - n / 2.0
        lam_new = lam + rho * viol
        h_curr = project_feasible(h_new, n)
        c5_curr = compute_c5(h_curr)
        if c5_curr < c5_best:
            c5_best = c5_curr
            h_best = h_curr.copy()
        
        if abs(viol) < 1e-10:
            lam = lam_new
            break
        lam = lam_new
    
    return h_best, c5_best

def penalty_search(h, n, time_limit):
    """Penalty-based search: iteratively lower the threshold."""
    h_curr = h.copy()
    h_best = h.copy()
    c5_best = compute_c5(h)
    start = time.time()
    dx = 2.0 / n
    
    t = c5_best + 0.005  # Start slightly above current
    n_outer = 40
    
    for outer in range(n_outer):
        if time.time() - start > time_limit:
            break
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        
        t = max(0.37, c5_best - 0.002)  # Target slightly below current best
        
        # Inner: projected gradient on penalty
        for inner in range(200):
            if time.time() - start > time_limit:
                break
            if rem < 3:
                break
            
            corr = compute_corr(h_curr)
            overlaps = corr * dx
            active = np.where(overlaps > t)[0]
            
            if len(active) == 0:
                break
            
            g = np.zeros(n)
            for k in active:
                excess = overlaps[k] - t
                offset = k - (n - 1)
                if offset >= 0:
                    npairs = n - offset
                    g[offset:] += 2.0 * excess * dx * (1.0 - h_curr[:npairs])
                    g[:npairs] -= 2.0 * excess * dx * h_curr[offset:]
                else:
                    a = -offset
                    npairs = n - a
                    g[:npairs] += 2.0 * excess * dx * (1.0 - h_curr[a:])
                    g[a:] -= 2.0 * excess * dx * h_curr[:npairs]
            
            gn = np.linalg.norm(g)
            if gn < 1e-15:
                break
            
            step = 0.02 / gn
            h_new = project_feasible(h_curr - step * g, n)
            
            c5_new = compute_c5(h_new)
            if c5_new <= c5_best:
                h_curr = h_new
                if c5_new < c5_best:
                    c5_best = c5_new
                    h_best = h_curr.copy()
            else:
                # Try smaller step
                step *= 0.3
                h_new = project_feasible(h_curr - step * g, n)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best:
                    h_curr = h_new
                    c5_best = c5_new
                    h_best = h_curr.copy()
        
        if compute_c5(h_curr) < c5_best:
            c5_best = compute_c5(h_curr)
            h_best = h_curr.copy()
    
    return h_best, c5_best

def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)
