import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_CUR_N = 0
_SM_BETA = 1000.0
_ACTIVE_IDX = None
_ACTIVE_BETA = 1000.0


def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_single_shift(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        np_ = n - off
        g[off:] += (1.0 - h[:np_])
        g[:np_] -= h[off:]
    else:
        a = -off
        np_ = n - a
        g[:np_] += (1.0 - h[a:])
        g[a:] -= h[:np_]
    return g


def _eq_fun(x):
    return np.sum(x) - _CUR_N / 2.0


def _eq_jac(x):
    return np.ones(_CUR_N)


def _active_obj(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    c_act = corr[_ACTIVE_IDX]
    c_max = np.max(c_act)
    beta = _ACTIVE_BETA
    z = np.clip(beta * (c_act - c_max), -50.0, 50.0)
    ez = np.exp(z)
    return (c_max + np.log(np.sum(ez)) / beta) * dx


def _active_grad(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    c_act = corr[_ACTIVE_IDX]
    c_max = np.max(c_act)
    beta = _ACTIVE_BETA
    z = np.clip(beta * (c_act - c_max), -50.0, 50.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for pos, k in enumerate(_ACTIVE_IDX):
        wk = w[pos] * dx
        if wk < 1e-15:
            continue
        g += wk * grad_single_shift(x, k, n)
    return g


def slsqp_active(h, active_idx, beta, maxiter=2000):
    global _CUR_N, _ACTIVE_IDX, _ACTIVE_BETA
    _CUR_N = len(h)
    _ACTIVE_IDX = active_idx
    _ACTIVE_BETA = beta
    h0 = project_feasible(h, _CUR_N)
    bounds = [(0.0, 1.0)] * _CUR_N
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        obj0 = _active_obj(h0)
        res = minimize(_active_obj, h0, jac=_active_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.fun < obj0 - 1e-14:
            return project_feasible(res.x, _CUR_N)
    except Exception:
        pass
    return h0


def active_set_loop(h, n, max_outer=30, betas=(5000, 20000, 50000, 200000, 500000, 2000000),

    best_c5 = compute_c5(h)

    outer = 0
    for beta in betas:

                best_c5 = c5_new

def multiplicative_weight(h, n, n_iters=50, tau_schedule=None, time_limit=120):
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    dx = 2.0 / n
    start = time.time()
    n_corr = 2 * n - 1
    w = np.ones(n_corr) / n_corr
    tau = 5.0 / dx

    for it in range(n_iters):
        if remaining_time() < 10 or time.time() - start > time_limit:
            break
        # Build objective: min sum_k w_k * corr[k] * dx
        # This is -max h^T B_w h + const, approximately linear for small dx
        # Use SLSQP on the linear approximation with quadratic correction
        def mw_obj(x):
            corr = compute_corr(x)
            return np.dot(w, corr) * dx

        def mw_grad(x):
            corr = compute_corr(x)
            g = np.zeros(n)
            for k in range(n_corr):
                if w[k] < 1e-10:
                    continue
                g += w[k] * dx * grad_single_shift(x, k, n)
            return g

        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
        global _CUR_N
        _CUR_N = n
        try:
            res = minimize(mw_obj, h, jac=mw_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-12})
            if res.success:
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
        except Exception:
            pass

        # Update weights
        corr = compute_corr(h)
        tau = max(tau * 0.85, 0.1 / dx)
        w = np.exp(corr / tau)
        w = w / np.sum(w)

    return best_h, best_c5


def binary_sa(h, n, time_limit=180, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h_bin = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        sort_idx = np.argsort(-h_bin)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0

    dx = 2.0 / n
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    curr_c5 = best_c5
    start = time.time()

    T_start = 10.0 / dx
    T_end = 0.0005 / dx
    total_iters = 50000
    min_T_ratio = T_end / T_start

    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    no_improve = 0
    best_no_improve_limit = 8000

    for it in range(total_iters):
        if time.time() - start > time_limit:
            break
        if remaining_time() < 5:
            break

        T = T_start * (min_T_ratio ** (it / max(1, total_iters)))

        i_idx = rng.randint(len(ones))
        j_idx = rng.randint(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        if i == j:
            continue

        h_bin[i] = 0.0
        h_bin[j] = 1.0
        new_c5 = compute_c5(h_bin)
        delta = new_c5 - curr_c5

        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / max(T, 1e-10))):
            ones[i_idx] = j
            zeros[j_idx] = i
            curr_c5 = new_c5
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_h = h_bin.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            no_improve += 1

        if no_improve > best_no_improve_limit:
            break

    return best_h, best_c5


def binary_multi_start_sa(h, n, n_starts=5, time_per_start=60, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    best_h = (h > 0.5).astype(float)
    best_c5 = compute_c5(best_h)

    for s in range(n_starts):
        if remaining_time() < time_per_start + 5:
            break
        # Different perturbation for each start
        h_start = best_h.copy()
        n_swap = rng.randint(n // 10, n // 4)
        ones_idx = np.where(h_start == 1.0)[0]
        zeros_idx = np.where(h_start == 0.0)[0]
        for _ in range(n_swap):
            i = ones_idx[rng.randint(len(ones_idx))]
            j = zeros_idx[rng.randint(len(zeros_idx))]
            h_start[i] = 0.0
            h_start[j] = 1.0

        h_res, c5_res = binary_sa(h_start, n, time_limit=time_per_start, rng=rng)
        if c5_res < best_c5 - 1e-15:
            best_c5 = c5_res
            best_h = h_res.copy()

    return best_h, best_c5


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def theta_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)


def block_structure(n, n_blocks=None):
    if n_blocks is None:
        n_blocks = n // 20
    h = np.zeros(n)
    block_len = n // n_blocks
    for b in range(n_blocks):
        start = b * block_len
        end = start + block_len
        pattern = np.random.RandomState(b * 7 + 3).rand(block_len)
        h[start:end] = (pattern > 0.5).astype(float)
    return project_feasible(h, n)


def top_k_descend(h, n, top_k=6, max_iters=500, time_limit=60):
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    start = time.time()
    for it in range(max_iters):
        if remaining_time() < 3 or time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        c_max = corr[idx_top[0]]
        c_min_top = corr[idx_top[-1]]
        if c_max - c_min_top < 1e-12:
            break
        g_dir = np.zeros(n)
        n_top = len(idx_top)
        for pos, k in enumerate(idx_top):
            w_k = 1.0 if pos == 0 else (-1.0 / (n_top - 1))
            g_dir += w_k * grad_single_shift(h, k, n)
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        d = g_dir / gn
        improved = False
        for step in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.002, 0.03]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def subgrad_escape(h, n, max_iters=6000, time_limit=90):
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    k_prev = None
    stale = 0

    for it in range(max_iters):
        if remaining_time() < 5 or time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        k_curr = int(np.argmax(corr))

        if k_prev is not None and k_curr != k_prev:
            g = 0.5 * (grad_single_shift(h, k_curr, n) + grad_single_shift(h, k_prev, n))
            k_prev = k_curr
        else:
            g = grad_single_shift(h, k_curr, n)
            if k_prev is None:
                k_prev = k_curr

        g_norm = np.linalg.norm(g)
        if g_norm > 1e-14:
            g = g / g_norm

        improved = False
        base = max(0.0005, 0.01 * (1.0 - 0.5 * it / max_iters))
        for step in [base * 0.3, base * 0.7, base, base * 2, base * 4, 0.001, 0.003, 0.01, 0.03, 0.1]:
            ht = project_feasible(h - step * g, n)
            c5_new = compute_c5(ht)
            if c5_new < best_c5 - 1e-15:
                h = ht
                best_c5 = c5_new
                best_h = h.copy()
                improved = True
                stale = 0
                break
        if not improved:
            stale += 1
            if stale > 300:
                break
            if stale % 100 == 0:
                rng = np.random.RandomState(stale + it)
                pert = rng.randn(n) * 0.01
                ht = project_feasible(h + pert, n)
                c5_new = compute_c5(ht)
                if c5_new < best_c5:
                    h = ht
                    best_c5 = c5_new
                    best_h = h.copy()
                    stale = 0

    return best_h, best_c5


def refine_cycle(h, n, time_limit=120):
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()

    # Active set SLSQP
    if remaining_time() > 20 and time.time() - start < time_limit:
        corr = compute_corr(h)
        top_idx = np.argsort(-corr)[:10]
        for beta in [10000, 100000, 1000000]:
            if remaining_time() < 10 or time.time() - start > time_limit:
                break
            h_new = slsqp_active(h, top_idx, beta, maxiter=1000)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
                corr = compute_corr(h)
                top_idx = np.argsort(-corr)[:10]

    # Top-k descend
    if remaining_time() > 15 and time.time() - start < time_limit:
        tl = min(30, max(5, int(time_limit - (time.time() - start))))
        h_eq, c5_eq = top_k_descend(h, n, top_k=8, max_iters=400, time_limit=tl)
        if c5_eq < best_c5 - 1e-15:

            h = h_eq.copy()

    # Subgradient
    if remaining_time() > 15 and time.time() - start < time_limit:

                if remaining_time() < 8:
                    break
                corr = compute_corr(h_final)
                top_idx = np.argsort(-corr)[:top_m]
                h_new = slsqp_active(h_final, top_idx, beta, maxiter=2000)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h_final = h_new.copy()

        # Final top-k and subgradient
        if remaining_time() > 10:
            h_eq, c5_eq = top_k_descend(h_final, n_final, top_k=5, max_iters=500,
                                         time_limit=min(20, int(remaining_time() - 5)))
            if c5_eq < best_c5 - 1e-15:
                best_c5 = c5_eq
                best_h = h_eq.copy()

        if remaining_time() > 10:
            h_sg, c5_sg = subgrad_escape(best_h, n_final, max_iters=4000,
                                          time_limit=min(20, int(remaining_time() - 3)))
            if c5_sg < best_c5 - 1e-15:
                best_c5 = c5_sg
                best_h = h_sg.copy()

        # Last polish
        if remaining_time() > 8:
            corr = compute_corr(best_h)
            top_idx = np.argsort(-corr)[:4]
            h_pol = slsqp_active(best_h, top_idx, 20000000, maxiter=2000)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5 - 1e-14:
                best_c5 = c5_pol
                best_h = h_pol.copy()

        print(f"Phase7: {best_c5:.6f} (t={time.time()-t7:.1f}s)")

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Validate with evaluator
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    print(f"FINAL: C5 = {best_c5:.6f}, n = {best_n}")
    return (best_h, best_c5, best_n)
