from scipy.optimize import minimize
from scipy.signal import fftconvolve
import time

_T0 = 0.0
_BUDGET = 1300.0
_CUR_N = 0
_SM_BETA = 1000.0

    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    return (m + np.log(np.sum(ez)) / beta) * dx


def sm_grad(x):

    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S  # weights over 2n-1 lags
    
    # Gradient via FFT convolution
    conv1 = fftconvolve(w, 1.0 - x, mode='full')
    conv2 = fftconvolve(w[::-1], x, mode='full')
    g = dx * (conv1[n-1:2*n-1] - conv2[n-1:2*n-1])
    return g

    return np.sum(x) - _CUR_N / 2.0


def eq_jac(x):
    return np.ones(_CUR_N)

    _CUR_N = n
    _SM_BETA = beta

                       bounds=bounds, constraints=cons,

def grad_single_shift(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        np_ = n - off
        g[off:] += (1.0 - h[:np_])
        g[:np_] -= h[off:]
    else:
        a = -off
        np_ = n - a
        g[:np_] += (1.0 - h[a:])
        g[a:] -= h[:np_]
    return g


def top_k_equalize(h, n, top_k=5, max_iters=500, time_limit=None):
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    start = time.time()
    for it in range(max_iters):
        if remaining_time() < 3:
            break
        if time_limit is not None and time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        c_max = corr[idx_top[0]]
        c_min_top = corr[idx_top[-1]]
        if c_max - c_min_top < 1e-12:
            break
        g_dir = np.zeros(n)
        n_top = len(idx_top)
        for pos, k in enumerate(idx_top):
            w_k = 1.0 if pos == 0 else (-1.0 / (n_top - 1))
            g_dir += w_k * grad_single_shift(h, k, n)
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        d = g_dir / gn
        improved = False
        steps = [0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.0005, 0.003, 0.008]
        for step in steps:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            for step in steps:
                ht = project_feasible(h - step * d, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-14:
                    h = ht
                    best_c5 = cn
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5


def nesterov_subgrad(h, n, max_iters=5000, time_limit=90):
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    momentum = np.zeros(n)
    beta = 0.9
    step_size = 0.05
    k_prev = None
    stale = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if remaining_time() < 5:
            break

        corr = compute_corr(h)
        k_curr = int(np.argmax(corr))

        if k_prev is not None and k_curr != k_prev:
            g = 0.5 * (grad_single_shift(h, k_curr, n) + grad_single_shift(h, k_prev, n))
        else:
            g = grad_single_shift(h, k_curr, n)
        k_prev = k_curr

        g_norm = np.linalg.norm(g)
        if g_norm > 1e-14:
            g = g / g_norm

        momentum = beta * momentum - step_size * g
        ht = project_feasible(h + momentum, n)
        c5_new = compute_c5(ht)

        if c5_new < best_c5 - 1e-15:
            h = ht
            best_c5 = c5_new
            best_h = h.copy()
            stale = 0
        else:
            h = project_feasible(h - 0.5 * momentum, n)
            momentum *= 0.5
            stale += 1

        step_size *= 0.999
        if stale > 300:
            break
        if stale % 100 == 0:
            rng = np.random.RandomState(stale + it)
            pert = rng.randn(n) * 0.003
            ht2 = project_feasible(h + pert, n)
            c5_2 = compute_c5(ht2)
            if c5_2 < best_c5:
                h = ht2
                best_c5 = c5_2
                best_h = h.copy()
                stale = 0
                momentum = np.zeros(n)

    return best_h, best_c5


def paired_coord_descent(h, n, max_rounds=500, time_limit=60, rng=None):
    if rng is None:
        rng = np.random.RandomState(123)
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()

    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        if remaining_time() < 5:
            break

        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if h[i] <= h[j]:
            i, j = j, i

        improved = False
        options = [0.01, 0.02, 0.05, 0.03, 0.005, 0.08, 0.002, 0.1]
        for delta in options:
            if h[i] - h[j] < delta:
                continue
            ht = h.copy()
            ht[i] -= delta
            ht[j] += delta
            ht = project_feasible(ht, n)
            c5_new = compute_c5(ht)
            if c5_new < best_c5 - 1e-14:
                h = ht
                best_c5 = c5_new
                best_h = h.copy()
                improved = True
                break
        if not improved:
            continue

    return best_h, best_c5

    best_c5 = curr_c5
    best_h = h_bin.copy()
    start = time.time()

    T_start = 5.0 / dx
    T_end = 0.001 / dx
    total_iters = 30000
    no_improve = 0

    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]

    for it in range(total_iters):
        if time.time() - start > time_limit:
            break
        if remaining_time() < 5:
            break

        T = T_start * (T_end / T_start) ** (it / max(1, total_iters))

        i_idx = rng.randint(len(ones))
        j_idx = rng.randint(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]

        h_bin[i] = 0.0
        h_bin[j] = 1.0
        new_c5 = compute_c5(h_bin)
        delta = new_c5 - curr_c5

        if delta < 0 or (T > 1e-10 and rng.random() < np.exp(-delta / T)):
            ones[i_idx] = j
            zeros[j_idx] = i
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_bin.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0

        if no_improve > 8000:
            break

    return best_h, best_c5


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def theta_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)


def interleave_pattern(n):
    h = np.zeros(n)
    for i in range(n):
        r = i % 4
        h[i] = 1.0 if r < 2 else 0.0
    return project_feasible(h, n)


def block_pattern(n, block_size=8):
    h = np.zeros(n)
    for i in range(n):
        phase = (i // block_size) % 2
        h[i] = 1.0 if phase == 0 else 0.0
    return project_feasible(h, n)


def refine(h, n, time_limit=60):
    """Multi-stage refinement pipeline."""
    best_h = project_feasible(h, n)
    best_c5 = compute_c5(best_h)

    for beta in [5000, 50000, 500000, 5000000]:
        if remaining_time() < 10:
            break
        h_new = slsqp_opt(best_h, n, beta, maxiter=800)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()

    if remaining_time() > 10:
        h_eq, c5_eq = top_k_equalize(best_h, n, top_k=8, max_iters=500,
                                      time_limit=min(30, int(remaining_time() * 0.3)))
        if c5_eq < best_c5 - 1e-15:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    if remaining_time() > 10:
        h_ns, c5_ns = nesterov_subgrad(best_h, n, max_iters=4000,
                                        time_limit=min(30, int(remaining_time() * 0.3)))
        if c5_ns < best_c5 - 1e-15:
            best_c5 = c5_ns
            best_h = h_ns.copy()

    if remaining_time() > 10:
        h_pc, c5_pc = paired_coord_descent(best_h, n, max_rounds=1000,
                                            time_limit=min(20, int(remaining_time() * 0.2)),
                                            rng=np.random.RandomState(99))
        if c5_pc < best_c5 - 1e-15:
            best_c5 = c5_pc
            best_h = h_pc.copy()

    for beta in [100000, 1000000, 10000000]:
        if remaining_time() < 10:
            break
        h_new = slsqp_opt(best_h, n, beta, maxiter=1000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()

    if remaining_time() > 5:
        h_eq, c5_eq = top_k_equalize(best_h, n, top_k=12, max_iters=400,
                                      time_limit=min(15, int(remaining_time() * 0.3)))
        if c5_eq < best_c5 - 1e-15:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _CUR_N
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = np.zeros(640)
        best_h[:320] = 1.0

    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick SLSQP on parent resolution
    for beta in [2000, 20000, 200000, 2000000]:
        if remaining_time() < 15:
            break
        h_new = slsqp_opt(best_h, best_n, beta, maxiter=1000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 2: Top-k and subgradient at parent resolution
    if remaining_time() > 20:
        h_eq, c5_eq = top_k_equalize(best_h, best_n, top_k=10, max_iters=600, time_limit=40)
        if c5_eq < best_c5 - 1e-15:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    if remaining_time() > 20:
        h_ns, c5_ns = nesterov_subgrad(best_h, best_n, max_iters=5000, time_limit=40)
        if c5_ns < best_c5 - 1e-15:
            best_c5 = c5_ns
            best_h = h_ns.copy()

    # Phase 3: Upsample to higher resolutions and optimize
    for n_target in [1280, 2048]:
        if remaining_time() < 60:
            break
        h_up = interp_to(best_h, best_n, n_target)
        h_up = project_feasible(h_up, n_target)
        c5_up = compute_c5(h_up)

        h_opt = refine(h_up, n_target, time_limit=min(120, int(remaining_time() * 0.4)))
        if h_opt[1] < best_c5 - 1e-10:
            best_c5 = h_opt[1]
            best_h = h_opt[0].copy()
            best_n = n_target

    # Phase 4: Binary SA at best resolution + continuous refinement
    if remaining_time() > 80:
        sa_time = min(120, int(remaining_time() * 0.25))
        h_bin_init = (best_h > 0.5).astype(float)
        h_b, c5_b = binary_sa(h_bin_init, best_n, time_limit=sa_time, rng=rng)

        h_b_c = project_feasible(h_b, best_n)
        h_b_c, c5_b_c = refine(h_b_c, best_n, time_limit=min(60, int(remaining_time() * 0.2)))

        if c5_b_c < best_c5 - 1e-15:
            best_c5 = c5_b_c
            best_h = h_b_c.copy()
        if c5_b < best_c5 - 1e-15:
            best_c5 = c5_b
            best_h = h_b.copy().astype(float)

    # Phase 5: Alternative starting points at current best resolution
    if remaining_time() > 60:
        alt_starts = [
            theta_morse(best_n),
            interleave_pattern(best_n),
            block_pattern(best_n, 8),
            block_pattern(best_n, 16),
        ]
        # Random balanced
        h_rand = np.zeros(best_n)
        h_rand[rng.choice(best_n, best_n // 2, replace=False)] = 1.0
        alt_starts.append(h_rand)

        for h_alt in alt_starts:
            if remaining_time() < 30:
                break
            h_opt, c5_opt = refine(h_alt, best_n, time_limit=min(40, int(remaining_time() * 0.25)))
            if c5_opt < best_c5 - 1e-10:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 6: Intensive final refinement at best resolution
    n = best_n
    h = best_h.copy()
    for beta in [50000, 500000, 5000000, 20000000]:
        if remaining_time() < 10:
            break
        h_new = slsqp_opt(h, n, beta, maxiter=2000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    if remaining_time() > 10:
        h_eq, c5_eq = top_k_equalize(h, n, top_k=15, max_iters=600,
                                      time_limit=min(25, int(remaining_time() * 0.3)))
        if c5_eq < best_c5 - 1e-15:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    if remaining_time() > 15:
        h_ns, c5_ns = nesterov_subgrad(h, n, max_iters=6000,
                                        time_limit=min(30, int(remaining_time() * 0.3)))
        if c5_ns < best_c5 - 1e-15:
            best_c5 = c5_ns
            best_h = h_ns.copy()
            h = h_ns.copy()

    if remaining_time() > 10:
        h_pc, c5_pc = paired_coord_descent(h, n, max_rounds=2000,
                                            time_limit=min(20, int(remaining_time() * 0.25)),
                                            rng=np.random.RandomState(777))
        if c5_pc < best_c5 - 1e-15:
            best_c5 = c5_pc
            best_h = h_pc.copy()

    # Final SLSQP polish
    if remaining_time() > 10:
        h_final = slsqp_opt(best_h, n, 50000000, maxiter=3000)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-15:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, n)
