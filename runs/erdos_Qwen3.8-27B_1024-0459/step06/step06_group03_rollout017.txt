def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_corr_k_vec(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        np_ = n - offset
        g[offset:] += (1.0 - h[:np_])
        g[:np_] -= h[offset:]
    else:
        a = -offset
        np_ = n - a
        g[:np_] += (1.0 - h[a:])
        g[a:] -= h[:np_]
    return g


def smmax_objective(h):
    n = _G_N
    corr = np.correlate(h, 1.0 - h, mode='full')
    beta = _G_BETA
    K = _G_K
    idx = np.argsort(-corr)[:K]
    c = corr[idx]
    m = c.max()
    sh = c - m
    cl = np.clip(beta * sh, -80.0, 80.0)
    ex = np.exp(cl)
    val = (m + np.log(ex.sum()) / beta) * (2.0 / n)
    return val


def smmax_gradient(h):
    n = _G_N
    beta = _G_BETA
    K = _G_K
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(-corr)[:K]
    c = corr[idx]
    m = c.max()
    sh = c - m
    cl = np.clip(beta * sh, -80.0, 80.0)
    ex = np.exp(cl)
    w = ex / ex.sum()
    g = np.zeros(n)
    for i in range(K):
        if w[i] < 1e-15:
            continue
        g += w[i] * grad_corr_k_vec(h, idx[i], n)
    return g


def sum_eq_fun(h):
    return np.sum(h) - _G_N / 2.0


def sum_eq_jac(h):
    return np.ones(_G_N)


def slsqop_smmax(h, n, beta, K, maxiter):
    global _G_N, _G_BETA, _G_K
    _G_N = n
    _G_BETA = beta
    _G_K = K
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}]
    try:
        res = minimize(smmax_objective, h, jac=smmax_gradient, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            if compute_c5(h_new) <= compute_c5(h) + 1e-12:
                return h_new
    except Exception:
        pass
    return h.copy()


def tmin_ineq_fun(v):
    n = _G_N
    h = v[:n]
    t = v[-1]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return t * n / 2.0 - corr[_G_KL]


def tmin_ineq_jac(v):
    n = _G_N
    h = v[:n]
    J = np.zeros((len(_G_KL), n + 1))
    for i, k in enumerate(_G_KL):
        g = grad_corr_k_vec(h, k, n)
        J[i, :n] = -g
        J[i, -1] = n / 2.0
    return J


def obj_t_fun(v):
    return v[-1]


def obj_t_jac(v):
    g = np.zeros(_G_N + 1)
    g[-1] = 1.0
    return g


def sum_eq_fun2(v):
    return np.sum(v[:_G_N]) - _G_N / 2.0


def sum_eq_jac2(v):
    g = np.zeros(_G_N + 1)
    g[:_G_N] = 1.0
    return g


def active_set_slsqp(h, n, K_list, maxiter):
    global _G_N, _G_KL
    _G_N = n
    _G_KL = K_list
    t0 = compute_c5(h)
    v0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'ineq', 'fun': tmin_ineq_fun, 'jac': tmin_ineq_jac},
        {'type': 'eq', 'fun': sum_eq_fun2, 'jac': sum_eq_jac2}
    ]
    try:
        res = minimize(obj_t_fun, v0, jac=obj_t_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x[:n], n)
            return h_new, compute_c5(h_new)
    except Exception:
        pass
    return h.copy(), compute_c5(h)


def subgradient_descent(h, n, time_limit, rng):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        progress = min(1.0, it / 600.0)
        K = 30
        beta = 30.0 + 100.0 * progress
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(-corr)[:K]
        sh = corr[idx] - corr[idx[0]]
        cl = np.clip(beta * sh, -60.0, 60.0)
        ex = np.exp(cl)
        w = ex / ex.sum()
        g = np.zeros(n)
        for i in range(K):
            if w[i] > 1e-12:
                g += w[i] * grad_corr_k_vec(h, idx[i], n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        step = 0.06 / (1.0 + 0.02 * it)
        step = min(step, 0.15)
        h_new = project_feasible(h - step * g, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            for factor in [0.5, 0.25, 0.1, 2.0]:
                h_try = project_feasible(h - step * factor * g, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    break
        it += 1
    return best_h, best_c5


def cutting_plane_step(h, n, K, delta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(-corr)[:K]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub_list = []
    b_ub_list = []
    for k in idx:
        g = grad_corr_k_vec(h, k, n)
        fk = corr[k]
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_ub_list.append(row)
        b_ub_list.append(float(np.dot(g, h)) - fk)
    I = np.eye(n)
    A_trust = np.vstack([
        np.hstack([I, np.zeros((n, 1))]),
        np.hstack([-I, np.zeros((n, 1))])
    ])
    b_trust = np.concatenate([h + delta, h + delta])
    A_ub = np.vstack([np.array(A_ub_list), A_trust])
    b_ub = np.concatenate([np.array(b_ub_list), b_trust])
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0, 1)] * n + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n], res.x[-1] * (2.0 / n)
    except Exception:
        pass
    return h.copy(), compute_c5(h)


def equalize_top_shifts(h, n, K, max_iters):
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(-corr)[:K]
        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        if c_max - c_min < 1e-10:
            break
        g_avg = np.zeros(n)
        for k in idx:
            g_avg += grad_corr_k_vec(h, k, n)
        g_avg /= K
        g_dir = -g_avg
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        g_dir /= gn
        improved = False
        for step in [0.02, 0.04, 0.01, 0.005, 0.06, 0.003, 0.03, 0.08, 0.002, 0.015]:
            ht = project_feasible(h + step * g_dir, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def coordinate_descent(h, n, time_limit, K):
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    rng = np.random.RandomState(123)
    while time.time() - start < time_limit:
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        j = rng.randint(n)
        free_mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        free_mask[j] = False
        free_idx = np.where(free_mask)[0]
        if len(free_idx) < 2:
            continue
        partner = free_idx[rng.randint(len(free_idx))]
        for delta_h in [0.01, 0.005, 0.002, 0.02, 0.001, 0.008, -0.01, -0.005, -0.002, -0.02]:
            h_try = h.copy()
            h_try[j] += delta_h
            h_try[partner] -= delta_h
            if h_try[j] < 0 or h_try[j] > 1 or h_try[partner] < 0 or h_try[partner] > 1:
                continue
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                h = h_try
                best_c5 = c5_try
                break
    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _G_N, _G_BETA, _G_K, _G_KL
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 512
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: Quick subgradient descent
    if remaining() > 40:
        sg_time = min(40.0, remaining() * 0.06)
        h_sg, c5_sg = subgradient_descent(h, n, sg_time, rng)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 2: Multi-stage smooth-max SLSQP with increasing K
    stages = [
        (beta, K, iters) for beta, K, iters in [
            (20, 30, 500), (50, 50, 500), (100, 60, 400),
            (200, 80, 400), (500, 100, 350), (1000, 120, 300),
            (2000, 150, 300), (5000, 100, 300), (10000, 80, 250),
            (20000, 60, 250), (50000, 50, 200),
        ]
    ]
    for beta, K, maxit in stages:
        if remaining() < 15:
            break
        h_new = slsqop_smmax(h, n, beta, K, maxiter)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 3: Cutting plane iterations
    if remaining() > 60:
        delta = 0.05
        for it in range(300):
            if remaining() < 5:
                break
            h_lp, t_lp = cutting_plane_step(h, n, K=25, delta=delta)
            c5_cur = compute_c5(h)
            if t_lp < c5_cur - 1e-10:
                improved = False
                for alpha in [1.0, 0.7, 0.5, 0.3, 0.9, 0.4]:
                    h_try = project_feasible(h + alpha * (h_lp - h), n)
                    c_try = compute_c5(h_try)
                    if c_try < best_c5 - 1e-12:
                        best_c5 = c_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
                        break
                if not improved:
                    delta = max(0.01, delta * 0.7)
            elif it > 50:
                delta = max(0.005, delta * 0.8)
            if delta < 0.005 and it > 100:
                break

    # Phase 4: Active-set min-max SLSQP
    if remaining() > 40:
        for K_active in [5, 8, 12, 16, 20, 25, 30, 40]:
            if remaining() < 10:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            idx_active = np.argsort(-corr)[:K_active]
            h_am, c5_am = active_set_slsqp(h, n, idx_active, maxiter=400)
            if c5_am < best_c5 - 1e-12:
                best_c5 = c5_am
                best_h = h_am.copy()
                h = h_am.copy()

    # Phase 5: Equalize top shifts
    if remaining() > 20:
        h_eq, c5_eq = equalize_top_shifts(h, n, K=10, max_iters=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 6: More smooth-max at very high beta
    for beta in [30000, 60000, 100000, 200000]:
        if remaining() < 10:
            break
        h = slsqop_smmax(h, n, beta, 60, maxiter=400)
        c5_tmp = compute_c5(h)
        if c5_tmp < best_c5:
            best_c5 = c5_tmp
            best_h = h.copy()

    # Phase 7: Try higher resolution n=768
    n2 = 768
    if remaining() > 200:
        h_interp = interp_to(best_h, best_n, n2)
        h_interp = project_feasible(h_interp, n2)
        c5_interp = compute_c5(h_interp)

        if c5_interp <= best_c5 + 0.002:
            h2 = h_interp.copy()
            # Subgradient at higher res
            if remaining() > 100:
                h2, c5_2 = subgradient_descent(h2, n2, min(30.0, remaining() * 0.1), rng)
                if c5_2 > c5_interp:
                    h2 = h_interp.copy()
                    c5_2 = c5_interp
            # Smooth-max stages
            for beta2, K2, it2 in [(50, 40, 300), (200, 60, 300), (500, 80, 250), (2000, 60, 250), (5000, 50, 200)]:
                if remaining() < 20:
                    break
                h2 = slsqop_smmax(h2, n2, beta2, K2, it2)
                c5_2 = compute_c5(h2)
            # Active set
            if remaining() > 40:
                for Ka in [8, 15, 25]:
                    if remaining() < 10:
                        break
                    corr2 = np.correlate(h2, 1.0 - h2, mode='full')
                    idx2 = np.argsort(-corr2)[:Ka]
                    h2, c5_2 = active_set_slsqp(h2, n2, idx2, maxiter=300)
            # Equalize
            if remaining() > 20:
                h2, c5_2 = equalize_top_shifts(h2, n2, K=10, max_iters=200)

            c5_2 = compute_c5(h2)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()
                best_n = n2
                h = h2.copy()
                n = n2

    # Phase 8: Intensive refinement cycles
    n = best_n
    h = best_h.copy()
    for cycle in range(4):
        if remaining() < 30:
            break
        # Subgradient
        if remaining() > 15:
            h, c5_t = subgradient_descent(h, n, min(15.0, remaining() * 0.1), rng)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h.copy()

        # SLSQP high beta
        if remaining() > 20:
            for beta in [10000, 30000, 80000]:
                if remaining() < 8:
                    break
                h = slsqop_smmax(h, n, beta, 50, maxiter=350)
            c5_t = compute_c5(h)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h.copy()

        # Active set
        if remaining() > 25:
            corr_c = np.correlate(h, 1.0 - h, mode='full')
            for Ka in [10, 20, 30, 40, 50]:
                if remaining() < 8:
                    break
                idx_a = np.argsort(-np.correlate(h, 1.0 - h, mode='full'))[:Ka]
                h, c5_t = active_set_slsqp(h, n, idx_a, maxiter=300)
                if c5_t < best_c5:
                    best_c5 = c5_t
                    best_h = h.copy()

        # Equalize
        if remaining() > 15:
            h, c5_t = equalize_top_shifts(h, n, K=12, max_iters=300)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h.copy()

        # Coordinate descent
        if remaining() > 20:
            h, c5_t = coordinate_descent(h, n, min(20.0, remaining() * 0.15), K=10)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h.copy()

    # Phase 9: Cutting plane again with very small delta
    if remaining() > 40:
        for it in range(200):
            if remaining() < 5:
                break
            delta_i = max(0.005, 0.02 * (0.95 ** it))
            h_lp, t_lp = cutting_plane_step(h, n, K=30, delta=delta_i)
            c5_cur = compute_c5(h)
            if t_lp < c5_cur - 1e-10:
                for alpha in [1.0, 0.6, 0.4, 0.8, 0.3, 0.5]:
                    h_try = project_feasible(h + alpha * (h_lp - h), n)
                    c_try = compute_c5(h_try)
                    if c_try < best_c5 - 1e-12:
                        best_c5 = c_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        break

    # Phase 10: Final ultra-high beta polish
    h = best_h.copy()
    for beta in [50000, 100000, 200000, 500000]:
        if remaining() < 5:
            break
        h = slsqop_smmax(h, best_n, beta, 40, maxiter=500)
        c5_f = compute_c5(h)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h.copy()

    # Phase 11: Very fine equalize
    if remaining() > 10:
        h_f, c5_f = equalize_top_shifts(best_h, best_n, K=15, max_iters=400)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 12: Tiny step refinement
    if remaining() > 10:
        h_t = best_h.copy()
        for it in range(100):
            if remaining() < 3:
                break
            corr_t = np.correlate(h_t, 1.0 - h_t, mode='full')
            idx_t = np.argsort(-corr_t)[:20]
            g_t = np.zeros(best_n)
            for k in idx_t:
                g_t += grad_corr_k_vec(h_t, k, best_n)
            g_t /= len(idx_t)
            gn_t = np.linalg.norm(g_t)
            if gn_t < 1e-14:
                break
            for step in [0.003, 0.001, 0.0005, 0.005, 0.002, 0.0008]:
                h_try = project_feasible(h_t - step * g_t / gn_t, best_n)
                c_try = compute_c5(h_try)
                if c_try < best_c5:
                    best_c5 = c_try
                    best_h = h_try.copy()
                    h_t = h_try.copy()
                    break

    # Final feasibility and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
