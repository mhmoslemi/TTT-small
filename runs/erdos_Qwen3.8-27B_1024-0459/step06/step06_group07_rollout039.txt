```python
def make_binary(h, n):
    """Convert h to binary with exactly n/2 ones."""
    # Round and fix
    h_bin = np.zeros(n)
    # Get the top n/2 values
    threshold = np.sort(h)[n//2]
    h_bin[h >= threshold] = 1.0
    # Fix count
    count = np.sum(h_bin)
    if count > n/2:
        # Turn off some 1s
        excess = int(count - n/2)
        ones = np.where(h_bin == 1)[0]
        # Turn off the ones with lowest original h value
        order = np.argsort(-h[ones])  # descending h, so last ones have lowest h
        for idx in order[-excess:]:
            h_bin[ones[idx]] = 0
    elif count < n/2:
        deficit = int(n/2 - count)
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])
        for idx in order[:deficit]:
            h_bin[zeros[idx]] = 1
    return h_bin
```

For the greedy descent:
- Repeatedly find the best swap that reduces max(C)
- If no such swap exists, stop
- "Best" = the swap that most reduces max(C), or if multiple reduce by the same amount, the one that also reduces the second-max, etc.

To find the best swap efficiently:
- For each i with h[i]=1 and each j with h[j]=0:
  - Compute ΔC = butterfly[i] - butterfly[j]
  - new_max = max(C + ΔC)
  - Keep track of the minimum new_max
- Apply the best swap

This is O(n² * n) = O(n³) per full pass. For n=512, that's about 134M operations, which in Python might take 10-30 seconds per pass. Too slow for many passes.

Faster approach: focus on "active" shifts. Let M = max(C). The shifts where C[m] is close to M are the "bottleneck" shifts. A swap is useful only if it reduces C[m] for at least one bottleneck shift without increasing any other shift above M.

For a swap (i,j) to reduce C[m_bottleneck]:
h[i+m] + h[i-m] < h[j+m] + h[j-m]

For binary h, this means the number of 1s in {i+m, i-m} is less than in {j+m, j-m}.

So for each bottleneck shift m, I want to find i where i±m have many 1s, and j where j±m have many 0s.

This gives a heuristic: for each bottleneck m, score each position p by:
score[p] = h[p+m] + h[p-m] (number of 1s adjacent to p at distance m)

Then for each bottleneck m, try swaps from high-score positions (where h=1) to low-score positions (where h=0).

This is much more focused and should be faster.

```python
def build_butterfly(h, n):
    """Precompute butterfly[p][m_idx] = h[p+m] + h[p-m] for all p, m_idx."""
    m_vals = np.arange(-(n-1), n)  # length 2n-1
    butterfly = np.zeros((n, 2*n-1), dtype=np.int8)
    for p in range(n):
        idx1 = p + m_vals
        mask1 = (idx1 >= 0) & (idx1 < n)
        butterfly[p, mask1] += h[idx1[mask1]]
        idx2 = p - m_vals
        mask2 = (idx2 >= 0) & (idx2 < n)
        butterfly[p, mask2] += h[idx2[mask2]]
    return butterfly

def update_butterfly(butterfly, h, i, j, n):
    """Update butterfly after swapping h[i]:1->0 and h[j]:0->1."""
    nfull = 2*n-1
    # h[i] goes from 1 to 0: butterfly[p] decreases by 1 where p±m = i
    for p in range(n):
        m_idx1 = i - p + (n-1)  # p + m = i → m = i-p → m_idx = i-p+n-1
        if 0 <= m_idx1 < nfull:
            butterfly[p, m_idx1] -= 1
        m_idx2 = p - i + (n-1)  # p - m = i → m = p-i → m_idx = p-i+n-1
        if 0 <= m_idx2 < nfull and m_idx2 != m_idx1:
            butterfly[p, m_idx2] -= 1
    # h[j] goes from 0 to 1: butterfly[p] increases by 1 where p±m = j
    for p in range(n):
        m_idx1 = j - p + (n-1)
        if 0 <= m_idx1 < nfull:
            butterfly[p, m_idx1] += 1
        m_idx2 = p - j + (n-1)
        if 0 <= m_idx2 < nfull and m_idx2 != m_idx1:
            butterfly[p, m_idx2] += 1
```

```python
    max_C = int(np.max(C))

    T_start = 2.0
    T_end = 0.01
    n_iters = int(time_limit * 100000)  # target ~100K iters/sec
    
    start = time.time()
    best_h = h.copy()
    best_max_C = max_C

            break
        if _BUDGET - (time.time() - _T0) < 10:

        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]

        delta = butterfly[i] - butterfly[j]

            h[i] = 0
            h[j] = 1

            if max_C < best_max_C:
                best_max_C = max_C
                best_h = h.copy()
```
