        best_i, best_j = -1, -1
        
        for ii in range(len(ones_idx)):
            i = ones_idx[ii]
            base = corr - dcorr[i]
            # For each zero j: new_corr = base + dcorr[j]
            # new_max = max(base + dcorr[j])
            new_maxs = base[None, :] + dcorr[zeros_idx]
            row_maxs = np.max(new_maxs, axis=1)
            local_best = int(np.argmin(row_maxs))
            local_improve = c_max - row_maxs[local_best]
            
            if local_improve > best_improve:
                best_improve = local_improve
                best_i = i
                best_j = int(zeros_idx[local_best])
        
        if best_improve < 1e-10:
            break
        
        h[best_i] = 0.0
        h[best_j] = 1.0
        new_c5 = compute_c5(h)
        if new_c5 < best_c5 - 1e-12:
            best_c5 = new_c5
        else:
            h[best_i] = 1.0
            h[best_j] = 0.0
            break
    
    return h, best_c5
```

Hmm, this is O(n_ones * n_zeros * 2n) per pass which for n=512 is 256*256*1023 ≈ 67M. In numpy, the vectorized max over axis for 256x1023 array is fast (~1ms). So 256 iterations * 1ms = 256ms per pass. We can do ~700 passes in 200s.

But actually, I realize the inner loop can be more carefully vectorized. For all ones at once:
new_maxs_all = np.max(corr[None,None,:] - dcorr[ones_idx,:,None] + dcorr[zeros_idx,None,:], axis=2)

This creates a (n_ones, n_zeros) array. For n=512: 256*256 = 65536 values, each requiring max over 1023 elements. That's 67M operations. In numpy, this should take maybe 50-100ms.

Let me try this vectorized approach:

```python
def local_search_vectorized(h, n, time_limit=200):
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    
    while time.time() - start < time_limit:
        if _BUDGET - (time.time() - _T0) < 10:
            break
        
        corr = compute_corr(h)
        c_max = float(np.max(corr))
        
        ones_idx = np.where(h >= 0.5)[0]
        zeros_idx = np.where(h <= 0.5)[0]
        
        if len(ones_idx) < 1 or len(zeros_idx) < 1:
            break
        
        dcorr = precompute_dcorr(h, n)
        
        # Vectorized: for all pairs (i,j), compute new max
        # new_corr[i,j,k] = corr[k] - dcorr[i,k] + dcorr[j,k]
        # We want min over (i,j) of max over k
        dc_i = dcorr[ones_idx]  # (n1, nfull)
        dc_j = dcorr[zeros_idx]  # (n0, nfull)
        
        # new_maxs[i,j] = max_k (corr[k] - dc_i[i,k] + dc_j[j,k])
        # = max_k (corr[k] - dc_i[i,k]) + dc_j[j,k] ... NO, max of sum ≠ sum of maxs
        
        # We need: new_maxs[i,j] = max_k (corr[k] - dc_i[i,k] + dc_j[j,k])
        # For fixed i: base[i,k] = corr[k] - dc_i[i,k]
        # new_maxs[i,j] = max_k (base[i,k] + dc_j[j,k])
        
        # This is a "max-plus" type operation. For each pair (i,j), we need
        # max over k of (base[i,k] + dc_j[j,k]).
        # Vectorized: reshape to (n1, nfull, 1) + (1, nfull, n0) → (n1, nfull, n0) → max over axis 1
        # But that's n1 * nfull * n0 = 256 * 1023 * 256 = 67M elements. 512MB. Too much.
        
        # Alternative: loop over i, vectorize over j
        best_improve = 0
        best_ii, best_jj = -1, -1
        
        for ii in range(len(ones_idx)):
            base = corr - dc_i[ii]  # (nfull,)
            new_maxs = np.max(base[None, :] + dc_j, axis=1)  # (n0,)
            local_best = int(np.argmin(new_maxs))
            local_improve = c_max - new_maxs[local_best]
            if local_improve > best_improve:
                best_improve = local_improve
                best_ii = ii
                best_jj = local_best
        
        if best_improve < 1e-10:
            break
        
        i = ones_idx[best_ii]
        j = zeros_idx[best_jj]
        h[i] = 0.0
        h[j] = 1.0
        best_c5 = compute_c5(h)
    
    return h, best_c5
```

The inner loop is 256 iterations, each doing a (256, 1023) → (256,) max reduction. That's 256*256*1023 ≈ 67M ops per pass, maybe 50-100ms in numpy. So ~10-20 passes per second. In 200s, that's 2000-4000 passes. Should be plenty for convergence.

Now, for 2-swap moves: after reaching a local optimum with 1-swaps, try 2-swaps. A 2-swap flips two 1s to 0s and two 0s to 1s. The new correlation is:
new_corr = corr - dcorr[i1] - dcorr[i2] + dcorr[j1] + dcorr[j2]

For efficiency, we can:
1. Do the first swap (best 1-swap)
2. Then do the second swap on the updated state
This is O(n₁ * n₀ * 2n) * 2 per 2-swap, same as 2 * 1-swap cost.

Or we can directly search: for all pairs (i1,i2) of ones and (j1,j2) of zeros, compute new max. This is O(n₁² * n₀² * 2n) which is way too expensive for n=512.

Better: sequential 2-swaps. First find best 1-swap, apply it, then find best 1-swap on the result. If the combined improvement is positive, accept.

Actually, for escaping local optima, a better approach is:
1. Find best 1-swap (improvement might be 0 at local optimum)
2. Find second-best 1-swap
3. Try both, see if combined improves
4. If not, try random 2-swaps

Or use simulated annealing: occasionally accept a worsening move.

Let me design the full algorithm:

```
Phase 1: Binarize parent (n=512), run best-improvement local search
Phase 2: Simulated annealing with 1-swaps and 2-swaps
Phase 3: Interpolate to n=1024, binarize, local search
Phase 4: Come back to n=512, binarize, local search  
Phase 5: Try n=256, local search
Phase 6: SLSQP continuous polish
Phase 7: Final local search
```

Let me also add a "kick" move: randomly flip k bits (k=2,4,6) and repair, then local search. This is like a perturbation in iterated local search.

```python
import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def binarize(h, n):
    idx = np.argsort(-h)[:n // 2]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        at_low = h <= 1e-12
        at_high = h >= 1.0 - 1e-12
        mid = ~(at_low | at_high)
        cnt = int(np.count_nonzero(mid))
        if cnt > 0:
            h[mid] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def precompute_dcorr(h, n):
    nfull = 2 * n - 1
    k_range = np.arange(nfull)
    dcorr = np.zeros((n, nfull))
    for i in range(n):
        j1 = i + k_range - (n - 1)
        j2 = i - k_range + (n - 1)
        v1 = (j1 >= 0) & (j1 < n)
        v2 = (j2 >= 0) & (j2 < n)
        dcorr[i, v1] += (1.0 - h[j1[v1]])
        dcorr[i, v2] -= h[j2[v2]]
    return dcorr

def local_search(h, n, time_limit=200, rng=None):
    """Best-improvement 1-swap local search for binary vectors."""
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    
    while time.time() - start < time_limit:
        if _BUDGET - (time.time() - _T0) < 10:
            break
        
        corr = compute_corr(h)
        c_max = float(np.max(corr))
        
        ones_idx = np.where(h >= 0.5)[0]
        zeros_idx = np.where(h <= 0.5)[0]
        
        if len(ones_idx) < 1 or len(zeros_idx) < 1:
            break
        
        dcorr = precompute_dcorr(h, n)
        dc_i = dcorr[ones_idx]
        dc_j = dcorr[zeros_idx]
        
        best_improve = 0.0
        best_ii, best_jj = -1, -1
        
        for ii in range(len(ones_idx)):
            base = corr - dc_i[ii]
            new_maxs = np.max(base[None, :] + dc_j, axis=1)
            local_best = int(np.argmin(new_maxs))
            local_improve = c_max - new_maxs[local_best]
            if local_improve > best_improve:
                best_improve = local_improve
                best_ii = ii
                best_jj = local_best
        
        if best_improve < 1e-10:
            break
        
        i = ones_idx[best_ii]
        j = zeros_idx[best_jj]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        if new_c5 < best_c5 - 1e-14:
            best_c5 = new_c5
        else:
            h[i] = 1.0
            h[j] = 0.0
            break
    
    return h, best_c5

def simulated_annealing(h, n, time_limit=100, rng=None):
    """SA with 1-swap and 2-swap moves."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    
    T_max = 0.05  # Initial temperature in C5 units
    T_min = 0.001
    n_total = max(500, int(time_limit * 5))
    
    for it in range(n_total):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 10:
            break
        
        progress = it / max(n_total, 1)
        T = T_max * (T_min / T_max) ** progress
        
        # Choose move type
        if rng.random() < 0.7:
            # 1-swap
            ones_idx = np.where(h >= 0.5)[0]
            zeros_idx = np.where(h <= 0.5)[0]
            i = ones_idx[rng.randint(len(ones_idx))]
            j = zeros_idx[rng.randint(len(zeros_idx))]
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = compute_c5(h)
            delta = new_c5 - best_c5
            if new_c5 < best_c5 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                h[i] = 1.0
                h[j] = 0.0
        else:
            # 2-swap
            ones_idx = np.where(h >= 0.5)[0]
            zeros_idx = np.where(h <= 0.5)[0]
            if len(ones_idx) < 2 or len(zeros_idx) < 2:
                continue
            i1, i2 = rng.choice(ones_idx, 2, replace=False)
            j1, j2 = rng.choice(zeros_idx, 2, replace=False)
            h[i1] = 0.0; h[i2] = 0.0
            h[j1] = 1.0; h[j2] = 1.0
            new_c5 = compute_c5(h)
            delta = new_c5 - best_c5
            if new_c5 < best_c5 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                h[i1] = 1.0; h[i2] = 1.0
                h[j1] = 0.0; h[j2] = 0.0
    
    return best_h, best_c5

def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    x_new = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(x_new, xp, src_h)

def smooth_max_value(x, beta, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50.0, 50.0)
    sm = m + np.log(np.sum(np.exp(shifted))) / beta
    return sm * (2.0 / n)

def smooth_max_grad(x, beta, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50.0, 50.0)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = w / w_sum
    
    g = np.zeros(n)
    dx = 2.0 / n
    nfull = 2 * n - 1
    for k in range(nfull):
        wk = w[k] * dx
        if wk < 1e-16:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            g[offset:offset + npairs] += wk * (1.0 - x[:npairs])
            g[:npairs] -= wk * x[offset:offset + npairs]
        else:
            a = -offset
            npairs = n - a
            g[:npairs] += wk * (1.0 - x[a:a + npairs])
            g[a:a + npairs] -= wk * x[:npairs]
    return g

def slsqp_optimize(h, n, beta, maxiter=500):
    try:
        res = minimize(
            lambda x: smooth_max_value(x, beta, n),
            h.copy(),
            jac=lambda x: smooth_max_grad(x, beta, n),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                         'jac': lambda x: np.ones(n)}],
            options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0}
        )
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def iterated_local_search(h, n, time_limit=200, rng=None):
    """ILS: local search + perturbation."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    
    while time.time() - start < time_limit:
        if _BUDGET - (time.time() - _T0) < 10:
            break
        
        # Local search
        h, c5 = local_search(h, n, time_limit=min(30.0, time_limit - (time.time() - start)))
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Perturbation: flip k random bits
        k = rng.randint(2, 7)
        ones_idx = np.where(h >= 0.5)[0]
        zeros_idx = np.where(h <= 0.5)[0]
        if len(ones_idx) < k // 2 or len(zeros_idx) < k // 2:
            break
        
        n_flip_1 = k // 2
        n_flip_0 = k - n_flip_1
        
        flip_1s = rng.choice(ones_idx, min(n_flip_1, len(ones_idx)), replace=False)
        flip_0s = rng.choice(zeros_idx, min(n_flip_0, len(zeros_idx)), replace=False)
        
        for idx in flip_1s:
            h[idx] = 0.0
        for idx in flip_0s:
            h[idx] = 1.0
        
        # Repair: ensure exactly n/2 ones
        current_ones = int(np.sum(h >= 0.5))
        target_ones = n // 2
        if current_ones > target_ones:
            extra = current_ones - target_ones
            ones_now = np.where(h >= 0.5)[0]
            to_flip = rng.choice(ones_now, extra, replace=False)
            h[to_flip] = 0.0
        elif current_ones < target_ones:
            deficit = target_ones - current_ones
            zeros_now = np.where(h <= 0.5)[0]
            to_flip = rng.choice(zeros_now, deficit, replace=False)
            h[to_flip] = 1.0
    
    return best_h, best_c5

    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    
    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)
    
    # Phase 0: Initialize
    try:
        parent_h = np.array(initial_h_values, dtype=float)
