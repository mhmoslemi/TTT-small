    h = h.copy()
    dx = 2.0/n
    best_cost = np.correlate(h, 1-h, mode='full').max() * dx
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    start = time.time()
    improved = True
    while improved and time.time() - start < time_limit:
        improved = False
        best_delta = 0
        best_swap = None
        corr = np.correlate(h, 1-h, mode='full')
        current_cost = corr.max() * dx
        
        for i in ones:
            for j in zeros:
                h[i] = 0
                h[j] = 1
                new_corr = np.correlate(h, 1-h, mode='full')
                new_cost = new_corr.max() * dx
                delta = new_cost - current_cost
                if delta < best_delta - 1e-15:

                    best_swap = (i, j)
                h[i] = 1
                h[j] = 0
        
        if best_swap is not None:
            i, j = best_swap
            h[i] = 0
            h[j] = 1

            best_cost = np.correlate(h, 1-h, mode='full').max() * dx

def eval_swap_delta(h, n, i, j):
    """Compute the change in correlation at each lag when swapping h[i]:1->0, h[j]:0->1."""
    # i is currently 1, j is currently 0
    delta = np.zeros(2*n - 1)  # indices 0 to 2n-2, offset = idx - (n-1)
    
    # We need to compute Δ_k for k = -(n-1) to n-1
    # delta[idx] corresponds to k = idx - (n-1)
    
    for k in range(-(n-1), n):
        d = 0.0
        # Term 1: m = i, need 0 <= i < n and 0 <= i+k < n
        if 0 <= i + k < n:
            d -= (1.0 - h[i+k])
        # Term 2: m = j, need 0 <= j < n and 0 <= j+k < n
        if 0 <= j + k < n:
            if j + k == i:
                d += 1.0
            else:
                d += (1.0 - h[j+k])
        # Term 3: m = i-k, need 0 <= i-k < n and 0 <= i < n (i is always valid)
        if 0 <= i - k < n:
            if i - k == j:
                d += 1.0
            else:
                d += h[i-k]
        # Term 4: m = j-k, need 0 <= j-k < n and 0 <= j < n (j is always valid)
        if 0 <= j - k < n:
            if j - k == i:
                d += 0.0
            else:
                d -= h[j-k]
        
        delta[k + (n-1)] = d
    
    return delta

Δ_k is non-zero only when at least one of the four terms is active:
- Term 1 active when: 0 ≤ i+k < n → -i ≤ k < n-i
- Term 2 active when: 0 ≤ j+k < n → -j ≤ k < n-j
- Term 3 active when: 0 ≤ i-k < n → i-n < k ≤ i
- Term 4 active when: 0 ≤ j-k < n → j-n < k ≤ j

So Δ_k is potentially non-zero for k in [-i, n-i-1] ∪ [-j, n-j-1] ∪ [i-n+1, i] ∪ [j-n+1, j].

The union of these is at most 4 intervals, each of length at most n. In the worst case, they cover all k from -(n-1) to n-1. So in the worst case, I still need to check O(n) lags.

But for each of these O(n) lags, the computation is O(1). So total is O(n) per swap. That's the 500x speedup I mentioned.

    """Compute Δ_k only for lags where it might be non-zero."""
    # Lags where term 1 is active: k in [-i, n-1-i]
    # Lags where term 2 is active: k in [-j, n-1-j]
    # Lags where term 3 is active: k in [i-n+1, i] = [-(n-1-i), i]
    # Lags where term 4 is active: k in [j-n+1, j] = [-(n-1-j), j]
    
    # Collect all relevant k values
    k_set = set()
    for k in range(-i, n-i):
        k_set.add(k)
    for k in range(-j, n-j):
        k_set.add(k)
    for k in range(-(n-1-i), i+1):
        k_set.add(k)
    for k in range(-(n-1-j), j+1):
        k_set.add(k)

            d -= (1.0 - h[i+k])
