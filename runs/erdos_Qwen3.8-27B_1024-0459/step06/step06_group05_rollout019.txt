    m = np.max(c)
    sh = np.clip(beta * (c - m), -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / ex.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for i in np.where(w > 1e-10)[0]:
        g += w[i] * dx * grad_corr_k(x, i, n)
    return g


def _slsqp_cfun(x):
    return np.sum(x) - _GNPOINTS / 2.0


def _slsqp_cjac(x):
    return np.ones(_GNPOINTS)


def slsqp_optimize(h, n, beta, maxiter=400):
    global _GNPOINTS, _GBETA
    _GNPOINTS = n
    _GBETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cfun, 'jac': _slsqp_cjac}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            return h_new
    except Exception:
        pass
    return h


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def topk_slsqp(h, n, K, beta, maxiter=400):
    """SLSQP using top-K smooth max (more efficient than full smooth max)."""
    global _GNPOINTS, _GBETA
    _GNPOINTS = n
    _GBETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cfun, 'jac': _slsqp_cjac}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            return h_new
    except Exception:
        pass
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

    # Get initial solution
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 512
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    # Phase 1: Projected gradient at parent resolution
    n = best_n
    h = best_h.copy()
    if remaining() > 30:
        t1 = min(120.0, remaining() * 0.2)
        h_pg, c5_pg = proj_grad_optimize(h, n, t1, K=8, base_step=0.1)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
            h = h_pg.copy()

    # Phase 2: SLSQP at parent resolution with various parameters
    for beta in [500, 1500, 5000, 15000, 50000]:
        if remaining() < 10:
            break
        h_new = slsqp_optimize(h, n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 3: Interpolate to n=1024 and optimize
    n2 = 1024
    if remaining() > 80:
        h2 = interp_to(best_h, best_n, n2)
        h2 = project_feasible(h2, n2)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5 + 0.005:
            # Projected gradient at n=1024
            t3 = min(200.0, remaining() * 0.3)
            h2, c5_2 = proj_grad_optimize(h2, n2, t3, K=10, base_step=0.06)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()
                best_n = n2

            # SLSQP at n=1024
            for beta in [1000, 5000, 20000, 80000]:
                if remaining() < 15:
                    break
                h2_new = slsqp_optimize(h2, n2, beta, maxiter=350)
                c5_2_new = compute_c5(h2_new)
                if c5_2_new < best_c5 - 1e-14:
                    best_c5 = c5_2_new
                    best_h = h2_new.copy()
                    best_n = n2
                    h2 = h2_new.copy()

    # Phase 4: Try n=2048 if time permits
    n3 = 2048
    if remaining() > 100:
        h3 = interp_to(best_h, best_n, n3)
        h3 = project_feasible(h3, n3)
        c5_3 = compute_c5(h3)
        if c5_3 < best_c5 + 0.003:
            t4 = min(180.0, remaining() * 0.3)
            h3, c5_3 = proj_grad_optimize(h3, n3, t4, K=12, base_step=0.05)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h3.copy()
                best_n = n3

            # SLSQP at n=2048 (fewer iterations due to higher dim)
            for beta in [5000, 20000]:
                if remaining() < 20:
                    break
                h3_new = slsqp_optimize(h3, n3, beta, maxiter=250)
                c5_3_new = compute_c5(h3_new)
                if c5_3_new < best_c5 - 1e-14:
                    best_c5 = c5_3_new
                    best_h = h3_new.copy()
                    best_n = n3
                    h3 = h3_new.copy()

    # Phase 5: Multi-start perturbation at best resolution
    n = best_n
    h = best_h.copy()
    if remaining() > 60:
        for pert_i in range(5):
            if remaining() < 25:
                break
            sigma = 0.01 + 0.008 * pert_i
            h_pert = h + rng.randn(n) * sigma
            h_pert = project_feasible(h_pert, n)
            c5_pert = compute_c5(h_pert)
            if c5_pert > best_c5 + 0.02:
                continue
            # Quick optimize from perturbation
            h_pert, c5_pert = proj_grad_optimize(h_pert, n, 20.0, K=8, base_step=0.05)
            if remaining() > 10:
                h_pert = slsqp_optimize(h_pert, n, 5000, maxiter=200)
                c5_pert = compute_c5(h_pert)
            if c5_pert < best_c5 - 1e-14:
                best_c5 = c5_pert
                best_h = h_pert.copy()
                h = h_pert.copy()

    # Phase 6: Intensive final SLSQP at best resolution
    n = best_n
    h = best_h.copy()
    for beta in [3000, 10000, 40000, 120000]:
        if remaining() < 8:
            break
        h_new = slsqp_optimize(h, n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 7: Very fine projected gradient (last tweaks)
    if remaining() > 15:
        h_fine, c5_fine = proj_grad_optimize(best_h, best_n,
                                              min(30.0, remaining() * 0.4),
                                              K=5, base_step=0.02)
        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_fine.copy()

    # Phase 8: Final SLSQP polish
    if remaining() > 5:
        h_final = slsqp_optimize(best_h, best_n, 100000, maxiter=300)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
