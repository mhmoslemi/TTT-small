def slsqp_smooth_max(h, n, K, beta, maxiter=300):
    corr_full = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n

    obj = corr_full.copy()
    m = np.max(obj)
    shifted = beta * (obj - m)
    clipped = np.clip(shifted, -40.0, 40.0)
    ex = np.exp(clipped)
    s = np.sum(ex)
    w = ex / s

    def obj_func(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        mm = np.max(c)
        sh = beta * (c - mm)
        cl = np.clip(sh, -40.0, 40.0)
        ee = np.exp(cl)
        return (mm + np.log(np.sum(ee)) / beta) * dx

    def grad_func(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        mm = np.max(c)
        sh = beta * (c - mm)
        cl = np.clip(sh, -40.0, 40.0)
        ee = np.exp(cl)
        ss = np.sum(ee)
        ww = ee / ss
        g = np.zeros(n)
        for k in range(len(c)):
            wk = ww[k]
            if wk > 1e-12:
                g += (wk * dx) * grad_corr_k(x, k, n)
        return g

    def con_fun(x):
        return np.sum(x) - n / 2.0

    def con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]

    try:
        res = minimize(obj_func, h, jac=grad_func, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            return h_new
    except Exception:
        pass
    return h.copy()


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def projected_gradient(h, n, K, beta, step, max_backtrack=10):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    shifted = beta * (corr - m)
    clipped = np.clip(shifted, -40.0, 40.0)
    ex = np.exp(clipped)
    s = np.sum(ex)
    w = ex / s

    g = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk > 1e-12:
            g += (wk * dx) * grad_corr_k(h, k, n)

    current_obj = compute_c5(h)
    lr = step
    for bt in range(max_backtrack):
        h_try = h - lr * g
        h_try = project_feasible(h_try, n)
        obj_try = compute_c5(h_try)
        if obj_try < current_obj:
            return h_try, obj_try
        lr *= 0.5
    return h, current_obj


def topk_slsqp(h, n, K, beta, maxiter=200):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    idx_top = np.argsort(-corr)[:K]
    d = 2.0 / n

    def obj_func(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        vals = c[idx_top]
        mm = np.max(vals)
        sh = beta * (vals - mm)
        cl = np.clip(sh, -40.0, 40.0)
        ee = np.exp(cl)
        return (mm + np.log(np.sum(ee)) / beta) * d

    def grad_func(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        vals = c[idx_top]
        mm = np.max(vals)
        sh = beta * (vals - mm)
        cl = np.clip(sh, -40.0, 40.0)
        ee = np.exp(cl)
        ss = np.sum(ee)
        ww = ee / ss
        g = np.zeros(n)
        for i, k in enumerate(idx_top):
            if ww[i] > 1e-12:
                g += (ww[i] * d) * grad_corr_k(x, k, n)
        return g

    def con_fun(x):
        return np.sum(x) - n / 2.0

    def con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]

    try:
        res = minimize(obj_func, h, jac=grad_func, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()


def structured_perturb(h, n, rng, scale=0.02):
    h_p = h.copy()
    x = np.arange(n) * 2.0 / n
    freq = rng.randint(1, 8)
    phase = rng.uniform(0, 2 * np.pi)
    h_p += scale * np.sin(2 * np.pi * freq * x + phase)
    h_p += scale * 0.3 * rng.randn(n)
    h_p += scale * 0.5 * rng.rand(n) * (rng.rand(n) - 0.5)
    return project_feasible(h_p, n)


def coordinate_polish(h, n, max_sweeps=5, time_limit=30.0, delta=0.01):
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    target = n / 2.0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        improved = False
        for i in range(n):
            if time.time() - start > time_limit:
                break
            old_val = h[i]
            for d in [delta, -delta, 2*delta, -2*delta, 0.005, -0.005, 0.0, 1.0]:
                c = old_val + d
                if c < 0.0 or c > 1.0:
                    continue
                h[i] = c
                diff = target - np.sum(h)
                if abs(diff) < 1e-12:
                    ccur = compute_c5(h)
                    if ccur < best_c5 - 1e-14:
                        return h, ccur
                h[i] = old_val
        if not improved:
            break
    return h, compute_c5(h)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)

    # Initialize from parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        parent_h = np.zeros(768)
        parent_h[:384] = 1.0

    n0 = len(parent_h)
    best_h = parent_h.copy()
    best_c5 = compute_c5(best_h)
    best_n = n0

    # Phase 1: Quick SLSQP polish at parent resolution
    for K in [10, 20]:
        for beta in [500, 2000, 10000]:
            if remaining() < 15:
                break
            h_try = topk_slsqp(best_h, n0, K, beta, maxiter=200)
            c_try = compute_c5(h_try)
            if c_try < best_c5:
                best_c5 = c_try
                best_h = h_try.copy()

    print(f"After Phase 1: C5={best_c5:.6f}, n={best_n}")

    # Phase 2: Upsample to higher resolution and optimize
    target_res = 2048
    if remaining() > 200:
        h_hi = interp_to(best_h, best_n, target_res)
        h_hi = project_feasible(h_hi, target_res)
        c_hi = compute_c5(h_hi)
        print(f"Interpolated to n={target_res}: C5={c_hi:.6f}")

        # Projected gradient descent at high resolution
        step = 0.05
        h_cur = h_hi.copy()
        n_cur = target_res
        stagnation = 0
        max_stag = 200

        for it in range(50000):
            if time.time() - _T0 > _BUDGET - 300:
                break
            K_cur = 20
            beta_cur = 50.0

            h_new, obj_new = projected_gradient(h_cur, n_cur, K_cur, beta_cur, step)
            if obj_new < compute_c5(h_cur) - 1e-14:
                h_cur = h_new
                stagnation = 0
                if obj_new < best_c5:
                    best_c5 = obj_new
                    best_h = h_cur.copy()
                    best_n = n_cur
                step = min(step * 1.05, 0.1)
            else:
                stagnation += 1
                step *= 0.9

            if stagnation >= max_stag:
                # Try SLSQP polish
                if remaining() > 60:
                    h_pol = topk_slsqp(h_cur, min(n_cur, 1024), 15, 2000, maxiter=200)
                    c_pol = compute_c5(h_pol)
                    if c_pol < best_c5:
                        best_c5 = c_pol
                        best_h = h_pol.copy()
                        best_n = n_pol if 'n_pol' in dir() else n_cur
                    # Upsample back and continue
                    if n_cur > 1024:
                        h_cur = interp_to(h_pol, 1024, n_cur)
                        h_cur = project_feasible(h_cur, n_cur)
                    else:
                        h_cur = h_pol
                    step = 0.05
                    stagnation = 0
                else:
                    break

            # Adaptive beta increase
            if it % 1000 == 0:
                beta_cur = 50.0 * (1 + it / 2000.0)

        print(f"After Phase 2 (PGD n={target_res}): C5={best_c5:.6f}")

    # Phase 3: Try different resolutions and structures
    for n_try in [1024, 1536, 2048, 3072]:
        if remaining() < 80:
            break
        if n_try % 2 != 0:
            continue

        h_t = interp_to(best_h, best_n, n_try)
        h_t = project_feasible(h_t, n_try)

        # SLSQP at this resolution (limit to reasonable n)
        if n_try <= 1536:
            for K in [10, 20]:
                for beta in [1000, 5000, 20000]:
                    if remaining() < 20:
                        break
                    h_s = topk_slsqp(h_t, n_try, K, beta, maxiter=150)
                    c_s = compute_c5(h_s)
                    if c_s < best_c5:
                        best_c5 = c_s
                        best_h = h_s.copy()
                        best_n = n_try
                        h_t = h_s.copy()

        # Projected gradient
        if remaining() > 30:
            step = 0.04
            h_cur = h_t.copy()
            for it in range(5000):
                if remaining() < 30:
                    break
                h_new, obj_new = projected_gradient(h_cur, n_try, 15, 50.0, step)
                if obj_new < compute_c5(h_cur) - 1e-14:
                    h_cur = h_new
                    if obj_new < best_c5:
                        best_c5 = obj_new
                        best_h = h_cur.copy()
                        best_n = n_try
                else:
                    step *= 0.9

    print(f"After Phase 3: C5={best_c5:.6f}, n={best_n}")

    # Phase 4: Random restarts with different structures
    for restart in range(5):
        if remaining() < 60:
            break
        h_r = structured_perturb(best_h, best_n, rng, scale=0.02 + restart * 0.01)
        n_r = min(best_n, 1024)
        h_r = interp_to(h_r, len(h_r), n_r)
        h_r = project_feasible(h_r, n_r)

        for K in [15, 30]:
            for beta in [2000, 10000]:
                if remaining() < 20:
                    break
                h_s = topk_slsqp(h_r, n_r, K, beta, maxiter=150)
                c_s = compute_c5(h_s)
                if c_s < best_c5:
                    best_c5 = c_s
                    best_h = h_s.copy()
                    best_n = n_r

    print(f"After Phase 4: C5={best_c5:.6f}")

    # Phase 5: High-resolution projected gradient on best solution
    n_wrk = max(best_n, 2048)
    h_wrk = interp_to(best_h, best_n, n_wrk)
    h_wrk = project_feasible(h_wrk, n_wrk)
    c_wrk = compute_c5(h_wrk)

    if c_wrk < best_c5:
        best_c5 = c_wrk
        best_h = h_wrk.copy()
        best_n = n_wrk

    # Long projected gradient run
    step = 0.03
    h_cur = h_wrk.copy()
    for it in range(100000):
        if remaining() < 60:
            break
        K_cur = 25 if it < 20000 else 15
        beta_cur = 80.0 + 20.0 * (it / 10000.0)
        h_new, obj_new = projected_gradient(h_cur, n_wrk, K_cur, beta_cur, step)
        prev_obj = compute_c5(h_cur)
        if obj_new < prev_obj - 1e-14:
            h_cur = h_new
            if obj_new < best_c5:
                best_c5 = obj_new
                best_h = h_cur.copy()
                best_n = n_wrk
            step = min(step * 1.03, 0.08)
        else:
            step *= 0.92
            if step < 0.001:
                break

    print(f"After Phase 5 (PGD n={n_wrk}): C5={best_c5:.6f}")

    # Phase 6: Final SLSQP polish at moderate resolution
    n_pol_final = min(best_n, 1536)
    h_pol = interp_to(best_h, best_n, n_pol_final)
    h_pol = project_feasible(h_pol, n_pol_final)

    for K in [10, 20, 40]:
        for beta in [5000, 20000, 100000]:
            if remaining() < 15:
                break
            h_s = topk_slsqp(h_pol, n_pol_final, K, beta, maxiter=300)
            c_s = compute_c5(h_s)
            if c_s < best_c5:
                best_c5 = c_s
                best_h = h_s.copy()
                best_n = n_pol_final
                h_pol = h_s.copy()

    print(f"After Phase 6: C5={best_c5:.6f}")

    # Phase 7: Final projected gradient at best_n
    if remaining() > 30:
        h_cur = project_feasible(best_h, best_n)
        step = 0.02
        for it in range(50000):
            if remaining() < 30:
                break
            h_new, obj_new = projected_gradient(h_cur, best_n, 20, 100.0, step)
            prev_obj = compute_c5(h_cur)
            if obj_new < prev_obj - 1e-14:
                h_cur = h_new
                if obj_new < best_c5:
                    best_c5 = obj_new
                    best_h = h_cur.copy()
                step = min(step * 1.02, 0.06)
            else:
                step *= 0.9
                if step < 0.001:
                    break

    print(f"After Phase 7: C5={best_c5:.6f}")

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    print(f"FINAL: C5={best_c5:.8f}, n={best_n}")
    return (best_h, best_c5, best_n)
