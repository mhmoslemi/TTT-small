def two_coord_minimax(h, n, i, j, corr):
    """Find optimal delta to move from j to i using 1D minimax.
    Returns (delta, new_corr) or (0, corr) if no improvement."""

    # For lag d (0 to 2n-2), offset = d - (n-1)
    # If offset >= 0: corr[d] = sum_{p=0}^{n-1-offset} h[p]*(1-h[p+offset])
    #   coeff of h[m]: (1-h[m+offset]) if m+offset<n and m <= n-1-offset
    #                   -h[m-offset] if m-offset >= 0
    # If offset < 0, let a = -offset: corr[d] = sum_{p=a}^{n-1} h[p]*(1-h[p-a])
    #   coeff of h[m]: (1-h[m-a]) if m-a >= 0 and m >= a
    #                   -h[m+a] if m+a < n
    
    # B[d] = c_i(d) - c_j(d)
    B = np.zeros(2*n - 1)
    for d in range(2*n - 1):
        off = d - (n - 1)
        ci = 0.0
        cj = 0.0
        if off >= 0:
            if i + off < n and i <= n - 1 - off:
                ci += (1.0 - h[i + off])
            if i - off >= 0:
                ci -= h[i - off]
            if j + off < n and j <= n - 1 - off:
                cj += (1.0 - h[j + off])
            if j - off >= 0:
                cj -= h[j - off]
        else:
            a = -off
            if i >= a:
                ci += (1.0 - h[i - a])
            if i + a < n:
                ci -= h[i + a]
            if j >= a:
                cj += (1.0 - h[j - a])
            if j + a < n:
                cj -= h[j + a]
        B[d] = ci - cj
    
    # Feasible range for delta
    L = max(-h[i], h[j] - 1.0)
    U = min(1.0 - h[i], h[j])
    if U - L < 1e-15:
        return 0.0, corr
    
    # Golden section search on [L, U]
    def obj(delta):
        return np.max(corr + delta * B)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = L, U
    c_ = b - gr * (b - a)
    d_ = a + gr * (b - a)
    fc = obj(c_)
    fd = obj(d_)
    
    for _ in range(60):
        if b - a < 1e-10:
            break
        if fc < fd:
            b = d_
            d_ = c_
            fd = fc
            c_ = b - gr * (b - a)
            fc = obj(c_)
        else:
            a = c_
            c_ = d_
            fc = fd
            d_ = a + gr * (b - a)
            fd = obj(d_)
    
    delta = (a + b) / 2.0
    new_corr = corr + delta * B
    if np.max(new_corr) < np.max(corr) - 1e-15:
        return delta, new_corr
    return 0.0, corr

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0

def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def project_simplex_box(v, target_sum):
    """Project v onto {h: 0<=h[i]<=1, sum(h)=target_sum} via bisection on lambda."""
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        h = np.clip(v + mid, 0.0, 1.0)
        s = np.sum(h)
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def smooth_max_obj_and_grad(h, n, beta):
    """Compute smoothed max objective and gradient."""
    dx = 2.0 / n
    corr = compute_corr(h)
    m = float(np.max(corr))
    z = np.clip(beta * (corr - m), -80.0, 80.0)
    ez = np.exp(z)
    S = np.sum(ez)
    obj = (m + np.log(S) / beta) * dx
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-16:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - h[a:])
            g[a:] -= wk * h[:np_]
    return obj, g

def nesterov_pg(h, n, betas, max_iter_per_beta=2000, time_limit=600):
    """Nesterov accelerated projected gradient with multi-beta annealing."""
    start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h = h.copy()
    n_half = n / 2.0

    for beta in betas:
        if remaining_time() < 10 or time.time() - start > time_limit:
            break
        y = h.copy()
        t_k = 1.0
        alpha = 1.0 / (beta * 0.5)
        alpha = min(alpha, 0.5)

        for it in range(max_iter_per_beta):
            if remaining_time() < 5 or (time.time() - start > time_limit):
                break
            # Nesterov extrapolation
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            theta = (1.0 - t_k) / t_new
            y = h + theta * (h - y)
            y = np.clip(y, 1e-10, 1.0 - 1e-10)

            obj, g = smooth_max_obj_and_grad(y, n, beta)
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-16:
                break

            # Projected gradient step
            v = y - alpha * g
            h_new = project_simplex_box(v, n_half)

            # Check true objective
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()

            # Adaptive step size
            if c5_new <= obj:
                alpha = min(alpha * 1.3, 2.0)
            else:
                alpha *= 0.7

            h = h_new
            t_k = t_new

            # Converged check
            if it > 50 and abs(c5_new - best_c5) < 1e-13 and np.linalg.norm(h_new - best_h) < 1e-10:
                break

    return best_h, best_c5

def two_coord_greedy(h, n, max_swaps=5000, time_limit=120):
    """Greedy 2-coordinate swap optimization."""
    start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h = h.copy()
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    dx = 2.0 / n
    rng = np.random.RandomState(123)

    for it in range(max_swaps):
        if remaining_time() < 5 or time.time() - start > time_limit:
            break

        # Pick indices toward and away from the max correlation
        # Find the active lag
        k_act = int(np.argmax(corr))
        off = k_act - (n - 1)

        # Compute gradient for the active lag
        g_act = np.zeros(n)
        if off >= 0:
            np_ = n - off
            g_act[off:] += (1.0 - h[:np_])
            g_act[:np_] -= h[off:]
        else:
            a = -off
            np_ = n - a
            g_act[:np_] += (1.0 - h[a:])
            g_act[a:] -= h[:np_]

        # We want to decrease corr[k_act]: move mass from high-g to low-g positions
        # But we need to check all lags, so use the full gradient
        # For efficiency, use random pairs biased by gradient
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        while i == j:
            j = int(rng.randint(n))

        # Compute B[d] = c_i(d) - c_j(d)
        B = np.zeros(2 * n - 1)
        for d in range(2 * n - 1):
            off_d = d - (n - 1)
            ci = 0.0
            cj = 0.0
            if off_d >= 0:
                if i + off_d < n:
                    ci += (1.0 - h[min(i + off_d, n - 1)])
                if i - off_d >= 0:
                    ci -= h[i - off_d]
                if j + off_d < n:
                    cj += (1.0 - h[min(j + off_d, n - 1)])
                if j - off_d >= 0:
                    cj -= h[j - off_d]
            else:
                a_d = -off_d
                if i >= a_d:
                    ci += (1.0 - h[i - a_d])
                if i + a_d < n:
                    ci -= h[i + a_d]
                if j >= a_d:
                    cj += (1.0 - h[j - a_d])
                if j + a_d < n:
                    cj -= h[j + a_d]
            B[d] = ci - cj

        # Find best delta via golden section
        L = max(-h[i], h[j] - 1.0)
        U = min(1.0 - h[i], h[j])
        if U - L < 1e-14:
            continue

        # Try a few candidate deltas
        best_delta = 0.0
        best_val = c_max
        candidates = [0.0, (U - L) * 0.1, (U - L) * 0.3, (U - L) * 0.5,
                      (U - L) * 0.7, (U - L) * 0.9,
                      -0.3 * (U - L), 0.3 * (U - L)]
        # Also try the steepest descent direction
        slope = np.dot(B, np.ones(2 * n - 1)) / (2 * n - 1)
        if slope < 0:
            candidates.append(min(U, L + 0.5 * (U - L)))
        else:
            candidates.append(max(L, L + 0.5 * (U - L)))

        for delta in candidates:
            if delta < L or delta > U:
                continue
            new_max = np.max(corr + delta * B)
            if new_max < best_val - 1e-14:
                best_val = new_max
                best_delta = delta

        if best_delta == 0.0:
            continue

        # Apply the swap
        h[i] += best_delta
        h[j] -= best_delta
        h = np.clip(h, 0.0, 1.0)
        corr = corr + best_delta * B
        c_max = float(np.max(corr))

        if c_max * dx < best_c5 - 1e-14:
            best_c5 = c_max * dx
            best_h = h.copy()

    return best_h, best_c5

def subgradient_zigzag(h, n, max_iters=3000, time_limit=60):
    """Subgradient zigzag method."""
    start = time.time()
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h = h.copy()
    n_half = n / 2.0
    k_prev = None
    stale = 0

    for it in range(max_iters):
        if remaining_time() < 5 or time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        k_curr = int(np.argmax(corr))

        if k_prev is not None and k_curr != k_prev:
            def grad(h, k, n):
                g = np.zeros(n)
                off = k - (n - 1)
                if off >= 0:
                    np_ = n - off
                    g[off:] += (1.0 - h[:np_])
                    g[:np_] -= h[off:]
                else:
                    a = -off
                    np_ = n - a
                    g[:np_] += (1.0 - h[a:])
                    g[a:] -= h[:np_]
                return g
            g = 0.5 * (grad(h, k_curr, n) + grad(h, k_prev, n))
            k_prev = k_curr
        else:
            g = np.zeros(n)
            off = k_curr - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:] += (1.0 - h[:np_])
                g[:np_] -= h[off:]
            else:
                a = -off
                np_ = n - a
                g[:np_] += (1.0 - h[a:])
                g[a:] -= h[:np_]
            if k_prev is None:
                k_prev = k_curr

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm

        base_step = max(0.002, 0.03 * (1.0 - 0.5 * it / max_iters))
        improved = False
        for step in [base_step * 0.3, base_step * 0.6, base_step, base_step * 2,
                     base_step * 4, 0.003, 0.008, 0.015, 0.03, 0.06, 0.1]:
            h_new = project_simplex_box(h - step * g, n_half)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                improved = True
                stale = 0
                break

        if not improved:
            stale += 1
            if stale > 300:
                break

    return best_h, best_c5

def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)

def top_k_direction(h, n, top_k=8):
    """Compute a direction that equalizes the top-k correlations."""
    dx = 2.0 / n
    corr = compute_corr(h)
    idx_top = np.argsort(-corr)[:top_k]
    c_max = corr[idx_top[0]]
    weights = np.zeros(2 * n - 1)
    for i, k in enumerate(idx_top):
        weights[k] = 1.0 if i == 0 else -1.0 / (top_k - 1)

    g = np.zeros(n)
    for k in idx_top:
        wk = weights[k] * dx
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - h[a:])
            g[a:] -= wk * h[:np_]
    gn = np.linalg.norm(g)
    if gn < 1e-14:
        return None
    return g / gn
