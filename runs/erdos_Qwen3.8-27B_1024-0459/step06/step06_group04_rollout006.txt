import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        interior = (h > 1e-13) & (h < 1.0 - 1e-13)
        if interior.any():
            h[interior] += diff / interior.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def slse_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = corr.max()
    s = np.clip(beta * (corr - m), -600.0, 600.0)
    e = np.exp(s)
    return (m + np.log(e.sum()) / beta) * dx


def slse_jac(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = corr.max()
    s = np.clip(beta * (corr - m), -600.0, 600.0)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx] * dx
        if wk < 1e-16:
            continue
        off = k_idx - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - h[ao:])
            g[ao:] -= wk * h[:np_]
    return g


def eq_func(x, n):
    return x.sum() - n / 2.0


def eq_jac(x, n):
    return np.ones(n)


def slsqp_smooth(h, beta, maxiter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(slse_obj, h, args=(beta,),
                       jac=slse_jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': eq_func,
                                     'jac': eq_jac, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def active_set_optimize(h, n, time_limit, K=40):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    for outer in range(200):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 20:
            break
        corr = compute_corr(best_h)
        top_k = np.argsort(-corr)[:K]
        top_vals = corr[top_k]
        top_max = top_vals[0]
        beta_w = 100.0
        exps = np.exp(beta_w * (top_vals - top_max))
        weights = exps / exps.sum()

        def qobj(x):
            c = compute_corr(x)
            return np.dot(weights, c[top_k]) * (2.0 / n)

        def qjac(x):
            g = np.zeros(n)
            for i in range(K):
                wi = weights[i]
                if abs(wi) < 1e-16:
                    continue
                k_idx = top_k[i]
                off = k_idx - (n - 1)
                if off >= 0:
                    np_ = n - off
                    g[off:] += wi * (1.0 - x[:np_])
                    g[:np_] -= wi * x[off:]
                else:
                    ao = -off
                    np_ = n - ao
                    g[:np_] += wi * (1.0 - x[ao:])
                    g[ao:] -= wi * x[:np_]
            g *= (2.0 / n)
            return g

        try:
            res = minimize(qobj, best_h, jac=qjac, method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq', 'fun': eq_func,
                                         'jac': eq_jac, 'args': (n,)}],

            if res.x is not None:

                c_new = compute_c5(h_new)
                if c_new < best_c5 - 1e-15:

def penalty_optimize(h, n, time_limit):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t = best_c5 + 0.01
    mu = 10.0
    start = time.time()

        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 20:
            break

        def pen_obj(x, _t, _mu):
            c = compute_corr(x) * (2.0 / n)
            val = _t
            excess = np.maximum(0, c - _t)
            val += _mu * np.sum(excess * excess)
            return val

        def pen_jac(x, _t, _mu):
            c = compute_corr(x) * (2.0 / n)
            excess = np.maximum(0, c - _t)
            w = np.zeros(2 * n - 1)
            w[excess > 0] = 2.0 * _mu * excess[excess > 0]
            g = np.zeros(n)
            for k_idx in range(2 * n - 1):
                wk = w[k_idx]
                if abs(wk) < 1e-16:
                    continue

                if off >= 0:
                    np_ = n - off
                    g[off:] += wk * (1.0 - x[:np_])
                    g[:np_] -= wk * x[off:]
                else:
                    ao = -off
                    np_ = n - ao
                    g[:np_] += wk * (1.0 - x[ao:])
                    g[ao:] -= wk * x[:np_]
            return g

        try:
            res = minimize(pen_obj, best_h, args=(t, mu), jac=pen_jac,
                           method='SLSQP', bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq', 'fun': eq_func,
                                         'jac': eq_jac, 'args': (n,)}],
                           options={'maxiter': 60, 'ftol': 1e-16, 'iprint': 0})
            if res.x is not None:
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new)
                t = c_new
                if c_new < best_c5 - 1e-15:
                    best_c5 = c_new
                    best_h = h_new
                mu *= 3.0
                if mu > 1e6:
                    break
        except Exception:
            break
    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def binary_local_refine(h, n, time_limit):
