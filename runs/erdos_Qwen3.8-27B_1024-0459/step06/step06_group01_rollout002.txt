def grad_single_corr(h, k, n):
    """Gradient of corr[k] w.r.t. h"""
    off = k - (n - 1)
    g = np.zeros(n)
    if off >= 0:
        np_ = n - off
        g[off:] += (1.0 - h[:np_])
        g[:np_] -= h[off:]
    else:
        a = -off
        np_ = n - a
        g[:np_] += (1.0 - h[a:])
        g[a:] -= h[:np_]
    return g

def topk_subgrad(h, n, top_k, max_iters):

def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)
