import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    # Project onto {0 ≤ h ≤ 1, sum(h) = n/2}
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        # Adjust only values not at boundary
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        adj = diff / cnt
        h[mask] += adj
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_overlap_k(h, k, n):
    """Gradient of overlap at shift k (full correlation index k)"""
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        g[offset:] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:]
    else:
        a = -offset
        npairs = n - a
        g[:npairs] += (1.0 - h[a:])
        g[a:] -= h[:npairs]
    return g

def active_set_gradient(h, n, K):
    """Compute gradient of sum of top-K correlations"""
    corr = compute_corr(h)
    top_idx = np.argsort(-corr)[:K]
    g = np.zeros(n)
    for k in top_idx:
        g += grad_overlap_k(h, k, n)
    return g / K, corr, top_idx

def active_set_descent(h, n, time_limit, K=5, max_iter=10000):
    """Active-set gradient descent with backtracking line search"""
    h_curr = h.copy()
    h_best = h.copy()
    c5_best = compute_c5(h)
    t0 = time.time()
    
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        
        g, corr, top_idx = active_set_gradient(h_curr, n, K)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        
        # Backtracking line search
        base_step = 1.0 / (gn + 1e-10)
        improved = False
        for factor in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125]:
            step = base_step * factor
            h_try = project_feasible(h_curr - step * g, n)
            c_try = compute_c5(h_try)
            if c_try < c5_best - 1e-14:
                c5_best = c_try
                h_best = h_try.copy()
                h_curr = h_try.copy()
                improved = True
                break
            elif c_try < compute_c5(h_curr) - 1e-12:
                # Accept even if not improving best (to escape)
                h_curr = h_try.copy()
                improved = True
                break
        
        if not improved and it % 1000 == 0:
            # Stuck: try smaller K or perturbation
            g, _, _ = active_set_gradient(h_curr, n, max(3, K//2))
            step = 0.01 / (np.linalg.norm(g) + 1e-10)
            h_try = project_feasible(h_curr - step * g, n)
            h_curr = h_try
    
    return h_best, c5_best

            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_overlap_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        if npairs <= 0:
            return g
        g[offset:] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:]
    else:
        a = -offset
        npairs = n - a
        if npairs <= 0:
            return g
        g[:npairs] += (1.0 - h[a:])
        g[a:] -= h[:npairs]
    return g

def active_set_gradient(h, n, K):
    corr = compute_corr(h)
    top_idx = np.argsort(-corr)[:K]
    g = np.zeros(n)
    for k in top_idx:
        g += grad_overlap_k(h, int(k), n)
    g /= K
    return g, corr, top_idx

def hinge_gradient(h, n, threshold, top_extra=0):
    corr = compute_corr(h)
    above = np.where(corr > threshold)[0]
    g = np.zeros(n)
    for k in above:
        weight = max(0.0, corr[k] - threshold)
        g += weight * grad_overlap_k(h, int(k), n)
    if top_extra > 0 and len(above) > 0:
        top_full = np.argsort(-corr)[:top_extra]
        for k in top_full:
            if k not in above:
                g += 0.5 * grad_overlap_k(h, int(k), n)
    if len(above) == 0 and top_extra == 0:
        top_idx = np.argsort(-corr)[:3]
        for k in top_idx:
            g += grad_overlap_k(h, int(k), n)
    return g, corr

def gradient_descent_core(h, n, time_limit, K=5, use_hinge=False, hinge_tau_frac=0.98):
    h_curr = h.copy()
    h_best = h.copy()
    c5_best = compute_c5(h)
    t0 = time.time()
    stall_count = 0
    step_scale = 1.0

    while time.time() - t0 < time_limit:
        corr = compute_corr(h_curr)
        c5_curr = np.max(corr) * (2.0 / n)
        if c5_curr < c5_best - 1e-12:
            c5_best = c5_curr
            h_best = h_curr.copy()
            stall_count = 0

        if use_hinge:
            thresh = c5_best * (n / 2.0) * hinge_tau_frac
            g, _ = hinge_gradient(h_curr, n, thresh, top_extra=K)
        else:
            g, _, _ = active_set_gradient(h_curr, n, K)

        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break

        base_step = (0.3 * step_scale) / gn
        improved = False
        for factor in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.01]:
            step = min(base_step * factor, 0.5)
            h_try = project_feasible(h_curr - step * g, n)
            c_try = compute_c5(h_try)
            if c_try < c5_best - 1e-13:
                c5_best = c_try
                h_best = h_try.copy()
                h_curr = h_try.copy()
                improved = True
                break
            elif c_try < c5_curr - 1e-10:
                h_curr = h_try.copy()
                improved = True
                break

        if not improved:
            stall_count += 1
            if stall_count > 200:
                step_scale *= 2.0
                if step_scale > 100.0:
                    rng2 = np.random.RandomState(int(time.time() * 1000) % 100000)
                    h_curr = project_feasible(h_curr + rng2.randn(n) * 0.01, n)
                    step_scale = 1.0
                    stall_count = 0
            elif stall_count > 50:
                K_new = max(3, K + 2)
                g, _, _ = active_set_gradient(h_curr, n, K_new)
                gn = np.linalg.norm(g)
                if gn > 1e-14:
                    step = min(0.1 / gn, 0.1)
                    h_try = project_feasible(h_curr - step * g, n)
                    h_curr = h_try
                stall_count = 0
        else:
            if stall_count > 0:
                step_scale = max(0.5, step_scale * 0.9)
            stall_count = 0

    return h_best, c5_best

def slsqp_polish(h, n, K, beta, maxiter=300):
    def obj_func(x):
        corr = compute_corr(x)
        idx = np.argsort(-corr)[:K]
        c_top = corr[idx]
        m = np.max(c_top)
        s = np.clip(beta * (c_top - m), -50, 50)
        ex = np.exp(s)
        return (m + np.log(ex.sum()) / beta) * (2.0 / n)

    def grad_func(x):
        corr = compute_corr(x)
        idx = np.argsort(-corr)[:K]
        c_top = corr[idx]
        m = np.max(c_top)
        s = np.clip(beta * (c_top - m), -50, 50)
        ex = np.exp(s)
        w = ex / ex.sum()
        g = np.zeros(n)
        dx = 2.0 / n
        for i in range(K):
            if w[i] > 1e-12:
                g += w[i] * dx * grad_overlap_k(x, int(idx[i]), n)
        return g

    def cfun(x):
        return np.sum(x) - n / 2.0

    def cjac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj_func, h, jac=grad_func, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            return h_new
    except Exception:
        pass
    return h

def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def run(seed=42, budget_s=1300, **kwargs):
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)

    # Get initial solution from parent
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 512
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_c5 = compute_c5(best_h)

    n = best_n
    h = best_h.copy()

    # Phase 1: Active-set gradient descent at parent resolution
    # Use multiple K values adaptively
    if remaining() > 60:
        t1 = min(120.0, remaining() * 0.35)
        for K_val in [3, 5, 8]:
            if remaining() < 20:
                break
            h_sg, c5_sg = gradient_descent_core(h, n, min(t1/3, 50.0), K=K_val, use_hinge=False)
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg.copy()
                h = h_sg.copy()

    # Phase 2: Hinge-based gradient (considers many active constraints)
    if remaining() > 60:
        t2 = min(100.0, remaining() * 0.3)
        h_hg, c5_hg = gradient_descent_core(h, n, t2, K=5, use_hinge=True, hinge_tau_frac=0.97)
        if c5_hg < best_c5:
            best_c5 = c5_hg
            best_h = h_hg.copy()
            h = h_hg.copy()

    # Phase 3: SLSQP polishing at parent resolution
    for K in [5, 8, 12]:
        for beta in [500, 2000, 8000]:
            if remaining() < 10:
                break
            h_new = slsqp_polish(h, n, K, beta, maxiter=250)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 4: Another round of gradient descent after SLSQP
    if remaining() > 40:
        t4 = min(60.0, remaining() * 0.25)
        h_gd, c5_gd = gradient_descent_core(h, n, t4, K=4, use_hinge=False)
        if c5_gd < best_c5:
            best_c5 = c5_gd
            best_h = h_gd.copy()
            h = h_gd.copy()
        if remaining() > 30:
            h_gd2, c5_gd2 = gradient_descent_core(h, n, min(30.0, remaining()*0.2), K=6, use_hinge=True, hinge_tau_frac=0.95)
            if c5_gd2 < best_c5:
                best_c5 = c5_gd2
                best_h = h_gd2.copy()
                h = h_gd2.copy()

    # Phase 5: Try higher resolution (n=768 or n=1024)
    n_hi = 1024
    if remaining() > 120 and n < n_hi:
        h_hi = project_feasible(interp_to(best_h, best_n, n_hi), n_hi)
        c5_hi = compute_c5(h_hi)

        if remaining() > 80:
            h_hi, c5_hi = gradient_descent_core(h_hi, n_hi, min(60.0, remaining()*0.4), K=5, use_hinge=False)

        if remaining() > 50:
            h_hi, c5_hi = gradient_descent_core(h_hi, n_hi, min(40.0, remaining()*0.3), K=6, use_hinge=True, hinge_tau_frac=0.96)

        if remaining() > 30:
            h_hi = slsqp_polish(h_hi, n_hi, 8, 2000, maxiter=200)
            c5_hi = compute_c5(h_hi)

        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi.copy()
            best_n = n_hi

    # Phase 6: Try n=768 as well (if we didn't already use that)
    n_mid = 768
    if remaining() > 100 and best_n != n_mid:
        h_mid = project_feasible(interp_to(best_h, best_n, n_mid), n_mid)
        c5_mid = compute_c5(h_mid)

        if remaining() > 60:
            h_mid, c5_mid = gradient_descent_core(h_mid, n_mid, min(50.0, remaining()*0.4), K=5, use_hinge=False)

        if remaining() > 40:
            h_mid, c5_mid = gradient_descent_core(h_mid, n_mid, min(30.0, remaining()*0.3), K=7, use_hinge=True, hinge_tau_frac=0.96)

        if remaining() > 25:
            h_mid = slsqp_polish(h_mid, n_mid, 8, 3000, maxiter=200)
            c5_mid = compute_c5(h_mid)

        if c5_mid < best_c5:
            best_c5 = c5_mid
            best_h = h_mid.copy()
            best_n = n_mid

    # Phase 7: Multi-start perturbation at best resolution
    n = best_n
    h = best_h.copy()
    if remaining() > 60:
        for pert_i in range(5):
            if remaining() < 20:
                break
            sigma = 0.008 + 0.007 * pert_i
            h_pert = project_feasible(h + rng.randn(n) * sigma, n)
            c5_pert = compute_c5(h_pert)
            if c5_pert > best_c5 + 0.02:
                continue
            t_pert = min(25.0, remaining() * 0.3)
            h_pert, c5_pert = gradient_descent_core(h_pert, n, t_pert, K=4, use_hinge=False)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()
                h = h_pert.copy()

    # Phase 8: Final intensive SLSQP polish
    n = best_n
    h = best_h.copy()
    for K in [4, 6, 10]:
        for beta in [3000, 10000, 30000]:
            if remaining() < 6:
                break
            h_new = slsqp_polish(h, n, K, beta, maxiter=350)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 9: Final gradient descent pass
    if remaining() > 15:
        h_fg, c5_fg = gradient_descent_core(best_h, best_n, min(20.0, remaining()*0.5), K=3, use_hinge=False)
        if c5_fg < best_c5:
            best_c5 = c5_fg
            best_h = h_fg.copy()

    # Phase 10: Final SLSQP with very high beta
    if remaining() > 8:
        h_final = slsqp_polish(best_h, best_n, 5, 50000, maxiter=400)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
