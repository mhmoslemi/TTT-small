def project_feasible(h, n):

    h0 = h0.copy()
    
    def obj(x):
        return smooth_max_value(x, beta)
    
    def grad(x):
        return smooth_max_gradient(x, beta)
    
    def constr_fun(x):
        return np.sum(x) - n / 2.0
    
    def constr_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constr_fun, 'jac': constr_jac}]
    
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',

def subgradient_descent(h0, n, max_iters=5000, time_limit=200.0):
    """Projected subgradient descent with adaptive step size."""
    h = h0.copy()
    c5_cur = compute_c5(h)
    best_c5 = c5_cur
    best_h = h.copy()
    start = time.time()
    
    # Use top-K subgradients with softmax weighting
    step_schedule = 0.05
    decay = 0.999
    
    corr = compute_corr(h)
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if time.time() - _T0 > _BUDGET - 30:
            break
        
        c5_cur = compute_c5(h)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
        
        progress = it / max_iters
        K = max(5, int(30 * (1 - 0.7 * progress)))
        beta_g = 20.0 + 80.0 * progress
        
        # Compute correlation and top-K
        corr = compute_corr(h)
        # Select top K correlations
        top_idx = np.argsort(-corr)[:K]
        top_vals = corr[top_idx]
        weights = np.exp(beta_g * (top_vals - top_vals.max()))
        weights /= weights.sum()
        
        # Compute combined subgradient
        g = np.zeros(n)
        for ii, idx in enumerate(top_idx):
            if weights[ii] < 1e-12:
                continue
            k_shift = idx - (n - 1)
            if k_shift == 0:
                g += weights[ii] * (1.0 - 2.0 * h)
            elif k_shift > 0:
                # corr_k = sum_{i=0}^{n-1-k} h[i+k](1-h[i])
                # d/dh[j] = (1-h[j-k])*1[j>=k] - h[j+k]*1[j<=n-1-k]
                g[k:] += weights[ii] * (1.0 - h[:n-k_shift])
                g[:n-k_shift] -= weights[ii] * h[k_shift:]
            else:
                kk = -k_shift
                g[:n-kk] += weights[ii] * (1.0 - h[kk:])
                g[kk:] -= weights[ii] * h[:n-kk]
        
        # Take step with projection
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        step = step_schedule * (1.0 / (1.0 + 0.01 * progress))
        h_trial = h - step * g / gn
        h_new = project_feasible(h_trial, n)
        
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur + 0.001 * step:
            h = h_new
        else:
            # Backtrack
            step *= 0.5
            h_trial = h - step * g / gn
            h_new = project_feasible(h_trial, n)
            h = h_new
            step_schedule *= 0.95
        step_schedule *= decay
    
    return best_h, best_c5
