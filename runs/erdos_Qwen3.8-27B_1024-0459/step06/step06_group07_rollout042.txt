def precompute_dcorr(h, n):

    dcorr = np.zeros((n, nfull), dtype=float)

        dcorr[i, v1] += (1.0 - h[j1[v1]])
        dcorr[i, v2] -= h[j2[v2]]
    return dcorr

def to_binary(h, n):
    h = np.array(h, dtype=float).copy()
    h_bin = (h >= 0.5).astype(float)
    ones = int(np.sum(h_bin))
    target = n // 2
    if ones > target:
        idx = np.where(h_bin == 1)[0]
        scores = h[idx]
        order = np.argsort(scores)
        for k in range(ones - target):
            h_bin[idx[order[k]]] = 0.0
    elif ones < target:
        idx = np.where(h_bin == 0)[0]
        scores = h[idx]
        order = np.argsort(-scores)
        for k in range(target - ones):
            h_bin[idx[order[k]]] = 1.0
    return h_bin


def thue_morse_bin(n):
    h = np.zeros(n, dtype=np.int8)
    for i in range(n):
        h[i] = bin(i).count('1') & 1
    return (h == 0).astype(float)


def make_random_binary(n, rng):
    h = np.zeros(n, dtype=float)

    h[perm[:n // 2]] = 1.0
    return h


def precompute_dcorr(h, n):
    nfull = 2 * n - 1
    k_range = np.arange(nfull)
    dcorr = np.zeros((n, nfull), dtype=float)
    for i in range(n):
        j1 = i + k_range - (n - 1)
        j2 = i - k_range + (n - 1)
        v1 = (j1 >= 0) & (j1 < n)
        v2 = (j2 >= 0) & (j2 < n)
        dcorr[i, v1] += (1.0 - h[j1[v1]])
        dcorr[i, v2] -= h[j2[v2]]
    return dcorr


def update_dcorr_flip(dcorr, i, delta, n):
    max_m = 2 * n - 2
    for k in range(n):
        m1 = i - k + (n - 1)
        if 0 <= m1 <= max_m:
            dcorr[k, m1] -= delta
        m2 = k - i + (n - 1)
        if 0 <= m2 <= max_m:
            dcorr[k, m2] -= delta


def greedy_fill(n):

    corr = np.zeros(2 * n - 1, dtype=float)
    dcorr = precompute_dcorr(h, n)
    for _step in range(n // 2):

        i = int(np.argmin(scores))
        h[i] = 1.0
        corr = corr + dcorr[i]
        update_dcorr_flip(dcorr, i, 1.0, n)
    return h
