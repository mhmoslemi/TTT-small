- To reduce corr at lag k, we want to move 1s to positions where h shifted by k is also 1

from scipy.optimize import minimize
import time

_BUDGET = 1300.0

    dx = 2.0 / n

    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):

        if abs(diff) < 1e-14:

def to_binary(h, n):

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = bin(i).count('1') % 2
    return h

def corr_fft(h):
    """Compute full correlation using FFT."""
    n = len(h)
    v = 1.0 - h
    size = 2 * n - 1
    H = np.fft.rfft(h, size)
    Vf = np.fft.rfft(v[::-1], size)
    return np.fft.irfft(H * Vf, size)

def smooth_max_val(corr, K, beta):
    """Compute smooth max of top K entries of corr."""
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    m = np.max(c_top)
    sh = np.clip(beta * (c_top - m), -50.0, 50.0)
    ex = np.exp(sh)
    return m + np.log(ex.sum()) / beta

def smooth_max_weights(corr, K, beta):
    """Get weights for the smooth max."""
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    m = np.max(c_top)
    sh = np.clip(beta * (c_top - m), -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / ex.sum()
    return idx, w

def grad_corr_k(h, k, n):
    """Gradient of correlation at lag k (where lag k means shift by k in the full correlate).
    
    full correlate output index = k + n - 1 for shift k.
    For shift k >= 0: C = sum_{i=k}^{n-1} h[i]*(1-h[i-k])
    For shift k < 0: C = sum_{i=0}^{n-1+k} h[i]*(1-h[i+k]) where k is negative, so n-1+k = n-1-|k|
    """
    g = np.zeros(n)
    if k >= 0:
        npairs = n - k
        # C = sum_{i=k}^{n-1} h[i] * (1 - h[i-k])
        # dC/dh[j]:
        #   If j >= k: h[j] appears as h[i] with i=j, contributes (1-h[j-k])
        #   If j < k: h[j] appears as h[i-k] with i=j+k, contributes -h[j+k]
        g[k:] += (1.0 - h[:npairs])
        g[:npairs] -= h[k:]
    else:
        a = -k  # positive
        npairs = n - a
        # C = sum_{i=0}^{npairs-1} h[i] * (1 - h[i+a])
        # dC/dh[j]:
        #   If j < npairs: h[j] appears as h[i] with i=j, contributes (1-h[j+a])
        #   If j >= npairs: h[j] appears as h[i+a] with i=j-a, contributes -h[j-a]
        g[:npairs] += (1.0 - h[a:])
        g[a:] -= h[:npairs]
    return g

def smooth_max_grad(h, n, K, beta):
    """Gradient of smooth_max(corr) * dx with respect to h."""
    corr = compute_corr(h)
    idx, w = smooth_max_weights(corr, K, beta)
    dx = 2.0 / n
    g = np.zeros(n)
    for i in range(K):
        wk = w[i] * dx
        if wk > 1e-15:
            k = idx[i] - (n - 1)  # convert output index to lag
            g += wk * grad_corr_k(h, k, n)
    return g

def slsqp_opt(h, n, K, beta, lambda_p, maxiter=300):
    """SLSQP optimization with binary penalty."""
    dx = 2.0 / n
    
    def obj(x):
        corr = compute_corr(x)
        val = smooth_max_val(corr, K, beta) * dx
        if lambda_p > 0:
            val += lambda_p * np.sum(x * (1 - x))
        return val

    def grad(x):
        g = smooth_max_grad(x, n, K, beta)
        if lambda_p > 0:
            g -= 2.0 * lambda_p * (2.0 * x - 1.0)
        return g

    def cfun(x):
        return np.sum(x) - n / 2.0

    def cjac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]

    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            return h_new
    except Exception:
        pass
    return h

def grad_corr_k(h, k, n):
    g = np.zeros(n)
    if k >= 0:
        # g[j] += (1-h[j-k]) for j >= k
        g[k:] += (1.0 - h[:n-k])
        # g[j] += -h[j+k] for j <= n-1-k
        g[:n-k] -= h[k:]
    else:
        a = -k
        # g[j] += (1-h[j+a]) for j < n-a
        g[:n-a] += (1.0 - h[a:])
        # g[j] += -h[j-a] for j >= a
        g[a:] -= h[:n-a]
    return g
