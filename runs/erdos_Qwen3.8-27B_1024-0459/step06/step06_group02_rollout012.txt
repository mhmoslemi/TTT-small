import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def project_sum_clip(v, target_sum):
    """Project v onto {h: sum(h)=target_sum, 0<=h<=1} via bisection on shift."""
    n = len(v)
    lo = v.min() - 2.0
    hi = v.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(v - tau, 0.0, 1.0)


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    g = np.zeros(n)
    if offset >= 0:
        npairs = n - offset
        if npairs > 0:
            g[:npairs] += (1.0 - h[offset:])
            g[offset:] -= h[:npairs]
    else:
        a = -offset
        npairs = n - a
        if npairs > 0:
            g[a:] += (1.0 - h[:npairs])
            g[:npairs] -= h[a:]
    return g


def smooth_max_obj(h, beta, K):
    corr = compute_corr(h)
    dx = 2.0 / len(h)
    idx = np.argsort(-corr)[:K]
    c_max = corr.max()
    shifted = beta * (corr[idx] - c_max)
    shifted = np.clip(shifted, -50.0, 50.0)
    ex = np.exp(shifted)
    val = (c_max + np.log(ex.sum()) / beta) * dx
    return val


def smooth_max_grad(h, beta, K):
    corr = compute_corr(h)
    dx = 2.0 / len(h)
    n = len(h)
    idx = np.argsort(-corr)[:K]
    c_max = corr.max()
    shifted = beta * (corr[idx] - c_max)
    shifted = np.clip(shifted, -50.0, 50.0)
    ex = np.exp(shifted)
    w = ex / ex.sum()
    g = np.zeros(n)
    for i, k in enumerate(idx):
        if w[i] > 1e-14:
            g += w[i] * grad_corr_k(h, k, n)
    g *= dx
    return g


def slsqp_smooth_max(h, n, beta, K, maxiter=500, ftol=1e-16):
    def obj(x):
        return smooth_max_obj(x, beta, K)

    def grad(x):
        return smooth_max_grad(x, beta, K)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]

    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            if compute_c5(h_new) <= compute_c5(h) + 1e-12:
                return h_new
    except Exception:
        pass
    return h.copy()


def block_slsqp(h, n, beta, K, block_size, maxiter_per_block=100):
    """Block coordinate descent: optimize one block at a time with SLSQP."""
    h = h.copy()
    best_c5 = compute_c5(h)
    n_blocks = (n + block_size - 1) // block_size
    for bi in range(n_blocks):
        start = bi * block_size
        end = min(start + block_size, n)
        B = end - start
        if B < 2:
            continue
        block_sum = np.sum(h[start:end])
        other_sum = n / 2.0 - block_sum

        def obj_block(x):
            h_trial = h.copy()
            h_trial[start:end] = x
            return smooth_max_obj(h_trial, beta, K)

        def grad_block(x):
            h_trial = h.copy()
            h_trial[start:end] = x
            full_grad = smooth_max_grad(h_trial, beta, K)
            return full_grad[start:end]

        x0 = h[start:end].copy()
        bounds = [(0.0, 1.0)] * B
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - block_sum,
                 'jac': lambda x: np.ones(B)}]

        try:

                           options={'maxiter': maxiter_per_block, 'ftol': 1e-14, 'iprint': 0})
            if res.x is not None:

                if c_new < best_c5:

                    best_c5 = c_new
        except Exception:

def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def subgradient_step(h, n, K, beta, step):
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:K]
    c_vals = corr[idx]
    c_max = c_vals.max()
    weights = np.exp(beta * (c_vals - c_max))
    s = weights.sum()
    if s < 1e-30:
        weights = np.ones(K) / K
    else:
        weights /= s
    g = np.zeros(n)
    for i, k in enumerate(idx):
        if weights[i] > 1e-12:
            g += weights[i] * grad_corr_k(h, k, n) * (2.0 / n)
    gn = np.linalg.norm(g)
    if gn < 1e-14:
        return h
    h_new = project_feasible(h - step * g / gn, n)
    return h_new


def nesterov_pgd(h, n, beta, K, max_iter, time_limit):
    """Nesterov accelerated projected gradient with backtracking."""
    start_time = time.time()
    target_sum = n / 2.0
    h_curr = h.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h_curr)
    t = 1.0
    alpha = 0.01
    f_curr = compute_c5(h_curr)

    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        momentum = (t - 1.0) / (t + 1.0)
        y = h_curr + momentum * (h_curr - h_prev)
        y_proj = np.clip(y, 0.0, 1.0)

        g = smooth_max_grad(y_proj, beta, K)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break

        for _ in range(20):
            h_trial = project_sum_clip(y - alpha * g, target_sum)
            f_trial = compute_c5(h_trial)
            if f_trial <= f_curr - 1e-18 * f_curr:

            alpha *= 0.5
        else:
            alpha = 0.001

        h_curr = h_trial
        t = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        f_curr = compute_c5(h_curr)

        if f_curr < best_c5:
            best_c5 = f_curr
            best_h = h_curr.copy()

    return best_h, best_c5

    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
