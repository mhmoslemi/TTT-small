        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        if npairs > 0:
            g[offset:] += (1.0 - h[:npairs])
            g[:npairs] -= h[offset:]
    else:
        a = -offset
        npairs = n - a
        if npairs > 0:
            g[:npairs] += (1.0 - h[a:])
            g[a:] -= h[:npairs]
    return g

def active_set_optimize(h, n, K, beta, maxiter=500, ftol=1e-16):
    """Optimize using active-set smooth max over top K lags."""
    dx = 2.0 / n
    
    def get_active_lags(corr):
        # Return indices of top K lags
        top_k = np.argsort(-corr)[:K]
        return top_k
    
    def obj(x):
        corr = compute_corr(x)
        top_k = np.argsort(-corr)[:K]
        c_active = corr[top_k]
        m = np.max(c_active)
        shifted = beta * (c_active - m)
        shifted = np.clip(shifted, -50.0, 50.0)
        ex = np.exp(shifted)
        s = np.sum(ex)
        val = (m + np.log(s) / beta) * dx
        return val
    
    def gradient(x):
        corr = compute_corr(x)
        top_k = np.argsort(-corr)[:K]
        c_active = corr[top_k]
        m = np.max(c_active)
        shifted = beta * (c_active - m)
        shifted = np.clip(shifted, -50.0, 50.0)
        ex = np.exp(shifted)
        s = np.sum(ex)
        w = ex / s
        g = np.zeros(n)
        for i, k in enumerate(top_k):
            if w[i] > 1e-14:
                g += w[i] * grad_corr_k(x, k, n)
        g *= dx
        return g
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0, 
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, jac=gradient, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        if res.x is not None and res.fun < obj(h) + 1e-10:
            h_new = project_feasible(res.x, n)
            return h_new
    except Exception:
        pass
    
    # Retry with slightly different settings
    try:
        res = minimize(obj, h, jac=gradient, method='SLSQP',

                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        if res.x is not None:

            return h_new

        pass

    # Phase 2: Try interpolation to higher resolutions
    for n_t in [1024, 1536, 2048]:
        if remaining() < 60:
            break
        h_t = project_feasible(interp_to(best_h, best_n, n_t), n_t)
        c_t = compute_c5(h_t)
        
        for K in [3, 5, 10]:
            for beta in [500, 5000, 30000]:
                if remaining() < 25:
                    break
                h_new = active_set_slsqp(h_t, n_t, K, beta, maxiter=350)
                c_new = compute_c5(h_new)
                if c_new < compute_c5(h_t) - 1e-12:
                    h_t = h_new
                if c_new < best_c5:
                    best_c5 = c_new
                    best_h = h_t.copy()
                    best_n = n_t
    
    # Phase 3: Try different resolutions from parent
    for n_t in [512, 1024]:
        if remaining() < 50:
            break
        h_t = project_feasible(interp_to(parent_h, n_parent, n_t), n_t)
        for K in [3, 5, 10]:
            for beta in [1000, 10000, 50000]:
                if remaining() < 20:
                    break
                h_new = active_set_slsqp(h_t, n_t, K, beta, maxiter=300)
                c_new = compute_c5(h_new)
                if c_new < compute_c5(h_t) - 1e-12:
                    h_t = h_new
                if c_new < best_c5:
                    best_c5 = c_new
                    best_h = h_t.copy()
                    best_n = n_t
    
    # Phase 4: Perturbation + re-optimization from best
    for p in range(8):
        if remaining() < 25:
            break
        scale = 0.005 + p * 0.005
        h_pert = structured_perturbation(best_h, best_n, rng, scale=scale)
        for K in [5, 10]:
            for beta in [5000, 30000, 100000]:
                if remaining() < 15:
                    break
                h_new = active_set_slsqp(h_pert, best_n, K, beta, maxiter=300)
                c_new = compute_c5(h_new)
                if c_new < compute_c5(h_pert) - 1e-12:
                    h_pert = h_new
                if c_new < best_c5:
                    best_c5 = c_new
                    best_h = h_pert.copy()
                    best_n = best_n
    
    # Phase 5: Threshold polishing
    if remaining() > 30:
        h_thr = threshold_polish(best_h, best_n, eps=0.08)
        c_thr = compute_c5(h_thr)
        if c_thr < best_c5:
            best_c5 = c_thr
            best_h = h_thr.copy()
        # Optimize from thresholded
        for K in [3, 5]:
            for beta in [20000, 100000]:
                if remaining() < 15:
                    break
                h_new = active_set_slsqp(h_thr, best_n, K, beta, maxiter=400)
                c_new = compute_c5(h_new)
                if c_new < best_c5:
                    best_c5 = c_new
                    best_h = h_new.copy()
    
    # Phase 6: Block initializations at best resolution
    for nb in [4, 6, 8, 12, 16]:
        if remaining() < 30:
            break
        h_blk = make_block_initial(best_n, rng, nb, pattern_seed=nb*7)
        for K in [5, 10]:
            for beta in [2000, 20000]:
                if remaining() < 15:
                    break
                h_new = active_set_slsqp(h_blk, best_n, K, beta, maxiter=250)
                c_new = compute_c5(h_new)
                if c_new < compute_c5(h_blk) - 1e-12:
                    h_blk = h_new
                if c_new < best_c5:
                    best_c5 = c_new
                    best_h = h_blk.copy()
    
    # Phase 7: Final ultra-fine polishing with various K
    h_final = best_h.copy()
    for K in [2, 3, 5, 8]:
        for beta in [100000, 300000, 1000000]:
            if remaining() < 12:
                break
            h_new = active_set_slsqp(h_final, best_n, K, beta, maxiter=800, ftol=1e-18)
            c_new = compute_c5(h_new)
            if c_new < compute_c5(h_final) - 1e-14:
                h_final = h_new
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_final.copy()
    
    # Phase 8: Multi-start sinusoidal
    for s in range(4):
        if remaining() < 25:
            break
        h_sin = make_sinusoidal_initial(best_n, rng)
        for K in [5, 10]:
            for beta in [5000, 50000]:
                if remaining() < 12:
                    break
                h_new = active_set_slsqp(h_sin, best_n, K, beta, maxiter=250)
                c_new = compute_c5(h_new)
                if c_new < compute_c5(h_sin) - 1e-12:
                    h_sin = h_new
                if c_new < best_c5:
                    best_c5 = c_new
                    best_h = h_sin.copy()
    
    # Phase 9: Final SLSQP squeeze
    if remaining() > 10:
        h_squeeze = active_set_slsqp(best_h, best_n, 3, 500000, maxiter=1000, ftol=1e-18)
        c_squeeze = compute_c5(h_squeeze)
        if c_squeeze < best_c5:
            best_c5 = c_squeeze
            best_h = h_squeeze.copy()
    
    # Phase 10: Very small perturbation
    for p in range(3):
        if remaining() < 12:
            break
        h_pert = best_h + rng.randn(best_n) * 0.002
        h_pert = project_feasible(h_pert, best_n)
        h_new = active_set_slsqp(h_pert, best_n, 3, 200000, maxiter=500)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
    
    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
