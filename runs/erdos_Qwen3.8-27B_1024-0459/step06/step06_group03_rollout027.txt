import numpy as np
from scipy.optimize import minimize, linprog
import time

_T0 = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)

def get_all_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_c_k(h, k, n):
    dx = 2.0 / n
    g = np.zeros(n)
    j = k - (n - 1)
    if j >= 0:
        cnt = n - j
        g[:cnt] += (1.0 - h[j:j+cnt])
        g[j:j+cnt] -= h[:cnt]
    else:
        a = -j
        cnt = n - a
        g[a:] += (1.0 - h[:cnt])
        g[:cnt] -= h[a:a+cnt]
    return g * dx

def active_set_lp(h, n, K, alpha):
    """Solve min t subject to linearized active constraints"""
    corr = get_all_corr(h)
    idx = np.argsort(-corr)[:K]
    
    # Build LP: min t s.t. g_k^T h <= t - b_k for k in active, sum(h)=n/2, 0<=h<=1
    # where b_k = c_k - g_k^T h (so g_k^T h + b_k = c_k at current point)
    # Actually: c_k(h_new) ≈ c_k(h) + g_k^T (h_new - h) = g_k^T h_new + (c_k - g_k^T h)
    # We want: g_k^T h_new + (c_k - g_k^T h) <= t
    # i.e.: g_k^T h_new - t <= g_k^T h - c_k
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    A_ub_rows = []
    b_ub_vals = []
    
    for k in idx:
        g = grad_c_k(h, k, n)
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_ub_rows.append(row)
        b_ub_vals.append(np.dot(g, h) - corr[k])
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0, 1)] * n + [(None, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            h_lp = res.x[:n]
            # Take a step: h_new = (1-alpha)*h + alpha*h_lp
            h_new = project_feasible((1 - alpha) * h + alpha * h_lp, n)
            return h_new
    except Exception:
        pass
    return None

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)

        cnt = np.count_nonzero(mask)

def grad_c_k(h, k, n):
    dx = 2.0 / n
    g = np.zeros(n)
    j = k - (n - 1)
    if j >= 0:
        cnt = n - j
        g[:cnt] += (1.0 - h[j:j + cnt])
        g[j:j + cnt] -= h[:cnt]
    else:
        a = -j
        cnt = n - a
        g[a:] += (1.0 - h[:cnt])
        g[:cnt] -= h[a:a + cnt]
    return g * dx

        e = np.exp(cl)
        return (cm + np.log(e.sum()) / _G_BETA) * (2.0 / nn)

    def smax_grad(v):
        nn = _G_N
        corr = np.correlate(v, 1.0 - v, mode='full')

        s = c - cm

        g = np.zeros(nn)

            if w[i] > 1e-14:
                g += w[i] * grad_c_k(v, idxx[i], nn)
        return g

        return np.sum(v) - _G_N / 2.0

        return np.ones(_G_N)

            h_low_s = slsqop_smmax(h_low, n_low, beta=2000, K=15, maxiter=400)
            c5_low3 = compute_c5(h_low_s)
            if c5_low3 < c5_low2:
                h_low, c5_low2 = (h_low_s, c5_low3)
            h_low_eq, c5_low_eq = equalize_top(h_low, n_low, K=6, max_iters=200)
            if c5_low_eq < c5_low2:
                h_low, c5_low2 = (h_low_eq, c5_low_eq)

            if c5_low2 < best_c5:
                best_c5 = c5_low2
                best_h = h_low.copy()
                best_n = n_low

    # Phase 8: Try higher resolution n=1024
    n_high = 1024
    if remaining() > 120:
        h_high = interp_to(best_h, best_n, n_high)
        h_high = project_feasible(h_high, n_high)
        c5_high = compute_c5(h_high)

        if c5_high <= best_c5 + 0.003:
            h_high, c5_h2, _ = bb_gradient_descent(h_high, n_high, min(30.0, remaining() * 0.1), beta_start=20, beta_end=200, K_start=20, K_end=5)
            if c5_h2 > c5_high:
                h_high, c5_h2 = (h_high.copy(), c5_high)
            if remaining() > 30:
                for beta in [100, 1000, 5000]:
                    if remaining() < 15:
                        break
                    h_high = slsqop_smmax(h_high, n_high, beta, K=20, maxiter=300)
                    c5_h2 = compute_c5(h_high)
            if remaining() > 15:
                h_high, c5_h2 = equalize_top(h_high, n_high, K=6, max_iters=150)

            if c5_h2 < best_c5:
                best_c5 = c5_h2
                best_h = h_high.copy()
                best_n = n_high

    # Phase 9: Final intensive polish at best resolution
    n = best_n
    h = best_h.copy()
    if remaining() > 30:
        h_f, c5_f = final_polish(h, n, min(remaining() * 0.4, 200.0))
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            h = h_f.copy()

    # Phase 10: Ultra-fine equalize + tiny steps
    if remaining() > 15:
        h_eq2, c5_eq2 = equalize_top(h, n, K=5, max_iters=500)
        if c5_eq2 < best_c5:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()
            h = h_eq2.copy()

    # Phase 11: Tiny gradient steps at the very end
    if remaining() > 10:
        h_t = best_h.copy()
        for it in range(200):
            if remaining() < 3:
                break
            corr_t = get_all_corr_scaled(h_t)
            idx_t = np.argsort(-corr_t)[:10]
            g_t = np.zeros(n)
            for k in idx_t:
                g_t += grad_c_k(h_t, k, n)
            g_t /= len(idx_t)
            gn_t = np.linalg.norm(g_t)
            if gn_t < 1e-14:
                break
            improved = False
            for step in [0.003, 0.001, 0.0005, 0.005, 0.002, 0.0008]:
                h_try = project_feasible(h_t - step * g_t / gn_t, n)
                c_try = compute_c5(h_try)
                if c_try < best_c5:
                    best_c5 = c_try
                    best_h = h_try.copy()
                    h_t = h_try.copy()
                    improved = True
                    break
            if not improved and it > 50:
                break

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            best_c5 = min(best_c5, float(eval_c5))
    except Exception:
        pass

    return (best_h, best_c5, best_n)
