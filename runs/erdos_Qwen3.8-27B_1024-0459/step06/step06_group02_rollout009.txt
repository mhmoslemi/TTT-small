min_h max_k |{i: h[i]=1, h[i+k]=0}| * dx

This is a combinatorial problem. For continuous h, it's a continuous relaxation.

One key insight: if h is very close to binary (values near 0 or 1), the problem becomes approximately combinatorial. The optimal binary pattern would minimize the maximum "disagreement" over all shifts.

Another insight: by the pigeonhole principle, for any h with integral 1 on [0,2], there exists a shift k such that the overlap is at least some value. The question is what's the tightest such lower bound.

The gradient of the soft-max objective is:
G = Σ_k w_k * d(corr_k)/d(h)

d(corr_k)/d(h) for the full correlation:
For shift s = k-(n-1):
- If s >= 0: it's a vector where position j has value (1-h(j-s)) for j>=s, minus h(j+s) for j<=n-1-s
- If s < 0: similar with shifted indices

This can be computed as:
G = Σ_k w_k * [corr_shift(1-h, s_k, 'flip') - corr_shift(h, s_k)]

Where corr_shift gives the appropriate shifted/correlated vector.

Actually, this is getting complicated. Let me just use the O(n^2) approach but vectorize better:

For all positive shifts s = 1, 2, ..., n-1:
- The contribution to grad from shift s (positive) is: w[n-1+s] * [(1-h)[shifted right by s] - h[shifted left by s]]
  where "shifted right by s" means the vector (1-h[s:], zeros(s)) for the first part
  
Actually, let me just do it the straightforward way but precompute (1-h) once:

```python
def compute_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -700, 700)
    w = np.exp(shifted) / np.exp(shifted).sum()
    
    one_minus_h = 1.0 - h
    g = np.zeros(n)
    
    for k in range(2*n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        if s == 0:
            g += wk * (1.0 - 2.0 * h)
        elif s > 0:
            g[s:] += wk * one_minus_h[:n-s]
            g[:n-s] -= wk * h[s:]
        else:
            a = -s
            g[:n-a] += wk * one_minus_h[a:]
            g[a:] -= wk * h[:n-a]
    
    return g
```

This is O(n * 2n) = O(n^2) but with small constant. For n=2048, that's about 4096 * 2048 ≈ 8M element operations, which numpy handles in about 10-20ms.

Now, for the projected gradient method:
1. Compute gradient G
2. Take step: h_new = h - alpha * G
3. Project h_new onto {0≤x≤1, sum(x)=n/2}
4. Check if objective decreased (Armijo condition)
5. If not, halve alpha and retry

The projection onto {0≤x≤1, sum(x)=c}:
- This is: min ||x - y||^2 subject to 0≤x≤1, sum(x)=c
- Solution: x_i = clip(y_i - λ, 0, 1) for the right λ
- Find λ by bisection: sum(clip(y - λ, 0, 1)) = c

Let me code this up:

```python
def project_box_sum(y, n, c):
    """Project y onto {0<=x<=1, sum(x)=c}"""
    # Bisection on lambda
    lo, hi = y.min() - c/n - 1, y.max() - c/n + 1
    # Actually, simpler: lambda ranges from -inf to +inf
    # sum(clip(y - lambda, 0, 1)) is decreasing in lambda
    # At lambda -> -inf: sum = n (all 1s)
    # At lambda -> +inf: sum = 0
    lo, hi = -10.0, 10.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(y - mid, 0, 1).sum()
        if s > c:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(y - lam, 0, 1)
```

This is clean and efficient.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_box_sum(y, n, c):
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > c:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(y - lam, 0.0, 1.0)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    return project_box_sum(h, n, n / 2.0)

def soft_max_val(h, beta):
    corr = compute_corr(h)
    n = len(h)
    dx = 2.0 / n
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    val = (m + np.log(np.sum(np.exp(shifted))) / beta) * dx
    return val

def soft_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    w = np.exp(shifted)
    sw = w.sum()
    if sw < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = w / sw
    one_minus_h = 1.0 - h
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        if s == 0:
            g += wk * (1.0 - 2.0 * h)
        elif s > 0:
            if s < n:
                g[s:] += wk * one_minus_h[:n - s]
                g[:n - s] -= wk * h[s:]
        else:
            a = -s
            if a < n:
                g[:n - a] += wk * one_minus_h[a:]
                g[a:] -= wk * h[:n - a]
    return g

def projected_gradient_descent(h, n, beta, n_steps, step_init, time_limit):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t_start = time.time()
    alpha = step_init
    corr_curr = compute_corr(h)
    m_curr = np.max(corr_curr)

    for it in range(n_steps):
        if time.time() - t_start > time_limit:
            break
        g = soft_max_grad(h, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-16:
            break
        g_normed = g / gn
        for inner in range(30):
            h_trial = project_box_sum(h - alpha * g, n, n / 2.0)
            f_trial = soft_max_val(h_trial, beta)
            f_curr = soft_max_val(h, beta)
            if f_trial <= f_curr - 1e-4 * alpha * gn:
                h = h_trial
                break
            alpha *= 0.5
            if alpha < 1e-12:
                break
        else:
            h = project_box_sum(h - alpha * g, n, n / 2.0)
        alpha = min(alpha * 1.05, step_init)

        if it % 50 == 0:
            c_exact = compute_c5(h)
            if c_exact < best_c5:
                best_c5 = c_exact
                best_h = h.copy()
    c_exact = compute_c5(h)
    if c_exact < best_c5:
        best_c5 = c_exact
        best_h = h.copy()
    return best_h, best_c5

def coordinate_descent(h, n, max_sweeps, time_limit):
    h = h.copy()
    best_c5 = compute_c5(h)
    t_start = time.time()
    target = n / 2.0
    for sweep in range(max_sweeps):
        if time.time() - t_start > time_limit:
            break
        improved = False
        for i in range(n):
            if time.time() - t_start > time_limit:
                break
            old = h[i]
            candidates = [old - 0.1, old - 0.05, old - 0.02, old - 0.01, old - 0.005,
                          old + 0.005, old + 0.01, old + 0.02, old + 0.05, old + 0.1,
                          0.0, 1.0]
            best_val = old
            best_c = best_c5
            for cv in candidates:
                cv = max(0.0, min(1.0, cv))
                if abs(cv - old) < 1e-10:
                    continue
                h[i] = cv
                h = project_box_sum(h, n, target)
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_val = cv
            h[i] = best_val
            h = project_box_sum(h, n, target)
            c_new = compute_c5(h)
            if c_new < best_c5:
                best_c5 = c_new
                improved = True
        if not improved:
            break
    return h, best_c5

def slsqp_polish(h, n, beta, maxiter=300):
    def obj(x):
        return soft_max_val(x, beta)
    def grad(x):
        return soft_max_grad(x, beta)
    def cfun(x):
        return np.sum(x) - n / 2.0
    def cjac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            if compute_c5(h_new) <= compute_c5(h) + 1e-10:
                return h_new
    except Exception:
        pass
    return h.copy()

def interp_to(src_h, n_dst):
    src_n = len(src_h)
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    return np.interp(x_new, xp, src_h)

def random_binary_init(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def optimize_resolution(h_init, n, time_avail, rng):
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t_start = time.time()

    def rem():
        return max(0.0, time_avail - (time.time() - t_start) - 5.0)

    beta_schedule = [100, 500, 2000, 8000, 30000, 100000]
    step_schedule = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002]

    for beta, step in zip(beta_schedule, step_schedule):
        if rem() < 15:
            break
        time_for_phase = min(rem() * 0.5, 120.0)
        n_steps = max(100, int(time_for_phase / 0.02))
        h_new, c_new = projected_gradient_descent(h, n, beta, n_steps, step, time_for_phase)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
            h = h_new.copy()

    if rem() > 30:
        h_cd, c_cd = coordinate_descent(best_h, n, max_sweeps=20, time_limit=min(rem() * 0.4, 120.0))
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd.copy()

    if rem() > 20:
        h_sl = slsqp_polish(best_h, n, 50000, maxiter=200)
        c_sl = compute_c5(h_sl)
        if c_sl < best_c5:
            best_c5 = c_sl
            best_h = h_sl.copy()

    if rem() > 15:
        h_final, c_final = projected_gradient_descent(best_h, n, 200000, 500, 0.001, rem() - 10)
        if c_final < best_c5:
            best_c5 = c_final
            best_h = h_final.copy()

    return best_h, best_c5
```
