def project_feasible(h, n):

    target = n / 2.0

        if abs(diff) < 1e-14:
            break
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if interior.any():
            h[interior] += diff / interior.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    # Final enforcement
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) > 1e-10:

        h = np.clip(h, 0.0, 1.0)
        deficit = target - h.sum()
        # Distribute among non-bound variables
        free = (h > 0) & (h < 1)
        if free.sum() > 0:
            h[free] += deficit / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def to_binary(h, n):

def smooth_max_func(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    s = np.clip(beta * (corr - m), -500.0, 500.0)
    e = np.exp(s)
    return (m + np.log(e.sum()) / beta) * dx


def smooth_max_jac(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    s = np.clip(beta * (corr - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx] * dx
        if wk < 1e-15:
            continue
        off = k_idx - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - x[ao:])
            g[ao:] -= wk * x[:np_]
    return g

def smooth_max_func(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    s = np.clip(beta * (corr - m), -500.0, 500.0)
    e = np.exp(s)
    return (m + np.log(e.sum()) / beta) * dx


def smooth_max_jac(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    s = np.clip(beta * (corr - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx] * dx
        if wk < 1e-15:

            g[:np_] += wk * (1.0 - x[ao:])

    tries = 0
    while flipped < n_flips and tries < n_flips * 50:
        tries += 1
        i = rng.randint(n)
        j = rng.randint(n)
        if h_new[i] == 1.0 and h_new[j] == 0.0:
            h_new[i], h_new[j] = 0.0, 1.0
            flipped += 1
        elif h_new[i] == 0.0 and h_new[j] == 1.0:
            h_new[i], h_new[j] = 1.0, 0.0
            flipped += 1
    return h_new


def optimize_binary_full(h, n, time_limit, rng):
    """Full binary optimization pipeline"""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    # Phase A: SA with high initial temp
    t_a = min(60.0, time_limit * 0.35)
    if t_a > 5:
        h_a, c_a = binary_sa(h, n, t_a, rng, T0=0.08, cooling=0.99997)
        if c_a < best_c5:
            best_c5 = c_a
            best_h = h_a.copy()
        time_limit -= t_a
    # Phase B: Local search
    if time_limit > 10:
        t_b = min(40.0, time_limit * 0.25)
        h_b, c_b = binary_local_search(best_h, n, t_b, rng)
        if c_b < best_c5:
            best_c5 = c_b
            best_h = h_b.copy()
        time_limit -= t_b
    # Phase C: SA with low temp (polish)
    if time_limit > 10:
        t_c = min(50.0, time_limit * 0.5)
        h_c, c_c = binary_sa(best_h, n, t_c, rng, T0=0.02, cooling=0.99999)
        if c_c < best_c5:
            best_c5 = c_c
            best_h = h_c.copy()
    return best_h, best_c5

        return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)
