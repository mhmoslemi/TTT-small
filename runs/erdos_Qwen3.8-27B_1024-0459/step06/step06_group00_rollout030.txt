            break

            # Single swap
            ones = np.where(h_bin == 1.0)[0]
            zeros = np.where(h_bin == 0.0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            h_bin[i] = 0.0
            h_bin[j] = 1.0
        elif move_type == 1:
            # Block swap (swap a block of 1s with a block of 0s)
            block_len = rng.randint(1, max(2, n // 32))
            ones = np.where(h_bin == 1.0)[0]
            zeros = np.where(h_bin == 0.0)[0]
            if len(ones) < block_len or len(zeros) < block_len:
                continue
            # Find a consecutive block of 1s
            i = ones[rng.randint(len(ones))]
            # Extend to find a block
            start_i = i
            while start_i > 0 and h_bin[start_i - 1] == 1.0 and (i - start_i + 1) < block_len:
                start_i -= 1
            if i - start_i + 1 < block_len:
                continue
            # Find a consecutive block of 0s
            j = zeros[rng.randint(len(zeros))]
            start_j = j
            while start_j > 0 and h_bin[start_j - 1] == 0.0 and (j - start_j + 1) < block_len:
                start_j -= 1
            if j - start_j + 1 < block_len:
                continue

            block_1 = h_bin[start_i:start_i+block_len].copy()
            block_0 = h_bin[start_j:start_j+block_len].copy()
            h_bin[start_i:start_i+block_len] = block_0
            h_bin[start_j:start_j+block_len] = block_1
        else:

            i = rng.randint(n)
            j = rng.randint(n)
            if i > j:

            h_bin[i:j+1] = seg[::-1]
            # Check that the number of 1s is preserved (it is, since we're just reversing)

        if no_improve > 5000:
