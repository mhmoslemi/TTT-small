    """Compute delta to correlation array when flipping h[pos]."""
    delta_full = np.zeros(2*n-1)
    h_val = h[pos]  # 0 or 1 for binary
    
    m_min = min(pos, n-1-pos)
    
    if m_min > 0:

        # h[pos+m] for m=1..m_min: h[pos+1:pos+m_min+1]

        if h_val > 0.5:  # 1 -> 0
            delta_full[n:n+m_min] += dm  # positive lags: indices n to n+m_min-1

        else:  # 0 -> 1

            delta_full[n-1-m_min:n] -= dm

        # Let me just compute directly

        m_end = pos
        if m_end >= m_start:

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    n_ones = n // 2
    h_bin = np.zeros(n)
    # Sort by h value, set top n_ones to 1
    idx = np.argsort(-h)
    h_bin[idx[:n_ones]] = 1.0
    return h_bin

def full_corr(h):
    return np.correlate(h, 1.0 - h, mode='full').astype(float)

def golden_ratio_binary(n):
    """Create binary sequence using golden ratio for low discrepancy."""
    n_ones = n // 2
    h = np.zeros(n)
    phi = (1 + np.sqrt(5)) / 2.0
    for i in range(n_ones):
        pos = int((i * phi) % 1.0 * n)
        if h[pos] == 0:
            h[pos] = 1.0
        else:
            # Find next available
            for j in range(1, n):
                p = (pos + j) % n
                if h[p] == 0:
                    h[p] = 1.0
                    break
    return h

def thue_morse_binary(n):
    """Create binary sequence from Thue-Morse."""
    h = np.zeros(n)
    n_ones = n // 2
    # Thue-Morse: t(i) = parity of number of 1-bits in i
    tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    # Select n_ones positions with tm=1 (or 0, whichever is better)
    ones_idx = np.where(tm == 1)[0]
    zeros_idx = np.where(tm == 0)[0]
    if len(ones_idx) >= n_ones:
        selected = ones_idx[:n_ones]
    else:
        selected = np.concatenate([ones_idx, zeros_idx[:n_ones - len(ones_idx)]])
    h[selected] = 1.0
    return h

def slsqp_refine(h, n, K, beta, maxiter=300):
    """SLSQP with smooth max."""
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        top = np.sort(corr)[-K:]
        m = top[-1]
        val = (m + np.log(np.exp(beta * (top - m)).sum()) / beta) * dx
        return val
    
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        idx = np.argsort(corr)[-K:]
        top = corr[idx]
        m = top[-1]
        sh = np.clip(beta * (top - m), -30, 30)
        ex = np.exp(sh)
        w = ex / ex.sum()
        g = np.zeros(n)
        for i in range(K):
            wk = w[i] * dx
            if wk < 1e-15:
                continue
            k = idx[i]
            offset = k - (n - 1)
            if offset >= 0:
                npairs = n - offset
                g[offset:] += wk * (1.0 - x[:npairs])
                g[:npairs] -= wk * x[offset:]
            else:
                a = -offset
                npairs = n - a
                g[:npairs] += wk * (1.0 - x[a:])
                g[a:] -= wk * x[:npairs]
        return g
    
    def cfun(x):
        return np.sum(x) - n / 2.0
    
    def cjac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h
