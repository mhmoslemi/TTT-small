    g = np.zeros(n)

    return g

            cl = np.clip(beta * sh, -50, 50)
            ex = np.exp(cl)
            w = ex / np.sum(ex)
            g_full = np.zeros(n)
            wc = w * (2.0 / n)
            for k in range(len(corr)):
                wk = wc[k]
                if wk < 1e-16:
                    continue
                g_full += wk * grad_corr_k(hh, k, n)
            # Map back to g
            g_half = 0.5 * (g_full[:n_half] + g_full[n_half:])
            return np.clip(g_half, -100, 100)

        bounds = [(0.0, 1.0)] * n_half
        try:
            res = minimize(full_obj, best_g, jac=full_grad, method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': 3000, 'ftol': 1e-20,
                                                    'gtol': 1e-14, 'maxfun': 20000})
            if res.success or res.x is not None:
                g_new = np.clip(res.x, 0, 1)
                s = np.sum(g_new) - n_half / 2.0
                g_new = np.clip(g_new - s / n_half, 0, 1)
                h_new = np.tile(g_new, 2)
                h_new = project_feasible(h_new, n)
                c_new = compute_c5(h_new)
                if c_new < best_c:
                    best_c = c_new
                    best_g = g_new.copy()
        except Exception:
            pass

    h_result = np.tile(best_g, 2)
    h_result = project_feasible(h_result, n)
    return h_result, compute_c5(h_result)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 512
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 0: Try periodic ansatz at n=512 as alternative
    if remaining_time() > 200:
        h_per, c5_per = try_periodic_ansatz(512, 60.0, rng)
        if h_per is not None and c5_per < best_c5:
            best_c5 = c5_per
            best_h = h_per.copy()
            best_n = 512

    # Phase 1: Interpolate to n=1024
    n_work = 1024
    h = interp_to(best_h, best_n, n_work)
    h = project_feasible(h, n_work)
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
        best_n = n_work

    # Phase 2: L-BFGS-B with smooth max (coarse to fine beta)
    for beta in [300, 1000, 3000]:
        if remaining_time() < 30:
            break
        h = lbfgsb_smooth_max(h, n_work, beta, 500.0, maxiter=5000)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n_work

    # Phase 3: Active set optimization
    for K in [30, 20, 15, 10, 8, 6]:
        if remaining_time() < 20:
            break
        h = active_set_lbfgs(h, n_work, K, maxiter=3000)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n_work
            h = best_h.copy()

    # Phase 4: SLSQP with high beta
    for beta in [5000, 10000, 20000]:
        if remaining_time() < 20:
            break
        h = slsqp_smooth(h, n_work, beta, maxiter=500)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n_work
            h = best_h.copy()

    # Phase 5: Equalize shifts
    if remaining_time() > 30:
        h, c5 = equalize_shifts(h, n_work, K=12, max_iters=500)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n_work

    # Phase 6: More L-BFGS with very high beta
    for beta in [15000, 30000, 50000]:
        if remaining_time() < 25:
            break
        h = lbfgsb_smooth_max(h, n_work, beta, 2000.0, maxiter=8000)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n_work
            h = best_h.copy()

    # Phase 7: Subgradient refinement
    if remaining_time() > 60:
        h, c5 = subgradient_refine(h, n_work, min(80.0, remaining_time() * 0.3), rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n_work
            h = best_h.copy()

    # Phase 8: Coordinate descent with fine sweep
    if remaining_time() > 60:
        h, c5 = coordinate_descent_fine(h, n_work, min(90.0, remaining_time() * 0.3), n_sweep=21)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n_work
            h = best_h.copy()

    # Phase 9: Final equalize + polish
    if remaining_time() > 20:
        h, c5 = equalize_shifts(h, n_work, K=8, max_iters=300)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n_work

    if remaining_time() > 30:
        h = slsqp_smooth(h, n_work, 30000, maxiter=800)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n_work

    # Phase 10: Try n=2048 if time permits
    if remaining_time() > 150 and best_n < 2048:
        n_vh = 2048
        h_vh = interp_to(best_h, best_n, n_vh)
        h_vh = project_feasible(h_vh, n_vh)
        c5_vh = compute_c5(h_vh)
        if c5_vh < best_c5:
            # Optimize at higher resolution
            h_vh = lbfgsb_smooth_max(h_vh, n_vh, 3000, 1000.0, maxiter=3000)
            c5_vh = compute_c5(h_vh)
            h_vh = slsqp_smooth(h_vh, n_vh, 10000, maxiter=400)
            c5_vh = compute_c5(h_vh)
            h_vh, c5_vh = equalize_shifts(h_vh, n_vh, K=10, max_iters=300)
            if c5_vh < best_c5:
                best_c5 = c5_vh
                best_h = h_vh.copy()
                best_n = n_vh

    # Phase 11: Final coordinate descent
    if remaining_time() > 40:
        h, c5 = coordinate_descent_fine(best_h, best_n, min(50.0, remaining_time() * 0.5), n_sweep=31)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = best_n

    # Phase 12: Ultra-fine active set
    if remaining_time() > 20:
        for K in [6, 5, 4, 3]:
            if remaining_time() < 10:
                break
            h = active_set_lbfgs(best_h, best_n, K, maxiter=3000)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_n = best_n

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
