    """Solve min max_k corr[k] by iteratively constraining the top shifts."""
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    dx = 2.0 / n
    start = time.time()
    
    for round in range(max_rounds):
        if time.time() - start > time_limit:
            break
        if remaining_time() < 30:
            break
        
        # Identify active set
        corr = compute_corr(h)
        active_idx = np.argsort(-corr)[:top_k]
        
        # Set up SLSQP problem
        def obj(x):
            return x[0]  # minimize t
        
        def obj_grad(x):
            g = np.zeros(n+1)
            g[0] = 1.0
            return g
        
        # Constraints: corr[k_j] - t <= 0 for each active shift
        cons = []
        for k_idx in active_idx:
            cons.append({
                'type': 'ineq',
                'fun': lambda x, ki=k_idx: -(corr_func(x, ki, n) - x[0]),
                'jac': lambda x, ki=k_idx: np.hstack([x[0], -grad_shift(x[1:], ki, n)])
            })

        cons.append({
            'type': 'eq',
            'fun': lambda x: np.sum(x[1:]) - n/2,
            'jac': lambda x: np.hstack([0.0, np.ones(n)])
        })
        
        bounds = [(0, None)] + [(0, 1)]*n
        
        x0 = np.concatenate([[compute_c5(h)/dx], h])

                      bounds=bounds, constraints=cons,

    off = k - (n - 1)
    if off >= 0:
        np_ = n - off
        g[off:] += (1.0 - h[:np_])
        g[:np_] -= h[off:]
    else:
        a = -off
        np_ = n - a
        g[:np_] += (1.0 - h[a:])
        g[a:] -= h[:np_]
    return g

def smooth_max_grad(h, n, beta):
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        g += wk * grad_single_shift(h, k, n)
    return g, m * dx

def projected_gradient_nesterov(h, n, beta, max_iters=5000, time_limit=120):
    """Projected gradient with Nesterov momentum."""
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    y = h.copy()  # Nesterov momentum variable
    momentum = 0.9
    step = 0.01
    start = time.time()
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if remaining_time() < 10:
            break
        
        g, _ = smooth_max_grad(y, n, beta)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        
        # Backtracking line search
        improved = False
        for trial in range(20):
            s = step * (0.5 ** trial)
            h_new = y - s * g
            h_proj = project_feasible(h_new, n)
            c5_new = compute_c5(h_proj)
            if c5_new < best_c5 + 1e-10:  # allow tiny increases for Nesterov
                h_proj_new = project_feasible(h_proj, n)
                # Nesterov update
                y_new = h_proj_new + momentum * (h_proj_new - h)
                y_new = np.clip(y_new, 0, 1)
                h = h_proj_new
                y = y_new
                if c5_new < best_c5 - 1e-15:
                    best_c5 = c5_new
                    best_h = h.copy()
                improved = True
                break
        
        if not improved:
            step *= 0.5
            if step < 1e-8:
                break
        
        if it % 100 == 0:
            momentum = min(0.99, 0.9 + 0.09 * it / max_iters)
    
    return best_h, best_c5

def active_set_opt(h, n, top_k=8, max_rounds=30, time_limit=200):
    """Active set: constrain top-k shifts, solve with SLSQP."""
    global _CUR_N, _SM_BETA
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        if remaining_time() < 30:
            break
        
        corr = compute_corr(h)
        active = np.argsort(-corr)[:top_k]
        
        _CUR_N = n
        _SM_BETA = 500000.0
        
        # Use smooth max but weighted toward active shifts
        def obj(x):
            corr_x = compute_corr(x)
            # Penalty on active shifts
            vals = np.array([corr_x[k] for k in active])
            m = np.max(vals)
            beta = 10000.0
            z = np.clip(beta * (vals - m), -30, 30)
            ez = np.exp(z)
            sm = (m + np.log(np.sum(ez)) / beta)
            return sm * (2.0 / n)
        
        def obj_grad(x):
            corr_x = compute_corr(x)
            vals = np.array([corr_x[k] for k in active])
            m = np.max(vals)
            beta = 10000.0
            z = np.clip(beta * (vals - m), -30, 30)
            ez = np.exp(z)
            S = np.sum(ez)
            w = ez / S
            dx = 2.0 / n
            gx = np.zeros(n)
            for i, k in enumerate(active):
                gx += w[i] * dx * grad_single_shift(x, k, n)
            return gx
        
        def eq_fun(x):
            return np.sum(x) - n / 2.0
        
        def eq_jac(x):
            return np.ones(n)
        
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
        
        try:
            res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': 300, 'ftol': 1e-14})
            if res.x is not None:
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
        except Exception:
            pass
        
        # Check convergence
        corr_new = compute_corr(h)
        new_active = np.argsort(-corr_new)[:top_k]
        if np.array_equal(np.sort(active), np.sort(new_active)):
            if np.max(corr_new) - np.min(corr_new[active]) < 1e-6:
                break
        
        # Gradually increase top_k
        top_k = min(top_k + 2, 20)
    
    return best_h, best_c5

def block_swap_sa(h, n, time_limit=200, rng=None):
    """Binary SA with block swaps."""
    if rng is None:
        rng = np.random.RandomState(42)
    h_bin = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        sort_idx = np.argsort(-h_bin)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0
    
    dx = 2.0 / n
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    curr_c5 = best_c5
    start = time.time()
    
    T_schedule = [(0.3, 2.0/dx), (0.7, 0.3/dx), (1.0, 0.05/dx)]
    # (fraction of time, temperature)
    
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    no_improve = 0
    total_budget = time_limit
    
    for frac, T in T_schedule:
        t_target = start + frac * total_budget
        block_sizes = [1, 2, 3, 4, 5] if T > 0.5/dx else [1, 2, 3, 4]
        
        while time.time() < min(t_target, start + total_budget) and remaining_time() > 10:
            # Choose block swap
            L = block_sizes[rng.randint(len(block_sizes))]
            # Pick two blocks to swap
            i_start = rng.randint(0, n - L + 1)
            j_start = rng.randint(0, n - L + 1)
            if abs(i_start - j_start) < L:
                continue
            
            i_block = np.arange(i_start, i_start + L)
            j_block = np.arange(j_start, j_start + L)
            
            # Ensure we're swapping 1-block with 0-block
            if np.sum(h_bin[i_block]) == 0 or np.sum(h_bin[j_block]) == L:
                continue
            if np.sum(h_bin[i_block]) == L and np.sum(h_bin[j_block]) == 0:
                # Valid swap: move 1-block to 0-region
                h_bin[i_block], h_bin[j_block] = h_bin[j_block].copy(), h_bin[i_block].copy()
            elif np.sum(h_bin[i_block]) == 0 and np.sum(h_bin[j_block]) == L:
                h_bin[i_block], h_bin[j_block] = h_bin[j_block].copy(), h_bin[i_block].copy()
            else:
                continue
            
            new_c5 = compute_c5(h_bin)
            delta = new_c5 - curr_c5
            
            if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / max(T, 1e-10))):
                curr_c5 = new_c5
