def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:

        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def _sm_obj(x):

    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)

def _sm_grad(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    return g


def _eq_fun(x):
    return np.sum(x) - _CUR_N / 2.0


def _eq_jac(x):

def slsqp_opt(h, n, beta, maxiter=2000):
    global _CUR_N, _SM_BETA
    _CUR_N = n
    _SM_BETA = beta
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_sm_obj, h0, jac=_sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.fun < _sm_obj(h0) - 1e-15:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def grad_single_shift(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        np_ = n - off
        g[off:] += (1.0 - h[:np_])
        g[:np_] -= h[off:]
    else:
        a = -off
        np_ = n - a
        g[:np_] += (1.0 - h[a:])
        g[a:] -= h[:np_]
    return g

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)


def structured_pattern(n, p, q):
    """h[i]=1 if i mod p < q."""
    h = np.zeros(n)
    for i in range(n):
        if (i % p) < q:
            h[i] = 1.0
    return project_feasible(h, n)


def subgradient_accelerated(h, n, max_iters=5000, time_limit=90):
    """Nesterov-accelerated subgradient with zigzag for non-smooth max."""
    h = project_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()

    y = h.copy()
    beta_nest = 0.5
    k_curr = None
    k_prev = None

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if remaining_time() < 5:
            break

        corr = np.correlate(y, 1.0 - y, mode='full')
        k_curr = int(np.argmax(corr))

        if k_prev is not None and k_curr != k_prev:
            g = 0.5 * (grad_single_shift(y, k_curr, n) + grad_single_shift(y, k_prev, n))
        else:
            g = grad_single_shift(y, k_curr, n)
        k_prev = k_curr

        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g = g / gn

        t_nest = 2.0 / (it + 2.0)
        step = 0.01 * t_nest

        x_new = project_feasible(y - step * g, n)

        if compute_c5(x_new) < best_c5 - 1e-15:
            best_c5 = compute_c5(x_new)
            best_h = x_new.copy()

        y = x_new + beta_nest * (x_new - h)
        y = project_feasible(y, n)
        h = x_new
        beta_nest = 0.5 + beta_nest * 0.25

    return best_h, best_c5

        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, n_final)
