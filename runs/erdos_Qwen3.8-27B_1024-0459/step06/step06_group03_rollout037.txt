from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            break
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        m = n - offset
        g[offset:] += (1.0 - h[:m])
        g[:m] -= h[offset:]
    else:
        a = -offset
        m = n - a
        g[:m] += (1.0 - h[a:])
        g[a:] -= h[:m]
    return g


def smax_objective(h):
    n = _G_N
    beta = _G_BETA
    K = _G_K
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(-corr)[:K]
    c = corr[idx] * dx
    m = c.max()
    sh = beta * (c - m)
    cl = np.clip(sh, -80.0, 80.0)
    ex = np.exp(cl)
    return m + np.log(ex.sum()) / beta


def smax_gradient(h):
    n = _G_N
    beta = _G_BETA
    K = _G_K
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(-corr)[:K]
    c = corr[idx] * dx
    m = c.max()
    sh = beta * (c - m)
    cl = np.clip(sh, -80.0, 80.0)
    ex = np.exp(cl)
    w = ex / ex.sum()
    g = np.zeros(n)
    for i in range(K):
        if w[i] < 1e-15:
            continue
        g += (w[i] * dx) * grad_corr_k(h, idx[i], n)
    return g


def sum_eq_fun(h):
    return np.sum(h) - _G_N / 2.0


def sum_eq_jac(h):
    return np.ones(_G_N)


def vec_ineq_fun(v):

    t = v[-1]
    n = _G_N
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * corr[_G_KLIST] - t


def vec_ineq_jac(v):
    h = v[:_G_N]
    n = _G_N
    dx = 2.0 / n
    K_list = _G_KLIST
    J = np.zeros((len(K_list), n + 1))
    for i, k in enumerate(K_list):
        g = grad_corr_k(h, k, n)
        J[i, :n] = dx * g
        J[i, -1] = -1.0
    return J

def cutting_plane_lp(h, n, K, delta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    c_max_val = np.max(corr) * dx
    idx = np.argsort(-corr)[:K]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub_list = []
    b_ub_list = []
    for k in idx:
        g = grad_corr_k(h, k, n) * dx
        fk = corr[k] * dx
        # Linearization: f_k(h_new) ≈ f_k(h) + g·(h_new - h) = g·h_new + (fk - g·h)
        # Constraint: g·h_new + (fk - g·h) <= t
        # => g·h_new - t <= -(fk - g·h)
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_ub_list.append(row)
        b_ub_list.append(-(fk - np.dot(g, h)))
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    I = np.eye(n)
    A_trust = np.vstack([np.hstack([I, np.zeros((n, 1))]),
                         np.hstack([-I, np.zeros((n, 1))])])
    b_trust = np.concatenate([h + delta, h + delta])
    A_ub = np.vstack([A_ub, A_trust])
    b_ub = np.concatenate([b_ub, b_trust])
    bounds = [(0, 1)] * n + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return h.copy(), c_max_val

        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        progress = min(1.0, it / 300.0)
        K = 30 if progress < 0.3 else (20 if progress < 0.6 else 10)

        corr = np.correlate(h, 1.0 - h, mode='full')

        idx = np.argsort(-corr)[:K]

        for i in range(K):

            g += (w[i] * dx) * grad_corr_k(h, idx[i], n)
        gn = np.linalg.norm(g)
