def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):

        if abs(diff) < 1e-15:
            break

        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        np_ = n - offset
        g[offset:offset + np_] += (1.0 - h[:np_])
        g[:np_] -= h[offset:offset + np_]
    else:
        a = -offset
        np_ = n - a
        g[a:] += (1.0 - h[:np_])
        g[:np_] -= h[a:]
    return g


def smooth_max_obj(h):

    K = _G_K

    ex = np.exp(cl)
    return (cmax + np.log(ex.sum()) / beta) * (2.0 / n)


def smooth_max_grad(h):
    n = _G_N
    beta = _G_BETA
    K = _G_K
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(-corr)[:K]

    for i in range(K):

        g += w[i] * grad_corr_k(h, int(idx[i]), n)

                       bounds=bounds, constraints=cons,

            # Quick SLSQP
            h_try = slsqp_optimize(h_try, best_n, 100, K=25, maxiter=200, ftol=1e-15)
            c5_try = compute_c5(h_try)
            h_try = slsqp_optimize(h_try, best_n, 500, K=25, maxiter=200, ftol=1e-16)
            c5_try = compute_c5(h_try)
            if remaining() > 10:
                h_try, c5_try = fine_polish(h_try, best_n, time_limit=8)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()

    # Phase 7: Final equalize + tiny steps
    if remaining() > 10:
        h_eq2, c5_eq2 = equalize_shifts(best_h, best_n, K=6,
                                        time_limit=min(8, remaining() * 0.3),
                                        max_iters=300)
        if c5_eq2 < best_c5:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()

    if remaining() > 5:
        h_fp2, c5_fp2 = fine_polish(best_h, best_n, time_limit=5)
        if c5_fp2 < best_c5:
            best_c5 = c5_fp2
            best_h = h_fp2.copy()

    # Final feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            best_c5 = min(best_c5, float(eval_c5))
    except Exception:
        pass

    return (best_h, best_c5, best_n)
