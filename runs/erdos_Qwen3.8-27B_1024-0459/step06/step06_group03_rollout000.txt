                g += w[i] * grad_corr_k(h, idx[i], n)
        g *= _G_DX

        gn = np.linalg.norm(g)
        if gn < 1e-16:
            break

        if s_prev is not None and g_prev is not None:
            sg = np.dot(s_prev, g - g_prev)
            ss = np.dot(s_prev, s_prev)
            if abs(sg) > 1e-14:
                alpha_bb = ss / abs(sg)
                alpha = np.clip(alpha_bb, 1e-5, 0.4)

        h_old = h.copy()
        h_new = project_feasible(h - alpha * g, n)
        c5_new = compute_c5(h_new)

        s_prev = h_new - h_old
        g_new = np.zeros_like(g)
        corr_new = compute_corrs(h_new)
        idx_new = np.argsort(-corr_new)[:_G_K]
        c_new = corr_new[idx_new]
        sh_new = c_new - c_new[0]
        cl_new = np.clip(_G_BETA * sh_new, -60.0, 60.0)
        ex_new = np.exp(cl_new)
        w_new = ex_new / ex_new.sum()
        for i in range(_G_K):
            if w_new[i] > 1e-15:
                g_new += w_new[i] * grad_corr_k(h_new, idx_new[i], n)
        g_new *= _G_DX
        g_prev = g_new

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            stall = 0
        else:
            stall += 1
            improved = False
            for f in [0.5, 0.25, 0.1, 0.05]:
                h_try = project_feasible(h_old - alpha * f * g, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    s_prev = h - h_old
                    improved = True
                    stall = 0
                    break
            if not improved:
                alpha *= 0.5
                if stall > 50:
                    # Reheat: add small noise
                    noise = rng.randn(n) * 0.005
                    h = project_feasible(h + noise, n)
                    best_h = h.copy()
                    best_c5 = compute_c5(h)
                    stall = 0

        it += 1

    _G_N = 0
    return best_h, best_c5


def slsqop_smax(h, n, beta, K, maxiter):
    global _G_N, _G_BETA, _G_K, _G_DX
    _G_N = n
    _G_BETA = beta
    _G_K = K
    _G_DX = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}]
    try:
        res = minimize(smax_obj, h, jac=smax_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            if compute_c5(h_new) <= compute_c5(h) + 1e-12:
                return h_new
    except Exception:
        pass
    _G_N = 0
    return h.copy()


def slsqop_tminmax(h, n, K_list, maxiter):
    global _G_N, _G_KLIST, _G_DX
    _G_N = n
    _G_KLIST = np.array(K_list, dtype=int)
    _G_DX = 2.0 / n
    t0 = compute_c5(h)
    v0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'ineq', 'fun': tmin_ineq_fun, 'jac': tmin_ineq_jac},
        {'type': 'eq', 'fun': sum_eq_fun2, 'jac': sum_eq_jac2}
    ]
    try:
        res = minimize(tmin_obj, v0, jac=tmin_obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x[:n], n)
            c5_new = compute_c5(h_new)
            if c5_new <= compute_c5(h) + 1e-12:
                return h_new, c5_new
    except Exception:
        pass
    _G_N = 0
    return h.copy(), compute_c5(h)


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def equalize_top(h, n, K, max_iters):
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        corr = compute_corrs(h)
        idx = np.argsort(-corr)[:K]
        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        if c_max - c_min < 1e-12:
            break
        g_avg = np.zeros(n)
        for k in idx:
            g_avg += grad_corr_k(h, k, n)
        g_avg /= K
        g_avg *= (2.0 / n)
        gn = np.linalg.norm(g_avg)
        if gn < 1e-15:
            break
        improved = False
        for step in [0.03, 0.01, 0.005, 0.02, 0.015, 0.008, 0.05, 0.003]:
            ht = project_feasible(h - step * g_avg, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-15:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def micro_polish(h, n, time_limit, rng):
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        rem = _BUDGET - (time.time() - _T0)
        if rem < 3:
            break
        corr = compute_corrs(h)
        idx = np.argsort(-corr)[:12]
        g = np.zeros(n)
        for k in idx:
            g += grad_corr_k(h, k, n)
        g /= len(idx)
        g *= (2.0 / n)
        gn = np.linalg.norm(g)
        if gn < 1e-16:
            break
        improved = False
        for step in [0.003, 0.001, 0.0005, 0.002, 0.0008, 0.0015]:
            ht = project_feasible(h - step * g / gn, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-16:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            # Try random micro-perturbation
            for _ in range(5):
                noise = rng.randn(n) * 0.001
                ht = project_feasible(h + noise, n)
                cn = compute_c5(ht)
                if cn < best_c5:
                    h = ht
                    best_c5 = cn
                    improved = True
                    break
        it += 1
    return h, best_c5


def symmetry_transform(h):
    """Apply h(x) -> 1-h(2-x)."""
    return 1.0 - h[::-1].copy()


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _G_N, _G_BETA, _G_K, _G_KLIST, _G_DX
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
    except Exception:
        parent_n = 512
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0

    best_h = project_feasible(parent_h, parent_n)
    best_c5 = compute_c5(best_h)
    best_n = parent_n

    # Phase 1: Perturbation search for better basin (30s)
    if remaining() > 30:
        perturb_time = min(25.0, remaining() * 0.05)
        n = best_n
        candidates = [best_h.copy()]
        # Symmetry
        h_sym = project_feasible(symmetry_transform(best_h), n)
        candidates.append(h_sym)
        # Random perturbations
        for level in [0.02, 0.01, 0.05, 0.015]:
            noise = rng.randn(n) * level
            h_p = project_feasible(best_h + noise, n)
            candidates.append(h_p)
        # Smooth then re-optimize
        from scipy.ndimage import gaussian_filter1d
        h_smooth = gaussian_filter1d(best_h, sigma=3)
        h_smooth = project_feasible(h_smooth, n)
        candidates.append(h_smooth)
        h_smooth2 = gaussian_filter1d(best_h, sigma=1)
        h_smooth2 = project_feasible(h_smooth2, n)
        candidates.append(h_smooth2)

        best_pert = best_c5
        for h_cand in candidates:
            c5_cand = compute_c5(h_cand)
            if c5_cand < best_pert:
                best_pert = c5_cand
                best_h = h_cand.copy()
                best_c5 = best_pert

    # Phase 2: BB gradient descent at parent resolution (150s)
    n = best_n
    h = best_h.copy()
    if remaining() > 100:
        sg_time = min(120.0, remaining() * 0.2)
        h_bb, c5_bb = bb_gradient_descent(h, n, sg_time, rng)
        if c5_bb < best_c5:
            best_c5 = c5_bb
            best_h = h_bb.copy()
            h = h_bb.copy()

    # Phase 3: SLSQP smooth-max stages (250s)
    if remaining() > 100:
        beta_stages = [20, 50, 100, 300, 1000, 3000, 8000, 20000, 50000]
        for beta in beta_stages:
            if remaining() < 15:
                break
            maxit = 500 if beta < 500 else 300
            K = 30 if beta < 1000 else 40
            h_new = slsqop_smax(h, n, beta, K, maxit)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 4: Active-set min-max (150s)
    if remaining() > 60:
        for K_active in [5, 8, 12, 16, 20, 30, 40]:
            if remaining() < 10:
                break
            corr = compute_corrs(h)
            idx_active = np.argsort(-corr)[:K_active]
            h_am, c5_am = slsqop_tminmax(h, n, idx_active, maxiter=300)
            if c5_am < best_c5:
                best_c5 = c5_am
                best_h = h_am.copy()
                h = h_am.copy()

    # Phase 5: Equalize at parent resolution (60s)
    if remaining() > 40:
        h_eq, c5_eq = equalize_top(h, n, K=10, max_iters=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 6: Try n=1024 (300s)
    n_high = 1024
    if remaining() > 150:
        h_hi = interp_to(best_h, best_n, n_high)
        h_hi = project_feasible(h_hi, n_high)
        c5_hi = compute_c5(h_hi)
        h_opt_hi = h_hi.copy()

        if c5_hi <= best_c5 + 0.002:
            # BB at high res
            if remaining() > 80:
                h_hi, c5_hi = bb_gradient_descent(h_hi, n_high, min(60.0, remaining() * 0.2), rng)

            # SLSQP at high res
            for beta in [30, 100, 500, 2000, 8000]:
                if remaining() < 15:
                    break
                K = 30 if beta < 500 else 40
                h_hi = slsqop_smax(h_hi, n_high, beta, K, maxiter=300)
                c5_hi = compute_c5(h_hi)

            # Active set at high res
            if remaining() > 40:
                for K_a in [8, 15, 25]:
                    if remaining() < 10:
                        break
                    corr_hi = compute_corrs(h_hi)
                    idx_a = np.argsort(-corr_hi)[:K_a]
                    h_hi, c5_hi = slsqop_tminmax(h_hi, n_high, idx_a, maxiter=250)

            # Equalize at high res
            if remaining() > 20:
                h_hi, c5_hi = equalize_top(h_hi, n_high, K=10, max_iters=200)
                c5_hi = compute_c5(h_hi)

            if c5_hi < best_c5:
                best_c5 = c5_hi
                best_h = h_hi.copy()
                best_n = n_high

    # Phase 7: Intensive cycling at best resolution (200s)
    n = best_n
    h = best_h.copy()
    cycle_time = min(180.0, remaining() * 0.3)
    if cycle_time > 30:
        cstart = time.time()
        K_cycle = [5, 8, 12, 16, 20, 25, 30, 20, 10, 8]
        ci = 0
        while time.time() - cstart < cycle_time:
            rem = _BUDGET - (time.time() - _T0)
            if rem < 10:
                break
            K_cur = K_cycle[ci % len(K_cycle)]
            # Equalize
            h_eq, c5_eq = equalize_top(h, n, K=K_cur, max_iters=50)
            if c5_eq < best_c5:
                best_c5 = c5_eq
                best_h = h_eq.copy()
            h = h_eq

            # Small SLSQP
            if rem > 15:
                beta_cur = 2000 + 1000 * ci
                h_s = slsqop_smax(h, n, beta_cur, K_cur, maxiter=200)
                c5_s = compute_c5(h_s)
                if c5_s < best_c5:
                    best_c5 = c5_s
                    best_h = h_s.copy()
                h = h_s

            # BB step
            if rem > 20:
                h_bb, c5_bb = bb_gradient_descent(h, n, min(10.0, rem * 0.15), rng)
                if c5_bb < best_c5:
                    best_c5 = c5_bb
                    best_h = h_bb.copy()
                h = h_bb

            ci += 1

    # Phase 8: Final micro-polish (80s)
    if remaining() > 30:
        polish_time = min(60.0, remaining() * 0.4)
        h_mp, c5_mp = micro_polish(best_h, best_n, polish_time, rng)
        if c5_mp < best_c5:
            best_c5 = c5_mp
            best_h = h_mp.copy()

    # Phase 9: Final SLSQP with very high beta
    if remaining() > 20:
        h_f = slsqop_smax(best_h, best_n, 100000, 50, maxiter=500)
        c5_f = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 10: Last micro steps
    if remaining() > 15:
        h_mp2, c5_mp2 = micro_polish(best_h, best_n, min(12.0, remaining() * 0.5), rng)
        if c5_mp2 < best_c5:
            best_c5 = c5_mp2
            best_h = h_mp2.copy()

    # Final feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
