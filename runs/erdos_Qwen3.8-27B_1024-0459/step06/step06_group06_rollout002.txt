import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        if npairs > 0:
            g[offset:n] += (1.0 - h[:npairs])
            g[:npairs] -= h[offset:n]
    else:
        a = -offset
        npairs = n - a
        if npairs > 0:
            g[:npairs] += (1.0 - h[a:n])
            g[a:n] -= h[:npairs]
    return g


def smooth_max_obj(x, n, beta, penalty):
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(beta * sh, -500.0, 500.0)
    ex = np.exp(cl)
    sm = (m + np.log(np.sum(ex) + 1e-30) / beta) * (2.0 / n)
    s = np.sum(x) - n / 2.0
    return sm + 0.5 * penalty * s * s


def smooth_max_grad(x, n, beta, penalty):
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(beta * sh, -500.0, 500.0)
    ex = np.exp(cl)
    w = ex / (np.sum(ex) + 1e-30)
    g = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk > 1e-12:
            g += wk * (2.0 / n) * grad_corr_k(x, k, n)
    s = np.sum(x) - n / 2.0
    g += penalty * s
    return g


def lbfgsb_opt(h, n, beta, penalty, maxiter):
    global _T0, _BUDGET
    def obj_fun(x):
        return smooth_max_obj(x, n, beta, penalty)

    def grad_fun(x):
        return smooth_max_grad(x, n, beta, penalty)

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fun, h, jac=grad_fun, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-22,
                                'gtol': 1e-16, 'maxfun': maxiter * 8})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def slsqp_opt(h, n, beta, maxiter):
    global _T0, _BUDGET

    def obj_fun(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -500.0, 500.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex) + 1e-30) / beta) * (2.0 / n)

    def grad_fun(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -500.0, 500.0)
        ex = np.exp(cl)
        w = ex / (np.sum(ex) + 1e-30)
        g = np.zeros(n)
        for k in range(len(corr)):
            wk = w[k]
            if wk > 1e-12:
                g += wk * (2.0 / n) * grad_corr_k(x, k, n)
        return g

    def sc_fun(x):
        return np.sum(x) - n / 2.0

    def sc_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sc_fun, 'jac': sc_jac}]
    try:
        res = minimize(obj_fun, h, jac=grad_fun, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def active_set_descend(h, n, time_limit, K=40, beta_w=80.0):
    global _T0, _BUDGET
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        it += 1
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:K]
        c_top = corr[idx]
        weights = np.exp(beta_w * (c_top - c_top.max()))
        sw = weights.sum()
        if sw < 1e-30:
            weights = np.ones(K) / K
        else:
            weights /= sw
        g = np.zeros(n)
        for i, k in enumerate(idx):
            if weights[i] > 1e-14:
                g += weights[i] * grad_corr_k(h, k, n)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        improved = False
        steps = [0.08, 0.04, 0.02, 0.01, 0.05, 0.005, 0.015, 0.03, 0.002]
        for step in steps:
            h_try = project_feasible(h - step * g, n)
            c_try = compute_c5(h_try)
            if c_try < best_c5 - 1e-16:
                h = h_try
                best_c5 = c_try
                improved = True
                break
        if not improved:
            for step in [0.04, 0.02, 0.01, 0.005]:
                h_try = project_feasible(h + step * g, n)
                c_try = compute_c5(h_try)
                if c_try < best_c5 - 1e-16:
                    h = h_try
                    best_c5 = c_try
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5


def equalize_push(h, n, K=12, time_limit=30.0):
    global _T0, _BUDGET

        c_min_val = corr[idx[-1]]

            break
        g = np.zeros(n)
        for k in idx:
            g -= grad_corr_k(h, k, n)

def block_coordinate(h, n, time_limit, block_size=8):
    global _T0, _BUDGET
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    nblocks = n // block_size
    while time.time() - start < time_limit:
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        improved = False
        for b in range(nblocks):
            if time.time() - start > time_limit:
                break
            s = b * block_size
            e = s + block_size
            orig_block = h[s:e].copy()
            sum_rest = np.sum(h[:s]) + np.sum(h[e:]) if e < n else np.sum(h[:s])
            target_block_sum = n / 2.0 - sum_rest
            # Try perturbing the block
            for trial in range(4):
                if trial == 0:
                    pert = rng_perturb(orig_block, 0.02)
                elif trial == 1:
                    pert = rng_perturb(orig_block, 0.05)
                elif trial == 2:
                    pert = np.clip(orig_block + (1.0 - orig_block) * 0.1, 0.0, 1.0)
                else:
                    pert = np.clip(orig_block - orig_block * 0.1, 0.0, 1.0)
                pert = np.clip(pert, 0.0, 1.0)
                diff = target_block_sum - np.sum(pert)
                mask = (pert > 1e-12) & (pert < 1.0 - 1e-12)
                if mask.sum() > 0:
                    pert[mask] += diff / mask.sum()
                else:
                    pert[:] += diff / block_size
                pert = np.clip(pert, 0.0, 1.0)
                h_new = h.copy()
                h_new[s:e] = pert
                h_new = project_feasible(h_new, n)
                c_try = compute_c5(h_new)
                if c_try < best_c5 - 1e-16:
                    h = h_new
                    best_c5 = c_try
                    improved = True
                    break
            else:
                h[s:e] = orig_block
        if not improved:
            break
    return h, best_c5


def rng_perturb(block, amp):
    return block + np.random.uniform(-amp, amp, size=block.shape)


def coordinate_descent(h, n, time_limit):
    global _T0, _BUDGET
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    passed_full = 0
    while time.time() - start < time_limit:
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        improved = False
        for i in range(n):
            if time.time() - start > time_limit:
                break
            old_val = h[i]
            candidates = []
            if old_val < 0.05:
                candidates = [0.0, 0.02, 0.05, 0.1]
            elif old_val > 0.95:
                candidates = [1.0, 0.98, 0.95, 0.9]
            else:
                candidates = [0.0, 1.0, 0.5,
                              max(0.0, old_val - 0.05),
                              min(1.0, old_val + 0.05),
                              max(0.0, old_val - 0.01),
                              min(1.0, old_val + 0.01)]
            best_c = best_c5
            best_v = old_val
            for cv in candidates:
                cv = float(np.clip(cv, 0.0, 1.0))
                if abs(cv - old_val) < 1e-12:
                    continue
                h[i] = cv
                h = project_feasible(h, n)
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_v = cv
            h[i] = best_v
            h = project_feasible(h, n)
            c_now = compute_c5(h)
            if c_now < best_c5 - 1e-15:
                best_c5 = c_now
                improved = True
        if not improved:
            passed_full += 1
            if passed_full >= 2:
                break
        else:
            passed_full = 0
    return h, best_c5


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

    return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)
