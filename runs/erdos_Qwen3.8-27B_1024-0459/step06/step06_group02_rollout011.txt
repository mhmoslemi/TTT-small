def topk_reduction(h, n, K=5, steps=500, step_size=0.01):

        g /= K
        # Project gradient to feasible space (remove mean component)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_lag(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        np_ = n - offset
        if np_ > 0:
            g[offset:] += (1.0 - h[:np_])
            g[:np_] -= h[offset:]
    else:
        a = -offset
        np_ = n - a
        if np_ > 0:
            g[:np_] += (1.0 - h[a:])
            g[a:] -= h[:np_]
    return g

def lse_obj(h, beta):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50.0, 50.0)
    return (m + np.log(np.exp(shifted).sum()) / beta) * (2.0 / n)

def lse_grad(h, beta):

    corr = compute_corr(h)
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(shifted)
    s = ex.sum()
    w = ex / s
    g = np.zeros(n)
    dx = 2.0 / n
    for k in range(len(corr)):
        wk = w[k] * dx
        if wk > 1e-13:
            g += wk * grad_lag(h, k, n)
    return g

    corr = compute_corr(h)

            best_h = h_tk.copy()
        if remaining() > 20:
            h_s = slsqp_lse(best_h, n, 100000, maxiter=200)
            c_s = compute_c5(h_s)
            if c_s < best_c5:
                best_c5 = c_s
                best_h = h_s.copy()

    # Phase 4: Upsample to 1024 and optimize
    if remaining() > 150:
        n_up = 1024
        h_up = project_feasible(interp_to(best_h, best_n, n_up), n_up)
        c_up = compute_c5(h_up)
        for beta in [2000, 20000, 80000, 200000]:
            if remaining() < 30:
                break
            h_new = slsqp_lse(h_up, n_up, beta, maxiter=300)
            c_new = compute_c5(h_new)
            if c_new < c_up:
                c_up = c_new
                h_up = h_new
        for p in [32, 64, 128]:
            if remaining() < 25:
                break
            h_new = slsqp_power(h_up, n_up, p, maxiter=250)
            c_new = compute_c5(h_new)
            if c_new < c_up:
                c_up = c_new
                h_up = h_new
        h_tk2, c_tk2 = topk_reduce(h_up, n_up, K=5, nsteps=200, init_step=0.003)
        if c_tk2 < c_up:
            c_up = c_tk2
            h_up = h_tk2
        if c_up < best_c5:
            best_c5 = c_up
            best_h = h_up.copy()
            best_n = n_up

    # Phase 5: Block coordinate descent
    if remaining() > 80:
        h_bc, c_bc = block_cd(best_h, best_n, block_size=32, n_sweeps=4, beta=50000)
        if c_bc < best_c5:
            best_c5 = c_bc
            best_h = h_bc.copy()
        h_s = slsqp_lse(best_h, best_n, 100000, maxiter=200)
        c_s = compute_c5(h_s)
        if c_s < best_c5:
            best_c5 = c_s
            best_h = h_s.copy()

    # Phase 6: Fourier perturbations + SLSQP
    for p in range(8):
        if remaining() < 35:
            break
        scale = 0.003 + p * 0.002
        h_p = fourier_perturb(best_h, best_n, rng, scale=scale)
        h_s = slsqp_lse(h_p, best_n, 50000, maxiter=250)
        c_s = compute_c5(h_s)
        if c_s < best_c5:
            best_c5 = c_s
            best_h = h_s.copy()

    # Phase 7: Subgradient + SLSQP alternation
    if remaining() > 120:
        h_sg = best_h.copy()
        n_sg = best_n
        for it in range(2000):
            if time.time() - _t0 > _budget - 60:
                break
            progress = it / 2000.0
            K = 15 if progress < 0.3 else (8 if progress < 0.7 else 4)
            beta_g = 40.0 if progress < 0.5 else 80.0
            step = 0.004 / (1.0 + 0.03 * it)
            g = subgrad_step(h_sg, n_sg, K=K, beta=beta_g)
            g -= np.mean(g)
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break
            h_new = project_feasible(h_sg - step * g / gn, n_sg)
            c_new = compute_c5(h_new)
            c_cur = compute_c5(h_sg)
            if c_new < c_cur - 1e-14:
                h_sg = h_new
            elif it % 300 == 299 and remaining() > 40:
                h_pol = slsqp_lse(h_sg, n_sg, 150000, maxiter=250)
                c_pol = compute_c5(h_pol)
                if c_pol < best_c5:
                    best_c5 = c_pol
                    best_h = h_pol.copy()
                h_sg = h_pol

    # Phase 8: Final high-beta SLSQP passes
    for beta in [200000, 500000, 1000000]:
        if remaining() < 15:
            break
        h_new = slsqp_lse(best_h, best_n, beta, maxiter=500)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Phase 9: Tiny perturbations at very small scale
    for p in range(5):
        if remaining() < 20:
            break
        h_p = fourier_perturb(best_h, best_n, rng, scale=0.001 + p * 0.001)
        h_s = slsqp_lse(h_p, best_n, 200000, maxiter=350)
        c_s = compute_c5(h_s)
        if c_s < best_c5:
            best_c5 = c_s
            best_h = h_s.copy()

    # Phase 10: Final top-K squeeze
    if remaining() > 20:
        h_tk, c_tk = topk_reduce(best_h, best_n, K=3, nsteps=500, init_step=0.001)
        if c_tk < best_c5:
            best_c5 = c_tk
            best_h = h_tk.copy()

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
