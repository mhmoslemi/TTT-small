def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_corr_k_vec(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        np_ = n - offset
        g[offset:] += (1.0 - h[:np_])
        g[:np_] -= h[offset:]
    else:
        a = -offset
        np_ = n - a
        g[:np_] += (1.0 - h[a:])
        g[a:] -= h[:np_]
    return g


def smooth_max_val(h, n, beta, K):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(-corr)[:K]
    c = corr[idx]
    cmax = c.max()
    shifted = beta * (c - cmax)
    shifted = np.clip(shifted, -80.0, 80.0)
    ex = np.exp(shifted)
    val = cmax + np.log(ex.sum()) / beta
    return val * (2.0 / n), idx, c


def smooth_max_grad(h, n, beta, K):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(-corr)[:K]
    c = corr[idx]
    cmax = c.max()
    shifted = beta * (c - cmax)
    shifted = np.clip(shifted, -80.0, 80.0)
    ex = np.exp(shifted)
    w = ex / ex.sum()
    dx_val = 2.0 / n
    g = np.zeros(n)
    for i in range(K):
        if w[i] < 1e-15:
            continue
        wk = w[i] * dx_val
        k = idx[i]
        offset = k - (n - 1)
        if offset >= 0:
            np_ = n - offset
            g[offset:] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[offset:]
        else:
            a = -offset
            np_ = n - a
            g[:np_] += wk * (1.0 - h[a:])
            g[a:] -= wk * h[:np_]
    return g

    _G_BETA = beta
    _G_K = K

def cutplane_step(h, n, K, delta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(-corr)[:K]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub_list = []
    b_ub_list = []
    for k in idx:
        g = grad_corr_k_vec(h, k, n)
        fk = corr[k]
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_ub_list.append(row)
        b_ub_list.append(-fk + np.dot(g, h))
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    I = np.eye(n)
    A_trust = np.vstack([np.hstack([I, np.zeros((n, 1))]),
                         np.hstack([-I, np.zeros((n, 1))])])
    b_trust = np.concatenate([h + delta, h + delta])
    A_ub_full = np.vstack([A_ub, A_trust])
    b_ub_full = np.concatenate([b_ub, b_trust])
    bounds = [(0, 1)] * n + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub_full, b_ub=b_ub_full, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n], res.x[-1], corr[idx[0]]
    except Exception:
        pass
    return h.copy(), corr[idx[0]], corr[idx[0]]


def interp_to(src_h, src_n, dst_n):

    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

        h, c5_9 = nesterov_smooth_max(h, n, min(30.0, remaining() * 0.4),
                                      beta_start=500, beta_end=3000, K_start=50, K_end=100)
        if c5_9 < best_c5:
            best_c5 = c5_9
            best_h = h.copy()
            h = h.copy()

    if remaining() > 20:
        for beta in [10000, 30000]:
            if remaining() < 8:
                break
            h, c5_10 = slsqp_optimize(h, n, beta, K=60, maxiter=400)
            if c5_10 < best_c5:
                best_c5 = c5_10
                best_h = h.copy()
                h = h.copy()

    if remaining() > 15:
        h, c5_11 = equalize_top(best_h, n, K=12, max_iters=200, time_limit=min(15.0, remaining() * 0.5))
        if c5_11 < best_c5:
            best_c5 = c5_11
            best_h = h.copy()

    # Phase 10: Micro-steps
    if remaining() > 10:
        h_t = best_h.copy()
        t10_start = time.time()
        t10_limit = min(15.0, remaining() * 0.6)
        for it in range(200):
            if time.time() - t10_start > t10_limit:
                break
            corr_t = np.correlate(h_t, 1.0 - h_t, mode='full')
            idx_t = np.argsort(-corr_t)[:20]
            g_t = np.zeros(n)
            for k in idx_t:
                g_t += grad_corr_k_vec(h_t, k, n)
            g_t /= len(idx_t)
            gn_t = np.linalg.norm(g_t)
            if gn_t < 1e-14:
                break
            for step in [0.003, 0.001, 0.005, 0.002, 0.0005]:
                h_try = project_feasible(h_t - step * g_t / gn_t, n)
                c_try = compute_c5(h_try)
                if c_try < best_c5:
                    best_c5 = c_try
                    best_h = h_try.copy()
                    h_t = h_try.copy()
                    break

    # Final feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
