The objective is: C₅ = max_k ∫₀² h(x)(1-h(x+k)) dx (with appropriate truncation)

For k=0: ∫₀² h(x)(1-h(x)) dx = ∫₀² h(x) dx - ∫₀² h(x)² dx = 1 - ∫₀² h(x)² dx

To minimize this, we want to maximize ∫₀² h(x)² dx, which means making h as close to {0,1} as possible (since ∫h = 1 is fixed).

For binary h: k=0 gives 0.
For continuous h: k=0 gives 1 - ∫h². To minimize, make h binary → k=0 term is 0.

So the optimal solution should be close to binary! This supports the binary local search approach.

For k≠0: The integral is over the overlap region. For binary h, this counts the number of positions where h[i]=1 and h[i+k]=0 (for positive k) or h[i]=0 and h[i+k]=1 (for negative k), scaled by dx.

So the problem reduces to: find a binary sequence of length n with exactly n/2 ones that minimizes the maximum overlap count across all shifts.

This is a well-studied problem in coding theory! It's related to "minimum overlap" sequences or "balanced sequences with minimum autocorrelation."

The key metric is: for each shift k, count the number of (1,0) transitions at distance k. We want to minimize the maximum of this count.

For a random binary sequence with n/2 ones, the expected count at each shift is approximately (n/4) * (n-1)/(n-1) ≈ n/4. With dx = 2/n, the expected C₅ is approximately (n/4) * (2/n) = 1/2.

The theoretical lower bound for the maximum overlap of a binary sequence with n/2 ones is related to the problem of "maximum size of a family of sets with bounded intersection" or similar combinatorial problems.

def binary_greedy_search(h_init, n, time_limit, rng):
    h = (h_init > 0.5).astype(float)
    target = n // 2
    ones = np.where(h == 1)[0]
    if len(ones) > target:
        idx = rng.choice(len(ones), len(ones) - target, replace=False)
        h[ones[idx]] = 0
    elif len(ones) < target:
        zeros = np.where(h == 0)[0]
        idx = rng.choice(len(zeros), target - len(ones), replace=False)
        h[zeros[idx]] = 1

    c5 = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()
    start = time.time()
    stall = 0

    while time.time() - start < time_limit:
        if _BUDGET - (time.time() - _T0) < 10:
            break
        improved = False
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        for _ in range(400):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            best_h[i], best_h[j] = best_h[j], best_h[i]
            c5_try = compute_c5(best_h)
            if c5_try < best_c5 - 1e-16:
                best_c5 = c5_try
                improved = True
                ones = np.where(best_h == 1)[0]
                zeros = np.where(best_h == 0)[0]
            else:
                best_h[i], best_h[j] = best_h[j], best_h[i]
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 15:
                n_flip = rng.randint(3, min(8, n // 8))
                positions = rng.choice(n, n_flip * 2, replace=False)
                for p in range(0, n_flip * 2, 2):
                    a, b = int(positions[p]), int(positions[p + 1])
                    if best_h[a] != best_h[b]:
                        best_h[a], best_h[b] = best_h[b], best_h[a]
                stall = 0
    return best_h, best_c5


def nesterov_subgrad(h, n, time_limit, K, beta):
    h_curr = h.copy()
    h_best = h.copy()
    c5_best = compute_c5(h)
    h_prev = h.copy()
    start = time.time()

    for it in range(200000):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 5:
            break
        if it == 0:
            y = h_curr.copy()
        else:
            t = float(it)
            gamma = (np.sqrt(1.0 + 4.0 * (t - 1.0) ** 2) - 1.0) / (2.0 * (t - 1.0)) if t > 1 else 1.0
            y = h_curr + gamma * (h_curr - h_prev)
            y = np.clip(y, 0.0, 1.0)

        _, g = smooth_max_and_grad(y, n, K, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        step = 0.12 / gn
        step = min(step, 0.25)
        h_new = project_feasible(y - step * g, n)
        c5_new = compute_c5(h_new)
        h_prev = h_curr.copy()
        h_curr = h_new
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_curr.copy()

    return h_best, c5_best


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)

    # Get initial solution
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 512
        best_h = np.zeros(best_n)
        positions = rng.choice(best_n, best_n // 2, replace=False)
        best_h[positions] = 1.0
        best_c5 = compute_c5(best_h)

    n = best_n
    h = best_h.copy()

    # Phase 1: Binary simulated annealing (explore discrete landscape)
    if remaining() > 50:
        sa_time = min(70.0, remaining() * 0.12)
        h_sa, c5_sa = binary_sim_anneal(h, n, sa_time, rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            h = h_sa.copy()

    # Phase 2: Binary greedy local search (polish the binary solution)
    if remaining() > 30:
        bl_time = min(40.0, remaining() * 0.08)
        h_bg, c5_bg = binary_greedy_search(h, n, bl_time, rng)
        if c5_bg < best_c5:
            best_c5 = c5_bg
            best_h = h_bg.copy()
            h = h_bg.copy()

    # Phase 3: Continuous SLSQP refinement (adaptive K, beta)
    K_beta_schedule = [
        (8, 500), (8, 2000), (12, 1000), (12, 5000),
        (15, 2000), (15, 10000), (20, 5000), (20, 20000),
        (30, 10000), (30, 50000), (50, 20000), (50, 100000),
    ]
    for K, beta in K_beta_schedule:
        if remaining() < 8:
            break
        h_new = slsqp_run(h, n, K, beta, maxiter=350)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 4: Binary search from the refined continuous solution
    if remaining() > 30:
        bl_time = min(40.0, remaining() * 0.10)
        h_bg2, c5_bg2 = binary_greedy_search(best_h, n, bl_time, rng)
        if c5_bg2 < best_c5:
            best_c5 = c5_bg2
            best_h = h_bg2.copy()
            h = h_bg2.copy()

    # Phase 5: Continuous refinement from the binary solution
    for K, beta in [(10, 5000), (15, 20000), (20, 50000), (25, 100000)]:
        if remaining() < 8:
            break
        h_new = slsqp_run(h, n, K, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 6: Nesterov subgradient (escape local optima)
    if remaining() > 40:
        sg_time = min(50.0, remaining() * 0.12)
        h_sg, c5_sg = nesterov_subgrad(h, n, sg_time, K=30, beta=150)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 7: Multi-resolution exploration
    res_list = []
    for ratio in [1.5, 2.0, 2.5]:
        r = int(round(best_n * ratio / 4) * 4)
        if r % 2 != 0:
            r += 1
        if 256 <= r <= 2560 and r != best_n:
            res_list.append(r)
    res_list = sorted(set(res_list))

    for n_t in res_list[:3]:
        if remaining() < 80:
            break

        h_t = project_feasible(interp_to(best_h, best_n, n_t), n_t)
        c5_t = compute_c5(h_t)

        # Binary SA at new resolution
        if remaining() > 30:
            h_t_sa, c5_t_sa = binary_sim_anneal(h_t, n_t, min(30.0, remaining() * 0.15), rng)
            if c5_t_sa < c5_t:
                h_t = h_t_sa
                c5_t = c5_t_sa

        # Binary greedy
        if remaining() > 20:
            h_t_bg, c5_t_bg = binary_greedy_search(h_t, n_t, min(20.0, remaining() * 0.10), rng)
            if c5_t_bg < c5_t:
                h_t = h_t_bg
                c5_t = c5_t_bg

        # SLSQP at new resolution
        for K, beta in [(15, 2000), (20, 10000), (30, 30000), (40, 100000)]:
            if remaining() < 10:
                break
            h_t = slsqp_run(h_t, n_t, K, beta, maxiter=300)
            c5_t = compute_c5(h_t)

        # Final binary polish
        if remaining() > 15:
            h_t_bg2, c5_t_bg2 = binary_greedy_search(h_t, n_t, min(15.0, remaining() * 0.10), rng)
            if c5_t_bg2 < c5_t:
                h_t = h_t_bg2
                c5_t = c5_t_bg2

        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            best_n = n_t

    # Phase 8: Final intensive SLSQP polish
    n = best_n
    h = best_h.copy()
    for K, beta in [(6, 20000), (8, 50000), (10, 100000), (12, 200000), (15, 50000), (20, 100000)]:
        if remaining() < 5:
            break
        h_new = slsqp_run(h, n, K, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 9: Final binary search (last chance)
    if remaining() > 15:
        h_bg3, c5_bg3 = binary_greedy_search(best_h, best_n, min(15.0, remaining() * 0.5), rng)
        if c5_bg3 < best_c5:
            best_c5 = c5_bg3
            best_h = h_bg3.copy()

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
