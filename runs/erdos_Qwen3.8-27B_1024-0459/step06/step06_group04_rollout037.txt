    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)

            h[interior] += diff / interior.sum()
        else:
            h += diff / n

def lse_smooth(obj, beta):
    """Log-sum-exp smooth approximation of max."""
    n = len(obj)
    m = obj.max()
    s = np.clip(beta * (obj - m), -600.0, 600.0)
    e = np.exp(s)
    return m + np.log(e.sum()) / beta
