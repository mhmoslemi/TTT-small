So corr_0 ≤ n/2 - n/4 = n/4.

For n=512: corr_0 ≤ 128. C5 ≤ 128/256 = 0.5.

For k≠0, corr_k = ∑_{i=0}^{n-1-|k|} h[i+|k|](1-h[i]).
The number of terms is n-|k|.
Max corr_k ≤ (n-|k|) * max(h[i+k](1-h[i])) ≤ n-|k|.

But with the constraint ∑h=n/2, the max is more constrained.

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)


def project_hypersimplex(v, target):
    """Project v onto {0<=h<=1, sum(h)=target} using bisection on lambda."""
    n = len(v)
    lo = v.min() - 2.0
    hi = v.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(free)
        if cnt > 0:
            h[free] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h


def grad_corr_k(h, k, n):
    """Gradient of corr_k (the k-th element of np.correlate(h,1-h,'full')) wrt h."""
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        m = n - offset
        g[offset:] += (1.0 - h[:m])
        g[:m] -= h[offset:]
    else:
        a = -offset
        m = n - a
        g[:m] += (1.0 - h[a:])
        g[a:] -= h[:m]
    return g


def smooth_max_obj_grad(h, n, beta, K):
    """Compute smooth max objective and gradient over top-K correlations."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    c_max = c_top[0]
    shifted = np.clip(beta * (c_top - c_max), -80.0, 80.0)
    ex = np.exp(shifted)
    w = ex / ex.sum()
    val = (c_max + np.log(ex.sum()) / beta) * (2.0 / n)
    g = np.zeros(n)
    for i in range(K):
        if w[i] < 1e-15:
            continue
        g += w[i] * grad_corr_k(h, idx[i], n)
    g *= (2.0 / n)
    return val, g

def cutting_plane_step(h, n, K, delta):
    """Solve one SCA/cutting plane LP step."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(-corr)[:K]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_rows = []
    b_rows = []
    for k in idx:
        g = grad_corr_k(h, k, n)
        A_row = np.zeros(n + 1)
        A_row[:n] = g
        A_row[-1] = -1.0
        A_rows.append(A_row)
        b_rows.append(corr[k] - np.dot(g, h))
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    I = np.eye(n)
    A_tr = np.vstack([np.hstack([I, np.zeros((n, 1))]),
                      np.hstack([-I, np.zeros((n, 1))])])
    b_tr = np.concatenate([h + delta, h + delta])
    A_ub = np.vstack([A_ub, A_tr])
    b_ub = np.concatenate([b_ub, b_tr])
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, None

        idx = np.where(h == 0.0)[0]
        rng.shuffle(idx)
        h[idx[:int(n / 2 - ones)]] = 1.0
    h = project_hypersimplex(h, n / 2.0)
    
    # Local search: try swapping pairs

    for it in range(max_iters):
        if _BUDGET - (time.time() - _T0) < 5:
            break
        i = rng.randint(n)

        h[i], h[j] = h[j], h[i]
        c5_new = compute_c5(h)
        if c5_new > best_c5:
            h[i], h[j] = h[j], h[i]

            best_c5 = c5_new
    return h, best_c5


def equalize_shifts(h, n, K, max_iters):
    """Gradient descent to equalize top-K correlations."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    target = n / 2.0
    for it in range(max_iters):
        if _BUDGET - (time.time() - _T0) < 3:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(-corr)[:K]
        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        if c_max - c_min < 1e-10:
            break
        g = np.zeros(n)
        for k in idx:
            g += grad_corr_k(h, k, n)
        g /= K
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        improved = False
        for step in [0.01, 0.005, 0.02, 0.002, 0.04, 0.001]:
            h_try = project_hypersimplex(h - step * g / gn, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                h = h_try
                best_c5 = c5_try
                best_h = h.copy()
                improved = True
                break
        if not improved:
            break
    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
