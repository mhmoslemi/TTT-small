def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(600):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        interior = (h > 1e-14) & (h < 1.0 - 1e-14)
        if interior.any():
            h[interior] += diff / interior.sum()
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_obj(x, beta, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    s = np.clip(beta * (corr - m), -500.0, 500.0)
    e = np.exp(s)
    return (m + np.log(e.sum()) / beta) * dx


def smooth_max_grad(x, beta, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()

def topk_weighted_grad(h, n, K, beta_w):

    if s < 1e-30:

    if time_left() > 30:
        for beta in [1000000, 3000000]:
            if time_left() < 20:
                break
            h_new = slsqp_solve(best_h, beta, maxiter=300, n=best_n)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new

    # Phase 4: Binary local search for structural improvement
    if time_left() > 60:
        t_bs = min(80.0, time_left() * 0.08)
        h_bin, c_bin = binary_local_search(best_h, best_n, t_bs, max_rounds=1500)
        if c_bin < best_c5 - 1e-8:
            # Use binary solution as new starting point for continuous opt
            best_h = h_bin.copy()
            best_c5 = min(best_c5, c_bin)
            if time_left() > 40:
                h_new, c_new = slsqp_curriculum(best_h, best_n, time.time() + 40,
                                                 betas=[5000, 50000, 300000, 1500000])
                if c_new < best_c5:
                    best_c5 = c_new
                    best_h = h_new
        elif c_bin < best_c5:
            best_h = h_bin.copy()
            best_c5 = c_bin

    # Phase 5: Try higher resolution (1024)
    if time_left() > 150:
        target_n = best_n * 2
        if target_n <= 2048:
            h_interp = interp_to(best_h, best_n, target_n)
            h_hi = project_feasible(h_interp, target_n)
            c_hi = compute_c5(h_hi)
            # Optimize at higher resolution
            if time_left() > 60:
                h_hi, c_hi = slsqp_curriculum(h_hi, target_n, time.time() + 60,
                                               betas=[1000, 10000, 50000, 200000])
            if time_left() > 30:
                h_sg, c_sg = subgradient_momentum(h_hi, target_n, min(30.0, time_left() * 0.1),
                                                   K_init=10, K_end=3,
                                                   beta_init=50.0, beta_end=300.0)
                if c_sg < c_hi:
                    c_hi = c_sg
                    h_hi = h_sg
            if c_hi < best_c5:
                best_c5 = c_hi
                best_h = h_hi.copy()
                best_n = target_n

    # Phase 6: Fractional refinement around best solution
    if time_left() > 30:
        h_frac, c_frac = fractional_refine(best_h, best_n, min(25.0, time_left() * 0.08))
        if c_frac < best_c5:
            best_c5 = c_frac
            best_h = h_frac

    # Phase 7: Final SLSQP polish
    if time_left() > 15:
        h_new = slsqp_solve(best_h, 5000000, maxiter=200, n=best_n)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    # Phase 8: Trust-constr attempt
    if time_left() > 20:
        h_tc = tc_solve(best_h, 2000000, maxiter=80, n=best_n)
        c_tc = compute_c5(h_tc)
        if c_tc < best_c5:
            best_c5 = c_tc
            best_h = h_tc

    # Phase 9: Another subgradient pass with different parameters
    if time_left() > 20:
        h_sg, c_sg = subgradient_momentum(best_h, best_n, min(20.0, time_left() * 0.3),
                                           K_init=5, K_end=2,
                                           beta_init=100.0, beta_end=800.0)
        if c_sg < best_c5:
            best_c5 = c_sg
            best_h = h_sg

    # Phase 10: Very fine local perturbation
    if time_left() > 10:
        h_loc, c_loc = fractional_refine(best_h, best_n, min(10.0, time_left() * 0.4))
        if c_loc < best_c5:
            best_c5 = c_loc
            best_h = h_loc

    # Final feasibility and evaluation
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
