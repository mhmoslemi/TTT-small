def swap_delta(h, i, j, N):
    """Return the change in the full correlation array when h[i]:1->0 and h[j]:0->1."""
    delta = np.zeros(2*N-1)
    
    # h[i]: 1 -> 0
    # Positive lags d=0..N-1-i: add h[i:i+N-i] to positions [N-1, N-1+N-i-1]
    if N - i > 0:
        delta[N-1 : N-1+N-i] += h[i : i+N-i]
    # Positive lags d=0..i: subtract (1-h[i-d]) to positions [N-1, N-1+i]
    #   i-d ranges from i down to 0, so (1-h[i-d]) = (1-h[i]), (1-h[i-1]), ..., (1-h[0])
    #   This is (1-h[0:i+1])[::-1]
    if i >= 0:
        delta[N-1 : N-1+i+1] -= (1.0 - h[0:i+1])[::-1]
    # Negative lags a=1..min(i,N-1): add h[i-a] to positions [N-1-min(i,N-1), N-2]
    #   i-a ranges from i-1 down to i-min(i,N-1)
    a_max = min(i, N-1)
    if a_max > 0:
        # positions N-1-a for a=1..a_max → indices N-2, N-3, ..., N-1-a_max
        # h[i-a] for a=1..a_max → h[i-1], h[i-2], ..., h[i-a_max]
        delta[N-1-a_max : N-1] += h[i-a_max : i][::-1]
    # Negative lags a=1..min(N-1-i, N-1): subtract (1-h[i+a]) to positions [N-1-min(N-1-i,N-1), N-2]
    a_max2 = min(N-1-i, N-1)
    if a_max2 > 0:
        # positions N-1-a for a=1..a_max2 → indices N-2, N-3, ..., N-1-a_max2
        # (1-h[i+a]) for a=1..a_max2 → (1-h[i+1]), ..., (1-h[i+a_max2])
        delta[N-1-a_max2 : N-1] -= (1.0 - h[i+1 : i+1+a_max2])[::-1]
    
    # h[j]: 0 -> 1
    # Positive lags d=0..N-1-j: subtract h[j:j+N-j] from positions [N-1, N-1+N-j-1]
    if N - j > 0:
        delta[N-1 : N-1+N-j] -= h[j : j+N-j]
    # Positive lags d=0..j: add (1-h[j-d]) to positions [N-1, N-1+j]
    if j >= 0:
        delta[N-1 : N-1+j+1] += (1.0 - h[0:j+1])[::-1]
    # Negative lags a=1..min(j,N-1): subtract h[j-a] from positions [N-1-min(j,N-1), N-2]
    a_max3 = min(j, N-1)
    if a_max3 > 0:
        delta[N-1-a_max3 : N-1] -= h[j-a_max3 : j][::-1]
    # Negative lags a=1..min(N-1-j, N-1): add (1-h[j+a]) to positions [N-1-min(N-1-j,N-1), N-2]
    a_max4 = min(N-1-j, N-1)
    if a_max4 > 0:
        delta[N-1-a_max4 : N-1] += (1.0 - h[j+1 : j+1+a_max4])[::-1]
    
    return delta
