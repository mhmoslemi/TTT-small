def project_feasible(h, n):

        if remaining() < 30:
            break
        scale = 0.003 + 0.004 * p
        h_pert = make_perturbation(best_h, best_n, rng, scale)
        # Quick SLSQP
        h_opt = slsqp_optimize(h_pert, best_n, 5000, maxiter=200)
        c_opt = compute_c5(h_opt)
        # Quick PG
        if remaining() > 15:
            h_pg, c_pg = projected_gradient_loop(h_opt, best_n, beta=5000, top_k=30,
                                                  max_iter=1500, time_limit=min(20.0, remaining() - 5.0))
            if c_pg < c_opt:
                c_opt = c_pg
                h_opt = h_pg.copy()
        if c_opt < best_c5:
            best_c5 = c_opt
            best_h = h_opt.copy()

    # Phase 7: Final projected gradient with high beta
    if remaining() > 20:
        tl = min(40.0, remaining() - 10.0)
        h_pg, c_pg = projected_gradient_loop(best_h, best_n, beta=10000, top_k=30,
                                              max_iter=5000, time_limit=tl)
        if c_pg < best_c5:
            best_c5 = c_pg
            best_h = h_pg.copy()

    # Phase 8: Final coordinate descent
    if remaining() > 15:
        h_cd, c_cd = targeted_coord_descent(best_h, best_n, max_sweeps=20,
                                             time_limit=min(30.0, remaining() - 8.0))
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd.copy()

    # Phase 9: Final SLSQP polish
    if remaining() > 15:
        h_final = slsqp_optimize(best_h, best_n, 20000, maxiter=500)
        c_final = compute_c5(h_final)
        if c_final < best_c5:
            best_c5 = c_final
            best_h = h_final.copy()
        # One more PG after SLSQP
        if remaining() > 15:
            h_pg, c_pg = projected_gradient_loop(best_h, best_n, beta=10000, top_k=20,
                                                  max_iter=2000, time_limit=min(20.0, remaining() - 8.0))
            if c_pg < best_c5:
                best_c5 = c_pg
                best_h = h_pg.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-12:
        # Slight degradation from projection, keep best_c5
        pass
    else:
        best_c5 = min(best_c5, final_c5)

    # Validate with evaluate_erdos_solution if available
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
