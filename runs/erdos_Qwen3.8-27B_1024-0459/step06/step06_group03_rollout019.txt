def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_corr_k_vec(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        np_ = n - offset
        g[offset:] += (1.0 - h[:np_])
        g[:np_] -= h[offset:]
    else:
        a = -offset
        np_ = n - a
        g[:np_] += (1.0 - h[a:])
        g[a:] -= h[:np_]
    return g


def _smmax_obj(h):

    ex = np.exp(cl)
    val = (m + np.log(ex.sum()) / beta) * (2.0 / n)

def _smmax_grad(h):

    n_alt = 768
    if remaining() > 100:
        h_alt = interp_to(best_h, best_n, n_alt)
        h_alt = project_feasible(h_alt, n_alt)
        c5_alt = compute_c5(h_alt)
        if c5_alt <= best_c5 + 0.002:
            h_alt = slsqop_smmax(h_alt, n_alt, maxiter=200)
            c5_alt = compute_c5(h_alt)
            h_alt_eq, c5_alt_eq = equalize_shifts(h_alt, n_alt, K=10, max_iters=150)
            if c5_alt_eq < c5_alt:
                h_alt = h_alt_eq
                c5_alt = c5_alt_eq
            if c5_alt < best_c5:
                best_c5 = c5_alt
                best_h = h_alt.copy()
                best_n = n_alt

    # Phase 9: Final intensive cutting plane at best resolution
    n = best_n
    h = best_h.copy()
    if remaining() > 40:
        h_fin, c5_fin = cutting_plane_phase(h, n, max_iters=200,
                                            delta_start=0.015, delta_end=0.03,
                                            remaining_limit=remaining() * 0.4)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Phase 10: Final SLSQP + equalize
    if remaining() > 15:
        h_fin = slsqop_smmax(best_h, best_n, maxiter=500)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
    if remaining() > 10:
        h_fin_eq, c5_fin_eq = equalize_shifts(best_h, best_n, K=14, max_iters=200)
        if c5_fin_eq < best_c5:
            best_c5 = c5_fin_eq
            best_h = h_fin_eq.copy()

    # Phase 11: Tiny gradient steps at the very end
    if remaining() > 8:
        h_t = best_h.copy()
        nt = best_n
        for it in range(100):
            if remaining() < 3:
                break
            corr_t = np.correlate(h_t, 1.0 - h_t, mode='full')
            idx_t = np.argsort(-corr_t)[:12]
            g_t = np.zeros(nt)
            for k in idx_t:
                g_t += grad_corr_k_vec(h_t, k, nt)
            g_t /= len(idx_t)
            gn_t = np.linalg.norm(g_t)
            if gn_t < 1e-14:
                break
            IMPROVED = False
            for step in [0.004, 0.002, 0.001, 0.006, 0.003]:
                h_try = project_feasible(h_t - step * g_t / gn_t, nt)
                c_try = compute_c5(h_try)
                if c_try < best_c5 - 1e-15:
                    best_c5 = c_try
                    best_h = h_try.copy()
                    h_t = h_try.copy()
                    IMPROVED = True
                    break
            if not IMPROVED:
                for step in [0.004, 0.002, 0.001]:
                    h_try = project_feasible(h_t + step * g_t / gn_t, nt)
                    c_try = compute_c5(h_try)
                    if c_try < best_c5 - 1e-15:
                        best_c5 = c_try
                        best_h = h_try.copy()
                        h_t = h_try.copy()
                        IMPROVED = True
                        break
            if not IMPROVED:
                break

    # Final feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    else:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
