def binary_from_h(h, n, rng):
    """Convert h to binary with exactly n/2 ones"""
    v = (h >= 0.5).astype(np.uint8)
    s = int(v.sum())
    target = n // 2
    if s > target:
        ones = np.where(v == 1)[0]
        rng.shuffle(ones)
        v[ones[:s-target]] = 0
    elif s < target:
        zeros = np.where(v == 0)[0]
        rng.shuffle(zeros)
        v[zeros[:target-s]] = 1
    return v.astype(float)

def sim_anneal(h, n, T0, Tfinal, n_iter, rng, time_limit):
    """Simulated annealing with pair swaps"""
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(np.max(corr))
    dx = 2.0 / n
    best_h = h.copy()
    best_max = cur_max
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    start = time.time()
    for it in range(n_iter):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 5:
            break
        
        T = T0 * (Tfinal / T0) ** (it / max(n_iter, 1))
        
        # Pick random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Perform swap
        h[i] = 0.0
        h[j] = 1.0
        corr_new = np.correlate(h, 1.0 - h, mode='full')
        new_max = float(np.max(corr_new))
        
        delta = new_max - cur_max
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            cur_max = new_max
            corr = corr_new
            # Update ones and zeros
            pos_i = np.where(ones == i)[0]
            pos_j = np.where(zeros == j)[0]
            if len(pos_i) > 0 and len(pos_j) > 0:
                ones[pos_i[0]] = j
                zeros[pos_j[0]] = i
            if cur_max < best_max - 1e-15:

                best_h = h.copy()

            h[j] = 0.0

        if _BUDGET - (time.time() - _T0) < 5:
            break
        improved = False
        rng.shuffle(ones)
        rng.shuffle(zeros)
        
        for i in ones:
            if time.time() - start > time_limit:
                break
            if h[i] != 1:
                continue
            best_j = -1
            best_new_max = cur_max
            
            for j in zeros:
                if h[j] != 0:
                    continue
                h[i] = 0.0
                h[j] = 1.0
                c = np.correlate(h, 1.0 - h, mode='full')
                nm = float(np.max(c))
                h[i] = 1.0
                h[j] = 0.0
                
                if nm < best_new_max - 1e-14:
                    best_new_max = nm
                    best_j = j
            
            if best_j >= 0:

                cur_max = float(np.max(corr))
                # Update lists
                ones.remove(i)
                ones.append(best_j)
                zeros.remove(best_j)
                zeros.append(i)
                best_c5 = cur_max * dx
                improved = True
    
    return h, best_c5

    dcorr = np.zeros((n, nfull), dtype=float)

```python
def precompute_dcorr(h, n):
    nfull = 2 * n - 1
    k_range = np.arange(nfull)
    dcorr = np.zeros((n, nfull), dtype=float)
    B = 1.0 - h  # B[i] = 1 - h[i]
    
    for i in range(n):
        # Term 1: (1-h[i-k'+(n-1)]) for valid k' where 0 ≤ i-k'+(n-1) < n
        m1 = i - k_range + (n - 1)
        valid1 = (m1 >= 0) & (m1 < n)
        dcorr[i, valid1] += B[m1[valid1]]
        
        # Term 2: -h[i+k'-(n-1)] for valid k' where 0 ≤ i+k'-(n-1) < n
        m2 = i + k_range - (n - 1)
        valid2 = (m2 >= 0) & (m2 < n)
        dcorr[i, valid2] -= h[m2[valid2]]
    
    return dcorr
```

        new_max = float(np.max(corr_new))
        
        delta = new_max - cur_max
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-12)):
            cur_max = new_max
            corr = corr_new
            # Swap in lists
            pos_i = ones.index(i) if i in ones else 0
            pos_j = zeros.index(j) if j in zeros else 0
            if pos_i == 0 and len(ones) > 1 and ones[pos_i] != i:
                try:
                    pos_i = ones.index(i)
                except ValueError:
                    pos_i = 0
            if pos_j == 0 and len(zeros) > 1 and zeros[pos_j] != j:
                try:
                    pos_j = zeros.index(j)
                except ValueError:
                    pos_j = 0
            ones[pos_i] = j
            zeros[pos_j] = i
            if cur_max < best_max - 1e-15:
                best_max = cur_max
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_max * dx


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    
    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)
    
    # Phase 0: Initialize from parent
    try:
        parent_h = np.array(initial_h_values, dtype=np.float64)
        n_parent = len(parent_h)
    except Exception:
        n_parent = 512
        parent_h = np.zeros(n_parent)
        parent_h[:n_parent // 2] = 1.0
    
    h = binary_from_h(parent_h, n_parent, rng)
    best_h = h.copy()
    best_n = n_parent
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Intensive local search at parent resolution
    if remaining() > 30:
        tl = min(250.0, remaining() * 0.45)
        h, c5 = fast_local_search(h, best_n, tl, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Phase 2: Second local search round (may find more after perturbations)
    if remaining() > 30:
        tl = min(150.0, remaining() * 0.3)
        h, c5 = fast_local_search(best_h, best_n, tl, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Phase 3: Simulated annealing to explore
    if remaining() > 60:
        tl = min(120.0, remaining() * 0.3)
        h_sa, c5_sa = simulated_anneal(best_h, best_n, tl, rng)
        # Polish the SA result
        if remaining() > 30:
            h_sa, c5_sa2 = fast_local_search(h_sa, best_n, min(40.0, remaining() * 0.2), rng)
            if c5_sa2 < c5_sa:
                c5_sa = c5_sa2
                h_sa = h_sa if c5_sa2 < c5_sa else np.array(h_sa)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
    
    # Phase 4: Try coarser resolution (n=256)
    if remaining() > 60:
        n_coarse = 256
        h_c = interp_to(best_h, best_n, n_coarse, rng)
        h_c, c5_c = fast_local_search(h_c, n_coarse, min(40.0, remaining() * 0.3), rng)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c.copy()
            best_n = n_coarse
    
    # Phase 5: Try finer resolution (n=1024)
    if remaining() > 80:
        n_fine = 1024
        h_f = interp_to(best_h, best_n, n_fine, rng)
        h_f, c5_f = fast_local_search(h_f, n_fine, min(50.0, remaining() * 0.4), rng)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            best_n = n_fine
    
    # Phase 6: Final intensive polish at best resolution
    if remaining() > 20:
        tl = min(remaining() - 10, 100.0)
        h, c5 = fast_local_search(best_h, best_n, tl, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Phase 7: Final micro-polish
    if remaining() > 10:
        tl = min(remaining() - 5, 30.0)
        h, c5 = fast_local_search(best_h, best_n, tl, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Ensure feasibility
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
