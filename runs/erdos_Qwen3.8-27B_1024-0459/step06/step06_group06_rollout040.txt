def project_feasible(h, n):

def grad_corr_k(h, k, n):
    """Gradient of the k-th correlation entry w.r.t. h."""
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        g[offset:n] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:n]
    else:
        a = -offset
        npairs = n - a
        g[:npairs] += (1.0 - h[a:n])
        g[a:n] -= h[:npairs]
    return g

    for k in range(len(corr)):
        wk = wc[k]
        if wk > 1e-14:
            g += wk * grad_corr_k(h, k, n)
    return g


def topk_obj(h, n, K, temp):
    corr = compute_corr(h)
    m = np.max(corr)
    w = np.exp(-temp * (m - corr))
    s = np.sum(w)
    if s < 1e-30:
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    return np.dot(w, corr) * (2.0 / n)


def topk_grad(h, n, K, temp):
    corr = compute_corr(h)
    m = np.max(corr)
    w = np.exp(-temp * (m - corr))
    s = np.sum(w)
    if s < 1e-30:
        idx = np.argmax(corr)
        return grad_corr_k(h, idx, n) * (2.0 / n)
    w /= s
    g = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk > 1e-14:
            g += wk * grad_corr_k(h, k, n)
    return g * (2.0 / n)


def lbfgsb_smooth(h, n, beta, penalty, maxiter=3000):
    def obj_fun(x):
        o = smooth_max_obj(x, n, beta)
        s = np.sum(x) - n / 2.0
        return o + 0.5 * penalty * s * s

    def grad_fun(x):
        g = smooth_max_grad(x, n, beta)
        s = np.sum(x) - n / 2.0
        g = g + penalty * s * np.ones(n)
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fun, h, jac=grad_fun, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-15, 'maxfun': 50000,
                                'maxls': 50})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def lbfgsb_topk(h, n, temp, penalty, maxiter=2000):
    def obj_fun(x):
        o = topk_obj(x, n, 10, temp)
        s = np.sum(x) - n / 2.0
        return o + 0.5 * penalty * s * s

    def grad_fun(x):
        g = topk_grad(x, n, 10, temp)
        s = np.sum(x) - n / 2.0
        g = g + penalty * s * np.ones(n)
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fun, h, jac=grad_fun, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-15, 'maxfun': 50000,
                                'maxls': 50})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def slsqp_smooth(h, n, beta, maxiter=500):
    def obj_fun(x):
        return smooth_max_obj(x, n, beta)

    def grad_fun(x):
        return smooth_max_grad(x, n, beta)

    def sc_fun(x):
        return np.sum(x) - n / 2.0

    def sc_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sc_fun, 'jac': sc_jac}]
    try:
        res = minimize(obj_fun, h, jac=grad_fun, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def coordinate_descent(h, n, time_limit):
    global _T0, _BUDGET
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    while time.time() - start < time_limit:
        if _BUDGET - (time.time() - _T0) < 5:
            break
        improved = False
        for i in range(n):
            if time.time() - start > time_limit:
                break
            old_val = h[i]
            candidates = [0.0, 1.0, 0.5,
                          max(0.0, old_val - 0.15), min(1.0, old_val + 0.15),
                          max(0.0, old_val - 0.05), min(1.0, old_val + 0.05)]
            candidates = [c for c in candidates if abs(c - old_val) > 1e-10]
            best_c = best_c5
            best_v = old_val
            for cv in candidates:
                h[i] = cv
                h_p = project_feasible(h, n)
                c = compute_c5(h_p)
                if c < best_c:
                    best_c = c
                    best_v = cv
            h[i] = best_v
            h = project_feasible(h, n)
            c_now = compute_c5(h)
            if c_now < best_c5 - 1e-14:
                best_c5 = c_now
                improved = True
        if not improved:
            break
    return h, best_c5


def bit_crush(h, n, threshold):
    h_cr = np.where(h > threshold, 1.0, 0.0)
    ones = int(np.sum(h_cr))
    target = int(n / 2)
    if ones > target:
        excess = ones - target
        idx_ones = np.where(h_cr == 1.0)[0]
        order = idx_ones[np.argsort(h[idx_ones])]
        h_cr[order[:excess]] = 0.0
    elif ones < target:
        deficit = target - ones
        idx_zeros = np.where(h_cr == 0.0)[0]
        order = idx_zeros[np.argsort(-h[idx_zeros])]
        h_cr[order[:deficit]] = 1.0
    return project_feasible(h_cr, n)


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)


def cutting_plane_step(h, n, K=30, delta=0.03):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    idx = np.argsort(-corr)[:K]
    nv = n + 1
    c_obj = np.zeros(nv)
    c_obj[-1] = 1.0
    A_ub_list = []
    b_ub_list = []
    for k in idx:
        g = grad_corr_k(h, k, n)
        fk = corr[k]
        row = np.zeros(nv)
        row[:n] = g
        row[-1] = -1.0
        A_ub_list.append(row)
        b_ub_list.append(-fk + np.dot(g, h))
    I = np.eye(n)
    A_trust = np.vstack([np.hstack([I, np.zeros((n, 1))]),
                         np.hstack([-I, np.zeros((n, 1))])])
    b_trust = np.concatenate([h + delta, -h + delta])
    A_ub = np.vstack([np.array(A_ub_list), A_trust])
    b_ub = np.concatenate([np.array(b_ub_list), b_trust])
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0, 1)] * n + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return h.copy(), c_max


def subgradient_refine(h, n, time_limit, rng):
    global _T0, _BUDGET
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        if _BUDGET - (time.time() - _T0) < 5:
            break
        it += 1
        rem = time_limit - (time.time() - start)
        if rem < 1:
            break
        K = max(5, 30 - it // 20)
        beta_g = min(200.0, 30.0 + it * 2.0)
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:K]
        c_top = corr[idx]
        weights = np.exp(beta_g * (c_top - c_top.max()))
        s = weights.sum()
        if s < 1e-30:
            weights = np.ones(K) / K
        else:
            weights /= s
        g = np.zeros(n)
        for i, k in enumerate(idx):
            if weights[i] > 1e-12:
                g += weights[i] * grad_corr_k(h, k, n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        step_base = 0.06 / (1.0 + 0.05 * it)
        step_base = min(step_base, 0.2)
        found = False
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_try = project_feasible(h - step_base * factor * g, n)
            c_try = compute_c5(h_try)
            if c_try < best_c5 - 1e-15:
                best_c5 = c_try
                best_h = h_try.copy()
                h = h_try.copy()
                found = True
                break
        if not found and it % 50 == 0:
            h = best_h.copy()
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 512
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Upsample to n=1024
    n = 1024
    h = interp_to(best_h, best_n, n)
    h = project_feasible(h, n)
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
        best_n = n

    # Phase 1: Warm-up with top-K (soft max) objective
    for temp in [20, 50, 100]:
        if remaining_time() < 40:
            break
        h = lbfgsb_topk(h, n, temp, penalty=1000, maxiter=1500)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: Smooth max L-BFGS-B with increasing beta
    for beta in [300, 1000, 3000, 8000]:
        if remaining_time() < 35:
            break
        h = lbfgsb_smooth(h, n, beta, penalty=5000, maxiter=2500)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 3: Subgradient refinement
    if remaining_time() > 50:
        h_sg, c5_sg = subgradient_refine(h, n, min(40.0, remaining_time() * 0.3), rng)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 4: Coordinate descent
    if remaining_time() > 60:
        h_cd, c5_cd = coordinate_descent(h, n, min(80.0, remaining_time() * 0.35))
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h = h_cd.copy()

    # Phase 5: Bit-crush exploration
    if remaining_time() > 120:
        for threshold in [0.5, 0.4, 0.6, 0.3, 0.7]:
            if remaining_time() < 35:
                break
            h_b = bit_crush(h, n, threshold)
            c5_b = compute_c5(h_b)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
            h_b = lbfgsb_smooth(h_b, n, 3000, penalty=5000, maxiter=1000)
            c5_b = compute_c5(h_b)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                h = h_b.copy()

    # Phase 6: Upsample to n=2048 and optimize
    if remaining_time() > 250:
        n_hi = 2048
        h_hi = interp_to(best_h, best_n, n_hi)
        h_hi = project_feasible(h_hi, n_hi)
        c5_hi = compute_c5(h_hi)
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi.copy()
            best_n = n_hi
            h = h_hi.copy()
            n = n_hi
        for beta in [2000, 5000, 10000]:
            if remaining_time() < 40:
                break
            h_hi = lbfgsb_smooth(h_hi, n_hi, beta, penalty=5000, maxiter=1500)
            c5_hi = compute_c5(h_hi)
            if c5_hi < best_c5:
                best_c5 = c5_hi
                best_h = h_hi.copy()
                best_n = n_hi
                h = h_hi.copy()
                n = n_hi
        # Coordinate descent at high res
        if remaining_time() > 40:
            h_hi, c5_hi = coordinate_descent(h_hi, n_hi, min(50.0, remaining_time() * 0.3))
            if c5_hi < best_c5:
                best_c5 = c5_hi
                best_h = h_hi.copy()
                best_n = n_hi
                h = h_hi.copy()
                n = n_hi

    # Phase 7: SLSQP polish
    for beta in [5000, 15000, 30000]:
        if remaining_time() < 25:
            break
        h = slsqp_smooth(h, n, beta, maxiter=400)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 8: Random restarts with structured perturbations
    if remaining_time() > 120:
        for trial in range(8):
            if remaining_time() < 35:
                break
            noise_scale = 0.03 + 0.01 * trial
            pert = rng.randn(n) * noise_scale
            h_r = project_feasible(h + pert, n)
            h_r = lbfgsb_smooth(h_r, n, 3000, penalty=5000, maxiter=800)
            c5_r = compute_c5(h_r)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                h = h_r.copy()
            else:
                h = best_h.copy()
                n = best_n

    # Phase 9: Cutting plane LP for lower bound + optimization
    if remaining_time() > 60:
        for it in range(50):
            if remaining_time() < 5:
                break
            h_lp, t_lp = cutting_plane_step(h, n, K=40, delta=0.02)
            cur_c5 = compute_c5(h)
            if t_lp < cur_c5 - 1e-8:
                best_alpha = 0
                best_val = cur_c5
                for alpha in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
                    h_try = project_feasible(h + alpha * (h_lp - h), n)
                    c_try = compute_c5(h_try)
                    if c_try < best_val:
                        best_val = c_try
                        best_alpha = alpha
                if best_alpha > 0:
                    h = project_feasible(h + best_alpha * (h_lp - h), n)
                    if compute_c5(h) < best_c5:
                        best_c5 = compute_c5(h)
                        best_h = h.copy()
            elif it > 20:
                break

    # Phase 10: Final high-beta smooth max
    for beta in [20000, 50000, 100000]:
        if remaining_time() < 20:
            break
        h = lbfgsb_smooth(h, n, beta, penalty=10000, maxiter=2000)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 11: Final SLSQP
    if remaining_time() > 20:
        h = slsqp_smooth(best_h, best_n, 50000, maxiter=500)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 12: Final coordinate descent
    if remaining_time() > 20:
        h_final, c5_final = coordinate_descent(best_h, best_n,
                                               min(30.0, remaining_time() * 0.7))
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
