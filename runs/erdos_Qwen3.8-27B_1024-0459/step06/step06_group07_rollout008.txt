            if wk < 1e-16:
                continue
            offset = k - (n - 1)
            if offset >= 0:
                npairs = n - offset
                g[offset:offset + npairs] += wk * (1.0 - x[:npairs])
                g[:npairs] -= wk * x[offset:offset + npairs]
            else:
                a = -offset
                npairs = n - a
                g[:npairs] += wk * (1.0 - x[a:a + npairs])
                g[a:a + npairs] -= wk * x[:npairs]
        return g

    def eq_fun(x):
        return np.sum(x) - n / 2.0

    def eq_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(smooth_max_value, h, jac=smooth_max_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    x_new = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(x_new, xp, src_h)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 10.0)

    # Phase 0: Initialize from parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        best_n = len(parent_h)
        best_h = project_feasible(parent_h, best_n)
    except Exception:
        best_n = 512
        best_h = np.zeros(best_n)
        best_h[::2] = 1.0
        best_h = project_feasible(best_h, best_n)

    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP to get near-binary, then binarize
    n = best_n
    h = best_h.copy()
    for beta in [500, 2000, 10000]:
        if remaining() < 30:
            break
        h_new = slsqp_smooth_max(h.copy(), n, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new.copy()

    # Binarize for local search
    h_bin = binarize(h, n)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    h = h_bin.copy()

    # Phase 2: Intensive local search at base resolution
    t_alloc = min(400.0, remaining() * 0.45)
    h_ls, c5_ls = local_search_intensive(h.copy(), n, time_limit=t_alloc, rng=rng)
    if c5_ls < best_c5:
        best_c5 = c5_ls
        best_h = h_ls.copy()
        h = h_ls.copy()

    # Phase 3: SLSQP polish on the binary result (allow slight non-binary)
    for beta in [20000, 80000]:
        if remaining() < 20:
            break
        h_pol = slsqp_smooth_max(h.copy(), n, beta, maxiter=400)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5 - 1e-13:
            best_c5 = c5_pol
            best_h = h_pol.copy()
        # Also try binarized version
        h_pol_b = binarize(h_pol, n)
        c5_pol_b = compute_c5(h_pol_b)
        if c5_pol_b < best_c5 - 1e-13:
            best_c5 = c5_pol_b
            best_h = h_pol_b.copy()
            h = h_pol_b.copy()

    # Phase 4: More local search after SLSQP
    if remaining() > 60:
        h_ls2, c5_ls2 = local_search_intensive(h.copy(), n, time_limit=min(200.0, remaining() * 0.35), rng=rng)
        if c5_ls2 < best_c5:
            best_c5 = c5_ls2
            best_h = h_ls2.copy()
            h = h_ls2.copy()

    # Phase 5: Try other resolutions
    for mult in [0.5, 1.5, 2.0]:
        if remaining() < 80:
            break
        n_t = int(round(best_n * mult / 16) * 16)
        if n_t < 64 or n_t > 2048 or n_t % 2 != 0 or n_t == best_n:
            continue

        h_t = interp_to(best_h, best_n, n_t)
        h_t = binarize(h_t, n_t)
        c5_t = compute_c5(h_t)

        # Quick local search at new resolution
        t_t = min(80.0, remaining() * 0.3)
        h_t_ls, c5_t_ls = local_search_intensive(h_t.copy(), n_t, time_limit=t_t, rng=rng)
        if c5_t_ls < c5_t:
            c5_t = c5_t_ls
            h_t = h_t_ls.copy()

        if c5_t < best_c5 - 1e-13:
            best_c5 = c5_t
            best_h = h_t.copy()
            best_n = n_t
            h = h_t.copy()

    # Phase 6: Final intensive local search at best resolution
    if remaining() > 40:
        n = best_n
        h = best_h.copy()
        t_final = min(200.0, remaining() * 0.6)
        h_fin, c5_fin = local_search_intensive(h.copy(), n, time_limit=t_final, rng=rng)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final SLSQP polish
    if remaining() > 15:
        h_pol = slsqp_smooth_max(best_h.copy(), best_n, 200000, maxiter=300)
        c5_pol = compute_c5(h_pol)
        h_pol_b = binarize(h_pol, best_n)
        c5_pol_b = compute_c5(h_pol_b)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
        if c5_pol_b < best_c5:
            best_c5 = c5_pol_b
            best_h = h_pol_b.copy()

    # Final local search
    if remaining() > 20 and best_n <= 1024:
        h_fin2, c5_fin2 = local_search_intensive(best_h.copy(), best_n, time_limit=min(60.0, remaining() * 0.7), rng=rng)
        if c5_fin2 < best_c5:
            best_c5 = c5_fin2
            best_h = h_fin2.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
