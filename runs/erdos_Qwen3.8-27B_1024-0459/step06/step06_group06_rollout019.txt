import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft, next_fast_len
import time

_T0 = 0.0
_BUDGET = 1300.0

def fast_full_corr(h):
    """Full linear cross-correlation h vs (1-h) via FFT."""
    n = len(h)
    b = 1.0 - h
    N = next_fast_len(2 * n)
    H = fft(h, N)
    B = fft(b, N)
    corr_full = np.real(ifft(H * np.conj(B)))
    return np.concatenate([corr_full[N - n + 1:], corr_full[:n]])

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = fast_full_corr(h)
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_for_lag(h, lag, n):
    """Gradient of corr at lag (positive or negative) w.r.t. h."""
    g = np.zeros(n)
    if lag >= 0:
        npairs = n - lag
        if npairs <= 0:
            return g
        g[:npairs] -= h[lag:]
        g[lag:] += (1.0 - h[:npairs])
    else:
        a = -lag
        npairs = n - a
        if npairs <= 0:
            return g
        g[:npairs] += (1.0 - h[a:])
        g[a:] -= h[:npairs]
    return g

def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

def minmax_subgrad(h, n, time_limit, rng):
    """Direct min-max via subgradient descent on top-K overlaps."""
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    it = 0
    
    while time.time() - start < time_limit and remaining_time() > 5:
        corr = fast_full_corr(h)
        c5_now = float(np.max(corr)) * (2.0 / n)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
        
        it += 1
        # Adaptive parameters
        progress = min(1.0, it / 500.0)
        K = int(30 * (1.0 - 0.5 * progress) + 10)  # 30 → 20
        beta_w = 20.0 + 100.0 * progress  # 20 → 120
        
        idx = np.argsort(-corr)[:K]
        c_top = corr[idx]
        weights = np.exp(beta_w * (c_top - c_top.max()))
        sw = weights.sum()
        if sw > 1e-30:
            weights /= sw
        else:
            weights = np.ones(K) / K
        
        g = np.zeros(n)
        for ii in range(K):
            if weights[ii] > 1e-15:
                lag = idx[ii] - (n - 1)
                g += weights[ii] * grad_for_lag(h, lag, n)
        
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        
        # Multiple trial steps
        base_step = 0.02 / (1.0 + 0.05 * progress * it)
        improved = False
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05, 2.0, 1.5]:
            step = base_step * factor
            h_new = project_feasible(h - step * g / gn, n)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-16:
                h = h_new
                best_c5 = c_new
                best_h = h.copy()
                improved = True
                break
        
        if not improved:
            # Occasionally reset
            if it % 200 == 0:
                h = best_h.copy()
    
    return best_h, best_c5

def coord_descent_fft(h, n, time_limit, rng):
    """Coordinate descent using FFT for objective evaluation."""
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    sweep = 0
    
    # Focus on "interesting" coordinates first (where 0 < h < 1)
    interesting = np.where((h > 0.01) & (h < 0.99))[0]
    indices = np.concatenate([interesting, np.setdiff1d(np.arange(n), interesting)])
    
    while time.time() - start < time_limit and remaining_time() > 5:
        sweep += 1
        any_improve = False
        for idx in indices:
            if time.time() - start > time_limit or remaining_time() < 3:
                return h, best_c5
            old = h[idx]
            if old < 1e-8 or old > 1.0 - 1e-8:
                # Try flipping to the other extreme
                candidates = [1.0 - old] if old < 0.5 else [old - 1.0]
                candidates = [c for c in candidates if 0 <= c <= 1]
            else:
                # Try a few perturbations
                step = max(0.01, 0.1 / (1 + 0.01 * sweep))
                candidates = [old + step, old - step, 
                             min(1.0, old + 2*step), max(0.0, old - 2*step), 0.5]
                candidates = list(set([round(c, 8) for c in candidates if 0 < c < 1 and abs(c - old) > 1e-8]))
            
            best_local = best_c5
            best_val = old
            for cv in candidates[:4]:  # Limit candidates
                h[idx] = cv
                h = project_feasible(h, n)
                c = compute_c5(h)
                if c < best_local - 1e-15:
                    best_local = c
                    best_val = cv
                h[idx] = old  # restore
            h[idx] = best_val
            if best_val != old:
                h = project_feasible(h, n)
                c_now = compute_c5(h)
                if c_now < best_c5:
                    best_c5 = c_now
                    any_improve = True
        
        if not any_improve:
            # Try a random perturbation to escape local optimum
            h_pert = h + 0.02 * rng.randn(n)
            h_pert = project_feasible(h_pert, n)
            c_pert = compute_c5(h_pert)
            if c_pert < best_c5:
                h = h_pert
                best_c5 = c_pert
            else:
                break
    
    return h, best_c5

def lbfgsb_minmax(h, n, beta, maxiter=2000):
    """L-BFGS-B on soft-min-max objective."""
    N_fft = next_fast_len(2 * n)
    
    def obj_fun(x):
        b = 1.0 - x
        H = fft(x, N_fft)
        B = fft(b, N_fft)
        corr = np.real(ifft(H * np.conj(B)))
        corr_full = np.concatenate([corr[N_fft - n + 1:], corr[:n]])
        dx = 2.0 / n
        m = np.max(corr_full)
        sh = beta * (corr_full - m)
        cl = np.clip(sh, -50.0, 50.0)
        ex = np.exp(cl)
        return (m + np.log(ex.sum()) / beta) * dx
    
    def grad_fun(x):
        b = 1.0 - x
        H = fft(x, N_fft)
        B = fft(b, N_fft)
        corr = np.real(ifft(H * np.conj(B)))
        corr_full = np.concatenate([corr[N_fft - n + 1:], corr[:n]])
        dx = 2.0 / n
        m = np.max(corr_full)
        sh = beta * (corr_full - m)
        cl = np.clip(sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / ex.sum() * dx
        
        g = np.zeros(n)
        # Only compute gradient for top lags to save time
        topK = min(50, 2 * n - 1)
        idx = np.argsort(-corr_full)[:topK]
        top_w = w[idx]
        for ii in range(topK):
            if top_w[ii] < 1e-18:
                continue
            lag = idx[ii] - (n - 1)
            lag_g = grad_for_lag(x, lag, n)
            g += top_w[ii] * lag_g
        return g
    
    # Feasibility constraint via penalty
    penalty = 1000.0
    def obj_pen(x):
        return obj_fun(x) + 0.5 * penalty * (np.sum(x) - n/2.0)**2
    
    def grad_pen(x):
        g = grad_fun(x)
        s = np.sum(x) - n/2.0
        return g + penalty * s
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_pen, h.copy(), jac=grad_pen, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-14, 'maxfun': 5000})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def fast_full_corr(h):
    n = len(h)
    b = 1.0 - h
    N = next_fast_len(2 * n)
    H = fft(h, N)
    B = fft(b, N)
    corr = np.real(ifft(H * np.conj(B)))
    return np.concatenate([corr[N - n + 1:], corr[:n]])


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = fast_full_corr(h)
    return float(np.max(corr)) * dx


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_for_lag(h, lag, n):
    g = np.zeros(n)
    if lag >= 0:
        npairs = n - lag
        if npairs <= 0:
            return g
        g[:npairs] -= h[lag:]
        g[lag:] += (1.0 - h[:npairs])
    else:
        a = -lag
        npairs = n - a
        if npairs <= 0:
            return g
        g[:npairs] += (1.0 - h[a:])
        g[a:] -= h[:npairs]
    return g


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def lbfgsb_minmax(h, n, beta, maxiter=1500):
    N_fft = next_fast_len(2 * n)
    dx = 2.0 / n
    penalty = 500.0
    topK = min(60, 2 * n - 1)

    def obj_fun(x):
        b = 1.0 - x
        H = fft(x, N_fft)
        B = fft(b, N_fft)
        corr_raw = np.real(ifft(H * np.conj(B)))
        corr_full = np.concatenate([corr_raw[N_fft - n + 1:], corr_raw[:n]])
        m = np.max(corr_full)
        sh = beta * (corr_full - m)
        cl = np.clip(sh, -50.0, 50.0)
        ex = np.exp(cl)
        s = np.sum(x) - n / 2.0
        return (m + np.log(ex.sum()) / beta) * dx + 0.5 * penalty * s * s

    def grad_fun(x):
        b = 1.0 - x
        H = fft(x, N_fft)
        B = fft(b, N_fft)
        corr_raw = np.real(ifft(H * np.conj(B)))
        corr_full = np.concatenate([corr_raw[N_fft - n + 1:], corr_raw[:n]])
        m = np.max(corr_full)
        sh = beta * (corr_full - m)
        cl = np.clip(sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / ex.sum() * dx

        g = np.zeros(n)
        idx_top = np.argsort(-corr_full)[:topK]
        for ii in range(topK):
            wk = w[idx_top[ii]]
            if wk < 1e-16:
                continue
            lag = idx_top[ii] - (n - 1)
            g += wk * grad_for_lag(x, lag, n)

        s = np.sum(x) - n / 2.0
        g += penalty * s
        return g

    bounds = [(0.0, 1.0)] * n
    try:

                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-13, 'maxfun': 4000})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

            if c_try < best_c5 - 1e-15:
                h = h_try
                best_c5 = c_try
                break

    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 512
        parent_h = np.full(parent_n, 0.5)
        best_h = parent_h.copy()
        best_c5 = compute_c5(parent_h)
        best_n = parent_n

    n_hi = 4096
    h_hi = interp_to(best_h, best_n, n_hi)
    h_hi = project_feasible(h_hi, n_hi)
    c5_hi = compute_c5(h_hi)
    if c5_hi < best_c5:
        best_c5 = c5_hi
        best_h = h_hi.copy()
        best_n = n_hi

    h = best_h.copy()
    n = best_n

    if remaining_time() > 50:
        for beta in [300, 2000]:
            if remaining_time() < 25:
                break
            h_new = lbfgsb_minmax(h, n, beta, maxiter=800)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    if remaining_time() > 80:
        t_lim = min(120.0, remaining_time() * 0.2)
        h_sg, c5_sg = minmax_subgrad(h, n, t_lim, rng)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    if remaining_time() > 60:
        h_eq, c5_eq = equalize_top(h, n, K=12, max_iters=200)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    if remaining_time() > 100:
        t_lim = min(200.0, remaining_time() * 0.3)
        h_cd, c5_cd = coord_descent_fft(h, n, t_lim, rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h = h_cd.copy()

    if remaining_time() > 50:
        for beta in [5000, 15000]:
            if remaining_time() < 20:
                break
            h_new = lbfgsb_minmax(h, n, beta, maxiter=1000)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    if remaining_time() > 60:
        t_lim = min(100.0, remaining_time() * 0.25)
        h_sg2, c5_sg2 = minmax_subgrad(h, n, t_lim, rng)
        if c5_sg2 < best_c5:
            best_c5 = c5_sg2
            best_h = h_sg2.copy()
            h = h_sg2.copy()

    if remaining_time() > 40:
        h_eq2, c5_eq2 = equalize_top(h, n, K=20, max_iters=300)
        if c5_eq2 < best_c5:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()
            h = h_eq2.copy()

    if remaining_time() > 80:
        t_lim = min(150.0, remaining_time() * 0.3)
        h_cd2, c5_cd2 = coord_descent_fft(h, n, t_lim, rng)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()
            h = h_cd2.copy()

    if remaining_time() > 40:
        t_lim = min(50.0, remaining_time() * 0.4)
        h_bp, c5_bp = block_perturbation(h, n, rng, t_lim)
        if c5_bp < best_c5:
            best_c5 = c5_bp
            best_h = h_bp.copy()
            h = h_bp.copy()

    if remaining_time() > 100 and best_n < 8192:
        n_vh = 8192
        h_vh = interp_to(best_h, best_n, n_vh)
        h_vh = project_feasible(h_vh, n_vh)
        c5_vh = compute_c5(h_vh)
        if c5_vh <= best_c5 + 0.002:
            h_vh = lbfgsb_minmax(h_vh, n_vh, 3000, maxiter=500)
            c5_vh = compute_c5(h_vh)
            if remaining_time() > 50:
                h_vh, c5_vh = minmax_subgrad(h_vh, n_vh, min(60.0, remaining_time() * 0.4), rng)
            c5_vh = compute_c5(h_vh)
            if c5_vh < best_c5:
                best_c5 = c5_vh
                best_h = h_vh.copy()
                best_n = n_vh

    if remaining_time() > 40:
        t_lim = min(60.0, remaining_time() * 0.5)
        h_final, c5_final = coord_descent_fft(best_h, best_n, t_lim, rng)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    if remaining_time() > 30:
        h_new = lbfgsb_minmax(best_h, best_n, 50000, maxiter=800)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    if remaining_time() > 20:
        h_eq3, c5_eq3 = equalize_top(best_h, best_n, K=25, max_iters=200)
        if c5_eq3 < best_c5:
            best_c5 = c5_eq3
            best_h = h_eq3.copy()

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
