    corr = compute_corr(h)
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    m = np.max(c_top)

    s = ex.sum()
    val = (m + np.log(s) / beta) * (2.0 / n)
    w = ex / s

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = int(np.count_nonzero(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-17:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] += diff
        h[idx] = np.clip(h[idx], 0.0, 1.0)
    return h


def grad_corr_k(h, k, n):
    g = np.zeros(n)
    if k >= 0:
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[k:] += (1.0 - h[:n - k])
            g[:n - k] -= h[k:]
    else:
        j = -k
        g[:n - j] += (1.0 - h[j:])
        g[j:] -= h[:n - j]
    return g


def smooth_max_value(h, n, K, beta):
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    m = np.max(c_top)
    sh = np.clip(beta * (c_top - m), -50.0, 50.0)
    ex = np.exp(sh)
    s = ex.sum()
    val = (m + np.log(s) / beta) * (2.0 / n)
    w = ex / s
    return val, idx, w


def smooth_max_grad(h, n, K, beta):
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    m = np.max(c_top)
    sh = np.clip(beta * (c_top - m), -50.0, 50.0)

    w = ex / ex.sum()
    dx = 2.0 / n

        if w[i] < 1e-15:
            continue
        g += w[i] * dx * grad_corr_k(h, int(idx[i] - (n - 1)), n)
    return g

        return v + mu * sv * sv

        sv = np.sum(x) - n / 2.0
        g += 2.0 * mu * sv
        return g

                       bounds=bounds,

            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 3: SLSQP refinement (handles constraint exactly)
    for K in [8, 12, 20, 35]:
        for beta in [5000, 20000, 80000]:
            if remaining() < 6:
                break
            h_new = slsqp_refine(h, n, K, beta, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 4: Higher resolution (n*2)
    if remaining() > 120:
        n2 = best_n * 2
        h2 = interp_to(best_h, best_n, n2)
        h2 = project_feasible(h2, n2)
        c5_2 = compute_c5(h2)

        for K in [20, 40, 70]:
            for beta in [500, 3000, 15000]:
                if remaining() < 8:
                    break
                h2_new = lbfgs_optimize(h2, n2, K, beta, maxiter=600, mu=30.0)
                c5_2_new = compute_c5(h2_new)
                if c5_2_new < c5_2:
                    h2 = h2_new
                    c5_2 = c5_2_new

        if remaining() > 40:
            h2, c5_2 = projected_subgrad(h2, n2, min(50.0, remaining() * 0.2), K=60, beta_w=40.0)

        for K in [30, 50]:
            for beta in [30000, 100000]:
                if remaining() < 8:
                    break
                h2_new = lbfgs_optimize(h2, n2, K, beta, maxiter=800, mu=30.0)
                c5_2_new = compute_c5(h2_new)
                if c5_2_new < c5_2:
                    h2 = h2_new
                    c5_2 = c5_2_new

        if remaining() > 20:
            h2_new = slsqp_refine(h2, n2, 30, 50000, maxiter=250)
            c5_2_new = compute_c5(h2_new)
            if c5_2_new < c5_2:
                h2 = h2_new
                c5_2 = c5_2_new

        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
            best_n = n2

    # Phase 5: Greedy lag reduction at best resolution
    n = best_n
    h = best_h.copy()
    if remaining() > 15:
        h, c5_g = greedy_lag_reduction(h, n, K_lags=15, max_rounds=200)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h.copy()

    # Phase 6: Perturbation multi-start
    n = best_n
    h = best_h.copy()
    if remaining() > 40:
        for pi in range(3):
            if remaining() < 20:
                break
            sigma = 0.01 + 0.008 * pi
            h_pert = project_feasible(h + rng.randn(n) * sigma, n)
            c5_pert = compute_c5(h_pert)
            if c5_pert > best_c5 + 0.02:
                continue
            for beta in [5000, 30000]:
                if remaining() < 8:
                    break
                h_pert = lbfgs_optimize(h_pert, n, 15, beta, maxiter=500, mu=30.0)
            c5_pert = compute_c5(h_pert)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()
                h = h_pert.copy()

    # Phase 7: Final intensive SLSQP + L-BFGS-B
    n = best_n
    h = best_h.copy()
    for K in [5, 8, 10, 15]:
        for beta in [50000, 150000, 400000]:
            if remaining() < 4:
                break
            h_new = lbfgs_optimize(h, n, K, beta, maxiter=1500, mu=50.0)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 8: Final SLSQP polish
    for beta in [100000, 300000]:
        if remaining() < 4:
            break
        h_new = slsqp_refine(h, n, 8, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 9: Final greedy
    if remaining() > 10:
        h_final, c5_final = greedy_lag_reduction(best_h, best_n, K_lags=20, max_rounds=150)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Final projection
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Verify with evaluator
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
