def coordinate_polish(h, n, max_sweeps=100, time_limit=120.0):
    """Fine coordinate descent on active lags."""
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    target = n / 2.0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        top_k = np.argsort(-corr)[:30]
        g = np.zeros(n)
        for k in top_k:
            g += _grad_corr_k(h, k, n)

        order = np.argsort(-np.abs(g))
        improved = False

        for idx in order:
            if time.time() - start > time_limit:
                break

            old_val = h[idx]
            if abs(g[idx]) < 1e-6:
                continue

            # Try gradient-based candidates
            candidates = []
            for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                cv = old_val - frac * g[idx]
                if 0.0 <= cv <= 1.0:
                    candidates.append(cv)
            candidates.append(0.0)
            candidates.append(1.0)

            best_cand = old_val
            best_c = best_c5

            for c in candidates:
                if abs(c - old_val) < 1e-10:
                    continue
                h[idx] = c
                h = project_feasible(h, n)
                c_cur = compute_c5(h)
                if c_cur < best_c - 1e-14:
                    best_c = c_cur
                    best_cand = c

            h[idx] = best_cand
            h = project_feasible(h, n)
            if best_cand != old_val:
                if compute_c5(h) < best_c5:
                    best_c5 = compute_c5(h)
                    best_h = h.copy()
                    improved = True

        if not improved:
            break

    return best_h, best_c5

    best_h = parent_h.copy()
    best_c5 = compute_c5(best_h)
    best_n = n_orig

    # Phase 1: Upsample to n=2048
    target_n = 2048
    h_work = interp_to(best_h, n_orig, target_n)
    h_work = project_feasible(h_work, target_n)
    c_work = compute_c5(h_work)

    # Phase 2: L-BFGS-B at n=2048 with increasing beta
    betas = [200, 1000, 5000, 20000, 100000]
    for beta in betas:
        if _remaining() < 30:
            break
        tl = min(80, _remaining())
        h_try = lbfgsb_optimize(h_work, target_n, beta, penalty=1e4, maxiter=800)
        c_try = compute_c5(h_try)
        if c_try < c_work:
            c_work = c_try
            h_work = h_try

    if c_work < best_c5:
        best_c5 = c_work
        best_h = h_work.copy()
        best_n = target_n

    # Phase 3: Projected gradient at n=2048
    if _remaining() > 60:
        tl = min(100, _remaining())
        h_pg, c_pg = proj_gradient(best_h, best_n, beta=10000,
                                   step_init=0.03, maxiter=3000, time_limit=tl)
        if c_pg < best_c5:
            best_c5 = c_pg
            best_h = h_pg.copy()

    # Phase 4: More L-BFGS-B with higher beta
    for beta in [200000, 500000, 1000000]:
        if _remaining() < 25:
            break
        h_try = lbfgsb_optimize(best_h, best_n, beta, penalty=1e4, maxiter=1000)
        c_try = compute_c5(h_try)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try.copy()

    # Phase 5: Active set optimization
    if _remaining() > 60:
        tl = min(120, _remaining())
        h_as, c_as = active_set_optimize(best_h, best_n, K=20,
                                         maxiter=3000, time_limit=tl)
        if c_as < best_c5:
            best_c5 = c_as
            best_h = h_as.copy()

    # Phase 6: Coordinate descent polish
    if _remaining() > 40:
        tl = min(100, _remaining())
        h_cd, c_cd = coordinate_polish(best_h, best_n, max_sweeps=50, time_limit=tl)
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd.copy()

    # Phase 7: Try perturbations + re-optimize
    for p in range(5):
        if _remaining() < 30:
            break
        scale = 0.005 + p * 0.005
        h_pert = structured_perturb(best_h, best_n, rng, scale)
        h_opt = lbfgsb_optimize(h_pert, best_n, 50000, penalty=1e4, maxiter=500)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c5:
            best_c5 = c_opt
            best_h = h_opt.copy()

    # Phase 8: Try higher resolution n=3072
    if _remaining() > 80:
        n_hi = 3072
        h_hi = interp_to(best_h, best_n, n_hi)
        h_hi = project_feasible(h_hi, n_hi)
        c_hi = compute_c5(h_hi)
        for beta in [5000, 50000, 200000]:
            if _remaining() < 20:
                break
            h_try = lbfgsb_optimize(h_hi, n_hi, beta, penalty=1e4, maxiter=600)
            c_try = compute_c5(h_try)
            if c_try < c_hi:
                c_hi = c_try
                h_hi = h_try
        if c_hi < best_c5:
            best_c5 = c_hi
            best_h = h_hi.copy()
            best_n = n_hi

    # Phase 9: Final active set + coordinate polish
    if _remaining() > 30:
        tl = min(60, _remaining())
        h_as2, c_as2 = active_set_optimize(best_h, best_n, K=15,
                                           maxiter=2000, time_limit=tl)
        if c_as2 < best_c5:
            best_c5 = c_as2
            best_h = h_as2.copy()

    if _remaining() > 20:
        tl = min(40, _remaining())
        h_cd2, c_cd2 = coordinate_polish(best_h, best_n, max_sweeps=30, time_limit=tl)
        if c_cd2 < best_c5:
            best_c5 = c_cd2
            best_h = h_cd2.copy()

    # Final L-BFGS-B
    if _remaining() > 15:
        h_final = lbfgsb_optimize(best_h, best_n, 1000000, penalty=1e4, maxiter=500)
        c_final = compute_c5(h_final)
        if c_final < best_c5:
            best_c5 = c_final
            best_h = h_final.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
