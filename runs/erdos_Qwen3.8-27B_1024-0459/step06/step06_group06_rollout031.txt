def compute_c5(h):

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    """Compute full correlation of h with 1-h"""
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max_obj(h, n, beta, penalty):
    """Compute smooth max objective + penalty"""
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    ex = np.exp(shifted)
    sm = (m + np.log(np.sum(ex) + 1e-30) / beta) * (2.0 / n)
    s = np.sum(h) - n / 2.0
    return sm + 0.5 * penalty * s * s

def smooth_max_grad(h, n, beta, penalty):
    """Compute gradient of smooth max objective + penalty"""
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    ex = np.exp(shifted)
    w = ex / (np.sum(ex) + 1e-30)
    
    g = np.zeros(n)

    threshold = 1e-10
    active = np.where(w > threshold)[0]

        s = k - (n - 1)
        if s >= 0:

                    h = project_feasible(h, n)
        if not improved:
            break
    return h, best_c5

def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)
