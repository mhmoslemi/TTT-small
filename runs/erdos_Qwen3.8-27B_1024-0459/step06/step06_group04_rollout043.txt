    dx = 2.0 / n

        return np.dot(lam, corr) * dx

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except:
        return h_init

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        interior = (h > 1e-14) & (h < 1.0 - 1e-14)
        if interior.any():
            h[interior] += diff / interior.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def corr_gradient(h, k_offset):
    n = len(h)
    g = np.zeros(n)
    if k_offset >= 0:
        np_ = n - k_offset
        g[k_offset:] += (1.0 - h[:np_])
        g[:np_] -= h[k_offset:]
    else:
        a = -k_offset
        np_ = n - a
        g[:np_] += (1.0 - h[a:])
        g[a:] -= h[:np_]
    return g


def topk_subgradient(h, K):

    corr = compute_corr(h)
    idx = np.argsort(-corr)[:K]
    g = np.zeros(n)
    for k_idx in idx:
        g += corr_gradient(h, k_idx - (n - 1))
    g /= K
    g *= (2.0 / n)
    return g, corr

        c_try = compute_c5(h_try)

        if c_try < best_c5 - 1e-15:
            h = h_try
            best_c5 = c_try
            best_h = h.copy()
            stall = 0
        else:
            h_try2 = project_feasible(h - 0.4 * lr * v, n)
            c_try2 = compute_c5(h_try2)
            if c_try2 < best_c5 - 1e-15:
                h = h_try2
                best_c5 = c_try2
                best_h = h.copy()
                lr *= 0.5
            else:
                lr = max(lr * 0.65, min_lr)
                v *= 0.2
                stall += 1
                if stall > 300:
                    v = g.copy() * 0.5
                    lr = lr_init * 0.3
                    stall = 0

        lr = max(lr * lr_decay, min_lr)

    return best_h, best_c5


def multi_phase_descent(h, n, time_limit):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()

    configs = [
        (30, 0.92, 0.03, 0.999, False, 0),
        (20, 0.88, 0.02, 0.9993, False, 0),
        (12, 0.84, 0.012, 0.9995, True, 50.0),
        (8, 0.78, 0.006, 0.9997, True, 100.0),
        (5, 0.70, 0.003, 0.9998, True, 200.0),
    ]

    total_weight = sum(1.0 / (c[3] - 1.0) for c in configs)
    remaining_frac = 1.0

    for K, mom, lr0, decay, use_w, bw in configs:
        remaining = time_limit - (time.time() - start)
        if remaining < 5:
            break
        phase_time = min(60.0, remaining * 0.25)
        if phase_time < 3:
            phase_time = remaining * 0.5
        h_opt, c_opt = momentum_descent(
            best_h, n, phase_time, K=K, momentum=mom,
            lr_init=lr0, lr_decay=decay,
            use_weighted=use_w, beta_w=bw)
        if c_opt < best_c5:
            best_c5 = c_opt
            best_h = h_opt

    return best_h, best_c5


def smooth_max_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    s = np.clip(beta * (corr - m), -500.0, 500.0)
    e = np.exp(s)
    return (m + np.log(e.sum()) / beta) * dx


def smooth_max_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    s = np.clip(beta * (corr - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-13:
            continue
        g += wk * corr_gradient(x, k_idx - (n - 1))
    return g * dx


def slsqp_optimize(h, n, beta, maxiter=150):
    bnds = Bounds(np.zeros(n), np.ones(n))
    A = np.ones((1, n))
    cons = LinearConstraint(A, n / 2.0, n / 2.0)
    try:
        res = minimize(smooth_max_obj, h, args=(beta,),
                       jac=smooth_max_grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)
