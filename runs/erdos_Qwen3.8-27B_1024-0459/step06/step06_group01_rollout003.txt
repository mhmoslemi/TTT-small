        res = minimize(sm_obj, h0, jac=sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.fun < sm_obj(h0) - 1e-15:
            return project(res.x, n)
    except Exception:
        pass
    return h0

def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)

def active_topk(h, n, top_k=4, max_iters=3000):
    h = project(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    lr = 0.02
    stall = 0
    for it in range(max_iters):
        if remaining_time() < 5:
            break
        corr = corr_vec(h)
        idx = np.argsort(-corr)[:top_k]
        c_vals = corr[idx]
        g = np.zeros(n)
        for ki in idx:
            off = ki - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:] += (1.0 - h[:np_])
                g[:np_] -= h[off:]
            else:
                a = -off
                np_ = n - a
                g[:np_] += (1.0 - h[a:])
                g[a:] -= h[:np_]
        g *= dx
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        d = -g / gn
        improved = False
        for step in [lr, lr * 0.5, lr * 0.25, lr * 2.0, lr * 0.1, lr * 4.0]:
            ht = project(h + step * d, n)
            ct = compute_c5(ht)
            if ct < best_c5 - 1e-14:
                h = ht
                best_c5 = ct
                improved = True
                break
        if improved:
            lr = min(lr * 1.05, 0.15)
            stall = 0
        else:
            lr *= 0.6
            stall += 1
            if stall > 80:
                break
    return h, best_c5

def fast_coord_descent(h, n, max_sweeps=30):
    h = project(h, n)
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if remaining_time() < 10:
            break
        any_improve = False
        for j in range(n):
            old = h[j]
            best_v = old
            best_o = compute_c5(h)
            a, b = max(0.0, old - 0.4), min(1.0, old + 0.4)
            if b - a < 1e-10:
                continue
            for _ in range(20):
                m1 = a + (b - a) * 0.381966
                m2 = a + (b - a) * 0.618034
                h[j] = m1
                c1 = compute_c5(h)
                h[j] = m2
                c2 = compute_c5(h)
                if c1 < c2:
                    b = m2
                    h[j] = m1
                    if c1 < best_o:
                        best_o = c1
                        best_v = m1
                else:
                    a = m1
                    h[j] = m2
                    if c2 < best_o:
                        best_o = c2
                        best_v = m2
            h[j] = best_v
            if best_o < best_c5 - 1e-13:
                best_c5 = best_o
                any_improve = True
        h = project(h, n)
        actual = compute_c5(h)
        if actual < best_c5:
            best_c5 = actual
        if not any_improve and sweep >= 5:
            break
    return h, best_c5

def binary_sa(h_init, n, time_limit, rng):
    h = (h_init > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        si = np.argsort(-h)
        h = np.zeros(n)
        h[si[:n_half]] = 1.0
    start = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    cur_c5 = best_c5
    T = 0.8
    no_imp = 0
    while time.time() - start < time_limit and remaining_time() > 10:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            no_imp += 1
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = h.copy()
                no_imp = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_imp += 1
        T *= 0.9997
        if no_imp > 5000:
            break
    return best_h, best_c5

def theta_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project(h, n)

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = np.zeros(640)
        best_h[:320] = 1.0
    best_n = len(best_h)
    best_h = project(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP polish at parent resolution with increasing beta
    h = best_h.copy()
    for beta in [200, 1000, 5000, 20000]:
        if remaining_time() < 30:
            break
        h_new = run_slsqp(h, best_n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Active-set top-k search
    if remaining_time() > 30:
        h_eq, c5_eq = active_topk(h, best_n, top_k=5, max_iters=5000)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 3: SLSQP with higher beta
    for beta in [50000, 200000]:
        if remaining_time() < 25:
            break
        h_new = run_slsqp(h, best_n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 4: Multi-resolution exploration
    resolutions = [512, 768, 1024]
    for n_t in resolutions:
        if n_t == best_n:
            continue
        if remaining_time() < 80:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project(h_interp, n_t)
        c5_int = compute_c5(h_feas)

        h_opt = h_feas
        for beta in [500, 5000, 50000]:
            if remaining_time() < 20:
                break
            h_opt = run_slsqp(h_opt, n_t, beta, maxiter=400)
        c5_opt = compute_c5(h_opt)
        h_opt, c5_opt = active_topk(h_opt, n_t, top_k=5, max_iters=3000)
        if c5_opt < best_c5 - 1e-14:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t
            h = h_opt.copy()

    # Phase 5: Coordinate descent at best resolution
    if remaining_time() > 60:
        h_cd, c5_cd = fast_coord_descent(best_h.copy(), best_n, max_sweeps=25)
        if c5_cd < best_c5 - 1e-14:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h = h_cd.copy()

    # Phase 6: Binary SA + continuous refinement
    if remaining_time() > 100:
        h_bin = (best_h > 0.5).astype(float)
        n_half = best_n // 2
        n_ones = int(np.sum(h_bin))
        if n_ones != n_half:
            si = np.argsort(-h_bin)
            h_bin = np.zeros(best_n)
            h_bin[si[:n_half]] = 1.0
        sa_time = min(120, max(20, int(remaining_time() * 0.25)))
        h_b, c5_b = binary_sa(h_bin, best_n, sa_time, rng)
        h_b_c = project(h_b, best_n)
        for beta in [5000, 50000]:
            if remaining_time() < 15:
                break
            h_b_c = run_slsqp(h_b_c, best_n, beta, maxiter=500)
        c5_bc = compute_c5(h_b_c)
        h_b_c, c5_bc = active_topk(h_b_c, best_n, top_k=5, max_iters=2000)
        if c5_bc < best_c5 - 1e-14:
            best_c5 = c5_bc
            best_h = h_b_c.copy()
        if c5_b < best_c5 - 1e-14:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 7: Try Thue-Morse start
    if remaining_time() > 60:
        h_tm = theta_morse(best_n)
        h_tm_opt = h_tm
        for beta in [1000, 10000]:
            if remaining_time() < 15:
                break
            h_tm_opt = run_slsqp(h_tm_opt, best_n, beta, maxiter=400)
        h_tm_opt, c5_tm = active_topk(h_tm_opt, best_n, top_k=5, max_iters=2000)
        if c5_tm < best_c5 - 1e-14:
            best_c5 = c5_tm
            best_h = h_tm_opt.copy()

    # Phase 8: Perturbation + re-optimize
    if remaining_time() > 50:
        h_pert = best_h.copy()
        for _ in range(3):
            h_pert = best_h.copy()
            perturb_mask = rng.rand(best_n) < 0.1
            h_pert[perturb_mask] += rng.uniform(-0.1, 0.1, pert=0) * 0
            # Random small perturbation
            idx_p = rng.choice(best_n, size=best_n // 5, replace=False)
            h_pert[idx_p] += rng.uniform(-0.05, 0.05, len(idx_p))
            h_pert = project(h_pert, best_n)
            h_pert = run_slsqp(h_pert, best_n, 10000, maxiter=300)
            h_pert, c5_p = active_topk(h_pert, best_n, top_k=4, max_iters=2000)
            if c5_p < best_c5 - 1e-14:
                best_c5 = c5_p
                best_h = h_pert.copy()

    # Phase 9: Final coordinate descent
    if remaining_time() > 30:
        h_cd2, c5_cd2 = fast_coord_descent(best_h.copy(), best_n, max_sweeps=20)
        if c5_cd2 < best_c5 - 1e-14:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()

    # Phase 10: Final SLSQP + active set polish
    if remaining_time() > 20:
        h_fin = run_slsqp(best_h, best_n, 500000, maxiter=400)
        h_fin, c5_fin = active_topk(h_fin, best_n, top_k=7, max_iters=3000)
        if c5_fin < best_c5 - 1e-14:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final feasibility check
    best_h = project(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
