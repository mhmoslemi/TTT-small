∂corr_{k*}/∂h[j] = (1-h[j-k*]) - h[j+k*] (with appropriate bounds)

We want to decrease corr_{k*}, so we move in the direction -∂corr_{k*}/∂h.

But we also need to keep other correlations from increasing. This gives a constrained direction finding problem.

A simple approximation: use the subgradient of the max, which is just the gradient of the argmax correlation. If multiple correlations are tied at the max, use the average of their gradients.

This is essentially what the top-k push method does. So I'll stick with that.

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_CUR_N = 0
_SM_BETA = 1000.0


def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def _sm_objective(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = beta * (corr - m)
    z = np.clip(z, -100.0, 100.0)
    ez = np.exp(z)
    return (m + np.log(np.sum(ez)) / beta) * dx


def _sm_gradient(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = beta * (corr - m)
    z = np.clip(z, -100.0, 100.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-16:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    return g


def _eq_fun(x):
    return np.sum(x) - _CUR_N / 2.0


def _eq_jac(x):
    return np.ones(_CUR_N)


def slsqp_opt(h, n, beta, maxiter=500):
    global _CUR_N, _SM_BETA
    _CUR_N = n
    _SM_BETA = beta
    h0 = project_feasible(h, n)
    obj0 = _sm_objective(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_sm_objective, h0, jac=_sm_gradient, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'maxls': 300})
        if res.fun < obj0 - 1e-14:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def topk_push(h, n, top_k=5, max_iters=5000):
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    stagnant = 0

    for it in range(max_iters):
        if remaining_time() < 5:
            break
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]

        g = np.zeros(n)
        for k in idx_top:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:] += (1.0 - h[:np_])
                g[:np_] -= h[off:]
            else:
                a = -off
                np_ = n - a
                g[:np_] += (1.0 - h[a:])
                g[a:] -= h[:np_]
        g *= dx

        g_feas = g - np.mean(g)
        gn = np.linalg.norm(g_feas)
        if gn < 1e-14:
            break

        d = -g_feas
        improved = False
        base_lr = 0.15 / (1.0 + 0.01 * it)
        for trial in range(20):
            step = base_lr * (0.5 ** trial)
            h_trial = project_feasible(h + step * d, n)
            c_trial = compute_c5(h_trial)
            if c_trial < best_c5 - 1e-15:
                h = h_trial
                best_c5 = c_trial
                best_h = h.copy()
                improved = True
                stagnant = 0
                break

        if not improved:
            stagnant += 1
            if stagnant > 100:
                break

    return best_h, best_c5


def weighted_topk_push(h, n, top_k=8, max_iters=3000):
    """Weighted subgradient that emphasizes larger correlations more."""
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    stagnant = 0

    for it in range(max_iters):
        if remaining_time() < 5:
            break
        corr = compute_corr(h)
        idx_sorted = np.argsort(-corr)
        idx_top = idx_sorted[:top_k]
        c_max = corr[idx_top[0]]
        c_floor = corr[idx_top[-1]]
        span = c_max - c_floor
        if span < 1e-12:
            span = 1e-12

        g = np.zeros(n)
        for pos, k in enumerate(idx_top):
            weight = (c_max - corr[k]) / span
            weight = weight ** 2 + 0.1
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:] += weight * (1.0 - h[:np_])
                g[:np_] -= weight * h[off:]
            else:
                a = -off
                np_ = n - a
                g[:np_] += weight * (1.0 - h[a:])
                g[a:] -= weight * h[:np_]
        g *= dx

        g_feas = g - np.mean(g)
        gn = np.linalg.norm(g_feas)
        if gn < 1e-14:
            break

        d = -g_feas
        improved = False
        base_lr = 0.1 / (1.0 + 0.01 * it)
        for trial in range(20):
            step = base_lr * (0.5 ** trial)
            h_trial = project_feasible(h + step * d, n)
            c_trial = compute_c5(h_trial)
            if c_trial < best_c5 - 1e-15:
                h = h_trial
                best_c5 = c_trial
                best_h = h.copy()
                improved = True
                stagnant = 0
                break

        if not improved:
            stagnant += 1
            if stagnant > 80:
                break

    return best_h, best_c5


def coord_descent(h, n, max_sweeps=20):
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)

    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        any_improve = False
        for j in range(n):
            old_val = h[j]
            best_v = old_val
            best_obj = compute_c5(h)
            a, b = 0.0, 1.0
            for _ in range(28):
                m1 = a + (b - a) * 0.381966
                m2 = a + (b - a) * 0.618034
                h[j] = m1
                c1 = compute_c5(h)
                h[j] = m2
                c2 = compute_c5(h)
                if c1 < c2:
                    b = m2
                    h[j] = m1
                    if c1 < best_obj:
                        best_obj = c1
                        best_v = m1
                else:
                    a = m1
                    h[j] = m2
                    if c2 < best_obj:
                        best_obj = c2
                        best_v = m2
            h[j] = best_v
            curr = compute_c5(h)
            if curr < best_c5 - 1e-13:
                best_c5 = curr
                any_improve = True

        h = project_feasible(h, n)
        actual = compute_c5(h)
        if actual < best_c5:
            best_c5 = actual
        if not any_improve and sweep > 2:
            break

    return h, best_c5


def nudge_restart(h, n, rng, nudge_scale=0.02, n_nudges=5):
    """Try small random perturbations and re-optimize briefly."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(n_nudges):
        if remaining_time() < 10:
            break
        noise = rng.randn(n) * nudge_scale
        h_nudged = project_feasible(h + noise, n)
        h_opt = slsqp_opt(h_nudged, n, 50000, maxiter=300)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5 - 1e-14:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    return best_h, best_c5
