def compute_corr(h):

def project_feasible(h, n):

def grad_corr_k(h, k, n):
    """Gradient of correlation at lag k with respect to h."""
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        g[offset:n] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:n]
    else:
        a = -offset
        npairs = n - a
        g[:npairs] += (1.0 - h[a:n])
        g[a:n] -= h[:npairs]
    return g

def active_set_gradient(h, n, K=20):

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()

        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        g[offset:n] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:n]
    else:
        a = -offset
        npairs = n - a
        g[:npairs] += (1.0 - h[a:n])
        g[a:n] -= h[:npairs]
    return g


def top_k_gradient(h, n, K=25):

    idx = np.argsort(-corr)[:K]
    g = np.zeros(n)
    for k in idx:

    g /= K
    return g, float(np.max(corr))

def smooth_max_obj_grad(h, n, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    s = ex.sum()
    obj = (m + np.log(s) / beta) * (2.0 / n)
    w = ex / s
    g = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k] * (2.0 / n)
        if wk > 1e-12:
            g += wk * grad_corr_k(h, k, n)
    return obj, g


def slsqp_smooth(h, n, beta, maxiter=500):
    def obj_fun(x):
        return smooth_max_obj_grad(x, n, beta)[0]

    def grad_fun(x):
        return smooth_max_obj_grad(x, n, beta)[1]

    def sc_fun(x):
        return np.sum(x) - n / 2.0

    def sc_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sc_fun, 'jac': sc_jac}]
    try:
        res = minimize(obj_fun, h, jac=grad_fun, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def lbfgsb_smooth(h, n, beta, penalty, maxiter=3000):

        corr = compute_corr(x)
        m = corr.max()
        sh = np.clip(beta * (corr - m), -50.0, 50.0)
        ex = np.exp(sh)
        sm = (m + np.log(ex.sum()) / beta) * (2.0 / n)
        s = np.sum(x) - n / 2.0
        return sm + 0.5 * penalty * s * s

    def grad_fun(x):
        corr = compute_corr(x)
        m = corr.max()
        sh = np.clip(beta * (corr - m), -50.0, 50.0)
        ex = np.exp(sh)

            if wk > 1e-12:
                g += wk * grad_corr_k(x, k, n)
        s = np.sum(x) - n / 2.0
        g += penalty * s
        return g

    try:
        res = minimize(obj_fun, h, jac=grad_fun, method='L-BFGS-B',

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 512
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Work at n=2048 for more detail
    n_work = 2048
    h_work = interp_to(best_h, best_n, n_work)
    h_work = project_feasible(h_work, n_work)
    c5_work = compute_c5(h_work)
    if c5_work < best_c5:
        best_c5 = c5_work
        best_h = h_work.copy()
        best_n = n_work

    h = h_work.copy()
    n = n_work

    # Phase 1: Minimax descent (100s)
    if remaining_time() > 120:
        t1 = min(100.0, remaining_time() * 0.25)
        h, c5 = minimax_descent(h, n, t1, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: L-BFGS-B with increasing beta
    for beta, pen in [(500, 200), (2000, 500), (5000, 1000), (10000, 2000)]:
        if remaining_time() < 25:
            break
        h_new = lbfgsb_smooth(h, n, beta, pen, maxiter=2000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 3: Equalize shifts
    if remaining_time() > 50:
        h_eq, c5_eq = equalize_shifts(h, n, K=15, max_iters=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 4: Binary push + re-optimize
    if remaining_time() > 60:
        bp, c5_bp = binary_push_and_reopt(h, n, min(40.0, remaining_time() * 0.15), rng)
        if c5_bp < best_c5:
            best_c5 = c5_bp
            best_h = bp.copy()
            h = bp.copy()

    # Phase 5: More minimax descent
    if remaining_time() > 80:
        t5 = min(80.0, remaining_time() * 0.2)
        h5, c5_5 = minimax_descent(h, n, t5, rng)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
            h = h5.copy()

    # Phase 6: SLSQP polish
    for beta in [3000, 8000, 15000, 30000]:
        if remaining_time() < 20:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 7: Coordinate descent
    if remaining_time() > 60:
        t7 = min(70.0, remaining_time() * 0.2)
        h_cd, c5_cd = coordinate_descent(h, n, t7)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h = h_cd.copy()

    # Phase 8: Final equalize + fine subgradient
    if remaining_time() > 40:
        h_eq, c5_eq = equalize_shifts(h, n, K=10, max_iters=200)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    if remaining_time() > 40:
        t8 = min(40.0, remaining_time() * 0.2)
        h8, c5_8 = fine_subgradient(h, n, t8, rng)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()

    # Phase 9: Try n=4096
    if remaining_time() > 120 and best_n < 4096:
        n_vh = 4096
        h_vh = interp_to(best_h, best_n, n_vh)
        h_vh = project_feasible(h_vh, n_vh)
        c5_vh = compute_c5(h_vh)
        if c5_vh < best_c5:
            h_vh, c5_vh = minimax_descent(h_vh, n_vh, min(60.0, remaining_time() * 0.3), rng)
            if c5_vh < best_c5:
                h_vh = slsqp_smooth(h_vh, n_vh, 8000, maxiter=300)
                c5_vh = compute_c5(h_vh)
            h_eq, c5_eq = equalize_shifts(h_vh, n_vh, K=10, max_iters=150)
            if c5_eq < c5_vh:
                h_vh, c5_vh = h_eq, c5_eq
            if c5_vh < best_c5:
                best_c5 = c5_vh
                best_h = h_vh.copy()
                best_n = n_vh

    # Phase 10: Final SLSQP polish
    if remaining_time() > 20:
        h_new = slsqp_smooth(best_h, best_n, 20000, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 11: Micro-equalize
    if remaining_time() > 15:
        h_eq, c5_eq = equalize_shifts(best_h, best_n, K=8, max_iters=100)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Final feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
