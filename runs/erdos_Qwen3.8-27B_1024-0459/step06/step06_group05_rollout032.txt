def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_corr_k(h, k_idx, n):
    """Gradient of the k_idx-th element of the full correlation."""
    g = np.zeros(n)
    offset = k_idx - (n - 1)
    if offset >= 0:
        npairs = n - offset
        if npairs <= 0:
            return g
        g[offset:] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:]
    else:
        a = -offset
        npairs = n - a
        if npairs <= 0:
            return g
        g[:npairs] += (1.0 - h[a:])
        g[a:] -= h[:npairs]
    return g

def softmax_obj_grad(h, n, K, beta):
    """Compute soft-max objective and gradient for top-K correlations."""
    corr = compute_corr(h)
    n_corr = len(corr)
    K_use = min(K, n_corr)
    idx = np.argpartition(-corr, K_use - 1)[:K_use]
    c_top = corr[idx]
    m = c_top.max()
    sh = np.clip(beta * (c_top - m), -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / ex.sum()
    dx = 2.0 / n
    val = (m + np.log(ex.sum()) / beta) * dx
    
    g = np.zeros(n)
    for i in range(K_use):
        wi = w[i] * dx
        if wi < 1e-16:
            continue
        g += wi * grad_corr_k(h, idx[i], n)
    return val, g

def slsqp_run(h, n, K, beta, maxiter=500):
    """Run SLSQP optimization."""
    def obj(x):
        v, _ = softmax_obj_grad(x, n, K, beta)
        return v

    def grad(x):
        _, g = softmax_obj_grad(x, n, K, beta)
        return g

    def cfun(x):
        return np.sum(x) - n / 2.0

    def cjac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]

    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            c5_old = compute_c5(h)
            if c5_new <= c5_old + 1e-10:
                return h_new
    except Exception:
        pass
    return h

def binary_local_search(h, n, rng, max_iters=20000):

    c5_best = c5_curr

    for it in range(max_iters):
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h_bin[i] = 0.0

            # Update lists

            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0

def multi_scale_binary_search(h, n, rng, max_rounds=5000):
    """Multi-scale binary search: start with coarse moves, refine with fine moves."""
    h_bin = np.zeros(n)
    sorted_idx = np.argsort(-h)
    h_bin[sorted_idx[:n//2]] = 1.0
    
    c5_best = compute_c5(h_bin)
    h_best = h_bin.copy()
    
    # Multi-scale: try swaps at different "distances"
    for scale in [n//4, n//8, n//16, n//32, n//64, 1]:
        for it in range(max_rounds // (len([n//4, n//8, n//16, n//32, n//64, 1])) + 1):
            i = rng.randint(n)
            if h_bin[i] != 1:
                continue
            # Find a zero at distance ~scale
            j_candidates = []
            for dj in [scale, -scale]:
                j = (i + dj) % n
                if 0 <= j < n and h_bin[j] == 0:
                    j_candidates.append(j)
            if not j_candidates:
                continue
            j = j_candidates[rng.randint(len(j_candidates))]
            
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            c5_new = compute_c5(h_bin)
            
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_bin.copy()
            else:
                h_bin[i] = 1.0
                h_bin[j] = 0.0
    
    return h_best, c5_best

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_corr_k(h, k_idx, n):
    g = np.zeros(n)
    offset = k_idx - (n - 1)
    if offset >= 0:
        npairs = n - offset
        if npairs <= 0:
            return g
        g[offset:] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:]
    else:
        a = -offset
        npairs = n - a
        if npairs <= 0:
            return g
        g[:npairs] += (1.0 - h[a:])
        g[a:] -= h[:npairs]
    return g


def softmax_obj(h, n, K, beta):
    corr = compute_corr(h)
    n_corr = len(corr)
    K_use = min(K, n_corr)
    if K_use == n_corr:
        idx = np.arange(n_corr)
    else:
        idx = np.argpartition(-corr, K_use - 1)[:K_use]
    c_top = corr[idx]
    m = c_top.max()
    sh = np.clip(beta * (c_top - m), -50.0, 50.0)
    ex = np.exp(sh)
    s = ex.sum()
    dx = 2.0 / n
    val = (m + np.log(s) / beta) * dx
    return val


def softmax_grad(h, n, K, beta):
    corr = compute_corr(h)
    n_corr = len(corr)
    K_use = min(K, n_corr)
    if K_use == n_corr:
        idx = np.arange(n_corr)
    else:
        idx = np.argpartition(-corr, K_use - 1)[:K_use]
    c_top = corr[idx]
    m = c_top.max()
    sh = np.clip(beta * (c_top - m), -50.0, 50.0)
    ex = np.exp(sh)
    s = ex.sum()
    if s < 1e-30:
        w = np.ones(K_use) / K_use
    else:
        w = ex / s
    dx = 2.0 / n
    g = np.zeros(n)
    for i in range(K_use):
        wi = w[i] * dx
        if wi < 1e-16:
            continue
        g += wi * grad_corr_k(h, idx[i], n)
    return g


def slsqp_run(h, n, K, beta, maxiter=500):
    def obj(x):
        return softmax_obj(x, n, K, beta)

    def grad(x):
        return softmax_grad(x, n, K, beta)

    def cfun(x):
        return np.sum(x) - n / 2.0

    def cjac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]

    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            c5_old = compute_c5(h)
            if c5_new <= c5_old + 1e-10:
                return h_new
    except Exception:
        pass
    return h


def binary_local_search(h, n, rng, max_iters=30000):
    h_bin = np.zeros(n)
    sorted_idx = np.argsort(-h)
    h_bin[sorted_idx[:n // 2]] = 1.0
    c5_best = compute_c5(h_bin)
    h_best = h_bin.copy()

    ones = list(np.where(h_bin == 1)[0])
    zeros = list(np.where(h_bin == 0)[0])

    for it in range(max_iters):
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        c5_new = compute_c5(h_bin)
        if c5_new < c5_best - 1e-15:
            c5_best = c5_new
            h_best = h_bin.copy()
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
    return h_best, c5_best


def annealed_binary_search(h, n, rng, n_rounds=50000):
    h_bin = np.zeros(n)
    sorted_idx = np.argsort(-h)
    h_bin[sorted_idx[:n // 2]] = 1.0
    c5_best = compute_c5(h_bin)
    h_best = h_bin.copy()
    c5_curr = c5_best

    T = 0.01
    T_min = 1e-6
    cooling = 0.9995

    for it in range(n_rounds):
        i_idx = rng.randint(n)
        j_idx = rng.randint(n)
        if h_bin[i_idx] == h_bin[j_idx]:
            continue
        h_bin[i_idx] = 1.0 - h_bin[i_idx]
        h_bin[j_idx] = 1.0 - h_bin[j_idx]
        c5_new = compute_c5(h_bin)
        delta = c5_new - c5_curr
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            c5_curr = c5_new
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_bin.copy()
        else:
            h_bin[i_idx] = 1.0 - h_bin[i_idx]
            h_bin[j_idx] = 1.0 - h_bin[j_idx]
        T = max(T * cooling, T_min)
    return h_best, c5_best


def interp_to_resolution(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)
