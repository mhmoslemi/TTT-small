```python
import numpy as np
from scipy.optimize import minimize, fmin_bfgs
import time

_T0 = 0.0
_BUDGET = 1300.0
_CUR_N = 0
_BETA = 1000.0
_PENALTY = 1e6

def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    beta = _BETA
    m = np.max(corr)
    z = np.clip(beta * (corr - m), -50, 50)
    ez = np.exp(z)
    S = np.sum(ez)
    obj = (m + np.log(S) / beta) * dx
    # Penalty for sum constraint
    pen = _PENALTY * (np.sum(x) - n/2.0)**2
    return obj + pen

def soft_max_grad(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    beta = _BETA
    m = np.max(corr)
    z = np.clip(beta * (corr - m), -50, 50)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2*n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        off = k - (n-1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    # Penalty gradient
    g += 2 * _PENALTY * (np.sum(x) - n/2.0)
    return g

def lbfgs_opt(h, n, beta, maxiter=2000):
    global _CUR_N, _BETA
    _CUR_N = n
    _BETA = beta
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(soft_max_obj, h0, jac=soft_max_grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
        if res.fun < soft_max_obj(h0) - 1e-10:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0
```

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_CUR_N = 0
_BETA = 1000.0
_PENALTY = 1e5


def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def _sm_obj(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    beta = _BETA
    m = np.max(corr)
    z = np.clip(beta * (corr - m), -60.0, 60.0)
    ez = np.exp(z)
    S = np.sum(ez)
    return (m + np.log(S) / beta) * dx + _PENALTY * (np.sum(x) - n / 2.0) ** 2


def _sm_grad(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    beta = _BETA
    m = np.max(corr)
    z = np.clip(beta * (corr - m), -60.0, 60.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-16:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    g += 2.0 * _PENALTY * (np.sum(x) - n / 2.0)
    return g


def lbfgs_opt(h, n, beta, maxiter=3000):
    global _CUR_N, _BETA, _PENALTY
    _CUR_N = n
    _BETA = beta
    _PENALTY = 1e5
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    f0 = _sm_obj(h0)
    try:
        res = minimize(_sm_obj, h0, jac=_sm_grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter,
                                               'ftol': 1e-18, 'gtol': 1e-14,
                                               'maxfun': maxiter * 2})
        if res.fun < f0 - 1e-12:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def active_set_grad(h, n, top_k=8, max_iters=2000):
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step0 = 0.01
    for it in range(max_iters):
        if remaining_time() < 5:
            break
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        c_vals = corr[idx_top]
        c_max = c_vals[0]
        if c_max - c_vals[-1] < 1e-11:
            break
        g_dir = np.zeros(n)
        for ki in idx_top:
            off = ki - (n - 1)
            if off >= 0:
                np_ = n - off
                g_dir[off:] += (1.0 - h[:np_])
                g_dir[:np_] -= h[off:]
            else:
                a = -off
                np_ = n - a
                g_dir[:np_] += (1.0 - h[a:])
                g_dir[a:] -= h[:np_]
        g_dir *= dx
        gn = np.linalg.norm(g_dir)
        if gn < 1e-15:
            break
        d = -g_dir / gn
        improved = False
        for step in [step0, step0*0.5, step0*2, step0*0.1, step0*0.25, step0*4, step0*0.05]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                best_h = ht.copy()
                improved = True
                step0 = min(step * 1.5, 0.1)
                break
        if not improved:
            for step in [step0*0.3, step0*0.1, step0*0.05]:
                ht = project_feasible(h - step * d, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-14:
                    h = ht
                    best_c5 = cn
                    best_h = ht.copy()
                    improved = True
                    break
            if not improved:
                step0 *= 0.5
                if step0 < 1e-6:
                    break
    return best_h, best_c5


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def binary_local_search(h, n, time_limit, rng, greedy=True):
    h = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        idx = np.argsort(-h)
        h = np.zeros(n)
        h[idx[:n_half]] = 1.0
    start = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    curr_c5 = best_c5
    T = 0.5 * dx if (dx := 2.0/n) > 0 else 0.001
    T = 0.003
    no_improve = 0
    max_no_improve = 5000
    while time.time() - start < time_limit and remaining_time() > 10:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - curr_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-12)):
            curr_c5 = new_c5
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
        T *= 0.9997
        if no_improve > max_no_improve:
            break
    return best_h, best_c5


def greedy_binary_improve(h, n, max_rounds=100):

def theta_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)


def random_binary(n, rng):
    h = np.zeros(n)
    n_half = n // 2
    idx = rng.choice(n, size=n_half, replace=False)
    h[idx] = 1.0
    return h
