from scipy.optimize import minimize, linprog
import time

_BUDGET = 1300.0
_G_N = 0
_G_BETA = 1.0
_G_K = 40

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        # Distribute the correction to coordinates that have room
        if diff > 0:
            # Need to increase sum: increase coordinates below 1
            mask = h < 1.0 - 1e-12
        else:
            # Need to decrease sum: decrease coordinates above 0
            mask = h > 1e-12
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_corr_k_vec(h, off, n):
    """Gradient of correlation at offset 'off' with respect to h."""
    g = np.zeros(n)
    if off >= 0:
        m = n - off
        # corr = sum_{i=0}^{m-1} h[i] * (1-h[i+off])
        # d/dh[i] for i in [0, m-1]: (1-h[i+off])
        # d/dh[i+off] for i in [0, m-1]: -h[i]
        g[:m] += (1.0 - h[off:])  # contribution from first factor
        g[off:] -= h[:m]           # contribution from second factor
    else:
        a = -off
        m = n - a
        # corr = sum_{i=0}^{m-1} h[i+a] * (1-h[i])
        # d/dh[i+a]: (1-h[i])
        # d/dh[i]: -h[i+a]
        g[a:] += (1.0 - h[:m])
        g[:m] -= h[a:]
    return g

def smax_obj(h):
    n = _G_N
    beta = _G_BETA
    K = _G_K
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr.copy()
    m = c.max()
    # Use log-sum-exp for numerical stability
    c_shifted = c - m
    log_sum_exp = m + np.log(np.exp(beta * c_shifted).sum()) / beta
    return log_sum_exp * (2.0 / n)

def smax_grad(h):
    n = _G_N
    beta = _G_BETA
    K = _G_K
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Top K
    idx = np.argsort(-corr)[:K]
    c = corr[idx]
    c_max = c.max()
    shifted = c - c_max
    clipped = np.clip(beta * shifted, -100, 100)
    exps = np.exp(clipped)
    w = exps / exps.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for i in range(K):

        g += w[i] * grad_corr_k_vec(h, idx[i] - (n-1), n)
    return g * dx

    return np.sum(h) - _G_N / 2.0

def smax_eq_jac(h):
    return np.ones(_G_N)

def slsqp_smax(h, n, beta, K, maxiter):

    _G_N = n
    _G_BETA = beta
    _G_K = K
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': smax_eq_fun, 'jac': smax_eq_jac}]
    try:

                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None and res.x is not h:

            c5_old = compute_c5(h)
            if c5_new <= c5_old + 1e-12:

    except Exception:

def tmin_ineq(v):
    # Returns -1 * (corr_k - t*n/2) for each k in KLIST
    # We need: corr_k <= t*n/2, i.e., t*n/2 - corr_k >= 0
    # SLSQP ineq: fun >= 0
    h = v[:_G_T_N]
    n = _G_T_N
    t = v[-1]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return t * n / 2.0 - corr[_G_T_KLIST]

def tmin_ineq_jac(v):

    n = _G_T_N
    K_list = _G_T_KLIST
    J = np.zeros((len(K_list), n + 1))

        g = grad_corr_k_vec(h, off, n)
        J[i, :n] = -g  # d(-corr_k)/dh = -grad
        J[i, -1] = n / 2.0  # d(t*n/2)/dt = n/2

def subgradient_descent(h, n, time_limit):

    best_h = h.copy()

    start = time.time()

    momentum = np.zeros(n)

        K = 30 if progress < 0.3 else (20 if progress < 0.6 else 10)
        beta = 30.0 if progress < 0.3 else (50.0 if progress < 0.6 else 80.0)

        idx = np.argsort(-corr)[:K]
        c = corr[idx]
        c_max = c.max()
        shifted = c - c_max
        clipped = np.clip(beta * shifted, -80, 80)
        exps = np.exp(clipped)
        w = exps / exps.sum()

def pair_swap_cd(h, n, time_limit):
    """Coordinate descent via pair swaps."""
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    rng = np.random.RandomState(12345)
    
    while time.time() - start < time_limit:
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        
        # Compute current correlations
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Top K offsets
        K = 20
        top_idx = np.argsort(-corr)[:K]
        
        improved = False
        # Random order of coordinates
        order = rng.permutation(n)
        
        for i in order:
            if time.time() - start > time_limit:
                break
            rem_inner = _BUDGET - (time.time() - _T0)
            if rem_inner < 3:
                break
            
            # Try swapping h[i] with a few random partners
            # Choose j ≠ i
            for attempt in range(3):
                j = rng.randint(n)
                if j == i:
                    continue
                
                # Feasible range for δ
                # h[i] + δ ∈ [0,1], h[j] - δ ∈ [0,1]
                delta_lo = max(-h[i], h[j] - 1.0)
                delta_hi = min(1.0 - h[i], h[j])
                if delta_hi - delta_lo < 1e-10:
                    continue
                
                # Compute gradients for top K offsets
                best_delta = 0.0
                best_val = compute_c5(h)
                
                # Sample a few δ values
                candidates = [0.001, -0.001, 0.005, -0.005, 0.01, -0.01, 0.05, -0.05]
                # Also try the boundaries
                candidates.append(delta_hi * 0.5)
                candidates.append(delta_lo * 0.5)
                candidates.append(delta_hi)
                candidates.append(delta_lo)
                
                # Filter to feasible
                feasible = [d for d in candidates if delta_lo <= d <= delta_hi]
                
                for d in feasible:
                    if abs(d) < 1e-12:
                        continue
                    h[i] += d
                    h[j] -= d
                    c5_try = compute_c5(h)
                    h[i] -= d  # undo
                    h[j] += d
                    if c5_try < best_val - 1e-14:
                        best_val = c5_try
                        best_delta = d
                
                if best_delta != 0.0:
                    h[i] += best_delta
                    h[j] -= best_delta
                    h = project_feasible(h, n)
                    c5_new = compute_c5(h)
                    if c5_new < best_c5 - 1e-14:
                        best_c5 = c5_new
                        best_h = h.copy()
                        improved = True
        
        if not improved:
            break
    
    return best_h, best_c5

                break

        if not improved:
            for step in [0.002, 0.001, 0.0005]:
                h_try = project_feasible(h - step * step_dir, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    break

        it += 1
    return best_h, best_c5


def pair_swap_cd(h, n, time_limit, rng_seed=42):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    rng = np.random.RandomState(rng_seed)

    while time.time() - start < time_limit and remaining_time() > 5:
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.time() - start > time_limit:
                break
            if remaining_time() < 3:
                break

            for attempt in range(4):
                j = rng.randint(n)
                if j == i:
                    continue

                delta_lo = max(-h[i], h[j] - 1.0)
                delta_hi = min(1.0 - h[i], h[j])
                if delta_hi - delta_lo < 1e-10:
                    continue

                best_delta = 0.0
                best_val = best_c5

                candidates = [0.002, -0.002, 0.005, -0.005, 0.01, -0.01,
                              0.03, -0.03, 0.08, -0.08, 0.001, -0.001]
                mid = (delta_lo + delta_hi) / 2.0
                candidates.append(mid)
                candidates.append(delta_hi * 0.3)
                candidates.append(delta_lo * 0.3)

                for d in candidates:
                    if d < delta_lo - 1e-12 or d > delta_hi + 1e-12:
                        continue
                    d = max(delta_lo, min(delta_hi, d))
                    if abs(d) < 1e-12:
                        continue
                    h[i] += d
                    h[j] -= d
                    c5_try = compute_c5(h)
                    h[i] -= d
                    h[j] += d
                    if c5_try < best_val - 1e-14:
                        best_val = c5_try
                        best_delta = d

                if best_delta != 0.0:
                    h[i] += best_delta
                    h[j] -= best_delta
                    h = project_feasible(h, n)
                    c5_new = compute_c5(h)
                    if c5_new < best_c5 - 1e-14:
                        best_c5 = c5_new
                        best_h = h.copy()
                        improved = True

        if not improved:
            # Try random perturbations to escape
            if time.time() - start < time_limit * 0.8:
                pert = rng.randn(n) * 0.01
                h_try = project_feasible(h + pert, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                else:
                    break

    return best_h, best_c5


def random_init(n, rng):
    h = rng.rand(n)
    h = project_feasible(h, n)
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    # Load initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
        c5_0 = compute_c5(h0)
    except Exception:
        n0 = 512
        h0 = np.ones(n0) * 0.5
        h0 = project_feasible(h0, n0)
        c5_0 = compute_c5(h0)

    best_h = h0.copy()
    best_c5 = c5_0
    best_n = n0

    # Phase 1: Quick subgradient at initial resolution
    if remaining_time() > 30:
        t1 = min(45.0, remaining_time() * 0.07)
        h1, c5_1 = subgradient_descent(h0, n0, t1)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: SLSQP smooth-max stages (increasing beta)
    h = best_h.copy()
    for beta in [20, 60, 200, 700, 2500, 8000, 25000]:
        if remaining_time() < 10:
            break
        maxit = 500 if beta < 200 else 350
        h = slsqp_smax(h, n0, beta, K=40, maxiter=maxit)
        c5_tmp = compute_c5(h)
        if c5_tmp < best_c5:
            best_c5 = c5_tmp
            best_h = h.copy()

    # Phase 3: Explicit min-max with expanding active set
    h = best_h.copy()
    if remaining_time() > 30:
        for K_a in [6, 10, 15, 20, 30, 40, 50]:
            if remaining_time() < 8:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            idx_a = np.argsort(-corr)[:K_a]
            h_new, c5_new = slsqp_tminmax(h, n0, idx_a, maxiter=350)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 4: Pair swap coordinate descent
    if remaining_time() > 40:
        t4 = min(80.0, remaining_time() * 0.12)
        h4, c5_4 = pair_swap_cd(best_h, n0, t4, rng_seed=99)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: SLSQP polish after pair swaps
    h = best_h.copy()
    for beta in [5000, 20000, 60000]:
        if remaining_time() < 8:
            break
        h = slsqp_smax(h, n0, beta, K=50, maxiter=500)
        c5_tmp = compute_c5(h)
        if c5_tmp < best_c5:
            best_c5 = c5_tmp
            best_h = h.copy()

    # Phase 6: More min-max
    h = best_h.copy()
    if remaining_time() > 25:
        for K_a in [25, 40, 60, 80]:
            if remaining_time() < 8:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            idx_a = np.argsort(-corr)[:K_a]
            h_new, c5_new = slsqp_tminmax(h, n0, idx_a, maxiter=400)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 7: Try different resolutions
    for n_try in [256, 384, 768, 1024]:
        if remaining_time() < 50:
            break
        if n_try == n0:
            continue

        h_interp = interp_to(best_h, best_n, n_try)
        h_interp = project_feasible(h_interp, n_try)
        c5_interp = compute_c5(h_interp)

        if c5_interp > best_c5 + 0.003:
            continue

        h_try = h_interp.copy()

        if remaining_time() > 35:
            t_sub = min(20.0, remaining_time() * 0.12)
            h_try, c5_try = subgradient_descent(h_try, n_try, t_sub)
            if compute_c5(h_try) > c5_interp:
                h_try = h_interp.copy()
                c5_try = c5_interp
            else:
                c5_try = compute_c5(h_try)

        for beta in [50, 500, 3000, 10000]:
            if remaining_time() < 10:
                break
            h_try = slsqp_smax(h_try, n_try, beta, K=40, maxiter=300)
            c5_try = compute_c5(h_try)

        if remaining_time() > 20:
            corr_t = np.correlate(h_try, 1.0 - h_try, mode='full')
            idx_t = np.argsort(-corr_t)[:25]
            h_try, c5_try = slsqp_tminmax(h_try, n_try, idx_t, maxiter=300)

        if remaining_time() > 25:
            h_try, c5_try = pair_swap_cd(h_try, n_try, min(20.0, remaining_time() * 0.15), rng_seed=n_try)

        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = n_try

    # Phase 8: Random restarts at lower resolution (to find different basins)
    if remaining_time() > 80:
        for n_small in [128, 192, 256]:
            if remaining_time() < 30:
                break
            best_small_c5 = compute_c5(best_h if best_n == n_small else interp_to(best_h, best_n, n_small))
            best_small_h = project_feasible(interp_to(best_h, best_n, n_small), n_small) if best_n != n_small else best_h.copy()

            n_restarts = 5
            for rs in range(n_restarts):
                if remaining_time() < 10:
                    break
                h_rand = random_init(n_small, rng)
                # Quick optimize
                h_r, c5_r = subgradient_descent(h_rand, n_small, min(8.0, remaining_time() * 0.05))
                h_r = slsqp_smax(h_r, n_small, 200, K=30, maxiter=200)
                c5_r = compute_c5(h_r)
                if c5_r < best_small_c5:
                    best_small_c5 = c5_r
                    best_small_h = h_r.copy()

            # If the small resolution solution is good, upscale and refine
            if best_small_c5 < best_c5 + 0.002:
                h_up = interp_to(best_small_h, n_small, max(n0, 512))
                h_up = project_feasible(h_up, max(n0, 512))
                c5_up = compute_c5(h_up)
                if c5_up < best_c5 + 0.001:
                    h_up = slsqp_smax(h_up, max(n0, 512), 500, K=40, maxiter=300)
                    c5_up = compute_c5(h_up)
                    if remaining_time() > 15:
                        corr_up = np.correlate(h_up, 1.0 - h_up, mode='full')
                        idx_up = np.argsort(-corr_up)[:20]
                        h_up, c5_up = slsqp_tminmax(h_up, max(n0, 512), idx_up, maxiter=250)
                    if c5_up < best_c5:
                        best_c5 = c5_up
                        best_h = h_up.copy()
                        best_n = max(n0, 512)

    # Phase 9: Final intensive min-max at best resolution
    h = best_h.copy()
    n = best_n
    if remaining_time() > 20:
        for K_a in [20, 35, 50, 70, 100]:
            if remaining_time() < 8:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            idx_a = np.argsort(-corr)[:K_a]
            h_new, c5_new = slsqp_tminmax(h, n, idx_a, maxiter=500)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 10: Final SLSQP with very high beta
    if remaining_time() > 15:
        for beta in [30000, 100000, 300000]:
            if remaining_time() < 5:
                break
            h = slsqp_smax(h, n, beta, K=60, maxiter=600)
            c5_f = compute_c5(h)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h.copy()

    # Phase 11: Final pair swap polish
    if remaining_time() > 15:
        h_ps, c5_ps = pair_swap_cd(best_h, n, min(25.0, remaining_time() * 0.4), rng_seed=777)
        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h_ps.copy()

    # Phase 12: Tiny gradient steps
    if remaining_time() > 10:
        h_t = best_h.copy()
        for it in range(300):
            if remaining_time() < 3:
                break
            corr_t = np.correlate(h_t, 1.0 - h_t, mode='full')
            idx_t = np.argsort(-corr_t)[:15]
            g_t = np.zeros(n)
            for k in idx_t:
                g_t += grad_corr_k_vec(h_t, k - (n - 1), n)
            g_t /= len(idx_t)
            gn_t = np.linalg.norm(g_t)
            if gn_t < 1e-15:
                break
            improved = False
            for step in [0.002, 0.001, 0.0005, 0.003, 0.0008, 0.0015]:
                h_try = project_feasible(h_t - step * g_t / gn_t, n)
                c_try = compute_c5(h_try)
                if c_try < best_c5 - 1e-15:
                    best_c5 = c_try
                    best_h = h_try.copy()
                    h_t = h_try.copy()
                    improved = True
                    break
            if not improved and it > 50:
                break

    # Final feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            best_c5 = min(best_c5, float(eval_c5))
    except Exception:
        pass

    return (best_h, best_c5, best_n)
