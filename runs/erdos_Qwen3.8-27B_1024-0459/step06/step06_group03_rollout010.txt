def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):

def grad_corr_k_vec(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        np_ = n - offset
        g[offset:] += (1.0 - h[:np_])
        g[:np_] -= h[offset:]
    else:
        a = -offset
        np_ = n - a
        g[:np_] += (1.0 - h[a:])
        g[a:] -= h[:np_]
    return g

def slsqop_tminmax(h, n, K_list, maxiter):
    global _G_BETA, _G_K
    # Minimize t subject to corr_k <= t*n/2 for k in K_list
    # Use SLSQP with explicit t variable
    t0 = compute_c5(h)
    v0 = np.concatenate([h, [t0]])
    n_plus = n + 1
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    # Constraint: corr_k - t*n/2 <= 0 for each k in K_list
    # SLSQP ineq: fun >= 0, so we need t*n/2 - corr_k >= 0
    
    def fun_ineq(v):
        hh = v[:n]
        t = v[-1]
        corr = get_corr(hh)
        return t * n / 2.0 - corr[K_list]
    
    def jac_ineq(v):
        hh = v[:n]
        J = np.zeros((len(K_list), n_plus))
        for i, k in enumerate(K_list):
            J[i, :n] = -grad_corr_k_vec(hh, k, n)
            J[i, -1] = n / 2.0
        return J
    
    def fun_eq(v):
        return np.sum(v[:n]) - n / 2.0
    
    def jac_eq(v):
        g = np.zeros(n_plus)
        g[:n] = 1.0
        return g
    
    def fun_obj(v):
        return v[-1]
    
    def jac_obj(v):
        g = np.zeros(n_plus)
        g[-1] = 1.0
        return g
    
    cons = [
        {'type': 'ineq', 'fun': fun_ineq, 'jac': jac_ineq},
        {'type': 'eq', 'fun': fun_eq, 'jac': jac_eq}
    ]
    try:
        res = minimize(fun_obj, v0, jac=jac_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x[:n], n)
            c5_new = compute_c5(h_new)
            if c5_new < t0 + 1e-12:
                return h_new, c5_new
    except Exception:
        pass
    return h.copy(), t0
