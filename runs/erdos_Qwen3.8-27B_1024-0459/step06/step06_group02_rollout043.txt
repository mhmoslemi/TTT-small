import numpy as np
from scipy.optimize import minimize
import time

_BETA = 1000.0
_PENALTY = 1000.0
_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argsort(h)
        if diff > 0:
            for i in range(len(idx)):
                j = idx[len(idx) - 1 - i]
                add = min(diff, 1.0 - h[j])
                h[j] += add
                diff -= add
                if diff <= 1e-15:
                    break
        else:
            for i in range(len(idx)):
                j = idx[i]
                sub = min(-diff, h[j])
                h[j] -= sub
                diff += sub
                if diff >= -1e-15:
                    break
    return h


def _grad_from_weights(x, weights, n):
    """Compute gradient given softmax weights over correlation lags."""
    g = np.zeros(n)
    for k in range(len(weights)):
        w = weights[k]
        if w < 1e-15:
            continue
        tau = k - (n - 1)
        if tau >= 0:
            npairs = n - tau
            if npairs > 0:
                g[tau:tau + npairs] += w * (1.0 - x[:npairs])
                g[:npairs] -= w * x[tau:tau + npairs]
        else:
            a = -tau
            npairs = n - a
            if npairs > 0:
                g[:npairs] += w * (1.0 - x[a:a + npairs])
                g[a:a + npairs] -= w * x[:npairs]
    return g


def smooth_max_obj(x):
    beta = _BETA
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = beta * (corr - m)
    clipped = np.clip(shifted, -80.0, 80.0)
    ex = np.exp(clipped)
    n = len(x)
    dx = 2.0 / n
    return (m + np.log(np.sum(ex)) / beta) * dx


def smooth_max_grad(x):
    beta = _BETA
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = beta * (corr - m)
    clipped = np.clip(shifted, -80.0, 80.0)
    ex = np.exp(clipped)
    s = np.sum(ex)
    n = len(x)
    dx = 2.0 / n
    weights = ex / s * dx
    return _grad_from_weights(x, weights, n)


def eq_fun(x):
    return np.sum(x) - len(x) / 2.0


def eq_jac(x):
    return np.ones(len(x))


def slsqp_optimize(h, n, beta, maxiter=500):
    global _BETA
    _BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    c_old = compute_c5(h)
    try:
        res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None and res.x.shape == (n,):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new)
            if c_new <= c_old + 1e-10:
                return h_new
    except Exception:
        pass
    return h.copy()

def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def nesterov_subgrad(h, n, max_iter=3000, time_limit=120.0, alpha0=0.04):
    start = time.time()
    x = project_feasible(h, n).copy()
    x_prev = x.copy()
    y = x.copy()
    best_x = x.copy()
    best_c = compute_c5(x)

    for k in range(max_iter):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(y)
        m = np.max(corr)
        K = min(40, len(corr))
        top_idx = np.argsort(-corr)[:K]
        top_vals = corr[top_idx]
        temp = 30.0
        w = np.exp(temp * (top_vals - top_vals.max()))
        ws = np.sum(w)
        if ws < 1e-30:
            w = np.ones(K) / K
        else:
            w = w / ws
        w_full = np.zeros(len(corr))
        w_full[top_idx] = w
        w_full *= 2.0 / n

        g = _grad_from_weights(y, w_full, n)

        alpha = alpha0 / (1.0 + 0.02 * k)
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (k + 1) ** 2)) / 2.0
        t_k1 = (1.0 + np.sqrt(1.0 + 4.0 * (k + 2) ** 2)) / 2.0

        x_new = project_feasible(y - alpha * g, n)
        mu = (t_k - 1.0) / t_k1
        y_new = x_new + mu * (x_new - x_prev)
        y = project_feasible(y_new, n)

        c = compute_c5(x_new)
        if c < best_c:
            best_c = c
            best_x = x_new.copy()

        x_prev = x_new
        x = x_new

    return best_x, best_c


def diverse_start(rng, n, best_h, strategy):

            h_div += 0.12 * np.cos(2 * np.pi * freq * x + phase)

        for _ in range(n // 32):
            start = rng.randint(0, n - 16)
            length = rng.randint(4, 20)
            h_div[start:start + length] = 1.0 - h_div[start:start + length]
    elif strategy == 6:
        # Shift the best solution
        shift = rng.randint(1, n // 4)
        h_div = np.roll(best_h, shift)
    else:
        h_div = best_h + 0.005 * rng.randn(n)
