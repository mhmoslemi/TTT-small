import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_binary(h, n, seed=42):
    h = np.array(h, dtype=float).copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    target = n // 2
    s = int(np.sum(h))
    rng = np.random.RandomState(seed)
    if s > target:
        ones = np.where(h == 1)[0]
        idx = rng.permutation(ones)
        h[idx[:s - target]] = 0.0
    elif s < target:
        zeros = np.where(h == 0)[0]
        idx = rng.permutation(zeros)
        h[idx[:target - s]] = 1.0
    return h


def compute_E(h, n):
    nfull = 2 * n - 1
    E = np.zeros((n, nfull))
    i_idx = np.arange(n)
    for k in range(n):
        idx = n - 1 + k
        v1 = i_idx + k < n
        E[v1, idx] += (1.0 - h[i_idx[v1] + k])
        v2 = i_idx - k >= 0
        E[v2, idx] -= h[i_idx[v2] - k]
    for a in range(1, n):
        idx = n - 1 - a
        v1 = i_idx - a >= 0
        E[v1, idx] += (1.0 - h[i_idx[v1] - a])
        v2 = i_idx + a < n
        E[v2, idx] -= h[i_idx[v2] + a]
    return E


def binary_local_search(h, n, time_limit=200, max_passes=500, seed=42):
    h = np.array(h, dtype=float).copy()
    rng = np.random.RandomState(seed)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    start = time.time()

        m1 = len(ones_idx)
        m0 = len(zeros_idx)

        order = rng.permutation(m1)
        improved = False
        for oi in order:
            i = ones_idx[oi]
            base = corr - E[i]
            if np.max(base) >= c_max + 0.5:

            new_maxs = np.max(base[None, :] + E[zeros_idx], axis=1)
            best_pos = int(np.argmin(new_maxs))

            if best_val < c_max - 1e-12:
                j = zeros_idx[best_pos]
                h[i] = 0.0

                corr = compute_corr(h)

def smooth_max_val(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -80.0, 80.0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * (2.0 / n)


def smooth_max_grad(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -80.0, 80.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= ws
    g = np.zeros(n)
    dx = 2.0 / n
    nf = 2 * n - 1
    for k in range(nf):
        wk = w[k]
        if abs(wk) < 1e-14:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:off + np_]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - h[a:a + np_])
            g[a:a + np_] -= wk * h[:np_]
    return g * dx


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        lo = h <= 1e-14
        hi = h >= 1.0 - 1e-14
        mid = ~(lo | hi)
        cnt = int(np.count_nonzero(mid))
        if cnt > 0:
            h[mid] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def slsqp_opt(h, n, beta, maxiter=400):
    for _attempt in range(3):
        try:
            x0 = h.copy() if _attempt == 0 else h + np.random.RandomState(_attempt + 7).randn(n) * 0.01
            x0 = project_feasible(x0, n)

            def _obj(x):
                return smooth_max_val(x, n, beta)

            def _grad(x):
                return smooth_max_grad(x, n, beta)

            def _eq(x):
                return np.sum(x) - n / 2.0

            def _ejac(x):
                return np.ones(n)

            res = minimize(_obj, x0, jac=_grad, method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq', 'fun': _eq, 'jac': _ejac}],
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            if res.x is not None and res.success:
                return project_feasible(res.x, n)
        except Exception:
            pass
    return h.copy()
