import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5_direct(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_h(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if mask.any():
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def autocorr_g_fft(g):
    n = len(g)
    N = 2 * n
    G = np.fft.rfft(g, n=N)
    acf = np.fft.irfft(G * np.conj(G), n=N)[:n]
    return acf


def grad_autocorr_k(g, k, n):
    if k == 0:
        return 2.0 * g.copy()
    if k >= n:
        return np.zeros(n)
    grad = np.zeros(n)
    grad[:n - k] += g[k:]
    grad[k:] += g[:n - k]
    return grad


def smooth_c5_obj_grad(g, alpha, dx, penalty):
    n = len(g)
    acf = autocorr_g_fft(g) * dx
    z = -alpha * acf
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    log_sum_exp = z_max + np.log(np.sum(exp_z))
    c5 = 0.5 + log_sum_exp / alpha
    w = exp_z / exp_z.sum()
    grad = np.zeros(n)
    thresh = 1e-8
    active = np.where(w > thresh)[0]
    for k in active:
        grad += w[k] * grad_autocorr_k(g, k, n)
    grad_c5 = -dx * grad
    s = g.sum()
    obj = c5 + 0.5 * penalty * s * s / n
    grad_total = grad_c5 + penalty * s / n
    return obj, grad_total


def lbfgsb_optimize(g, alpha, dx, penalty, maxiter=400):
    n = len(g)

    def fun(x):
        return smooth_c5_obj_grad(x, alpha, dx, penalty)[0]

    def jac(x):
        return smooth_c5_obj_grad(x, alpha, dx, penalty)[1]

    bounds = [(-0.5, 0.5)] * n
    try:
        res = minimize(fun, g, jac=jac, method='L-BFGS-B',
                       bounds=bounds, maxiter=maxiter,
                       options={'ftol': 1e-18, 'gtol': 1e-12,
                                'maxfun': maxiter * 5,
                                'maxls': 50})
        return res.x
    except Exception:
        return g.copy()

        idx = np.argsort(acf)[:K]
