Hmm, let me be more careful with the gradient. 

corr[k] (the k-th element of the full correlation from np.correlate) = ∑_i h[i] * (1 - h[i + k - (n-1)]) for valid i.

Let offset = k - (n-1). Then:
corr[k] = ∑_i h[i] * (1 - h[i + offset]) where i ranges so that both i and i+offset are in [0, n-1].

If offset >= 0: i ranges from 0 to n-1-offset
If offset < 0: i ranges from -offset to n-1

∂corr[k]/∂h[j]:
Case offset >= 0:
- j appears as h[i] when i = j, valid if j ∈ [0, n-1-offset]: contributes (1 - h[j+offset])
- j appears as h[i+offset] when i = j-offset, valid if j-offset ∈ [0, n-1-offset], i.e., j ∈ [offset, n-1]: contributes -h[j-offset]

So:
∂corr[k]/∂h[j] = (1 - h[j+offset]) * 1[j ∈ [0, n-1-offset]] - h[j-offset] * 1[j ∈ [offset, n-1]]

Case offset < 0: let a = -offset > 0
- j appears as h[i] when i = j, valid if j ∈ [a, n-1]: contributes (1 - h[j-a])
- j appears as h[i+offset] = h[i-a] when i-a = j, i.e., i = j+a, valid if j+a ∈ [a, n-1], i.e., j ∈ [0, n-1-a]: contributes -h[j+a]

So:
∂corr[k]/∂h[j] = (1 - h[j-a]) * 1[j ∈ [a, n-1]] - h[j+a] * 1[j ∈ [0, n-1-a]]

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def corr_fft(a, v):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    V = np.fft.rfft(v, N)
    C = np.fft.irfft(A * np.conj(V), N)
    return np.concatenate([C[n - 1::-1], C[N - 1:N - n:-1]])


def compute_c5_fast(h):
    n = len(h)
    dx = 2.0 / n
    corr = corr_fft(h, 1.0 - h)
    return float(np.max(corr)) * dx


def compute_c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr_fast(h):
    return corr_fft(h, 1.0 - h)


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        g[:npairs] += (1.0 - h[offset:])
        g[offset:] -= h[:npairs]
    else:
        a = -offset
        npairs = n - a
        g[a:] += (1.0 - h[:npairs])
        g[:npairs] -= h[a:]
    return g


def smooth_obj_grad(h, n, beta, penalty):
    corr = compute_corr_fast(h)
    m = np.max(corr)
    sh = beta * (corr - m)
    sh = np.clip(sh, -500.0, 500.0)
    ex = np.exp(sh)
    sm = np.sum(ex)
    if sm < 1e-300:
        w = np.ones_like(corr) / len(corr)
        obj = (m + np.log(len(corr)) / beta) * (2.0 / n)
    else:
        w = ex / sm
        obj = (m + np.log(sm) / beta) * (2.0 / n)
    s = np.sum(h) - n / 2.0
    obj += 0.5 * penalty * s * s
    g = np.zeros(n)
    wc = w * (2.0 / n)
    thresh = 1e-12
    active = np.where(wc > thresh)[0]
    for k in active:
        g += wc[k] * grad_corr_k(h, k, n)
    g += penalty * s
    return obj, g


def lbfgsb_smooth(h, n, beta, penalty, maxiter=3000):
    def obj_fun(x):
        return smooth_obj_grad(x, n, beta, penalty)[0]

    def grad_fun(x):
        return smooth_obj_grad(x, n, beta, penalty)[1]

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fun, h, jac=grad_fun, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-14, 'maxfun': 50000,
                                'maxls': 30})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

        K = 25
        idx = np.argsort(-corr)[:K]

        beta_g = 60.0 + 2.0 * it
        weights = np.exp(beta_g * (c_top - c_top.max()))
        s = weights.sum()
        if s < 1e-30:

                g += weights[i] * grad_corr_k(h, k, n)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        step = 0.025 / (1.0 + 0.03 * it)
        h_new = project_feasible(h - step * g, n)
        c5_new = compute_c5_fast(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            found = False
            for factor in [0.5, 0.25, 0.1]:
                h_try = project_feasible(h - step * factor * g, n)
                c_try = compute_c5_fast(h_try)
                if c_try < best_c5:
                    best_c5 = c_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    found = True
                    break
            if not found and it % 100 == 0:
                h = best_h.copy()
    return best_h, best_c5


def coordinate_descent(h, n, time_limit):
    global _T0, _BUDGET
    h = h.copy()
    best_c5 = compute_c5_fast(h)
    start = time.time()
    improved_any = True
    while improved_any and time.time() - start < time_limit:
        if _BUDGET - (time.time() - _T0) < 5:
            break
        improved_any = False
        for i in range(n):
            if time.time() - start > time_limit:
                break
            old_val = h[i]
            best_c = best_c5
            best_v = old_val
            for cv in [0.0, 1.0]:
                if abs(cv - old_val) < 1e-10:
                    continue
                h[i] = cv
                h = project_feasible(h, n)
                c = compute_c5_fast(h)
                if c < best_c - 1e-15:
                    best_c = c
                    best_v = cv
            if best_v != old_val:
                h[i] = best_v
                h = project_feasible(h, n)
                improved_any = True
            else:
                h[i] = old_val
    return h, best_c5


def equalize_shifts(h, n, K=10, max_iters=300):
    global _T0, _BUDGET
    h = h.copy()
    best_c5 = compute_c5_fast(h)
    for it in range(max_iters):
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        corr = compute_corr_fast(h)
        idx = np.argsort(-corr)[:K]
        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        if c_max - c_min < 1e-10:
            break
        g_avg = np.zeros(n)
        for k in idx:
            g_avg += grad_corr_k(h, k, n)
        g_avg /= K
        g_dir = -g_avg
        gn = np.linalg.norm(g_dir)
        if gn < 1e-15:
            break
        g_dir /= gn
        improved = False
        for step in [0.01, 0.02, 0.005, 0.015, 0.03, 0.008]:
            ht = project_feasible(h + step * g_dir, n)
            cn = compute_c5_fast(ht)
            if cn < best_c5 - 1e-15:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            for step in [0.01, 0.02, 0.005, 0.015]:
                ht = project_feasible(h - step * g_dir, n)
                cn = compute_c5_fast(ht)
                if cn < best_c5 - 1e-15:
                    h = ht
                    best_c5 = cn
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5


def cutting_plane_lp(h, n, K=50, delta=0.03):
    corr = compute_corr_fast(h)
    c_max = float(np.max(corr))
    idx = np.argsort(-corr)[:K]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub_list = []
    b_ub_list = []
    for k in idx:
        g = grad_corr_k(h, k, n)
        fk = corr[k]
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_ub_list.append(row)
        b_ub_list.append(-fk + np.dot(g, h))
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    I = np.eye(n)
    A_trust = np.vstack([np.hstack([I, np.zeros((n, 1))]),
                         np.hstack([-I, np.zeros((n, 1))])])
    b_trust = np.concatenate([h + delta, np.maximum(1.0 - h, delta)])
    A_ub = np.vstack([A_ub, A_trust])
    b_ub = np.concatenate([b_ub, b_trust])
    bounds = [(0, 1)] * n + [(None, None)]
    from scipy.optimize import linprog
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return h.copy(), c_max


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)


def binary_local_search(h, n, time_limit, rng):
    global _T0, _BUDGET
    h = h.copy()
    best_c5 = compute_c5_fast(h)
    best_h = h.copy()
    start = time.time()
    while time.time() - start < time_limit:
        if _BUDGET - (time.time() - _T0) < 3:
            break
        zeros = np.where(h < 0.15)[0]
        ones = np.where(h > 0.85)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        try_best = False
        for _ in range(50):
            if time.time() - start > time_limit:
                break
            zi = zeros[rng.randint(len(zeros))]
            oi = ones[rng.randint(len(ones))]
            h[zi] = 1.0
            h[oi] = 0.0
            h = project_feasible(h, n)
            c_try = compute_c5_fast(h)
            if c_try < best_c5 - 1e-15:
                best_c5 = c_try
                best_h = h.copy()
                try_best = True
            else:
                h = best_h.copy()
                break
        if not try_best:
            if rng.rand() < 0.3:
                return best_h, best_c5
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5_fast(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 512
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5_fast(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Interpolate to n=4096
    n_main = 4096
    h_main = interp_to(best_h, best_n, n_main)
    h_main = project_feasible(h_main, n_main)
    c5_main = compute_c5_fast(h_main)
    if c5_main < best_c5:
        best_c5 = c5_main
        best_h = h_main.copy()
        best_n = n_main

    h = best_h.copy()
    n = best_n

    # Phase 1: Quick subgradient descent (15s)
    if remaining_time() > 25:
        h_sg, c5_sg = subgradient_descent(h, n, 15.0, rng)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 2: L-BFGS-B with increasing beta (main optimization, ~300s)
    beta_schedule = [100, 300, 800, 2000, 5000, 10000, 20000]
    for beta in beta_schedule:
        if remaining_time() < 25:
            break
        h_new = lbfgsb_smooth(h, n, beta, 5000, maxiter=2500)
        c5_new = compute_c5_fast(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 3: Equalize shifts (30s)
    if remaining_time() > 40:
        h_eq, c5_eq = equalize_shifts(h, n, K=12, max_iters=200)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 4: More L-BFGS-B with very high beta
    for beta in [30000, 60000, 100000]:
        if remaining_time() < 20:
            break
        h_new = lbfgsb_smooth(h, n, beta, 10000, maxiter=3000)
        c5_new = compute_c5_fast(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 5: Cutting plane LP refinement (60s)
    if remaining_time() > 70:
        h_cp = h.copy()
        c5_cp = best_c5
        cp_start = time.time()
        for it in range(200):
            if time.time() - cp_start > 60:
                break
            if _BUDGET - (time.time() - _T0) < 5:
                break
            h_lp, t_lp = cutting_plane_lp(h_cp, n, K=40, delta=0.02)
            cur_c5 = compute_c5_fast(h_cp)
            if t_lp < cur_c5 - 1e-10:
                best_val = cur_c5
                best_alpha = 0
                for alpha in [1.0, 0.7, 0.5, 0.3, 0.2, 0.8, 0.6, 0.4]:
                    h_try = project_feasible(h_cp + alpha * (h_lp - h_cp), n)
                    c_try = compute_c5_fast(h_try)
                    if c_try < best_val:
                        best_val = c_try
                        best_alpha = alpha
                if best_alpha > 0:
                    h_cp = project_feasible(h_cp + best_alpha * (h_lp - h_cp), n)
                    c_try = compute_c5_fast(h_cp)
                    if c_try < c5_cp:
                        c5_cp = c_try
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = h_cp.copy()
                        h = h_cp.copy()
            elif it > 80:
                break

    # Phase 6: Subgradient refinement (20s)
    if remaining_time() > 30:
        h_sg2, c5_sg2 = subgradient_descent(h, n, 20.0, rng)
        if c5_sg2 < best_c5:
            best_c5 = c5_sg2
            best_h = h_sg2.copy()
            h = h_sg2.copy()

    # Phase 7: L-BFGS-B final with extreme beta
    if remaining_time() > 30:
        h_new = lbfgsb_smooth(h, n, 200000, 20000, maxiter=3000)
        c5_new = compute_c5_fast(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 8: Coordinate descent (40s)
    if remaining_time() > 50:
        h_cd, c5_cd = coordinate_descent(h, n, min(40.0, remaining_time() * 0.35))
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h = h_cd.copy()

    # Phase 9: Try n=8192
    if remaining_time() > 120 and best_n < 8192:
        n_vh = 8192
        h_vh = interp_to(best_h, best_n, n_vh)
        h_vh = project_feasible(h_vh, n_vh)
        c5_vh = compute_c5_fast(h_vh)
        if c5_vh <= best_c5 + 0.001:
            h_vh = lbfgsb_smooth(h_vh, n_vh, 500, 5000, maxiter=1000)
            c5_vh = compute_c5_fast(h_vh)
            if remaining_time() > 40:
                h_vh = lbfgsb_smooth(h_vh, n_vh, 5000, 10000, maxiter=2000)
                c5_vh = compute_c5_fast(h_vh)
            if remaining_time() > 30:
                h_vh = lbfgsb_smooth(h_vh, n_vh, 20000, 10000, maxiter=1500)
                c5_vh = compute_c5_fast(h_vh)
            if c5_vh < best_c5:
                best_c5 = c5_vh
                best_h = h_vh.copy()
                best_n = n_vh

    # Phase 10: Equalize shifts on best solution
    if remaining_time() > 25:
        h_eq2, c5_eq2 = equalize_shifts(best_h, best_n, K=15, max_iters=200)
        if c5_eq2 < best_c5:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()

    # Phase 11: Final L-BFGS-B polish
    if remaining_time() > 20:
        h_new = lbfgsb_smooth(best_h, best_n, 50000, 20000, maxiter=2000)
        c5_new = compute_c5_fast(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 12: Binary local search
    if remaining_time() > 30:
        h_bs, c5_bs = binary_local_search(best_h, best_n,
                                          min(25.0, remaining_time() * 0.5), rng)
        if c5_bs < best_c5:
            best_c5 = c5_bs
            best_h = h_bs.copy()

    # Phase 13: Final equalize + L-BFGS-B
    if remaining_time() > 30:
        h_eq3, c5_eq3 = equalize_shifts(best_h, best_n, K=8, max_iters=100)
        if c5_eq3 < best_c5:
            best_c5 = c5_eq3
            best_h = h_eq3.copy()
        if remaining_time() > 15:
            h_new = lbfgsb_smooth(best_h, best_n, 100000, 30000, maxiter=1500)
            c5_new = compute_c5_fast(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Final feasibility check and exact evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5_exact(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
