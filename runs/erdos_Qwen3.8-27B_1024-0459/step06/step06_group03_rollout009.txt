import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_G_N = 512
_G_ACTIVE = None


def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(600):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_corr_k(h, k_idx, n):
    g = np.zeros(n)
    offset = k_idx - (n - 1)
    if offset >= 0:
        np_ = n - offset
        g[offset:] += (1.0 - h[:np_])
        g[:np_] -= h[offset:]
    else:
        a = -offset
        np_ = n - a
        g[:np_] += (1.0 - h[a:])
        g[a:] -= h[:np_]
    return g


def get_remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

def am_obj(v):
    return v[-1]

def am_obj_jac(v):

    return g

    n = _G_N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return t * n / 2.0 - corr[_G_ACTIVE]

def am_ineq_jac(v):
    h = v[:_G_N]
    n = _G_N
    active = _G_ACTIVE
    J = np.zeros((len(active), n + 1))
    for i, k in enumerate(active):

        if offset >= 0:
            np_ = n - offset
            J[i, offset:] += (1.0 - h[:np_])
            J[i, :np_] -= h[offset:]
        else:
            a = -offset
            np_ = n - a
            J[i, :np_] += (1.0 - h[a:])
            J[i, a:] -= h[:np_]
        J[i, -1] = n / 2.0
    return J

def am_eq_fun(v):
    return np.sum(v[:_G_N]) - _G_N / 2.0

def am_eq_jac(v):
    g = np.zeros(_G_N + 1)
    g[:_G_N] = 1.0
    return g

def solve_active_minmax(h, n, active_set, maxiter=500):
    global _G_N, _G_ACTIVE
    _G_N = n
    _G_ACTIVE = active_set
    c5 = compute_c5(h)
    v0 = np.concatenate([h, [c5]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'ineq', 'fun': am_ineq_fun, 'jac': am_ineq_jac},
        {'type': 'eq', 'fun': am_eq_fun, 'jac': am_eq_jac}
    ]
    try:
        res = minimize(am_obj, v0, jac=am_obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:

            return h_new, compute_c5(h_new)
    except Exception:

def sm_obj(v):
    n = _G_N
    h = v[:n]
    corr = np.correlate(h, 1.0 - h, mode='full')
    beta = 50000.0
    K = 60
    m = corr.max()

    c = corr[idx]
    sh = c - c.max()
    cl = np.clip(beta * sh, -50.0, 50.0)
    ex = np.exp(cl)
    return (c.max() + np.log(ex.sum()) / beta) * (2.0 / n)

def sm_jac(v):
    n = _G_N
    h = v[:n]
    corr = np.correlate(h, 1.0 - h, mode='full')
    beta = 50000.0
    K = 60
    idx = np.argsort(-corr)[:K]
    c = corr[idx]
    sh = c - c.max()
    cl = np.clip(beta * sh, -50.0, 50.0)
    ex = np.exp(cl)
    w = ex / ex.sum()
    g = np.zeros(n)
    dx = 2.0 / n

            np_ = n - a
            g[:np_] += wk * (1.0 - h[a:])
            g[a:] -= h[:np_]
    return g

                    h = project_feasible(h, n)
