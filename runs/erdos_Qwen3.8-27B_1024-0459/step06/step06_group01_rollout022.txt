```python
            if off >= 0:

                g[off:] += wi * (1.0 - h[:np_])
                g[:np_] -= wi * h[off:]
            else:
                a = -off
                np_ = n - a
                g[:np_] += wi * (1.0 - h[a:])
                g[a:] -= wi * h[:np_]
        g *= dx
        
        # Project to tangent space of sum constraint
        g -= np.mean(g)
        
        # Adam
        m_buf = b1 * m_buf + (1 - b1) * g
        v_buf = b2 * v_buf + (1 - b2) * g * g
        m_hat = m_buf / (1 - b1 ** t_cnt)
        v_hat = v_buf / (1 - b2 ** t_cnt)
        
        lr_t = lr / (1 + 0.005 * t_cnt)
        h_new = project_feasible(h - lr_t * m_hat / (np.sqrt(v_hat) + eps), n)

        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            stuck = 0
        else:
            stuck += 1
        h = h_new
        if stuck > 2000:
            break
    return best_h, best_c5

def theta_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)
```

For off ≥ 0:
```python

```

