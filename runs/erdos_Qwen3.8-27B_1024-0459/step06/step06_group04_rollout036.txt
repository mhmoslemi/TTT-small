def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        # Distribute diff to interior points
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_bisect_subproblem(c, B, h_cur):
    """Solve 1D subproblem: find t in [0,1] minimizing max_k (c[k] + B[k]*(t - h_cur))
    
    c: array of correlation values (length 2n-1)
    B: array of slopes (length 2n-1)  
    h_cur: current value of h[i]
    
    Returns optimal t.
    """
    d_lo = -h_cur  # t = 0
    d_hi = 1.0 - h_cur  # t = 1
    
    def f_at(d):
        return np.max(c + B * d)
    
    # Check endpoints
    f_lo = f_at(d_lo)
    f_hi = f_at(d_hi)
    
    if f_lo <= f_hi:
        # Try to see if minimum is at d_lo or to the right
        # Subgradient at d_lo: B[k] for k where c[k] + B[k]*d_lo = f_lo
        active = np.abs(c + B * d_lo - f_lo) < 1e-12
        if active.any():
            min_B = B[active].min()
            max_B = B[active].max()
            if min_B <= 0 <= max_B:
                return 0.0  # d_lo is optimal

                # Function decreasing at d_lo, minimum is to the right
                pass
        # Minimum is interior or at d_hi

        lo, hi = d_lo, d_hi
        for _ in range(30):

            f_mid = f_at(mid)
            active = np.abs(c + B * mid - f_mid) < 1e-10

            min_B = B[active].min()
            if min_B > 0:
                # All active slopes positive, minimum is to the left
                hi = mid

            return 0.0
        else:
            return 1.0

            min_B = B[active].min()
            max_B = B[active].max()
            if min_B <= 0 <= max_B:

    """Run coordinate descent with exact 1D subproblems."""
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 10:
            break
        
        # Compute current correlations
        c = compute_corr(h)
        
        # Cycle through variables in random order (or sequential)
        order = np.arange(n)
        np.random.shuffle(order)
        
        improved = False
        for i in order:
            o_range = np.arange(-(n-1), n)  # offsets from -(n-1) to n-1
            m_idx = o_range + (n - 1)  # corresponding correlation indices
            
            # Compute B_m for this i
            # B_m = (1-h[i-o]) * first_valid - h[i+o] * second_valid
            # where h[j] = 0 for j outside [0, n-1]
            
            B = np.zeros(n_corr)
            
            for j, o in enumerate(o_range):
                if o >= 0:
                    first_valid = (o <= i <= n - 1)
                    second_valid = (0 <= i <= n - 1 - o)
                    idx1 = i - o
                    idx2 = i + o
                    v1 = (1.0 - h[idx1]) if 0 <= idx1 < n else 1.0
                    v2 = h[idx2] if 0 <= idx2 < n else 0.0
                    B[m_idx[j]] = first_valid * v1 - second_valid * v2
                else:
                    j_neg = -o
                    first_valid = (0 <= i <= n - 1 - j_neg)
                    second_valid = (j_neg <= i <= n - 1)
                    idx1 = i - o  # = i + j_neg
                    idx2 = i + o  # = i - j_neg
                    v1 = (1.0 - h[idx1]) if 0 <= idx1 < n else 1.0
                    v2 = h[idx2] if 0 <= idx2 < n else 0.0
                    B[m_idx[j]] = first_valid * v1 - second_valid * v2
            
            # Solve 1D subproblem
            t_opt = coord_bisect_subproblem(c, B, h[i])
            
            if abs(t_opt - h[i]) > 1e-10:

        if c5_now < best_c5:

def solve_1d_fast(c, B_col, h_cur_val):
    """Find t in [0,1] minimizing max_m(c[m] + B_col[m]*(t - h_cur_val))"""
    d_lo = -h_cur_val
    d_hi = 1.0 - h_cur_val
    
    vals_lo = c + B_col * d_lo
    f_lo = vals_lo.max()
    vals_hi = c + B_col * d_hi
    f_hi = vals_hi.max()
    
    if f_lo <= f_hi:
        # Check if d_lo is optimal
        active = vals_lo >= f_lo - 1e-14
        if active.any():
            max_slope = B_col[active].max()
            min_slope = B_col[active].min()
            if min_slope <= 0:
                # 0 is in or below subgradient → d_lo is (approximately) optimal
                return 0.0
        # Need to search to the right
        lo, hi = d_lo, d_hi
        for _ in range(25):
            mid = (lo + hi) / 2
            vals = c + B_col * mid
            f_mid = vals.max()
            if f_mid >= f_hi:
                hi = mid
            else:
                # Check subgradient
                active = vals >= f_mid - 1e-14
                if active.any():
                    min_slope = B_col[active].min()
                    if min_slope > 0:
                        hi = mid
                        continue
                lo = mid
        t = (lo + hi) / 2 + h_cur_val
        # Verify
        if c + B_col * (t - h_cur_val) is not NaN:  # always true, just checking logic
            f_check = (c + B_col * (t - h_cur_val)).max()
            if f_check > f_lo:
                return 0.0
        return np.clip(t, 0.0, 1.0)
    else:
        # f_hi < f_lo
        active = vals_hi >= f_hi - 1e-14
        if active.any():
            max_slope = B_col[active].max()
            min_slope = B_col[active].min()
            if max_slope >= 0:
                return 1.0
        lo, hi = d_lo, d_hi
        for _ in range(25):
            mid = (lo + hi) / 2
            vals = c + B_col * mid
            f_mid = vals.max()
            if f_mid >= f_lo:
                lo = mid
            else:
                active = vals >= f_mid - 1e-14
                if active.any():
                    max_slope = B_col[active].max()
                    if max_slope < 0:

                        continue

        t = (lo + hi) / 2 + h_cur_val
        f_check = (c + B_col * (t - h_cur_val)).max()
        if f_check > f_hi:
            return 1.0
        return np.clip(t, 0.0, 1.0)

    v2[valid2] = h[idx2[valid2]]

    return first_valid * v1 - second_valid * v2


def solve_1d_subproblem(c, B_col, h_cur):
    """Find t in [0,1] minimizing max_m(c[m] + B_col[m]*(t - h_cur))."""
    d_lo = -h_cur
    d_hi = 1.0 - h_cur

    vals_lo = c + B_col * d_lo
    f_lo = vals_lo.max()
    vals_hi = c + B_col * d_hi
    f_hi = vals_hi.max()

    if f_lo <= f_hi:
        active = vals_lo >= f_lo - 1e-14
        if active.any():
            min_slope = B_col[active].min()
            if min_slope <= 0:
                return 0.0
        # Bisection
        lo, hi = d_lo, d_hi
        for _ in range(30):
            mid = (lo + hi) / 2.0
            f_mid = (c + B_col * mid).max()
            if f_mid >= f_hi:
                hi = mid
            else:
                vals = c + B_col * mid
                active = vals >= vals.max() - 1e-14
                if active.any() and B_col[active].min() > 0:
                    hi = mid
                else:
                    lo = mid
        t = lo + h_cur
        f_check = (c + B_col * (t - h_cur)).max()
        if f_check > f_lo + 1e-15:
            return 0.0
        return np.clip(t, 0.0, 1.0)
    else:
        active = vals_hi >= f_hi - 1e-14
        if active.any():
            max_slope = B_col[active].max()
            if max_slope >= 0:
                return 1.0
        lo, hi = d_lo, d_hi
        for _ in range(30):
            mid = (lo + hi) / 2.0
            f_mid = (c + B_col * mid).max()
            if f_mid >= f_lo:
                lo = mid
            else:
                vals = c + B_col * mid
                active = vals >= vals.max() - 1e-14
                if active.any() and B_col[active].max() < 0:
                    lo = mid
                else:
                    hi = mid
        t = hi + h_cur
        f_check = (c + B_col * (t - h_cur)).max()
        if f_check > f_hi + 1e-15:
            return 1.0
        return np.clip(t, 0.0, 1.0)


def coord_descent_cont(h, n, max_sweeps, time_limit, rng):
    """Continuous coordinate descent with exact 1D subproblems."""
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 10:
            break

        c = compute_corr(h)
        order = rng.permutation(n)
        total_change = 0.0

        for i in order:
            B_col = compute_B_col(h, i, n)
            t_opt = solve_1d_subproblem(c, B_col, h[i])
            delta = t_opt - h[i]
            if abs(delta) > 1e-12:
                h[i] = t_opt
                total_change += abs(delta)

        h = project_feasible(h, n)
        c5_now = compute_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()

        if total_change < 1e-8 * n and sweep > 3:
            break

    return best_h, best_c5


def binary_coord_descent(h, n, max_sweeps, time_limit, rng):
    """Binary coordinate descent: for each i, try 0 and 1."""
    h = (h > 0.5).astype(float)
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 10:
            break

        c = compute_corr(h)
        order = rng.permutation(n)
        improved = False

        for i in order:
            if h[i] > 0.5:
                t_try = 0.0
            else:
                t_try = 1.0
            B_col = compute_B_col(h, i, n)
            d = t_try - h[i]
            f_try = (c + B_col * d).max()
            f_cur = c.max()
            if f_try < f_cur - 1e-14:
                h[i] = t_try
                improved = True
                # Update c incrementally (first order)
                c += d * B_col

        h = project_feasible(h, n)
        c5_now = compute_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
        elif not improved and sweep > 5:
            break

    return best_h, best_c5


def smooth_max_func(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = corr.max()
    s = np.clip(beta * (corr - m), -500.0, 500.0)
    e = np.exp(s)
    return (m + np.log(e.sum()) / beta) * dx


def smooth_max_jac(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = corr.max()
    s = np.clip(beta * (corr - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    o = np.arange(-(n - 1), n)
    # Vectorized gradient computation
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        off = o[k]
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - h[ao:])
            g[ao:] -= wk * h[:np_]
    return g


def slsqp_optimize(h, beta, maxiter, n):
    from scipy.optimize import minimize
    try:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
                 'jac': lambda x: np.ones(n)}]
        res = minimize(smooth_max_func, h, args=(beta,),
                       jac=smooth_max_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None and res.success:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def make_thue_morse(n):
    """Create Thue-Morse inspired balanced sequence of length n."""
    t = np.zeros(n)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    t = t.astype(float)
    # Adjust to have exactly n/2 ones
    ones = int(t.sum())
    target = n // 2
    if ones > target:
        idx = np.where(t == 1)[0]
        np.random.shuffle(idx)
        t[idx[:ones - target]] = 0
    elif ones < target:
        idx = np.where(t == 0)[0]
        np.random.shuffle(idx)
        t[idx[:target - ones]] = 1
    return t


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def rem():
        return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

    # Get initial solution
    try:
        best_h = np.array(initial_h_values, dtype=float).copy()
        best_n = len(best_h)
        best_c5 = compute_c5(best_h)
    except Exception:
        best_n = 512
        best_h = make_thue_morse(best_n)
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)

    # Phase 1: Coordinate descent at parent resolution
    if rem() > 30:
        t1 = min(120.0, rem() * 0.15)
        h1, c1 = coord_descent_cont(best_h, best_n, max_sweeps=100, time_limit=t1, rng=rng)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1.copy()

    # Phase 2: Binary coordinate descent at parent resolution
    if rem() > 30:
        t2 = min(60.0, rem() * 0.1)
        h2, c2 = binary_coord_descent(best_h, best_n, max_sweeps=200, time_limit=t2, rng=rng)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()

    # Phase 3: Interpolate to higher resolution and optimize
    target_n = 1024
    if rem() > 60:
        h_interp = interp_to(best_h, best_n, target_n)
        h_interp = project_feasible(h_interp, target_n)
        c_interp = compute_c5(h_interp)

        if rem() > 50:
            t3 = min(200.0, rem() * 0.25)
            h3, c3 = coord_descent_cont(h_interp, target_n, max_sweeps=80, time_limit=t3, rng=rng)
            if c3 < best_c5:
                best_c5 = c3
                best_h = h3.copy()
                best_n = target_n

        if rem() > 30 and best_n == target_n:
            t4 = min(60.0, rem() * 0.1)
            h4, c4 = binary_coord_descent(best_h, target_n, max_sweeps=300, time_limit=t4, rng=rng)
            if c4 < best_c5:
                best_c5 = c4
                best_h = h4.copy()

    # Phase 4: SLSQP polish with smooth max
    if rem() > 40:
        for beta in [5000, 20000, 100000]:
            if rem() < 15:
                break
            h_s = slsqp_optimize(best_h, beta, maxiter=200, n=best_n)
            c_s = compute_c5(h_s)
            if c_s < best_c5:
                best_c5 = c_s
                best_h = h_s.copy()

    # Phase 5: More coordinate descent at best resolution
    if rem() > 30:
        t5 = min(120.0, rem() * 0.2)
        h5, c5 = coord_descent_cont(best_h, best_n, max_sweeps=100, time_limit=t5, rng=rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h5.copy()

    # Phase 6: Binary refinement at final resolution
    if rem() > 30:
        t6 = min(80.0, rem() * 0.15)
        h6, c6 = binary_coord_descent(best_h, best_n, max_sweeps=500, time_limit=t6, rng=rng)
        if c6 < best_c5:
            best_c5 = c6
            best_h = h6.copy()

    # Phase 7: Try at even higher resolution if time allows
    if rem() > 100 and best_n < 2048:
        target_n2 = min(2048, best_n * 2)
        h_int2 = interp_to(best_h, best_n, target_n2)
        h_int2 = project_feasible(h_int2, target_n2)
        t7 = min(150.0, rem() * 0.4)
        h7, c7 = coord_descent_cont(h_int2, target_n2, max_sweeps=60, time_limit=t7, rng=rng)
        if c7 < best_c5:
            best_c5 = c7
            best_h = h7.copy()
            best_n = target_n2
        if rem() > 30:
            h8, c8 = binary_coord_descent(best_h, best_n, max_sweeps=300, time_limit=min(60, rem() * 0.2), rng=rng)
            if c8 < best_c5:
                best_c5 = c8
                best_h = h8.copy()

    # Phase 8: Final SLSQP polish
    if rem() > 20:
        for beta in [200000, 500000]:
            if rem() < 10:
                break
            h_f = slsqp_optimize(best_h, beta, maxiter=300, n=best_n)
            c_f = compute_c5(h_f)
            if c_f < best_c5:
                best_c5 = c_f
                best_h = h_f.copy()

    # Phase 9: Final coordinate descent
    if rem() > 15:
        h_f2, c_f2 = coord_descent_cont(best_h, best_n, max_sweeps=50, time_limit=min(30, rem() * 0.5), rng=rng)
        if c_f2 < best_c5:
            best_c5 = c_f2
            best_h = h_f2.copy()

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
