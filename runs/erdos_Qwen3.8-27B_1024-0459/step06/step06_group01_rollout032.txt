```python
import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def topk_grad_direct(h, n, top_k=5):
    dx = 2.0 / n
    corr = compute_corr(h)
    idx_top = np.argsort(-corr)[:top_k]
    g = np.zeros(n)
    for ki in idx_top:
        off = ki - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += (1.0 - h[:np_])
            g[:np_] -= h[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += (1.0 - h[a:])
            g[a:] -= h[:np_]
    g *= dx / top_k
    return g


def adam_optimize(h, n, max_iters, lr=0.02, top_k=5, beta1=0.9, beta2=0.999, eps=1e-8, rng_seed=42):
    h = project_feasible(h, n).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    rng = np.random.RandomState(rng_seed)
    max_no_improve = 800

    for t in range(1, max_iters + 1):
        if remaining_time() < 10:
            break
        g = topk_grad_direct(h, n, top_k)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h - step, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            h_half = project_feasible(h - 0.5 * step, n)
            c5_half = compute_c5(h_half)
            if c5_half < best_c5 - 1e-15:
                best_c5 = c5_half
                best_h = h_half.copy()
                h = h_half.copy()
                no_improve = 0
            else:
                h = h_half
        if no_improve > max_no_improve:
            pert = rng.uniform(-0.03, 0.03, n)
            h = project_feasible(best_h + pert, n)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
    return best_h, best_c5


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def topk_equalize(h, n, top_k=5, max_iters=500):
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining_time() < 3:
            break
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        c_max = corr[idx_top[0]]
        c_min_top = corr[idx_top[-1]]
        if c_max - c_min_top < 1e-10:
            break
        g_dir = np.zeros(n)
        for pos, k in enumerate(idx_top):
            w_k = 1.0 if pos == 0 else (-1.0 / (len(idx_top) - 1))
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_dir[off:] += w_k * (1.0 - h[:np_])
                g_dir[:np_] -= w_k * h[off:]
            else:
                a = -off
                np_ = n - a
                g_dir[:np_] += w_k * (1.0 - h[a:])
                g[a:] -= w_k * h[:np_]
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        d = g_dir / gn
        improved = False
        steps = [0.002, 0.005, 0.01, 0.02, 0.001, 0.008, 0.03, 0.05, 0.08, 0.12]
        for step in steps:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-13:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            for step in [0.005, 0.01, 0.02]:
                ht = project_feasible(h - step * d, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-13:
                    h = ht
                    best_c5 = cn
                    improved = True
                    break
            if not improved:
                break
    return h, best_c5


def _slsqp_obj(x, n, beta):
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    return (m + np.log(np.sum(ez)) / beta) * dx


def _slsqp_grad(x, n, beta):
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    return g


def _eq_fun(x, n):
    return np.sum(x) - n / 2.0


def _eq_jac(x, n):
    return np.ones(n)


def slsqp_smooth_max(h, n, beta, maxiter=300):
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq',
             'fun': lambda x: _eq_fun(x, n),
             'jac': lambda x: _eq_jac(x, n)}]
    try:
        res = minimize(lambda x: _slsqp_obj(x, n, beta), h0,
                       jac=lambda x: _slsqp_grad(x, n, beta),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.fun < _slsqp_obj(h0, n, beta) - 1e-14:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def theta_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)


def binary_sa(h, n, time_limit, rng):
    h = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        sort_idx = np.argsort(-h)
        h = np.zeros(n)
        h[sort_idx[:n_half]] = 1.0
    start = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    current_c5 = best_c5
    T = 0.5
    cool_rate = 0.9997
    no_improve = 0
    max_no_improve = 2000
    while time.time() - start < time_limit and remaining_time() > 10:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            current_c5 = new_c5
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
        T *= cool_rate
        if no_improve > max_no_improve:
            break
    return best_h, best_c5
```
