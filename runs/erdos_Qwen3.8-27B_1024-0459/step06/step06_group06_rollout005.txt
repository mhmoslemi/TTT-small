import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = free.sum()
        if cnt > 0:
            h[free] += diff / cnt
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_corr_k(h, k_idx, n):
    lag = k_idx - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        m = n - lag
        g[lag:n] += (1.0 - h[:m])
        g[:m] -= h[lag:]
    else:
        a = -lag
        m = n - a
        g[:m] += (1.0 - h[a:])
        g[a:] -= h[:m]
    return g


def pnorm_objective(h, n, p, n_active=60):
    corr = compute_corr(h)
    dx = 2.0 / n
    M = np.max(corr)
    if M < 1e-20:
        return M * dx, corr
    r = corr / M
    r_clip = np.clip(r, 0.0, 1.0)
    rp = np.power(r_clip, p)
    sum_rp = np.sum(rp)
    if sum_rp < 1e-300:
        obj = M * dx
    else:
        obj = M * np.power(sum_rp, 1.0 / p) * dx
    return obj, corr


def pnorm_objective_grad(h, n, p, n_active=60):
    corr = compute_corr(h)
    dx = 2.0 / n
    M = np.max(corr)
    if M < 1e-20:
        return M * dx, np.zeros(n)
    r = np.clip(corr / M, 1e-16, 1.0)
    rp = np.power(r, p)
    sum_rp = np.sum(rp)
    if sum_rp < 1e-300:
        return M * dx, np.zeros(n)
    obj = M * np.power(sum_rp, 1.0 / p) * dx
    w_base = np.power(r, p - 1) / (p * np.power(sum_rp, 1.0 - 1.0 / p)) * dx
    g = np.zeros(n)
    order = np.argsort(-w_base)
    for k_idx in order[:n_active]:
        wk = w_base[k_idx]
        if wk < 1e-18:
            break
        g += wk * grad_corr_k(h, k_idx, n)
    return obj, g


def lbfgs_pnorm(h, n, p, maxiter, ftol, penalty, remaining):
    def obj_pen(x):
        base, _ = pnorm_objective(x, n, p)
        s = np.sum(x) - n / 2.0
        return base + 0.5 * penalty * s * s

    def grad_pen(x):
        base, g = pnorm_objective_grad(x, n, p)
        s = np.sum(x) - n / 2.0
        g = g + penalty * s
        return g

    x0 = h.copy()
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj_pen, x0, jac=grad_pen, method='L-BFGS-B',
                   bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': ftol,
                            'gtol': 1e-12, 'maxfun': maxiter * 3,
                            'iprint': 0})
    if time.time() - _T0 > _BUDGET - 10:
        return h
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, n)
    return h


def slsqp_pnorm(h, n, p, maxiter, penalty):
    def sc_fun(x):
        return np.sum(x) - n / 2.0

    def sc_jac(x):
        return np.ones(n)

    def obj(x):
        return pnorm_objective(x, n, p)[0]

    def grad(x):
        return pnorm_objective_grad(x, n, p)[1]

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sc_fun, 'jac': sc_jac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def cutting_plane_step(h, n, K=40, delta=0.03):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    idx = np.argsort(-corr)[:K]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub_list = []
    b_ub_list = []
    for k in idx:
        g = grad_corr_k(h, k, n)
        fk = corr[k]
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_ub_list.append(row)
        b_ub_list.append(-(fk + 1e-15))
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    I = np.eye(n)
    A_trust = np.vstack([np.hstack([I, np.zeros((n, 1))]),
                         np.hstack([-I, np.zeros((n, 1))])])
    b_trust = np.concatenate([h + delta, -h + delta])
    A_ub = np.vstack([A_ub, A_trust])
    b_ub = np.concatenate([b_ub, b_trust])
    bounds = [(0, 1)] * n + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return h.copy(), c_max

        alloc = min(120.0, remaining_time() * 0.5)
        h_opt, c5_opt = optimize_at_resolution(h_new, n_target, alloc, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_target

    # Phase 3: Interpolate to n=4096 and optimize
    n_target = 4096
    h_new = interp_to(best_h, best_n, n_target)
    h_new = project_feasible(h_new, n_target)

    if remaining_time() > 120:
        alloc = min(200.0, remaining_time() * 0.5)
        h_opt, c5_opt = optimize_at_resolution(h_new, n_target, alloc, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_target

    # Phase 4: Try n=8192 if time permits and solution improved
    if remaining_time() > 150:
        n_target = 8192
        h_new = interp_to(best_h, best_n, n_target)
        h_new = project_feasible(h_new, n_target)
        alloc = min(180.0, remaining_time() * 0.5)
        h_opt, c5_opt = optimize_at_resolution(h_new, n_target, alloc, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_target

    # Phase 5: Coordinate descent on the best solution
    if remaining_time() > 30:
        alloc = min(60.0, remaining_time() * 0.3)
        h_cd, c5_cd = coordinate_descent(best_h, best_n, alloc)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 6: Final L-BFGS-B with high p
    if remaining_time() > 20:
        for p in [500, 1000]:
            if remaining_time() < 15:
                break
            maxiter = int(min(500, remaining_time() * 3))
            h_new = lbfgs_pnorm(best_h, best_n, p, maxiter, 1e-18, 1000.0, remaining_time())
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 7: Final equalize
    if remaining_time() > 15:
        h_eq, c5_eq = equalize_top_shifts(best_h, best_n, K=20, max_iters=200)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Phase 8: SLSQP final polish
    if remaining_time() > 15:
        h_new = slsqp_pnorm(best_h, best_n, 500, maxiter=200, penalty=1000.0)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 9: Try binary projection
    h_bin = binary_project(best_h, best_n)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # Phase 10: PERTURBATION RESTART - try a few random perturbations
    if remaining_time() > 40:
        for attempt in range(3):
            if remaining_time() < 20:
                break
            h_pert = best_h.copy()
            mask = (h_pert > 0.05) & (h_pert < 0.95)
            cnt = mask.sum()
            if cnt > 0:
                h_pert[mask] += rng.randn(cnt) * 0.1
            h_pert = project_feasible(h_pert, best_n)
            h_opt, c5_opt = optimize_at_resolution(h_pert, best_n, min(20.0, remaining_time() * 0.3), rng)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 11: Final subgradient with tiny steps
    if remaining_time() > 15:
        h_fine, c5_fine = subgradient_refine(best_h, best_n, min(12.0, remaining_time() * 0.5), rng)
        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_fine.copy()

    # Final feasibility and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
