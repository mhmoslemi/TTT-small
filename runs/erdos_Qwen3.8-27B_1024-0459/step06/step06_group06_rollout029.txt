def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    # Projects h to {0<=h<=1, sum(h)=n/2}
    h = np.array(h, dtype=float).copy()
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        diff = n/2.0 - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.count_nonzero(mask)
        else:
            h[:] += diff / n
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -100, 100)
    exps = np.exp(shifted)
    logsumexp = m + np.log(np.sum(exps)) / beta
    obj = logsumexp * dx
    weights = exps / np.sum(exps)
    # Gradient
    grad = np.zeros_like(h)

    h_prev = h.copy()
    y = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t = 1.0
    start = time.time()
    
    for it in range(10000):
        if time.time() - start > time_limit:
            break
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        
        obj, grad_y = smooth_max_obj_grad(y, beta)
        # Projected gradient step
        lr = 1.0 / (it + 1) ** 0.7  # decreasing step size
        h_new = project_feasible(y - lr * grad_y, n)
        
        # Nesterov momentum
        t_new = (1 + np.sqrt(1 + 4*t*t)) / 2
        theta = (t - 1) / t_new
        y = project_feasible(h_new + theta * (h_new - h_prev), n)
        
        c5 = compute_c5(y)
        if c5 < best_c5:
            best_c5 = c5
            best_h = y.copy()
        
        h_prev = h_new
        t = t_new
    
    return best_h, best_c5

    n = len(h)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
    return np.clip(h, 0.0, 1.0)


def smooth_max_and_grad(h, beta, penalty=0.0):
    """Compute smooth max objective and gradient.
    Returns (obj, grad) where obj includes penalty for sum constraint."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exps = np.exp(shifted)
    logsum = m + np.log(np.sum(exps)) / beta
    obj = logsum * dx
    
    # Add penalty for sum constraint
    s = np.sum(h) - n / 2.0
    obj += 0.5 * penalty * s * s
    
    weights = exps / np.sum(exps)
    
    # Gradient of smooth max
    grad = np.zeros(n)
    for k_raw in range(len(corr)):
        w = weights[k_raw] * dx
        if w < 1e-18:
            continue
        k = k_raw - (n - 1)
        if k >= 0:
            a = k
            npairs = n - a
            if npairs > 0:
                grad[:npairs] += w * (1.0 - h[a:])
                grad[a:] -= w * h[:npairs]
        else:
            a = -k
            npairs = n - a
            if npairs > 0:
                grad[a:] += w * (1.0 - h[:npairs])
                grad[:npairs] -= w * h[a:]
    
    # Add penalty gradient
    if penalty > 0:
        grad += penalty * s
    
    return obj, grad


def lbfgsb_optimize(h, beta, penalty, maxiter=3000):
    n = len(h)
    
    def obj_fun(x):
        return smooth_max_and_grad(x, beta, penalty)[0]
    
    def grad_fun(x):
        return smooth_max_and_grad(x, beta, penalty)[1]
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fun, h, jac=grad_fun, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-14, 'maxfun': maxiter * 5})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()
