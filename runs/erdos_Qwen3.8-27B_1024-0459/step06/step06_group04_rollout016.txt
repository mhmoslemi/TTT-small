def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0

            i1 = np.random.randint(0, n)
            i0 = np.random.randint(0, n)
            if h_bin[i1] == h_bin[i0]:
                continue
            h_try = h_bin.copy()
            h_try[i1] = 0.0
            h_try[i0] = 1.0
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try)
            if c_try < best_c5 - 1e-12:
                h_bin = h_try
                best_c5 = c_try
                best_h = h_try.copy()
                improved = True
                break
        if not improved and rnd > 100:
            break
    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def optimize_phase(h, n, time_limit):
    if time_limit <= 0:
        return h, compute_c5(h)
    best_c5 = compute_c5(h)
    best_h = h.copy()

    # SLSQP with increasing beta
    alphas = [500, 2000, 10000, 50000, 200000]
    for beta in alphas:
        if time_limit <= 5:
            break
        t_alloc = min(80.0, time_limit * 0.3)
        maxit = max(50, min(300, int(50 + beta / 500)))
        h_new = slsqp_solve(best_h, beta, maxiter=maxit)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
        time_limit -= t_alloc
        if time_limit <= 5:
            break

    # Projected gradient
    if time_limit > 15:
        t_pg = min(60.0, time_limit * 0.4)
        h_new, c_new = projected_gradient(best_h, n, t_pg)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
        time_limit -= t_pg

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    np.random.seed(seed)

    def rem():
        return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)

    # Load parent
    try:
        best_h = np.array(initial_h_values, dtype=float).copy()
        best_n = len(best_h)
    except Exception:
        best_n = 512
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Optimize at parent resolution
    if rem() > 30:
        t1 = min(250.0, rem() * 0.18)
        h1, c1 = optimize_phase(best_h, best_n, t1)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1.copy()

    # Phase 2: Binary local search at parent resolution
    if rem() > 30:
        t2 = min(80.0, rem() * 0.06)
        h_bin, c_bin = binary_local_search(best_h, best_n, t2)
        if c_bin < best_c5:
            best_c5 = c_bin
            best_h = h_bin.copy()
        # Use binary as additional starting point for continuous
        if c_bin < best_c5 + 0.01 and rem() > 20:
            h_cont = project_feasible(best_h.copy(), best_n)
            t_cont = min(40.0, rem() * 0.03)
            h_c, c_c = optimize_phase(h_cont, best_n, t_cont)
            if c_c < best_c5:
                best_c5 = c_c
                best_h = h_c.copy()

    # Phase 3: Try n=1024
    n_alt = 1024
    if rem() > 60 and n_alt != best_n:
        h_interp = interp_to(best_h, best_n, n_alt)
        h_interp = project_feasible(h_interp, n_alt)
        c_interp = compute_c5(h_interp)
        t3 = min(350.0, rem() * 0.28)
        h_new, c_new = optimize_phase(h_interp, n_alt, t3)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
            best_n = n_alt

    # Phase 4: Try n=2048
    n_alt2 = 2048
    if rem() > 80 and n_alt2 != best_n:
        h_interp2 = interp_to(best_h, best_n, n_alt2)
        h_interp2 = project_feasible(h_interp2, n_alt2)
        t4 = min(300.0, rem() * 0.25)
        h_new2, c_new2 = optimize_phase(h_interp2, n_alt2, t4)
        if c_new2 < best_c5:
            best_c5 = c_new2
            best_h = h_new2.copy()
            best_n = n_alt2

    # Phase 5: Coordinate descent polish at best resolution
    if rem() > 20:
        t5 = min(120.0, rem() * 0.12)
        h_cd, c_cd = coordinate_descent(best_h, best_n, t5, max_passes=10)
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd.copy()

    # Phase 6: Another projected gradient pass with different parameters
    if rem() > 20:
        t6 = min(80.0, rem() * 0.08)
        h_pg, c_pg = projected_gradient(best_h, best_n, t6, K=20, beta_w=150.0)
        if c_pg < best_c5:
            best_c5 = c_pg
            best_h = h_pg.copy()

    # Phase 7: Final SLSQP polish with very high beta
    if rem() > 15:
        for beta_final in [500000, 1000000]:
            if rem() < 10:
                break
            t_sf = min(60.0, rem() * 0.06)
            # Temporarily reduce budget for this call
            old_budget = _BUDGET
            h_new_f = slsqp_solve(best_h, beta_final, maxiter=100)
            c_new_f = compute_c5(h_new_f)
            if c_new_f < best_c5:
                best_c5 = c_new_f
                best_h = h_new_f.copy()

    # Phase 8: Final coordinate descent
    if rem() > 15:
        t8 = min(60.0, rem() * 0.05)
        h_final_cd, c_final_cd = coordinate_descent(best_h, best_n, t8, max_passes=5)
        if c_final_cd < best_c5:
            best_c5 = c_final_cd
            best_h = h_final_cd.copy()

    # Final feasibility and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
