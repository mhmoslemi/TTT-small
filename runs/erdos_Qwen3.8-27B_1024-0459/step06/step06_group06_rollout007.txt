def fast_grad(h, w_full):
    """
    h: array of length n
    w_full: array of length 2n-1, w_full[j] corresponds to lag j-(n-1)
    Returns: gradient of sum_j w_full[j] * corr_j w.r.t. h, length n
    """
    n = len(h)
    h_rev = h[::-1]
    # Compute convolutions via FFT
    N_fft = 1
    while N_fft < 2*n:
        N_fft *= 2
    H = np.fft.rfft(h, N_fft)
    H_rev = np.fft.rfft(h_rev, N_fft)
    W = np.fft.rfft(w_full, N_fft)
    wh = np.fft.irfft(W * H, N_fft)[:n]  # (w*h)[0..n-1]
    wh_rev = np.fft.irfft(W * H_rev, N_fft)[:n]  # (w*h_rev)[0..n-1]
    # grad[i] = sum(w) - wh[i] - wh_rev[n-1-i]
    grad = np.sum(w_full) - wh - wh_rev[::-1]
    return grad

    # Phase 2: Multi-beta refinement (budget: ~200s)
    if time_left() > 50:
        h2, c5_2 = multi_beta_descent(h, n, [100, 300, 1000, 3000, 8000], 80, 0.08)
        if c5_2 < best_c5:
            best_h = h2.copy()
            best_c5 = c5_2
            best_n = n
            h = h2.copy()

    # Phase 3: SLSQP polish at increasing beta (budget: ~100s)
    for b in [2000, 5000, 10000]:
        if time_left() < 30:
            break
        h3 = slsqp_polish(h, n, b, maxiter=300)
        c5_3 = compute_c5(h3)
        if c5_3 < best_c5:
            best_h = h3.copy()
            best_c5 = c5_3
            best_n = n
            h = h3.copy()

    # Phase 4: L-BFGS-B (budget: ~100s)
    for b, pen in [(3000, 500), (8000, 2000), (15000, 5000)]:
        if time_left() < 30:
            break
        h4 = lbfgsb_polish(h, n, b, pen, maxiter=1500)
        c5_4 = compute_c5(h4)
        if c5_4 < best_c5:
            best_h = h4.copy()
            best_c5 = c5_4
            best_n = n
            h = h4.copy()

    # Phase 5: Random perturbations + re-optimize (budget: ~100s)
    if time_left() > 60:
        for mag in [0.05, 0.03, 0.02, 0.01]:
            if time_left() < 20:
                break
            h_pert = random_perturb(h, n, rng, mag)
            h_pert, c5_pert = gradient_descent_loop(h_pert, n, 500.0, 5000.0, 100, 0.05)
            if c5_pert < best_c5:
                best_h = h_pert.copy()
                best_c5 = c5_pert
                best_n = n
                h = h_pert.copy()

    # Phase 6: Coordinate descent fine-tuning (budget: ~80s)
    if time_left() > 40:
        h_cd, c5_cd = coordinate_descent_fine(h, n, min(60.0, time_left() * 0.5))
        if c5_cd < best_c5:
            best_h = h_cd.copy()
            best_c5 = c5_cd
            best_n = n
            h = h_cd.copy()

    # Phase 7: Try n=4096 if time permits (budget: ~100s)
    if time_left() > 100:
        n2 = 4096
        h2 = interp_resample(best_h, best_n, n2)
        h2 = project_feasible(h2, n2)
        c5_h2 = compute_c5(h2)
        h2, c5_h2 = gradient_descent_loop(h2, n2, 200.0, 3000.0, 150, 0.08)
        h2 = slsqp_polish(h2, n2, 5000, maxiter=200)
        c5_h2 = compute_c5(h2)
        if c5_h2 < best_c5:
            best_h = h2.copy()
            best_c5 = c5_h2
            best_n = n2

    # Phase 8: Final fine gradient descent (budget: remaining)
    if time_left() > 20:
        h, c5 = gradient_descent_loop(h, best_n, 5000.0, 20000.0,
                                      max(50, int(time_left() * 2)), 0.03)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            best_n = best_n

    # Phase 9: Final SLSQP
    if time_left() > 15:
        h_final = slsqp_polish(best_h, best_n, 20000, maxiter=300)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_h = h_final.copy()
            best_c5 = c5_final

    # Final feasibility and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
