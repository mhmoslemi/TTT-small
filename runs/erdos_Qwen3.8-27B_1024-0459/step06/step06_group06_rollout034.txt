def topk_smooth_obj_grad(h, n, K, beta, penalty):

    dx = 2.0 / n
    # Get top K indices
    idx = np.argpartition(-corr, K)[:K]
    c_top = corr[idx]
    m = np.max(c_top)
    sh = np.clip(beta * (c_top - m), -50.0, 50.0)
    ex = np.exp(sh)
    sm = np.sum(ex)
    w = ex / sm
    obj = (m + np.log(sm) / beta) * dx

        if wk < 1e-16:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            g[offset:n] += wk * (1.0 - h[:npairs])
            g[:npairs] -= wk * h[offset:n]
        else:
            a = -offset
            npairs = n - a
            g[:npairs] += wk * (1.0 - h[a:n])
            g[a:n] -= wk * h[:npairs]
    
    s = np.sum(h) - n / 2.0
    obj += 0.5 * penalty * s * s
    g += penalty * s
    
    return obj, g

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.where((h > 1e-14) & (h < 1.0 - 1e-14))[0]
        if len(idx) > 0:
            h[idx[0]] += diff * 0.5

    def obj_fun(x):

        return smooth_max_obj_grad(x, n, beta, penalty)[1]

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fun, h, jac=grad_fun, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-22,
                                'gtol': 1e-14, 'maxfun': 200000,
                                'maxls': 50})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h
