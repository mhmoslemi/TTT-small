For the autocorrelation formulation, we want a set of measure 1/2 whose autocorrelation is as flat as possible (minimizing the maximum).

def compute_c5(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def _sm_obj_func(x):
    n = _G_N
    dx = 2.0 / n
    x = np.clip(x, 0.0, 1.0)
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    z = np.clip(_G_BETA * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    sm = (m + np.log(np.sum(ez)) / _G_BETA) * dx
    pen = _G_MU * (np.sum(x) - n / 2.0) ** 2
    return sm + pen


def _sm_grad_func(x):
    n = _G_N
    dx = 2.0 / n
    x = np.clip(x, 0.0, 1.0)
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    z = np.clip(_G_BETA * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        off = k - (n - 1)

            g[off:] += wk * (1.0 - x[:np_])

        else:
            a = -off
            np_ = n - a

            g[a:] -= wk * x[:np_]
    g += 2.0 * _G_MU * (np.sum(x) - n / 2.0)
    return g

def interp_to(src, src_n, dst_n):

                    improved_sweep = True
            h[i] = best_val
            h = project_feasible(h, n)
        if not improved_sweep:
            break
    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _G_N, _G_BETA, _G_MU
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = np.zeros(640)
        best_h[:320] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    h = best_h.copy()
    n = best_n

    # Phase 1: L-BFGS-B at parent resolution with increasing beta
    for beta in [200, 1000, 5000, 20000, 100000]:
        if remaining_time() < 30:
            break
        h_new = lbfgsb_opt(h, n, beta, mu=10.0, maxiter=3000, time_limit=60)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: SLSQP polish at parent resolution
    for beta in [50000, 200000, 1000000]:
        if remaining_time() < 20:
            break
        h_new = slsqp_opt(h, n, beta, maxiter=2000, time_limit=80)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 3: Topo gradient descent
    if remaining_time() > 30:
        h_eq, c5_eq = topo_gradient(h, n, top_k=5, max_iters=2000, time_limit=60)
        if c5_eq < best_c5 - 1e-15:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()
        h_eq, c5_eq = topo_gradient(h, n, top_k=10, max_iters=1500, time_limit=45)
        if c5_eq < best_c5 - 1e-15:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 4: Upscale to higher resolution
    n_hi = 1280
    if n_hi != n and remaining_time() > 100:
        h_interp = interp_to(best_h, n, n_hi)
        h_interp = project_feasible(h_interp, n_hi)
        c5_int = compute_c5(h_interp)
        h_hi = h_interp.copy()
        for beta in [1000, 10000, 100000]:
            if remaining_time() < 30:
                break
            h_new = lbfgsb_opt(h_hi, n_hi, beta, mu=10.0, maxiter=3000, time_limit=80)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_n = n_hi
                best_h = h_new.copy()
                h = h_new.copy()
                h_hi = h_new.copy()
            else:
                h_hi = n_hi and h_new
        if remaining_time() > 30 and best_n == n_hi:
            h_new = slsqp_opt(h_hi, n_hi, 500000, maxiter=2000, time_limit=80)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 5: Try even higher resolution
    n_hi2 = 2048
    if n_hi2 != best_n and remaining_time() > 120:
        h_interp2 = interp_to(best_h, best_n, n_hi2)
        h_interp2 = project_feasible(h_interp2, n_hi2)
        h_hi2 = h_interp2.copy()
        for beta in [5000, 50000]:
            if remaining_time() < 40:
                break
            h_new = lbfgsb_opt(h_hi2, n_hi2, beta, mu=10.0, maxiter=2000, time_limit=100)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_n = n_hi2
                best_h = h_new.copy()
                h = h_new.copy()
                h_hi2 = h_new.copy()
            else:
                h_hi2 = h_new
        if best_n == n_hi2 and remaining_time() > 40:
            h_new = slsqp_opt(h_hi2, n_hi2, 200000, maxiter=1500, time_limit=80)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 6: Binary search at best resolution
    if remaining_time() > 80:
        n_cur = best_n
        h_bin = (best_h > 0.5).astype(float)
        sa_time = min(180, int(remaining_time() * 0.3))
        h_b, c5_b = binary_swap_sa(h_bin, n_cur, time_limit=sa_time, rng=rng, restarts=3)
        if c5_b < best_c5 - 1e-15:
            best_c5 = c5_b
            best_h = h_b.copy().astype(float)
            h = h_b.copy().astype(float)

        if remaining_time() > 40:
            h_g, c5_g = binary_swap_greedy(h_b, n_cur, time_limit=40, rng=rng)
            if c5_g < best_c5 - 1e-15:
                best_c5 = c5_g
                best_h = h_g.copy().astype(float)
                h = h_g.copy().astype(float)

    # Phase 7: Continuous refinement after binary
    if remaining_time() > 40:
        n_cur = best_n
        h_c = best_h.copy()
        for beta in [50000, 500000]:
            if remaining_time() < 20:
                break
            h_new = lbfgsb_opt(h_c, n_cur, beta, mu=10.0, maxiter=2000, time_limit=60)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
                h_c = h_new.copy()
        if remaining_time() > 30:
            h_eq, c5_eq = topo_gradient(h, n_cur, top_k=8, max_iters=1500, time_limit=40)
            if c5_eq < best_c5 - 1e-15:
                best_c5 = c5_eq
                best_h = h_eq.copy()
                h = h_eq.copy()

    # Phase 8: Coordinate refinement
    if remaining_time() > 20:
        n_cur = best_n
        h_cr, c5_cr = coordinate_refine(best_h, n_cur, max_sweeps=10, time_limit=min(50, int(remaining_time() * 0.4)))
        if c5_cr < best_c5 - 1e-15:
            best_c5 = c5_cr
            best_h = h_cr.copy()
            h = h_cr.copy()

    # Phase 9: Final SLSQP polish
    if remaining_time() > 15:
        n_cur = best_n
        h_final = slsqp_opt(best_h, n_cur, 2000000, maxiter=2000, time_limit=min(60, int(remaining_time() * 0.7)))
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-15:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Phase 10: Topo gradient final
    if remaining_time() > 10:
        n_cur = best_n
        h_eq, c5_eq = topo_gradient(best_h, n_cur, top_k=6, max_iters=1000, time_limit=min(30, int(remaining_time() * 0.6)))
        if c5_eq < best_c5 - 1e-15:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Final feasibility check
    n_final = best_n
    best_h = project_feasible(best_h, n_final)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, n_final)
