def compute_swap_delta(h, a, b, n):
    """Compute delta in full correlation when swapping h[a]:1->0, h[b]:0->1.
    Returns delta array of length 2n-1."""
    delta = np.zeros(2*n-1)
    idx = n - 1  # offset: array index = p + idx
    
    # Term 1: p in [-a, n-1-a]
    p_lo1 = -a
    p_hi1 = n - 1 - a
    if p_lo1 <= p_hi1:
        p_range = np.arange(p_lo1, p_hi1 + 1)
        positions = a + p_range  # positions of second factor
        vals = 1.0 - h[positions]
        vals[positions == b] = 1.0  # special case
        delta[p_range + idx] -= vals
    
    # Term 2: p in [-b, n-1-b]
    p_lo2 = -b
    p_hi2 = n - 1 - b
    if p_lo2 <= p_hi2:
        p_range = np.arange(p_lo2, p_hi2 + 1)
        positions = b + p_range
        vals = 1.0 - h[positions]
        vals[positions == a] = 1.0  # special case
        delta[p_range + idx] += vals
    
    # Term 3: p in [a-n+1, a]
    p_lo3 = a - n + 1
    p_hi3 = a
    if p_lo3 <= p_hi3:
        p_range = np.arange(p_lo3, p_hi3 + 1)
        positions = a - p_range
        vals = h[positions].copy()
        vals[positions == b] = 1.0  # special case
        delta[p_range + idx] += vals
