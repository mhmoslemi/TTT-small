from scipy.optimize import minimize, LinearConstraint, Bounds
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_c5_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)), corr


def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        if interior.any():
            h[interior] += diff / interior.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def to_binary(h, n):
    """Convert h to binary with exactly n/2 ones using threshold + adjustment"""
    h_bin = (h >= 0.5).astype(float)
    ones = int(h_bin.sum())
    target = n // 2
    if ones > target:
        # Turn off the smallest values
        idx = np.where(h_bin == 1)[0]
        order = idx[np.argsort(-h[idx])]
        for i in range(ones - target):
            h_bin[order[i]] = 0
    elif ones < target:
        idx = np.where(h_bin == 0)[0]
        order = idx[np.argsort(-h[idx])]
        for i in range(target - ones):
            h_bin[order[i]] = 1
    return h_bin


def smooth_max_func(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    s = np.clip(beta * (corr - m), -500.0, 500.0)
    e = np.exp(s)
    return (m + np.log(e.sum()) / beta) * dx


def smooth_max_jac(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    s = np.clip(beta * (corr - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx] * dx
        if wk < 1e-15:
            continue
        off = k_idx - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - x[ao:])
            g[ao:] -= wk * x[:np_]
    return g


def slsqp_smooth(h, beta, maxiter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_max_func, h, args=(beta,),
                       jac=smooth_max_jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _eq_fun,
                                     'jac': _eq_jac, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def _eq_fun(x, n):
    return x.sum() - n / 2.0


def _eq_jac(x, n):
    return np.ones(n)


def topk_subgradient(h, n, time_limit, k_top=5, beta=100.0, use_momentum=False):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    momentum = np.zeros(n) if use_momentum else None
    lr = 0.05
    stall = 0
    for it in range(100000):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 15:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = corr.max()
        s = np.clip(beta * (corr - m), -500.0, 500.0)
        e = np.exp(s)
        w = e / e.sum()
        g = np.zeros(n)
        for k_idx in range(2 * n - 1):
            wk = w[k_idx] * (2.0 / n)
            if wk < 1e-15:
                continue
            off = k_idx - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:] += wk * (1.0 - h[:np_])
                g[:np_] -= wk * h[off:]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - h[ao:])
                g[ao:] -= wk * h[:np_]
        if momentum is not None:
            momentum = 0.85 * momentum + 0.15 * g
            g = momentum
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        accepted = False
        for scale in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125]:
            step = lr * scale / (gn + 1e-15)
            h_try = project_feasible(h - step * g, n)
            c_try = compute_c5(h_try)
            if c_try < best_c5 - 1e-16:
                h = h_try
                best_c5 = c_try
                best_h = h_try.copy()
                accepted = True
                break
        if accepted:
            lr = min(lr * 1.1, 0.5)
            stall = 0
        else:
            lr *= 0.5
            stall += 1
            if stall > 50:
                break
    return best_h, best_c5


def binary_sa(h_bin, n, time_limit, rng, T_init=0.05, cool_rate=0.9995):
    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin)
    curr_c5 = best_c5
    start = time.time()
    T = T_init
    idx_ones = np.where(h_bin == 1)[0]
    idx_zeros = np.where(h_bin == 0)[0]
    while time.time() - start < time_limit and _BUDGET - (time.time() - _T0) > 15:
        i1 = idx_ones[rng.randint(len(idx_ones))]
        i0 = idx_zeros[rng.randint(len(idx_zeros))]
        if i1 == i0:
            continue
        h_try = h_bin.copy()
        h_try[i1] = 0.0
        h_try[i0] = 1.0
        c_try = compute_c5(h_try)
        delta = c_try - curr_c5
        if delta < 0 or (T > 1e-10 and rng.random() < np.exp(-delta / T)):
            h_bin = h_try
            curr_c5 = c_try
            # Update index arrays
            idx_ones = np.where(h_bin == 1)[0]
            idx_zeros = np.where(h_bin == 0)[0]
            if c_try < best_c5:
                best_c5 = c_try
                best_h = h_try.copy()
        T *= cool_rate
    return best_h, best_c5


def binary_targeted_search(h_bin, n, time_limit, rng):
    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin)
    start = time.time()
    patience = 0
    MAX_PATIENCE = 3000
    while time.time() - start < time_limit and _BUDGET - (time.time() - _T0) > 15:
        corr = np.correlate(h_bin, 1.0 - h_bin, mode='full')
        corr_max = corr.max()
        # Find top lags
        top_lags = np.argsort(-corr)[:8]
        improved = False
        for lag_idx in top_lags:
            off = lag_idx - (n - 1)
            # Find contributing positions
            if off >= 0:
                bad = []
                for i in range(n - off):
                    if h_bin[i] == 1 and h_bin[i + off] == 0:
                        bad.append(i)
            else:
                ao = -off
                bad = []
                for i in range(ao, n):
                    if h_bin[i] == 1 and h_bin[i - ao] == 0:
                        bad.append(i)
            if not bad:
                continue
            # Try flipping from bad positions
            for i in bad[:15]:
                # Flip h[i] from 1 to 0, and flip a random 0 to 1
                zeros = np.where(h_bin == 0)[0]
                for _ in range(5):
                    j = zeros[rng.randint(len(zeros))]
                    if j == i:
                        continue
                    h_try = h_bin.copy()
                    h_try[i] = 0.0
                    h_try[j] = 1.0
                    c_try = compute_c5(h_try)
                    if c_try < best_c5 - 1e-15:
                        h_bin = h_try
                        best_c5 = c_try
                        best_h = h_try.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            patience += 1
            if patience > MAX_PATIENCE:
                # Perturb
                for _ in range(5):
                    i1 = rng.randint(n)
                    if h_bin[i1] == 1:
                        zeros = np.where(h_bin == 0)[0]
                        i0 = zeros[rng.randint(len(zeros))]
                        h_bin[i1] = 0.0
                        h_bin[i0] = 1.0
                        break
                patience = 0
    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)
