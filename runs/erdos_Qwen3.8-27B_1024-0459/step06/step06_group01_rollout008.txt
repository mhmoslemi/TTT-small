def _sm_objective(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = beta * (corr - m)
    z = np.clip(z, -50.0, 50.0)
    ez = np.exp(z)
    return (m + np.log(np.sum(ez)) / beta) * dx


def _sm_gradient(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = beta * (corr - m)
    z = np.clip(z, -50.0, 50.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    return g


def _eq_fun(x):
    return np.sum(x) - _CUR_N / 2.0


def _eq_jac(x):
    return np.ones(_CUR_N)


def slsqp_opt(h, n, beta, maxiter=500):
    global _CUR_N, _SM_BETA
    _CUR_N = n
    _SM_BETA = beta
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_sm_objective, h0, jac=_sm_gradient, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.fun < _sm_objective(h0) - 1e-14:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def wide_topk_equalize(h, n, top_k=20, max_iters=1000, adaptive=True):
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining_time() < 3:
            break
        corr = compute_corr(h)
        sorted_idx = np.argsort(-corr)
        if adaptive:
            k_eff = min(top_k, 3 + it // 50)
        else:
            k_eff = top_k
        idx_top = sorted_idx[:k_eff]
        # Check convergence: if top and 5th in set are close, and spread is small
        spread = corr[idx_top[0]] - corr[idx_top[-1]]
        if spread < 1e-10 and k_eff >= top_k:
            break
        g_dir = np.zeros(n)
        # Weighted: top gets +1, others get -1/(k_eff-1) each
        for pos in range(k_eff):
            k = idx_top[pos]
            w_k = 1.0 if pos == 0 else (-1.0 / max(k_eff - 1, 1))
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_dir[off:] += w_k * (1.0 - h[:np_])
                g_dir[:np_] -= w_k * h[off:]
            else:
                a = -off
                np_ = n - a
                g_dir[:np_] += w_k * (1.0 - h[a:])
                g_dir[a:] -= w_k * h[:np_]
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        d = g_dir / gn
        improved = False
        steps = [0.002, 0.005, 0.01, 0.001, 0.02, 0.003, 0.04, 0.008]
        for step in steps:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-13:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            for step in [0.003, 0.008, 0.015]:
                ht = project_feasible(h - step * d, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-13:
                    h = ht
                    best_c5 = cn
                    improved = True
                    break
            if not improved:
                # Try smaller steps
                for step in [0.0005, 0.001, 0.002]:
                    ht = project_feasible(h + step * d, n)
                    cn = compute_c5(ht)
                    if cn < best_c5 - 1e-14:
                        h = ht
                        best_c5 = cn
                        improved = True
                        break
                if not improved:
                    break
    return h, best_c5


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def pair_mix_descent(h, n, max_iters=20000, rng=None):
    """Randomized pair-mixing local search"""
    if rng is None:
        rng = np.random.RandomState(0)
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    no_improve = 0
    max_no_improve = 2000
    for it in range(max_iters):
        if remaining_time() < 5:
            break
        if no_improve > max_no_improve:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        old_i, old_j = h[i], h[j]
        # Try mixing: new_i = (1-t)*old_i + t*old_j, new_j = (1-t)*old_j + t*old_i
        t = rng.uniform(0.0, 1.0)
        h[i] = (1.0 - t) * old_i + t * old_j
        h[j] = (1.0 - t) * old_j + t * old_i
        h = project_feasible(h, n)
        c = compute_c5(h)
        if c < best_c5 - 1e-14:
            best_c5 = c
            no_improve = 0
        else:
            h[i] = old_i
            h[j] = old_j
            no_improve += 1
    return h, best_c5


def targeted_gradient_descent(h, n, max_iters=2000, top_k=10):
    """Gradient descent targeting the top-k correlations specifically"""
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining_time() < 5:
            break
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        # Build gradient to reduce top correlations
        g = np.zeros(n)
        for ki in idx_top:
            off = ki - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:] += (1.0 - h[:np_])
                g[:np_] -= h[off:]
            else:
                a = -off
                np_ = n - a
                g[:np_] += (1.0 - h[a:])
                g[a:] -= h[:np_]
        g *= dx / top_k
        # Project gradient to feasible direction (zero sum, within [0,1])
        g -= np.mean(g)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        d = -g / gn
        # Line search
        improved = False
        for step in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.2]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def multi_start_optimize(h_starts, n, rng, max_starts=5):
    """Try multiple perturbations and optimize each"""
    best_h = None
    best_c5 = float('inf')
    for s in range(max_starts):
        if remaining_time() < 40:
            break
        h0 = h_starts[s].copy()
        h0 = project_feasible(h0, n)
        c0 = compute_c5(h0)
        # SLSQP
        for beta in [5000, 100000]:
            if remaining_time() < 20:
                break
            h0 = slsqp_opt(h0, n, beta, maxiter=400)
        # Wide top-k
        h0, c0 = wide_topk_equalize(h0, n, top_k=20, max_iters=500)
        # Targeted gradient
        h0, c0 = targeted_gradient_descent(h0, n, max_iters=500, top_k=10)
        if c0 < best_c5:
            best_c5 = c0
            best_h = h0.copy()
    return best_h, best_c5


def theta_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-10:
        sort_idx = np.argsort(-h)
        if diff > 0:
            add = int(round(diff))
            for idx in sort_idx[:add]:
                if h[idx] < 1.0:
                    h[idx] = min(1.0, h[idx] + diff / max(add, 1))
        else:
            rem = int(round(-diff))
            for idx in sort_idx[-rem:]:
                if h[idx] > 0.0:
                    h[idx] = max(0.0, h[idx] + diff / max(rem, 1))
    return project_feasible(h, n)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _CUR_N, _SM_BETA
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    # Get initial values
    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = np.zeros(640)
        best_h[:320] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Optimize at parent resolution
    h = best_h.copy()

    # SLSQP with adaptive beta
    for beta in [500, 5000, 50000, 200000]:
        if remaining_time() < 40:
            break
        h_new = slsqp_opt(h, best_n, beta, maxiter=800)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Wide top-k equalization (much wider than before)
    if remaining_time() > 40:
        h_eq, c5_eq = wide_topk_equalize(h, best_n, top_k=50, max_iters=1500, adaptive=True)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Targeted gradient descent
    if remaining_time() > 30:
        h_tg, c5_tg = targeted_gradient_descent(h, best_n, max_iters=1000, top_k=15)
        if c5_tg < best_c5 - 1e-14:
            best_c5 = c5_tg
            best_h = h_tg.copy()
            h = h_tg.copy()

    # Pair mixing
    if remaining_time() > 30:
        h_pm, c5_pm = pair_mix_descent(h, best_n, max_iters=8000, rng=rng)
        if c5_pm < best_c5 - 1e-14:
            best_c5 = c5_pm
            best_h = h_pm.copy()
            h = h_pm.copy()

    # Phase 2: Upscale to n=1024
    n_high = 1024
    if remaining_time() > 150:
        h_high = interp_to(best_h, best_n, n_high)
        h_high = project_feasible(h_high, n_high)

        # Optimize at high resolution
        for beta in [2000, 20000, 200000]:
            if remaining_time() < 30:
                break
            h_high = slsqp_opt(h_high, n_high, beta, maxiter=500)

        h_high, c5_high = wide_topk_equalize(h_high, n_high, top_k=30, max_iters=800, adaptive=True)
        h_high, c5_high = targeted_gradient_descent(h_high, n_high, max_iters=800, top_k=15)

        if c5_high < best_c5 - 1e-14:
            best_c5 = c5_high
            best_h = h_high.copy()
            best_n = n_high
            h = h_high.copy()

    # Phase 3: Multi-start at best resolution
    if remaining_time() > 100:
        starts = [best_h.copy()]
        # Add perturbed versions
        for scale in [0.05, 0.1, 0.15]:
            pert = best_h + scale * rng.randn(best_n)
            starts.append(project_feasible(pert, best_n))
        # Thue-Morse
        starts.append(theta_morse(best_n))
        h_ms, c5_ms = multi_start_optimize(starts, best_n, rng, max_starts=5)
        if h_ms is not None and c5_ms < best_c5 - 1e-14:
            best_c5 = c5_ms
            best_h = h_ms.copy()
            h = h_ms.copy()

    # Phase 4: Fine polishing at current best resolution
    for beta in [500000, 2000000, 10000000]:
        if remaining_time() < 30:
            break
        h_new = slsqp_opt(h, best_n, beta, maxiter=1000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 5: Iterative wide top-k with different parameters
    for k_val in [10, 20, 50, 100]:
        if remaining_time() < 20:
            break
        h_eq, c5_eq = wide_topk_equalize(h, best_n, top_k=k_val, max_iters=600, adaptive=False)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 6: Try n=1536
    if remaining_time() > 100:
        n_alt = 1536
        h_alt = interp_to(best_h, best_n, n_alt)
        h_alt = project_feasible(h_alt, n_alt)
        for beta in [5000, 50000, 200000]:
            if remaining_time() < 25:
                break
            h_alt = slsqp_opt(h_alt, n_alt, beta, maxiter=400)
        h_alt, c5_alt = wide_topk_equalize(h_alt, n_alt, top_k=25, max_iters=500, adaptive=True)
        if c5_alt < best_c5 - 1e-14:
            best_c5 = c5_alt
            best_h = h_alt.copy()
            best_n = n_alt
            h = h_alt.copy()

    # Phase 7: More pair mixing for final refinement
    if remaining_time() > 30:
        h_pm2, c5_pm2 = pair_mix_descent(h, best_n, max_iters=15000, rng=rng)
        if c5_pm2 < best_c5 - 1e-14:
            best_c5 = c5_pm2
            best_h = h_pm2.copy()
            h = h_pm2.copy()

    # Phase 8: Final targeted gradient + SLSQP
    if remaining_time() > 30:
        h_tg2, c5_tg2 = targeted_gradient_descent(h, best_n, max_iters=1000, top_k=5)
        if c5_tg2 < best_c5 - 1e-14:
            best_c5 = c5_tg2
            best_h = h_tg2.copy()
            h = h_tg2.copy()

    if remaining_time() > 20:
        h_final = slsqp_opt(h, best_n, 5000000, maxiter=600)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-14:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Phase 9: Ultra-fine top-k equalization
    if remaining_time() > 15:
        h_eq_f, c5_eq_f = wide_topk_equalize(best_h, best_n, top_k=200, max_iters=500, adaptive=True)
        if c5_eq_f < best_c5 - 1e-14:
            best_c5 = c5_eq_f
            best_h = h_eq_f.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
