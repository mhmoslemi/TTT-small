def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(800):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        at_low = h <= 1e-14
        at_high = h >= 1.0 - 1e-14
        mid = ~(at_low | at_high)
        cnt = int(np.count_nonzero(mid))
        if cnt > 0:
            h[mid] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def to_binary(h, n):
    h = np.array(h, dtype=float).copy()
    target = n // 2
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n, dtype=float)
    h_bin[sorted_idx[:target]] = 1.0
    return h_bin


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    x_new = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(x_new, xp, src_h)


def slsqp_obj(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50.0, 50.0)
    sm = m + np.log(np.sum(np.exp(shifted))) / beta
    return sm * (2.0 / n)


def slsqp_grad(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50.0, 50.0)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        w = np.zeros(len(corr))
    else:
        w = w / w_sum
    g = np.zeros(n)
    nfull = 2 * n - 1
    for k in range(nfull):
        wk = w[k]
        if wk < 1e-10:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            g[offset:offset + npairs] += wk * (1.0 - x[:npairs])
            g[:npairs] -= wk * x[offset:offset + npairs]
        else:
            a = -offset
            npairs = n - a
            g[:npairs] += wk * (1.0 - x[a:a + npairs])
            g[a:a + npairs] -= wk * x[:npairs]
    return g * (2.0 / n)


def slsqp_eq(x, n):
    return np.sum(x) - n / 2.0


def slsqp_eq_jac(x, n):
    return np.ones(n)


def slsqp_optimize(h, n, beta, maxiter=500):
    h0 = np.array(h, dtype=float).copy()
    bounds = [(0.0, 1.0)] * n

    def obj_func(x):
        return slsqp_obj(x, n, beta)

    def grad_func(x):
        return slsqp_grad(x, n, beta)

    def eq_fun(x):
        return slsqp_eq(x, n)

    def eq_jac(x):
        return slsqp_eq_jac(x, n)

    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(obj_func, h0, jac=grad_func, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def compute_dcorr_matrix(h, n):
    """Compute for each position i, the contribution vector to all correlations."""
    nfull = 2 * n - 1
    k_range = np.arange(nfull)
    dcorr = np.zeros((n, nfull), dtype=float)
    for i in range(n):
        j1 = i + k_range - (n - 1)
        v1 = (j1 >= 0) & (j1 < n)
        dcorr[i, v1] += (1.0 - h[j1[v1]])
        j2 = i - k_range + (n - 1)
        v2 = (j2 >= 0) & (j2 < n)
        dcorr[i, v2] -= h[j2[v2]]
    return dcorr

def subgrad_step(h, n, K=30, beta_w=50.0):
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    weights = np.exp(beta_w * (c_top - c_top.max()))
    w_sum = np.sum(weights)
    if w_sum < 1e-30:
        weights = np.ones(K) / K
    else:
        weights /= w_sum

    g = np.zeros(n)
    nfull = 2 * n - 1
    for wt, k in zip(weights, idx):
        if wt < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            np_ = n - offset
            g[offset:offset + np_] += wt * (1.0 - h[:np_])
            g[:np_] -= wt * h[offset:offset + np_]
        else:
            a = -offset
            np_ = n - a
            g[:np_] += wt * (1.0 - h[a:a + np_])
            g[a:a + np_] -= wt * h[:np_]
    return g

            # Quick SLSQP after subgradient

                h_pol = slsqp_optimize(h_feas.copy(), n_t, 30000, maxiter=200)
                c5_pol = compute_c5(h_pol)
                if c5_pol < c5_int:
                    c5_int = c5_pol
                    h_feas = h_pol.copy()

        # Final binary local search
        if n_t <= 2048 and remaining_budget() > 15:
            h_bin_t2 = to_binary(h_feas, n_t)
            h_lt2, c5_lt2 = binary_local_search(h_bin_t2, n_t,
                                                 time_limit=min(25.0, remaining_budget() * 0.2),
                                                 max_passes=30, top_k_frac=0.1)
            if c5_lt2 < c5_int:
                c5_int = c5_lt2
                h_feas = h_lt2.copy()

        if c5_int < best_c5 - 1e-12:
            best_c5 = c5_int
            best_h = h_feas.copy()
            best_n = n_t
            n = n_t
            h = h_feas.copy()

    # Phase 6: Final intensive refinement at best resolution
    n = best_n
    h = best_h.copy()

    for cycle in range(4):
        if remaining_budget() < 25:
            break

        # SLSQP with very high beta
        for beta in [50000, 200000, 500000]:
            if remaining_budget() < 10:
                break
            h_new = slsqp_optimize(h.copy(), n, beta, maxiter=min(400, int(remaining_budget() // 0.03)))
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

        # Binary local search
        if n <= 2048 and remaining_budget() > 15:
            h_bi = to_binary(h, n)
            h_lb, c5_lb = binary_local_search(h_bi, n, time_limit=min(30.0, remaining_budget() * 0.25),
                                              max_passes=40, top_k_frac=0.15)
            if c5_lb < best_c5 - 1e-13:
                best_c5 = c5_lb
                best_h = h_lb.copy()
                h = h_lb.copy()

        # Subgradient escape
        if remaining_budget() > 15:
            h_sge, c5_sge = subgrad_descent(h.copy(), n, time_limit=min(15.0, remaining_budget() * 0.15))
            if c5_sge < best_c5:
                best_c5 = c5_sge
                best_h = h_sge.copy()
                h = h_sge.copy()

    # Phase 7: Final polish
    for beta in [100000, 300000, 1000000]:
        if remaining_budget() < 8:
            break
        h_new = slsqp_optimize(best_h.copy(), best_n, beta, maxiter=min(500, int(remaining_budget() // 0.02)))
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final binary local search
    if best_n <= 2048 and remaining_budget() > 10:
        h_bi_final = to_binary(best_h, best_n)
        h_final, c5_final = binary_local_search(h_bi_final, best_n,
                                                 time_limit=min(20.0, remaining_budget() * 0.4),
                                                 max_passes=50, top_k_frac=0.2)
        if c5_final < best_c5 - 1e-14:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
