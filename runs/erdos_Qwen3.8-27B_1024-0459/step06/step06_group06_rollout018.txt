    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_corr_k(h, k_idx, n):
    g = np.zeros(n)
    s = k_idx - (n - 1)
    if s >= 0:
        npairs = n - s
        if npairs > 0:
            g[:npairs] += (1.0 - h[s:s + npairs])
            g[s:] -= h[:npairs]
    else:
        t = -s
        npairs = n - t
        if npairs > 0:
            g[t:] += (1.0 - h[:npairs])
            g[:npairs] -= h[t:t + npairs]
    return g

        res = minimize(_obj_lbfgsb, h, jac=_grad_lbfgsb, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-15, 'maxfun': maxiter * 3,
                                'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def slsqp_optimize(h, n, beta, maxiter):
    global _CUR_N, _CUR_BETA
    _CUR_N = n
    _CUR_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_obj_slsqp, h, jac=_grad_slsqp, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def top_k_weighted_gradient(h, n, K, beta_w):
    corr = compute_corr(h)
    top_idx = np.argsort(-corr)[:K]
    c_top = corr[top_idx]
    w = np.exp(beta_w * (c_top - c_top.max()))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(K) / K
    else:
        w = w / w_sum
    g = np.zeros(n)
    for i in range(K):
        if w[i] > 1e-12:
            g += w[i] * grad_corr_k(h, top_idx[i], n)
    return g, corr[top_idx]


def equalize_shifts(h, n, K, max_iters, time_limit):
    global _T0, _BUDGET
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        g, c_top = top_k_weighted_gradient(h, n, K, beta_w=120.0)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g_dir = g / gn
        improved = False
        for step in [0.06, 0.03, 0.015, 0.008, 0.04, 0.01, 0.02, 0.05, 0.003, 0.005]:
            ht = project_feasible(h - step * g_dir, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-15:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            for step in [0.04, 0.02, 0.01, 0.005, 0.03, 0.06]:
                ht = project_feasible(h + step * g_dir, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-15:
                    h = ht
                    best_c5 = cn
                    improved = True
                    break
        if not improved and it % 50 == 0:
            K = max(4, K - 1)
    return h, best_c5


def subgradient_descent(h, n, time_limit):
    global _T0, _BUDGET
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        if _BUDGET - (time.time() - _T0) < 5:
            break
        it += 1
        progress = min(1.0, it / 400.0)
        K = 25 if progress < 0.3 else (15 if progress < 0.7 else 8)
        g, _ = top_k_weighted_gradient(h, n, K, beta_w=100.0)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        step = 0.04 / (1.0 + 0.06 * it)
        step = min(step, 0.12)
        h_new = project_feasible(h - step * g / gn, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            found = False
            for f in [0.5, 0.25, 0.1, 0.05]:
                h_try = project_feasible(h - step * f * g / gn, n)
                c_try = compute_c5(h_try)
                if c_try < best_c5:
                    best_c5 = c_try
                    best_h = h_try.copy()
                    h = h_try
                    found = True
                    break
            if not found and it % 150 == 0:
                h = best_h.copy()
    return best_h, best_c5


def coordinate_descent(h, n, time_limit):
    global _T0, _BUDGET
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    while time.time() - start < time_limit:
        if _BUDGET - (time.time() - _T0) < 5:
            break
        improved = False
        for i in range(n):
            if time.time() - start > time_limit:
                break
            old = h[i]
            candidates = []
            if old > 0.005:
                candidates.append(0.0)
            if old < 0.995:
                candidates.append(1.0)
            best_c = best_c5
            best_v = old
            for cv in candidates:
                h[i] = cv
                h = project_feasible(h, n)
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_v = cv
            h[i] = best_v
            h = project_feasible(h, n)
            c_now = compute_c5(h)
            if c_now < best_c5 - 1e-14:
                best_c5 = c_now
                improved = True
        if not improved:
            break
    return h, best_c5


def pair_exchange_descent(h, n, time_limit, n_pairs_try=5000):
    global _T0, _BUDGET
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    rng = np.random.RandomState(12345)
    attempts = 0
    while time.time() - start < time_limit:
        if _BUDGET - (time.time() - _T0) < 5:
            break
        attempts += 1
        i, j = rng.randint(0, n), rng.randint(0, n)
        if i == j:
            continue
        di = h[i]
        dj = h[j]
        if di <= dj:
            continue
        for delta in [0.1, 0.2, 0.3, 0.5, 0.33, 0.42]:
            new_di = di - delta
            new_dj = dj + delta
            if new_di < 0 or new_dj > 1:
                continue
            h[i] = new_di
            h[j] = new_dj
            c = compute_c5(h)
            if c < best_c5 - 1e-15:
                best_c5 = c
                break
        else:
            h[i] = di
            h[j] = dj
    return h, best_c5


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xq = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xq, xp, src_h)


def time_remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _CUR_N, _CUR_BETA, _CUR_PENALTY
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    # Get parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        parent_h = np.zeros(512)
        parent_h[:256] = 1.0

    parent_n = len(parent_h)

    # Work at n=2048
    n = 2048
    h = interp_to(parent_h, parent_n, n)
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()

    # Phase 1: L-BFGS-B coarse
    if time_remaining() > 80:
        h = lbfgsb_optimize(h, n, 300, 500, 2000)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: Subgradient
    if time_remaining() > 50:
        h_sg, c5_sg = subgradient_descent(best_h, n, min(30.0, time_remaining() * 0.2))
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 3: L-BFGS-B medium
    if time_remaining() > 80:
        h = lbfgsb_optimize(best_h, n, 2000, 2000, 2500)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 4: Equalize shifts (broad)
    if time_remaining() > 40:
        h_eq, c5_eq = equalize_shifts(best_h, n, K=20, max_iters=400,
                                      time_limit=min(35.0, time_remaining() * 0.3))
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Phase 5: L-BFGS-B fine
    if time_remaining() > 80:
        h = lbfgsb_optimize(best_h, n, 8000, 5000, 3000)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 6: Equalize shifts (narrow)
    if time_remaining() > 40:
        h_eq, c5_eq = equalize_shifts(best_h, n, K=12, max_iters=400,
                                      time_limit=min(35.0, time_remaining() * 0.3))
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Phase 7: L-BFGS-B very fine
    if time_remaining() > 80:
        h = lbfgsb_optimize(best_h, n, 30000, 10000, 3000)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 8: Coordinate descent (binary moves)
    if time_remaining() > 50:
        h_cd, c5_cd = coordinate_descent(best_h, n, time_limit=min(80.0, time_remaining() * 0.4))
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 9: Equalize shifts (very narrow)
    if time_remaining() > 30:
        h_eq, c5_eq = equalize_shifts(best_h, n, K=8, max_iters=300,
                                      time_limit=min(40.0, time_remaining() * 0.3))
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Phase 10: SLSQP polish
    if time_remaining() > 30:
        h = slsqp_optimize(best_h, n, 20000, 500)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 11: L-BFGS-B ultra-fine
    if time_remaining() > 60:
        h = lbfgsb_optimize(best_h, n, 80000, 20000, 3000)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 12: More equalization
    if time_remaining() > 30:
        h_eq, c5_eq = equalize_shifts(best_h, n, K=6, max_iters=300,
                                      time_limit=min(30.0, time_remaining() * 0.4))
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Phase 13: Coordinate descent again
    if time_remaining() > 40:
        h_cd, c5_cd = coordinate_descent(best_h, n, time_limit=min(60.0, time_remaining() * 0.4))
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 14: Pair exchange
    if time_remaining() > 40:
        h_pe, c5_pe = pair_exchange_descent(best_h, n, time_limit=min(40.0, time_remaining() * 0.4),
                                            n_pairs_try=20000)
        if c5_pe < best_c5:
            best_c5 = c5_pe
            best_h = h_pe.copy()

    # Phase 15: Final L-BFGS-B + equalize
    if time_remaining() > 40:
        h = lbfgsb_optimize(best_h, n, 50000, 15000, 2000)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if time_remaining() > 20:
            h_eq, c5_eq = equalize_shifts(best_h, n, K=5, max_iters=200,
                                          time_limit=min(20.0, time_remaining() * 0.5))
            if c5_eq < best_c5:
                best_c5 = c5_eq
                best_h = h_eq.copy()

    # Phase 16: Try n=1024 path as alternative
    if time_remaining() > 60:
        n_lo = 1024
        h_lo = interp_to(best_h, n, n_lo)
        h_lo = project_feasible(h_lo, n_lo)
        c5_lo = compute_c5(h_lo)
        if c5_lo < best_c5:
            h_lo = lbfgsb_optimize(h_lo, n_lo, 20000, 10000, 2000)
            c5_lo = compute_c5(h_lo)
            h_eq_lo, c5_eq_lo = equalize_shifts(h_lo, n_lo, K=10, max_iters=200, time_limit=20.0)
            if c5_eq_lo < c5_lo:
                h_lo = h_eq_lo
                c5_lo = c5_eq_lo
            if c5_lo < best_c5:
                best_c5 = c5_lo
                best_h = h_lo.copy()
                n = n_lo

    # Final feasibility check
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, float(best_c5), n)
