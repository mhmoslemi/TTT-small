import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        g[offset:] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:]
    else:
        a = -offset
        npairs = n - a
        g[:npairs] += (1.0 - h[a:])
        g[a:] -= h[:npairs]
    return g


def smooth_max_obj_grad(h, n, K, beta):
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    m = np.max(c_top)
    sh = np.clip(beta * (c_top - m), -60.0, 60.0)
    ex = np.exp(sh)
    s = ex.sum()
    if s < 1e-30:
        w = np.ones(K) / K
    else:
        w = ex / s
    dx = 2.0 / n
    val = (m + np.log(s) / beta) * dx
    g = np.zeros(n)
    for i in range(K):
        wk = w[i] * dx
        if wk > 1e-15:
            g += wk * grad_corr_k(h, idx[i], n)
    return val, g


def lbfgsb_optimize(h, n, K, beta, maxiter=500, mu=50.0):
    def obj(x):
        h_p = project_feasible(x, n)
        v, _ = smooth_max_obj_grad(h_p, n, K, beta)
        penalty = mu * (np.sum(h_p) - n / 2.0) ** 2
        return v + penalty

    def grad(x):
        h_p = project_feasible(x, n)
        _, g = smooth_max_obj_grad(h_p, n, K, beta)
        g = g + 2.0 * mu * (np.sum(h_p) - n / 2.0) * np.ones(n)
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-12, 'maxls': 50, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return h


def slsqp_optimize(h, n, K, beta, maxiter=400):
    def obj(x):
        v, _ = smooth_max_obj_grad(x, n, K, beta)
        return v

    def grad(x):
        _, g = smooth_max_obj_grad(x, n, K, beta)
        return g

    def cfun(x):
        return np.sum(x) - n / 2.0

    def cjac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

            y = h_curr + gamma * (h_curr - h_prev)
            y = np.clip(y, 0.0, 1.0)

def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def thue_morse_h(n):
    tm = np.zeros(n, dtype=int)
    for i in range(1, n):
        c = i
        parity = 0
        while c:
            parity ^= (c & 1)
            c >>= 1
        tm[i] = parity
    h = (1 - tm).astype(float)
    return project_feasible(h, n)


def binary_local_search(h, n, max_flips=200):
    h_curr = (h > 0.5).astype(float)
    h_curr = project_feasible(h_curr, n)
    c5_curr = compute_c5(h_curr)
    best = h_curr.copy()
    best_c5 = c5_curr
    for _ in range(max_flips):
        improved = False
        for i in range(n):
            h_try = h_curr.copy()
            h_try[i] = 1.0 - h_try[i]
            h_try = project_feasible(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-10:
                h_curr = h_try
                best = h_try.copy()
                best_c5 = c5_try
                improved = True
                break
        if not improved:
            break
    return best, best_c5
