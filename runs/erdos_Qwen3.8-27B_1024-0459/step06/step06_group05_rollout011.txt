import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def penalty_objective(h, n, C_target, alpha=50.0, K=30):
    corr = compute_corr(h)
    dx = 2.0 / n
    C_corr = C_target * n / 2.0
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    exceed = np.maximum(c_top - C_corr, 0.0)
    val = np.sum(exceed**2) * dx * dx
    sdiff = np.sum(h) - n / 2.0
    val += alpha * sdiff * sdiff
    return val


def penalty_gradient(h, n, C_target, alpha=50.0, K=30):
    corr = compute_corr(h)
    dx = 2.0 / n
    C_corr = C_target * n / 2.0
    idx = np.argsort(-corr)[:K]
    g = np.zeros(n)
    for i in range(K):
        m = idx[i]
        k = m - (n - 1)
        cval = corr[m]
        if cval <= C_corr:
            continue
        w = 2.0 * (cval - C_corr) * dx * dx
        if k >= 0:
            npairs = n - k
            g[k:k + npairs] += w * (1.0 - h[:npairs])
            g[:npairs] -= w * h[k:k + npairs]
        else:
            a = -k
            npairs = n - a
            g[:npairs] += w * (1.0 - h[a:a + npairs])
            g[a:a + npairs] -= w * h[:npairs]
    sdiff = np.sum(h) - n / 2.0
    g += 2.0 * alpha * sdiff
    return g


def smoothmax_objective(h, n, beta=500.0, K=20):
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    m = np.max(c_top)
    sh = np.clip(beta * (c_top - m), -50.0, 50.0)
    ex = np.exp(sh)
    val = (m + np.log(ex.sum() + 1e-30) / beta) * (2.0 / n)
    sdiff = np.sum(h) - n / 2.0
    val += 10.0 * sdiff * sdiff
    return val


def smoothmax_gradient(h, n, beta=500.0, K=20):
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    m = np.max(c_top)
    sh = np.clip(beta * (c_top - m), -50.0, 50.0)
    ex = np.exp(sh)
    s = ex.sum() + 1e-30
    w = ex / s
    dx = 2.0 / n
    g = np.zeros(n)
    for i in range(K):
        wk = w[i] * dx
        if wk < 1e-15:
            continue
        m2 = idx[i]
        k = m2 - (n - 1)
        if k >= 0:

            g[:npairs] -= wk * h[k:k + npairs]
        else:
            a = -k
            npairs = n - a
            g[:npairs] += wk * (1.0 - h[a:a + npairs])
            g[a:a + npairs] -= wk * h[:npairs]
    sdiff = np.sum(h) - n / 2.0
    g += 20.0 * sdiff
    return g


def lbfgs_optimize(h, n, C_target, maxiter=300, K=30, alpha=50.0):
    def obj(x):
        return penalty_objective(x, n, C_target, alpha, K)

    def grad(x):
        return penalty_gradient(x, n, C_target, alpha, K)

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            return h_new
    except Exception:
        pass
    return h


def smooth_lbfgs(h, n, beta=1000.0, K=20, maxiter=300):
    def obj(x):
        return smoothmax_objective(x, n, beta, K)

    def grad(x):
        return smoothmax_gradient(x, n, beta, K)

    bounds = [(0.0, 1.0)] * n

                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8, 'iprint': 0})
        if res.x is not None:

            return h_new
    except Exception:
        pass
    return h

    c5_curr = compute_c5(h_curr)
    h_best = h_curr.copy()

    start = time.time()
    dx = 2.0 / n
    corr = compute_corr(h_curr)

    for sweep in range(2000):
        if time.time() - start > time_limit or remaining_time() < 3:
            break
        for j in range(n):
            if time.time() - start > time_limit:
                break
            h_old = h_curr[j]
            best_val = c5_curr
            best_hj = h_old
            for trial in [-0.02, -0.01, -0.005, 0.005, 0.01, 0.02]:

            c5_curr = best_val
            if c5_curr < c5_best:

                h_best = h_curr.copy()
