_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_and_grad(h, n, K, beta):
    """Compute smooth max objective and gradient."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    # Find top K
    if K >= len(corr):
        idx = np.arange(len(corr))
    else:
        idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    m = np.max(c_top)
    sh = np.clip(beta * (c_top - m), -50.0, 50.0)
    ex = np.exp(sh)
    s = ex.sum()
    w = ex / s
    val = (m + np.log(s) / beta) * dx
    
    # Gradient
    g = np.zeros(n)
    for i in range(len(idx)):
        wk = w[i] * dx
        if wk < 1e-15:
            continue
        k = idx[i] - (n - 1)
        if k >= 0:
            npairs = n - k
            g[k:] += wk * (1.0 - h[:npairs])
            g[:npairs] -= wk * h[k:]
        else:
            a = -k
            npairs = n - a
            g[:npairs] += wk * (1.0 - h[a:])
            g[a:] -= wk * h[:npairs]
    
    return val, g

    g_prev = None

        if _BUDGET - (time.time() - _T0) < 5:

        # Two-loop recursion for L-BFGS
        if len(s_mem) == 0:
            p = -g
        else:
            alpha = [0.0] * len(s_mem)
            q = g.copy()
            for i in range(len(s_mem) - 1, -1, -1):

                q -= alpha[i] * y_mem[i]

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_lag(h, k, n):
    dx = 2.0 / n
    g = np.zeros(n)
    lag = k - (n - 1)
    if lag >= 0:
        npairs = n - lag
        g[lag:] += (1.0 - h[:npairs])
        g[:npairs] -= h[lag:]
    else:
        a = -lag
        npairs = n - a
        g[:npairs] += (1.0 - h[a:])
        g[a:] -= h[:npairs]
    return g * dx


def smooth_max_val_grad(h, n, K, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    nc = len(corr)
    if K >= nc:
        idx = np.arange(nc)
    else:
        idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    m = np.max(c_top)
    sh = np.clip(beta * (c_top - m), -50.0, 50.0)
    ex = np.exp(sh)
    s = ex.sum()
    w = ex / s
    dx = 2.0 / n
    val = (m + np.log(s) / beta) * dx
    g = np.zeros(n)
    for i in range(len(idx)):
        wk = w[i] * dx
        if wk < 1e-15:
            continue
        g += wk * grad_lag(h, idx[i], n)
    return val, g


def slsqp_smooth(h, n, K, beta, maxiter=400):
    def obj(x):
        v, _ = smooth_max_val_grad(x, n, K, beta)
        return v

    def grad(x):
        _, g = smooth_max_val_grad(x, n, K, beta)
        return g

    def cfun(x):
        return np.sum(x) - n / 2.0

    def cjac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-17, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def direct_max_slsqp(h, n, K, maxiter=350):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(-corr)[:K]
    dx = 2.0 / n

    def obj(x):
        hv = x[:n]
        t = x[n]
        cv = np.correlate(hv, 1.0 - hv, mode='full')
        c_sub = cv[idx] * dx
        viol = np.maximum(c_sub - t, 0.0)
        return t + 50.0 * np.sum(viol ** 2)

    def grad(x):
        hv = x[:n]
        t = x[n]
        cv = np.correlate(hv, 1.0 - hv, mode='full')
        c_sub = cv[idx] * dx
        viol = np.maximum(c_sub - t, 0.0)
        g_h = np.zeros(n)
        for i in range(K):
            if viol[i] > 1e-15:
                g_h += 100.0 * viol[i] * grad_lag(hv, idx[i], n)
        g_t = 1.0 - 100.0 * np.sum(viol)
        return np.concatenate([g_h, [g_t]])

    t0 = compute_c5(h) + 0.05
    x0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [{'type': 'eq',
             'fun': lambda x: np.sum(x[:n]) - n / 2.0,
             'jac': lambda x: np.concatenate([np.ones(n), [0.0]])}]
    try:
        res = minimize(obj, x0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x[:n], n)
    except Exception:
        pass
    return h


def nesterov_opt(h, n, K, beta, time_limit):

    h_prev = h.copy()
    h_best = h.copy()
    c5_best = compute_c5(h)
    start = time.time()

    for it in range(500000):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 5:
            break
        if it == 0:
            y = h_curr.copy()
        else:
            t_it = (1.0 + np.sqrt(1.0 + 4.0 * (it ** 2))) / 2.0
            t_prev = (1.0 + np.sqrt(1.0 + 4.0 * ((it - 1) ** 2))) / 2.0
            gamma = (t_prev - 1.0) / t_it
            y = h_curr + gamma * (h_curr - h_prev)
            y = np.clip(y, 0.0, 1.0)

        f_y, g_y = smooth_max_val_grad(y, n, K, beta)
        gn = np.linalg.norm(g_y)
        if gn < 1e-14:
            break

        step = 2.0 / ((it + 1) ** 2 * gn)
        step = min(step, 0.8 / gn)
        h_new = project_feasible(y - step * g_y, n)

        h_prev = h_curr.copy()
        h_curr = h_new
        c5_curr = compute_c5(h_curr)
        if c5_curr < c5_best:

def momentum_gd(h, n, K, beta, time_limit, lr0=0.3, decay=0.9995):
    h_curr = h.copy()
    h_best = h.copy()
    c5_best = compute_c5(h)
    v = np.zeros(n)
    lr = lr0
    mu = 0.85
    start = time.time()

    for it in range(1000000):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 5:
            break

        f, g = smooth_max_val_grad(h_curr, n, K, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break

        v = mu * v + (1.0 - mu) * g
        vn = np.linalg.norm(v)
        if vn < 1e-14:
            break

        step = lr / vn
        h_try = project_feasible(h_curr - step * v, n)
        f_try = compute_c5(h_try)

        if f_try < c5_best:
            c5_best = f_try
            h_best = h_try.copy()
            h_curr = h_try
        else:
            # Backtrack
            improved = False
            for bt in range(10):
                step *= 0.5
                h_try = project_feasible(h_curr - step * v, n)
                f_try2, _ = smooth_max_val_grad(h_try, n, K, beta)
                if f_try2 < f:
                    h_curr = h_try
                    improved = True
                    break
            if not improved:
                lr *= 0.9
                if lr < 1e-6:
                    break

        lr *= decay

    return h_best, c5_best
