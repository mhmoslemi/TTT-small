```python
def project_feasible_v2(h, n):
    h = np.asarray(h, dtype=float)
    target = n / 2.0
    # Find lambda such that sum(clip(h - lambda, 0, 1)) = target
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(h - lam, 0, 1)
```

```python
def project_feasible(h, n):
    """Project h onto {0 <= h <= 1, sum(h) = n/2} using binary search on Lagrange multiplier."""
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return np.clip(h, 0.0, 1.0)
    # Binary search for lambda
    lo = h.min() - 1.0
    hi = h.max() + 1.0

        if s_mid > target:
            lo = mid
        else:
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_AS_N = 0
_AS_KLIST = None


def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)


def project_feasible(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2} via binary search on Lagrange multiplier."""
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-13:
        return np.clip(h, 0.0, 1.0)
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s_mid = np.clip(h - mid, 0.0, 1.0).sum()
        if s_mid > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)


def grad_corr_k_vec(h, k, n):
    """Gradient of the k-th correlation w.r.t. h."""
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        length = n - offset
        g[offset:] += (1.0 - h[:length])
        g[:length] -= h[offset:]
    else:
        a = -offset
        length = n - a
        g[:length] += (1.0 - h[a:])
        g[a:] -= h[:length]
    return g


def smax_obj_grad(h, n, beta, K):
    """Smooth-max (log-sum-exp) objective and gradient using top-K shifts."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(-corr)[:K]
    c = corr[idx]
    c_max = c.max()
    sh = c - c_max
    cl = np.clip(beta * sh, -80.0, 80.0)
    ex = np.exp(cl)
    ws = ex / ex.sum()
    obj = (c_max + np.log(ex.sum()) / beta) * (2.0 / n)
    g = np.zeros(n)
    dx = 2.0 / n
    for i in range(K):
        if ws[i] < 1e-15:
            continue
        wk = ws[i] * dx
        g += wk * grad_corr_k_vec(h, idx[i], n)
    return obj, g

def slsqp_tminmax(h, n, K_list, maxiter):
    """SLSQP min-max: minimize t s.t. corr_k*dx <= t for k in K_list."""
    global _AS_N, _AS_KLIST
    _AS_N = n
    _AS_KLIST = K_list
    t0 = compute_c5(h)
    v0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    try:
        res = minimize(_tmin_obj, v0, jac=_tmin_jac, method='SLSQP',
                       bounds=bounds, constraints=[
                           {'type': 'ineq', 'fun': _tmin_ineq, 'jac': _tmin_ineq_jac},
                           {'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac}
                       ],
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x[:n], n)
            c5_new = compute_c5(h_new)
            if c5_new <= compute_c5(h) + 1e-12:
                return h_new, c5_new
    except Exception:
        pass
    return h.copy(), compute_c5(h)


def _tmin_obj(v):
    return v[-1]


def _tmin_jac(v):
    g = np.zeros(len(v))
    g[-1] = 1.0
    return g


def _tmin_ineq(v):
    h = v[:-1]
    t = v[-1]
    n = _AS_N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return t * n / 2.0 - corr[_AS_KLIST]


def _tmin_ineq_jac(v):
    h = v[:-1]
    n = _AS_N
    K_list = _AS_KLIST
    J = np.zeros((len(K_list), n + 1))
    for i, k in enumerate(K_list):
        g = grad_corr_k_vec(h, k, n)
        J[i, :n] = -g
        J[i, -1] = n / 2.0
    return J


def _sum_eq(v):
    return np.sum(v[:-1]) - _AS_N / 2.0


def _sum_eq_jac(v):
    g = np.zeros(len(v))
    g[:-1] = 1.0
    return g

                h5, c5_5 = fista_optimize(h_hi, n_high, t5,
                                          beta_start=30, beta_end=400, K=30)
                if c5_5 < best_c5:
                    best_c5 = c5_5
                    best_h = h5.copy()
                    h = h5.copy()
                    n = n_high
                else:
                    h = h_hi
                    n = n_high

            # SLSQP at high res
            if remaining() > 60:
                corr_hi = np.correlate(h, 1.0 - h, mode='full')
                for K_a in [10, 20, 30, 40]:
                    if remaining() < 15:
                        break
                    idx_a = np.argsort(-np.correlate(h, 1.0 - h, mode='full'))[:K_a]
                    h6, c5_6 = slsqp_tminmax(h, n, idx_a, maxiter=400)
                    if c5_6 < best_c5:
                        best_c5 = c5_6
                        best_h = h6.copy()
                        h = h6.copy()

            # Active set at high res
            if remaining() > 40:
                h7, c5_7 = active_set_opt(h, n, K=10, maxiter=600)
                if c5_7 < best_c5:
                    best_c5 = c5_7
                    best_h = h7.copy()
                    h = h7.copy()

            # FISTA high beta at high res
            if remaining() > 50:
                h8, c5_8 = fista_optimize(h, n, min(80.0, remaining() * 0.1),
                                          beta_start=400, beta_end=3000, K=40)
                if c5_8 < best_c5:
                    best_c5 = c5_8
                    best_h = h8.copy()
                    h = h8.copy()

    # Phase 6: Try n=2048
    n_vhigh = 2048
    if remaining() > 250:
        h_vh = interp_to(best_h, best_n, n_vhigh)
        c5_vh = compute_c5(h_vh)

        if c5_vh <= best_c5 + 0.003:
            if remaining() > 150:
                t6 = min(200.0, remaining() * 0.3)
                h9, c5_9 = fista_optimize(h_vh, n_vhigh, t6,
                                          beta_start=30, beta_end=300, K=30)
                if c5_9 < best_c5:
                    best_c5 = c5_9
                    best_h = h9.copy()
                    h = h9.copy()
                    n = n_vhigh
                    best_n = n_vhigh

                if remaining() > 50:
                    h10, c5_10 = active_set_opt(h, n, K=8, maxiter=500)
                    if c5_10 < best_c5:
                        best_c5 = c5_10
                        best_h = h10.copy()
                        h = h10.copy()

                if remaining() > 40:
                    h11, c5_11 = fista_optimize(h, n, min(50.0, remaining() * 0.1),
                                                beta_start=300, beta_end=2000, K=35)
                    if c5_11 < best_c5:
                        best_c5 = c5_11
                        best_h = h11.copy()
                        h = h11.copy()

    # Phase 7: Final refinement cycles at best resolution
    n = best_n
    h = best_h.copy()
    for cycle in range(3):
        if remaining() < 30:
            break

        # FISTA polish
        if remaining() > 25:
            hf, cf = fista_optimize(h, n, min(40.0, remaining() * 0.25),
                                    beta_start=1000, beta_end=5000, K=50)
            if cf < best_c5:
                best_c5 = cf
                best_h = hf.copy()
                h = hf.copy()

        # Active set
        if remaining() > 20:
            ha, ca = active_set_opt(h, n, K=12, maxiter=500)
            if ca < best_c5:
                best_c5 = ca
                best_h = ha.copy()
                h = ha.copy()

    # Phase 8: Tiny steps polish
    if remaining() > 20:
        hp, cp = tiny_steps_polish(best_h, best_n, max_iters=500)
        if cp < best_c5:
            best_c5 = cp
            best_h = hp.copy()

    # Phase 9: Final SLSQP polish
    if remaining() > 20:
        corr_f = np.correlate(best_h, 1.0 - best_h, mode='full')
        for K_f in [15, 25]:
            if remaining() < 10:
                break
            idx_f = np.argsort(-corr_f)[:K_f]
            hs, cs = slsqp_tminmax(best_h, best_n, idx_f, maxiter=300)
            if cs < best_c5:
                best_c5 = cs
                best_h = hs.copy()

    # Final feasibility and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
```
