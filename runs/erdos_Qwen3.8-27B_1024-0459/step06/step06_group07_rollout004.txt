import numpy as np
import time

_T0 = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def to_binary(h, n):
    order = np.argsort(-h)
    binary = np.zeros(n)
    binary[order[:n//2]] = 1.0
    return binary

def project_binary(h, n):
    """Project to exactly n/2 ones"""
    ones = np.sum(h >= 0.5)
    if ones == n // 2:
        return (h >= 0.5).astype(float)
    # Need exactly n/2 ones
    vals = h.copy()
    # Sort by value descending, take top n/2
    order = np.argsort(-vals)
    binary = np.zeros(n)
    binary[order[:n//2]] = 1.0
    return binary

def simulate_annealing(h, n, time_limit=60.0, T0=10.0, T1=0.01, cooling=0.995):
    """Simulated annealing on binary configurations"""
    h = h.copy()
    corr = compute_corr(h)
    max_corr = float(np.max(corr))
    best_h = h.copy()
    best_max_corr = max_corr
    ones_idx = np.where(h == 1.0)[0]
    zeros_idx = np.where(h == 0.0)[0]
    start = time.time()
    T = T0
    rng = np.random.RandomState(42)
    
    while time.time() - start < time_limit and _BUDGET - (time.time() - _T0) > 10:
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        
        # Random swap
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        
        # Perform swap
        h[i] = 0.0
        h[j] = 1.0
        new_corr = compute_corr(h)
        new_max = float(np.max(new_corr))
        
        delta = new_max - max_corr
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            # Accept
            corr = new_corr
            max_corr = new_max
            ones_idx = np.where(h == 1.0)[0]
            zeros_idx = np.where(h == 0.0)[0]
            
            if max_corr < best_max_corr:
                best_max_corr = max_corr
                best_h = h.copy()
        else:
            # Reject
            h[i] = 1.0
            h[j] = 0.0
        
        T *= cooling
    
    return best_h, best_max_corr * (2.0 / n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def to_binary(h, n):
    order = np.argsort(-h)
    binary = np.zeros(n)
    binary[order[:n // 2]] = 1.0
    return binary


def project_binary(h, n):
    vals = np.array(h, dtype=float)
    order = np.argsort(-vals)
    binary = np.zeros(n)
    binary[order[:n // 2]] = 1.0
    return binary


def upsample_binary(h, n_src, n_dst):
    if n_dst == n_src:
        return h.copy()
    xp = np.linspace(0.0, 2.0, n_src, endpoint=False) + 1.0 / n_src
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False) + 1.0 / n_dst
    interp = np.interp(x_new, xp, h)
    return project_binary(interp, n_dst)

def theue_morse_init(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i.bit_count() % 2 == 1) else 0.0

    for i in range(n):
        if pow(i + 1, (p - 1) // 2, p) == 1:
            h[i] = 1.0
    h = project_binary(h, n)
    return h
