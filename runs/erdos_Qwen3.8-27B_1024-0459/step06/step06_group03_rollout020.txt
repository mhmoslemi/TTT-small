                h_opt, c5_opt = sg_descent(h_opt, n_hi, min(35.0, rem() * 0.15), rng)
            if rem() > 25:
                h_opt = do_slsqp(h_opt, n_hi, 20000, K=50, maxit=500)
                c5_opt = compute_c5(h_opt)
            if rem() > 40:
                for it in range(100):
                    if rem() < 5:
                        break
                    h_lp, t_lp = cut_plane_lp(h_opt, n_hi, K=30, delta=0.03)
                    c_cur = compute_c5(h_opt)
                    if t_lp < c_cur - 1e-9:
                        for alpha in [1.0, 0.6, 0.4, 0.8]:
                            h_try = project_feasible(h_opt + alpha * (h_lp - h_opt), n_hi)
                            c_try = compute_c5(h_try)
                            if c_try < c5_opt:
                                h_opt = h_try
                                c5_opt = c_try
                                break
                    elif it > 60:
                        break
            if rem() > 20:
                h_opt, c5_opt = equalize(h_opt, n_hi, K=10, max_it=150)
            if rem() > 15:
                h_opt = do_slsqp(h_opt, n_hi, 50000, K=70, maxit=600)
                c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_hi

    # Phase 7: Refinement cycles at best resolution
    n = best_n
    h = best_h.copy()
    for cycle in range(3):
        if rem() < 25:
            break
        if rem() > 15:
            h, c5 = sg_descent(h, n, min(20.0, rem() * 0.12), rng)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        if rem() > 12:
            h = do_slsqp(h, n, beta=30000 + 30000 * cycle, K=70, maxit=600)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        if rem() > 12:
            h, c5 = equalize(h, n, K=15, max_it=150)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 8: Micro-steps
    h_f = best_h.copy()
    if rem() > 12:
        for it in range(120):
            if rem() < 3:
                break
            corr_f = np.correlate(h_f, 1.0 - h_f, mode='full')
            idx_f = np.argsort(-corr_f)[:25]
            g_f = np.zeros(n)
            for k in idx_f:
                g_f += grad_k(h_f, k, n)
            g_f /= len(idx_f)
            gn_f = np.linalg.norm(g_f)
            if gn_f < 1e-14:
                break
            for step in [0.002, 0.001, 0.0005, 0.004, 0.0015, 0.0008, 0.003]:
                h_try = project_feasible(h_f - step * g_f / gn_f, n)
                c_try = compute_c5(h_try)
                if c_try < best_c5 - 1e-16:
                    best_c5 = c_try
                    best_h = h_try.copy()
                    h_f = h_try.copy()
                    break

    # Phase 9: Final SLSQP with very high beta
    if rem() > 8:
        h = do_slsqp(best_h, best_n, 200000, K=80, maxit=800)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Final projection
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
