def subgradient_descent(h, n, time_limit, K=15):
    start = time.time()
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(20000):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 10:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = corr.max() * (2.0/n)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Compute weighted subgradient
        top_k = np.argsort(-corr)[:K]
        # Use softmax weights
        c_top = corr[top_k]
        w = np.exp(50 * (c_top - c_top.max()))
        s = w.sum()
        if s < 1e-30:
            w = np.ones(K) / K
        else:
            w /= s
        
        g = np.zeros(n)
        for ii, k_idx in enumerate(top_k):
            if w[ii] < 1e-12:
                continue
            off = k_idx - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:] += w[ii] * (1.0 - h[:np_])
                g[:np_] -= w[ii] * h[off:]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += w[ii] * (1.0 - h[ao:])
                g[ao:] -= w[ii] * h[:np_]
        g *= (2.0/n)
        
        # Project gradient
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        
        # Backtracking line search
        alpha = 0.1
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_try = project_feasible(h - mult * alpha * g / gn, n)
            c_try = compute_c5(h_try)
            if c_try < c5_val:
                h = h_try
                break
        else:
            break
    
    return best_h, best_c5

    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        if interior.any():
            h[interior] += diff / interior.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    s = np.clip(beta * (corr - m), -500.0, 500.0)
    e = np.exp(s)
    return (m + np.log(e.sum()) / beta) * dx

def smooth_max_jac(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    s = np.clip(beta * (corr - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-15:
            continue
        off = k_idx - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - h[ao:])
            g[ao:] -= wk * h[:np_]
    return g * dx

def eq_fun(x, n):
    return x.sum() - n / 2.0

def eq_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, beta, maxiter, n):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_max_val, h, args=(beta,),
                      jac=smooth_max_jac, method='SLSQP',
                      bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (n,)}],
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def tc_opt(h, beta, maxiter, n):
    try:
        bnds = Bounds(np.zeros(n), np.ones(n))
        A_eq = np.ones((1, n))
        cons = LinearConstraint(A_eq, n/2.0, n/2.0)
        res = minimize(smooth_max_val, h, args=(beta,),
                      jac=smooth_max_jac, method='trust-constr',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'gtol': 1e-8, 'xtol': 1e-10})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def solve_1d_max(a, b, lo, hi):
    """Minimize max_k (a[k] + b[k]*t) for t in [lo, hi] via bisection."""
    if lo >= hi:
        return lo
    for _ in range(50):
        t_mid = (lo + hi) * 0.5
        vals = a + b * t_mid
        g_val = vals.max()
        active = (vals > g_val - 1e-14)
        if not active.any():
            active[idx] = True where idx = argmax
        b_act = b[active]
        b_min = b_act.min()
        b_max = b_act.max()
        if b_min > 1e-14:
            hi = t_mid
        elif b_max < -1e-14:
            lo = t_mid
        else:
            break
    return (lo + hi) * 0.5
