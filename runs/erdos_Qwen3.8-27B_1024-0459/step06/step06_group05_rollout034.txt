import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0

def next_pow2(n):
    size = 1
    while size < n:
        size *= 2
    return size

def correlate_fft(h):
    """Fast computation of np.correlate(h, 1-h, 'full') using FFT."""
    n = len(h)
    size = 2 * n - 1
    size_fft = next_pow2(size)
    H = np.fft.rfft(h, size_fft)
    O = np.fft.rfft((1.0 - h)[::-1], size_fft)
    corr = np.fft.irfft(H * O, size_fft)
    return corr[:size]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate_fft(h)
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        # Distribute the correction to non-saturated coordinates
        not_sat = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(not_sat)
        if cnt > 0:
            h[not_sat] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_corr_k(h, k, n):
    """Gradient of corr[k] with respect to h."""
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        g[offset:] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:]
    else:
        a = -offset
        npairs = n - a
        g[:npairs] += (1.0 - h[a:])
        g[a:] -= h[:npairs]
    return g

def smooth_max_grad(h, K, beta):
    """Compute gradient of smooth max top-K of correlation."""
    n = len(h)
    corr = correlate_fft(h)
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    m = np.max(c_top)
    sh = np.clip(beta * (c_top - m), -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / ex.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for i in range(K):
        if w[i] * dx < 1e-15:
            continue
        g += w[i] * dx * grad_corr_k(h, idx[i], n)
    return g

def slsqp_optimize(h, n, K, beta, maxiter=400):
    def obj(x):
        corr = correlate_fft(x)
        idx = np.argsort(-corr)[:K]
        c_top = corr[idx]
        m = np.max(c_top)
        sh = np.clip(beta * (c_top - m), -50.0, 50.0)
        ex = np.exp(sh)
        return (m + np.log(ex.sum()) / beta) * (2.0 / n)
    
    def grad(x):
        return smooth_max_grad(x, K, beta)
    
    def cfun(x):
        return np.sum(x) - n / 2.0
    
    def cjac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    
    try:

                       bounds=bounds, constraints=cons,

        if res.x is not None:

def next_pow2(n):
    size = 1
    while size < n:
        size *= 2
    return size

def correlate_fft(h):
    n = len(h)
    size = 2 * n - 1
    size_fft = next_pow2(size)
    H = np.fft.rfft(h, size_fft)
    O = np.fft.rfft((1.0 - h)[::-1], size_fft)
    corr = np.fft.irfft(H * O, size_fft)
    return corr[:size]

def compute_c5(h):

    dx = 2.0 / n

    return float(np.max(corr)) * dx

def project_feasible(h, n):

        return smooth_max_grad(x, K, beta)
