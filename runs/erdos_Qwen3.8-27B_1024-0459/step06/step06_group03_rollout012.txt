For very small δ (like 0.001), this redistribution is tiny and unlikely to push anything to bounds. So the projection is essentially: h[i] += δ, h[j] -= δ/n for j ≠ i.

Let me optimize this:

```python
def try_coord_move(h, n, i, delta):
    """Try changing h[i] by delta, maintaining sum constraint."""
    if h[i] + delta < 0 or h[i] + delta > 1:
        return None
    h_new = h.copy()
    h_new[i] += delta
    # Redistribute -delta across other coordinates
    others = np.ones(n)
    others[i] = 0
    n_others = n - 1
    h_new -= (delta / n_others) * others
    # Clip and fix
    h_new = np.clip(h_new, 0.0, 1.0)
    # Final sum fix
    diff = n / 2.0 - np.sum(h_new)
    if abs(diff) > 1e-15:
        mask = (h_new > 1e-15) & (h_new < 1.0 - 1e-15)
        if np.sum(mask) > 0:
            h_new[mask] += diff / np.sum(mask)
            h_new = np.clip(h_new, 0.0, 1.0)
    return h_new
```

This is much faster than the full project_feasible.

Now, for the active-set LP refinement:

The nonlinear problem is:
min t
s.t. corr_k(h) ≤ t for k in active set
     sum(h) = n/2
     0 ≤ h ≤ 1

Since corr_k is quadratic in h (it's h·(1-h) at shifted positions), the problem is non-convex. But we can linearize:
corr_k(h + Δ) ≈ corr_k(h) + g_k · Δ (first order)

So the LP is:
min t
s.t. corr_k(h) + g_k · Δ ≤ t for k in active set
     sum(Δ) = 0
     0 ≤ h + Δ ≤ 1

This is a linear program with n+1 variables and K+1+n*2 constraints. For n=512, K=20, this is manageable with HiGHS.

Let me also think about a "pair move" for local search:
- Pick two coordinates i, j
- Move: h[i] += δ, h[j] -= δ (maintains sum)
- This is a 2D search

For the perturbation approach:
- Take current h
- Add random noise: h_noise = h + σ * (random - random) (zero-mean, zero-sum)
- Clip to [0,1]
- Project to feasible
- Run quick SLSQP
- Keep if improved

The key question is: what σ to use? If we're very close to the optimum, σ should be small (like 0.01-0.05). If we're in a local minimum, σ should be larger.

1. Less emphasis on SLSQP (which is already well-done in the previous program)
2. More emphasis on structured local search (coordinate descent, pair moves)

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

# Global state for SLSQP callbacks
_G_N = 0
_G_BETA = 0
_G_K = 0
_G_KLIST = None

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)

def get_rem(T0, budget):
    return max(0.0, budget - (time.time() - T0) - 3.0)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        np_ = n - offset
        g[offset:] += (1.0 - h[:np_])
        g[:np_] -= h[offset:]
    else:
        a = -offset
        np_ = n - a
        g[:np_] += (1.0 - h[a:])
        g[a:] -= h[:np_]
    return g

def _smax_obj(x):
    n = _G_N
    beta = _G_BETA
    K = _G_K
    corr = np.correlate(x, 1.0 - x, mode='full')
    idx = np.argsort(-corr)[:K]
    c = corr[idx]
    m = c.max()
    sh = c - m
    cl = np.clip(beta * sh, -80.0, 80.0)
    ex = np.exp(cl)
    return (m + np.log(ex.sum()) / beta) * (2.0 / n)

def _smax_grad(x):
    n = _G_N
    beta = _G_BETA
    K = _G_K
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    idx = np.argsort(-corr)[:K]
    c = corr[idx]
    m = c.max()
    sh = c - m
    cl = np.clip(beta * sh, -80.0, 80.0)
    ex = np.exp(cl)
    w = ex / ex.sum()
    g = np.zeros(n)
    for i in range(K):
        if w[i] > 1e-15:
            g += w[i] * grad_corr_k(x, idx[i], n)
    g *= dx
    return g

def _sum_eq(x):
    return np.sum(x) - _G_N / 2.0

def _sum_eq_jac(x):
    return np.ones(_G_N)

def slsqop_smax(h, n, beta, K, maxiter):
    global _G_N, _G_BETA, _G_K
    _G_N = n
    _G_BETA = beta
    _G_K = K
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac}]
    try:
        res = minimize(_smax_obj, h, jac=_smax_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            c5_old = compute_c5(h)
            if c5_new <= c5_old + 1e-12:
                return h_new, c5_new
    except Exception:
        pass
    return h.copy(), compute_c5(h)

def coord_descent(h, n, K, n_sweeps, T0, budget):
    h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(0)
    
    for sweep in range(n_sweeps):
        if get_rem(T0, budget) < 5:
            break
        
        # Shuffle order
        order = np.arange(n)
        rng.shuffle(order)
        
        for i in order:
            if get_rem(T0, budget) < 2:
                break
            # Try moves
            for delta in [0.005, -0.005, 0.002, -0.002, 0.001, -0.001, 0.01, -0.01]:
                hi = h[i] + delta
                if hi < 0 or hi > 1:
                    continue
                h_trial = h.copy()
                h_trial[i] = hi
                # Fix sum: redistribute -delta across all others
                others_sum = np.sum(h_trial[:i]) + np.sum(h_trial[i+1:])
                target = n / 2.0
                diff = target - np.sum(h_trial)
                if abs(diff) < 1e-15:
                    continue
                # Simple redistribution
                mask = (h_trial > 1e-15) & (h_trial < 1.0 - 1e-15)
                mask[i] = False  # don't change i further
                cnt = np.count_nonzero(mask)
                if cnt > 0:
                    h_trial[mask] += diff / cnt
                h_trial = np.clip(h_trial, 0.0, 1.0)
                # Verify sum
                if abs(np.sum(h_trial) - n/2.0) > 1e-10:
                    continue
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5 - 1e-15:
                    h = h_trial
                    best_c5 = c5_trial
                    break
    
    return h, best_c5

def active_set_lp(h, n, K, delta_bound, T0, budget):
    """Solve linearized active-set problem."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(-corr)[:K]
    c_active = corr[idx]
    c_max = c_active[0]
    
    # Build LP: min t
    # s.t. c_k + g_k·Δ ≤ t  (K constraints)
    #      sum(Δ) = 0
    #      -delta_bound ≤ Δ_i ≤ delta_bound (trust region)
    #      0 ≤ h_i + Δ_i ≤ 1 (bounds)
    
    dim = n + 1  # Δ (n) + t (1)
    c_obj = np.zeros(dim)
    c_obj[-1] = 1.0
    
    # Inequality constraints: c_k + g_k·Δ - t ≤ 0  →  g_k·Δ - t ≤ -c_k
    A_ub_rows = []
    b_ub_rows = []
    for i in range(K):
        g = grad_corr_k(h, idx[i], n)
        row = np.zeros(dim)
        row[:n] = g
        row[-1] = -1.0
        A_ub_rows.append(row)
        b_ub_rows.append(-c_active[i])
    
    # Trust region: |Δ_i| ≤ delta_bound
    I_n = np.eye(n)
    A_trust = np.vstack([np.hstack([I_n, np.zeros((n,1))]),
                          np.hstack([-I_n, np.zeros((n,1))])])
    b_trust = np.concatenate([np.full(n, delta_bound), np.full(n, delta_bound)])
    
    # Bounds: 0 ≤ h_i + Δ_i ≤ 1 → -h_i ≤ Δ_i ≤ 1 - h_i
    A_bound = np.vstack([np.hstack([I_n, np.zeros((n,1))]),
                          np.hstack([-I_n, np.zeros((n,1))])])
    b_bound = np.concatenate([1.0 - h, h])
    
    A_ub = np.vstack([A_ub_rows, A_trust, A_bound])
    b_ub = np.concatenate([b_ub_rows, b_trust, b_bound])
    
    A_eq = np.zeros((1, dim))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    bounds = [(None, None)] * n + [(None, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            delta = res.x[:n]
            t_new = res.x[-1]
            h_new = project_feasible(h + delta, n)
            c5_new = compute_c5(h_new)
            return h_new, c5_new, t_new
    except Exception:
        pass
    return h.copy(), compute_c5(h), c_max

def perturb_reoptimize(h, n, sigma, beta, K, maxiter, T0, budget):
    rng = np.random.RandomState(int(time.time() * 1000) % 100000)
    h_noise = h + sigma * (rng.random(n) - rng.random(n))  # zero-sum noise
    h_noise = project_feasible(h_noise, n)
    c5_before = compute_c5(h)
    h_new, c5_new = slsqop_smax(h_noise, n, beta, K, maxiter)
    if c5_new < c5_before:
        return h_new, c5_new
    return h.copy(), c5_before

Each correlation is O(n) with the full mode being O(n) for a single lag but O(n²) for all lags. Wait, np.correlate with mode='full' for two arrays of length n is O(n²). So each compute_c5 call is O(n²) = O(262K). With n*2 = 1024 pairs and S=4 steps, that's about 4096 correlation calls per sweep, each O(n²). Total: 4096 * 262K ≈ 1B operations per sweep. That's too slow!

I need to be smarter. Instead of computing the full correlation each time, I can compute only the top K correlation values. Or better, I can compute the change in the top K correlations analytically.

For a pair move (i, j, δ):
- h[i] → h[i] + δ
- h[j] → h[j] - δ

The change in corr_k:
Δcorr_k = δ * (grad_k[i] - grad_k[j])

where grad_k = grad_corr_k(h, k, n).

So if I precompute grad_k for the top K shifts, I can evaluate the change in each in O(K) time per pair+step. Much faster!
