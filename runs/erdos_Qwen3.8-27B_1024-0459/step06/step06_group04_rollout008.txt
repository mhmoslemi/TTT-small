    m = corr.max()

        pass
    return h


def active_set_descend(h, n, time_limit, K_start=5, K_end=30, momentum=0.3):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    v = np.zeros(n)
    patience = 0

    for it in range(200000):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 5:
            break

        corr = all_corrs(h)
        c5_raw = corr.max()
        c5 = c5_raw * (2.0 / n)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1

        if patience > 2000:
            K_use = K_end
        else:
            frac = min(1.0, it / 5000.0)
            K_use = int(K_start + frac * (K_end - K_start))

        top_idx = np.argsort(-corr)[:K_use]
        top_vals = corr[top_idx]
        top_max = top_vals.max()
        sig = top_max - top_vals.min()
        if sig > 1e-10:
            weights = np.exp(80.0 * (top_vals - top_max) / sig)
        else:
            weights = np.ones(K_use) / K_use
        weights /= weights.sum()

        g = np.zeros(n)
        for i, k_idx in enumerate(top_idx):
            g += weights[i] * grad_lag(h, k_idx)
        g *= (2.0 / n)

        g_feas = g - g.mean()
        v = momentum * v + (1.0 - momentum) * g_feas
        v_feas = v - v.mean()

        gn = np.linalg.norm(v_feas)
        if gn < 1e-14:
            break

        step_found = False
        for step in [0.3, 0.15, 0.07, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_try = project_feasible(h - step * v_feas / gn, n)
            c_try = compute_c5(h_try)
            if c_try < best_c5:
                h = h_try
                best_c5 = c_try
                best_h = h.copy()
                patience = 0
                step_found = True
                break

        if not step_found and it > 10000:
            h_try = project_feasible(h - 1e-5 * v_feas / gn, n)
            c_try = compute_c5(h_try)
            if c_try < best_c5:
                h = h_try
                best_h = h.copy()

    return best_h, best_c5


def coord_local_search(h, n, time_limit):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    rng = np.random.RandomState(12345)
    patience = 0

    for rnd in range(500000):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 5:
            break

        if rnd % 3 == 0:
            i = rng.randint(n)
            if h[i] > 0.05 and h[i] < 0.95:
                h_try = h.copy()
                h_try[i] = 1.0 if h[i] > 0.5 else 0.0
                h_try = project_feasible(h_try, n)
                c_try = compute_c5(h_try)
                if c_try < best_c5 - 1e-14:
                    h = h_try
                    best_c5 = c_try
                    best_h = h.copy()
                    patience = 0
                    continue

        j = rng.randint(n)
        candidates = []
        if h[j] > 0.1:
            candidates.append(h[j] - 0.15)
            candidates.append(h[j] - 0.05)
            candidates.append(0.0)
        if h[j] < 0.9:
            candidates.append(h[j] + 0.15)
            candidates.append(h[j] + 0.05)
            candidates.append(1.0)
        for val in candidates:
            val = float(np.clip(val, 0.0, 1.0))
            if abs(val - h[j]) < 1e-6:
                continue
            h_try = h.copy()
            h_try[j] = val
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try)
            if c_try < best_c5 - 1e-14:
                h = h_try
                best_c5 = c_try
                best_h = h.copy()
                patience = 0
                break
        else:
            patience += 1

        if rnd % 500 == 250:
            i = rng.randint(n)
            j2 = rng.randint(n)
            if i != j2 and h[i] != h[j2]:
                h_try = h.copy()
                h_try[i], h_try[j2] = h[j2], h[i]
                h_try = project_feasible(h_try, n)
                c_try = compute_c5(h_try)
                if c_try < best_c5 - 1e-14:
                    h = h_try
                    best_c5 = c_try
                    best_h = h.copy()
                    patience = 0

        if patience > 5000:
            break

    return best_h, best_c5


def weighted_topk_descent(h, n, time_limit, K=15, beta_w=100.0):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    v = np.zeros(n)
    mom = 0.4

    for it in range(300000):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 5:
            break

        corr = all_corrs(h)
        top_idx = np.argsort(-corr)[:K]
        top_vals = corr[top_idx]
        top_max = top_vals.max()

        if top_max > 0:
            weights = np.exp(beta_w * (top_vals - top_max) / top_max)
        else:
            weights = np.ones(K) / K
        sw = weights.sum()
        if sw < 1e-30:
            weights = np.ones(K) / K
        else:
            weights /= sw

        g = np.zeros(n)
        for i, k_idx in enumerate(top_idx):
            g += weights[i] * grad_lag(h, k_idx)
        g *= (2.0 / n)
        g_feas = g - g.mean()

        v = mom * v + (1.0 - mom) * g_feas
        v_feas = v - v.mean()
        gn = np.linalg.norm(v_feas)
        if gn < 1e-14:
            break

        improved = False
        for step in [0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
            h_try = project_feasible(h - step * v_feas / gn, n)
            c_try = compute_c5(h_try)
            if c_try < best_c5:
                h = h_try
                best_c5 = c_try
                best_h = h.copy()
                improved = True
                break

        if not improved and it > 20000:
            h_try = project_feasible(h - 1e-6 * v_feas / gn, n)
            c_try = compute_c5(h_try)
            if c_try < best_c5:
                best_c5 = c_try
                best_h = h.copy()

    return best_h, best_c5


def interp_up(h, n_src, n_dst):
    xp = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_new = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    return np.interp(x_new, xp, h)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s

    def rem():
        return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

    try:
        best_h = np.array(initial_h_values, dtype=float).copy()
        best_n = len(best_h)
    except Exception:
        best_n = 512
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP at original resolution with increasing beta
    for beta in [500, 5000, 50000]:
        if rem() < 30:
            break
        t_start = time.time()
        maxit = int(min(300, max(50, (min(60.0, rem() * 0.4)) / 0.5)))
        h_new = slsqp_opt(best_h, beta, maxit)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    # Phase 2: Upsample to 2x resolution
    n2 = best_n * 2
    if n2 <= 4096 and rem() > 60:
        h_up = interp_up(best_h, best_n, n2)
        h_up = project_feasible(h_up, n2)
        c_up = compute_c5(h_up)

        for beta in [1000, 10000, 100000]:
            if rem() < 30:
                break
            h_new = slsqp_opt(h_up, beta, maxiter=int(min(200, max(30, rem() * 0.2 / 0.5))))
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new
                best_n = n2
            elif c_new < c_up:
                h_up = h_new
        if rem() > 20:
            h_new = project_feasible(h_up, n2)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new
                best_n = n2

    # Phase 3: Active-set descent (long run)
    if rem() > 60:
        t_alloc = min(350.0, rem() * 0.3)
        h_new, c_new = active_set_descend(best_h, best_n, t_alloc, K_start=5, K_end=25, momentum=0.25)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    # Phase 4: Weighted top-K descent
    if rem() > 40:
        t_alloc = min(200.0, rem() * 0.2)
        h_new, c_new = weighted_topk_descent(best_h, best_n, t_alloc, K=12, beta_w=150.0)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    # Phase 5: Coordinate local search
    if rem() > 30:
        t_alloc = min(200.0, rem() * 0.2)
        h_new, c_new = coord_local_search(best_h, best_n, t_alloc)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    # Phase 6: Another round of active-set with fewer active constraints
    if rem() > 30:
        t_alloc = min(150.0, rem() * 0.2)
        h_new, c_new = active_set_descend(best_h, best_n, t_alloc, K_start=3, K_end=10, momentum=0.15)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    # Phase 7: Final SLSQP polish
    if rem() > 20:
        for beta in [500000, 2000000]:
            if rem() < 10:
                break
            h_new = slsqp_opt(best_h, beta, maxiter=100)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new

    # Phase 8: Final coord search
    if rem() > 20:
        t_alloc = min(100.0, rem() * 0.3)
        h_new, c_new = coord_local_search(best_h, best_n, t_alloc)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    # Final check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
