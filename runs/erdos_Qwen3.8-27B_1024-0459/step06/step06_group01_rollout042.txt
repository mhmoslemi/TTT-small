import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_GH = {}
_CONS_N = 0

    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def _inner_obj(x):
    n = _GH['n']
    w = _GH['w']
    active = _GH['active']
    dx = 2.0 / n
    corr = compute_corr(x)
    return float(np.dot(w, corr[active]) * dx)


def _inner_grad(x):
    n = _GH['n']
    w = _GH['w']
    active = _GH['active']
    dx = 2.0 / n
    grad = np.zeros(n)
    for pos, k in enumerate(active):
        wk = w[pos]
        lag = k - (n - 1)
        if lag >= 0:
            np_ = n - lag
            grad[lag:] += wk * (1.0 - x[:np_])
            grad[:np_] -= wk * x[lag:]
        else:
            a = -lag
            np_ = n - a
            grad[:np_] += wk * (1.0 - x[a:])
            grad[a:] -= wk * x[:np_]
    grad *= dx
    return grad


def _eq_fun(x):
    return np.sum(x) - _CONS_N / 2.0


def _eq_jac(x):
    return np.ones(_CONS_N)


def slsqp_inner(h, n, w, active, maxiter=80):

    _GH = {'n': n, 'w': w, 'active': active}
    _CONS_N = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    h0 = project_feasible(h, n)
    f0 = _inner_obj(h0)
    try:
        res = minimize(_inner_obj, h0, jac=_inner_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if res.fun < f0 - 1e-14:

    except Exception:
        pass
    return h0

    corr = compute_corr(x)
    m = np.max(corr)

def smooth_max_grad_fun(x):

            np_ = n - a

def minmax_optimize(h, n, time_limit, eta_init=0.5, inner_maxiter=60):
    dx = 2.0 / n
    n_lags = 2 * n - 1
    w = np.ones(n_lags) / n_lags
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    n_active = max(16, min(n_lags, n * 3 // 8))
    iteration = 0

    while time.time() - start < time_limit and remaining_time() > 15:
        corr = compute_corr(h)
        c = corr * dx
        c5_val = float(np.max(c))

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()

        m = float(np.max(c))
        if m <= 1e-12:
            break

        eta = eta_init * (0.9995 ** iteration)
        eta = max(eta, 0.05)
        w = w * np.exp(eta * (c - m) / m)
        wsum = np.sum(w)

            w = np.ones(n_lags) / n_lags
        else:
            w = w / wsum

        active = np.argsort(-w)[:n_active]

        h_new = slsqp_inner(h, n, w_act, active, maxiter=inner_maxiter)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = project_feasible(h_new, n)
        iteration += 1

    return best_h, best_c5


def projected_gradient_minmax(h, n, time_limit, eta_init=0.4, n_inner=15):
    dx = 2.0 / n
    n_lags = 2 * n - 1
    w = np.ones(n_lags) / n_lags
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    n_active = max(16, min(n_lags, n * 3 // 8))
    outer_it = 0

    while time.time() - start < time_limit and remaining_time() > 15:
        corr = compute_corr(h)
        c = corr * dx
        c5_val = float(np.max(c))

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()

        m = float(np.max(c))
        if m <= 1e-12:
            break

        eta = eta_init * (0.999 ** outer_it)
        eta = max(eta, 0.1)
        w = w * np.exp(eta * (c - m) / m)
        wsum = np.sum(w)
        if wsum < 1e-30:
            w = np.ones(n_lags) / n_lags
        else:
            w = w / wsum

        active = np.argsort(-w)[:n_active]
        w_act = w[active].copy()
        w_act = w_act / w_act.sum()

        lr = n / 4.0
        for _ in range(n_inner):
            grad = np.zeros(n)
            for pos, k in enumerate(active):
                wk = w_act[pos]
                lag = k - (n - 1)
                if lag >= 0:
                    np_ = n - lag
                    grad[lag:] += wk * (1.0 - h[:np_])
                    grad[:np_] -= wk * h[lag:]
                else:
                    a = -lag
                    np_ = n - a
                    grad[:np_] += wk * (1.0 - h[a:])
                    grad[a:] -= wk * h[:np_]
            grad *= dx
            gn = np.linalg.norm(grad)
            if gn < 1e-15:
                break
            step = lr * 0.5 * gn / (np.max(np.abs(grad)) + 1e-15)
            step = min(step, 0.3)
            h = project_feasible(h - step * grad, n)

        c5_after = compute_c5(h)
        if c5_after < best_c5 - 1e-15:
            best_c5 = c5_after
            best_h = h.copy()
        outer_it += 1

def coord_descent(h, n, max_sweeps=15, time_limit=None):
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    start = time.time()
    for sweep in range(max_sweeps):
        if time_limit is not None and time.time() - start > time_limit:
            break
        if remaining_time() < 10:
            break
        any_improve = False
        for j in range(n):
            old_val = h[j]
            curr_c5 = compute_c5(h)
            a, b = 0.0, 1.0
            best_v = old_val
            for _ in range(25):
                m1 = a + (b - a) * 0.381966
                m2 = a + (b - a) * 0.618034
                h[j] = m1
                c1 = compute_c5(h)
                h[j] = m2
                c2 = compute_c5(h)
                if c1 < c2:
                    b = m2
                    h[j] = m1
                    if c1 < curr_c5:
                        curr_c5 = c1
                        best_v = m1
                else:
                    a = m1
                    h[j] = m2
                    if c2 < curr_c5:
                        curr_c5 = c2
                        best_v = m2
            h[j] = best_v
            c_test = compute_c5(h)
            if c_test < best_c5 - 1e-14:
                best_c5 = c_test
                any_improve = True
        h = project_feasible(h, n)
        actual = compute_c5(h)
        if actual < best_c5:
            best_c5 = actual
        if not any_improve and sweep > 2:
            break
    return h, best_c5


def pairwise_guided(h, n, time_limit, rng):
    dx = 2.0 / n
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()

            best_h = h.copy()

        pressure = np.zeros(n)
        for k in top_idx:
            lag = k - (n - 1)
            if lag >= 0:
                np_ = n - lag
                pressure[lag:] += (1.0 - h[:np_])
                pressure[:np_] -= h[lag:]
            else:
                a = -lag
                np_ = n - a
                pressure[:np_] += (1.0 - h[a:])
                pressure[a:] -= h[:np_]

        donors = np.where((pressure < -0.005) & (h > 0.05))[0]
        receivers = np.where((pressure > 0.005) & (h < 0.95))[0]

        if len(donors) < 1 or len(receivers) < 1:
            break

        n_d = min(len(donors), 30)
        n_r = min(len(receivers), 30)
        donor_sel = donors[np.random.choice(len(donors), n_d, replace=False)]
        recv_sel = receivers[np.random.choice(len(receivers), n_r, replace=False)]

        improved = False
        for i in donor_sel:
            for j in recv_sel:
                delta_max = min(h[i], 1.0 - h[j])
                if delta_max < 0.01:
                    continue
                lo, hi = 0.0, delta_max
                best_d = 0.0
                best_c = c5_val
                h[i] -= lo
                h[j] += lo
                for _ in range(8):
                    mid = (lo + hi) * 0.5
                    h[i] -= mid
                    h[j] += mid
                    c_mid = compute_c5(h)
                    if c_mid < best_c:
                        best_c = c_mid
                        best_d = mid
                        lo = mid
                    else:
                        hi = mid
                h[i] = hi - h[i] + h[i]
                h[i] += hi
                h[j] -= hi
                if best_d > 0.005 and best_c < c5_val - 1e-13:
                    h[i] = best_h if False else h[i]
                    h[i] -= best_d
                    h[j] += best_d
                    h = np.clip(h, 0.0, 1.0)
                    h = project_feasible(h, n)
                    c5_val = compute_c5(h)
                    improved = True

        if not improved:
            break

    best_c5 = min(best_c5, compute_c5(best_h))
    return best_h, best_c5

def theta_morse_start(n):

        starts.append(theta_morse_start(n_c))

            starts.append(random_binary_start(n_c, rng))

        for start_h in starts[:4]:
            if remaining_time() < 10:
                break
            start_h = project_feasible(start_h, n_c)
            t_alloc = min(8.0, remaining_time() * 0.6)
            h_opt, c5_opt = minmax_optimize(start_h, n_c, t_alloc, eta_init=0.8, inner_maxiter=40)
            if c5_opt < best_c5 - 1e-10:
                # Upscale to track
                pass
            # Also try projected gradient
            if remaining_time() > 5:
                h_pg, c5_pg = projected_gradient_minmax(start_h, n_c, 3.0, eta_init=0.6, n_inner=10)
                if c5_pg < best_c5 - 1e-10:
                    pass

    # Phase 2: Progressive refinement
    resolutions = []
    if parent_h is not None:
        resolutions.append(parent_n)
    for r in [128, 256, 384, 512, 640, 768, 1024, 1280, 1536, 2048]:
        if r % 2 == 0 and r >= 64:
            resolutions.append(r)
    resolutions = sorted(set(resolutions))
    # Limit to a manageable set
    if len(resolutions) > 8:
        # Pick a subset that's well-spaced
        idx = np.linspace(0, len(resolutions) - 1, 8).astype(int)
        resolutions = [resolutions[i] for i in idx]
        resolutions = sorted(set(resolutions))

    # Assign time proportional to resolution
    total_res_time = min(700.0, remaining_time() * 0.7)
    res_sum = float(sum(resolutions))
    res_time = {r: total_res_time * r / res_sum for r in resolutions}
    # Cap max time per resolution
    for r in resolutions:
        res_time[r] = min(res_time[r], 250.0)

    cur_best_h = best_h
    cur_best_n = best_n

    for n_r in resolutions:
        if remaining_time() < 30:
            break
        t_alloc = min(res_time.get(n_r, 50.0), remaining_time() * 0.4)
        if t_alloc < 10:
            t_alloc = min(20.0, remaining_time() * 0.3)

        if t_alloc < 5:
            continue

        # Interpolate from current best
        h_interp = interp_to(cur_best_h, cur_best_n, n_r)
        h_interp = project_feasible(h_interp, n_r)

        # Also try a fresh random start if time allows
        h_random = random_binary_start(n_r, rng) if remaining_time() > 40 else None

        # Optimize interpolated start
        t1 = t_alloc * 0.6
        h_opt1, c5_opt1 = minmax_optimize(h_interp, n_r, t1, eta_init=0.5, inner_maxiter=50)

        # Try projected gradient for diversity
        h_opt2, c5_opt2 = projected_gradient_minmax(h_opt1, n_r, t_alloc * 0.15, eta_init=0.4, n_inner=12)

        # Try random start if available
        c5_best_local = min(c5_opt1, c5_opt2)
        h_best_local = h_opt1 if c5_opt1 <= c5_opt2 else h_opt2

        if h_random is not None and remaining_time() > 15:
            h_opt3, c5_opt3 = minmax_optimize(h_random, n_r, t_alloc * 0.15, eta_init=0.6, inner_maxiter=40)
            if c5_opt3 < c5_best_local:
                c5_best_local = c5_opt3
                h_best_local = h_opt3

        # Smooth max SLSQP polish
        if remaining_time() > 10:
            h_polish = smooth_max_slsqp(h_best_local, n_r, 5000, maxiter=200)
            c5_polish = compute_c5(h_polish)
            if c5_polish < c5_best_local:
                c5_best_local = c5_polish
                h_best_local = h_polish

        if c5_best_local < best_c5 - 1e-12:
            best_c5 = c5_best_local
            best_h = h_best_local.copy()
            best_n = n_r
            cur_best_h = h_best_local.copy()
            cur_best_n = n_r

    # Phase 3: Fine polishing at best resolution
    if remaining_time() > 30:
        # Multiple SLSQP with different betas
        for beta in [2000, 10000, 50000, 200000]:
            if remaining_time() < 15:
                break
            h_p = smooth_max_slsqp(best_h, best_n, beta, maxiter=200)
            c5_p = compute_c5(h_p)
            if c5_p < best_c5 - 1e-14:
                best_c5 = c5_p
                best_h = h_p.copy()

        # Coordinate descent
        if remaining_time() > 30:
            cd_time = min(60.0, remaining_time() * 0.3)
            h_cd, c5_cd = coord_descent(best_h, best_n, max_sweeps=12, time_limit=cd_time)
            if c5_cd < best_c5 - 1e-14:
                best_c5 = c5_cd
                best_h = h_cd.copy()

        # Pairwise guided search
        if remaining_time() > 20:
            pw_time = min(40.0, remaining_time() * 0.25)
            h_pw, c5_pw = pairwise_guided(best_h, best_n, pw_time, rng)
            if c5_pw < best_c5 - 1e-14:
                best_c5 = c5_pw
                best_h = h_pw.copy()

    # Phase 4: Try upscaling to a finer resolution if time remains
    if remaining_time() > 120:
        n_fine = best_n * 2
        n_fine = max(n_fine, 1024)
        n_fine = min(n_fine, 4096)
        if n_fine % 2 != 0:
            n_fine += 1
        h_up = interp_to(best_h, best_n, n_fine)
        h_up = project_feasible(h_up, n_fine)
        t_fine = min(60.0, remaining_time() * 0.3)
        h_up_opt, c5_up = minmax_optimize(h_up, n_fine, t_fine, eta_init=0.4, inner_maxiter=40)
        if remaining_time() > 20:
            h_up_opt = smooth_max_slsqp(h_up_opt, n_fine, 10000, maxiter=150)
            c5_up = compute_c5(h_up_opt)
        if c5_up < best_c5 - 1e-14:
            best_c5 = c5_up
            best_h = h_up_opt.copy()
            best_n = n_fine

    # Phase 5: Final polish
    if remaining_time() > 15:
        h_final = smooth_max_slsqp(best_h, best_n, 500000, maxiter=300)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-14:
            best_c5 = c5_final
            best_h = h_final.copy()

    if remaining_time() > 15:
        h_cd2, c5_cd2 = coord_descent(best_h, best_n, max_sweeps=8, time_limit=min(20.0, remaining_time() * 0.4))
        if c5_cd2 < best_c5 - 1e-14:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
