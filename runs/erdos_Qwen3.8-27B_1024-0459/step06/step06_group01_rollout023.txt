    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    z = beta * (corr - m)
    z = np.clip(z, -100.0, 100.0)
    ez = np.exp(z)
    return (m + np.log(np.sum(ez)) / beta) * dx

def smooth_max_gradient(x, beta, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    z = beta * (corr - m)
    z = np.clip(z, -100.0, 100.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):

Note: ∫h(x)(1-h(x+k))dx = ∫h(x)dx - ∫h(x)h(x+k)dx (over the overlap)

Actually more precisely, for the discrete case with dx:
corr[k] * dx = Σ_i h[i]*(1-h[i+k-offset])*dx (for valid i)

This is the integral approximated by the Riemann sum.

For k=0: corr[0] = Σ h[i](1-h[i]) = Σ h[i] - Σ h[i]²
Since Σh[i] = n/2, this is n/2 - Σh[i]².
To minimize this, we want to maximize Σh[i]², which means making h as extreme (0 or 1) as possible.
So for k=0, the minimum is achieved by binary h, giving corr[0] = n/2 - n/2 = 0.

For k=1: corr[1] = Σ_{i=0}^{n-2} h[i]*(1-h[i+1])
For binary h, this is the number of 1→0 transitions (from position i to i+1).

So for binary solutions, the problem becomes: minimize the maximum over k of the number of "mismatches" between h and its k-shift, scaled by dx.

This is related to theULONGEST common subsequence or string matching problems.

For the binary case with n points:
- k=0: 0 (always, for binary)
- k=1: number of 1→0 plus 0→1 transitions... wait, it's h[i]*(1-h[i+1]) which is 1 only when h[i]=1 and h[i+1]=0. So it counts 1→0 transitions.
- k=-1 (offset -1): h[i]*(1-h[i-1]) which is 1 only when h[i]=1 and h[i-1]=0. So it counts 0→1 transitions.

For a periodic binary string with period p, the transitions per period are fixed. The maximum correlation over all k is bounded by the structure.

def coord_optimal_j(h, n, j):
    """Find optimal h[j] given all other h fixed. Returns new value."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    old = h[j]
    
    # For each k (offset d), compute how corr changes with h[j] = old + δ
    # corr_d = Σ_i h[i](1-h[i+d])
    # Terms involving j: h[j](1-h[j+d]) if j+d valid, h[j-d](1-h[j]) if j-d valid
    # So: corr_d(δ) = corr_d + δ*(1-h[j+d]) - h[j-d]*δ  [for d≠0]
    # Or: corr_d(δ) = corr_d + δ - h[j-d]*δ - h[j]δ + h[j-d]h[j]... 
    # 
    # Actually let me be more careful:
    # corr_d = Σ_{valid i} h[i] * (1 - h[i+d])
    # 
    # For a specific j, h[j] appears:
    # 1. As h[i] when i = j: contributes h[j]*(1-h[j+d]) if j+d is valid
    # 2. As h[i+d] when i+d = j, i = j-d: contributes h[j-d]*(1-h[j]) if j-d is valid
    #
    # So: corr_d = (other terms) + h[j]*(1-h[j+d]) + h[j-d]*(1-h[j])
    # where the h[j] terms above are only included if indices are valid.
    #
    # If we change h[j] by δ (h[j] → h[j]+δ):
    # corr_d(δ) = (other terms) + (h[j]+δ)*(1-h[j+d]) + h[j-d]*(1-h[j]-δ)
    #            = corr_d + δ*(1-h[j+d]) - h[j-d]*δ
    #            = corr_d + δ*(1-h[j+d]-h[j-d])
    #
    # For d=0: corr_0 = Σ h[i](1-h[i]), h[j] contributes h[j]-h[j]²
    # corr_0(δ) = corr_0 + δ - (h[j]+δ)² + h[j]² = corr_0 + δ - 2h[j]δ - δ²
    #
    # So for d=0: quadratic in δ
    # For d≠0: linear in δ
    #
    # The C5(δ) = max_d corr_d(δ) * dx
    # Since most are linear and one is quadratic, the max is a convex piecewise function.
    # We can find the minimum by checking breakpoints.
    
    # For d=0:
    c0_slope = 1.0 - 2.0 * old  # coefficient of δ
    c0_quad = -1.0  # coefficient of δ²
    # corr_0(δ) = corr[0] + c0_slope*δ + c0_quad*δ²
    
    # For d≠0:
    # corr_d(δ) = corr[d] + slope_d * δ
    # where slope_d = (1-h[j+d]) - h[j-d] (with appropriate index handling)
    
    # Actually, let me just evaluate C5 at several δ values and pick the best.
    # δ ranges from -old to 1-old.
    n_evals = 30
    deltas = np.linspace(-old, 1.0 - old, n_evals)
    best_v = old
    best_c = compute_c5(h)
    for dv in deltas:
        h[j] = old + dv
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_v = old + dv
    h[j] = best_v
    return best_v, best_c

import numpy as np
from scipy.optimize import minimize

_G_BETA = 0.0
_G_MU = 0.0


def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=np.float64).flatten(), 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h + mid, 0.0, 1.0)) > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-15:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(free)
        if cnt > 0:
            h[free] += residual / cnt
            h = np.clip(h, 0.0, 1.0)
    return h


def _sm_obj(x):
    n, beta = _G_N, _G_BETA
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    return (m + np.log(np.sum(np.exp(z))) / beta) * dx


def _sm_grad(x):
    n, beta = _G_N, _G_BETA
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    w = ez / np.sum(ez)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            p = n - off
            g[off:] += wk * (1.0 - x[:p])
            g[:p] -= wk * x[off:]
        else:
            a = -off
            p = n - a
            g[:p] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:p]
    return g


def _pen_obj(x):
    c = np.sum(x) - _G_N / 2.0
    return _sm_obj(x) + _G_MU * c * c


def _pen_grad(x):
    c = np.sum(x) - _G_N / 2.0
    return _sm_grad(x) + 2.0 * _G_MU * c


def _eq_f(x):
    return np.sum(x) - _G_N / 2.0


def _eq_j(x):
    return np.ones(_G_N)


def lbfgsb_opt(h, n, beta, maxiter=3000):
    global _G_N, _G_BETA, _G_MU
    _G_N, _G_BETA, _G_MU = n, beta, 1e5
    h0 = project_feasible(h, n)
    try:
        res = minimize(_pen_obj, h0, jac=_pen_grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-9, 'maxls': 40})
        if np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def slsqp_opt(h, n, beta, maxiter=300):
    global _G_N, _G_BETA
    _G_N, _G_BETA = n, beta
    h0 = project_feasible(h, n)

        res = minimize(_sm_obj, h0, jac=_sm_grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _eq_f, 'jac': _eq_j}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)

def binary_sa(h, n, time_limit, rng):
    h = (h > 0.5).astype(np.float64).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        si = np.argsort(-h)
        h = np.zeros(n)
        h[si[:n_half]] = 1.0
    start = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    current_c5 = best_c5
    T = 0.3
    no_improve = 0
    while time.time() - start < time_limit and remaining_time() > 10:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            current_c5 = new_c5
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
        T *= 0.9997
        if no_improve > 4000:
            break
    return best_h, best_c5


def theta_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)

    target = n / 2.0

def binary_sa(h, n, time_limit, rng):
    h = (h > 0.5).astype(np.float64).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        si = np.argsort(-h)
        h = np.zeros(n)
        h[si[:n_half]] = 1.0
    start = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    current_c5 = best_c5
    T = 0.3
    no_improve = 0
    while time.time() - start < time_limit and remaining_time() > 10:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        i = int(ones[rng.randint(len(ones))])
        j = int(zeros[rng.randint(len(zeros))])
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5
        if delta < -1e-15 or (rng.rand() < np.exp(-delta / max(T, 1e-10))):
            current_c5 = new_c5
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
        T *= 0.9997
        if no_improve > 5000:
            break
    return best_h, best_c5

        h_t = project_feasible(h_t, n_t)

        for beta in [3000, 15000, 60000]:
            if remaining_time() < 60:
                break
            h_t = lbfgsb_opt(h_t, n_t, beta, maxiter=3000)

        if remaining_time() > 40:
            h_t, c5_t = nesterov_gd(h_t, n_t, 30000, lr0=0.15, momentum=0.92, maxit=1500)
        else:
            c5_t = compute_c5(h_t)

        if remaining_time() > 30:
            h_t, c5_t = coord_sweep_fast(h_t, n_t, max_sweeps=5)

        if c5_t < best_c5 - 1e-14:
            best_c5, best_h, best_n = c5_t, h_t.copy(), n_t

    # Phase 7: Return to best and intensify
    for beta in [50000, 200000]:
        if remaining_time() < 50:
            break
        h_new = lbfgsb_opt(best_h, best_n, beta, maxiter=4000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5, best_h = c5_new, h_new.copy()

    # Phase 8: Binary SA + continuous refinement
    if remaining_time() > 100:
        h_bin = (best_h > 0.5).astype(np.float64)
        n_half = best_n // 2
        if int(np.sum(h_bin)) != n_half:
            si = np.argsort(-h_bin)
            h_bin = np.zeros(best_n)
            h_bin[si[:n_half]] = 1.0

        sa_time = min(80, int(remaining_time() * 0.25))
        h_b, c5_b_bin = binary_sa(h_bin, best_n, sa_time, rng)

        h_b_cont = project_feasible(h_b, best_n)
        for beta in [10000, 50000]:
            if remaining_time() < 20:
                break
            h_b_cont = lbfgsb_opt(h_b_cont, best_n, beta, maxiter=2000)
        if remaining_time() > 20:
            h_b_cont, c5_b_cont = coord_sweep_fast(h_b_cont, best_n, max_sweeps=4)
        else:
            c5_b_cont = compute_c5(h_b_cont)

        if c5_b_cont < best_c5 - 1e-14:
            best_c5, best_h = c5_b_cont, h_b_cont.copy()
        if c5_b_bin < best_c5 - 1e-14:
            best_c5, best_h = c5_b_bin, h_b.copy()

    # Phase 9: Try different starting structures
    if remaining_time() > 80:
        for init_func, arg in [(periodic_init, 12), (theta_morse, None)]:
            if remaining_time() < 40:
                break
            if arg is not None:
                h_s = init_func(best_n, arg)
            else:
                h_s = init_func(best_n)
            h_s = lbfgsb_opt(h_s, best_n, 10000, maxiter=2000)
            h_s, c5_s = coord_sweep_fast(h_s, best_n, max_sweeps=3)
            if c5_s < best_c5 - 1e-14:
                best_c5, best_h = c5_s, h_s.copy()

    # Phase 10: Final SLSQP + coordinate descent polish
    if remaining_time() > 25 and best_n <= 800:
        h_fin = slsqp_opt(best_h, best_n, 300000, maxiter=200)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5 - 1e-14:
            best_c5, best_h = c5_fin, h_fin.copy()

    if remaining_time() > 15:
        h_cd, c5_cd = coord_sweep_fast(best_h, best_n, max_sweeps=4)
        if c5_cd < best_c5 - 1e-14:
            best_c5, best_h = c5_cd, h_cd.copy()

    if remaining_time() > 15:
        h_fin = lbfgsb_opt(best_h, best_n, 500000, maxiter=3000)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5 - 1e-14:
            best_c5, best_h = c5_fin, h_fin.copy()

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        ec = evaluate_erdos_solution(best_h)
        if ec is not None and np.isfinite(ec) and ec < best_c5:
            best_c5 = ec
    except Exception:
        pass

    return (best_h, best_c5, best_n)
