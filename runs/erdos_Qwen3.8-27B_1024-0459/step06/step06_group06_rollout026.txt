```python
def optimal_descent_direction(h, n, K=20):
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:K]
    # Try all equal weights first, then optimize
    grads = np.stack([grad_corr_k(h, k, n) for k in idx])
    # Optimize weights p to minimize ||sum p_k g_k||
    # This is: min_p ||G p|| s.t. p >= 0, sum(p) = 1
    # Can use scipy.optimize.minimize
    from scipy.optimize import minimize as sp_min
    def obj(p):
        return np.linalg.norm(grads @ p)
    cons = [{'type': 'eq', 'fun': lambda p: np.sum(p) - 1}]
    bounds = [(0, 1)] * K
    p0 = np.ones(K) / K
    res = sp_min(obj, p0, method='SLSQP', bounds=bounds, constraints=cons)
    p_star = res.x
    g_avg = grads @ p_star
    if np.linalg.norm(g_avg) < 1e-14:
        return np.zeros(n), 0.0
    d = -g_avg / np.linalg.norm(g_avg)
    return d, np.linalg.norm(g_avg)
```

```python
def grad_corr_k(h, k_idx, n):
    g = np.zeros(n)
    offset = k_idx - (n - 1)
    if offset >= 0:
        npairs = n - offset
        g[offset:n] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:n]
    else:
        a = -offset
        npairs = n - a
        g[:npairs] += (1.0 - h[a:n])
        g[a:n] -= h[:npairs]
    return g

def optimal_direction(h, n, K=20):
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:K]
    grads = []
    for k in idx:
        grads.append(grad_corr_k(h, k, n))
    grads = np.array(grads)
    def norm_obj(p):
        return np.linalg.norm(grads @ p)
    cons_wp = [(0.0, 1.0)] * K
    def sum_eq(p):
        return np.sum(p) - 1.0
    def sum_eq_jac(p):
        return np.ones(K)
    p0 = np.ones(K) / K
    try:
        res = minimize(norm_obj, p0, method='SLSQP', bounds=cons_wp,
                       constraints=[{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac}],
                       options={'maxiter': 200, 'ftol': 1e-12})
        p_star = res.x
        p_star = np.maximum(p_star, 0)
        p_star /= p_star.sum()
    except Exception:
        p_star = np.ones(K) / K
    g_avg = grads @ p_star
    gn = np.linalg.norm(g_avg)
    if gn < 1e-14:
        return np.zeros(n), 0.0
    d = -g_avg / gn
    return d, gn

            best_c5 = c5_sym
            best_h = h_sym.copy()

    # Phase 8: More optimal direction refinement
    if remaining_time() > 40:
        h_opt = best_h.copy()
        c5_opt = best_c5
        for it in range(200):
            rem = _BUDGET - (time.time() - _T0)
            if rem < 5:
                break
            d, gn = optimal_direction(h_opt, n_main, K=20)
            if gn < 1e-12:
                break
            for alpha in [0.03, 0.015, 0.01, 0.02, 0.005, 0.04]:
                h_try = project_feasible(h_opt + alpha * d, n_main)
                c_try = compute_c5(h_try)
                if c_try < c5_opt - 1e-15:
                    h_opt = h_try
                    c5_opt = c_try
                    break
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 9: Fine coordinate descent
    if remaining_time() > 50:
        h_cd, c5_cd = coordinate_descent_fine(best_h, n_main,
                                              min(50.0, remaining_time() * 0.5))
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 10: Another shake with smaller perturbation
    if remaining_time() > 40:
        h_shake, c5_shake = shake_reoptimize(best_h, n_main, rng,
                                             n_shakes=3, shake_std=0.01,
                                             budget_each=12.0)
        if c5_shake < best_c5:
            best_c5 = c5_shake
            best_h = h_shake.copy()

    # Phase 11: Final SLSQP with very high beta
    for beta in [15000, 30000]:
        if remaining_time() < 15:
            break
        h = slsqp_smooth(best_h, n_main, beta=beta, K=30, maxiter=500)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 12: Try n=4096 if time permits
    if remaining_time() > 100:
        n_hv = 4096
        h_hv = interp_to(best_h, best_n, n_hv)
        h_hv = project_feasible(h_hv, n_hv)
        c5_hv = compute_c5(h_hv)
        if c5_hv < best_c5:
            h_hv = lbfgsb_smooth(h_hv, n_hv, beta=3000, K=60, penalty=500, maxiter=1500)
            c5_hv = compute_c5(h_hv)
            h_hv = slsqp_smooth(h_hv, n_hv, beta=8000, K=40, maxiter=300)
            c5_hv = compute_c5(h_hv)
            if c5_hv < best_c5:
                best_c5 = c5_hv
                best_h = h_hv.copy()
                best_n = n_hv

    # Final: optimal direction fine-tuning
    if remaining_time() > 20:
        h_final = best_h.copy()
        n_final = best_n
        c5_final = best_c5
        for it in range(100):
            rem = _BUDGET - (time.time() - _T0)
            if rem < 3:
                break
            d, gn = optimal_direction(h_final, n_final, K=25)
            if gn < 1e-12:
                break
            for alpha in [0.01, 0.005, 0.002, 0.02, 0.008]:
                h_try = project_feasible(h_final + alpha * d, n_final)
                c_try = compute_c5(h_try)
                if c_try < c5_final - 1e-16:
                    h_final = h_try
                    c5_final = c_try
                    break
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Final feasibility and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
```
