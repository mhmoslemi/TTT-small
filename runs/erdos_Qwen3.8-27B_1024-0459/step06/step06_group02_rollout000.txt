    if lag >= 0:
        npairs = n - lag
        if npairs > 0:
            g[:npairs] += (1.0 - h[lag:])
            g[lag:] -= h[:npairs]
    else:
        a = -lag
        npairs = n - a
        if npairs > 0:
            g[a:] += (1.0 - h[:npairs])
            g[:npairs] -= h[a:]
    return g


def smooth_max_objective(h, n, beta, top_k=300):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = len(corr)
    if m <= top_k:
        top_idx = np.arange(m)
    else:
        top_idx = np.argpartition(-corr, top_k)[:top_k]
    c_top = corr[top_idx]
    c_max = c_top.max()
    shifted = beta * (c_top - c_max)
    clipped = np.clip(shifted, -50.0, 50.0)
    ex = np.exp(clipped)
    s = ex.sum()
    if s < 1e-30:
        return c_max * dx, np.zeros(n)
    logsumexp = c_max + np.log(s) / beta
    obj = logsumexp * dx
    w = ex / s
    mask = w > 1e-12
    idx_sel = top_idx[mask]
    w_sel = w[mask]
    g = np.zeros(n)
    for j in range(len(idx_sel)):
        if w_sel[j] > 1e-12:
            g += w_sel[j] * dx * grad_corr_k(h, int(idx_sel[j]), n)
    return obj, g


def nesterov_projected_gradient(h, n, beta, n_iter=5000, time_limit=100.0, top_k=300):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    y = h.copy()
    t = 1.0
    start = time.time()
    
    for it in range(n_iter):
        if time.time() - start > time_limit:
            break
        obj, g = smooth_max_objective(y, n, beta, top_k)
        step = 2.0 / (1.0 + t)
        y_new = y - step * g
        h_new = project_feasible(y_new, n)
        t = (1.0 + math.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = h_new + (t - 1.0) / t * (h_new - y)
        y = np.clip(y, 0.0, 1.0)
        
        if it % 50 == 0:
            c_cur = compute_c5(h_new)
            if c_cur < best_c5:
                best_c5 = c_cur
                best_h = h_new.copy()
    
    c_final = compute_c5(best_h)
    return best_h, best_c5


def slsqp_smooth_max(h, n, beta, maxiter=500, ftol=1e-16, top_k=200):
    def obj(x):
        val, _ = smooth_max_objective(x, n, beta, top_k)
        return val

    def grad(x):
        _, g = smooth_max_objective(x, n, beta, top_k)
        return g

    def constraint_fun(x):
        return np.sum(x) - n / 2.0

    def constraint_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]

    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            if compute_c5(h_new) <= compute_c5(h) + 1e-10:
                return h_new
    except Exception:
        pass
    return h.copy()


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def binary_init(n, seed=0):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    half = n // 2
    idx = rng.choice(n, half, replace=False)
    h[idx] = 1.0
    return h


def periodic_init(n, period=8, seed=0):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    per_pattern = []
    for p in range(period):
        if p < period // 2:
            per_pattern.append(1.0)
        else:
            per_pattern.append(0.0)
    rng.shuffle(per_pattern)
    for i in range(n):
        h[i] = per_pattern[i % period]
    target = n / 2.0
    diff = target - h.sum()
    fractional = int(round(diff))
    one_positions = np.where(h == 1.0)[0]
    zero_positions = np.where(h == 0.0)[0]
    if fractional > 0:
        for i in range(min(fractional, len(zero_positions))):
            h[zero_positions[i]] = 1.0
    elif fractional < 0:
        for i in range(min(-fractional, len(one_positions))):
            h[one_positions[i]] = 0.0
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)

    # Initialize from parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        parent_h = None

    if parent_h is not None:
        n_start = len(parent_h)
        best_h = parent_h.copy()
        best_c5 = compute_c5(best_h)
        best_n = n_start
    else:
        n_start = 768
        best_h = periodic_init(n_start, period=8, seed=seed)
        best_c5 = compute_c5(best_h)
        best_n = n_start

    # Phase 1: Interpolate to n=1024 and do initial gradient descent
    n1 = 1024
    h1 = interp_to(best_h, best_n, n1)
    h1 = project_feasible(h1, n1)
    c1 = compute_c5(h1)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h1.copy()
        best_n = n1
    else:
        # Keep parent resolution if interpolation doesn't help
        pass

    # SLSQP at n=1024 with moderate beta
    for beta in [500, 2000, 8000]:
        if remaining() < 20:
            break
        h_s = slsqp_smooth_max(h1, n1, beta, maxiter=300, top_k=200)
        c_s = compute_c5(h_s)
        if c_s < best_c5:
            best_c5 = c_s
            best_h = h_s.copy()
            best_n = n1
            h1 = h_s.copy()

    # Phase 2: Nesterov projected gradient at n=1024
    if remaining() > 60:
        for beta in [2000, 5000, 15000, 50000]:
            if remaining() < 30:
                break
            h_ng, c_ng = nesterov_projected_gradient(best_h, best_n, beta,
                                                      n_iter=3000, time_limit=40, top_k=300)
            if c_ng < best_c5:
                best_c5 = c_ng
                best_h = h_ng.copy()

    # Phase 3: SLSQP refinement at higher beta
    for beta in [15000, 50000, 150000]:
        if remaining() < 20:
            break
        h_s = slsqp_smooth_max(best_h, best_n, beta, maxiter=500, top_k=300)
        c_s = compute_c5(h_s)
        if c_s < best_c5:
            best_c5 = c_s
            best_h = h_s.copy()
            best_n = len(h_s)

    # Phase 4: Try n=2048
    n2 = 2048
    if remaining() > 120:
        h2 = interp_to(best_h, best_n, n2)
        h2 = project_feasible(h2, n2)
        c2 = compute_c5(h2)

        # SLSQP at n=2048 (fewer iterations since it's larger)
        for beta in [2000, 8000, 30000]:
            if remaining() < 30:
                break
            h2s = slsqp_smooth_max(h2, n2, beta, maxiter=200, top_k=250)
            c2s = compute_c5(h2s)
            if c2s < c2:
                c2 = c2s
                h2 = h2s

        # Nesterov at n=2048
        if remaining() > 60:
            for beta in [5000, 20000, 80000]:
                if remaining() < 30:
                    break
                h2g, c2g = nesterov_projected_gradient(h2, n2, beta,
                                                        n_iter=2000, time_limit=30, top_k=250)
                if c2g < c2:
                    c2 = c2g
                    h2 = h2g

        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()
            best_n = n2

    # Phase 5: Multi-start exploration at best resolution
    n_explore = best_n
    if remaining() > 80:
        # Try periodic initialization
        for period in [6, 10, 12, 16]:
            if remaining() < 20:
                break
            h_p = periodic_init(n_explore, period=period, seed=seed + period)
            h_p = project_feasible(h_p, n_explore)
            for beta in [5000, 30000]:
                if remaining() < 15:
                    break
                h_ps = slsqp_smooth_max(h_p, n_explore, beta, maxiter=200, top_k=250)
                c_ps = compute_c5(h_ps)
                if c_ps < best_c5:
                    best_c5 = c_ps
                    best_h = h_ps.copy()
                    best_n = n_explore

        # Try binary + smooth
        h_b = binary_init(n_explore, seed=seed + 100)
        # Significant smoothing
        kernel = np.ones(7) / 7.0
        h_b = np.convolve(h_b, kernel, mode='same')
        h_b = project_feasible(h_b, n_explore)
        for beta in [5000, 30000]:
            if remaining() < 15:
                break
            h_bs = slsqp_smooth_max(h_b, n_explore, beta, maxiter=200, top_k=250)
            c_bs = compute_c5(h_bs)
            if c_bs < best_c5:
                best_c5 = c_bs
                best_h = h_bs.copy()
                best_n = n_explore

    # Phase 6: Final SLSQP polishing
    for beta in [50000, 200000, 500000]:
        if remaining() < 15:
            break
        h_f = slsqp_smooth_max(best_h, best_n, beta, maxiter=600, top_k=300)
        c_f = compute_c5(h_f)
        if c_f < best_c5:
            best_c5 = c_f
            best_h = h_f.copy()

    # Phase 7: Very fine Nesterov at high beta
    if remaining() > 30:
        for beta in [100000, 500000]:
            if remaining() < 20:
                break
            h_fin, c_fin = nesterov_projected_gradient(best_h, best_n, beta,
                                                        n_iter=5000, time_limit=25, top_k=400)
            if c_fin < best_c5:
                best_c5 = c_fin
                best_h = h_fin.copy()

    # Phase 8: Coordinate descent polish
    if remaining() > 20:
        h_cd = best_h.copy()
        n_cd = best_n
        target_sum = n_cd / 2.0
        start_cd = time.time()
        improved_any = True
        while improved_any and time.time() - start_cd < min(remaining() - 5, 30):
            improved_any = False
            for i in range(n_cd):
                if time.time() - start_cd > min(remaining() - 5, 30):
                    break
                old = h_cd[i]
                for delta in [-0.02, -0.01, -0.005, -0.002, 0.002, 0.005, 0.01, 0.02]:
                    new_val = old + delta
                    if new_val < 0.0 or new_val > 1.0:
                        continue
                    h_try = h_cd.copy()
                    h_try[i] = new_val
                    # Fix sum via neighbor
                    diff = target_sum - h_try.sum()
                    j = (i + 1) % n_cd
                    h_try[j] = h_try[j] - diff
                    if h_try[j] < -0.01 or h_try[j] > 1.01:
                        continue
                    h_try = project_feasible(h_try, n_cd)
                    c_try = compute_c5(h_try)
                    if c_try < best_c5 - 1e-10:
                        best_c5 = c_try
                        best_h = h_try.copy()
                        best_n = n_cd
                        h_cd = h_try.copy()
                        improved_any = True
                        break

    # Ensure final feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Validate with evaluator
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
