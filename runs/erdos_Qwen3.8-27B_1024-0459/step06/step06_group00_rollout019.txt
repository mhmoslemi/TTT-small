```python
def compute_smooth_max_topk(h, k, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')  # O(n²)
    idx = np.argpartition(corr, -k)[-k:]  # top-k indices
    topk_vals = corr[idx]
    # Smooth max of top-k
    m = np.max(topk_vals)
    z = beta * (topk_vals - m)
    ez = np.exp(z - np.max(z))  # numerical stability
    sm = (m + np.log(np.sum(ez)) / beta) * dx
    weights = ez / np.sum(ez)
    return sm, idx, weights

def compute_grad_topk(h, idx, weights):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for pos, k in enumerate(idx):
        wk = weights[pos] * dx
        off = k - (n-1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - h[a:])
            g[a:] -= wk * h[:np_]
    return g
```

```python
        s = np.sum(x) - n / 2.0
        return cm + penalty_lambda * s * s
    
    def grad(x):
        cm, g, _, _ = smooth_max_topk(x, k, beta)
        s = np.sum(x) - n / 2.0
        g += 2 * penalty_lambda * s
        return g
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(fun, h, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
    return project_feasible(res.x, n)

def grad_single_shift(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        np_ = n - off
        g[off:] += (1.0 - h[:np_])
        g[:np_] -= h[off:]
    else:
        a = -off
        np_ = n - a
        g[:np_] += (1.0 - h[a:])
        g[a:] -= h[:np_]
    return g

def top_k_grad_descent(h, n, k, lr_schedule='anneal', max_iters=3000, time_limit=60):
    """Direct gradient descent on top-k correlations with line search."""
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = 0.01
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if remaining_time() < 5:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argpartition(corr, -k)[-k:]
        # Equal-weighted gradient of top-k
        g = np.zeros(n)
        for ki in idx:
            g += grad_single_shift(h, ki, n)
        g /= k
        
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        
        # Line search
        improved = False
        steps = [lr, lr*0.5, lr*0.2, lr*2, lr*0.1, 0.005, 0.01, 0.02, 0.05]
        for step in steps:
            ht = project_feasible(h - step * g, n)
            c5_new = compute_c5(ht)
            if c5_new < best_c5 - 1e-15:
                h = ht
                best_c5 = c5_new
                best_h = h.copy()
                improved = True
                break
        
        if not improved:
            lr *= 0.8
            if lr < 1e-5:
                break
        else:
            lr = min(lr * 1.05, 0.1)
    
    return best_h, best_c5

def binary_block_swap_sa(h, n, time_limit=120, rng=None, max_block_len=8):
    """Binary simulated annealing with block swaps."""
    if rng is None:
        rng = np.random.RandomState(42)
    
    h_bin = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        sort_idx = np.argsort(-h_bin)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0
    
    dx = 2.0 / n
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    curr_c5 = best_c5
    start = time.time()
    
    T_start = 0.05
    T_end = 0.0001
    no_improve = 0
    total_evals = 0
    
    while time.time() - start < time_limit and remaining_time() > 5:
        T = T_start * (T_end / T_start) ** (total_evals / 5000.0)
        T = max(T, T_end)
        
        # Choose block length
        r = rng.random()
        if r < 0.5:
            block_len = 1
        elif r < 0.7:
            block_len = 2
        elif r < 0.85:
            block_len = rng.randint(3, 5)
        else:
            block_len = rng.randint(5, max_block_len + 1)
        
        # Find a block of 1s and a block of 0s of length block_len
        # For block_len=1, this is just a single swap
        valid = False
        for _ in range(20):  # Try 20 times to find valid blocks
            i = rng.randint(0, n - block_len + 1)
            j = rng.randint(0, n - block_len + 1)
            if np.all(h_bin[i:i+block_len] == 1.0) and np.all(h_bin[j:j+block_len] == 0.0):
                valid = True
                break
            elif np.all(h_bin[i:i+block_len] == 0.0) and np.all(h_bin[j:j+block_len] == 1.0):
                valid = True
                break
        
        if not valid:
            continue
        
        old_block1 = h_bin[i:i+block_len].copy()
        old_block2 = h_bin[j:j+block_len].copy()
        
        # Swap
        h_bin[i:i+block_len] = old_block2
        h_bin[j:j+block_len] = old_block1
        
        new_c5 = compute_c5(h_bin)
        delta = new_c5 - curr_c5
        total_evals += 1
        
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_h = h_bin.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Revert
            h_bin[i:i+block_len] = old_block1
            h_bin[j:j+block_len] = old_block2
            no_improve += 1
        
        if no_improve > 3000:

                a = rng.randint(n)
                b = rng.randint(n)
                if h_bin[a] != h_bin[b]:
                    h_bin[a], h_bin[b] = h_bin[b], h_bin[a]
            best_c5 = min(best_c5, compute_c5(h_bin))
            best_h = h_bin.copy() if compute_c5(h_bin) < best_c5 else best_h
            curr_c5 = compute_c5(h_bin)
            no_improve = 0
    
    return best_h, best_c5

def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)

def coordinate_min(h, n, max_sweeps=50):
    """Coordinate-wise minimization with constraint projection."""
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        improved = False
        for i in range(n):
            old = h[i]
            # Try: 0, 1, and neighbors
            cands = [0.0, 1.0]
            if 0.1 < old < 0.9:
                cands.extend([old-0.1, old+0.1, 0.3, 0.7])
            cands = [c for c in cands if 0<=c<=1 and abs(c-old)>1e-8]
            
            best_cand = old
            best_c = best_c5
            for c in cands:
                h[i] = c
                h = project_feasible(h, n)
                c5_v = compute_c5(h)
                if c5_v < best_c - 1e-13:
                    best_c = c5_v
                    best_cand = c
                    improved = True
            h[i] = best_cand
            h = project_feasible(h, n)
        if not improved:
            break
    return h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_CUR_N = 0
_K_TOP = 8
_BETA_SM = 5000.0


def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def corr_and_grad_full(h, n, beta):
    """Compute smooth max value and full gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    z = beta * (corr - m)
    z -= np.max(z)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    sm_val = (m + np.max(z) + np.log(S)) / beta * dx
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-13:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - h[a:])
            g[a:] -= wk * h[:np_]
    return sm_val, g, corr


def lbfgs_smooth(h, n, k, beta, maxiter=400, penalty=1e5):
    """L-BFGS-B with smooth max (top-k via heavy beta) + penalty constraint."""
    global _CUR_N, _K_TOP, _BETA_SM
    _CUR_N = n
    _K_TOP = k
    _BETA_SM = max(beta, 50000)

    def fun(x):
        cm, _, corr = corr_and_grad_full(x, n, _BETA_SM)
        s = np.sum(x) - n / 2.0
        return cm + penalty * s * s

    def grad(x):
        cm, g, corr = corr_and_grad_full(x, n, _BETA_SM)
        s = np.sum(x) - n / 2.0
        g = g + 2.0 * penalty * s
        return g

    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h0, jac=grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter,
                                               'ftol': 1e-14, 'gtol': 1e-8,
                                               'maxls': 30})
        return project_feasible(res.x, n)
    except Exception:
        return h0


def grad_shift(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        np_ = n - off
        g[off:] += (1.0 - h[:np_])
        g[:np_] -= h[off:]
    else:
        a = -off
        np_ = n - a
        g[:np_] += (1.0 - h[a:])
        g[a:] -= h[:np_]
    return g


def topk_grad_descent(h, n, k, max_iters=6000, time_limit=80):
    """Gradient descent on top-k correlations with adaptive step."""
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = 0.02
    stall = 0

    for it in range(max_iters):
        if time.time() - start > time_limit or remaining_time() < 5:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argpartition(corr, -k)[-k:]
        g = np.zeros(n)
        for ki in idx:
            g += grad_shift(h, ki, n)
        g /= k
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        improved = False
        for step in [lr, lr * 0.5, lr * 2, lr * 0.2, 0.005, 0.01, 0.03, 0.06]:
            ht = project_feasible(h - step * g, n)
            c5n = compute_c5(ht)
            if c5n < best_c5 - 1e-15:
                h = ht
                best_c5 = c5n
                best_h = h.copy()
                improved = True
                break
        if improved:
            lr = min(lr * 1.03, 0.08)
            stall = 0
        else:
            lr *= 0.85
            stall += 1
            if stall > 80:
                break
            if stall % 20 == 0:
                pert = np.random.randn(n) * 0.003
                ht = project_feasible(h + pert, n)
                if compute_c5(ht) < best_c5:
                    h = ht
                    best_c5 = compute_c5(h)
                    best_h = h.copy()
                    stall = 0
    return best_h, best_c5


def binary_block_sa(h_init, n, time_limit=150, rng=None, max_blk=10, T_mult=0.7):
    """Binary SA with variable-length block swaps."""
    if rng is None:
        rng = np.random.RandomState(42)
    h_bin = (h_init > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        idx = np.argsort(-h_bin)
        h_bin = np.zeros(n)
        h_bin[idx[:n_half]] = 1.0

    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    curr_c5 = best_c5
    start = time.time()
    T = 0.03
    no_imp = 0
    eval_count = 0
    ones_list = np.where(h_bin == 1.0)[0]
    zeros_list = np.where(h_bin == 0.0)[0]

    while time.time() - start < time_limit and remaining_time() > 5:
        T *= T_mult ** 0.01
        if T < 0.00005:
            T = 0.00005

        r = rng.random()
        if r < 0.45:
            bl = 1
        elif r < 0.65:
            bl = 2
        elif r < 0.80:
            bl = rng.randint(3, 6)
        else:
            bl = rng.randint(6, max_blk + 1)

        valid = False
        for _ in range(30):
            i = rng.randint(0, max(1, n - bl))
            j = rng.randint(0, max(1, n - bl))
            if np.all(h_bin[i:i+bl] == 1.0) and np.all(h_bin[j:j+bl] == 0.0):
                valid = True
                break
            elif np.all(h_bin[i:i+bl] == 0.0) and np.all(h_bin[j:j+bl] == 1.0):
                valid = True
                break

        if not valid:
            continue

        old1 = h_bin[i:i+bl].copy()
        old2 = h_bin[j:j+bl].copy()
        h_bin[i:i+bl] = old2
        h_bin[j:j+bl] = old1

        new_c5 = compute_c5(h_bin)
        delta = new_c5 - curr_c5
        eval_count += 1

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            curr_c5 = new_c5
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_h = h_bin.copy()
                no_imp = 0
            else:
                no_imp += 1
            ones_list = np.where(h_bin == 1.0)[0]
            zeros_list = np.where(h_bin == 0.0)[0]
        else:
            h_bin[i:i+bl] = old1
            h_bin[j:j+bl] = old2
            no_imp += 1

        if no_imp > 4000:
            for _ in range(15):
                a = rng.randint(n)
                b = rng.randint(n)
                if h_bin[a] != h_bin[b]:
                    h_bin[a], h_bin[b] = h_bin[b], h_bin[a]
            curr_c5 = compute_c5(h_bin)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h_bin.copy()
            ones_list = np.where(h_bin == 1.0)[0]
            zeros_list = np.where(h_bin == 0.0)[0]
            no_imp = 0

    return best_h, best_c5


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def coordinate_min(h, n, max_sweeps=40):
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    for sw in range(max_sweps):
        if remaining_time() < 5:
            break
        imp = False
        for i in range(n):
            old = h[i]
            cands = [0.0, 1.0]
            if 0.05 < old < 0.95:
                cands += [old - 0.1, old + 0.1, 0.25, 0.75]
            cands = [c for c in cands if 0 <= c <= 1 and abs(c - old) > 1e-8]
            bc = old
            bc5 = best_c5
            for c in cands:
                h[i] = c
                h = project_feasible(h, n)
                v = compute_c5(h)
                if v < bc5 - 1e-13:
                    bc5 = v
                    bc = c
                    imp = True
            h[i] = bc
            h = project_feasible(h, n)
            if bc5 < best_c5:
                best_c5 = bc5
        if not imp:
            break
    return h, best_c5


def multi_scale_optimize(h, n, time_limit=120):
    """L-BFGS with increasing beta, then top-k grad descent."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    st = time.time()

    for beta in [5000, 50000, 500000, 3000000]:
        if time.time() - st > time_limit * 0.6 or remaining_time() < 10:
            break
        h_new = lbfgs_smooth(best_h, n, 8, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()

    if time.time() - st < time_limit * 0.5 and remaining_time() > 10:
        h_g, c5_g = topk_grad_descent(best_h, n, k=10,
                                      max_iters=3000,
                                      time_limit=int((time_limit - (time.time() - st)) * 0.7))
        if c5_g < best_c5 - 1e-14:
            best_c5 = c5_g
            best_h = h_g.copy()

    return best_h, best_c5
```
