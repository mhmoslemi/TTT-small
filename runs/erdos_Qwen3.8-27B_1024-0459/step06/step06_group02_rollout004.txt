    g = np.zeros(n)
    for i, k in enumerate(idx):
        if weights[i] > 1e-10:
            g += weights[i] * grad_corr_k(h, k, n)
    return g * (2.0 / n)


def slsqp_optimize(h, n, beta, maxiter=300, ftol=1e-18):
    dx = 2.0 / n

    def obj(x):
        return smooth_max_val(x, n, beta)

    def grad(x):
        return smooth_max_grad(x, n, beta)

    def con_fun(x):
        return np.sum(x) - n / 2.0

    def con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        if res.x is not None and res.x is not h:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()


def subgrad_descent(h, n, steps=300, K=5, sharpness=20.0):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(steps):
        if time_left() < 2.0:
            break
        g = exact_subgrad(h, n, K, sharpness)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        t = 0.08 / (1.0 + 0.01 * it)
        h_new = h - t * g / gn
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def perturb_h(h, n, rng, scale=0.02):
    h_p = h.copy()
    x = np.arange(n) * 2.0 / n
    n_freqs = rng.randint(2, 5)
    for _ in range(n_freqs):
        freq = rng.randint(1, 8)
        phase = rng.uniform(0, 2 * np.pi)
        amp = scale * 0.4 * rng.uniform(0.5, 1.5)
        h_p += amp * np.sin(2 * np.pi * freq * x + phase)
    h_p += scale * 0.2 * rng.randn(n)
    return project_feasible(h_p, n)


def perturb_h_small(h, n, rng, scale=0.005):
    h_p = h.copy()
    h_p += scale * rng.randn(n)
    return project_feasible(h_p, n)


def nesterov_subgrad(h, n, steps=400, K=5, sharpness=30.0):
    h = h.copy().astype(float)
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(steps):
        if time_left() < 2.0:
            break
        t_k = 2.0 / (it + 2.0)
        h_y = h + (t_k - 1.0) / t_k * v
        h_y = np.clip(h_y, 0.0, 1.0)
        g = exact_subgrad(h_y, n, K, sharpness)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        h_new = h_y - t_k * g / gn
        v = h_new - h
        h = np.clip(h_new, 0.0, 1.0)
        h = project_feasible(h, n)
        c_new = compute_c5(h)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h.copy()
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _DEADLINE
    _DEADLINE = time.time() + budget_s
    rng = np.random.RandomState(seed)

    # Initialize from parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        n = 512
        parent_h = np.zeros(n)
        parent_h[:n // 2] = 1.0

    n = len(parent_h)
    best_h = project_feasible(parent_h, n)
    best_c5 = compute_c5(best_h)
    best_n = n

    # Phase 1: SLSQP polish at parent resolution
    for beta in [500, 2000, 10000, 50000]:
        if time_left() < 20:
            break
        h_try = slsqp_optimize(best_h, n, beta, maxiter=200)
        c_try = compute_c5(h_try)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try.copy()

    # Phase 2: Nesterov subgradient at parent resolution
    if time_left() > 80:
        h_ns, c_ns = nesterov_subgrad(best_h, n, steps=400, K=5, sharpness=30.0)
        if c_ns < best_c5:
            best_c5 = c_ns
            best_h = h_ns.copy()
        # Polish
        if time_left() > 20:
            h_pol = slsqp_optimize(best_h, n, 20000, maxiter=300)
            c_pol = compute_c5(h_pol)
            if c_pol < best_c5:
                best_c5 = c_pol
                best_h = h_pol.copy()

    # Phase 3: Upsample to higher resolution and optimize
    resolutions = [1024, 1536, 2048]
    resolutions = [r for r in resolutions if r > n and r <= 3072]
    if n > 1024:
        resolutions = [r for r in resolutions if r > n]
    if not resolutions:
        resolutions = [n * 2]
    resolutions = sorted(set(resolutions))

    for n_t in resolutions[:3]:
        if time_left() < 120:
            break

        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        c_inter = compute_c5(h_feas)

        # SLSQP at new resolution
        for beta in [300, 1000, 5000, 20000]:
            if time_left() < 25:
                break
            h_new = slsqp_optimize(h_feas, n_t, beta, maxiter=200)
            c_new = compute_c5(h_new)
            if c_new < c_inter:
                c_inter = c_new
                h_feas = h_new

        if c_inter < best_c5:
            best_c5 = c_inter
            best_h = h_feas.copy()
            best_n = n_t

        # Nesterov subgradient at new resolution
        if time_left() > 60:
            h_ns2, c_ns2 = nesterov_subgrad(h_feas, n_t, steps=300, K=4, sharpness=25.0)
            if c_ns2 < best_c5:
                best_c5 = c_ns2
                best_h = h_ns2.copy()
                best_n = n_t

        # Final SLSQP polish at new resolution
        if time_left() > 20:
            h_pol2 = slsqp_optimize(best_h, best_n, 50000, maxiter=300)
            c_pol2 = compute_c5(h_pol2)
            if c_pol2 < best_c5:
                best_c5 = c_pol2
                best_h = h_pol2.copy()

    # Phase 4: Perturbation cycles at best resolution
    for p in range(12):
        if time_left() < 30:
            break
        if p < 4:
            h_pert = perturb_h(best_h, best_n, rng, scale=0.015 + 0.01 * p)
        else:
            h_pert = perturb_h_small(best_h, best_n, rng, scale=0.005)

        c_pert = compute_c5(h_pert)
        # Try SLSQP from perturbation
        for beta in [5000, 30000]:
            if time_left() < 15:
                break
            h_s = slsqp_optimize(h_pert, best_n, beta, maxiter=200)
            c_s = compute_c5(h_s)
            if c_s < best_c5:
                best_c5 = c_s
                best_h = h_s.copy()

    # Phase 5: High-beta SLSQP for very fine polishing
    for beta in [100000, 300000, 1000000]:
        if time_left() < 15:
            break
        h_new = slsqp_optimize(best_h, best_n, beta, maxiter=500)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Phase 6: Final Nesterov + SLSQP
    if time_left() > 40:
        h_ns3, c_ns3 = nesterov_subgrad(best_h, best_n, steps=300, K=3, sharpness=50.0)
        if c_ns3 < best_c5:
            best_c5 = c_ns3
            best_h = h_ns3.copy()
    if time_left() > 15:
        h_fin = slsqp_optimize(best_h, best_n, 500000, maxiter=600, ftol=1e-20)
        c_fin = compute_c5(h_fin)
        if c_fin < best_c5:
            best_c5 = c_fin
            best_h = h_fin.copy()

    # Phase 7: Last small perturbations + polish
    for p in range(6):
        if time_left() < 12:
            break
        h_sp = perturb_h_small(best_h, best_n, rng, scale=0.002)
        h_sp = slsqp_optimize(h_sp, best_n, 200000, maxiter=200)
        c_sp = compute_c5(h_sp)
        if c_sp < best_c5:
            best_c5 = c_sp
            best_h = h_sp.copy()

    # Final feasibility and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
