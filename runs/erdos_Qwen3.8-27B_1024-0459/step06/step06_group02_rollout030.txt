def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)

        if cnt > 0:
            h[mask] += diff / cnt
        else:

        h = np.clip(h, 0.0, 1.0)

def grad_corr_k(h, k, n):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        if npairs > 0:
            g[offset:] += (1.0 - h[:npairs])
            g[:npairs] -= h[offset:]
    else:
        a = -offset
        npairs = n - a
        if npairs > 0:
            g[:npairs] += (1.0 - h[a:])
            g[a:] -= h[:npairs]
    return g


def cutting_plane_lp(h, n, top_k):
    dx = 2.0 / n
    corr = compute_corr(h)
    f_vals = corr * dx

    K = min(top_k, len(f_vals))
    top_idx = np.argsort(-f_vals)[:K]

    grads = np.zeros((K, n))
    for i, k in enumerate(top_idx):
        grads[i] = grad_corr_k(h, k, n)

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    A1 = np.hstack([grads, -np.ones((K, 1))])
    b1 = -f_vals[top_idx]

    A2 = np.hstack([-np.eye(n), np.zeros((n, 1))])
    b2 = h.copy()
    A3 = np.hstack([np.eye(n), np.zeros((n, 1))])
    b3 = 1.0 - h

    A_ub = np.vstack([A1, A2, A3])
    b_ub = np.concatenate([b1, b2, b3])

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  method='highs', options={'maxiter': 10000})

    if res.success:
        delta = res.x[:n]
        t_val = res.x[-1]
        return delta, t_val, f_vals[top_idx]
    return None, None, f_vals[top_idx]


def cut_plane_optimize(h, n, max_iters, top_k, time_limit, rng):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    stuck_count = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        delta, t_val, _ = cutting_plane_lp(h, n, top_k)
        if delta is None:
            stuck_count += 1
            if stuck_count > 10:
                break
            h = project_feasible(h + 0.02 * rng.randn(n), n)
            continue

        delta_norm = np.linalg.norm(delta)
        if delta_norm < 1e-10:
            stuck_count += 1
            if stuck_count > 20:
                break
            h = project_feasible(h + 0.01 * rng.randn(n), n)
            continue

        improved = False
        for alpha in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05]:
            h_try = np.clip(h + alpha * delta, 0.0, 1.0)
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try)
            if c_try < best_c5 - 1e-14:
                h = h_try
                best_c5 = c_try
                best_h = h.copy()
                stuck_count = 0
                improved = True
                break

        if not improved:
            stuck_count += 1
            if stuck_count > 30:
                # Try larger K
                top_k = min(top_k * 2, 2 * n - 1)
                stuck_count = 0
                h = project_feasible(h + 0.005 * rng.randn(n), n)

    return best_h, best_c5


def slsqp_smooth_max(h, n, beta, maxiter=400, ftol=1e-18):
    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        shifted = beta * (corr - m)
        clipped = np.clip(shifted, -60.0, 60.0)
        ex = np.exp(clipped)
        val = (m + np.log(np.sum(ex)) / beta) * (2.0 / n)
        return val

    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        shifted = beta * (corr - m)
        clipped = np.clip(shifted, -60.0, 60.0)
        ex = np.exp(clipped)
        s = np.sum(ex)
        w = ex / s
        dx = 2.0 / n
        g = np.zeros(n)
        for k in range(len(corr)):
            wk = w[k] * dx
            if wk < 1e-14:
                continue
            g += wk * grad_corr_k(x, k, n)
        return g

    def constraint_fun(x):
        return np.sum(x) - n / 2.0

    def constraint_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]

    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            if compute_c5(h_new) <= compute_c5(h) + 1e-10:
                return h_new
    except Exception:
        pass
    return h.copy()


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def structured_perturbation(h, n, rng, scale=0.02):
    h_p = h.copy()
    freq = rng.randint(1, 6)
    phase = rng.uniform(0, 2 * np.pi)
    x = np.arange(n) * 2.0 / n
    noise = scale * np.sin(2 * np.pi * freq * x + phase)
    h_p += noise
    noise2 = scale * 0.3 * rng.randn(n)
    h_p += noise2
    return project_feasible(h_p, n)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)

    # Initialize from parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        parent_h = np.zeros(768)
        parent_h[:384] = 1.0

    n = len(parent_h)
    best_h = parent_h.copy()
    best_c5 = compute_c5(best_h)
    best_n = n

    # Phase 1: Cutting-plane optimization at parent resolution
    time_limit1 = min(remaining() * 0.35, 400.0)
    h1, c5_1 = cut_plane_optimize(best_h, n, max_iters=20000, top_k=150,
                                  time_limit=time_limit1, rng=rng)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: More cutting-plane with increased K
    if remaining() > 30:
        time_limit2 = min(remaining() * 0.2, 200.0)
        h2, c5_2 = cut_plane_optimize(best_h, n, max_iters=15000, top_k=400,
                                      time_limit=time_limit2, rng=rng)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: SLSQP polish at parent resolution
    if remaining() > 30:
        h3 = slsqp_smooth_max(best_h, n, beta=50000, maxiter=300)
        c5_3 = compute_c5(h3)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

        if remaining() > 20:
            h3b = slsqp_smooth_max(best_h, n, beta=200000, maxiter=300)
            c5_3b = compute_c5(h3b)
            if c5_3b < best_c5:
                best_c5 = c5_3b
                best_h = h3b.copy()

    # Phase 4: Higher resolution
    res_targets = []
    for m in [1.5, 2.0, 2.5, 3.0]:
        nt = int(round(n * m))
        if nt > n and nt <= 4096 and nt % 2 == 0:
            res_targets.append(nt)
    res_targets = sorted(set(res_targets))[:3]

    for n_t in res_targets:
        if remaining() < 60:
            break

        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        c_inter = compute_c5(h_feas)

        # Cutting-plane at new resolution
        time_limit_cp = min(remaining() * 0.25, 150.0)
        if time_limit_cp > 20:
            h_cp, c5_cp = cut_plane_optimize(h_feas, n_t, max_iters=10000,
                                             top_k=200, time_limit=time_limit_cp, rng=rng)
            if c5_cp < c_inter:
                h_feas = h_cp
                c_inter = c5_cp

        # SLSQP polish at new resolution
        if remaining() > 30:
            h_sls = slsqp_smooth_max(h_feas, n_t, beta=30000, maxiter=200)
            c_sls = compute_c5(h_sls)
            if c_sls < c_inter:
                h_feas = h_sls
                c_inter = c_sls

        if c_inter < best_c5:
            best_c5 = c_inter
            best_h = h_feas.copy()
            best_n = n_t

    # Phase 5: Final cutting-plane at best resolution
    if remaining() > 30:
        time_limit5 = min(remaining() * 0.4, 200.0)
        h5, c5_5 = cut_plane_optimize(best_h, best_n, max_iters=20000,
                                      top_k=300, time_limit=time_limit5, rng=rng)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Phase 6: Final SLSQP polish
    if remaining() > 20:
        h6 = slsqp_smooth_max(best_h, best_n, beta=500000, maxiter=500)
        c5_6 = compute_c5(h6)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Phase 7: Small perturbations + quick polish
    for p in range(5):
        if remaining() < 20:
            break
        h_pert = structured_perturbation(best_h, best_n, rng, scale=0.005)
        h_cp, c5_cp = cut_plane_optimize(h_pert, best_n, max_iters=3000,
                                         top_k=200, time_limit=10.0, rng=rng)
        if c5_cp < best_c5:
            best_c5 = c5_cp
            best_h = h_cp.copy()

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
