def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.count_nonzero(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            if diff > 0:
                idx = np.argmin(h)
                h[idx] += diff
            else:
                idx = np.argmax(h)
                h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h


def binarize(h, n):
    """Round to binary and fix the sum constraint."""
    h_bin = (h >= 0.5).astype(float)
    ones = int(np.count_nonzero(h_bin))
    target = n // 2
    diff = target - ones
    if diff > 0:
        # Need more ones: flip the largest 0s to 1
        zeros = np.where(h_bin == 0)[0]
        zeros_sorted = zeros[np.argsort(-h[zeros])][:diff]
        h_bin[zeros_sorted] = 1.0
    elif diff < 0:
        # Need fewer ones: flip the smallest 1s to 0
        ones_idx = np.where(h_bin == 1)[0]
        ones_sorted = ones_idx[np.argsort(h[ones_idx])[:abs(diff)]]
        h_bin[ones_sorted] = 0.0
    return h_bin


def smooth_max_value(x, beta, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50.0, 50.0)
    sm = m + np.log(np.sum(np.exp(shifted))) / beta
    return sm * (2.0 / n)


def smooth_max_grad(x, beta, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50.0, 50.0)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = w / w_sum

    g = np.zeros(n)
    dx = 2.0 / n
    nfull = 2 * n - 1
    for k in range(nfull):
        wk = w[k] * dx
        if wk < 1e-16:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            g[offset:offset + npairs] += wk * (1.0 - x[:npairs])
            g[:npairs] -= wk * x[offset:offset + npairs]
        else:
            a = -offset
            npairs = n - a
            g[:npairs] += wk * (1.0 - x[a:a + npairs])
            g[a:a + npairs] -= wk * x[:npairs]
    return g


def slsqp_optimize(h, n, beta, maxiter=500):
    def obj_func(x):
        return smooth_max_value(x, beta, n)

    def grad_func(x):
        return smooth_max_grad(x, beta, n)

    def eq_fun(x):
        return np.sum(x) - n / 2.0

    def eq_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]

    try:
        res = minimize(obj_func, h, jac=grad_func, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()


def precompute_dcorr(h, n):
    """Precompute dcorr[i, k] = d(corr[k])/d(h[i]) for each position i and lag k."""
    nfull = 2 * n - 1
    dcorr = np.zeros((n, nfull))
    lags = np.arange(nfull) - (n - 1)
    for i in range(n):
        # i as first element: h[i]*(1-h[i+lag]), valid when 0<=i+lag<n
        valid_pos = (i + lags >= 0) & (i + lags < n)
        idx_pos = np.where(valid_pos)[0]
        dcorr[i, idx_pos] += (1.0 - h[i + lags[idx_pos]])
        # i as second element: h[i-lag]*(1-h[i]), d/dh[i] = -h[i-lag], valid when 0<=i-lag<n
        valid_neg = (i - lags >= 0) & (i - lags < n)
        idx_neg = np.where(valid_neg)[0]
        dcorr[i, idx_neg] -= h[i - lags[idx_neg]]
    return dcorr


def best_swap_local_search(h, n, time_limit=120, max_passes=200):
    """Intensive best-swap local search for binary h."""
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    nfull = 2 * n - 1

    for _pass in range(max_passes):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 20:
            break

        ones_mask = h >= 0.999
        zeros_mask = h <= 0.001
        ones_idx = np.where(ones_mask)[0]
        zeros_idx = np.where(zeros_mask)[0]

        if len(ones_idx) < 2 or len(zeros_idx) < 2:
            # Not fully binary, use threshold
            ones_mask = h >= 0.5
            zeros_mask = h <= 0.5
            ones_idx = np.where(ones_mask)[0]
            zeros_idx = np.where(zeros_mask)[0]
            if len(ones_idx) < 1 or len(zeros_idx) < 1:
                break

        corr = compute_corr(h)
        c_max_val = float(np.max(corr))

        dcorr = precompute_dcorr(h, n)

        best_improvement = 1e-15
        best_i = -1
        best_j = -1

        for i in ones_idx:
            base = corr - dcorr[i]
            base_max = float(np.max(base))
            if base_max >= c_max_val - 1e-12:
                continue
            # For each zero j, new corr = base + dcorr[j]
            new_maxs = np.max(base[None, :] + dcorr[zeros_idx], axis=1)
            best_pos = int(np.argmin(new_maxs))
            best_new_max = float(new_maxs[best_pos])
            if best_new_max < c_max_val - best_improvement:
                best_improvement = c_max_val - best_new_max
                best_i = i
                best_j = zeros_idx[best_pos]

        if best_i >= 0 and best_j >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            h = project_feasible(h, n)
            c5_new = compute_c5(h)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
            else:
                # Undo
                h[best_i] = 1.0
                h[best_j] = 0.0
                h = project_feasible(h, n)
                break
        else:
            break

    return h, best_c5


def subgradient_descent(h, n, time_limit=60):

            break

        corr = compute_corr(h)
        progress = it / max(n_iter, 1)
        K = 40 if progress < 0.2 else (25 if progress < 0.6 else 15)
        beta_w = 30.0 if progress < 0.2 else (60.0 if progress < 0.6 else 120.0)

        idx = np.argsort(-corr)[:K]
        c_top = corr[idx]
        weights = np.exp(beta_w * (c_top - c_top.max()))
        w_sum = np.sum(weights)
        if w_sum < 1e-30:
            weights = np.ones(K) / K
        else:
            weights /= w_sum

        g = np.zeros(n)
        for wt, k in zip(weights, idx):
            if wt < 1e-14:
                continue
            offset = k - (n - 1)
            if offset >= 0:
                np_ = n - offset
                g[offset:offset + np_] += wt * (1.0 - h[:np_])
                g[:np_] -= wt * h[offset:offset + np_]
            else:
                a = -offset
                np_ = n - a
                g[:np_] += wt * (1.0 - h[a:a + np_])
                g[a:a + np_] -= wt * h[:np_]

        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break

        base_step = 0.05 / (1.0 + 0.03 * it)
        step = min(base_step, 0.15) * n / (gn + 1e-14)

        h_new = project_feasible(h - step * g, n)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            for factor in [0.5, 0.25, 0.1]:
                h_try = project_feasible(h - step * factor * g, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    break

    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
