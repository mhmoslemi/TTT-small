def project_feasible(h, n):

    offset = k - (n - 1)
    if offset >= 0:

        if npairs > 0:

            g[:npairs] -= h[offset:]
    else:
        a = -offset
        npairs = n - a
        if npairs > 0:
            g[:npairs] += (1.0 - h[a:])
            g[a:] -= h[:npairs]
    return g

def smooth_max_value(h, n, beta):

    """Create discrete representation of a step function.
    jump_points: sorted array of m jump locations in [0, 2]
    heights: array of m+1 heights
    n: number of discrete points

    h = np.zeros(n)

            mask = x >= jump_points[-1]

    return h

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - float(np.sum(h))
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.count_nonzero(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)

def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        if npairs > 0:
            g[offset:] += (1.0 - h[:npairs])
            g[:npairs] -= h[offset:]
    else:
        a = -offset
        npairs = n - a
        if npairs > 0:
            g[:npairs] += (1.0 - h[a:])
            g[a:] -= h[:npairs]
    return g

        self.beta = beta
        self.pl = pl
        self.top_k = top_k
        self.target = n / 2.0

def smobj(x, p):
    n = p.n
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(corr))
    sh = np.clip(p.beta * (corr - m), -100, 100)
    ex = np.exp(sh)
    val = (m + float(np.log(float(np.sum(ex)))) / p.beta) * dx
    val += p.pl * (float(np.sum(x)) - p.target) ** 2
    return val

def smgrad(x, p):
    n = p.n
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(corr))
    sh = np.clip(p.beta * (corr - m), -100, 100)
    ex = np.exp(sh)
    s = float(np.sum(ex))
    w = ex / s
    top_idx = np.argsort(-w)[:p.top_k]
    ws = w[top_idx].copy()
    ts = float(ws.sum())
    if ts > 1e-30:

    g = np.zeros(n)
    for ii in range(len(top_idx)):
        if ws[ii] < 1e-15:
            continue
        g += ws[ii] * grad_corr_k(x, int(top_idx[ii]), n)
    g *= dx
    if p.pl > 0.0:
        g += 2.0 * p.pl * (float(np.sum(x)) - p.target)
    return g

                       bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-14, 'iprint': 0})

                best_x = res.x.copy()
    except Exception:
        pass
