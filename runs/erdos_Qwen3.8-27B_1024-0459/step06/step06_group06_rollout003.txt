            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = n
            h = h_new.copy()
    
    # Phase 7: Top lag equalization
    if remaining_time() > 30:
        h_eq, c5_eq = top_lag_equalization(h, n, max_iters=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            best_n = n
            h = h_eq.copy()
    
    # Phase 8: Final SLSQP
    if remaining_time() > 15:
        h_new = slsqp_optimize(h, n, 50000, maxiter=600)

            h = h_new.copy()

            best_c5 = c5_bf2
