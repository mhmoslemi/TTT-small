def _tmin_obj(v):
    return v[-1]


def _tmin_obj_jac(v):
    g = np.zeros(len(v))
    g[-1] = 1.0
    return g


def _tmin_ineq(v):
    h = v[:-1]
    t = v[-1]
    n = _G_N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return t * n / 2.0 - corr[_G_KLIST]


def _tmin_ineq_jac(v):
    h = v[:-1]
    n = _G_N
    KL = _G_KLIST
    m = len(KL)
    J = np.zeros((m, n + 1))
    for i, k in enumerate(KL):
        g = grad_corr_k_vec(h, k, n)
        J[i, :n] = -g
        J[i, -1] = n / 2.0
    return J


def _tmin_eq(v):
    return np.sum(v[:-1]) - _G_N / 2.0


def _tmin_eq_jac(v):
    g = np.zeros(len(v))
    g[:-1] = 1.0
    return g


def active_set_opt(h, n, K_active, maxiter=300):
    global _G_N, _G_KLIST
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx_active = np.argsort(-corr)[:K_active]
    _G_N = n
    _G_KLIST = idx_active
    t0 = compute_c5(h)
    v0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'ineq', 'fun': _tmin_ineq, 'jac': _tmin_ineq_jac},
        {'type': 'eq', 'fun': _tmin_eq, 'jac': _tmin_eq_jac}
    ]
    try:
        res = minimize(_tmin_obj, v0, jac=_tmin_obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x[:n], n)
            c5_new = compute_c5(h_new)
            return h_new, c5_new
    except Exception:
        pass
    return h.copy(), compute_c5(h)

    """Compute np.correlate(a, b, 'full') using FFT."""
    n = len(a)
    size = 2 * n
    fa = np.fft.rfft(a, size)

    return c[idx]
