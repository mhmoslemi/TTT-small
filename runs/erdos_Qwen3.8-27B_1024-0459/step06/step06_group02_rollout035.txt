import numpy as np
from scipy.optimize import minimize
import time

def next_power_of_2(n):
    return 1 << (n - 1).bit_length()

def fft_full_correlation(h):
    """Compute full cross-correlation of h and (1-h) using FFT.
    Returns array of length 2n-1 matching np.correlate(h, 1-h, 'full').
    """
    n = len(h)
    N = next_power_of_2(4 * n)  # Safe for no aliasing
    H = np.fft.rfft(h, N)
