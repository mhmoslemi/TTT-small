def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_lag(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        if npairs > 0:
            g[:npairs] += (1.0 - h[offset:offset + npairs])
            g[offset:] -= h[:npairs]
    else:
        a = -offset
        npairs = n - a
        if npairs > 0:
            g[a:] += (1.0 - h[:npairs])
            g[:npairs] -= h[a:a + npairs]
    return g

def interp_rescale(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def _obj_general(x):
    n = _OPT_N
    K = _OPT_K
    beta = _OPT_BETA
    hx = np.clip(x, 0.0, 1.0)
    cx = np.correlate(hx, 1.0 - hx, mode='full')

    ca = cx[idx]
    shifted = np.clip(beta * (ca - ca.max()), -50.0, 50.0)
    ex = np.exp(shifted)
    return (ca.max() + np.log(np.sum(ex)) / beta) * (2.0 / n)

def _grad_general(x):
    n = _OPT_N
    K = _OPT_K
    beta = _OPT_BETA
    hx = np.clip(x, 0.0, 1.0)
    cx = np.correlate(hx, 1.0 - hx, mode='full')

    ca = cx[idx]

    s = np.sum(ex)
    w = ex / s
    dx_val = 2.0 / n
    g = np.zeros(n)
    for i in range(K):
        if w[i] > 1e-14:
            g += w[i] * dx_val * grad_lag(hx, idx[i], n)
    return g

    return np.sum(x) - _OPT_N / 2.0

    return np.ones(_OPT_N)

def _obj_sym(x):

    K = _OPT_K
    beta = _OPT_BETA

    set_opt_params(n, K, beta)

    except Exception:
        pass
    return h.copy()
