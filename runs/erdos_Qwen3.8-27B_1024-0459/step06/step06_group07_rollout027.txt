def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_binary(h, n):
    """Project to binary with exactly n/2 ones."""
    h = (h >= 0.5).astype(float)
    ones = int(np.sum(h))
    target = n // 2
    if ones > target:
        # Need to flip (ones - target) ones to zeros
        # Flip the ones that are closest to 0.5 (i.e., least "committed")
        # For now, just pick random ones

        idx = rng.permutation(idx)
        for i in range(ones - target):
            h[idx[i]] = 0.0
    elif ones < target:
        idx = np.where(h == 0.0)[0]
        rng = np.random.default_rng()
        idx = rng.permutation(idx)
        for i in range(target - ones):

def slsqp_refine(h_init, n, time_limit):
    """SLSQP optimization with smooth max."""
    h = project_feasible(h_init.copy(), n)
    best_c5 = compute_c5(h)
    start = time.time()
    
    for beta in [500, 2000, 8000, 30000]:
        if time.time() - start > time_limit - 5:
            break
        if _BUDGET - (time.time() - _T0) < 10:
            break
        
        h_new = _slsqp_single(h, n, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            h = h_new.copy()
    
    return h, best_c5

def _slsqp_single(h, n, beta, maxiter=300):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -50.0, 50.0)
        return (m + np.log(np.sum(np.exp(shifted))) / beta) * (2.0 / n)
    
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -50.0, 50.0)
        w = np.exp(shifted)
        w_sum = np.sum(w)
        if w_sum < 1e-30:
            w = np.ones(2*n-1) / (2*n-1)
        else:
            w /= w_sum
        
        g = np.zeros(n)
        dx = 2.0 / n
        nfull = 2*n - 1
        for k in range(nfull):
            wk = w[k] * dx
            if wk < 1e-16:
                continue
            offset = k - (n-1)
            if offset >= 0:
                np_ = n - offset
                g[offset:offset+np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[offset:offset+np_]
            else:
                a = -offset
                np_ = n - a
                g[:np_] += wk * (1.0 - x[a:a+np_])
                g[a:a+np_] -= wk * x[:np_]
        return g
    
    def eq_fun(x):
        return np.sum(x) - n / 2.0
    
    def eq_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    
    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)
    
    # Phase 0: Get initial solution
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        best_n = len(parent_h)
    except Exception:
        best_n = 512
        parent_h = np.zeros(best_n)
        parent_h[:best_n//2] = 1.0
    
    best_h = project_feasible(parent_h, best_n)
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Binary simulated annealing at parent resolution
    h_bin = (best_h >= 0.5).astype(float)
    h_bin = binary_project(h_bin, best_n, rng)
    c5_bin = compute_c5(h_bin)
    
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    
    # SA on binary
    if remaining() > 60:
        t_sa = min(300.0, remaining() * 0.4)
        h_sa, c5_sa, _ = sa_binary(h_bin, best_n, rng, t_sa)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
    
    # Phase 1b: Targeted SA
    if remaining() > 60:
        h_t, c5_t = targeted_sa(h_sa, best_n, rng, min(200.0, remaining() * 0.25))
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
    
    # Phase 2: Continuous refinement
    if remaining() > 30:
        t_cont = min(200.0, remaining() * 0.3)
        h_cont, c5_cont = continuous_refine(best_h.copy(), best_n, t_cont)
        if c5_cont < best_c5:
            best_c5 = c5_cont
            best_h = h_cont
