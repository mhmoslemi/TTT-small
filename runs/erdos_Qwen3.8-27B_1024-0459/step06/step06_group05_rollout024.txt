def project_sum_box(v, target, n):
    """Project v onto {x : sum(x) = target, 0 <= x_i <= 1}."""
    # Binary search on the threshold
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        # Project each component: x_i = clip(v_i - mid, 0, 1)
        x = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(x)
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2, 0.0, 1.0)
    # Adjust for floating point
    diff = target - np.sum(x)
    if abs(diff) > 1e-15:
        mask = (x > 0) & (x < 1)
        if np.any(mask):
            x[mask] += diff / np.count_nonzero(mask)
    return np.clip(x, 0.0, 1.0)
