        best_alpha = 0
        best_c = best_c5
        scale = max(0.1, 0.5 / (1 + 0.01 * it))
        for alpha in [scale, scale * 0.5, scale * 2, scale * 0.25, scale * 4,
                      -scale, -scale * 0.5, -scale * 2]:
            ht = project_feasible(h + alpha * d, n)
            ct = compute_c5(ht)
            if ct < best_c - 1e-14:
                best_c = ct
                best_alpha = alpha
        if best_alpha != 0:
            h = project_feasible(h + best_alpha * d, n)
            if best_c < best_c5:
                best_c5 = best_c
                best_h = h.copy()
    return best_h, best_c5


def coordinate_descent_continuous(h, n, time_limit, rng):
    global _T0, _BUDGET
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    order = rng.permutation(n)
    it = 0
    while time.time() - start < time_limit:
        if _BUDGET - (time.time() - _T0) < 3:
            break
        it += 1
        improved = False
        for _pass in range(min(3, n)):
            for idx_pos in range(n):
                if time.time() - start > time_limit:
                    break
                if _BUDGET - (time.time() - _T0) < 3:
                    break
                i = order[(it * n + idx_pos) % n]
                old_val = h[i]
                candidates = [0.0, 1.0]
                if old_val > 0.05 and old_val < 0.95:
                    candidates.extend([0.25, 0.5, 0.75])
                best_c = compute_c5(h)
                best_v = old_val
                for cv in candidates:
                    if abs(cv - old_val) < 1e-12:
                        continue
                    h[i] = cv
                    h = project_feasible(h, n)
                    c = compute_c5(h)
                    if c < best_c - 1e-14:
                        best_c = c
                        best_v = cv
                h[i] = best_v
                h = project_feasible(h, n)
                c_now = compute_c5(h)
                if c_now < best_c5 - 1e-14:
                    best_c5 = c_now
                    improved = True
        if not improved:
            break
    return h, best_c5


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 512
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: Interpolate to n=1024
    n1 = 1024
    h1 = interp_to(best_h, best_n, n1)
    h1 = project_feasible(h1, n1)
    c5_1 = compute_c5(h1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
        best_n = n1

    h_curr = h1.copy()
    n_curr = n1

    # Phase 2: Quick subgradient at n=1024
    if remaining_time() > 60:
        h_sg, c5_sg = subgradient_descent(h_curr, n_curr, min(25.0, remaining_time() * 0.15), rng)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h_curr = h_sg.copy()

    # Phase 3: L-BFGS-B with increasing beta at n=1024
    for beta in [1000, 3000, 10000]:
        if remaining_time() < 30:
            break
        h_new = lbfgsb_smooth(h_curr, n_curr, beta, 500, maxiter=2000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_curr = h_new.copy()

    # Phase 4: Sign algorithm at n=1024
    if remaining_time() > 50:
        h_sa, c5_sa = sign_algorithm(h_curr, n_curr, min(40.0, remaining_time() * 0.15), rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            h_curr = h_sa.copy()

    # Phase 5: SLSQP polish at n=1024
    for beta in [5000, 15000]:
        if remaining_time() < 20:
            break
        h_new = slsqp_smooth(h_curr, n_curr, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_curr = h_new.copy()

    # Phase 6: Move to n=2048
    n2 = 2048
    h2 = interp_to(best_h, best_n, n2)
    h2 = project_feasible(h2, n2)
    c5_2 = compute_c5(h2)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
        best_n = n2

    h_curr2 = h2.copy()
    n_curr2 = n2

    # Phase 7: Subgradient at n=2048
    if remaining_time() > 60:
        h_sg2, c5_sg2 = subgradient_descent(h_curr2, n_curr2, min(30.0, remaining_time() * 0.15), rng)
        if c5_sg2 < best_c5:
            best_c5 = c5_sg2
            best_h = h_sg2.copy()
            h_curr2 = h_sg2.copy()

    # Phase 8: L-BFGS-B at n=2048
    for beta in [2000, 8000, 20000]:
        if remaining_time() < 40:
            break
        maxit = 3000 if remaining_time() > 100 else 1500
        h_new = lbfgsb_smooth(h_curr2, n_curr2, beta, 1000, maxiter=maxit)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_curr2 = h_new.copy()

    # Phase 9: Sign algorithm at n=2048
    if remaining_time() > 40:
        h_sa2, c5_sa2 = sign_algorithm(h_curr2, n_curr2, min(35.0, remaining_time() * 0.15), rng)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()
            h_curr2 = h_sa2.copy()

    # Phase 10: SLSQP at n=2048
    for beta in [10000, 30000]:
        if remaining_time() < 20:
            break
        h_new = slsqp_smooth(h_curr2, n_curr2, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_curr2 = h_new.copy()

    # Phase 11: Random direction search at n=2048
    if remaining_time() > 40:
        h_rd, c5_rd = random_direction_search(h_curr2, n_curr2, min(35.0, remaining_time() * 0.15), rng)
        if c5_rd < best_c5:
            best_c5 = c5_rd
            best_h = h_rd.copy()
            h_curr2 = h_rd.copy()

    # Phase 12: L-BFGS-B high beta at n=2048
    for beta in [50000, 100000]:
        if remaining_time() < 30:
            break
        h_new = lbfgsb_smooth(h_curr2, n_curr2, beta, 5000, maxiter=2000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_curr2 = h_new.copy()

    # Phase 13: Try n=4096 if time permits
    if remaining_time() > 100:
        n3 = 4096
        h3 = interp_to(best_h, best_n, n3)
        h3 = project_feasible(h3, n3)
        c5_3 = compute_c5(h3)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
            best_n = n3
        h_curr3 = h3.copy()
        # Quick optimization at n=4096
        if remaining_time() > 60:
            h_sg3, c5_sg3 = subgradient_descent(h_curr3, n3, min(20.0, remaining_time() * 0.2), rng)
            if c5_sg3 < best_c5:
                best_c5 = c5_sg3
                best_h = h_sg3.copy()
                h_curr3 = h_sg3.copy()
        if remaining_time() > 40:
            h_new = lbfgsb_smooth(h_curr3, n3, 5000, 2000, maxiter=1500)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_curr3 = h_new.copy()
        if remaining_time() > 20:
            h_new = slsqp_smooth(h_curr3, n3, 10000, maxiter=200)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 14: Final polish on best solution
    if remaining_time() > 20:
        h_new = slsqp_smooth(best_h, best_n, 50000, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 15: Final sign algorithm
    if remaining_time() > 15:
        h_sa_f, c5_sa_f = sign_algorithm(best_h, best_n, min(12.0, remaining_time() * 0.6), rng)
        if c5_sa_f < best_c5:
            best_c5 = c5_sa_f
            best_h = h_sa_f.copy()

    # Phase 16: Final coordinate descent
    if remaining_time() > 15:
        h_cd, c5_cd = coordinate_descent_continuous(best_h, best_n, min(12.0, remaining_time() * 0.6), rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Final feasibility and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
