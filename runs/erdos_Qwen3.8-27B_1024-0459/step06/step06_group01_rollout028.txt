def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            adj = diff / cnt
            if adj > 0:
                h[mask] = np.minimum(h[mask] + adj, 1.0)
            else:
                h[mask] = np.maximum(h[mask] + adj, 0.0)
        else:

            h = np.clip(h, 0.0, 1.0)
    return h


def corr_grad_single(h, k, n):
    """Gradient of corr[k] w.r.t. h, where k is the index in np.correlate output."""
    off = k - (n - 1)
    g = np.zeros(n)
    if off >= 0:
        m = n - off
        g[off:] += 1.0
        g[off:] -= h[:m]
        g[:m] -= h[off:]
    else:
        a = -off
        m = n - a
        g[:m] += 1.0
        g[:m] -= h[a:]
        g[a:] -= h[:m]
    return g


def topk_direction(h, n, top_k=5):
    """Compute the average gradient direction of top-k correlations."""
    corr = compute_corr(h)

    g = np.zeros(n)
