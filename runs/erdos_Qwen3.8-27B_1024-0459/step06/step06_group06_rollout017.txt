def smooth_max_obj(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(beta * sh, -50.0, 50.0)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * (2.0 / n)

def smooth_max_grad(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(beta * sh, -50.0, 50.0)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    dx = 2.0 / n
    for k in range(len(corr)):
        wk = w[k] * dx
        if wk < 1e-16:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            g[offset:n] += wk * (1.0 - h[:npairs])
            g[:npairs] -= wk * h[offset:n]
        else:
            a = -offset
            npairs = n - a
            g[:npairs] += wk * (1.0 - h[a:n])
            g[a:n] -= wk * h[:npairs]
    return g

        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        sm = (m + np.log(np.sum(ex)) / beta) * (2.0 / n)
        s = np.sum(x) - n / 2.0
        return sm + 0.5 * penalty * s * s
    
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        dx = 2.0 / n
        for k in range(len(corr)):
            wk = w[k] * dx
            if wk < 1e-16:
                continue
            offset = k - (n - 1)
            if offset >= 0:
                npairs = n - offset
                g[offset:n] += wk * (1.0 - x[:npairs])
                g[:npairs] -= wk * x[offset:n]
            else:
                a = -offset
                npairs = n - a
                g[:npairs] += wk * (1.0 - x[a:n])
                g[a:n] -= wk * x[:npairs]
        s = np.sum(x) - n / 2.0
        g += penalty * s * np.ones(n)
        return g
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14, 'maxfun': 100000})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import coo_matrix
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)


def solve_lp_for_n(n):
    """Solve the exact LP: min T s.t. for each shift k, sum of z_pairs <= T,
    with z linearization constraints, sum(h)=n/2, 0<=h<=1."""
    nshifts = 2 * n - 1
    total_z = n * n
    nvars = n + total_z + 1
    z_start = n
    t_idx = n + total_z

    # Build z mapping
    z_map = []
    shift_z_indices = []
    z_count = 0
    for k in range(nshifts):
        offset = k - (n - 1)
        z_indices = []
        if offset >= 0:
            npairs = n - offset
            for i in range(npairs):
                z_map.append((offset + i, i))
                z_indices.append(z_count)
                z_count += 1
        else:
            a = -offset
            npairs = n - a
            for i in range(npairs):
                z_map.append((i, a + i))
                z_indices.append(z_count)
                z_count += 1
        shift_z_indices.append(z_indices)

    assert z_count == total_z

    # Build sparse constraint matrix
    rows_ub = []
    cols_ub = []
    vals_ub = []
    b_ub = []
    row = 0

    # Set 1: z[m] - h[i] <= 0
    for m in range(total_z):
        i, j = z_map[m]
        zm = z_start + m
        rows_ub.extend([row, row])
        cols_ub.extend([zm, i])
        vals_ub.extend([1.0, -1.0])
        b_ub.append(0.0)
        row += 1

    # Set 2: z[m] + h[j] <= 1
    for m in range(total_z):
        i, j = z_map[m]
        zm = z_start + m
        rows_ub.extend([row, row])
        cols_ub.extend([zm, j])
        vals_ub.extend([1.0, 1.0])
        b_ub.append(1.0)
        row += 1

    # Set 3: -z[m] + h[i] - h[j] <= 0
    for m in range(total_z):
        i, j = z_map[m]
        zm = z_start + m
        rows_ub.extend([row, row, row])
        cols_ub.extend([zm, i, j])
        vals_ub.extend([-1.0, 1.0, -1.0])
        b_ub.append(0.0)
        row += 1

    # Set 4: sum_z for shift k - T <= 0
    for k in range(nshifts):
        sz = shift_z_indices[k]
        rows_ub.extend([row] * (len(sz) + 1))
        cols_ub.extend([z_start + m for m in sz] + [t_idx])
        vals_ub.extend([1.0] * len(sz) + [-1.0])
        b_ub.append(0.0)
        row += 1

    assert row == 3 * total_z + nshifts

    A_ub = coo_matrix((vals_ub, (rows_ub, cols_ub)), shape=(row, nvars)).tocsr()
    b_ub_arr = np.array(b_ub)

    A_eq = coo_matrix((np.ones(n), (np.zeros(n, dtype=int), np.arange(n))), shape=(1, nvars)).tocsr()
    b_eq = np.array([n / 2.0])

    c = np.zeros(nvars)
    c[t_idx] = 1.0

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)] * total_z + [(0.0, None)]

    start = time.time()
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub_arr, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        elapsed = time.time() - start
        if res.success:
            h_opt = res.x[:n]
            T_opt = res.x[t_idx]
            return project_feasible(h_opt, n), T_opt * (2.0 / n), elapsed
    except Exception:
        pass
    return None, None, time.time() - start


def lbfgsb_smooth(h, n, beta, penalty, maxiter):
    def obj_fun(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        sm = (m + np.log(np.sum(ex)) / beta) * (2.0 / n)
        s = np.sum(x) - n / 2.0
        return sm + 0.5 * penalty * s * s

    def grad_fun(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        dx = 2.0 / n
        for k in range(len(corr)):
            wk = w[k] * dx
            if wk < 1e-16:
                continue
            offset = k - (n - 1)
            if offset >= 0:
                npairs = n - offset
                g[offset:n] += wk * (1.0 - x[:npairs])
                g[:npairs] -= wk * x[offset:n]
            else:
                a = -offset
                npairs = n - a
                g[:npairs] += wk * (1.0 - x[a:n])
                g[a:n] -= wk * x[:npairs]
        s = np.sum(x) - n / 2.0
        g += penalty * s * np.ones(n)
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fun, h, jac=grad_fun, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-15, 'maxfun': 80000})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

                g[:npairs] -= weights[i] * h[offset:n]
            else:
                a = -offset
                npairs = n - a
                g[:npairs] += weights[i] * (1.0 - h[a:n])
                g[a:n] -= weights[i] * h[:npairs]

        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break

        base_step = 0.03 / (1.0 + 0.03 * it)
        improved = False
        for s_f in [base_step, base_step * 0.5, base_step * 0.2, base_step * 1.5]:
            ht = project_feasible(h - s_f * g / gn, n)
            ct = compute_c5(ht)
            if ct < best_c5 - 1e-15:
                h = ht
                best_c5 = ct
                best_h = h.copy()
                improved = True
                break
        if not improved and it % 100 == 0:
            h = best_h.copy()
        elif not improved and it > 500:
            break
    return best_h, best_c5


def equalize_top_shifts(h, n, time_limit):
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    for it in range(300):
        if time.time() - start > time_limit or remaining_time() < 5:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        K = 12
        idx = np.argsort(-corr)[:K]
        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        if c_max - c_min < 1e-10:
            break

        g_avg = np.zeros(n)
        for k in idx:
            offset = k - (n - 1)
            if offset >= 0:
                npairs = n - offset
                g_avg[offset:n] += (1.0 - h[:npairs])
                g_avg[:npairs] -= h[offset:n]
            else:
                a = -offset
                npairs = n - a
                g_avg[:npairs] += (1.0 - h[a:n])
                g_avg[a:n] -= h[:npairs]
        g_avg /= K

        gn = np.linalg.norm(g_avg)
        if gn < 1e-14:
            break
        g_dir = g_avg / gn

        improved = False
        for step in [0.01, 0.02, 0.005, 0.03, 0.015, 0.04]:
            ht = project_feasible(h - step * g_dir, n)
            ct = compute_c5(ht)
            if ct < best_c5 - 1e-15:
                h = ht
                best_c5 = ct
                improved = True
                break
        if not improved:
            for step in [0.01, 0.02, 0.005, 0.03]:
                ht = project_feasible(h + step * g_dir, n)
                ct = compute_c5(ht)
                if ct < best_c5 - 1e-15:
                    h = ht
                    best_c5 = ct
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def random_binary_init(n, rng, frac):
    h = np.zeros(n)
    n_ones = int(n * frac)
    idx = rng.permutation(n)[:n_ones]
    h[idx] = 1.0
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent or create default
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 512
        parent_h = random_binary_init(parent_n, rng, 0.5)
        best_h = parent_h.copy()
        best_c5 = compute_c5(best_h)
        best_n = parent_n

    # Phase 1: LP at n=256
    if remaining_time() > 150:
        n_lp1 = 256
        h_256 = interp_to(best_h, best_n, n_lp1)
        h_256 = project_feasible(h_256, n_lp1)
        h_sol, c5_sol, elapsed = solve_lp_for_n(n_lp1)
        if h_sol is not None:
            c5_eval = compute_c5(h_sol)
            print(f"LP n={n_lp1}: C5={c5_eval:.6f} in {elapsed:.1f}s")
            if c5_eval < best_c5:
                best_c5 = c5_eval
                best_h = h_sol.copy()
                best_n = n_lp1
            # Also try from parent interpolation
            h_256b = interp_to(parent_h, parent_n, n_lp1)
            h_256b = project_feasible(h_256b, n_lp1)
            h_sol2, c5_sol2, elapsed2 = solve_lp_for_n(n_lp1)
            if h_sol2 is not None:
                c2 = compute_c5(h_sol2)
                if c2 < best_c5:
                    best_c5 = c2
                    best_h = h_sol2.copy()
                    best_n = n_lp1

    # Phase 2: LP at n=384 (different resolution)
    if remaining_time() > 200:
        n_lp2 = 384
        h_384 = interp_to(best_h, best_n, n_lp2)
        h_384 = project_feasible(h_384, n_lp2)
        h_sol3, c5_sol3, elapsed3 = solve_lp_for_n(n_lp2)
        if h_sol3 is not None:
            c3 = compute_c5(h_sol3)
            print(f"LP n={n_lp2}: C5={c3:.6f} in {elapsed3:.1f}s")
            if c3 < best_c5:
                best_c5 = c3
                best_h = h_sol3.copy()
                best_n = n_lp2

    # Phase 3: LP at n=512
    if remaining_time() > 250:
        n_lp3 = 512
        h_512 = interp_to(best_h, best_n, n_lp3)
        h_512 = project_feasible(h_512, n_lp3)
        h_sol4, c5_sol4, elapsed4 = solve_lp_for_n(n_lp3)
        if h_sol4 is not None:
            c4 = compute_c5(h_sol4)
            print(f"LP n={n_lp3}: C5={c4:.6f} in {elapsed4:.1f}s")
            if c4 < best_c5:
                best_c5 = c4
                best_h = h_sol4.copy()
                best_n = n_lp3

    # Phase 4: Interpolate to n=1024 and refine
    n_ref = 1024
    h_ref = interp_to(best_h, best_n, n_ref)
    h_ref = project_feasible(h_ref, n_ref)
    c5_ref = compute_c5(h_ref)
    if c5_ref < best_c5:
        best_c5 = c5_ref
        best_h = h_ref.copy()
        best_n = n_ref

    h = h_ref.copy()
    n = n_ref

    # Phase 5: Smooth max L-BFGS-B at n=1024
    for beta in [800, 3000, 8000, 15000]:
        if remaining_time() < 25:
            break
        h_new = lbfgsb_smooth(h, n, beta, 2000, maxiter=4000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 6: Subgradient refinement
    if remaining_time() > 50:
        h_sg, c5_sg = subgradient_refine(h, n, min(40.0, remaining_time() * 0.35), rng)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 7: Equalize shifts
    if remaining_time() > 30:
        h_eq, c5_eq = equalize_top_shifts(h, n, min(25.0, remaining_time() * 0.4))
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 8: More L-BFGS-B at high beta
    for beta in [20000, 40000]:
        if remaining_time() < 20:
            break
        h_new = lbfgsb_smooth(h, n, beta, 5000, maxiter=3000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 9: Try n=2048
    if remaining_time() > 100 and best_n < 2048:
        n_vh = 2048
        h_vh = interp_to(best_h, best_n, n_vh)
        h_vh = project_feasible(h_vh, n_vh)
        c5_vh = compute_c5(h_vh)
        if c5_vh < best_c5:
            h_vh = lbfgsb_smooth(h_vh, n_vh, 3000, 2000, maxiter=2000)
            c5_vh = compute_c5(h_vh)
            h_vh = lbfgsb_smooth(h_vh, n_vh, 8000, 5000, maxiter=3000)
            c5_vh = compute_c5(h_vh)
            if c5_vh < best_c5:
                best_c5 = c5_vh
                best_h = h_vh.copy()
                best_n = n_vh

    # Phase 10: Random restart at n=512 with different structures
    if remaining_time() > 120:
        for trial in range(3):
            if remaining_time() < 40:
                break
            # Create perturbed version
            h_r = best_h.copy()
            n_r = len(h_r)
            # Random perturbation
            mask = rng.random(n_r) < 0.3
            h_r[mask] = h_r[mask] + rng.randn(mask.sum()) * 0.1
            h_r = project_feasible(h_r, n_r)
            # Quick optimization
            h_r = lbfgsb_smooth(h_r, n_r, 5000, 2000, maxiter=2000)
            c5_r = compute_c5(h_r)
            if c5_r < best_c5:
                h_r = lbfgsb_smooth(h_r, n_r, 15000, 5000, maxiter=2000)
                c5_r = compute_c5(h_r)
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = h_r.copy()
                    best_n = n_r

    # Phase 11: Final polish
    if remaining_time() > 15:
        h_final = best_h.copy()
        n_final = best_n
        h_final = lbfgsb_smooth(h_final, n_final, 50000, 10000, maxiter=3000)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Final equalization polish
    if remaining_time() > 10:
        h_eq2, c5_eq2 = equalize_top_shifts(best_h, best_n, 8.0)
        if c5_eq2 < best_c5:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        best_c5 = min(best_c5, eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
