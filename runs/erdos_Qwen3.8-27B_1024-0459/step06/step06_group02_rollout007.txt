def grad_smooth_max(h, n, beta):
    # Compute all correlations
    corr = np.correlate(h, 1.0-h, mode='full')  # length 2n-1
    # Weights
    shifted = beta * corr
    shifted -= shifted.max()  # numerical stability
    w = np.exp(shifted)
    w /= w.sum()
    # Now compute gradient
    dx = 2.0 / n
    g = np.zeros(n)
    for j in range(n):
        s = 0.0
        for k in range(2*n-1):
            m = k - (n-1)  # lag
            if m >= 0:
                # c_m contribution
                if m <= j <= n-1:
                    s += w[k] * (1.0 - h[j-m])
                if j <= n-1-m:
                    s += w[k] * (-h[j+m])
            else:
                p = -m
                if j <= n-1-p:
                    s += w[k] * (1.0 - h[j+p])
                if j >= p:
                    s += w[k] * (-h[j-p])
        g[j] = dx * s
    return g

            g[m:] += wi * (1.0 - h[:n - m])
            g[:n - m] -= wi * h[m:]
        else:
            p = -m
            g[:n - p] += wi * (1.0 - h[p:])
            g[p:] -= wi * h[:n - p]
    gn = np.linalg.norm(g)
    if gn < 1e-14:
        return h.copy()
    h_new = project_feasible(h - step * g / gn, n)
    return h_new


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)

    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        parent_h = np.zeros(512)
        parent_h[:256] = 1.0

    n_init = len(parent_h)
    best_h = project_feasible(parent_h.copy(), n_init)
    best_c5 = compute_c5(best_h)
    best_n = n_init

    # Phase 1: L-BFGS-B at parent resolution with increasing beta
    betas_p1 = [50, 200, 1000, 5000, 20000, 100000]
    for beta in betas_p1:
        if remaining() < 30:
            break
        h_try = lbfgs_run(best_h, best_n, beta, 1000.0, maxiter=300, maxfun=1500, ftol=1e-14)
        c_try = compute_c5(h_try)
        if c_try < best_c5 - 1e-12:
            best_c5 = c_try
            best_h = h_try.copy()

    # Phase 2: Multiple random perturbations + optimize
    for p in range(6):
        if remaining() < 40:
            break
        scale = 0.01 + 0.01 * p
        h_pert = random_perturb(best_h, best_n, rng, scale)
        for beta in [500, 5000]:
            if remaining() < 20:
                break
            h_s = lbfgs_run(h_pert, best_n, beta, 1000.0, maxiter=250, maxfun=1000, ftol=1e-13)
            c_s = compute_c5(h_s)
            if c_s < best_c5 - 1e-12:
                best_c5 = c_s
                best_h = h_s.copy()

    # Phase 3: Multi-resolution exploration
    res_options = []
    if best_n > 384:
        res_options.append(max(256, best_n // 2))
    if best_n < 1024:
        res_options.append(1024)
    if best_n < 768:
        res_options.append(768)
    if best_n < 512:
        res_options.append(512)
    res_options = [r for r in res_options if r != best_n and 128 <= r <= 2048]
    res_options = [r for r in res_options if r % 2 == 0]
    res_options = sorted(set(res_options))

    for n_t in res_options[:4]:
        if remaining() < 60:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        c_inter = compute_c5(h_feas)

        for beta in [200, 1000, 5000, 20000]:
            if remaining() < 25:
                break
            h_new = lbfgs_run(h_feas, n_t, beta, 1000.0, maxiter=300, maxfun=1500, ftol=1e-14)
            c_new = compute_c5(h_new)
            if c_new < c_inter:
                c_inter = c_new
                h_feas = h_new

        if c_inter < best_c5 - 1e-12:
            best_c5 = c_inter
            best_h = h_feas.copy()
            best_n = n_t

        if remaining() > 30:
            h_pert2 = random_perturb(h_feas, n_t, rng, 0.015)
            h_new2 = lbfgs_run(h_pert2, n_t, 5000, 1000.0, maxiter=250, maxfun=1000, ftol=1e-13)
            c_new2 = compute_c5(h_new2)
            if c_new2 < best_c5 - 1e-12:
                best_c5 = c_new2
                best_h = h_new2.copy()
                best_n = n_t

    # Phase 4: Subgradient descent for broader search
    if remaining() > 60:
        h_sg = best_h.copy()
        n_sg = best_n
        c_sg = compute_c5(h_sg)
        n_iter_sg = int(min(3000, remaining() * 30))
        for it in range(n_iter_sg):
            if time.time() - _T0 > _BUDGET - 40:
                break
            progress = it / max(1, n_iter_sg)
            K = 40 if progress < 0.3 else (25 if progress < 0.6 else 15)
            top_beta = 50.0 if progress < 0.3 else (80.0 if progress < 0.6 else 120.0)
            step = 0.05 / (1.0 + 0.03 * it)
            h_sg_new = subgrad_step(h_sg, n_sg, K, top_beta, step)
            c_new = compute_c5(h_sg_new)
            if c_new < c_sg - 1e-13:
                c_sg = c_new
                h_sg = h_sg_new
            elif it > 200 and it % 150 == 0:
                if remaining() > 30:
                    h_pol = lbfgs_run(h_sg, n_sg, 5000, 1000.0, maxiter=150, maxfun=500, ftol=1e-12)
                    c_pol = compute_c5(h_pol)
                    if c_pol < best_c5 - 1e-12:
                        best_c5 = c_pol
                        best_h = h_pol.copy()
                        best_n = n_sg
                    h_sg = h_pol
                    c_sg = compute_c5(h_sg)

        c_sg_final = compute_c5(h_sg)
        if c_sg_final < best_c5 - 1e-12:
            best_c5 = c_sg_final
            best_h = h_sg.copy()
            best_n = n_sg

    # Phase 5: High-beta L-BFGS-B polish at best resolution
    for beta in [50000, 200000, 500000]:
        if remaining() < 15:
            break
        h_new = lbfgs_run(best_h, best_n, beta, 1000.0, maxiter=500, maxfun=2000, ftol=1e-15)
        c_new = compute_c5(h_new)
        if c_new < best_c5 - 1e-12:
            best_c5 = c_new
            best_h = h_new.copy()

    # Phase 6: Small perturbations + final polish
    for p in range(5):
        if remaining() < 20:
            break
        h_pert = random_perturb(best_h, best_n, rng, 0.005 + 0.003 * p)
        h_pol = lbfgs_run(h_pert, best_n, 20000, 1000.0, maxiter=300, maxfun=1200, ftol=1e-14)
        c_pol = compute_c5(h_pol)
        if c_pol < best_c5 - 1e-12:
            best_c5 = c_pol
            best_h = h_pol.copy()

    # Phase 7: Final high-beta run
    if remaining() > 15:
        h_final = lbfgs_run(best_h, best_n, 500000, 1000.0, maxiter=800, maxfun=3000, ftol=1e-16)
        c_final = compute_c5(h_final)
        if c_final < best_c5 - 1e-12:
            best_c5 = c_final
            best_h = h_final.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
