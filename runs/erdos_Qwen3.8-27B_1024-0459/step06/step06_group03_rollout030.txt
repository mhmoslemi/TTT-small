        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        if c_max - c_min < 1e-12:
            break
        g_avg = np.zeros(n)
        for k in idx:
            g_avg += grad_corr_k(h, k, n)
        g_avg /= K
        gn = np.linalg.norm(g_avg)
        if gn < 1e-14:
            break
        improved = False
        for step in [0.01, 0.02, 0.005, 0.04, 0.002, 0.008, 0.015, 0.025]:
            ht = project_feasible(h - step * g_avg / gn, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-15:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            # Try smaller steps
            for step in [0.001, 0.003, 0.0005, 0.002]:
                ht = project_feasible(h - step * g_avg / gn, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-15:
                    h = ht
                    best_c5 = cn
                    improved = True
                    break
            if not improved:
                break
    return h, best_c5


def lp_active_step(h, n, K, delta):
    """LP-based step: linearize and solve for direction that reduces max."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    idx = np.argsort(-corr)[:K]
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub_list = []
    b_ub_list = []
    for k in idx:
        g = grad_corr_k(h, k, n)
        fk = corr[k]
        row = g.copy()
        row = np.concatenate([row, [-1.0]])
        A_ub_list.append(row)
        b_ub_list.append(-fk)
    
    # Trust region
    I = np.eye(n)
    A_trust = np.vstack([
        np.hstack([I, np.zeros((n, 1))]),
        np.hstack([-I, np.zeros((n, 1))])
    ])
    b_trust = np.concatenate([delta * np.ones(n), delta * np.ones(n)])
    
    A_ub = np.vstack([np.array(A_ub_list), A_trust])
    b_ub = np.concatenate([np.array(b_ub_list), b_trust])
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, c_max


def subgrad_with_bb(h, n, time_limit, rng):
    """Subgradient descent with Barzilai-Borwein step size."""
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    it = 0
    h_prev = h.copy()
    g_prev = None
    x_diff_prev = None
    g_diff_prev = None
    
    while time.time() - start < time_limit:
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        progress = min(1.0, it / 300.0)
        K = 30 if progress < 0.3 else (20 if progress < 0.6 else 12)
        beta = 30.0 if progress < 0.3 else (60.0 if progress < 0.6 else 120.0)
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(-corr)[:K]
        sh = corr[idx] - corr[idx[0]]
        cl = np.clip(beta * sh, -60.0, 60.0)
        ex = np.exp(cl)
        w = ex / ex.sum()
        
        g = np.zeros(n)
        for i in range(K):
            if w[i] < 1e-13:
                continue
            g += w[i] * grad_corr_k(h, idx[i], n)
        
        g_n = np.linalg.norm(g)
        if g_n < 1e-14:
            break
        
        # BB step size
        if g_prev is not None and x_diff_prev is not None:
            num = np.dot(x_diff_prev, g_diff_prev)
            den = np.dot(g_diff_prev, g_diff_prev)
            if den > 1e-14:
                step = num / den
                step = np.clip(step, 0.001, 0.3)
            else:
                step = 0.02
        else:
            step = 0.02
        
        h_new = project_feasible(h - step * g, n)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            x_diff = h_new - h
            h = h_new
            g_diff = g - g_prev
            x_diff_prev = x_diff
            g_prev = g
            g_diff_prev = g_diff
        else:
            # Backtracking
            for factor in [0.5, 0.25, 0.1, 0.05]:
                h_try = project_feasible(h - step * factor * g, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    x_diff = h_try - h
                    h = h_try
                    g_diff = g - g_prev
                    x_diff_prev = x_diff
                    g_prev = g
                    g_diff_prev = g_diff
                    break
            else:
                if g_prev is not None:
                    x_diff_prev = None
                    g_diff_prev = None
        
        it += 1
    
    return best_h, best_c5


def fine_polish(h, n, max_iters):
    """Very fine polishing using precise line search on top-K average gradient."""
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        rem = _BUDGET - (time.time() - _T0)
        if rem < 3:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max_val = corr.max()
        # Find shifts within 1% of max
        threshold = c_max_val - 0.005 * c_max_val
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.argsort(-corr)[:5]
        if len(active) > 15:
            active = np.argsort(-corr)[:15]
        
        g = np.zeros(n)
        for k in active:
            g += grad_corr_k(h, k, n)
        g /= len(active)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        
        # Fine line search
        improved = False
        for step in [0.003, 0.001, 0.005, 0.002, 0.0005, 0.004, 0.0015, 0.0008]:
            h_try = project_feasible(h - step * g / gn, n)
            c_try = compute_c5(h_try)
            if c_try < best_c5 - 1e-16:
                h = h_try
                best_c5 = c_try
                improved = True
                break
        if not improved:
            # Try even smaller
            for step in [0.0003, 0.0001, 0.0002, 0.0005]:
                h_try = project_feasible(h - step * g / gn, n)
                c_try = compute_c5(h_try)
                if c_try < best_c5 - 1e-16:
                    h = h_try
                    best_c5 = c_try
                    improved = True
                    break
            if not improved:
                break
    return h, best_c5


def perturb_and_optimize(h, n, rng, time_limit):
    """Perturb h slightly and re-optimize to try to escape local minima."""
    best = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    
    for trial in range(8):
        if time.time() - start > time_limit or _BUDGET - (time.time() - _T0) < 10:
            break
        # Random perturbation
        noise = rng.normal(0, 0.01, n)
        h_p = project_feasible(h + noise, n)
        
        # Quick optimization
        h_p = slsqp_smmax(h_p, n, beta=100, K=30, maxiter=150)
        c5_p = compute_c5(h_p)
        
        if c5_p < best_c5:
            best = h_p.copy()
            best_c5 = c5_p
            h = h_p.copy()
    
    return best, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _G_N, _G_BETA, _G_K, _G_KLIST
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 512
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: Subgradient with BB step (quick improvement)
    if remaining() > 40:
        h_sg, c5_sg = subgrad_with_bb(h, n, min(40.0, remaining() * 0.12), rng)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 2: Multi-stage smooth-max SLSQP
    beta_schedule = [20, 50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000]
    for beta in beta_schedule:
        if remaining() < 15:
            break
        maxit = 800 if beta < 200 else (500 if beta < 2000 else 300)
        K_use = 50 if beta < 1000 else (40 if beta < 10000 else 30)
        h_new = slsqp_smmax(h, n, beta=beta, K=K_use, maxiter=maxit)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 3: Active-set min-max SLSQP
    if remaining() > 40:
        for K_active in [5, 8, 10, 15, 20, 30, 40]:
            if remaining() < 15:
                break
            corr_a = np.correlate(h, 1.0 - h, mode='full')
            idx_a = np.argsort(-corr_a)[:K_active]
            h_am, c5_am = slsqp_tminmax(h, n, idx_a, maxiter=400)
            if c5_am < best_c5:
                best_c5 = c5_am
                best_h = h_am.copy()
                h = h_am.copy()
            # Refresh active set
            corr_a = np.correlate(h, 1.0 - h, mode='full')
            idx_a = np.argsort(-corr_a)[:K_active]

    # Phase 4: Equalize top shifts
    if remaining() > 30:
        for K_eq in [5, 8, 10, 15]:
            if remaining() < 10:
                break
            h_eq, c5_eq = equalize_shifts(h, n, K=K_eq, max_iters=300)
            if c5_eq < best_c5:
                best_c5 = c5_eq
                best_h = h_eq.copy()
                h = h_eq.copy()

    # Phase 5: LP-based active set refinement
    if remaining() > 50:
        for it in range(200):
            if remaining() < 5:
                break
            h_lp, t_lp = lp_active_step(h, n, K=20, delta=0.03)
            if h_lp is not None:
                for alpha in [1.0, 0.7, 0.5, 0.3, 0.8]:
                    h_try = project_feasible(h + alpha * h_lp, n)
                    c_try = compute_c5(h_try)
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        break
                else:
                    if it > 100:
                        break

    # Phase 6: SLSQP polish after LP
    for beta in [10000, 30000, 80000]:
        if remaining() < 10:
            break
        h = slsqp_smmax(h, n, beta=beta, K=40, maxiter=500)
        c5_tmp = compute_c5(h)
        if c5_tmp < best_c5:
            best_c5 = c5_tmp
            best_h = h.copy()
            h = h.copy()

    # Phase 7: Fine polish
    if remaining() > 15:
        h_f, c5_f = fine_polish(h, n, max_iters=500)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            h = h_f.copy()

    # Phase 8: Perturbation restarts
    if remaining() > 60:
        h_p, c5_p = perturb_and_optimize(h, n, rng, min(60.0, remaining() * 0.3))
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()
            h = h_p.copy()
            # Re-polish after perturbation
            if remaining() > 20:
                h = slsqp_smmax(h, n, beta=10000, K=40, maxiter=300)
                c5_r = compute_c5(h)
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = h.copy()
                if remaining() > 10:
                    h_f2, c5_f2 = fine_polish(h, n, max_iters=200)
                    if c5_f2 < best_c5:
                        best_c5 = c5_f2
                        best_h = h_f2.copy()

    # Phase 9: Try higher resolution n=1024
    n_high = 1024
    if remaining() > 150:
        h_interp = interp_to(best_h, best_n, n_high)
        h_interp = project_feasible(h_interp, n_high)
        c5_interp = compute_c5(h_interp)

        if c5_interp <= best_c5 + 0.002:
            h_h = h_interp.copy()
            # Subgradient at high res
            if remaining() > 80:
                h_h, c5_h = subgrad_with_bb(h_h, n_high, min(30.0, remaining() * 0.15), rng)
                if compute_c5(h_h) > c5_interp:
                    h_h = h_interp.copy()
            
            # SLSQP at high res
            for beta_h in [50, 200, 1000, 5000, 15000]:
                if remaining() < 20:
                    break
                h_h = slsqp_smmax(h_h, n_high, beta=beta_h, K=40, maxiter=300)
            
            # Active set at high res
            if remaining() > 40:
                for K_a in [8, 12, 20]:
                    if remaining() < 15:
                        break
                    corr_h = np.correlate(h_h, 1.0 - h_h, mode='full')
                    idx_h = np.argsort(-corr_h)[:K_a]
                    h_h, c5_h = slsqp_tminmax(h_h, n_high, idx_h, maxiter=250)
            
            # Equalize at high res
            if remaining() > 20:
                h_h, c5_h = equalize_shifts(h_h, n_high, K=8, max_iters=200)
            
            # Fine polish at high res
            if remaining() > 15:
                h_h, c5_h = fine_polish(h_h, n_high, max_iters=300)
            
            c5_h_final = compute_c5(h_h)
            if c5_h_final < best_c5:
                best_c5 = c5_h_final
                best_h = h_h.copy()
                best_n = n_high

    # Phase 10: Final intensive refinement
    n_final = best_n
    h_final = best_h.copy()
    if remaining() > 30:
        # More SLSQP with very high beta
        for beta_f in [50000, 100000, 200000]:
            if remaining() < 8:
                break
            h_final = slsqp_smmax(h_final, n_final, beta=beta_f, K=50, maxiter=400)
            c5_f = compute_c5(h_final)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_final.copy()

        # Active set final
        if remaining() > 20:
            for K_f in [5, 8, 12, 16, 20]:
                if remaining() < 8:
                    break
                    corr_f = np.correlate(h_final, 1.0 - h_final, mode='full')
                    idx_f = np.argsort(-corr_f)[:K_f]
                    h_final, c5_f = slsqp_tminmax(h_final, n_final, idx_f, maxiter=300)
                    if c5_f < best_c5:
                        best_c5 = c5_f
                        best_h = h_final.copy()

        # Final fine polish
        if remaining() > 10:
            h_final, c5_f = fine_polish(best_h, n_final, max_iters=1000)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_final.copy()

    # Phase 11: ULTRA fine steps at the very end
    if remaining() > 5:
        h_u = best_h.copy()
        for it in range(200):
            if remaining() < 2:
                break
            corr_u = np.correlate(h_u, 1.0 - h_u, mode='full')
            c_max_u = corr_u.max()
            threshold_u = c_max_u - 0.002 * c_max_u
            active_u = np.where(corr_u >= threshold_u)[0]
            if len(active_u) > 20:
                active_u = np.argsort(-corr_u)[:20]
            if len(active_u) == 0:
                active_u = np.argsort(-corr_u)[:5]
            
            g_u = np.zeros(n_final)
            for k in active_u:
                g_u += grad_corr_k(h_u, k, n_final)
            g_u /= len(active_u)
            gn_u = np.linalg.norm(g_u)
            if gn_u < 1e-16:
                break
            
            improved = False
            for step in [0.001, 0.0005, 0.002, 0.0003, 0.0001, 0.0015, 0.0008, 0.0002]:
                h_try = project_feasible(h_u - step * g_u / gn_u, n_final)
                c_try = compute_c5(h_try)
                if c_try < best_c5 - 1e-17:
                    h_u = h_try
                    best_c5 = c_try
                    best_h = h_u.copy()
                    improved = True
                    break
            if not improved:
                break

    # Final feasibility enforcement
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
