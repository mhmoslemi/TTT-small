```python
def coord_descent_step(h, i, corr, n, n_trials=200):
    """Find optimal value for h[i] given other coordinates fixed."""
    # Compute gradient of corr wrt h[i]
    grad = np.zeros(2*n-1)
    s_val = np.arange(-n+1, n)
    m_val = np.arange(2*n-1)
    
    # First factor: valid when -i <= s < n-i
    s_first = s_val[(s_val >= -i) & (s_val < n-i)]
    m_first = s_first + n - 1
    grad[m_first] += (1.0 - h[i + s_first])
    
    # Second factor: valid when i-n < s <= i, i.e., i-s in [0, n)
    s_second = s_val[(s_val > i-n) & (s_val <= i)]
    m_second = s_second + n - 1
    grad[m_second] += -h[i - s_second]
    
    # Now find optimal x
    x_trials = np.linspace(0, 1, n_trials)
    delta = x_trials - h[i]  # shape (n_trials,)
    
    # Linear part: corr + delta * grad, shape (n_trials, 2n-1)
    new_corr = corr[None, :] + delta[:, None] * grad[None, :]
    
    # Quadratic correction at m = n-1
    new_corr[:, n-1] -= delta**2
    
    # Objective: max over m
    obj = np.max(new_corr, axis=1)
    
    best_idx = np.argmin(obj)
    return x_trials[best_idx]
```

```python
        df = x_fine - h[i]
        nc = corr[None, :] + df[:, None] * grad[None, :]
        nc[:, n - 1] -= df ** 2
        obj_f = np.max(nc, axis=1)
        x_best = x_fine[np.argmin(obj_f)]
    return float(np.clip(x_best, 0.0, 1.0))

def update_corr_inplace(corr, h, i, new_val, n):
    delta = new_val - h[i]
    if abs(delta) < 1e-16:
        return
    s1 = np.arange(-i, n - i)
    m1 = n - 1 + s1
    corr[m1] += delta * (1.0 - h[i + s1])
    s2 = np.arange(i - n + 1, i + 1)
    m2 = n - 1 + s2
    corr[m2] += -delta * h[i - s2]
    corr[n - 1] -= delta ** 2
    h[i] = new_val

def coord_descent_sweep(h, n, corr, n_trials=400, rng=None):
    h = h.copy()
    if rng is not None:
        idx_order = rng.permutation(n)
    else:
        idx_order = np.arange(n)
    for i in idx_order:
        x_new = coord_update(h, i, corr, n, n_trials)
        if abs(x_new - h[i]) > 1e-15:
            update_corr_inplace(corr, h, i, x_new, n)
    h = project_feasible(h, n)
    corr = compute_corr(h)
    return h, corr

def _smmax_obj(hv):
    n = _G_N
    corr = np.correlate(hv, 1.0 - hv, mode='full')
    beta = _G_BETA
    K = _G_K
    idx = np.argsort(-corr)[:K]
    c = corr[idx]
    sh = c - c.max()
    cl = np.clip(beta * sh, -80.0, 80.0)
    ex = np.exp(cl)
    return (c.max() + np.log(ex.sum()) / beta) * (2.0 / n)

def _smmax_grad(hv):
    n = _G_N
    beta = _G_BETA
    K = _G_K
    corr = np.correlate(hv, 1.0 - hv, mode='full')
    idx = np.argsort(-corr)[:K]
    c = corr[idx]
    sh = c - c.max()
    cl = np.clip(beta * sh, -80.0, 80.0)
    ex = np.exp(cl)
    w = ex / ex.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(K):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        k = idx[ii]
        offset = k - (n - 1)
        if offset >= 0:
            np_ = n - offset
            g[offset:] += wk * (1.0 - hv[:np_])
            g[:np_] -= wk * hv[offset:]
        else:
            a = -offset
            np_ = n - a
            g[:np_] += wk * (1.0 - hv[a:])
            g[a:] -= wk * hv[:np_]
    return g

def _sum_eq_fun(hv):
    return np.sum(hv) - _G_N / 2.0

def _sum_eq_jac(hv):
    return np.ones(_G_N)

def slsqop_smmax(h, n, beta, K, maxiter):
    global _G_N, _G_BETA, _G_K
    _G_N = n
    _G_BETA = beta
    _G_K = K
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq_fun, 'jac': _sum_eq_jac}]
    try:
        res = minimize(_smmax_obj, h, jac=_smmax_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            c5_old = compute_c5(h)
            if c5_new <= c5_old + 1e-14:
                return h_new
    except Exception:
        pass
    return h.copy()

def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def equalize_top_shifts(h, n, K, max_iters):
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:K]
        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        if c_max - c_min < 1e-12:
            break
        g_avg = np.zeros(n)
        for k in idx:
            offset = k - (n - 1)
            if offset >= 0:
                np_ = n - offset
                g_avg[offset:] += (1.0 - h[:np_])
                g_avg[:np_] -= h[offset:]
            else:
                a = -offset
                np_ = n - a
                g_avg[:np_] += (1.0 - h[a:])
                g_avg[a:] -= h[:np_]
        g_avg /= K
        g_dir = -g_avg
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        g_dir /= gn
        improved = False
        for step in [0.02, 0.04, 0.01, 0.005, 0.06, 0.003, 0.03, 0.08, 0.015]:
            ht = project_feasible(h + step * g_dir, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-15:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5

def active_set_slsqp(h, n, K_list, maxiter):
    global _G_N, _G_K
    _G_N = n
    _G_K = K_list
    t0 = compute_c5(h)
    
    def tmin_ineq_fun(v):
        mat = v[:n]
        tt = v[-1]
        cc = np.correlate(mat, 1.0 - mat, mode='full')
        return cc[K_list] - tt * (n / 2.0)
    
    def tmin_ineq_jac(v):
        mat = v[:n]
        J = np.zeros((K_list, n + 1))
        for ii in range(K_list):
            k = K_list[ii]
            offset = k - (n - 1)
            if offset >= 0:
                np_ = n - offset
                J[ii, offset:] += (1.0 - mat[:np_])
                J[ii, :np_] -= mat[offset:]
            else:
                a = -offset
                np_ = n - a
                J[ii, :np_] += (1.0 - mat[a:])
                J[ii, a:] -= mat[:np_]
            J[ii, -1] = -n / 2.0
        return J
    
    def tmin_ineq_fun_neg(v):
        return -tmin_ineq_fun(v)
    
    def tmin_ineq_jac_neg(v):
        return -tmin_ineq_jac(v)
    
    def obj_t(v):
        return v[-1]
    
    def obj_t_jac(v):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def sum_eq2(v):
        return np.sum(v[:n]) - n / 2.0
    
    def sum_eq2_jac(v):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    v0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'ineq', 'fun': tmin_ineq_fun_neg, 'jac': tmin_ineq_jac_neg},
        {'type': 'eq', 'fun': sum_eq2, 'jac': sum_eq2_jac}
    ]
    try:
        res = minimize(obj_t, v0, jac=obj_t_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x[:n], n)
            c5_new = compute_c5(h_new)
            if c5_new <= compute_c5(h) + 1e-14:
                return h_new, c5_new
    except Exception:
        pass
    return h.copy(), compute_c5(h)

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    
    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 512
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    
    n = best_n
    h = best_h.copy()
    
    # Phase 1: Quick SLSQP polish at original resolution
    if remaining() > 30:
        h = slsqop_smmax(h, n, beta=500, K=30, maxiter=300)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Phase 2: Upscale to n=1024
    n1 = 1024
    h1 = interp_to(h, n, n1)
    h1 = project_feasible(h1, n1)
    c5_1 = compute_c5(h1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
        best_n = n1
    
    # Phase 3: Coordinate descent at n=1024 (main workhorse)
    corr1 = compute_corr(h1)
    n_trials = 400
    n_sweeps_target = 300
    sweep_start = time.time()
    for sweep in range(n_sweeps_target):
        if remaining() < 60:
            break
        h1, corr1 = coord_descent_sweep(h1, n1, corr1, n_trials=n_trials, rng=rng)
        c5_1 = compute_c5(h1)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()
            best_n = n1
        # Periodic SLSQP polish
        if (sweep + 1) % 40 == 0 and remaining() > 80:
            h1 = slsqop_smmax(h1, n1, beta=3000, K=40, maxiter=200)
            c5_1 = compute_c5(h1)
            if c5_1 < best_c5:
                best_c5 = c5_1
                best_h = h1.copy()
                best_n = n1
            corr1 = compute_corr(h1)
    
    # Phase 4: Equalize and active-set at n=1024
    if remaining() > 40:
        h1, c5_eq = equalize_top_shifts(h1, n1, K=10, max_iters=200)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h1.copy()
            best_n = n1
    
    if remaining() > 30:
        corr1 = compute_corr(h1)
        for K_a in [8, 12, 16, 20]:
            if remaining() < 10:
                break
            idx_a = np.argsort(-corr1)[:K_a]
            h1, c5_a = active_set_slsqp(h1, n1, K_a, maxiter=200)
            if c5_a < best_c5:
                best_c5 = c5_a
                best_h = h1.copy()
                best_n = n1
            corr1 = compute_corr(h1)
    
    # Phase 5: SLSQP polish at n=1024
    if remaining() > 20:
        for beta in [5000, 15000, 30000]:
            if remaining() < 10:
                break
            h1 = slsqop_smmax(h1, n1, beta=beta, K=50, maxiter=300)
            c5_1 = compute_c5(h1)
            if c5_1 < best_c5:
                best_c5 = c5_1
                best_h = h1.copy()
                best_n = n1
    
    # Phase 6: Try n=2048
    n2 = 2048
    if remaining() > 150:
        h2 = interp_to(h1, n1, n2)
        h2 = project_feasible(h2, n2)
        c5_2 = compute_c5(h2)
        if c5_2 <= best_c5 + 0.002:
            corr2 = compute_corr(h2)
            n_trials_2 = 300
            for sweep in range(150):
                if remaining() < 60:
                    break
                h2, corr2 = coord_descent_sweep(h2, n2, corr2, n_trials=n_trials_2, rng=rng)
                c5_2 = compute_c5(h2)
                if c5_2 < best_c5:
                    best_c5 = c5_2
                    best_h = h2.copy()
                    best_n = n2
                if (sweep + 1) % 30 == 0 and remaining() > 80:
                    h2 = slsqop_smmax(h2, n2, beta=5000, K=40, maxiter=200)
                    c5_2 = compute_c5(h2)
                    if c5_2 < best_c5:
                        best_c5 = c5_2
                        best_h = h2.copy()
                        best_n = n2
                    corr2 = compute_corr(h2)
            if remaining() > 30:
                h2, c5_eq2 = equalize_top_shifts(h2, n2, K=10, max_iters=150)
                if c5_eq2 < best_c5:
                    best_c5 = c5_eq2
                    best_h = h2.copy()
                    best_n = n2
            if remaining() > 20:
                for beta in [10000, 20000]:
                    if remaining() < 10:
                        break
                    h2 = slsqop_smmax(h2, n2, beta=beta, K=50, maxiter=200)
                    c5_2 = compute_c5(h2)
                    if c5_2 < best_c5:
                        best_c5 = c5_2
                        best_h = h2.copy()
                        best_n = n2
    
    # Phase 7: More coordinate descent at best resolution with finer grid
    n_best = best_n
    h_best = best_h.copy()
    if remaining() > 30:
        corr_b = compute_corr(h_best)
        n_trials_fine = 600
        for sweep in range(100):
            if remaining() < 10:
                break
            h_best, corr_b = coord_descent_sweep(h_best, n_best, corr_b, n_trials=n_trials_fine, rng=rng)
            c5_b = compute_c5(h_best)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_best.copy()
    
    # Phase 8: Final fine equalization
    if remaining() > 15:
        h_f, c5_f = equalize_top_shifts(best_h, best_n, K=15, max_iters=300)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    
    # Phase 9: Tiny-step refinement
    if remaining() > 10:
        h_t = best_h.copy()
        for it in range(200):
            if remaining() < 3:
                break
            corr_t = compute_corr(h_t)
            idx_t = np.argsort(-corr_t)[:20]
            g_t = np.zeros(best_n)
            for k in idx_t:
                offset = k - (best_n - 1)
                if offset >= 0:
                    np_ = best_n - offset
                    g_t[offset:] += (1.0 - h_t[:np_])
                    g_t[:np_] -= h_t[offset:]
                else:
                    a = -offset
                    np_ = best_n - a
                    g_t[:np_] += (1.0 - h_t[a:])
                    g_t[a:] -= h_t[:np_]
            g_t /= len(idx_t)
            gn_t = np.linalg.norm(g_t)
            if gn_t < 1e-14:
                break
            for step in [0.003, 0.001, 0.0005, 0.005, 0.002, 0.0015]:
                h_try = project_feasible(h_t - step * g_t / gn_t, best_n)
                c_try = compute_c5(h_try)
                if c_try < best_c5:
                    best_c5 = c_try
                    best_h = h_try.copy()
                    h_t = h_try.copy()
                    break
    
    # Final feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
```
