def grad_corr_k_vec(h, k, n):
    """Gradient of the k-th correlation value with respect to h."""

    return g

    modes = rng.choice(np.arange(start, end), size=min(n_modes, end - start), replace=False)

        else:

    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        if remaining_time() < 3:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max_val = np.max(corr) * (2.0 / n)
        if c_max_val < best_c5 - 1e-12:
            best_c5 = c_max_val
            h = h.copy()
        # Get gradient of the max correlation
        k_max = np.argmax(corr)
        g = grad_corr_k_vec(h, k_max, n)
        # Also consider top few
        idx_top = np.argsort(-corr)[:5]
        g_avg = np.zeros(n)
        for k in idx_top:
            g_avg += grad_corr_k_vec(h, k, n)
        g_avg /= len(idx_top)

        # Try moves in direction of -g_avg and individual coordinates
        improved = False
        # Directional move
        for step in [0.005, 0.002, 0.001, 0.01]:
            h_try = project_feasible(h - step * g_avg, n)
            c_try = compute_c5(h_try)
            if c_try < best_c5:
                best_c5 = c_try
                h = h_try.copy()
                improved = True
                break
        if not improved:
            # Coordinate moves
            for trial in range(20):
                i = rng.randint(n)
                for delta in [0.01, -0.01, 0.02, -0.02, 0.005, -0.005]:
                    h_try = h.copy()
                    h_try[i] = np.clip(h_try[i] + delta, 0.0, 1.0)
                    h_try = project_feasible(h_try, n)
                    c_try = compute_c5(h_try)
                    if c_try < best_c5:
                        best_c5 = c_try
                        h = h_try.copy()
                        improved = True
                        break
                if improved:
                    break
        if not improved and it > 200:
            break
        it += 1
    return h, best_c5


def low_res_optimize(n_low, n_high, rng):
    """Optimize at lower resolution and upscale."""
    # Start with a simple initialization at low res
    h_low = np.full(n_low, 0.5)
    # Add some structure
    for i in range(n_low):
        h_low[i] = 0.5 + 0.3 * np.sin(2 * np.pi * i / n_low * 3) + 0.1 * np.sin(2 * np.pi * i / n_low * 7)
    h_low = project_feasible(h_low, n_low)

    # Optimize at low res with subgradient
    h_low, c5_low = subgradient_descent(h_low, n_low, 15.0, rng)

    # SLSQP at low res
    for beta in [50, 500, 5000]:
        if remaining_time() < 10:
            break
        h_low = slsqop_smmax(h_low, n_low, beta, K=30, maxiter=300)
    c5_low = compute_c5(h_low)

    # Active set at low res
    if remaining_time() > 15:
        corr = np.correlate(h_low, 1.0 - h_low, mode='full')
        idx = np.argsort(-corr)[:15]
        h_low, c5_low = slsqop_tminmax2(h_low, n_low, idx, maxiter=200)

    # Upscale
    h_high = interp_to(h_low, n_low, n_high)
    h_high = project_feasible(h_high, n_high)
    c5_high = compute_c5(h_high)

    return h_high, c5_high


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _G_N, _G_BETA, _G_K, _G_KLIST
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get initial solution from parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 512
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: Quick subgradient to enter the basin (short since we start good)
    if remaining_time() > 20:
        h_sg, c5_sg = subgradient_descent(h, n, min(15.0, remaining_time() * 0.06), rng)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 2: SLSQP smooth max with increasing beta
    beta_stages = [30, 100, 300, 1000, 3000, 10000, 30000]
    for beta in beta_stages:
        if remaining_time() < 12:
            break
        maxit = 500 if beta < 500 else 350
        h_new = slsqop_smmax(h, n, beta, K=50, maxiter=maxit)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 3: Active-set min-max with diverse cycling
    if remaining_time() > 60:
        for cycle in range(4):
            if remaining_time() < 15:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            idx_all = np.argsort(-corr)
            for K_active in [10, 20, 35]:
                if remaining_time() < 10:
                    break
                # Different offset each cycle to explore different subsets
                offset = (cycle * 5) % max(1, K_active - 5)
                start_idx = offset
                end_idx = min(len(idx_all), K_active + offset)
                if end_idx <= start_idx:
                    continue
                idx_active = idx_all[start_idx:end_idx]
                h_am, c5_am = slsqop_tminmax2(h, n, idx_active, maxiter=250)
                if c5_am < best_c5:
                    best_c5 = c5_am
                    best_h = h_am.copy()
                    h = h_am.copy()
                    corr = np.correlate(h, 1.0 - h, mode='full')
                    idx_all = np.argsort(-corr)

    # Phase 4: Multi-restart perturbation + re-optimization (key new phase)
    if remaining_time() > 80:
        n_restarts = 8
        for r in range(n_restarts):
            if remaining_time() < 20:
                break
            # Choose perturbation type
            pert_type = r % 4
            if pert_type == 0:
                # Small Gaussian
                h_pert = np.clip(h + 0.015 * rng.randn(n), 0, 1)
                h_pert = project_feasible(h_pert, n)
            elif pert_type == 1:
                # Frequency perturbation
                h_pert = freq_perturb(h, n, rng, n_modes=10, amp=0.05)
            elif pert_type == 2:
                # Swap perturbation
                h_pert = swap_perturb(h, n, rng, n // 8)
            else:
                # Medium Gaussian
                h_pert = np.clip(h + 0.03 * rng.randn(n), 0, 1)
                h_pert = project_feasible(h_pert, n)

            c5_pert = compute_c5(h_pert)
            # Only proceed if perturbation didn't destroy too much
            if c5_pert > best_c5 + 0.005:
                continue

            # Quick optimization from perturbed state
            h_opt, c5_opt = perturbed_optimize(h_pert, n, rng, 15.0)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                h = h_opt.copy()

    # Phase 5: Low-resolution exploration then upscale
    if remaining_time() > 80:
        h_lr, c5_lr = low_res_optimize(128, n, rng)
        c5_lr_up = compute_c5(h_lr)
        if c5_lr_up < best_c5 + 0.01:
            # Refine the upscaled solution
            h_lr = h_lr
            if remaining_time() > 30:
                h_lr, c5_lr2 = subgradient_descent(h_lr, n, min(12.0, remaining_time() * 0.2), rng)
            if remaining_time() > 15:
                h_lr = slsqop_smmax(h_lr, n, beta=5000, K=50, maxiter=300)
                c5_lr2 = compute_c5(h_lr)
            if remaining_time() > 10:
                corr_lr = np.correlate(h_lr, 1.0 - h_lr, mode='full')
                idx_lr = np.argsort(-corr_lr)[:25]
                h_lr, c5_lr2 = slsqop_tminmax2(h_lr, n, idx_lr, maxiter=200)
            c5_lr2 = compute_c5(h_lr)
            if c5_lr2 < best_c5:
                best_c5 = c5_lr2
                best_h = h_lr.copy()
                h = h_lr.copy()

    # Phase 6: Try higher resolution n=1024
    n_high = 1024
    if remaining_time() > 100:
        h_interp = interp_to(best_h, best_n, n_high)
        h_interp = project_feasible(h_interp, n_high)
        c5_interp = compute_c5(h_interp)

        if c5_interp <= best_c5 + 0.003:
            h_high = h_interp.copy()
            if remaining_time() > 40:
                h_high, c5_high = subgradient_descent(h_high, n_high, min(20.0, remaining_time() * 0.15), rng)
                if c5_high > c5_interp:
                    h_high = h_interp.copy()
                    c5_high = c5_interp
            for beta in [200, 2000, 10000, 30000]:
                if remaining_time() < 15:
                    break
                h_high = slsqop_smmax(h_high, n_high, beta, K=50, maxiter=300)
                c5_high = compute_c5(h_high)
            if remaining_time() > 25:
                corr_h = np.correlate(h_high, 1.0 - h_high, mode='full')
                for K_a in [15, 30, 45]:
                    if remaining_time() < 10:
                        break
                    idx_a = np.argsort(-np.correlate(h_high, 1.0 - h_high, mode='full'))[:K_a]
                    h_high, c5_high = slsqop_tminmax2(h_high, n_high, idx_a, maxiter=200)
            c5_high = compute_c5(h_high)
            if c5_high < best_c5:
                best_c5 = c5_high
                best_h = h_high.copy()
                best_n = n_high
                h = h_high.copy()
                n = n_high

    # Phase 7: Intensive refinement at best resolution
    n = best_n
    h = best_h.copy()
    for cycle in range(3):
        if remaining_time() < 25:
            break
        for beta in [10000, 30000, 80000]:
            if remaining_time() < 8:
                break
            h = slsqop_smmax(h, n, beta, K=60, maxiter=400)
            c5_t = compute_c5(h)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h.copy()
        if remaining_time() > 15:
            corr_c = np.correlate(h, 1.0 - h, mode='full')
            for K_a in [15, 30, 50]:
                if remaining_time() < 8:
                    break
                idx_a = np.argsort(-corr_c)[:K_a]
                h, c5_t = slsqop_tminmax2(h, n, idx_a, maxiter=300)
                if c5_t < best_c5:
                    best_c5 = c5_t
                    best_h = h.copy()
                    corr_c = np.correlate(h, 1.0 - h, mode='full')

    # Phase 8: Greedy coordinate descent for fine-tuning
    if remaining_time() > 20:
        h_gc, c5_gc = greedy_coordinate_descent(best_h, best_n, min(30.0, remaining_time() * 0.3), rng)
        if c5_gc < best_c5:
            best_c5 = c5_gc
            best_h = h_gc.copy()

    # Phase 9: Final SLSQP with very high beta
    h = best_h.copy()
    for beta in [50000, 150000, 300000]:
        if remaining_time() < 8:
            break
        h = slsqop_smmax(h, best_n, beta, K=80, maxiter=500)
    c5_f = compute_c5(h)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h.copy()

    # Phase 10: Ultra-fine equalization
    if remaining_time() > 15:
        h_eq = best_h.copy()
        best_eq = best_c5
        for it in range(300):
            if remaining_time() < 3:
                break
            corr = np.correlate(h_eq, 1.0 - h_eq, mode='full')
            idx = np.argsort(-corr)[:15]
            c_max = corr[idx[0]]
            c_min_top = corr[idx[-1]]
            if c_max - c_min_top < 1e-11:
                break
            g_avg = np.zeros(best_n)
            for k in idx:
                g_avg += grad_corr_k_vec(h_eq, k, best_n)
            g_avg /= len(idx)
            gn = np.linalg.norm(g_avg)
            if gn < 1e-14:
                break
            improved = False
            for step in [0.002, 0.001, 0.003, 0.0005, 0.005]:
                ht = project_feasible(h_eq - step * g_avg / gn * best_n, best_n)
                cn = compute_c5(ht)
                if cn < best_eq - 1e-14:
                    h_eq = ht
                    best_eq = cn
                    improved = True
                    break
            if not improved:
                # Try individual top-k descent
                k_top = idx[0]
                g_top = grad_corr_k_vec(h_eq, k_top, best_n)
                gn_t = np.linalg.norm(g_top)
                if gn_t > 1e-14:
                    for step in [0.002, 0.001, 0.0005]:
                        ht = project_feasible(h_eq - step * g_top / gn_t * best_n, best_n)
                        cn = compute_c5(ht)
                        if cn < best_eq - 1e-14:
                            h_eq = ht
                            best_eq = cn
                            improved = True
                            break
            if not improved:
                break
        if best_eq < best_c5:
            best_c5 = best_eq
            best_h = h_eq.copy()

    # Final: one more round of greedy on the very best
    if remaining_time() > 10:
        h_g2, c5_g2 = greedy_coordinate_descent(best_h, best_n, min(15.0, remaining_time() * 0.5), rng)
        if c5_g2 < best_c5:
            best_c5 = c5_g2
            best_h = h_g2.copy()

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
