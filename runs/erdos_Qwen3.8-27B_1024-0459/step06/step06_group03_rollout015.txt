    global _G_N, _G_BETA, _G_K
    _G_N = n
    _G_BETA = beta
    _G_K = K
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}]
    try:
        res = minimize(smmax_objective, h, jac=smmax_gradient, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            if compute_c5(h_new) <= compute_c5(h) + 1e-12:
                return h_new
    except Exception:
        pass
    return h.copy()


def slsqop_tminmax(h, n, K_list, maxiter):
    global _G_N, _G_KLIST
    _G_N = n
    _G_KLIST = np.array(K_list, dtype=int)
    t0 = compute_c5(h)
    v0 = np.concatenate([h, [t0 * n / 2.0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'ineq', 'fun': tmin_ineq_fun_global, 'jac': tmin_ineq_jac_global},
        {'type': 'eq', 'fun': sum_eq_fun2, 'jac': sum_eq_jac2}
    ]
    try:
        res = minimize(obj_t_fun, v0, jac=obj_t_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x[:n], n)
            c5_new = compute_c5(h_new)
            if c5_new <= compute_c5(h) + 1e-12:
                return h_new, c5_new
    except Exception:
        pass
    return h.copy(), compute_c5(h)


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def subgradient_step(h, n, K, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    sh = c_top - c_top.max()
    cl = np.clip(beta * sh, -60.0, 60.0)
    ex = np.exp(cl)
    w = ex / ex.sum()
    g = np.zeros(n)
    for i in range(K):
        if w[i] < 1e-12:
            continue
        g += w[i] * grad_corr_k_vec(h, idx[i], n)
    return g, np.max(corr) * (2.0 / n)


def subgradient_descent(h, n, time_limit, rng):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    it = 0
    # Anderson acceleration memory
    H_hist = [h.copy()]
    G_hist = [None]
    M = 5  # Anderson memory

    while time.time() - start < time_limit:
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        progress = min(1.0, it / 400.0)
        K = 25 if progress < 0.3 else (18 if progress < 0.6 else 12)
        beta = 50.0 if progress < 0.3 else (80.0 if progress < 0.6 else 150.0)
        g, c_max = subgradient_step(h, n, K, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g = g / gn
        # Backtracking line search
        base_step = 0.05 * n / (1.0 + 0.02 * it)
        base_step = min(base_step, 0.12)
        improved = False
        for factor in [1.0, 0.6, 0.4, 0.25, 0.15, 0.1, 0.05]:
            h_new = project_feasible(h - base_step * factor * g, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                h = h_new
                best_h = h_new.copy()
                best_c5 = c5_new
                improved = True
                break
        if not improved:
            # Try very small steps
            for step in [0.005, 0.002, 0.001]:
                h_new = project_feasible(h - step * g, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    h = h_new
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    break
        it += 1
    return best_h, best_c5


def active_set_newton(h, n, K, max_iters, tol):
    """Newton-type refinement on active set to equalize top K correlations."""
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(-corr)[:K]
        c_active = corr[idx]
        c_spread = c_active.max() - c_active.min()
        if c_spread < tol:
            break
        # Compute gradient of each active correlation
        grads = []
        for k in idx:
            grads.append(grad_corr_k_vec(h, k, n))
        grads = np.array(grads)  # shape (K, n)
        # We want to move h in direction d s.t.:
        #   grads[i] . d = -(c_active[i] - c_mean)  for all i
        # where c_mean = mean(c_active)
        # Also: sum(d) = 0 (preserve constraint)
        c_mean = c_active.mean()
        targets = -(c_active - c_mean)
        # Build system: [grads; ones] d = [targets; 0]
        A = np.vstack([grads, np.ones((1, n))])
        b = np.concatenate([targets, [0.0]])
        # Solve least squares (overdetermined if K+1 < n)
        try:
            d, _, _, _ = np.linalg.lstsq(A, b, rcond=None)
        except Exception:
            break
        dn = np.linalg.norm(d)
        if dn < 1e-15:
            break
        d = d / dn
        # Line search
        improved = False
        for alpha in [1.0, 0.5, 0.25, 0.125, 0.625, 0.3125, 0.0625]:
            h_new = project_feasible(h + alpha * d, n)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-16:
                h = h_new
                best_c5 = c_new
                improved = True
                break
        if not improved:
            # Try negative direction
            for alpha in [0.5, 0.25, 0.125]:
                h_new = project_feasible(h - alpha * d, n)
                c_new = compute_c5(h_new)
                if c_new < best_c5 - 1e-16:
                    h = h_new
                    best_c5 = c_new
                    improved = True
                    break
        if not improved and it > 5:
            break
    return h, best_c5


def equalize_top_shifts(h, n, K, max_iters):
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(-corr)[:K]
        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        if c_max - c_min < 1e-12:
            break
        g_avg = np.zeros(n)
        for k in idx:
            g_avg += grad_corr_k_vec(h, k, n)
        g_avg /= K
        g_dir = -g_avg
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        g_dir /= gn
        improved = False
        for step in [0.03, 0.05, 0.015, 0.008, 0.04, 0.004, 0.02, 0.06, 0.01]:
            ht = project_feasible(h + step * g_dir, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-15:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            # Try opposite
            for step in [0.03, 0.015, 0.008]:
                ht = project_feasible(h - step * g_dir, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-15:
                    h = ht
                    best_c5 = cn
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5


def fine_polish(h, n, max_iters):
    """Very fine polishing with adaptive step sizes."""
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        rem = _BUDGET - (time.time() - _T0)
        if rem < 3:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(-corr)[:20]
        # Weighted gradient based on correlation values
        c_top = corr[idx]
        c_max = c_top.max()
        # Weights proportional to how much each is above the min of top-K
        c_min_top = c_top.min()
        if c_max - c_min_top < 1e-14:
            # Try reducing the common level
            w = np.ones(len(idx)) / len(idx)
        else:
            w = (c_top - c_min_top + 1e-10)
            w = w / w.sum()
        g = np.zeros(n)
        for i in range(len(idx)):
            if w[i] < 1e-14:
                continue
            g += w[i] * grad_corr_k_vec(h, idx[i], n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g = -g / gn
        improved = False
        for step in [0.01, 0.005, 0.002, 0.001, 0.02, 0.008, 0.004, 0.003]:
            h_new = project_feasible(h + step * g, n)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-16:
                h = h_new
                best_c5 = c_new
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def random_perturbation(h, n, rng, strength):
    """Add small random perturbation to escape local minima."""
    noise = rng.randn(n) * strength
    h_new = project_feasible(h + noise, n)
    return h_new


def perturb_and_optimize(h, n, rng, time_limit):
    """Try random perturbations followed by quick optimization."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    attempts = 0
    while time.time() - start < time_limit and attempts < 30:
        strength = 0.01 * (1.0 + 0.5 * np.exp(-attempts / 10.0))
        h_pert = random_perturbation(best_h, n, rng, strength)
        # Quick SLSQP polish
        c5_pert = compute_c5(h_pert)
        if c5_pert < best_c5:
            best_h = h_pert.copy()
            best_c5 = c5_pert
        # Brief subgradient
        h_sg, c5_sg = subgradient_descent(h_pert, n, min(3.0, time_limit * 0.1), rng)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
        # Brief active set
        if time.time() - start < time_limit * 0.8:
            h_as, c5_as = active_set_newton(h_sg, n, K=15, max_iters=50, tol=1e-10)
            if c5_as < best_c5:
                best_c5 = c5_as
                best_h = h_as.copy()
        attempts += 1
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _G_N, _G_BETA, _G_K, _G_KLIST
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 512
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: Subgradient basin finding
    if remaining() > 40:
        sg_time = min(40.0, remaining() * 0.08)
        h_sg, c5_sg = subgradient_descent(h, n, sg_time, rng)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 2: Progressive smooth-max SLSQP
    beta_stages = [20, 60, 200, 600, 2000, 6000, 15000, 40000, 100000]
    for beta in beta_stages:
        if remaining() < 12:
            break
        K_use = 50 if beta < 1000 else 40
        maxit = 800 if beta < 500 else (500 if beta < 5000 else 400)
        h_new = slsqop_smmax(h, n, beta, K=K_use, maxiter=maxit)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        if remaining() < 60:
            break

    # Phase 3: Active-set min-max SLSQP
    if remaining() > 40:
        for K_active in [8, 12, 16, 20, 28, 40]:
            if remaining() < 12:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            idx_active = np.argsort(-corr)[:K_active]
            h_am, c5_am = slsqop_tminmax(h, n, idx_active, maxiter=400)
            if c5_am < best_c5:
                best_c5 = c5_am
                best_h = h_am.copy()
                h = h_am.copy()

    # Phase 4: Newton active-set refinement
    if remaining() > 30:
        h_ns, c5_ns = active_set_newton(h, n, K=15, max_iters=100, tol=1e-11)
        if c5_ns < best_c5:
            best_c5 = c5_ns
            best_h = h_ns.copy()
            h = h_ns.copy()

    # Phase 5: Equalize top shifts
    if remaining() > 25:
        h_eq, c5_eq = equalize_top_shifts(h, n, K=10, max_iters=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 6: SLSQP polish at very high beta
    for beta in [20000, 50000, 100000, 200000]:
        if remaining() < 10:
            break
        h = slsqop_smmax(h, n, beta, K=60, maxiter=600)
        c5_tmp = compute_c5(h)
        if c5_tmp < best_c5:
            best_c5 = c5_tmp
            best_h = h.copy()

    # Phase 7: Multi-resolution - try n=1024
    n_high = 1024
    if remaining() > 200:
        h_interp = interp_to(best_h, best_n, n_high)
        h_interp = project_feasible(h_interp, n_high)
        c5_interp = compute_c5(h_interp)

        if c5_interp <= best_c5 + 0.002:
            h_h = h_interp.copy()
            # Subgradient at high res
            if remaining() > 80:
                h_h, c5_h = subgradient_descent(h_h, n_high, min(30.0, remaining() * 0.12), rng)
                if c5_h > c5_interp:
                    h_h = h_interp.copy()
                    c5_h = c5_interp

            # SLSQP at high res
            for beta in [100, 500, 2000, 10000, 50000]:
                if remaining() < 20:
                    break
                h_h = slsqop_smmax(h_h, n_high, beta, K=50, maxiter=400)
                c5_h = compute_c5(h_h)

            # Active set at high res
            if remaining() > 50:
                corr_h = np.correlate(h_h, 1.0 - h_h, mode='full')
                for K_a in [10, 20, 30]:
                    if remaining() < 15:
                        break
                    idx_a = np.argsort(-np.correlate(h_h, 1.0 - h_h, mode='full'))[:K_a]
                    h_h, c5_h = slsqop_tminmax(h_h, n_high, idx_a, maxiter=300)

            # Newton refinement at high res
            if remaining() > 30:
                h_h, c5_h = active_set_newton(h_h, n_high, K=20, max_iters=80, tol=1e-11)

            # Equalize at high res
            if remaining() > 20:
                h_h, c5_h = equalize_top_shifts(h_h, n_high, K=10, max_iters=200)

            # Fine polish
            c5_h = compute_c5(h_h)
            c5_h = min(c5_h, c5_interp)
            if c5_h < best_c5:
                best_c5 = c5_h
                best_h = h_h.copy()
                best_n = n_high

    # Phase 8: Also try n=256 for potentially different local optimum
    n_low = 256
    if remaining() > 150:
        h_low = interp_to(best_h, best_n, n_low)
        h_low = project_feasible(h_low, n_low)
        c5_low = compute_c5(h_low)
        h_l = h_low.copy()
        # Optimize at low res
        for beta in [100, 1000, 10000, 50000]:
            if remaining() < 15:
                break
            h_l = slsqop_smmax(h_l, n_low, beta, K=30, maxiter=400)
        if remaining() > 30:
            h_l, c5_l = active_set_newton(h_l, n_low, K=12, max_iters=100, tol=1e-11)
        if remaining() > 20:
            h_l, c5_l = equalize_top_shifts(h_l, n_low, K=8, max_iters=200)
        c5_l = compute_c5(h_l)
        # Interpolate back and compare
        h_back = interp_to(h_l, n_low, best_n)
        h_back = project_feasible(h_back, best_n)
        c5_back = compute_c5(h_back)
        if c5_back < best_c5:
            best_c5 = c5_back
            best_h = h_back.copy()

    # Phase 9: Intensive refinement cycles at best resolution
    n = best_n
    h = best_h.copy()
    for cycle in range(4):
        if remaining() < 40:
            break
        # Subgradient
        if remaining() > 20:
            h, c5_t = subgradient_descent(h, n, min(20.0, remaining() * 0.12), rng)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h.copy()

        # SLSQP
        if remaining() > 20:
            for beta in [5000, 20000, 80000]:
                if remaining() < 10:
                    break
                h = slsqop_smmax(h, n, beta, K=60, maxiter=500)
            c5_t = compute_c5(h)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h.copy()

        # Active set Newton
        if remaining() > 20:
            h, c5_t = active_set_newton(h, n, K=20, max_iters=100, tol=1e-12)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h.copy()

        # Equalize
        if remaining() > 15:
            h, c5_t = equalize_top_shifts(h, n, K=12, max_iters=300)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h.copy()

        # Fine polish
        if remaining() > 10:
            h, c5_t = fine_polish(h, n, max_iters=200)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h.copy()

    # Phase 10: Random perturbations to escape potential local minima
    if remaining() > 60:
        h_p, c5_p = perturb_and_optimize(best_h, best_n, rng, min(50.0, remaining() * 0.5))
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()

    # Phase 11: Final intensive SLSQP + Newton + polish
    h = best_h.copy()
    if remaining() > 30:
        for beta in [50000, 150000, 500000]:
            if remaining() < 8:
                break
            h = slsqop_smmax(h, best_n, beta, K=80, maxiter=800)
        if remaining() > 15:
            h, c5_t = active_set_newton(h, best_n, K=25, max_iters=200, tol=1e-13)
        if remaining() > 10:
            h, c5_t = equalize_top_shifts(h, best_n, K=15, max_iters=300)
        if remaining() > 8:
            h, c5_t = fine_polish(h, best_n, max_iters=300)
        c5_f = compute_c5(h)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h.copy()

    # Phase 12: Ultra-fine final steps
    h_t = best_h.copy()
    c5_t = best_c5
    for it in range(100):
        if remaining() < 3:
            break
        corr_t = np.correlate(h_t, 1.0 - h_t, mode='full')
        idx_t = np.argsort(-corr_t)[:20]
        g_t = np.zeros(best_n)
        c_vals = corr_t[idx_t]
        w_t = c_vals - c_vals.min() + 1e-10
        w_t = w_t / w_t.sum()
        for i in range(len(idx_t)):
            if w_t[i] > 1e-14:
                g_t += w_t[i] * grad_corr_k_vec(h_t, idx_t[i], best_n)
        gn_t = np.linalg.norm(g_t)
        if gn_t < 1e-14:
            break
        g_t = -g_t / gn_t
        improved = False
        for step in [0.003, 0.001, 0.0005, 0.002, 0.0015, 0.0008, 0.0003]:
            h_try = project_feasible(h_t + step * g_t, best_n)
            c_try = compute_c5(h_try)
            if c_try < c5_t - 1e-16:
                h_t = h_try
                c5_t = c_try
                best_h = h_try.copy()
                best_c5 = c5_t
                improved = True
                break
        if not improved:
            break

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
