def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        m = n - offset
        if m > 0:
            g[offset:] += (1.0 - h[:m])
            g[:m] -= h[offset:]
    else:
        a = -offset
        m = n - a
        if m > 0:
            g[:m] += (1.0 - h[a:])
            g[a:] -= h[:m]
    return g

def smooth_max_obj_grad(h, n, beta, K=20):
    corr = compute_corr(h)
    top_idx = np.argsort(-corr)[:min(K, len(corr))]
    c_top = corr[top_idx]
    m = c_top.max()
    sh = c_top - m
    cl = np.clip(beta * sh, -50, 50)
    ex = np.exp(cl)
    s = ex.sum()
    if s < 1e-30:
        ex = np.ones(len(top_idx))
        s = len(top_idx)
    obj = (m + np.log(s) / beta) * (2.0 / n)
    w = ex / s
    g = np.zeros(n)
    for i, k in enumerate(top_idx):
        wi = w[i] * (2.0 / n)
        if wi < 1e-15:
            continue
        g += wi * grad_corr_k(h, k, n)
    return obj, g

def lbfgsb_smooth(h, n, beta, penalty, maxiter=3000):
    def obj_fun(x):
        corr = compute_corr(x)
        top_k = min(30, len(corr))
        top_idx = np.argsort(-corr)[:top_k]
        c_top = corr[top_idx]
        m = c_top.max()
        sh = np.clip(beta * (c_top - m), -50, 50)
        ex = np.exp(sh)
        sm = (m + np.log(ex.sum()) / beta) * (2.0 / n)
        s = np.sum(x) - n / 2.0
        return sm + 0.5 * penalty * s * s

    def grad_fun(x):
        corr = compute_corr(x)
        top_k = min(30, len(corr))
        top_idx = np.argsort(-corr)[:top_k]
        c_top = corr[top_idx]
        m = c_top.max()
        sh = np.clip(beta * (c_top - m), -50, 50)
        ex = np.exp(sh)
        w = ex / ex.sum()
        g = np.zeros(n)
        for i, k in enumerate(top_idx):
            if w[i] > 1e-14:
                g += w[i] * (2.0 / n) * grad_corr_k(x, k, n)
        s = np.sum(x) - n / 2.0
        g += penalty * s
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fun, h, jac=grad_fun, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-14, 'maxfun': 20000, 'maxls': 30})

    except Exception:
        pass
    return h
