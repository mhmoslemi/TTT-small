def compute_swap_D(h_bin, n):
    """Compute D matrix for O(1) per-lag swap delta.
    D[p, k] such that for swap (i: 1→0, j: 0→1):
    delta_corr[k] = D[i, k] - D[j, k]
    """
    D = np.zeros((n, 2 * n - 1))
    one_minus_h = 1.0 - h_bin
    
    for p in range(n):
        # R part: h[p+s] for valid s (0 <= p+s < n)
        # s from -p to n-p-1, corresponding to k from n-1-p to 2n-2-p

            D[p, k_start:k_start + n_l] -= one_minus_h[::-1]
    
    return D
