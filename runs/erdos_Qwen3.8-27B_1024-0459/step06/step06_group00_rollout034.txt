        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)

    """Soft-max objective and gradient."""
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    S = np.sum(ez)
    obj = (m + np.log(S) / beta) * dx

        if wk < 1e-15:

        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])

            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            c_new = compute_c5(h_bin)
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            delta = c_new - best_c5
            if delta < best_delta:
                best_delta = delta
                best_pair = (i, j)
        
        if best_pair is not None:
            i, j = best_pair
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            best_c5 = compute_c5(h_bin)
            best_h = h_bin.copy()
            improved = True
        else:
            no_improve_rounds += 1
            if no_improve_rounds > 10:
                break
            # Random perturbation
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            best_c5 = compute_c5(h_bin)
            best_h = h_bin.copy()
    
    return best_h, best_c5

def grad_single_shift(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        np_ = n - off
        g[off:] += (1.0 - h[:np_])
        g[:np_] -= h[off:]
    else:
        a = -off
        np_ = n - a
        g[:np_] += (1.0 - h[a:])
        g[a:] -= h[:np_]
    return g

def coordinate_refine(h, n, max_sweeps=20):
    """Coordinate descent: for each i, try setting h[i] to 0 or 1 (or small perturbation)."""
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        improved = False
        for i in range(n):
            old_val = h[i]
            # Try a few candidate values
            candidates = [0.0, 1.0]
            if 0.01 < old_val < 0.99:
                candidates.extend([old_val - 0.1, old_val + 0.1, old_val - 0.05, old_val + 0.05])
            
            best_cand = old_val
            best_cand_c5 = best_c5
            
            for c in candidates:
                if c < 0 or c > 1 or abs(c - old_val) < 1e-6:
                    continue
                h[i] = c
                h = project_feasible(h, n)
                c5_v = compute_c5(h)
                if c5_v < best_cand_c5 - 1e-13:
                    best_cand_c5 = c5_v
                    best_cand = c
                    improved = True
            h[i] = best_cand
            h = project_feasible(h, n)
        
        if not improved:
            break
    
    return h, best_c5
