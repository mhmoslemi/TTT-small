def solve_ilp(n, time_limit=120):
    """Solve MILP for optimal binary h with n points."""
    try:
        if n % 2 != 0 or n > 192:
            return None
        
        h_var = cp.Variable(n, boolean=True)
        t_var = cp.Variable(nonneg=True)
        
        constraints = [h_var.sum() == n // 2]
        dx = 2.0 / n
        
        # For each offset, we need: sum_{valid i} h[i]*(1-h[i+off]) * dx <= t
        # For binary: h[i]*(1-h[i+off]) = h[i] - h[i]*h[i+off]
        # Need z[i,j] = h[i]*h[j] for all pairs (i,j)
        
        # Build z for all pairs
        z_list = []
        z_dict = {}
        z_cons = []
        for i in range(n):
            for j in range(i + 1, n):
                z = cp.Variable(nonneg=True)
                z_dict[(i, j)] = z
                z_list.append(z)
                z_cons.append(z <= h_var[i])
                z_cons.append(z <= h_var[j])
                z_cons.append(z >= h_var[i] + h_var[j] - 1)
        
        constraints.extend(z_cons)
        
        # Offset constraints
        for off in range(1, n):
            # Positive offset: sum_{i=0}^{n-1-off} h[i]*(1-h[i+off])
            pos_expr = 0
            for i in range(n - off):
                a, b = i, i + off
                pos_expr += h_var[a] - z_dict[(min(a,b), max(a,b))]
            constraints.append(pos_expr * dx <= t_var)
            
            # Negative offset: sum_{i=0}^{n-1-off} h[i+off]*(1-h[i])
            neg_expr = 0
            for i in range(n - off):
                a, b = i + off, i
                neg_expr += h_var[a] - z_dict[(min(a,b), max(a,b))]
            constraints.append(neg_expr * dx <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        
        start_time = time.time()
        for solver in [cp.CBC, cp.SCIP, cp.GLPK]:
            try:
                if time.time() - start_time > time_limit:
                    break
                prob.solve(solver=solver, gap_rel=1e-8, presolve=True)
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    h_val = np.round(np.array(h_var.value)).astype(float)
                    h_val = project_feasible(h_val, n)
                    return h_val
            except Exception:
                continue
        
        return None
    except Exception:
        return None

            h_try = project_feasible_simple(h - step * g / gn, n)
            c_try = compute_c5(h_try)
            if c_try < best_c:
                best_c = c_try
                best_step = h_try
        
        if best_step is not None:
            h = best_step.copy()
            best_c5 = best_c
            best_h = h.copy()
        else:
            # No improvement, try restarting with perturbation
            if it % 100 == 99:
                perturb = np.random.RandomState(it * 7 + 3).randn(n) * 0.01
                h_try = project_feasible_simple(h + perturb, n)
                c_try = compute_c5(h_try)
                if c_try < best_c5:
                    h = h_try
                    best_c5 = c_try
                    best_h = h.copy()
    
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    
    def rem():
        return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)
    
    # Initialize from parent
    try:
        best_h = np.array(initial_h_values, dtype=float).copy()
        best_n = len(best_h)
        best_c5 = compute_c5(best_h)
    except Exception:
        best_n = 512
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_c5 = compute_c5(best_h)
    
    # Phase 1: Quick SLSQP polish on parent
    if rem() > 60:
        h1, c1 = slsqp_polish(best_h, best_n, min(120.0, rem() * 0.15))
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1.copy()
    
    # Phase 2: ILP at smaller resolution for good binary starting point
    ilp_h = None
    if rem() > 180:
        h_ilp = solve_ilp(128, time_limit=min(90.0, rem() * 0.1))
        if h_ilp is not None:
            c_ilp_small = compute_c5(h_ilp)
            # Interpolate to best_n
            h_interp = interp_to(h_ilp, 128, best_n)
            h_feas = project_feasible(h_interp, best_n)
            c_interp = compute_c5(h_feas)
            if c_interp < best_c5:
                best_c5 = c_interp
                best_h = h_feas.copy()
            ilp_h = h_ilp
    
    # Phase 3: Try ILP at 192 if time permits
    if rem() > 120 and ilp_h is not None:
        h_ilp192 = solve_ilp(192, time_limit=min(90.0, rem() * 0.1))
        if h_ilp192 is not None:
            h_interp = interp_to(h_ilp192, 192, best_n)
            h_feas = project_feasible(h_interp, best_n)
            c_interp = compute_c5(h_feas)
            if c_interp < best_c5:
                best_c5 = c_interp
                best_h = h_feas.copy()
    
    # Phase 4: Simulated annealing on binary at best_n
    if rem() > 100:
        t_sa = min(80.0, rem() * 0.1)
        h_sa, c_sa = simulated_annealing_binary(best_h, best_n, t_sa, seed=seed + 1)
        if c_sa < best_c5:
            # Refine the binary solution
            h_sa_feas = project_feasible(h_sa, best_n)
            h_ref, c_ref = slsqp_polish(h_sa_feas, best_n, min(60.0, rem() * 0.1))
            if c_ref < best_c5:
                best_c5 = c_ref
                best_h = h_ref.copy()
            elif c_sa < best_c5:
                best_c5 = c_sa
                best_h = h_sa_feas.copy()
    
    # Phase 5: Gradient descent refinement
    if rem() > 60:
        t_gd = min(120.0, rem() * 0.15)
        h_gd, c_gd = smooth_grad_desc(best_h, best_n, t_gd, beta_start=500, beta_end=100000, n_iters=20000)
        if c_gd < best_c5:
            best_c5 = c_gd
            best_h = h_gd.copy()
    
    # Phase 6: Top-K descent
    if rem() > 40:
        h_tk, c_tk = topk_descent(best_h, best_n, min(80.0, rem() * 0.1), K=20, beta_w=80.0)
        if c_tk < best_c5:
            best_c5 = c_tk
            best_h = h_tk.copy()
    
    # Phase 7: Try higher resolution
    if rem() > 100:
        for mult in [2, 3]:
            if rem() < 60:
                break
            rn = best_n * mult
            if rn > 4096:
                break
            h_interp = interp_to(best_h, best_n, rn)
            h_feas = project_feasible(h_interp, rn)
            c_int = compute_c5(h_feas)
            t_alloc = min(80.0, rem() * 0.15)
            h_hires, c_hires = slsqp_polish(h_feas, rn, t_alloc)
            if c_hires < best_c5:
                best_c5 = c_hires
                best_h = h_hires.copy()
                best_n = rn
    
    # Phase 8: Final SLSQP with very high beta
    if rem() > 30:
        for beta in [500000, 2000000, 5000000]:
            if rem() < 15:
                break
            h_new = slsqp_optimize(best_h, best_n, beta, maxiter=150)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()
    
    # Phase 9: Final top-K with high beta
    if rem() > 20:
        h_f, c_f = topk_descent(best_h, best_n, min(40.0, rem() * 0.4), K=10, beta_w=200.0)
        if c_f < best_c5:
            best_c5 = c_f
            best_h = h_f.copy()
    
    # Final feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
