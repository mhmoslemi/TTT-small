13. **Projected gradient approach**:
    - Minimize f_β(h) subject to 0 ≤ h[i] ≤ 1 and ∑h[i] = n/2
    - Lagrangian: L(h, λ) = f_β(h) + λ(∑h[i] - n/2)
    - Projected gradient step: h_new[i] = clip(h[i] - α(∂f_β/∂h[i] + λ), 0, 1)
    - Bisect on λ to satisfy ∑h_new[i] = n/2
    
    This is clean and fast! Each iteration is O(n log n) for the FFT + O(n log n) for the bisection (which needs O(log(1/ε)) steps, each O(n)).

    Actually, the bisection on λ: for a given λ, the mapping h[i] - α(∂f_β/∂h[i] + λ) → clip(..., 0, 1) is monotonically decreasing in λ (since we're subtracting a larger value). So the sum is monotonically decreasing in λ, and we can bisect.

    Each bisection step is O(n), and we need O(log(n/ε)) steps, so O(n log(n/ε)) per gradient step. The FFT part is O(n log n). Total per iteration: O(n log n).

    if mode == 'block':
        # Random block structure
        n_blocks = rng.randint(3, 15)
        boundaries = np.sort(rng.choice(np.arange(1, n), n_blocks - 1, replace=False))
        boundaries = np.concatenate([[0], boundaries, [n]])
        h = np.zeros(n)
        for i in range(n_blocks):
            if rng.rand() < 0.5:
                h[boundaries[i]:boundaries[i+1]] = 1.0
        # Adjust to get exactly n/2 ones
        n_ones = np.sum(h > 0.5)
        if n_ones > n / 2:
            # Turn off some
            idx = np.where(h > 0.5)[0]
            rng.shuffle(idx)
            h[idx[:n_ones - n//2]] = 0.0
        elif n_ones < n / 2:
            idx = np.where(h <= 0.5)[0]
            rng.shuffle(idx)
            h[idx[:n//2 - n_ones]] = 1.0
        return h
    elif mode == 'smooth':
        # Smooth random function
        freqs = rng.randint(1, 10, size=5)

        return h
    else:

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr_fft(h):
    n = len(h)
    v = 1.0 - h
    L = 1
    while L < 2 * n - 1:
        L *= 2
    H = np.fft.rfft(h, L)
    V = np.fft.rfft(v, L)
    corr = np.fft.irfft(H * np.conj(V), L)
    return corr[:2 * n - 1]


def compute_grad_fft(h, weights, beta, dx):
    n = len(h)
    L = 1
    while L < 3 * n - 1:
        L *= 2
    Wf = np.fft.rfft(weights, L)
    Hf = np.fft.rfft(h, L)
    conv1 = np.fft.irfft(Wf * Hf, L)
    A = conv1[n - 1: 2 * n - 1]
    W_rev = weights[::-1]
    Wrevf = np.fft.rfft(W_rev, L)
    conv2 = np.fft.irfft(Wrevf * Hf, L)
    B = conv2[n - 1: 2 * n - 1]
    return (1.0 - A - B) * dx / beta


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h


def projected_gradient_step(h, grad, alpha, n):
    target = n / 2.0

    def sum_at_lam(lam):
        return np.sum(np.clip(h - alpha * (grad + lam), 0.0, 1.0))

    lo = -5.0
    hi = 5.0
    s_lo = sum_at_lam(lo)
    s_hi = sum_at_lam(hi)
    expand = 5.0
    for _ in range(30):
        if s_lo >= target >= s_hi:
            break
        if s_lo < target:
            lo -= expand
            s_lo = sum_at_lam(lo)
        if s_hi > target:
            hi += expand
            s_hi = sum_at_lam(hi)
        expand *= 2.0

    for _ in range(60):
        mid = (lo + hi) / 2.0
        s_mid = sum_at_lam(mid)
        if s_mid > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-15:
            break

    lam = (lo + hi) / 2.0
    return np.clip(h - alpha * (grad + lam), 0.0, 1.0)


def fft_optimize(h0, n, betas, max_iters, time_limit, alpha_scale=1.0):
    h = h0.copy()
    start_t = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    beta_idx = 0
    beta = betas[0]
    alpha = alpha_scale / np.sqrt(beta)

    for it in range(max_iters):
        if time.time() - start_t > time_limit:
            break
        if beta_idx < len(betas) - 1 and it % 800 == 0 and it > 0:
            beta_idx += 1
            beta = betas[beta_idx]
            alpha = alpha_scale / np.sqrt(beta)

        dx = 2.0 / n
        corr = compute_corr_fft(h)
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -50.0, 50.0)
        exp_s = np.exp(shifted)
        s = np.sum(exp_s)
        if s < 1e-30:
            weights = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            weights = exp_s / s

        grad = compute_grad_fft(h, weights, beta, dx)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        alpha_adapt = alpha / (1.0 + 0.0005 * it)
        h_new = projected_gradient_step(h, grad, alpha_adapt, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        if it > 500 and it % 500 == 0:
            c5_cur = compute_c5(h)
            if c5_cur > best_c5 * 1.001 + 1e-6:
                h = best_h.copy()

    return best_h, best_c5

def smooth_init(n, rng):
    n_harm = rng.randint(3, 10)
    x = np.arange(n) * 2.0 / n

def slsqp_polish(h, n, beta, maxiter=200):
    dx = 2.0 / n

    def obj(x):
        corr = compute_corr_fft(x)
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -50.0, 50.0)
        return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

    def grad(x):
        corr = compute_corr_fft(x)
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -50.0, 50.0)
        exp_s = np.exp(shifted)
        s = np.sum(exp_s)
        if s < 1e-30:
            w = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            w = exp_s / s
        return compute_grad_fft(x, w, beta, dx)

    def con_fun(x):
        return np.sum(x) - n / 2.0

    def con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            if compute_c5(h_new) <= compute_c5(h) + 1e-10:
                return h_new
    except Exception:
        pass
    return h.copy()
