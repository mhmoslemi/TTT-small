One more important optimization trick: since the problem is essentially binary at the optimum, I can:
1. Do continuous optimization
2. Threshold to binary (h[i] = 1 if h[i] > 0.5 else 0)
3. Adjust to get exactly n/2 ones
4. Do binary local search
5. Re-add small continuous perturbations (if helpful)
6. Repeat

This thresholding approach often finds much better solutions than pure continuous optimization.

Let me also think about the "top-K targeting" approach more carefully. The idea is:
1. Compute all 2n-1 correlation values
2. Find the top K (say K=5)
3. Optimize to minimize the MAX of just these K values (not all of them)
4. After a few iterations, the "top K" changes
5. Repeat

This is more focused than the full log-sum-exp and can converge faster.

For the objective: min max_{k in TopK} corr[k] * dx

I can use the smooth max of just these K values:
f(x) = (1/beta) * log(sum_{k in TopK} exp(beta * corr[k]))

This is a smooth upper bound on max_{k in TopK} corr[k].

The gradient involves only K terms, making it much faster than the full problem.

But there's a non-smoothness issue: the "TopK" set changes. A workaround: use the TopK from the previous iteration (stale gradient). This works well in practice.

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_shift(h, lag, n):
    k = lag - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        L = n - k
        if k == 0:
            g[:] = (1.0 - 2.0 * h)
        else:
            g[k:] += (1.0 - h[:L])
            g[:L] += (-h[k:])
    else:
        a = -k
        L = n - a
        g[:L] += (1.0 - h[a:])
        g[a:] += (-h[:L])
    return g


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)


def lbfgs_optimize(h, n, beta, penalty, maxiter=2000):
    def obj_fun(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = corr.max()
        sh = np.clip(beta * (corr - m), -50.0, 50.0)
        ex = np.exp(sh)
        sm = (m + np.log(ex.sum()) / beta) * (2.0 / n)
        s = np.sum(x) - n / 2.0
        return sm + 0.5 * penalty * s * s

    def grad_fun(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = corr.max()
        sh = np.clip(beta * (corr - m), -50.0, 50.0)
        ex = np.exp(sh)
        w = ex / ex.sum()
        dx = 2.0 / n
        g = np.zeros(n)
        for lag in range(len(corr)):
            wk = w[lag]
            if wk > 1e-12:
                g += wk * dx * grad_shift(x, lag, n)
        s = np.sum(x) - n / 2.0
        g += penalty * s
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fun, h.copy(), jac=grad_fun, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-14, 'maxfun': maxiter * 2})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()


def targeted_optimize(h, n, time_limit, rng):
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta_list = [100, 500, 2000, 8000]
    K_list = [3, 5, 8]

    for beta in beta_list:
        for K in K_list:
            if time.time() - start > time_limit:
                break
            if remaining_time() < 10:
                break
            for _ in range(5):
                if time.time() - start > time_limit:
                    break
                corr = np.correlate(h, 1.0 - h, mode='full')
                top_idx = np.argsort(-corr)[:K]
                top_vals = corr[top_idx]

                def t_obj(x):
                    tx = np.correlate(x, 1.0 - x, mode='full')
                    tv = tx[top_idx]
                    m = tv.max()
                    sh = np.clip(beta * (tv - m), -50.0, 50.0)
                    ex = np.exp(sh)
                    val = (m + np.log(ex.sum()) / beta) * (2.0 / n)
                    s = np.sum(x) - n / 2.0
                    return val + 0.5 * 500.0 * s * s

                def t_grad(x):
                    tx = np.correlate(x, 1.0 - x, mode='full')
                    tv = tx[top_idx]
                    m = tv.max()
                    sh = np.clip(beta * (tv - m), -50.0, 50.0)
                    ex = np.exp(sh)
                    w = ex / ex.sum()
                    dxv = 2.0 / n
                    g = np.zeros(n)
                    for i, lag in enumerate(top_idx):
                        if w[i] > 1e-12:
                            g += w[i] * dxv * grad_shift(x, lag, n)
                    s = np.sum(x) - n / 2.0
                    g += 500.0 * s
                    return g

                bounds = [(0.0, 1.0)] * n
                try:
                    res = minimize(t_obj, h.copy(), jac=t_grad, method='L-BFGS-B',
                                   bounds=bounds,
                                   options={'maxiter': 500, 'ftol': 1e-16,
                                            'gtol': 1e-12, 'maxfun': 1000})
                    if res.x is not None:
                        h_new = project_feasible(res.x, n)
                        c5_new = compute_c5(h_new)
                        if c5_new < best_c5:
                            best_c5 = c5_new
                            h = h_new
                except Exception:
                    pass
    return h, best_c5


def local_search_binary(h, n, time_limit, rng):

    n_ones = int(np.sum(h))

    if n_ones > target:
        one_idx = np.where(h == 1.0)[0]
        for i in range(n_ones - target):
            j = rng.choice(one_idx[target:])
            h[j] = 0.0
    elif n_ones < target:
        zero_idx = np.where(h == 0.0)[0]
        for i in range(target - n_ones):
            j = rng.choice(zero_idx[target - n_ones:])
            h[j] = 1.0

def thue_morse_init(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1')
        if c % 2 == 0:
            h[i] = 1.0
    return h


def golden_ratio_init(n, shift=0.0):
    phi = 0.5 + 1.0 / np.sqrt(5.0)
    x = (np.arange(n) * phi + shift) % 1.0
    h = (x < 0.5).astype(float)
    return h


def quadratic_residue_init(n):
    p = n
    while p > 100:
        prime = True
        for f in range(2, int(np.sqrt(p)) + 1):
            if p % f == 0:
                prime = False
                break
        if prime and p % 4 == 1:
            break
        p -= 1
    h = np.zeros(n)
    for x in range(1, p):
        r = (x * x) % p
        if r < n:
            h[r] = 1.0
    h[0] = 1.0
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        idx = np.where(h == 1.0)[0]
        rng = np.random.RandomState(0)
        for i in range(n_ones - target):
            j = idx[rng.randint(target, len(idx))]
            h[j] = 0.0
    elif n_ones < target:
        idx = np.where(h == 0.0)[0]
        rng = np.random.RandomState(0)
        for i in range(target - n_ones):
            j = idx[rng.randint(0, len(idx))]
            h[j] = 1.0
    return h
