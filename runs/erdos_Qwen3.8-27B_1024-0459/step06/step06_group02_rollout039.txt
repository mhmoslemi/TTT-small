def smooth_grad_fft(x, n, beta, lam):
    corr = compute_corr_array(x)
    m = np.max(corr)
    shifted = beta * (corr - m)
    clipped = np.clip(shifted, -200, 200)
    ex = np.exp(clipped)
    s = np.sum(ex)
    if s < 1e-30:
        w = np.ones(len(corr)) / len(corr)
    else:
        w = ex / s
    dx_const = 2.0 / n
    f_smooth = (m + np.log(s) / beta) * dx_const
    
    sum_dev = np.sum(x) - n / 2.0
    f_pen = 0.5 * lam * sum_dev ** 2
    val = f_smooth + f_pen
    
    # FFT-based gradient
    N_fft = 1
    while N_fft < 2*n - 1:
        N_fft *= 2
    
    x_fft = np.fft.rfft(x, N_fft)
    omx = 1.0 - x
    omx_fft = np.fft.rfft(omx, N_fft)
    
    # P[d] = w[n-1+d] for d = 0, ..., n-1
    P = w[n-1:n-1+n]  # w[n-1], w[n], ..., w[2n-2], length n
    # Q[d] = w[n-1-d] for d = 0, ..., n-1 (Q[0] = w[n-1] = P[0])
    Q = w[n-1::-1]  # w[n-1], w[n-2], ..., w[0], length n
    
    P_fft = np.fft.rfft(P, N_fft)
    Q_fft = np.fft.rfft(Q, N_fft)
    
    # conv(P, x)[i] = ifft(P_fft * x_fft)[i] for i = 0, ..., n-1
    conv_P_x = np.fft.irfft(P_fft * x_fft, N_fft)[:n]
    conv_P_omx = np.fft.irfft(P_fft * omx_fft, N_fft)[:n]
    conv_Q_x = np.fft.irfft(Q_fft * x_fft, N_fft)[:n]
    conv_Q_omx = np.fft.irfft(Q_fft * omx_fft, N_fft)[:n]
    
    # cross_corr(P, f)[i] = Σ_d P[d] * f[i+d]
    # = conv(flip(P), f)[i + n - 1]
    P_flip = P[::-1]
    Q_flip = Q[::-1]
    P_flip_fft = np.fft.rfft(P_flip, N_fft)
    Q_flip_fft = np.fft.rfft(Q_flip, N_fft)
    
    cross_P_omx = np.fft.irfft(P_flip_fft * omx_fft, N_fft)[n-1:2*n-1]
    cross_P_x = np.fft.irfft(P_flip_fft * x_fft, N_fft)[n-1:2*n-1]
    cross_Q_omx = np.fft.irfft(Q_flip_fft * omx_fft, N_fft)[n-1:2*n-1]
    cross_Q_x = np.fft.irfft(Q_flip_fft * x_fft, N_fft)[n-1:2*n-1]

    #       + conv_Q_omx[i] - Q[0]*omx[i] - cross_Q_x[i] + Q[0]*x[i]
    P0 = P[0]

    g = (cross_P_omx - conv_P_x + P0 * x

    # Penalty gradient
    g += lam * sum_dev
    
    return val, g
