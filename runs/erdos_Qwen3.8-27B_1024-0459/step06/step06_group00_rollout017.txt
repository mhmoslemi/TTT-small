**New idea: Use the structure of the Thue-Morse sequence**

The Thue-Morse sequence t(n) = popcount(n) mod 2 has the property that at every scale, it's balanced. Specifically, the autocorrelation has a fractal structure.

For our problem, we want h and 1-h to have low cross-correlation at all shifts. If h = T (Thue-Morse), then 1-h = complement of T. The cross-correlation of T with complement of T at shift k is:

For Thue-Morse, this should be close to half the overlap (by the self-similarity), giving a correlation of about (n-1-k)/2 on average. But the maximum over k might be higher.

def binary_hillclimb(h, n, time_limit=60, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h = (h > 0.5).astype(float).copy()
    n_half = n // 2
    if int(np.sum(h)) != n_half:
        idx = np.argsort(-h)
        h = np.zeros(n)
        h[idx[:n_half]] = 1.0
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    no_improve = 0
    
    while time.time() - start < time_limit and remaining_time() > 5:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        
        # Try random 2-swaps
        best_delta = 0
        best_swap = None
        n_trials = 500
        for _ in range(n_trials):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            c5_new = compute_c5(h)
            h[i] = 1.0
            h[j] = 0.0
            delta = c5_new - best_c5
            if delta < best_delta:
                best_delta = delta
                best_swap = (i, j)
        
        if best_swap is not None:
            i, j = best_swap
            h[i] = 0.0
            h[j] = 1.0
            best_c5 += best_delta
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                break
        
        # Occasionally try 3-swaps
        if no_improve > 5 and no_improve % 5 == 0:
            # Try a random 3-swap
            i1 = ones[rng.randint(len(ones))]
            i2 = ones[rng.randint(len(ones))]
            i3 = ones[rng.randint(len(ones))]
            j1 = zeros[rng.randint(len(zeros))]
            j2 = zeros[rng.randint(len(zeros))]
            j3 = zeros[rng.randint(len(zeros))]
            if len(set([i1,i2,i3,j1,j2,j3])) < 6:
                continue
            h[i1]=0; h[i2]=0; h[i3]=0
            h[j1]=1; h[j2]=1; h[j3]=1
            c5_new = compute_c5(h)
            h[i1]=1; h[i2]=1; h[i3]=1

    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 0.0)
    return m + np.log(np.sum(np.exp(z))) / beta

def _sm_grad_slq(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 0.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    return g

def _eq_fun(x):
    return np.sum(x) - _CUR_N / 2.0

def _eq_jac(x):
    return np.ones(_CUR_N)

def lbfgsb_opt(h, n, beta, maxiter=3000):
    global _CUR_N, _SM_BETA, _PEN_LAM
    _CUR_N = n
    _SM_BETA = beta
    _PEN_LAM = 1e8
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_sm_obj_pen, h0, jac=_sm_grad_pen, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                                'maxfun': 10000})
        h_new = project_feasible(res.x, n)
        c5_new = compute_c5(h_new)
        c5_0 = compute_c5(h0)
        if c5_new < c5_0 - 1e-15:
            return h_new
    except Exception:
        pass
    return h0

def slsqp_opt(h, n, beta, maxiter=2000):
    global _CUR_N, _SM_BETA
    _CUR_N = n
    _SM_BETA = beta
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_sm_obj_slq, h0, jac=_sm_grad_slq, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c5_new = compute_c5(h_new)
        c5_0 = compute_c5(h0)
        if c5_new < c5_0 - 1e-15:
            return h_new
    except Exception:
        pass
    return h0

def top_k_equalize(h, n, top_k=5, max_iters=600, time_limit=60):
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    start = time.time()
    for it in range(max_iters):
        if remaining_time() < 3:
            break
        if time.time() - start > time_limit:
            break
        corr = compute_corr_arr(h)
        idx_top = np.argsort(-corr)[:top_k]
        corr_top = corr[idx_top]
        c_max = corr_top[0]
        c_min_top = corr_top[-1]
        if c_max - c_min_top < 1e-12:
            break
        g_dir = np.zeros(n)
        n_top = len(idx_top)
        for pos, k in enumerate(idx_top):
            if pos == 0:
                w_k = 1.0
            else:
                w_k = -1.0 / (n_top - 1)
            g_dir += w_k * _grad_single(h, k, n)
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        d = g_dir / gn
        improved = False
        for step in [0.001, 0.003, 0.01, 0.03, 0.1, 0.3, 0.5]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5

def _grad_single(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        np_ = n - off
        g[off:] += (1.0 - h[:np_])
        g[:np_] -= h[off:]
    else:
        a = -off
        np_ = n - a
        g[:np_] += (1.0 - h[a:])
        g[a:] -= h[:np_]
    return g

def binary_hillclimb(h, n, time_limit=90, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h = (h > 0.5).astype(float).copy()
    n_half = n // 2
    if int(np.sum(h)) != n_half:
        idx = np.argsort(-h)
        h = np.zeros(n)
        h[idx[:n_half]] = 1.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    no_improve = 0
    while time.time() - start < time_limit and remaining_time() > 5:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_delta = 0.0
        best_swap = None
        n_trials = 800
        for _ in range(n_trials):
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])
            h[i] = 0.0
            h[j] = 1.0
            c5_new = compute_c5(h)
            h[i] = 1.0
            h[j] = 0.0
            delta = c5_new - best_c5
            if delta < best_delta - 1e-15:
                best_delta = delta
                best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            h[i] = 0.0
            h[j] = 1.0
            best_c5 = compute_c5(h)
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 15:
                break
    return best_h, best_c5

def binary_3swap_climb(h, n, time_limit=60, rng=None):
    if rng is None:
        rng = np.random.RandomState(123)
    h = (h > 0.5).astype(float).copy()
    n_half = n // 2
    if int(np.sum(h)) != n_half:
        idx = np.argsort(-h)
        h = np.zeros(n)
        h[idx[:n_half]] = 1.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    no_improve = 0
    while time.time() - start < time_limit and remaining_time() > 5:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_delta = 0.0
        best_swap = None
        n_trials = 300
        for _ in range(n_trials):
            trio_1 = rng.choice(len(ones), size=3, replace=False)
            trio_0 = rng.choice(len(zeros), size=3, replace=False)
            idx1 = ones[trio_1]
            idx0 = zeros[trio_0]
            if len(set(list(idx1) + list(idx0))) < 6:
                continue
            h[idx1] = 0.0
            h[idx0] = 1.0
            c5_new = compute_c5(h)
            h[idx1] = 1.0
            h[idx0] = 0.0
            delta = c5_new - best_c5
            if delta < best_delta - 1e-15:
                best_delta = delta
                best_swap = (idx1, idx0)
        if best_swap is not None:
            idx1, idx0 = best_swap
            h[idx1] = 0.0
            h[idx0] = 1.0
            best_c5 = compute_c5(h)
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 10:
                break
    return best_h, best_c5

def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)

def theta_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)

def periodic_pattern(n, period, duty):
    h = np.zeros(n)
    for i in range(n):
        if (i % period) < duty:
            h[i] = 1.0
    return project_feasible(h, n)

def continuous_sa(h, n, time_limit=60, rng=None):
    if rng is None:
        rng = np.random.RandomState(77)
    h = project_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    T = 0.1
    no_improve = 0
    while time.time() - start < time_limit and remaining_time() > 5:
        T *= 0.995
        n_pert = rng.randint(5, 40)
        idx = rng.choice(n, size=n_pert, replace=False)
        perturb = rng.randn(n_pert) * 0.15
        h_new = h.copy()
        h_new[idx] = np.clip(h[idx] + perturb, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        delta = c5_new - best_c5
        if delta < 0 or (T > 1e-6 and rng.random() < np.exp(-delta / T)):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        if T < 1e-8 and no_improve > 50:
            break
    return best_h, best_c5

def multi_beta_optimize(h, n, betas, maxiter_each=2000, method='lbfgsb'):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for beta in betas:
        if remaining_time() < 15:
            break
        if method == 'lbfgsb':
            h_new = lbfgsb_opt(best_h, n, beta, maxiter=maxiter_each)
        else:
            h_new = slsqp_opt(best_h, n, beta, maxiter=maxiter_each)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = theta_morse(640)
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    n_work = best_n

    # Phase 1: L-BFGS-B at parent resolution with increasing beta
    if remaining_time() > 30:
        h = lbfgsb_opt(best_h, n_work, 100, maxiter=2000)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
        h = lbfgsb_opt(best_h, n_work, 500, maxiter=2000)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
        h = lbfgsb_opt(best_h, n_work, 5000, maxiter=2000)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: SLSQP polish (exact constraint)
    if remaining_time() > 30:
        h = slsqp_opt(best_h, n_work, 10000, maxiter=2000)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
        h = slsqp_opt(best_h, n_work, 100000, maxiter=2000)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()

    # Phase 3: Top-k equalization
    if remaining_time() > 20:
        for tk in [4, 8, 15]:
            if remaining_time() < 15:
                break
            h_eq, c5_eq = top_k_equalize(best_h, n_work, top_k=tk, max_iters=500, time_limit=40)
            if c5_eq < best_c5 - 1e-15:
                best_c5 = c5_eq
                best_h = h_eq.copy()

    # Phase 4: L-BFGS-B with very high beta
    if remaining_time() > 30:
        h = lbfgsb_opt(best_h, n_work, 50000, maxiter=3000)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
        h = lbfgsb_opt(best_h, n_work, 500000, maxiter=2000)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()

    # Phase 5: Binary hill climbing (2-swap)
    if remaining_time() > 40:
        h_bin = (best_h > 0.5).astype(float)
        h_b, c5_b = binary_hillclimb(h_bin, n_work, time_limit=min(80, int(remaining_time() * 0.3)), rng=rng)
        if c5_b < best_c5 - 1e-15:
            best_c5 = c5_b
            best_h = h_b.copy()
            h_b2, c5_b2 = binary_hillclimb(h_b, n_work, time_limit=40, rng=rng)
            if c5_b2 < best_c5 - 1e-15:
                best_c5 = c5_b2
                best_h = h_b2.copy()

    # Phase 6: Binary 3-swap climb
    if remaining_time() > 30:
        h_bin = (best_h > 0.5).astype(float)
        h_3, c5_3 = binary_3swap_climb(h_bin, n_work, time_limit=min(50, int(remaining_time() * 0.3)), rng=rng)
        if c5_3 < best_c5 - 1e-15:
            best_c5 = c5_3
            best_h = h_3.copy()

    # Phase 7: SLSQP after binary search
    if remaining_time() > 30:
        h = slsqp_opt(best_h, n_work, 50000, maxiter=2000)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
        h = slsqp_opt(best_h, n_work, 500000, maxiter=2000)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()

    # Phase 8: Continuous SA for exploration
    if remaining_time() > 40:
        h_sa, c5_sa = continuous_sa(best_h, n_work, time_limit=min(50, int(remaining_time() * 0.25)), rng=rng)
        if c5_sa < best_c5 - 1e-15:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            h_new = lbfgsb_opt(h_sa, n_work, 10000, maxiter=2000)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 9: Multi-resolution - try higher n
    n_hi = min(n_work * 2, 1280)
    if remaining_time() > 80 and n_hi > n_work:
        h_hi = interp_to(best_h, n_work, n_hi)
        h_hi = project_feasible(h_hi, n_hi)
        h_hi = lbfgsb_opt(h_hi, n_hi, 500, maxiter=2000)
        h_hi = lbfgsb_opt(h_hi, n_hi, 5000, maxiter=2000)
        c5_hi = compute_c5(h_hi)
        h_hi, c5_hi = top_k_equalize(h_hi, n_hi, top_k=6, max_iters=300, time_limit=40)
        if remaining_time() > 20:
            h_hi = slsqp_opt(h_hi, n_hi, 100000, maxiter=1500)
            c5_hi = compute_c5(h_hi)
        if c5_hi < best_c5 - 1e-15:
            best_c5 = c5_hi
            best_h = h_hi.copy()
            best_n = n_hi
            n_work = n_hi

    # Phase 10: Try structured alternatives at current n
    if remaining_time() > 60:
        alts = []
        alts.append(theta_morse(n_work))
        alts.append(periodic_pattern(n_work, 7, 3))
        alts.append(periodic_pattern(n_work, 11, 5))
        h_rand = np.zeros(n_work)
        h_rand[rng.choice(n_work, n_work // 2, replace=False)] = 1.0
        alts.append(h_rand)

        for h_alt in alts:
            if remaining_time() < 30:
                break
            h_alt = lbfgsb_opt(h_alt, n_work, 1000, maxiter=1500)
            h_alt = lbfgsb_opt(h_alt, n_work, 10000, maxiter=1500)
            c5_alt = compute_c5(h_alt)
            h_alt, c5_alt = top_k_equalize(h_alt, n_work, top_k=5, max_iters=200, time_limit=20)
            if c5_alt < best_c5 - 1e-15:
                best_c5 = c5_alt
                best_h = h_alt.copy()

    # Phase 11: Final intensive refinement
    if remaining_time() > 20:
        h = lbfgsb_opt(best_h, n_work, 1000000, maxiter=3000)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()

    if remaining_time() > 15:
        h_eq, c5_eq = top_k_equalize(best_h, n_work, top_k=10, max_iters=500, time_limit=25)
        if c5_eq < best_c5 - 1e-15:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    if remaining_time() > 15:
        h = slsqp_opt(best_h, n_work, 5000000, maxiter=2000)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()

    # Final binary polish
    if remaining_time() > 20:
        h_bin = (best_h > 0.5).astype(float)
        h_b, c5_b = binary_hillclimb(h_bin, n_work, time_limit=min(30, int(remaining_time() * 0.4)), rng=rng)
        if c5_b < best_c5 - 1e-15:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, n_work)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, n_work)
