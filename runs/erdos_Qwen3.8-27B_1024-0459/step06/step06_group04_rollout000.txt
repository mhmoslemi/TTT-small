import numpy as np
from scipy.optimize import minimize, LinearConstraint, Bounds
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if interior.any():
            h[interior] += diff / interior.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def smax_val(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    s = np.clip(beta * (corr - m), -500.0, 500.0)
    e = np.exp(s)
    return (m + np.log(e.sum()) / beta) * dx


def smax_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    s = np.clip(beta * (corr - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx] * dx
        if wk < 1e-15:
            continue
        off = k_idx - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - h[ao:])
            g[ao:] -= wk * h[:np_]
    return g

def trust_constr_step(h, beta, maxiter):

    bnds = Bounds(np.zeros(n), np.ones(n))
    cons = LinearConstraint(np.ones((1, n)), n / 2.0, n / 2.0)

    return h


def subgrad_topk(h, K):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(-corr)[:K]
    c_top = corr[top_idx]
    spread = c_top[0] - c_top[-1] + 1e-10
    w = np.exp((c_top - c_top[0]) / (spread * 0.5))
    w /= w.sum()
    g = np.zeros(n)
    for i in range(K):
        wk = w[i]
        if wk < 1e-12:
            continue
        k_idx = top_idx[i]
        off = k_idx - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - h[ao:])
            g[ao:] -= wk * h[:np_]
    g *= dx
    return g

def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
