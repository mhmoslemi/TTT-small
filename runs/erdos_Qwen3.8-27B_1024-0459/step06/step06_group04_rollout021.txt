```python
def active_set_lp(h, top_k, n, dx):
    """Solve active-set LP to find improvement direction."""
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:top_k]
    
    # Compute gradients for active lags
    n_vars = n + 1  # h[0..n-1] and t
    # c = [0]*n + [1] (minimize t)
    c_obj = np.zeros(n_vars)
    c_obj[-1] = 1.0
    
    # Inequality constraints: g_k^T h - t ≤ c_k for each active lag
    # g_k^T h - t ≤ t - f_k(h) + g_k^T h ... wait
    # f_k(h_new) ≈ f_k(h) + g_k^T (h_new - h)
    # We want f_k(h_new) ≤ t
    # f_k(h) + g_k^T (h_new - h) ≤ t
    # g_k^T h_new - t ≤ t - f_k(h) + g_k^T h  ... no
    # g_k^T h_new - t ≤ -f_k(h) + g_k^T h
    # Let rhs_k = g_k^T h - f_k(h)
    # g_k^T h_new - t ≤ rhs_k
    
    A_ub_rows = []
    b_ub_vals = []
    for k_idx in idx:
        off = k_idx - (n - 1)
        g = np.zeros(n)
        if off >= 0:
            np_ = n - off
            g[off:] += (1.0 - h[:np_])
            g[:np_] -= h[off:]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += (1.0 - h[ao:])
            g[ao:] -= h[:np_]
        # Full gradient including t: [g, -1]
        row = np.zeros(n_vars)
        row[:n] = g
        row[n] = -1.0
        A_ub_rows.append(row)
        rhs = g @ h - corr[k_idx]  # = -f_k(h) + g^T h
        b_ub_vals.append(rhs)
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n/2.0])
    
    # Bounds: 0 ≤ h_i ≤ 1, t ≥ 0
    bounds_list = [(0, 1)] * n + [(0, None)]
    
    from scipy.optimize import linprog
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds_list, method='highs')
    if res.success:
        return res.x[:n].copy(), res.x[n], idx
    return None, None, idx
```

        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 10.0:
            break

        target = active_set_lp_step(h, n, top_k)
        if target is None:
            break

        direction = target - h
        dnorm = np.linalg.norm(direction)
        if dnorm < 1e-12:
            break

        best_alpha = 0.0
        best_val = compute_c5(h)
        for alpha in [1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.05]:
            h_try = project_feasible(h + alpha * direction, n)
            c_try = compute_c5(h_try)
            if c_try < best_val - 1e-13:
                best_val = c_try
                best_alpha = alpha
                break

        if best_alpha == 0.0:
            for alpha in [0.02, 0.03, 0.04]:
                h_try = project_feasible(h + alpha * direction, n)
                c_try = compute_c5(h_try)
                if c_try < best_val - 1e-13:
                    best_val = c_try
                    best_alpha = alpha
                    break

        if best_alpha > 0.0:
            h = project_feasible(h + best_alpha * direction, n)
            c_new = compute_c5(h)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h.copy()
        else:
            break

    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def pair_swap_search(h, n, time_limit, max_rounds=500):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    m = 2 * n - 1
    rng = np.random.RandomState(999)

    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 10.0:
            break

        raw_corr = compute_raw_corr(h)
        improved = False
        one_mask = h > 0.5
        zero_mask = h < 0.5
        ones_idx = np.where(one_mask)[0]
        zeros_idx = np.where(zero_mask)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break

        n_try = min(300, len(ones_idx) * len(zeros_idx))
        for _ in range(n_try):
            i1 = ones_idx[rng.randint(len(ones_idx))]
            i0 = zeros_idx[rng.randint(len(zeros_idx))]
            # Swap: h[i1]→0-ish, h[i0]→1-ish
            # Compute effect on correlations
            delta = h[i1]  # amount to remove from i1, add to i0
            if delta < 0.05:
                continue

            # Change: h[i1] -= delta, h[i0] += delta
            # Effect on raw_corr[idx] for offset o = idx - (n-1):
            # Remove from i1: -delta*(1-h[i1+o]) + delta*h[i1-o]
            # Add to i0: +delta*(1-h[i0+o]) - delta*h[i0-o]
            # Net: delta*[h[i1-o] - (1-h[i1+o]) + (1-h[i0+o]) - h[i0-o]]
            new_c5 = raw_corr.max()
            # Quick check: does the max lag improve?
            k_max = np.argmax(raw_corr)
            o_max = k_max - (n - 1)
            eff = 0.0
            jp = i1 + o_max
            jm = i1 - o_max
            if 0 <= jp < n:
                eff -= (1.0 - h[jp])
            if 0 <= jm < n:
                eff += h[jm]
            jp2 = i0 + o_max
            jm2 = i0 - o_max
            if 0 <= jp2 < n:
                eff += (1.0 - h[jp2])
            if 0 <= jm2 < n:
                eff -= h[jm2]
            eff *= delta

            if eff >= -1e-10:
                continue  # doesn't reduce the max

            # Try the move
            h[i1] -= delta
            h[i0] += delta
            h[i1] = max(0.0, h[i1])
            h[i0] = min(1.0, h[i0])
            # Fix sum
            diff = n / 2.0 - h.sum()
            if abs(diff) > 1e-12:
                h += diff / n
                h = np.clip(h, 0.0, 1.0)
            c_try = float(raw_corr.max()) * (2.0 / n)
            # Recompute properly
            c_try = compute_c5(h)
            if c_try < best_c5 - 1e-14:
                best_c5 = c_try
                best_h = h.copy()
                improved = True
            else:
                # Revert
                h[i1] += delta
                h[i0] -= delta
                h[i1] = min(1.0, h[i1])
                h[i0] = max(0.0, h[i0])
                diff2 = n / 2.0 - h.sum()
                if abs(diff2) > 1e-12:
                    h += diff2 / n
                    h = np.clip(h, 0.0, 1.0)

        if not improved and rnd > 50:
            break

    return best_h, best_c5
