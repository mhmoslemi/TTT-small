            if len(idx) > 0:
                h[idx[0]] = 1.0 if diff > 0 else 0.0
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    sorted_idx = np.argsort(-h)
    b = np.zeros(n)
    b[sorted_idx[:n // 2]] = 1.0
    return b

def precompute_dcorr(h, n):
    nfull = 2 * n - 1
    dcorr = np.zeros((n, nfull))
    for i in range(n):
        d = np.arange(nfull) - (n - 1)
        j1 = i + d
        v1 = (j1 >= 0) & (j1 < n)
        dcorr[i, v1] += (1.0 - h[j1[v1]])
        j2 = i - d
        v2 = (j2 >= 0) & (j2 < n)
        dcorr[i, v2] -= h[j2[v2]]
    return dcorr

def flip_dcorr(dcorr, i, delta, n):
    """Update dcorr after h[i] changes by delta."""
    nfull = 2 * n - 1
    # First term: dcorr[m,k] -= delta when m+d(k)=i, m=i-k+n-1, k∈[i, i+n-1]
    k_lo1 = i
    k_hi1 = min(i + n - 1, 2 * n - 2)
    if k_lo1 <= k_hi1:
        ks = np.arange(k_lo1, k_hi1 + 1)
        ms = (i + n - 1) - ks
        dcorr[ms, ks] -= delta
    
    # Second term: dcorr[m,k] += delta when m-d(k)=i, m=i+k-n+1, k∈[n-1-i, 2n-2-i]
    k_lo2 = n - 1 - i
    k_hi2 = min(2 * n - 2, 2 * n - 2 - i)
    if k_lo2 <= k_hi2:
        ks = np.arange(k_lo2, k_hi2 + 1)
        ms = i + ks - (n - 1)
        dcorr[ms, ks] += delta

def binary_local_search(h, n, time_limit=120):
    h = to_binary(h, n).copy()
    nfull = 2 * n - 1
    corr = compute_corr(h).astype(float)
    c_max = float(np.max(corr))
    best_c5 = c_max * (2.0 / n)
    dcorr = precompute_dcorr(h, n)
    ones_idx = np.where(h == 1.0)[0]
    zeros_idx = np.where(h == 0.0)[0]
    start = time.time()
    
    improved = True
    while improved and time.time() - start < time_limit and _BUDGET - (time.time() - _T0) > 20:
        improved = False
        for i in ones_idx:
            if time.time() - start > time_limit or _BUDGET - (time.time() - _T0) < 20:
                break
            base = corr - dcorr[i]
            base_max = float(np.max(base))
            if base_max >= c_max - 1e-10:
                continue
            
            # Vectorized: for each j in zeros_idx, compute max(base + dcorr[j])
            dcorr_zeros = dcorr[zeros_idx]  # (n_zeros, nfull)
            new_maxs = np.maximum.reduce(base[None, :] + dcorr_zeros, axis=1)
            best_j_local = int(np.argmax(-new_maxs))
            best_new_max = float(new_maxs[best_j_local])
            
            if best_new_max < c_max - 1e-10:
                j = zeros_idx[best_j_local]
                # Perform swap
                h[i] = 0.0
                h[j] = 1.0
                corr = base + dcorr[j]
                c_max = best_new_max
                # Update dcorr
                flip_dcorr(dcorr, i, -1.0, n)  # h[i]: 1→0, delta=-1
                flip_dcorr(dcorr, j, +1.0, n)  # h[j]: 0→1, delta=+1
                ones_idx = np.where(h == 1.0)[0]
                zeros_idx = np.where(h == 0.0)[0]
                best_c5 = c_max * (2.0 / n)
                improved = True
                break
    
    return h, best_c5

def simulated_annealing(h, n, time_limit=300, seed=42):
    h = to_binary(h, n).copy()
    nfull = 2 * n - 1
    corr = compute_corr(h).astype(float)
    c_max = float(np.max(corr))
    best_h = h.copy()
    best_c5 = c_max * (2.0 / n)
    dcorr = precompute_dcorr(h, n)
    
    T = 5.0
    T_min = 0.01
    cooling = 0.99999
    rng = np.random.RandomState(seed)
    start = time.time()
    
    ones_idx = np.where(h == 1.0)[0].tolist()
    zeros_idx = np.where(h == 0.0)[0].tolist()
    n_ones_half = n // 2
    
    step = 0
    while time.time() - start < time_limit and _BUDGET - (time.time() - _T0) > 20:
        step += 1
        # Pick random positions
        idx_o = rng.randint(n_ones_half)
        idx_z = rng.randint(n_ones_half)
        i = ones_idx[idx_o]
        j = zeros_idx[idx_z]
        
        # Compute new corr
        new_corr = corr - dcorr[i] + dcorr[j]
        new_max = float(np.max(new_corr))
        delta = new_max - c_max
        
        if delta <= 0 or rng.rand() < np.exp(-delta / T):
            # Accept
            old_i = i
            h[i] = 0.0
            h[j] = 1.0
            corr = new_corr
            c_max = new_max
            
            # Update dcorr
            flip_dcorr(dcorr, i, -1.0, n)
            flip_dcorr(dcorr, j, +1.0, n)
            
            # Update index lists
            ones_idx[idx_o] = j
            zeros_idx[idx_z] = i
            
            if c_max * (2.0 / n) < best_c5:
                best_c5 = c_max * (2.0 / n)
                best_h = h.copy()
        
        if step % 10000 == 0:
            T = max(T_min, T * cooling ** 10000)
        
        T = max(T_min, T * cooling)
    
    return best_h, best_c5

def thue_morse(n):
    """Generate Thue-Morse sequence of length n."""
    tm = np.zeros(n, dtype=int)
    for i in range(n):
        tm[i] = bin(i).count('1') % 2
    return tm.astype(float)

def slsqp_polish(h, n, beta=10000, maxiter=300):
    """SLSQP optimization with smooth max."""
    nfull = 2 * n - 1
    dx = 2.0 / n
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        s = np.clip(beta * (c - m), -50, 50)
        return (m + np.log(np.sum(np.exp(s))) / beta) * dx
    
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        s = np.clip(beta * (c - m), -50, 50)
        w = np.exp(s)
        w /= np.sum(w)
        g = np.zeros(n)
        for k in range(nfull):
            wk = w[k] * dx
            if wk < 1e-15:
                continue
            offset = k - (n - 1)
            if offset >= 0:
                np_ = n - offset
                g[offset:offset+np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[offset:offset+np_]
            else:
                a = -offset
                np_ = n - a
                g[:np_] += wk * (1.0 - x[a:a+np_])
                g[a:a+np_] -= wk * x[:np_]
        return g
    
    def eq_fun(x):
        return np.sum(x) - n / 2.0
    
    def eq_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    x_new = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(x_new, xp, src_h)

            if c5_alt_ls < c5_alt:

        c5_s = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
    
    # Ensure feasibility
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, len(best_h))

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        at_low = h <= 1e-12
        at_high = h >= 1.0 - 1e-12
        mid = ~(at_low | at_high)
        cnt = int(np.count_nonzero(mid))
        if cnt > 0:
            h[mid] += diff / cnt
        else:
            if diff > 0:
                idx = np.where(at_low)[0]
            else:
                idx = np.where(at_high)[0]
            if len(idx) > 0:
                h[idx[0]] = 1.0 if diff > 0 else 0.0
        h = np.clip(h, 0.0, 1.0)
    return h


def to_binary(h, n):
    sorted_idx = np.argsort(-h)
    b = np.zeros(n)
    b[sorted_idx[:n // 2]] = 1.0
    return b


def precompute_dcorr(h, n):
    nfull = 2 * n - 1
    i_vals = np.arange(n)[:, None]
    d_vals = (np.arange(nfull) - (n - 1))[None, :]
    j1 = i_vals + d_vals
    v1 = (j1 >= 0) & (j1 < n)
    j1_clip = np.clip(j1, 0, n - 1)
    part1 = (1.0 - h[j1_clip]) * v1
    j2 = i_vals - d_vals
    v2 = (j2 >= 0) & (j2 < n)
    j2_clip = np.clip(j2, 0, n - 1)
    part2 = h[j2_clip] * v2
    return part1 - part2


def flip_dcorr(dcorr, i, delta, n):
    nfull = 2 * n - 1
    k_lo1 = i
    k_hi1 = i + n - 1
    if k_hi1 >= nfull:
        k_hi1 = nfull - 1
    if k_lo1 <= k_hi1:
        ks = np.arange(k_lo1, k_hi1 + 1)
        ms = (n - 1) + i - ks
        dcorr[ms, ks] -= delta
    k_lo2 = n - 1 - i

    if k_lo2 <= k_hi2:
        ks = np.arange(k_lo2, k_hi2 + 1)

def binary_local_search(h, n, time_limit=120):
    h = to_binary(h, n).copy()
    nfull = 2 * n - 1
    corr = compute_corr(h).astype(float)
    c_max = float(np.max(corr))
    best_c5 = c_max * (2.0 / n)
    dcorr = precompute_dcorr(h, n)
    ones_idx = np.where(h == 1.0)[0]
    zeros_idx = np.where(h == 0.0)[0]
    start = time.time()

    improved = True
    while improved and time.time() - start < time_limit and _BUDGET - (time.time() - _T0) > 20:
        improved = False
        for i in ones_idx:
            if time.time() - start > time_limit or _BUDGET - (time.time() - _T0) < 20:
                break
            base = corr - dcorr[i]
            base_max = float(np.max(base))
            if base_max >= c_max - 1e-10:
                continue
            dcorr_zeros = dcorr[zeros_idx]
            new_maxs = np.max(base[None, :] + dcorr_zeros, axis=1)
            best_j_local = int(np.argmin(new_maxs))
            best_new_max = float(new_maxs[best_j_local])
            if best_new_max < c_max - 1e-10:
                j = zeros_idx[best_j_local]
                h[i] = 0.0
                h[j] = 1.0
                corr = base + dcorr[j]
                c_max = best_new_max
                flip_dcorr(dcorr, i, -1.0, n)
                flip_dcorr(dcorr, j, 1.0, n)
                ones_idx = np.where(h == 1.0)[0]
                zeros_idx = np.where(h == 0.0)[0]

def thue_morse(n):
    tm = np.zeros(n, dtype=int)
    for i in range(n):
        tm[i] = bin(i).count('1') % 2
    return tm.astype(float)


def slsqp_polish(h, n, beta=10000, maxiter=300):
    nfull = 2 * n - 1
    dx = 2.0 / n

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        s = np.clip(beta * (c - m), -50, 50)
        return (m + np.log(np.sum(np.exp(s))) / beta) * dx

    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        s = np.clip(beta * (c - m), -50, 50)
        w = np.exp(s)
        w_sum = np.sum(w)
        if w_sum < 1e-30:
            w = np.ones(nfull) / nfull
        else:
            w /= w_sum
        g = np.zeros(n)
        for k in range(nfull):
            wk = w[k] * dx
            if wk < 1e-15:
                continue
            offset = k - (n - 1)
            if offset >= 0:
                np_ = n - offset
                g[offset:offset + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[offset:offset + np_]
            else:
                a = -offset
                np_ = n - a
                g[:np_] += wk * (1.0 - x[a:a + np_])
                g[a:a + np_] -= wk * x[:np_]
        return g

    def eq_fun(x):
        return np.sum(x) - n / 2.0

    def eq_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    x_new = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(x_new, xp, src_h)

    rng = np.random.RandomState(123)
    start = time.time()

    while time.time() - start < time_limit and _BUDGET - (time.time() - _T0) > 20:
        # Pick two random ones and two random zeros
        perm_o = rng.permutation(n_half)

        j1, j2 = zeros_idx[perm_z[0]], zeros_idx[perm_z[1]]
        if i1 == i2 or j1 == j2:

        new_corr = corr - dcorr[i1] - dcorr[i2] + dcorr[j1] + dcorr[j2]

            flip_dcorr(dcorr, i1, -1.0, n)

            flip_dcorr(dcorr, j1, 1.0, n)

            ones_idx = list(np.where(h == 1.0)[0])
