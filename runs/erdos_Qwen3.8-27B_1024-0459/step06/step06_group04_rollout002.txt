import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def sm_val(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    s = np.clip(beta * (corr - m), -100.0, 100.0)
    e = np.exp(s)
    return (m + np.log(e.sum()) / beta) * dx


def sm_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    s = np.clip(beta * (corr - m), -100.0, 100.0)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-14:
            continue
        off = k_idx - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - h[ao:])
            g[ao:] -= wk * h[:np_]
    return g * dx


def sm_eq(x, n):
    return x.sum() - n / 2.0


def sm_eq_jac(x, n):
    return np.ones(n)


def slsqp_optimize(h, beta, maxiter, ftol=1e-15):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': sm_eq, 'jac': sm_eq_jac, 'args': (n,)}]
    try:
        res = minimize(sm_val, h, args=(beta,),
                       jac=sm_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def momentum_gd(h, n, time_limit, K=30, lr_init=0.02, momentum=0.95):
    h = h.copy().astype(float)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    lr = lr_init
    for it in range(50000):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 10:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = corr.max() * (2.0 / n)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
        idx = np.argsort(-corr)[:K]
        c_top = corr[idx]
        weights = np.exp(40.0 * (c_top - c_top.max()))
        sw = weights.sum()
        if sw < 1e-30:
            weights = np.ones(K) / K
        else:
            weights /= sw
        g = np.zeros(n)
        for i, k_idx in enumerate(idx):
            wk = weights[i]
            if wk < 1e-12:
                continue
            off = k_idx - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:] += wk * (1.0 - h[:np_])
                g[:np_] -= wk * h[off:]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - h[ao:])
                g[ao:] -= wk * h[:np_]
        g *= (2.0 / n)
        g -= g.mean()

        h = project_feasible(h + v, n)

    return best_h, best_c5


def swap_search(h, n, time_limit, max_rounds=200, n_tries=400):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    rng = np.random.RandomState(777)
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 10:
            break
        improved = False
        for _ in range(n_tries):
            i1 = rng.randint(n)
            i2 = rng.randint(n)
            if i1 == i2 or abs(h[i1] - h[i2]) < 1e-8:
                continue
            h[i1], h[i2] = h[i2], h[i1]
            c_try = compute_c5(h)
            if c_try < best_c5 - 1e-14:
                best_c5 = c_try
                best_h = h.copy()
                improved = True
            else:
                h[i1], h[i2] = h[i2], h[i1]
        if not improved and rnd > 40:
            break
    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def perturb(h, n, rng, mag=0.05):
    h_new = h + rng.normal(0, mag, n)
    return project_feasible(h_new, n)


def subgrad_projected(h, n, time_limit, K=20):
    h = h.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    for it in range(30000):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 10:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = corr.max() * (2.0 / n)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
        idx = np.argsort(-corr)[:K]
        c_top = corr[idx]
        weights = np.exp(80.0 * (c_top - c_top.max()))
        sw = weights.sum()
        if sw < 1e-30:
            weights = np.ones(K) / K
        else:
            weights /= sw
        g = np.zeros(n)
        for i, k_idx in enumerate(idx):
            wk = weights[i]
            if wk < 1e-12:
                continue
            off = k_idx - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:] += wk * (1.0 - h[:np_])
                g[:np_] -= wk * h[off:]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - h[ao:])
                g[ao:] -= wk * h[:np_]
        g *= (2.0 / n)
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        gap = best_c5 * n / 2.0 - corr[idx[0]]
        alpha = max(1e-8, gap / (gn * gn + 1e-30))
        alpha = min(alpha, 0.05)
        accepted = False
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_try = project_feasible(h - mult * alpha * g / gn, n)
            c_try = compute_c5(h_try)
            if c_try < best_c5:
                h = h_try
                best_c5 = c_try
                best_h = h.copy()
                accepted = True
                break
        if not accepted:
            pass
    return best_h, best_c5
