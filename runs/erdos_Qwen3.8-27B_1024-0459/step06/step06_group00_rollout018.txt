def compute_c5(h):

        for pos, k in enumerate(idx_top):
            if pos == 0:
                w_k = 1.0
            else:
                w_k = -1.0 / (n_top - 1)
            g += w_k * grad_single_shift(h, int(k), n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        d = g / gn
        improved = False
        steps = [0.001, 0.003, 0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3]
        for step in steps:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-15:
                h = ht
                best_c5 = cn
                best_h = h.copy()
                improved = True
                break
        if not improved:
            # Try negative direction too
            for step in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05]:
                ht = project_feasible(h + step * (-d), n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-15:
                    h = ht
                    best_c5 = cn
                    best_h = h.copy()
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c5


def subgradient_zigzag(h, n, max_iters=10000, time_limit=90):
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    k_curr = None
    k_prev = None
    stale_count = 0
    for it in range(max_iters):
        if remaining_time() < 5:
            break
        if time.time() - start > time_limit:
            break
        corr = fft_corr(h)
        k_curr = int(np.argmax(corr))
        if k_prev is not None and k_curr != k_prev:
            g = 0.5 * (grad_single_shift(h, k_curr, n) + grad_single_shift(h, k_prev, n))
            k_prev = k_curr
        else:
            g = grad_single_shift(h, k_curr, n)
            if k_prev is None:
                k_prev = k_curr
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-14:
            g = g / g_norm
        improved = False
        base_step = max(0.001, 0.02 * (1.0 - 0.3 * it / max_iters))
        for step in [base_step * 0.3, base_step * 0.5, base_step, base_step * 2.0,
                     base_step * 4.0, 0.001, 0.003, 0.005, 0.01, 0.02, 0.05, 0.1, 0.15]:
            ht = project_feasible(h - step * g, n)
            c5_new = compute_c5(ht)
            if c5_new < best_c5 - 1e-15:
                h = ht
                best_c5 = c5_new
                best_h = h.copy()
                improved = True
                stale_count = 0
                break
        if not improved:
            stale_count += 1
            if stale_count > 300:
                break
            if stale_count % 80 == 0:
                rng = np.random.RandomState(stale_count + it)
                pert = rng.randn(n) * 0.003
                ht = project_feasible(h + pert, n)
                c5_new = compute_c5(ht)
                if c5_new < best_c5:
                    h = ht
                    best_c5 = c5_new
                    best_h = h.copy()
                    stale_count = 0
    return best_h, best_c5


def binary_sa(h, n, time_limit=120, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h_bin = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        sort_idx = np.argsort(-h_bin)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0
    dx = 2.0 / n
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    curr_c5 = best_c5
    start = time.time()
    T_start = 3.0 / dx
    T_end = 0.001 / dx
    total_iters = max(5000, int(time_limit * 200))
    min_T_ratio = T_end / T_start
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    no_improve = 0
    for it in range(total_iters):
        if time.time() - start > time_limit:
            break
        if remaining_time() < 5:
            break
        T = T_start * (min_T_ratio ** (it / max(1, total_iters)))
        i_idx = rng.randint(len(ones))
        j_idx = rng.randint(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        new_c5 = compute_c5(h_bin)
        delta = new_c5 - curr_c5
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / max(T, 1e-10))):
            ones[i_idx] = j
            zeros[j_idx] = i
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_bin.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
        if no_improve > 8000:
            break
    return best_h, best_c5


def binary_greedy_local(h_bin, n, time_limit=60, rng=None):
    if rng is None:
        rng = np.random.RandomState(123)
    h_bin = h_bin.copy()
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    start = time.time()
    no_improve_rounds = 0
    while time.time() - start < time_limit and remaining_time() > 5:
        improved = False
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        n_ones = len(ones)
        n_zeros = len(zeros)
        n_pairs = n_ones * n_zeros
        n_trials = min(n_pairs, 500)
        best_delta = 1e-10
        best_pair = None
        for _ in range(n_trials):
            i = ones[rng.randint(n_ones)]
            j = zeros[rng.randint(n_zeros)]
            if i == j:
                continue
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            c_new = compute_c5(h_bin)
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            delta = c_new - best_c5
            if delta < best_delta:
                best_delta = delta
                best_pair = (i, j)
        if best_pair is not None:
            i, j = best_pair
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            best_c5 = compute_c5(h_bin)
            best_h = h_bin.copy()
            improved = True
        if not improved:
            no_improve_rounds += 1
            if no_improve_rounds > 15:
                break
            i = ones[rng.randint(n_ones)]
            j = zeros[rng.randint(n_zeros)]
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            best_c5 = compute_c5(h_bin)
            best_h = h_bin.copy()
    return best_h, best_c5


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def theta_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)


def structured_pattern(n, variant=0):
    h = np.zeros(n)
    if variant == 0:
        third = n // 3
        h[:third] = 1.0
        h[2*third:] = 1.0
    elif variant == 1:
        quarter = n // 4
        h[:quarter] = 1.0
        h[quarter:2*quarter] = 0.0
        h[2*quarter:3*quarter] = 1.0
        h[3*quarter:] = 0.0
    elif variant == 2:
        fifth = n // 5
        h[:fifth] = 1.0
        h[2*fifth:3*fifth] = 1.0
    elif variant == 3:
        half = n // 2
        h[half::2] = 1.0
    else:
        h[n//4:n//2] = 1.0
        h[3*n//4:n] = 1.0
    return project_feasible(h, n)


def coordinate_refine(h, n, max_sweeps=30, time_limit=60):
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    start = time.time()
    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        if time.time() - start > time_limit:
            break
        improved_sweep = False
        for i in range(n):
            old = h[i]
            if 0.05 < old < 0.95:
                cands = [old + 0.05, old - 0.05, old + 0.15, old - 0.15, 0.0, 1.0, 0.5]
            else:
                cands = [0.0, 1.0, 0.3, 0.7]
            best_cand_val = old
            best_cand_c5 = best_c5
            for c in cands:
                if c < 0.0 or c > 1.0 or abs(c - old) < 1e-7:
                    continue
                h[i] = c
                h = project_feasible(h, n)
                c5_v = compute_c5(h)
                if c5_v < best_cand_c5 - 1e-14:
                    best_cand_c5 = c5_v
                    best_cand_val = c
                    improved_sweep = True
            h[i] = best_cand_val
            h = project_feasible(h, n)
        if not improved_sweep:
            break
    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _CUR_N, _BETA, _TOPK
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    # Load parent
    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = np.zeros(640)
        best_h[:320] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    n = best_n
    h = best_h.copy()

    # Phase 1: SLSQP with increasing beta on parent resolution
    for beta in [2000, 10000, 50000, 200000, 1000000, 5000000, 20000000]:
        if remaining_time() < 25:
            break
        h_new = slsqp_opt(h, n, beta, topk=20, maxiter=1500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Projected gradient descent (fast, multiple betas)
    for beta in [5000, 50000, 500000]:
        if remaining_time() < 15:
            break
        h_pg, c5_pg = proj_grad_descent(h, n, beta=beta, max_iters=20000, time_limit=45, topk=50)
        if c5_pg < best_c5 - 1e-15:
            best_c5 = c5_pg
            best_h = h_pg.copy()
            h = h_pg.copy()

    # Phase 3: Top-k equalization with increasing k
    for tk in [10, 20, 40, 60]:
        if remaining_time() < 15:
            break
        h_eq, c5_eq = top_k_equalize(h, n, top_k=tk, max_iters=3000, time_limit=40)
        if c5_eq < best_c5 - 1e-15:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 4: Subgradient zigzag
    if remaining_time() > 30:
        h_sg, c5_sg = subgradient_zigzag(h, n, max_iters=15000, time_limit=80)
        if c5_sg < best_c5 - 1e-15:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 5: More SLSQP with very high beta
    for beta in [5000000, 20000000, 100000000]:
        if remaining_time() < 20:
            break
        h_new = slsqp_opt(h, n, beta, topk=30, maxiter=2000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 6: Resolution increase to n=1280
    n_high = 1280
    if remaining_time() > 80:
        h_interp = interp_to(best_h, n, n_high)
        h_feas = project_feasible(h_interp, n_high)
        c5_int = compute_c5(h_feas)
        h_opt = h_feas.copy()
        for beta in [5000, 50000, 500000, 5000000]:
            if remaining_time() < 20:
                break
            h_opt = slsqp_opt(h_opt, n_high, beta, topk=25, maxiter=800)
        c5_opt = compute_c5(h_opt)
        h_opt, c5_opt = proj_grad_descent(h_opt, n_high, beta=100000, max_iters=15000, time_limit=40, topk=50)
        h_opt, c5_opt = top_k_equalize(h_opt, n_high, top_k=30, max_iters=2000, time_limit=30)
        h_opt, c5_opt = subgradient_zigzag(h_opt, n_high, max_iters=8000, time_limit=30)
        if c5_opt < best_c5 - 1e-15:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            n = n_high

    # Phase 7: Resolution 960 as alternative
    n_mid = 960
    if remaining_time() > 50:
        h_interp = interp_to(best_h, n, n_mid)
        h_feas = project_feasible(h_interp, n_mid)
        h_opt = h_feas.copy()
        for beta in [10000, 100000, 1000000]:
            if remaining_time() < 15:
                break
            h_opt = slsqp_opt(h_opt, n_mid, beta, topk=20, maxiter=1000)
        h_opt, c5_opt = proj_grad_descent(h_opt, n_mid, beta=200000, max_iters=10000, time_limit=30, topk=40)
        h_opt, c5_opt = top_k_equalize(h_opt, n_mid, top_k=20, max_iters=1500, time_limit=25)
        if c5_opt < best_c5 - 1e-15:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            n = n_mid

    # Phase 8: Binary simulated annealing
    if remaining_time() > 80:
        h_bin = (best_h > 0.5).astype(float)
        sa_time = min(120, int(remaining_time() * 0.3))
        h_b, c5_b = binary_sa(h_bin, n, time_limit=sa_time, rng=rng)
        # Refine binary result continuously
        h_b_c = h_b.copy()
        for beta in [50000, 500000, 5000000]:
            if remaining_time() < 15:
                break
            h_b_c = slsqp_opt(h_b_c, n, beta, topk=20, maxiter=1000)
        c5_b_c = compute_c5(h_b_c)
        h_b_c, c5_b_c = proj_grad_descent(h_b_c, n, beta=200000, max_iters=10000, time_limit=30, topk=40)
        h_b_c, c5_b_c = top_k_equalize(h_b_c, n, top_k=20, max_iters=1500, time_limit=25)

        if c5_b_c < best_c5 - 1e-15:
            best_c5 = c5_b_c
            best_h = h_b_c.copy()
        if c5_b < best_c5 - 1e-15:
            best_c5 = c5_b
            best_h = h_b.copy().astype(float)
            n = len(best_h)

    # Phase 9: Binary greedy local search
    if remaining_time() > 40:
        h_bin = (best_h > 0.5).astype(float)
        h_g, c5_g = binary_greedy_local(h_bin, n, time_limit=min(50, int(remaining_time()*0.3)), rng=rng)
        h_g_c = h_g.copy()
        for beta in [50000, 500000, 5000000]:
            if remaining_time() < 15:
                break
            h_g_c = slsqp_opt(h_g_c, n, beta, topk=25, maxiter=1000)
        h_g_c, c5_g_c = top_k_equalize(h_g_c, n, top_k=20, max_iters=1000, time_limit=20)
        if c5_g_c < best_c5 - 1e-15:
            best_c5 = c5_g_c
            best_h = h_g_c.copy()
            n = len(best_h)

    # Phase 10: Diverse starting points
    if remaining_time() > 60:
        alt_starts = [
            theta_morse(n),
            structured_pattern(n, 0),
            structured_pattern(n, 2),
        ]
        # Random balanced
        h_rand = np.zeros(n)
        h_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        alt_starts.append(h_rand)

        for h_alt in alt_starts:
            if remaining_time() < 30:
                break
            h_opt = h_alt.copy()
            h_opt = slsqp_opt(h_opt, n, 10000, topk=20, maxiter=800)
            h_opt = slsqp_opt(h_opt, n, 100000, topk=20, maxiter=800)
            h_opt, c5_opt = proj_grad_descent(h_opt, n, beta=50000, max_iters=10000, time_limit=25, topk=40)
            h_opt, c5_opt = top_k_equalize(h_opt, n, top_k=20, max_iters=1000, time_limit=20)
            if c5_opt < best_c5 - 1e-15:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                n = len(best_h)

    # Phase 11: Final intensive SLSQP refinement
    h = best_h.copy()
    for beta in [5000000, 20000000, 100000000]:
        if remaining_time() < 12:
            break
        h_new = slsqp_opt(h, n, beta, topk=30, maxiter=2000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 12: Final top-k equalization
    if remaining_time() > 15:
        h_eq, c5_eq = top_k_equalize(best_h, n, top_k=30, max_iters=2000, time_limit=30)
        if c5_eq < best_c5 - 1e-15:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Phase 13: Subgradient final
    if remaining_time() > 20:
        h_sg, c5_sg = subgradient_zigzag(best_h, n, max_iters=10000, time_limit=min(50, int(remaining_time()*0.5)))
        if c5_sg < best_c5 - 1e-15:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 14: Coordinate refinement
    if remaining_time() > 20:
        h_cr, c5_cr = coordinate_refine(best_h, n, max_sweeps=20, time_limit=min(50, int(remaining_time()*0.5)))
        if c5_cr < best_c5 - 1e-15:
            best_c5 = c5_cr
            best_h = h_cr.copy()

    # Phase 15: Final SLSQP polish
    if remaining_time() > 12:
        h_final = slsqp_opt(best_h, n, 50000000, topk=40, maxiter=1500)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-15:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Final projection and verification
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, n)
