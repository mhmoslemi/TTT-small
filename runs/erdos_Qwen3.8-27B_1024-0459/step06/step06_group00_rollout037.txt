import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0

def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, n, beta):
    """Compute objective and gradient of smooth max."""
    dx = 2.0 / n
    corr = compute_corr(h)
    c_max = np.max(corr)
    z = np.clip(beta * (corr - c_max), -50.0, 50.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S  # weights for each lag
    obj = (c_max + np.log(S) / beta) * dx
    
    # Gradient
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - h[a:])
            g[a:] -= wk * h[:np_]
    return obj, g

def prox_step(x, n, step):
    """Take a proximal step: minimize step*0.5*||h-x||^2 + indicator({0<=h<=1, sum(h)=n/2})"""
    # This is: find h = argmin 0.5*||h-x||^2 s.t. 0<=h<=1, sum(h)=n/2
    # KKT: h = clip(x - tau, 0, 1) for some tau, with sum(h) = n/2
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(60):
        tau = (lo + hi) / 2
        h = np.clip(x - tau, 0.0, 1.0)
        s = np.sum(h)
        if s > n / 2.0:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) / 2
    return np.clip(x - tau, 0.0, 1.0)

def fista_optimize(h0, n, beta, max_iter=5000, time_limit=None, start_time=None):
    """FISTA optimization of smooth max."""
    if start_time is None:
        start_time = time.time()
    h = project_feasible(h0.copy(), n)
    h_prev = h.copy()
    y = h.copy()
    t_k = 1.0
    best_h = h.copy()
    best_obj = compute_c5(h)
    
    # Initial step size
    step = 0.5 / (beta * n)  # conservative initial step
    
    for it in range(max_iter):
        if time_limit is not None and time.time() - start_time > time_limit:
            break
        if remaining_time() < 5:
            break
        
        obj, g = smooth_max_obj_grad(y, n, beta)
        
        # Backtracking line search
        while True:
            y_new = y - step * g
            h_new = prox_step(y_new, n, step)
            # Check sufficient decrease (using current obj as reference is not quite right for FISTA)
            # For FISTA, we can just take the step if it's reasonable
            if np.all(np.abs(h_new) <= 1.01) and np.sum(np.isnan(h_new)) == 0:
                break
            step *= 0.5
            if step < 1e-15:
                break
        
        h_next = h_new
        t_k_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = h_next + ((t_k - 1.0) / t_k_new) * (h_next - h)
        h = h_next
        t_k = t_k_new
        
        c5_curr = compute_c5(h)
        if c5_curr < best_obj - 1e-12:
            best_obj = c5_curr
            best_h = h.copy()
        
        # Adaptive step: increase if we're making progress
        if it % 100 == 0 and it > 0:
            step *= 1.1
    
    return best_h, best_obj

def binary_local_search(h, n, time_limit=60, rng=None):

        if no_improve > max_no_improve:
            break
        
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        
        # Random 2-swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Evaluate
        corr_old = compute_corr(h_bin)
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        corr_new = compute_corr(h_bin)
        c5_new = np.max(corr_new) * dx
        
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_bin.copy()
            ones[np.where(ones == i)[0][0]] = j
            zeros[np.where(zeros == j)[0][0]] = i
            no_improve = 0
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            no_improve += 1
    
    return best_h, best_c5

def binary_block_swap(h, n, block_size=2, time_limit=60, rng=None):
    """Binary block swap local search."""
    if rng is None:
        rng = np.random.RandomState(42)
    
    h_bin = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        sort_idx = np.argsort(-h_bin)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0
    
    dx = 2.0 / n
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    start = time.time()
    no_improve = 0
    
    while time.time() - start < time_limit and remaining_time() > 5:
        if no_improve > 5000:
            break
        
        # Find a block of 'block_size' consecutive 1s
        # and a block of 'block_size' consecutive 0s
        ones_mask = (h_bin == 1.0)
        zeros_mask = (h_bin == 0.0)
        
        # Find consecutive blocks
        one_blocks = []
        i = 0
        while i < n:
            if ones_mask[i]:
                j = i
                while j < n and ones_mask[j]:
                    j += 1
                if j - i >= block_size:
                    one_blocks.append(i)
                i = j
            else:
                i += 1
        
        zero_blocks = []
        i = 0
        while i < n:
            if zeros_mask[i]:
                j = i
                while j < n and zeros_mask[j]:
                    j += 1
                if j - i >= block_size:
                    zero_blocks.append(i)
                i = j
            else:
                i += 1
        
        if not one_blocks or not zero_blocks:
            break
        
        oi = one_blocks[rng.randint(len(one_blocks))]
        zj = zero_blocks[rng.randint(len(zero_blocks))]
        
        # Swap blocks
        for b in range(block_size):
            h_bin[oi + b] = 0.0
            h_bin[zj + b] = 1.0
        
        c5_new = compute_c5(h_bin)
        
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_bin.copy()
            no_improve = 0
        else:
            for b in range(block_size):
                h_bin[oi + b] = 1.0
                h_bin[zj + b] = 0.0
            no_improve += 1
    
    return best_h, best_c5

def thue_morse(n):
    """Generate Thue-Morse sequence of length n."""
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)

def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)

    def sm_grad(x):
        return grad_smooth_max(x, n, beta)

        return np.sum(x) - n / 2.0

        return np.ones(n)

    if c5_1 < best_c5 - 1e-14:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Top-k directional on parent
    h2, c5_2 = top_k_directional(best_h, best_n, top_k=8, max_iters=500, time_limit=60)
    if c5_2 < best_c5 - 1e-14:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Binary 2-swap SA
    h_bin = (best_h > 0.5).astype(float)
    h_b, c5_b = binary_2swap(h_bin, best_n, time_limit=90, rng=rng, sa=True)
    if c5_b < best_c5 - 1e-14:
        best_c5 = c5_b
        best_h = h_b.copy().astype(float)

    # Refine binary result continuously
    h_br = h_b.copy().astype(float)
    h_br, c5_br = proximal_linear_method(h_br, best_n, max_iter=2000, time_limit=60)
    if c5_br < best_c5 - 1e-14:
        best_c5 = c5_br
        best_h = h_br.copy()
    h_br, c5_br = top_k_directional(h_br, best_n, top_k=6, max_iters=300, time_limit=30)
    if c5_br < best_c5 - 1e-14:
        best_c5 = c5_br
        best_h = h_br.copy()

    # Phase 4: Block swap
    if remaining_time() > 30:
        h_bs, c5_bs = binary_block_swap(best_h, best_n, block_size=2, time_limit=45, rng=rng)
        if c5_bs < best_c5 - 1e-14:
            best_c5 = c5_bs
            best_h = h_bs.copy().astype(float)
        h_bs3, c5_bs3 = binary_block_swap(best_h, best_n, block_size=3, time_limit=45, rng=rng)
        if c5_bs3 < best_c5 - 1e-14:
            best_c5 = c5_bs3
            best_h = h_bs3.copy().astype(float)

    # Phase 5: Multi-resolution
    n_candidates = [800, 768, 512, 1024]
    for n_t in n_candidates:
        if remaining_time() < 60:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        h_opt = h_feas
        c5_opt = compute_c5(h_opt)

        # Proximal at new resolution
        h_opt, c5_opt = proximal_linear_method(h_opt, n_t, max_iter=2000, time_limit=40,
                                               beta_start=1000, beta_end=50000)
        h_opt, c5_opt = top_k_directional(h_opt, n_t, top_k=6, max_iters=300, time_limit=30)

        # SLSQP polish
        if remaining_time() > 20:
            h_s = slsqp_opt(h_opt, n_t, 50000, maxiter=500)
            c5_s = compute_c5(h_s)
            if c5_s < c5_opt - 1e-14:
                h_opt = h_s
                c5_opt = c5_s

        # Binary search at new resolution
        h_bin_t = (h_opt > 0.5).astype(float)
        h_bi, c5_bi = binary_2swap(h_bin_t, n_t, time_limit=30, rng=rng)
        if c5_bi < c5_opt - 1e-14:
            h_opt = h_bi.astype(float)
            c5_opt = c5_bi

        if c5_opt < best_c5 - 1e-14:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t

    # Phase 6: Alternative starts
    if remaining_time() > 80:
        starts = [
            thue_morse(best_n),
            np.roll(best_h, 1),
            np.roll(best_h, best_n // 96),
        ]
        # Random balanced
        h_rand = np.zeros(best_n)
        h_rand[rng.choice(best_n, best_n // 2, replace=False)] = 1.0
        starts.append(h_rand)

        for h_s in starts:
            if remaining_time() < 40:
                break
            h_s = project_feasible(h_s, best_n)
            h_opt, c5_opt = multi_start_optimize(h_s, best_n, time_limit=min(60, remaining_time() * 0.3))
            if c5_opt < best_c5 - 1e-14:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 7: Final intensive refinement
    n = best_n
    h = best_h.copy()

    # Multiple proximal passes with different betas
    for beta_s, beta_e in [(1000, 10000), (5000, 100000), (10000, 500000)]:
        if remaining_time() < 30:
            break
        h_new, c5_new = proximal_linear_method(h, n, max_iter=3000, time_limit=60,
                                               beta_start=beta_s, beta_end=beta_e, mu_init=0.5)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Top-k with larger k
    if remaining_time() > 20:
        h_eq, c5_eq = top_k_directional(h, n, top_k=15, max_iters=500, time_limit=40)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # SLSQP final polish
    for beta in [100000, 1000000, 5000000]:
        if remaining_time() < 15:
            break
        h_new = slsqp_opt(h, n, beta, maxiter=1000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Final binary refinement
    if remaining_time() > 20:
        h_fin_bin = (best_h > 0.5).astype(float)
        h_fb, c5_fb = binary_2swap(h_fin_bin, n, time_limit=min(40, remaining_time() * 0.3), rng=rng)
        if c5_fb < best_c5 - 1e-14:
            best_c5 = c5_fb
            best_h = h_fb.copy().astype(float)

        # Re-refine continuously
        if remaining_time() > 15:
            h_rec, c5_rec = proximal_linear_method(best_h, n, max_iter=2000, time_limit=30,
                                                   beta_start=5000, beta_end=200000)
            if c5_rec < best_c5 - 1e-14:
                best_c5 = c5_rec
                best_h = h_rec.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, n)
