import numpy as np
from scipy.optimize import minimize
from scipy.signal import fftconvolve
import time

_T0 = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_smooth_max(corr, dx, top_k=50, beta=200):
    """Log-sum-exp smooth max of corr * dx"""
    c = corr[:top_k] if len(corr) >= top_k else corr
    m = np.max(c)
    if beta > 1000:
        # Very sharp: just use top few
        val = m * dx
        return val
    sh = np.clip(beta * (c - m), -100, 100)
    ex = np.exp(sh)
    val = (m + np.log(ex.sum()) / beta) * dx
    return val

def grad_smooth_max(h, n, top_k=50, beta=200):
    """Gradient of smooth max of correlation"""
    corr = compute_corr_full(h)
    dx = 2.0 / n
    # Use full correlation but weight by softmax
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -100, 100)
    ex = np.exp(sh)
    w = ex / ex.sum()
    
    g = np.zeros(n)
    # For each lag, compute gradient of that correlation term
    # corr[j] = sum_i h[i] * (1-h)[i+j-(n-1)]
    # = sum_i h[i] * (1-h[i+j-(n-1)])
    # d/dh[x]:
    #   if x appears in h[i]: +1 * (1-h[x+j-(n-1)]) for valid indices
    #   if x appears in (1-h)[i+j-(n-1)]: -h[x] for valid indices
    
    # More efficiently:
    # corr[j] = sum_i h[i] - sum_i h[i]*h[i+j-(n-1)]
    # Let lag = j - (n-1)
    # If lag >= 0: corr[j] = sum_{i=0}^{n-1-lag} h[i] - sum_{i=0}^{n-1-lag} h[i]*h[i+lag]
    # If lag < 0: corr[j] = sum_{i=-lag}^{n-1} h[i] - sum_{i=-lag}^{n-1} h[i]*h[i+lag]
    
    for j in range(len(corr)):
        if w[j] * dx < 1e-14:
            continue
        wk = w[j] * dx
        lag = j - (n - 1)
        if lag >= 0:
            np_ = n - lag
            # d/dh[x] of sum_{i=0}^{np_-1} h[i] * (1-h[i+lag])
            # = (1 if x < np_) - sum of h[i]*... terms
            # For term h[i]*h[i+lag]:
            #   d/dh[x]: h[x+lag] if x < np_, h[x-lag] if x >= lag, combined
            for x in range(n):
                val = 0.0
                if x < np_:
                    val += (1.0 - (h[x + lag] if x + lag < n else 0.0))
                if x >= lag:
                    val -= h[x - lag]
                g[x] += wk * val
        else:
            a = -lag
            np_ = n - a
            for x in range(n):
                val = 0.0
                if x >= a:
                    val += (1.0 - (h[x - a] if x - a >= 0 else 0.0))
                if x < np_:
                    val -= h[x + a]

            g[lag:] -= wj * h[:np_]
        else:
            a = -lag
            np_ = n - a
            g[a:] += wj * (1.0 - h[:np_])
            g[:np_] -= wj * h[a:]
    return g

def lse_value(h, n, beta):
    corr = compute_corr_full(h)
    dx = 2.0 / n
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -200.0, 200.0)
    ex = np.exp(sh)
    s = ex.sum()
    if s < 1e-300:
        return m * dx
    return (m + np.log(s) / beta) * dx

def gradient_descent(h, n, time_limit, beta_start=50, beta_end=500, K=20, max_iter=2000):

    h_best = h.copy()
    c_best = compute_c5(h)
    start = time.time()
    n_iters = max_iter
    
    for it in range(n_iters):
        elapsed_local = time.time() - start
        if elapsed_local > time_limit:
            break
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        
        # Adaptive beta
        frac = min(1.0, it / (0.8 * n_iters))
        beta = beta_start + (beta_end - beta_start) * frac
        
        g = grad_lse_topk(h_curr, n, beta, K)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        
        # Line search
        best_step = 0
        best_val = lse_value(h_curr, n, beta)
        base_alpha = 1.0 / gn
        for factor in [1.0, 0.5, 0.25, 0.1, 2.0, 0.05]:
            alpha = min(base_alpha * factor, 0.5)
            h_try = project_feasible(h_curr - alpha * g, n)
            val_try = lse_value(h_try, n, beta)
            if val_try < best_val - 1e-15:
                best_val = val_try
                best_step = alpha
                h_curr = h_try
                break
        
        c_curr = compute_c5(h_curr)
        if c_curr < c_best:
            c_best = c_curr
            h_best = h_curr.copy()

def binary_sa(h_init, n, time_limit, T_init=0.5, T_final=0.01):
    # Round to binary
    h_bin = (h_init > 0.5).astype(float)
    # Fix sum
    target = n // 2
    ones = np.sum(h_bin)
    if ones > target:
        # Remove excess ones
        excess = target - int(ones)  # negative
        idx_ones = np.where(h_bin == 1)[0]
        remove = np.random.choice(idx_ones, -excess, replace=False)
        h_bin[remove] = 0
    elif ones < target:
        deficit = target - int(ones)
        idx_zeros = np.where(h_bin == 0)[0]
        add = np.random.choice(idx_zeros, deficit, replace=False)
        h_bin[add] = 1
    
    h_curr = h_bin.copy()
    h_best = h_bin.copy()
    c_curr = compute_c5(h_curr)
    c_best = c_curr
    start = time.time()
    rng = np.random.RandomState(42)
    
    n_iters = int(time_limit * 800)  # target iteration rate
    total_iters = max(n_iters, 1000)
    cooling = (T_final / T_init) ** (1.0 / total_iters)
    
    for it in range(total_iters):
        elapsed = time.time() - start
        if elapsed > time_limit:
            break
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        
        T = T_init * (cooling ** it)
        
        # Generate move: flip a 1 to 0 and a 0 to 1 (maintain sum)
        ones_idx = np.where(h_curr == 1)[0]
        zeros_idx = np.where(h_curr == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        
        # Random group flip: flip 1-3 pairs
        n_pairs = rng.randint(1, 4)
        flip_from = rng.choice(ones_idx, n_pairs, replace=False)
        flip_to = rng.choice(zeros_idx, n_pairs, replace=False)
        
        h_try = h_curr.copy()
        h_try[flip_from] = 0
        h_try[flip_to] = 1
        
        c_try = compute_c5(h_try)
        delta = c_try - c_curr
        
        if delta < 0 or (T > 1e-10 and rng.rand() < np.exp(-delta / T)):
            h_curr = h_try
            c_curr = c_try
            if c_curr < c_best:
                c_best = c_curr
                h_best = h_curr.copy()
    
    return h_best, c_best

def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def slsqp_refine(h, n, K, beta, maxiter=300):
    def obj(x):
        return lse_value(x, n, beta)
    
    def grad(x):
        return grad_lse_topk(x, n, beta, K)
    
    def cfun(x):
        return np.sum(x) - n / 2.0
    
    def cjac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            c5_old = compute_c5(h)
            if c5_new <= c5_old + 1e-12:
                return h_new
    except Exception:
        pass
    return h

        if c1 < best_c5:
            best_c5 = c1
            best_h = h1.copy()
            best_n = n

    # Phase 2: SLSQP refinement (multiple parameter sets)
    for K in [10, 15, 25]:
        for beta in [300, 1000, 5000]:
            if remaining() < 12:
                break
            h_new = slsqp_refine(best_h, best_n, K, beta, maxiter=300)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-13:
                best_c5 = c_new
                best_h = h_new.copy()

    # Phase 3: Binary SA (main exploration)
    if remaining() > 120:
        sa_time = min(320.0, remaining() * 0.35)
        h_sa, c5_sa = binary_sa(best_h, best_n, sa_time, T_init=0.4, T_final=0.002, rng_seed=seed + 1)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            best_n = len(h_sa)

    # Phase 3b: Relax SA result back to continuous and gradient-optimize
    if remaining() > 80:
        h_relax = best_h.copy()
        # Add small continuous perturbation around binary
        rng2 = np.random.RandomState(seed + 99)
        h_relax += rng2.randn(best_n) * 0.03
        h_relax = project_feasible(h_relax, best_n)
        t_alloc3b = min(80.0, remaining() * 0.2)
        h_gr, c_gr = gradient_descent(h_relax, best_n, t_alloc3b,
                                      beta_start=100, beta_end=800, K=15)
        if c_gr < best_c5:
            best_c5 = c_gr
            best_h = h_gr.copy()
            best_n = len(h_gr)

    # Phase 4: SLSQP refinement at higher beta
    for K in [8, 12, 20]:
        for beta in [2000, 8000, 30000]:
            if remaining() < 10:
                break
            h_new = slsqp_refine(best_h, best_n, K, beta, maxiter=350)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-13:
                best_c5 = c_new
                best_h = h_new.copy()

    # Phase 5: Second binary SA (finer temperature)
    if remaining() > 80:
        sa_time2 = min(150.0, remaining() * 0.3)
        h_sa2, c5_sa2 = binary_sa(best_h, best_n, sa_time2,
                                  T_init=0.15, T_final=0.001, rng_seed=seed + 42)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()
            best_n = len(h_sa2)

    # Phase 6: Final gradient pass
    if remaining() > 40:
        t6 = min(80.0, remaining() * 0.3)
        h6, c6 = gradient_descent(best_h, best_n, t6,
                                  beta_start=300, beta_end=2000, K=12)
        if c6 < best_c5:
            best_c5 = c6
            best_h = h6.copy()
            best_n = len(h6)

    # Phase 7: Final SLSQP polish
    for beta in [10000, 30000, 100000]:
        if remaining() < 8:
            break
        h_new = slsqp_refine(best_h, best_n, 10, beta, maxiter=500)
        c_new = compute_c5(h_new)
        if c_new < best_c5 - 1e-13:
            best_c5 = c_new
            best_h = h_new.copy()

    # Phase 8: Try n=2048 if time allows
    if remaining() > 250:
        n_hi = 2048
        h_hi = project_feasible(interp_to(best_h, best_n, n_hi), n_hi)
        c5_hi = compute_c5(h_hi)
        if c5_hi < best_c5 + 0.003:
            if remaining() > 100:
                h_hi, c5_hi = gradient_descent(h_hi, n_hi, min(80.0, remaining() * 0.2),
                                               beta_start=50, beta_end=300, K=20)
            if remaining() > 40:
                h_hi = slsqp_refine(h_hi, n_hi, 15, 2000, maxiter=250)
                c5_hi = compute_c5(h_hi)
            if remaining() > 30:
                h_hi, c5_hi = gradient_descent(h_hi, n_hi, min(30.0, remaining() * 0.2),
                                               beta_start=500, beta_end=2000, K=15)
            if c5_hi < best_c5:
                best_c5 = c5_hi
                best_h = h_hi.copy()
                best_n = n_hi

    # Phase 9: Final fine SA on best result
    if remaining() > 60:
        h_sa3, c5_sa3 = binary_sa(best_h, best_n, min(60.0, remaining() * 0.5),
                                  T_init=0.05, T_final=0.0005, rng_seed=seed + 200)
        if c5_sa3 < best_c5:
            best_c5 = c5_sa3
            best_h = h_sa3.copy()
            best_n = len(h_sa3)

    # Final SLSQP
    if remaining() > 5:
        h_new = slsqp_refine(best_h, best_n, 10, 50000, maxiter=400)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Final feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
