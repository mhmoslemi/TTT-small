def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_all_overlaps(h):
    """Return the array of all overlap values (before scaling by dx)."""
    return np.correlate(h, 1.0 - h, mode='full')

def grad_at_shift(h, k_idx, n):
    """Gradient of overlap at shift k_idx (index into full correlation array)."""
    # k_idx = 0 corresponds to shift -(n-1), ..., k_idx = n-1 corresponds to shift 0, ...
    # Actually in np.correlate full mode, index n-1 is shift 0
    shift = k_idx - (n - 1)
    g = np.zeros(n)
    if shift == 0:
        g = 1.0 - 2.0 * h
    elif shift > 0:
        npairs = n - shift
        g[:npairs] += (1.0 - h[shift:])
        g[shift:] -= h[:npairs]
    else:
        a = -shift
        npairs = n - a
        g[:npairs] += (1.0 - h[a:])
        g[a:] -= h[:npairs]
    return g

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max(corr, beta):
    m = np.max(corr)
    shifted = corr - m
    clamped = np.clip(beta * shifted, -50, 50)
    exp_vals = np.exp(clamped)
    logsumexp = m + np.log(exp_vals.sum()) / beta
    weights = exp_vals / exp_vals.sum()
    return logsumexp, weights

                nv = old + d
                if nv < -1e-6 or nv > 1.0 + 1e-6:
                    continue
                h_cur[i] = nv
                h_cur = project_feasible(h_cur, n)
                c = compute_c5(h_cur)
                if c < best_c5 - 1e-16:
                    best_c5 = c
                    improved = True
                    stall = 0
                    break
                else:
                    h_cur[i] = old
            else:
                h_cur[i] = old
                h_cur = project_feasible(h_cur, n)
        if not improved:
            stall += 1
            if stall > 3:
                # Try random pairwise swaps
                for _ in range(200):
                    if time.time() - t0 > time_limit:
                        break
                    i, j = rng.randint(0, n), rng.randint(0, n)
                    if i == j:
                        continue
                    tmp = h_cur[i]
                    h_cur[i] = h_cur[j]
                    h_cur[j] = tmp + old  # won't work, just swap
                    # Actually just do a clean swap
                    h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
                    h_cur = project_feasible(h_cur, n)
                    c = compute_c5(h_cur)
                    if c < best_c5 - 1e-16:
                        best_c5 = c
                        improved = True
                        stall = 0
                        break
                if not improved:
                    break
    return h_cur, best_c5


def slsqp_polish(h, n, beta, maxiter):
    def obj_fn(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = float(np.max(corr))
        sh = corr - m
        cl = np.clip(beta * sh, -50, 50)
        ex = np.exp(cl)
        return (m + np.log(ex.sum()) / beta) * (2.0 / n)

    def grad_fn(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = float(np.max(corr))
        sh = corr - m
        cl = np.clip(beta * sh, -50, 50)
        ex = np.exp(cl)
        w = ex / ex.sum()
        w *= 2.0 / n
        g = np.zeros(n)
        for k_idx in range(len(corr)):
            wk = w[k_idx]
            if wk < 1e-18:
                continue
            g += wk * grad_overlap(x, k_idx, n)
        return g

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj_fn, h, jac=grad_fn, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def exact_subgrad_polish(h, n, time_limit):
    h_cur = h.copy()
    best_c5 = compute_c5(h_cur)
    t0 = time.time()
    for it in range(2000):
        if time.time() - t0 > time_limit:
            break
        rem = _BUDGET - (time.time() - _T0) - 5
        if rem < 2:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr))
        g = grad_overlap(h_cur, k_star, n)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn
        step = 0.001 * (1.0 / (1.0 + 0.01 * it))
        h_try = project_feasible(h_cur - step * g, n)
        c_try = compute_c5(h_try)
        if c_try < best_c5 - 1e-16:
            best_c5 = c_try
            h_cur = h_try.copy()
        # Also try half step
        else:
            h_try2 = project_feasible(h_cur - step * 0.5 * g, n)
            c_try2 = compute_c5(h_try2)
            if c_try2 < best_c5 - 1e-16:
                best_c5 = c_try2
                h_cur = h_try2.copy()
            else:
                # Try opposite direction (maybe we hit a boundary)
                h_try3 = project_feasible(h_cur + step * 0.5 * g, n)
                c_try3 = compute_c5(h_try3)
                if c_try3 < best_c5 - 1e-16:
                    best_c5 = c_try3
                    h_cur = h_try3.copy()
                else:
                    if it > 100 and it % 200 == 0:
                        break
    return h_cur, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def rem_time():
        return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)

    # Initialize from parent
    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: LBFGS with moderate beta (400s budget)
    for beta in [300, 800, 2000, 5000]:
        if rem_time() < 80:
            break
        h_new = lbfgs_smooth(h, n, beta, 200, maxiter=4000)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Momentum subgradient (200s budget)
    if rem_time() > 150:
        h_sg, c5_sg = momentum_subgrad(h, n, min(200.0, rem_time() * 0.6), rng)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 3: LBFGS with high beta (200s budget)
    for beta in [10000, 25000, 50000]:
        if rem_time() < 50:
            break
        h_new = lbfgs_smooth(h, n, beta, 5000, maxiter=2000)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 4: SLSQP polish (100s budget)
    if rem_time() > 60:
        h_new = slsqp_polish(h, n, 20000, maxiter=800)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 5: Fine local search (200s budget)
    if rem_time() > 80:
        h_ls, c5_ls = fine_local_search(h, n, min(200.0, rem_time() * 0.55), rng)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
            h = h_ls.copy()

    # Phase 6: Exact subgradient polish (100s budget)
    if rem_time() > 50:
        h_es, c5_es = exact_subgrad_polish(h, n, min(100.0, rem_time() * 0.5))
        if c5_es < best_c5:
            best_c5 = c5_es
            best_h = h_es.copy()
            h = h_es.copy()

    # Phase 7: Another round of fine local search
    if rem_time() > 60:
        h_ls2, c5_ls2 = fine_local_search(h, n, min(80.0, rem_time() * 0.6), rng)
        if c5_ls2 < best_c5:
            best_c5 = c5_ls2
            best_h = h_ls2.copy()

    # Phase 8: Final exact subgradient
    if rem_time() > 30:
        h_es2, c5_es2 = exact_subgrad_polish(h, n, min(40.0, rem_time() * 0.7))
        if c5_es2 < best_c5:
            best_c5 = c5_es2
            best_h = h_es2.copy()

    # Final feasibility check
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = best_h.copy()

    try:
        ev = evaluate_erdos_solution(best_h)
        if ev < best_c5:
            best_c5 = ev
    except Exception:
        pass

    return (best_h, best_c5, n)
