def compute_c5(h):

def grad_corr_k_vec(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        np_ = n - offset
        g[offset:] += (1.0 - h[:np_])
        g[:np_] -= h[offset:]
    else:
        a = -offset
        np_ = n - a
        g[:np_] += (1.0 - h[a:])
        g[a:] -= h[:np_]
    return g


def smmax_objective(h):
    n = _G_N
    corr = np.correlate(h, 1.0 - h, mode='full')
    beta = _G_BETA
    K = _G_K
    m = corr.max()
    idx = np.argsort(-corr)[:K]
    c = corr[idx]
    sh = c - c.max()
    cl = np.clip(beta * sh, -80.0, 80.0)
    ex = np.exp(cl)
    val = (c.max() + np.log(ex.sum()) / beta) * (2.0 / n)
    return val


def smmax_gradient(h):
    n = _G_N
    beta = _G_BETA
    K = _G_K
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(-corr)[:K]
    c = corr[idx]
    sh = c - c.max()
    cl = np.clip(beta * sh, -80.0, 80.0)
    ex = np.exp(cl)
    w = ex / ex.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for i in range(K):
        if w[i] < 1e-15:
            continue
        wk = w[i] * dx
        k = idx[i]

        else:

            np_ = n - a
            g[:np_] += wk * (1.0 - h[a:])

    return g

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def binary_with_exact_count(h, n, rng):
    """Snap h to binary with exactly n/2 ones."""
    h_bin = (h > 0.5).astype(float)
    cnt = int(np.sum(h_bin))
    target = n // 2
    if cnt > target:
        positions = np.where(h_bin == 1)[0]
        order = np.argsort(h[positions])
        for idx in order[:cnt - target]:
            h_bin[positions[idx]] = 0
    elif cnt < target:
        positions = np.where(h_bin == 0)[0]
        order = np.argsort(-h[positions])
        for idx in order[:target - cnt]:
            h_bin[positions[idx]] = 1
    return h_bin
