def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):

        if abs(diff) < 1e-13:

        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if interior.any():
            h[interior] += diff / interior.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):

    s = np.clip(beta * (corr - m), -500.0, 500.0)

import numpy as np
from scipy.optimize import minimize, LinearConstraint, Bounds
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / n)


def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        if interior.any():
            h[interior] += diff / interior.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_eq_func(x, n):
    return x.sum() - n / 2.0


def smooth_eq_jac(x, n):
    return np.ones(n)


def smooth_max_func(h, beta):
    corr = compute_corr(h)
    n = len(h)
    dx = 2.0 / n
    m = corr.max()
    s = np.clip(beta * (corr - m), -500.0, 500.0)
    e = np.exp(s)
    return (m + np.log(e.sum()) / beta) * dx


def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = corr.max()
    s = np.clip(beta * (corr - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        d = k - (n - 1)
        if d == 0:
            g += wk * (1.0 - 2.0 * h)
        elif d > 0:
            g[d:] += wk * (1.0 - h[:n - d])
            g[:n - d] -= wk * h[d:]
        else:
            e2 = -d
            g[:n - e2] += wk * (1.0 - h[e2:])
            g[e2:] -= wk * h[:n - e2]
    return g * dx

            break
        if _BUDGET - (time.time() - _T0) < 10:
            break

        corr = compute_corr(h)
        top_idx = np.argsort(-corr)[:K]
        c_top = corr[top_idx]
        w = np.exp(beta_w * (c_top - c_top.max()))
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(K) / K
        else:
            w /= w_sum

        g = np.zeros(n)
        for i in range(K):
            wk = w[i]
            if wk < 1e-13:
                continue
            k_idx = top_idx[i]
            d = k_idx - (n - 1)
            if d == 0:
                g += wk * (1.0 - 2.0 * h)
            elif d > 0:
                g[d:] += wk * (1.0 - h[:n - d])
                g[:n - d] -= wk * h[d:]
            else:
                e2 = -d
                g[:n - e2] += wk * (1.0 - h[e2:])
                g[e2:] -= wk * h[:n - e2]
        g *= (2.0 / n)

def slsqp_smooth(h, beta, maxiter=200):
    n = len(h)
    try:
        res = minimize(smooth_max_func, h, args=(beta,),
                       jac=smooth_max_grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': smooth_eq_func,
                                     'jac': smooth_eq_jac, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def tc_smooth(h, beta, maxiter=100):
    n = len(h)
    try:
        bnds = Bounds(np.zeros(n), np.ones(n))
        cons = LinearConstraint(np.ones((1, n)), n / 2.0, n / 2.0)
        res = minimize(smooth_max_func, h, args=(beta,),
                       jac=smooth_max_grad, method='trust-constr',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'gtol': 1e-10,
                                'xtol': 1e-14, 'barrier_tol': 1e-14})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src)

            break

        top_k = np.argsort(-corr)[:8]

        improved_this_round = False
        n_indices = len(h)
        for _ in range(min(200, n_indices)):
            i1 = np.random.randint(n_indices)

                    break
