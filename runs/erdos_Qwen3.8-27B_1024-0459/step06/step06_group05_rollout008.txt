import numpy as np
from scipy.optimize import minimize
from scipy.signal import fftconvolve
import time

_T0 = 0.0
_BUDGET = 1300.0

def compute_corr(h):
    """Full correlation of h with (1-h)."""
    return np.correlate(h, 1.0 - h, mode='full')

def compute_corr_fft(h):
    """FFT-accelerated correlation."""
    return fftconvolve(h, (1.0 - h)[::-1], mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr)) * dx

def grad_corr_k(h, k, n):
    """Gradient of correlation at shift k with respect to h."""
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        g[offset:] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:]
    else:
        a = -offset
        npairs = n - a
        g[:npairs] += (1.0 - h[a:])
        g[a:] -= h[:npairs]
    return g

def subgrad_topk(h, n, K=5, beta=100):
    """Subgradient of smooth max of top-K correlations."""
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:K]
    m = np.max(corr)
    # Soft weights based on proximity to max
    sh = np.clip(beta * (corr[idx] - m), -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / ex.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for i in range(K):
        g += w[i] * dx * grad_corr_k(h, idx[i], n)
    val = (m + np.log(ex.sum()) / beta) * dx
    return g, val, np.max(corr) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def lbfgsb_optimize(h, n, K=5, beta=200, maxiter=2000, maxfun=5000):
    """L-BFGS-B optimization with smooth max objective."""
    def obj(x):
        _, v, _ = subgrad_topk(x, n, K, beta)
        return v
    
    def grad(x):
        g, _, _ = subgrad_topk(x, n, K, beta)
        return g
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                   bounds=bounds,
                   options={'maxiter': maxiter, 'maxfun': maxfun, 'ftol': 1e-16, 'gtol': 1e-14})
    h_new = project_feasible(res.x, n)
    return h_new

def compute_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')


def compute_corr_fast(h):
    n = len(h)
    if n > 600:
        return fftconvolve(h, (1.0 - h)[::-1], mode='full')
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr_fast(h)
    return float(np.max(corr)) * dx


def grad_corr_k(h, k_shift, n):
    g = np.zeros(n)
    if k_shift >= 0:
        npairs = n - k_shift
        if npairs <= 0:
            return g
        g[k_shift:] += (1.0 - h[:npairs])
        g[:npairs] -= h[k_shift:]
    else:
        a = -k_shift
        npairs = n - a
        if npairs <= 0:
            return g
        g[:npairs] += (1.0 - h[a:])
        g[a:] -= h[:npairs]
    return g


def topk_subgrad(h, n, K, beta):
    corr = compute_corr_fast(h)
    idx = np.argsort(-corr)
    top_idx = idx[:K]
    c_vals = corr[top_idx]
    m = np.max(c_vals)
    sh = np.clip(beta * (c_vals - m), -60.0, 60.0)
    ex = np.exp(sh)
    w = ex / ex.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for i in range(K):
        if w[i] < 1e-12:
            continue
        g += w[i] * grad_corr_k(h, top_idx[i] - (n - 1), n)
    val = (m + np.log(ex.sum()) / beta) * dx
    actual_max = np.max(corr) * dx
    return g * dx, val, actual_max


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()

    target = n / 2.0

        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def slsqp_opt(h, n, K, beta, maxiter=400):
    def obj(x):
        _, v, _ = topk_subgrad(x, n, K, beta)
        return v

    def grad_fn(x):
        g, _, _ = topk_subgrad(x, n, K, beta)
        return g

    def cfun(x):
        return np.sum(x) - n / 2.0

    def cjac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, jac=grad_fn, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return h


def lbfgs_penalty(h, n, K, beta, penalty=1e6, maxiter=3000, maxfun=10000):
    def obj(x):
        _, v, _ = topk_subgrad(x, n, K, beta)
        s = x.sum() - n / 2.0
        return v + penalty * s * s

    def grad_fn(x):
        g, _, _ = topk_subgrad(x, n, K, beta)
        s = x.sum() - n / 2.0
        return g + 2.0 * penalty * s

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad_fn, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'maxfun': maxfun,
                                'ftol': 1e-18, 'gtol': 1e-15, 'factr': 1e12})

    except Exception:

    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)
