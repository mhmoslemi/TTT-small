        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_value(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = float(corr.max())
    s = corr - cmax
    cl = np.clip(beta * s, -60.0, 60.0)
    e = np.exp(cl)
    return (cmax + np.log(e.sum()) / beta) * (2.0 / n)


def smooth_max_grad(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = float(corr.max())
    s = corr - cmax
    cl = np.clip(beta * s, -60.0, 60.0)
    e = np.exp(cl)
    tot = e.sum()
    w = e / tot
    g = np.zeros(n)
    sig = np.where(w > 1e-10)[0]
    for k in sig:
        wk = w[k]
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:n] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:n]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - h[a:n])
            g[a:n] -= wk * h[:np_]
    g *= (2.0 / n)
    return g


def lbfgsb_opt(h, n, beta, penalty, maxiter, time_limit):
    def fun(x):
        v = smooth_max_value(x, n, beta)
        s = np.sum(x) - n / 2.0
        return v + 0.5 * penalty * s * s

    def gfun(x):
        g = smooth_max_grad(x, n, beta)
        s = np.sum(x) - n / 2.0
        g += penalty * s
        return g

    bounds = [(0.0, 1.0)] * n
    start = time.time()
    try:
        res = minimize(fun, h, jac=gfun, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-14, 'maxfun': maxiter * 4,
                                'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def slsqp_opt(h, n, beta, maxiter, time_limit):
    def fun(x):
        return smooth_max_value(x, n, beta)

    def gfun(x):
        return smooth_max_grad(x, n, beta)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(fun, h, jac=gfun, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14,
                                'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def interp_to(src, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    fn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(fn, xp, src)


def rem_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 8.0)


def equalize_top(h, n, K, time_limit):
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    for it in range(500):
        if time.time() - start > time_limit or rem_time() < 5:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(-corr)[:K]
        c_vals = corr[idx]
        c_spread = c_vals.max() - c_vals.min()
        if c_spread < 1e-10:
            break
        g_avg = np.zeros(n)
        for k in idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_avg[off:n] += (1.0 - h[:np_])
                g_avg[:np_] -= h[off:n]
            else:
                a = -off
                np_ = n - a
                g_avg[:np_] += (1.0 - h[a:n])
                g_avg[a:n] -= h[:np_]
        g_avg /= K
        g_avg *= (2.0 / n)
        gn = np.linalg.norm(g_avg)
        if gn < 1e-14:
            break
        g_avg /= gn
        improved = False
        for step in [0.02, 0.01, 0.005, 0.03, 0.015, 0.008, 0.04, 0.025]:
            ht = project_feasible(h - step * g_avg, n)
            ct = compute_c5(ht)
            if ct < best_c5 - 1e-14:
                h = ht
                best_c5 = ct
                improved = True
                break
            ht2 = project_feasible(h + step * g_avg, n)
            ct2 = compute_c5(ht2)
            if ct2 < best_c5 - 1e-14:
                h = ht2
                best_c5 = ct2
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def coordinate_refine(h, n, time_limit):
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    n_improve = 0
    while time.time() - start < time_limit and rem_time() > 5:
        improved = False
        for i in range(n):
            if time.time() - start > time_limit or rem_time() < 5:
                break
            old = h[i]
            cands = [0.0, 1.0]
            if old > 0.15 and old < 0.85:
                cands.append(0.5)
            if old > 0.3 and old < 0.7:
                cands.append(0.25)
                cands.append(0.75)
            best_v = old
            best_c = best_c5
            for cv in cands:
                if abs(cv - old) < 1e-12:
                    continue
                h[i] = cv
                h = project_feasible(h, n)
                ct = compute_c5(h)
                if ct < best_c - 1e-14:
                    best_c = ct
                    best_v = cv
            h[i] = best_v
            h = project_feasible(h, n)
            cn = compute_c5(h)
            if cn < best_c5 - 1e-14:
                best_c5 = cn
                improved = True
                n_improve += 1
        if not improved:
            break
    return h, best_c5


def multi_start_optimize(h, n, beta, penalty, maxiter, time_limit, rng, n_starts=3):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    per_start = time_limit / n_starts
    for s in range(n_starts):
        if rem_time() < per_start + 10:
            break
        if s == 0:
            h_cur = h.copy()
        else:
            noise = rng.randn(n) * 0.1
            h_cur = project_feasible(h + noise, n)
        h_new = lbfgsb_opt(h_cur, n, beta, penalty, maxiter, per_start - 5)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    return best_h, best_c5

                best_c5, best_h, best_n = c5_eq, h2.copy(), n2

            h2 = slsqp_opt(h2, n2, 20000, 300, 45)

        h3 = interp_to(best_h, best_n, n3)
        h3 = project_feasible(h3, n3)
        c5 = compute_c5(h3)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h3.copy(), n3

        if rem_time() > 60:
            h3 = lbfgsb_opt(h3, n3, 5000, 5000, 2000, 50)

            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h3.copy(), n3

        c5 = compute_c5(h2)
        if c5 < best_c5:
            best_c5, best_h = c5, h2.copy()

    # Try 1-h symmetry
    h_sym = projekt_sym(1.0 - best_h, best_n)
    c5_sym = compute_c5(h_sym)
    if c5_sym < best_c5:
        best_c5, best_h = c5_sym, h_sym


def projekt_sym(h, n):
