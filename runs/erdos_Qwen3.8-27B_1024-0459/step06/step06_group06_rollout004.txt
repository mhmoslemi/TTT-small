    total = na + nb - 1
    N = 1
    while N < total:
        N *= 2
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    r = np.fft.irfft(A * np.conj(B), N)
    result = np.concatenate([r[:na][::-1], r[N-na+1:N][::-1]])
    return result


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = fft_correlate(h, 1.0 - h)
    return float(np.max(corr)) * dx


def compute_corr(h):
    return fft_correlate(h, 1.0 - h)
