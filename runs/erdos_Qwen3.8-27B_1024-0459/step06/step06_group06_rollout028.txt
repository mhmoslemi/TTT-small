        # Remove excess ones
        idx = np.where(h_bin == 1)[0]
        rng.shuffle(idx)
        h_bin[idx[:n_ones - target]] = 0
    elif n_ones < target:
        # Add missing ones
        idx = np.where(h_bin == 0)[4][0]
        rng.shuffle(idx)
        h_bin[idx[:target - n_ones]] = 1
    
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    start = time.time()
    
    # Simulated annealing with swaps
    T = 0.01  # Initial "temperature"
    T_min = 1e-6
    
    while time.time() - start < time_limit:
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        
        # Try some random swaps
        n_ones = int(np.sum(h_bin))
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        improved = False
        for _ in range(100):
            i = rng.choice(zeros)
            j = rng.choice(ones)
            h_bin[i] = 1
            h_bin[j] = 0
            c_new = compute_c5(h_bin)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_bin.copy()
                improved = True
            elif c_new - best_c5 < T:
                # Accept with probability (simulated annealing)
                if rng.random() < np.exp(-(c_new - best_c5) / T):
                    pass  # keep the change
                else:
                    h_bin[i] = 0
                    h_bin[j] = 1
            else:
                h_bin[i] = 0
                h_bin[j] = 1
        
        # Also try systematic swaps (greedy)
        # For each zero position, find the best one to swap with
        for _ in range(50):
            i = rng.choice(zeros)
            best_j = -1
            best_c = best_c5
            for j_idx in range(min(20, len(ones))):
                j = ones[rng.randint(len(ones))]
                h_bin[i] = 1
                h_bin[j] = 0
                c = compute_c5(h_bin)
                h_bin[i] = 0
                h_bin[j] = 1
                if c < best_c:
                    best_c = c
                    best_j = j
            if best_j >= 0:
                h_bin[i] = 1
                h_bin[best_j] = 0
                best_c5 = best_c
                best_h = h_bin.copy()
                improved = True
                ones = np.where(h_bin == 1)[0]
                zeros = np.where(h_bin == 0)[0]
        
        T *= 0.95
        if T < T_min:
            T = T_min
        
        if not improved and T < 0.001:
            # Try a few more rounds of pure greedy
            any_improve = False
            for _ in range(200):
                i = rng.choice(zeros)
                best_j = -1
                best_c = best_c5
                for _ in range(min(30, len(ones))):
                    j = ones[rng.randint(len(ones))]
                    h_bin[i] = 1[0]
                    h_bin[j] = 0
                    c = compute_c5(h_bin)
                    h_bin[i] = 0
                    h_bin[j] = 1
                    if c < best_c:
                        best_c = c
                        best_j = j
                if best_j >= 0:
                    h_bin[i] = 1
                    h_bin[best_j] = 0
                    best_c5 = best_c
                    best_h = h_bin.copy()
                    any_improve = True
                    ones = np.where(h_bin == 1)[0]
                    zeros = np.where(h_bin == 0)[0]
            if not any_improve:
                break
    
    return best_h, best_c5
