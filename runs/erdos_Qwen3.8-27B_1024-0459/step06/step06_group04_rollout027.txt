import numpy as np
from scipy.optimize import minimize, LinearConstraint, Bounds

_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = h.astype(float).copy()
    target = n / 2.0
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        interior = (h > 1e-14) & (h < 1.0 - 1e-14)
        if interior.any():
            h[interior] += diff / interior.sum()
        else:
            h += diff / n
    h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_topk(h, K, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    w = np.exp(beta * (c_top - c_top.max()))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(K) / K
    else:
        w /= w_sum
    g = np.zeros(n)
    for i in range(K):
        wk = w[i] * dx
        if wk < 1e-15:
            continue
        k_idx = idx[i]
        off = k_idx - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - h[ao:])
            g[ao:] -= wk * h[:np_]
    return g, float(corr.max()) * dx

def interp_to(src, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n

def smooth_max_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    s = np.clip(beta * (corr - m), -500.0, 500.0)
    e = np.exp(s)
    return (m + np.log(e.sum()) / beta) * dx

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    s = np.clip(beta * (corr - m), -500.0, 500.0)

        else:

            g[:np_] += wk * (1.0 - h[ao:])

    viol = max(0.0, x.sum() - n / 2.0, n / 2.0 - x.sum())
    if viol > 1e-10:
        return smooth_max_obj(x, beta) + 1000.0 * viol * viol
    return smooth_max_obj(x, beta)

def _constrained_grad(x, n, beta):
    g = smooth_max_grad(x, beta)
    viol = x.sum() - n / 2.0
    if abs(viol) > 1e-10:
        g += 2000.0 * viol * np.ones(n)
    return g

def slsqp_refine(h, beta, maxiter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            _constrained_obj, h, args=(n, beta),
            jac=_constrained_grad,
            method='SLSQP',
            bounds=bounds,
            options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0}
        )
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def trust_constr_refine(h, beta, maxiter):
    n = len(h)
    try:
        bnds = Bounds(np.zeros(n), np.ones(n))
        A_eq = np.ones((1, n))
        cons = LinearConstraint(A_eq, n / 2.0 - 1e-10, n / 2.0 + 1e-10)
        res = minimize(
            smooth_max_obj, h, args=(beta,),
            jac=smooth_max_grad,
            method='trust-constr',
            bounds=bnds, constraints=cons,
            options={'maxiter': maxiter, 'gtol': 1e-10, 'xtol': 1e-12, 'barrier_tol': 1e-12}
        )
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h
