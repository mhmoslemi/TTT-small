    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    return g


def _eq_fun(x):
    return [np.sum(x) - _CUR_N / 2.0]


def _eq_jac(x):
    return np.ones((1, _CUR_N))


def slsqp_opt(h, n, beta, maxiter=2000):
    global _CUR_N, _SM_BETA
    _CUR_N = n
    _SM_BETA = beta
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_sm_obj, h0, jac=_sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        if res.fun < _sm_obj(h0) - 1e-16:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def grad_single_shift(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        np_ = n - off
        g[off:] += (1.0 - h[:np_])
        g[:np_] -= h[off:]
    else:
        a = -off
        np_ = n - a
        g[:np_] += (1.0 - h[a:])
        g[a:] -= h[:np_]
    return g


def active_set_minimize(h, n, top_k=5, max_iters=500, time_limit=60):
    """Directly minimize the max of top-k active correlations using SLSQP with inequality constraints."""
    global _CUR_N, _SM_BETA
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if remaining_time() < 5:
            break

        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        c_max = corr[idx_top[0]]

        # Find common descent direction for all active shifts
        g_dir = np.zeros(n)
        for k in idx_top:
            g_dir += grad_single_shift(h, k, n)
        g_dir /= top_k
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break

        # Try steps along negative gradient of average active corr
        improved = False
        for step in [0.001, 0.002, 0.005, 0.01, 0.02, 0.0005, 0.0001]:
            ht = project_feasible(h - step * g_dir, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-15:
                h = ht
                best_c5 = cn
                best_h = h.copy()
                improved = True
                break

        if not improved:
            # Try individual shift gradients
            found = False
            for k in idx_top:
                gk = grad_single_shift(h, k, n)
                gkn = np.linalg.norm(gk)
                if gkn < 1e-14:
                    continue
                for step in [0.005, 0.01, 0.02, 0.05]:
                    ht = project_feasible(h - step * gk / gkn, n)
                    cn = compute_c5(ht)
                    if cn < best_c5 - 1e-15:
                        h = ht
                        best_c5 = cn
                        best_h = h.copy()
                        found = True
                        break
                if found:
                    break
            if not found:
                break

    return best_h, best_c5


def nudge_refine(h, n, max_iters=5000, time_limit=60):
    """Very fine-grained random nudging to escape tiny local optima."""
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    rng = np.random.RandomState(12345)
    start = time.time()
    no_improve = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if remaining_time() < 5:
            break

        # Small random perturbation
        scale = max(0.0005, 0.01 * (1.0 - it / max_iters))
        pert = rng.randn(n) * scale
        ht = project_feasible(h + pert, n)
        cn = compute_c5(ht)

        if cn < best_c5 - 1e-15:
            h = ht
            best_c5 = cn
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break

    return best_h, best_c5


def binary_swap_sa(h, n, time_limit=100, rng=None):
    """Simulated annealing on binary representations with 1-swap moves."""
    if rng is None:
        rng = np.random.RandomState(42)

    h_bin = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        sort_idx = np.argsort(-h_bin)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0

    dx = 2.0 / n
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    curr_c5 = best_c5
    start = time.time()

    T_start = 3.0 / dx
    T_end = 0.0005 / dx
    total_iters = 30000
    min_T_ratio = T_end / T_start

    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    no_improve = 0

    for it in range(total_iters):
        if time.time() - start > time_limit:
            break
        if remaining_time() < 5:
            break

        T = T_start * (min_T_ratio ** (it / max(1, total_iters)))

        i_idx = rng.randint(len(ones))
        j_idx = rng.randint(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]

        h_bin[i] = 0.0
        h_bin[j] = 1.0
        new_c5 = compute_c5(h_bin)
        delta = new_c5 - curr_c5

        if delta < 0 or (T > 1e-15 and rng.random() < np.exp(-delta / max(T, 1e-15))):
            ones[i_idx] = j
            zeros[j_idx] = i
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_bin.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0

        if no_improve > 8000:
            break

    return best_h, best_c5


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def optimize_at_resolution(h, n, time_budget=120):
    """Full optimization pipeline at a given resolution."""
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()

    # Smooth-max SLSQP with increasing beta
    for beta in [500, 2000, 10000, 50000, 200000, 500000, 2000000, 5000000, 20000000]:
        if time.time() - start > time_budget * 0.6:
            break
        if remaining_time() < 15:
            break
        h_new = slsqp_opt(best_h, n, beta, maxiter=3000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Active set refinement
    if time.time() - start < time_budget * 0.75:
        tl = min(30, max(5, int(time_budget * 0.15)))
        h_eq, c5_eq = active_set_minimize(best_h, n, top_k=5, max_iters=800, time_limit=tl)
        if c5_eq < best_c5 - 1e-15:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # More SLSQP after active set
    if time.time() - start < time_budget * 0.85:
        for beta in [10000000, 50000000, 100000000]:
            if remaining_time() < 10:
                break
            if time.time() - start > time_budget:
                break
            h_new = slsqp_opt(best_h, n, beta, maxiter=2000)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Nudge refinement
    if time.time() - start < time_budget:
        tl = min(20, max(5, int(time_budget * 0.1)))
        h_nd, c5_nd = nudge_refine(best_h, n, max_iters=3000, time_limit=tl)
        if c5_nd < best_c5 - 1e-15:
            best_c5 = c5_nd
            best_h = h_nd.copy()

    # Final SLSQP polish
    if remaining_time() > 10:
        h_final = slsqp_opt(best_h, n, 50000000, maxiter=2000)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-15:
            best_c5 = c5_final
            best_h = h_final.copy()

    return project_feasible(best_h, n), best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _CUR_N, _SM_BETA
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        parent_h = np.zeros(640)
        parent_h[:320] = 1.0
    parent_n = len(parent_h)
    parent_h = project_feasible(parent_h, parent_n)
    parent_c5 = compute_c5(parent_h)

    best_c5 = parent_c5
    best_h = parent_h.copy()
    best_n = parent_n

    # Phase 1: Optimize at parent resolution
    h_opt, c5_opt = optimize_at_resolution(parent_h, parent_n, time_budget=180)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt.copy()
        best_n = parent_n

    # Phase 2: Try higher resolutions
    higher_resolutions = [parent_n * 2, parent_n * 2 + 128, parent_n * 3]
    for n_t in higher_resolutions:
        if remaining_time() < 60:
            break
        if n_t % 2 != 0:
            n_t += 1
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        c5_int = compute_c5(h_feas)

        tb = min(120, int(remaining_time() * 0.25))
        h_opt2, c5_opt2 = optimize_at_resolution(h_feas, n_t, time_budget=tb)
        if c5_opt2 < best_c5:
            best_c5 = c5_opt2
            best_h = h_opt2.copy()
            best_n = n_t

    # Phase 3: Try medium resolutions
    med_resolutions = [960, 800, 1120]
    for n_t in med_resolutions:
        if remaining_time() < 50:
            break
        if n_t % 2 != 0:
            n_t += 1
        if n_t == best_n:
            continue
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        tb = min(80, int(remaining_time() * 0.2))
        h_opt3, c5_opt3 = optimize_at_resolution(h_feas, n_t, time_budget=tb)
        if c5_opt3 < best_c5:
            best_c5 = c5_opt3
            best_h = h_opt3.copy()
            best_n = n_t

    # Phase 4: Binary SA at best resolution
    if remaining_time() > 60:
        h_bin = (best_h > 0.5).astype(float)
        sa_time = min(120, int(remaining_time() * 0.3))
        h_b, c5_b = binary_swap_sa(h_bin, best_n, time_limit=sa_time, rng=rng)

        # Refine binary result continuously
        h_b_c = h_b.copy()
        for beta in [100000, 1000000, 10000000]:
            if remaining_time() < 10:
                break
            h_b_c = slsqp_opt(h_b_c, best_n, beta, maxiter=1500)
        c5_b_c = compute_c5(h_b_c)
        h_b_c, c5_b_c = active_set_minimize(h_b_c, best_n, top_k=6, max_iters=400, time_limit=20)
        h_b_c, c5_b_c = nudge_refine(h_b_c, best_n, max_iters=2000, time_limit=15)

        if c5_b_c < best_c5:
            best_c5 = c5_b_c
            best_h = h_b_c.copy()
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy().astype(float)
            best_n = best_n

        # Also try SA with different seed
        if remaining_time() > 40:
            rng2 = np.random.RandomState(seed + 1)
            h_bin2 = (best_h > 0.5).astype(float)
            h_b2, c5_b2 = binary_swap_sa(h_bin2, best_n, time_limit=60, rng=rng2)
            h_b2_c = h_b2.copy()
            for beta in [500000, 5000000]:
                if remaining_time() < 10:
                    break
                h_b2_c = slsqp_opt(h_b2_c, best_n, beta, maxiter=1000)
            c5_b2_c = compute_c5(h_b2_c)
            h_b2_c, c5_b2_c = active_set_minimize(h_b2_c, best_n, top_k=5, max_iters=300, time_limit=15)
            if c5_b2_c < best_c5:
                best_c5 = c5_b2_c
                best_h = h_b2_c.copy()

    # Phase 5: Restart from different initializations
    if remaining_time() > 50:
        # Thue-Morse initialization
        h_tm = np.zeros(best_n)
        for i in range(best_n):
            v = i
            cnt = 0
            while v:
                cnt += v & 1
                v >>= 1
            h_tm[i] = 1.0 if cnt % 2 == 1 else 0.0
        h_tm = project_feasible(h_tm, best_n)

        tb = min(60, int(remaining_time() * 0.3))
        h_tm_opt, c5_tm = optimize_at_resolution(h_tm, best_n, time_budget=tb)
        if c5_tm < best_c5:
            best_c5 = c5_tm
            best_h = h_tm_opt.copy()

    # Phase 6: Final intensive refinement
    n = best_n
    h = best_h.copy()

    # Very high beta SLSQP
    for beta in [50000000, 100000000, 200000000]:
        if remaining_time() < 10:
            break
        h_new = slsqp_opt(h, n, beta, maxiter=3000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-16:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Active set final
    if remaining_time() > 15:
        h_eq, c5_eq = active_set_minimize(h, n, top_k=8, max_iters=600, time_limit=25)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Nudge final
    if remaining_time() > 15:
        h_nd, c5_nd = nudge_refine(h, n, max_iters=5000, time_limit=min(30, int(remaining_time() * 0.4)))
        if c5_nd < best_c5:
            best_c5 = c5_nd
            best_h = h_nd.copy()

    # Final SLSQP
    if remaining_time() > 10:
        h_final = slsqp_opt(best_h, n, 100000000, maxiter=2000)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, n)
