def project_feasible(h, n):

def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        g[offset:n] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:n]
    else:
        a = -offset
        npairs = n - a
        g[:npairs] += (1.0 - h[a:n])
        g[a:n] -= h[:npairs]
    return g

def smooth_max_grad(h, n, beta, K=None):

    corr = compute_corr(h)
    m = np.max(corr)

        cl = np.clip(beta * sh, -50.0, 50.0)

    return g, float(np.max(corr))

        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 6: Fine subgradient
    if remaining_time() > 30:
        h_fine = h.copy()
        c5_fine = best_c5
        for it in range(300):
            if remaining_time() < 3:
                break
            g, c_max = top_k_subgradient(h_fine, n, K=12, beta=150.0)
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break
            g = g / gn
            for step in [0.003, 0.001, 0.002, 0.005, 0.0005]:
                ht = project_feasible(h_fine - step * g, n)
                ct = compute_c5(ht)
                if ct < c5_fine - 1e-15:
                    h_fine = ht
                    c5_fine = ct
                    break
            if c5_fine < best_c5:
                best_c5 = c5_fine
                best_h = h_fine.copy()

    # Phase 7: Try n=2048
    if remaining_time() > 100:
        n_vh = 2048
        h_vh = interp_to(best_h, best_n, n_vh)
        h_vh = project_feasible(h_vh, n_vh)
        c5_vh = compute_c5(h_vh)
        if c5_vh <= best_c5 + 0.001:  # Only proceed if not much worse
            h_vh, c5_vh = subgradient_descent(h_vh, n_vh, min(20.0, remaining_time() * 0.2), rng)
            h_vh = lbfgsb_smooth(h_vh, n
