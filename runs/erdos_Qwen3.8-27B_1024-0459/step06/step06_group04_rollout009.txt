import numpy as np
from scipy.optimize import minimize, LinearConstraint, Bounds
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if mask.any():
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_obj(x, beta, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    shifted = beta * (corr - corr.max())
    shifted = np.clip(shifted, -500.0, 500.0)
    val = corr.max() + np.log(np.exp(shifted).sum()) / beta
    return val * dx


def smooth_max_grad(x, beta, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    shifted = beta * (corr - corr.max())
    shifted = np.clip(shifted, -500.0, 500.0)
    e = np.exp(shifted)
    w = e / e.sum()
    g = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-14:
            continue
        off = k_idx - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - x[ao:])
            g[ao:] -= wk * x[:np_]
    return g * dx


def sum_eq_func(x, n):
    return x.sum() - n / 2.0


def sum_eq_jac_func(x, n):
    return np.ones(n)


def slsqp_optimize(h, beta, maxiter, n):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_func, 'jac': sum_eq_jac_func, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h, args=(beta, n),
                       jacobian=lambda x, b, nn: smooth_max_grad(x, b, nn),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def projected_subgrad(h, n, time_limit, K=8):

    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    step_scale = 1.0

        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 8:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(corr.max()) * (2.0 / n)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
        idx = np.argsort(-corr)[:K]
        g = np.zeros(n)
        for k_idx in idx:
            off = k_idx - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:] += (1.0 - h[:np_])
                g[:np_] -= h[off:]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += (1.0 - h[ao:])
                g[ao:] -= h[:np_]
        g /= K
        g *= (2.0 / n)
        g -= np.mean(g)
        gn = np.linalg.norm(g)

def thue_morse_binary(n):
    t = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        t[i] = 1.0 if cnt % 2 == 1 else 0.0
    ones = int(t.sum())
    target = n // 2
    if ones != target:
        idx = np.where(t == 1.0)[0]
        idx0 = np.where(t == 0.0)[0]
        if ones > target:
            for i in range(ones - target):
                t[idx[i]] = 0.0
        else:
            for i in range(target - ones):
                t[idx0[i]] = 1.0
    return t


def beatty_binary(n, alpha):
    h = np.zeros(n)
    for i in range(n):
        val = int((i + 1) * alpha) - int(i * alpha)
        h[i] = 1.0 if val > 0 else 0.0
    ones = int(h.sum())
    target = n // 2
    if ones > target:
        idx = np.where(h == 1.0)[0]
        np.random.shuffle(idx)
        for i in range(ones - target):
            h[idx[i]] = 0.0
    elif ones < target:
        idx = np.where(h == 0.0)[0]
        np.random.shuffle(idx)
        for i in range(target - ones):
            h[idx[i]] = 1.0
    return h

    start = time.time()

    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 8:
            break
        improved = False
        for _ in range(300):
            b1 = rng.randint(n_blocks)
            b2 = rng.randint(n_blocks)
            if b1 == b2:
                continue
            h_try = h.copy()
            s1, e1 = b1 * block_size, (b1 + 1) * block_size
            s2, e2 = b2 * block_size, (b2 + 1) * block_size
            h_try[s1:e1], h_try[s2:e2] = h_try[s2:e2].copy(), h_try[s1:e1].copy()
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try)
            if c_try < best_c5 - 1e-13:
                h = h_try
                best_c5 = c_try
                best_h = h_try.copy()
                improved = True
                break
        if not improved and rnd > 50:
            break
    return best_h, best_c5

    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src)

    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    lr = 0.02
    for it in range(20000):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 8:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(-corr)[:K]
        c_top = corr[idx]
        w = np.exp(beta_w * (c_top - c_top.max()))
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(K) / K
        else:
            w /= wsum
        g = np.zeros(n)
        for i, k_idx in enumerate(idx):
            wk = w[i]
            if wk < 1e-12:
                continue
            off = k_idx - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:] += wk * (1.0 - h[:np_])
                g[:np_] -= wk * h[off:]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - h[ao:])
                g[ao:] -= wk * h[:np_]
        g *= (2.0 / n)
        g -= np.mean(g)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        h_try = project_feasible(h - lr * g / gn, n)
        c_try = compute_c5(h_try)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try.copy()
            h = h_try
        else:
            lr *= 0.7
    return best_h, best_c5
