import time

_TB = 1300.0

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def proj(h, n):
    h = h.astype(float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    tgt = n * 0.5
    for _ in range(2000):
        d = tgt - h.sum()
        if abs(d) < 1e-16:
            break
        mk = (h > 1e-16) & (h < 1.0 - 1e-16)
        c = int(mk.sum())
        if c > 0:
            h[mk] += d / c
        else:
            h[:] += d / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

def gk(h, k, n):
    g = np.zeros(n)
    o = k - (n - 1)
    if o >= 0:
        m = n - o
        g[o:] += 1.0 - h[:m]
        g[:m] -= h[o:]
    else:
        a = -o
        m = n - a
        g[:m] += 1.0 - h[a:]
        g[a:] -= h[:m]
    return g

def interp(src, sn, dn):
    xp = (np.arange(sn) + 0.5) * 2.0 / sn
    return np.interp((np.arange(dn) + 0.5) * 2.0 / dn, xp, src)

def adam(h, n, tlim, K, b0, b1, lr0, rdec):
    h = h.astype(float).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    t0 = time.time()
    bc = compute_c5(h)
    bh = h.copy()
    while time.time() - t0 < tlim:
        if rem_secs() < 10:
            break
        t += 1
        f = min(1.0, t / max(1, int(tlim * 20)))
        beta = b0 * (b1 / b0) ** f
        lr = max(lr0 * rdec ** t, lr0 * 0.01)
        cr = np.correlate(h, 1.0 - h, mode='full')
        ki = np.argsort(-cr)[:K]
        cv = cr[ki]
        cl = np.clip(beta * (cv - cv.max()), -40.0, 0.0)
        ex = np.exp(cl)
        ws = ex / ex.sum()
        g = np.zeros(n)
        for i in range(K):
            if ws[i] > 1e-14:
                g += ws[i] * gk(h, ki[i], n)
        g *= (2.0 / n)
        m = 0.9 * m + 0.1 * g
        v = 0.999 * v + 0.001 * g * g
        b1c = 1.0 - 0.9 ** t
        b2c = 1.0 - 0.999 ** t
        st = -lr * (m / b1c) / (np.sqrt(v / b2c) + 1e-8)
        hn = proj(h + st, n)
        cn = compute_c5(hn)
        if cn < bc:
            bc = cn
            bh = hn.copy()
            h = hn.copy()
        else:
            ok = False
            for fac in (0.5, 0.2, 0.05):
                ht = proj(h + fac * st, n)
                ct = compute_c5(ht)
                if ct < bc:
                    bc = ct
                    bh = ht.copy()
                    h = ht.copy()
                    ok = True
                    break

                rr = np.random.RandomState(42 + t)
                h = proj(h + 0.002 * (rr.rand(n) - 0.5), n)
    return bh, bc

    return v[-1]

def s_obj_j(v):
    g = np.zeros(_NG + 1)
    g[-1] = 1.0
    return g

    hh = v[:_NG]
    cr = np.correlate(hh, 1.0 - hh, mode='full')
    return v[-1] * (_NG / 2.0) - cr[_KL]

def s_ineq_j(v):
    hh = v[:_NG]
    n = _NG
    J = np.zeros((len(_KL), n + 1))

            J[i, o:] += 1.0 - hh[:m]

            J[i, :m] += 1.0 - hh[a:]
            J[i, a:] -= hh[:m]
        J[i, -1] = n / 2.0
    return J

    return v[:_NG].sum() - _NG / 2.0

def s_eq_j(v):
    g = np.zeros(_NG + 1)

    return g

def slsq(h, n, K, iters):
    global _NG, _KL
    _NG = n
    cr = np.correlate(h, 1.0 - h, mode='full')
    _KL = np.argsort(-cr)[:K]
    t0 = compute_c5(h)
    v0 = np.concatenate([h, [t0 * n / 2.0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, None)]

            hn = proj(r.x[:n], n)
            cn = compute_c5(hn)
            return hn, cn
    except Exception:
        pass
    return h.copy(), compute_c5(h)
