    """Use QP to find a small adjustment that reduces top peak correlations."""
    corr = compute_corr(h)
    idx_top = np.argsort(-corr)[:top_k]
    c_max = np.max(corr)
    
    # Build gradient matrix
    A = np.zeros((top_k, n))
    for i, k in enumerate(idx_top):
        A[i] = grad_single_shift(h, k, n)
    
    # QP: min 0.5*d'*d s.t. A*d <= -epsilon, sum(d)=0, -h <= d <= 1-h, |d|_inf <= step_limit
    d_var = cp.Variable(n)
    eps = 0.01
    constraints = [
        A @ d_var <= -eps * np.ones(top_k),
        cp.sum(d_var) == 0,
        d_var >= -h,
        d_var <= 1.0 - h,
        d_var >= -step_limit,
        d_var <= step_limit

    obj = cp.Minimize(cp.sum_squares(d_var))

    try:

        if d_var.value is not None:
            h_new = project_feasible(h + d_var.value, n)
            return h_new

_SM_BETA = 1000.0


def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def _sm_obj(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    return (m + np.log(np.sum(ez)) / beta) * dx


def _sm_grad(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    return g


def _eq_fun(x):
    return np.sum(x) - _CUR_N / 2.0


def _eq_jac(x):
    return np.ones(_CUR_N)


def slsqp_opt(h, n, beta, maxiter=1000):
    global _CUR_N, _SM_BETA
    _CUR_N = n
    _SM_BETA = beta
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_sm_obj, h0, jac=_sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'eps': 1e-8})
        if res.fun < _sm_obj(h0) - 1e-15:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def grad_single_shift(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        np_ = n - off
        g[off:] += (1.0 - h[:np_])
        g[:np_] -= h[off:]
    else:
        a = -off
        np_ = n - a
        g[:np_] += (1.0 - h[a:])
        g[a:] -= h[:np_]
    return g

def binary_sa(h, n, time_limit=150, rng=None, tabu_size=50):
    if rng is None:
        rng = np.random.RandomState(42)
    h_bin = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        sort_idx = np.argsort(-h_bin)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0

    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    curr_c5 = best_c5
    start = time.time()

    T_start = 8.0
    T_end = 0.001
    total_iters = 50000
    tabu = {i: 0 for i in range(n)}
    no_improve = 0

    for it in range(total_iters):
        if time.time() - start > time_limit:
            break
        if remaining_time() < 5:
            break

        T = T_start * (T_end / T_start) ** (it / max(1, total_iters))

        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

        if tabu[i] > it or tabu[j] > it:

            curr_c5 = new_c5

                best_c5 = new_c5
                best_h = h_bin.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            no_improve += 1

def qp_peak_shave(h, n, top_k=5, n_iters=20, time_limit=60):
    """Iteratively shave top peaks using QP."""
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    start = time.time()

    for it in range(n_iters):
        if time.time() - start > time_limit:
            break
        if remaining_time() < 10:
            break

        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        c_max = corr[idx_top[0]]

        A = np.zeros((top_k, n))
        for i, k in enumerate(idx_top):
            A[i] = grad_single_shift(h, k, n)

        d_var = cp.Variable(n)
        eps = 0.005 * c_max if c_max > 0 else 0.001
        constraints = [
            A @ d_var <= -eps * np.ones(top_k),
            cp.sum(d_var) == 0,
            d_var >= -h - 1e-12,
            d_var <= 1.0 - h + 1e-12,
            d_var >= -0.05,
            d_var <= 0.05
        ]
        obj = cp.Minimize(cp.sum_squares(d_var))
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            if d_var.value is not None:
                d_val = d_var.value
                h_new = project_feasible(h + d_val, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-15:
                    h = h_new
                    best_c5 = c5_new
                else:
                    break
            else:
                break
        except Exception:
            break

    return h, best_c5


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def theta_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)


def kronecker_seq(n, alpha=None):
    if alpha is None:
        alpha = (1.0 + np.sqrt(5.0)) / 2.0 - 1.0
    phis = np.arange(n) * alpha
    phis = phis - np.floor(phis)
    h = (phis < 0.5).astype(float)
    return project_feasible(h, n)

                h_ms_c = slsqp_opt(h_ms_c, best_n, beta, maxiter=800)
            c5_ms_c = compute_c5(h_ms_c)
            h_ms_c, c5_ms_c = top_k_equalize(h_ms_c, best_n, top_k=5, max_iters=300, time_limit=25)
            if c5_ms_c < best_c5 - 1e-15:
                best_c5 = c5_ms_c
                best_h = h_ms_c.copy()
                h = h_ms_c.copy()

    # Phase 5: Fine coordinate descent (KEY improvement)
    if remaining_time() > 60:
        h_cd, c5_cd = fine_coord_descent(best_h, best_n, max_sweeps=25, n_grid=25)
        if c5_cd < best_c5 - 1e-15:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h = h_cd.copy()

    # Phase 6: Mid-sweep coordinate
    if remaining_time() > 40:
        h_ms2, c5_ms2 = midsweep_coord(best_h, best_n, max_sweeps=30)
        if c5_ms2 < best_c5 - 1e-15:
            best_c5 = c5_ms2
            best_h = h_ms2.copy()
            h = h_ms2.copy()

    # Phase 7: QP peak shaving
    if remaining_time() > 50:
        h_qp, c5_qp = qp_peak_shave(best_h, best_n, top_k=5, n_iters=15, time_limit=60)
        if c5_qp < best_c5 - 1e-15:
            best_c5 = c5_qp
            best_h = h_qp.copy()
            h = h_qp.copy()
        if remaining_time() > 30:
            h_qp, c5_qp = qp_peak_shave(best_h, best_n, top_k=8, n_iters=10, time_limit=40)
            if c5_qp < best_c5 - 1e-15:
                best_c5 = c5_qp
                best_h = h_qp.copy()
                h = h_qp.copy()

    # Phase 8: More SLSQP
    for beta in [500000, 2000000, 10000000, 50000000]:
        if remaining_time() < 20:
            break
        h_new = slsqp_opt(h, best_n, beta, maxiter=2000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 9: Subgradient refined
    if remaining_time() > 40:
        h_sg, c5_sg = subgradient_refined(h, best_n, max_iters=6000, time_limit=50)
        if c5_sg < best_c5 - 1e-15:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 10: Multi-resolution
    for n_t in [1024, 768, 512, 960]:
        if remaining_time() < 50:
            break
        if n_t == best_n:
            continue
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        c5_int = compute_c5(h_feas)

        h_opt = h_feas
        for beta in [5000, 50000, 500000]:
            if remaining_time() < 20:
                break
            h_opt = slsqp_opt(h_opt, n_t, beta, maxiter=800)
        c5_opt = compute_c5(h_opt)
        if remaining_time() > 30:
            h_opt, c5_opt = top_k_equalize(h_opt, n_t, top_k=5, max_iters=400, time_limit=30)
        if remaining_time() > 25:
            h_opt, c5_opt = fine_coord_descent(h_opt, n_t, max_sweeps=10, n_grid=15)
        if remaining_time() > 20:
            h_opt, c5_opt = subgradient_refined(h_opt, n_t, max_iters=3000, time_limit=20)

        if c5_opt < best_c5 - 1e-15:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t
            h = h_opt.copy()

    # Phase 11: Alternative starts (Kronecker, Thue-Morse)
    if remaining_time() > 100:
        alphas = [(1.0 + np.sqrt(5.0)) / 2.0 - 1.0, np.sqrt(2.0) - 1.0, np.sqrt(3.0) - 1.0]
        for alpha in alphas[:2]:
            if remaining_time() < 50:
                break
            h_kr = kronecker_seq(best_n, alpha)
            c5_kr = compute_c5(h_kr)
            h_kr = slsqp_opt(h_kr, best_n, 10000, maxiter=1000)
            c5_kr = compute_c5(h_kr)
            h_kr, c5_kr = top_k_equalize(h_kr, best_n, top_k=5, max_iters=300, time_limit=25)
            h_kr, c5_kr = fine_coord_descent(h_kr, best_n, max_sweeps=8, n_grid=15)
            if c5_kr < best_c5 - 1e-15:
                best_c5 = c5_kr
                best_h = h_kr.copy()
                h = h_kr.copy()

        if remaining_time() > 50:
            h_tm = theta_morse(best_n)
            c5_tm = compute_c5(h_tm)
            h_tm = slsqp_opt(h_tm, best_n, 10000, maxiter=800)
            c5_tm = compute_c5(h_tm)
            h_tm, c5_tm = fine_coord_descent(h_tm, best_n, max_sweeps=10, n_grid=15)
            if c5_tm < best_c5 - 1e-15:
                best_c5 = c5_tm
                best_h = h_tm.copy()
                h = h_tm.copy()

    # Phase 12: Final intensive polish
    n = best_n
    h = best_h.copy()
    for beta in [5000000, 20000000, 100000000]:
        if remaining_time() < 10:
            break
        h_new = slsqp_opt(h, n, beta, maxiter=2000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    if remaining_time() > 15:
        h_cd, c5_cd = fine_coord_descent(best_h, n, max_sweeps=15, n_grid=20)
        if c5_cd < best_c5 - 1e-15:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    if remaining_time() > 15:
        h_eq, c5_eq = top_k_equalize(best_h, n, top_k=8, max_iters=500, time_limit=20)
        if c5_eq < best_c5 - 1e-15:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    if remaining_time() > 10:
        h_qp, c5_qp = qp_peak_shave(best_h, n, top_k=6, n_iters=10, time_limit=15)
        if c5_qp < best_c5 - 1e-15:
            best_c5 = c5_qp
            best_h = h_qp.copy()

    if remaining_time() > 10:
        h_sg, c5_sg = subgradient_refined(best_h, n, max_iters=4000, time_limit=15)
        if c5_sg < best_c5 - 1e-15:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Final SLSQP polish
    if remaining_time() > 8:
        h_final = slsqp_opt(best_h, n, 100000000, maxiter=1000)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-15:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, n)
