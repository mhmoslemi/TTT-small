    except Exception:
        pass
    return h.copy(), c_max


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def equalize_and_descent(h, n, K, max_iters):
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if time.time() - _T0 > _BUDGET - 8:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(-corr)[:K]
        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        if c_max - c_min < 1e-12:
            break
        g_avg = np.zeros(n)
        for k in idx:
            g_avg += grad_corr_k_vec(h, k, n)
        g_avg /= K
        g_dir = -g_avg
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        g_dir /= gn
        improved = False
        for step in [0.01, 0.02, 0.005, 0.04, 0.001, 0.002, 0.05, 0.003]:
            ht = project_feasible(h + step * g_dir, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-15:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def subgradient_descent(h, n, time_limit, rng):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        if time.time() - _T0 > _BUDGET - 8:
            break
        progress = min(1.0, it / 400.0)
        K = 30 if progress < 0.3 else (20 if progress < 0.6 else 12)
        beta = 50.0 if progress < 0.3 else (80.0 if progress < 0.6 else 150.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(-corr)[:K]
        c_top = corr[idx]
        sh = c_top - c_top.max()
        cl = np.clip(beta * sh, -60.0, 60.0)
        ex = np.exp(cl)
        w = ex / ex.sum()
        g = np.zeros(n)
        for i in range(K):
            if w[i] < 1e-12:
                continue
            g += w[i] * grad_corr_k_vec(h, idx[i], n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        step = 0.03 * n / gn / (1.0 + 0.03 * it)
        step = min(step, 0.12)
        h_new = project_feasible(h - step * g, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            for factor in [0.5, 0.25, 0.1]:
                h_try = project_feasible(h - step * factor * g, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    break
        it += 1
    return best_h, best_c5


def coordinate_line_search(h, n, top_k, max_passes):
    h = h.copy()
    best_c5 = compute_c5(h)
    for p in range(max_passes):
        if time.time() - _T0 > _BUDGET - 8:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        active_idx = np.argsort(-corr)[:top_k]
        improved = False
        for i in range(n):
            if time.time() - _T0 > _BUDGET - 8:
                break
            old_val = h[i]
            best_local = best_c5
            best_local_val = old_val
            candidates = []
            for act_k in active_idx[:8]:
                offset = act_k - (n - 1)
                if offset >= 0:
                    if i + offset < n:
                        candidates.append(0.0)
                        candidates.append(1.0)
                    if i >= offset and i - offset < n:
                        pass
                else:
                    a = -offset
                    if i + a < n:
                        candidates.append(0.0)
                        candidates.append(1.0)
            candidates = [0.0, 1.0, max(0, old_val - 0.05), max(0, old_val - 0.01),
                          min(1, old_val + 0.05), min(1, old_val + 0.01)]
            for trial in candidates:
                if trial < -1e-12 or trial > 1 + 1e-12:
                    continue
                h[i] = trial
                c5_try = compute_c5(h)
                if c5_try < best_local - 1e-15:
                    best_local = c5_try
                    best_local_val = trial
            h[i] = best_local_val
            if best_local_val != old_val:
                improved = True
            h = project_feasible(h, n)
        if not improved:
            break
        c5_now = compute_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
    return h, best_c5


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _G_N, _G_BETA, _G_K, _G_KLIST
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 512
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: Quick SLSQP at parent resolution
    for beta in [100, 500, 2000, 10000, 50000, 100000]:
        if remaining() < 10:
            break
        maxit = 400 if beta > 10000 else 600
        h_new = slsqop_smmax(h, n, beta, K=50, maxiter=maxit)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Active set min-max SLSQP (progressive K)
    for K_active in [4, 6, 8, 10, 15, 20, 30, 40]:
        if remaining() < 10:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx_active = np.argsort(-corr)[:K_active]
        h_am, c5_am = slsqop_tminmax(h, n, idx_active, maxiter=400)
        if c5_am < best_c5:
            best_c5 = c5_am
            best_h = h_am.copy()
            h = h_am.copy()

    # Phase 3: Equalize top shifts
    if remaining() > 20:
        h_eq, c5_eq = equalize_and_descent(h, n, K=10, max_iters=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 4: Aggressive cutting plane LP at parent resolution
    if remaining() > 60:
        h_cp = h.copy()
        for it in range(600):
            if remaining() < 5:
                break
            corr_cp = np.correlate(h_cp, 1.0 - h_cp, mode='full')
            idx_cp = np.argsort(-corr_cp)[:30]
            c5_cur = compute_c5(h_cp)
            delta = max(0.005, 0.04 * (1.0 - it / 600.0) ** 0.5)
            h_lp, t_lp = cutting_plane_lp(h_cp, n, K=30, delta=delta)
            t_lp_c5 = t_lp * 2.0 / n
            if t_lp_c5 < c5_cur - 1e-12:
                best_move = h_cp.copy()
                best_move_c5 = c5_cur
                for alpha in [1.0, 0.7, 0.5, 0.3, 0.8, 0.4, 0.6]:
                    h_try = project_feasible(h_cp + alpha * (h_lp - h_cp), n)
                    c_try = compute_c5(h_try)
                    if c_try < best_move_c5:
                        best_move_c5 = c_try
                        best_move = h_try.copy()
                h_cp = best_move
                c5_cp = compute_c5(h_cp)
                if c5_cp < best_c5:
                    best_c5 = c5_cp
                    best_h = h_cp.copy()
                    h = h_cp.copy()
            elif it > 150:
                break

    # Phase 5: SLSQP polish after CP at parent res
    for beta in [50000, 100000, 200000]:
        if remaining() < 8:
            break
        h = slsqop_smmax(h, n, beta, K=60, maxiter=500)
        c5_tmp = compute_c5(h)
        if c5_tmp < best_c5:
            best_c5 = c5_tmp
            best_h = h.copy()

    # Phase 6: Interpolate to n=1024 and optimize
    n_high = 1024
    if remaining() > 200:
        h_high = interp_to(best_h, best_n, n_high)
        h_high = project_feasible(h_high, n_high)
        c5_high = compute_c5(h_high)

        # Subgradient warm-up at high res
        if remaining() > 40:
            h_high, c5_high = subgradient_descent(h_high, n_high, min(20.0, remaining() * 0.1), rng)

        # SLSQP stages at high res
        for beta in [100, 500, 2000, 10000, 50000, 100000]:
            if remaining() < 15:
                break
            maxit = 300 if beta > 10000 else 500
            h_high = slsqop_smmax(h_high, n_high, beta, K=50, maxiter=maxit)
            c5_high = compute_c5(h_high)
            if c5_high < best_c5:
                best_c5 = c5_high
                best_h = h_high.copy()
                best_n = n_high

        # Active set at high res
        for K_a in [4, 8, 12, 20, 30, 40]:
            if remaining() < 12:
                break
            corr_h = np.correlate(h_high, 1.0 - h_high, mode='full')
            idx_a = np.argsort(-corr_h)[:K_a]
            h_high, c5_high = slsqop_tminmax(h_high, n_high, idx_a, maxiter=350)
            if c5_high < best_c5:
                best_c5 = c5_high
                best_h = h_high.copy()
                best_n = n_high

        # Cutting plane at high res
        if remaining() > 80:
            for it in range(400):
                if remaining() < 5:
                    break
                corr_cp2 = np.correlate(h_high, 1.0 - h_high, mode='full')
                idx_cp2 = np.argsort(-corr_cp2)[:25]
                c5_cur2 = compute_c5(h_high)
                delta2 = max(0.003, 0.03 * (1.0 - it / 400.0) ** 0.5)
                h_lp2, t_lp2 = cutting_plane_lp(h_high, n_high, K=25, delta=delta2)
                t_lp2_c5 = t_lp2 * 2.0 / n_high
                if t_lp2_c5 < c5_cur2 - 1e-12:
                    best_move2 = h_high.copy()
                    best_move2_c5 = c5_cur2
                    for alpha in [1.0, 0.7, 0.5, 0.3]:
                        h_try2 = project_feasible(h_high + alpha * (h_lp2 - h_high), n_high)
                        c_try2 = compute_c5(h_try2)
                        if c_try2 < best_move2_c5:
                            best_move2_c5 = c_try2
                            best_move2 = h_try2.copy()
                    h_high = best_move2
                    c5_high = compute_c5(h_high)
                    if c5_high < best_c5:
                        best_c5 = c5_high
                        best_h = h_high.copy()
                        best_n = n_high
                elif it > 100:
                    break

        # Equalize at high res
        if remaining() > 15:
            h_high, c5_high = equalize_and_descent(h_high, n_high, K=10, max_iters=200)
            if c5_high < best_c5:
                best_c5 = c5_high
                best_h = h_high.copy()
                best_n = n_high

        # Final SLSQP at high res
        if remaining() > 10:
            for beta in [50000, 150000]:
                if remaining() < 5:
                    break
                h_high = slsqop_smmax(h_high, n_high, beta, K=60, maxiter=400)
            c5_high = compute_c5(h_high)
            if c5_high < best_c5:
                best_c5 = c5_high
                best_h = h_high.copy()
                best_n = n_high

    # Phase 7: Coordinate line search at best resolution
    if remaining() > 30:
        best_n_cur = best_n
        h_cur = best_h.copy()
        h_cd, c5_cd = coordinate_line_search(h_cur, best_n_cur, top_k=10, max_passes=5)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 8: Intensive refinement cycles at best resolution
    n_best = best_n
    h_best = best_h.copy()
    for cycle in range(4):
        if remaining() < 25:
            break
        # Active set with fresh top-K
        corr_c = np.correlate(h_best, 1.0 - h_best, mode='full')
        for K_a in [6, 10, 15, 25]:
            if remaining() < 8:
                break
            idx_a = np.argsort(-np.correlate(h_best, 1.0 - h_best, mode='full'))[:K_a]
            h_best, c5_t = slsqop_tminmax(h_best, n_best, idx_a, maxiter=300)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_best.copy()

        # SLSQP polish
        if remaining() > 10:
            for beta in [30000, 80000, 150000]:
                if remaining() < 5:
                    break
                h_best = slsqop_smmax(h_best, n_best, beta, K=60, maxiter=400)
            c5_t = compute_c5(h_best)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_best.copy()

        # Equalize
        if remaining() > 12:
            h_best, c5_t = equalize_and_descent(h_best, n_best, K=12, max_iters=200)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_best.copy()

        # Cutting plane
        if remaining() > 30:
            for it in range(100):
                if remaining() < 5:
                    break
                corr_cp3 = np.correlate(h_best, 1.0 - h_best, mode='full')
                idx_cp3 = np.argsort(-corr_cp3)[:20]
                c5_cur3 = compute_c5(h_best)
                delta3 = max(0.003, 0.02 * (1.0 - it / 100.0))
                h_lp3, t_lp3 = cutting_plane_lp(h_best, n_best, K=20, delta=delta3)
                t_lp3_c5 = t_lp3 * 2.0 / n_best
                if t_lp3_c5 < c5_cur3 - 1e-12:
                    for alpha in [1.0, 0.6, 0.4]:
                        h_try3 = project_feasible(h_best + alpha * (h_lp3 - h_best), n_best)
                        c_try3 = compute_c5(h_try3)
                        if c_try3 < c5_cur3:
                            h_best = h_try3
                            break
                    c5_t3 = compute_c5(h_best)
                    if c5_t3 < best_c5:
                        best_c5 = c5_t3
                        best_h = h_best.copy()
                elif it > 50:
                    break

    # Phase 9: Final ultra-fine polishing
    if remaining() > 12:
        for beta in [200000, 500000]:
            if remaining() < 5:
                break
            h_fin = slsqop_smmax(best_h, best_n, beta, K=80, maxiter=600)
            c5_fin = compute_c5(h_fin)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()

    # Phase 10: Tiny equalize steps
    if remaining() > 8:
        h_t, c5_t = equalize_and_descent(best_h, best_n, K=15, max_iters=200)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()

    # Final feasibility check and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            final_c5 = min(final_c5, float(eval_c5))
    except Exception:
        pass

    return (best_h, final_c5, best_n)
