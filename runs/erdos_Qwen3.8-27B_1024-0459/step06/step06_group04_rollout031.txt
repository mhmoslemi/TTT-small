              top_k = np.argsort(-corr)[:K]
              active_set = set(top_k.tolist())

              A_rows = []

                          a[j2] += dx * (-h[j1])

      from scipy.optimize import linprog, minimize
      import time

      _T0 = 0.0
      _BUDGET = 1300.0

      def compute_c5(h):
          dx = 2.0 / len(h)
          corr = np.correlate(h, 1.0 - h, mode='full')
          return float(np.max(corr)) * dx

      def project_feasible(h, n):
          h = np.clip(h.astype(float).copy(), 0.0, 1.0)
          target = n / 2.0
          for _ in range(500):
              diff = target - h.sum()
              if abs(diff) < 1e-12:
                  break
              mask = (h > 1e-10) & (h < 1.0 - 1e-10)
              if mask.any():
                  h[mask] += diff / mask.sum()
              else:
                  h += diff / n
              h = np.clip(h, 0.0, 1.0)
          return h

      def get_lag_gradient(h, L, n):
          """Compute gradient of f_L at h, and the constant for linearization."""
          dx = 2.0 / n
          a = np.zeros(n)
          b_const = 0.0
          if L >= 0:
              for i in range(n - L):
                  j1 = i + L
                  j2 = i
                  a[j1] += dx * (1.0 - h[j2])
                  a[j2] -= dx * h[j1]
                  b_const += dx * h[j1] * h[j2]
          else:
              M = -L
              for i in range(n - M):
                  j1 = i
                  j2 = i + M
                  a[j1] += dx * (1.0 - h[j2])
                  a[j2] -= dx * h[j1]
                  b_const += dx * h[j1] * h[j2]
          return a, b_const

      def solve_lp_active(h, n, active_lags, max_t=None):

          num_vars = n + 1

          c_obj[n] = 1.0  # minimize t

          A_rows = []
          b_list = []
          for lag_idx in active_lags:

              a, b_const = get_lag_gradient(h, L, n)

              row = np.append(a, -1.0)
              A_rows.append(row)
              b_list.append(-b_const)

          b_ub = np.array(b_list)
