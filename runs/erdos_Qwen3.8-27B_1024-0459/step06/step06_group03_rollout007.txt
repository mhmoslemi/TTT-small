    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def refine_active_set(h, n, K, max_iters):
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(-corr)[:K]
        c_max = corr[idx[0]]
        c_min_top = corr[idx[-1]]
        spread = c_max - c_min_top
        if spread < 1e-12:
            break
        g_avg = np.zeros(n)
        for k in idx:
            g_avg += grad_corr_k_vec(h, k, n)
        g_avg /= K
        # Direction that decreases all top-K simultaneously
        g_dir = -g_avg
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        g_dir /= gn
        improved = False
        steps = [0.01, 0.005, 0.002, 0.02, 0.001, 0.03, 0.0005, 0.015]
        for step in steps:
            ht = project_feasible(h + step * g_dir, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-15:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            # Try individual shift reduction
            for k in idx[:min(K, 6)]:
                gk = grad_corr_k_vec(h, k, n)
                gkn = np.linalg.norm(gk)
                if gkn < 1e-14:
                    continue
                for step in [0.005, 0.002, 0.001]:
                    ht = project_feasible(h - step * gk / gkn, n)
                    cn = compute_c5(ht)
                    if cn < best_c5 - 1e-15:
                        h = ht
                        best_c5 = cn
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                break
    return h, best_c5


def perturbation_search(h, n, rng, time_limit, t0, budget):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    while time.time() - t0 < budget - 5:
        # Small random perturbation
        scale = 0.01 / np.sqrt(n)
        h_pert = h + scale * rng.standard_normal(n)
        h_pert = project_feasible(h_pert, n)
        # Quick Adam refinement
        for beta in [50, 200]:
            h_pert, c5_pert = adam_smooth_max(h_pert, n, beta, K=20,
                                               n_iter=50, lr_base=0.02,
                                               t0=t0, budget=budget)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()
                h = h_pert.copy()
        # SLSQP polish
        h_pert = slsqp_smmax(h_pert, n, beta=500, K=30, maxiter=200)
        c5_pert = compute_c5(h_pert)
        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_pert.copy()
            h = h_pert.copy()
    return best_h, best_c5


def multi_beta_slsqp(h, n, t0, budget, betas=None):
    if betas is None:
        betas = [20, 50, 100, 300, 1000, 3000, 10000, 30000]
    h_cur = h.copy()
    best_c5 = compute_c5(h_cur)
    for beta in betas:
        if time.time() - t0 > budget - 8:
            break
        maxit = 800 if beta < 100 else (500 if beta < 3000 else 300)
        h_cur = slsqp_smmax(h_cur, n, beta=beta, K=40, maxiter=maxit)
        c5_cur = compute_c5(h_cur)
        if c5_cur < best_c5:
            best_c5 = c5_cur
    return h_cur, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 512
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_c5 = compute_c5(best_h)

    n = best_n
    h = best_h.copy()

    # Phase 1: Adam-based smooth max (different from previous subgradient)
    if remaining() > 40:
        adam_time = min(30.0, remaining() * 0.1)
        h, c5_adam = adam_smooth_max(h, n, beta=30, K=30,
                                     n_iter=2000, lr_base=0.05,
                                     t0=_T0, budget=_T0 + adam_time + 1300)
        # Run multiple Adam stages with increasing beta
        for beta in [80, 200, 500]:
            if remaining() < 8:
                break
            h, c5_adam = adam_smooth_max(h, n, beta=beta, K=30,
                                         n_iter=1500, lr_base=0.02,
                                         t0=_T0, budget=_T0 + remaining() * 0.08 + 1300)
            c5_adam = compute_c5(h)
            if c5_adam < best_c5:
                best_c5 = c5_adam
                best_h = h.copy()

    # Phase 2: Multi-stage SLSQP smooth-max
    if remaining() > 60:
        h, c5_slsqp = multi_beta_slsqp(h, n, _T0, _T0 + remaining() * 0.35,
                                       betas=[50, 200, 800, 3000, 10000, 30000])
        if c5_slsqp < best_c5:
            best_c5 = c5_slsqp
            best_h = h.copy()

    # Phase 3: Active-set min-max SLSQP with varying K
    if remaining() > 30:
        for K_active in [4, 6, 8, 10, 12, 15, 20, 25, 35]:
            if remaining() < 10:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            idx_active = np.argsort(-corr)[:K_active]
            # Also try slightly different active sets
            if K_active < len(corr):
                idx_extra = np.argsort(-corr)[K_active:K_active + 5]
                all_idx = list(idx_active) + list(idx_extra[:3])
            else:
                all_idx = list(idx_active)
            h_am, c5_am = slsqp_tminmax(h, n, all_idx, maxiter=400)
            if c5_am < best_c5:
                best_c5 = c5_am
                best_h = h_am.copy()
                h = h_am.copy()

    # Phase 4: Fine active-set refinement
    if remaining() > 15:
        h_ref, c5_ref = refine_active_set(h, n, K=10, max_iters=500)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()
            h = h_ref.copy()

    # Phase 5: Perturbation search (try to escape local minima)
    if remaining() > 60:
        h_pert, c5_pert = perturbation_search(h, n, rng,
                                              min(40.0, remaining() * 0.25),
                                              _T0, _T0 + remaining() * 0.25)
        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_pert.copy()
            h = h_pert.copy()
            n = n

    # Phase 6: SLSQP polish after perturbation
    if remaining() > 30:
        for beta in [5000, 15000, 50000]:
            if remaining() < 8:
                break
            h = slsqp_smmax(h, n, beta=beta, K=50, maxiter=400)
        c5_pol = compute_c5(h)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h.copy()

    # Phase 7: Try n=256 (lower resolution, potentially better structure)
    n_low = 256
    if remaining() > 80:
        h_low = interp_to(best_h, best_n, n_low)
        h_low = project_feasible(h_low, n_low)
        c5_low = compute_c5(h_low)

        if c5_low <= best_c5 + 0.002:
            # Adam at low res
            if remaining() > 40:
                h_low, c5_low = adam_smooth_max(h_low, n_low, beta=50, K=25,
                                                n_iter=1500, lr_base=0.04,
                                                t0=_T0, budget=_T0 + remaining() * 0.2)
            # SLSQP at low res
            if remaining() > 30:
                for beta in [100, 500, 2000, 10000]:
                    if remaining() < 10:
                        break
                    h_low = slsqp_smmax(h_low, n_low, beta=beta, K=30, maxiter=400)
            # Active set at low res
            if remaining() > 20:
                corr_low = np.correlate(h_low, 1.0 - h_low, mode='full')
                for K_a in [6, 10, 15]:
                    if remaining() < 8:
                        break
                    idx_a = np.argsort(-corr_low)[:K_a]
                    h_low, c5_low = slsqp_tminmax(h_low, n_low, idx_a, maxiter=300)
            # Refine
            if remaining() > 10:
                h_low, c5_low = refine_active_set(h_low, n_low, K=8, max_iters=300)

            if c5_low < best_c5:
                best_c5 = c5_low
                best_h = h_low.copy()
                best_n = n_low

    # Phase 8: Try n=1024 (higher resolution)
    n_high = 1024
    if remaining() > 100:
        h_high = interp_to(best_h, best_n, n_high)
        h_high = project_feasible(h_high, n_high)
        c5_high = compute_c5(h_high)

        if c5_high <= best_c5 + 0.003:
            if remaining() > 50:
                h_high, c5_high = adam_smooth_max(h_high, n_high, beta=40, K=30,
                                                  n_iter=1000, lr_base=0.03,
                                                  t0=_T0, budget=_T0 + remaining() * 0.2)
            if remaining() > 30:
                for beta in [100, 500, 2000]:
                    if remaining() < 12:
                        break
                    h_high = slsqp_smmax(h_high, n_high, beta=beta, K=35, maxiter=250)
            if remaining() > 20:
                corr_high = np.correlate(h_high, 1.0 - h_high, mode='full')
                for K_a in [10, 20]:
                    if remaining() < 10:
                        break
                    idx_a = np.argsort(-corr_high)[:K_a]
                    h_high, c5_high = slsqp_tminmax(h_high, n_high, idx_a, maxiter=200)
            if remaining() > 10:
                h_high, c5_high = refine_active_set(h_high, n_high, K=8, max_iters=200)

            if c5_high < best_c5:
                best_c5 = c5_high
                best_h = h_high.copy()
                best_n = n_high

    # Phase 9: Cycle of refinement at best resolution
    n = best_n
    h = best_h.copy()
    for cycle in range(4):
        if remaining() < 25:
            break
        # Adam
        if remaining() > 15:
            h, c5_c = adam_smooth_max(h, n, beta=200, K=30,
                                      n_iter=500, lr_base=0.01,
                                      t0=_T0, budget=_T0 + remaining() * 0.15)
            c5_c = compute_c5(h)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h.copy()
        # SLSQP
        if remaining() > 15:
            for beta in [5000, 20000]:
                if remaining() < 8:
                    break
                h = slsqp_smmax(h, n, beta=beta, K=50, maxiter=300)
            c5_c = compute_c5(h)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h.copy()
        # Active set
        if remaining() > 15:
            corr_c = np.correlate(h, 1.0 - h, mode='full')
            for K_a in [8, 15, 25]:
                if remaining() < 8:
                    break
                idx_a = np.argsort(-corr_c)[:K_a]
                h, c5_c = slsqp_tminmax(h, n, idx_a, maxiter=250)
                if c5_c < best_c5:
                    best_c5 = c5_c
                    best_h = h.copy()
        # Fine refine
        if remaining() > 10:
            h, c5_c = refine_active_set(h, n, K=12, max_iters=300)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h.copy()

    # Phase 10: Ultra-fine final steps
    h = best_h.copy()
    if remaining() > 10:
        # Very tight active set refinement
        h_f, c5_f = refine_active_set(h, best_n, K=6, max_iters=1000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

        # Tiny gradient steps on the max value
        h_t = best_h.copy()
        for it in range(200):
            if remaining() < 3:
                break
            corr_t = np.correlate(h_t, 1.0 - h_t, mode='full')
            idx_t = np.argsort(-corr_t)[:8]
            g_t = np.zeros(best_n)
            for k in idx_t:
                g_t += grad_corr_k_vec(h_t, k, best_n)
            g_t /= len(idx_t)
            gn_t = np.linalg.norm(g_t)
            if gn_t < 1e-14:
                break
            for step in [0.003, 0.001, 0.0005, 0.002, 0.0008, 0.0015]:
                h_try = project_feasible(h_t - step * g_t / gn_t, best_n)
                c_try = compute_c5(h_try)
                if c_try < best_c5:
                    best_c5 = c_try
                    best_h = h_try.copy()
                    h_t = h_try.copy()
                    break

    # Phase 11: Final SLSQP with very high beta
    if remaining() > 8:
        h_fin = slsqp_smmax(best_h, best_n, beta=100000, K=60, maxiter=500)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final feasibility projection
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
