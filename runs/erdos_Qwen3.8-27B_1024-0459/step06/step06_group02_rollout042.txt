def grad_corr_k(h, k, n):
    """Gradient of correlation at lag k (k is the index in the full correlation output)."""
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        if npairs > 0:
            g[offset:] += (1.0 - h[:npairs])
            g[:npairs] -= h[offset:]
    else:
        a = -offset
        npairs = n - a
        if npairs > 0:
            g[:npairs] += (1.0 - h[a:])
            g[a:] -= h[:npairs]
    return g

def smooth_max_objective(x, n, beta, lam, rho):
    """Augmented Lagrangian objective with smooth max."""
    corr = compute_corr_full(x)
    m = np.max(corr)
    shifted = beta * (corr - m)
    clipped = np.clip(shifted, -60.0, 60.0)
    ex = np.exp(clipped)
    smax = m + np.log(np.sum(ex)) / beta
    val = smax * (2.0 / n)
    # Augmented Lagrangian for equality constraint
    s = np.sum(x) - n / 2.0
    val += lam * s + 0.5 * rho * s * s
    return val

def smooth_max_gradient(x, n, beta, lam, rho):
    corr = compute_corr_full(x)
    m = np.max(corr)
    shifted = beta * (corr - m)
    clipped = np.clip(shifted, -60.0, 60.0)
    ex = np.exp(clipped)
    s_exp = np.sum(ex)
    w = ex / s_exp
    dx = 2.0 / n
    
    g = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        g += wk * grad_corr_k(x, k, n)
    
    # Equality constraint gradient
    s = np.sum(x) - n / 2.0
    g += (lam + rho * s) * np.ones(n)
    
    return g

def aug_lagrangian_lbfgs(h, n, beta, maxiter=500, ftol=1e-16, time_limit=60):
    """Augmented Lagrangian + L-BFGS-B optimization."""
    lam = 0.0
    rho = 1.0
    start_time = time.time()
    
    for outer in range(20):
        if time.time() - start_time > time_limit:
            break
        if rho > 1e6:
            break
        
        bounds = [(0.0, 1.0)] * n
        
        def obj(x):
            return smooth_max_objective(x, n, beta, lam, rho)
        
        def grad(x):
            return smooth_max_gradient(x, n, beta, lam, rho)
        
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
            h_new = res.x
        except Exception:
            break
        
        h_new = project_feasible(h_new, n)
        constraint_violation = abs(np.sum(h_new) - n / 2.0)
        
        lam = lam + rho * (np.sum(h_new) - n / 2.0)
        
        if constraint_violation < 1e-8:
            rho *= 2.0
        else:
            rho *= 10.0
        
        h = h_new
    
    return h

def slsqp_smooth_max(h, n, beta, maxiter=500, ftol=1e-18, time_limit=60):
    start_time = time.time()
    
    def obj(x):
        corr = compute_corr_full(x)
        m = np.max(corr)
        shifted = beta * (corr - m)
        clipped = np.clip(shifted, -60.0, 60.0)
        ex = np.exp(clipped)
        val = (m + np.log(np.sum(ex)) / beta) * (2.0 / n)
        return val

    def grad(x):
        corr = compute_corr_full(x)
        m = np.max(corr)
        shifted = beta * (corr - m)
        clipped = np.clip(shifted, -60.0, 60.0)
        ex = np.exp(clipped)
        s = np.sum(ex)
        w = ex / s
        dx_val = 2.0 / n
        g = np.zeros(n)
        for k in range(len(corr)):
            wk = w[k] * dx_val
            if wk < 1e-14:
                continue
            g += wk * grad_corr_k(x, k, n)
        return g

    def constraint_fun(x):
        return np.sum(x) - n / 2.0

    def constraint_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]

    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        if res.x is not None and time.time() - start_time < time_limit:
            h_new = project_feasible(res.x, n)
            if compute_c5(h_new) <= compute_c5(h) + 1e-10:
                return h_new
    except Exception:
        pass
    return h.copy()

def active_set_polish(h, n, top_k=20, beta=50000, maxiter=300):

        # = C_lg + t*(1-h[i+offset]) * I_valid1 + h[i-offset]*I_valid2 - t*h[i-offset]*I_valid2
        # = (C_lg + h[i-offset]*I_valid2) + t*((1-h[i+offset])*I_valid1 - h[i-offset]*I_valid2)
        
        # So slope_lg = (1-h[i+offset])*I_valid1 - h[i-offset]*I_valid2
        # intercept_lg = C_lg + h[i-offset]*I_valid2
        
        # But C_lg = C(lg) - old_terms, where old_terms is the contribution of h[i] to C(lg)
        # old_terms = old_val*(1-h[i+offset])*I_valid1 + h[i-offset]*(1-old_val)*I_valid2
        
        # This is getting complex. Let me just compute the new C(lg) directly.
        
        # Actually, a simpler approach: for each lag, compute how C(lag) changes when h[i] changes.
        # ΔC(lag) = new_C(lag) - old_C(lag)
        # This is linear in t = h[i]:
        # ΔC(lag) = slope_lag * (t - old_val)
        
        # So C(lag; t) = C(lag; old_val) + slope_lag * (t - old_val)
        #                = (C(lag) - slope_lag * old_val) + slope_lag * t
        
        # The slope:
        # For offset = lag - (n-1):
        # If 0 ≤ i+offset < n: slope gets (1 - h[i+offset]) [from the h[i]*(1-h[i+offset]) term]
        # If 0 ≤ i-offset < n: slope gets -h[i-offset] [from the -h[i-offset]*h[i] term]
        
        # Let me compute slopes for all lags efficiently.
        
        offset_range = np.arange(-(n-1), n)  # all possible offsets
        # For offset ℓ: 
        # valid1 = (0 ≤ i+ℓ < n) i.e., -i ≤ ℓ < n-i
        # valid2 = (0 ≤ i-ℓ < n) i.e., i-n < ℓ ≤ i
        # slope = (1-h[i+ℓ])*valid1 - h[i-ℓ]*valid2
        
        slopes = np.zeros(2*n-1)
        for lg in range(2*n-1):
            off = lg - (n-1)
            s = 0.0
            if 0 <= i + off < n:
                s += (1.0 - h[i + off])
            if 0 <= i - off < n:
                s -= h[i - off]
            slopes[lg] = s
        
        # Now C(lg; t) = corr[lg] - slopes[lg] * old_val + slopes[lg] * t
        # = corr[lg] + slopes[lg] * (t - old_val)
        
        # We want to minimize max_lg (corr[lg] + slopes[lg] * (t - old_val)) * dx
        # over t ∈ [0, 1]
        
        # This is min_t max_lg (a_lg * t + b_lg) where a_lg = slopes[lg], b_lg = corr[lg] - slopes[lg]*old_val
        # (times dx, which is a constant so we can ignore it)
        
        # This is a 1D convex optimization. The optimal t is where the two maximum lines cross,
        # or at a boundary.
        
        # Quick approach: ternary search or bisection on the derivative.
        # The function f(t) = max_lg (a_lg * t + b_lg) is convex and piecewise linear.
        # Its subdifferential at t is the convex hull of {a_lg : lg is argmax}.
        # The optimal t has 0 in its subdifferential, or is at a boundary.
        
        # Simple approach: try a few candidate values.
        # The "knee" points are where two lines cross: t = (b_lg2 - b_lg1) / (a_lg1 - a_lg2)
        # But there are O(n²) such points. Instead, use a simple approach:
        
        # Evaluate f(t) at t = 0, 1, and the "center" where the weighted average slope is 0.
        # Or just do a golden section search.
        
        # Actually, the simplest robust approach: evaluate at many points and pick the best.
        candidates = np.concatenate([
            np.linspace(0, 1, 20),
            [old_val, 0.0, 1.0]
        ])
        candidates = np.clip(candidates, 0, 1)
        
        best_t = old_val
        best_f = np.inf
        for t in candidates:
            vals = corr + slopes * (t - old_val)
            f = np.max(vals)
            if f < best_f:
                best_f = f
                best_t = t
        
        h[i] = best_t
    
    return h
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        if npairs > 0:
            g[offset:] += (1.0 - h[:npairs])
            g[:npairs] -= h[offset:]
    else:
        a = -offset
        npairs = n - a
        if npairs > 0:
            g[:npairs] += (1.0 - h[a:])
            g[a:] -= h[:npairs]
    return g

def slsqp_optimize(h, n, beta, maxiter=500, ftol=1e-18, time_limit=60):
    start_time = time.time()
    
    def obj(x):
        corr = compute_corr_full(x)
        m = np.max(corr)
        shifted = beta * (corr - m)
        shifted = np.clip(shifted, -60.0, 60.0)
        ex = np.exp(shifted)
        val = (m + np.log(np.sum(ex)) / beta) * (2.0 / n)
        return val

    def grad(x):
        corr = compute_corr_full(x)
        m = np.max(corr)
        shifted = beta * (corr - m)
        shifted = np.clip(shifted, -60.0, 60.0)
        ex = np.exp(shifted)
        s = np.sum(ex)
        w = ex / s
        dx_val = 2.0 / n
        g = np.zeros(n)
        for k in range(len(corr)):
            wk = w[k] * dx_val
            if wk < 1e-13:
                continue
            g += wk * grad_corr_k(x, k, n)
        return g

    def constraint_fun(x):
        return np.sum(x) - n / 2.0

    def constraint_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]

    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        if time.time() - start_time < time_limit:
            h_new = project_feasible(res.x, n)
            if compute_c5(h_new) <= compute_c5(h) + 1e-10:
                return h_new
    except Exception:
        pass
    return h.copy()

def lbfgs_aug_lagrangian(h, n, beta, time_limit=60, max_outer=30):
    """Augmented Lagrangian method with L-BFGS-B."""
    start_time = time.time()
    lam = 0.0
    rho = 10.0
    
    def obj(x):
        corr = compute_corr_full(x)
        m = np.max(corr)
        shifted = beta * (corr - m)
        shifted = np.clip(shifted, -60.0, 60.0)
        ex = np.exp(shifted)
        val = (m + np.log(np.sum(ex)) / beta) * (2.0 / n)
        s = np.sum(x) - n / 2.0
        val += lam * s + 0.5 * rho * s * s
        return val

    def grad(x):
        corr = compute_corr_full(x)
        m = np.max(corr)
        shifted = beta * (corr - m)
        shifted = np.clip(shifted, -60.0, 60.0)
        ex = np.exp(shifted)
        s_exp = np.sum(ex)
        w = ex / s_exp
        dx_val = 2.0 / n
        g = np.zeros(n)
        for k in range(len(corr)):
            wk = w[k] * dx_val
            if wk < 1e-13:
                continue
            g += wk * grad_corr_k(x, k, n)
        s = np.sum(x) - n / 2.0
        g += (lam + rho * s)
        return g
    
    bounds = [(0.0, 1.0)] * n
    
    for outer in range(max_outer):
        if time.time() - start_time > time_limit:
            break
        
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': 200, 'ftol': 1e-15, 'disp': False})
            h = res.x
        except Exception:
            break
        
        h = project_feasible(h, n)
        violation = abs(np.sum(h) - n / 2.0)
        
        lam += rho * (np.sum(h) - n / 2.0)
        
        if violation < 1e-10:
            rho *= 2.0
        else:
            rho *= 5.0
        
        if rho > 1e8:
            break
    
    return h

def coordinate_descent(h, n, max_sweeps=100, time_limit=60, n_candidates=50):
    """Coordinate descent with exact 1D minimization at each step."""
    h = h.copy()
    start_time = time.time()
    best_c5 = compute_c5(h)
    
    # Precompute candidate t values
    t_candidates = np.linspace(0.0, 1.0, n_candidates)
    
    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        
        corr = compute_corr_full(h)
        
        for i in range(n):
            if time.time() - start_time > time_limit:
                    break
            
            old_val = h[i]
            off = np.arange(2*n-1) - (n-1)
            
            # Compute slopes
            idx1 = i + off
            valid1 = (idx1 >= 0) & (idx1 < n)
            idx2 = i - off
            valid2 = (idx2 >= 0) & (idx2 < n)
            
            slopes = np.zeros(2*n-1)
            if np.any(valid1):
                slopes[valid1] += (1.0 - h[idx1[valid1]])
            if np.any(valid2):
                slopes[valid2] -= h[idx2[valid2]]
            
            # Evaluate at candidates
            # C(lg; t) = corr[lg] + slopes[lg] * (t - old_val)
            diffs = t_candidates - old_val  # shape: (n_candidates,)
            # vals: (n_candidates, 2n-1)
            vals = corr[None, :] + slopes[None, :] * diffs[:, None]
            max_vals = np.max(vals, axis=1)
            
            best_idx = np.argmin(max_vals)
            best_t = t_candidates[best_idx]
            
            if best_t != old_val:
                h[i] = best_t
        
        # Project after each sweep
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
    
    return h, best_c5
