def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:

    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    return (m + np.log(np.sum(ez)) / beta) * dx

    m = np.max(corr)

    S = np.sum(ez)

        # SLSQP refinement (time-bounded)
        if remaining_time() - (time.time() - start) > 40:
            h_slq = slsqp_opt(h, n, 5000, maxiter=300)
            c5_slq = compute_c5(h_slq)
            if c5_slq < best_c5 - 1e-14:
                best_c5 = c5_slq
                best_h = h_slq.copy()
                h = h_slq.copy()
                improved_this_round = True
            else:
                h = h_slq.copy()
            h_cd2, c5_cd2 = coord_fast(h, n, max_sweeps=40, rng=rng)
            if c5_cd2 < best_c5 - 1e-14:
                best_c5 = c5_cd2
                best_h = h_cd2.copy()
                h = h_cd2.copy()
                improved_this_round = True
            else:
                h = h_cd2.copy()

        if improved_this_round:
            stall_rounds = 0
        else:
            stall_rounds += 1

        # Perturbation to escape local optima
        elapsed = time.time() - start
        frac = min(1.0, elapsed / time_limit)
        pert_size = 0.02 + 0.08 * frac + 0.05 * stall_rounds / max_stall
        pert = rng.randn(n) * pert_size
        h_pert = project_feasible(h + pert, n)
        c5_pert = compute_c5(h_pert)
        # Accept perturbation if not too bad
        if c5_pert < best_c5 + 0.01 or stall_rounds >= max_stall:
            h = h_pert
            stall_rounds = 0

    return best_h, best_c5


def subgradient_fast(h, n, top_k=5, max_iters=100000, time_limit=100):
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = 0.5
    no_improve = 0

    for it in range(max_iters):
        if time.time() - start > time_limit or remaining_time() < 10:
            break
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        for ki in idx_top:
            off = ki - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:] += (1.0 - h[:np_])
                g[:np_] -= h[off:]
            else:
                a = -off
                np_ = n - a
                g[:np_] += (1.0 - h[a:])
                g[a:] -= h[:np_]
        g *= dx / top_k
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        d = -g / gn
        improved = False
        for alpha in [lr * 0.1, lr * 0.3, lr, lr * 3]:
            ht = project_feasible(h + alpha * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-15:
                best_c5 = cn
                best_h = ht.copy()
                h = ht.copy()
                improved = True
                no_improve = 0
                break
        if not improved:
            no_improve += 1
            lr *= 0.95
            if no_improve > 5000:
                break
        else:
            lr = min(lr * 1.02, 2.0)

    return best_h, best_c5


def theta_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    try:
        best_h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        best_h = np.zeros(640)
        best_h[:320] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Initial SLSQP
    h = best_h.copy()
    for beta in [1000, 10000, 100000]:
        if remaining_time() < 60:
            break
        h_new = slsqp_opt(h, best_n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Fast coordinate descent
    if remaining_time() > 30:
        h_cd, c5_cd = coord_fast(h, best_n, max_sweeps=80, rng=rng)
        if c5_cd < best_c5 - 1e-14:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h = h_cd.copy()
        else:
            h = h_cd.copy()
            h_cd2, c5_cd2 = coord_fast(best_h, best_n, max_sweeps=80, rng=rng)
            if c5_cd2 < best_c5 - 1e-14:
                best_c5 = c5_cd2
                best_h = h_cd2.copy()
            h = best_h.copy()

    # Phase 3: Subgradient fast
    if remaining_time() > 60:
        h_sg, c5_sg = subgradient_fast(h, best_n, top_k=5, max_iters=50000, time_limit=60)
        if c5_sg < best_c5 - 1e-14:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 4: Iterative refinement (main workhorse)
    if remaining_time() > 200:
        time_for_iter = min(remaining_time() - 200, 800)
        h_iter, c5_iter = iterative_refine(h, best_n, time_for_iter, rng)
        if c5_iter < best_c5 - 1e-14:
            best_c5 = c5_iter
            best_h = h_iter.copy()
            h = h_iter.copy()
        else:
            h = best_h.copy()

    # Phase 5: Multi-resolution
    resolutions = [512, 800, 1024, 768]
    resolutions = [r for r in resolutions if r % 2 == 0 and r != best_n]
    for n_t in resolutions[:3]:
        if remaining_time() < 80:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        h_opt = h_feas
        for beta in [5000, 50000]:
            if remaining_time() < 30:
                break
            h_opt = slsqp_opt(h_opt, n_t, beta, maxiter=300)
        h_opt, c5_opt = coord_fast(h_opt, n_t, max_sweeps=50, rng=rng)
        if remaining_time() > 30:
            h_sg2, c5_sg2 = subgradient_fast(h_opt, n_t, top_k=5, max_iters=30000, time_limit=30)
            if c5_sg2 < c5_opt:
                c5_opt = c5_sg2
                h_opt = h_sg2
            h_opt, c5_opt2 = coord_fast(h_opt, n_t, max_sweeps=30, rng=rng)
            if c5_opt2 < c5_opt:
                c5_opt = c5_opt2
                h_opt = h_opt
        if c5_opt < best_c5 - 1e-14:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t
            h = h_opt.copy()

    # Phase 6: Try alternative starts
    if remaining_time() > 100:
        h_tm = theta_morse(best_n)
        h_tm_opt = slsqp_opt(h_tm, best_n, 5000, maxiter=300)
        h_tm_opt, c5_tm = coord_fast(h_tm_opt, best_n, max_sweeps=40, rng=rng)
        if c5_tm < best_c5 - 1e-14:
            best_c5 = c5_tm
            best_h = h_tm_opt.copy()
            h = h_tm_opt.copy()

    # Phase 7: Final polish - coordinate descent + SLSQP
    if remaining_time() > 40:
        h_final = best_h.copy()
        h_final, c5_f = coord_fast(h_final, best_n, max_sweeps=100, rng=rng)
        if c5_f < best_c5 - 1e-14:
            best_c5 = c5_f
            best_h = h_final.copy()
        if remaining_time() > 20:
            h_slq = slsqp_opt(best_h, best_n, 200000, maxiter=500)
            c5_slq = compute_c5(h_slq)
            if c5_slq < best_c5 - 1e-14:
                best_c5 = c5_slq
                best_h = h_slq.copy()
        if remaining_time() > 15:
            h_final2, c5_f2 = coord_fast(best_h, best_n, max_sweeps=80, rng=rng)
            if c5_f2 < best_c5 - 1e-14:
                best_c5 = c5_f2
                best_h = h_final2.copy()

    # Phase 8: Final subgradient micro-optimization
    if remaining_time() > 20:
        h_sg_fin, c5_sf = subgradient_fast(best_h, best_n, top_k=7, max_iters=50000, time_limit=20)
        if c5_sf < best_c5 - 1e-14:
            best_c5 = c5_sf
            best_h = h_sg_fin.copy()
        h_cd_fin, c5_cf = coord_fast(best_h, best_n, max_sweeps=50, rng=rng)
        if c5_cf < best_c5 - 1e-14:
            best_c5 = c5_cf
            best_h = h_cd_fin.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
