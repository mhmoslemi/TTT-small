    g = np.zeros(n)

            # More SLSQP
            if remaining() > 60:
                for beta in [5000, 30000, 100000]:
                    if remaining() < 20:
                        break
                    h_new = slsqp_smooth(h_high, n_high, beta, K=100, maxiter=300)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_high:
                        c5_high = c5_new
                        h_high = h_new.copy()

            if c5_high < best_c5:
                best_c5 = c5_high
                best_h = h_high.copy()
                best_n = n_high

    # Phase 4: Final intensive refinement at best resolution
    n = best_n
    h = best_h.copy()
    for cycle in range(4):
        if remaining() < 20:
            break
        # SLSQP with high beta
        for beta in [10000, 50000, 200000]:
            if remaining() < 10:
                break
            h_new = slsqp_smooth(h, n, beta, K=100, maxiter=400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

        # Active set
        if remaining() > 20:
            for K_active in [10, 20, 30, 40]:
                if remaining() < 10:
                    break
                corr_a = np.correlate(h, 1.0 - h, mode='full')
                idx_a = np.argsort(-corr_a)[:K_active]
                h_new = slsqp_minmax(h, n, idx_a, maxiter=250)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()

        # Equalize
        if remaining() > 15:
            h, c5_eq = equalize_shifts(h, n, K=12, max_iters=200)
            if c5_eq < best_c5:
                best_c5 = c5_eq
                best_h = h.copy()

        # Adam fine-tune
        if remaining() > 30:
            h, c5_ad = adam_descent(h, n, beta=500, K=80,
                                    steps=500, lr_init=0.002, time_limit=40)
            if c5_ad < best_c5:
                best_c5 = c5_ad
                best_h = h.copy()

    # Phase 5: Tiny step refinement
    h = best_h.copy()
    n = best_n
    if remaining() > 10:
        for it in range(200):
            if remaining() < 3:
                break
            corr_t = np.correlate(h, 1.0 - h, mode='full')
            idx_t = np.argsort(-corr_t)[:20]
            g_t = np.zeros(n)
            for k in idx_t:
                g_t += grad_corr_k(h, k, n)
            g_t /= len(idx_t)
            gn_t = np.linalg.norm(g_t)
            if gn_t < 1e-14:
                break
            improved = False
            for step in [0.003, 0.001, 0.005, 0.002, 0.0005, 0.008, 0.004]:
                h_try = project_feasible(h - step * g_t / gn_t, n)
                c_try = compute_c5(h_try)
                if c_try < best_c5:
                    best_c5 = c_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    improved = True
                    break
            if not improved:
                for step in [0.002, 0.001, 0.004, 0.0005]:
                    h_try = project_feasible(h + step * g_t / gn_t, n)
                    c_try = compute_c5(h_try)
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
                        break
            if not improved:
                break

    # Final feasibility and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            best_c5 = min(best_c5, float(eval_c5))
    except Exception:
        pass

    return (best_h, best_c5, best_n)
