def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        # Distribute to coordinates with room
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            adj = diff / cnt
            h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
        else:
            break
    return h

def logsumexp_topk(corr, K, beta):
    """Compute smooth approximation to max of top K correlations."""
    idx = np.argsort(-corr)[:K]
    c = corr[idx]
    m = np.max(c)
    shifted = beta * (c - m)
    shifted = np.clip(shifted, -100, 100)
    ex = np.exp(shifted)
    val = m + np.log(ex.sum()) / beta
    weights = ex / ex.sum()
    return val, idx, weights

def smooth_obj_and_grad(h, n, K, beta):
    dx = 2.0 / n
    corr = get_corr(h)
    val, idx, w = logsumexp_topk(corr, K, beta)
    
    # Gradient: for each active shift, compute grad of that correlation
    g = np.zeros(n)
    for i in range(K):

        if wk < 1e-15:

        offset = k - (n - 1)

            g[:npairs] -= wk * h[offset:]
        else:
            a = -offset

            g[:npairs] += wk * (1.0 - h[a:])
            g[a:] -= wk * h[:npairs]
    
    return val, g

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = int(np.count_nonzero(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
            h[mask] = np.minimum(h[mask], 1.0)
        else:
            mask = h > 1e-12
            cnt = int(np.count_nonzero(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
            h[mask] = np.maximum(h[mask], 0.0)
    return h


def smooth_val_grad(h, n, K, beta):
    dx = 2.0 / n
    corr = get_corr(h)
    idx = np.argpartition(-corr, min(K, len(corr) - 1))[:K]
    c_top = corr[idx]
    m = float(np.max(c_top))
    sh = np.clip(beta * (c_top - m), -80.0, 80.0)
    ex = np.exp(sh)
    s = float(ex.sum())
    w = ex / s
    val = (m + np.log(s) / beta) * dx
    g = np.zeros(n)
    for i in range(K):
        wk = w[i] * dx
        if wk < 1e-16:
            continue
        k = int(idx[i])
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            g[offset:] += wk * (1.0 - h[:npairs])
            g[:npairs] -= wk * h[offset:]
        else:
            a = -offset
            npairs = n - a
            g[:npairs] += wk * (1.0 - h[a:])
            g[a:] -= wk * h[:npairs]
    return val, g


def lbfgs_optimize(h, n, K, beta, maxiter, penalty=1e8):
    def obj_fn(x):
        val, _ = smooth_val_grad(x, n, K, beta)
        pen = penalty * (np.sum(x) - n / 2.0) ** 2
        return val + pen

    def grad_fn(x):
        _, g = smooth_val_grad(x, n, K, beta)
        g = g + 2.0 * penalty * (np.sum(x) - n / 2.0)
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fn, h.copy(), jac=grad_fn, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-20,
                                               'gtol': 1e-16, 'maxls': 50})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def slsqp_optimize(h, n, K, beta, maxiter=300):
    def obj_fn(x):
        v, _ = smooth_val_grad(x, n, K, beta)
        return v

    def grad_fn(x):
        _, g = smooth_val_grad(x, n, K, beta)
        return g

    def cfun(x):
        return np.sum(x) - n / 2.0

    def cjac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj_fn, h.copy(), jac=grad_fn, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def slsqp_active_set(h, n, active_indices, beta, maxiter=300):
    dx = 2.0 / n

    def obj_fn(x):
        corr = get_corr(x)
        c_top = corr[active_indices]
        m = float(np.max(c_top))
        sh = np.clip(beta * (c_top - m), -80.0, 80.0)
        ex = np.exp(sh)
        s = float(ex.sum())
        return (m + np.log(s) / beta) * dx

    def grad_fn(x):
        corr = get_corr(x)
        c_top = corr[active_indices]
        m = float(np.max(c_top))
        sh = np.clip(beta * (c_top - m), -80.0, 80.0)
        ex = np.exp(sh)
        w = ex / ex.sum()
        g = np.zeros(n)
        for i in range(len(active_indices)):
            wk = w[i] * dx
            if wk < 1e-16:
                continue
            k = int(active_indices[i])
            offset = k - (n - 1)
            if offset >= 0:
                npairs = n - offset
                g[offset:] += wk * (1.0 - h[:npairs])
                g[:npairs] -= wk * h[offset:]
            else:
                a = -offset
                npairs = n - a
                g[:npairs] += wk * (1.0 - h[a:])
                g[a:] -= wk * h[:npairs]
        return g

    def cfun(x):
        return np.sum(x) - n / 2.0

    def cjac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj_fn, h.copy(), jac=grad_fn, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def nesterov_subgrad(h, n, time_limit, K, beta):
    h_curr = h.copy()
    h_prev = h.copy()
    h_best = h.copy()
    c5_best = compute_c5(h)
    start = time.time()
    for it in range(200000):
        if time.time() - start > time_limit:
            break
        if time.time() - _T0 > _BUDGET - 10:
            break
        t = it + 1
        if t > 1:
            gam = (t - 1.0 + np.sqrt(1.0 + 4.0 * (t - 1.0) ** 2)) / (2.0 * t)
        else:
            gam = 1.0
        y = np.clip(h_curr + gam * (h_curr - h_prev), 0.0, 1.0)
        _, g = smooth_val_grad(y, n, K, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        step = 0.12 / gn
        step = min(step, 0.25) * (1.0 + 0.5 / (1.0 + t * 0.005))
        h_new = project_feasible(y - step * g, n)
        c5_new = compute_c5(h_new)
        h_prev[:] = h_curr
        h_curr = h_new
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_curr.copy()
    return h_best, c5_best


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def coord_descent_polish(h, n, max_sweeps, rng):
    h = h.copy()
    c5_best = compute_c5(h)
    corr = get_corr(h)
    # Focus on coordinates that affect the top correlations
    top_idx = np.argpartition(-corr, 20)[:20]
    affected = set()
    for k in top_idx:
        offset = k - (n - 1)
        if offset >= 0:
            for i in range(n - offset):
                affected.add(i)
                affected.add(i + offset)
        else:
            a = -offset
            for i in range(n - a):
                affected.add(i)
                affected.add(i + a)
    affected_list = list(affected)
    for sweep in range(max_sweeps):
        improved = False
        order = list(affected_list)
        rng.shuffle(order)
        for j in order:
            old_val = h[j]
            best_t = old_val
            best_c = c5_best
            # Try perturbed values around current
            for delta in [0.005, -0.005, 0.01, -0.01, 0.02, -0.02, 0.5]:
                t = old_val + delta
                if t < 0.0 or t > 1.0:
                    continue
                h[j] = t
                h_try = project_feasible(h, n)
                c_try = compute_c5(h_try)
                if c_try < best_c - 1e-13:
                    best_c = c_try
                    best_t = t
                    h[:] = h_try
                h[j] = old_val
            if best_t != old_val:
                h[j] = best_t
                h = project_feasible(h, n)
                c5_best = compute_c5(h)
                corr = get_corr(h)
                improved = True
        if not improved:
            break
        if time.time() - _T0 > _BUDGET - 15:
            break
    return h, c5_best
