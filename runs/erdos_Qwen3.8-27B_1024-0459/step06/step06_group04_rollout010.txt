import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        if interior.any():
            h[interior] += diff / interior.sum()
        else:
            if diff > 0:
                z = np.where(h < 0.5)[0]
                if len(z) > 0:
                    h[z[0]] = min(1.0, h[z[0]] + diff)
                else:
                    h += diff / n
            else:
                o = np.where(h > 0.5)[0]
                if len(o) > 0:
                    h[o[0]] = max(0.0, h[o[0]] + diff)
                else:
                    h += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h


def grad_topk(h, n, K, beta_w):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_val = float(corr.max()) * dx
    idx = np.argsort(-corr)[:K]
    top_vals = corr[idx]
    w = np.exp(beta_w * (top_vals - top_vals.max()))
    s = w.sum()
    if s < 1e-30:
        w = np.ones(K) / K
    else:
        w /= s
    g = np.zeros(n)
    for wi, k_idx in zip(w, idx):
        if wi < 1e-12:
            continue
        off = k_idx - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wi * (1.0 - h[:np_])
            g[:np_] -= wi * h[off:]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wi * (1.0 - h[ao:])
            g[ao:] -= wi * h[:np_]
    g *= dx
    return g, c5_val, idx


def momentum_descent(h, n, time_limit, K=5, beta_w=40.0, lr=0.03, momentum=0.7):
    h = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    start = time.time()
    patience = 0
    max_patience = 300

    for it in range(50000):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 10:
            break

        g, c5_val, _ = grad_topk(h, n, K, beta_w)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
            patience = 0
        else:
            patience += 1

        if patience > max_patience:
            # Try perturbation
            h = h + 0.05 * np.random.randn(n)
            h = project_feasible(h, n)
            patience = 0
            continue

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        v = momentum * v + (1 - momentum) * g / g_norm
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break
        v /= v_norm

        # Backtracking line search
        improved = False
        for mult in [1.0, 0.7, 0.5, 0.3, 0.15, 0.08]:
            step = lr * mult
            h_try = project_feasible(h - step * v, n)
            c_try = compute_c5(h_try)
            if c_try < best_c5 - 1e-12:
                h = h_try
                best_c5 = c_try
                best_h = h.copy()
                improved = True
                break
        if not improved:
            lr *= 0.85
            if lr < 1e-10:
                break
        else:
            lr = min(lr / 0.95, 0.1)

    return best_h, best_c5


def binary_local_search(h, n, time_limit, max_rounds=500):
    start = time.time()
    h_bin = (h > 0.5).astype(float)
    ones_count = int(h_bin.sum())
    target_ones = n // 2
    if ones_count > target_ones:
        idx_one = np.where(h_bin == 1)[0]
        rng = np.random.RandomState(123)
        rng.shuffle(idx_one)
        h_bin[idx_one[:ones_count - target_ones]] = 0
    elif ones_count < target_ones:
        idx_zero = np.where(h_bin == 0)[0]
        rng = np.random.RandomState(123)
        rng.shuffle(idx_zero)
        h_bin[idx_zero[:target_ones - ones_count]] = 1
    h_bin = project_feasible(h_bin, n)
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    rng = np.random.RandomState(42)
    n_one = np.where(h_bin == 1)[0]
    n_zero = np.where(h_bin == 0)[0]

    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 10:
            break
        improved = False
        n_try = min(400, max(50, len(n_one) * len(n_zero) // 100))
        for _ in range(n_try):
            if len(n_one) == 0 or len(n_zero) == 0:
                break
            i1 = n_one[rng.randint(len(n_one))]
            i0 = n_zero[rng.randint(len(n_zero))]
            h_try = h_bin.copy()
            h_try[i1] = 0.0
            h_try[i0] = 1.0
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try)
            if c_try < best_c5 - 1e-13:
                h_bin = h_try
                best_c5 = c_try
                best_h = h_try.copy()
                n_one = np.where(h_bin == 1)[0]
                n_zero = np.where(h_bin == 0)[0]
                improved = True
                break
        if not improved and rnd > 80:
            break
    return best_h, best_c5


def smooth_max_optimize(h, n, beta, maxiter, time_limit):
    dx = 2.0 / n
    start = time.time()

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = corr.max()
        s = np.clip(beta * (corr - m), -300.0, 300.0)
        e = np.exp(s)
        return (m + np.log(e.sum()) / beta) * dx

    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = corr.max()
        s = np.clip(beta * (corr - m), -300.0, 300.0)
        e = np.exp(s)
        w = e / e.sum()
        g = np.zeros(n)
        for k_idx in range(2 * n - 1):
            wk = w[k_idx] * dx
            if wk < 1e-14:
                continue
            off = k_idx - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:])
                g[ao:] -= wk * x[:np_]
        return g

    def eq_fun(x):
        return x.sum() - n / 2.0

    def eq_jac(x):
        return np.ones(n)

    try:
        res = minimize(obj, h, jac=jac, method='trust-constr',
                       bounds=Bounds(np.zeros(n), np.ones(n)),
                       constraints=[LinearConstraint(np.ones((1, n)), n / 2.0, n / 2.0)],
                       options={'maxiter': maxiter, 'gtol': 1e-8, 'xtol': 1e-10})
        if time.time() - start > time_limit:
            pass
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            return h_new, compute_c5(h_new)
    except Exception:
        pass
    return h, compute_c5(h)


def thue_morse_init(n, seed):
    rng = np.random.RandomState(seed)
    # Generate Thue-Morse sequence
    tm = np.zeros(n)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    # Shuffle to break periodicity
    perm = rng.permutation(n)
    tm = tm[perm]
    # Make it have exactly n/2 ones
    ones = np.where(tm == 1.0)[0]
    zeros = np.where(tm == 0.0)[0]
    if len(ones) > n // 2:
        swap = ones[len(ones) - (len(ones) - n // 2):]
        tm[swap] = 0.0
    elif len(ones) < n // 2:
        swap = zeros[:n // 2 - len(ones)]
        tm[swap] = 1.0
    # Add small noise to make it non-binary
    h = tm + 0.1 * rng.randn(n)
    return project_feasible(h, n)


def random_init(n, seed):
    rng = np.random.RandomState(seed)
    h = rng.rand(n)
    # Ensure exactly n/2 sum: use water-filling
    h = np.sort(h)
    target = n / 2.0
    # Threshold to get right count
    thresh = h[n // 2]
    h = (h > thresh).astype(float)
    # Add noise
    h = h + 0.2 * rng.randn(n)
    return project_feasible(h, n)


def interpolated_init(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def multi_phase_optimize(h, n, time_limit):
    best_c5 = compute_c5(h)
    best_h = h.copy()

    # Phase A: Momentum descent with varying K and beta
    phases = [
        (3, 20.0, 0.05, 0.8, 0.25),
        (5, 50.0, 0.04, 0.75, 0.2),
        (8, 100.0, 0.03, 0.7, 0.2),
        (5, 200.0, 0.02, 0.8, 0.15),
        (3, 500.0, 0.015, 0.85, 0.15),
    ]
    for K, beta, lr, mom, frac in phases:
        if time_limit <= 5:
            break
        t_alloc = min(time_limit * frac, 120.0)
        h_new, c_new = momentum_descent(best_h, n, t_alloc, K=K, beta_w=beta, lr=lr, momentum=mom)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new
        time_limit -= t_alloc

    # Phase B: Binary projection + local search
    if time_limit > 20:
        t_alloc = min(60.0, time_limit * 0.3)
        h_bin, c_bin = binary_local_search(best_h, n, t_alloc)
        if c_bin < best_c5:
            best_c5 = c_bin
            best_h = h_bin.copy()
        time_limit -= t_alloc

    # Phase C: Smooth max optimization
    if time_limit > 10:
        for beta in [5000, 20000, 80000]:
            if time_limit < 8:
                break
            t_alloc = min(40.0, time_limit * 0.3)
            h_s, c_s = smooth_max_optimize(best_h, n, beta, maxiter=200, time_limit=t_alloc)
            if c_s < best_c5:
                best_c5 = c_s
                best_h = h_s
            time_limit -= t_alloc

    # Phase D: Fine-tune with very focused descent
    if time_limit > 10:
        t_alloc = min(60.0, time_limit * 0.5)
        h_f, c_f = momentum_descent(best_h, n, t_alloc, K=3, beta_w=300.0, lr=0.01, momentum=0.9)
        if c_f < best_c5:
            best_c5 = c_f
            best_h = h_f
        time_limit -= t_alloc

    return best_h, best_c5
