def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        g[offset:n] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:n]
    else:
        a = -offset
        npairs = n - a
        g[:npairs] += (1.0 - h[a:n])
        g[a:n] -= h[:npairs]
    return g


def top_k_subgradient(h, n, K=20, beta=50.0):
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    weights = np.exp(beta * (c_top - c_top.max()))
    s = weights.sum()
    if s < 1e-30:
        weights = np.ones(K) / K
    else:
        weights /= s
    g = np.zeros(n)
    for i, k in enumerate(idx):
        if weights[i] > 1e-12:
            g += weights[i] * grad_corr_k(h, k, n)
    return g, float(np.max(corr))


def smooth_max_and_grad(h, n, beta):

    obj = (m + np.log(np.sum(ex)) / beta) * (2.0 / n)

    return obj, g

def cutting_plane_lp(h, n, K=30, delta=0.05):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    idx = np.argsort(-corr)[:K]

    c_obj[-1] = 1.0

    b_ub_list = []
    for k in idx:
        g = grad_corr_k(h, k, n)
        fk = corr[k]
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_ub_list.append(row)
        b_ub_list.append(-fk + np.dot(g, h))
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    I = np.eye(n)
    A_trust = np.vstack([np.hstack([I, np.zeros((n, 1))]),
                         np.hstack([-I, np.zeros((n, 1))])])
    b_trust = np.concatenate([h + delta, -h + delta])
    A_ub = np.vstack([A_ub, A_trust])
    b_ub = np.concatenate([b_ub, b_trust])
    bounds = [(0, 1)] * n + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return h.copy(), c_max

    c5_main = compute_c5(h_main)
    if c5_main < best_c5:
        best_c5 = c5_main
        best_h = h_main.copy()
        best_n = n_main

    h = best_h.copy()
    n = best_n

    # Phase 2: Adaptive subgradient descent (~120s)
    if remaining_time() > 150:
        h_sg, c5_sg = subgradient_descent_adaptive(h, n, min(120.0, remaining_time() * 0.35), rng)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 3: L-BFGS-B with smooth max (~200s)
    for beta in [300, 1000, 3000, 8000]:
        if remaining_time() < 40:
            break
        for penalty in [500, 2000]:
            if remaining_time() < 25:
                break
            h_new = lbfgsb_smooth(h, n, beta, penalty, maxiter=2000)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 4: Equalize shifts (~60s)
    if remaining_time() > 80:
        h_eq, c5_eq = equalize_shifts(h, n, K=10, max_iters=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 5: Fine coordinate descent (~200s)
    if remaining_time() > 220:
        h_cd, c5_cd = fine_coordinate_descent(h, n, min(200.0, remaining_time() * 0.35), n_grid=9)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h = h_cd.copy()

    # Phase 6: Pair moves (~60s)
    if remaining_time() > 80:
        h_pm, c5_pm = pair_move_descent(h, n, min(60.0, remaining_time() * 0.25), max_pairs=80)
        if c5_pm < best_c5:
            best_c5 = c5_pm
            best_h = h_pm.copy()
            h = h_pm.copy()

    # Phase 7: Targeted descent (~60s)
    if remaining_time() > 70:
        h_td, c5_td = targeted_descent(h, n, min(60.0, remaining_time() * 0.2))
        if c5_td < best_c5:
            best_c5 = c5_td
            best_h = h_td.copy()
            h = h_td.copy()

    # Phase 8: SLSQP polish with high beta
    for beta in [5000, 15000, 30000]:
        if remaining_time() < 25:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 9: Fine coordinate descent again (tighter grid)
    if remaining_time() > 120:
        h_cd2, c5_cd2 = fine_coordinate_descent(h, n, min(100.0, remaining_time() * 0.35), n_grid=21)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()
            h = h_cd2.copy()

    # Phase 10: Try n=2048 if time permits
    if remaining_time() > 200 and best_n < 2048:
        n_vh = 2048
        h_vh = interp_to(best_h, best_n, n_vh)
        h_vh = project_feasible(h_vh, n_vh)
        c5_vh = compute_c5(h_vh)
        if c5_vh <= best_c5 + 0.001:
            # Subgradient at higher res
            if remaining_time() > 100:
                h_vh, c5_vh = subgradient_descent_adaptive(h_vh, n_vh, min(60.0, remaining_time() * 0.25), rng)
            # L-BFGS-B
            if remaining_time() > 60:
                h_vh = lbfgsb_smooth(h_vh, n_vh, 3000, 2000, maxiter=1500)
                c5_vh = compute_c5(h_vh)
            # Fine coord descent
            if remaining_time() > 60:
                h_vh, c5_vh = fine_coordinate_descent(h_vh, n_vh, min(50.0, remaining_time() * 0.3), n_grid=7)
            # Equalize
            if remaining_time() > 30:
                h_vh, c5_vh = equalize_shifts(h_vh, n_vh, K=8, max_iters=100)
            if c5_vh < best_c5:
                best_c5 = c5_vh
                best_h = h_vh.copy()
                best_n = n_vh

    # Phase 11: Final targeted descent + fine polish
    if remaining_time() > 40:
        h_td2, c5_td2 = targeted_descent(best_h, best_n, min(30.0, remaining_time() * 0.5))
        if c5_td2 < best_c5:
            best_c5 = c5_td2
            best_h = h_td2.copy()

    if remaining_time() > 30:
        h_cd3, c5_cd3 = fine_coordinate_descent(best_h, best_n, min(25.0, remaining_time() * 0.5), n_grid=15)
        if c5_cd3 < best_c5:
            best_c5 = c5_cd3
            best_h = h_cd3.copy()

    # Final SLSQP
    if remaining_time() > 20:
        h_new = slsqp_smooth(best_h, best_n, 50000, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
