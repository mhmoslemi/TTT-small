    v_hat = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    """Returns (objective_value, gradient) for the smooth max objective."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    z = np.clip(beta * (corr - m), -50.0, 50.0)

        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    obj = (m + np.log(S) / beta) * dx
    g = np.zeros(n)
    wdx = w * dx
    for k in range(2 * n - 1):
        wk = wdx[k]
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - h[a:])
            g[a:] -= wk * h[:np_]
    return obj, g


def adam_run(h0, n, beta_start, beta_end, num_iters, lr=0.005, check_interval=200):
    h = project_feasible(h0, n)
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    best_h = h.copy()
    best_c5 = compute_c5(h)
    beta_ratio = beta_end / beta_start
    for t in range(1, num_iters + 1):
        if t % 2000 == 0 and remaining_time() < 30:
            break
        frac = (t - 1) / max(1, num_iters - 1)
        beta = beta_start * (beta_ratio ** frac)
        obj, g = smooth_max_grad(h, n, beta)
        m1 = b1 * m1 + (1.0 - b1) * g
        m2 = b2 * m2 + (1.0 - b2) * g * g
        m1h = m1 / (1.0 - b1 ** t)
        m2h = m2 / (1.0 - b2 ** t)
        lr_t = lr * np.sqrt(1.0 - b2 ** t) / (1.0 - b1 ** t)
        h_new = h - lr_t * m1h / (np.sqrt(m2h) + eps)
        h = project_feasible(h_new, n)
        if t % check_interval == 0:
            c5_t = compute_c5(h)
            if c5_t < best_c5 - 1e-14:
                best_c5 = c5_t
                best_h = h.copy()
    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5


def topk_push(h, n, top_k=5, max_iters=600):
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = 0.03
    for it in range(max_iters):
        if remaining_time() < 3:
            break
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        for ki in idx_top:
            off = ki - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:] += (1.0 - h[:np_])
                g[:np_] -= h[off:]
            else:
                a = -off
                np_ = n - a
                g[:np_] += (1.0 - h[a:])
                g[a:] -= h[:np_]
        g *= dx / top_k
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        d = -g / gn
        improved = False
        steps = [lr, lr * 0.5, lr * 2.0, lr * 0.25, lr * 0.75]
        for step in steps:
            ht = project_feasible(h + step * d * gn, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-13:
                h = ht
                best_c5 = cn
                best_h = h.copy()
                improved = True
                break
        if not improved:
            for step in [lr * 0.5, lr * 0.25, lr * 0.1]:
                ht = project_feasible(h - step * d * gn, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-13:
                    h = ht
                    best_c5 = cn
                    best_h = h.copy()
                    improved = True
                    break
        if not improved:
            lr *= 0.95
            if lr < 1e-7:
                break
    return best_h, best_c5


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def _sm_obj(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    z = np.clip(_SM_BETA * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    return (m + np.log(np.sum(ez)) / _SM_BETA) * dx


def _sm_jac(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    z = np.clip(_SM_BETA * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    w = ez / np.sum(ez)
    g = np.zeros(n)
    wdx = w * dx
    for k in range(2 * n - 1):
        wk = wdx[k]
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    return g


def _eq_fun(x):
    return np.sum(x) - _CUR_N / 2.0


def _eq_jac(x):
    return np.ones(_CUR_N)


def slsqp_refine(h, n, beta, maxiter=400):
    global _CUR_N, _SM_BETA
    _CUR_N = n
    _SM_BETA = beta
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_sm_obj, h0, jac=_sm_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        obj_final = _sm_obj(res.x)
        obj_init = _sm_obj(h0)
        if obj_final < obj_init - 1e-14:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def sinusoidal_perturb(h, n, rng, amp):
    h_p = h.copy()
    freq = rng.uniform(1.0, 5.0)
    phase = rng.uniform(0.0, 2.0 * np.pi)
    x = np.linspace(0.0, 2.0, n)
    h_p += amp * np.sin(2.0 * np.pi * freq * x + phase)
    return project_feasible(h_p, n)


def morse_perturb(h, n, rng, amp):
    h_p = h.copy()
    v = np.zeros(n)
    for i in range(n):
        bit = i
        cnt = 0
        while bit:
            cnt += bit & 1
            bit >>= 1
        v[i] = 1.0 if cnt % 2 == 1 else -1.0
    h_p += amp * v * 0.33
    return project_feasible(h_p, n)


def binary_sa(h, n, time_limit, rng):
    h = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        sort_idx = np.argsort(-h)
        h = np.zeros(n)
        h[sort_idx[:n_half]] = 1.0
    start = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    current_c5 = best_c5
    T = 0.5
    cool_rate = 0.9997
    no_improve = 0
    max_no_improve = 5000
    while time.time() - start < time_limit and remaining_time() > 10:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            current_c5 = new_c5
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
        T *= cool_rate
        if no_improve > max_no_improve:
            break
    return best_h, best_c5


def multistart_optimize(h_base, n, rng, num_restarts, iters_each, beta_start, beta_end):
    best_h = h_base.copy()
    best_c5 = compute_c5(h_base)
    for r in range(num_restarts):
        if remaining_time() < 30:
            break
        amp = 0.03 + 0.04 * rng.rand()
        if r % 3 == 0:
            h_start = sinusoidal_perturb(h_base, n, rng, amp)
        elif r % 3 == 1:
            h_start = morse_perturb(h_base, n, rng, amp)
        else:
            noise = rng.randn(n) * amp * 0.5
            h_start = project_feasible(h_base + noise, n)
        h_r, c5_r = adam_run(h_start, n, beta_start, beta_end, iters_each, lr=0.005, check_interval=200)
        h_r, c5_r = topk_push(h_r, n, top_k=3, max_iters=200)
        if c5_r < best_c5 - 1e-14:
            best_c5 = c5_r
            best_h = h_r.copy()
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _CUR_N, _SM_BETA
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    try:
        best_h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        best_h = np.zeros(640)
        best_h[:320] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: ADAM with beta annealing at native resolution
    h = best_h.copy()
    h, c5 = adam_run(h, best_n, 50.0, 20000.0, 5000, lr=0.006, check_interval=100)
    if c5 < best_c5 - 1e-14:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: Top-k equalization
    if remaining_time() > 20:
        h, c5 = topk_push(best_h, best_n, top_k=5, max_iters=600)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()

    # Phase 3: Higher-beta ADAM for fine-tuning
    if remaining_time() > 100:
        h, c5 = adam_run(best_h, best_n, 5000.0, 200000.0, 4000, lr=0.003, check_interval=100)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()

    # Phase 4: Top-k again with more iterations
    if remaining_time() > 20:
        h, c5 = topk_push(best_h, best_n, top_k=7, max_iters=500)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()

    # Phase 5: Multi-resolution search
    resolutions = [960, 512, 1024, 768, 896]
    resolutions = [r for r in resolutions if r % 2 == 0 and 256 <= r <= 2048]
    resolutions = sorted(set(resolutions), key=lambda x: abs(x - best_n))

    for n_t in resolutions[:4]:
        if remaining_time() < 80:
            break
        h_int = project_feasible(interp_to(best_h, best_n, n_t), n_t)
        h_t, c5_t = adam_run(h_int, n_t, 200.0, 50000.0, 3000, lr=0.005, check_interval=200)
        h_t, c5_t = topk_push(h_t, n_t, top_k=5, max_iters=400)
        if c5_t < best_c5 - 1e-14:
            best_c5 = c5_t
            best_h = h_t.copy()
            best_n = n_t
            if remaining_time() > 40:
                h_t2, c5_t2 = adam_run(best_h, best_n, 10000.0, 300000.0, 2000, lr=0.002, check_interval=200)
                if c5_t2 < best_c5 - 1e-14:
                    best_c5 = c5_t2
                    best_h = h_t2.copy()

    # Phase 6: Multi-start perturbation search
    if remaining_time() > 120:
        h_ms, c5_ms = multistart_optimize(best_h, best_n, rng, 6, 2000, 500.0, 80000.0)
        if c5_ms < best_c5 - 1e-14:
            best_c5 = c5_ms
            best_h = h_ms.copy()

    # Phase 7: SLSQP polish at high beta
    for beta in [200000, 800000]:
        if remaining_time() < 25:
            break
        h_s = slsqp_refine(best_h, best_n, beta, maxiter=400)
        c5_s = compute_c5(h_s)
        if c5_s < best_c5 - 1e-14:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 8: Binary SA + continuous refinement
    if remaining_time() > 100:
        h_bin = (best_h > 0.5).astype(float)
        n_half = best_n // 2
        n_ones = int(np.sum(h_bin))
        if n_ones != n_half:
            sort_idx = np.argsort(-h_bin)
            h_bin = np.zeros(best_n)
            h_bin[sort_idx[:n_half]] = 1.0
        sa_time = min(80, max(15, int(remaining_time() * 0.15)))
        h_b, c5_b = binary_sa(h_bin, best_n, sa_time, rng)

        h_b_c = project_feasible(h_b, best_n)
        h_b_c, c5_bc = adam_run(h_b_c, best_n, 1000.0, 100000.0, 2000, lr=0.004, check_interval=200)
        h_b_c, c5_bc = topk_push(h_b_c, best_n, top_k=5, max_iters=300)
        if c5_bc < best_c5 - 1e-14:
            best_c5 = c5_bc
            best_h = h_b_c.copy()
        if c5_b < best_c5 - 1e-14:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 9: Another round of ADAM + top-k on best
    if remaining_time() > 60:
        h, c5 = adam_run(best_h, best_n, 20000.0, 500000.0, 3000, lr=0.002, check_interval=100)
        h, c5 = topk_push(h, best_n, top_k=5, max_iters=400)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()

    # Phase 10: Final SLSQP
    if remaining_time() > 15:
        h_f = slsqp_refine(best_h, best_n, 1000000, maxiter=300)
        c5_f = compute_c5(h_f)
        if c5_f < best_c5 - 1e-14:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Ensure feasibility and final evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
