def active_set_lp(h, n, K, delta, time_limit=None):
    """Solve LP with active set constraints."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(-corr)[:K]
    
    # Minimize t subject to:
    # grad_k * h_new <= t - (corr_k - grad_k * h) for each k in active set
    # Actually: corr_k(h_new) ≈ corr_k(h) + grad_k * (h_new - h) <= t
    # So: grad_k * h_new <= t - corr_k(h) + grad_k * h
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = []
    b_ub = []
    for k in idx:
        g = grad_corr_k(h, k, n)
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_ub.append(row)
        b_ub.append(-corr[k])
    
    # Trust region
    I = np.eye(n)
    A_trust = np.vstack([np.hstack([I, np.zeros((n, 1))]),
                         np.hstack([-I, np.zeros((n, 1))])])
    b_trust = np.concatenate([h + delta, h + delta])
    
    A_ub = np.vstack([np.array(A_ub), A_trust])
    b_ub = np.concatenate([np.array(b_ub), b_trust])
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0, 1)] * n + [(None, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return h.copy(), np.max(corr)


def smooth_max_optimize(h, n, beta, K, maxiter):
    """Smooth max optimization using SLSQP."""
    global _G_N, _G_BETA, _G_K
    _G_N = n
    _G_BETA = beta
    _G_K = K
    
    def obj(h):

        m = corr.max()

        c = corr[idx]
        sh = c - c.max()
        cl = np.clip(beta * sh, -80.0, 80.0)
        ex = np.exp(cl)
        return (c.max() + np.log(ex.sum()) / beta) * (2.0 / n)
    
    def grad(h):

        idx = np.argsort(-corr)[:K]
        c = corr[idx]
        sh = c - c.max()
        cl = np.clip(beta * sh, -80.0, 80.0)
        ex = np.exp(cl)
        w = ex / ex.sum()
        g = np.zeros(n)
        dx = 2.0 / n
        for i in range(K):

                continue

                g[:np_] -= wk * h[offset:]
            else:
                a = -offset
                np_ = n - a
                g[:np_] += wk * (1.0 - h[a:])
                g[a:] -= wk * h[:np_]
        return g
    
    bounds = [(0.0, 1.0)] * n
    
    def sum_fun(h):
        return np.sum(h) - n / 2.0
    
    def sum_jac(h):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:

            if compute_c5(h_new) <= compute_c5(h) + 1e-12:

def subgradient_descent(h, n, time_limit, rng):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    it = 0
    vel = np.zeros(n)
    while time.time() - start < time_limit:
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        progress = min(1.0, it / 300.0)
        K = int(15 + 25 * (1.0 - progress))
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(-corr)[:K]
        g = np.zeros(n)
        for k in idx:
            g += grad_corr_k(h, k, n)
        g /= K
        step = 0.08 * n / (np.linalg.norm(g) + 1e-10) / (1.0 + 0.02 * it)
        step = min(step, 0.12)
        vel = 0.7 * vel + 0.3 * g
        h_new = project_feasible(h - step * vel, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            for f in [0.5, 0.25, 0.1]:
                h_try = project_feasible(h - step * f * vel, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    break
        it += 1
    return best_h, best_c5


def _smmax_obj(h):
    n = _G_N
    corr = np.correlate(h, 1.0 - h, mode='full')
    beta = _G_BETA
    K = _G_K
    m = corr.max()
    idx = np.argsort(-corr)[:K]
    c = corr[idx]
    sh = c - c.max()
    cl = np.clip(beta * sh, -80.0, 80.0)
    ex = np.exp(cl)
    return (c.max() + np.log(ex.sum()) / beta) * (2.0 / n)


def _smmax_grad(h):
    n = _G_N
    beta = _G_BETA
    K = _G_K
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(-corr)[:K]
    c = corr[idx]
    sh = c - c.max()
    cl = np.clip(beta * sh, -80.0, 80.0)
    ex = np.exp(cl)
    w = ex / ex.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for i in range(K):
        if w[i] < 1e-15:
            continue
        wk = w[i] * dx
        k = idx[i]
        offset = k - (n - 1)
        if offset >= 0:
            np_ = n - offset
            g[offset:] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[offset:]
        else:
            a = -offset
            np_ = n - a
            g[:np_] += wk * (1.0 - h[a:])
            g[a:] -= wk * h[:np_]
    return g


def _sum_eq(h):
    return np.sum(h) - _G_N / 2.0


def _sum_eq_jac(h):
    return np.ones(_G_N)


def slsqop_smmax(h, n, beta, K, maxiter):
    global _G_N, _G_BETA, _G_K
    _G_N = n
    _G_BETA = beta
    _G_K = K
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac}]
    try:
        res = minimize(_smmax_obj, h, jac=_smmax_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            if compute_c5(h_new) <= compute_c5(h) + 1e-12:
                return h_new
    except Exception:
        pass
    return h.copy()


def _tmin_ineq_fun(v):
    h = v[:_G_N]
    t = v[-1]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return t * (_G_N / 2.0) - corr[_G_KLIST]


def _tmin_ineq_jac(v):
    h = v[:_G_N]
    n = _G_N
    K_list = _G_KLIST
    J = np.zeros((len(K_list), n + 1))
    for i, k in enumerate(K_list):
        offset = k - (n - 1)
        if offset >= 0:
            np_ = n - offset
            J[i, offset:] -= (1.0 - h[:np_])
            J[i, :np_] += h[offset:]
        else:
            a = -offset
            np_ = n - a
            J[i, :np_ -= 0] += 0
            J[i, :n] += 0
            J[i, :np_] -= (1.0 - h[a:])
            J[i, a:] += h[:np_]
        J[i, -1] = n / 2.0
    return J


def _tmin_obj(v):
    return v[-1]


def _tmin_obj_jac(v):
    g = np.zeros(_G_N + 1)
    g[-1] = 1.0
    return g


def _tmin_sum_eq(v):
    return np.sum(v[:_G_N]) - _G_N / 2.0


def _tmin_sum_eq_jac(v):
    g = np.zeros(_G_N + 1)
    g[:_G_N] = 1.0
    return g


def slsqop_tminmax(h, n, K_list, maxiter):
    global _G_N, _G_KLIST
    _G_N = n
    _G_KLIST = K_list
    t0 = np.max(np.correlate(h, 1.0 - h, mode='full')) / (n / 2.0)
    v0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'ineq', 'fun': _tmin_ineq_fun, 'jac': _tmin_ineq_jac},
        {'type': 'eq', 'fun': _tmin_sum_eq, 'jac': _tmin_sum_eq_jac}
    ]
    try:
        res = minimize(_tmin_obj, v0, jac=_tmin_obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x[:n], n)
            c5_new = compute_c5(h_new)
            if c5_new <= compute_c5(h) + 1e-12:
                return h_new, c5_new
    except Exception:
        pass
    return h.copy(), compute_c5(h)


def benders_lp(h, n, K, delta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(-corr)[:K]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub_list = []
    b_ub_list = []
    for k in idx:
        g = grad_corr_k(h, k, n)
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_ub_list.append(row)
        b_ub_list.append(-corr[k])
    I = np.eye(n)
    A_trust = np.vstack([np.hstack([I, np.zeros((n, 1))]),
                         np.hstack([-I, np.zeros((n, 1))])])
    b_trust = np.concatenate([h + delta, h + delta])
    A_ub = np.vstack([np.array(A_ub_list), A_trust])
    b_ub = np.concatenate([np.array(b_ub_list), b_trust])
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0, 1)] * n + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return h.copy(), np.max(corr)


def equalize_top(h, n, K, max_iters, time_limit):
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(-corr)[:K]
        if corr[idx[0]] - corr[idx[-1]] < 1e-12:
            break
        g_avg = np.zeros(n)
        for k in idx:
            g_avg += grad_corr_k(h, k, n)
        g_avg /= K
        g_dir = -g_avg
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        g_dir /= gn
        improved = False
        for step in [0.01, 0.02, 0.005, 0.04, 0.003, 0.015, 0.008, 0.03]:
            ht = project_feasible(h + step * g_dir, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-15:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def fine_polish(h, n, time_limit):
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    for it in range(5000):
        if time.time() - start > time_limit:
            break
        rem = _BUDGET - (time.time() - _T0)
        if rem < 3:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(-corr)[:20]
        g = np.zeros(n)
        for k in idx:
            g += grad_corr_k(h, k, n)
        g /= len(idx)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        for step in [0.003, 0.001, 0.002, 0.0005, 0.005, 0.0015]:
            h_try = project_feasible(h - step * g / gn, n)
            c_try = compute_c5(h_try)
            if c_try < best_c5:
                best_c5 = c_try
                h = h_try.copy()
                break
    return h, best_c5


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _G_N, _G_BETA, _G_K, _G_KLIST
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 512
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: Quick subgradient to find basin
    if remaining() > 20:
        h_sg, c5_sg = subgradient_descent(h, n, min(20.0, remaining() * 0.08), rng)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 2: SLSQP smooth-max
    if remaining() > 60:
        for beta in [50, 200, 1000, 5000]:
            if remaining() < 10:
                break
            h = slsqop_smmax(h, n, beta, K=40, maxiter=400)
            c5_tmp = compute_c5(h)
            if c5_tmp < best_c5:
                best_c5 = c5_tmp
                best_h = h.copy()

    # Phase 3: Coordinate descent (main improvement)
    if remaining() > 120:
        h_cd, c5_cd = fast_coord_descent(h, n, min(120.0, remaining() * 0.4), K=40)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h = h_cd.copy()

    # Phase 4: Active-set SLSQP
    if remaining() > 60:
        corr_a = np.correlate(h, 1.0 - h, mode='full')
        for K_a in [8, 15, 25, 40]:
            if remaining() < 10:
                break
            idx_a = np.argsort(-np.correlate(h, 1.0 - h, mode='full'))[:K_a]
            h_am, c5_am = slsqop_tminmax(h, n, idx_a, maxiter=300)
            if c5_am < best_c5:
                best_c5 = c5_am
                best_h = h_am.copy()
                h = h_am.copy()

    # Phase 5: Benders LP
    if remaining() > 80:
        for it in range(300):
            if remaining() < 5:
                break
            h_lp, t_lp = benders_lp(h, n, K=25, delta=0.03)
            c5_cur = compute_c5(h)
            if t_lp < c5_cur * (n / 2.0) / (n / 2.0) - 1e-10:
                # LP found a lower max; move toward it
                best_val = c5_cur
                for alpha in [1.0, 0.7, 0.5, 0.3, 0.8]:
                    h_try = project_feasible(h + alpha * (h_lp - h), n)
                    c_try = compute_c5(h_try)
                    if c_try < best_val:
                        best_val = c_try
                        h = h_try.copy()
                    else:
                        break
                c5_now = compute_c5(h)
                if c5_now < best_c5:
                    best_c5 = c5_now
                    best_h = h.copy()
            elif it > 150:
                break

    # Phase 6: More coordinate descent after LP
    if remaining() > 80:
        h_cd2, c5_cd2 = fast_coord_descent(h, n, min(80.0, remaining() * 0.3), K=50)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()
            h = h_cd2.copy()

    # Phase 7: Equalize top shifts
    if remaining() > 30:
        h_eq, c5_eq = equalize_top(h, n, K=10, max_iters=300,
                                   time_limit=min(30.0, remaining() * 0.4))
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 8: Try higher resolution
    n_hi = 1024
    if remaining() > 200:
        h_hi = interp_to(best_h, best_n, n_hi)
        h_hi = project_feasible(h_hi, n_hi)
        c5_hi = compute_c5(h_hi)
        if c5_hi <= best_c5 + 0.002:
            h_work = h_hi.copy()
            # Subgradient at high res
            if remaining() > 150:
                h_work, c5_w = subgradient_descent(h_work, n_hi, min(20.0, remaining() * 0.1), rng)
            # SLSQP at high res
            for beta in [100, 1000, 5000]:
                if remaining() < 20:
                    break
                h_work = slsqop_smmax(h_work, n_hi, beta, K=40, maxiter=300)
            # Coord descent at high res
            if remaining() > 80:
                h_work, c5_w = fast_coord_descent(h_work, n_hi, min(80.0, remaining() * 0.3), K=40)
            # Equalize at high res
            if remaining() > 20:
                h_work, c5_w = equalize_top(h_work, n_hi, K=10, max_iters=200,
                                            time_limit=min(20.0, remaining() * 0.3))
            c5_final_hi = compute_c5(h_work)
            if c5_final_hi < best_c5:
                best_c5 = c5_final_hi
                best_h = h_work.copy()
                best_n = n_hi

    # Phase 9: Final coordinate descent
    n = best_n
    h = best_h.copy()
    if remaining() > 40:
        h_cd3, c5_cd3 = fast_coord_descent(h, n, min(40.0, remaining() * 0.4), K=50)
        if c5_cd3 < best_c5:
            best_c5 = c5_cd3
            best_h = h_cd3.copy()

    # Phase 10: Fine polish
    if remaining() > 20:
        h = best_h.copy()
        h_fp, c5_fp = fine_polish(h, best_n, min(20.0, remaining() * 0.6))
        if c5_fp < best_c5:
            best_c5 = c5_fp
            best_h = h_fp.copy()

    # Phase 11: Final SLSQP polish
    if remaining() > 15:
        h = best_h.copy()
        for beta in [20000, 50000]:
            if remaining() < 8:
                break
            h = slsqop_smmax(h, best_n, beta, K=60, maxiter=500)
        c5_f = compute_c5(h)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h.copy()

    # Final checks
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
