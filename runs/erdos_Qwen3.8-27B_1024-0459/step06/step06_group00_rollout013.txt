def projected_gradient_lbfgs(h, n, beta, maxiter=5000, time_limit=120):

    _SM_BETA = beta

    dx = 2.0 / n

            break

        s = x - x_prev
        y = sm_grad(x) - g
        sy = np.dot(s, y)

        else:

                    best_c5 = cn

                break
        else:
            x = project_feasible(x - 0.001 * y, n)
        
        x_prev = x.copy()
        g = y
    
    return best_h, best_c5

def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def binary_sa(h, n, time_limit=120, rng=None):

def active_set_newton(h, n, top_k=8, max_iters=500, time_limit=60):
    """Newton-like method for equalizing top-k correlations"""
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    start = time.time()
    
    for it in range(max_iters):
        if time.time() - start > time_limit or remaining_time() < 5:
            break
        
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        c_vals = corr[idx_top]
        c_target = np.mean(c_vals)
        
        # Residuals: corr[k_i] - c_target for i in top_k
        residuals = corr[idx_top] - c_target
        if np.max(np.abs(residuals)) < 1e-12:
            break
        
        # Jacobian: d(corr[k_i])/d(h_j) for i in top_k, j in [0,n)
        # This is a top_k x n matrix
        # J[i,j] = d(corr[k_i])/d(h_j) = grad_single_shift(h, k_i, n)[j]

        # with the constraint sum(delta_h) = 0 (to preserve integral)
        # and ||delta_h|| minimized
        # This is a least squares with constraint sum=0
        # Use: minimize ||J @ d + r||² s.t. sum(d) = 0

        JJT = J @ J.T  # top_k x top_k
        # d = -J^T @ (JJT^{-1} @ r)

            alpha = np.linalg.solve(JJT_reg, residuals)
            d = -J.T @ alpha
            # Project to sum=0
            d -= np.sum(d) / n
            d_norm = np.linalg.norm(d)
            if d_norm < 1e-14:
                break
            d /= d_norm
        except np.linalg.LinAlgError:
            break

        for step in [0.001, 0.003, 0.01, 0.03, 0.1, 0.3, 0.0003, 0.001, 0.01]:

        corr = compute_corr(h)
        k_curr = int(np.argmax(corr))
        recent_k.append(k_curr)
        if len(recent_k) > n_recent:
            recent_k.pop(0)

        g = np.zeros(n)
        for k in recent_k:
            g += grad_single_shift(h, k, n)
        g /= len(recent_k)

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g /= g_norm

        base_step = 0.008 / (1.0 + 0.0005 * it)
        vel = momentum * vel + (1 - momentum) * g
        v_norm = np.linalg.norm(vel)
        if v_norm > 1e-10:
            vel /= v_norm
        else:
            vel = g

        improved = False
        for step in [base_step * 0.1, base_step * 0.3, base_step, base_step * 3,
                     base_step * 10, 0.001, 0.003, 0.01, 0.03, 0.1]:
            ht = project_feasible(h - step * vel, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-15:
                h = ht
                best_c5 = cn
                best_h = h.copy()
                improved = True
                stale = 0
                break

        if not improved:
            stale += 1
            if stale > 400:
                rng = np.random.RandomState(it)
                pert = rng.randn(n) * 0.008
                ht = project_feasible(h + pert, n)
                cn = compute_c5(ht)
                if cn < best_c5:
                    h = ht
                    best_c5 = cn
                    best_h = h.copy()
                    stale = 0
                    vel = np.zeros(n)
                else:
                    if stale > 1500:
                        break
    return best_h, best_c5


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def binary_sa(h, n, time_limit=120, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h_bin = (h > 0.5).astype(float).copy()
    n_half = n // 2
    if int(np.sum(h_bin)) != n_half:
        sort_idx = np.argsort(-h_bin)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0

    dx = 2.0 / n
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    curr_c5 = best_c5
    start = time.time()

    T_start = 15.0 / dx
    T_end = 0.00005 / dx
    total_iters = 60000
    min_T_ratio = T_end / T_start

    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    no_improve = 0

    for it in range(total_iters):
        if time.time() - start > time_limit or remaining_time() < 5:
            break

        T = T_start * (min_T_ratio ** (it / max(1, total_iters)))

        i_idx = rng.randint(len(ones))
        j_idx = rng.randint(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]

        h_bin[i] = 0.0
        h_bin[j] = 1.0
        new_c5 = compute_c5(h_bin)
        delta = new_c5 - curr_c5

        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / max(T, 1e-10))):
            ones[i_idx] = j
            zeros[j_idx] = i
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_bin.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0

        if no_improve > 12000:
            break

    return best_h, best_c5


def coordinate_refine(h, n, max_sweeps=40, time_limit=60):
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    start = time.time()

    for sweep in range(max_sweeps):
        if remaining_time() < 5 or time.time() - start > time_limit:
            break
        improved_sweep = False
        for i in range(n):
            old = h[i]
            if 0.05 < old < 0.95:
                cands = [old + 0.02, old - 0.02, old + 0.05, old - 0.05,
                         old + 0.1, old - 0.1, old + 0.2, old - 0.2, 0.0, 1.0]
            else:
                cands = [0.0, 1.0, old + 0.02, old - 0.02, old + 0.05, old - 0.05]
            cands = [c for c in cands if -0.01 <= c <= 1.01 and abs(c - old) > 1e-8]

            best_val = old
            best_c5_val = best_c5

            for c in cands:
                h[i] = min(max(c, 0.0), 1.0)
                h_proj = project_feasible(h, n)
                c5_v = compute_c5(h_proj)
                if c5_v < best_c5_val - 1e-15:
                    best_c5_val = c5_v
                    best_val = min(max(c, 0.0), 1.0)
                    improved_sweep = True

            h[i] = best_val
            h = project_feasible(h, n)

        if not improved_sweep:
            break
    return h, best_c5


def theta_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _CUR_N, _SM_BETA
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    # Load parent
    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = theta_morse(640)
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B at parent resolution with increasing beta
    h = best_h.copy()
    for beta in [5000, 50000, 500000, 5000000, 50000000]:
        if remaining_time() < 30:
            break
        h_new = lbfgsb_opt(h, best_n, beta, maxiter=3000, penalty=1e8)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Active-set Newton for top-k
    if remaining_time() > 30:
        for tk in [5, 8, 12, 20]:
            if remaining_time() < 15:
                break
            h_as, c5_as = active_set_newton(h, best_n, top_k=tk, max_iters=400, time_limit=40)
            if c5_as < best_c5 - 1e-15:
                best_c5 = c5_as
                best_h = h_as.copy()
                h = h_as.copy()

    # Phase 3: Top-k gradient equalization
    if remaining_time() > 30:
        for tk in [5, 10, 20, 40]:
            if remaining_time() < 20:
                break
            h_eq, c5_eq = top_k_equalize(h, best_n, top_k=tk, max_iters=2000, time_limit=45)
            if c5_eq < best_c5 - 1e-15:
                best_c5 = c5_eq
                best_h = h_eq.copy()
                h = h_eq.copy()

    # Phase 4: Subgradient with momentum
    if remaining_time() > 40:
        h_sg, c5_sg = subgradient_momentum(h, best_n, max_iters=20000, time_limit=100, n_recent=5, momentum=0.7)
        if c5_sg < best_c5 - 1e-15:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 5: SLSQP refinement at parent
    for beta in [10000000, 100000000]:
        if remaining_time() < 20:
            break
        h_new = slsqp_opt(h, best_n, beta, maxiter=2000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 6: Upsample to higher resolution
    for n_t in [1280, 1920]:
        if remaining_time() < 80:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)

        h_opt = h_feas
        for beta in [10000, 100000, 1000000, 10000000]:
            if remaining_time() < 25:
                break
            h_opt = lbfgsb_opt(h_opt, n_t, beta, maxiter=2000, penalty=1e8)
        c5_opt = compute_c5(h_opt)

        if remaining_time() > 25:
            for tk in [8, 15]:
                if remaining_time() < 10:
                    break
                h_opt, c5_opt = active_set_newton(h_opt, n_t, top_k=tk, max_iters=300, time_limit=35)
        if remaining_time() > 20:
            h_opt, c5_opt = top_k_equalize(h_opt, n_t, top_k=10, max_iters=1000, time_limit=35)
        if remaining_time() > 20:
            h_opt, c5_opt = subgradient_momentum(h_opt, n_t, max_iters=5000, time_limit=40, n_recent=4, momentum=0.7)

        if c5_opt < best_c5 - 1e-15:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t
            h = h_opt.copy()

    # Phase 7: Binary SA at current best resolution
    if remaining_time() > 80:
        sa_time = min(150, int(remaining_time() * 0.3))
        h_bin = (best_h > 0.5).astype(float)
        h_b, c5_b = binary_sa(h_bin, best_n, time_limit=sa_time, rng=rng)
        # Continuous refinement of binary result
        h_b_c = h_b.copy()
        for beta in [100000, 1000000]:
            if remaining_time() < 15:
                break
            h_b_c = lbfgsb_opt(h_b_c, best_n, beta, maxiter=1000, penalty=1e8)
        c5_b_c = compute_c5(h_b_c)
        if remaining_time() > 15:
            h_b_c, c5_b_c = top_k_equalize(h_b_c, best_n, top_k=10, max_iters=500, time_limit=30)
        if remaining_time() > 15:
            h_b_c, c5_b_c = active_set_newton(h_b_c, best_n, top_k=8, max_iters=200, time_limit=30)

        if c5_b_c < best_c5 - 1e-15:
            best_c5 = c5_b_c
            best_h = h_b_c.copy()
            h = h_b_c.copy()
        if c5_b < best_c5 - 1e-15:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 8: Coordinate refinement
    if remaining_time() > 30:
        h_cr, c5_cr = coordinate_refine(best_h, best_n, max_sweeps=30,
                                        time_limit=min(80, int(remaining_time() * 0.4)))
        if c5_cr < best_c5 - 1e-15:
            best_c5 = c5_cr
            best_h = h_cr.copy()
            h = h_cr.copy()

    # Phase 9: Final intensive optimization
    n = best_n
    h = best_h.copy()
    for beta in [50000000, 500000000]:
        if remaining_time() < 15:
            break
        h_new = lbfgsb_opt(h, n, beta, maxiter=3000, penalty=1e9)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    if remaining_time() > 20:
        for tk in [5, 10]:
            if remaining_time() < 10:
                break
            h_as, c5_as = active_set_newton(h, n, top_k=tk, max_iters=300, time_limit=30)
            if c5_as < best_c5 - 1e-15:
                best_c5 = c5_as
                best_h = h_as.copy()
                h = h_as.copy()

    if remaining_time() > 15:
        h_eq, c5_eq = top_k_equalize(h, n, top_k=12, max_iters=1000, time_limit=30)
        if c5_eq < best_c5 - 1e-15:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    if remaining_time() > 20:
        h_sg, c5_sg = subgradient_momentum(best_h, n, max_iters=8000,
                                           time_limit=min(50, int(remaining_time() * 0.5)),
                                           n_recent=3, momentum=0.8)
        if c5_sg < best_c5 - 1e-15:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Final SLSQP polish
    if remaining_time() > 15:
        h_final = slsqp_opt(best_h, n, 100000000, maxiter=2000)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-15:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = project_feasible(best_h, n)

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, n)
