```python
def compute_delta(h, i, j, n):
    """Compute delta_s for swapping h[i]:1->0, h[j]:0->1."""
    # s ranges from -(n-1) to (n-1)
    s = np.arange(-(n-1), n)
    
    # h[i+s] valid when 0 <= i+s < n
    idx_pos = i + s
    valid_pos = (idx_pos >= 0) & (idx_pos < n)
    a1 = np.zeros(2*n-1)
    a1[valid_pos] = h[idx_pos[valid_pos]]
    
    # h[i-s] valid when 0 <= i-s < n
    idx_neg = i - s
    valid_neg = (idx_neg >= 0) & (idx_neg < n)
    a2 = np.zeros(2*n-1)
    a2[valid_neg] = h[idx_neg[valid_neg]]
    
    # h[j+s]
    idx_jp = j + s
    valid_jp = (idx_jp >= 0) & (idx_jp < n)
    b1 = np.zeros(2*n-1)
    b1[valid_jp] = h[idx_jp[valid_jp]]
    
    # h[j-s]
    idx_jm = j - s
    valid_jm = (idx_jm >= 0) & (idx_jm < n)
    b2 = np.zeros(2*n-1)
    b2[valid_jm] = h[idx_jm[valid_jm]]
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def binarize(h, n):
    target = n // 2
    idx = np.argpartition(h, -target)[-target:]
    b = np.zeros(n)
    b[idx] = 1.0
    return b
```


```python
def smooth_max_val(h, n, K, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    m = c_top[0]
    sh = np.clip(beta * (c_top - m), -50.0, 50.0)
    ex = np.exp(sh)
    dx = 2.0 / n
    return (m + np.log(ex.sum()) / beta) * dx


def smooth_max_grad(h, n, K, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    m = c_top[0]
    sh = np.clip(beta * (c_top - m), -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / ex.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for i in range(K):
        wk = w[i] * dx
        if wk < 1e-15:
            continue
        k = idx[i]
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            g[offset:] += wk * (1.0 - h[:npairs])
            g[:npairs] -= wk * h[offset:]
        else:
            a = -offset
            npairs = n - a
            g[:npairs] += wk * (1.0 - h[a:])
            g[a:] -= wk * h[:npairs]
    return g


def slsqp_optimize(h, n, K, beta, maxiter=300):
    def obj(x):
        return smooth_max_val(x, n, K, beta)

    def grad(x):
        return smooth_max_grad(x, n, K, beta)

    def cfun(x):
        return np.sum(x) - n / 2.0

    def cjac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            c5_old = compute_c5(h)
            if c5_new <= c5_old + 1e-12:
                return h_new
    except Exception:
        pass
    return h

            break

        T = T_init * (T_final / T_init) ** frac

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

        h[i] = 0.0
        h[j] = 1.0
        new_corr = np.correlate(h, 1.0 - h, mode='full')
```
```python
        new_max = float(new_corr.max()) * dx
        old_max = float(curr_corr.max()) * dx
```

```python
def nesterov_subgrad(h, n, time_limit, K=20, beta=80):

    h_best = h.copy()

    h_prev = h.copy()

    n_iter = max(int(time_limit * 30), 50)

            break

            break

        if it == 0:
            y = h_curr.copy()
        else:
            gamma = (it - 1.0 + np.sqrt(1.0 + 4.0 * (it - 1.0) ** 2)) / (2.0 * it)
            y = h_curr + gamma * (h_curr - h_prev)
            y = np.clip(y, 0.0, 1.0)

        gy_n = np.linalg.norm(g_y)

        h_new = project_feasible(y - step_size * g_y, n)

        h_curr = h_new
```
