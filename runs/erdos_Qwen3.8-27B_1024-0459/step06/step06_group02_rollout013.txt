import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def lag_gradient(h, offset, n):
    g = np.zeros(n)
    if offset >= 0:
        npairs = n - offset
        if npairs > 0:
            g[:npairs] += (1.0 - h[offset:offset + npairs])
            g[offset:offset + npairs] -= h[:npairs]
    else:
        a = -offset
        npairs = n - a
        if npairs > 0:
            g[a:a + npairs] += (1.0 - h[:npairs])
            g[:npairs] -= h[a:a + npairs]
    return g

def weighted_lag_gradient(h, active_lags, weights, n):
    g = np.zeros(n)
    for i, lag_idx in enumerate(active_lags):
        w = weights[i]
        if w < 1e-16:
            continue
        offset = lag_idx - (n - 1)
        g += w * lag_gradient(h, offset, n)
    return g

    m = corr.max()
    shifted = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(shifted)
    s = ex.sum()
    return m + np.log(s) / beta

    idx = np.argsort(-corr)[:K]
    vals = corr[idx]
    shifted = np.clip(beta * (vals - vals.max()), -50.0, 50.0)
    w = np.exp(shifted)
    s = w.sum()
    if s < 1e-30:
        w = np.ones(K)
    else:
        w = w / s
    return idx, w

    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    fp = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(fp, xp, src)

def projected_gradient_step(h, g, step, n):
    x = h - step * g
    x = np.clip(x, 0.0, 1.0)
    x = project_feasible(x, n)
    return x

def bb_step(s, y, alpha_max, alpha_min):
    sy = np.dot(s, y)
    yy = np.dot(y, y)
    ss = np.dot(s, s)
    if yy < 1e-20:
        return alpha_max
    if sy > 1e-10 * yy:
        alpha_plus = ss / sy
        alpha_minus = sy / yy
        return min(max(alpha_plus, alpha_min), alpha_max)
    return alpha_max
