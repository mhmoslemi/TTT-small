import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

_T0 = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        interior = (h > 1e-15) & (h < 1.0 - 1e-15)
        if interior.any():
            h[interior] += diff / interior.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(h, beta, penalty_M):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = beta * (corr * dx)
    # Use logsumexp for numerical stability
    m = scaled.max()
    log_sum = m + np.log(np.exp(scaled - m).sum())
    obj = log_sum / beta + penalty_M * (h.sum() - n/2.0)**2
    
    # Gradient of smooth max
    w = np.exp(scaled - log_sum)  # softmax weights, sum to 1
    # w[k] = exp(beta*corr[k]*dx) / sum(exp(beta*corr[j]*dx))
    
    g = np.zeros(n)
    for k_idx in range(2*n - 1):
        wk = w[k_idx] * dx
        if wk < 1e-15:
            continue
        off = k_idx - (n - 1)
        if off >= 0:
            np_ = n - off
            # c = sum_{m=0}^{n-1-off} h[m+off]*(1-h[m])
            # dc/dh[i] = (1-h[i-off])*[off<=i<=n-1] + (-h[i+off])*[0<=i<=n-1-off]
            g[off:] += wk * (1.0 - h[:np_])  # i from off to n-1, 1-h[i-off]
            g[:np_] -= wk * h[off:]  # i from 0 to n-1-off, -h[i+off]
        else:
            ao = -off
            np_ = n - ao
            # c = sum_{m=0}^{n-1-ao} h[m]*(1-h[m+ao])
            # dc/dh[i] = (1-h[i+ao])*[0<=i<=n-1-ao] + (-h[i-ao])*[ao<=i<=n-1]
            g[:np_] += wk * (1.0 - h[ao:])  # i from 0 to n-1-ao, 1-h[i+ao]
            g[ao:] -= wk * h[:np_]  # i from ao to n-1, -h[i-ao]
    
    g += 2 * penalty_M * (h.sum() - n/2.0)
    return obj, g

        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - h[ao:])
            g[ao:] -= wk * h[:np_]
    return f, g


def gradient_descent_projected(h, beta, K, time_limit, n_steps_max=5000):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    step = 0.05
    for it in range(n_steps_max):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 10:
            break
        f, g = topk_smooth_grad(h, beta, K)
        g = g + (h.sum() - n / 2.0) / n
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            alpha = step * mult / (gn * gn + 1e-30)
            alpha = min(alpha, 0.5 / (gn + 1e-15))
            h_try = h - alpha * g
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try)
            if c_try < best_c5:
                h = h_try
                best_c5 = c_try
                best_h = h.copy()
                step = min(step * 1.1, 0.5)
                break
            elif mult == 0.05:
                step = max(step * 0.7, 0.001)
    return best_h, best_c5


def binary_local_search(h, n, time_limit, max_rounds=500):
    start = time.time()
    h_bin = (h > 0.5).astype(float)
    ones = int(np.sum(h_bin))
    target_ones = n // 2
    if ones > target_ones:
        idx_one = np.where(h_bin == 1)[0]
        np.random.shuffle(idx_one)
        for i in range(ones - target_ones):
            h_bin[idx_one[i]] = 0.0
    elif ones < target_ones:
        idx_zero = np.where(h_bin == 0)[0]
        np.random.shuffle(idx_zero)
        for i in range(target_ones - ones):
            h_bin[idx_zero[i]] = 1.0
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    no_improve = 0
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 10:
            break
        corr = np.correlate(h_bin, 1.0 - h_bin, mode='full')
        k_max = np.argmax(corr)
        off = k_max - (n - 1)
        n_one = np.where(h_bin == 1)[0]
        n_zero = np.where(h_bin == 0)[0]
        if len(n_one) == 0 or len(n_zero) == 0:
            break
        n_try = min(300, len(n_one) * len(n_zero))
        improved = False
        for _ in range(n_try):
            i1 = n_one[np.random.randint(len(n_one))]
            i0 = n_zero[np.random.randint(len(n_zero))]
            dx = 2.0 / n
            delta = 0.0
            for kk in [-3, -2, -1, 0, 1, 2, 3]:
                ki = kk + (n - 1)
                if 0 <= ki < len(corr):
                    pass
            h_try = h_bin.copy()
            h_try[i1] = 0.0
            h_try[i0] = 1.0
            c_try = compute_c5(h_try)
            if c_try < best_c5 - 1e-14:
                h_bin = h_try
                best_c5 = c_try
                best_h = h_try.copy()
                n_one = np.where(h_bin == 1)[0]
                n_zero = np.where(h_bin == 0)[0]
                improved = True
                no_improve = 0
                break
        if not improved:
            no_improve += 1
            if no_improve > 100:
                break
    return best_h, best_c5


def quasiperiodic_init(n, alpha, rng):
    x = (np.arange(n) + 0.5) * 2.0 / n
    h = (x * alpha % 1.0 < 0.5).astype(float)
    return project_feasible(h, n)


def optimize_at_resolution(h, n, time_limit):
    best_c5 = compute_c5(h)
    best_h = h.copy()
    betas = [50, 200, 1000, 5000, 20000, 100000]
    for beta in betas:
        if time.time() - _T0 > _BUDGET - 30:
            break
        elapsed = time.time() - _T0
        remaining = _BUDGET - elapsed - 30
        if remaining < 5:
            break
        M = max(1000.0, beta * 5.0)
        maxiter = 150 if n <= 1024 else 100
        h_new = lbfgs_run(best_h, beta, M, maxiter=maxiter)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new
    if time.time() - _T0 < _BUDGET - 20:
        t_rem = _BUDGET - time.time() - 15
        t_gd = min(30.0, t_rem * 0.3)
        if t_gd > 3:
            h_gd, c_gd = gradient_descent_projected(best_h, 50000, 30, t_gd, n_steps_max=3000)
            if c_gd < best_c5:
                best_c5 = c_gd
                best_h = h_gd
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    try:
        best_h = np.array(initial_h_values, dtype=float).copy()
        best_n = len(best_h)
        best_c5 = compute_c5(best_h)
    except Exception:
        best_n = 512
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_c5 = compute_c5(best_h)

    # Phase 1: Optimize at parent resolution with L-BFGS
    h_work = best_h.copy()
    n_work = best_n
    for beta in [50, 200, 1000, 5000, 20000]:
        if time.time() - _T0 > _BUDGET - 300:
            break
        M = max(1000.0, beta * 5.0)
        maxiter = 200
        h_work = lbfgs_run(h_work, beta, M, maxiter=maxiter)
        c = compute_c5(h_work)
        if c < best_c5:
            best_c5 = c
            best_h = h_work.copy()

    # Phase 2: Higher resolution n=1024
    if time.time() - _T0 < _BUDGET - 200:
        n2 = 1024
        h2 = interp_to(best_h, best_n, n2)
        h2 = project_feasible(h2, n2)
        for beta in [100, 500, 2000, 10000, 50000]:
            if time.time() - _T0 > _BUDGET - 100:
                break
            M = max(5000.0, beta * 5.0)
            maxiter = 150
            h2 = lbfgs_run(h2, beta, M, maxiter=maxiter)
            c = compute_c5(h2)
            if c < best_c5:
                best_c5 = c
                best_h = h2.copy()
                best_n = n2

    # Phase 3: Higher resolution n=2048
    if time.time() - _T0 < _BUDGET - 120:
        n3 = 2048
        h3 = interp_to(best_h, best_n, n3)
        h3 = project_feasible(h3, n3)
        for beta in [100, 500, 2000, 10000]:
            if time.time() - _T0 > _BUDGET - 60:
                break
            M = max(5000.0, beta * 5.0)
            maxiter = 120
            h3 = lbfgs_run(h3, beta, M, maxiter=maxiter)
            c = compute_c5(h3)
            if c < best_c5:
                best_c5 = c
                best_h = h3.copy()
                best_n = n3

    # Phase 4: Try quasi-periodic initializations at current best resolution
    alphas = [1.0 / (2.0 * (1.0 + np.sqrt(5.0)) % 1.0), 0.381966, 0.618034,
              0.311_228, 0.726_543, 0.309_017, 0.707_107]
    for alpha in alphas:
        if time.time() - _T0 > _BUDGET - 40:
            break
        h_qp = quasiperiodic_init(best_n, alpha, rng)
        c_qp = compute_c5(h_qp)
        if c_qp < best_c5 + 0.05:
            for beta in [500, 5000, 50000]:
                if time.time() - _T0 > _BUDGET - 30:
                    break
                M = max(5000.0, beta * 5.0)
                h_qp = lbfgs_run(h_qp, beta, M, maxiter=100)
                c = compute_c5(h_qp)
                if c < best_c5:
                    best_c5 = c
                    best_h = h_qp.copy()
                    break

    # Phase 5: Gradient descent refinement
    if time.time() - _T0 < _BUDGET - 20:
        t_rem = _BUDGET - time.time() - 15
        t_gd = min(60.0, t_rem * 0.4)
        if t_gd > 5:
            h_gd, c_gd = gradient_descent_projected(best_h, 100000, 20, t_gd, n_steps_max=5000)
            if c_gd < best_c5:
                best_c5 = c_gd
                best_h = h_gd.copy()

    # Phase 6: Very high beta L-BFGS polish
    if time.time() - _T0 < _BUDGET - 15:
        for beta in [200000, 500000]:
            if time.time() - _T0 > _BUDGET - 10:
                break
            M = max(10000.0, beta * 3.0)
            h_new = lbfgs_run(best_h, beta, M, maxiter=100)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()

    # Phase 7: Binary local search
    if time.time() - _T0 < _BUDGET - 30:
        t_bs = min(60.0, _BUDGET - time.time() - 25)
        if t_bs > 5:
            h_bin, c_bin = binary_local_search(best_h, best_n, t_bs, max_rounds=300)
            if c_bin < best_c5:
                best_c5 = c_bin
                best_h = h_bin.copy()
                # Polish the binary solution with continuous methods
                if time.time() - _T0 < _BUDGET - 15:
                    for beta in [5000, 50000]:
                        if time.time() - _T0 > _BUDGET - 10:
                            break
                        M = max(10000.0, beta * 3.0)
                        h_new = lbfgs_run(best_h, beta, M, maxiter=80)
                        c_new = compute_c5(h_new)
                        if c_new < best_c5:
                            best_c5 = c_new
                            best_h = h_new.copy()

    # Phase 8: Final exact gradient steps
    if time.time() - _T0 < _BUDGET - 10:
        for _ in range(50):
            if time.time() - _T0 > _BUDGET - 8:
                break
            g = exact_max_grad(best_h)
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break
            for step_mult in [0.01, 0.005, 0.001, 0.0005]:
                h_try = project_feasible(best_h - step_mult * g, best_n)
                c_try = compute_c5(h_try)
                if c_try < best_c5:
                    best_c5 = c_try
                    best_h = h_try.copy()
                    break

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
