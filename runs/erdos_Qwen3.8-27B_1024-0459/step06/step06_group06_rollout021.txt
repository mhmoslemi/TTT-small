The minimum over the hypercube [0,1]^n of such a function is achieved at a vertex (all h[j] ∈ {0,1}). This is because along each coordinate line, the function is linear, so the minimum on [0,1] is at an endpoint.

Therefore: min_{0≤h≤1} max_k f_k(h) = min_{h∈{0,1}^n} max_k f_k(h)

The constraint sum(h) = n/2 is automatically satisfiable with a binary sequence (choose any n/2 positions to be 1).

So the problem reduces to: arrange n/2 ones and n/2 zeros in a sequence of length n to minimize max_k (number of 0→1 pairs at lag k).

This is a pure combinatorial problem! And the "nearly binary" solutions from the continuous optimization are essentially approximations to this combinatorial optimum.

Now, how to solve this combinatorial problem?
- Brute force: impossible for n=512
- Local search: start with a binary sequence and flip bits one at a time
- The "flip" of bit j (0→1 or 1→0) changes f_k by a definite amount for each k

For a binary sequence, flipping h[j] from 0 to 1:
- f_k changes by: +(1 if h[j-o_k]=0 else 0) - (1 if h[j+o_k]=1 else 0)
  Wait, let me recompute.

f_k = sum_{i∈S_k} h[i+o_k]*(1-h[i])

If h[j] goes from 0 to 1:
- Terms where j = i+o_k (i.e., i = j-o_k, if j-o_k ∈ S_k): was 0 (since h[j]=0), becomes 1-h[j-o_k]
  → Δf_k += (1-h[j-o_k]) if j-o_k is valid
- Terms where j = i (i.e., i = j, if j ∈ S_k): was h[j+o_k]*(1-0) = h[j+o_k], becomes h[j+o_k]*(1-1) = 0
  → Δf_k -= h[j+o_k] if j is valid

So Δf_k = (1-h[j-o_k]) - h[j+o_k] (with validity checks) = 1 - h[j-o_k] - h[j+o_k]

If h[j] goes from 1 to 0:
Δf_k = -(1-h[j-o_k]) + h[j+o_k] = -1 + h[j-o_k] + h[j+o_k]

This is the same as what I derived before for the gradient.

So the local search is:
1. Compute all f_k
2. Find the current max
3. For each j, compute Δf_k for all k if we flip h[j]
4. Find the flip that most reduces the max

This is essentially what the coordinate descent in the previous code does, but restricted to binary values.

- We can find it by sorting the breakpoints

```python
import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1-h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.count_nonzero(mask)
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def make_binary_feasible(h, n):
    """Convert to binary while maintaining sum = n/2"""
    h = np.array(h, dtype=float).copy()
    # Sort by value, set top n/2 to 1, rest to 0
    target = int(n / 2)
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:target]] = 1.0
    return h_bin


def init_thue_morse(n):
    """Thue-Morse based initialization"""
    h = np.zeros(n)
    for i in range(n):
        # Count set bits
        c = 0
        x = i
        while x:
            c += x & 1
            x >>= 1
        h[i] = c % 2
    return project_feasible(h, n)


def init_random(rng, n):
    """Random binary initialization"""
    target = n // 2
    h = np.zeros(n)
    idx = rng.choice(n, size=target, replace=False)
    h[idx] = 1.0
    return h


def init_vander_corput(n):
    """Van der Corput low discrepancy"""
    h = np.zeros(n)
    target = n // 2
    # Van der Corput sequence in [0,1)
    vdc = np.zeros(n)
    for i in range(1, n+1):
        f = 0.0
        d = 0.5
        x = i
        while x > 0:
            f += (x % 2) * d
            x //= 2
            d *= 0.5
        vdc[i-1] = f
    # Select the "middle" n/2 values (closest to 0.5)
    dist = np.abs(vdc - 0.5)
    order = np.argsort(dist)
    h[order[:target]] = 1.0
    return h


def grad_f_k(h, k_idx, n):
    """Gradient of f_k with respect to h, where k_idx is the index into the full correlation"""
    g = np.zeros(n)
    offset = k_idx - (n - 1)
    if offset >= 0:
        npairs = n - offset
        # f_k = sum_{i=0}^{npairs-1} h[i+offset]*(1-h[i])
        # df_k/dh[j] = (1-h[j-offset]) if j>=offset and j-offset < npairs
        #            - h[j+offset] if j < npairs and j+offset < n
        g[offset:offset+npairs] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:offset+npairs]
    else:
        a = -offset
        npairs = n - a
        # f_k = sum_{i=a}^{n-1} h[i]*(1-h[i-a])
        #     = sum_{j=0}^{npairs-1} h[j+a]*(1-h[j])
        g[a:a+npairs] += (1.0 - h[:npairs])
        g[:npairs] -= h[a:a+npairs]
    return g


def f_k_value(h, k_idx, n):
    """Compute f_k for a specific k index"""
    offset = k_idx - (n - 1)
    if offset >= 0:
        npairs = n - offset
        return np.sum(h[offset:offset+npairs] * (1 - h[:npairs]))
    else:
        a = -offset
        npairs = n - a
        return np.sum(h[a:a+npairs] * (1 - h[:npairs]))


def top_k_gradient(h, n, K=20):
    """Compute weighted gradient of top-K correlations"""
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:K]
    c_max = corr[idx[0]]
    # Soft weights: exp(beta * (c_i - c_max))
    beta = 50.0
    w = np.exp(beta * (corr[idx] - c_max))
    w /= w.sum()
    g = np.zeros(n)
    for i, k in enumerate(idx):
        g += w[i] * grad_f_k(h, k, n)
    return g, float(c_max)

def smooth_max_obj_grad(h, n, beta):
    """Smooth max of correlations and its gradient"""
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -50, 50)
    exp_vals = np.exp(shifted)
    logsum = m + np.log(np.sum(exp_vals)) / beta
    w = exp_vals / np.sum(exp_vals)
    
    g = np.zeros(n)
    dx = 2.0 / n
    wc = w * dx
    for k in range(len(corr)):
        wk = wc[k]
        if wk > 1e-14:
            g += wk * grad_f_k(h, k, n)
    return logsum * dx, g


def lbfgs_smooth(h, n, beta, penalty, maxiter=500, time_limit=60):
    global _T0, _BUDGET
    def obj(x):
        val, _ = smooth_max_obj_grad(x, n, beta)
        s = np.sum(x) - n/2.0
        return val + 0.5 * penalty * s * s
    
    def grad(x):
        _, g = smooth_max_obj_grad(x, n, beta)
        s = np.sum(x) - n/2.0
        g += penalty * s
        return g
    
    bounds = [(0,1)] * n
    start = time.time()
    options = {'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12, 'maxfun': 10000}
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds, options=options)
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def slsqp_smooth(h, n, beta, maxiter=300):
    def obj(x):
        return smooth_max_obj_grad(x, n, beta)[0]
    
    def grad(x):
        return smooth_max_obj_grad(x, n, beta)[1]
    
    def con_fun(x):
        return np.sum(x) - n/2.0
    
    def con_jac(x):
        return np.ones(n)
    
    bounds = [(0,1)] * n
    cons = [{'type':'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def anneal_step(h, n, rng, temperature):
    """One simulated annealing step"""
    h_new = h.copy()
    # Random perturbation
    n_perturb = max(1, n // 50)
    idx = rng.choice(n, size=n_perturb, replace=False)
    perturbation = rng.uniform(-0.3, 0.3, size=n_perturb)
    h_new[idx] += perturbation
    h_new = np.clip(h_new, 0, 1)
    h_new = project_feasible(h_new, n)
    return h_new


def simulate_anneal(h, n, time_limit, rng):
    global _T0, _BUDGET
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    
    T0 = 0.1
    T_end = 0.001
    it = 0
    while time.time() - start < time_limit:
        rem = _BUDGET - (time.time() - _T0)
        if rem < 10:
            break
        it += 1
        progress = min(1.0, (time.time() - start) / time_limit)
        T = T0 * (T_end/T0) ** progress
        
        h_new = anneal_step(h, n, rng, T)
        c_old = compute_c5(h)
        c_new = compute_c5(h_new)
        
        delta = c_new - c_old
        if delta < 0 or rng.uniform() < np.exp(-delta / max(T, 1e-10)):
            h = h_new
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()
        else:
            # Rejected: try a smaller perturbation
            if it % 10 == 0:
                h_new2 = h.copy()
                j = rng.randint(n)
                h_new2[j] = 1.0 - h[j]
                h_new2 = project_feasible(h_new2, n)
                c_try = compute_c5(h_new2)
                if c_try < best_c5:
                    best_c5 = c_try
                    best_h = h_new2.copy()
                    h = h_new2
```

```python
    deltas = np.zeros((n, n_full))
    for j in range(n):
        # First term indices: j-o[k] in [0, n)
        idx1 = j - o  # shape (2n-1,)
        valid1 = (idx1 >= 0) & (idx1 < n)
        # Second term indices: j+o[k] in [0, n)
        idx2 = j + o
        valid2 = (idx2 >= 0) & (idx2 < n)
        
        d = np.zeros(n_full)
        d[valid1] = (1.0 - h[idx1[valid1]])
        d[valid2] -= h[idx2[valid2]]
        deltas[j] = d
    return deltas
```

This is O(n * 2n) = O(2n²) with vectorized inner loop. For n=2048, each row is 4095 elements, and we have 2048 rows. Total: ~8M operations. In numpy, this should take <10ms.

Now, for the swap search:
- Given deltas (n × (2n-1)) and current corr (2n-1):
- For swap (a→0, b→1): new_corr = corr - deltas[a] + deltas[b]
- new_max = max(new_corr)
- We want: min over (a,b) with h[a]=1, h[b]=0 of new_max

This is the hard part. Naively O(n² * 2n). For n=2048: 2048² * 4095 ≈ 17B. Too slow in one shot.

But we can be smarter:
1. For each a, find the best b: O(n * 2n) per a, total O(n² * 2n) = O(n² * 4095) for n=2048 ≈ 17B. Still too slow.

Alternative: use the fact that we only need to check if new_max < current_max.
new_max < current_max ⟺ max_k(corr[k] + delta_b[k] - delta_a[k]) < current_max

This is hard to vectorize over (a,b) pairs.

Better approach: **greedy with look-ahead**.
1. Find the current top-K correlations
2. For each of these K positions k*, find the best swap that reduces corr[k*]
3. Check the global effect

Or even simpler: **randomized local search**.
1. Repeat:
   a. Pick a random pair (a,b) with h[a]=1, h[b]=0
   b. Compute new_corr = corr - delta[a] + delta[b]
   c. If max(new_corr) < max(corr): accept
   d. Else with probability exp(-Δ/T): accept (simulated annealing)

Each step is O(2n) to compute new_corr (just array ops). We can do millions of steps per second.

For n=2048: each step is O(4095) ≈ 4K operations. At 1GHz: 250M steps per second. Way more than enough.

Let me go with this approach for the main optimization loop.

Additionally, I can use a "focal" method:
1. Identify the top K=10 correlations
2. For each top correlation k*, find the single flip (or swap) that most reduces corr[k*] while keeping other top correlations from increasing too much
3. This is a 1D/2D subproblem

Let me code up the complete solution now. I'll use a multi-phase approach:

Phase 0: Multiple initializations (parent, Thue-Morse, random, etc.)
Phase 1: Quick optimization at n=256 or n=512
Phase 2: Upscale and optimize at n=1024
Phase 3: Upscale and optimize at n=2048
Phase 4: Final polish

For each optimization phase:
- Binary swap local search with simulated annealing
- Followed by L-BFGS-B on the smooth max (to allow near-binary values)
- Then SLSQP for final constraint satisfaction

Let me also add a "targeted flip" step:
- Find the k with maximum corr[k]
- Identify the position j that contributes most to corr[k] (i.e., h[j+o_k]*(1-h[j]) is large)
- Try flipping that position

OK, I think I have a solid plan. Let me write the final code. I'll aim for clarity and correctness, with the key optimization being the vectorized swap-based local search.

One more thought: the problem says "step function h: [0,2] → [0,1]". In the discrete setting, this means h is piecewise constant on intervals of length dx. The values h[0], h[1], ..., h[n-1] are the step heights. The binary solution corresponds to a step function that jumps between 0 and 1.

Let me also make sure I handle the edge cases correctly. The "full" correlation of two length-n arrays gives 2n-1 values. The dx factor is 2/n. So the objective is max(corr) * 2/n.

```python
import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def make_binary_feasible(h, n):
    target = int(n / 2)
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:target]] = 1.0
    return h_bin


def init_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1')
        h[i] = c % 2
    return h


def init_random(rng, n):
    target = n // 2
    h = np.zeros(n)
    idx = rng.choice(n, size=target, replace=False)
    h[idx] = 1.0
    return h


def init_block(n):
    h = np.zeros(n)
    h[:n//2] = 1.0
    return h


def compute_deltas(h, n):
    """Compute delta[j,k]: change in corr[k] when h[j] goes 0->1 (others fixed)"""
    n_full = 2 * n - 1
    o = np.arange(n_full) - (n - 1)
    deltas = np.zeros((n, n_full))
    for j in range(n):
        idx1 = j - o
        valid1 = (idx1 >= 0) & (idx1 < n)
        idx2 = j + o
        valid2 = (idx2 >= 0) & (idx2 < n)
        d = np.zeros(n_full)
        d[valid1] = (1.0 - h[idx1[valid1]])
        d[valid2] -= h[idx2[valid2]]
        deltas[j] = d
    return deltas


def swap_local_search(h, n, time_limit, rng, sa=True, T0=0.05):
    """Binary swap-based local search with simulated annealing"""
    global _T0, _BUDGET
    h = np.array(h, dtype=float).copy()
    h = make_binary_feasible(h, n)
    corr = compute_corr(h)
    best_h = h.copy()
    best_corr = corr.copy()
    best_max = float(np.max(corr))
    start = time.time()
    it = 0
    no_improve = 0
    
    while time.time() - start < time_limit:
        rem = _BUDGET - (time.time() - _T0)
        if rem < 10:
            break
        it += 1
        
        # Recompute deltas periodically
        if it % 50 == 1:
            deltas = compute_deltas(h, n)
        
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        improved = False
        progress = min(1.0, (time.time() - start) / time_limit)
        T = T0 * (0.001 / T0) ** progress if sa else 0.0
        
        # Try random swaps
        n_tries = max(100, 500 // max(1, it % 10))
        for _ in range(n_tries):
            a = rng.choice(ones)
            b = rng.choice(zeros)
            if a == b:
                continue
            new_corr = corr - deltas[a] + deltas[b]
            new_max = float(np.max(new_corr))
            cur_max = float(np.max(corr))
            
            if new_max < cur_max - 1e-12:
                h[a], h[b] = h[b], h[a]
                corr = new_corr
                improved = True
                no_improve = 0
                if new_max < best_max:
                    best_max = new_max
                    best_h = h.copy()
                    best_corr = corr.copy()
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h < 0.5)[0]
            elif new_max - cur_max < T and rng.uniform() < np.exp(-(new_max - cur_max) / max(T, 1e-10)):
                h[a], h[b] = h[b], h[a]
                corr = new_corr
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h < 0.5)[0]
        
        # Also try targeted flips: find worst k, try to reduce it
        if it % 10 == 0:
            k_worst = int(np.argmax(corr))
            o_w = k_worst - (n - 1)
            # Find positions contributing most to corr[k_worst]
            # corr[k_worst] = sum h[i+o_w]*(1-h[i]) for valid i
            if o_w >= 0:
                valid_i = np.arange(0, n - o_w)
                contrib = h[valid_i + o_w] * (1 - h[valid_i])
            else:
                a0 = -o_w
                valid_i = np.arange(0, n + o_w)
                contrib = h[valid_i + o_w] * (1 - h[valid_i])
            
            if len(contrib) > 0:
                top_contrib = np.where(contrib > 0)[0]
                if len(top_contrib) > 0:
                    # Try flipping the position that contributes most
                    tc = top_contrib
                    for _ in range(min(20, len(tc))):
                        ci = tc[rng.randint(len(tc))]
                        if o_w >= 0:
                            j = valid_i[ci] + o_w  # flip h[j] from 1 to 0
                        else:
                            j = valid_i[ci] + o_w
                        
                        if h[j] > 0.5:
                            # Try setting to 0, and some other 0 to 1
                            if len(zeros) > 0:
                                b = rng.choice(zeros)
                                if b != j:
                                    new_corr = corr - deltas[j] + deltas[b]
                                    new_max = float(np.max(new_corr))
                                    cur_max = float(np.max(corr))
                                    if new_max < cur_max - 1e-12:
                                        h[j], h[b] = h[b], h[j]
                                        corr = new_corr
                                        improved = True
                                        no_improve = 0
                                        if new_max < best_max:
                                            best_max = new_max
                                            best_h = h.copy()
                                            best_corr = corr.copy()
                                        ones = np.where(h > 0.5)[0]
                                        zeros = np.where(h < 0.5)[0]
        
        if not improved:
            no_improve += 1
            if no_improve > 20:
                # Restart from best
                h = best_h.copy()
                corr = best_corr.copy()
                deltas = compute_deltas(h, n)
                no_improve = 0
                if it > 500 and progress > 0.8:
                    break
    
    return h, best_h, best_max * (2.0 / n)


def smooth_max_and_grad(h, n, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    sh = beta * (corr - m)
    sh = np.clip(sh, -50, 50)
    ex = np.exp(sh)
    s = ex.sum()
    obj = (m + np.log(s) / beta) * (2.0 / n)
    w = ex / s
    g = np.zeros(n)
    dx = 2.0 / n
    wc = w * dx
    for k in range(len(corr)):
        wk = wc[k]
        if wk > 1e-14:
            o = k - (n - 1)
            if o >= 0:
                npairs = n - o
                g[o:o+npairs] += wk * (1.0 - h[:npairs])
                g[:npairs] -= wk * h[o:o+npairs]
            else:
                a0 = -o
                npairs = n - a0
                g[a0:a0+npairs] += wk * (1.0 - h[:npairs])
                g[:npairs] -= wk * h[a0:a0+npairs]
    return obj, g


def lbfgs_smooth(h, n, beta, penalty, maxiter=2000, time_limit=60):
    global _T0, _BUDGET
    def obj(x):
        val, _ = smooth_max_and_grad(x, n, beta)
        ss = np.sum(x) - n / 2.0
        return val + 0.5 * penalty * ss * ss

    def grad(x):
        _, g = smooth_max_and_grad(x, n, beta)
        ss = np.sum(x) - n / 2.0
        g += penalty * ss
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                                'maxfun': 20000})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def slsqp_smooth(h, n, beta, maxiter=500):
    def obj(x):
        return smooth_max_and_grad(x, n, beta)[0]

    def grad(x):
        return smooth_max_and_grad(x, n, beta)[1]

    def cfun(x):
        return np.sum(x) - n / 2.0

    def cjac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)
```
