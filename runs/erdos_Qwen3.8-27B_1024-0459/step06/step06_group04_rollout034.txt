import numpy as np
from scipy.optimize import minimize, LinearConstraint, Bounds
import time

_T0 = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if interior.any():
            h[interior] += diff / interior.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_func(x, beta):
    dx = 2.0 / len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    s = beta * (corr - m)
    s = np.clip(s, -500.0, 500.0)
    e = np.exp(s)
    return (m + np.log(e.sum()) / beta) * dx

def smooth_max_jac(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    s = beta * (corr - m)
    s = np.clip(s, -500.0, 500.0)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx] * dx
        if wk < 1e-15:
            continue
        off = k_idx - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - x[ao:])

def thue_morse(n):
    """Generate Thue-Morse sequence of length n (padded or truncated)."""
    m = 1
    seq = np.array([0.0])
    while m < n:
        m *= 2
        seq = np.concatenate([seq, 1.0 - seq])
    return seq[:n]

def rudin_shapiro(n):

    m = 1
    a = np.array([1.0])
    b = np.array([1.0])

    #                          for k in range(fft_size-n+1, fft_size): wraparound = 0
    # 
    # Actually, let me just verify with a small example.
    # h = [a, b, c], 1-h = [1-a, 1-b, 1-c]
    # np.correlate(h, 1-h, 'full') = [a(1-c)+b(1-b)+c(1-a), a(1-b)+b(1-a), a(1-a)]
    # Wait no: np.correlate(a, v, 'full')[k] = sum_n a[n+k-(len(v)-1)] * v[n]
    # For a = [h0, h1, h2], v = [v0, v1, v2], len(v) = 3
    # result[0] = a[0-2]*v[0] + a[1-2]*v[1] + a[2-2]*v[2] = a[-2]*v0 + a[-1]*v1 + a[0]*v2 = 0 + 0 + h0*v2
    # Hmm, that doesn't seem right either.
    # 
    # Let me just check numpy docs: 
    # "Full output of the linear cross-correlation (or correlation product) of one_1 and one_2.
    # Its length is (N + M - 1)."
    # "The default mode 'full' returns output of length N+M-1 (full convolution)
    # and... out[k] = sum_m a[m] * v[m+k-(M-1)]"
    # So out[k] = sum_m h[m] * (1-h)[m+k-(M-1)] where M = len(1-h) = N
    # = sum_m h[m] * (1-h)[m+k-N+1]
    # For k=0: sum_m h[m] * (1-h)[m-N+1] = h[N-1]*(1-h)[0] (only m=N-1 gives valid index)
    # For k=N-1: sum_m h[m] * (1-h)[m] = sum h[m]*(1-h[m])
    # For k=2N-2: sum_m h[m] * (1-h)[m+N-1] = h[0]*(1-h)[N-1]
    
    # So the full correlation vector has index k from 0 to 2N-2,
    # corresponding to lag = k - (N-1), ranging from -(N-1) to +(N-1).
    # corr_full[k] = sum_m h[m] * (1-h)[m + (k-(N-1))]
    # = sum_{m: 0<=m<N, 0<=m+k-N+1<N} h[m] * (1-h)[m+k-N+1]
    
    # The FFT approach:
    # Ifft(fft(h) * conj(fft(1-h))) gives the cross-correlation:
    # c[k] = sum_n h[n] * (1-h)[(n-k) mod fft_size]
    # With zero padding, for k in [0, N-1]:
    # c[k] = sum_{n=k}^{N-1} h[n] * (1-h)[n-k]
    # For k in [fft_size-N+1, fft_size-1] (i.e., k = -N+1 to -1 mod fft_size):
    # c[k] = sum_{n=0}^{fft_size-1-k} h[n] * (1-h)[n-k+fft_size]
    #       But 1-h is zero-padded, so (1-h)[n-k+fft_size] = 0 unless n-k+fft_size < N
    #       This gives c[k] = sum_{n=0}^{k-fft_size+N-1} ... hmm this is getting confusing.
    #
    # Let me use a different approach. The standard formula:
    # np.correlate(a, v, 'full') = np.convolve(a, v[::-1], 'full')
    # So I can compute it as: ifft(fft(h, fft_size) * fft((1-h)[::-1], fft_size))
    
    H = np.fft.rfft(h, fft_size)
    O_rev = np.fft.rfft(one_minus_h[::-1], fft_size)
    conv = np.fft.irfft(H * O_rev, fft_size)
    # conv[0:size] is the linear convolution = the full correlation
    return conv[:size].copy()

import numpy as np
from scipy.optimize import minimize, LinearConstraint, Bounds
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if interior.any():
            h[interior] += diff / interior.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_func(x, beta):
    dx = 2.0 / len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    s = beta * (corr - m)
    s = np.clip(s, -600.0, 600.0)
    e = np.exp(s)
    return (m + np.log(e.sum()) / beta) * dx


def smooth_max_jac(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    s = beta * (corr - m)
    s = np.clip(s, -600.0, 600.0)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx] * dx
        if wk < 1e-15:
            continue
        off = k_idx - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - x[ao:])
            g[ao:] -= wk * x[:np_]
    return g


def eq_func(x, n):
    return x.sum() - n / 2.0


def eq_jac(x, n):
    return np.ones(n)


def slsqp_opt(h, beta, maxiter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_max_func, h, args=(beta,),
                       jac=smooth_max_jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': eq_func,
                                     'jac': eq_jac, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def trust_constr_opt(h, beta, maxiter):
    n = len(h)
    try:
        bnds = Bounds(np.zeros(n), np.ones(n))
        A_eq = np.ones((1, n))
        cons = LinearConstraint(A_eq, n / 2.0, n / 2.0)
        res = minimize(smooth_max_func, h, args=(beta,),
                       jac=smooth_max_jac, method='trust-constr',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'gtol': 1e-10, 'xtol': 1e-14})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def thue_morse(n):
    m = 1
    seq = np.array([0.0])
    while m < n:
        seq = np.concatenate([seq, 1.0 - seq])
        m = len(seq)
    return seq[:n]


def rudin_shapiro(n):
    m = 1
    a = np.array([1.0])
    b = np.array([1.0])
    while m < n:
        a, b = np.concatenate([a, b]), np.concatenate([a, -b])
        m = len(a)
    return (1.0 + a[:n]) / 2.0


def weighted_subgradient(h, n, time_limit, K=15, beta_w=50.0):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    for it in range(200000):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 5:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = corr.max() * (2.0 / n)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        idx = np.argsort(-corr)[:K]
        c_top = corr[idx]
        weights = np.exp(beta_w * (c_top - c_top.max()))
        s = weights.sum()
        if s < 1e-30:
            weights = np.ones(K) / K
        else:
            weights /= s
        g = np.zeros(n)
        for i, k_idx in enumerate(idx):
            wk = weights[i]
            if wk < 1e-12:
                continue
            off = k_idx - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:] += wk * (1.0 - h[:np_])
                g[:np_] -= wk * h[off:]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - h[ao:])
                g[ao:] -= wk * h[:np_]
        g *= (2.0 / n)

    return best_h, best_c5

    elif ones < target:
        idx = np.where(h_bin == 0)[0]
        np.random.shuffle(idx)
        for i in range(int(target - ones)):
            h_bin[idx[i]] = 1
    h_bin = project_feasible(h_bin, n)
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    start = time.time()

        if time.time() - start > time_limit:
            break

        improved = False
        n_one = np.where(h_bin == 1)[0]

def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)
