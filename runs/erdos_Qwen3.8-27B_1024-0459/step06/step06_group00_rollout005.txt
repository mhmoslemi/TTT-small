Since the problem is: min_h max_k sum_i h[i+k](1-h[i]) s.t. sum h = n/2, 0≤h≤1

This is a min-max problem. We can use the von Neumann minimax theorem:

- The inner problem becomes: min_h sum_k p[k] * corr(h,k) s.t. constraints
- This is a QP (quadratic objective)
- Update p based on which lags are "active"

This is essentially the "multiplicative weights" or "exponentiated gradient" approach to the outer max.

        h_b, c5_b = binary_greedy_swap(h_bin, n, time_limit=min(25, int(remaining_time() * 0.5)))
        h_bc = h_b.copy()
        if remaining_time() > 15:
            h_bc = slsqp_smoothmax(h_bc, n, 5000000, maxiter=1000)
            c5_bc = compute_c5(h_bc)
            h_bc, c5_bc = topk_equalize(h_bc, n, top_k=8, max_iters=300, time_limit=10)
            if c5_bc < best_c5 - 1e-14:
                best_c5 = c5_bc
                best_h = h_bc.copy()
        if c5_b < best_c5 - 1e-14:
            best_c5 = c5_b
            best_h = h_b.copy().astype(float)

    # Final SLSQP
    if remaining_time() > 10:
        h_final = slsqp_smoothmax(best_h, n, 10000000, maxiter=2000)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-14:
            best_c5 = c5_final
            best_h = h_final.copy()

    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, n)
