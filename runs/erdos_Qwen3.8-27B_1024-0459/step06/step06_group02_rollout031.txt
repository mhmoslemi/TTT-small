def smooth_max_obj_and_grad(x, n, beta, topk_cutoff=200):
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = beta * (corr - m)
    clipped = np.clip(shifted, -50.0, 50.0)
    ex = np.exp(clipped)
    s = np.sum(ex)
    w = ex / s
    dx = 2.0 / n
    val = (m + np.log(s) / beta) * dx
    
    # Gradient: only top lags
    g = np.zeros(n)
    thresh = 1.0 / (len(corr) * 10)
    for k in range(len(corr)):
        wk = w[k] * dx
        if wk < thresh:
            continue
        g += wk * grad_corr_k(x, k, n)
    return val, g


def slsqp_smooth_max(h0, n, beta, maxiter=300):
    def obj(x):
        v, _ = smooth_max_obj_and_grad(x, n, beta)
        return v

    def grad(x):
        _, g = smooth_max_obj_and_grad(x, n, beta)
        return g

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq',
             'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]

    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            return h_new
    except Exception:
        pass
    return h0.copy()


def weighted_gradient_step(h, n, top_k=20, step=0.01):
    """Step in direction that reduces top-k correlations."""
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:top_k]
    vals = corr[idx]
    # Weight by how much above the median of top-k
    v_med = np.median(vals)
    weights = np.maximum(vals - v_med, 0.01) * 0.5 + np.ones(len(idx)) * 0.5
    weights /= np.sum(weights)
    
    g = np.zeros(n)
    for i, k in enumerate(idx):
        g += weights[i] * grad_corr_k(h, k, n)
    
    h_new = project_feasible(h - step * g, n)
    return h_new


def gradient_descent_loop(h0, n, max_iter=500, base_step=0.005, top_k_start=50, top_k_end=5):
    h = h0.copy()
    best_c = compute_c5(h)
    stagnation = 0
    for it in range(max_iter):
        progress = it / max_iter
        K = max(top_k_end, int(top_k_start * (1 - progress) ** 0.5))
        step = base_step / (1.0 + 0.05 * it)
        
        h_new = weighted_gradient_step(h, n, top_k=K, step=step)
        c_new = compute_c5(h_new)
        
        if c_new < best_c - 1e-14:
            best_c = c_new
            h = h_new
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 200:
                break
    
    return h, best_c


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def make_feasible_random(n, rng):
    h = rng.rand(n)
    return project_feasible(h, n)


def make_sinusoidal(n, freq, amp, phase, rng):
    x = (np.arange(n) + 0.5) * 2.0 / n
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x + phase)
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

    # Initialize from parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        n = 512
        parent_h = np.zeros(n)
        parent_h[:n // 2] = 1.0

    n_init = len(parent_h)
    best_h = parent_h.copy()
    best_c5 = compute_c5(best_h)
    best_n = n_init

    # Also try the complement (by symmetry should give same C5 but may lead elsewhere)
    comp_h = 1.0 - parent_h
    comp_h = project_feasible(comp_h, n_init)
    comp_c5 = compute_c5(comp_h)

    # Phase 1: Polish parent at increasing resolution
    # First at native resolution
    for beta in [500, 2000, 10000, 50000]:
        if remaining() < 20:
            break
        h_try = slsqp_smooth_max(best_h, best_n, beta, maxiter=400)
        c_try = compute_c5(h_try)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try.copy()

    # Gradient descent at native resolution
    if remaining() > 40:
        h_gd, c_gd = gradient_descent_loop(best_h, best_n, max_iter=300, base_step=0.004)
        if c_gd < best_c5:
            best_c5 = c_gd
            best_h = h_gd.copy()

    # Phase 2: Multi-resolution approach
    resolutions = [1024, 1536, 2048]
    for target_n in resolutions:
        if remaining() < 150:
            break

        # Upsample
        h_up = interp_to(best_h, best_n, target_n)
        h_up = project_feasible(h_up, target_n)
        c_up = compute_c5(h_up)

        # SLSQP polish at new resolution
        for beta in [2000, 10000, 50000, 200000]:
            if remaining() < 30:
                break
            h_new = slsqp_smooth_max(h_up, target_n, beta, maxiter=300)
            c_new = compute_c5(h_new)
            if c_new < c_up:
                c_up = c_new
                h_up = h_new
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_up.copy()
                best_n = target_n

        # Gradient descent at new resolution
        if remaining() > 80:
            h_gd, c_gd = gradient_descent_loop(h_up, target_n, max_iter=400, base_step=0.003)
            if c_gd < c_up:
                c_up = c_gd
                h_up = h_gd
            if c_up < best_c5:
                best_c5 = c_up
                best_h = h_up.copy()
                best_n = target_n

        # Final SLSQP at this resolution
        if remaining() > 20:
            h_new = slsqp_smooth_max(h_up, target_n, 500000, maxiter=400)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()
                best_n = target_n

    # Phase 3: Random restarts with different structures
    n_use = best_n
    restarts = 25
    for r in range(restarts):
        if remaining() < 30:
            break

        # Different perturbation strategies
        strategy = r % 5
        if strategy == 0:
            # Gaussian perturbation
            scale = 0.003 + 0.002 * (r // 5)
            h_pert = best_h + scale * rng.randn(n_use)
        elif strategy == 1:
            # Sinusoidal perturbation
            freq = rng.randint(1, 8)
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.02 + 0.01 * (r // 5)
            x = (np.arange(n_use) + 0.5) * 2.0 / n_use
            noise = amp * np.sin(2 * np.pi * freq * x * 2.0 / 2.0 + phase)
            h_pert = best_h + noise
        elif strategy == 2:
            # Block perturbation
            h_pert = best_h.copy()
            block = rng.randint(10, max(11, n_use // 10))
            start = rng.randint(0, n_use - block)
            sign = 1.0 if rng.rand() > 0.5 else -1.0
            h_pert[start:start + block] += sign * 0.05
        elif strategy == 3:
            # High-frequency noise
            scale = 0.005 + 0.003 * (r // 5)
            h_pert = best_h + scale * rng.randn(n_use)
        else:
            # Very small perturbation + complement direction
            scale = 0.002
            h_pert = best_h + scale * rng.randn(n_use)

        h_pert = project_feasible(h_pert, n_use)
        c_pert = compute_c5(h_pert)

        # Optimize from perturbed point
        h_opt = slsqp_smooth_max(h_pert, n_use, 50000, maxiter=300)
        c_opt = compute_c5(h_opt)

        # If SLSQP didn't help much, try gradient descent
        if c_opt > c_pert + 0.0005 and remaining() > 20:
            h_gd, c_gd = gradient_descent_loop(h_pert, n_use, max_iter=200, base_step=0.003)
            if c_gd < c_opt:
                c_opt = c_gd
                h_opt = h_gd

        if c_opt < best_c5:
            best_c5 = c_opt
            best_h = h_opt.copy()

    # Phase 4: Try different initial constructions at a good resolution
    if remaining() > 100:
        n_try = best_n
        # Try a few structured starts
        starts = []
        # Random feasible
        starts.append(make_feasible_random(n_try, rng))
        # Low frequency sinusoid
        starts.append(make_sinusoidal(n_try, 1, 0.3, 0, rng))
        starts.append(make_sinusoidal(n_try, 2, 0.25, np.pi / 4, rng))
        # The complement of parent (upsampled)
        if n_try != n_init:
            comp_up = interp_to(comp_h, n_init, n_try)
            comp_up = project_feasible(comp_up, n_try)
            starts.append(comp_up)

        for s_idx, h_start in enumerate(starts):
            if remaining() < 30:
                break
            c_start = compute_c5(h_start)
            h_opt = slsqp_smooth_max(h_start, n_try, 10000, maxiter=300)
            c_opt = compute_c5(h_opt)
            if remaining() > 20 and c_opt < best_c5 + 0.01:
                h_opt2 = slsqp_smooth_max(h_opt, n_try, 100000, maxiter=300)
                c_opt2 = compute_c5(h_opt2)
                if c_opt2 < c_opt:
                    c_opt = c_opt2
                    h_opt = h_opt2
            if c_opt < best_c5:
                best_c5 = c_opt
                best_h = h_opt.copy()
                best_n = n_try

    # Phase 5: Final intensive polish
    # Multiple SLSQP passes with very high beta
    for beta in [500000, 2000000, 10000000]:
        if remaining() < 15:
            break
        h_new = slsqp_smooth_max(best_h, best_n, beta, maxiter=500)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Final gradient descent polish
    if remaining() > 30:
        h_gd, c_gd = gradient_descent_loop(best_h, best_n, max_iter=500, base_step=0.002,
                                           top_k_start=20, top_k_end=3)
        if c_gd < best_c5:
            best_c5 = c_gd
            best_h = h_gd.copy()

    # One more SLSQP after gradient descent
    if remaining() > 15:
        h_new = slsqp_smooth_max(best_h, best_n, 5000000, maxiter=500)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Verify with evaluator
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
