def theta_morse(n):

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_CUR_N = 0
_SM_BETA = 1000.0


def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def _sm_obj(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    return (m + np.log(np.sum(ez)) / beta) * dx


def _sm_grad(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    return g


def _eq_fun(x):
    return np.sum(x) - _CUR_N / 2.0


def _eq_jac(x):
    return np.ones(_CUR_N)


def slsqp_opt(h, n, beta, maxiter=1000):
    global _CUR_N, _SM_BETA
    _CUR_N = n
    _SM_BETA = beta
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_sm_obj, h0, jac=_sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.fun < _sm_obj(h0) - 1e-15:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def grad_single_shift(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        np_ = n - off
        g[off:] += (1.0 - h[:np_])
        g[:np_] -= h[off:]
    else:
        a = -off
        np_ = n - a
        g[:np_] += (1.0 - h[a:])
        g[a:] -= h[:np_]
    return g


def top_k_equalize(h, n, top_k=5, max_iters=800, time_limit=None):
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    start = time.time()
    for it in range(max_iters):
        if remaining_time() < 3:
            break
        if time_limit is not None and time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        c_max = corr[idx_top[0]]
        c_min_top = corr[idx_top[-1]]
        if c_max - c_min_top < 1e-12:
            break
        g_dir = np.zeros(n)
        n_top = len(idx_top)
        for pos, k in enumerate(idx_top):
            w_k = 1.0 if pos == 0 else (-1.0 / (n_top - 1))
            g_dir += w_k * grad_single_shift(h, k, n)
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        d = g_dir / gn
        improved = False
        steps = [0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.003, 0.007]
        for step in steps:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def subgradient_zigzag(h, n, max_iters=5000, time_limit=90):
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    k_curr = None
    k_prev = None
    stale_count = 0
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if remaining_time() < 5:
            break
        corr = compute_corr(h)
        k_curr = int(np.argmax(corr))
        if k_prev is not None and k_curr != k_prev:
            g = 0.5 * (grad_single_shift(h, k_curr, n) + grad_single_shift(h, k_prev, n))
            k_prev = k_curr
        else:
            g = grad_single_shift(h, k_curr, n)
            if k_prev is None:
                k_prev = k_curr
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-14:
            g = g / g_norm
        improved = False
        base_step = max(0.001, 0.02 * (1.0 - 0.5 * it / max_iters))
        for step in [base_step * 0.2, base_step * 0.5, base_step, base_step * 2.0,
                     base_step * 5.0, 0.001, 0.003, 0.005, 0.01, 0.02, 0.05, 0.1]:
            ht = project_feasible(h - step * g, n)
            c5_new = compute_c5(ht)
            if c5_new < best_c5 - 1e-15:
                h = ht
                best_c5 = c5_new
                best_h = h.copy()
                improved = True
                stale_count = 0
                break
        if not improved:
            stale_count += 1
            if stale_count > 300:
                break
            if stale_count % 40 == 0:
                rng = np.random.RandomState(stale_count * 7 + 13)
                pert = rng.randn(n) * 0.003
                ht = project_feasible(h + pert, n)
                c5_new = compute_c5(ht)
                if c5_new < best_c5:
                    h = ht
                    best_c5 = c5_new
                    best_h = h.copy()
                    stale_count = 0
    return best_h, best_c5


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def theta_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)


def pair_descent(h, n, max_sweeps=200, time_limit=60, rng=None):
    """Targeted pair coordinate descent focusing on max-correlation contributors."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    start = time.time()
    no_improve_sweeps = 0

    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        improved = False

        # Focus on the top shifts
        top_shifts = np.argsort(-corr)[:5]
        k_star = top_shifts[0]

        # Find positions contributing to the max correlation
        off = k_star - (n - 1)
        if off >= 0:
            contrib_ones = np.where((h[:n-off] > 0.5) & (h[off:] < 0.5))[0]
            contrib_zeros_off = off
        else:
            a = -off
            contrib_ones = np.where((h[a:] > 0.5) & (h[:n-a] < 0.5))[0]
            contrib_zeros_off = 0

        n_pairs = 200
        for _ in range(n_pairs):
            # Choose i from contributors, j from non-contributors
            if len(contrib_ones) > 0 and rng.random() < 0.7:
                i = contrib_ones[rng.randint(len(contrib_ones))]
                if off >= 0:
                    i = i  # position in h
                else:
                    i = i + a
            else:
                i = rng.randint(n)

            j = rng.randint(n)
            if i == j:
                continue

            t_min = max(-h[i], h[j] - 1.0)
            t_max = min(1.0 - h[i], h[j])
            if t_max - t_min < 1e-10:
                continue

            # Try to reduce: increase h[i] if i contributes, or decrease h[j]
            # Determine direction: we want to reduce corr at k_star
            # Increasing h[i] when h[i] contributes (h[i]=1, h[i+off]=0) increases corr
            # So we should DECREASE h[i] or INCREASE h[j] (to make h[j+off]=1)
            # net effect: move mass from i to j means h[i]-=t, h[j]+=t
            # Let's try both directions

            best_t = 0.0
            best_val = best_c5
            h_si, h_sj = h[i], h[j]

            # Try several t values in both directions
            for frac in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]:
                for direction in [1.0, -1.0]:
                    t_cand = direction * frac * 0.1 * (t_max - t_min + 1e-10)
                    t_cand = np.clip(t_cand, t_min, t_max)
                    if abs(t_cand) < 1e-12:
                        continue
                    h[i] = h_si + t_cand
                    h[j] = h_sj - t_cand
                    new_c5 = float(np.max(compute_corr(h)) * dx)
                    if new_c5 < best_val:
                        best_val = new_c5
                        best_t = t_cand

            h[i] = h_si
            h[j] = h_sj

            if best_t != 0 and best_val < best_c5 - 1e-15:
                h[i] += best_t
                h[j] -= best_t
                best_c5 = best_val
                improved = True

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 15:
                # Perturb and retry
                h = project_feasible(h + rng.randn(n) * 0.002, n)
                best_c5 = compute_c5(h)
                if best_c5 > compute_c5(project_feasible(h, n)):
                    break
                no_improve_sweeps = 0
        else:
            no_improve_sweeps = 0

        # Update contributor set periodically
        if sweep % 20 == 0:
            corr = compute_corr(h)
            top_shifts = np.argsort(-corr)[:5]
            k_star = top_shifts[0]
            off = k_star - (n - 1)
            if off >= 0:
                contrib_ones = np.where((h[:n-off] > 0.5) & (h[off:] < 0.5))[0]
            else:
                a = -off
                contrib_ones = np.where((h[a:] > 0.5) & (h[:n-a] < 0.5))[0]

    return h, best_c5


def binary_sa_targeted(h, n, time_limit=120, rng=None):
    """Binary SA with targeted moves based on correlation structure."""
    if rng is None:
        rng = np.random.RandomState(42)
    h_bin = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        sort_idx = np.argsort(-h_bin)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0

    dx = 2.0 / n
    corr = compute_corr(h_bin)
    best_c5 = float(np.max(corr) * dx)
    best_h = h_bin.copy()
    curr_c5 = best_c5
    start = time.time()

    T_start = 3.0 / dx
    T_end = 0.001 / dx
    total_iters = 40000
    min_T_ratio = T_end / T_start

    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    no_improve = 0

    for it in range(total_iters):
        if time.time() - start > time_limit:
            break
        if remaining_time() < 5:
            break

        T = T_start * (min_T_ratio ** (it / max(1, total_iters)))

        # Targeted move: focus on positions contributing to max correlation
        if it % 50 == 0:
            corr = compute_corr(h_bin)
            k_max = int(np.argmax(corr))

        off = k_max - (n - 1)
        if off >= 0:
            mask = (h_bin[:n-off] == 1.0) & (h_bin[off:] == 0.0)
        else:
            a = -off
            mask = (h_bin[a:] == 1.0) & (h_bin[:n-a] == 0.0)

        contrib = np.where(mask)[0]

        if len(contrib) > 0 and rng.random() < 0.6:
            if off >= 0:
                i = contrib[rng.randint(len(contrib))]
                j = i + off
            else:
                a = -off
                i = contrib[rng.randint(len(contrib))] + a
                j = contrib[rng.randint(len(contrib))]
            # Swap i (a 1) with a random zero
            j_new = zeros[rng.randint(len(zeros))]
            if j_new == i:
                j_new = zeros[rng.randint(len(zeros))]
                if j_new == i:
                    continue
        else:
            i = ones[rng.randint(len(ones))]
            j_new = zeros[rng.randint(len(zeros))]
            if i == j_new:
                continue

        h_bin[i] = 0.0
        h_bin[j_new] = 1.0
        new_corr = compute_corr(h_bin)
        new_c5 = float(np.max(new_corr) * dx)
        delta = new_c5 - curr_c5

        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / max(T, 1e-10))):
            ones[np.where(ones == i)[0][0]] = j_new
            zeros[np.where(zeros == j_new)[0][0]] = i
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_bin.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h_bin[i] = 1.0
            h_bin[j_new] = 0.0

        if no_improve > 10000:
            break

    return best_h, best_c5

        h_rand = np.zeros(best_n)
        h_rand[rng.choice(best_n, best_n // 2, replace=False)] = 1.0
        alt_starts.append(h_rand)
        alt_starts.append(project_feasible(np.roll(best_h, 5), best_n))
        alt_starts.append(project_feasible(np.roll(best_h, 13), best_n))

        for h_alt in alt_starts:
            if remaining_time() < 30:
                break
            h_opt = h_alt
            for beta in [5000, 50000, 500000]:
                if remaining_time() < 15:
                    break
                h_opt = slsqp_opt(h_opt, best_n, beta, maxiter=1000)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5 - 1e-15:
                h_opt, c5_opt = top_k_equalize(h_opt, best_n, top_k=8, max_iters=500, time_limit=20)
            if c5_opt < best_c5 - 1e-15:
                h_opt, c5_opt = subgradient_zigzag(h_opt, best_n, max_iters=3000, time_limit=20)
            if c5_opt < best_c5 - 1e-15:
                h_opt, c5_opt = pair_descent(h_opt, best_n, max_sweeps=50, time_limit=15, rng=rng)
            if c5_opt < best_c5 - 1e-15:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                h = h_opt.copy()

    # Phase 9: Final intensive refinement
    n = best_n
    h = best_h.copy()
    for beta in [5000000, 20000000, 80000000]:
        if remaining_time() < 10:
            break
        h_new = slsqp_opt(h, n, beta, maxiter=3000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    if remaining_time() > 15:
        h_eq, c5_eq = top_k_equalize(h, n, top_k=20, max_iters=1500, time_limit=25)
        if c5_eq < best_c5 - 1e-15:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    if remaining_time() > 15:
        h_pd, c5_pd = pair_descent(best_h, n, max_sweeps=200, time_limit=25, rng=rng)
        if c5_pd < best_c5 - 1e-15:
            best_c5 = c5_pd
            best_h = h_pd.copy()

    if remaining_time() > 20:
        h_sg, c5_sg = subgradient_zigzag(best_h, n, max_iters=8000, time_limit=min(50, int(remaining_time()*0.5)))
        if c5_sg < best_c5 - 1e-15:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Final SLSQP polish
    if remaining_time() > 10:
        h_final = slsqp_opt(best_h, n, 80000000, maxiter=2000)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-15:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Validate with evaluator
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
