                    h_bin[i] = 1.0
                    h_bin[j] = 0.0
                    delta = c_new - best_c5
                    if delta < best_delta:
                        best_delta = delta
                        best_pair = (i, j)
            if best_pair is not None:
                i, j = best_pair
                h_bin[i] = 0.0
                h_bin[j] = 1.0
                best_c5 = compute_c5(h_bin)
                best_h = h_bin.copy()
                improved = True
        else:
            n_trials = min(300, n_pairs)
            best_delta = 1e-10
            best_pair = None
            for _ in range(n_trials):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_bin[i] = 0.0
                h_bin[j] = 1.0
                c_new = compute_c5(h_bin)
                h_bin[i] = 1.0
                h_bin[j] = 0.0
                delta = c_new - best_c5
                if delta < best_delta:
                    best_delta = delta
                    best_pair = (i, j)
            if best_pair is not None:
                i, j = best_pair
                h_bin[i] = 0.0
                h_bin[j] = 1.0
                best_c5 = compute_c5(h_bin)
                best_h = h_bin.copy()
                improved = True
        if not improved:
            no_improve_rounds += 1
            if no_improve_rounds > 30:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            best_c5 = compute_c5(h_bin)
    return best_h, best_c5


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def golden_ratio_h(n, offset=0.0):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    for i in range(n):
        v = (i + offset) * phi
        h[i] = 1.0 if v - int(v) < 0.5 else 0.0
    return project_feasible(h, n)


def multi_freq_h(n, freqs=None, phases=None, rng=None):
    if rng is None:
        rng = np.random.RandomState(123)
    if freqs is None:
        freqs = [3, 7, 11, 19, 31]
    if phases is None:
        phases = rng.uniform(0, 2 * np.pi, len(freqs))
    x = np.arange(n)
    s = np.zeros(n)
    for f, p in zip(freqs, phases):
        s += np.sin(2.0 * np.pi * f * x / n + p)
    h = (s > 0).astype(float)
    return project_feasible(h, n)


def near_binary_perturb(h_bin, n, num_frac, rng):
    h = h_bin.copy()
    n_frac = int(num_frac)
    if n_frac > 0:
        frac_positions = rng.choice(n, min(n_frac, n), replace=False)
        for p in frac_positions:
            h[p] = rng.uniform(0.1, 0.9)
    return project_feasible(h, n)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _CUR_N
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = golden_ratio_h(640)
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick L-BFGS-B at parent resolution
    h = best_h.copy()
    for beta in [2000, 20000, 200000, 2000000]:
        if remaining_time() < 25:
            break
        h_new = lbfgsb_opt(h, best_n, beta, maxiter=3000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Top-k equalization
    if remaining_time() > 20:
        for tk in [4, 8, 16]:
            if remaining_time() < 15:
                break
            h_eq, c5_eq = top_k_equalize(h, best_n, top_k=tk, max_iters=800, time_limit=50)
            if c5_eq < best_c5 - 1e-15:
                best_c5 = c5_eq
                best_h = h_eq.copy()
                h = h_eq.copy()

    # Phase 3: Subgradient zigzag
    if remaining_time() > 30:
        h_sg, c5_sg = subgradient_zigzag(h, best_n, max_iters=8000, time_limit=90)
        if c5_sg < best_c5 - 1e-15:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 4: Binary SA at parent resolution
    if remaining_time() > 80:
        h_bin = (best_h > 0.5).astype(float)
        sa_time = min(120, int(remaining_time() * 0.28))
        h_b, c5_b = binary_sa(h_bin, best_n, time_limit=sa_time, rng=rng)
        if c5_b < best_c5 - 1e-15:
            best_c5 = c5_b
            best_h = h_b.copy()
        # Continuous refinement of binary result
        h_bc = h_b.copy()
        for beta in [5000, 50000, 500000, 5000000]:
            if remaining_time() < 15:
                break
            h_bc = lbfgsb_opt(h_bc, best_n, beta, maxiter=2000)
        c5_bc = compute_c5(h_bc)
        h_bc, c5_bc = top_k_equalize(h_bc, best_n, top_k=8, max_iters=600, time_limit=40)
        h_bc, c5_bc = subgradient_zigzag(h_bc, best_n, max_iters=5000, time_limit=40)
        if c5_bc < best_c5 - 1e-15:
            best_c5 = c5_bc
            best_h = h_bc.copy()
            h = h_bc.copy()

    # Phase 5: Binary greedy
    if remaining_time() > 50:
        h_bin = (best_h > 0.5).astype(float)
        h_g, c5_g = binary_greedy(h_bin, best_n, time_limit=min(50, int(remaining_time()*0.25)), rng=rng)
        h_gc = h_g.copy()
        for beta in [50000, 500000]:
            if remaining_time() < 15:
                break
            h_gc = lbfgsb_opt(h_gc, best_n, beta, maxiter=1500)
        c5_gc = compute_c5(h_gc)
        h_gc, c5_gc = top_k_equalize(h_gc, best_n, top_k=8, max_iters=500, time_limit=30)
        h_gc, c5_gc = subgradient_zigzag(h_gc, best_n, max_iters=4000, time_limit=30)
        if c5_gc < best_c5 - 1e-15:
            best_c5 = c5_gc
            best_h = h_gc.copy()
            h = h_gc.copy()

    # Phase 6: Multi-resolution optimization
    n_candidates = [int(best_n * 1.2), int(best_n * 1.5), int(best_n * 2)]
    for n_t in n_candidates:
        if remaining_time() < 50:
            break
        if n_t == best_n or n_t > 2500:
            continue
        n_t = int(n_t)
        # Ensure n_t allows even split
        if n_t % 2 == 1:
            n_t += 1
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        h_opt = h_feas
        for beta in [5000, 50000, 500000]:
            if remaining_time() < 20:
                break
            h_opt = lbfgsb_opt(h_opt, n_t, beta, maxiter=2000)
        c5_opt = compute_c5(h_opt)
        h_opt, c5_opt = top_k_equalize(h_opt, n_t, top_k=8, max_iters=500, time_limit=40)
        h_opt, c5_opt = subgradient_zigzag(h_opt, n_t, max_iters=4000, time_limit=35)
        # Super-resolution: try SLSQP for final polish
        if remaining_time() > 20:
            h_s = h_opt.copy()
            for beta in [2000000, 10000000]:
                if remaining_time() < 12:
                    break
                h_s = slsqp_opt(h_s, n_t, beta, maxiter=1500)
            c5_s = compute_c5(h_s)
            if c5_s < c5_opt:
                h_opt = h_s
                c5_opt = c5_s
        if c5_opt < best_c5 - 1e-15:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t
            h = h_opt.copy()

    # Phase 7: Golden ratio and structured starts
    if remaining_time() > 60:
        starts = []
        starts.append(golden_ratio_h(best_n, offset=0.0))
        starts.append(golden_ratio_h(best_n, offset=0.37))
        starts.append(multi_freq_h(best_n, freqs=[3, 7, 13], rng=rng))
        starts.append(multi_freq_h(best_n, freqs=[5, 11, 17], rng=rng))
        # Near-binary perturbation of current best
        h_bin_cur = (best_h > 0.5).astype(float)
        starts.append(near_binary_perturb(h_bin_cur, best_n, max(5, best_n // 40), rng))
        starts.append(near_binary_perturb(h_bin_cur, best_n, max(10, best_n // 20), rng))

        for h_alt in starts:
            if remaining_time() < 25:
                break
            h_opt = h_alt.copy()
            for beta in [2000, 20000, 200000]:
                if remaining_time() < 12:
                    break
                h_opt = lbfgsb_opt(h_opt, best_n, beta, maxiter=1500)
            c5_opt = compute_c5(h_opt)
            h_opt, c5_opt = top_k_equalize(h_opt, best_n, top_k=8, max_iters=400, time_limit=25)
            h_opt, c5_opt = subgradient_zigzag(h_opt, best_n, max_iters=3000, time_limit=20)
            if c5_opt < best_c5 - 1e-15:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                h = h_opt.copy()

    # Phase 8: Final intensive refinement
    n = best_n
    h = best_h.copy()
    for beta in [100000, 1000000, 5000000, 20000000]:
        if remaining_time() < 10:
            break
        h_new = lbfgsb_opt(h, n, beta, maxiter=3000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    if remaining_time() > 15:
        h_eq, c5_eq = top_k_equalize(h, n, top_k=12, max_iters=800, time_limit=min(35, int(remaining_time()*0.6)))
        if c5_eq < best_c5 - 1e-15:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    if remaining_time() > 20:
        h_sg, c5_sg = subgradient_zigzag(h, n, max_iters=6000, time_limit=min(50, int(remaining_time()*0.5)))
        if c5_sg < best_c5 - 1e-15:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 9: SLSQP final polish with high beta
    if remaining_time() > 12:
        h_final = slsqp_opt(best_h, n, 50000000, maxiter=2000)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-15:
            best_c5 = c5_final
            best_h = h_final.copy()

    if remaining_time() > 10:
        h_eq2, c5_eq2 = top_k_equalize(best_h, n, top_k=6, max_iters=500, time_limit=20)
        if c5_eq2 < best_c5 - 1e-15:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()

    # Final feasibility check
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, n)
