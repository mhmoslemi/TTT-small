        # So we need sum lambda_k G_proj[k] = 0 (the mu*ones component is free)
        # This is: find lambda >= 0, sum lambda = 1, such that G_proj.T @ lambda = 0
        # i.e., lambda is in null(G_proj.T) with lambda >= 0, sum = 1
        # Use SVD approach
        U, s, Vt = np.linalg.svd(G_proj, full_matrices=False)
        # Find null space of G_proj.T (which is right null space of G_proj)
        # null space dimension = K - rank
        # The last column of Vt gives a null direction (if rank < K)
        # Better: find lambda in null(G_proj.T) using least squares
        # min ||G_proj.T @ lambda||^2 s.t. lambda >= 0, sum lambda = 1
        # Use iterative projection
        lam = np.ones(K) / K
        for _ in range(100):
            resid = G_proj.T @ lam
            if np.linalg.norm(resid) < 1e-8:
                break
            lam -= 0.5 * resid * (1.0 / K)  # step in direction of decreasing resid
            lam = np.maximum(lam, 0)
            lam_sum = np.sum(lam)
            if lam_sum < 1e-10:
                lam = np.ones(K) / K
            else:
                lam /= lam_sum
        # Direction: -G_proj.T @ lam (should be ~ parallel to ones)
        direction = -G_proj.T @ lam
        # Remove component along ones to maintain sum constraint
        direction = direction - np.mean(direction) * ones
        dn = np.linalg.norm(direction)
        if dn < 1e-12:
            break
        direction /= dn
        # Line search
        improved = False
        for step in [0.01, 0.005, 0.02, 0.002, 0.001, 0.04, 0.008, 0.003]:
            h_try = project_feasible(h + step * direction, n)
            c_try = compute_c5(h_try)
            if c_try < best_c5 - 1e-14:
                h = h_try
                best_c5 = c_try
                improved = True
                break
        if not improved:
            # Try the old equalization: average gradient
            g_avg = np.zeros(n)
            for k in idx:
                g_avg += grad_corr_k_vec(h, k, n)
            g_avg /= K
            g_avg = g_avg - np.mean(g_avg) * ones
            gn = np.linalg.norm(g_avg)
            if gn < 1e-12:
                break
            for step in [0.01, 0.005, 0.02, 0.003, 0.001]:
                h_try = project_feasible(h - step * g_avg / gn, n)
                c_try = compute_c5(h_try)
                if c_try < best_c5 - 1e-14:
                    h = h_try
                    best_c5 = c_try
                    improved = True
                    break
            if not improved:
                break
    return h, best_c5

# Subgradient descent
def subgradient_descent(h, n, time_limit, K, beta):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(-corr)[:K]
        c_vals = corr[idx]
        sh = c_vals - c_vals.max()
        cl = np.clip(beta * sh, -60.0, 60.0)
        ex = np.exp(cl)
        w = ex / ex.sum()
        g = np.zeros(n)
        for i in range(K):
            if w[i] < 1e-12:
                continue
            g += w[i] * grad_corr_k_vec(h, idx[i], n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        progress = min(1.0, it / 300.0)
        step = 0.03 * n / gn / (1.0 + 0.03 * it)
        step = min(step, 0.1)
        h_new = project_feasible(h - step * g, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            for factor in [0.5, 0.25, 0.1, 0.05]:
                h_try = project_feasible(h - step * factor * g, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    break
        it += 1
    return best_h, best_c5

# Cutting plane LP
def cutting_plane_lp(h, n, K, delta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    idx = np.argsort(-corr)[:K]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub_list = []
    b_ub_list = []
    for k in idx:
        g = grad_corr_k_vec(h, k, n)
        fk = corr[k]
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_ub_list.append(row)
        b_ub_list.append(float(-fk + np.dot(g, h)))
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    I = np.eye(n)
    A_trust = np.vstack([np.hstack([I, np.zeros((n, 1))]),
                         np.hstack([-I, np.zeros((n, 1))])])
    b_trust = np.concatenate([h + delta, h + delta])
    A_ub = np.vstack([A_ub, A_trust])
    b_ub = np.concatenate([b_ub, b_trust])
    bounds_lp = [(0, 1)] * n + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds_lp, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n].copy(), float(res.x[-1])
    except Exception:
        pass
    return h.copy(), c_max

# Active-set min-max with t variable
_AT_N = 0
_AT_KLIST = None

def at_ineq_fun(v):
    h = v[:_AT_N]
    t = v[-1]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return t * (_AT_N / 2.0) - corr[_AT_KLIST]

def at_ineq_jac(v):
    h = v[:_AT_N]
    n = _AT_N
    K_list = _AT_KLIST
    J = np.zeros((len(K_list), n + 1))
    for i, k in enumerate(K_list):
        offset = k - (n - 1)
        if offset >= 0:
            np_ = n - offset
            J[i, offset:] -= (1.0 - h[:np_])
            J[i, :np_] += h[offset:]
        else:
            a = -offset
            np_ = n - a
            J[i, :np_] -= (1.0 - h[a:])
            J[i, a:] += h[:np_]
        J[i, -1] = n / 2.0
    return J

def at_t_obj(v):
    return v[-1]

def at_t_jac(v):
    g = np.zeros(_AT_N + 1)
    g[-1] = 1.0
    return g

def at_eq_fun(v):
    return float(np.sum(v[:_AT_N]) - _AT_N / 2.0)

def at_eq_jac(v):
    g = np.zeros(_AT_N + 1)
    g[:_AT_N] = 1.0
    return g

def solve_active_set(h, n, K_list, maxiter):
    global _AT_N, _AT_KLIST
    _AT_N = n
    _AT_KLIST = np.array(K_list)
    t0 = compute_c5(h)
    v0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'ineq', 'fun': at_ineq_fun, 'jac': at_ineq_jac},
        {'type': 'eq', 'fun': at_eq_fun, 'jac': at_eq_jac}
    ]
    try:
        res = minimize(at_t_obj, v0, jac=at_t_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x[:n], n)
            c5_new = compute_c5(h_new)
            if c5_new <= compute_c5(h) + 1e-12:
                return h_new, c5_new
    except Exception:
        pass
    return h.copy(), compute_c5(h)

# Main optimization at a given resolution
def optimize_at_resolution(h, n, time_budget, rng):
    best_h = h.copy()
    best_c5 = compute_c5(h)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

    # Phase 1: Subgradient for basin finding
    if remaining() > 20:
        t_sg = min(20.0, time_budget * 0.06)
        for K in [15, 25, 40]:
            if remaining() < 10:
                break
            h_sg, c5_sg = subgradient_descent(h, n, min(8.0, remaining() * 0.1), K, 50.0)
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg.copy()
                h = h_sg.copy()

    # Phase 2: Sigmoid SLSQP with increasing beta
    beta_list = [20, 50, 100, 300, 1000, 3000, 10000]
    K_list_s = [20, 30, 40, 50, 60, 80, 100]
    for bi, beta in enumerate(beta_list):
        if remaining() < 15:
            break
        K = K_list_s[min(bi, len(K_list_s) - 1)]
        maxit = 800 if beta < 100 else 500
        h_new = solve_sigmoid_slsqp(h, n, beta, K, maxit)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 3: Pixel-space SLSQP
    for beta in [100, 500, 2000, 5000]:
        if remaining() < 10:
            break
        K = min(30 + 10 * beta // 500, 80)
        h_new = solve_pixel_slsqp(h, n, beta, K, 600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 4: KKT equalization
    if remaining() > 15:
        for K_eq in [5, 8, 10, 12, 15, 20]:
            if remaining() < 8:
                break
            h_eq, c5_eq = kkt_equalize(h, n, K_eq, 200)
            if c5_eq < best_c5:
                best_c5 = c5_eq
                best_h = h_eq.copy()
                h = h_eq.copy()

    # Phase 5: Active-set min-max
    if remaining() > 20:
        for K_a in [5, 8, 12, 16, 20, 30]:
            if remaining() < 10:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            idx_a = np.argsort(-corr)[:K_a]
            h_am, c5_am = solve_active_set(h, n, idx_a.tolist(), 400)
            if c5_am < best_c5:
                best_c5 = c5_am
                best_h = h_am.copy()
                h = h_am.copy()

    # Phase 6: Cutting plane LP
    if remaining() > 30:
        for it in range(150):
            if remaining() < 5:
                break
            h_lp, t_lp = cutting_plane_lp(h, n, K=25, delta=0.03)
            c5_cur = compute_c5(h)
            if t_lp < c5_cur - 1e-10:
                best_line = compute_c5(h)
                for alpha in [1.0, 0.7, 0.5, 0.8, 0.3]:
                    h_try = project_feasible(h + alpha * (h_lp - h), n)
                    c_try = compute_c5(h_try)
                    if c_try < best_line:
                        best_line = c_try
                        h = h_try.copy()
                    else:
                        break
                if compute_c5(h) < best_c5:
                    best_c5 = compute_c5(h)
                    best_h = h.copy()
            elif it > 60:
                break

    # Phase 7: Final SLSQP polish
    for beta in [5000, 15000, 50000]:
        if remaining() < 8:
            break
        h = solve_sigmoid_slsqp(h, n, beta, 60, 600)
        h = solve_pixel_slsqp(h, n, beta, 60, 400)
        c5_t = compute_c5(h)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h.copy()

    # Phase 8: Final KKT equalization
    if remaining() > 10:
        h_eq, c5_eq = kkt_equalize(best_h, n, 15, 300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 512
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)

    # Strategy: multi-resolution with progressive refinement
    # Start at parent resolution, then try higher resolutions
    
    # Main optimization at parent resolution
    if remaining() > 60:
        h_opt, c5_opt = optimize_at_resolution(best_h, best_n, remaining() * 0.45, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Try higher resolution: n=1024
    n_high = 1024
    if remaining() > 120:
        h_interp = interp_h(best_h, best_n, n_high)
        h_interp = project_feasible(h_interp, n_high)
        c5_interp = compute_c5(h_interp)
        if c5_interp <= best_c5 + 0.002:
            h_h, c5_h = optimize_at_resolution(h_interp, n_high, remaining() * 0.35, rng)
            if c5_h < best_c5:
                best_c5 = c5_h
                best_h = h_h.copy()
                best_n = n_high

    # Try even higher resolution: n=2048
    n_vhigh = 2048
    if remaining() > 150:
        h_vh = interp_h(best_h, best_n, n_vhigh)
        h_vh = project_feasible(h_vh, n_vhigh)
        c5_vh = compute_c5(h_vh)
        if c5_vh <= best_c5 + 0.001:
            h_vh_opt, c5_vh_opt = optimize_at_resolution(h_vh, n_vhigh, remaining() * 0.3, rng)
            if c5_vh_opt < best_c5:
                best_c5 = c5_vh_opt
                best_h = h_vh_opt.copy()
                best_n = n_vhigh

    # Final refinement cycle at best resolution
    if remaining() > 30:
        h_final, c5_final = optimize_at_resolution(best_h, best_n, remaining() * 0.5, rng)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Tiny final steps
    if remaining() > 10:
        h_f = best_h.copy()
        for it in range(100):
            if remaining() < 3:
                break
            corr_f = np.correlate(h_f, 1.0 - h_f, mode='full')
            idx_f = np.argsort(-corr_f)[:12]
            g_f = np.zeros(best_n)
            for k in idx_f:
                g_f += grad_corr_k_vec(h_f, k, best_n)
            g_f /= len(idx_f)
            ones = np.ones(best_n)
            g_f = g_f - np.mean(g_f) * ones
            gn_f = np.linalg.norm(g_f)
            if gn_f < 1e-14:
                break
            h_f = g_f  # temp
            improved = False
            for step in [0.003, 0.001, 0.005, 0.002, 0.0005]:
                h_try = project_feasible(best_h - step * g_f / gn_f, best_n)
                c_try = compute_c5(h_try)
                if c_try < best_c5:
                    best_c5 = c_try
                    best_h = h_try.copy()
                    improved = True
                    break
            if not improved:
                break

    # Final feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
