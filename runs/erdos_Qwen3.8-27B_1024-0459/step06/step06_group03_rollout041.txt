def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def grad_corr_k_vec(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        np_ = n - offset
        g[offset:] += (1.0 - h[:np_])
        g[:np_] -= h[offset:]
    else:
        a = -offset
        np_ = n - a
        g[:np_] += (1.0 - h[a:])
        g[a:] -= h[:np_]
    return g


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def smmax_objective(h):
    n = _G_N
    corr = np.correlate(h, 1.0 - h, mode='full')
    beta = _G_BETA
    K = _G_K
    m = corr.max()
    idx = np.argsort(-corr)[:K]
    c = corr[idx]
    sh = c - c.max()
    cl = np.clip(beta * sh, -50.0, 50.0)
    ex = np.exp(cl)
    val = (c.max() + np.log(ex.sum()) / beta) * (2.0 / n)
    return val


def smmax_gradient(h):
    n = _G_N
    beta = _G_BETA
    K = _G_K
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(-corr)[:K]
    c = corr[idx]
    sh = c - c.max()
    cl = np.clip(beta * sh, -50.0, 50.0)
    ex = np.exp(cl)
    w = ex / ex.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for i in range(K):
        if w[i] < 1e-15:
            continue
        wk = w[i] * dx
        k = idx[i]
        offset = k - (n - 1)
        if offset >= 0:
            np_ = n - offset
            g[offset:] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[offset:]
        else:
            a = -offset
            np_ = n - a
            g[:np_] += wk * (1.0 - h[a:])
            g[a:] -= wk * h[:np_]
    return g


def sum_eq_fun(h):
    return np.sum(h) - _G_N / 2.0


def sum_eq_jac(h):
    return np.ones(_G_N)


def vec_ineq_fun(v):
    h = v[:_G_N]
    t = v[-1]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[_G_KLIST] - t * (_G_N / 2.0)


def vec_ineq_jac(v):
    h = v[:_G_N]
    n = _G_N
    K_list = _G_KLIST
    J = np.zeros((len(K_list), n + 1))
    for i, k in enumerate(K_list):
        offset = k - (n - 1)
        if offset >= 0:
            np_ = n - offset
            J[i, offset:] += (1.0 - h[:np_])
            J[i, :np_] -= h[offset:]
        else:
            a = -offset
            np_ = n - a
            J[i, :np_] += (1.0 - h[a:])
            J[i, a:] -= h[:np_]
        J[i, -1] = -n / 2.0
    return J


def obj_t_fun(v):
    return v[-1]


def obj_t_jac(v):
    g = np.zeros(_G_N + 1)
    g[-1] = 1.0
    return g


def sum_eq_fun2(v):
    return np.sum(v[:_G_N]) - _G_N / 2.0


def sum_eq_jac2(v):
    g = np.zeros(_G_N + 1)
    g[:_G_N] = 1.0
    return g


def tmin_ineq_fun_global(v):
    return -vec_ineq_fun(v)


def tmin_ineq_jac_global(v):
    return -vec_ineq_jac(v)


def slsqop_smmax(h, n, beta, K, maxiter):
    global _G_N, _G_BETA, _G_K
    _G_N = n
    _G_BETA = beta
    _G_K = K
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}]
    try:
        res = minimize(smmax_objective, h, jac=smmax_gradient, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            if compute_c5(h_new) <= compute_c5(h) + 1e-10:
                return h_new
    except Exception:
        pass
    return h.copy()


def slsqop_tminmax(h, n, K_list, maxiter):
    global _G_N, _G_KLIST
    _G_N = n
    _G_KLIST = K_list
    t0 = compute_c5(h) * (n / 2.0)
    v0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'ineq', 'fun': tmin_ineq_fun_global, 'jac': tmin_ineq_jac_global},
        {'type': 'eq', 'fun': sum_eq_fun2, 'jac': sum_eq_jac2}
    ]
    try:
        res = minimize(obj_t_fun, v0, jac=obj_t_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x[:n], n)
            c5_new = compute_c5(h_new)
            if c5_new <= compute_c5(h) + 1e-10:
                return h_new, c5_new
    except Exception:
        pass
    return h.copy(), compute_c5(h)


def subgradient_momentum(h, n, time_limit, rng):
    h = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    it = 0
    mu = 0.9
    while time.time() - start < time_limit:
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:25]
        g = np.zeros(n)
        for k in idx:
            g += grad_corr_k_vec(h, k, n)
        g /= len(idx)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        progress = min(1.0, it / 800.0)
        step = (0.06 * n / gn) / (1.0 + 0.1 * progress)
        step = min(step, 0.2)
        v = mu * v - step * g
        h_new = project_feasible(h + v, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            v = 0.5 * v
            h_try = project_feasible(h + v, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try.copy()
        it += 1
    return best_h, best_c5


def greedy_swap_local_search(h, n, time_limit, max_rounds=5000):
    """
    Greedy local search: repeatedly find the best single-pair swap
    that reduces the maximum cross-correlation.
    """
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    rounds_since_improve = 0

    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break

        corr = compute_corr(h)
        c_max_rot = np.max(corr) * (2.0 / n)
        if c_max_rot < best_c5:
            best_c5 = c_max_rot

        # Find top lags
        K_check = min(30, 2 * n - 1)
        idx_top = np.argsort(-corr)[:K_check]
        c_top = corr[idx_top]

        # Focus on the worst lag
        k_star = idx_top[0]
        grad_k = grad_corr_k_vec(h, k_star, n)

        # Find best pair: increase h[a], decrease h[b]
        # Effect on c_{k_star}: delta * (grad[a] - grad[b])
        # We want grad[b] > grad[a] to reduce c_{k_star}
        # Also check effect on other top lags

        gt = np.argsort(grad_k)  # ascending: gt[0] has lowest grad (best to increase)
        gt_desc = np.argsort(-grad_k)  # descending: gt_desc[0] has highest grad (best to decrease)

        best_delta_c = 0
        best_a = -1
        best_b = -1
        best_other_max = 0

        # Try top candidates
        candidates_a = gt[:20]
        candidates_b = gt_desc[:20]

        for a in candidates_a:
            for b in candidates_b:
                if a == b:
                    continue
                if h[a] >= 1.0 - 1e-9 or h[b] <= 1e-9:
                    continue
                # Delta in c_{k_star}
                dc = grad_k[a] - grad_k[b]
                if dc >= 0:
                    continue  # would increase c_{k_star}
                # Check effect on other top lags (approximate)
                ok = True
                for k2 in idx_top[1:min(10, len(idx_top))]:
                    g2 = grad_corr_k_vec(h, k2, n)
                    dc2 = g2[a] - g2[b]
                    if corr[k2] + dc2 > corr[k_star] + dc + 0.5 * abs(dc):
                        # This swap might make k2 the new max
                        ok = False
                        break
                if not ok:
                    continue
                if dc < best_delta_c:
                    best_delta_c = dc
                    best_a = a
                    best_b = b

        if best_a < 0:
            # No good pair found; try single variable adjustment
            # Decrease h at position with highest grad, increase at lowest
            a = gt[0]
            b = gt_desc[0]
            if a == b or h[a] >= 1.0 - 1e-9 or h[b] <= 1e-9:
                rounds_since_improve += 1
                if rounds_since_improve > 200:
                    break
                continue

        # Line search: find best step size
        improved = False
        for step in [0.01, 0.005, 0.02, 0.003, 0.015, 0.008, 0.05, 0.002]:
            h_try = h.copy()
            h_try[best_a] += step
            h_try[best_b] -= step
            h_try = project_feasible(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-12:
                h = h_try
                best_c5 = c5_try
                rounds_since_improve = 0
                improved = True
                break

        if not improved:
            rounds_since_improve += 1
            if rounds_since_improve > 300:
                break

    return h, best_c5


def targeted_descent(h, n, time_limit, K=15):
    """
    More targeted descent: focus on the top K lags simultaneously.
    Use a projected gradient step toward reducing all top lags.
    """
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    v = np.zeros(n)
    mu = 0.85
    it = 0

    while time.time() - start < time_limit:
        rem = _BUDGET - (time.time() - _T0)
        if rem < 3:
            break

        corr = compute_corr(h)
        idx = np.argsort(-corr)[:K]
        c_vals = corr[idx]
        c_max = c_vals[0]

        # Weighted gradient: weight by how close each lag is to the max
        sh = c_vals - c_max
        beta_local = 50.0
        cl = np.clip(beta_local * sh, -40.0, 40.0)
        ex = np.exp(cl)
        w = ex / ex.sum()

        g = np.zeros(n)
        for i in range(K):
            if w[i] < 1e-10:
                continue
            g += w[i] * grad_corr_k_vec(h, idx[i], n)

        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break

        progress = min(1.0, it / 1000.0)
        step = (0.05 * n / gn) / (1.0 + 0.05 * it)
        step = min(step, 0.1)

        v = mu * v - step * g
        h_new = project_feasible(h + v, n)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5:
            best_c5 = c5_new
            h = h_new.copy()
        else:
            v = 0.5 * v
            h_try = project_feasible(h + v, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                h = h_try.copy()
            else:
                v = np.zeros(n)

        it += 1
    return h, best_c5


def binary_snap_and_search(h, n, time_limit):
    """
    Snap to binary, do local search on binary space, then unfreeze.
    """
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()

    # Snap to binary
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    # Ensure exactly n/2 ones
    num_ones = np.sum(h_bin)
    if num_ones != n // 2:
        # Adjust
        if num_ones > n // 2:
            # Remove excess ones (those with lowest h value)
            one_indices = np.where(h_bin == 1.0)[0]
            order = np.argsort(h[one_indices])
            for idx in order[:int(num_ones - n // 2)]:
                h_bin[one_indices[idx]] = 0.0
        else:
            zero_indices = np.where(h_bin == 0.0)[0]
            order = np.argsort(-h[zero_indices])
            for idx in order[:int(n // 2 - num_ones)]:
                h_bin[zero_indices[idx]] = 1.0

    h_bin = project_feasible(h_bin, n)
    c5_bin = compute_c5(h_bin)

    # Binary local search: try flipping pairs
    improved_any = False
    for _ in range(500):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 5:
            break

        corr = compute_corr(h_bin)
        idx_top = np.argsort(-corr)[:20]

        best_flip_gain = 0
        best_flip_a = -1
        best_flip_b = -1

        # Try flipping: h[a] 1->0, h[b] 0->1 (or vice versa)
        one_pos = np.where(h_bin == 1.0)[0]
        zero_pos = np.where(h_bin == 0.0)[0]

        # Sample candidates for speed
        if len(one_pos) > 30:
            np.random.shuffle(one_pos)
            one_pos = one_pos[:30]
        if len(zero_pos) > 30:
            np.random.shuffle(zero_pos)
            zero_pos = zero_pos[:30]

        for a in one_pos:
            for b in zero_pos:
                h_try = h_bin.copy()
                h_try[a] = 0.0
                h_try[b] = 1.0
                corr_try = compute_corr(h_try)
                max_try = np.max(corr_try)
                gain = np.max(corr) - max_try
                if gain > best_flip_gain:
                    best_flip_gain = gain
                    best_flip_a = a
                    best_flip_b = b

        if best_flip_a >= 0 and best_flip_gain > 0:
            h_bin[best_flip_a] = 0.0
            h_bin[best_flip_b] = 1.0
            c5_new = compute_c5(h_bin)
            if c5_new < best_c5:
                best_c5 = c5_new
                improved_any = True
        else:
            # Try also 0->1, 1->0 direction
            for a in zero_pos[:20]:
                for b in one_pos[:20]:
                    h_try = h_bin.copy()
                    h_try[a] = 1.0
                    h_try[b] = 0.0
                    corr_try = compute_corr(h_try)
                    max_try = np.max(corr_try)
                    gain = np.max(corr) - max_try
                    if gain > best_flip_gain:
                        best_flip_gain = gain
                        best_flip_a = a
                        best_flip_b = b

            if best_flip_a >= 0 and best_flip_gain > 0:
                h_bin[best_flip_a] = 1.0
                h_bin[best_flip_b] = .0
                c5_new = compute_c5(h_bin)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    improved_any = True
            else:
                break

    # Unfreeze: blend binary solution with original
    h_blend = project_feasible(0.7 * h_bin + 0.3 * h, n)
    c5_blend = compute_c5(h_blend)

    if c5_blend < best_c5:
        best_c5 = c5_blend
        return h_blend, best_c5
    elif c5_bin < best_c5:
        best_c5 = c5_bin
        return h_bin, best_c5

    # Try continuous optimization from binary starting point
    if _BUDGET - (time.time() - _T0) > 30:
        h_cont = h_bin.copy()
        h_cont = slsqop_smmax(h_cont, n, beta=100, K=30, maxiter=200)
        c5_cont = compute_c5(h_cont)
        if c5_cont < best_c5:
            best_c5 = c5_cont
            return h_cont, best_c5

    return h, best_c5


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def multi_freq_init(n, rng):
    """Create diverse initializations using different frequency structures."""
    inits = []

    # Sawtooth-like
    x = np.linspace(0, 2, n)
    h1 = np.where((x * 3.0) % 1.0 < 0.5, 1.0, 0.0)
    h1 = project_feasible(h1, n)
    inits.append(h1)

    # Multi-scale
    h2 = np.zeros(n)
    for freq, amp in [(1.5, 0.8), (3.0, 0.5), (6.0, 0.3)]:
        h2 += amp * np.sin(2 * np.pi * freq * x / 2.0 + rng.uniform(0, 2 * np.pi))
    h2 = np.clip(h2, 0, 1)
    h2 = project_feasible(h2, n)
    inits.append(h2)

    # Block structure with varying sizes
    h3 = np.zeros(n)
    pos = 0
    while pos < n:
        run_1 = rng.randint(5, 30)
        run_0 = rng.randint(5, 30)
        h3[pos:pos + run_1] = 1.0
        pos += run_1
        h3[pos:pos + run_0] = 0.0
        pos += run_0
    h3 = project_feasible(h3, n)
    inits.append(h3)

    # Random with smoothing
    h4 = rng.rand(n)
    # Apply moving average to create structure
    k = 7
    h4 = np.convolve(h4, np.ones(k) / k, mode='same')
    h4 = np.clip(h4, 0, 1)
    h4 = project_feasible(h4, n)
    inits.append(h4)

    return inits


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _G_N, _G_BETA, _G_K, _G_KLIST
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 512
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # === Phase 1: Quick subgradient with momentum ===
    if remaining() > 40:
        t1 = min(40.0, remaining() * 0.06)
        h1, c5_1 = subgradient_momentum(h, n, t1, rng)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()
            h = h1.copy()

    # === Phase 2: SLSQP smooth-max stages ===
    for beta in [20, 80, 300, 1000, 4000]:
        if remaining() < 15:
            break
        maxit = 800 if beta < 100 else 500
        h = slsqop_smmax(h, n, beta, K=45, maxiter=maxit)
        c5_t = compute_c5(h)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h.copy()

    # === Phase 3: Targeted descent ===
    if remaining() > 60:
        t3 = min(60.0, remaining() * 0.1)
        h3, c5_3 = targeted_descent(h, n, t3, K=20)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
            h = h3.copy()

    # === Phase 4: Greedy swap local search (KEY NEW PHASE) ===
    if remaining() > 80:
        t4 = min(150.0, remaining() * 0.2)
        h4, c5_4 = greedy_swap_local_search(h, n, t4)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
            h = h4.copy()

    # === Phase 5: Active set min-max ===
    if remaining() > 40:
        for K_a in [8, 12, 20, 30]:
            if remaining() < 10:
                break
            corr_a = np.correlate(h, 1.0 - h, mode='full')
            idx_a = np.argsort(-corr_a)[:K_a]
            h_a, c5_a = slsqop_tminmax(h, n, idx_a, maxiter=300)
            if c5_a < best_c5:
                best_c5 = c5_a
                best_h = h_a.copy()
                h = h_a.copy()

    # === Phase 6: Binary snap and search ===
    if remaining() > 80:
        t6 = min(80.0, remaining() * 0.1)
        h6, c5_6 = binary_snap_and_search(h, n, t6)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
            h = h6.copy()

    # === Phase 7: SLSQP polish ===
    for beta in [5000, 15000, 40000]:
        if remaining() < 10:
            break
        h = slsqop_smmax(h, n, beta, K=50, maxiter=600)
        c5_t = compute_c5(h)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h.copy()

    # === Phase 8: More greedy swaps ===
    if remaining() > 60:
        t8 = min(80.0, remaining() * 0.1)
        h8, c5_8 = greedy_swap_local_search(h, n, t8)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()
            h = h8.copy()

    # === Phase 9: Try higher resolution n=1024 ===
    n_high = 1024
    if remaining() > 250:
        h_interp = interp_to(best_h, best_n, n_high)
        h_interp = project_feasible(h_interp, n_high)
        c5_interp = compute_c5(h_interp)

        if c5_interp <= best_c5 + 0.002:
            h_h = h_interp.copy()

            # Subgradient at high res
            if remaining() > 100:
                h_h, c5_h = subgradient_momentum(h_h, n_high, min(40.0, remaining() * 0.08), rng)
                if c5_h > c5_interp:
                    h_h = h_interp.copy()
                    c5_h = c5_interp

            # SLSQP stages
            for beta in [50, 300, 2000, 10000]:
                if remaining() < 20:
                    break
                h_h = slsqop_smmax(h_h, n_high, beta, K=40, maxiter=400)
                c5_h = compute_c5(h_h)
                if c5_h < best_c5:
                    best_c5 = c5_h
                    best_h = h_h.copy()

            # Targeted descent
            if remaining() > 60:
                h_h, c5_h = targeted_descent(h_h, n_high, min(50.0, remaining() * 0.12), K=20)
                if c5_h < best_c5:
                    best_c5 = c5_h
                    best_h = h_h.copy()

            # Greedy swaps at high res
            if remaining() > 60:
                h_h, c5_h = greedy_swap_local_search(h_h, n_high, min(60.0, remaining() * 0.12))
                if c5_h < best_c5:
                    best_c5 = c5_h
                    best_h = h_h.copy()

            # Active set
            if remaining() > 30:
                for K_a in [10, 20, 35]:
                    if remaining() < 10:
                        break
                    corr_h = np.correlate(h_h, 1.0 - h_h, mode='full')
                    idx_a = np.argsort(-corr_h)[:K_a]
                    h_h, c5_h = slsqop_tminmax(h_h, n_high, idx_a, maxiter=250)
                    if c5_h < best_c5:
                        best_c5 = c5_h
                        best_h = h_h.copy()

            # Polish
            for beta in [20000, 60000]:
                if remaining() < 10:
                    break
                h_h = slsqop_smmax(h_h, n_high, beta, K=55, maxiter=500)
                c5_h = compute_c5(h_h)
                if c5_h < best_c5:
                    best_c5 = c5_h
                    best_h = h_h.copy()

            if best_c5 < compute_c5(np.interp((np.arange(best_n) + 0.5) * 2.0 / best_n,
                                              (np.arange(n_high) + 0.5) * 2.0 / n_high, best_h)):
                best_n = n_high
        else:
            # Try different initialization at high res
            inits_h = multi_freq_init(n_high, rng)
            for init_h in inits_h[:2]:
                if remaining() < 80:
                    break
                c5_init = compute_c5(init_h)
                if c5_init < best_c5 + 0.05:
                    h_try = init_h.copy()
                    h_try, c5_t = subgradient_momentum(h_try, n_high, min(30.0, remaining() * 0.1), rng)
                    for beta in [100, 1000, 5000]:
                        if remaining() < 15:
                            break
                        h_try = slsqop_smmax(h_try, n_high, beta, K=40, maxiter=300)
                    c5_t = compute_c5(h_try)
                    if c5_t < best_c5:
                        best_c5 = c5_t
                        best_h = h_try.copy()
                        best_n = n_high

    # === Phase 10: Final intensive refinement at best resolution ===
    n = best_n
    h = best_h.copy()
    for cycle in range(4):
        if remaining() < 30:
            break

        # Targeted descent
        if remaining() > 20:
            h, c5_t = targeted_descent(h, n, min(25.0, remaining() * 0.12), K=15)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h.copy()

        # Greedy swaps
        if remaining() > 20:
            h, c5_t = greedy_swap_local_search(h, n, min(30.0, remaining() * 0.12))
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h.copy()

        # SLSQP
        if remaining() > 15:
            h = slsqop_smmax(h, n, beta=30000, K=50, maxiter=400)
            c5_t = compute_c5(h)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h.copy()

    # === Phase 11: Tiny steps final tuning ===
    h = best_h.copy()
    n = best_n
    if remaining() > 15:
        for it in range(200):
            if remaining() < 3:
                break
            corr_t = np.correlate(h, 1.0 - h, mode='full')
            idx_t = np.argsort(-corr_t)[:12]
            g_t = np.zeros(n)
            for k in idx_t:
                g_t += grad_corr_k_vec(h, k, n)
            g_t /= len(idx_t)
            gn_t = np.linalg.norm(g_t)
            if gn_t < 1e-14:
                break
            for step in [0.003, 0.001, 0.002, 0.005, 0.0005]:
                h_try = project_feasible(h - step * g_t / gn_t, n)
                c_try = compute_c5(h_try)
                if c_try < best_c5:
                    best_c5 = c_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    break
            else:
                if it > 50:
                    break

    # Final feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
