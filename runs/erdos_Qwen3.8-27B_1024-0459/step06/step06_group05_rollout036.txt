    """

def binary_local_search(h_init, n, time_limit=60.0, rng=None):
    # Start from binary
    h = h_init.copy()
    # Round to binary
    h_bin = (h > 0.5).astype(float)
    # Adjust to have exactly n/2 ones

    if num_ones > n // 2:
        # Remove excess ones (those with lowest h values)
        ones = np.where(h_bin == 1)[0]
        excess = num_ones - n // 2
        # Remove the ones with lowest h values
        order = np.argsort(h[ones])[:excess]

        zeros = np.where(h_bin == 0)[0]

            zeros = np.where(h == 0)[0]
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h[i] = 0
            h[j] = 1
    
    return h_best, c5_best
