import time

_T0 = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        if interior.any():
            h[interior] += diff / interior.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    s = np.clip(beta * (corr - m), -500.0, 500.0)
    e = np.exp(s)
    return (m + np.log(e.sum()) / beta) * dx

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    s = np.clip(beta * (corr - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx] * dx
        if wk < 1e-15:
            continue
        off = k_idx - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - h[ao:])
            g[ao:] -= wk * h[:np_]
    return g

def eq_func(x, n):
    return x.sum() - n / 2.0

def eq_jac(x, n):
    return np.ones(n)

def run_slsqp(h, beta, maxiter):

    bounds = [(0.0, 1.0)] * n
    try:

def bundle_direction(h, active_indices, beta_for_weights):
    """
    Find a direction d that decreases f_k for all k in active_indices.
    Solves: max ε s.t. ∇f_k · d ≤ -ε for all k, ||d||_∞ ≤ 1
    """
    n = len(h)
    m = len(active_indices)
    
    # Compute gradients for active shifts
    grads = []
    for k_idx in active_indices:
        g = np.zeros(n)
        off = k_idx - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += (1.0 - h[:np_])
            g[:np_] -= h[off:]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += (1.0 - h[ao:])
            g[ao:] -= h[:np_]
        grads.append(g * (2.0 / n))
    
    grads = np.array(grads)  # shape (m, n)
    
    # LP: max ε s.t. grads @ d ≤ -ε, -1 ≤ d ≤ 1
    # Variables: d_0, ..., d_{n-1}, ε  (total n+1 variables)
    # Objective: -ε (minimize)
    c = np.zeros(n + 1)
    c[-1] = -1  # minimize -ε = maximize ε
    
    # Constraints: grads @ d + ε ≤ 0, i.e., grads @ d - ε ≤ 0... wait
    # We want: ∇f_k · d ≤ -ε for all k
    # i.e., ∇f_k · d + ε ≤ 0
    # In standard form: A @ [d; ε] ≤ b
    A = np.hstack([grads, -np.ones((m, 1))])  # shape (m, n+1)
    b = np.zeros(m)
    
    # Bounds: -1 ≤ d_i ≤ 1, ε ≥ 0 (or -inf ≤ ε)
    bounds = [(-1, 1)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    
    if res.status == 0:
        return res.x[:n], res.x[-1]
    else:
        return None, 0.0

import numpy as np
from scipy.optimize import minimize, LinearConstraint, Bounds
import time

_T0 = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        if interior.any():
            h[interior] += diff / interior.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    s = np.clip(beta * (corr - m), -500.0, 500.0)
    e = np.exp(s)
    return (m + np.log(e.sum()) / beta) * dx

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    s = np.clip(beta * (corr - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx] * dx
        if wk < 1e-15:
            continue
        off = k_idx - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - h[ao:])
            g[ao:] -= wk * h[:np_]
    return g

def eq_func(x, n):
    return x.sum() - n / 2.0

def eq_jac(x, n):
    return np.ones(n)

def slsqp_solve(h, beta, maxiter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_max_val, h, args=(beta,),
                       jac=smooth_max_grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': eq_func,
                                     'jac': eq_jac, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def tc_solve(h, beta, maxiter):

        cons = LinearConstraint(np.ones((1, n)), n / 2.0, n / 2.0)

                       jac=smooth_max_grad, method='trust-constr',

def weighted_grad_descent(h, time_limit, K=30, beta_w=100.0):
    """Gradient descent using weighted sum of worst-K shift gradients, with line search."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    
    for it in range(20000):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 5:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = corr.max() * dx
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Get top K shifts
        top_idx = np.argsort(corr)[-K:]
        c_top = corr[top_idx]
        # Weights: softmax
        w = np.exp(beta_w * (c_top - c_top.max()))
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(K) / K
        else:
            w /= w_sum
        
        # Compute weighted gradient
        g = np.zeros(n)
        for i, k_idx in enumerate(top_idx):
            wi = w[i]
            if wi < 1e-14:
                continue
            off = k_idx - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:] += wi * (1.0 - h[:np_])
                g[:np_] += (-wi) * h[off:]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wi * (1.0 - h[ao:])
                g[ao:] += (-wi) * h[:np_]
        g *= dx
        
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        
        # Line search (backtracking)
        direction = -g / gn
        step = 0.1
        for trial in range(20):
            h_try = project_feasible(h + step * direction, n)
            c_try = compute_c5(h_try)
            if c_try < c5_val:
                h = h_try
                break
            step *= 0.5
        else:
            # No improvement found with this direction; try plain gradient
            h_try = project_feasible(h + step * direction, n)
            h = h_try
    
    return best_h, best_c5
