def find_descent_direction(h, n, K, delta):
    """Find descent direction using LP on linearized objectives."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(-corr)[:K]
    
    # Build LP: min t s.t. grad_k^T d <= t - f_k, sum(d)=0, |d_i| <= delta
    # Variables: d (n), t (1) -> total n+1
    # Objective: minimize t -> c = [0,...,0,1]
    # Constraints:
    #   grad_k^T d - t <= -f_k for each active k (K constraints)
    #   sum(d) = 0 (equality)
    #   -delta <= d_i <= delta for each i (2n bounds)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((K, n + 1))
    b_ub = np.zeros(K)
    for i, k in enumerate(idx):
        g = grad_corr_k(h, k, n)
        A_ub[i, :n] = g
        A_ub[i, -1] = -1.0
        b_ub[i] = -corr[k]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    bounds = [(-delta, delta)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1] - np.max(corr)
    return None, None

_BUDGET = 1300.0

_G_BETA = 10000.0
_G_K = 50


def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        nfree = np.count_nonzero(mask)
        if nfree > 0:
            h[mask] += diff / nfree
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_corr_k_vec(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        np_ = n - offset
        g[offset:] += (1.0 - h[:np_])
        g[:np_] -= h[offset:]
    else:
        a = -offset
        np_ = n - a
        g[:np_] += (1.0 - h[a:])
        g[a:] -= h[:np_]
    return g


def lp_step(h, n, K, delta, tol_frac):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    threshold = c_max - tol_frac

    if len(idx_all) == 0:

    for i, k in enumerate(idx):

    c = corr[idx]
    sh = c - m
    cl = np.clip(beta * sh, -80.0, 80.0)
    ex = np.exp(cl)
    val = (m + np.log(ex.sum()) / beta) * (2.0 / n)
    return val


def smax_gradient(h):

    beta = _G_BETA
    K = _G_K
    corr = np.correlate(h, 1.0 - h, mode='full')
