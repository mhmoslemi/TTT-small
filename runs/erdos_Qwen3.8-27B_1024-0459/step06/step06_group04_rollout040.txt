def efficient_swap_search(h, n, num_iterations, seed):
    rng = np.random.RandomState(seed)
    h = h.copy()
    corr = np.correlate(h, 1-h, mode='full')  # length 2n-1
    best_max = corr.max()
    
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h < 0.5)[0])
    
    for it in range(num_iterations):
        if not ones or not zeros:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Compute delta for swap (i: 1->0, j: 0->1)
        # delta(k) = h[i+k] + h[i-k] - h[j+k] - h[j-k] for k in -(n-1)..(n-1)
        delta = np.zeros(2*n-1)
        # A_i(k) = h[i+k] + h[i-k]
        for k in range(-(n-1), n):
            val = 0.0
            if 0 <= i+k < n: val += h[i+k]
            if 0 <= i-k < n: val += h[i-k]
            if 0 <= j+k < n: val -= h[j+k]
            if 0 <= j-k < n: val -= h[j-k]
            delta[k+n-1] = val
        
        new_corr = corr + delta
        new_max = new_corr.max()
        
        if new_max < best_max - 1e-12:
            # Accept swap
            h[i] = 0
            h[j] = 1
            ones.remove(i)
            zeros.remove(j)
            zeros.append(i)
            ones.append(j)
            corr = new_corr
            best_max = new_max
    
    return h, best_max

    """Compute delta for all lags when swapping i(1->0) and j(0->1)"""
    k = np.arange(-(n-1), n)
    # A_i(k) = h[i+k] (if valid) + h[i-k] (if valid)

    valid_ik = (ik_idx >= 0) & (ik_idx < n)

def smooth_max_func(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    s = np.clip(beta * (corr - m), -500.0, 500.0)
    e = np.exp(s)
    return (m + np.log(e.sum()) / beta) * dx

def smooth_max_jac(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    s = np.clip(beta * (corr - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx] * dx
        if wk < 1e-15:
            continue
        off = k_idx - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - x[ao:])
            g[ao:] -= wk * x[:np_]
    return g

def eq_func(x, n):
    return x.sum() - n / 2.0

def eq_jac(x, n):
    return np.ones(n)

    start = time.time()
    try:
        res = minimize(smooth_max_func, h, args=(beta,),
                       jac=smooth_max_jac, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': eq_func, 'jac': eq_jac, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:

    return h

    rng = np.random.RandomState(42)

    best_c5 = compute_c5(h)

        sw = weights.sum()
        if sw < 1e-30:

        else:

                g[:np_] -= wk * h[off:]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - h[ao:])
                g[ao:] -= wk * h[:np_]

            best_h = h_try.copy()
            h = h_try

        else:
