    return float(np.max(corr)) * dx

    target = n / 2.0
    for _ in range(500):

def smooth_max_value(h, beta):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Scale corr by dx to get the actual objective values
    vals = corr * dx
    m = vals.max()
    # Softmax for stability
    s = beta * (vals - m)
    s = np.clip(s, -500, 500)
    e = np.exp(s)
    return m + np.log(e.sum()) / beta

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = vals.max()
    s = beta * (vals - m)
    s = np.clip(s, -500, 500)
    e = np.exp(s)
    w = e / e.sum()  # weights for each lag
    g = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-15:
            continue
        off = k_idx - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - h[ao:])
            g[ao:] -= wk * h[:np_]
    g *= dx
    return g

def lbfgs_optimize(h, n, beta, maxiter, penalty_lam=1000.0):
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        f = smooth_max_value(x, beta)
        pen = penalty_lam * (x.sum() - n/2.0)**2
        return f + pen
    
    def grad(x):
        g = smooth_max_grad(x, beta)
        g += 2.0 * penalty_lam * (x.sum() - n/2.0) * np.ones(n)
        return g
    
    res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
    return project_feasible(res.x, n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        if interior.any():
            h[interior] += diff / interior.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_value(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = vals.max()
    s = np.clip(beta * (vals - m), -500.0, 500.0)
    e = np.exp(s)
    return m + np.log(e.sum()) / beta


def smooth_max_grad(h, beta):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def lbfgs_solve(h, n, beta, maxiter, penalty_lam, time_limit):
    bounds = [(0.0, 1.0)] * n
    target_sum = n / 2.0

    def obj(x):
        f = smooth_max_value(x, beta)
        pen = penalty_lam * (x.sum() - target_sum) ** 2
        return f + pen

    def grad(x):
        g = smooth_max_grad(x, beta)
        g += 2.0 * penalty_lam * (x.sum() - target_sum) * np.ones(n)
        return g

    start = time.time()
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
        if time.time() - start > time_limit:
            return project_feasible(h, n)
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)


def slsqp_solve(h, n, beta, maxiter, time_limit):
    bounds = [(0.0, 1.0)] * n
    target_sum = n / 2.0

    def obj(x):
        return smooth_max_value(x, beta)

    def obj_grad(x):
        return smooth_max_grad(x, beta)

    def eq_fun(x):
        return x.sum() - target_sum

    def eq_jac(x):
        return np.ones(n)

    start = time.time()
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if time.time() - start > time_limit:
            return project_feasible(h, n)
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)
