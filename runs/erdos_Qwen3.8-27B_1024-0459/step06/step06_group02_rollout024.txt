        if p % 3 == 0:
            h_pert = perturb_smooth(best_h, rng, scale=0.012 + 0.008 * p, n_modes=3, max_freq=8)
        elif p % 3 == 1:
            h_pert = perturb_fourier(best_h, rng, scale=0.025)
        else:
            h_pert = perturb_flip(best_h, rng, frac=0.03)
        for beta in [60, 200, 500]:
            if remaining() < 20:
                break
            h_s = opt_lbfgsb(h_pert, best_n, beta, maxiter=200, gtol=1e-8, top_k=400)
            c_s = compute_c5(h_s)
            if c_s < best_c5 - 1e-14:
                best_c5 = c_s
                best_h = h_s.copy()

    # Phase 3: Upsample to n=1536 and optimize
    if remaining() > 90:
        n_t = 1536
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_box_sum(h_interp, n_t / 2.0)

            c_new = compute_c5(h_new)
            if c_new < c_inter - 1e-14:
                c_inter = c_new
                h_feas = h_new

        if c_inter < best_c5 - 1e-14:
            best_c5 = c_inter
            best_h = h_feas.copy()
            best_n = n_t

        # Perturbation restarts at n=1536
        for p in range(5):
            if remaining() < 35:
                break
                break
            if p % 2 == 0:
                h_pert = perturb_smooth(best_h, rng, scale=0.008 + 0.005 * p, n_modes=2, max_freq=6)
            else:
                h_pert = perturb_fourier(best_h, rng, scale=0.02)
            for beta in [60, 250]:
                if remaining() < 20:
                    break
                h_s = opt_lbfgsb(h_pert, n_t, beta, maxiter=250, gtol=1e-8, top_k=500)
                c_s = compute_c5(h_s)
                if c_s < best_c5 - 1e-14:
                    best_c5 = c_s
                    best_h = h_s.copy()
                    best_n = n_t

    # Phase 4: Try n=2048 if time permits
    if remaining() > 120:
        n_t = 2048
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_box_sum(h_interp, n_t / 2.0)
        c_inter = compute_c5(h_feas)

        for beta in [15, 30, 60, 120, 250]:
            if remaining() < 30:
                break
            h_new = opt_lbfgsb(h_feas, n_t, beta, maxiter=250, gtol=1e-8, top_k=500)
            c_new = compute_c5(h_new)
            if c_new < c_inter - 1e-14:
                c_inter = c_new
                h_feas = h_new

        if c_inter < best_c5 - 1e-14:
            best_c5 = c_inter
            best_h = h_feas.copy()
            best_n = n_t

            h_pert = perturb_smooth(h_feas, rng, scale=0.008, n_modes=2, max_freq=5)
            for beta in [60, 250]:
                if remaining() < 20:
                    break
                h_s = opt_lbfgsb(h_pert, n_t, beta, maxiter=200, gtol=1e-8, top_k=500)
                c_s = compute_c5(h_s)
                if c_s < best_c5 - 1e-14:
                    best_c5 = c_s

    for beta in [200, 500, 1000, 2000, 5000]:

    for p in range(10):
        if remaining() < 18:

        elif p % 3 == 1:

            h_pert = perturb_flip(best_h, rng, frac=0.01)
        h_new = opt_lbfgsb(h_pert, best_n, 300, maxiter=300, gtol=1e-10, top_k=600)

    if remaining() > 15:

        c_new = compute_c5(h_new)
        if c_new < best_c5 - 1e-15:
            best_c5 = c_new
            best_h = h_new.copy()

    # Ensure feasibility
    best_h = project_box_sum(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, float(best_c5), best_n)
