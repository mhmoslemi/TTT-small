        elif max_d < -tol:
            lo = mid
        else:
            return mid
    return (lo + hi) * 0.5

def compute_cd_coeffs(h, i, active_idx, n, corr):
    """Compute C_k and d_k for coordinate descent at position i."""
    K = len(active_idx)
    C_k = np.empty(K)
    d_k = np.empty(K)
    for j in range(K):
        k = active_idx[j]
        offset = k - (n - 1)
        if offset > 0:
            o = offset
            d = 0.0
            if i < n - o:
                d += (1.0 - h[i + o])
            if i >= o:
                d -= h[i - o]
        elif offset < 0:
            o = -offset
            d = 0.0
            if i >= o:
                d += (1.0 - h[i - o])
            if i < n - o:
                d -= h[i + o]
        else:
            d = 1.0 - 2.0 * h[i]
        d_k[j] = d
        C_k[j] = corr[k] - h[i] * d
    return C_k, d_k

def coord_descent(h, n, K, max_sweeps, start_time=None):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    max_no_improve = 20
    
    if start_time is None:
        start_time = time.time()
    
    for sweep in range(max_sweeps):
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        
        corr = get_corr(h)
        active_idx = np.argsort(-corr)[:K]
        
        h_new = h.copy()
        for i in range(n):
            C_k, d_k = compute_cd_coeffs(h, i, active_idx, n, corr)
            h_new[i] = solve_1d_minimax(C_k, d_k)
        
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
            h = h_new.copy()
    
    return best_h, best_c5
```

Wait, I realize there's a problem with my coordinate descent. When I use the stale correlation (computed at the start of the sweep), the C_k values I compute are based on the OLD h. But after I update h[0], h[1], ..., h[i-1], the actual correlation has changed. So my C_k values are stale.

This means that when I solve for h[i], I'm optimizing based on a "snapshot" of the problem, not the current state. This is the "Jacobi" coordinate descent, which is valid but might converge slower.

For better convergence, I could:
1. Use Gauss-Seidel: after each coordinate update, update the correlation incrementally.
2. Or: recompute the correlation more frequently (e.g., every 32 coordinates).

Let me implement the incremental correlation update for Gauss-Seidel:

When h[i] changes from old to new, the change in corr[k] is:
For offset > 0 (offset = o):
  corr_o changes by (new - old) * [(1 - h[i+o]) - h[i-o]] (with appropriate boundary checks)
  = (new - old) * d_k

For offset < 0 (offset = -o):
  corr_{-o} changes by (new - old) * [(1 - h[i-o]) - h[i+o]] (with appropriate boundary checks)
  = (new - old) * d_k

So I can update corr[k] += (new - old) * d_k for each active k.

But wait, this only updates the active correlations. The inactive ones also change (but I don't care about them for the minimax). However, when I re-select the active set, I need the full correlation. So I should either:
1. Update all n-1 correlations (expensive: O(n) per coordinate)
2. Only update active ones and recompute the full correlation periodically

Option 2 is a good compromise. Let me update the active correlations incrementally and recompute the full correlation every, say, 64 coordinates.

Actually, for n=512 and K=40, updating K correlations per coordinate is 40 operations, times 512 coordinates = 20480 operations per sweep. Plus one full correlation recompute (O(n²) = 262K). Total per sweep: ~282K operations. For 100 sweeps: ~28M. Should be fine.

Let me implement this:

```python
def coord_descent_gs(h, n, K, max_sweeps):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    
    for sweep in range(max_sweeps):
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        
        corr = get_corr(h)
        active_idx = np.argsort(-corr)[:K]
        
        h_old = h.copy()
        
        for i in range(n):
            # Compute C_k and d_k using current h (Gauss-Seidel)
            C_k = np.empty(K)
            d_k = np.empty(K)
            for j in range(K):
                k = active_idx[j]
                offset = k - (n - 1)
                if offset > 0:
                    o = offset
                    d = 0.0
                    if i < n - o:
                        d += (1.0 - h[i + o])
                    if i >= o:
                        d -= h[i - o]
                elif offset < 0:
                    o = -offset
                    d = 0.0
                    if i >= o:
                        d += (1.0 - h[i - o])
                    if i < n - o:
                        d -= h[i + o]
                else:
                    d = 1.0 - 2.0 * h[i]
                d_k[j] = d
                # C_k is corr[k] - h_old[i] * d_k (since corr was computed with h_old)
                # But wait, corr has been updated for previous coordinates...
                # Actually, corr should reflect the current state (with h[0..i-1] updated)
                # So C_k = corr_as_if_h_i_were_zero
                # corr_current = C_k + h[i] * d_k (approximately, if corr tracks current state)
                # So C_k = corr_current - h[i] * d_k
                C_k[j] = corr[k] - h[i] * d_k
            
            t_opt = solve_1d_minimax(C_k, d_k)
            old_val = h[i]
            h[i] = t_opt
            
            # Update corr incrementally
            delta = t_opt - old_val
            if abs(delta) > 1e-15:
                for j in range(K):
                    corr[active_idx[j]] += delta * d_k[j]
        
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 15:
                break
    
    return best_h, best_c5
```

Hmm wait, there's a subtle issue with the incremental correlation update. When I change h[i] from old_val to new_val, the change in corr[k] is:

corr_new[k] - corr_old[k] = (new_val - old_val) * d_k

where d_k is computed using the CURRENT values of other h's (before the change to h[i]).

But d_k depends on h[i+offset] and h[i-offset], which are the current values (some updated, some not). So d_k is correct for the current state.

After the update, corr[k] += delta * d_k gives the correct new correlation value.

This is correct! The incremental update is exact because corr[k] is linear in h[i] (with the coefficient d_k depending on other coordinates).

Let me verify: corr[k] = sum_j h[j]*(1-h[j+offset_k])
The terms involving h[i]:
- h[i]*(1-h[i+offset_k]) (if i is in range for the first index)
- h[i-offset_k]*(1-h[i]) (if i-offset_k is in range for the first index)

So corr[k] = [terms not involving h[i]] + h[i]*(1-h[i+offset_k]) - h[i-offset_k]*h[i]
= C + h[i] * [(1-h[i+offset_k]) - h[i-offset_k]]
= C + h[i] * d_k

So corr[k] - h[i]*d_k = C (independent of h[i]). ✓

And corr[k] + delta * d_k = C + h[i]*d_k + delta*d_k = C + (h[i]+delta)*d_k = C + new_h[i]*d_k. ✓

Great, the incremental update is exact.

Now, there's one more issue: after the sweep, I project to satisfy the sum constraint. This changes all h values, so the corr values I've been tracking are now stale. But that's fine because I recompute corr at the start of the next sweep.

Let me also think about the efficiency. The inner loop is:
- For each i (n = 512 iterations):
  - For each j (K = 40 iterations): compute d_k and C_k
  - Solve 1D minimax (50 iterations of binary search, each O(K))
  - Update corr (K operations)

Total per sweep: n * (K + K*50 + K) = n * K * 52 ≈ 512 * 40 * 52 ≈ 1M operations.
In Python, this might take 5-10 seconds per sweep. For 100 sweeps, that's 500-1000 seconds. That's too slow!

I need to optimize. The bottleneck is the 1D minimax binary search (50 iterations * K operations). Let me reduce the binary search iterations to 30 (still gives precision of 1/2^30 ≈ 1e-9, more than enough).

With 30 iterations: n * K * 32 ≈ 512 * 40 * 32 ≈ 655K operations per sweep. In Python, maybe 3-5 seconds per sweep. For 100 sweeps: 300-500 seconds. Better, but still tight.

Let me also vectorize the d_k and C_k computation. Instead of looping over K for each i, I can precompute arrays.

Actually, let me think about this differently. The inner loop for a given i:
- For each j in range(K): compute d_k[j] and C_k[j]
- This requires h[i ± offset_j] for each j.

I can precompute the offsets and use numpy indexing:
- offsets = active_idx - (n-1), then take absolute values and signs.
- For each j: if offset_j > 0, d_j = (1-h[i+offset_j]) - h[i-offset_j] (with boundary checks)
- etc.

The boundary checks make vectorization tricky. But I can handle it with masking.

Let me try a vectorized approach:

```python
def coord_descent_vec(h, n, K, max_sweeps):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    
    for sweep in range(max_sweeps):
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        
        corr = get_corr(h)
        active_idx = np.argsort(-corr)[:K]
        offsets = active_idx - (n - 1)  # range: -(n-1) to n-1
        
        for i in range(n):
            # Vectorized computation of d_k
            d_k = np.zeros(K)
            
            pos_mask = offsets > 0
            neg_mask = offsets < 0
            zero_mask = offsets == 0
            
            # For positive offsets: d = [i < n-o](1-h[i+o]) - [i>=o]h[i-o]
            if np.any(pos_mask):
                o = offsets[pos_mask]
                valid1 = (i < n - o)  # i+o < n
                valid2 = (i >= o)  # i-o >= 0
                vals1 = np.where(valid1, 1.0 - h[i + o], 0.0)
                vals2 = np.where(valid2, h[i - o], 0.0)
                d_k[pos_mask] = vals1 - vals2
            
            # For negative offsets: d = [i>=o](1-h[i-o]) - [i<n-o]h[i+o] where o=-offset
            if np.any(neg_mask):
                o = -offsets[neg_mask]
                valid1 = (i >= o)
                valid2 = (i < n - o)
                vals1 = np.where(valid1, 1.0 - h[i - o], 0.0)
                vals2 = np.where(valid2, h[i + o], 0.0)
                d_k[neg_mask] = vals1 - vals2
            
            # For zero offset: d = 1 - 2*h[i]
            if np.any(zero_mask):
                d_k[zero_mask] = 1.0 - 2.0
