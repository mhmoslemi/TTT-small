def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        g[offset:] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:]
    else:
        a = -offset
        npairs = n - a
        g[:npairs] += (1.0 - h[a:])
        g[a:] -= h[:npairs]
    return g

def smooth_max_topk(h, n, K, beta):
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    m = np.max(c_top)
    sh = np.clip(beta * (c_top - m), -50.0, 50.0)
    ex = np.exp(sh)
    val = (m + np.log(ex.sum()) / beta)
    return val * (2.0 / n)

def smooth_max_topk_grad(h, n, K, beta):
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    m = np.max(c_top)
    sh = np.clip(beta * (c_top - m), -50.0, 50.0)
    ex = np.exp(sh)
    s = ex.sum()
    if s < 1e-30:
        w = np.ones(K) / K
    else:
        w = ex / s

    for i in range(K):
        wk = w[i] * dx
        if wk < 1e-15:
            continue
        g += wk * grad_corr_k(h, idx[i], n)
    return g

    """L-BFGS-B with quadratic penalty on the sum constraint."""
    h_curr = h.copy()
    best_c5 = compute_c5(h_curr)
    best_h = h_curr.copy()
    
    penalty = penalty_start
    
    for outer in range(5):
        def obj(x):
            sm = smooth_max_topk(x, n, K, beta)
            penalty_term = penalty * (np.sum(x) - n/2.0)**2
            return sm + penalty_term
        
        def grad(x):
            g_sm = smooth_max_topk_grad(x, n, K, beta)
            g_pen = 2.0 * penalty * (np.sum(x) - n/2.0) * np.ones(n)
            return g_sm + g_pen
        
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj, h_curr, jac=grad, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-12})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_curr = h_new.copy()
            
            # Check constraint satisfaction
            sum_err = abs(np.sum(h_curr) - n/2.0)
            if sum_err < 1e-10:
                break
            penalty *= penalty_mult
            
        except Exception:
            break
    
    return best_h, best_c5

def slsqp_optimize(h, n, K, beta, maxiter=400):
    def obj(x):
        return smooth_max_topk(x, n, K, beta)
    
    def grad(x):
        return smooth_max_topk_grad(x, n, K, beta)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:

            c5_old = compute_c5(h)
            if c5_new <= c5_old + 1e-12:

    except Exception:

        h_curr = h_new
    
    return best_h, best_c5
