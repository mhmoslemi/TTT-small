def swap_delta(h, i, j, n):
    """Compute corr + delta after swapping h[i](=1) and h[j](=0).
    Returns the new max correlation (without dx factor)."""
    nfull = 2 * n - 1
    # h[i] changes from 1 to 0, h[j] from 0 to 1
    # delta[k] = -∂corr[k]/∂h[i] + ∂corr[k]/∂h[j]
    
    # ∂corr[k]/∂h[i] = (1-h[i+off_k])*[0<=i+off_k<n] - h[i-off_k]*[0<=i-off_k<n]
    # where off_k = k-(n-1)
    
    # As first factor: i+off in [0,n-1] → off in [-i, n-1-i] → k in [n-1-i, 2n-2-i]
    # value: (1-h[i+off_k]) = (1-h[k-(n-1-i)]) for k in [n-1-i, 2n-2-i]
    
    # As second factor: i-off in [0,n-1] → off in [i-(n-1), i] → k in [i, 2n-2-i]  
    # value: -h[i-off_k] = -h[n-1+i-k] for k in [i, 2n-2-i]
    
    delta = np.zeros(nfull)
    
    # First factor contribution for i: -(1-h[k-(n-1-i)]) for k in [n-1-i, 2n-2-i]
    k_start = n-1-i
    k_end = 2*n-2-i  # inclusive
    idx = np.arange(k_start, k_end+1)
    positions = idx - (n-1-i)  # = k-(n-1-i), ranges from 0 to n-1-i
    delta[idx] -= (1.0 - h[positions])
    
    # Second factor contribution for i: +h[n-1+i-k] for k in [i, 2n-2-i]
    k_start = i
    k_end = 2*n-2-i
    idx = np.arange(k_start, k_end+1)
    positions = n-1+i-idx  # = n-1+i-k, ranges from n-1 down to 0
    delta[idx] += h[positions]
    
    # First factor contribution for j: +(1-h[k-(n-1-j)]) for k in [n-1-j, 2n-2-j]
    k_start = n-1-j
    k_end = 2*n-2-j
    idx = np.arange(k_start, k_end+1)
    positions = idx - (n-1-j)
    delta[idx] += (1.0 - h[positions])
    
    # Second factor contribution for j: -h[n-1+j-k] for k in [j, 2n-2-j]
    k_start = j
    k_end = 2*n-2-j
    idx = np.arange(k_start, k_end+1)
    positions = n-1+j-idx
    delta[idx] -= h[positions]
    
    return delta

    cur_max = float(np.max(corr))
    best_max = cur_max
    best_h = h.copy()
    start = time.time()
    ones = np.where(h == 1.0)[0].tolist()
    zeros = np.where(h == 0.0)[0].tolist()
    nfull = 2 * n - 1
    n_iter = int(time_limit * 300)
    for it in range(n_iter):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 10:
            break
        prog = it / max(n_iter, 1)
        T = T_start * (T_end / T_start) ** prog
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        d = swap_delta_vec(h, i, j, n)
        new_corr = corr + d
        new_max = float(np.max(new_corr))
        delta = new_max - cur_max
        if delta < 0 or (T > 1e-8 and np.random.random() < np.exp(-delta / T)):
            h[i] = 0.0
            h[j] = 1.0
            corr = new_corr
            cur_max = new_max
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
    return best_h, best_max * 2.0 / n, compute_corr(best_h)


def focused_search(h, n, corr, time_limit=40, n_trials=5000):
    h = h.copy()
    corr = corr.copy()
    cur_max = float(np.max(corr))
    start = time.time()
    ones = np.where(h == 1.0)[0].tolist()
    zeros = np.where(h == 0.0)[0].tolist()
    improved = True
    while improved and time.time() - start < time_limit:
        if _BUDGET - (time.time() - _T0) < 10:
            break
        improved = False
        corr_now = np.max(corr)
        if abs(corr_now - cur_max) > 1e-10:
            cur_max = float(corr_now)
        top_k = np.where(corr >= cur_max - 2.0)[0]
        best_gain = 1e-10
        best_i = -1
        best_j = -1
        best_delta = None
        tr = min(n_trials, len(ones) * len(zeros))
        for _t in range(tr):
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
            d = swap_delta_vec(h, i, j, n)
            top_new = np.max(corr[top_k] + d[top_k])
            global_new = float(np.max(corr + d))
            nm = max(top_new, global_new)
            g = cur_max - nm
            if g > best_gain:
                best_gain = g
                best_i = i
                best_j = j
                best_delta = d
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            corr += best_delta
            cur_max -= best_gain
            ones.remove(best_i)
            ones.append(best_j)
            zeros.remove(best_j)
            zeros.append(best_i)
            improved = True
    return h, cur_max * 2.0 / n, corr


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    x_new = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(x_new, xp, src_h)


def smooth_max_value(x, beta, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50.0, 50.0)
    sm = m + np.log(np.sum(np.exp(shifted))) / beta
    return sm * (2.0 / n)


def smooth_max_grad(x, beta, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50.0, 50.0)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = w / w_sum
    g = np.zeros(n)
    dx = 2.0 / n
    nfull = 2 * n - 1
    for k in range(nfull):
        wk = w[k] * dx
        if wk < 1e-16:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            np_ = n - offset
            g[offset:offset + np_] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[offset:offset + np_]
        else:
            a = -offset
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:a + np_])
            g[a:a + np_] -= wk * x[:np_]
    return g


def slsqp_optimize(h, n, beta, maxiter=500):
    def obj_func(x):
        return smooth_max_value(x, beta, n)

    def grad_func(x):
        return smooth_max_grad(x, beta, n)

    def eq_fun(x):
        return np.sum(x) - n / 2.0

    def eq_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(obj_func, h, jac=grad_func, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    np.random.seed(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)

    # Phase 0: Initialize
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        n0 = len(parent_h)
        best_h = project_feasible(parent_h, n0)
    except Exception:
        n0 = 512
        best_h = np.zeros(n0)
        best_h[:n0 // 2] = 1.0

    best_c5 = compute_c5(best_h)

    # Phase 1: Continuous SLSQP to refine parent
    n = n0
    h = best_h.copy()
    for beta in [500, 3000, 15000]:
        if remaining() < 20:
            break
        h_new = slsqp_optimize(h.copy(), n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Round to binary and start combinatorial search
    h_bin = project_to_binary(h, n)
    corr_bin = compute_corr(h_bin)
    max_bin = float(np.max(corr_bin))
    c5_bin = max_bin * 2.0 / n
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # Phase 3: Aggressive greedy local search (many passes)
    if remaining() > 30:
        h_g, c5_g, corr_g = greedy_bin(h_bin, n, corr_bin, time_limit=min(40.0, remaining() * 0.3), n_trials=5000)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()
            h_bin = h_g.copy()
            corr_bin = corr_g.copy()

    # Phase 4: Focused search on worst shifts
    if remaining() > 30:
        h_f, c5_f, corr_f = focused_search(h_bin, n, corr_bin, time_limit=min(50.0, remaining() * 0.3), n_trials=8000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            h_bin = h_f.copy()
            corr_bin = corr_f.copy()

    # Phase 5: Simulated annealing (multiple restarts)
    for sa_round in range(3):
        if remaining() < 40:
            break
        h_a, c5_a, corr_a = anneal_bin(h_bin, n, corr_bin,
                                        time_limit=min(50.0, remaining() * 0.25),
                                        T_start=1.5, T_end=0.005)
        if c5_a < best_c5:
            best_c5 = c5_a
            best_h = h_a.copy()
            h_bin = h_a.copy()
            corr_bin = corr_a.copy()
        # Follow up with greedy
        if remaining() > 20:
            h_g2, c5_g2, corr_g2 = greedy_bin(h_bin, n, corr_bin, time_limit=min(20.0, remaining() * 0.2), n_trials=4000)
            if c5_g2 < best_c5:
                best_c5 = c5_g2
                best_h = h_g2.copy()
                h_bin = h_g2.copy()
                corr_bin = corr_g2.copy()

    # Phase 6: More annealing with different schedules
    for sa_round in range(3):
        if remaining() < 35:
            break
        h_a, c5_a, corr_a = anneal_bin(h_bin, n, corr_bin,
                                        time_limit=min(40.0, remaining() * 0.25),
                                        T_start=0.8, T_end=0.002)
        if c5_a < best_c5:
            best_c5 = c5_a
            best_h = h_a.copy()
            h_bin = h_a.copy()
            corr_bin = corr_a.copy()
        if remaining() > 20:
            h_f2, c5_f2, corr_f2 = focused_search(h_bin, n, corr_bin, time_limit=min(20.0, remaining() * 0.2))
            if c5_f2 < best_c5:
                best_c5 = c5_f2
                best_h = h_f2.copy()
                h_bin = h_f2.copy()
                corr_bin = corr_f2.copy()

    # Phase 7: Try higher resolutions with binary search
    res_list = []
    for mult in [1.25, 1.5, 2.0]:
        r = int(round(n * mult / 64) * 64)
        if 64 <= r <= 2048 and r % 2 == 0 and r != n:
            res_list.append(r)
    res_list = sorted(set(res_list))

    for n_t in res_list:
        if remaining() < 80:
            break
        h_interp = interp_to(best_h, n, n_t)
        h_t = project_to_binary(h_interp, n_t)
        corr_t = compute_corr(h_t)
        c5_t = float(np.max(corr_t)) * 2.0 / n_t

        if remaining() > 40:
            h_t, c5_t, corr_t = greedy_bin(h_t, n_t, corr_t, time_limit=min(30.0, remaining() * 0.3), n_trials=4000)
        if remaining() > 30:
            h_t, c5_t, corr_t = focused_search(h_t, n_t, corr_t, time_limit=min(25.0, remaining() * 0.3), n_trials=6000)
        if remaining() > 25:
            h_t, c5_t, corr_t = anneal_bin(h_t, n_t, corr_t, time_limit=min(20.0, remaining() * 0.2), T_start=1.0, T_end=0.005)
        if remaining() > 15:
            h_t, c5_t, corr_t = greedy_bin(h_t, n_t, corr_t, time_limit=min(10.0, remaining() * 0.15), n_trials=3000)

        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            best_n = n_t
            n = n_t
            h_bin = h_t.copy()
            corr_bin = corr_t

    # Phase 8: Final continuous polish (SLSQP on best continuous interp)
    if remaining() > 30:
        h_cont = project_feasible(best_h, n)
        for beta in [5000, 20000, 80000]:
            if remaining() < 10:
                break
            h_new = slsqp_optimize(h_cont, n, beta, maxiter=800)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_cont = h_new.copy()

    # Phase 9: Final binary polish at best resolution
    h_bin_final = project_to_binary(best_h, n)
    corr_final = compute_corr(h_bin_final)
    c5_final = float(np.max(corr_final)) * 2.0 / n
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h_bin_final.copy()

    if remaining() > 20:
        h_fin, c5_fin, corr_fin = greedy_bin(h_bin_final, n, corr_final, time_limit=min(15.0, remaining() * 0.5), n_trials=6000)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, n)
