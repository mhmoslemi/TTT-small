def fast_swap_eval(h, corr, i, j):
    """Evaluate swap h[i]=1->0, h[j]=0->1. Returns delta array and new_max.
    corr is the current correlation array (length 2n-1).
    """
    n = len(h)
    # For shift k (correlation index = n-1+k):
    # delta[k] = -(1-h[i+k])*[valid] + h[i-k]*[valid] + (1-h[j+k])*[valid] - h[j-k]*[valid]
    
    # Vectorized computation:
    k_vals = np.arange(-(n-1), n)  # shift values
    # i+k valid: 0 <= i+k < n
    ik = i + k_vals
    jk = j + k_vals
    ik_valid = (ik >= 0) & (ik < n)
    jk_valid = (jk >= 0) & (jk < n)
    
    i_minus_k = i - k_vals
    j_minus_k = j - k_vals
    imk_valid = (i_minus_k >= 0) & (i_minus_k < n)
    jmk_valid = (j_minus_k >= 0) & (j_minus_k < n)
    
    delta = np.zeros(2*n-1)
    delta[ik_valid] += -(1.0 - h[ik[ik_valid]])
    delta[imk_valid] += h[i_minus_k[imk_valid]]
    delta[jk_valid] += (1.0 - h[jk[jk_valid]])
    delta[jmk_valid] += -h[j_minus_k[jmk_valid]]
    
    new_corr = corr + delta
    new_max = float(np.max(new_corr))
    return delta, new_max, new_corr

The gradient of corr[k] with respect to h is:
- For k ≥ 0 (correlation index n-1+k ≥ n-1):
  d(corr[k])/dh[i] = (1-h[i+k]) if i+k < n, plus (-h[i-k]) if i-k ≥ 0
  
  More precisely, corr[k] = ∑_{i=0}^{n-1-k} h[i](1-h[i+k])
  d/dh[j] = (1-h[j+k])·[j ≤ n-1-k] + (-h[j-k])·[j-k ≥ 0, j ≥ k]
  
  For j in [0, n):
  - Term 1: (1-h[j+k]) if j+k < n, i.e., j < n-k
  - Term 2: -h[j-k] if j-k ≥ 0, i.e., j ≥ k

So the gradient at position j is:
(1-h[j+k])·[j < n-k] - h[j-k]·[j ≥ k]

For the weighted sum over k:
g[j] = ∑_k w_k·dx · [(1-h[j+k])·[j < n-k] - h[j-k]·[j ≥ k]]

This is a convolution-like operation! We can use FFT to compute it in O(n log n).

A[j] = ∑_k w_k · (1-h[j+k]) · [j+k < n, j < n-k]
B[j] = ∑_k w_k · h[j-k] · [j-k ≥ 0]

Then g = A - B (times dx).

```python
def compute_swap_delta(h, i, j):
    """Compute delta in correlation when h[i]: 1->0, h[j]: 0->1."""
    n = len(h)
    delta = np.zeros(2*n-1)
    
    # Effect of h[i] going from 1 to 0:
    # corr[k] includes h[i](1-h[i+k]) for valid i+k, and h[i-k](1-h[i]) for valid i-k
    # After change: 0*(1-h[i+k]) and h[i-k]*(1-0)
    # Delta from i: -(1-h[i+k])*[valid] + h[i-k]*[valid]
    
    # Effect of h[j] going from 0 to 1:
    # Before: 0*(1-h[j+k]) and h[j-k]*(1-0)
    # After: 1*(1-h[j+k]) and h[j-k]*(1-1)=0
    # Delta from j: (1-h[j+k])*[valid] - h[j-k]*[valid]
    
    k_vals = np.arange(-(n-1), n)
    
    # i+k valid
    ik = i + k_vals
    mask_ik = (ik >= 0) & (ik < n)
    delta[mask_ik] += -(1.0 - h[ik[mask_ik]])
    
    # i-k valid
    ik2 = i - k_vals
    mask_ik2 = (ik2 >= 0) & (ik2 < n)
    delta[mask_ik2] += h[ik2[mask_ik2]]
    
    # j+k valid
    jk = j + k_vals
    mask_jk = (jk >= 0) & (jk < n)
    delta[mask_jk] += (1.0 - h[jk[mask_jk]])
    
    # j-k valid
    jk2 = j - k_vals
    mask_jk2 = (jk2 >= 0) & (jk2 < n)
    delta[mask_jk2] += -h[jk2[mask_jk2]]
    
    return delta
```

This is O(n) with vectorized numpy operations. For n=640, it involves 4 masked operations on arrays of length 1279. Should be very fast (~10μs).

So we can evaluate ~100K swaps per second. Over 120 seconds, that's 12M swap evaluations. With a batch of 500 and keeping the best, that's 24K "rounds" of best-of-500 selection. Should be plenty.

            h[mask] += diff / cnt
        else:

def compute_swap_delta(h, i, j):

def smooth_max_obj_grad(x, n, beta):
    """Objective and gradient for smooth max of correlations."""
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(corr))
    z = beta * (corr - m)
    z = np.clip(z, -50.0, 50.0)
    ez = np.exp(z)
    S = float(np.sum(ez))
    obj = (m + np.log(S) / beta) * dx
    
    # Gradient
    w = ez / S
    g = np.zeros(n)
    for k in range(2*n-1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        off = k - (n-1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    
    return obj, g

def projected_grad_momentum(h, n, time_limit=120, beta=5000):
    """Projected gradient descent with momentum for continuous h."""
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    dx = 2.0/n
    lr = 0.05
    mu = 0.7  # momentum
    v = np.zeros(n)
    
    start = time.time()
    patience = 0
    
    while time.time() - start < time_limit and remaining_time() > 5:
        obj, g = smooth_max_obj_grad(h, n, beta)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        
        # Momentum update
        v = mu * v - lr * g
        v_norm = np.linalg.norm(v)
        if v_norm > 0.5:
            v = v * 0.5 / v_norm
        
        # Line search
        improved = False
        for alpha in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            h_new = project_feasible(h + alpha * v, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                improved = True
                patience = 0
                break
        
        if not improved:
            patience += 1
            lr *= 0.7
            if patience > 30:
                # Restart with smaller beta
                beta = max(100, beta * 0.5)
                v = np.zeros(n)
                lr = 0.05
                patience = 0
                if beta < 100:
                    break
    
    return best_h, best_c5

def slsqp_opt(h, n, beta=1e6, maxiter=2000):
    dx = 2.0/n
    h0 = project_feasible(h, n)
    
    def obj(x):
        o, _ = smooth_max_obj_grad(x, n, beta)
        return o
    
    def grad(x):
        _, gd = smooth_max_obj_grad(x, n, beta)
        return gd
    
    def eq_fun(x):
        return np.sum(x) - n/2.0
    
    def eq_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_res = project_feasible(res.x, n)
            return h_res
    except Exception:
        pass
    return h0

def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)

def top_k_equalize(h, n, top_k=5, max_iters=500, time_limit=60):
    h = project_feasible(h, n)
    dx = 2.0/n
    best_c5 = compute_c5(h)
    start = time.time()
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if remaining_time() < 5:
            break
        
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        c_max = corr[idx_top[0]]
        c_min_top = corr[idx_top[-1]]
        if c_max - c_min_top < 1e-10:
            break
        
        # Direction: decrease top, increase bottom of top-k
        g_dir = np.zeros(n)
        for pos, k in enumerate(idx_top):
            off = k - (n-1)
            if off >= 0:
                np_ = n - off
                if pos == 0:
                    g_dir[off:] += (1.0 - h[:np_])
                    g_dir[:np_] -= h[off:]
                else:
                    g_dir[off:] -= (1.0 - h[:np_])
                    g_dir[:np_] += h[off:]
            else:
                a = -off
                np_ = n - a
                if pos == 0:
                    g_dir[:np_] += (1.0 - h[a:])
                    g_dir[a:] -= h[:np_]
                else:
                    g_dir[:np_] -= (1.0 - h[a:])
                    g_dir[a:] += h[:np_]
        
        g_dir = g_dir / len(idx_top)
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        d = g_dir / gn
        
        improved = False
        for step in [0.01, 0.005, 0.02, 0.05, 0.001, 0.002, 0.1]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    
    return h, best_c5
