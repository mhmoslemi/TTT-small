def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        at_low = h <= 1e-12
        at_high = h >= 1.0 - 1e-12
        mid = ~(at_low | at_high)
        cnt = int(np.count_nonzero(mid))
        if cnt > 0:
            h[mid] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def project_binary(h, n):
    """Project to binary: choose n/2 positions to be 1, preferring those with high h values."""
    target = n // 2
    idx = np.argsort(-h)
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def smooth_max_value(x, beta, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50.0, 50.0)
    sm = m + np.log(np.sum(np.exp(shifted))) / beta
    return sm * (2.0 / n)

def smooth_max_grad(x, beta, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50.0, 50.0)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = w / w_sum
    g = np.zeros(n)
    dx = 2.0 / n
    nfull = 2 * n - 1
    for k in range(nfull):
        wk = w[k] * dx
        if wk < 1e-16:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            np_ = n - offset
            g[offset:offset + np_] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[offset:offset + np_]
        else:
            a = -offset
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:a + np_])
            g[a:a + np_] -= wk * x[:np_]
    return g

def slsqp_optimize(h, n, beta, maxiter=500):
    def obj_func(x):
        return smooth_max_value(x, beta, n)
    def grad_func(x):
        return smooth_max_grad(x, beta, n)

        return np.sum(x) - n / 2.0
    def eq_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n

    try:

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        at_low = h <= 1e-12
        at_high = h >= 1.0 - 1e-12
        mid = ~(at_low | at_high)
        cnt = int(np.count_nonzero(mid))
        if cnt > 0:
            h[mid] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def precompute_dcorr(h, n):
    nfull = 2 * n - 1
    k_range = np.arange(nfull)
    dcorr = np.zeros((n, nfull))
    for i in range(n):
        j1 = i + k_range - (n - 1)
        j2 = i - k_range + (n - 1)
        v1 = (j1 >= 0) & (j1 < n)
        v2 = (j2 >= 0) & (j2 < n)
        dcorr[i, v1] += (1.0 - h[j1[v1]])
        dcorr[i, v2] -= h[j2[v2]]
    return dcorr


def smooth_max_value(x, beta, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50.0, 50.0)
    sm = m + np.log(np.sum(np.exp(shifted))) / beta
    return sm * (2.0 / n)


def smooth_max_grad(x, beta, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50.0, 50.0)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = w / w_sum
    g = np.zeros(n)
    dx = 2.0 / n
    nfull = 2 * n - 1
    for k in range(nfull):
        wk = w[k] * dx
        if wk < 1e-16:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            np_ = n - offset
            g[offset:offset + np_] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[offset:offset + np_]
        else:
            a = -offset
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:a + np_])
            g[a:a + np_] -= wk * x[:np_]
    return g

    def obj_func(x):
        return smooth_max_value(x, beta, n)

    def grad_func(x):
        return smooth_max_grad(x, beta, n)

    def eq_fun(x):
        return np.sum(x) - n / 2.0

    def eq_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(obj_func, h, jac=grad_func, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def structured_init(n, pattern='period1_blocks'):

            candidate[half + blk:half + blk + blk // 2] = 1.0

        if c5_ils < best_c5 - 1e-14:
            best_c5 = c5_ils
            best_h = h_ils.copy()
            h = h_ils.copy()

    # Phase 7: Try different structured initializations at n=512
    n_arch = 512
    if remaining() > 100:
        for pattern in ['period1_blocks', 'alternating']:
            if remaining() < 50:
                break
            h_struct = structured_init(n_arch, pattern)
            h_struct = project_feasible(h_struct, n_arch)
            c5_struct = compute_c5(h_struct)
            if c5_struct < best_c5:
                best_c5 = c5_struct
                best_h = h_struct.copy()
                best_n = n_arch

            h_st_opt, c5_st = steep_descent_local_search(h_struct, n_arch,
                                                          max_passes=200,
                                                          time_limit=min(40.0, remaining() * 0.2))
            if c5_st < best_c5 - 1e-14:
                best_c5 = c5_st
                best_h = h_st_opt.copy()
                best_n = n_arch

    # Phase 8: Multi-start random binary at n=512
    if remaining() > 60:
        for trial in range(4):
            if remaining() < 30:
                break
            h_rand = random_binary_init(512, rng)
            c5_r = compute_c5(h_rand)
            h_r_opt, c5_r_opt = steep_descent_local_search(h_rand, 512,
                                                            max_passes=150,
                                                            time_limit=min(20.0, remaining() * 0.2))
            if c5_r_opt < best_c5 - 1e-14:
                best_c5 = c5_r_opt
                best_h = h_r_opt.copy()
                best_n = 512

    # Phase 9: Final intensive polish at best resolution
    if remaining() > 40:
        h_final, c5_final = steep_descent_local_search(best_h.copy(), best_n,
                                                        max_passes=300,
                                                        time_limit=min(80.0, remaining() * 0.5))
        if c5_final < best_c5 - 1e-15:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Phase 10: Final SLSQP + binary round
    if remaining() > 20:
        h_sl_final = slsqp_optimize(best_h.copy(), best_n, 200000, maxiter=1000)
        c5_slf = compute_c5(h_sl_final)
        if c5_slf < best_c5 - 1e-14:
            best_c5 = c5_slf
            best_h = h_sl_final.copy()

        h_bin_final = binary_round(best_h, best_n)
        c5_bf = compute_c5(h_bin_final)
        if c5_bf < best_c5 - 1e-15:
            best_c5 = c5_bf
            best_h = h_bin_final.copy()

        # One more local search
        if remaining() > 10:
            h_last, c5_last = steep_descent_local_search(best_h, best_n,
                                                          max_passes=100,
                                                          time_limit=min(20.0, remaining() * 0.5))
            if c5_last < best_c5 - 1e-15:
                best_c5 = c5_last
                best_h = h_last.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    final_c5 = min(final_c5, best_c5)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            final_c5 = min(final_c5, eval_c5)
    except Exception:
        pass

    return (best_h, final_c5, best_n)
