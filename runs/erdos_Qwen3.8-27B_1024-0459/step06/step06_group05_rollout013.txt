```python
def grad_corr_k(h, k_shift, n):
    """
    Gradient of corr[k_shift] with respect to h.
    k_shift is 0-indexed in the full correlation array (length 2n-1).
    The actual shift is k_shift - (n-1).
    """
    g = np.zeros(n)
    shift = k_shift - (n - 1)
    if shift >= 0:
        # corr = sum_i h[i+shift] * (1-h[i]) for i in [0, n-shift-1]
        npairs = n - shift
        # d/dh[j] for j in [shift, n-1]: (1-h[j-shift])
        g[shift:] += (1.0 - h[:npairs])
        # d/dh[j] for j in [0, npairs-1]: -h[j+shift]
        g[:npairs] -= h[shift:]
    else:

        npairs = n - a
        # corr = sum_i h[i] * (1-h[i+a]) for i in [0, n-a-1]
        g[:npairs] += (1.0 - h[a:])
        g[a:] -= h[:npairs]
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            if diff > 0:
                # Need to increase some h[i]
                idx = np.where(h < 1.0)[0]
                if len(idx) > 0:
                    h[idx] += diff / len(idx)
                else:
                    break
            else:
                idx = np.where(h > 0.0)[0]
                if len(idx) > 0:
                    h[idx] += diff / len(idx)
                else:
                    break
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_corr_shift(h, shift, n):
    """Gradient of corr at a given shift (in the full correlation array, 0-indexed)."""
    g = np.zeros(n)
    # shift here is the actual lag: shift = k_index - (n-1)
    if shift >= 0:
        npairs = n - shift
        if npairs <= 0:
            return g
        # corr = sum_i h[i+shift]*(1-h[i]) for i=0..npairs-1
        g[shift:] += (1.0 - h[:npairs])
        g[:npairs] -= h[shift:]
    else:
        a = -shift
        npairs = n - a
        if npairs <= 0:
            return g
        # corr = sum_i h[i]*(1-h[i+a]) for i=0..npairs-1
        g[:npairs] += (1.0 - h[a:])
        g[a:] -= h[:npairs]
    return g

def smooth_max_obj_grad(h, n, K, beta):
    corr = compute_corr(h)
    dx = 2.0 / n
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    m = np.max(c_top)
    sh = np.clip(beta * (c_top - m), -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / ex.sum()
    val = (m + np.log(ex.sum()) / beta) * dx
    g = np.zeros(n)
    for i in range(K):
        if w[i] < 1e-14:
            continue
        wk = w[i] * dx
        shift = idx[i] - (n - 1)
        g += wk * grad_corr_shift(h, shift, n)
    return val, g

def nesterov_optimize(h, n, time_limit, K_start=10, K_end=40, beta_start=30, beta_end=300, push_interval=30):
    h_curr = h.copy()
    h_prev = h.copy()
    h_best = h.copy()
    c5_best = compute_c5(h)
    start = time.time()
    max_iter = int(time_limit * 500)  # generous upper bound
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        
        # Adaptive parameters
        frac = min(1.0, it / (max_iter * 0.5))
        K = int(K_start + (K_end - K_start) * frac)
        beta = beta_start + (beta_end - beta_start) * frac
        
        # Nesterov extrapolation
        if it == 0:
            y = h_curr.copy()
        else:
            momentum = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
            momentum = min(momentum, 5.0)  # Cap momentum
            y = h_curr + (momentum - 1.0) / momentum * (h_curr - h_prev)
            y = np.clip(y, 0.0, 1.0)
        
        # Compute objective and gradient at y
        obj_val, g_y = smooth_max_obj_grad(y, n, K, beta)
        gn = np.linalg.norm(g_y)
        if gn < 1e-16:
            break
        
        # Step size with backtracking
        step = 0.1 / gn
        step = min(step, 0.5)
        
        h_new = project_feasible(y - step * g_y, n)
        c5_new = compute_c5(h_new)
        
        # Backtracking if needed
        for bt in range(5):
            if c5_new < obj_val * 1.0 + 0.001:  # Acceptable
                break
            step *= 0.5
            h_new = project_feasible(y - step * g_y, n)
            c5_new = compute_c5(h_new)
        
        h_prev = h_curr.copy()
        h_curr = h_new
        
        # Binary push every push_interval iterations
        if it % push_interval == push_interval - 1 and it > push_interval:
            h_pushed = push_to_binary(h_curr, n, strength=1.5 + 0.5 * frac)
            c5_pushed = compute_c5(h_pushed)
            if c5_pushed < c5_best:
```

```python
                                              K_start=5, K_end=20,
                                              beta_start=100, beta_end=800,
                                              push_every=30)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Final SLSQP polish
    if remaining() > 10:
        h_final = slsqp_opt(best_h, best_n, 8, 80000, maxiter=500)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
```
