    else:
        a = -d
        np_ = n - a
        g[a:] += 1.0 - h[:np_]
        g[:np_] -= h[a:]
    return g


def _sm_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    z = beta * (corr - m)
    z = np.clip(z, -50.0, 50.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    f = (m + np.log(S) / beta) * dx
    gap = np.sum(h) - n / 2.0
    f += 0.5 * _PEN_LAMBDA * gap * gap
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        g += wk * _corr_grad_one(h, k)
    g += _PEN_LAMBDA * gap
    return f, g


def lbfgs_opt(h, n, beta, maxiter=2000, gtol=1e-9):
    global _CUR_N, _SM_BETA
    _CUR_N = n
    _SM_BETA = beta
    h0 = project_feasible(h.copy(), n)
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return _sm_obj_grad(x, beta)[0]

    def grad(x):
        return _sm_obj_grad(x, beta)[1]

    try:
        res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': gtol, 'maxls': 40,
                                'maxfun': maxiter * 5})
        h_new = project_feasible(res.x, n)
        if compute_c5(h_new) < compute_c5(h0):
            return h_new
    except Exception:
        pass
    return h0


def slsqp_opt(h, n, beta, maxiter=600):
    global _CUR_N, _SM_BETA
    _CUR_N = n
    _SM_BETA = beta
    h0 = project_feasible(h.copy(), n)
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return _sm_obj_grad(x, beta)[0]

    def grad(x):
        return _sm_obj_grad(x, beta)[1]

    def cfun(x):
        return np.sum(x) - n / 2.0

    def cjac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        if compute_c5(h_new) < compute_c5(h0):
            return h_new
    except Exception:
        pass
    return h0


def topk_descent(h, n, top_k=8, max_iters=2000, lr0=0.4):
    h = project_feasible(h.copy(), n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(max_iters):
        if remaining_time() < 8:
            break
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        for ki in idx_top:
            g += _corr_grad_one(h, ki)
        g *= dx
        # Add constraint gradient
        gap = np.sum(h) - n / 2.0
        g += 5.0 * gap
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        d = -g / gn
        improved = False
        lr = lr0 / (1.0 + 0.005 * it)
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05, 2.0, 3.0]:
            step = lr * mult * gn * 0.05
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-15:
                best_c5 = cn
                best_h = ht.copy()
                h = ht.copy()
                improved = True
                break
        if not improved:
            for mult in [0.1, 0.05, 0.02, 0.01]:
                step = lr * mult * gn * 0.05
                ht = project_feasible(h - step * d, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-15:
                    best_c5 = cn
                    best_h = ht.copy()
                    h = ht.copy()
                    improved = True
                    break
        if not improved and it > 50:
            lr *= 0.5
            if lr < 0.001:
                break
    return best_h, best_c5


def pair_swap_descent(h, n, max_iters=8000, rng=None):
    h = project_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    if rng is None:
        rng = np.random.RandomState(42)
    for it in range(max_iters):
        if remaining_time() < 8:
            break
        corr = compute_corr(h)
        curr_max = np.max(corr)
        # Find positions where h is near 1 and near 0
        ones_mask = h > 0.7
        zeros_mask = h < 0.3
        ones_idx = np.where(ones_mask)[0]
        zeros_idx = np.where(zeros_mask)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        # Try a random swap
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        if i == j:
            continue
        h[i], h[j] = h[j], h[i]
        new_max = np.max(compute_corr(h))
        if new_max < curr_max - 1e-14:
            best_c5 = new_max * (2.0 / n)
            best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_c5


def coord_descent_fast(h, n, max_sweeps=30, top_k=5):
    h = project_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if remaining_time() < 8:
            break
        any_improve = False
        corr = compute_corr(h)
        top_idx = set(np.argsort(-corr)[:top_k].tolist())
        for j in range(n):
            if remaining_time() < 5:
                break
            old_val = h[j]
            best_v = old_val
            best_obj = compute_c5(h)
            a, b = 0.0, 1.0
            for _ in range(25):
                m1 = a + (b - a) * 0.381966
                m2 = a + (b - a) * 0.618034
                h[j] = m1
                c1 = compute_c5(h)
                h[j] = m2
                c2 = compute_c5(h)
                if c1 < c2:
                    b = m2
                    h[j] = m1
                    if c1 < best_obj:
                        best_obj = c1
                        best_v = m1
                else:
                    a = m1
                    h[j] = m2
                    if c2 < best_obj:
                        best_obj = c2
                        best_v = m2
            h[j] = best_v
            curr = compute_c5(h)
            if curr < best_c5 - 1e-14:
                best_c5 = curr
                any_improve = True
        h = project_feasible(h, n)
        actual = compute_c5(h)
        if actual < best_c5:
            best_c5 = actual
        if not any_improve and sweep > 2:
            break
    return h, best_c5


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return project_feasible(src.copy(), dst_n)
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    h = np.interp(xn, xp, src)
    return project_feasible(h, dst_n)


def theta_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _PEN_LAMBDA
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    # Initialize from parent or default
    try:
        best_h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        best_h = np.zeros(640)
        best_h[:320] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B at parent resolution with increasing beta
    h = best_h.copy()
    for beta in [200, 1000, 5000, 20000, 100000]:
        if remaining_time() < 30:
            break
        h_new = lbfgs_opt(h, best_n, beta, maxiter=1500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Top-k descent
    if remaining_time() > 30:
        h_tk, c5_tk = topk_descent(h, best_n, top_k=8, max_iters=3000, lr0=0.5)
        if c5_tk < best_c5 - 1e-14:
            best_c5 = c5_tk
            best_h = h_tk.copy()
            h = h_tk.copy()

    # Phase 3: SLSQP refinement
    for beta in [10000, 50000, 200000]:
        if remaining_time() < 25:
            break
        h_new = slsqp_opt(h, best_n, beta, maxiter=800)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 4: Multi-resolution search
    resolutions = [1024, 1280, 1536, 768, 512]
    for n_t in resolutions:
        if remaining_time() < 60:
            break
        if n_t % 2 != 0:
            n_t += 1
        h_interp = interp_to(best_h, best_n, n_t)
        c5_int = compute_c5(h_interp)

        h_opt = h_interp.copy()
        for beta in [2000, 20000, 100000]:
            if remaining_time() < 25:
                break
            h_opt = lbfgs_opt(h_opt, n_t, beta, maxiter=1000)

        if remaining_time() > 20:
            h_opt, c5_opt = topk_descent(h_opt, n_t, top_k=8, max_iters=1500, lr0=0.4)
        else:
            c5_opt = compute_c5(h_opt)

        if remaining_time() > 20:
            h_opt2 = slsqp_opt(h_opt, n_t, 100000, maxiter=500)
            c5_opt2 = compute_c5(h_opt2)
            if c5_opt2 < c5_opt:
                h_opt = h_opt2
                c5_opt = c5_opt2

        if c5_opt < best_c5 - 1e-14:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t
            h = h_opt.copy()

    # Phase 5: Further refinement at best resolution
    if remaining_time() > 40:
        h_new = lbfgs_opt(h, best_n, 50000, maxiter=2000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 6: Top-k descent with more iterations
    if remaining_time() > 30:
        h_tk2, c5_tk2 = topk_descent(h, best_n, top_k=10, max_iters=4000, lr0=0.3)
        if c5_tk2 < best_c5 - 1e-14:
            best_c5 = c5_tk2
            best_h = h_tk2.copy()
            h = h_tk2.copy()

    # Phase 7: Coordinate descent on the best solution
    if remaining_time() > 30:
        h_cd, c5_cd = coord_descent_fast(h, best_n, max_sweeps=20, top_k=6)
        if c5_cd < best_c5 - 1e-14:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h = h_cd.copy()

    # Phase 8: Binary pair swaps (if near-binary)
    if remaining_time() > 20:
        h_bin = (best_h > 0.5).astype(float).copy()
        n_half = best_n // 2
        n_ones = int(np.sum(h_bin))
        if n_ones != n_half:
            sort_idx = np.argsort(-h_bin)
            h_bin = np.zeros(best_n)
            h_bin[sort_idx[:n_half]] = 1.0
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5 + 0.001:
            h_ps, c5_ps = pair_swap_descent(h_bin, best_n, max_iters=5000, rng=rng)
            if c5_ps < best_c5 - 1e-14:
                best_c5 = c5_ps
                best_h = h_ps.copy()
            # Try continuous refinement from binary
            if remaining_time() > 15:
                h_cref = project_feasible(h_ps, best_n)
                h_cref = lbfgs_opt(h_cref, best_n, 5000, maxiter=500)
                c5_cr = compute_c5(h_cref)
                if c5_cr < best_c5 - 1e-14:
                    best_c5 = c5_cr
                    best_h = h_cref.copy()

    # Phase 9: Alternative start - Thue-Morse
    if remaining_time() > 40:
        h_tm = theta_morse(best_n)
        h_tm = lbfgs_opt(h_tm, best_n, 5000, maxiter=800)
        h_tm, c5_tm = topk_descent(h_tm, best_n, top_k=6, max_iters=1500, lr0=0.3)
        c5_tm = compute_c5(h_tm)
        if c5_tm < best_c5 - 1e-14:
            best_c5 = c5_tm
            best_h = h_tm.copy()

    # Phase 10: Perturbation + re-optimization
    if remaining_time() > 40:
        for trial in range(3):
            if remaining_time() < 15:
                break
            pert = best_h + 0.02 * rng.randn(best_n)
            pert = project_feasible(pert, best_n)
            h_p = lbfgs_opt(pert, best_n, 20000, maxiter=800)
            h_p, c5_p = topk_descent(h_p, best_n, top_k=8, max_iters=1000, lr0=0.3)
            c5_p = compute_c5(h_p)
            if c5_p < best_c5 - 1e-14:
                best_c5 = c5_p
                best_h = h_p.copy()

    # Phase 11: Final L-BFGS-B polish
    if remaining_time() > 15:
        h_final = lbfgs_opt(best_h, best_n, 200000, maxiter=1500, gtol=1e-10)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-14:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Phase 12: Final top-k squeeze
    if remaining_time() > 10:
        h_s, c5_s = topk_descent(best_h, best_n, top_k=12, max_iters=2000, lr0=0.2)
        if c5_s < best_c5 - 1e-14:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Ensure feasibility and compute final
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            if eval_c5 < best_c5:
                best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
