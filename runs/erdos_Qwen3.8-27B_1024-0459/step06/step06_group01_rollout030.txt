Wait, I need to double-check the gradient computation. Let me re-derive it carefully.

The correlation is:
corr[k] = sum_i h[i] * (1 - h[i + k - (n-1)])

where the sum is over valid indices (both i and i+k-(n-1) in [0, n-1]).

Let me define offset = k - (n-1).

Case 1: offset >= 0
corr[k] = sum_{i=0}^{n-1-offset} h[i+offset] * (1 - h[i])
        = sum_{i=0}^{n-1-offset} h[i+offset] - sum_{i=0}^{n-1-offset} h[i+offset] * h[i]

d(corr[k])/d(h[j]):
- For the first sum: h[j] appears when j = i+offset, i.e., i = j-offset. Valid when 0 <= j-offset <= n-1-offset, i.e., offset <= j <= n-1.
  Contribution: +1
- For the second sum: h[j] appears as h[i+offset] when j = i+offset (i.e., i = j-offset), or as h[i] when j = i.
  - As h[i+offset]: derivative = -h[i] = -h[j-offset], valid when offset <= j <= n-1
  - As h[i]: derivative = -h[i+offset] = -h[j+offset], valid when 0 <= j <= n-1-offset

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_T0 = 0.0
_BUDGET = 1300.0

def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def weighted_gradient(h, n, weights):
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if abs(wk) < 1e-18:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - h[a:])
            g[a:] -= wk * h[:np_]
    return g * dx

def slsqp_smooth_max(h, n, beta, maxiter=400):
    dx = 2.0 / n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        z = beta * (corr - m)
        z = np.clip(z, -50.0, 50.0)
        ez = np.exp(z)
        return (m + np.log(np.sum(ez)) / beta) * dx

    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        z = beta * (corr - m)
        z = np.clip(z, -50.0, 50.0)
        ez = np.exp(z)
        S = np.sum(ez)
        w = ez / S
        return weighted_gradient(x, n, w)

    def eq_fun(x):
        return np.sum(x) - n / 2.0

    def eq_jac(x):
        return np.ones(n)

    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0

def lbfgsb_penalty(h, n, beta, maxiter=2000):
    dx = 2.0 / n
    target = n / 2.0
    mu = 100.0

    def objective(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        z = beta * (corr - m)
        z = np.clip(z, -50.0, 50.0)
        ez = np.exp(z)
        sm = (m + np.log(np.sum(ez)) / beta) * dx
        pen = mu * (np.sum(x) - target) ** 2
        return sm + pen

    def gradient(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        z = beta * (corr - m)
        z = np.clip(z, -50.0, 50.0)
        ez = np.exp(z)
        S = np.sum(ez)
        w = ez / S
        g = weighted_gradient(x, n, w)
        g += 2.0 * mu * (np.sum(x) - target)
        return g

    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(objective, h0, jac=gradient, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxfun': 5000})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0

def active_set_lp(h, n, top_k, max_iter=80):
    h = project_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    dx = 2.0 / n

    for iteration in range(max_iter):
        if remaining_time() < 15:
            break

        corr = compute_corr(h)
        top_indices = np.argsort(-corr)[:top_k]

        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]

        for k in top_indices:
            off = k - (n - 1)
            grad_k = np.zeros(n)
            if off >= 0:
                np_ = n - off
                grad_k[off:] = (1.0 - h[:np_])
                grad_k[:np_] -= h[off:]
            else:
                a = -off
                np_ = n - a
                grad_k[:np_] = (1.0 - h[a:])
                grad_k[a:] -= h[:np_]

            const = (corr[k] - np.dot(grad_k, h)) * dx
            coeff = grad_k * dx
            constraints.append(t_var >= const + coeff @ h_var)

        prob = cp.Problem(cp.Minimize(t_var), constraints)
        solved = False
        for solver in ['ECOS', 'CLARABEL', 'GLPK']:
            try:
                prob.solve(solver=solver, verbose=False)
                solved = True
                break
            except Exception:
                continue

        if not solved or h_var.value is None:
            break

        h_lp = np.array(h_var.value).flatten()
        h_lp = project_feasible(h_lp, n)

        c5_lp = compute_c5(h_lp)
        if c5_lp < best_c5 - 1e-14:
            best_c5 = c5_lp
            best_h = h_lp.copy()
            h = h_lp.copy()
        else:
            improved = False
            for alpha in [0.7, 0.5, 0.3, 0.1, 0.05]:
                h_frac = project_feasible(h + alpha * (h_lp - h), n)
                c5_frac = compute_c5(h_frac)
                if c5_frac < best_c5 - 1e-14:
                    best_c5 = c5_frac
                    best_h = h_frac.copy()
                    h = h_frac.copy()
                    improved = True
                    break
            if not improved:
                break

    return best_h, best_c5

def topk_equalize(h, n, top_k=7, max_iters=600):
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining_time() < 5:
            break
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        c_max = corr[idx_top[0]]
        c_min_top = corr[idx_top[-1]]
        if c_max - c_min_top < 1e-10:
            break
        g_dir = np.zeros(n)
        for pos, k in enumerate(idx_top):
            w_k = 1.0 if pos == 0 else (-1.0 / (len(idx_top) - 1))
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_dir[off:] += w_k * (1.0 - h[:np_])
                g_dir[:np_] -= w_k * h[off:]
            else:
                a = -off
                np_ = n - a
                g_dir[:np_] += w_k * (1.0 - h[a:])
                g_dir[a:] -= w_k * h[:np_]
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        d = g_dir / gn
        improved = False
        for step in [0.002, 0.005, 0.01, 0.02, 0.001, 0.008, 0.03, 0.05, 0.08]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-13:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            for step in [0.005, 0.01, 0.02]:
                ht = project_feasible(h - step * d, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-13:
                    h = ht
                    best_c5 = cn
                    improved = True
                    break
            if not improved:
                break
    return h, best_c5

def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)

def binary_sa(h, n, time_limit, rng):
    h = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        sort_idx = np.argsort(-h)
        h = np.zeros(n)
        h[sort_idx[:n_half]] = 1.0
    start = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    current_c5 = best_c5
    T = 1.0
    cool_rate = 0.9995
    no_improve = 0
    max_no_improve = 5000
    while time.time() - start < time_limit and remaining_time() > 15:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            current_c5 = new_c5
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
        T *= cool_rate
        if no_improve > max_no_improve:
            break
    return best_h, best_c5

def theta_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)

def subgradient_descend(h, n, top_k=10, max_iters=3000):
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr0 = 0.5
    for it in range(max_iters):
        if remaining_time() < 5:
            break
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        weights = np.zeros(2 * n - 1)
        for ki in idx_top:
            weights[ki] = 1.0 / top_k
        g = weighted_gradient(h, n, weights)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        d = -g / gn
        lr = lr0 / (1.0 + 0.005 * it)
        step = lr * dx
        ht = project_feasible(h + step * d, n)
        c5_new = compute_c5(ht)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = ht.copy()
            h = ht.copy()
        else:
            h = project_feasible(h + 0.5 * step * d, n)
    return best_h, best_c5

            h_new, c5_tk = topk_equalize(h_new, n_t, top_k=7, max_iters=300)
            if c5_tk < c5_new:
                c5_new = c5_tk

        if remaining_time() > 40:
            h_lp, c5_lp = active_set_lp(h_new, n_t, top_k=30, max_iter=40)
            if c5_lp < c5_new:
                h_new = h_lp
                c5_new = c5_lp

        if remaining_time() > 30:
            h_new = lbfgsb_penalty(h_new, n_t, 50000, maxiter=800)
            c5_new = compute_c5(h_new)

        print(f"  n={n_t}: C5={c5_new:.8f}")

        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = n_t
            h = h_new.copy()

    # Phase 7: Binary SA + continuous refinement
    if remaining_time() > 80:
        h_bin = (best_h > 0.5).astype(float)
        sa_time = min(80, max(15, int(remaining_time() * 0.2)))
        h_b, c5_b = binary_sa(h_bin, best_n, sa_time, rng)

        h_bc = project_feasible(h_b, best_n)
        h_bc = lbfgsb_penalty(h_bc, best_n, 10000, maxiter=800)
        c5_bc = compute_c5(h_bc)
        h_bc, c5_bc = topk_equalize(h_bc, best_n, top_k=7, max_iters=300)

        if c5_bc < best_c5 - 1e-14:
            best_c5 = c5_bc
            best_h = h_bc.copy()
        if c5_b < best_c5 - 1e-14:
            best_c5 = c5_b
            best_h = h_b.copy()
        print(f"  SA+refine: binary={c5_b:.8f}, continuous={c5_bc:.8f}")

    # Phase 8: Alternative initializations at best resolution
    if remaining_time() > 60:
        h_tm = theta_morse(best_n)
        h_tm = lbfgsb_penalty(h_tm, best_n, 5000, maxiter=800)
        c5_tm = compute_c5(h_tm)
        h_tm, c5_tm = topk_equalize(h_tm, best_n, top_k=7, max_iters=200)
        if remaining_time() > 30:
            h_tm, c5_tm = active_set_lp(h_tm, best_n, top_k=20, max_iter=30)
            c5_tm = compute_c5(h_tm)
        if c5_tm < best_c5 - 1e-14:
            best_c5 = c5_tm
            best_h = h_tm.copy()
        print(f"  Morse: C5={c5_tm:.8f}")

    # Phase 9: Final polish
    if remaining_time() > 20:
        h_final = lbfgsb_penalty(best_h, best_n, 100000, maxiter=1000)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-14:
            best_c5 = c5_final
            best_h = h_final.copy()

    if remaining_time() > 15:
        h_eq2, c5_eq2 = topk_equalize(best_h, best_n, top_k=10, max_iters=300)
        if c5_eq2 < best_c5 - 1e-14:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()

    if remaining_time() > 15:
        h_lp2, c5_lp2 = active_set_lp(best_h, best_n, top_k=40, max_iter=30)
        if c5_lp2 < best_c5 - 1e-14:
            best_c5 = c5_lp2
            best_h = h_lp2.copy()

    # Final feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    print(f"Final: n={best_n}, C5={best_c5:.8f}")

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
