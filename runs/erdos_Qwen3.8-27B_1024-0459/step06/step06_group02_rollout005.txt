    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):

        if cnt > 0:

            h = np.clip(h, 0.0, 1.0)
        else:
            break
    return h

def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        if npairs > 0:
            g[offset:offset+npairs] += (1.0 - h[:npairs])
            g[:npairs] -= h[offset:offset+npairs]

        if npairs > 0:

            g[a:a+npairs] -= h[:npairs]
    return g

def smooth_max_obj(x, p, penalty):
    corr = compute_corr(x)
    # p-norm approximation of max
    m = len(corr)
    val = (np.sum(np.abs(corr)**p) / m)**(1.0/p)
    # Add penalty for constraint violation
    target = len(x) / 2.0
    pen = penalty * (np.sum(x) - target)**2
    return val * (2.0/len(x)) + pen

def smooth_max_grad(x, p, penalty):
    corr = compute_corr(x)
    m = len(corr)
    # Gradient of (sum |corr_k|^p / m)^(1/p) wrt x
    # = (1/p) * (sum |corr_k|^p / m)^(1/p - 1) * (1/m) * p * |corr_k|^(p-1) * sign(corr_k) * dcorr_k/dx
    # = (sum |corr_k|^p / m)^(1/p - 1) * (1/m) * |corr_k|^(p-1) * sign(corr_k) * dcorr_k/dx
    base = (np.sum(np.abs(corr)**p) / m)**(1.0/p - 1.0)
    weights = (np.abs(corr)**(p-1) * np.sign(corr)) / m * base
    # For very large p, weights concentrate on the max
