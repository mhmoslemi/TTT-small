        c_max = cvals[0]
        c_min = cvals[-1]
        if c_max - c_min < 1e-13:
            break

        g_dir = np.zeros(n)
        for k in idx:
            gk = grad_corr_k(h, k, n)
            if cvals[idx.tolist().index(k)] >= c_max - (c_max - c_min) * 0.3:
                g_dir += gk
            elif cvals[idx.tolist().index(k)] <= c_min + (c_max - c_min) * 0.3:
                g_dir -= gk

        g_norm = np.linalg.norm(g_dir)
        if g_norm < 1e-15:
            break
        g_dir /= g_norm

        improved = False
        for step in [0.03, 0.01, 0.005, 0.015, 0.05, 0.002, 0.008]:
            h_try = project_feasible(h - step * g_dir, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                h = h_try
                best_c5 = c5_try
                improved = True
                break
        if not improved:
            h_try = project_feasible(h + 0.01 * g_dir, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                h = h_try
                best_c5 = c5_try
                improved = True
            if not improved:
                break
    return h, best_c5


def refined_scan_optimize(h, n, time_limit, K, wbeta):
    """Ultra-fine optimization with very small steps focused on top K."""
    global _T0, _BUDGET
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()

    for it in range(5000):
        if time.time() - start > time_limit or remaining_time() < 3:
            break
        g, cmax = get_subgrad(h, n, K, wbeta)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        d = -g / g_norm
        improved = False
        for s in [0.001, 0.002, 0.005, 0.003, 0.008, 0.0005, 0.01]:
            h_try = project_feasible(h + s * d, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-16:
                h = h_try
                best_c5 = c5_try
                best_h = h.copy()
                improved = True
                break
        if not improved:
            frac_idx = np.where((h > 0.02) & (h < 0.98))[0]
            if len(frac_idx) > 0:
                idx_p = np.random.choice(frac_idx, size=min(5, len(frac_idx)), replace=False)
                for i in idx_p:
                    h[i] = np.clip(h[i] + np.random.randn() * 0.05, 0, 1)
                h = project_feasible(h, n)
            else:
                break
            if it % 200 == 0 and it > 0:
                h = best_h.copy()
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    np.random.seed(seed)

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
    except Exception:
        parent_n = 512
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0

    best_h = parent_h.copy()
    best_c5 = compute_c5(parent_h)
    best_n = parent_n

    n = 1024
    h = interp_to(best_h, best_n, n)
    h = project_feasible(h, n)
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
        best_n = n

    # Phase 1: L-BFGS-B global (beta=200)
    if remaining_time() > 80:
        h_new = smooth_max_lbfgs(h, n, beta=200, penalty=500, maxiter=3000, time_limit=70)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Gradient scan K=25
    if remaining_time() > 100:
        h, c5 = gradient_optimize(h, n, min(100.0, remaining_time() - 5), K=25, wbeta=30.0, init_step=0.08)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 3: L-BFGS-B (beta=1000)
    if remaining_time() > 60:
        h_new = smooth_max_lbfgs(h, n, beta=1000, penalty=2000, maxiter=4000, time_limit=55)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 4: Gradient scan K=12
    if remaining_time() > 80:
        h, c5 = gradient_optimize(h, n, min(80.0, remaining_time() - 5), K=12, wbeta=60.0, init_step=0.04)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 5: L-BFGS-B (beta=5000)
    if remaining_time() > 50:
        h_new = smooth_max_lbfgs(h, n, beta=5000, penalty=5000, maxiter=4000, time_limit=45)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 6: Gradient scan K=6
    if remaining_time() > 60:
        h, c5 = gradient_optimize(h, n, min(60.0, remaining_time() - 5), K=6, wbeta=100.0, init_step=0.02)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 7: Equalize top 4
    if remaining_time() > 40:
        h_eq, c5_eq = equalize_top(h, n, K=4, time_limit=min(40.0, remaining_time() - 5))
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 8: L-BFGS-B (beta=20000)
    if remaining_time() > 40:
        h_new = smooth_max_lbfgs(h, n, beta=20000, penalty=10000, maxiter=3000, time_limit=35)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 9: Ultra-fine K=3
    if remaining_time() > 40:
        h_fine, c5_fine = refined_scan_optimize(h, n, min(40.0, remaining_time() - 5), K=3, wbeta=200.0)
        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_fine.copy()

    # Phase 10: Equalize top 3
    if remaining_time() > 25:
        h_eq, c5_eq = equalize_top(best_h, best_n, K=3, time_limit=min(25.0, remaining_time() - 5))
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Phase 11: Try n=2048
    if remaining_time() > 80 and best_n < 2048:
        n2 = 2048
        h2 = interp_to(best_h, best_n, n2)
        h2 = project_feasible(h2, n2)
        c5_2 = compute_c5(h2)
        if remaining_time() > 40:
            h2, c5_2 = gradient_optimize(h2, n2, min(50.0, remaining_time() * 0.5), K=15, wbeta=50.0, init_step=0.04)
        if c5_2 < best_c5:
            if remaining_time() > 20:
                h2_new = smooth_max_lbfgs(h2, n2, beta=5000, penalty=5000, maxiter=2000, time_limit=min(15.0, remaining_time() - 5))
                c5_2_new = compute_c5(h2_new)
                if c5_2_new < c5_2:
                    h2 = h2_new
                    c5_2 = c5_2_new
            best_c5 = c5_2
            best_h = h2.copy()
            best_n = n2

    # Phase 12: Final L-BFGS-B polish
    if remaining_time() > 25:
        h_new = smooth_max_lbfgs(best_h, best_n, beta=50000, penalty=20000, maxiter=2000,
                                  time_limit=min(20.0, remaining_time() - 5))
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 13: Final ultra-fine
    if remaining_time() > 15:
        h_fine, c5_fine = refined_scan_optimize(best_h, best_n, min(12.0, remaining_time() - 3), K=3, wbeta=300.0)
        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_fine.copy()

    # Final projection
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
