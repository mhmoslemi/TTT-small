                if h[i] == 1 and h[i + off] == 0:
                    # Swapping h[i] to 0 would reduce corr at this lag
                    candidates.append((i, i + off))
        else:
            ao = -off
            for i in range(n - ao):
                if h[i + ao] == 1 and h[i] == 0:
                    candidates.append((i + ao, i))
        
        if not candidates:
            break
        
        # Try the top candidates
        rng.shuffle(candidates)
        improved = False
        for (i1, i0) in candidates[:20]:
            h_try = h.copy()
            h_try[i1] = 0
            h_try[i0] = 1
            c_try = compute_c5(h_try)
            if c_try < best_c5 - 1e-15:
                h = h_try
                best_c5 = c_try
                best_h = h.copy()
                ones = set(np.where(h == 1)[0].tolist())
                zeros = set(np.where(h == 0)[0].tolist())
                improved = True
                break
        
        if not improved:
            # Try random swap
            if ones and zeros:
                i1 = list(ones)[rng.randint(len(ones))]
                i0 = list(zeros)[rng.randint(len(zeros))]
                h[i1] = 0
                h[i0] = 1
                ones.discard(i1)
                zeros.discard(i0)
                ones.add(i0)
                zeros.add(i1)
            else:
                break
    
    return best_h, best_c5
