        m = n - a
        g[:m] += (1.0 - h[a:])

            if w[i] > 1e-14:
                g += w[i] * grad_corr_k(h, idx[i], n)
        g /= n * 2.0
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        if h_prev is not None and g_prev is not None:
            alpha = barzilai_borwein_step(h_prev, g_prev, h, g, n)
        h_prev = h.copy()
        g_prev = g.copy()
        h_new = project_feasible(h - alpha * g / gn, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            alpha *= 0.5
            h_new = project_feasible(h - alpha * g / gn, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
        it += 1
    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def nelder_mead_polish(h, n, max_iter=200, time_limit=60.0):
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    # Perturb K nearest-neighbor points to create initial simplex
    K = 8
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(-corr)[:K]
    g_avg = np.zeros(n)
    for k in idx:
        g_avg += grad_corr_k(h, k, n)
    g_avg /= K
    gn = np.linalg.norm(g_avg)
    if gn < 1e-14:
        return h, best_c5
    direction = g_avg / gn
    # Try small perturbations
    for step_size in [0.01, 0.005, 0.02, 0.002, 0.008]:
        for sign in [-1.0, 1.0]:
            h_try = project_feasible(h + sign * step_size * direction, n)
            c_try = compute_c5(h_try)
            if c_try < best_c5:
                h = h_try
                best_c5 = c_try
    # Local random search
    rng = np.random.RandomState(42)
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 5:
            break
        # Random direction weighted toward top-K gradient
        r = rng.randn(n)
        r = project_feasible(r - r.mean(), n)
        h_try = project_feasible(h + 0.005 * r / (np.linalg.norm(r) + 1e-10), n)
        c_try = compute_c5(h_try)
        if c_try < best_c5:
            h = h_try
            best_c5 = c_try
    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _G_n, _G_beta, _G_K, _G_klist
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 512
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # === Phase 1: Optimization at parent resolution ===
    n = best_n
    h = best_h.copy()

    # 1a: Projected gradient descent (quick basin finding)
    if remaining() > 20:
        h_pg, c5_pg = projected_gradient_descent(h, n, min(25.0, remaining() * 0.08), K=30, beta=40.0)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
            h = h_pg.copy()

    # 1b: Multi-stage smooth-max SLSQP
    beta_stages = [30, 80, 200, 600, 1500, 4000, 10000, 25000]
    for beta in beta_stages:
        if remaining() < 12:
            break
        K_use = min(50, max(20, n // 12))
        maxit = 500 if beta < 500 else 350
        h_new = slsqop_smmax(h, n, beta, K_use, maxiter=maxit)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # 1c: Active-set min-max SLSQP (grow active set)
    if remaining() > 40:
        for K_active in [5, 8, 12, 16, 24, 32, 42]:
            if remaining() < 10:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            idx_active = np.argsort(-corr)[:K_active]
            h_am, c5_am = slsqop_minmax(h, n, idx_active, maxiter=350)
            if c5_am < best_c5:
                best_c5 = c5_am
                best_h = h_am.copy()
                h = h_am.copy()

    # 1d: Equalize top shifts
    if remaining() > 20:
        h_eq, c5_eq = equalize_top_shifts(h, n, K=10, max_iters=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # 1e: Another round of SLSQP after equalization
    for beta in [5000, 15000, 40000]:
        if remaining() < 10:
            break
        h = slsqop_smmax(h, n, beta, K=50, maxiter=450)
        c5_t = compute_c5(h)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h.copy()
            h = h.copy()

    # 1f: Final min-max with large active set
    if remaining() > 25:
        corr = np.correlate(h, 1.0 - h, mode='full')
        for K_a in [40, 55, 70, 90]:
            if remaining() < 8:
                break
            idx_a = np.argsort(-corr)[:min(K_a, 2 * n - 1)]
            h_am, c5_am = slsqop_minmax(h, n, idx_a, maxiter=400)
            if c5_am < best_c5:
                best_c5 = c5_am
                best_h = h_am.copy()
                h = h_am.copy()

    # === Phase 2: Try higher resolution ===
    n_high = 2 * best_n
    if remaining() > 150:
        h_hi = interp_to(best_h, best_n, n_high)
        h_hi = project_feasible(h_hi, n_high)
        c5_hi = compute_c5(h_hi)
        if c5_hi <= best_c5 + 0.002:
            # Optimize at high resolution
            if remaining() > 20:
                h_hi, c5_hi = projected_gradient_descent(h_hi, n_high, min(20.0, remaining() * 0.08), K=35, beta=40.0)

            for beta in [50, 200, 800, 3000, 10000, 30000]:
                if remaining() < 12:
                    break
                h_hi = slsqop_smmax(h_hi, n_high, beta, K=min(50, n_high // 12), maxiter=350)
                c5_hi = compute_c5(h_hi)

            if remaining() > 30:
                corr_hi = np.correlate(h_hi, 1.0 - h_hi, mode='full')
                for K_a in [8, 14, 20, 30, 40]:
                    if remaining() < 10:
                        break
                    idx_a = np.argsort(-corr_hi)[:K_a]
                    h_hi, c5_hi = slsqop_minmax(h_hi, n_high, idx_a, maxiter=300)

            if remaining() > 20:
                h_hi, c5_hi = equalize_top_shifts(h_hi, n_high, K=10, max_iters=250)

            # More SLSQP at high res
            for beta in [10000, 30000, 80000]:
                if remaining() < 10:
                    break
                h_hi = slsqop_smmax(h_hi, n_high, beta, K=50, maxiter=400)
                c5_hi = compute_c5(h_hi)

            if c5_hi < best_c5:
                best_c5 = c5_hi
                best_h = h_hi.copy()
                best_n = n_high

    # === Phase 3: Try another resolution ===
    n_alt = 4 * best_n // (best_n // 256)  # Try to get ~1024 or 2048
    n_alt = min(max(n_alt, 256), 2048)
    if n_alt != best_n and remaining() > 120:
        h_alt = interp_to(best_h, best_n, n_alt)
        h_alt = project_feasible(h_alt, n_alt)
        c5_alt = compute_c5(h_alt)
        if c5_alt <= best_c5 + 0.003:
            if remaining() > 20:
                h_alt, c5_alt = projected_gradient_descent(h_alt, n_alt, min(15.0, remaining() * 0.1), K=35, beta=50.0)
            for beta in [100, 500, 2000, 8000, 25000]:
                if remaining() < 10:
                    break
                h_alt = slsqop_smmax(h_alt, n_alt, beta, K=min(50, n_alt // 12), maxiter=300)
                c5_alt = compute_c5(h_alt)
            if remaining() > 25:
                corr_alt = np.correlate(h_alt, 1.0 - h_alt, mode='full')
                for K_a in [10, 20, 35]:
                    if remaining() < 8:
                        break
                    idx_a = np.argsort(-corr_alt)[:K_a]
                    h_alt, c5_alt = slsqop_minmax(h_alt, n_alt, idx_a, maxiter=250)
            if remaining() > 15:
                h_alt, c5_alt = equalize_top_shifts(h_alt, n_alt, K=10, max_iters=200)
            if c5_alt < best_c5:
                best_c5 = c5_alt
                best_h = h_alt.copy()
                best_n = n_alt

    # === Phase 4: Intensive refinement at best resolution ===
    n_best = best_n
    h_best = best_h.copy()

    # Cycle of alternating methods
    for cycle in range(4):
        if remaining() < 30:
            break

        # Smooth-max SLSQP with high beta
        if remaining() > 15:
            for beta in [8000, 25000]:
                if remaining() < 8:
                    break
                h_best = slsqop_smmax(h_best, n_best, beta, K=50, maxiter=500)
            c5_cycle = compute_c5(h_best)
            if c5_cycle < best_c5:
                best_c5 = c5_cycle
                best_h = h_best.copy()

        # Min-max with growing active set
        if remaining() > 20:
            corr_c = np.correlate(h_best, 1.0 - h_best, mode='full')
            for K_a in [15, 25, 40, 60]:
                if remaining() < 8:
                    break
                idx_a = np.argsort(-corr_c)[:K_a]
                h_best, c5_cycle = slsqop_minmax(h_best, n_best, idx_a, maxiter=350)
                if c5_cycle < best_c5:
                    best_c5 = c5_cycle
                    best_h = h_best.copy()

        # Equalize
        if remaining() > 12:
            h_best, c5_cycle = equalize_top_shifts(h_best, n_best, K=12, max_iters=250)
            if c5_cycle < best_c5:
                best_c5 = c5_cycle
                best_h = h_best.copy()
                h_best = h_best.copy()

    # === Phase 5: Final fine polish ===
    h_final = best_h.copy()

    # Tiny gradient steps
    if remaining() > 15:
        for it in range(200):
            if remaining() < 5:
                break
            corr_f = np.correlate(h_final, 1.0 - h_final, mode='full')
            idx_f = np.argsort(-corr_f)[:20]
            # Check if top shifts are well-equalized
            c_top = corr_f[idx_f]
            c_gap = c_top.max() - c_top.min()
            if c_gap < 1e-10:
                break
            g_f = np.zeros(n_best)
            # Weight by "violations" (how much above the minimum of top K)
            for i, k in enumerate(idx_f):
                w_i = max(0, corr_f[k] - c_top.min()) + 0.1
                g_f += w_i * grad_corr_k(h_final, k, n_best)
            g_f /= (np.linalg.norm(g_f) + 1e-15)
            improved = False
            for step in [0.005, 0.002, 0.001, 0.008, 0.003, 0.0005]:
                h_try = project_feasible(h_final - step * g_f, n_best)
                c_try = compute_c5(h_try)
                if c_try < best_c5:
                    best_c5 = c_try
                    best_h = h_try.copy()
                    h_final = h_try.copy()
                    improved = True
                    break
            if not improved:
                break

    # Nelder-Mead style random polish
    if remaining() > 20:
        h_final, c5_nm = nelder_mead_polish(best_h, best_n, max_iter=150, time_limit=min(40.0, remaining() * 0.3))
        if c5_nm < best_c5:
            best_c5 = c5_nm
            best_h = h_final.copy()

    # Final SLSQP polish
    if remaining() > 15:
        for beta in [50000, 150000]:
            if remaining() < 6:
                break
            h_polish = slsqop_smmax(best_h, best_n, beta, K=60, maxiter=600)
            c5_polish = compute_c5(h_polish)
            if c5_polish < best_c5:
                best_c5 = c5_polish
                best_h = h_polish.copy()

    # Final equalization
    if remaining() > 10:
        h_fin, c5_fin = equalize_top_shifts(best_h, best_n, K=15, max_iters=300, tolerance=1e-13)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final projection and validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
