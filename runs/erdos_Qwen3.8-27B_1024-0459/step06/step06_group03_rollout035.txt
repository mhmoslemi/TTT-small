def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        np_ = n - offset
        g[offset:] += (1.0 - h[:np_])
        g[:np_] -= h[offset:]
    else:
        a = -offset
        np_ = n - a
        g[:np_] += (1.0 - h[a:])
        g[a:] -= h[:np_]
    return g


def smooth_max_obj(h):
    n = _N
    corr = np.correlate(h, 1.0 - h, mode='full')
    beta = _BETA
    K = _K
    idx = np.argsort(-corr)[:K]

    cshift = c - c.max()
    cl = np.clip(beta * cshift, -80, 80)
    ex = np.exp(cl)
    return (c.max() + np.log(ex.sum()) / beta) * (2.0 / n)

    n = _N
    beta = _BETA

    I = np.eye(n)
    A_trust = np.vstack([np.hstack([I, np.zeros((n, 1))]),
                         np.hstack([-I, np.zeros((n, 1))])])
    b_trust = np.concatenate([h + delta, h + delta])
    A_ub = np.vstack([A_ub, A_trust])
    b_ub = np.concatenate([b_ub, b_trust])
    bounds = [(0, 1)] * n + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return h.copy(), np.max(corr)


def equalize_top_shifts(h, n, K, max_iters):
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if (_BUDGET - (time.time() - _T0)) < 5:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(-corr)[:K]
        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        if c_max - c_min < 1e-10:
            break
        g_avg = np.zeros(n)
        for k in idx:
            g_avg += grad_corr_k(h, k, n)
        g_avg /= K
        gn = np.linalg.norm(g_avg)
        if gn < 1e-14:
            break
        g_dir = -g_avg / gn
        improved = False
        for step in [0.01, 0.02, 0.005, 0.03, 0.015, 0.04, 0.008]:
            ht = project_feasible(h + step * g_dir, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _K, _KLIST
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
    except Exception:
        parent_n = 512
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0

    h = project_feasible(parent_h, parent_n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_n = parent_n
    n = parent_n

    # Phase 1: Subgradient descent for basin finding
    if remaining() > 20:
        h_sg, c5_sg = subgradient_descent(h, n, min(25.0, remaining() * 0.1), rng)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 2: SLSQP smooth max with increasing beta
    for beta, K in [(50, 30), (200, 40), (500, 50), (1000, 60), (3000, 70)]:
        if remaining() < 15:
            break
        h_new, c5_new = slsqop_smmax(h, n, beta, K, maxiter=800)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 3: Active set min-max
    if remaining() > 30:
        for K_active in [8, 15, 25, 40]:
            if remaining() < 10:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            idx_active = np.argsort(-corr)[:K_active]
            h_am, c5_am = slsqop_tminmax(h, n, idx_active, maxiter=400)
            if c5_am < best_c5:
                best_c5 = c5_am
                best_h = h_am.copy()
                h = h_am.copy()

    # Phase 4: Gradient-guided local search (key innovation)
    if remaining() > 60:
        ls_time = min(150.0, remaining() * 0.25)
        h_ls, c5_ls = gradient_guided_search(h, n, ls_time, rng)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
            h = h_ls.copy()

    # Phase 5: SLSQP after local search
    if remaining() > 30:
        for beta, K in [(2000, 60), (5000, 80)]:
            if remaining() < 10:
                break
            h_new, c5_new = slsqop_smmax(h, n, beta, K, maxiter=800)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 6: Equalize + more local search
    if remaining() > 40:
        h_eq, c5_eq = equalize_top_shifts(h, n, K=10, max_iters=200)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

        if remaining() > 40:
            h_ls, c5_ls = gradient_guided_search(h, n, min(60.0, remaining() * 0.3), rng)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()
                h = h_ls.copy()

    # Phase 7: Cutting plane LP
    if remaining() > 40:
        for it in range(100):
            if remaining() < 5:
                break
            h_lp, t_lp = cutting_plane_lp(h, n, K=25, delta=0.03)
            if t_lp < compute_c5(h) - 1e-10:
                best_val = compute_c5(h)
                for alpha in [1.0, 0.7, 0.5, 0.8, 0.3]:
                    h_try = project_feasible(h + alpha * (h_lp - h), n)
                    c_try = compute_c5(h_try)
                    if c_try < best_val:
                        best_val = c_try
                        h = h_try.copy()
                    else:
                        break
                if compute_c5(h) < best_c5:
                    best_c5 = compute_c5(h)
                    best_h = h.copy()
            elif it > 50:
                break

    # Phase 8: Higher resolution (1024)
    n_high = 1024
    if remaining() > 100:
        h_interp = interp_to(best_h, best_n, n_high)
        h_interp = project_feasible(h_interp, n_high)
        c5_interp = compute_c5(h_interp)

        if c5_interp <= best_c5 + 0.003:
            h_h = h_interp.copy()
            # Subgradient at high res
            if remaining() > 50:
                h_h, c5_h = subgradient_descent(h_h, n_high, min(20.0, remaining() * 0.1), rng)
                if c5_h > c5_interp:
                    h_h = h_interp.copy()
                    c5_h = c5_interp

            # SLSQP at high res
            for beta, K in [(100, 40), (500, 50), (2000, 60)]:
                if remaining() < 15:
                    break
                h_h, c5_h = slsqop_smmax(h_h, n_high, beta, K, maxiter=400)

            # Active set at high res
            if remaining() > 25:
                corr_h = np.correlate(h_h, 1.0 - h_h, mode='full')
                for K_a in [10, 20, 30]:
                    if remaining() < 10:
                        break
                    idx_a = np.argsort(-np.correlate(h_h, 1.0 - h_h, mode='full'))[:K_a]
                    h_h, c5_h = slsqop_tminmax(h_h, n_high, idx_a, maxiter=300)

            # Gradient-guided search at high res
            if remaining() > 40:
                h_h, c5_h = gradient_guided_search(h_h, n_high, min(50.0, remaining() * 0.3), rng)

            # Equalize at high res
            if remaining() > 15:
                h_h, c5_h = equalize_top_shifts(h_h, n_high, K=10, max_iters=150)

            if c5_h < best_c5:
                best_c5 = c5_h
                best_h = h_h.copy()
                best_n = n_high

    # Phase 9: Final intensive cycles at best resolution
    n = best_n
    h = best_h.copy()
    for cycle in range(3):
        if remaining() < 20:
            break
        # SLSQP
        if remaining() > 15:
            h_new, c5_new = slsqop_smmax(h, n, 5000, K=70, maxiter=600)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

        # Gradient-guided search
        if remaining() > 20:
            h_ls, c5_ls = gradient_guided_search(h, n, min(30.0, remaining() * 0.3), rng)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()
                h = h_ls.copy()

        # Active set
        if remaining() > 15:
            corr_c = np.correlate(h, 1.0 - h, mode='full')
            for K_a in [15, 25, 40]:
                if remaining() < 8:
                    break
                idx_a = np.argsort(-np.correlate(h, 1.0 - h, mode='full'))[:K_a]
                h_am, c5_am = slsqop_tminmax(h, n, idx_a, maxiter=300)
                if c5_am < best_c5:
                    best_c5 = c5_am
                    best_h = h_am.copy()
                    h = h_am.copy()

    # Phase 10: Final very high beta SLSQP
    if remaining() > 15:
        for beta in [15000, 30000, 50000]:
            if remaining() < 8:
                break
            h_final, c5_final = slsqop_smmax(best_h, best_n, beta, K=80, maxiter=800)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final.copy()

    # Phase 11: Tiny final steps
    if remaining() > 8:
        h_t = best_h.copy()
        n_t = best_n
        for it in range(100):
            if remaining() < 3:
                break
            corr_t = np.correlate(h_t, 1.0 - h_t, mode='full')
            idx_t = np.argsort(-corr_t)[:12]
            c_top = corr_t[idx_t]
            cshift = c_top - c_top.max()
            ex = np.exp(100.0 * cshift)
            w = ex / ex.sum()
            g_t = np.zeros(n_t)
            for i in range(len(idx_t)):
                if w[i] > 1e-10:
                    g_t += w[i] * grad_corr_k(h_t, idx_t[i], n_t)
            gn_t = np.linalg.norm(g_t)
            if gn_t < 1e-14:
                break
            for step in [0.003, 0.001, 0.005, 0.002, 0.008]:
                h_try = project_feasible(h_t - step * g_t / gn_t, n_t)
                c_try = compute_c5(h_try)
                if c_try < best_c5:
                    best_c5 = c_try
                    best_h = h_try.copy()
                    h_t = h_try.copy()
                    break

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
