    _CUR_N = n
    _SM_BETA = beta

        corr = compute_corr(x)
        m = np.max(corr)
        z = np.clip(beta * (corr - m), -50.0, 50.0)
        ez = np.exp(z)
        sm = (m + np.log(np.sum(ez)) / beta)
        return sm + lam * (np.sum(x) - n/2.0)
    
    def grad_lag(x, lam):
        corr = compute_corr(x)
        m = np.max(corr)
        z = np.clip(beta * (corr - m), -50.0, 50.0)
        ez = np.exp(z)
        S = np.sum(ez)
        w = ez / S
        g = np.zeros(n)
        for k in range(2 * n - 1):

            if wk < 1e-14: continue

            return obj_lag(x, lam)

def binary_2swap_sa(h, n, time_limit=120, rng=None):

    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    curr_c5 = best_c5
    start = time.time()

    total_iters = 30000
    min_T_ratio = T_end / T_start
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    no_improve = 0
    for it in range(total_iters):
        if time.time() - start > time_limit or remaining_time() < 5:
            break
        T = T_start * (min_T_ratio ** (it / max(1, total_iters)))
        i_idx = rng.randint(len(ones))

        i = ones[i_idx]
        j = zeros[j_idx]
        if i == j:

        h_bin[i] = 0.0
        h_bin[j] = 1.0
        new_c5 = compute_c5(h_bin)

        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / max(T, 1e-10))):
            ones[i_idx] = j
            zeros[j_idx] = i
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_bin.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            no_improve += 1
        if no_improve > 8000:
            break
    return best_h, best_c5


def interp_to(src, src_n, dst_n):

    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def theta_morse(n):

            best_h = h_new.copy()
            h = h_new.copy()

    # L-BFGS-B dual
    if remaining_time() > 30 and time.time() - start < phase_time * 0.7:
        h_d, c5_d = lbfgsb_dual(h, n, beta=500000, max_outer=30,
                                time_limit=min(60, remaining_time() * 0.2))
        if c5_d < best_c5 - 1e-15:
            best_c5 = c5_d
            best_h = h_d.copy()
            h = h_d.copy()

    # Top-k equalization
    if remaining_time() > 15 and time.time() - start < phase_time * 0.85:
        for tk in [3, 6, 12]:
            h_eq, c5_eq = top_k_equalize(h, n, top_k=tk, max_iters=400,
                                         time_limit=min(20, remaining_time() * 0.1))
            if c5_eq < best_c5 - 1e-15:
                best_c5 = c5_eq
                best_h = h_eq.copy()
                h = h_eq.copy()

    # Subgradient
    if remaining_time() > 15 and time.time() - start < phase_time * 0.95:
        h_sg, c5_sg = subgradient_zigzag(h, n, max_iters=3000,
                                         time_limit=min(30, remaining_time() * 0.15))
        if c5_sg < best_c5 - 1e-15:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Final SLSQP polish
    if remaining_time() > 10:
        h_new = slsqp_opt(best_h, n, 50000000, maxiter=2000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    # Load parent
    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = np.zeros(640)
        best_h[:320] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Optimize at parent resolution
    h1, c5_1 = optimize_at_n(best_h, best_n, phase_time=100, rng=rng)
    if c5_1 < best_c5:
        best_h, best_c5, best_n = h1, c5_1, len(h1)

    # Phase 2: Try higher resolutions
    higher_ns = [960, 1024, 1280, 1920]
    for n_t in higher_ns:
        if remaining_time() < 80:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        h_opt, c5_opt = optimize_at_n(h_feas, n_t, phase_time=90, rng=rng)
        if c5_opt < best_c5 - 1e-15:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t

    # Phase 3: Binary SA at current best resolution
    if remaining_time() > 80:
        h_bin = (best_h > 0.5).astype(float)
        h_b, c5_b = binary_2swap_sa(h_bin, best_n, time_limit=min(120, int(remaining_time() * 0.3)), rng=rng)
        # Refine continuously from binary
        h_b_c = h_b.copy()
        for beta in [100000, 1000000, 5000000]:
            if remaining_time() < 15:
                break
            h_b_c = slsqp_opt(h_b_c, best_n, beta, maxiter=1500)
        c5_b_c = compute_c5(h_b_c)
        if c5_b_c < best_c5 - 1e-15:
            best_c5 = c5_b_c
            best_h = h_b_c.copy()
        if c5_b < best_c5 - 1e-15:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 4: Multi-start
    alt_starts = []
    alt_starts.append(theta_morse(best_n))
    h_rand = np.zeros(best_n)
    h_rand[rng.choice(best_n, best_n // 2, replace=False)] = 1.0
    alt_starts.append(h_rand)
    alt_starts.append(project_feasible(np.roll(best_h, 1), best_n))
    alt_starts.append(project_feasible(1.0 - best_h + best_h[::-1] * 0.0, best_n))
    # Symmetric starts
    h_sym = np.zeros(best_n)
    h_sym[:best_n // 4] = 1.0
    h_sym[best_n // 2:best_n * 3 // 4] = 1.0
    alt_starts.append(project_feasible(h_sym, best_n))

    for h_alt in alt_starts:
        if remaining_time() < 40:
            break
        h_opt = h_alt
        for beta in [10000, 500000]:
            if remaining_time() < 15:
                break
            h_opt = slsqp_opt(h_opt, best_n, beta, maxiter=1000)
        c5_opt = compute_c5(h_opt)
        h_opt, c5_opt = top_k_equalize(h_opt, best_n, top_k=5, max_iters=300, time_limit=20)
        h_opt, c5_opt = subgradient_zigzag(h_opt, best_n, max_iters=2000, time_limit=20)
        if c5_opt < best_c5 - 1e-15:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 5: Final intensive refinement
    n = best_n
    h = best_h.copy()
    if remaining_time() > 20:
        h_new = slsqp_opt(h, n, 100000000, maxiter=3000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    if remaining_time() > 10:
        h_eq, c5_eq = top_k_equalize(h, n, top_k=8, max_iters=600, time_limit=20)
        if c5_eq < best_c5 - 1e-15:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    if remaining_time() > 15:
        h_sg, c5_sg = subgradient_zigzag(best_h, n, max_iters=4000,
                                         time_limit=min(50, int(remaining_time() * 0.4)))
        if c5_sg < best_c5 - 1e-15:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Final SLSQP polish
    if remaining_time() > 10:
        h_final = slsqp_opt(best_h, n, 500000000, maxiter=2000)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-15:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Validate with evaluator
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, n)
