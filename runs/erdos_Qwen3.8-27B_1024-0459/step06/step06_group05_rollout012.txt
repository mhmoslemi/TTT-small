def compute_d(h, j, n):
    """Compute d(k) for all k = -(n-1), ..., 0, ..., n-1 for coordinate j."""
    ks = np.arange(-(n-1), n)  # length 2n-1
    d = np.zeros(2*n-1)
    
    # k >= 0: indices n-1, n, ..., 2n-2 in d
    k_pos = np.arange(0, n)  # 0 to n-1
    idx_pos = k_pos + (n-1)  # indices in d
    part1 = np.zeros(n)
    part2 = np.zeros(n)
    valid1 = k_pos <= j  # j >= k
    part1[valid1] = 1 - h[j - k_pos[valid1]]
    valid2 = (j + k_pos) < n
    part2[valid2] = -h[j + k_pos[valid2]]
    d[idx_pos] = part1 + part2
    
    # k < 0: indices 0, 1, ..., n-2 in d
    k_neg = np.arange(-(n-1), 0)  # -(n-1) to -1
    idx_neg = k_neg + (n-1)  # indices in d (0 to n-2)
    part1 = np.zeros(len(k_neg))
    part2 = np.zeros(len(k_neg))
    valid1 = (j < n + k_neg)  # j < n + k (k is negative, so n+k < n)
    # j - k_neg[j] = j + |k|, need this < n for h[j-k] to be valid
    # j - k = j + |k|
    jp = j - k_neg  # = j + |k|, ranges from j+1 to j+n-1
    valid1 = jp < n
    part1[valid1] = 1 - h[jp[valid1]]
    jn = j + k_neg  # = j - |k|, ranges from j-(n-1) to j-1
    valid2 = jn >= 0
    part2[valid2] = -h[jn[valid2]]
    d[idx_neg] = part1 + part2
    
    return d
