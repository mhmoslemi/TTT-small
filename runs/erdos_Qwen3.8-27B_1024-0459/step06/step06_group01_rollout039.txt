    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)

        else:
            h[:] = h[:] + diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient(h, k):
    """Gradient of correlation at index k with respect to h."""
    n = len(h)
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        np_ = n - off
        g[off:] += (1.0 - h[:np_])
        g[:np_] -= h[off:]
    else:
        a = -off
        np_ = n - a
        g[:np_] += (1.0 - h[a:])
        g[a:] -= h[:np_]
    return g

def compute_corr(h):

def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.count_nonzero(mask)

        h = np.clip(h, 0.0, 1.0)

def corr_grad_single(h, k):
    n = len(h)

        g[:np_] -= h[off:]
    else:
        a = -off
        np_ = n - a
        g[:np_] += (1.0 - h[a:])
        g[a:] -= h[:np_]
    return g

def binary_local_search(h, n, time_limit=60):
    """Local search on binary projection with continuous refinement."""
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    n_half = n // 2
    if n_ones != n_half:
        if n_ones > n_half:
            ones = np.where(h_bin == 1.0)[0]
            for i in range(n_ones - n_half):
                h_bin[ones[i]] = 0.0
        else:
            zeros = np.where(h_bin == 0.0)[0]
            for i in range(n_half - n_ones):
                h_bin[zeros[i]] = 1.0

    start = time.time()
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    current_c5 = best_c5
    no_improve = 0

    while time.time() - start < time_limit and remaining_time() > 10:
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        new_c5 = compute_c5(h_bin)
        if new_c5 < current_c5:
            current_c5 = new_c5
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = h_bin.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            no_improve += 1
        if no_improve > 5000:
            break
    return best_h, best_c5

        h_final = slsqp_opt(best_h, best_n, 1000000, maxiter=400)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-14:
            best_c5 = c5_final
            best_h = h_final.copy()

    if remaining_time() > 15:
        h_eqf, c5_eqf = active_set_v2(best_h, best_n, max_iters=1500)
        if c5_eqf < best_c5 - 1e-14:
            best_c5 = c5_eqf
            best_h = h_eqf.copy()

    # Final feasibility and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
