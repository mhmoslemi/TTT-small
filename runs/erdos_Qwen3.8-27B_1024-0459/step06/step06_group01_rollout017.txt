def compute_c5(h):

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:

        if diff > 0:
            idx = np.argsort(h)
            for i in idx:
                room = 1.0 - h[i]
                if room > 1e-15:
                    add = min(diff, room)
                    h[i] += add
                    diff -= add
                    if abs(diff) < 1e-14:

        else:
            idx = np.argsort(-h)
            for i in idx:
                room = h[i]
                if room > 1e-15:
                    sub = min(-diff, room)
                    h[i] -= sub
                    diff += sub
                    if abs(diff) < 1e-14:
                        break
    h = np.clip(h, 0.0, 1.0)

    _GN = n
    _PM_M = M
    h0 = project_feasible(h, n)

def nesterov_topk(h, n, top_k=7, max_iters=3000):
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    v = np.zeros(n)

                no_improve = 0

            h[j] = 0.0
            no_improve += 1

        if no_improve > max_no_improve:

            if remaining_time() < 30:
                break
            h_cur = slsqp_pm(h_cur, n_t, M, maxiter=400)
        c5_cur = compute_c5(h_cur)
        # Nesterov
        if remaining_time() > 20:
            h_cur2, c5_cur2 = nesterov_topk(h_cur, n_t, top_k=7, max_iters=1000)
            if c5_cur2 < c5_cur:
                h_cur = h_cur2
                c5_cur = c5_cur2
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            best_n = n_t
            h = h_cur.copy()
            _GN = n_t

    # Phase 4: Binary SA at current best resolution
    if remaining_time() > 100:
        h_bin = (best_h > 0.5).astype(float)
        sa_time = min(200, max(30, int(remaining_time() * 0.25)))
        h_b, c5_b = binary_sa(h_bin, best_n, sa_time, rng)
        if c5_b < best_c5 - 1e-14:
            best_c5 = c5_b
            best_h = h_b.copy()
            h = h_b.copy()
        # Continuous refinement from SA result
        if remaining_time() > 60:
            h_ref, c5_ref = continuous_refine(h_b, best_n, min(120, int(remaining_time() * 0.4)))
            if c5_ref < best_c5 - 1e-14:
                best_c5 = c5_ref
                best_h = h_ref.copy()
                h = h_ref.copy()

    # Phase 5: Random restart binary SA
    if remaining_time() > 120:
        for attempt in range(3):
            if remaining_time() < 40:
                break
            h_rand = random_binary(best_n, rng)
            sa_t = min(60, max(15, int(remaining_time() * 0.15)))
            h_ra, c5_ra = binary_sa(h_rand, best_n, sa_t, rng)
            if c5_ra < best_c5 - 1e-14:
                best_c5 = c5_ra
                best_h = h_ra.copy()
                h = h_ra.copy()
            # Quick continuous refine from this binary
            if remaining_time() > 30:
                h_rc, c5_rc = continuous_refine(h_ra, best_n, min(40, int(remaining_time() * 0.2)))
                if c5_rc < best_c5 - 1e-14:
                    best_c5 = c5_rc
                    best_h = h_rc.copy()
                    h = h_rc.copy()

    # Phase 6: Thue-Morse based start
    if remaining_time() > 60:
        h_tm = theta_morse(best_n)
        h_tm_opt = h_tm
        for M in [10, 100, 500]:
            if remaining_time() < 20:
                break
            h_tm_opt = slsqp_pm(h_tm_opt, best_n, M, maxiter=300)
        c5_tm = compute_c5(h_tm_opt)
        h_tm_opt, c5_tm2 = nesterov_topk(h_tm_opt, best_n, top_k=5, max_iters=800)
        c5_tm = min(c5_tm, c5_tm2)
        if c5_tm2 < c5_tm:
            h_tm_opt = h_tm_opt  # already the nesterov result
        if c5_tm < best_c5 - 1e-14:
            best_c5 = c5_tm
            best_h = h_tm_opt.copy()
            h = h_tm_opt.copy()

    # Phase 7: Final high-M SLSQP polish at best resolution
    if remaining_time() > 30:
        for M in [2000, 10000, 50000]:
            if remaining_time() < 15:
                break
            h_pol = slsqp_pm(h, best_n, M, maxiter=500)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5 - 1e-14:
                best_c5 = c5_pol
                best_h = h_pol.copy()
                h = h_pol.copy()

    # Phase 8: Final Nesterov polish
    if remaining_time() > 20:
        h_fin, c5_fin = nesterov_topk(h, best_n, top_k=9, max_iters=2000)
        if c5_fin < best_c5 - 1e-14:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Phase 9: Try interpolation to slightly different resolutions
    if remaining_time() > 40:
        for n_try in [best_n * 2, best_n // 2]:
            if n_try < 64 or n_try > 4096 or n_try % 2 != 0:
                continue
            if remaining_time() < 20:
                break
            h_t = interp_to(best_h, best_n, n_try)
            h_t = project_feasible(h_t, n_try)
            for M in [50, 500, 5000]:
                if remaining_time() < 10:
                    break
                h_t = slsqp_pm(h_t, n_try, M, maxiter=300)
            c5_t = compute_c5(h_t)
            if c5_t < best_c5 - 1e-14:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = n_try

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
