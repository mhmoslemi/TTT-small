import time

_BUDGET = 1300.0
_CUR_N = 0
_SM_BETA = 1000.0

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)

def _sm_obj(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -60.0, 60.0)
    ez = np.exp(z)
    return (m + np.log(np.sum(ez)) / beta) * dx

def _sm_grad(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -60.0, 60.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    return g

def _eq_fun(x):
    return np.sum(x) - _CUR_N / 2.0

def _eq_jac(x):
    return np.ones(_CUR_N)

def slsqp_opt(h, n, beta, maxiter=1000):

    _CUR_N = n
    _SM_BETA = beta
    h0 = project_feasible(h, n)

    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]

        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        c_max = corr[idx_top[0]]
        
        # Check convergence
        c_min_top = corr[idx_top[-1]]
        if c_max - c_min_top < 1e-10:
            break
        
        # Compute direction: reduce max, increase min of top-k
        g_dir = np.zeros(n)
        n_top = len(idx_top)
        for pos, k in enumerate(idx_top):
            if pos == 0:
                g_dir += grad_single_shift(h, k, n)
            else:
                g_dir -= grad_single_shift(h, k, n) / (n_top - 1)
        
        # Project to satisfy constraint
        g_dir -= np.mean(g_dir)
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        d = g_dir / gn
        
        # Adaptive step size
        improved = False
        step0 = 0.01 * (1.0 + 0.1 * (it // 100))
        for step in [step0, step0*0.5, step0*2, step0*0.1, step0*5, 0.001, 0.01, 0.05, 0.1]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                best_h = h.copy()
                improved = True
                break
        
        if not improved:
            # Try negative direction
            for step in [0.001, 0.005, 0.01, 0.05]:
                ht = project_feasible(h - step * d, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-14:
                    h = ht
                    best_c5 = cn
                    best_h = h.copy()
                    improved = True
                    break
            if not improved:
                break
    
    return best_h, best_c5

def subgradient_zigzag(h, n, max_iters=10000, time_limit=90):
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    k_curr = None
    k_prev = None
    stale_count = 0
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if remaining_time() < 5:
            break

        corr = compute_corr(h)
        k_curr = int(np.argmax(corr))

        if k_prev is not None and k_curr != k_prev:
            g = 0.5 * (grad_single_shift(h, k_curr, n) + grad_single_shift(h, k_prev, n))
            k_prev = k_curr
        else:
            g = grad_single_shift(h, k_curr, n)
            if k_prev is None:
                k_prev = k_curr

        g = g - np.mean(g)  # Project to maintain constraint
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-14:
            g = g / g_norm

        improved = False
        base_step = max(0.0005, 0.01 * (1.0 - 0.5 * it / max_iters))
        for step in [base_step * 0.1, base_step * 0.3, base_step, base_step * 3,
                     base_step * 10, 0.0005, 0.001, 0.005, 0.01, 0.05]:
            ht = project_feasible(h - step * g, n)
            c5_new = compute_c5(ht)
            if c5_new < best_c5 - 1e-15:
                h = ht
                best_c5 = c5_new
                best_h = h.copy()
                improved = True
                stale_count = 0
                break
        
        if not improved:
            stale_count += 1
            if stale_count > 300:
                break
            if stale_count % 100 == 0:
                rng = np.random.RandomState(it)
                pert = rng.randn(n) * 0.002
                ht = project_feasible(h + pert, n)
                c5_new = compute_c5(ht)
                if c5_new < best_c5:
                    h = ht
                    best_c5 = c5_new
                    best_h = h.copy()
                    stale_count = 0

    return best_h, best_c5

def binary_sa_block(h, n, time_limit=120, rng=None):
    """Binary SA with single-bit and block moves."""
    if rng is None:
        rng = np.random.RandomState(42)
    h_bin = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        sort_idx = np.argsort(-h_bin)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0

    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    curr_c5 = best_c5
    start = time.time()

    T_start = 2.0
    T_end = 0.0001
    total_iters = max(10000, int(time_limit * 50))
    no_improve = 0
    
    for it in range(total_iters):
        if time.time() - start > time_limit:
            break
        if remaining_time() < 5:
            break

        T = T_start * (T_end / T_start) ** (it / max(1, total_iters))
        
        # Choose move type
        if rng.random() < 0.7:
            # Single swap
            i = rng.randint(n)
            j = rng.randint(n)
            if h_bin[i] == h_bin[j]:
                continue
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        else:
            # Block swap: swap a contiguous block
            block_len = rng.choice([2, 3, 4, 5, 8])
            i = rng.randint(n - block_len + 1)
            j = rng.randint(n - block_len + 1)
            if abs(i - j) < block_len:
                continue
            # Check balance
            diff = h_bin[i:i+block_len].sum() - h_bin[j:j+block_len].sum()
            if diff != 0:
                continue
            h_bin[i:i+block_len], h_bin[j:j+block_len] = h_bin[j:j+block_len].copy(), h_bin[i:i+block_len].copy()
        
        new_c5 = compute_c5(h_bin)
        delta = new_c5 - curr_c5

        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / max(T, 1e-10))):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_bin.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Undo
            if rng.random() < 0.7:  # was single swap
                h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
            else:  # was block swap
                h_bin[i:i+block_len], h_bin[j:j+block_len] = h_bin[j:j+block_len].copy(), h_bin[i:i+block_len].copy()
            no_improve += 1

        if no_improve > 5000:
            break

    return best_h, best_c5

def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)

def muller_sequence(n, p):
    """Muller sequence: positions where (p*i)^2 mod q < q/2"""
    h = np.zeros(n)
    q = 2 * n + 1  # or some large odd number
    for i in range(n):
        val = (p * i) % q
        h[i] = 1.0 if (val * val) % q < q // 2 else 0.0
    return project_feasible(h, n)

def theta_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)

def structured_start(n, freq):
    """Start based on a specific frequency pattern."""
    x = np.linspace(0, 2, n, endpoint=False) + 1.0/n
    h = 0.5 + 0.4 * np.cos(2*np.pi*freq*x) + 0.1 * np.cos(2*np.pi*freq*3*x)
    h = np.clip(h, 0, 1)
    return project_feasible(h, n)

def coordinate_descent_fine(h, n, max_sweeps=50, time_limit=60):
    """Fine coordinate descent with quadratic approximation."""
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    start = time.time()
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if remaining_time() < 5:
            break
        improved_any = False
        for i in range(n):
            old = h[i]
            # Try small perturbations
            for delta in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02, 0.001, -0.001]:
                new_val = old + delta
                if new_val < 0 or new_val > 1:
                    continue
                h[i] = new_val
                h = project_feasible(h, n)
                c5_new = compute_c5(h)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    old = new_val
                    improved_any = True
                    break
            h[i] = old
            h = project_feasible(h, n)
        if not improved_any:
            break
    
    return h, best_c5

def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0

        if cnt > 0:
            h[mask] += diff / cnt
        else:

        h = np.clip(h, 0.0, 1.0)

    return np.ones(_CUR_N)


def slsqp_opt(h, n, beta, maxiter=1000):
    global _CUR_N, _SM_BETA
    _CUR_N = n
    _SM_BETA = beta
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_sm_obj, h0, jac=_sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        if res.fun < _sm_obj(h0) - 1e-15:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def grad_single_shift(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        np_ = n - off
        g[off:] += (1.0 - h[:np_])
        g[:np_] -= h[off:]
    else:
        a = -off
        np_ = n - a
        g[:np_] += (1.0 - h[a:])
        g[a:] -= h[:np_]
    return g


def projected_gradient_descent(h, n, beta, max_iters=20000, time_limit=90):
    global _CUR_N, _SM_BETA
    _CUR_N = n
    _SM_BETA = beta
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if remaining_time() < 5:
            break

        g = _sm_grad(h)
        g = g - np.mean(g)

        improved = False
        for step in [0.05, 0.02, 0.01, 0.005, 0.001, 0.0005, 0.1, 0.2, 0.0001]:
            ht = project_feasible(h - step * g, n)
            c5_new = compute_c5(ht)
            if c5_new < best_c5 - 1e-15:
                h = ht
                best_c5 = c5_new
                best_h = h.copy()
                improved = True
                break

        if not improved:
            for step in [0.00005, 0.00001, 0.000005]:
                ht = project_feasible(h - step * g, n)
                c5_new = compute_c5(ht)
                if c5_new < best_c5 - 1e-16:
                    h = ht
                    best_c5 = c5_new
                    best_h = h.copy()
                    improved = True
                    break
            if not improved:
                break

    return best_h, best_c5


def top_k_equalize_fast(h, n, top_k=5, max_iters=5000, time_limit=60):
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if remaining_time() < 3:
            break

        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        c_max = corr[idx_top[0]]
        c_min_top = corr[idx_top[-1]]
        if c_max - c_min_top < 1e-12:
            break

        g_dir = np.zeros(n)
        n_top = len(idx_top)
        for pos, k in enumerate(idx_top):
            if pos == 0:
                g_dir += grad_single_shift(h, k, n)
            else:
                g_dir -= grad_single_shift(h, k, n) / (n_top - 1)

        g_dir -= np.mean(g_dir)
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        d = g_dir / gn

        improved = False
        step0 = max(0.001, 0.02 * (1.0 - 0.3 * it / max_iters))
        for step in [step0, step0 * 0.5, step0 * 2.0, step0 * 0.1, step0 * 5.0,
                     0.001, 0.005, 0.01, 0.05, 0.1, 0.2]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                best_h = h.copy()
                improved = True
                break

        if not improved:
            for step in [0.001, 0.005, 0.01, 0.05]:
                ht = project_feasible(h - step * d, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-14:
                    h = ht
                    best_c5 = cn
                    best_h = h.copy()
                    improved = True
                    break
            if not improved:
                break

    return best_h, best_c5


def subgradient_zigzag(h, n, max_iters=10000, time_limit=90):
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    k_curr = None
    k_prev = None
    stale_count = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if remaining_time() < 5:
            break

        corr = compute_corr(h)
        k_curr = int(np.argmax(corr))

        if k_prev is not None and k_curr != k_prev:
            g = 0.5 * (grad_single_shift(h, k_curr, n) + grad_single_shift(h, k_prev, n))
            k_prev = k_curr
        else:
            g = grad_single_shift(h, k_curr, n)
            if k_prev is None:
                k_prev = k_curr

        g = g - np.mean(g)
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-14:
            g = g / g_norm

        improved = False
        base_step = max(0.0005, 0.015 * (1.0 - 0.5 * it / max_iters))
        for step in [base_step * 0.1, base_step * 0.3, base_step, base_step * 3,
                     base_step * 10, 0.0005, 0.001, 0.005, 0.01, 0.03, 0.1]:
            ht = project_feasible(h - step * g, n)
            c5_new = compute_c5(ht)
            if c5_new < best_c5 - 1e-15:
                h = ht
                best_c5 = c5_new
                best_h = h.copy()
                improved = True
                stale_count = 0
                break

        if not improved:
            stale_count += 1
            if stale_count > 400:
                break
            if stale_count % 100 == 0:
                rng2 = np.random.RandomState(it * 7 + 3)
                pert = rng2.randn(n) * 0.003
                ht = project_feasible(h + pert, n)
                c5_new = compute_c5(ht)
                if c5_new < best_c5:
                    h = ht
                    best_c5 = c5_new
                    best_h = h.copy()
                    stale_count = 0

    return best_h, best_c5


def binary_sa_advanced(h, n, time_limit=120, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h_bin = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        sort_idx = np.argsort(-h_bin)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0

    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    curr_c5 = best_c5
    start = time.time()
    total_iters = max(20000, int(time_limit * 80))
    no_improve = 0

    for it in range(total_iters):
        if time.time() - start > time_limit:
            break
        if remaining_time() < 5:
            break

        T = 2.0 * (0.0001 / 2.0) ** (it / max(1, total_iters))

        move_type = rng.choice(['swap', 'block'], p=[0.65, 0.35])

        if move_type == 'swap':
            i = rng.randint(n)
            j = rng.randint(n)
            if h_bin[i] == h_bin[j]:
                continue
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
            undo = ('swap', i, j)
        else:
            blen = rng.choice([2, 3, 4, 5, 6, 8, 10])
            i = rng.randint(n - blen + 1)
            j = rng.randint(n - blen + 1)
            if abs(i - j) < blen:
                continue
            if np.sum(h_bin[i:i+blen]) != np.sum(h_bin[j:j+blen]):
                continue
            tmp = h_bin[i:i+blen].copy()
            h_bin[i:i+blen] = h_bin[j:j+blen]
            h_bin[j:j+blen] = tmp
            undo = ('block', i, j, blen)

        new_c5 = compute_c5(h_bin)
        delta = new_c5 - curr_c5

        if delta < 0 or (T > 1e-10 and rng.random() < np.exp(-delta / max(T, 1e-10))):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_bin.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            if undo[0] == 'swap':
                h_bin[undo[1]], h_bin[undo[2]] = h_bin[undo[2]], h_bin[undo[1]]
            else:
                i2, j2, blen2 = undo[1], undo[2], undo[3]
                tmp2 = h_bin[i2:i2+blen2].copy()
                h_bin[i2:i2+blen2] = h_bin[j2:j2+blen2]
                h_bin[j2:j2+blen2] = tmp2
            no_improve += 1

        if no_improve > 8000:
            break

    return best_h, best_c5


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def theta_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)


def muller_sequence(n, p=7):
    h = np.zeros(n)
    q = 4 * n + 3
    for i in range(n):
        val = (p * i) % q
        h[i] = 1.0 if (val * val) % q < q // 2 else 0.0
    return project_feasible(h, n)


def structured_start(n, freq):
    x = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
    h = 0.5 + 0.35 * np.cos(2 * np.pi * freq * x / 2) + 0.15 * np.cos(2 * np.pi * freq * 2 * x / 2)
    h = np.clip(h, 0, 1)
    return project_feasible(h, n)


def coordinate_descent_fine(h, n, max_sweeps=40, time_limit=60):
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    start = time.time()

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if remaining_time() < 5:
            break
        improved_any = False
        for i in range(n):
            old = h[i]
            for delta in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02, 0.001, -0.001, 0.002, -0.002]:
                new_val = old + delta
                if new_val < 0 or new_val > 1:
                    continue
                h[i] = new_val
                h = project_feasible(h, n)
                c5_new = compute_c5(h)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    old = new_val
                    improved_any = True
                    break
            h[i] = old
            h = project_feasible(h, n)
        if not improved_any:
            break

    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _CUR_N, _SM_BETA
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    # Load parent
    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = theta_morse(640)
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Progressive SLSQP (moderate beta for equalization)
    h = best_h.copy()
    for beta in [500, 5000, 50000, 500000]:
        if remaining_time() < 50:
            break
        h_new = slsqp_opt(h, best_n, beta, maxiter=3000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Top-k equalization (the key refinement)
    if remaining_time() > 40:
        for tk in [5, 8, 12, 16, 20, 30]:
            if remaining_time() < 15:
                break
            h_eq, c5_eq = top_k_equalize_fast(h, best_n, top_k=tk, max_iters=5000, time_limit=30)
            if c5_eq < best_c5 - 1e-15:
                best_c5 = c5_eq
                best_h = h_eq.copy()
                h = h_eq.copy()

    # Phase 3: Subgradient zigzag
    if remaining_time() > 40:
        h_sg, c5_sg = subgradient_zigzag(h, best_n, max_iters=15000, time_limit=60)
        if c5_sg < best_c5 - 1e-15:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 4: High-beta SLSQP (sharp focus on max)
    for beta in [5000000, 50000000, 500000000]:
        if remaining_time() < 30:
            break
        h_new = slsqp_opt(h, best_n, beta, maxiter=4000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 5: PGR with very high beta (fine-tuning)
    if remaining_time() > 40:
        h_pgd, c5_pgd = projected_gradient_descent(h, best_n, beta=100000000, max_iters=30000, time_limit=60)
        if c5_pgd < best_c5 - 1e-15:
            best_c5 = c5_pgd
            best_h = h_pgd.copy()
            h = h_pgd.copy()

    # Phase 6: More top-k equalization after all the above
    if remaining_time() > 30:
        for tk in [10, 15, 25, 40]:
            if remaining_time() < 10:
                break
            h_eq, c5_eq = top_k_equalize_fast(h, best_n, top_k=tk, max_iters=5000, time_limit=25)
            if c5_eq < best_c5 - 1e-15:
                best_c5 = c5_eq
                best_h = h_eq.copy()
                h = h_eq.copy()

    # Phase 7: Binary SA with block moves
    if remaining_time() > 100:
        sa_time = min(120, int(remaining_time() * 0.25))
        h_bin = (best_h > 0.5).astype(float)
        h_b, c5_b = binary_sa_advanced(h_bin, best_n, time_limit=sa_time, rng=rng)
        # Continuous refinement from best binary
        h_b_c = h_b.copy()
        for beta in [100000, 1000000, 10000000]:
            if remaining_time() < 15:
                break
            h_b_c = slsqp_opt(h_b_c, best_n, beta, maxiter=2000)
        c5_b_c = compute_c5(h_b_c)
        h_b_c, c5_b_c = top_k_equalize_fast(h_b_c, best_n, top_k=10, max_iters=3000, time_limit=30)
        h_b_c, c5_b_c = subgradient_zigzag(h_b_c, best_n, max_iters=10000, time_limit=30)

        if c5_b_c < best_c5 - 1e-15:
            best_c5 = c5_b_c
            best_h = h_b_c.copy()
            h = h_b_c.copy()
        if c5_b < best_c5 - 1e-15:
            best_c5 = c5_b
            best_h = h_b.copy().astype(float)

    # Phase 8: Multi-resolution (try one alternative resolution)
    if remaining_time() > 80:
        n_t = int(best_n * 1.5) if best_n <= 700 else int(best_n * 0.75)
        n_t = max(320, min(n_t, 1280))
        if n_t != best_n:
            h_interp = interp_to(best_h, best_n, n_t)
            h_feas = project_feasible(h_interp, n_t)
            h_opt = h_feas
            for beta in [10000, 100000, 1000000]:
                if remaining_time() < 20:
                    break
                h_opt = slsqp_opt(h_opt, n_t, beta, maxiter=2000)
            c5_opt = compute_c5(h_opt)
            h_opt, c5_opt = top_k_equalize_fast(h_opt, n_t, top_k=8, max_iters=3000, time_limit=30)
            h_opt, c5_opt = subgradient_zigzag(h_opt, n_t, max_iters=8000, time_limit=25)
            h_opt, c5_opt = projected_gradient_descent(h_opt, n_t, beta=100000000, max_iters=15000, time_limit=25)

            if c5_opt < best_c5 - 1e-15:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_t
                h = h_opt.copy()

    # Phase 9: Diverse starting points
    if remaining_time() > 80:
        alt_starts = [theta_morse(best_n), muller_sequence(best_n, 7), structured_start(best_n, 3)]
        h_rand = np.zeros(best_n)
        h_rand[rng.choice(best_n, best_n // 2, replace=False)] = 1.0
        alt_starts.append(project_feasible(h_rand, best_n))

        for h_alt in alt_starts:
            if remaining_time() < 40:
                break
            h_opt = h_alt
            for beta in [5000, 50000, 500000]:
                if remaining_time() < 15:
                    break
                h_opt = slsqp_opt(h_opt, best_n, beta, maxiter=1500)
            c5_opt = compute_c5(h_opt)
            h_opt, c5_opt = top_k_equalize_fast(h_opt, best_n, top_k=10, max_iters=2000, time_limit=25)
            h_opt, c5_opt = subgradient_zigzag(h_opt, best_n, max_iters=6000, time_limit=20)

            if c5_opt < best_c5 - 1e-15:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                h = h_opt.copy()

    # Phase 10: Final intensive refinement on best solution
    n = best_n
    h = best_h.copy()

    for beta in [100000000, 1000000000]:
        if remaining_time() < 15:
            break
        h_new = slsqp_opt(h, n, beta, maxiter=5000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    if remaining_time() > 20:
        h_pgd, c5_pgd = projected_gradient_descent(h, n, beta=1000000000, max_iters=40000, time_limit=45)
        if c5_pgd < best_c5 - 1e-15:
            best_c5 = c5_pgd
            best_h = h_pgd.copy()
            h = h_pgd.copy()

    if remaining_time() > 15:
        h_eq, c5_eq = top_k_equalize_fast(h, n, top_k=20, max_iters=5000, time_limit=30)
        if c5_eq < best_c5 - 1e-15:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    if remaining_time() > 20:
        h_sg, c5_sg = subgradient_zigzag(best_h, n, max_iters=10000, time_limit=min(40, int(remaining_time() * 0.6)))
        if c5_sg < best_c5 - 1e-15:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Coordinate descent for very fine adjustments
    if remaining_time() > 20:
        h_cd, c5_cd = coordinate_descent_fine(best_h, n, max_sweeps=30, time_limit=min(40, int(remaining_time() * 0.6)))
        if c5_cd < best_c5 - 1e-15:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Final SLSQP polish
    if remaining_time() > 10:
        h_final = slsqp_opt(best_h, n, 500000000, maxiter=5000)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-15:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, n)
