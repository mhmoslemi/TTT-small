Hmm, let me think more carefully about the smooth-max. The standard approach is:

soft_max(c; β) = (1/β) * log(Σ exp(β * c_i))

As β → ∞, this approaches max(c_i). The gradient is:
∂ soft_max / ∂ c_j = exp(β * c_j) / Σ exp(β * c_i) = softmax(β * c)_j

So the gradient of soft_max * (2/n) with respect to h is:
(2/n) * Σ_j softmax_j * (∂corr_j/∂h)

This is what I had. Good.

But there's a numerical issue: if β is very large, exp(β * c_i) can overflow. The fix is to subtract the max:
soft_max(c; β) = (max(c) + 1/β) * log(Σ exp(β * (c_i - max(c))))

This is numerically stable.

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _G_N, _G_BETA, _G_K
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def rem():
        return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=np.float64)
        n0 = len(h0)
    except Exception:
        n0 = 512
        h0 = np.zeros(n0)
        h0[:n0 // 2] = 1.0

    c5_0 = compute_c5(h0)
    best_h = h0.copy()
    best_c5 = c5_0
    best_n = n0

    # Interpolate to n=1024
    n_target = 1024
    h_work = h0.copy()
    n_work = n0

    if rem() > 150 and n_target > n0:
        h_interp = interp_to(h0, n0, n_target)
        h_interp = exact_project(h_interp, n_target / 2.0)
        c5_interp = compute_c5(h_interp)
        if c5_interp <= c5_0 + 0.005:
            best_h = h_interp.copy()
            best_c5 = c5_interp
            best_n = n_target
            h_work = h_interp.copy()
            n_work = n_target

    n = n_work
    h = h_work.copy()

    # Phase 1: Quick projected subgradient
    if rem() > 30:
        h = pg_descent(h, n, min(25.0, rem() * 0.04), rng)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: SLSQP smooth-max progressive beta
    ssluges = [(10, 20, 300), (50, 30, 400), (200, 40, 400),
               (1000, 50, 500), (5000, 60, 500), (20000, 80, 600)]
    for beta, K, maxit in ssluges:
        if rem() < 15:
            break
        h_new = slsqop_optimize(h, n, beta, K, maxit)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 3: LP cutting plane refinement
    if rem() > 40:
        lp_time = min(rem() * 0.25, 200.0)
        h = lp_refine(h, n, lp_time, rng)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 4: More SLSQP after LP
    for beta, K, maxit in [(5000, 80, 500), (20000, 100, 500), (50000, 120, 600)]:
        if rem() < 10:
            break
        h_new = slsqop_optimize(h, n, beta, K, maxit)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 5: Another LP round
    if rem() > 30:
        lp_time = min(rem() * 0.2, 100.0)
        h = lp_refine(h, n, lp_time, rng)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 6: Final SLSQP polish
    for beta, K, maxit in [(10000, 100, 600), (50000, 150, 800)]:
        if rem() < 8:
            break
        h_new = slsqop_optimize(h, n, beta, K, maxit)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 7: Tiny step polish
    if rem() > 15:
        h = polish(h, n, min(rem() * 0.3, 60.0))
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 8: Multi-start perturbations
    if rem() > 60:
        for attempt in range(5):
            if rem() < 20:
                break
            h_try = best_h + rng.normal(0, 0.02, size=best_n)
            h_try = exact_project(h_try, best_n / 2.0)
            h_try = slsqop_optimize(h_try, best_n, 5000, 80, 400)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()

    # Phase 9: Final LP + polish
    if rem() > 20:
        h = lp_refine(best_h, best_n, min(rem() * 0.3, 50.0), rng)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if rem() > 10:
            h = polish(h, best_n, min(rem() * 0.4, 30.0))
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Final feasibility check
    best_h = exact_project(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        ec = evaluate_erdos_solution(best_h)
        if ec is not None and ec < best_c5:
            best_c5 = float(ec)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
