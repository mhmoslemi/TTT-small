        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def get_betas(h, m, n):
    d = np.arange(-(n-1), n)
    b = np.zeros(2*n-1)
    mask1 = (d >= -m) & (d <= n-1-m)
    if np.any(mask1):
        b[mask1] += (1.0 - h[m + d[mask1]])
    mask2 = (d >= m-n+1) & (d <= m)
    if np.any(mask2):
        b[mask2] += (-h[m - d[mask2]])
    return b
