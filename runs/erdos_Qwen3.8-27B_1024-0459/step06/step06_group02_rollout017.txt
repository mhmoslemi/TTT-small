import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.count_nonzero(mask)
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_and_grad_topk(h, n, K=200):
    """Compute correlation and top-K gradient."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get top K lags by correlation value
    if K < len(corr):
        top_idx = np.argsort(-corr)[:K]
    else:
        top_idx = np.arange(len(corr))
    # Compute gradient for top K lags
    g = np.zeros(n)
    for k in top_idx:
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                g[offset:] += (1.0 - h[:npairs])
                g[:npairs] -= h[offset:]
        else:
            a = -offset
            npairs = n - a
            if npairs > 0:
                g[:npairs] += (1.0 - h[a:])
                g[a:] -= h[:npairs]
    return corr, g

def smooth_max_obj(h, beta, n, penalty=1e4):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    shifted = beta * (corr * dx - m * dx)
    shifted = np.clip(shifted, -50, 50)
    ex = np.exp(shifted)
    val = (m + np.log(np.sum(ex)) / beta) * dx
    # Add penalty for sum constraint
    sum_diff = np.sum(h) - n / 2.0
    val += penalty * sum_diff * sum_diff
    return val

def smooth_max_grad(h, beta, n, K=200, penalty=1e4):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    shifted = beta * (corr * dx - m * dx)
    shifted = np.clip(shifted, -50, 50)
    ex = np.exp(shifted)
    s = np.sum(ex)
    w = ex / s  # softmax weights
    
    # Only compute gradient for top K weighted lags
    top_idx = np.argsort(-w)[:K]
    g = np.zeros(n)

        wk = w[k] * dx
        if wk < 1e-15:

        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                g[offset:] += wk * (1.0 - h[:npairs])

            npairs = n - a
            if npairs > 0:
                g[:npairs] += wk * (1.0 - h[a:])
                g[a:] -= wk * h[:npairs]
    # Add penalty gradient
    sum_diff = np.sum(h) - n / 2.0
    g += 2 * penalty * sum_diff
    return g
