import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        at_low = h <= 1e-12
        at_high = h >= 1.0 - 1e-12
        mid = ~(at_low | at_high)
        cnt = int(np.count_nonzero(mid))
        if cnt > 0:
            h[mid] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def binarize(h, n):
    """Convert to binary with exactly n/2 ones."""
    h = h.copy()
    # Sort by value, set top n/2 to 1, rest to 0
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:n//2]] = 1.0
    return b

def precompute_flip_effects(h, n):
    """Precompute flip1 and flip0 matrices."""
    nfull = 2 * n - 1
    flip1 = np.zeros((n, nfull))
    flip0 = np.zeros((n, nfull))
    
    offsets = np.arange(nfull) - (n - 1)  # -（n-1) to n-1
    
    for a in range(n):
        if h[a] >= 0.5:  # It's a 1
            # Flipping 1→0: change in corr[k]
            # Term from n=a: h[a+offset] if valid
            mask1 = (a + offsets >= 0) & (a + offsets < n)
            flip1[a, mask1] = h[a + offsets[mask1]]
            # Term from n=a-offset: (h[a-offset]-1) if valid
            mask2 = (a - offsets >= 0) & (a - offsets < n)
            flip1[a, mask2] += h[a - offsets[mask2]] - 1.0
        else:  # It's a 0
            # Flipping 0→1: change in corr[k]
            mask1 = (a + offsets >= 0) & (a + offsets < n)
            flip0[a, mask1] = -h[a + offsets[mask1]]
            mask2 = (a - offsets >= 0) & (a - offsets < n)
            flip0[a, mask2] += 1.0 - h[a - offsets[mask2]]
    
    return flip1, flip0

def binary_steepest_descent(h, n, max_iters=200, time_limit=300):
    """Steepest descent on binary swaps."""
    h = h.copy().astype(float)
    best_c5 = compute_c5(h)
    start = time.time()
    
    ones_idx = np.where(h >= 0.5)[0]
    zeros_idx = np.where(h < 0.5)[0]
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 30:
            break
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        
        corr = compute_corr(h)
        c_max = float(np.max(corr))
        
        flip1, flip0 = precompute_flip_effects(h, n)
        
        best_improvement = 0
        best_a = -1
        best_b = -1
        best_new_max = c_max
        
        # For each 1-position, find best 0-position to swap with
        for a in ones_idx:
            c_a = corr + flip1[a]
            # For each 0-position b, compute new max
            new_maxs = np.max(c_a[None, :] + flip0[zeros_idx, :], axis=1)
            idx_best = int(np.argmin(new_maxs))
            nm = float(new_maxs[idx_best])
            
            if nm < best_new_max - 1e-12:
                best_new_max = nm
                best_a = a
                best_b = zeros_idx[idx_best]
                best_improvement = c_max - nm
        
        if best_improvement < 1e-12:
            break
        
        # Apply the swap
        h[best_a] = 0.0
        h[best_b] = 1.0
        ones_idx = np.where(h >= 0.5)[0]
        zeros_idx = np.where(h < 0.5)[0]
        
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
    
    return h, best_c5

def binary_simulated_annealing(h, n, time_limit=300, T_start=0.5, T_end=0.01):
    """Simulated annealing on binary swaps."""
    h = h.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    
    nfull = 2 * n - 1
    n_steps = int(time_limit / 0.005)  # estimate
    
    for step in range(n_steps):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 30:
            break
        
        progress = step / max(n_steps, 1)
        T = T_start * (T_end / T_start) ** progress
        
        ones_idx = np.where(h >= 0.5)[0]
        zeros_idx = np.where(h < 0.5)[0]
        
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        
        # Random swap
        a = ones_idx[np.random.randint(len(ones_idx))]
        b = zeros_idx[np.random.randint(len(zeros_idx))]
        
        corr = compute_corr(h)
        old_max = float(np.max(corr))
        
        # Compute new corr after swap
        h[a] = 0.0
        h[b] = 1.0
        new_corr = compute_corr(h)
        new_max = float(np.max(new_corr))
        
        delta = new_max - old_max
        
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            # Accept
            c5_new = new_max * 2.0 / n
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            # Reject
            h[a] = 1.0
            h[b] = 0.0
    
    return best_h, best_c5
