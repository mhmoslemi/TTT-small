        h_new = project_feasible(h_new, n)

        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new.copy()

    return best_h, best_c5


def exponential_weights_loop(h, n, time_limit=120, n_inner=30):
    h = h.copy()
    nfull = 2 * n - 1
    weights = np.ones(nfull) / nfull
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    learning_rate = 2.0

    iteration = 0
    while time.time() - start < time_limit and _BUDGET - (time.time() - _T0) > 20:
        iteration += 1
        # Inner: FW steps with current weights
        progress = min(1.0, iteration / 50.0)
        n_fw = max(5, int(n_inner * (1.0 - 0.5 * progress)))
        h_fw, c5_fw = frank_wolfe_weighted(h, n, weights, n_iter=n_fw,
                                            time_limit=min(20.0, time_limit - (time.time() - start)))
        if c5_fw < best_c5:
            best_c5 = c5_fw
            best_h = h_fw.copy()
        h = h_fw.copy()

        # Outer: update weights via exponential weights
        corr = compute_corr(h)
        values = corr * (2.0 / n)
        max_val = np.max(values)
        if max_val > 0:
            weights = weights * np.exp(learning_rate * values / (max_val + 1e-15))
            weight_sum = np.sum(weights)
            if weight_sum > 0:
                weights /= weight_sum
            else:
                weights = np.ones(nfull) / nfull
        # Decay learning rate
        learning_rate *= 0.95

    return best_h, best_c5


def precompute_dcorr(h, n):
    nfull = 2 * n - 1
    k_range = np.arange(nfull)
    dcorr = np.zeros((n, nfull))
    for i in range(n):
        j1 = i + k_range - (n - 1)
        j2 = i - k_range + (n - 1)
        v1 = (j1 >= 0) & (j1 < n)
        v2 = (j2 >= 0) & (j2 < n)
        dcorr[i, v1] += (1.0 - h[j1[v1]])
        dcorr[i, v2] -= h[j2[v2]]
    return dcorr


def local_pair_search(h, n, time_limit=60, max_passes=30):
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()

    for _pass in range(max_passes):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 10:
            break

        corr = compute_corr(h)
        c_max_val = float(np.max(corr))

        ones_mask = h >= 0.5
        zeros_mask = h <= 0.5
        ones_count = int(np.count_nonzero(ones_mask))
        zeros_count = int(np.count_nonzero(zeros_mask))

        if ones_count < 2 or zeros_count < 2:
            break

        dcorr = precompute_dcorr(h, n)
        ones_idx = np.where(ones_mask)[0]
        zeros_idx = np.where(zeros_mask)[0]

        improved = False
        nfull = 2 * n - 1

        for i in ones_idx:
            if _BUDGET - (time.time() - _T0) < 10:
                break
            base = corr - dcorr[i]
            base_max = np.max(base)
            if base_max >= c_max_val - 1e-13:
                continue

            new_maxs = np.max(base[None, :] + dcorr[zeros_idx], axis=1)
            best_pos = int(np.argmin(new_maxs))
            best_new_max = float(new_maxs[best_pos])

            if best_new_max < c_max_val - 1e-12:
                j = zeros_idx[best_pos]
                h[i] = 0.0
                h[j] = 1.0
                h = project_feasible(h, n)
                c5_new = compute_c5(h)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    improved = True
                    break

        if not improved:
            break

    return h, best_c5


def single_point_adjustment(h, n, time_limit=30, max_passes=50):
    """Try adjusting individual h values slightly to reduce the worst correlation."""
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()

    for _pass in range(max_passes):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 10:
            break

        corr = compute_corr(h)
        c_max_val = float(np.max(corr))
        worst_lag = int(np.argmax(corr))
        offset = worst_lag - (n - 1)

        # Try small adjustments to h values that affect the worst lag
        if offset >= 0:
            affected = list(range(offset)) + list(range(offset, n))
            pairs = [(i, i + offset) for i in range(n - offset)]
        else:
            a = -offset
            pairs = [(i, i + a) for i in range(n - a)]

        improved = False
        for (i, j) in pairs:
            for delta in [0.01, -0.01, 0.02, -0.02, 0.05, -0.05]:
                h_old_i = h[i]
                h[i] = np.clip(h[i] + delta, 0.0, 1.0)
                h[i] = np.clip(h[i] - delta, 0.0, 1.0)
                h_new = project_feasible(h, n)
                c5_try = compute_c5(h_new)
                if c5_try < best_c5 - 1e-12:
                    best_c5 = c5_try
                    h = h_new.copy()
                    improved = True
                    break
            if improved:
                break
            h[i] = h_old_i

        if not improved:
            break

    return h, best_c5


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    x_new = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(x_new, xp, src_h)


def subgradient_descent(h, n, time_limit=40):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()

    for it in range(max(50, int(time_limit * 50))):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 5:
            break

        corr = compute_corr(h)
        m = np.max(corr)
        progress = it / max(100, int(time_limit * 50))
        K = max(5, int(40 * (1.0 - progress)))
        idx = np.argsort(-corr)[:K]
        c_top = corr[idx]
        beta_w = 20.0 + 60.0 * progress
        weights = np.exp(beta_w * (c_top - c_top.max()))
        w_sum = np.sum(weights)
        if w_sum < 1e-30:
            weights = np.ones(K) / K
        else:
            weights /= w_sum

        # Build full weight vector
        full_w = np.zeros(2 * n - 1)
        for wt, k in zip(weights, idx):
            full_w[k] = wt

        g = weighted_gradient(h, n, full_w)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break

        base_step = 0.03 / (1.0 + 0.03 * it)
        step = min(base_step, 0.15) * n / (gn + 1e-14)

        h_new = project_feasible(h - step * g, n)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            found = False
            for factor in [0.5, 0.25, 0.1]:
                h_try = project_feasible(h - step * factor * g, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    found = True
                    break

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)

    # Phase 0: Initialize
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        best_n = len(parent_h)
        best_h = project_feasible(parent_h, best_n)
    except Exception:
        best_n = 512
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_h = project_feasible(best_h, best_n)

    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP at parent resolution
    n = best_n
    h = best_h.copy()
    for beta in [500, 2000, 10000, 30000]:
        if remaining() < 8:
            break
        h_new = slsqp_optimize(h.copy(), n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Exponential weights + Frank-Wolfe
    if remaining() > 30:
        t_limit = min(150.0, remaining() * 0.25)
        h_e, c5_e = exponential_weights_loop(h.copy(), n, time_limit=t_limit)
        if c5_e < best_c5:
            best_c5 = c5_e
            best_h = h_e.copy()
            h = h_e.copy()

    # Phase 3: Subgradient
    if remaining() > 20:
        h_sg, c5_sg = subgradient_descent(h.copy(), n, time_limit=min(40.0, remaining() * 0.15))
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 4: Local pair search
    if n <= 1024 and remaining() > 15:
        h_lp, c5_lp = local_pair_search(h.copy(), n, time_limit=min(60.0, remaining() * 0.25))
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp.copy()
            h = h_lp.copy()

    # SLSQP polish
    for beta in [20000, 80000]:
        if remaining() < 8:
            break
        h_new = slsqp_optimize(h.copy(), n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 5: Try different resolutions
    res_candidates = []
    for mult in [0.5, 0.75, 1.5, 2.0]:
        r = int(round(best_n * mult / 64) * 64)
        if 128 <= r <= 2048 and r % 2 == 0 and abs(r - best_n) > 64:
            res_candidates.append(r)
    res_candidates = sorted(set(res_candidates))

    # Also try fixed resolutions
    for r in [256, 384, 768, 1024]:
        if r not in res_candidates and 128 <= r <= 2048:
            res_candidates.append(r)
    res_candidates = sorted(set(res_candidates))

    for n_t in res_candidates:
        if remaining() < 50:
            break

        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        c5_int = compute_c5(h_feas)

        # SLSQP at new resolution
        for beta in [1000, 5000, 20000]:
            if remaining() < 12:
                break
            h_cur = slsqp_optimize(h_feas.copy(), n_t, beta, maxiter=300)
            c5_cur = compute_c5(h_cur)
            if c5_cur < c5_int:
                c5_int = c5_cur
                h_feas = h_cur.copy()

        # Exponential weights at new resolution
        if remaining() > 30:
            h_e2, c5_e2 = exponential_weights_loop(h_feas.copy(), n_t,
                                                    time_limit=min(60.0, remaining() * 0.2))
            if c5_e2 < c5_int:
                c5_int = c5_e2
                h_feas = h_e2.copy()

        # Local pair search (only for manageable n)
        if n_t <= 640 and remaining() > 20:
            h_lp2, c5_lp2 = local_pair_search(h_feas.copy(), n_t,
                                               time_limit=min(30.0, remaining() * 0.2))
            if c5_lp2 < c5_int:
                c5_int = c5_lp2
                h_feas = h_lp2.copy()

        # Final SLSQP
        if remaining() > 10:
            h_final = slsqp_optimize(h_feas.copy(), n_t, 50000, maxiter=400)
            c5_fin = compute_c5(h_final)
            if c5_fin < c5_int:
                c5_int = c5_fin
                h_feas = h_final.copy()

        if c5_int < best_c5 - 1e-12:
            best_c5 = c5_int
            best_h = h_feas.copy()
            best_n = n_t

    # Phase 6: Intensive multi-cycle refinement at best resolution
    n = best_n
    h = best_h.copy()

    for cycle in range(4):
        if remaining() < 25:
            break

        # SLSQP with high beta
        for beta in [50000, 150000, 400000]:
            if remaining() < 6:
                break
            h_new = slsqp_optimize(h.copy(), n, beta, maxiter=400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

        # Local pair search
        if n <= 768 and remaining() > 15:
            h_lp, c5_lp = local_pair_search(h.copy(), n, time_limit=min(40.0, remaining() * 0.2))
            if c5_lp < best_c5 - 1e-12:
                best_c5 = c5_lp
                best_h = h_lp.copy()
                h = h_lp.copy()

        # Subgradient
        if remaining() > 15:
            h_sg, c5_sg = subgradient_descent(h.copy(), n, time_limit=min(20.0, remaining() * 0.15))
            if c5_sg < best_c5 - 1e-12:
                best_c5 = c5_sg
                best_h = h_sg.copy()
                h = h_sg.copy()

    # Phase 7: Try random binary restarts at manageable resolutions
    if remaining() > 30:
        for n_r in [best_n, 256, 512, 1024]:
            if remaining() < 15:
                break
            if n_r > 1024 or n_r % 2 != 0:
                continue

            # Binary random with structure
            h_rand = np.zeros(n_r)
            n_ones = n_r // 2
            # Use a structured pattern: blocks
            block_size = max(1, n_ones // 8)
            for _ in range(20):
                h_try = np.zeros(n_r)
                positions = rng.choice(n_r, size=n_ones, replace=False)
                h_try[positions] = 1.0
                h_try = project_feasible(h_try, n_r)
                c5_try = compute_c5(h_try)

                if c5_try < 0.45:  # Only pursue promising starts
                    # Quick SLSQP
                    if remaining() > 15:
                        h_opt = slsqp_optimize(h_try, n_r, 5000, maxiter=200)
                        c5_opt = compute_c5(h_opt)
                    else:
                        h_opt, c5_opt = h_try, c5_try

                    if c5_opt < best_c5 - 1e-12:
                        best_c5 = c5_opt
                        best_h = h_opt.copy()
                        best_n = n_r
                        h = h_opt.copy()

    # Phase 8: Final polish
    n = best_n
    h = best_h.copy()
    for beta in [100000, 300000, 800000]:
        if remaining() < 5:
            break
        h_new = slsqp_optimize(h.copy(), n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Final local pair search
    if best_n <= 768 and remaining() > 10:
        h_lp, c5_lp = local_pair_search(best_h.copy(), best_n,
                                         time_limit=min(30.0, remaining() * 0.4))
        if c5_lp < best_c5 - 1e-13:
            best_c5 = c5_lp
            best_h = h_lp.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
