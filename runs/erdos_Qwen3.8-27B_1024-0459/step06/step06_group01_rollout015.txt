import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_CUR_N = 0
_SM_BETA = 1000.0


def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def to_binary(h, n):
    h = np.array(h, dtype=float).flatten()
    n_half = n // 2
    idx = np.argpartition(-h, n_half)[:n_half]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin


def _sm_objective(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = beta * (corr - m)
    z = np.clip(z, -60.0, 60.0)
    ez = np.exp(z)
    return (m + np.log(np.sum(ez)) / beta) * dx


def _sm_gradient(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = beta * (corr - m)
    z = np.clip(z, -60.0, 60.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-16:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    return g


def _eq_fun(x):
    return np.sum(x) - _CUR_N / 2.0


def _eq_jac(x):
    return np.ones(_CUR_N)


def slsqp_opt(h, n, beta, maxiter=500):
    global _CUR_N, _SM_BETA
    _CUR_N = n
    _SM_BETA = beta
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_sm_objective, h0, jac=_sm_gradient, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.fun < _sm_objective(h0) - 1e-14:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0

def topk_equalize(h, n, top_k=5, max_iters=500):
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining_time() < 3:
            break
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        c_max = corr[idx_top[0]]
        c_min_top = corr[idx_top[-1]]
        if c_max - c_min_top < 1e-10:
            break
        g_dir = np.zeros(n)
        for pos, k in enumerate(idx_top):
            w_k = 1.0 if pos == 0 else (-1.0 / (len(idx_top) - 1))
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_dir[off:] += w_k * (1.0 - h[:np_])
                g_dir[:np_] -= w_k * h[off:]
            else:
                a = -off
                np_ = n - a
                g_dir[:np_] += w_k * (1.0 - h[a:])
                g_dir[a:] -= w_k * h[:np_]
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        d = g_dir / gn
        improved = False
        steps = [0.002, 0.005, 0.01, 0.02, 0.001, 0.008, 0.03, 0.05]
        for step in steps:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-13:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            for step in [0.005, 0.01, 0.02]:
                ht = project_feasible(h - step * d, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-13:
                    h = ht
                    best_c5 = cn
                    improved = True
                    break
            if not improved:
                break
    return h, best_c5


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def coord_descent(h, n, max_sweeps=40):
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        any_improve = False
        for j in range(n):
            old_val = h[j]
            a, b = 0.0, 1.0
            best_v = old_val
            best_obj = compute_c5(h)
            for _ in range(30):
                m1 = a + (b - a) * 0.381966
                m2 = a + (b - a) * 0.618034
                h[j] = m1
                c1 = compute_c5(h)
                h[j] = m2
                c2 = compute_c5(h)
                if c1 < c2:
                    b = m2
                    h[j] = m1
                    if c1 < best_obj:
                        best_obj = c1
                        best_v = m1
                else:
                    a = m1
                    h[j] = m2
                    if c2 < best_obj:

            curr_c5 = compute_c5(h)
            if curr_c5 < best_c5 - 1e-14:

        if c5_opt < best_c5 - 1e-14:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t

    # Phase 8: Coordinate descent on best
    if remaining_time() > 40:
        h_cd, c5_cd = coord_descent(best_h.copy(), best_n, max_sweeps=30)
        if c5_cd < best_c5 - 1e-14:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 9: Final SLSQP polish with very high beta
    for beta in [1000000, 5000000]:
        if remaining_time() < 20:
            break
        h_final = slsqp_opt(best_h, best_n, beta, maxiter=500)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-14:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Phase 10: Final top-k equalize
    if remaining_time() > 15:
        h_eq2, c5_eq2 = topk_equalize(best_h, best_n, top_k=9, max_iters=300)
        if c5_eq2 < best_c5 - 1e-14:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()

    # Final feasibility and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
