def smooth_max_jac_vec(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    s = np.clip(beta * (corr - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for ao in range(n):  # ao = 0, 1, ..., n-1
        if ao == 0:
            w_pos = w[n-1]  # k = n-1, offset = 0
            g += w_pos * dx * (1.0 - 2.0 * x)
        else:
            w_pos = w[n-1+ao]  # offset = +ao
            w_neg = w[n-1-ao]  # offset = -ao
            # offset = +ao: (1-h[j-ao]) for j>=ao, -h[j+ao] for j<n-ao
            g[ao:] += w_pos * dx * (1.0 - x[:n-ao])
            g[:n-ao] -= w_pos * dx * x[ao:]
            # offset = -ao: (1-h[j+ao]) for j<n-ao, -h[j-ao] for j>=ao
            g[:n-ao] += w_neg * dx * (1.0 - x[ao:])
            g[ao:] -= w_neg * dx * x[:n-ao]
    return g

The objective is max_k sum_i h[i](1-h[i+k]). If I fix all but one index j, the problem becomes:

- If k corresponds to offset ao > 0 and j < n-ao: -h[j+ao]*... wait, this is getting complicated.

Let me think differently. For a fixed j:
corr[k] = A_k + B_k * h[j] + C_k * h[j]² ... no, there's no h[j]² term (unless k=0).

Actually for k≠0: corr[k] is linear in h[j] (since h[j] appears multiplied by either 1 or h[j±ao]).
For k=0: corr[0] contains h[j]² term.

So the problem for a fixed j is:
min max_k (a_k + b_k * h[j])  (for k≠0)
         (a_0 + h[j] - h[j]²)  (for k=0)

subject to 0 <= h[j] <= 1 and the sum constraint.

This is a 1D min-max problem which can be solved by bisection! For a given t, check if there exists h[j] in [0,1] such that all a_k + b_k*h[j] <= t and a_0 + h[j] - h[j]² <= t.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if interior.any():
            h[interior] += diff / interior.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    s = np.clip(beta * (corr - m), -500.0, 500.0)
    e = np.exp(s)
    return (m + np.log(e.sum()) / beta) * dx

def smooth_max_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    s = np.clip(beta * (corr - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for ao in range(n):
        if ao == 0:
            wk = w[n-1] * dx
            if wk < 1e-15:
                continue
            g += wk * (1.0 - 2.0 * x)
        else:
            wp = w[n-1+ao] * dx
            wn = w[n-1-ao] * dx
            if wp < 1e-15 and wn < 1e-15:
                continue
            if wp >= 1e-15:
                g[ao:] += wp * (1.0 - x[:n-ao])
                g[:n-ao] -= wp * x[ao:]
            if wn >= 1e-15:
                g[:n-ao] += wn * (1.0 - x[ao:])
                g[ao:] -= wn * x[:n-ao]
    return g

    return x.sum() - n / 2.0

def eq_jac(x, n):
    return np.ones(n)

def slsqp_run(h, beta, maxiter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_max_obj, h, args=(beta,),
                       jac=smooth_max_grad, method='SLSQP',
                       bounds=bounds,

                                     'jac': eq_jac, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})

            return project_feasible(res.x, n)
    except Exception:
        pass
    return h
