    """Focus on reducing the top K correlation lags specifically."""
    h = h.copy()
    best_c5 = compute_c5(h)
    
    for step in range(steps):
        if _BUDGET - (time.time() - _T0) < 5:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        
        # Get top K lags
        top_k = np.argsort(-corr)[:num_lags]
        # Weight by how close to max
        weights = np.exp(50.0 * (corr[top_k] - c_max))
        weights /= np.sum(weights)
        
        # Compute gradient for weighted sum of top K lags
        g = np.zeros(n)
        dx = 2.0 / n
        for wt, k in zip(weights, top_k):
            if wt < 1e-14:
                continue
            offset = k - (n - 1)
            if offset >= 0:
                npairs = n - offset
                g[offset:offset + npairs] += wt * (1.0 - h[:npairs])
                g[:npairs] -= wt * h[offset:offset + npairs]
            else:
                a = -offset
                npairs = n - a
                g[:npairs] += wt * (1.0 - h[a:a + npairs])
                g[a:a + npairs] -= wt * h[:npairs]
        
        g *= dx
        
        # Projected gradient step
        step_size = lr / (np.linalg.norm(g) + 1e-10)
        h_new = project_feasible(h - step_size * g, n)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            h = h_new
        else:
            # Try smaller step
            for factor in [0.5, 0.25, 0.1]:
                h_try = project_feasible(h - step_size * factor * g, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    h = h_try
                    break
    
    return h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] += diff / np.sum(free)
        else:
            if diff > 0:
                idx = np.argmin(h)
            else:
                idx = np.argmax(h)
            h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_value(x, beta, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50.0, 50.0)
    sm = m + np.log(np.sum(np.exp(shifted))) / beta
    return sm * (2.0 / n)


def smooth_max_grad(x, beta, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50.0, 50.0)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = w / w_sum
    g = np.zeros(n)
    dx = 2.0 / n
    nfull = 2 * n - 1
    for k in range(nfull):
        wk = w[k] * dx
        if wk < 1e-16:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            g[offset:offset + npairs] += wk * (1.0 - x[:npairs])
            g[:npairs] -= wk * x[offset:offset + npairs]
        else:
            a = -offset
            npairs = n - a
            g[:npairs] += wk * (1.0 - x[a:a + npairs])
            g[a:a + npairs] -= wk * x[:npairs]
    return g


def slsqp_optimize(h, n, beta, maxiter=500):
    def obj_func(x):
        return smooth_max_value(x, beta, n)
    def grad_func(x):
        return smooth_max_grad(x, beta, n)
    def eq_fun(x):
        return np.sum(x) - n / 2.0
    def eq_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(obj_func, h, jac=grad_func, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()


def top_lag_gradient_step(h, n, K=15, beta_w=80.0):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    weights = np.exp(beta_w * (c_top - c_top.max()))
    w_sum = np.sum(weights)
    if w_sum < 1e-30:
        weights = np.ones(K) / K
    else:
        weights /= w_sum
    g = np.zeros(n)
    dx = 2.0 / n
    for wt, k in zip(weights, idx):
        if wt < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            np_ = n - offset
            g[offset:offset + np_] += wt * (1.0 - h[:np_])
            g[:np_] -= wt * h[offset:offset + np_]
        else:
            a = -offset
            np_ = n - a
            g[:np_] += wt * (1.0 - h[a:a + np_])
            g[a:a + np_] -= wt * h[:np_]
    return g * dx

        gn = np.linalg.norm(g)

            h_tl, c5_tl = top_lag_descent(h.copy(), n, time_limit=min(15.0, remaining() * 0.2))
            if c5_tl < best_c5 - 1e-14:
                best_c5 = c5_tl
                best_h = h_tl.copy()
                h = h_tl.copy()

    # Phase 8: Final local search
    if best_n <= 1024 and remaining() > 10:
        h_ls, c5_ls = local_swap_search(best_h.copy(), best_n, time_limit=min(20.0, remaining() * 0.4))
        if c5_ls < best_c5 - 1e-14:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Final SLSQP polish
    if remaining() > 5:
        h_new = slsqp_optimize(best_h.copy(), best_n, 500000, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
