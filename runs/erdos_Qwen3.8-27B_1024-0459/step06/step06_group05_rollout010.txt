from scipy.optimize import minimize
import time

_T0 = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        g[offset:] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:]
    else:
        a = -offset
        npairs = n - a
        g[:npairs] += (1.0 - h[a:])
        g[a:] -= h[:npairs]
    return g

def smooth_max_eval(h, n, idx, beta):
    corr = compute_corr(h)
    c_top = corr[idx]
    m = np.max(c_top)
    ex = np.exp(beta * (c_top - m))
    return (m + np.log(ex.sum()) / beta) * (2.0 / n), c_top, ex

def smooth_max_grad(h, n, idx, beta):
    corr = compute_corr(h)
    c_top = corr[idx]
    m = np.max(c_top)
    ex = np.exp(np.clip(beta * (c_top - m), -50, 50))
    w = ex / ex.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for i in range(len(idx)):
        wk = w[i] * dx
        if wk > 1e-15:
            g += wk * grad_corr_k(h, idx[i], n)
    return g, c_top.max() * dx

def level_penalty(h, n, idx, beta, gamma):
    """Penalty for variance of top-k correlation values"""
    corr = compute_corr(h)
    c_top = corr[idx]
    m = c_top.max()
    ex = np.exp(np.clip(beta * (c_top - m), -50, 50))
    w = ex / ex.sum()
    weighted_mean = w @ c_top
    var = (w * (c_top - weighted_mean)**2).sum()
    return gamma * var * (2.0/n)**2

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        g[offset:] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:]
    else:
        a = -offset
        npairs = n - a
        g[:npairs] += (1.0 - h[a:])
        g[a:] -= h[:npairs]
    return g

def get_top_indices(corr, K):
    return np.argsort(-corr)[:K]

def smooth_max_and_grad(h, n, K, beta):
    corr = compute_corr(h)
    idx = get_top_indices(corr, K)
    c_top = corr[idx]
    m = c_top.max()
    shifted = beta * (c_top - m)
    shifted = np.clip(shifted, -50, 50)
    ex = np.exp(shifted)
    w = ex / ex.sum()
    val = (m + np.log(ex.sum()) / beta) * (2.0 / n)
    
    dx = 2.0 / n
    g = np.zeros(n)
    for i in range(K):
        wi = w[i] * dx
        if wi > 1e-15:
            g += wi * grad_corr_k(h, idx[i], n)
    return val, g, idx, c_top

def nesterov_optimize(h, n, time_limit, K=50, beta=200, momentum_init=0.9):
    """Nesterov accelerated projected gradient descent"""
    h_curr = h.copy()
    h_best = h.copy()
    c5_best = compute_c5(h)
    start = time.time()
    
    # Initial step size estimation
    g0, _, idx0, _ = smooth_max_and_grad(h_curr, n, K, beta)
    g0_norm = np.linalg.norm(g0)
    if g0_norm < 1e-15:
        return h_best, c5_best
    alpha0 = 0.1 / g0_norm
    alpha0 = min(alpha0, 0.3)
    
    h_prev = h_curr.copy()
    t = 1.0
    
    for it in range(20000):
        if time.time() - start > time_limit:
            break
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        
        # Nesterov momentum
        gamma = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / (2.0 * t + 1.0)
        y = h_curr + (gamma - 1.0) / gamma * (h_curr - h_prev)
        y = np.clip(y, 0.0, 1.0)
        
        val, g, idx, c_top = smooth_max_and_grad(y, n, K, beta)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        
        # Backtracking line search
        alpha = alpha0
        for ls in range(15):
            h_try = project_feasible(y - alpha * g, n)
            val_try, g_try, _, _ = smooth_max_and_grad(h_try, n, K, beta)
            if val_try <= val - 1e-10 * np.max([np.linalg.norm(g), 1e-10]) * alpha:
                break
            alpha *= 0.5
        
        h_new = project_feasible(y - alpha * g, n)
        c5_new = compute_c5(h_new)
        
        h_prev = h_curr.copy()
        h_curr = h_new
        t += 1
        
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_curr.copy()
    
    return h_best, c5_best

def slsqp_optimize(h, n, K, beta, maxiter=200):
    """SLSQP with smooth max objective"""
    def obj(x):
        corr = compute_corr(x)
        idx = get_top_indices(corr, K)
        c_top = corr[idx]
        m = c_top.max()
        ex = np.exp(np.clip(beta * (c_top - m), -50, 50))
        return (m + np.log(ex.sum()) / beta) * (2.0 / n)
    
    def grad(x):
        corr = compute_corr(x)
        idx = get_top_indices(corr, K)
        c_top = corr[idx]
        m = c_top.max()
        ex = np.exp(np.clip(beta * (c_top - m), -50, 50))
        w = ex / ex.sum()
        dx = 2.0 / n
        g = np.zeros(n)
        for i in range(K):
            wi = w[i] * dx
            if wi > 1e-15:
                g += wi * grad_corr_k(x, idx[i], n)
        return g
    
    def cfun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            c5_old = compute_c5(h)
            if c5_new <= c5_old + 1e-12:
                return h_new
    except Exception:
        pass
    return h

def make_tent(n):
    x = np.linspace(0, 2, n, endpoint=False) + 1.0/n
    h = 1.0 - np.abs(x - 1.0)
    h = np.clip(h, 0, 1)
    return project_feasible(h, n)

def make_parabola(n):
    x = np.linspace(0, 2, n, endpoint=False) + 1.0/n
    h = 1.0 - 0.75 * (x - 1.0)**2
    h = np.clip(h, 0, 1)
    return project_feasible(h, n)

def make_step(n):
    h = np.zeros(n)
    h[n//4:3*n//4] = 1.0
    return project_feasible(h, n)
