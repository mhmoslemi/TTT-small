from scipy.fft import fft, ifft, rfft, irfft

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr_array(h):
    return np.correlate(h, 1.0 - h, mode='full')

def fft_corr(h, one_minus_h):
    """Compute full correlation using FFT."""
    n = len(h)
    m = 2 * n - 1
    size = 1
    while size < m:
        size *= 2
    H = fft(h, size)
    G = fft(one_minus_h, size)
    corr = irfft(H * np.conj(G), size)[:m]
    return corr

def grad_corr_at_lag(h, j, n):
    """Gradient of correlation at lag j (j = k - (n-1), can be negative)."""
    g = np.zeros(n)
    if j >= 0:

        # d(c)/d(h[i]) for i in [0, n-1-j]: (1-h[i+j])
        # d(c)/d(h[i]) for i in [j, n-1]: -h[i-j]
        for i in range(n - j):
            g[i] += (1.0 - h[i + j])
            g[i + j] -= h[i]
    else:
        jj = -j
        # c = sum_{i=jj}^{n-1} h[i] * (1-h[i-jj])
        for i in range(jj, n):
            g[i] += (1.0 - h[i - jj])
            g[i - jj] -= h[i]

def slsqp_optimize(h, n, maxiter=200):
    """SLSQP optimization targeting the max correlation."""
    def obj(x):
        corr = compute_corr_array(x)
        return np.max(corr) * (2.0 / n)
    
    def grad(x):
        corr = compute_corr_array(x)
        k_max = np.argmax(corr)
        j = k_max - (n - 1)
        g = grad_corr_at_lag_vec(x, j, n)
        return g * (2.0 / n)
    
    def cfun(x):
        return np.sum(x) - n / 2.0
    
    def cjac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            c5_old = compute_c5(h)
            if c5_new <= c5_old + 1e-12:
                return h_new
    except Exception:
        pass
    return h

def smooth_max_objective(h, n, K, beta):
    """Smooth max of top-K correlations."""
    corr = compute_corr_array(h)
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    m = np.max(c_top)
    sh = np.clip(beta * (c_top - m), -50.0, 50.0)
    ex = np.exp(sh)
    s = ex.sum()
    if s < 1e-30:
        return m * (2.0 / n)
    return (m + np.log(s) / beta) * (2.0 / n)

def smooth_max_grad(h, n, K, beta):
    """Gradient of smooth max objective."""
    corr = compute_corr_array(h)
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    m = np.max(c_top)
    sh = np.clip(beta * (c_top - m), -50.0, 50.0)
    ex = np.exp(sh)
    s = ex.sum()
    if s < 1e-30:
        w = np.ones(K) / K
    else:
        w = ex / s
    
    dx = 2.0 / n
    g = np.zeros(n)
    for i in range(K):
        if w[i] < 1e-15:
            continue
        j = idx[i] - (n - 1)
        g += w[i] * dx * grad_corr_at_lag_vec(h, j, n)
    return g

def nesterov_optimize(h, n, time_limit, K=10, beta=200):
    """Nesterov accelerated gradient with adaptive step."""
    h_curr = h.copy()
    h_best = h.copy()
    c5_best = compute_c5(h)
    h_prev = h.copy()
    start = time.time()
    
    it = 0
    while time.time() - start < time_limit and (_BUDGET - (time.time() - _T0)) > 5:
        it += 1
        # Nesterov step
        if it == 1:
            y = h_curr.copy()
        else:
            gamma = (1.0 + np.sqrt(1.0 + 4.0 * (it - 1.0) ** 2)) / (2.0 * it)
            y = h_curr + gamma * (h_curr - h_prev)
            y = np.clip(y, 0.0, 1.0)
        
        g = smooth_max_grad(y, n, K, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        
        # Adaptive step size
        step = 0.1 / gn
        step = min(step, 0.3)
        
        h_new = project_feasible(y - step * g, n)
        c_new = compute_c5(h_new)
        
        h_prev = h_curr.copy()
        h_curr = h_new
        
        if c_new < c5_best:
            c5_best = c_new
            h_best = h_curr.copy()
    
    return h_best, c5_best

Let me think about this from a different angle. The problem is equivalent to:
minimize max_j sum_i h[i](1-h[i+j])
subject to 0 ≤ h[i] ≤ 1, sum h[i] = n/2

This is a convex optimization problem. Let me think about the dual.

The primal is: min max_j f_j(h) s.t. g(h) = 0, 0 ≤ h ≤ 1
where f_j(h) = sum_i h[i](1-h[i+j]) and g(h) = sum h[i] - n/2.

The dual would involve Lagrange multipliers for each constraint. At the optimum:
∇_h [max_j f_j(h)] = λ * 1 + μ (where μ handles the box constraints)

For the active set of lags A (where f_j is maximal):
∇_h [sum_{j∈A} α_j f_j(h)] = λ * 1 + μ (for some α_j ≥ 0, sum α_j = 1)

This gives: for each coordinate i that's not at a boundary:
sum_{j∈A} α_j * ∂f_j/∂h[i] = λ

where ∂f_j/∂h[i] = (1-h[i+j]) - h[i-j] (for appropriate j, with boundary handling).

This is a system of equations that determines the structure of the optimal h.

For the binary case, all h[i] are at boundaries (0 or 1), so the KKT conditions only apply to the constraint sum h[i] = n/2. The optimality condition becomes: there exist α_j ≥ 0 such that for all i with h[i]=1:
sum_{j∈A} α_j * (∂f_j/∂h[i]) ≤ λ
and for all i with h[i]=0:
sum_{j∈A} α_j * (∂f_j/∂h[i]) ≥ λ

This is a complex condition but it gives insight into the structure.
