import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_CUR_N = 0
_SM_BETA = 1000.0


def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def _sm_obj(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    return (m + np.log(np.sum(ez)) / beta) * dx


def _sm_grad(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    return g


def _eq_fun(x):
    return np.sum(x) - _CUR_N / 2.0


def _eq_jac(x):
    return np.ones(_CUR_N)


def slsqp_opt(h, n, beta, maxiter=1000):
    global _CUR_N, _SM_BETA
    _CUR_N = n
    _SM_BETA = beta
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_sm_obj, h0, jac=_sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.fun < _sm_obj(h0) - 1e-15:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def grad_single_shift(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        np_ = n - off
        g[off:] += (1.0 - h[:np_])
        g[:np_] -= h[off:]
    else:
        a = -off
        np_ = n - a
        g[:np_] += (1.0 - h[a:])
        g[a:] -= h[:np_]
    return g


def fast_projected_gradient(h, n, max_iters=5000, time_limit=120, beta=10000):
    """Custom projected gradient with backtracking on actual C5."""
    global _CUR_N, _SM_BETA
    _CUR_N = n
    _SM_BETA = beta
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    step_size = 0.01

    for it in range(max_iters):
        if time.time() - start > time_limit or remaining_time() < 5:
            break
        g = _sm_grad(h)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = -g / g_norm * step_size
        improved = False
        for _ in range(10):
            h_new = project_feasible(h + g, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                improved = True
                break
            g *= 0.5
        if not improved:
            step_size *= 0.8
        else:
            step_size = min(step_size * 1.2, 0.05)
    return best_h, best_c5


def directed_binary_search(h_bin, n, time_limit=120, rng=None):

    if rng is None:
        rng = np.random.RandomState(42)
    h = (h_bin > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:

        h = np.zeros(n)
        h[sort_idx[:n_half]] = 1.0

    dx = 2.0 / n
    corr = compute_corr(h)
    current_max = np.max(corr)
    best_c5 = current_max * dx
    best_h = h.copy()

    stale = 0

    zeros = np.where(h == 0.0)[0].tolist()

        top_ks = np.argsort(-corr)[:3]

        n_candidates = 0
        best_swap = None
        best_new_max = current_max

        if 0 <= s_max < n:
            directed_ones = []
            directed_zeros = []
            for i in range(n - s_max - 1 if s_max > 0 else n):
                j = i + s_max
                if h[i] == 1.0 and h[j] == 0.0 and j < n:
                    directed_zeros.append(j)
                if h[i] == 0.0 and h[j] == 1.0 and j < n:
                    directed_ones.append(j)
                    directed_zeros.append(i)
            if directed_ones and directed_zeros:
                for _ in range(min(100, len(directed_ones) * len(directed_zeros))):
                    a = directed_ones[rng.randint(len(directed_ones))]
                    b = directed_zeros[rng.randint(len(directed_zeros))]
                    if a == b or h[a] != 0.0 or h[b] != 1.0:
                        continue
                    h[a] = 1.0
                    h[b] = 0.0
                    new_corr = compute_corr(h)
                    new_max = np.max(new_corr)
                    h[a] = 0.0
                    h[b] = 1.0
                    n_candidates += 1
                    if new_max < best_new_max:
                        best_new_max = new_max
                        best_swap = (a, b)

        # Also try random swaps

            a = zeros[rng.randint(len(zeros))]
            b = ones[rng.randint(len(ones))]
            h[a] = 1.0
            h[b] = 0.0

                break

                    a = zeros[rng.randint(len(zeros))]
                    b = ones[rng.randint(len(ones))]
                    h[a] = 1.0
                    h[b] = 0.0

            a = zeros[rng.randint(len(zeros))]
            b = ones[rng.randint(len(ones))]
            h[a] = 1.0

                # Update lists
                ia = zeros.index(a)
                ib = ones.index(b)
                zeros[ia] = b
                ones[ib] = a
                current_max = new_max
            else:
                h[a] = 0.0
                h[b] = 1.0
            continue

        # Accept best swap
        a, b = best_swap
        h[a] = 1.0
        h[b] = 0.0
        ia = zeros.index(a)
        ib = ones.index(b)
        zeros[ia] = b
        ones[ib] = a

def top_k_equalize(h, n, top_k=5, max_iters=800, time_limit=60):

        c_min_top = corr[idx_top[-1]]
        if c_max - c_min_top < 1e-12:
            break
        g_dir = np.zeros(n)
        n_top = len(idx_top)
        for pos, k in enumerate(idx_top):
            w_k = 1.0 if pos == 0 else (-1.0 / (n_top - 1))

        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:

        d = g_dir / gn

def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def theta_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)


def sturmian_word(n, alpha):
    """Generate a Sturmian word (balanced binary sequence) with density alpha."""
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (int((i + 1) * alpha) - int(i * alpha)) == 1 else 0.0
    h = (h > 0.5).astype(float)
    # Fix to exactly n/2 ones
    n_ones = int(np.sum(h))
    n_half = n // 2
    if n_ones != n_half:
        if n_ones > n_half:
            ones = np.where(h == 1.0)[0]
            rng = np.random.RandomState(0)
            rng.shuffle(ones)
            h[ones[n_half:]] = 0.0
        else:
            zeros = np.where(h == 0.0)[0]
            rng = np.random.RandomState(0)
            rng.shuffle(zeros)
            h[zeros[:n_half - n_ones]] = 1.0
    return h


def block_pattern(n, block_sizes):
    """Create a periodic block pattern."""
    h = np.zeros(n)
    total = sum(block_sizes)
    for i in range(n):
        pos = i % total
        cum = 0
        val = 0
        for bs in block_sizes:
            if pos < cum + bs:
                val = 0
                break
            cum += bs
            val = 1 - val
        h[i] = 1.0 if val == 1 else 0.0
    # Fix count
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        if n_ones > n_half:
            ones = np.where(h == 1.0)[0]
            h[ones[n_half:]] = 0.0
        else:
            zeros = np.where(h == 0.0)[0]
            h[zeros[:n_half - n_ones]] = 1.0
    return h

                break
            h_c2 = slsqp_opt(h_c2, n, beta, maxiter=500)
        c5_c2 = compute_c5(h_c2)
        if c5_c2 < c5 - 1e-14:
            h = h_c2.copy()
            c5 = c5_c2
        if c5_b2 < c5 - 1e-14:
            h = h_b2.copy().astype(float)
            c5 = c5_b2

    return h, c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _CUR_N
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    # Load parent
    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = theta_morse(640)
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Optimize parent at native resolution
    h1, c5_1 = multi_start_optimize(best_h, best_n, time_per_start=180, rng=rng)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Try higher resolutions
    for n_t in [1024, 2048, 960]:
        if remaining_time() < 80:
            break
        if n_t == best_n:
            continue
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        h_t, c5_t = multi_start_optimize(h_feas, n_t, time_per_start=100, rng=rng)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            best_n = n_t

    # Phase 3: Try different starting structures at best_n
    starts = []
    # Thue-Morse
    starts.append(theta_morse(best_n))
    # Sturmian with alpha=0.5
    starts.append(sturmian_word(best_n, 0.5))
    # Block patterns
    starts.append(block_pattern(best_n, [3, 1]))
    starts.append(block_pattern(best_n, [2, 1, 1]))
    # Random balanced
    h_rand = np.zeros(best_n)
    h_rand[rng.choice(best_n, best_n // 2, replace=False)] = 1.0
    starts.append(h_rand)
    # Shifted parent
    starts.append(project_feasible(np.roll(best_h, best_n // 3), best_n))

    for h_start in starts:
        if remaining_time() < 50:
            break
        h_opt, c5_opt = multi_start_optimize(h_start, best_n, time_per_start=50, rng=rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 4: Intensive final refinement on best solution
    if remaining_time() > 30:
        n = best_n
        h = best_h.copy()

        # SLSQP with very high beta
        for beta in [2000000, 10000000, 50000000]:
            if remaining_time() < 10:
                break
            h_new = slsqp_opt(h, n, beta, maxiter=3000)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

        # Top-k with larger k
        if remaining_time() > 20:
            h_eq, c5_eq = top_k_equalize(h, n, top_k=10, max_iters=1000, time_limit=40)
            if c5_eq < best_c5 - 1e-15:
                best_c5 = c5_eq
                best_h = h_eq.copy()
                h = h_eq.copy()

        # Directed binary search (long)
        if remaining_time() > 20:
            h_bin = (h > 0.5).astype(float)
            h_b, c5_b = directed_binary_search(h_bin, n, time_limit=60, rng=rng)
            h_c = h_b.copy()
            for beta in [1000000, 10000000]:
                if remaining_time() < 10:
                    break
                h_c = slsqp_opt(h_c, n, beta, maxiter=1000)
            c5_c = compute_c5(h_c)
            if c5_c < best_c5 - 1e-15:
                best_c5 = c5_c
                best_h = h_c.copy()
            if c5_b < best_c5 - 1e-15:
                best_c5 = c5_b
                best_h = h_b.copy().astype(float)

        # Subgradient final
        if remaining_time() > 20:
            h_sg, c5_sg = subgradient_zigzag(best_h, n, max_iters=8000, time_limit=60)
            if c5_sg < best_c5 - 1e-15:
                best_c5 = c5_sg
                best_h = h_sg.copy()

    # Phase 5: Try a couple more resolutions if time
    if remaining_time() > 60:
        for n_t in [1280, 768]:
            if remaining_time() < 40:
                break
            if n_t == best_n:
                continue
            h_interp = interp_to(best_h, best_n, n_t)
            h_feas = project_feasible(h_interp, n_t)
            h_t, c5_t = multi_start_optimize(h_feas, n_t, time_per_start=50, rng=rng)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = n_t

    # Phase 6: Final polish
    if remaining_time() > 15:
        n = best_n
        h_final = slsqp_opt(best_h, n, 50000000, maxiter=2000)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-15:
            best_c5 = c5_final
            best_h = h_final.copy()

    if remaining_time() > 10:
        h_eq, c5_eq = top_k_equalize(best_h, best_n, top_k=8, max_iters=500, time_limit=20)
        if c5_eq < best_c5 - 1e-15:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
