            h_bin[j] = 0.0

def interp_to(src, src_n, dst_n):

    return np.interp(xn, xp, src)

    # Phase 3: Subgradient accelerated
    if remaining_time() > 20:
        h_sg, c5_sg = subgradient_accelerated(h, best_n, max_iters=10000, time_limit=40)
        if c5_sg < best_c5 - 1e-15:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 4: Multi-resolution - go UP to n=1024
    n_high = 1024
    if remaining_time() > 120:
        h_high = interp_to(best_h, best_n, n_high)
        h_high = project_feasible(h_high, n_high)
        c5_high = compute_c5(h_high)

        for beta in [100, 1000, 10000, 50000, 200000, 1000000]:
            if remaining_time() < 20:
                break
            h_high = slsqp_opt(h_high, n_high, beta, maxiter=2000)
            c5_high = compute_c5(h_high)

        if remaining_time() > 30:
            h_high, c5_high = bottleneck_reduce(h_high, n_high, n_top=5, max_iters=3000, time_limit=50)
        if remaining_time() > 20:
            h_high, c5_high = subgradient_accelerated(h_high, n_high, max_iters=5000, time_limit=30)

        # Compare
        if c5_high < best_c5 - 1e-15:
            best_c5 = c5_high
            best_h = h_high.copy()
            best_n = n_high
            h = h_high.copy()
        else:
            # Project back and try to improve
            h_back = interp_to(h_high, n_high, best_n)
            h_back = project_feasible(h_back, best_n)
            c5_back = compute_c5(h_back)
            if c5_back < best_c5 - 1e-15:
                best_c5 = c5_back
                best_h = h_back.copy()
                h = h_back.copy()

    # Phase 5: Try n=768 and n=896
    for n_try in [768, 896]:
        if remaining_time() < 50 or n_try == best_n:
            continue
        h_try = interp_to(best_h, best_n, n_try)
        h_try = project_feasible(h_try, n_try)
        for beta in [500, 10000, 100000]:
            if remaining_time() < 15:
                break
            h_try = slsqp_opt(h_try, n_try, beta, maxiter=1500)
        if remaining_time() > 20:
            h_try, c5_try = bottleneck_reduce(h_try, n_try, n_top=5, max_iters=2000, time_limit=30)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-15:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = n_try
            h = h_try.copy()

    # Phase 6: Binary SA + fractional relaxation
    if remaining_time() > 60:
        h_bin = (best_h > 0.5).astype(float)
        h_sa, c5_sa = binary_sa(h_bin, best_n, time_limit=min(80, int(remaining_time() * 0.4)), rng=rng)
        # Fractional relaxation
        h_frac = h_sa.copy()
        for beta in [1000, 10000, 100000, 1000000]:
            if remaining_time() < 10:
                break
            h_frac = slsqp_opt(h_frac, best_n, beta, maxiter=1500)
        if remaining_time() > 15:
            h_frac, c5_frac = bottleneck_reduce(h_frac, best_n, n_top=5, max_iters=2000, time_limit=25)
        c5_frac = compute_c5(h_frac)
        if c5_frac < best_c5 - 1e-15:
            best_c5 = c5_frac
            best_h = h_frac.copy()
            h = h_frac.copy()
        if c5_sa < best_c5 - 1e-15:
            best_c5 = c5_sa
            best_h = h_sa.copy().astype(float)

    # Phase 7: Random restarts at current best resolution
    if remaining_time() > 80:
        for restart in range(4):
            if remaining_time() < 30:
                break
            if restart == 0:
                h_r = theta_morse_start(best_n)
            elif restart == 1:
                h_r = np.zeros(best_n)
                h_r[rng.choice(best_n, best_n // 2, replace=False)] = 1.0
            elif restart == 2:
                h_r = best_h.copy()
                h_r[rng.choice(best_n, best_n // 8, replace=False)] = 1.0 - h_r[rng.choice(best_n, best_n // 8, replace=False)]
                h_r = project_feasible(h_r, best_n)
            else:
                # Structured blocks
                h_r = np.zeros(best_n)
                block_size = max(4, best_n // 32)
                for b in range(0, best_n, block_size):
                    if rng.random() > 0.5:
                        h_r[b:b+block_size] = 1.0
                h_r = project_feasible(h_r, best_n)

            for beta in [500, 10000, 100000]:
                if remaining_time() < 15:
                    break
                h_r = slsqp_opt(h_r, best_n, beta, maxiter=1500)
            if remaining_time() > 15:
                h_r, c5_r = bottleneck_reduce(h_r, best_n, n_top=5, max_iters=2000, time_limit=20)
            c5_r = compute_c5(h_r)
            if c5_r < best_c5 - 1e-15:
                best_c5 = c5_r
                best_h = h_r.copy()
                h = h_r.copy()

    # Phase 8: Fractional local optimization
    if remaining_time() > 20:
        h_fl, c5_fl = frac_local_opt(h, best_n, n_frac=30, time_limit=min(30, int(remaining_time() * 0.3)))
        if c5_fl < best_c5 - 1e-15:
            best_c5 = c5_fl
            best_h = h_fl.copy()
            h = h_fl.copy()

    # Phase 9: Final intensive SLSQP polish
    n = best_n
    h = best_h.copy()
    for beta in [500000, 2000000, 10000000, 50000000]:
        if remaining_time() < 10:
            break
        h_new = slsqp_opt(h, n, beta, maxiter=5000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 10: Final bottleneck + subgradient
    if remaining_time() > 20:
        h_final, c5_final = bottleneck_reduce(h, n, n_top=3, max_iters=3000, time_limit=min(30, int(remaining_time() * 0.4)))
        if c5_final < best_c5 - 1e-15:
            best_c5 = c5_final
            best_h = h_final.copy()

    if remaining_time() > 15:
        h_sg, c5_sg = subgradient_accelerated(best_h, n, max_iters=5000, time_limit=min(25, int(remaining_time() * 0.4)))
        if c5_sg < best_c5 - 1e-15:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Final SLSQP
    if remaining_time() > 10:
        h_last = slsqp_opt(best_h, n, 50000000, maxiter=3000)
        c5_last = compute_c5(h_last)
        if c5_last < best_c5 - 1e-15:
            best_c5 = c5_last
            best_h = h_last.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)


def theta_morse_start(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    # Adjust to exactly n/2 ones
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones < target:
        zeros = np.where(h == 0.0)[0]
        extra = target - n_ones
        h[zeros[:extra]] = 1.0
    elif n_ones > target:
        ones = np.where(h == 1.0)[0]
        extra = n_ones - target
        h[ones[:extra]] = 0.0
    return project_feasible(h, n)
