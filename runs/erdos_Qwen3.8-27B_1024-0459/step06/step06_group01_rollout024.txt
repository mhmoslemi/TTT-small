def nesterov_subgrad(h, n, max_iters=50000):
    h = project_feasible(h, n).copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    y = h.copy()
    t_val = 1.0
    # Step size: use diminishing step
    # For minimizing f(x) = max_k corr_k(x) * dx over the feasible set
    # Lipschitz constant L ≈ 2 * dx (rough estimate)
    L = 2.0 * dx
    
    for it in range(max_iters):
        if remaining_time() < 3:
            break
        
        # Evaluate subgradient at y
        hy = project_feasible(y, n)
        corr_y = compute_corr(hy)
        max_idx = np.argmax(corr_y)
        
        off = max_idx - (n - 1)
        g = np.zeros(n)
        if off >= 0:
            np_ = n - off
            if np_ > 0:
                g[off:] += (1.0 - hy[:np_])
                g[:np_] -= hy[off:]
        else:
            a = -off
            np_ = n - a
            if np_ > 0:
                g[:np_] += (1.0 - hy[a:])
                g[a:] -= hy[:np_]
        g *= dx
        
        # Nesterov step
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        step = 2.0 / (t_new + 1.0)
        
        x_new = project_feasible(h - step * g / (L + 1e-10), n)
        
        alpha = (t_val - 1.0) / t_new
        y = x_new + alpha * (x_new - h)
        
        h = x_new
        t_val = t_new
        
        # Track best
        if (it + 1) % 10 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                best_h = h.copy()
    
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    v = np.zeros(n)  # momentum
    mu = 0.9  # momentum coefficient
    lr = 0.5
    
    for it in range(max_iters):
        if remaining_time() < 3:
            break
        
        # Pick top-k lags
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        
        # Average gradient over top-k
        g = np.zeros(n)
        for ki in idx_top:
            off = ki - (n - 1)
            if off >= 0:
                np_ = n - off
                if np_ > 0:
                    g[off:] += (1.0 - h[:np_])
                    g[:np_] -= h[off:]
            else:
                a = -off
                np_ = n - a
                if np_ > 0:
                    g[:np_] += (1.0 - h[a:])
                    g[a:] -= h[:np_]
        g *= dx / top_k
        
        # Momentum update
        v = mu * v - lr * g
        h_new = project_feasible(h + v, n)
        
        # Check improvement
        if (it + 1) % 5 == 0:
            c5 = compute_c5(h_new)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                best_h = h_new.copy()
            h = h_new
        else:
            h = h_new
        
        # Decay learning rate
        lr *= 0.99999
    
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    return best_h, best_c5

def topk_directed(h, n, top_k=5, max_iters=2000):
    """Directly target top-k correlations with a smarter step selection."""
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if remaining_time() < 3:
            break
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        c_max = corr[idx_top[0]]
        
        # Direction: reduce the max, increase the min of top-k
        g_dir = np.zeros(n)
        for pos, ki in enumerate(idx_top):
            if pos == 0:
                w_k = 1.0
            else:
                w_k = -0.5 / (top_k - 1)
            off = ki - (n - 1)
            if off >= 0:
                np_ = n - off
                if np_ > 0:
                    g_dir[off:] += w_k * (1.0 - h[:np_])
                    g_dir[:np_] -= w_k * h[off:]
            else:
                a = -off
                np_ = n - a
                if np_ > 0:
                    g_dir[:np_] += w_k * (1.0 - h[a:])
                    g_dir[a:] -= w_k * h[:np_]
        
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        d = g_dir / gn
        
        # Binary search for best step
        best_step = 0.0
        best_c5_step = best_c5
        lo, hi = 0.0, 0.5
        for _ in range(20):
            mid = (lo + hi) / 2
            ht = project_feasible(h - mid * d, n)
            cn = compute_c5(ht)
            if cn < best_c5_step:
                best_c5_step = cn
                best_step = mid
                lo = mid
            else:
                hi = mid
        # Also try positive direction
        lo, hi = 0.0, 0.5
        for _ in range(20):
            mid = (lo + hi) / 2
            ht = project_feasible(h + mid * d, n)
            cn = compute_c5(ht)
            if cn < best_c5_step:
                best_c5_step = cn
                best_step = -mid
                lo = mid
            else:
                hi = mid
        
        if best_c5_step < best_c5 - 1e-14:
            h = project_feasible(h - best_step * d, n)
            best_c5 = compute_c5(h)
        else:
            # Try smaller steps
            improved = False
            for step in [0.001, 0.002, 0.005, 0.01]:
                ht = project_feasible(h - step * d, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-14:
                    h = ht
                    best_c5 = cn
                    improved = True
                    break
                ht = project_feasible(h + step * d, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-14:
                    h = ht
                    best_c5 = cn
                    improved = True
                    break
            if not improved:
                break
    
    return h, best_c5

def binary_sa(h, n, time_limit, rng):
    """Binary simulated annealing with block moves."""
    h_bin = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        sort_idx = np.argsort(-h)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0
    
    start = time.time()
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    current_c5 = best_c5
    T = 1.0
    cool_rate = 0.9998
    no_improve = 0
    max_no_improve = 5000
    
    while time.time() - start < time_limit and remaining_time() > 10:
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        
        # Random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        new_c5 = compute_c5(h_bin)
        delta = new_c5 - current_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            current_c5 = new_c5
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = h_bin.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            no_improve += 1
        
        T *= cool_rate
        if no_improve > max_no_improve:
            break
    
    return best_h, best_c5

def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)

def continuous_sa(h, n, time_limit, rng):
    """Continuous simulated annealing with random perturbations."""
    h = project_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    current_c5 = best_c5
    T = 0.01
    cool_rate = 0.999
    no_improve = 0
    max_no_improve = 3000
    start = time.time()
    
    while time.time() - start < time_limit and remaining_time() > 10:
        # Perturb a random subset
        n_perturb = rng.randint(5, 20)
        idx = rng.choice(n, size=n_perturb, replace=False)
        old_vals = h[idx].copy()
        h[idx] = np.clip(h[idx] + rng.normal(0, 0.1, n_perturb), 0, 1)
        h = project_feasible(h, n)
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            current_c5 = new_c5
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[idx] = old_vals
            h = project_feasible(h, n)
            no_improve += 1
        
        T *= cool_rate
        if no_improve > max_no_improve:
            break
    
    return best_h, best_c5
