def thue_morse(n):
    """Generate Thue-Morse sequence of length n (padded if needed)."""
    tm = np.zeros(1)

        for k_idx in idx:
            off = k_idx - (n - 1)
            if off >= 0:
                np_ = n - off
                g[:np_] += (1.0 - h[off:off+np_])
                g[off:off+np_] -= h[:np_]
            else:
                ao = -off
                np_ = n - ao
                g[ao:] += (1.0 - h[:np_])
                g[:np_] -= h[ao:]
        
        g /= K
        g *= (2.0 / n)
        
        # Line search
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        
        gap = best_c5 * n / 2.0 - corr[idx[0]]
        alpha = max(0.01, gap / (gn * gn + 1e-10))
        alpha = min(alpha, 0.5)
        
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_try = project_feasible(h - mult * alpha * g / gn, n)
            c_try = compute_c5(h_try)
            if c_try < best_c5:
                h = h_try
                best_c5 = c_try
                best_h = h_try.copy()
                break
            elif mult == 0.05:
                h = h_try
    
    return best_h, best_c5

def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break

            h[interior] += diff / interior.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    s = np.clip(beta * (corr - m), -500, 500)
