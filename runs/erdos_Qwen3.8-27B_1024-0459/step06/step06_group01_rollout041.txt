def compute_gradient(x, n, beta, mu):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')  # length 2n-1
    m = np.max(corr)
    z = np.clip(beta * (corr - m), -50, 50)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S  # length 2n-1, w[k] for k=0,...,2n-2
    
    penalty_term = 2.0 * mu * (np.sum(x) - n/2.0)
    
    g = np.zeros(n)
    for j in range(n):
        # First sum: k from max(0, j-(n-1)) to min(2n-2, j), weight w[k] * (1-h[j-k])
        # j-k ranges from 0 to n-1
        k_min = max(0, j - (n-1))
        k_max = min(2*n-2, j)
        if k_min <= k_max:
            p = j - np.arange(k_min, k_max+1)  # p = j-k, ranges over valid values
            g[j] += np.dot(w[k_min:k_max+1], 1.0 - x[p])
        
        # Second sum: k from max(0, -j) to min(2n-2, n-1-j), weight w[k] * h[j+k]
        # j+k ranges from 0 to n-1
        k_min2 = max(0, -j)  # j+k = 0 => k = -j
        k_max2 = min(2*n-2, n-1-j)  # j+k = n-1 => k = n-1-j
        if k_min2 <= k_max2:
            q = j + np.arange(k_min2, k_max2+1)  # q = j+k
            g[j] -= np.dot(w[k_min2:k_max2+1], x[q])
        
        g[j] *= dx
        g[j] += penalty_term
    
    return g

            val -= np.dot(w[k_lo2:k_hi2+1], x[q])
        g[j] = val * dx + penalty_grad
    
    return f, g

        return f

        return g

def contrast_move(h, n, rng):
    """Increase or decrease contrast in a random block while preserving mean."""
    L = rng.randint(3, min(30, n//8))
    a = rng.randint(0, n - L)

    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def smooth_max_and_weights(corr, beta):
    m = np.max(corr)
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    sm = m + np.log(S) / beta
    return sm, w


def obj_and_grad(x, n, beta, mu):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    sm, w = smooth_max_and_weights(corr, beta)
    sum_dev = np.sum(x) - n / 2.0
    f = sm * dx + mu * sum_dev * sum_dev

    penalty_grad = 2.0 * mu * sum_dev
    g = np.zeros(n)

    for j in range(n):
        val = 0.0
        k_lo = max(0, j - (n - 1))
        k_hi = min(2 * n - 2, j)
        if k_lo <= k_hi:
            p = j - np.arange(k_lo, k_hi + 1)
            val += np.dot(w[k_lo:k_hi + 1], 1.0 - x[p])
        k_lo2 = max(0, -j)
        k_hi2 = min(2 * n - 2, n - 1 - j)
        if k_lo2 <= k_hi2:
            q = j + np.arange(k_lo2, k_hi2 + 1)
            val -= np.dot(w[k_lo2:k_hi2 + 1], x[q])
        g[j] = val * dx + penalty_grad

    return f, g


def lbfgsb_opt(h, n, beta, mu, maxiter=2000, time_limit=60):
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n

    def fun(x):
        f, _ = obj_and_grad(x, n, beta, mu)
        return f

    def grad(x):
        _, g = obj_and_grad(x, n, beta, mu)
        return g

    t_start = time.time()
    best_local = h0.copy()
    best_local_c5 = compute_c5(h0)
    try:
        res = minimize(fun, h0, jac=grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8})
        if time.time() - t_start > time_limit:
            return best_local
        result = project_feasible(res.x, n)
        c5_res = compute_c5(result)
        if c5_res < best_local_c5:
            best_local = result
    except Exception:
        pass
    return best_local


def topk_equalize(h, n, top_k=5, max_iters=500):
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining_time() < 3:
            break
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        c_max = corr[idx_top[0]]
        c_min_top = corr[idx_top[-1]]
        if c_max - c_min_top < 1e-10:
            break
        g_dir = np.zeros(n)
        cnt = len(idx_top)
        for pos, k in enumerate(idx_top):
            if pos == 0:
                w_k = 1.0
            else:
                w_k = -1.0 / (cnt - 1)
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_dir[off:] += w_k * (1.0 - h[:np_])
                g_dir[:np_] -= w_k * h[off:]
            else:
                a = -off
                np_ = n - a
                g_dir[:np_] += w_k * (1.0 - h[a:])
                g_dir[a:] -= w_k * h[:np_]
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        d = g_dir / gn
        improved = False
        steps = [0.002, 0.005, 0.01, 0.02, 0.001, 0.008, 0.03, 0.05, 0.08]
        for step in steps:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-13:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            for step in [0.005, 0.01, 0.02]:
                ht = project_feasible(h - step * d, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-13:
                    h = ht
                    best_c5 = cn
                    improved = True
                    break
            if not improved:
                break
    return h, best_c5


def block_swap_sa(h, n, time_limit, rng, initial_T=0.01):

    best_c5 = compute_c5(h)
    best_h = h.copy()

    T = initial_T
    cool_rate = 0.999
    no_improve = 0
    max_no_improve = 5000

        L = rng.randint(1, max_L)
        a = rng.randint(0, n - L)
        b = rng.randint(0, n - L)
        if abs(a - b) < L:
            continue

        h[a:a + L], h[b:b + L] = h[b:b + L], h[a:a + L]
        new_c5 = compute_c5(h)
        delta = new_c5 - curr_c5

        if delta < 0 or (T > 1e-12 and rng.rand() < np.exp(-delta / T)):
            curr_c5 = new_c5

                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[a:a + L], h[b:b + L] = h[b:b + L], h[a:a + L]

def contrast_move(h, n, rng):

    if cnt > 0:
        new_block[mask] += diff / cnt
        new_block = np.clip(new_block, 0, 1)
    h[a:a + L] = new_block
    return h

        # Final L-BFGS-B at new resolution
        if remaining_time() > 15:
            h_opt = lbfgsb_opt(h_opt, n_t, 500000, 100.0, maxiter=800, time_limit=30)
            c5_opt = compute_c5(h_opt)

        if c5_opt < best_c5 - 1e-15:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t
            h = h_opt.copy()

    # Phase 6: Try alternative periodic initializations at best resolution
    if remaining_time() > 80:
        periods = [4, 6, 8, 10, 12, 16, 20]
        for per in periods:
            if remaining_time() < 20:
                break
            per_eff = min(per, best_n)
            h_p = periodic_init(best_n, per_eff, 0.5)
            h_p_opt = lbfgsb_opt(h_p, best_n, 5000, 50.0, maxiter=500, time_limit=20)
            h_p_opt, c5_p = topk_equalize(h_p_opt, best_n, top_k=5, max_iters=200)
            if c5_p < best_c5 - 1e-15:
                best_c5 = c5_p
                best_h = h_p_opt.copy()
                h = h_p_opt.copy()

    # Phase 7: Contrast moves + L-BFGS-B polish
    if remaining_time() > 40:
        h_c = best_h.copy()
        for _ in range(300):
            if remaining_time() < 10:
                break
            h_c = contrast_move(h_c, best_n, rng)
            c5_c = compute_c5(h_c)
            if c5_c < best_c5 - 1e-15:
                best_c5 = c5_c
                best_h = h_c.copy()
            else:
                h_c = best_h.copy()
        if remaining_time() > 20:
            h_new = lbfgsb_opt(best_h, best_n, 1000000, 200.0, maxiter=1500, time_limit=40)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 8: Final block swap polish
    if remaining_time() > 30:
        t_limit = min(60, int(remaining_time() * 0.5))
        h_b, c5_b = block_swap_sa(best_h, best_n, t_limit, rng, initial_T=0.005)
        if c5_b < best_c5 - 1e-15:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 9: Final top-k equalization
    if remaining_time() > 15:
        h_eq2, c5_eq2 = topk_equalize(best_h, best_n, top_k=7, max_iters=300)
        if c5_eq2 < best_c5 - 1e-15:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
