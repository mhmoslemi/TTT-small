def grad_corr_k(h, k, n):
    """Gradient of corr[k] with respect to h.
    corr[k] = sum_{i} h[i] * (1-h[i+k]) for appropriate i depending on k's position in the full correlation.
    For mode='full', the output has length 2n-1.
    Index k in the output corresponds to shift (k - (n-1)).

      If s < 0: corr = sum_{i=0}^{n-1+s} h[i-s] * (1-h[i])

                h = project_feasible(h, n)
                c = compute_c5(h)
                if c < best_c - 1e-16:
                    best_c = c
                    best_v = cv

            h[i] = best_v
            h = project_feasible(h, n)
            c_now = compute_c5(h)
            if c_now < best_c5 - 1e-15:
                best_c5 = c_now
                improved = True

        if not improved:
            break
        order = rng_cd.permutation(n)

    return h, best_c5


def subgrad_refine(h, n, time_limit, rng):
    global _T0, _BUDGET
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    it = 0

    while time.time() - start < time_limit:
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        it += 1
        progress = min(1.0, it / 500.0)
        K = 30 if progress < 0.3 else (15 if progress < 0.6 else 8)
        beta_g = 50.0 if progress < 0.3 else (100.0 if progress < 0.6 else 200.0)

        corr = compute_corr(h)
        idx = np.argsort(-corr)[:K]
        c_top = corr[idx]
        weights = np.exp(beta_g * (c_top - c_top.max()))
        sw = weights.sum()
        if sw < 1e-30:
            weights = np.ones(K) / K
        else:
            weights /= sw

        g = np.zeros(n)
        for i in range(K):
            if weights[i] > 1e-12:
                g += weights[i] * grad_corr_k(h, idx[i], n)

        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break

        step = 0.03 / (1.0 + 0.05 * it)
        h_new = project_feasible(h - step * g, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            h = h_new
        else:
            found = False
            for factor in [0.5, 0.25, 0.1]:
                h_try = project_feasible(h - step * factor * g, n)
                c_try = compute_c5(h_try)
                if c_try < best_c5:
                    best_c5 = c_try
                    h = h_try
                    found = True
                    break
            if not found and it % 200 == 0:
                h = project_feasible(h, n)

    return h, best_c5


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
    except Exception:
        parent_n = 512
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0

    parent_c5 = compute_c5(parent_h)
    best_h = parent_h.copy()
    best_c5 = parent_c5
    best_n = parent_n

    # Interpolate to n=2048
    n = 2048
    h = interp_to(best_h, best_n, n)
    h = project_feasible(h, n)
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
        best_n = n

    # Phase 1: Subgradient to quickly improve (60s)
    if remaining_time() > 60:
        h, c5 = subgrad_refine(h, n, 60.0, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: L-BFGS-B with moderate beta (200s)
    if remaining_time() > 180:
        h = lbfgsb_opt(h, n, beta=500, penalty=5000, maxiter=3000)
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 3: L-BFGS-B with higher beta (200s)
    if remaining_time() > 180:
        h = lbfgsb_opt(h, n, beta=2000, penalty=10000, maxiter=3000)
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 4: L-BFGS-B with even higher beta (150s)
    if remaining_time() > 140:
        h = lbfgsb_opt(h, n, beta=8000, penalty=20000, maxiter=2500)
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 5: Coordinate descent (250s)
    if remaining_time() > 200:
        t_cd = min(220.0, remaining_time() * 0.5)
        h_cd, c5_cd = coord_descent(h, n, t_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h = h_cd.copy()

    # Phase 6: L-BFGS-B with very high beta (120s)
    if remaining_time() > 100:
        h = lbfgsb_opt(h, n, beta=30000, penalty=50000, maxiter=2000)
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 7: More coordinate descent (100s)
    if remaining_time() > 80:
        h_cd, c5_cd = coord_descent(h, n, min(80.0, remaining_time() * 0.5))
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h = h_cd.copy()

    # Phase 8: Subgradient refinement with small steps (80s)
    if remaining_time() > 60:
        h_sg, c5_sg = subgrad_refine(h, n, min(60.0, remaining_time() * 0.5), rng)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 9: SLSQP polish (60s)
    if remaining_time() > 50:
        h = slsqp_opt(h, n, beta=20000, maxiter=400)
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 10: Final coordinate descent (50s)
    if remaining_time() > 40:
        h_cd, c5_cd = coord_descent(best_h, best_n, min(40.0, remaining_time() * 0.5))
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            best_n = best_n

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
