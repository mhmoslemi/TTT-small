        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_corr = compute_corr(h)
        new_c5 = float(np.max(new_corr) * dx)
        delta = new_c5 - current_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            current_c5 = new_c5
            corr = new_corr
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
        T *= 0.9998
        if no_improve > max_no_improve:
            T = max(T, 0.1)
            no_improve = 0
    return best_h, best_c5


def multi_start_gradient(h, n, rng, time_limit):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    n_starts = 5
    for s in range(n_starts):
        if time.time() - start > time_limit * (s + 1) / n_starts or remaining_time() < 30:
            break
        if s == 0:
            h_s =best_h.copy()
        elif s == 1:
            h_s = (best_h > 0.5).astype(float)
            n_half = n // 2
            n_ones = int(np.sum(h_s))
            if n_ones != n_half:
                idx = np.argsort(-h_s)
                h_s = np.zeros(n)
                h_s[idx[:n_half]] = 1.0
            h_s = project_feasible(h_s, n)
        else:
            noise = rng.randn(n) * 0.05
            h_s = project_feasible(best_h + noise, n)
        for beta in [500, 5000, 50000, 200000]:
            if remaining_time() < 20:
                break
            h_s, c5_s = projected_gradient(h_s, n, beta, lr0=0.02, max_iters=3000, momentum=0.85)
            if c5_s < best_c5 - 1e-14:
                best_c5 = c5_s
                best_h = h_s.copy()
        h_s, c5_s = topk_equalize(h_s, n, top_k=6, max_iters=300)
        if c5_s < best_c5 - 1e-14:
            best_c5 = c5_s
            best_h = h_s.copy()
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _CUR_N, _SM_BETA
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    try:
        best_h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        best_h = np.zeros(640)
        best_h[:320] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Projected gradient with multiple betas
    h = best_h.copy()
    for beta in [200, 1000, 5000, 20000, 100000, 500000]:
        if remaining_time() < 30:
            break
        h_new, c5_new = projected_gradient(h, best_n, beta, lr0=0.03, max_iters=4000, momentum=0.85)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            h = best_h.copy()

    # Phase 2: Top-k equalization (aggressive)
    if remaining_time() > 30:
        h_eq, c5_eq = topk_equalize(h, best_n, top_k=8, max_iters=1000)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 3: SLSQP polish
    for beta in [50000, 200000, 1000000]:
        if remaining_time() < 25:
            break
        h_new = slsqp_opt(h, best_n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 4: More top-k after SLSQP
    if remaining_time() > 20:
        h_eq, c5_eq = topk_equalize(h, best_n, top_k=10, max_iters=500)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 5: Multi-start at parent resolution
    if remaining_time() > 120:
        h_ms, c5_ms = multi_start_gradient(best_h.copy(), best_n, rng, min(200, remaining_time() * 0.3))
        if c5_ms < best_c5 - 1e-14:
            best_c5 = c5_ms
            best_h = h_ms.copy()
            h = h_ms.copy()

    # Phase 6: Try different resolutions
    resolutions = [512, 768, 1024, 576, 896]
    resolutions = [r for r in resolutions if r % 2 == 0 and 256 <= r <= 2048]
    resolutions = sorted(resolutions, key=lambda x: abs(x - best_n))

    for n_t in resolutions[:3]:
        if remaining_time() < 60:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        c5_int = compute_c5(h_feas)

        h_opt = h_feas
        for beta in [2000, 20000, 100000, 500000]:
            if remaining_time() < 20:
                break
            h_opt, c5_opt = projected_gradient(h_opt, n_t, beta, lr0=0.03, max_iters=3000, momentum=0.85)
            if c5_opt < c5_int - 1e-14:
                c5_int = c5_opt
        h_opt, c5_opt = topk_equalize(h_opt, n_t, top_k=8, max_iters=400)
        if remaining_time() > 20:
            h_opt = slsqp_opt(h_opt, n_t, 200000, maxiter=500)
            c5_opt = compute_c5(h_opt)
        if remaining_time() > 10:
            h_opt, c5_opt = topk_equalize(h_opt, n_t, top_k=8, max_iters=300)

        if c5_opt < best_c5 - 1e-14:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t

    # Phase 7: Binary SA at best resolution
    if remaining_time() > 100:
        h_bin = (best_h > 0.5).astype(float)
        n_half = best_n // 2
        n_ones = int(np.sum(h_bin))
        if n_ones != n_half:
            sort_idx = np.argsort(-h_bin)
            h_bin = np.zeros(best_n)
            h_bin[sort_idx[:n_half]] = 1.0
        sa_time = min(120, max(20, int(remaining_time() * 0.15)))
        h_b, c5_b = binary_sa_fast(h_bin, best_n, sa_time, rng)

        # Refine binary solution continuously
        h_b_c = project_feasible(h_b, best_n)
        for beta in [5000, 50000, 500000]:
            if remaining_time() < 15:
                break
            h_b_c, c5_bc = projected_gradient(h_b_c, best_n, beta, lr0=0.02, max_iters=2000, momentum=0.85)
            if c5_bc < best_c5 - 1e-14:
                best_c5 = c5_bc
                best_h = h_b_c.copy()
        if best_n == len(h_b_c):
            h_b_c, c5_bc = topk_equalize(h_b_c, best_n, top_k=8, max_iters=300)
            if c5_bc < best_c5 - 1e-14:
                best_c5 = c5_bc
                best_h = h_b_c.copy()
        if c5_b < best_c5 - 1e-14:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 8: Final SLSQP + top-k polish
    if remaining_time() > 20:
        h_final = slsqp_opt(best_h, best_n, 5000000, maxiter=400)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-14:
            best_c5 = c5_final
            best_h = h_final.copy()

    if remaining_time() > 10:
        h_eq2, c5_eq2 = topk_equalize(best_h, best_n, top_k=12, max_iters=400)
        if c5_eq2 < best_c5 - 1e-14:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()

    # Final feasibility and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
