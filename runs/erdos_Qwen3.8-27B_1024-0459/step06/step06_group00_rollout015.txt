import time

_CUR_N = 0
_SM_BETA = 1000.0

    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr_array(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-16:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def _sm_obj(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr_array(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -100.0, 100.0)
    ez = np.exp(z)
    return (m + np.log(np.sum(ez)) / beta) * dx

def _sm_grad(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr_array(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -100.0, 100.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    return g

def _eq_fun(x):
    return np.sum(x) - _CUR_N / 2.0

def _eq_jac(x):
    return np.ones(_CUR_N)

def slsqp_opt(h, n, beta, maxiter=2000):

    _CUR_N = n
    _SM_BETA = beta
    h0 = project_feasible(h, n)

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        if res.fun < _sm_obj(h0) - 1e-16:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0

def binary_swap_delta(h, i, j, n):
    """Compute change in C5 when swapping h[i] (1->0) and h[j] (0->1)."""
    # For the full correlation, changing h[i] affects corr[k] for all k
    # where the term involves position i in either factor.
    # corr[k] = sum_p h[p] * (1-h[p+k-(n-1)])
    # Change in h[i] from 1 to 0:
    #   - Term where p=i: -(1-h[i+k-(n-1)]) for valid k
    #   - Term where p+k-(n-1)=i, i.e., p=i-k+(n-1): changes from h[p]*(1-1)=0 to h[p]*(1-0)=h[p]
    # Change in h[j] from 0 to 1:
    #   + Term where p=j: +(1-h[j+k-(n-1)]) for valid k
    #   - Term where p+k-(n-1)=j: changes from h[p]*(1-0)=h[p] to h[p]*(1-1)=0, so change is -h[p]
    
    # This is O(n) to compute exactly. But for simplicity and correctness,
    # let's just recompute the full correlation. For n=640-1280 it's fast enough.
    # Actually for binary swaps in a loop, O(n²) per swap is too slow.
    # Let me compute the delta in O(n).
    
    delta_corr = np.zeros(2 * n - 1)
    
    # Effect of changing h[i] from 1 to 0:
    # corr[k] has term h[i]*(1-h[i+k-(n-1)]) when i+k-(n-1) is valid
    # and term h[p]*(1-h[i]) when p+k-(n-1)=i, i.e., p = i-k+(n-1)
    
    # Term 1: p = i, so k must satisfy 0 <= i + k - (n-1) < n, i.e., n-1-i <= k <= n-1+i
    k_min1 = max(0, n - 1 - i)
    k_max1 = min(2*n-2, n - 1 + i)
    for k in range(k_min1, k_max1 + 1):
        j2 = i + k - (n - 1)  # valid index
        delta_corr[k] -= (1.0 - h[j2])  # loss from p=i term going away
    
    # Term 2: p + k - (n-1) = i, so p = i - k + (n-1)
    # p must be in [0, n-1], k in [0, 2n-2]
    k_min2 = max(0, n - 1 - i)  # p = i - k + (n-1) >= 0 => k <= i + (n-1)
    k_max2 = min(2*n-2, i + n - 1)
    for k in range(k_min2, k_max2 + 1):
        p = i - k + (n - 1)  # valid index
        # Before: h[p]*(1-h[i]) = h[p]*0 = 0
        # After: h[p]*(1-0) = h[p]
        delta_corr[k] += h[p]
    
    # Effect of changing h[j] from 0 to 1:
    # Term 1: p = j
    k_min3 = max(0, n - 1 - j)
    k_max3 = min(2*n-2, n - 1 + j)
    for k in range(k_min3, k_max3 + 1):
        j2 = j + k - (n - 1)
        delta_corr[k] += (1.0 - h[j2])
    
    # Term 2: p + k - (n-1) = j, p = j - k + (n-1)
    k_min4 = max(0, n - 1 - j)
    k_max4 = min(2*n-2, j + n - 1)
    for k in range(k_min4, k_max4 + 1):
        p = j - k + (n - 1)
        # Before: h[p]*(1-h[j]) = h[p]*1 = h[p]
        # After: h[p]*(1-1) = 0
        delta_corr[k] -= h[p]
    
    return delta_corr

def binary_swap_optimize(h_bin, n, time_limit=120, rng=None, max_rounds=50000):
    """Optimize binary h using 2-swap with exact C5 delta computation."""
    if rng is None:
        rng = np.random.RandomState(42)
    
    h = h_bin.astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        sort_idx = np.argsort(-h)
        h = np.zeros(n)
        h[sort_idx[:n_half]] = 1.0
    
    corr = compute_corr_array(h)
    best_c5 = float(np.max(corr) * (2.0 / n))
    curr_c5 = best_c5
    best_h = h.copy()
    
    ones = np.where(h == 1.0)[0].tolist()
    zeros = np.where(h == 0.0)[0].tolist()
    
    start = time.time()
    no_improve = 0
    T_curr = 0.02  # tentative "temperature" for random acceptance
    dx = 2.0 / n
    
    for it in range(max_rounds):
        if time.time() - start > time_limit:
            break
        if remaining_time() < 5:
            break
        
        # Choose a swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Compute delta in corr
        delta = binary_swap_delta(h, i, j, n)
        
        # New corr would be corr + delta
        new_corr_max = np.max(corr + delta)
        new_c5 = new_corr_max * dx
        
        if new_c5 < curr_c5 - 1e-15:
            # Accept
            h[i] = 0.0
            h[j] = 1.0
            corr += delta
            curr_c5 = new_c5
            ones[ones.index(i)] = j
            zeros[zeros.index(j)] = i
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        elif T_curr > 0 and rng.random() < np.exp(-(new_c5 - curr_c5) / T_curr):
            # Accept with probability (simulated annealing)
            h[i] = 0.0
            h[j] = 1.0
            corr += delta
            curr_c5 = new_c5
            ones[ones.index(i)] = j
            zeros[zeros.index(j)] = i
            no_improve += 1
        else:
            no_improve += 1
        
        # Cooling
        if no_improve > 500:
            T_curr *= 0.95
            no_improve = 0
        if T_curr < 1e-6:
            T_curr = 0.0
        
        if no_improve > 50000:
            break
    
    return best_h, best_c5

        if remaining_time() < 5:
            break
        gsz = group_sizes[rng.randint(len(group_sizes))]
        ones_list = list(np.where(h == 1.0)[0])
        zeros_list = list(np.where(h == 0.0)[0])
        if len(ones_list) < gsz or len(zeros_list) < gsz:
            break
        swap_from = list(rng.choice(ones_list, gsz, replace=False))
        swap_to = list(rng.choice(zeros_list, gsz, replace=False))
        h_trial = h.copy()
        for s in swap_from:
            h_trial[s] = 0.0
        for s in swap_to:
            h_trial[s] = 1.0
        c5_new = compute_c5(h_trial)
        if c5_new < best_c5 - 1e-16:
            h = h_trial
            best_c5 = c5_new
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 3000:
            break
    return best_h, best_c5


def subgradient_zigzag(h, n, max_iters=5000, time_limit=80):
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    k_curr = None
    k_prev = None
    stale_count = 0
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if remaining_time() < 5:
            break
        corr = compute_corr_array(h)
        k_curr = int(np.argmax(corr))
        if k_prev is not None and k_curr != k_prev:
            g = 0.5 * (grad_single_shift(h, k_curr, n) + grad_single_shift(h, k_prev, n))
            k_prev = k_curr
        else:
            g = grad_single_shift(h, k_curr, n)
            if k_prev is None:
                k_prev = k_curr
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-14:
            g = g / g_norm
        improved = False
        base_step = max(0.001, 0.02 * (1.0 - 0.5 * it / max_iters))
        for step in [base_step * 0.2, base_step * 0.5, base_step, base_step * 2.0,
                     base_step * 5.0, 0.001, 0.005, 0.01, 0.02, 0.05, 0.1]:
            ht = project_feasible(h - step * g, n)
            c5_new = compute_c5(ht)
            if c5_new < best_c5 - 1e-16:
                h = ht
                best_c5 = c5_new
                best_h = h.copy()
                improved = True
                stale_count = 0
                break
        if not improved:
            stale_count += 1
            if stale_count > 300:
                break
            if stale_count % 80 == 0:
                rng = np.random.RandomState(stale_count)
                pert = rng.randn(n) * 0.005
                ht = project_feasible(h + pert, n)
                c5_new = compute_c5(ht)
                if c5_new < best_c5:
                    h = ht
                    best_c5 = c5_new
                    best_h = h.copy()
                    stale_count = 0
    return best_h, best_c5


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def theta_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)


def coordinate_refine(h, n, max_sweeps=30):
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        improved_sweep = False
        for i in range(n):
            old = h[i]
            if 0.05 < old < 0.95:
                cands = [old + 0.05, old - 0.05, old + 0.15, old - 0.15, 0.0, 1.0]
            else:
                cands = [0.0, 1.0]
            cands = [c for c in cands if 0.0 <= c <= 1.0 and abs(c - old) > 1e-7]
            best_cand_val = old
            best_cand_c5 = best_c5
            for c in cands:
                h[i] = c
                h = project_feasible(h, n)
                c5_v = compute_c5(h)
                if c5_v < best_cand_c5 - 1e-14:
                    best_cand_c5 = c5_v
                    best_cand_val = c
                    improved_sweep = True
            h[i] = best_cand_val
            h = project_feasible(h, n)
        if not improved_sweep:
            break
    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _CUR_N
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = np.zeros(640)
        best_h[:320] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Optimize parent at its native resolution
    h_work = best_h.copy()
    n_work = best_n
    for beta in [1000, 10000, 100000, 1000000]:
        if remaining_time() < 20:
            break
        h_new = slsqp_opt(h_work, n_work, beta, maxiter=2000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-16:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_work = h_new.copy()

    if remaining_time() > 20:
        h_eq, c5_eq = top_k_equalize(h_work, n_work, top_k=12, max_iters=800, time_limit=40)
        if c5_eq < best_c5 - 1e-16:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h_work = h_eq.copy()

    # Phase 2: Binary swap optimization at parent resolution
    if remaining_time() > 60:
        h_bin = (best_h > 0.5).astype(float)
        h_b, c5_b = binary_optimize(h_bin, n_work, time_limit=min(100, int(remaining_time() * 0.25)), rng=rng)
        if c5_b < best_c5 - 1e-16:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 3: Upscale to higher resolution and optimize
    high_n = min(2048, max(1280, 2 * n_work))
    if remaining_time() > 120:
        h_high = interp_to(best_h, best_n, high_n)
        h_high = project_feasible(h_high, high_n)
        c5_high = compute_c5(h_high)

        if c5_high > best_c5:
            h_high = project_feasible(interp_to(best_h, best_n, high_n), high_n)

        # Binary optimization at high resolution
        if remaining_time() > 80:
            h_bin_high = (h_high > 0.5).astype(float)
            h_bh, c5_bh = binary_optimize(h_bin_high, high_n, time_limit=min(120, int(remaining_time() * 0.3)),
                                          rng=np.random.RandomState(seed + 1))
            if c5_bh < best_c5 - 1e-16:
                best_c5 = c5_bh
                best_h = h_bh.copy()
                best_n = high_n
                h_high = h_bh.copy()

        # Multi-swap at high resolution
        if remaining_time() > 40:
            h_bin_h = (h_high > 0.5).astype(float)
            h_mh, c5_mh = binary_multi_swap(h_bin_h, high_n, time_limit=min(50, int(remaining_time() * 0.2)),
                                            rng=np.random.RandomState(seed + 2))
            if c5_mh < best_c5 - 1e-16:
                best_c5 = c5_mh
                best_h = h_mh.copy()
                best_n = high_n
                h_high = h_mh.copy()

        # SLSQP at high resolution
        for beta in [10000, 100000, 1000000, 5000000]:
            if remaining_time() < 20:
                break
            h_opt = slsqp_opt(h_high, high_n, beta, maxiter=2000)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5 - 1e-16:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = high_n
                h_high = h_opt.copy()

        # Top-k at high resolution
        if remaining_time() > 30:
            h_eq_h, c5_eq_h = top_k_equalize(h_high, high_n, top_k=15, max_iters=800,
                                              time_limit=min(50, int(remaining_time() * 0.2)))
            if c5_eq_h < best_c5 - 1e-16:
                best_c5 = c5_eq_h
                best_h = h_eq_h.copy()
                best_n = high_n
                h_high = h_eq_h.copy()

        # Subgradient at high resolution
        if remaining_time() > 30:
            h_sg, c5_sg = subgradient_zigzag(h_high, high_n, max_iters=5000,
                                             time_limit=min(50, int(remaining_time() * 0.2)))
            if c5_sg < best_c5 - 1e-16:
                best_c5 = c5_sg
                best_h = h_sg.copy()
                best_n = high_n
                h_high = h_sg.copy()

    # Phase 4: Back to original resolution for final polish if beneficial
    if remaining_time() > 30 and best_n != n_work:
        h_back = interp_to(best_h, best_n, n_work)
        h_back = project_feasible(h_back, n_work)
        c5_back = compute_c5(h_back)
        if c5_back < best_c5 - 1e-6:
            # Try optimizing at original resolution too
            for beta in [100000, 1000000]:
                if remaining_time() < 15:
                    break
                h_back = slsqp_opt(h_back, n_work, beta, maxiter=1500)
            c5_back = compute_c5(h_back)
            if c5_back < best_c5 - 1e-16:
                # Keep both, return the better one at the end
                pass

    # Phase 5: Alternative starting points at best resolution
    n_best = best_n
    if remaining_time() > 60:
        # Thiessen/Morse sequence start
        h_morse = theta_morse(n_best)
        c5_morse = compute_c5(h_morse)
        if c5_morse < best_c5 + 0.05:
            h_morse = slsqp_opt(h_morse, n_best, 100000, maxiter=1000)
            c5_morse = compute_c5(h_morse)
            h_morse, c5_morse = top_k_equalize(h_morse, n_best, top_k=10, max_iters=500, time_limit=30)
            if c5_morse < best_c5 - 1e-16:
                best_c5 = c5_morse
                best_h = h_morse.copy()

        # Random balanced binary
        h_rand = np.zeros(n_best)
        h_rand[rng.choice(n_best, n_best // 2, replace=False)] = 1.0
        h_rand = project_feasible(h_rand, n_best)
        if remaining_time() > 30:
            h_rand = slsqp_opt(h_rand, n_best, 50000, maxiter=800)
            c5_rand = compute_c5(h_rand)
            if c5_rand < best_c5 - 1e-16:
                best_c5 = c5_rand
                best_h = h_rand.copy()

    # Phase 6: Final intensive SLSQP with progressive beta
    n_fin = best_n
    h_fin = best_h.copy()
    for beta in [50000, 200000, 1000000, 5000000, 20000000]:
        if remaining_time() < 10:
            break
        h_fin = slsqp_opt(h_fin, n_fin, beta, maxiter=2000)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5 - 1e-16:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Phase 7: Final top-k and subgradient
    if remaining_time() > 15:
        h_fin, c5_fin = top_k_equalize(best_h, n_fin, top_k=20, max_iters=600, time_limit=30)
        if c5_fin < best_c5 - 1e-16:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    if remaining_time() > 20:
        h_fin, c5_fin = subgradient_zigzag(best_h, n_fin, max_iters=4000,
                                            time_limit=min(50, int(remaining_time() * 0.5)))
        if c5_fin < best_c5 - 1e-16:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Phase 8: Binary polish
    if remaining_time() > 30:
        h_bin_fin = (best_h > 0.5).astype(float)
        h_bf, c5_bf = binary_optimize(h_bin_fin, n_fin, time_limit=min(40, int(remaining_time() * 0.3)),
                                       rng=np.random.RandomState(seed + 100))
        if c5_bf < best_c5 - 1e-16:
            best_c5 = c5_bf
            best_h = h_bf.copy()
            # Refine fractionally
            h_rf = slsqp_opt(h_bf, n_fin, 1000000, maxiter=1000)
            c5_rf = compute_c5(h_rf)
            if c5_rf < best_c5 - 1e-16:
                best_c5 = c5_rf
                best_h = h_rf.copy()

    # Coordinate refinement
    if remaining_time() > 15:
        h_cr, c5_cr = coordinate_refine(best_h, n_fin, max_sweeps=15)
        if c5_cr < best_c5 - 1e-16:
            best_c5 = c5_cr
            best_h = h_cr.copy()

    # Final SLSQP polish
    if remaining_time() > 8:
        h_final = slsqp_opt(best_h, n_fin, 10000000, maxiter=1500)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-16:
            best_c5 = c5_final
            best_h = h_final.copy()

    best_h = project_feasible(best_h, n_fin)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, n_fin)
