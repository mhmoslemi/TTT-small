For j in [a+off, b+off]: -= h[j-off] (if off > 0)

For j in [a, b] ∩ [a+off, b+off]: both contributions apply

                        f_best = c2
                        v_best = m2
            h[j] = v_best

        h = project_feasible(h, n)
        curr = compute_c5(h)
        if curr < best_c5 - 1e-14:
            best_c5 = curr
            best_h = h.copy()
            improved = True
        elif not improved and curr >= best_c5 - 1e-12:
            # Check if we should stop
            if it > 200:
                break

    return best_h, best_c5

def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)

def random_perturb(h, n, sigma, rng):
    h_pert = h + sigma * rng.randn(n)
    return project_feasible(h_pert, n)

def multi_beta_nesterov(h, n, time_limit):
    """Run Nesterov with increasing beta."""
    h = project_feasible(h, n).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    total_time = time_limit * 0.85
    betas = [50, 200, 500, 2000, 5000, 20000]
    per_beta = total_time / len(betas)

    for beta in betas:
        if remaining_time() < 30:
            break
        tl = min(per_beta, remaining_time() - 20)
        if tl < 5:
            break
        h_new, c5_new = nesterov_opt(h, n, beta, tl)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new  # Chain regardless

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    # Get initial solution
    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = np.zeros(640)
        best_h[:320] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Nesterov at parent resolution
    if remaining_time() > 100:
        h1, c51 = multi_beta_nesterov(best_h.copy(), best_n, min(300, remaining_time() * 0.3))
        if c51 < best_c5 - 1e-14:
            best_c5 = c51
            best_h = h1.copy()

    # Phase 2: Top-k directional descent
    if remaining_time() > 60:
        h2, c52 = topk_dir_descent(best_h.copy(), best_n, top_k=5, time_limit=min(150, remaining_time() * 0.2))
        if c52 < best_c5 - 1e-14:
            best_c5 = c52
            best_h = h2.copy()

    # Phase 3: Random perturbations + re-optimize
    if remaining_time() > 120:
        for sigma in [0.05, 0.1, 0.03, 0.15]:
            if remaining_time() < 60:
                break
            h_pert = random_perturb(best_h, best_n, sigma, rng)
            h_opt, c5_opt = multi_beta_nesterov(h_pert, best_n, min(100, remaining_time() * 0.2))
            if c5_opt < best_c5 - 1e-14:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            h_opt2, c5_opt2 = topk_dir_descent(h_opt, best_n, top_k=5, time_limit=30)
            if c5_opt2 < best_c5 - 1e-14:
                best_c5 = c5_opt2
                best_h = h_opt2.copy()

    # Phase 4: Higher resolution
    for n_t in [960, 1280, 800]:
        if remaining_time() < 120:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        c5_int = compute_c5(h_feas)

        h_opt, c5_opt = multi_beta_nesterov(h_feas, n_t, min(200, remaining_time() * 0.3))
        if remaining_time() > 60:
            h_opt, c5_opt = topk_dir_descent(h_opt, n_t, top_k=7, time_limit=min(100, remaining_time() * 0.25))

        if c5_opt < best_c5 - 1e-14:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t
            h = h_opt.copy()

    # Phase 5: More Nesterov at current best resolution
    if remaining_time() > 80:
        h5, c55 = nesterov_opt(best_h.copy(), best_n, 10000, min(150, remaining_time() * 0.4))
        if c55 < best_c5 - 1e-14:
            best_c5 = c55
            best_h = h5.copy()
        h5b, c55b = nesterov_opt(best_h.copy(), best_n, 50000, min(100, remaining_time() * 0.3))
        if c55b < best_c5 - 1e-14:
            best_c5 = c55b
            best_h = h5b.copy()

    # Phase 6: Coordinate descent on subsets
    if remaining_time() > 100:
        h6, c56 = coord_descent_fast(best_h.copy(), best_n, time_limit=min(200, remaining_time() * 0.4), subset_size=80)
        if c56 < best_c5 - 1e-14:
            best_c5 = c56
            best_h = h6.copy()

    # Phase 7: Final Nesterov polish with very high beta
    if remaining_time() > 50:
        h7, c57 = nesterov_opt(best_h.copy(), best_n, 100000, min(100, remaining_time() * 0.5))
        if c57 < best_c5 - 1e-14:
            best_c5 = c57
            best_h = h7.copy()

    # Phase 8: Final top-k
    if remaining_time() > 30:
        h8, c58 = topk_dir_descent(best_h.copy(), best_n, top_k=7, time_limit=min(60, remaining_time() * 0.6))
        if c58 < best_c5 - 1e-14:
            best_c5 = c58
            best_h = h8.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Evaluate with the official evaluator if available
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
