    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        if interior.any():
            h[interior] += diff / interior.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_val(h, beta):
    n = len(h)
    corr = compute_corr(h)
    m = corr.max()
    s = np.clip(beta * (corr - m), -500.0, 500.0)
    return (m + np.log(np.exp(s).sum()) / beta) * (2.0 / n)


def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = corr.max()
    s = np.clip(beta * (corr - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx] * dx
        if wk < 1e-15:
            continue
        off = k_idx - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - h[ao:])
            g[ao:] -= wk * h[:np_]
    return g


def _penalty_obj(x, beta, lam):
    n = len(x)
    return smooth_max_val(x, beta) + lam * (x.sum() - n / 2.0) ** 2


def _penalty_grad(x, beta, lam):
    n = len(x)
    g = smooth_max_grad(x, beta)
    scale = 2.0 * lam * (x.sum() - n / 2.0)
    g += scale
    return g


def _eq_fun(x, n):
    return x.sum() - n / 2.0


def _eq_jac(x, n):
    return np.ones(n)


def lbfgsb_opt(h, beta, time_limit):
    n = len(h)
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    lam = 1.0
    for _ in range(30):
        if time.time() - start > time_limit or _BUDGET - (time.time() - _T0) < 10:
            break
        bnds = Bounds(np.zeros(n), np.ones(n))
        try:
            res = minimize(_penalty_obj, h, args=(beta, lam),
                           jac=_penalty_grad, method='L-BFGS-B',
                           bounds=bnds,
                           options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-9})
            h = project_feasible(res.x, n)
            c = compute_c5(h)
            if c < best_c5:
                best_c5 = c
                best_h = h.copy()
            viol = abs(h.sum() - n / 2.0)
            if viol < 1e-12:
                break
            lam = min(lam * 5.0, 1e12)
        except Exception:
            break
    return best_h, best_c5

                best_h = h2.copy()
    except Exception:
        pass
    return best_h, best_c5

def make_binary(h, n):
    h_bin = (h > 0.5).astype(float)
    target = n // 2
    ones = int(np.sum(h_bin == 1))
    if ones > target:
        idx = np.where(h_bin == 1)[0]
        np.random.shuffle(idx)
        for i in range(ones - target):
            h_bin[idx[i]] = 0
    elif ones < target:
        idx = np.where(h_bin == 0)[0]
        np.random.shuffle(idx)
        for i in range(target - ones):
            h_bin[idx[i]] = 1
    return h_bin


def binary_sa(h_start, n, time_limit, temp_scale=0.15, decay=0.9998):
    start = time.time()
    h_bin = make_binary(h_start, n)
    corr = compute_corr(h_bin)
    cur_max = float(corr.max())
    best_max = cur_max
    best_h = h_bin.copy()
    T0 = max(cur_max * temp_scale, 5.0)
    T = T0
    no_improve = 0
    patience = 3000
    it = 0
    while time.time() - start < time_limit and _BUDGET - (time.time() - _T0) > 10:
        it += 1
        ones_idx = np.where(h_bin == 1)[0]
        zeros_idx = np.where(h_bin == 0)[0]
        p = ones_idx[np.random.randint(len(ones_idx))]
        q = zeros_idx[np.random.randint(len(zeros_idx))]
        h_try = h_bin.copy()
        h_try[p] = 0
        h_try[q] = 1
        c_try = float(compute_corr(h_try).max())
        delta = c_try - cur_max
        if delta <= 0:
            h_bin = h_try
            cur_max = c_try
            if c_try < best_max:
                best_max = c_try
                best_h = h_bin.copy()
                no_improve = 0
            else:
                no_improve += 1
        elif np.random.random() < np.exp(-delta / max(T, 1e-10)):
            h_bin = h_try
            cur_max = c_try
            no_improve += 1
        else:
            no_improve += 1
        T = max(T * decay, 0.5)
        if no_improve > patience:
            h_bin = make_binary(h_start + np.random.randn(n) * 0.3, n)
            cur_max = float(compute_corr(h_bin).max())
            T = T0 * 0.5
            no_improve = 0
            patience = max(1000, patience // 2)
    return best_h, best_max * (2.0 / n)
