import numpy as np

def compute_c5(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def _sm_objective(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = beta * (corr - m)
    z = np.clip(z, -50.0, 50.0)
    ez = np.exp(z)
    return (m + np.log(np.sum(ez)) / beta) * dx


def _sm_gradient(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = beta * (corr - m)
    z = np.clip(z, -50.0, 50.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])

def binary_sa_swap(h, n, time_limit, rng, top_k=3):
    """Binary simulated annealing with swap moves (preserves sum)."""
    h = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        sort_idx = np.argsort(-h)
        h = np.zeros(n)
        h[sort_idx[:n_half]] = 1.0
    
    start = time.time()
    corr = compute_corr(h)
    best_c5 = float(np.max(corr) * (2.0 / n))
    best_h = h.copy()
    current_c5 = best_c5
    T = 0.5
    cool_rate = 0.9997
    no_improve = 0
    max_no_improve = 5000
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    while time.time() - start < time_limit and remaining_time() > 15:
        if len(ones) < 1 or len(zeros) < 1:
            break
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Swap: h[i] 1->0, h[j] 0->1
        h[i] = 0.0
        h[j] = 1.0
        corr_new = compute_corr(h)
        new_c5 = float(np.max(corr_new) * (2.0 / n))
        delta = new_c5 - current_c5
        
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-12)):
            current_c5 = new_c5
            corr = corr_new
            # Update ones/zeros
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
        
        T *= cool_rate
        if no_improve > max_no_improve:
            # Reheat
            T = min(T * 10.0, 0.5)
            no_improve = 0
    
    return best_h, best_c5


def subgradient_momentum(h, n, max_iters=3000, top_k=3, mu=0.7):
    """Subgradient method with momentum for minimax."""
    h = project_feasible(h, n).copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    v = np.zeros(n)
    
    for it in range(max_iters):
        if remaining_time() < 5:
            break
        
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        
        # Weighted subgradient (emphasize max)
        g = np.zeros(n)
        weights = np.ones(top_k)
        weights[0] = 2.0
        weights /= weights.sum()
        
        for pos, ki in enumerate(idx_top):
            off = ki - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:] += weights[pos] * (1.0 - h[:np_])
                g[:np_] -= weights[pos] * h[off:]
            else:
                a = -off
                np_ = n - a
                g[:np_] += weights[pos] * (1.0 - h[a:])
                g[a:] -= weights[pos] * h[:np_]
        
        g *= dx
        
        # Momentum
        v = mu * v - g
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break
        
        # Step with backtracking
        lr = 0.5 / (1.0 + 0.005 * it)
        improved = False
        for trial in range(20):
            step = lr * v_norm
            d = -v / v_norm
            ht = project_feasible(h + step * d, n)

                best_c5 = cn
                best_h = h.copy()
                improved = True

            if lr < 1e-10:
                break

            pert = np.random.randn(n) * 0.005

def interp_to(src, src_n, dst_n):

    return np.interp(xn, xp, src)

            h_b_cont, c5_bc = fast_coord_descent(h_b_cont, best_n, max_sweeps=100)
            if c5_bc < best_c5 - 1e-14:
                best_c5 = c5_bc
                best_h = h_b_cont.copy()
                h = h_b_cont.copy()
            h_b_cont, c5_bc2 = topk_equalize_line_search(h_b_cont, best_n, top_k=5, max_iters=300)
            if c5_bc2 < best_c5 - 1e-14:
                best_c5 = c5_bc2
                best_h = h_b_cont.copy()
                h = h_b_cont.copy()
            
            if c5_b < best_c5 - 1e-14:
                best_c5 = c5_b
                best_h = h_b.copy()

    # Phase 7: Multi-alternation (cycle through methods)
    if remaining_time() > 100:
        alt_time = min(200, int(remaining_time() * 0.5))
        h_alt, c5_alt = multi_alt_optimization(best_h.copy(), best_n, alt_time)
        if c5_alt < best_c5 - 1e-14:
            best_c5 = c5_alt
            best_h = h_alt.copy()

    # Phase 8: Try a different resolution
    if remaining_time() > 100:
        n_t = best_n + 128  # try slightly higher resolution
        if n_t % 2 == 0 and n_t <= 1024:
            h_interp = interp_to(best_h, best_n, n_t)
            h_feas = project_feasible(h_interp, n_t)
            h_t, c5_t = fast_coord_descent(h_feas, n_t, max_sweeps=80)
            h_t, c5_t = topk_equalize_line_search(h_t, n_t, top_k=5, max_iters=300)
            if c5_t < best_c5 - 1e-14:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = n_t

    # Phase 9: Final SLSQP polish
    if remaining_time() > 20:
        h_final = slsqp_opt(best_h, best_n, 500000, maxiter=400)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-14:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Phase 10: Final coordinate descent polish
    if remaining_time() > 15:
        h_final2, c5_final2 = fast_coord_descent(best_h, best_n, max_sweeps=50)
        if c5_final2 < best_c5 - 1e-14:
            best_c5 = c5_final2
            best_h = h_final2.copy()

    # Final top-k equalize
    if remaining_time() > 10:
        h_eq2, c5_eq2 = topk_equalize_line_search(best_h, best_n, top_k=7, max_iters=200)
        if c5_eq2 < best_c5 - 1e-14:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
