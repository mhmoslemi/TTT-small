One more thought: the "interaction" issue. When I compute base = corr - dcorr[i], this is the correlation if h[i] were 0 (approximately, ignoring the d=0 case). Then adding dcorr[j] gives the approximate new correlation if h[j] were 1. But if |i-j| is small, the terms interact.

For the PURPOSE OF RANKING (which swap is best), the approximate formula should be sufficient. We then verify by actually computing the new correlation. If the actual result is worse than expected (due to interaction), we reject that swap and try the next best.

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        at_low = h <= 1e-12
        at_high = h >= 1.0 - 1e-12
        mid = ~(at_low | at_high)
        cnt = int(np.count_nonzero(mid))
        if cnt > 0:
            h[mid] += diff / cnt
        else:
            if diff > 0:
                h[:] += diff / n
            else:
                h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n, rng):
    """Convert to binary with exactly n//2 ones."""
    h = np.array(h, dtype=float).copy()
    # Sort by value, set top n//2 to 1
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:n//2]] = 1.0
    return h_bin

def precompute_dcorr(h, n):
    nfull = 2 * n - 1
    k_range = np.arange(nfull)
    dcorr = np.zeros((n, nfull))
    for i in range(n):
        j1 = i + k_range - (n - 1)
        j2 = i - k_range + (n - 1)
        v1 = (j1 >= 0) & (j1 < n)
        v2 = (j2 >= 0) & (j2 < n)
        dcorr[i, v1] += (1.0 - h[j1[v1]])
        dcorr[i, v2] -= h[j2[v2]]
    return dcorr

def binary_local_search(h, n, time_limit=60, rng=None):

        max_indices = np.where(corr >= c_max - 1e-10)[0]

        ones_idx = np.where(h == 1.0)[0]

            break

        best_new_max = c_max

def simulated_annealing_binary(h, n, time_limit=60, rng=None):

    if rng is None:
        rng = np.random.RandomState(1)

    # Fix count

    target_count = n // 2
    while ones_count > target_count:
        i = rng.choice(np.where(h == 1)[0])
        h[i] = 0
        ones_count -= 1
    while ones_count < target_count:
        j = rng.choice(np.where(h == 0)[0])
        h[j] = 1
        ones_count += 1
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    cur_c5 = best_c5
    start = time.time()
    
    # Temperature schedule
    T0_temp = 5.0  # corresponds to accepting swaps that increase max_corr by ~5
    T_end = 0.1
    total_steps = max(1000, int(time_limit * 500))  # ~500 steps/sec
    
    for step in range(total_steps):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 10:
            break
        
        progress = step / total_steps
        T = T0_temp * (T_end / T0_temp) ** progress
        
        # Random swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = rng.choice(ones)
        j = rng.choice(zeros)
        
        h[i] = 0
        h[j] = 1
        new_c5 = compute_c5(h)
        
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            delta = new_c5 - cur_c5
            # delta is in units of dx = 2/n, so raw delta * n/2
            if rng.random() < np.exp(-delta * n / (2.0 * T)):
                cur_c5 = new_c5
            else:
                h[i] = 1
                h[j] = 0
    
    h[:] = best_h
    return h, best_c5

def smooth_max_value(x, beta, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50.0, 50.0)
    sm = m + np.log(np.sum(np.exp(shifted))) / beta
    return sm * (2.0 / n)

def smooth_max_grad(x, beta, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50.0, 50.0)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = w / w_sum
    
    g = np.zeros(n)
    dx = 2.0 / n
    nfull = 2 * n - 1
    for k in range(nfull):
        wk = w[k] * dx
        if wk < 1e-16:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            g[offset:offset + npairs] += wk * (1.0 - x[:npairs])
            g[:npairs] -= wk * x[offset:offset + npairs]
        else:
            a = -offset
            npairs = n - a
            g[:npairs] += wk * (1.0 - x[a:a + npairs])
            g[a:a + npairs] -= wk * x[:npairs]
    return g

def slsqp_optimize(h, n, beta, maxiter=500):
    def obj_func(x):
        return smooth_max_value(x, beta, n)
    def grad_func(x):
        return smooth_max_grad(x, beta, n)
    def eq_fun(x):
        return np.sum(x) - n / 2.0
    def eq_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    
    try:
        res = minimize(obj_func, h, jac=grad_func, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def continuous_local_search(h, n, time_limit=60, rng=None):
    """Local search allowing small continuous adjustments."""
    if rng is None:
        rng = np.random.RandomState(2)
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    nfull = 2 * n - 1
    
    while time.time() - start < time_limit:
        if _BUDGET - (time.time() - _T0) < 10:
            break
        
        corr = compute_corr(h)
        c_max = float(np.max(corr))
        dcorr = precompute_dcorr(h, n)
        
        # Find the lag with max correlation
        k_max = int(np.argmax(corr))
        
        # Try small adjustments at positions that most affect k_max
        # dcorr[:, k_max] shows how changing h[i] affects corr[k_max]
        # To DECREASE corr[k_max], we want to decrease h[i] where dcorr[i,k_max] > 0
        # and increase h[i] where dcorr[i,k_max] < 0
        # But we also need to check other lags don't increase
        
        best_improvement = 1e-12
        best_move = None
        
        # Try pair adjustments: decrease h[i] by eps, increase h[j] by eps
        # Focus on positions with large |dcorr[:, k_max]|
        affected = np.abs(dcorr[:, k_max])
        top_positions = np.argsort(-affected)[:30]
        
        for i in top_positions:
            if h[i] <= 1e-10:
                continue
            # Decrease h[i] by small amount
            for j in top_positions:
                if j == i or h[j] >= 1.0 - 1e-10:
                    continue
                # New corr ≈ corr - eps*dcorr[i] + eps*dcorr[j]
                # We want max to decrease
                for eps in [0.01, 0.05, 0.1, 0.2]:
                    new_corr_approx = corr + eps * (dcorr[j] - dcorr[i])
                    new_max = float(np.max(new_corr_approx))
                    if new_max < c_max - best_improvement:
                        best_improvement = c_max - new_max
                        best_move = (i, j, eps)
        
        if best_move is not None:
            i, j, eps = best_move
            h[i] -= eps
            h[j] += eps
            h = np.clip(h, 0, 1)
            h = project_feasible(h, n)
            new_c5 = compute_c5(h)
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
            else:
                # Revert
                h[i] += eps
                h[j] -= eps
                h = project_feasible(h, n)
        
        # Also try random small perturbations
        if rng.random() < 0.3:
            i = rng.randint(n)
            if 0 < h[i] < 1:
                delta = 0.05 * rng.choice([-1, 1])
                h_trial = h.copy()
                h_trial[i] += delta
                h_trial = np.clip(h_trial, 0, 1)
                # Find another position to compensate
                deficit = h.sum() - h_trial[i] - (h_trial.sum() - delta)
                # Actually just project
                h_trial = project_feasible(h_trial, n)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5 - 1e-14:
                    h = h_trial
                    best_c5 = c5_trial
    
    return h, best_c5

def interp_to(src_h, src_n, dst_n):

    x_new = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(x_new, xp, src_h)
