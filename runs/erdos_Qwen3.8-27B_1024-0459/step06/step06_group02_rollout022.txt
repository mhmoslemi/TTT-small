```python
        proj = np.clip(h - tau, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = tau
        else:
            hi = tau
    h = np.clip(h - tau, 0.0, 1.0)
    # Final correction for numerical issues
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_all_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def lag_grad_vec(h, L, n):
    """Gradient of the correlation at shift L (L can be negative)."""
    g = np.zeros(n)
    if L >= 0:
        npairs = n - L
        if npairs > 0:
            g[:npairs] += (1.0 - h[L:])
            g[L:] -= h[:npairs]
    else:
        A = -L
        npairs = n - A
        if npairs > 0:
            g[A:] += (1.0 - h[:npairs])
            g[:npairs] -= h[A:]
    return g

def smoothed_obj_and_grad(h, beta, topK=50):
    """Compute smooth max objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_all_corr(h)
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -50.0, 50.0)
    ex = np.exp(shifted)
    total = np.sum(ex)
    w = ex / total  # softmax weights
    
    # Objective: (m + log(sum(exp))/beta) * dx
    #             = (m + log(total)/beta) * dx  [since we shifted by m]
    # Actually: smooth_max = (1/beta) * log(sum_k exp(beta * corr_k))
    #           = (1/beta) * log(sum_k exp(beta * (corr_k - m) + beta*m))
    #           = m + (1/beta) * log(sum_k exp(beta * (corr_k - m)))
    #           = m + (1/beta) * log(total)
    obj = (m + np.log(total) / beta) * dx
    
    # Gradient: sum_k w_k * grad_corr_k * dx
    # Only compute for top K lags
    k_idx = np.argsort(-w)[:topK]
    w_k = w[k_idx].copy()
    s = w_k.sum()
    if s > 0:
        w_k /= s
    else:
        # Uniform fallback
        w_k = np.ones(topK) / topK
    
    g = np.zeros(n)
    nhalf = n - 1
    for i, k in enumerate(k_idx):
        if w_k[i] < 1e-15:
            continue
        L = k - nhalf
        g += w_k[i] * lag_grad_vec(h, L, n)
    g *= dx
    return obj, g
```

```python
    best_h = h.copy()

    moment = np.zeros(n)

        if time.time() - start > time_limit:
            break

        # Anneal beta
        beta = beta_start * (beta_end / beta_start) ** progress
        # Nonlinear: beta = beta_start * (beta_end/beta_start)**(it/n_iters)
        
        obj, g = smoothed_obj_and_grad(y, beta, topK=30)
        
        # BB step size (simplified): use a decreasing schedule
        alpha = 0.01 / (1.0 + 0.1 * progress)
        
        # Momentum update (Nesterov)

        y_new = h + s * (h - moment) - alpha * g
        y_new = project_feasible(y_new, n)
        
        # Check improvement
        c_new = compute_c5(y_new)
        if c_new < best_c5:
            best_h = y_new.copy()
            best_c5 = c_new
        
        moment = h
        h = y_new
        y = h
        
    return best_h, best_c5
```

```python
import time
```

_G_beta = 1.0
_G_dx = 1.0
_G_topK = 50
_G_T0 = 0.0
_G_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    # Clip first
    h = np.clip(h, 0.0, 1.0)
    # Water-filling projection
    lo, hi = -1.0, 1.0
    tau = 0.0
    for _ in range(80):
        tau = 0.5 * (lo + hi)
        proj = np.clip(h - tau, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = tau
        else:
            hi = tau
    res = np.clip(h - tau, 0.0, 1.0)
    # Final adjustment
    diff = target - np.sum(res)
    if abs(diff) > 1e-13:
        mask = (res > 1e-14) & (res < 1.0 - 1e-14)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            res[mask] += diff / cnt
        else:
            res[:] += diff / n
        res = np.clip(res, 0.0, 1.0)

    g = np.zeros(n)
    if L >= 0:
        npairs = n - L
        if npairs > 0:
            g[:npairs] += (1.0 - h[L:])
            g[L:] -= h[:npairs]
    else:
        A = -L
        npairs = n - A
        if npairs > 0:
            g[A:] += (1.0 - h[:npairs])
            g[:npairs] -= h[A:]
    return g

    return np.ones(_G_n)

def slsqp_polish(h, n, beta, maxiter=300, ftol=1e-16, topK=50):
    global _G_n, _G_beta, _G_dx, _G_topK
    _G_n = n
    _G_beta = float(beta)
    _G_dx = 2.0 / n
    _G_topK = topK
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    
    best_c5 = compute_c5(h)
    try:
        res = minimize(_obj_slsqp, h, jac=_grad_slsqp, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                return h_new
    except Exception:
        pass
    
    # Try again with slightly perturbed start
    try:
        res = minimize(_obj_slsqp, h, jac=_grad_slsqp, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                return h_new
    except Exception:
        pass
    
    return h.copy()

def upsample(h, new_n):
    n = len(h)
    xp = (np.arange(n) + 0.5) * 2.0 / n
    x_new = (np.arange(new_n) + 0.5) * 2.0 / new_n
    h_new = np.interp(x_new, xp, h)
    return project_feasible(h_new, new_n)

def perturb_structured(h, n, rng, scale):
    h_p = h.copy()
    # Low frequency perturbation
    x = np.arange(n) * 2.0 / n
    for freq in rng.choice([1, 2, 3, 4, 5], size=2, replace=False):
        phase = rng.uniform(0, 2 * np.pi)
        h_p += scale * rng.uniform(0.5, 1.0) * np.sin(np.pi * freq * x + phase)
    # Small random perturbation
    h_p += scale * 0.3 * rng.randn(n)
    return project_feasible(h_p, n)

def subgradient_hybrid(h, n, iters, beta_start, beta_end, time_limit):
    """Projected subgradient with adaptive step size."""
    start_t = time.time()
    h_cur = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h_cur)
    dx = 2.0 / n
    
    # Running parameters
    alpha = 0.05  # initial step size
    beta = beta_start
    prev_grad_norm = 1.0
    
    for it in range(iters):
        if time.time() - start_t > time_limit:
            break
        progress = it / max(iters, 1)
        beta = beta_start + (beta_end - beta_start) * progress
        
        # Compute correlation and identify top lags
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        m = np.max(corr)
        shifted = beta * (corr - m)
        shifted = np.clip(shifted, -50.0, 50.0)
        ex = np.exp(shifted)
        w = ex / np.sum(ex)
        
        # Gradient (top K)
        tk = min(30, len(corr))
        k_idx = np.argsort(-w)[:tk]
        w_k = w[k_idx].copy()
        s = w_k.sum()
        if s > 1e-30:
            w_k /= s
        else:
            w_k = np.ones(tk) / tk
        
        g = np.zeros(n)
        nhalf = n - 1
        for i in range(tk):
            if w_k[i] < 1e-14:
                continue
            L = k_idx[i] - nhalf
            g += w_k[i] * lag_grad_vec(h_cur, L, n)
        g *= dx
        
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        
        # BB step size
        alpha_new = 0.05 / (1.0 + 0.5 * progress)
        h_next = project_feasible(h_cur - alpha_new * g / gn, n)
        
        c_new = compute_c5(h_next)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_next.copy()
        
        h_cur = h_next
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _G_T0, _G_BUDGET
    _G_T0 = time.time()
    _G_BUDGET = float(budget_s)
    
    def remaining():
        return max(0.0, _G_BUDGET - (time.time() - _G_T0) - 3.0)
    
    rng = np.random.RandomState(seed)
    
    # Initialize from parent
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        parent_h = np.zeros(512)
        parent_h[:256] = 1.0
    
    n0 = len(parent_h)
    best_h = project_feasible(parent_h, n0)
    best_c5 = compute_c5(best_h)
    best_n = n0
    
    # Phase 1: SLSQP at parent resolution (n=768)
    for beta in [2000, 10000, 50000]:
        if remaining() < 20:
            break
        h_try = slsqp_polish(best_h, n0, beta, maxiter=300, ftol=1e-16)
        c_try = compute_c5(h_try)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try.copy()
    
    # Phase 2: Try a perturbed start at same resolution
    for p in range(3):
        if remaining() < 30:
            break
        h_pert = perturb_structured(best_h, n0, rng, scale=0.01 + 0.005 * p)
        for beta in [5000, 30000]:
            if remaining() < 15:
                break
            h_try = slsqp_polish(h_pert, n0, beta, maxiter=200, ftol=1e-16)
            c_try = compute_c5(h_try)
            if c_try < best_c5:
                best_c5 = c_try
                best_h = h_try.copy()
    
    # Phase 3: Upsample to n=1536
    n1 = 1536
    if remaining() > 100:
        h_up = upsample(best_h, n1)
        c_up = compute_c5(h_up)
        
        # SLSQP at new resolution
        for beta in [2000, 10000, 50000, 200000]:
            if remaining() < 30:
                break
            h_try = slsqp_polish(h_up, n1, beta, maxiter=250, ftol=1e-16, topK=40)
            c_try = compute_c5(h_try)
            if c_try < c_up:
                c_up = c_try
                h_up = h_try
        
        if c_up < best_c5:
            best_c5 = c_up
            best_h = h_up.copy()
            best_n = n1
    
    # Phase 4: Subgradient at best resolution
    if remaining() > 100:
        h_sg, c_sg = subgradient_hybrid(best_h, best_n, 
                                         iters=2000, beta_start=30, beta_end=500,
                                         time_limit=min(120, remaining() - 30))
        if c_sg < best_c5:
            best_c5 = c_sg
            best_h = h_sg.copy()
        
        # SLSQP from subgradient result
        if remaining() > 20:
            h_try = slsqp_polish(h_sg, best_n, 50000, maxiter=200, ftol=1e-16)
