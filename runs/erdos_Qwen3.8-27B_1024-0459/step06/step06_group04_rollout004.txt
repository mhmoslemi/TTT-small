    g = np.zeros(n)

            np_ = n - ao
            g[:np_] += wk * (1.0 - x[ao:])

            if rem() < 10:
                break
            h_p = slsqp_polish(best_h, best_n, beta, maxiter=150)
            c_p = compute_c5(h_p)
            if c_p < best_c5 - 1e-14:
                best_c5 = c_p
                best_h = h_p.copy()

    # Phase 5: TC polish
    if rem() > 15:
        h_tc = tc_polish(best_h, best_n, 300000, maxiter=100)
        c_tc = compute_c5(h_tc)
        if c_tc < best_c5 - 1e-14:
            best_c5 = c_tc
            best_h = h_tc.copy()

    # Final check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
