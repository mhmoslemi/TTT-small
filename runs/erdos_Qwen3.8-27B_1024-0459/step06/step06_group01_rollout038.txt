```python
def compute_corr_fft(h, n):
    # We need corr[k] = sum_i h[i] * (1-h[i+k]) for all k
    # = sum_i h[i] - sum_i h[i]*h[i+k]
    # = n*h_mean - autocorr_h(k) ... no wait
    # Actually: corr[k] = sum_i h[i]*(1-h[i+k]) for valid i
    # Let's use the definition: full correlation of h and (1-h)
    # full_corr[j] = sum_i h[i] * (1-h[j-i]) for j = -(n-1), ..., n-1
    # Using FFT: pad to length >= 2n-1, zero-pad, FFT, multiply, IFFT
    N = 1
    while N < 2*n - 1:
        N *= 2
    f1 = np.fft.rfft(h, N)
    f2 = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(f1 * np.conj(f2), N)[:2*n-1]
    return corr
```

```python
    target = 2 * n - 1

        N *= 2

    n = len(h)
    N = get_fft_size(n)
    f_h = np.fft.rfft(h, N)
    f_1mh = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(f_h * np.conj(f_1mh), N)
    # Rearrange to match np.correlate(h, 1-h, mode='full')
    return np.concatenate([corr[N-n+1:N], corr[0:n]])

    corr = compute_corr_fft(h)
    dx = 2.0 / len(h)
    return float(np.max(corr) * dx)

        if cnt > 0:

        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
```

```python
import numpy as np
import time

_T0 = 0.0
_BUDGET = 1300.0

def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)

def compute_effect_vec(h, j, n):
    """Compute d(corr)/d(h[j]) using the correlation structure."""
    e = np.zeros(2 * n - 1)
    # For lag ell >= 0 (index d = n-1+ell):
    # e[d] = (1-h[j+ell])*[j+ell<n] - h[j-ell]*[j>=ell and j-ell<n]
    ell = np.arange(n)
    # Positive contribution: j+ell < n => ell < n-j
    mask_pos = ell < (n - j)
    if mask_pos.any():
        e[n-1 + ell[mask_pos]] += (1.0 - h[j + ell[mask_pos]])
    # Negative contribution: j-ell >= 0 => ell <= j
    mask_neg = ell <= j
    if mask_neg.any():
        e[n-1 + ell[mask_neg]] -= h[j - ell[mask_neg]]
    
    # For lag ell < 0, let a = -ell > 0 (index d = n-1-a):
    # e[d] = (1-h[j-a])*[j>=a and j-a<n] - h[j+a]*[j+a<n]
    a = np.arange(1, n)
    d_idx = n - 1 - a
    # (1-h[j-a]) valid when j >= a
    mask1 = a <= j
    if mask1.any():
        e[d_idx[mask1]] += (1.0 - h[j - a[mask1]])
    # -h[j+a] valid when j+a < n
    mask2 = a < (n - j)
    if mask2.any():
        e[d_idx[mask2]] -= h[j + a[mask2]]
    
    return e

def coord_descent_fast(h, n, max_sweeps, n_gs=25):
    """Fast coordinate descent with incremental correlation updates."""
    h = np.array(h, dtype=float).copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr) * dx)
    
    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        improved = False
        for j in range(n):
            old_v = h[j]
            e = compute_effect_vec(h, j, n)
            
            # Golden section on [0,1]
            a, b = 0.0, 1.0
            c = 0.6180339887  # (1 - 1/phi) / ... actually 1/phi^2 = 0.381966
            d = 1.0 - c  # = 0.381966
            
            x1 = a + d * (b - a)  # = 0.381966
            x2 = a + c * (b - a)  # = 0.618034
            f1 = np.max(corr + (x1 - old_v) * e) * dx
            f2 = np.max(corr + (x2 - old_v) * e) * dx
            
            for _ in range(n_gs):
                if f1 < f2:
                    b = x2
                    x2 = x1
                    f2 = f1
                    x1 = a + d * (b - a)
                    f1 = np.max(corr + (x1 - old_v) * e) * dx
                else:
                    a = x1
                    x1 = x2
                    f1 = f2
                    x2 = a + c * (b - a)
                    f2 = np.max(corr + (x2 - old_v) * e) * dx
            
            # Determine best
            best_v = x1 if f1 < f2 else x2
            best_f = min(f1, f2)
            
            # Check endpoints
            f0 = np.max(corr + (0.0 - old_v) * e) * dx
            if f0 < best_f:
                best_v = 0.0
                best_f = f0
            f1v = np.max(corr + (1.0 - old_v) * e) * dx
            if f1v < best_f:
                best_v = 1.0
                best_f = f1v
            
            # Update
            if abs(best_v - old_v) > 1e-15:
                corr += (best_v - old_v) * e
                h[j] = best_v
                curr = float(np.max(corr) * dx)
                if curr < best_c5 - 1e-14:
                    best_c5 = curr
                    improved = True
        
        # Recompute corr periodically to avoid drift
        if sweep % 100 == 99:
            corr = np.correlate(h, 1.0 - h, mode='full')
            curr = float(np.max(corr) * dx)
            if curr < best_c5:
                best_c5 = curr
        
        if not improved and sweep > 5:
            break
    
    return h, best_c5
```
