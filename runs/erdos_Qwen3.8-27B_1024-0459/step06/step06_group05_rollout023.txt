        k_star = np.argmax(corr)

        K = min(50, 2*n-1)
        top_k = np.argsort(-corr)[:K]

        # c = g_{k*} (gradient of f_{k*} at h)
        # A_ub: rows are g_j for top K correlations
        # b_ub: f_j(h) + tol for j ≠ k*, f_{k*}(h) - decrease for j = k*
        # A_eq: ones(n)

import time

_T0 = 0.0
_BUDGET = 1300.0

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_corr_k(h, k, n):
    """Gradient of correlation at lag k with respect to h.
    k is the index in the full correlation array (0 to 2n-2)."""
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        # c = sum_n h[n+offset] * (1-h[n])
        # grad[j] = (1-h[j-offset]) if j >= offset, + (-h[j+offset]) if j+offset < n
        npairs = n - offset
        g[offset:] += (1.0 - h[:npairs])
        g[:npairs] += -h[offset:]
    else:
        a = -offset
        # c = sum_n h[n-a] * (1-h[n])
        # grad[j] = (1-h[j+a]) if j+a < n, + (-h[j-a]) if j >= a
        npairs = n - a
        g[:npairs] += (1.0 - h[a:a+npairs])
        g[a:] += -h[:npairs]
    return g

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

        top_idx = np.argsort(-corr_raw)[:K_constr]

        b_vals = []

            if idx == k_star:

            else:

            if res.success and res.x is not None:

                c5_new = compute_c5(h_new)
                if c5_new < c5_best:
                    c5_best = c5_new
                    h_best = h_new.copy()
                    h = h_new.copy()
                else:
                    # LP didn't help; try smaller step
                    # (this shouldn't happen often with good tolerance)
                    pass
            else:

                break
        except Exception:

        # Adaptive: if no progress for a while, reduce tol/decrease
        if it > 20 and compute_c5(h) >= c5_best - 1e-12:
            tol *= 0.5
            decrease *= 0.5
            if tol < 1e-6:
                break
    
    return h_best, c5_best

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        if npairs > 0:
            g[offset:] += (1.0 - h[:npairs])
            g[:npairs] += -h[offset:]
    else:
        a = -offset
        npairs = n - a
        if npairs > 0:
            g[:npairs] += (1.0 - h[a:a + npairs])
            g[a:] += -h[:npairs]
    return g

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def sm_objective(x, n, K, beta):
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:K]
    ct = corr[idx]
    m = ct[0]
    ex = np.exp(np.clip(beta * (ct - m), -40, 40))
    s = ex.sum()
    return (m + np.log(s) / beta) * (2.0 / n)

def sm_grad(x, n, K, beta):
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:K]
    ct = corr[idx]
    m = ct[0]
    ex = np.exp(np.clip(beta * (ct - m), -40, 40))
    w = ex / ex.sum()
    dx = 2.0 / n

    for i in range(K):
        if w[i] < 1e-14:
            continue
        g += w[i] * dx * grad_corr_k(x, idx[i], n)
    return g

def do_slsqp(h0, n, K, beta, maxiter):

    h_best = h0.copy()
    c5_best = compute_c5(h0)
    start = time.time()
    tol = tol_init
    dec = dec_init
    stall = 0
    it = 0
    
    while True:
        if time.time() - start > time_limit:
            break
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        
        it += 1
        corr_raw = compute_corr(h)
        k_star = int(np.argmax(corr_raw))
        
        # Top K correlations for constraints
        top_idx = np.argsort(-corr_raw)[:K_constr]
        
        # LP objective: gradient of f_{k*}
        c_obj = grad_corr_k(h, k_star, n)
        
        # LP constraints
        A_rows = []
        b_vals = []
        for idx in top_idx:
            g_k = grad_corr_k(h, idx, n)
            A_rows.append(g_k)
            if idx == k_star:
                b_vals.append(corr_raw[idx] - dec)
            else:
                b_vals.append(corr_raw[idx] + tol)
        
        A_ub = np.array(A_rows)
        b_ub = np.array(b_vals)
        A_eq = np.ones((1, n))
        b_eq = np.array([n / 2.0])
        bounds = [(0.0, 1.0)] * n

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        if npairs > 0:
            g[offset:] += (1.0 - h[:npairs])
            g[:npairs] += -h[offset:]
    else:
        a = -offset
        npairs = n - a
        if npairs > 0:
            g[:npairs] += (1.0 - h[a:a + npairs])
            g[a:] += -h[:npairs]
    return g

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def _sm_obj(x, n, K, beta):
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:K]
    ct = corr[idx]
    m = ct[0]
    ex = np.exp(np.clip(beta * (ct - m), -40.0, 40.0))
    s = ex.sum()
    if s < 1e-30:
        return m * (2.0 / n)
    return (m + np.log(s) / beta) * (2.0 / n)

def _sm_grad(x, n, K, beta):
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:K]
    ct = corr[idx]
    m = ct[0]
    ex = np.exp(np.clip(beta * (ct - m), -40.0, 40.0))

    return np.sum(x) - n / 2.0

        if res.x is not None:

    except Exception:
        pass
    return h
