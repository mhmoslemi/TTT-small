    return float(np.max(corr)) * dx

def project_feasible(h, n):
    # Project onto {0 <= h <= 1, sum(h) = n/2}
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        # Distribute the correction to interior points
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if interior.any():
            h[interior] += diff / interior.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, beta):
    """Compute smooth max of correlations and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_scaled = corr * dx  # This is what we're maximizing
    m = np.max(corr_scaled)
    s = np.clip(beta * (corr_scaled - m), -500.0, 500.0)
    e = np.exp(s)
    val = (m + np.log(e.sum()) / beta)
    
    # Gradient
    w = e / e.sum()
    g = np.zeros(n)

- For off = k-(n-1) ≥ 0: j ∈ [0, n-1-off] (as first factor) OR j ∈ [off, n-1] (as second factor)
  - First: off ≤ j, so k-(n-1) ≤ j, k ≤ j+(n-1)
  - Second: off ≤ j ≤ n-1+off, so k-(n-1) ≤ j ≤ n-1+k-(n-1), i.e., k ≥ j and off ≤ n-1-j, i.e., k-(n-1) ≤ n-1-j, k ≤ 2n-2-j

  Combined for off ≥ 0: k ∈ [max(n-1, n-1+off... )] ... this is getting complicated.

Let me just say: h[j] appears in at most 2n-1 different corr[k] values (all of them, in the worst case). And in each, the partial derivative is bounded by 1 in absolute value.

So |g_j| ≤ dx * sum_k w_k * 1 * (number of times j appears in corr[k])
≤ dx * 1 * (2n-1) = 2*(2n-1)/n ≈ 4

So L ≈ 4 (rough bound). A step size of 1/4 = 0.25 would be safe.

But this is a very loose bound. In practice, the weights w_k are concentrated on a few k values (for large β), so the actual gradient norm is much smaller.

Let me just use a practical approach:
- Start with step = 0.1
- If function increases, halve the step
- If function decreases by > 50% of expected, double the step (capped)

For the case where only a few k are active (high β), the loop is fine. For low β (many active k), I might want to vectorize.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                h = np.clip(h + diff / n, 0, 1)
            else:
                h = h + room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                h = np.clip(h + diff / n, 0, 1)
            else:
                h = h + room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(h, beta):
    corr = compute_corr_vec(h)
    m = np.max(corr)
    s = np.clip(beta * (corr - m), -500.0, 500.0)
    e = np.exp(s)
    return m + np.log(e.sum()) / beta

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    m = np.max(c)
    s = np.clip(beta * (c - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / e.sum()
    
    g = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for k_idx in active:
        wk = w[k_idx] * dx
        off = k_idx - (n - 1)
        if off >= 0:
            np_ = n - off
            g[:np_] += wk * (1.0 - h[off:off+np_])
            g[off:off+np_] -= wk * h[:np_]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - h[ao:ao+np_])
            g[ao:ao+np_] -= wk * h[:np_]
    return g

def topk_grad(h, K, beta_w):
    """Compute gradient using only top-K correlations with exponential weights."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    idx = np.argsort(-c)[:K]
    c_top = c[idx]
    weights = np.exp(beta_w * (c_top - c_top.max()))
    s = weights.sum()
    if s < 1e-30:
        weights = np.ones(K) / K
    else:
        weights /= s
    
    g = np.zeros(n)
    for i, k_idx in enumerate(idx):
        wk = weights[i]
        if wk < 1e-12:
            continue
        wkc = wk * dx
        off = k_idx - (n - 1)
        if off >= 0:
            np_ = n - off
            g[:np_] += wkc * (1.0 - h[off:off+np_])
            g[off:off+np_] -= wkc * h[:np_]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wkc * (1.0 - h[ao:ao+np_])
            g[ao:ao+np_] -= wkc * h[:np_]
    return g

        # Check function value

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_momentum * t_momentum)) / 2.0
        y = h_new + ((t_momentum - 1.0) / t_new) * (h_new - h_prev)
        y = np.clip(y, 0.0, 1.0)  # Keep y in bounds (soft)

        h = h_new

def interp_to(src_h, src_n, dst_n):

    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

        c_max = np.max(corr)
        
        # Find t: water level such that sum(max(0, c_k - t)) = 1/mu
        # Use bisection
        t_lo = np.median(corr)
        t_hi = c_max
        for _ in range(50):
            t_mid = (t_lo + t_hi) / 2
            excess = np.sum(np.maximum(0, corr - t_mid))
            if excess > 1.0 / mu:
                t_hi = t_mid
            else:
                t_lo = t_mid
        t = (t_lo + t_hi) / 2
        
        # Gradient: mu * sum_{k: c_k > t} (c_k - t) * grad(c_k)
        active = np.where(corr > t)[0]
        if len(active) == 0:
            break
        
        weights = mu * (corr[active] - t)
        total_w = weights.sum()
        # Normalize to get a direction
        weights = weights / total_w if total_w > 0 else np.ones(len(active)) / len(active)
        
        g = np.zeros(n)

                g[ao:ao+np_] -= wkc * h[:np_]
