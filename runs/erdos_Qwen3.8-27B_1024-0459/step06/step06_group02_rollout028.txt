```python
def compute_gradient_direct(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    g_k = corr * dx
    
    # Softmax
    shifted = beta * g_k
    shifted -= shifted.max()
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()  # length 2n-1
    
    # g[j] = dx * sum_k a[k] * w[n-1+|j-k|]
    a = 1.0 - 2.0 * h
    indices = np.arange(n)
    dist = np.abs(indices[:, None] - indices[None, :])  # n x n
    w_mat = w[n-1 + dist]  # n x n
    g = dx * (w_mat @ a)
    
    return g, w, corr
```
