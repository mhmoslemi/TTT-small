def _compute_c5(h):

def _project(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        d = target - np.sum(h)

    return h

def _sm_obj_grad(h, beta):
    """Returns (objective, gradient) for smooth-max of corr * dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)

    S = np.sum(ez)
    obj = (m + np.log(S) / beta) * dx
    w = ez / S
    
    # Gradient via W_full

    W_full[0] = w[n-1]

    h0 = _project(h, n).copy()
    
    def obj_fn(x):
        o, _ = _sm_obj_grad(x, beta)
        return o
    
    def grad_fn(x):
        _, g = _sm_obj_grad(x, beta)
        return g

        return np.sum(x) - n/2.0

        return np.ones(n)
