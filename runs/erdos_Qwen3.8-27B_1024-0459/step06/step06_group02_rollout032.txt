import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_onto_simplex(h, n):
    """Project h onto {0 <= x <= 1, sum(x) = n/2} efficiently using bisection on tau."""
    target = n / 2.0
    # Find tau such that sum(clip(h - tau, 0, 1)) = target
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        clipped = np.clip(h - mid, 0, 1)
        s = clipped.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    clipped = np.clip(h - (lo + hi) / 2, 0, 1)
    # Fine-tune to get exact sum
    diff = target - clipped.sum()
    if abs(diff) > 1e-10:
        # Distribute the difference
        mask = (clipped > 1e-10) & (clipped < 1 - 1e-10)
        if mask.sum() > 0:
            clipped[mask] += diff / mask.sum()
            clipped = np.clip(clipped, 0, 1)
    return clipped

def grad_corr_all(h, n, active_lags):
    """Compute gradient of correlation at specified lags.
    active_lags: list of offsets m (where lag index = n-1+m)"""
    g = np.zeros(n)
    for m in active_lags:
        if m >= 0:
            npairs = n - m
            if npairs > 0:
                # corr = sum_{i=0}^{n-1-m} h[i+m] * (1-h[i])
                # d/dh[j] = (1-h[j-m]) for m<=j<n, minus h[j+m] for 0<=j<n-m
                g[m:] += (1.0 - h[:npairs])
                g[:npairs] -= h[m:]
        else:
            a = -m
            npairs = n - a
            if npairs > 0:
                # corr = sum_{i=0}^{n-1-a} h[i] * (1-h[i+a])
                # d/dh[j] = (1-h[j+a]) for 0<=j<n-a, minus h[j-a] for a<=j<n
                g[:npairs] += (1.0 - h[a:])
                g[a:] -= h[:npairs]
    return g

def fista_smooth_max(h0, n, beta, max_iter, time_limit, progress_cb=None):
    """FISTA for smooth max approximation."""
    h = h0.copy()
    y = h0.copy()
    t = 1.0
    dx = 2.0 / n
    
    # Compute Lipschitz constant (approximate)
    # ||∇f||_L ≈ (2/n) * sum_k w_k * ||∇corr_k||_inf
    # ||∇corr_k||_inf ≤ 2 (each component is at most 1 in magnitude from each term)
    # Actually for a single lag, max |gradient component| ≤ 1
    # For weighted sum with weights summing to 1: max |gradient| ≤ 1
    # So L ≈ 2/n
    # But this is an upper bound; let's be safe with L = 4/n
    L = 4.0 / n
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        # Compute corr and weights
        corr = np.correlate(y, 1.0 - y, mode='full')
        m = corr.max()
        shifted = beta * (corr - m)
        shifted = np.clip(shifted, -50, 50)
        w = np.exp(shifted)
        w /= w.sum()
        
        # Find active lags (weight > threshold)
        threshold = 1e-6
        active_mask = w > threshold
        active_indices = np.where(active_mask)[0]
        active_offsets = active_indices - (n - 1)
        
        # Gradient
        g = grad_corr_all(y, n, active_offsets) * w[active_mask]
        g *= dx  # scale by dx
        
        # Gradient step
        h_new = y - g / L
        h_new = project_onto_simplex(h_new, n)
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = h_new + ((t - 1.0) / t_new) * (h_new - h)
        y = project_onto_simplex(y, n)
        h = h_new
        t = t_new
        
        # Track best
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        if progress_cb and it % 100 == 0:
            progress_cb(it, best_c5)
    
    return best_h, best_c5

4. This gradient points in the direction that increases all ceiling lags simultaneously
5. Step in the -g direction (to reduce all ceiling lags)
6. Project to feasible set

def ceiling_descent(h, n, epsilon=0.5, step=0.01, max_iter=10000):
    """Reduce the ceiling (max correlation) using projected gradient on ceiling lags."""
    h = h.copy()
    dx = 2.0 / n
    target = n / 2.0
    
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        M = corr.max()
        
        # Find ceiling lags
        ceiling_mask = corr > M - epsilon
        ceiling_indices = np.where(ceiling_mask)[0]
        
        if len(ceiling_indices) == 0:
            break
        
        # Compute aggregated gradient
        g = np.zeros(n)
        for idx in ceiling_indices:
            m = idx - (n - 1)
            # grad of corr at this lag
            gm = grad_corr_k(h, m, n)  # need to define this
            g += gm
        
        g /= len(ceiling_indices)
        
        # Project negative gradient to feasible direction
        # We need: h + step*(-g) stays in [0,1]^n and sum = n/2
        h_trial = h - step * g
        h_trial = project_onto_simplex(h_trial, n)
        
        # Check if it improved
        corr_new = np.correlate(h_trial, 1.0 - h_trial, mode='full')
        M_new = corr_new.max()
        
        if M_new < M - 1e-10:
            h = h_trial
        else:
            # Reduce step size
            step *= 0.9
            if step < 1e-10:
                break
    
    return h

Average overlap for a random balanced sequence: (n-|m|)/4 for lag m.
For small m: ≈ 768/4 = 192.
So we need to reduce from 192 down to 146, a 24% reduction. That's significant!

This means the optimal sequence is highly structured - not just random.

**Alternative approach: Use known good sequences as starting points**

There are known sequences with good correlation properties:
- Ternary sequences (Walsh functions)
- Binary m-sequences
- Costas arrays (permutation-based)

But these are designed for autocorrelation, not for the one-sided overlap metric. Still, they might provide good starting points.

Actually, let me think about this more carefully. The overlap at lag m counts 1→0 transitions (shifted by m). This is different from autocorrelation which counts both 1→0 and 0→1 transitions (for the off-diagonal terms).

For a binary sequence with n/2 ones:
- Autocorrelation at lag m: ∑ h[i]h[i+m] - (known terms)
- Overlap at lag m: ∑ h[i](1-h[i+m]) = ∑ h[i] - ∑ h[i]h[i+m]

So overlap_m = (number of ones in positions 0 to n-1-m) - autocorr_m

The total ones in the relevant range depends on m, so it's not exactly "autocorrelation + constant".

For a periodic extension (which is what the problem effectively does), the overlap is:
overlap_m = (n/2)(1 - autocorr_m/n) ... no, this isn't quite right either.

Let me just focus on the practical approach: use the parent solution, apply targeted optimization, and try to squeeze out the last bit of improvement.

**Let me finalize my approach:**

The key insight is that the gap (0.380898 vs 0.38085) is very small. The most likely ways to close it:
1. Better local optimization (ceiling descent is promising)
2. Slightly different resolution (try n that gives better grid alignment)
3. Multi-start from diverse points

Let me also try a "random restart" approach:
- Occasionally perturb the solution significantly

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_onto_simplex(h, n):
    target = n / 2.0
    lo = -2.0
    hi = 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    result = np.clip(h - tau, 0.0, 1.0)
    diff = target - result.sum()
    if abs(diff) > 1e-12:
        mask = (result > 1e-12) & (result < 1.0 - 1e-12)
        if mask.sum() > 0:
            result[mask] += diff / mask.sum()
            result = np.clip(result, 0.0, 1.0)
    return result

def grad_corr_k(h, m, n):
    g = np.zeros(n)
    if m >= 0:
        npairs = n - m
        if npairs > 0:
            g[m:] += (1.0 - h[:npairs])
            g[:npairs] -= h[m:]
    else:
        a = -m
        npairs = n - a
        if npairs > 0:
            g[:npairs] += (1.0 - h[a:])
            g[a:] -= h[:npairs]
    return g

def ceiling_descent(h, n, epsilon, max_iter, time_limit):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step = 0.02
    start = time.time()
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        M = corr.max()
        c5 = M * (2.0 / n)
        
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 1000:
                break
        
        ceiling_mask = corr >= M - epsilon
        ceiling_idx = np.where(ceiling_mask)[0]
        
        g = np.zeros(n)
        for idx in ceiling_idx:
            m_lag = int(idx) - (n - 1)
            g += grad_corr_k(h, m_lag, n)
        g /= max(1, len(ceiling_idx))
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        
        h_trial = project_onto_simplex(h - step * g, n)
        corr_t = np.correlate(h_trial, 1.0 - h_trial, mode='full')
        
        if corr_t.max() < M - 1e-14:
            h = h_trial
            step = min(step * 1.02, 0.15)
        else:
            step *= 0.85
            if step < 1e-8:
                h = best_h
                step = 0.01
                no_improve = 0
    
    return best_h, best_c5

def fista_smooth(h0, n, beta, max_iter, time_limit):
    h = h0.copy()
    y = h0.copy()
    t = 1.0
    dx = 2.0 / n
    L = 6.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(y, 1.0 - y, mode='full')
        m_val = corr.max()
        shifted = np.clip(beta * (corr - m_val), -50, 50)
        w = np.exp(shifted)
        w /= w.sum()
        
        g = np.zeros(n)

            m_lag = int(idx) - (n - 1)

        h = h_new
        t = t_new
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def slsqp_polish(h, n, beta, maxiter):
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -60, 60)
        ex = np.exp(shifted)
        return (m + np.log(np.sum(ex)) / beta) * dx
    
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -60, 60)
        ex = np.exp(shifted)
        s = np.sum(ex)
        w = ex / s
        g = np.zeros(n)
        for idx in np.where(w > 1e-10)[0]:
            m_lag = int(idx) - (n - 1)
            g += w[int(idx)] * grad_corr_k(x, m_lag, n)
        g *= dx
        return g
    
    def con_fun(x):
        return np.sum(x) - n / 2.0
    
    def con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'iprint': 0})
        if res.x is not None:
            return project_onto_simplex(res.x, n)
    except:
        pass
    return h.copy()

def binary_search(h, n, time_limit, seed):
    rng = np.random.RandomState(seed)
    h_bin = (h > 0.5).astype(float)
    n_ones = int(h_bin.sum())
    target = n // 2
    if n_ones > target:
        ones_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones_idx])
        h_bin[ones_idx[order[:n_ones - target]]] = 0
    elif n_ones < target:
        zeros_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros_idx])
        h_bin[zeros_idx[order[:target - n_ones]]] = 1
    
    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin)
    start = time.time()
    temp = 0.5
    cooling = 0.9998
    
    while time.time() - start < time_limit:
        ones_idx = np.where(h_bin == 1)[0]
        zeros_idx = np.where(h_bin == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = int(rng.choice(ones_idx))
        j = int(rng.choice(zeros_idx))
        
        h_bin[i] = 0
        h_bin[j] = 1
        c5_new = compute_c5(h_bin)
        delta = c5_new - best_c5
        
        if delta < 0 or (temp > 1e-6 and rng.random() < np.exp(-delta / temp)):
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_bin.copy()
        else:
            h_bin[i] = 1
            h_bin[j] = 0
        temp *= cooling
    
    return best_h, best_c5
