import numpy as np
import time

def compute_c5_binary(h):
    """Compute C5 for a binary (or general) sequence."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(float), 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def binarize(h, n):
    """Convert h to binary with exactly n/2 ones."""
    h = np.array(h, dtype=float).copy()
    # Sort by value, set top n/2 to 1, rest to 0
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:n//2]] = 1.0
    return h_bin

def sa_optimize_binary(h, n, start_temp, end_temp, time_limit, seed=42):
    """Simulated annealing for binary sequence optimization."""
    rng = np.random.RandomState(seed)
    h = h.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5_binary(h)
    curr_c5 = best_c5
    curr_corr = np.correlate(h, 1.0 - h, mode='full')
    curr_max = float(np.max(curr_corr))
    
    dx = 2.0 / n
    ones_idx = np.where(h == 1.0)[0]
    zeros_idx = np.where(h == 0.0)[0]
    
    start_time = time.time()
    total_iterations = int(time_limit / 0.001)  # estimate
    n_iterations = 0
    
    while time.time() - start_time < time_limit:
        n_iterations += 1
        # Compute temperature
        progress = min(n_iterations / max(total_iterations, 1), 1.0)
        T = start_temp * (end_temp / start_temp) ** progress
        
        # Random swap
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        
        # Apply swap
        h[i] = 0.0
        h[j] = 1.0
        
        # Compute new correlation
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_max = float(np.max(new_corr))
        delta = (new_max - curr_max) * dx
        
        # Accept?
        if delta < 0 or (T > 1e-15 and rng.rand() < np.exp(-delta / T)):
            # Accept
            curr_max = new_max
            curr_corr = new_corr
            curr_c5 = new_max * dx
            # Update index sets
            ones_idx = np.where(h == 1.0)[0]
            zeros_idx = np.where(h == 0.0)[0]
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h.copy()
        else:
            # Reject
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_c5

            inv_ones[j] = oi
            inv_ones[i] = -1
            zeros_positions[zi] = i
            inv_zeros[i] = zi
            inv_zeros[j] = -1
            c5 = curr_max * dx

                best_h = h.copy()

    rng = np.random.RandomState(seed)
    h = h.astype(np.float64).copy()
    half_n = n // 2

    best_c5 = curr_max * dx
    best_h = h.copy()
    
    ones_pos = np.where(h == 1.0)[0]
    zeros_pos = np.where(h == 0.0)[0]

    # Temperature: work in C5 units
    T_start = 0.005
    T_end = 1e-8
    
    # Estimate total iterations for cooling
    # We'll use time-based cooling
    n_est = int(time_limit * 500)  # estimate ~500 iters/sec
    cool_factor = (T_end / T_start) ** (1.0 / max(n_est, 1))
    T = T_start
    
    count = 0
    while True:
        now = time.time()
        if now - start_time >= time_limit:
            break
        if deadline is not None and now >= deadline:
            break
        count += 1
        
        # Temperature
        T *= cool_factor
        
        # Random swap
        oi = rng.randint(half_n)
        zi = rng.randint(half_n)
        i = ones_pos[oi]
        j = zeros_pos[zi]
        
        # Apply swap
        h[i] = 0.0
        h[j] = 1.0
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_max = float(np.max(new_corr))
        delta = (new_max - curr_max) * dx
        
        # Accept/reject
        if delta <= 0.0:
            accept = True
        elif T > 1e-15:
            accept = (rng.rand() < np.exp(-delta / T))

            accept = False

            curr_max = new_max

            zeros_pos[zi] = i

                best_h = h.copy()

            h[j] = 0.0

    for r in [256, 384, 512, 768, 1024]:
        if r not in res_list:
            res_list.append(r)
    res_list = sorted(set(res_list))

    for n_t in res_list:
        if remaining() < 80:
            break
        if n_t == best_n:
            continue

        # Create starting point at new resolution
        if parent_h is not None and n_t != len(parent_h):
            h_interp = interp_to(parent_h, len(parent_h), n_t)
        else:
            h_interp = interp_to(best_h, best_n, n_t)
        h_bin_t = binarize(h_interp, n_t)
        c5_t = compute_c5(h_bin_t)

        # SA at new resolution
        if remaining() > 60:
            alloc_sa = min(120.0, remaining() * 0.35)
            h_sa_t, c5_sa_t = sa_optimize(h_bin_t.copy(), n_t, alloc_sa,
                                          seed=seed + 100 + n_t,
                                          deadline=_T0 + _BUDGET - 30)
            c5_t = min(c5_t, c5_sa_t)
            h_bin_t = h_sa_t.copy() if c5_sa_t < c5_t else h_bin_t

        # Local search at new resolution
        if remaining() > 40:
            alloc_ls = min(60.0, remaining() * 0.25)
            h_ls_t, c5_ls_t = local_search(h_bin_t.copy(), n_t, alloc_ls,
                                            seed=seed + 200 + n_t,
                                            deadline=_T0 + _BUDGET - 30)
            if c5_ls_t < c5_t:
                c5_t = c5_ls_t
                h_bin_t = h_ls_t.copy()

        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_bin_t.copy()
            best_n = n_t

    # Phase 5: Second SA round at best resolution with different seed
    if remaining() > 80:
        alloc = min(200.0, remaining() * 0.4)
        h_sa2, c5_sa2 = sa_optimize(best_h.copy(), best_n, alloc, seed=seed + 1000,
                                    deadline=_T0 + _BUDGET - 20)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()

    # Phase 6: Final local search
    if remaining() > 30:
        alloc = min(100.0, remaining() * 0.5)
        h_final, c5_final = local_search(best_h.copy(), best_n, alloc,
                                         seed=seed + 2000,
                                         deadline=_T0 + _BUDGET - 10)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Phase 7: Targeted local search at the end
    if remaining() > 20:
        alloc = min(60.0, remaining() * 0.6)
        h_ts2, c5_ts2 = targeted_local_search(best_h.copy(), best_n, alloc,
                                              deadline=_T0 + _BUDGET - 5)
        if c5_ts2 < best_c5:
            best_c5 = c5_ts2
            best_h = h_ts2.copy()

    # Final evaluation
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
