def coordinate_descent_exact(h, n, rho, time_limit):
    """Coordinate descent with exact 1D minimization using smooth max."""
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    
    while time.time() - start < time_limit:
        improved = False
        for i in range(n):
            if time.time() - start > time_limit:
                break
            
            old_val = h[i]
            S = np.sum(h) - n/2.0  # current constraint violation
            
            def obj_1d(v):
                h[i] = v
                corr = compute_corr(h)
                c_max = np.max(corr)
                S_new = np.sum(h) - n/2.0
                return c_max * (2.0/n) + 0.5 * rho * S_new**2
            
            # Find best v in [0, 1]
            from scipy.optimize import minimize_scalar
            res = minimize_scalar(obj_1d, bounds=(0, 1), method='bounded',
                                  options={'xatol': 1e-8})
            if res.x is not None and res.fun < obj_1d(old_val):
                h[i] = res.x
                c_new = compute_c5(h)
                if c_new < best_c5 - 1e-14:
                    best_c5 = c_new
                    improved = True
        
        if not improved:
            break
    
    return h, best_c5

def project_feasible(h, n):

def compute_d_vec(h, i):

    d = np.zeros(2 * n - 1)
    m_min1 = max(-(n - 1), i - (n - 1))

def smooth_max_val_and_grad(h, n, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(beta * sh, -50.0, 50.0)
    ex = np.exp(cl)
    obj = (m + np.log(np.sum(ex)) / beta) * (2.0 / n)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k] * (2.0 / n)
        if wk < 1e-16:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            g[offset:n] += wk * (1.0 - h[:npairs])
            g[:npairs] -= wk * h[offset:n]
        else:
            a = -offset
            npairs = n - a
            g[:npairs] += wk * (1.0 - h[a:n])
            g[a:n] -= wk * h[:npairs]
    return obj, g


def lbfgsb_smooth(h, n, beta, penalty, maxiter=2000):
    def obj_fun(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        sm = (m + np.log(np.sum(ex)) / beta) * (2.0 / n)
        s = np.sum(x) - n / 2.0
        return sm + 0.5 * penalty * s * s

    def grad_fun(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        for k in range(len(corr)):
            wk = w[k] * (2.0 / n)
            if wk < 1e-16:
                continue
            offset = k - (n - 1)
            if offset >= 0:
                npairs = n - offset
                g[offset:n] += wk * (1.0 - x[:npairs])
                g[:npairs] -= wk * x[offset:n]
            else:
                a = -offset
                npairs = n - a
                g[:npairs] += wk * (1.0 - x[a:n])
                g[a:n] -= wk * x[:npairs]
        s = np.sum(x) - n / 2.0
        g += penalty * s
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fun, h, jac=grad_fun, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-15, 'maxfun': 10000})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

        return smooth_max_val_and_grad(x, n, beta)[1]

    try:

    except Exception:

            for factor in [0.5, 0.25, 0.1, 0.05]:
                h_try = project_feasible(h - step * factor * g, n)
                c_try = compute_c5(h_try)
                if c_try < best_c5:
                    best_c5 = c_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    found = True
                    break
            if not found and it % 200 == 0:
                h = best_h.copy()
    return best_h, best_c5


def equalize_top_k(h, n, K=10, max_iters=300):
    global _T0, _BUDGET
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(max_iters):
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:K]
        g_avg = np.zeros(n)
        for k in idx:
            offset = k - (n - 1)
            if offset >= 0:
                npairs = n - offset
                g_avg[offset:n] += 1.0 - h[:npairs]
                g_avg[:npairs] -= h[offset:n]
            else:
                a = -offset
                npairs = n - a
                g_avg[:npairs] += 1.0 - h[a:n]
                g_avg[a:n] -= h[:npairs]
        g_avg /= K
        g_dir = -g_avg
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        g_dir /= gn
        improved = False
        for step in [0.01, 0.02, 0.005, 0.015, 0.03, 0.008, 0.04, 0.003]:
            ht = project_feasible(h + step * g_dir, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-15:
                h = ht
                best_c5 = cn
                best_h = ht.copy()
                improved = True
                break
        if not improved:
            for step in [0.01, 0.02, 0.005, 0.015, 0.03]:
                ht = project_feasible(h - step * g_dir, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-15:
                    h = ht
                    best_c5 = cn
                    best_h = ht.copy()
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c5


def make_symmetric_init(n, rng, mode=0):
    x = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
    if mode == 0:
        h = 0.5 + 0.25 * np.sin(np.pi * x)
    elif mode == 1:
        h = 0.5 + 0.2 * np.cos(np.pi * x) + 0.1 * np.sin(2 * np.pi * x)
    elif mode == 2:
        h = 0.5 + 0.3 * np.sin(np.pi * x / 2)
    else:
        h = 0.5 + 0.15 * np.sin(np.pi * x) + 0.1 * np.sin(3 * np.pi * x)
    h = np.clip(h, 0.0, 1.0)
    h = project_feasible(h, n)
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 512
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n_hi = 1024
    h_hi = interp_to(best_h, best_n, n_hi)
    h_hi = project_feasible(h_hi, n_hi)
    c5_hi = compute_c5(h_hi)
    if c5_hi < best_c5:
        best_c5 = c5_hi
        best_h = h_hi.copy()
        best_n = n_hi

    h = best_h.copy()
    n = best_n

    # Phase 1: Fast coordinate descent (primary method)
    if remaining_time() > 60:
        h_cd, c5_cd = coordinate_descent_fast(h, n, min(80.0, remaining_time() * 0.4), rho=50.0)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h = h_cd.copy()

    # Phase 2: Smooth-max L-BFGS-B polish
    for beta in [2000, 5000, 10000]:
        if remaining_time() < 30:
            break
        h_new = lbfgsb_smooth(h, n, beta, 500, maxiter=2000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 3: More coordinate descent with different rho
    if remaining_time() > 40:
        h_cd2, c5_cd2 = coordinate_descent_fast(h, n, min(50.0, remaining_time() * 0.35), rho=20.0)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()
            h = h_cd2.copy()

    # Phase 4: Subgradient descent for broader exploration
    if remaining_time() > 30:
        h_sg, c5_sg = subgradient_descent(h, n, min(25.0, remaining_time() * 0.25), rng)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 5: Equalize top correlations
    if remaining_time() > 20:
        h_eq, c5_eq = equalize_top_k(h, n, K=8, max_iters=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 6: Coordinate descent with higher rho for constraint enforcement
    if remaining_time() > 30:
        h_cd3, c5_cd3 = coordinate_descent_fast(h, n, min(40.0, remaining_time() * 0.3), rho=100.0)
        if c5_cd3 < best_c5:
            best_c5 = c5_cd3
            best_h = h_cd3.copy()
            h = h_cd3.copy()

    # Phase 7: SLSQP polish
    for beta in [5000, 10000, 20000]:
        if remaining_time() < 20:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 8: Try alternative initializations
    if remaining_time() > 80:
        for mode in range(4):
            if remaining_time() < 20:
                break
            h_alt = make_symmetric_init(n, rng, mode=mode)
            h_alt = coordinate_descent_fast(h_alt, n, min(20.0, remaining_time() * 0.2), rho=50.0)[0]
            c5_alt = compute_c5(h_alt)
            if c5_alt < best_c5:
                best_c5 = c5_alt
                best_h = h_alt.copy()
                h = h_alt.copy()

    # Phase 9: High-resolution attempt (n=2048)
    if remaining_time() > 100 and best_n < 2048:
        n_vh = 2048
        h_vh = interp_to(best_h, best_n, n_vh)
        h_vh = project_feasible(h_vh, n_vh)
        c5_vh = compute_c5(h_vh)
        if c5_vh < best_c5 or c5_vh < best_c5 + 0.005:
            h_vh, c5_vh = coordinate_descent_fast(h_vh, n_vh, min(60.0, remaining_time() * 0.4), rho=50.0)
            if remaining_time() > 20:
                h_vh = lbfgsb_smooth(h_vh, n_vh, 5000, 500, maxiter=1000)
            c5_vh = compute_c5(h_vh)
            if c5_vh < best_c5:
                best_c5 = c5_vh
                best_h = h_vh.copy()
                best_n = n_vh

    # Phase 10: Final coordinate descent polish
    if remaining_time() > 20:
        h_final, c5_final = coordinate_descent_fast(h, n, min(30.0, remaining_time() * 0.5), rho=50.0)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Phase 11: Final SLSQP with very high beta
    if remaining_time() > 10:
        h_new = slsqp_smooth(best_h, best_n, 30000, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
