def lbfgsb_optimize(h, n, beta, penalty, maxiter=3000):
    def obj_fun(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        sm = (m + np.log(np.sum(ex)) / beta) * (2.0 / n)
        s = np.sum(x) - n / 2.0
        return sm + 0.5 * penalty * s * s
    
    def grad_fun(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        for k in range(len(corr)):
            wk = w[k] * (2.0 / n)
            if wk < 1e-15:
                continue
            g += wk * grad_corr_k(x, k, n)
        s = np.sum(x) - n / 2.0
        g += penalty * s
        return g
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fun, h, jac=grad_fun, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-22,

                                'factr': 1e12, 'm': 30})

    except Exception:
        pass
    return h

def subgradient_bb(h, n, time_limit, K=20, beta_sm=100.0):
    """Subgradient with Barzilai-Borwein step size."""
    global _T0, _BUDGET
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    it = 0
    h_prev = h.copy()
    g_prev = None
    
    while time.time() - start < time_limit:
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        it += 1
        
        corr = compute_corr(h)
        cov = np.max(corr) * (2.0 / n)
        
        # Find top K active shifts
        idx = np.argsort(-corr)[:K]
        c_top = corr[idx]
        weights = np.exp(beta_sm * (c_top - c_top.max()))
        s = weights.sum()
        weights /= s
        
        # Compute subgradient
        g = np.zeros(n)
        for i in range(K):
            if weights[i] > 1e-14:
                g += weights[i] * grad_corr_k(h, idx[i], n)
        
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        
        # BB step size
        if g_prev is not None:
            s_vec = h - h_prev
            y_vec = g - g_prev
            sy = np.dot(s_vec, y_vec)
            yy = np.dot(y_vec, y_vec)
            ss = np.dot(s_vec, s_vec)
            if yy > 1e-20 and sy > 1e-20:
                step = ss / sy
            elif ss > 1e-20:
                step = ss / yy if yy > 1e-20 else 0.01
            else:
                step = 0.01
            step = np.clip(step, 0.001, 0.1)
        else:
            step = 0.02
        
        h_prev = h.copy()
        g_prev = g.copy()
        
        h_new = project_feasible(h - step * g, n)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            # Try smaller steps
            found = False
            for factor in [0.5, 0.25, 0.1]:
                h_try = project_feasible(h - step * factor * g, n)
                c_try = compute_c5(h_try)
                if c_try < best_c5:
                    best_c5 = c_try
                    best_h = h_try.copy()
                    h = h_try
                    found = True
                    break
            if not found and it % 50 == 0:
                h = best_h.copy()
    
    return best_h, best_c5

def cutting_plane_lp(h, n, K=50, delta=0.03):
    corr = compute_corr(h)
    c_max = float(np.max(corr)) * (2.0 / n)
    idx = np.argsort(-corr)[:K]
    
    from scipy.optimize import linprog
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub_list = []
    b_ub_list = []
    for k in idx:
        g = grad_corr_k(h, k, n)
        fk = corr[k] * (2.0 / n)
        row = np.zeros(n + 1)
        row[:n] = g * (2.0 / n)
        row[-1] = -1.0
        A_ub_list.append(row)
        b_ub_list.append(fk - np.dot(g * (2.0 / n), h))
    
    # Trust region
    I = np.eye(n)
    A_trust = np.vstack([np.hstack([I, np.zeros((n, 1))]),
                         np.hstack([-I, np.zeros((n, 1))])])
    b_trust = np.concatenate([h + delta, -h + delta])
    
    A_ub = np.vstack([A_ub_list, A_trust])
    b_ub = np.concatenate([b_ub_list, b_trust])
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0, 1)] * n + [(None, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return h.copy(), c_max

def coordinate_descent(h, n, time_limit):
    global _T0, _BUDGET
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    
    while time.time() - start < time_limit:
        rem = _BUDGET - (time.time() - _T0)
        if rem < 3:
            break
        improved = False
        for i in range(n):
            if time.time() - start > time_limit:
                break
            old_val = h[i]
            # Try ternary-like search
            candidates = [0.0, 1.0]
            if 0.01 < old_val < 0.99:
                candidates.extend([old_val - 0.05, old_val + 0.05, 0.5])
            candidates = [c for c in candidates if 0 <= c <= 1 and abs(c - old_val) > 1e-10]
            
            best_c = best_c5
            best_v = old_val
            for cv in candidates:
                h[i] = cv
                h_proj = project_feasible(h, n)
                c = compute_c5(h_proj)
                if c < best_c:
                    best_c = c
                    best_v = cv
                h[i] = old_val
            
            if best_v != old_val:
                h[i] = best_v
                h = project_feasible(h, n)
                best_c5 = compute_c5(h)
                improved = True
        
        if not improved:
            break
    
    return h, best_c5

        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        if npairs > 0:
            g[offset:n] += (1.0 - h[:npairs])
            g[:npairs] -= h[offset:n]
    else:
        a = -offset
        npairs = n - a
        if npairs > 0:
            g[:npairs] += (1.0 - h[a:n])
            g[a:n] -= h[:npairs]
    return g


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)


def _smooth_obj(x, n, beta, penalty):
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(beta * sh, -50.0, 50.0)
    ex = np.exp(cl)
    sm = (m + np.log(np.sum(ex)) / beta) * (2.0 / n)
    s = np.sum(x) - n / 2.0
    return sm + 0.5 * penalty * s * s


def _smooth_grad(x, n, beta, penalty):
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(beta * sh, -50.0, 50.0)
    ex = np.exp(cl)
    s_ex = np.sum(ex)
    w = ex / s_ex
    g = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k] * (2.0 / n)
        if wk < 1e-14:
            continue
        g += wk * grad_corr_k(x, k, n)
    s = np.sum(x) - n / 2.0
    g += penalty * s
    return g


def lbfgsb_opt(h, n, beta, penalty, maxiter=3000, m_lbfgs=40):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, method='L-BFGS-B',
                       bounds=bounds, args=(n, beta, penalty),
                       options={'maxiter': maxiter, 'ftol': 1e-22,
                                'gtol': 1e-18, 'maxfun': 200000, 'm': m_lbfgs})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def slsqp_opt(h, n, beta, maxiter=500):
    def _obj_slsqp(x):
        return _smooth_obj(x, n, beta, 0.0)

    def _grad_slsqp(x):
        return _smooth_grad(x, n, beta, 0.0)

    def _con_fun(x):
        return np.sum(x) - n / 2.0

    def _con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_obj_slsqp, h, jac=_grad_slsqp, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def subgradient_descent(h, n, time_limit, K=25, beta_sm=100.0):
    global _T0, _BUDGET
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    it = 0
    h_prev = h.copy()
    g_prev = None

    while time.time() - start < time_limit:
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        it += 1
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:K]
        c_top = corr[idx]
        weights = np.exp(beta_sm * (c_top - c_top.max()))
        s = weights.sum()
        weights /= s
        g = np.zeros(n)
        for i in range(K):
            if weights[i] > 1e-14:
                g += weights[i] * grad_corr_k(h, idx[i], n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        if g_prev is not None:
            s_vec = h - h_prev
            y_vec = g - g_prev
            sy = np.dot(s_vec, y_vec)
            ss = np.dot(s_vec, s_vec)
            yy = np.dot(y_vec, y_vec)
            if sy > 1e-20:
                step = ss / sy
            elif yy > 1e-20:
                step = ss / yy
            else:
                step = 0.01
            step = np.clip(step, 0.002, 0.08)
        else:
            step = 0.015
        h_prev = h.copy()
        g_prev = g.copy()
        h_new = project_feasible(h - step * g, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            found = False
            for factor in [0.5, 0.25, 0.1, 0.05]:
                h_try = project_feasible(h - step * factor * g, n)
                c_try = compute_c5(h_try)
                if c_try < best_c5:
                    best_c5 = c_try
                    best_h = h_try.copy()
                    h = h_try
                    found = True
                    break
            if not found and it % 100 == 0:
                h = best_h.copy()
    return best_h, best_c5


def cutting_plane_lp(h, n, K=50, delta=0.02):
    dx = 2.0 / n
    corr = compute_corr(h)
    c_max = float(np.max(corr)) * dx
    idx = np.argsort(-corr)[:K]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub_list = []
    b_ub_list = []
    for k in idx:
        g = grad_corr_k(h, k, n)
        fk = corr[k] * dx
        row = np.zeros(n + 1)
        row[:n] = g * dx
        row[-1] = -1.0
        A_ub_list.append(row)
        b_ub_list.append(fk - np.dot(g * dx, h))
    I = np.eye(n)
    A_trust = np.vstack([np.hstack([I, np.zeros((n, 1))]),
                         np.hstack([-I, np.zeros((n, 1))])])
    b_trust = np.concatenate([h + delta, -h + delta])
    A_ub = np.vstack([A_ub_list, A_trust])
    b_ub = np.concatenate([b_ub_list, b_trust])
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0, 1)] * n + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return h.copy(), c_max
