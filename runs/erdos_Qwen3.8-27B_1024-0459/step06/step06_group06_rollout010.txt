Actually, let me simplify even further. Since g(t) = max_k (a_k t + b_k) is convex and piecewise linear, its minimum on [0,1] is at:
- t=0 if g'(0+) ≥ 0 (all active slopes at t=0 are ≥ 0)
- t=1 if g'(1-) ≤ 0 (all active slopes at t=1 are ≤ 0)
- Otherwise, at a breakpoint where the slope changes from negative to positive

The breakpoints are where a_k t + b_k = a_{k'} t + b_{k'} for some k, k'.

But finding all breakpoints is expensive. Ternary search is simpler and good enough.

Let me just do a simple version: evaluate g at t=0, t=1, and a few points in between, pick the best. Then refine with a few bisection steps.

OK let me just write the code now. I'll keep it practical.

Let me also think about the binary snapping strategy. After continuous optimization:
1. Identify coordinates with h[i] > 0.95 or h[i] < 0.05
2. Snap them to 1 or 0
3. Fix the sum constraint by adjusting the remaining "free" coordinates
4. This reduces the effective dimensionality
5. Then do local optimization only on the free coordinates

This could be very effective if the optimal solution is nearly binary.

_BUDGET = 1300.0
_OPT = {'n': 0, 'beta': 1000.0, 'penalty': 100.0, 'top_K': 50}


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        g[:npairs] += (1.0 - h[offset:offset + npairs])
        g[offset:offset + npairs] -= h[:npairs]
    else:
        a = -offset
        npairs = n - a
        g[a:n] += (1.0 - h[:npairs])
        g[:npairs] -= h[a:n]
    return g


def smooth_max_and_grad(h, n, beta, top_K=None):
    corr = compute_corr(h)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(beta * sh, -50.0, 50.0)
    ex = np.exp(cl)
    s = np.sum(ex)
    obj = (m + np.log(s) / beta) * (2.0 / n)
    w = ex / s
    g = np.zeros(n)
    wc = w * (2.0 / n)
    if top_K is not None and top_K < len(corr):
        top_idx = np.argsort(-wc)[:top_K]
        for ki in top_idx:
            wk = wc[ki]
            if wk > 1e-14:
                g += wk * grad_corr_k(h, ki, n)
    else:
        for k in range(len(corr)):
            wk = wc[k]
            if wk > 1e-14:
                g += wk * grad_corr_k(h, k, n)
    return obj, g


def _opt_fun(x):

    beta = _OPT['beta']
    penalty = _OPT['penalty']
    top_K = _OPT['top_K']
    obj, _ = smooth_max_and_grad(x, n, beta, top_K)
    s = np.sum(x) - n / 2.0
    return obj + 0.5 * penalty * s * s


def _opt_grad(x):
    n = _OPT['n']
    beta = _OPT['beta']
    penalty = _OPT['penalty']
    top_K = _OPT['top_K']
    _, g = smooth_max_and_grad(x, n, beta, top_K)
    s = np.sum(x) - n / 2.0
    g = g + penalty * s
    return g
