def binary_sa(h, n, time_limit, rng):

    n_ones = int(np.sum(h))

    best_c5 = compute_c5(h)
    best_h = h.copy()
    current_c5 = best_c5
    T = 0.5
    cool_rate = 0.9998
    no_improve = 0
    max_no_improve = 5000
    while time.time() - start < time_limit and remaining_time() > 10:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            current_c5 = new_c5
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
        T *= cool_rate
        if no_improve > max_no_improve:
            break
    return best_h, best_c5

def subgradient_topk(h, n, top_k=7, max_iters=5000, lr0=0.5):
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(max_iters):
        if remaining_time() < 5:
            break
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        for ki in idx_top:
            off = ki - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:] += (1.0 - h[:np_])
                g[:np_] -= h[off:]
            else:
                a = -off
                np_ = n - a
                g[:np_] += (1.0 - h[a:])
                g[a:] -= h[:np_]
        g *= dx
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        d = -g / gn
        lr = lr0 / (1.0 + 0.005 * it)
        step = lr * dx
        ht = project_feasible(h + step * d, n)
        c5_new = compute_c5(ht)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = ht.copy()
            h = ht.copy()
        else:
            h = project_feasible(h + 0.3 * step * d, n)
    return best_h, best_c5

    max_no_improve = 8000
    while time.time() - start < time_limit and remaining_time() > 10:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            current_c5 = new_c5
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
        T *= cool_rate
        if no_improve > max_no_improve:
            break
    return best_h, best_c5


def subgradient_topk(h, n, top_k=7, max_iters=5000, lr0=0.4):
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(max_iters):
        if remaining_time() < 5:
            break
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        for ki in idx_top:
            off = ki - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:] += (1.0 - h[:np_])
                g[:np_] -= h[off:]
            else:
                a = -off
                np_ = n - a
                g[:np_] += (1.0 - h[a:])
                g[a:] -= h[:np_]
        g *= dx
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        d = -g / gn
        lr = lr0 / (1.0 + 0.003 * it)
        step = lr * dx
        ht = project_feasible(h + step * d, n)
        c5_new = compute_c5(ht)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = ht.copy()
            h = ht.copy()
        else:
            h = project_feasible(h + 0.4 * step * d, n)
    return best_h, best_c5


def perturb_and_optimize(h, n, rng, noise_scale, beta, maxiter):
    h_pert = np.clip(h + rng.randn(n) * noise_scale, 0.0, 1.0)
    h_pert = project_feasible(h_pert, n)
    h_opt = slsqp_opt(h_pert, n, beta, maxiter=maxiter)
    h_opt, c5_opt = topk_equalize(h_opt, n, top_k=7, max_iters=300)
    return h_opt, c5_opt


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _CUR_N, _SM_BETA
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = np.zeros(384)
        best_h[:192] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP at parent resolution with adaptive beta
    h = best_h.copy()
    for beta in [2000, 20000, 200000, 2000000]:
        if remaining_time() < 30:
            break
        h_new = slsqp_opt(h, best_n, beta, maxiter=800)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Top-k equalization
    if remaining_time() > 30:
        h_eq, c5_eq = topk_equalize(h, best_n, top_k=9, max_iters=1000)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 3: Multi-resolution search (key improvement: try higher res first)
    resolutions_to_try = [1024, 1280, 640, 512, 2048]
    resolutions_to_try = [r for r in resolutions_to_try if r % 2 == 0 and 128 <= r <= 4096]

    for n_t in resolutions_to_try:
        if remaining_time() < 50:
            break
        if n_t == best_n:
            continue
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)

        h_opt = h_feas
        for beta in [5000, 50000, 500000]:
            if remaining_time() < 20:
                break
            h_opt = slsqp_opt(h_opt, n_t, beta, maxiter=500)

        if remaining_time() > 15:
            h_opt, c5_opt = topk_equalize(h_opt, n_t, top_k=9, max_iters=600)
        else:
            c5_opt = compute_c5(h_opt)

        if c5_opt < best_c5 - 1e-14:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t
            h = h_opt.copy()

    # Phase 4: Subgradient at best resolution
    if remaining_time() > 60:
        h_sg, c5_sg = subgradient_topk(h, best_n, top_k=9, max_iters=5000, lr0=0.5)
        if c5_sg < best_c5 - 1e-14:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 5: Intensive SLSQP with very high beta (key for tightening)
    for beta in [2000000, 20000000, 100000000]:
        if remaining_time() < 20:
            break
        h_new = slsqp_opt(h, best_n, beta, maxiter=1200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 6: Top-k with wider active set
    if remaining_time() > 20:
        h_eq, c5_eq = topk_equalize(h, best_n, top_k=13, max_iters=1200)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 7: Binary SA for diversification
    if remaining_time() > 120:
        h_bin = (best_h > 0.5).astype(float)
        n_half = best_n // 2
        n_ones = int(np.sum(h_bin))
        if n_ones != n_half:
            sort_idx = np.argsort(-h_bin)
            h_bin = np.zeros(best_n)
            h_bin[sort_idx[:n_half]] = 1.0

        sa_time = min(100, max(20, int(remaining_time() * 0.18)))
        h_b, c5_b = binary_sa(h_bin, best_n, sa_time, rng)

        # Continuous refinement
        h_b_c = project_feasible(h_b, best_n)
        for beta in [20000, 200000, 2000000]:
            if remaining_time() < 15:
                break
            h_b_c = slsqp_opt(h_b_c, best_n, beta, maxiter=600)
        c5_bc = compute_c5(h_b_c)
        if remaining_time() > 15:
            h_b_c, c5_bc = topk_equalize(h_b_c, best_n, top_k=9, max_iters=600)

        if c5_bc < best_c5 - 1e-14:
            best_c5 = c5_bc
            best_h = h_b_c.copy()
            h = h_b_c.copy()
        if c5_b < best_c5 - 1e-14:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 8: Randomized perturbation restarts
    if remaining_time() > 60:
        for scale in [0.05, 0.02, 0.01]:
            if remaining_time() < 25:
                break
            h_p, c5_p = perturb_and_optimize(best_h, best_n, rng, scale, 1000000, 500)
            if c5_p < best_c5 - 1e-14:
                best_c5 = c5_p
                best_h = h_p.copy()
                h = h_p.copy()

    # Phase 9: Coordinate descent
    if remaining_time() > 40:
        h_cd, c5_cd = coord_descent(best_h.copy(), best_n, max_sweeps=50)
        if c5_cd < best_c5 - 1e-14:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 10: Final SLSQP polish
    if remaining_time() > 15:
        h_final = slsqp_opt(best_h, best_n, 200000000, maxiter=800)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-14:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Phase 11: Final top-k
    if remaining_time() > 10:
        h_eq, c5_eq = topk_equalize(best_h, best_n, top_k=15, max_iters=600)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Phase 12: One more subgradient pass
    if remaining_time() > 20:
        h_sg2, c5_sg2 = subgradient_topk(best_h, best_n, top_k=11, max_iters=3000, lr0=0.3)
        if c5_sg2 < best_c5 - 1e-14:
            best_c5 = c5_sg2
            best_h = h_sg2.copy()

    # Final feasibility and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
