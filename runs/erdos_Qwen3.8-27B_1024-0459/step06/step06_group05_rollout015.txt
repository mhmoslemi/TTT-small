```python
import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_slsqp_params = {}


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        g[offset:] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:]
    else:
        a = -offset
        npairs = n - a
        g[:npairs] += (1.0 - h[a:])
        g[a:] -= h[:npairs]
    return g


def smooth_max_val_grad(x, n, K, beta):
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    m = np.max(c_top)
    sh = np.clip(beta * (c_top - m), -50.0, 50.0)
    ex = np.exp(sh)
    s = ex.sum()
    if s < 1e-30:
        w = np.ones(K) / K
    else:
        w = ex / s
    val = (m + np.log(s) / beta) * (2.0 / n)
    dx_val = 2.0 / n
    g = np.zeros(n)
    for i in range(K):
        wk = w[i] * dx_val
        if wk < 1e-15:
            continue
        g += wk * grad_corr_k(x, idx[i], n)
    return val, g


def _slsqp_obj(x):
    return smooth_max_val_grad(x, _slsqp_params['n'], _slsqp_params['K'], _slsqp_params['beta'])[0]


def _slsqp_grad(x):
    return smooth_max_val_grad(x, _slsqp_params['n'], _slsqp_params['K'], _slsqp_params['beta'])[1]


def _slsqp_cfun(x):
    return np.sum(x) - _slsqp_params['n'] / 2.0


def _slsqp_cjac(x):
    return np.ones(_slsqp_params['n'])


def slsqp_run(h, n, K, beta, maxiter=400):
    _slsqp_params['n'] = n
    _slsqp_params['K'] = K
    _slsqp_params['beta'] = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cfun, 'jac': _slsqp_cjac}]
    c5_old = compute_c5(h)
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new <= c5_old + 1e-12:
                return h_new
    except Exception:
        pass
    return h


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def remaining_budget():
    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)


def active_set_refine(h, n, time_limit):
    h_curr = h.copy()
    h_best = h.copy()
    c5_best = compute_c5(h)
    start = time.time()
    for phase in range(20):

        if remaining_budget() < 10:
            break
        corr = compute_corr(h_curr)
        c_max = np.max(corr)
        thresh = c_max - 0.003 * c_max
        active = np.where(corr >= thresh)[0]
        if len(active) < 3:
            active = np.argsort(-corr)[:10]
        active = active[np.argsort(-corr[active])]
        K = min(len(active), 25)
        beta = max(2000, 100 * K)

def projected_subgrad(h, n, time_limit, K=20, beta=300):

    h_prev = h.copy()

            break
        if remaining_budget() < 3:
            break
        if it == 0:
            y = h_curr.copy()
        else:
            t = float(it - 1)
            gamma = (np.sqrt(1.0 + 4.0 * t * t) - 1.0) / (2.0 * t) if t > 0 else 0.5
            y = h_curr + gamma * (h_curr - h_prev)
            y = np.clip(y, 0.0, 1.0)
        _, g_y = smooth_max_val_grad(y, n, K, beta)
        gn = np.linalg.norm(g_y)
        if gn < 1e-15:
            break
        step = min(0.2 / gn, 0.3)
        h_new = project_feasible(y - step * g_y, n)
        c5_new = compute_c5(h_new)
        h_prev = h_curr.copy()
        h_curr = h_new
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_curr.copy()
    return h_best, c5_best

def try_symmetry(h, n):
    """Try the symmetrized version h(x) = (h(x) + 1 - h(2-x))/2."""

    h_sym = project_feasible(h_sym, n)
    c5_sym = compute_c5(h_sym)
    return h_sym, c5_sym
