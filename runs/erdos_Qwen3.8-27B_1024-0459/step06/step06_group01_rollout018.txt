from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_CUR_N = 0
_SM_BETA = 1000.0


def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def _sm_objective(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = beta * (corr - m)
    z = np.clip(z, -50.0, 50.0)
    ez = np.exp(z)
    return (m + np.log(np.sum(ez)) / beta) * dx


def _sm_gradient(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = beta * (corr - m)
    z = np.clip(z, -50.0, 50.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    return g


def _eq_fun(x):
    return np.sum(x) - _CUR_N / 2.0


def _eq_jac(x):
    return np.ones(_CUR_N)


def slsqp_opt(h, n, beta, maxiter=500):
    global _CUR_N, _SM_BETA
    _CUR_N = n
    _SM_BETA = beta
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(_sm_objective, h0, jac=_sm_gradient, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.fun < _sm_objective(h0) - 1e-14:

    except Exception:
        pass
    return h0

    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def binary_local_search(h, n, time_limit, rng):
    h = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        sort_idx = np.argsort(-h)
        h = np.zeros(n)
        h[sort_idx[:n_half]] = 1.0

    start = time.time()
    corr = compute_corr(h)
    best_c5 = float(np.max(corr) * 2.0 / n)
    best_h = h.copy()
    current_c5 = best_c5
    T = 0.5
    cool_rate = 0.9997
    no_improve = 0
    max_no_improve = 5000
    n_corr = 2 * n - 1
    off_mid = n - 1

    while time.time() - start < time_limit and remaining_time() > 10:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break

        # Choose move type
        r = rng.rand()
        if r < 0.7:
            # Single swap
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            # Compute effect on all correlations
            # corr[k] = sum_i h[i+k']*(1-h[i]) where k' = k-(n-1)
            # Flip h[i]: 1->0, h[j]: 0->1
            # For shift s = k - (n-1):
            #   original contribution from x=i: h[i]*(1-h[i+s]) = 1*(1-h[i+s]) if i+s in range
            #   original contribution from x where x+s=i: h[i-s]*(1-h[i]) = h[i-s]*0 = 0
            #   new contribution from x=i: 0
            #   new contribution from x where x+s=i: h[i-s]*1
            #   original from x=j: h[j]*(1-h[j+s]) = 0
            #   original from x where x+s=j: h[j-s]*(1-h[j]) = h[j-s]*1
            #   new from x=j: 1*(1-h[j+s])
            #   new from x where x+s=j: h[j-s]*0 = 0
            # Delta for shift s:
            #   from i: 0 - (1-h[i+s]) = -(1-h[i+s])  [if 0<=i+s<n]
            #   from i: +h[i-s]  [if 0<=i-s<n, i.e., x=i-s, x+s=i]
            #   from j: +(1-h[j+s])  [if 0<=j+s<n]
            #   from j: -h[j-s]  [if 0<=j-s<n]

            # Vectorized: compute delta for all shifts at once
            s_arr = np.arange(-(n-1), n)  # shift values
            # For position i (1->0):
            #   -(1-h[i+s]) where 0<=i+s<n, plus h[i-s] where 0<=i-s<n
            vals = np.zeros(n_corr)
            # i+s in [0, n-1] => s in [-i, n-1-i]
            s_lo_i = max(-(n-1), -i)
            s_hi_i = min(n-1, n-1-i)
            if s_lo_i <= s_hi_i:
                idx_lo = s_lo_i + (n-1)
                idx_hi = s_hi_i + (n-1)
                vals[idx_lo:idx_hi+1] -= (1.0 - h[i+s_lo_i:i+s_hi_i+1])
            # i-s in [0, n-1] => s in [i-(n-1), i]
            s_lo_i2 = max(-(n-1), i-(n-1))
            s_hi_i2 = min(n-1, i)
            if s_lo_i2 <= s_hi_i2:
                idx_lo2 = s_lo_i2 + (n-1)
                idx_hi2 = s_hi_i2 + (n-1)
                vals[idx_lo2:idx_hi2+1] += h[i+s_lo_i2:i+s_hi_i2+1]
            # j+s in [0, n-1] => s in [-j, n-1-j]
            s_lo_j = max(-(n-1), -j)
            s_hi_j = min(n-1, n-1-j)
            if s_lo_j <= s_hi_j:
                idx_lo = s_lo_j + (n-1)
                idx_hi = s_hi_j + (n-1)
                vals[idx_lo:idx_hi+1] += (1.0 - h[j+s_lo_j:j+s_hi_j+1])
            # j-s in [0, n-1] => s in [j-(n-1), j]
            s_lo_j2 = max(-(n-1), j-(n-1))
            s_hi_j2 = min(n-1, j)
            if s_lo_j2 <= s_hi_j2:
                idx_lo2 = s_lo_j2 + (n-1)
                idx_hi2 = s_hi_j2 + (n-1)
                vals[idx_lo2:idx_hi2+1] -= h[j+s_lo_j2:j+s_hi_j2+1]

            new_corr = corr + vals
            new_max = float(np.max(new_corr))
            new_c5 = new_max * 2.0 / n
            delta = new_c5 - current_c5

            if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
                h[i] = 0.0
                h[j] = 1.0
                corr = new_corr
                current_c5 = new_c5
                if new_c5 < best_c5 - 1e-15:
                    best_c5 = new_c5
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1

        elif r < 0.85 and n > 100:
            # Segment reversal
            a = rng.randint(0, n - 10)
            b = a + rng.randint(10, min(80, n - a))
            b = min(b, n)
            # Reverse h[a:b]
            seg = h[a:b].copy()
            h[a:b] = seg[::-1]
            new_corr = compute_corr(h)
            new_c5 = float(np.max(new_corr)) * 2.0 / n
            delta = new_c5 - current_c5
            if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
                corr = new_corr
                current_c5 = new_c5
                if new_c5 < best_c5 - 1e-15:
                    best_c5 = new_c5
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[a:b] = seg  # undo
                no_improve += 1

        else:
            # Double swap
            if len(ones) >= 2 and len(zeros) >= 2:
                idx1 = rng.choice(len(ones), size=2, replace=False)
                idx2 = rng.choice(len(zeros), size=2, replace=False)
                i1, i2 = ones[idx1[0]], ones[idx1[1]]
                j1, j2 = zeros[idx2[0]], zeros[idx2[1]]
                h[i1] = 0; h[i2] = 0
                h[j1] = 1; h[j2] = 1
                new_corr = compute_corr(h)
                new_c5 = float(np.max(new_corr)) * 2.0 / n
                delta = new_c5 - current_c5
                if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
                    corr = new_corr
                    current_c5 = new_c5
                    if new_c5 < best_c5 - 1e-15:
                        best_c5 = new_c5
                        best_h = h.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                else:
                    h[i1] = 1; h[i2] = 1
                    h[j1] = 0; h[j2] = 0
                    no_improve += 1

        T *= cool_rate
        if no_improve > max_no_improve:
            # Reheat
            T = max(T, 0.3)
            no_improve = 0
            max_no_improve = int(max_no_improve * 1.3)

    return best_h, best_c5


def topk_equalize(h, n, top_k=5, max_iters=800):

        g_dir = np.zeros(n)
        for pos, k in enumerate(idx_top):
            w_k = 1.0 if pos == 0 else (-1.0 / (len(idx_top) - 1))
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_dir[off:] += w_k * (1.0 - h[:np_])

                a = -off

                g_dir[:np_] += w_k * (1.0 - h[a:])
                g_dir[a:] -= w_k * h[:np_]
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        d = g_dir / gn
        improved = False
        steps = [0.002, 0.005, 0.01, 0.02, 0.001, 0.008, 0.03, 0.05, 0.08, 0.12, 0.15, 0.2]

def coord_descent_focused(h, n, max_sweeps=40):
    """Coordinate descent focusing on positions that affect top correlations most."""
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        corr = compute_corr(h)
        top_idx = np.argsort(-corr)[:8]
        # Identify positions most in play
        in_play = set()
        for k in top_idx:
            off = k - (n - 1)
            if off >= 0:
                in_play.update(range(off, n))
                in_play.update(range(0, n - off))
            else:
                a = -off
                in_play.update(range(0, n - a))
                in_play.update(range(a, n))
        play_list = sorted(in_play)[:min(len(in_play), n)]
        if len(play_list) == 0:
            play_list = list(range(n))
        any_improve = False
        for j in play_list:
            if remaining_time() < 3:
                break
            old_val = h[j]
            best_v = old_val
            best_obj = compute_c5(h)
            a, b = 0.0, 1.0
            for _ in range(25):
                m1 = a + (b - a) * 0.381966
                m2 = a + (b - a) * 0.618034
                h[j] = m1
                c1 = compute_c5(h)
                h[j] = m2
                c2 = compute_c5(h)
                if c1 < c2:
                    b = m2
                    h[j] = m1
                    if c1 < best_obj:
                        best_obj = c1
                        best_v = m1
                else:
                    a = m1
                    h[j] = m2
                    if c2 < best_obj:
                        best_obj = c2
                        best_v = m2
            h[j] = best_v
            curr_c5 = compute_c5(h)
            if curr_c5 < best_c5 - 1e-13:
                best_c5 = curr_c5
                any_improve = True
        h = project_feasible(h, n)
        actual = compute_c5(h)
        if actual < best_c5:
            best_c5 = actual
        if not any_improve and sweep > 3:
            break
    return h, best_c5


def theta_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)

            sort_idx = np.argsort(-h_bin)
            h_bin = np.zeros(n_big)
            h_bin[sort_idx[:n_half]] = 1.0

        sa_time = min(500, max(30, int(remaining_time() * 0.35)))
        h_b, c5_b = binary_local_search(h_bin, n_big, sa_time, rng)
        if c5_b < best_c5 - 1e-14:
            best_c5 = c5_b
            best_h = h_b.copy()
            best_n = n_big

        # Continuous refinement of binary solution
        if remaining_time() > 100:
            h_cont = project_feasible(h_b, n_big)
            for beta in [2000, 20000, 200000]:
                if remaining_time() < 30:
                    break
                h_cont = slsqp_opt(h_cont, n_big, beta, maxiter=800)
                c5_c = compute_c5(h_cont)
                if c5_c < best_c5 - 1e-14:
                    best_c5 = c5_c
                    best_h = h_cont.copy()
                    best_n = n_big

            if remaining_time() > 30:
                h_eq, c5_eq = topk_equalize(h_cont, n_big, top_k=5, max_iters=500)
                if c5_eq < best_c5 - 1e-14:
                    best_c5 = c5_eq
                    best_h = h_eq.copy()
                    best_n = n_big

            if remaining_time() > 30:
                h_cd, c5_cd = coord_descent_focused(best_h.copy(), best_n, max_sweeps=30)
                if c5_cd < best_c5 - 1e-14:
                    best_c5 = c5_cd
                    best_h = h_cd.copy()

    # Phase 3: Try n=960 binary search
    n_mid = 960
    if remaining_time() > 200:
        h_interp960 = interp_to(best_h, best_n, n_mid)
        h_bin960 = (h_interp960 > 0.5).astype(float)
        n_half960 = n_mid // 2
        n_ones960 = int(np.sum(h_bin960))
        if n_ones960 != n_half960:
            sort_idx = np.argsort(-h_bin960)
            h_bin960 = np.zeros(n_mid)
            h_bin960[sort_idx[:n_half960]] = 1.0

        sa_time960 = min(300, max(20, int(remaining_time() * 0.25)))
        h_b960, c5_b960 = binary_local_search(h_bin960, n_mid, sa_time960, rng)
        if c5_b960 < best_c5 - 1e-14:
            best_c5 = c5_b960
            best_h = h_b960.copy()
            best_n = n_mid

        if remaining_time() > 60:
            h_c960 = project_feasible(h_b960, n_mid)
            for beta in [5000, 50000]:
                if remaining_time() < 20:
                    break
                h_c960 = slsqp_opt(h_c960, n_mid, beta, maxiter=600)
            c5_c960 = compute_c5(h_c960)
            h_c960, c5_c960 = topk_equalize(h_c960, n_mid, top_k=5, max_iters=400)
            if c5_c960 < best_c5 - 1e-14:
                best_c5 = c5_c960
                best_h = h_c960.copy()
                best_n = n_mid

    # Phase 4: Alternative starting points at n=1280
    if remaining_time() > 150:
        # Thue-Morse start
        h_tm = (theta_morse(n_big) > 0.5).astype(float)
        sa_tm = min(200, max(20, int(remaining_time() * 0.15)))
        h_tm_b, c5_tm_b = binary_local_search(h_tm, n_big, sa_tm, rng)
        if c5_tm_b < best_c5 - 1e-14:
            best_c5 = c5_tm_b
            best_h = h_tm_b.copy()
            best_n = n_big

    # Phase 5: Random binary start at n=1280
    if remaining_time() > 100:
        h_rand = random_binary(n_big, rng)
        sa_r = min(150, max(15, int(remaining_time() * 0.12)))
        h_r, c5_r = binary_local_search(h_rand, n_big, sa_r, rng)
        if c5_r < best_c5 - 1e-14:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_big

    # Phase 6: Final continuous polish
    if remaining_time() > 40:
        h_final = project_feasible(best_h, best_n)
        for beta in [100000, 500000, 2000000]:
            if remaining_time() < 15:
                break
            h_final = slsqp_opt(h_final, best_n, beta, maxiter=800)
            c5_f = compute_c5(h_final)
            if c5_f < best_c5 - 1e-14:
                best_c5 = c5_f
                best_h = h_final.copy()

    if remaining_time() > 20:
        h_eq2, c5_eq2 = topk_equalize(best_h.copy(), best_n, top_k=4, max_iters=400)
        if c5_eq2 < best_c5 - 1e-14:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()

    # Phase 7: Final coordinate descent
    if remaining_time() > 20:
        h_cd2, c5_cd2 = coord_descent_focused(best_h.copy(), best_n, max_sweeps=25)
        if c5_cd2 < best_c5 - 1e-14:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
