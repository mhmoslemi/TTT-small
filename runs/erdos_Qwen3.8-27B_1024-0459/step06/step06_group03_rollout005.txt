def sqp_step(h, n, K, delta):
    """One SQP iteration for the min-max problem."""
    corr = compute_corr(h)
    top_idx = np.argsort(-corr)[:K]
    
    # Build the QP subproblem
    # Variables: d (n) + t (1) = n+1 variables
    # Minimize: t
    # Subject to: 
    #   grad_k · d + corr_k - t <= 0 for k in top_idx
    #   -delta <= d_i <= delta for all i
    #   (sum(h) + sum(d)) = n/2 (equality)
    #   0 <= h_i + d_i <= 1
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = []
    b_ub = []
    for k in top_idx:
        g = grad_corr_k(h, k)
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_ub.append(row)
        b_ub.append(-corr[k])
    
    # Trust region: ||d||_inf <= delta
    I = np.eye(n)
    A_trust = np.vstack([np.hstack([I, np.zeros((n, 1))]),
                          np.hstack([-I, np.zeros((n, 1))])])
    b_trust = np.concatenate([delta * np.ones(n), delta * np.ones(n)])
    
    # Box constraints: 0 <= h_i + d_i <= 1 => -h_i <= d_i <= 1 - h_i
    # These are handled by bounds in linprog
    
    A_ub_full = np.vstack([np.array(A_ub), A_trust])
    b_ub_full = np.concatenate([np.array(b_ub), b_trust])
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0 - np.sum(h)])
    
    bounds = [(-h[i], 1.0 - h[i])] * n + [(None, None)]
    
    res = linprog(c_obj, A_ub=A_ub_full, b_ub=b_ub_full, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        d = res.x[:n]
        return d, res.x[-1]
    return None, None

        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_ub_list.append(row)
        b_ub_list.append(-corr[k_idx])
    I = np.eye(n)
    A_trust = np.vstack([np.hstack([I, np.zeros((n, 1))]),
                         np.hstack([-I, np.zeros((n, 1))])])
    b_trust = np.concatenate([delta * np.ones(n), delta * np.ones(n)])
    A_ub_full = np.vstack([np.array(A_ub_list), A_trust])
    b_ub_full = np.concatenate([np.array(b_ub_list), b_trust])
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0 - np.sum(h)])
    bounds = [(-h[i], 1.0 - h[i])] * n + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub_full, b_ub=b_ub_full, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return h.copy(), float(np.max(corr))


def sqp_main_loop(h, n, max_iters, delta_init):
    h = h.copy()
    best_c5 = compute_c5(h)
    delta = delta_init
    start = time.time()
    for it in range(max_iters):
        rem = _G_BUDGET - (time.time() - _G_T0)
        if rem < 5:
            break
        if time.time() - start > 600:
            break
        corr = compute_corr(h)
        c_max = np.max(corr)
        top_K = np.argsort(-corr)[:30]
        d, t_lp = sqp_trust(h, n, top_K, delta)
        h_new = project_feasible(h + d, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            h = h_new.copy()
            best_c5 = c5_new
            if c5_new < t_lp * (2.0 / n) * 0.9:
                delta = min(delta * 1.5, 0.5)
            else:
                delta = min(delta * 1.2, 0.5)
        else:
            for factor in [0.5, 0.25, 0.1]:
                h_try = project_feasible(h + factor * d, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    h = h_try.copy()
                    best_c5 = c5_try
                    delta = min(delta * 1.1, 0.5)
                    break
            else:
                delta = max(delta * 0.5, 1e-6)
                if delta < 1e-5:
                    break
    return h, best_c5


def equalize_top_shifts(h, n, K, max_iters):
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        rem = _G_BUDGET - (time.time() - _G_T0)
        if rem < 5:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:K]
        c_vals = corr[idx]
        if c_vals.max() - c_vals.min() < 1e-12:
            break
        g_avg = np.zeros(n)
        for k in idx:
            g_avg += grad_corr_k(h, k)
        g_avg /= K
        g_dir = -g_avg
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        g_dir /= gn
        improved = False
        for step in [0.02, 0.04, 0.01, 0.005, 0.06, 0.003, 0.03, 0.08, 0.015, 0.05]:
            ht = project_feasible(h + step * g_dir, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def subgradient_descent(h, n, time_limit):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        rem = _G_BUDGET - (time.time() - _G_T0)
        if rem < 5:
            break
        progress = min(1.0, it / 400.0)
        K = 30 if progress < 0.3 else (20 if progress < 0.6 else 10)
        beta = 30.0 if progress < 0.3 else (50.0 if progress < 0.6 else 80.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        sidx = np.argsort(-corr)[:K]
        c_top = corr[sidx]
        sh = c_top - c_top.max()
        cl = np.clip(beta * sh, -60.0, 60.0)
        ex = np.exp(cl)
        w = ex / ex.sum()
        g = np.zeros(n)
        for i in range(K):
            if w[i] < 1e-12:
                continue
            g += w[i] * grad_corr_k(h, sidx[i])
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        step = 0.05 * n / gn / (1.0 + 0.03 * it)
        step = min(step, 0.2)
        h_new = project_feasible(h - step * g, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            for factor in [0.5, 0.25, 0.1]:
                h_try = project_feasible(h - step * factor * g, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    break
        it += 1
    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def active_set_slsqp(h, n, K_list, maxiter):
    global _G_N, _G_KLIST
    _G_N = n
    _G_KLIST = K_list

    def tmin_obj(v):
        return v[-1]

    def tmin_jac(v):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def tmin_ineq(v):
        hv = v[:n]
        t = v[-1]
        corr = np.correlate(hv, 1.0 - hv, mode='full')
        return t * (n / 2.0) - corr[_G_KLIST]

    def tmin_ineq_jac(v):
        hv = v[:n]
        J = np.zeros((len(_G_KLIST), n + 1))
        for i, k in enumerate(_G_KLIST):
            offset = k - (n - 1)
            if offset >= 0:
                m = n - offset
                J[i, offset:] += (1.0 - hv[:m])
                J[i, :m] -= hv[offset:]
            else:
                a = -offset
                m = n - a
                J[i, :m] += (1.0 - hv[a:])
                J[i, a:] -= hv[:m]
            J[i, -1] = n / 2.0
        return J

    def t_eq_fun(v):
        return np.sum(v[:n]) - n / 2.0

    def t_eq_jac(v):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    t0 = compute_c5(h)
    v0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'ineq', 'fun': tmin_ineq, 'jac': tmin_ineq_jac},
        {'type': 'eq', 'fun': t_eq_fun, 'jac': t_eq_jac}
    ]
    try:
        res = minimize(tmin_obj, v0, jac=tmin_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None and res.x[-1] < t0 + 1e-12:
            h_new = project_feasible(res.x[:n], n)
            c5_new = compute_c5(h_new)
            if c5_new < compute_c5(h) + 1e-12:
                return h_new, c5_new
    except Exception:
        pass
    return h.copy(), compute_c5(h)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _G_BUDGET, _G_N, _G_BETA, _G_K, _G_PEN, _G_KLIST
    _T0 = time.time()
    _G_BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 512
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    def remaining():
        return max(0.0, _G_BUDGET - (time.time() - _T0) - 3.0)

    # Phase 1: Quick subgradient to establish basin
    if remaining() > 25:
        h, c5_t = subgradient_descent(h, n, min(20.0, remaining() * 0.06))
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h.copy()

    # Phase 2: L-BFGS-B smooth-max with increasing beta (fast, no equality constraint needed due to penalty)
    beta_schedule = [20, 50, 150, 500, 1500, 5000]
    for beta in beta_schedule:
        if remaining() < 10:
            break
        maxit = 80 if beta > 1000 else 120
        h = lbfgs_smooth_max(h, n, beta, K=40, maxiter=maxit, pen=1e5)
        c5_t = compute_c5(h)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h.copy()

    # Phase 3: SLSQP smooth-max (handles equality exactly)
    for beta in [100, 500, 2000, 8000]:
        if remaining() < 10:
            break
        maxit = 400 if beta < 500 else 300
        h = slsqp_smooth_max(h, n, beta, K=40, maxiter=maxit)
        c5_t = compute_c5(h)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h.copy()

    # Phase 4: SQP with trust region (main optimization)
    if remaining() > 60:
        h_sq, c5_sq = sqp_main_loop(h, n, max_iters=300, delta_init=0.03)
        if c5_sq < best_c5:
            best_c5 = c5_sq
            best_h = h_sq.copy()
            h = h_sq.copy()

    # Phase 5: Active-set SLSQP with dynamic K
    for K_a in [8, 12, 16, 20, 25]:
        if remaining() < 10:
            break
        corr = compute_corr(h)
        idx_a = np.argsort(-corr)[:K_a]
        h_am, c5_am = active_set_slsqp(h, n, idx_a, maxiter=250)
        if c5_am < best_c5:
            best_c5 = c5_am
            best_h = h_am.copy()
            h = h_am.copy()

    # Phase 6: Equalize top shifts
    if remaining() > 15:
        h_eq, c5_eq = equalize_top_shifts(h, n, K=10, max_iters=200)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 7: More SQP iterations
    if remaining() > 60:
        h_sq, c5_sq = sqp_main_loop(h, n, max_iters=200, delta_init=0.01)
        if c5_sq < best_c5:
            best_c5 = c5_sq
            best_h = h_sq.copy()
            h = h_sq.copy()

    # Phase 8: Higher resolution n=1024
    n_high = 1024
    if remaining() > 100:
        h_interp = interp_to(best_h, best_n, n_high)
        h_interp = project_feasible(h_interp, n_high)
        c5_interp = compute_c5(h_interp)

        # Subgradient at high res
        if remaining() > 40:
            h_h, c5_h = subgradient_descent(h_interp, n_high, min(20.0, remaining() * 0.1))
            if c5_h > c5_interp:
                h_h = h_interp.copy()
                c5_h = c5_interp

        # L-BFGS-B stages
        for beta in [50, 200, 800, 3000]:
            if remaining() < 15:
                break
            h_h = lbfgs_smooth_max(h_h, n_high, beta, K=40, maxiter=60, pen=1e5)
            c5_h = compute_c5(h_h)

        # SLSQP
        for beta in [500, 2000, 8000]:
            if remaining() < 12:
                break
            h_h = slsqp_smooth_max(h_h, n_high, beta, K=40, maxiter=250)
            c5_h = compute_c5(h_h)

        # SQP at high res
        if remaining() > 60:
            h_h, c5_h = sqp_main_loop(h_h, n_high, max_iters=200, delta_init=0.02)

        # Active set
        if remaining() > 20:
            corr_h = compute_corr(h_h)
            for K_a in [10, 15, 20]:
                if remaining() < 10:
                    break
                idx_a = np.argsort(-corr_h)[:K_a]
                h_h, c5_h = active_set_slsqp(h_h, n_high, idx_a, maxiter=200)

        # Equalize
        if remaining() > 12:
            h_h, c5_h = equalize_top_shifts(h_h, n_high, K=10, max_iters=150)

        if c5_h < best_c5:
            best_c5 = c5_h
            best_h = h_h.copy()
            best_n = n_high
            h = h_h.copy()
            n = n_high

    # Phase 9: Intensive refinement cycles
    n = best_n
    h = best_h.copy()
    for cycle in range(4):
        if remaining() < 25:
            break
        # SQP
        if remaining() > 40:
            h_s, c5_s = sqp_main_loop(h, n, max_iters=150, delta_init=0.008)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()
                h = h_s.copy()
        # SLSQP polish
        if remaining() > 15:
            for beta in [5000, 15000]:
                if remaining() < 8:
                    break
                h = slsqp_smooth_max(h, n, beta, K=50, maxiter=300)
            c5_t = compute_c5(h)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h.copy()
        # Active set
        if remaining() > 15:
            corr_c = compute_corr(h)
            for K_a in [10, 15, 20]:
                if remaining() < 8:
                    break
                idx_a = np.argsort(-corr_c)[:K_a]
                h, c5_t = active_set_slsqp(h, n, idx_a, maxiter=200)
                if c5_t < best_c5:
                    best_c5 = c5_t
                    best_h = h.copy()
        # Equalize
        if remaining() > 10:
            h, c5_t = equalize_top_shifts(h, n, K=12, max_iters=150)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h.copy()

    # Phase 10: Very fine L-BFGS-B polish
    h_f = best_h.copy()
    if remaining() > 10:
        for beta in [20000, 50000, 100000]:
            if remaining() < 5:
                break
            h_f = lbfgs_smooth_max(h_f, best_n, beta, K=60, maxiter=100, pen=1e7)
        c5_f = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 11: Micro-steps on top shifts
    h_t = best_h.copy()
    if remaining() > 8:
        for it in range(100):
            if remaining() < 3:
                break
            corr_t = compute_corr(h_t)
            idx_t = np.argsort(-corr_t)[:12]
            g_t = np.zeros(best_n)
            for k in idx_t:
                g_t += grad_corr_k(h_t, k)
            g_t /= len(idx_t)
            gn_t = np.linalg.norm(g_t)
            if gn_t < 1e-15:
                break
            improved = False
            for step in [0.003, 0.001, 0.002, 0.0005, 0.005, 0.0015]:
                h_try = project_feasible(h_t - step * g_t / gn_t, best_n)
                c_try = compute_c5(h_try)
                if c_try < best_c5 - 1e-15:
                    best_c5 = c_try
                    best_h = h_try.copy()
                    h_t = h_try.copy()
                    improved = True
                    break
            if not improved:
                break

    # Phase 12: Final SQP micro-iterations
    if remaining() > 8:
        h_fine, c5_fine = sqp_main_loop(best_h, best_n, max_iters=50, delta_init=0.002)
        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_fine.copy()

    # Final feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
