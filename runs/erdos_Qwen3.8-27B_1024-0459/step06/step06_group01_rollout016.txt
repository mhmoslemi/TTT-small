    for it in range(max_iters):
        if remaining_time() < 5:
            break
        dx = 2.0 / n
        corr = compute_corr(h_curr)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        
        # Check if top-k are well-equalized
        if c_max - corr[idx[-1]] < 1e-10:
            # Only optimize the max
            idx = np.argsort(-corr)[:1]
        
        g = np.zeros(n)
        nm1 = n - 1
        w = 1.0 / len(idx)
        for ki in idx:
            off = ki - nm1
            if off >= 0:
                np_ = n - off
                g[off:] += w * (1.0 - h_curr[:np_])
                g[:np_] -= w * h_curr[off:]
            else:
                a = -off
                np_ = n - a
                g[:np_] += w * (1.0 - h_curr[a:])
                g[a:] -= w * h_curr[:np_]
        g *= dx
        
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        
        lr = 0.01 / (1.0 + 0.01 * it)
        obj_curr = compute_c5(h_curr)
        improved = False
        for step_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            lr_t = lr * step_mult
            h_new = project_feasible(h_curr - lr_t * g, n)
            obj_new = compute_c5(h_new)
            if obj_new < obj_curr - 1e-15:
                h_curr = h_new
                improved = True
                if obj_new < best_c5:
                    best_c5 = obj_new
                    best_h = h_new.copy()
                break
        if not improved:
            # Try smaller steps
            for step_mult in [0.01, 0.005, 0.001]:
                lr_t = lr * step_mult
                h_new = project_feasible(h_curr - lr_t * g, n)
                obj_new = compute_c5(h_new)
                if obj_new < obj_curr - 1e-15:
                    h_curr = h_new
                    improved = True
                    if obj_new < best_c5:
                        best_c5 = obj_new
                        best_h = h_new.copy()
                    break
            if not improved:
                break
    
    return best_h, best_c5

def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)
