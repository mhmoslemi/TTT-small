def smooth_max_objective(x, n, beta, top_k, penalty_mu):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    # Top-k selection
    if top_k is not None and top_k < len(corr):
        idx = np.argsort(-corr)[:top_k]
        top_vals = corr[idx]
    else:
        top_vals = corr
    m = np.max(top_vals)
    z = beta * (top_vals - m)
    z = np.clip(z, -50, 50)
    ez = np.exp(z)
    smax = (m + np.log(np.sum(ez)) / beta) * dx
    # Penalty for constraint
    penalty = penalty_mu * (np.sum(x) - n/2.0)**2
    return smax + penalty

def smooth_max_gradient(x, n, beta, top_k, penalty_mu):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    if top_k is not None and top_k < len(corr):
        idx = np.argsort(-corr)[:top_k]
        top_vals = corr[idx]
        w = np.zeros(len(corr))
        w[idx] = np.exp(beta * (top_vals - top_vals[0]))
        w /= np.sum(w)
    else:
        m = np.max(corr)
        w = np.exp(beta * (corr - m))
        w /= np.sum(w)
    
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-14:
            continue
        wk = w[k] * dx
        off = k - (n-1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    
    # Penalty gradient
    g += 2 * penalty_mu * (np.sum(x) - n/2.0)
    
    return g

            for c in cands:
                h[i] = c
                h = project_feasible(h, n)
                c5_v = compute_c5(h)
                if c5_v < best_c - 1e-13:
                    best_c = c5_v
                    best_val = c
                    improved = True
            h[i] = best_val
            h = project_feasible(h, n)
        if not improved:
            break
    return h, best_c5

def gen_theta(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)

def gen_random_bin(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n//2, replace=False)] = 1.0
    return h
