def compute_corr(h):
    """Compute full correlation of h with 1-h."""
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(corr, n):
    """Compute C5 from correlation array."""
    return float(np.max(corr)) * (2.0 / n)

def project_binary(h, n):
    """Project to binary with exactly n/2 ones."""
    h_bin = (h > 0.5).astype(float)
    ones = int(h_bin.sum())
    target = n // 2
    if ones > target:
        idx = np.where(h_bin == 1)[0]
        np.random.shuffle(idx)
        h_bin[idx[:ones-target]] = 0
    elif ones < target:
        idx = np.where(h_bin == 0)[0]
        np.random.shuffle(idx)
        h_bin[idx[:target-ones]] = 1
    return h_bin

            h_new, c_new = binary_sa(h_tm_hi, target_n, min(100.0, rem() * 0.3), seed=seed + 600, K=10,
                                     T_start=8.0, T_min=0.01, cooling=0.99999)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()
                best_n = target_n

    # Phase 4: Continue SA at best resolution with different K values
    if rem() > 50:
        h_bin = to_binary(best_h, best_n, rng)
        # K=5 for sharper focus
        t_alloc = min(120.0, rem() * 0.3)
        h_new, c_new = binary_sa(h_bin, best_n, t_alloc, seed=seed + 900, K=5,
                                 T_start=3.0, T_min=0.005, cooling=0.999997)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Phase 5: K=20 for smoother landscape
    if rem() > 40:
        h_bin = to_binary(best_h, best_n, rng)
        t_alloc = min(100.0, rem() * 0.25)
        h_new, c_new = binary_sa(h_bin, best_n, t_alloc, seed=seed + 1100, K=20,
                                 T_start=6.0, T_min=0.02, cooling=0.999995)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Phase 6: Multi-restart SA
    if rem() > 60:
        h_bin = to_binary(best_h, best_n, rng)
        t_alloc = min(rem() - 10, 200.0)
        h_new, c_new = multi_restart_sa(h_bin, best_n, t_alloc, seed=seed + 1500, K=10, n_restarts=3)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Phase 7: Final targeted refinement
    if rem() > 30:
        h_bin = to_binary(best_h, best_n, rng)
        t_alloc = min(rem() - 5, 60.0)
        h_new, c_new = targeted_sa(h_bin, best_n, t_alloc, seed=seed + 2000, K=3)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Final: project and evaluate
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        best_c5 = min(best_c5, eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
