        a = -offset

        return v + pen_mult * eq_viol ** 2

        eq_viol = np.sum(x) - n / 2.0
        g += 2 * pen_mult * eq_viol * np.ones(n)
        return g

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        if npairs <= 0:
            return g
        g[offset:] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:]
    else:
        a = -offset
        npairs = n - a
        if npairs <= 0:
            return g
        g[:npairs] += (1.0 - h[a:])
        g[a:] -= h[:npairs]
    return g


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def smooth_max_val(h, n, K, beta):
    corr = compute_corr(h)
    dx = 2.0 / n
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    m = c_top[0]
    sh = np.clip(beta * (c_top - m), -400.0, 400.0)
    ex = np.exp(sh)
    return (m + np.log(ex.sum()) / beta) * dx, idx


def smooth_max_grad(h, n, K, beta):
    corr = compute_corr(h)
    dx = 2.0 / n
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    m = c_top[0]
    sh = np.clip(beta * (c_top - m), -400.0, 400.0)
    ex = np.exp(sh)
    w = ex / ex.sum()
    g = np.zeros(n)
    for i in range(K):
        if w[i] < 1e-15:
            continue
        g += (w[i] * dx) * grad_corr_k(h, idx[i], n)
    return g


def lbfgs_with_penalty(h, n, K, beta, maxiter, pen_eq, pen_ineq, active_idx):
    dx = 2.0 / n
    target = n / 2.0

    def obj(x):
        v, _ = smooth_max_val(x, n, K, beta)
        eq_v = np.sum(x) - target
        penalty = pen_eq * eq_v * eq_v
        if len(active_idx) > 0 and pen_ineq > 0:
            corr = compute_corr(x)
            c_act = corr[active_idx]
            c_max = np.max(corr)
            viol = np.maximum(0.0, c_act - c_max)
            penalty += pen_ineq * np.sum(viol * viol)
        return v + penalty

    def grad(x):
        g = smooth_max_grad(x, n, K, beta)
        eq_v = np.sum(x) - target
        g += 2.0 * pen_eq * eq_v * np.ones(n)
        if len(active_idx) > 0 and pen_ineq > 0:
            corr = compute_corr(x)
            c_act = corr[active_idx]
            c_max = np.max(corr)
            viol = np.maximum(0.0, c_act - c_max)
            gv = 2.0 * pen_ineq * np.sum(viol)
            if gv > 0:
                for i in range(len(active_idx)):
                    if viol[i] > 0:
                        g += 2.0 * pen_ineq * viol[i] * grad_corr_k(x, active_idx[i], n) * dx
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-14, 'iprint': 0,
                                'maxfun': maxiter * (n + 10)})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def slsqp_optimize(h, n, K, beta, maxiter):
    dx = 2.0 / n
    target = n / 2.0

    def obj(x):
        v, _ = smooth_max_val(x, n, K, beta)
        return v

    def grad(x):
        return smooth_max_grad(x, n, K, beta)

    def cfun(x):
        return np.sum(x) - target

    def cjac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            c5_old = compute_c5(h)
            if c5_new <= c5_old + 1e-10:
                return h_new
    except Exception:
        pass
    return h

def nesterov_subgrad(h, n, time_limit, K, beta):
    h_curr = h.copy()
    h_best = h.copy()
    c5_best = compute_c5(h)
    h_prev = h.copy()
    start = time.time()

    for it in range(200000):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 5:
            break

        if it == 0:
            y = h_curr.copy()
        else:
            t = it + 1
            gamma = np.sqrt(1.0 + 4.0 * t * t) / (2.0 * t) - 1.0
            y = h_curr + gamma * (h_curr - h_prev)
            y = np.clip(y, 0.0, 1.0)
            y = project_feasible(y, n)

        g = smooth_max_grad(y, n, K, beta)
        eq_v = np.sum(y) - n / 2.0
        g += eq_v / n * np.ones(n)

        gn = np.linalg.norm(g)
        if gn < 1e-16:
            break

        step = 0.8 / (gn * np.sqrt(1.0 + 0.1 * it))
        step = min(step, 0.3)

        h_new = project_feasible(y - step * g, n)
        c5_new = compute_c5(h_new)

        h_prev = h_curr.copy()
        h_curr = h_new

        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_curr.copy()

    return h_best, c5_best
