    # Phase 5: Random restart exploration
    if remaining() > 40:
        h_r, c5_r = random_restart_optimize(best_h, best_n, min(120, remaining() * 0.25),
                                             rng, K=15, beta=500)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            h = h_r.copy()
            n = best_n

    # Phase 6: Intensive SLSQP polish with very high beta
    for K in [5, 8, 10, 15, 20]:
        for beta in [30000, 100000, 300000]:
            if remaining() < 5:
                break
            h_new = slsqp_optimize(best_h, best_n, K, beta, maxiter=800)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 7: Final Adam with very sharp smooth max
    if remaining() > 20:
        h_fin, c5_fin = adam_optimize(best_h, best_n, min(40, remaining() * 0.3),
                                       K=8, beta=5000, lr=0.001)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Phase 8: One more SLSQP pass
    for beta in [100000, 500000]:
        if remaining() < 5:
            break
        h_new = slsqp_optimize(best_h, best_n, 8, beta, maxiter=1000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 9: Final local flip search
    if remaining() > 10:
        h_flip, c5_flip = local_flip_search(best_h, best_n, max_rounds=4)
        if c5_flip < best_c5:
            best_c5 = c5_flip
            best_h = h_flip.copy()

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
