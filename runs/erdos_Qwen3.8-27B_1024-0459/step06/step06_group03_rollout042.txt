        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 5:
            break

        g = np.zeros(n)
        for k in idx:
            g += grad_corr_k_vec(h, k, n)
        g /= len(idx)

        if g_prev is not None and x_prev is not None:
            ys = g - g_prev
            ss = h - x_prev
            yn = np.dot(ys, ys)
            yss = np.dot(ys, ss)
            if yn > 1e-20 and yss > 1e-15:
                alpha_new = np.dot(ss, ss) / yss
                alpha_new = max(0.05 * alpha, min(2.0 * alpha, alpha_new))
                alpha = alpha_new

        c5_curr = compute_c5(h)
        improved = False
        a = alpha
        for _ in range(30):
            h_try = project_feasible(h - a * g, n)
            c5_try = compute_c5(h_try)
            if c5_try < c5_curr - 1e-15:
                h = h_try
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
                improved = True
                alpha = a
                break
            a *= 0.5
            if a < 1e-14:
                break

        if not improved:
            stall += 5
            if stall > 100:
                break

        g_prev = g.copy()
        x_prev = h.copy()

    return best_h, best_c5


def equalize_and_descent(h_init, n, K, time_limit):
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    h = h_init.copy()
    start = time.time()

    for it in range(50000):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 5:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(-corr)[:K]
        c_top = corr[idx]
        c_spread = c_top.max() - c_top.min()

        if c_spread < 1e-12:
            break

        g = np.zeros(n)
        w = np.ones(K) / K
        for i in range(K):
            g += w[i] * grad_corr_k_vec(h, idx[i], n)

        c5_curr = compute_c5(h)
        improved = False
        for step in [0.03, 0.015, 0.05, 0.01, 0.005, 0.08, 0.003, 0.012]:
            h_try = project_feasible(h - step * g, n)
            c5_try = compute_c5(h_try)
            if c5_try < c5_curr - 1e-15:
                h = h_try
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                improved = True
                break

        if not improved:
            for step in [0.002, 0.001, 0.004]:
                h_try = project_feasible(h - step * g, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try
                    improved = True
                    break
            if not improved:
                break

    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def multi_beta_fista(h, n, time_limit, beta_list, K_list):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    for i, (beta, K) in enumerate(zip(beta_list, K_list)):
        rem = time_limit - (time.time() - start)
        if rem < 10:
            break
        frac = (i + 1) / len(beta_list)
        alloc = rem * (1.0 - frac) * 0.7
        alloc = max(alloc, 5.0)
        h_new, c5_new = fista_opt(h, n, beta, K, alloc)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            h = h.copy()
    return best_h, best_c5


def active_set_refine(h, n, time_limit, K_seq=None):
    if K_seq is None:
        K_seq = [5, 8, 12, 16, 20, 30, 40]
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    for K in K_seq:
        rem = time_limit - (time.time() - start)
        if rem < 10:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(-corr)[:K]
        alloc = rem * 0.4
        h_new, c5_new = proj_grad_active(h, n, idx, alloc)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 512
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    h = best_h.copy()
    n = best_n

    # Phase A: FISTA at original resolution (quick warm-up)
    if remaining() > 50:
        beta_seq = [20, 50, 100, 300, 800]
        K_seq = [20, 30, 40, 50, 60]
        h_a, c5_a = multi_beta_fista(h, n, min(40.0, remaining() * 0.06), beta_seq, K_seq)
        if c5_a < best_c5:
            best_c5 = c5_a
            best_h = h_a.copy()
            h = h_a.copy()

    # Phase B: Equalize at original resolution
    if remaining() > 40:
        h_b, c5_b = equalize_and_descent(h, n, K=10, time_limit=min(30.0, remaining() * 0.05))
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            h = h_b.copy()

    # Phase C: Interpolate to n=1024 and optimize
    n_mid = 1024
    if remaining() > 200:
        h_mid = interp_to(best_h, best_n, n_mid)
        h_mid = project_feasible(h_mid, n_mid)
        c5_mid = compute_c5(h_mid)
        if c5_mid <= best_c5 + 0.002:
            beta_seq_mid = [30, 80, 200, 500, 1500, 5000]
            K_seq_mid = [25, 35, 45, 55, 65, 80]
            h_mid, c5_mid = multi_beta_fista(h_mid, n_mid, min(150.0, remaining() * 0.2), beta_seq_mid, K_seq_mid)
            if c5_mid < best_c5:
                best_c5 = c5_mid
                best_h = h_mid.copy()
                best_n = n_mid
                h = h_mid.copy()
                n = n_mid

            if remaining() > 100:
                h_mid, c5_mid = active_set_refine(h, n_mid, min(100.0, remaining() * 0.15))
                if c5_mid < best_c5:
                    best_c5 = c5_mid
                    best_h = h_mid.copy()
                    best_n = n_mid
                    h = h_mid.copy()
                    n = n_mid

            if remaining() > 50:
                h_mid, c5_mid = equalize_and_descent(h, n, K=12, time_limit=min(40.0, remaining() * 0.08))
                if c5_mid < best_c5:
                    best_c5 = c5_mid
                    best_h = h_mid.copy()
                    best_n = n_mid
                    h = h_mid.copy()
                    n = n_mid

    # Phase D: Interpolate to n=2048 and optimize
    n_high = 2048
    if remaining() > 300:
        h_high = interp_to(best_h, best_n, n_high)
        h_high = project_feasible(h_high, n_high)
        c5_high = compute_c5(h_high)
        if c5_high <= best_c5 + 0.003:
            beta_seq_high = [20, 50, 100, 300, 800, 2000, 6000]
            K_seq_high = [20, 30, 40, 50, 60, 70, 80]
            h_high, c5_high = multi_beta_fista(h_high, n_high, min(250.0, remaining() * 0.3), beta_seq_high, K_seq_high)
            if c5_high < best_c5:
                best_c5 = c5_high
                best_h = h_high.copy()
                best_n = n_high
                h = h_high.copy()
                n = n_high

            if remaining() > 150:
                h_high, c5_high = active_set_refine(h, n, min(150.0, remaining() * 0.2), K_seq=[8, 12, 16, 20, 30, 40])
                if c5_high < best_c5:
                    best_c5 = c5_high
                    best_h = h_high.copy()
                    best_n = n_high
                    h = h_high.copy()
                    n = n_high

            if remaining() > 60:
                h_high, c5_high = equalize_and_descent(h, n, K=12, time_limit=min(50.0, remaining() * 0.1))
                if c5_high < best_c5:
                    best_c5 = c5_high
                    best_h = h_high.copy()
                    best_n = n_high
                    h = h_high.copy()
                    n = n_high

    # Phase E: SLSQP polish at best resolution
    n = best_n
    h = best_h.copy()
    if remaining() > 60 and n <= 2048:
        for beta, K, maxit in [(2000, 50, 200), (8000, 60, 200), (20000, 70, 150)]:
            if remaining() < 20:
                break
            h_new = slsqop_smmax(h, n, beta, K, maxit)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase F: Final active-set refinement at best resolution
    n = best_n
    h = best_h.copy()
    if remaining() > 40:
        h_f, c5_f = active_set_refine(h, n, min(60.0, remaining() * 0.3), K_seq=[8, 12, 16, 24, 32])
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            h = h_f.copy()

    # Phase G: Very fine equalization
    if remaining() > 20:
        h_g, c5_g = equalize_and_descent(h, n, K=15, time_limit=min(25.0, remaining() * 0.2))
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()
            h = h_g.copy()

    # Phase H: Tiny step refinement
    if remaining() > 10:
        h_t = best_h.copy()
        n_t = best_n
        for it in range(200):
            if remaining() < 3:
                break
            corr_t = np.correlate(h_t, 1.0 - h_t, mode='full')
            idx_t = np.argsort(-corr_t)[:12]
            g_t = np.zeros(n_t)
            for k in idx_t:
                g_t += grad_corr_k_vec(h_t, k, n_t)
            g_t /= len(idx_t)
            gn = np.linalg.norm(g_t)
            if gn < 1e-15:
                break
            improved = False
            for step in [0.003, 0.001, 0.005, 0.002, 0.0005, 0.008]:
                h_try = project_feasible(h_t - step * g_t / gn, n_t)
                c_try = compute_c5(h_try)
                if c_try < best_c5 - 1e-15:
                    best_c5 = c_try
                    best_h = h_try.copy()
                    h_t = h_try.copy()
                    improved = True
                    break
            if not improved:
                if it > 50:
                    break

    # Phase I: Try n=4096 if time permits
    n_vhigh = 4096
    if remaining() > 120 and best_n < n_vhigh:
        h_vh = interp_to(best_h, best_n, n_vhigh)
        h_vh = project_feasible(h_vh, n_vhigh)
        c5_vh = compute_c5(h_vh)
        if c5_vh <= best_c5 + 0.002:
            beta_vh = [30, 100, 400, 1500]
            K_vh = [25, 40, 55, 70]
            h_vh, c5_vh = multi_beta_fista(h_vh, n_vhigh, min(80.0, remaining() * 0.5), beta_vh, K_vh)
            if remaining() > 30:
                h_vh, c5_vh = active_set_refine(h_vh, n_vhigh, min(30.0, remaining() * 0.3), K_seq=[10, 16, 24])
            c5_vh = compute_c5(h_vh)
            if c5_vh < best_c5:
                best_c5 = c5_vh
                best_h = h_vh.copy()
                best_n = n_vhigh

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
