def project_feasible(h, n):

                                restart_sigma=0.002)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()

    # Phase 7: Perturbation multi-start
    if remaining_budget() > 40:
        n = best_n
        for sigma in [0.002, 0.004, 0.008]:
            if remaining_budget() < 20:
                break
            h_pert = project_feasible(best_h + rng.randn(n) * sigma, n)
            c5_pert = compute_c5(h_pert)
            if c5_pert < best_c5 + 0.006:
                h_pert, c5_pert = nesterov_bb(h_pert, n, min(30, remaining_budget() * 0.35),
                                              K_min=2, K_max=15, beta=30000, lr_scale=0.5)
                # Quick SLSQP polish
                h_pert = slsqp_refine(h_pert, n, 3, 50000, maxiter=250)
                c5_pert = compute_c5(h_pert)
                if c5_pert < best_c5:
                    best_c5 = c5_pert
                    best_h = h_pert.copy()

    # Phase 8: Final SLSQP polish with very high beta
    n = best_n
    for K in [2, 3]:
        for beta in [500000, 2000000]:
            if remaining_budget() < 4:
                break
            h_t = slsqp_refine(best_h, n, K, beta, maxiter=500)
            c5_t = compute_c5(h_t)
            if c5_t < best_c5 - 1e-14:
                best_c5 = c5_t
                best_h = h_t.copy()

    # Phase 9: Tiny Nesterov for final convergence
    if remaining_budget() > 15:
        h_t, c5_t = nesterov_bb(best_h, n, min(20, remaining_budget() * 0.6),
                                K_min=2, K_max=5, beta=500000, lr_scale=0.15,
                                restart_sigma=0.001)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
