def slsqp_smooth_max(h, n, beta, maxiter=400):
    dx = 2.0 / n
    h0 = project(h.copy(), n)

    def _obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        z = np.clip(beta * (corr - m), -50.0, 50.0)
        ez = np.exp(z)
        return (m + np.log(np.sum(ez)) / beta) * dx

    def _grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        z = np.clip(beta * (corr - m), -50.0, 50.0)
        ez = np.exp(z)
        S = np.sum(ez)
        w = ez / S
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k] * dx
            if wk < 1e-14:
                continue
            off = k - (n - 1)
            a = abs(off)
            if a == 0:
                g += wk * (1.0 - 2.0 * x)
            elif off > 0:
                g[a:] += wk * (1.0 - x[:n - a])
                g[:n - a] -= wk * x[a:]
            else:
                g[:n - a] += wk * (1.0 - x[a:])
                g[a:] -= wk * x[:n - a]
        return g

    def _eq(x):
        return np.sum(x) - n / 2.0

    def _eqj(x):
        return np.ones(n)

    from scipy.optimize import minimize
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq, 'jac': _eqj}]
    try:
        res = minimize(_obj, h0, jac=_grad, method='SLSQP',

    strategies.append(h)
    # Block structure
    h = np.zeros(n)
    block = n // 8
    for i in range(0, n, block * 2):
        h[i:i + block] = 1.0
    strategies.append(project(h, n))
    # Low freq sine threshold
    x = np.linspace(0, 4 * np.pi, n, endpoint=False)
    h = (np.sin(x) > 0).astype(float)
    strategies.append(project(h, n))
    # High freq
    x = np.linspace(0, 32 * np.pi, n, endpoint=False)
    h = (np.sin(x) > 0).astype(float)
    strategies.append(project(h, n))
    # Combined freqs
    x = np.linspace(0, 2 * np.pi, n, endpoint=False)
    h = ((np.sin(x) + 0.3 * np.sin(3 * x)) > 0).astype(float)
    strategies.append(project(h, n))
    # Thue-Morse
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    strategies.append(project(h, n))
    return strategies


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = np.zeros(640)
        best_h[:320] = 1.0
    best_n = len(best_h)
    best_h = project(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Projected subgradient from parent
    h = best_h.copy()
    h, c5 = projected_subgradient(h, best_n, max_iters=800, top_k=15, lr_init=0.05)
    if c5 < best_c5 - 1e-14:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: SLSQP with smooth max
    for beta in [5000, 50000, 500000]:
        if remaining_time() < 40:
            break
        h_new = slsqp_smooth_max(best_h, best_n, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 3: Binary local search
    if remaining_time() > 60:
        h_b, c5_b = binary_local_search(best_h, best_n, max_iters=15000)
        if c5_b < best_c5 - 1e-14:
            best_c5 = c5_b
            best_h = h_b.copy()
        # Refine binary with continuous
        if remaining_time() > 40:
            h_ref = project(h_b, best_n)
            h_ref, c5_ref = projected_subgradient(h_ref, best_n, max_iters=500, top_k=10, lr_init=0.03)
            if c5_ref < best_c5 - 1e-14:
                best_c5 = c5_ref
                best_h = h_ref.copy()

    # Phase 4: Multi-resolution
    for n_t in [best_n * 2, best_n * 3, best_n]:
        if n_t == best_n and best_n != 640:
            continue
        if remaining_time() < 100:
            break
        if n_t < 256 or n_t > 4096:
            continue
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project(h_interp, n_t)
        h_opt = h_feas
        for beta in [5000, 100000]:
            if remaining_time() < 40:
                break
            h_opt = slsqp_smooth_max(h_opt, n_t, beta, maxiter=300)
        h_opt, c5_opt = projected_subgradient(h_opt, n_t, max_iters=600, top_k=20, lr_init=0.03)
        if c5_opt < best_c5 - 1e-14:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t

    # Phase 5: Multi-start at parent resolution
    if remaining_time() > 200:
        starts = multi_start_h(best_n, rng)
        for s in starts:
            if remaining_time() < 80:
                break
            s_opt = project(s, best_n)
            s_opt, c5_s = projected_subgradient(s_opt, best_n, max_iters=400, top_k=12, lr_init=0.04)
            if remaining_time() > 30:
                s_opt = slsqp_smooth_max(s_opt, best_n, 100000, maxiter=200)
                c5_s = compute_c5(s_opt)
            if c5_s < best_c5 - 1e-14:
                best_c5 = c5_s
                best_h = s_opt.copy()

    # Phase 6: Final polish cycle
    if remaining_time() > 50:
        h = best_h.copy()
        h, c5 = projected_subgradient(h, best_n, max_iters=1000, top_k=25, lr_init=0.02, momentum=0.7)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()

    if remaining_time() > 30:
        h_new = slsqp_smooth_max(best_h, best_n, 2000000, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final subgradient with small lr
    if remaining_time() > 20:
        h, c5 = projected_subgradient(best_h, best_n, max_iters=2000, top_k=30, lr_init=0.01, momentum=0.8)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()

    best_h = project(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
