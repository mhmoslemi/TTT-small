def precompute_dcorr(h, n):
    nfull = 2 * n - 1
    dcorr = np.zeros((n, nfull))
    for k in range(nfull):
        m = k - (n - 1)
        if m >= 0:
            np_ = n - m
            dcorr[m:, k] += (1.0 - h[:np_])
            dcorr[:np_, k] -= h[m:m+np_]
        else:
            a = -m
            np_ = n - a
            dcorr[:np_, k] += (1.0 - h[a:a+np_])
            dcorr[a:, k] -= h[:np_]
    return dcorr
