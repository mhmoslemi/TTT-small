def binary_local_search(h_bin, n, time_limit=60.0):
    """Local search on binary patterns via swaps."""
    dx = 2.0 / n
    h = h_bin.astype(float).copy()
    best_c5 = np.max(np.correlate(h, 1-h, mode='full')) * dx
    start = time.time()
    
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1-h, mode='full')
        c_max = np.max(corr)
        
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        
        best_swap = None
        best_new_c = c_max
        
        # Try a random subset of swaps
        n_tries = min(500, len(zeros) * len(ones))
        for _ in range(n_tries):
            i = zeros[np.random.randint(len(zeros))]
            j = ones[np.random.randint(len(ones))]
            
            # Effect of setting h[i]=1, h[j]=0
            # For shift s: change in corr[n-1+s]
            # h[i] goes 0→1: adds (1-h[i+s]) to corr at shifts where i+s valid
            #                        subtracts h[i-s] from corr at shifts where i-s valid (as 1-h[j] term)
            # Wait, I need to be more careful.
            
            # corr[n-1+s] = sum_j h[j+s]*(1-h[j])
            # Changing h[i] from 0 to 1:
            #   As h[j+s] term (j+s=i, j=i-s): adds (1-h[i-s]) if 0≤i-s<n
            #   As (1-h[j]) term (j=i): h[i+s] * (1-1) - h[i+s]*(1-0) = -h[i+s] if 0≤i+s<n
            #   Net change: (1-h[i-s])*I(0≤i-s<n) - h[i+s]*I(0≤i+s<n)
            
            # Changing h[j] from 1 to 0:
            #   As h[j+s] term (j+s=j... wait, the "j" in the sum is the free variable)
            #   Let me use different notation.
            #   corr[n-1+s] = sum_{t} h[t+s]*(1-h[t])
            #   h[j] appears as:
            #     As h[t+s] when t+s=j, i.e., t=j-s: contributes h[j]*(1-h[j-s]) → becomes 0
            #       Change: -h[j]*(1-h[j-s]) = -(1-h[j-s]) if 0≤j-s<n (since h[j]=1)
            #     As (1-h[t]) when t=j: contributes h[j+s]*(1-h[j]) = 0 → becomes h[j+s]*1 = h[j+s]
            #       Change: +h[j+s] if 0≤j+s<n
            
            # Total change in corr[n-1+s]:
            # From h[i]: 0→1:
            #   + (1-h[i-s]) if 0≤i-s<n
            #   - h[i+s] if 0≤i+s<n
            # From h[j]: 1→0:
            #   - (1-h[j-s]) if 0≤j-s<n
            #   + h[j+s] if 0≤j+s<n
            
            # For the new max, we need to check all shifts.
            # This is O(n) per swap.
            
            new_corr = corr.copy()
            
            # h[i]: 0→1
            for s in range(-(n-1), n):
                k = n - 1 + s
                delta = 0.0
                if 0 <= i-s < n:
                    delta += (1 - h[i-s])
                if 0 <= i+s < n:
                    delta -= h[i+s]
                # h[j]: 1→0
                if 0 <= j-s < n:
                    delta -= (1 - h[j-s])
                if 0 <= j+s < n:
                    delta += h[j+s]
                new_corr[k] += delta
            
            new_c = np.max(new_corr)
            if new_c < best_new_c:
                best_new_c = new_c
                best_swap = (i, j)
        
        if best_swap is None or best_new_c >= c_max - 1e-14:
            break
        
        i, j = best_swap
        h[i] = 1.0
        h[j] = 0.0
        best_c5 = best_new_c * dx
    
    return h, best_c5

def swap_delta(h, i, j, n):
    """Compute the change in all correlations when h[i]:0→1 and h[j]:1→0."""
    delta = np.zeros(2*n-1)
    
    # h[i]: 0→1
    # For shift s (output index k = n-1+s):
    #   + (1-h[i-s]) if 0≤i-s<n  → s = i-t for t in [0,n-1], so k = n-1+i-t for t in [0,min(i,n-1)]
    #   - h[i+s] if 0≤i+s<n     → s = t-i for t in [0,n-1], so k = n-1+t-i for t in [i, min(i+n-1, n-1)]
    
    # Term 1: (1-h[i-s]) for 0≤i-s<n → let t=i-s, t in [max(0,i-(n-1)), min(i,n-1)]
    t_min = max(0, i-(n-1))
    t_max = min(i, n-1)
    if t_min <= t_max:
        t_range = np.arange(t_min, t_max+1)
        k_range = n-1 + i - t_range  # k = n-1+s = n-1+(i-t)
        delta[k_range] += (1 - h[t_range])
    
    # Term 2: -h[i+s] for 0≤i+s<n → let t=i+s, t in [max(0,i-(n-1)), min(i+n-1, n-1)]
    # Wait, i+s in [0,n-1], s = t-i, k = n-1+t-i
    t_min = max(0, i-(n-1))  # s >= -(n-1) → t-i >= -(n-1) → t >= i-(n-1)
    t_max = min(n-1, i+n-1)  # s <= n-1 → t-i <= n-1 → t <= i+n-1
    # Also need i+s >= 0 → t >= 0 (already satisfied if i>=0)
    # And i+s <= n-1 → t <= n-1
    t_range = np.arange(max(t_min, 0), min(t_max, n-1)+1)
    if len(t_range) > 0:
        k_range = n-1 + t_range - i
        delta[k_range] -= h[t_range]
    
    # h[j]: 1→0
    # Term 3: -(1-h[j-s]) for 0≤j-s<n
    t_min = max(0, j-(n-1))
    t_max = min(j, n-1)
    if t_min <= t_max:
        t_range = np.arange(t_min, t_max+1)
        k_range = n-1 + j - t_range
        delta[k_range] -= (1 - h[t_range])
    
    # Term 4: +h[j+s] for 0≤j+s<n
    t_range = np.arange(max(0, j-(n-1)), min(n-1, j+n-1)+1)
    if len(t_range) > 0:
        k_range = n-1 + t_range - j
        delta[k_range] += h[t_range]
    
    return delta

    def grad_fun(x):
        _, g = smooth_max_obj_grad(x, n, beta)
        s = np.sum(x) - n / 2.0
        g = g + penalty * s
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fun, h, jac=grad_fun, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-30,
                                'gtol': 1e-20, 'maxfun': 200000})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def to_binary(h, n):
    h = np.array(h, dtype=float).copy()
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:n // 2]] = 1.0
    return h_bin


def swap_delta(h, i, j, n):
    delta = np.zeros(2 * n - 1)
    # h[i]: 0 -> 1
    tA = np.arange(0, min(i + 1, n))
    if len(tA) > 0:
        kA = n - 1 + i - tA
        delta[kA] += (1.0 - h[tA])
    tB = np.arange(i, n)
    if len(tB) > 0:
        kB = n - 1 + tB - i
        delta[kB] -= h[tB]
    # h[j]: 1 -> 0
    tC = np.arange(0, min(j + 1, n))
    if len(tC) > 0:
        kC = n - 1 + j - tC
        delta[kC] -= (1.0 - h[tC])
    tD = np.arange(j, n)
    if len(tD) > 0:
        kD = n - 1 + tD - j
        delta[kD] += h[tD]
    return delta


def binary_local_search(h, n, time_limit=120.0, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    global _T0, _BUDGET
    dx = 2.0 / n
    h = h.copy()
    best_c5 = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
    start = time.time()
    stall = 0

    while time.time() - start < time_limit:
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)

        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) < 2 or len(ones) < 2:
            break

        best_swap = None
        best_improvement = 1e-15

        # Adaptive number of trials
        elapsed_local = time.time() - start
        if elapsed_local < time_limit * 0.3:
            n_tries = min(2000, len(zeros) * len(ones))
        elif elapsed_local < time_limit * 0.7:
            n_tries = min(1000, len(zeros) * len(ones))
        else:
            n_tries = min(500, len(zeros) * len(ones))

        tried = set()
        for _ in range(n_tries):
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            key = (i, j)
            if key in tried:
                continue
            tried.add(key)

            delta = swap_delta(h, i, j, n)
            new_max = np.max(corr + delta)
            improvement = c_max - new_max
            if improvement > best_improvement:
                best_improvement = improvement
                best_swap = (i, j, delta)

        if best_swap is None:
            stall += 1
            if stall > 10:
                break
            # Perturb: random swap even if no improvement (restart)
            if rng.random() < 0.3:
                i = zeros[rng.randint(len(zeros))]
                j = ones[rng.randint(len(ones))]
                h[i] = 1.0
                h[j] = 0.0
                corr = np.correlate(h, 1.0 - h, mode='full')
                c_max = np.max(corr)
                best_c5 = min(best_c5, c_max * dx)
            continue

        i, j, delta = best_swap
        h[i] = 1.0
        h[j] = 0.0
        corr = corr + delta
        c_max = np.max(corr)
        new_c5 = c_max * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
        stall = 0

    final_c5 = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
    best_c5 = min(best_c5, final_c5)
    return h, best_c5


def continuous_to_binary_round(h_cont, n, time_for_search=60.0, rng=None):
    global _T0, _BUDGET
    if rng is None:
        rng = np.random.RandomState(0)

    best_h = h_cont.copy()
    best_c5 = compute_c5(h_cont)

    for threshold in [0.5, 0.4, 0.6]:
        if time.time() - _T0 > _BUDGET - 30:
            break
        h_bin = to_binary(h_cont, n)
        # Fine-tune: for near-threshold values, try both assignments
        h_tuned = h_bin.copy()
        idx = np.argsort(-np.abs(h_cont - threshold))[:10]
        for i in idx:
            h_tuned[i] = 0.0
            h_t2 = h_tuned.copy()
            h_t2[i] = 1.0
            if np.sum(h_t2) == n / 2:
                c2 = compute_c5(h_t2)
                if c2 < compute_c5(h_tuned):
                    h_tuned = h_t2

        # Ensure exact count
        cnt = int(np.sum(h_tuned))
        if cnt > n // 2:
            excess = cnt - n // 2
            zero_candidates = np.where(h_tuned == 0)[0]
            if len(zero_candidates) >= excess:
                # Flip the ones with lowest h_cont values
                one_idx = np.where(h_tuned == 1)[0]
                flip = np.argsort(h_cont[one_idx])[:excess]
                h_tuned[one_idx[flip]] = 0.0
        elif cnt < n // 2:
            deficit = n // 2 - cnt
            zero_candidates = np.where(h_tuned == 0)[0]
            if len(zero_candidates) >= deficit:
                flip = np.argsort(-h_cont[zero_candidates])[:deficit]
                h_tuned[zero_candidates[flip]] = 1.0

        c_tuned = compute_c5(h_tuned)
        if c_tuned < best_c5:
            h_bin = h_tuned
            best_c5 = c_tuned
            best_h = h_tuned.copy()

        # Binary local search
        rem_search = min(time_for_search, _BUDGET - 30 - (time.time() - _T0))
        if rem_search > 10:
            h_bs, c_bs = binary_local_search(h_bin, n, time_limit=rem_search * 0.5, rng=rng)
            if c_bs < best_c5:
                best_c5 = c_bs
                best_h = h_bs.copy()

    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)


def slsqp_optimize(h, n, beta, maxiter=500):
    def obj_fun(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -60.0, 60.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * (2.0 / n)

    def grad_fun(x):
        return smooth_max_obj_grad(x, n, beta)[1]

    def sc_fun(x):
        return np.sum(x) - n / 2.0

    def sc_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sc_fun, 'jac': sc_jac}]
    try:
        res = minimize(obj_fun, h, jac=grad_fun, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def subgradient_refine(h, n, time_limit=30.0, K=15, beta_g=80.0):
    global _T0, _BUDGET
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()

    while time.time() - start < time_limit:
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(-corr)[:K]
        c_top = corr[idx]
        weights = np.exp(beta_g * (c_top - c_top.max()))
        s = weights.sum()
        if s < 1e-30:
            weights = np.ones(K) / K
        else:
            weights /= s

        g = np.zeros(n)
        for wi, k in zip(weights, idx):
            if wi < 1e-12:
                continue
            s_val = k - (n - 1)
            if s_val >= 0:
                g[s_val:n] += wi * (1.0 - h[:n - s_val])
                g[:n - s_val] -= wi * h[s_val:n]
            else:
                a = -s_val
                g[:n - a] += wi * (1.0 - h[a:n])
                g[a:n] -= wi * h[:n - a]

        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break

        for step in [0.02, 0.01, 0.005, 0.03, 0.008]:
            ht = project_feasible(h - step * g / gn, n)
            ct = compute_c5(ht)
            if ct < best_c5 - 1e-15:
                h = ht
                best_c5 = ct
                break

    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 512
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: Work at n=1024
    n1 = 1024
    h1 = interp_to(best_h, best_n, n1)
    h1 = project_feasible(h1, n1)
    c5_1 = compute_c5(h1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
        best_n = n1

    # Phase 2: L-BFGS-B continuous optimization at n=1024
    h1 = lbfgsb_optimize(h1, n1, beta=200, penalty=500, maxiter=2000)
    c5_1 = compute_c5(h1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
        best_n = n1

    if remaining_time() > 100:
        h1 = lbfgsb_optimize(h1, n1, beta=500, penalty=1000, maxiter=3000)
        c5_1 = compute_c5(h1)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()
            best_n = n1

    # Phase 3: Binary local search at n=1024
    if remaining_time() > 100:
        h_bin1, c5_bin1 = continuous_to_binary_round(h1, n1, time_for_search=90.0, rng=rng)
        if c5_bin1 < best_c5:
            best_c5 = c5_bin1
            best_h = h_bin1.copy()
            best_n = n1

    # Phase 4: Continuous refinement after binary
    if remaining_time() > 100:
        # Interpolate binary back to continuous with slight smoothing
        h1c = best_h.astype(float).copy()
        h1c = project_feasible(h1c, n1)
        h1c = lbfgsb_optimize(h1c, n1, beta=1000, penalty=2000, maxiter=2000)
        c5_1c = compute_c5(h1c)
        if c5_1c < best_c5:
            best_c5 = c5_1c
            best_h = h1c.copy()
            best_n = n1

    # Phase 5: Subgradient refinement
    if remaining_time() > 60:
        h1s, c5_1s = subgradient_refine(best_h, n1, time_limit=min(30.0, remaining_time() * 0.4), K=20, beta_g=100.0)
        if c5_1s < best_c5:
            best_c5 = c5_1s
            best_h = h1s.copy()
            best_n = n1

    # Phase 6: Work at n=2048
    n2 = 2048
    if remaining_time() > 200:
        h2 = interp_to(best_h, best_n, n2)
        h2 = project_feasible(h2, n2)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
            best_n = n2

        # L-BFGS-B at n=2048
        if remaining_time() > 150:
            h2 = lbfgsb_optimize(h2, n2, beta=200, penalty=500, maxiter=1500)
            c5_2 = compute_c5(h2)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()
                best_n = n2

        if remaining_time() > 120:
            h2 = lbfgsb_optimize(h2, n2, beta=500, penalty=1000, maxiter=2000)
            c5_2 = compute_c5(h2)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()
                best_n = n2

        # Binary local search at n=2048
        if remaining_time() > 100:
            h_bin2, c5_bin2 = continuous_to_binary_round(h2, n2, time_for_search=min(80.0, remaining_time() * 0.5), rng=rng)
            if c5_bin2 < best_c5:
                best_c5 = c5_bin2
                best_h = h_bin2.copy()
                best_n = n2

        # More continuous at n=2048
        if remaining_time() > 60:
            h2c = lbfgsb_optimize(best_h, n2, beta=1000, penalty=2000, maxiter=1500)
            c5_2c = compute_c5(h2c)
            if c5_2c < best_c5:
                best_c5 = c5_2c
                best_h = h2c.copy()
                best_n = n2

    # Phase 7: Try n=4096 if time permits
    n3 = 4096
    if remaining_time() > 200 and best_n < n3:
        h3 = interp_to(best_h, best_n, n3)
        h3 = project_feasible(h3, n3)
        c5_3 = compute_c5(h3)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
            best_n = n3

        if remaining_time() > 150:
            h3 = lbfgsb_optimize(h3, n3, beta=200, penalty=500, maxiter=1000)
            c5_3 = compute_c5(h3)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h3.copy()
                best_n = n3

        if remaining_time() > 100:
            h_bin3, c5_bin3 = continuous_to_binary_round(h3, n3, time_for_search=min(60.0, remaining_time() * 0.4), rng=rng)
            if c5_bin3 < best_c5:
                best_c5 = c5_bin3
                best_h = h_bin3.copy()
                best_n = n3

    # Phase 8: Final SLSQP polish
    if remaining_time() > 30:
        h_final = slsqp_optimize(best_h, best_n, beta=5000, maxiter=500)
        c5_f = compute_c5(h_final)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_final.copy()

    # Phase 9: Final subgradient
    if remaining_time() > 20:
        h_sg, c5_sg = subgradient_refine(best_h, best_n, time_limit=min(15.0, remaining_time() * 0.5), K=10, beta_g=150.0)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
