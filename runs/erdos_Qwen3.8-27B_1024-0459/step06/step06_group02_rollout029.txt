def compute_corr(h):

def compute_c5(h):

    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    """Project h onto {x : 0<=x<=1, sum(x)=n/2}"""
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            # All at bounds, need to move some from 0 or 1
            if diff > 0:
                # Need to increase sum, move some 0s up
                zeros = np.where(h < 1e-10)[0]
                if len(zeros) > 0:
                    h[zeros[0]] = min(1.0, diff)
                else:
                    break
            else:
                ones = np.where(h > 1.0 - 1e-10)[0]
                if len(ones) > 0:
                    h[ones[0]] = max(0.0, h[ones[0]] + diff)
                else:
                    break
        h = np.clip(h, 0.0, 1.0)
    return h

    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -700.0, 0.0)  # numerical stability
    ex = np.exp(shifted)
    s = np.sum(ex)
    return (m + np.log(s) / beta) * dx

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -700.0, 0.0)

For the gradient method, instead of SLSQP (which is a constrained optimizer), let me use a **primal-dual / augmented Lagrangian** approach:

The problem: min F(h) = smooth_max(h)

The box constraints 0<=h<=1 are handled by projection.

Algorithm:
1. Initialize h, λ=0, μ=1
2. Inner loop: minimize L(h, λ) over h (using projected gradient, with projection handling both box and the augmented term)
3. Update λ += μ * (sum(h) - n/2)
4. Sometimes increase μ
5. Repeat

The projection for the augmented Lagrangian is a bit tricky because of the quadratic penalty term. For small μ, the projection is approximately the same as the original.

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            adj = diff / cnt
            # Check if adjustment would push any out of bounds
            new_vals = h[mask] + adj
            if new_vals.min() >= -1e-15 and new_vals.max() <= 1.0 + 1e-15:
                h[mask] = new_vals
            else:
                # Partial adjustment
                if adj > 0:
                    room = 1.0 - h[mask]
                    total_room = room.sum()
                    if total_room < abs(diff):

                        h[mask] = 1.0

                        if len(zeros) > 0:

                            h[zeros] = np.clip(h[zeros] + per, 0, 1)
                    else:

        if l >= 0:

            g[:n-l] -= wi * h[l:]
        else:
            a = -l
            g[:n-a] += wi * one_minus_h[a:]
            g[a:] -= wi * h[:n-a]
    
    return g

def projected_gradient(h, n, beta, max_iter, time_limit, verbose=False):
    h = h.copy()
    start = time.time()
    best_f = smooth_max_val(h, beta)
    best_c5 = compute_c5(h)
    alpha = 0.5
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        f = smooth_max_val(h, beta)
        g = smooth_max_grad(h, beta)
        
        # Project gradient to be orthogonal to constraint normal
        g_proj = g - np.mean(g)
        
        gnorm = np.linalg.norm(g_proj)
        if gnorm < 1e-12:
            break
        
        # Backtracking line search
        accepted = False
        for bt in range(50):
            h_new = project_feasible(h - alpha * g_proj, n)
            f_new = smooth_max_val(h_new, beta)
            if f_new < f - 1e-4 * alpha * gnorm * gnorm:
                accepted = True
                break
            alpha *= 0.5
        
        if accepted:
            h = h_new
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_f = f_new
            alpha = min(alpha * 1.5, 5.0)
        else:
            alpha *= 0.5
            if alpha < 1e-10:
                break
        
    return h, best_c5

def binary_round(h, n, rng):
    """Round h to binary with exactly n/2 ones."""
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    num_ones = int(np.sum(h_bin))
    target = n // 2
    if num_ones > target:
        ones = np.where(h_bin == 1)[0]
        # Preferentially remove the ones that are "most isolated"
        flip = rng.choice(ones, num_ones - target, replace=False)
        h_bin[flip] = 0
    elif num_ones < target:
        zeros = np.where(h_bin == 0)[0]
        flip = rng.choice(zeros, target - num_ones, replace=False)
        h_bin[flip] = 1
    return h_bin

def flip_optimize(h, n, max_sweeps, time_limit, rng, start_temp=0.01):
    """Simulated annealing on binary flips."""
    h = binary_round(h, n, rng).copy()
    corr = compute_corr(h)
    current_max = np.max(corr)
    current_c5 = current_max * (2.0 / n)
    best_c5 = current_c5
    best_h = h.copy()
    
    dx = 2.0 / n
    temperature = start_temp
    lags = np.arange(-(n-1), n)
    
    start = time.time()
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        temperature *= 0.99
        order = rng.permutation(n)
        sweep_improved = False
        
        for i in order:
            if time.time() - start > time_limit:
                break
            
            # Compute effect of flipping position i
            old_val = h[i]
            delta = 1.0 - 2.0 * old_val  # +1 if 0->1, -1 if 1->0
            
            # Change in correlation
            i_minus_l = i - lags
            i_plus_l = i + lags
            
            change = np.zeros(2*n - 1)
            mask1 = (i_minus_l >= 0) & (i_minus_l < n)
            mask2 = (i_plus_l >= 0) & (i_plus_l < n)
            change[mask1] += delta * (1.0 - h[i_minus_l[mask1]])
            change[mask2] -= delta * h[i_plus_l[mask2]]
            
            new_corr = corr + change
            new_max = np.max(new_corr)
            new_c5 = new_max * dx
            
            # Accept/reject (simulated annealing)
            if new_c5 <= current_c5 or rng.random() < np.exp(-(new_c5 - current_c5) / max(temperature, 1e-10)):
                h[i] = 1.0 - old_val
                corr = new_corr
                current_max = new_max
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                    sweep_improved = True
                    no_improve = 0
            # else: reject, h[i] stays the same
            
        if not sweep_improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0
    
    return best_h, best_c5

def interp_up(h, n_low, n_high):
    """Interpolate h from n_low to n_high points."""
    x_old = (np.arange(n_low) + 0.5) * (2.0 / n_low)
    x_new = (np.arange(n_high) + 0.5) * (2.0 / n_high)
    return np.interp(x_new, x_old, h)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:

        else:
            if diff > 0:
                idx = np.argmin(h)
                h[idx] = min(1.0, h[idx] + diff)
            else:
                idx = np.argmax(h)
                h[idx] = max(0.0, h[idx] + diff)
        h = np.clip(h, 0.0, 1.0)

    num_ones = int(np.sum(h_b))
    target = n // 2
    if num_ones > target:
        ones = np.where(h_b == 1)[0]
        n_flip = num_ones - target
        flip_idx = rng.choice(ones, n_flip, replace=False)
        h_b[flip_idx] = 0.0
    elif num_ones < target:
        zeros = np.where(h_b == 0)[0]
        n_flip = target - num_ones
        flip_idx = rng.choice(zeros, n_flip, replace=False)
        h_b[flip_idx] = 1.0
    return h_b

def flip_anneal(h, n, max_sweeps, time_limit, rng, start_temp=0.008):
    h = to_binary(h, n, rng).copy()
    corr = compute_corr(h)
    dx = 2.0 / n
    current_c5 = float(np.max(corr)) * dx
    best_c5 = current_c5
    best_h = h.copy()
    lags = np.arange(-(n - 1), n).astype(np.int64)
    start = time.time()
    temp = start_temp
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        temp *= 0.992
        if temp < 1e-6:
            temp = 1e-6
        order = rng.permutation(n)
        sweep_improved = False
        
        for i in order:
            if time.time() - start > time_limit:
                break
            old_val = h[i]
            delta = 1.0 - 2.0 * old_val
            i_minus_l = i - lags
            i_plus_l = i + lags
            change = np.zeros(2 * n - 1)
            m1 = (i_minus_l >= 0) & (i_minus_l < n)
            m2 = (i_plus_l >= 0) & (i_plus_l < n)
            if np.any(m1):
                change[m1] += delta * (1.0 - h[i_minus_l[m1]])
            if np.any(m2):
                change[m2] -= delta * h[i_plus_l[m2]]
            new_corr = corr + change
            new_c5 = float(np.max(new_corr)) * dx
            
            if new_c5 <= current_c5 or (temp > 1e-8 and rng.random() < np.exp(-(new_c5 - current_c5) / temp)):
                h[i] = 1.0 - old_val
                corr = new_corr
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                    sweep_improved = True
        if sweep_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                break
    return best_h, best_c5

def swap_local_search(h, n, time_limit, rng):
    h = to_binary(h, n, rng).copy()
    corr = compute_corr(h)
    dx = 2.0 / n
    current_c5 = float(np.max(corr)) * dx
    best_c5 = current_c5
    best_h = h.copy()
    lags = np.arange(-(n - 1), n).astype(np.int64)
    start = time.time()
    
    for sweep in range(200):
        if time.time() - start > time_limit:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        num_pairs = min(len(ones) * len(zeros), 3000)
        
        for _ in range(num_pairs):
            if time.time() - start > time_limit:
                break
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
            change = np.zeros(2 * n - 1)
            # Position i: 1 -> 0 (delta = -1)
            iml = i - lags
            ipl = i + lags
            m1 = (iml >= 0) & (iml < n)
            m2 = (ipl >= 0) & (ipl < n)
            if np.any(m1):
                change[m1] += -(1.0 - h[iml[m1]])
            if np.any(m2):
                change[m2] -= -h[ipl[m2]]
            # Position j: 0 -> 1 (delta = +1)
            jml = j - lags
            jpl = j + lags
            m3 = (jml >= 0) & (jml < n)
            m4 = (jpl >= 0) & (jpl < n)
            if np.any(m3):
                change[m3] += (1.0 - h[jml[m3]])
            if np.any(m4):
                change[m4] -= h[jpl[m4]]
            
            new_corr = corr + change
            new_c5 = float(np.max(new_corr)) * dx
            if new_c5 < current_c5 - 1e-14:
                h[i] = 0.0
                h[j] = 1.0
                corr = new_corr
                current_c5 = new_c5
                improved = True
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
        if not improved:
            break
    return best_h, best_c5

def interp_up(h, n_low, n_high):
    x_old = (np.arange(n_low) + 0.5) * (2.0 / n_low)
    x_new = (np.arange(n_high) + 0.5) * (2.0 / n_high)
    return np.interp(x_new, x_old, h)

def multiplicative_update(h, n, time_limit):
    """Coordinated descent with multiplicative updates."""
    h = h.copy()
    start = time.time()
    best_c5 = compute_c5(h)
    corr = compute_corr(h)
    dx = 2.0 / n
    target = n / 2.0
    
    for sweep in range(500):
        if time.time() - start > time_limit:
            break
        improved = False
        for i in range(n):
            if time.time() - start > time_limit:
                break
            # Try small adjustments
            for dv in [0.001, -0.001, 0.005, -0.005, 0.01, -0.01]:
                new_val = h[i] + dv
                if new_val < 0 or new_val > 1:
                    continue
                # Update correlation incrementally
                old_val = h[i]
                delta = new_val - old_val
                if abs(delta) < 1e-10:
                    continue
                lags = np.arange(-(n - 1), n)
                change = np.zeros(2 * n - 1)
                iml = i - lags
                ipl = i + lags
                m1 = (iml >= 0) & (iml < n)
                m2 = (ipl >= 0) & (ipl < n)
                if np.any(m1):
                    change[m1] += delta * (1.0 - h[iml[m1]])
                if np.any(m2):
                    change[m2] -= delta * h[ipl[m2]]
                
                new_corr = corr + change
                new_max = float(np.max(new_corr))
                new_c5 = new_max * dx
                
                # Check sum constraint approximately
                new_sum = np.sum(h) + delta
                if abs(new_sum - target) > 0.1:
                    continue
                
                if new_c5 < best_c5 - 1e-12:
                    h[i] = new_val
                    corr = new_corr
                    best_c5 = new_c5
                    improved = True
                    break
        if not improved:
            # Polish sum constraint
            h = project_feasible(h, n)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
            break
    h = project_feasible(h, n)
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    
    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 8.0)
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
    except (NameError, Exception):
        parent_h = None
    
    if parent_h is not None and len(parent_h) > 0:
        n = len(parent_h)
        best_h = parent_h.copy()
    else:
        n = 1024
        best_h = rng.rand(n)
        best_h = project_feasible(best_h, n)
    
    best_c5 = compute_c5(best_h)
    best_n = n
    
    # Phase 1: Smooth-max gradient at parent resolution
    for beta in [500, 2000, 8000, 20000]:
        if remaining() < 25:
            break
        h_try, c_try = projected_gradient_step(best_h, best_n, beta, max_iter=600,
                                               time_limit=min(90, remaining() / 4))
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try
            best_n = best_n
    
    # Phase 2: Binary flip annealing at parent resolution
    if remaining() > 80:
        h_bin, c_bin = flip_anneal(best_h, best_n, max_sweeps=2000,
                                   time_limit=min(350, remaining() * 0.4),
                                   rng=rng, start_temp=0.01)
        if c_bin < best_c5:
            best_c5 = c_bin
            best_h = h_bin.copy()
    
    # Phase 3: Swap local search
    if remaining() > 50:
        h_swap, c_swap = swap_local_search(best_h, best_n,
                                           time_limit=min(150, remaining() * 0.3),
                                           rng=rng)
        if c_swap < best_c5:
            best_c5 = c_swap
            best_h = h_swap.copy()
    
    # Phase 4: Gradient refinement after binary search
    for beta in [5000, 20000, 80000]:
        if remaining() < 20:
            break
        h_try, c_try = projected_gradient_step(best_h, best_n, beta, max_iter=500,
                                               time_limit=min(50, remaining() / 3))
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try
    
    # Phase 5: Fine coordinate descent
    if remaining() > 40:
        h_cd, c_cd = multiplicative_update(best_h, best_n,
                                           time_limit=min(100, remaining() * 0.4))
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd
        # Re-round and do more flips
        if remaining() > 30:
            h_bin2, c_bin2 = flip_anneal(best_h, best_n, max_sweeps=1000,
                                         time_limit=min(80, remaining() * 0.5),
                                         rng=rng, start_temp=0.003)
            if c_bin2 < best_c5:
                best_c5 = c_bin2
                best_h = h_bin2
    
    # Phase 6: Higher resolution
    for n_high in [1024, 1536, 2048]:
        if remaining() < 80:
            break
        if n_high <= best_n:
            continue
        
        h_interp = interp_up(best_h, best_n, n_high)
        h_feas = project_feasible(h_interp, n_high)
        c_inter = compute_c5(h_feas)
        
        for beta in [1000, 5000, 20000]:
            if remaining() < 25:
                break
            h_new, c_new = projected_gradient_step(h_feas, n_high, beta, max_iter=400,
                                                   time_limit=min(60, remaining() / 3))
            if c_new < c_inter:
                c_inter = c_new
                h_feas = h_new
        
        if remaining() > 50:
            h_bin3, c_bin3 = flip_anneal(h_feas, n_high, max_sweeps=500,
                                         time_limit=min(100, remaining() * 0.4),
                                         rng=rng, start_temp=0.005)
            if c_bin3 < c_inter:
                c_inter = c_bin3
                h_feas = h_bin3
        
        if remaining() > 15:
            h_new2, c_new2 = projected_gradient_step(h_feas, n_high, 50000, max_iter=300,
                                                     time_limit=min(30, remaining() * 0.5))
            if c_new2 < c_inter:
                c_inter = c_new2
                h_feas = h_new2
        
        if c_inter < best_c5:
            best_c5 = c_inter
            best_h = h_feas.copy()
            best_n = n_high
    
    # Phase 7: Final polish at best resolution
    if remaining() > 15:
        h_final, c_final = projected_gradient_step(best_h, best_n, 100000, max_iter=500,
                                                   time_limit=min(60, remaining()))
        if c_final < best_c5:
            best_c5 = c_final
            best_h = h_final
    
    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    # Validate with provided evaluator
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
