def grad_corr_k(h, k, n):
    """Gradient of O(offset) with respect to h, where offset = k - (n-1)"""
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        m = n - offset
        g[offset:] += (1.0 - h[:m])
        g[:m] -= h[offset:]
    else:
        a = -offset
        m = n - a
        g[:m] += (1.0 - h[a:])
        g[a:] -= h[:m]
    return g

    idx = np.argsort(-corr)[:K]
    c = corr[idx]

    sh = c - c_max
    ex = np.exp(np.clip(beta * sh, -80, 80))
    val = (c_max + np.log(ex.sum()) / beta) * (2.0 / n)

def cutting_plane_step(h, n, K, delta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max_val = np.max(corr) * (2.0 / n)
    idx = np.argsort(-corr)[:K]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub_list = []
    b_ub_list = []
    for k in idx:
        g = grad_corr_k_vec(h, k, n)
        fk = np.dot(g, h)
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_ub_list.append(row)
        b_ub_list.append(max(fk, corr[k]) - 1e-10)
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    I = np.eye(n)
    A_trust = np.vstack([
        np.hstack([I, np.zeros((n, 1))]),
        np.hstack([-I, np.zeros((n, 1))])
    ])
    b_trust = np.concatenate([
        np.clip(h + delta, 0, 1),
        np.clip(-h + 1 + delta, 0, 1)
    ])
    A_ub = np.vstack([A_ub, A_trust])
    b_ub = np.concatenate([b_ub, b_trust])
    bounds = [(0, 1)] * n + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': max(1, _BUDGET - (time.time() - _T0) - 5)})
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return h.copy(), c_max_val


def subgradient_descent(h, n, time_limit):
    h = h.copy()
    best_h = h.copy()

        if rem < 3:
            break
        progress = min(1.0, it / 300.0)
        K = 25 if progress < 0.3 else (15 if progress < 0.6 else 10)
        beta = 50.0 if progress < 0.3 else (80.0 if progress < 0.6 else 150.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(-corr)[:K]
        c_top = corr[idx]
        sh = c_top - c_top.max()
        cl = np.clip(beta * sh, -60.0, 60.0)
        ex = np.exp(cl)
        w = ex / ex.sum()
        g = np.zeros(n)
        for i in range(K):
            if w[i] < 1e-12:
                continue
            g += w[i] * grad_corr_k_vec(h, idx[i], n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        base_step = 0.05 * n / gn
        base_step = min(base_step, 0.2)
        base_step /= (1.0 + 0.03 * it)
        h_new = project_feasible(h - base_step * g, n)

                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()

        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: Optimize at parent resolution
    h_opt, c5_opt = optimize_at_resolution(best_h, best_n, min(300.0, remaining() * 0.25))
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt.copy()

    # Phase 2: Try higher resolution n=1024
    n_high = 1024
    if remaining() > 200:
        h_interp = interp_to(best_h, best_n, n_high)
        h_interp = project_feasible(h_interp, n_high)
        c5_interp = compute_c5(h_interp)

        if c5_interp <= best_c5 + 0.002:
            h_high, c5_high = optimize_at_resolution(h_interp, n_high, min(400.0, remaining() * 0.4))
            if c5_high < best_c5:
                best_c5 = c5_high
                best_h = h_high.copy()
                best_n = n_high

    # Phase 3: Try even higher resolution n=2048
    n_vhigh = 2048
    if remaining() > 200:
        h_interp2 = interp_to(best_h, best_n, n_vhigh)
        h_interp2 = project_feasible(h_interp2, n_vhigh)
        c5_interp2 = compute_c5(h_interp2)

        if c5_interp2 <= best_c5 + 0.002:
            h_vh, c5_vh = optimize_at_resolution(h_interp2, n_vhigh, min(350.0, remaining() * 0.4))
            if c5_vh < best_c5:
                best_c5 = c5_vh
                best_h = h_vh.copy()
                best_n = n_vhigh

    # Phase 4: Multi-restart at best resolution with different initializations
    n_cur = best_n
    if remaining() > 60:
        # Restart 1: from best with noise
        rng = np.random.RandomState(seed)
        for restart in range(3):
            if remaining() < 30:
                break
            h_r = best_h + rng.randn(n_cur) * 0.02
            h_r = project_feasible(h_r, n_cur)
            c5_r = compute_c5(h_r)
            if c5_r < best_c5 + 0.01:
                h_ro, c5_ro = optimize_at_resolution(h_r, n_cur, min(60.0, remaining() * 0.3))
                if c5_ro < best_c5:
                    best_c5 = c5_ro
                    best_h = h_ro.copy()

    # Phase 5: Final intensive polish at best resolution
    if remaining() > 30:
        h_fin, c5_fin = optimize_at_resolution(best_h, best_n, min(remaining() * 0.5, 200.0))
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Phase 6: Ultra-fine final steps
    h = best_h.copy()
    n = best_n
    if remaining() > 10:
        for beta in [50000, 100000, 200000]:
            if remaining() < 5:
                break
            h = slsqop_smax(h, n, beta, K=min(80, n // 6), maxiter=800)
            c5_t = compute_c5(h)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h.copy()

    if remaining() > 8:
        h_eq, c5_eq = equalize_top_shifts(best_h, n, K=15, max_iters=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Tiny step polish
    h_t = best_h.copy()
    for it in range(100):
        if remaining() < 3:
            break
        corr_t = np.correlate(h_t, 1.0 - h_t, mode='full')
        idx_t = np.argsort(-corr_t)[:20]
        g_t = np.zeros(n)
        for k in idx_t:
            g_t += grad_corr_k_vec(h_t, k, n)
        g_t /= len(idx_t)
        gn_t = np.linalg.norm(g_t)
        if gn_t < 1e-14:
            break
        for step in [0.003, 0.001, 0.002, 0.005, 0.0005, 0.0008]:
            h_try = project_feasible(h_t - step * g_t / gn_t, n)
            c_try = compute_c5(h_try)
            if c_try < best_c5:
                best_c5 = c_try
                best_h = h_try.copy()
                h_t = h_try.copy()
                break

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
