    while binary.sum() < k:
        zeros = np.where(binary == 0)[0]

def compute_S(h, n):
    nfull = 2*n - 1
    d = np.arange(-(n-1), n)  # lag values
    p = np.arange(n)  # positions
    # S[p, k] = h[p - d[k]] + h[p + d[k]] with out-of-bounds = 0
    idx1 = p[:, None] - d[None, :]  # (n, nfull)
    idx2 = p[:, None] + d[None, :]  # (n, nfull)
    S = np.zeros((n, nfull))
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    S[valid1] += h[idx1[valid1]]
    S[valid2] += h[idx2[valid2]]
    return S
