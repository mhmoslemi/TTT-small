            f_try = compute_c5(h_try)
            if f_try < f_curr - 1e-16:
                h_curr = h_try
                if f_try < c5_best:
                    c5_best = f_try
                    h_best = h_try.copy()
                    stalled = 0
                else:
                    stalled += 1
                accepted = True
                if st <= 3:
                    step = min(step * 1.2, 0.3 / (gn + 1e-10))
                break
            elif st >= 15:
                # Accept anyway to make progress
                h_curr = h_try
                if f_try < c5_best:
                    c5_best = f_try
                    h_best = h_try.copy()
                    stalled = 0
                accepted = True
                step = step * 0.5
                break
        
        if not accepted:
            step = step * 0.5
            if step < 1e-15:
                break
        
        if stalled > 100:
            # Try a larger step or restart
            step = 0.3 / (gn + 1e-10) * 2
            stalled = 0
    
    return h_best, c5_best
