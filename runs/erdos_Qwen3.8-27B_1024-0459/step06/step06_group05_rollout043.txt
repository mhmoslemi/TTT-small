def solve_1d_minimax(coeffs, b, x_min=0, x_max=1):
    """
    Minimize max_k (coeffs[k]*x + b[k]) for x in [x_min, x_max].
    coeffs: array of slopes (length M)
    b: array of intercepts (length M)
    Returns optimal x.
    """
    M = len(coeffs)
    
    # Candidate points
    candidates = [x_min, x_max]
    
    # All pairwise crossings
    for k in range(M):
        for l in range(k+1, M):
            if abs(coeffs[k] - coeffs[l]) > 1e-15:
                x_cross = (b[l] - b[k]) / (coeffs[k] - coeffs[l])
                if x_min < x_cross < x_max:
                    candidates.append(x_cross)
    
    # Also add points where a single line is the unique max (possible if M=1)
    # For M > 1, the optimum is at a crossing or endpoint
    
    # Evaluate f at all candidates
    best_x = x_min
    best_val = float('inf')
    for x in candidates:
        val = np.max(coeffs * x + b)
        if val < best_val:
            best_val = val
            best_x = x
    
    return best_x, best_val

def compute_c5(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    # Clip and adjust sum
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            # All at boundaries; redistribute
            if diff > 0:
                # Need more sum; increase some 0s
                zeros = np.where(h == 0)[0]
                h[zeros[:cnt]] += diff / max(cnt, 1)
            else:
                # Need less sum; decrease some 1s
                ones = np.where(h == 1)[0]
                h[ones[:cnt]] += diff / max(cnt, 1)
        h = np.clip(h, 0.0, 1.0)
    # Final exact adjustment
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        # Distribute among non-boundary elements
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.count_nonzero(mask) > 0:
            h[mask] += diff / np.count_nonzero(mask)
        else:
            h += diff / n

    return h
