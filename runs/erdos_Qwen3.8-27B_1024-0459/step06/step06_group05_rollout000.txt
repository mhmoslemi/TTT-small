        g += w[i] * dx * gk
    return g


def lbfgs_smooth_max(h, n, top_lags, beta, maxiter=500, ftol=1e-16):
    """L-BFGS-B optimization with smooth max objective and projection for equality."""
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        v, _ = smooth_max_obj(x, top_lags, n, beta)
        return v

    def gradient(x):
        return smooth_max_grad(x, top_lags, n, beta)

    try:
        res = minimize(obj, h, jac=gradient, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-14,
                                'maxfun': maxiter * 2, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            return h_new
    except Exception:
        pass
    return h


def projected_gradient_step(h, n, top_lags, beta, lr_scale=1.0):
    g = smooth_max_grad(h, top_lags, n, beta)
    gn = np.linalg.norm(g)
    if gn < 1e-16:
        return h, 0.0
    # Backtracking line search
    c5_curr = compute_c5(h)
    step = lr_scale / (gn + 1e-12)
    for _ in range(30):
        h_try = project_feasible(h - step * g, n)
        c5_try = compute_c5(h_try)
        if c5_try < c5_curr - 1e-15:
            return h_try, step
        step *= 0.5
    return h, 0.0


def full_gradient_descent(h, n, time_limit, beta=2000, K=10):
    h_curr = project_feasible(h.copy(), n)
    h_best = h_curr.copy()
    c5_best = compute_c5(h_best)
    start = time.time()
    max_iter = max(200, int(time_limit * 100))

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 5:
            break

        # Recompute top lags periodically
        corr = compute_corr_full(h_curr)
        top_lags = np.argsort(-corr)[:K]

        g = smooth_max_grad(h_curr, top_lags, n, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break

        base_lr = 0.05 / gn
        improved = False
        for factor in [1.0, 0.5, 0.25, 2.0, 0.1, 4.0, 0.05]:
            step = min(base_lr * factor, 0.3)
            h_try = project_feasible(h_curr - step * g, n)
            c5_try = compute_c5(h_try)
            if c5_try < c5_best - 1e-14:
                c5_best = c5_try
                h_best = h_try.copy()
                h_curr = h_try.copy()
                improved = True
                break
        if not improved:
            # Accept small step to move
            step = base_lr * 0.1
            h_curr = project_feasible(h_curr - step * g, n)

    return h_best, c5_best


def nesterov_approach(h, n, time_limit, K=15, beta=300):
    h_curr = project_feasible(h.copy(), n)
    h_prev = h_curr.copy()
    h_best = h_curr.copy()
    c5_best = compute_c5(h_best)
    start = time.time()
    max_iter = max(100, int(time_limit * 60))

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 5:
            break

        if it == 0:
            y = h_curr.copy()
        else:
            t = it + 1.0
            t_prev = it
            mu = (1.0 - np.sqrt(1.0 + 4.0 / (t_prev ** 2))) / 2.0 if t_prev > 0 else 0.0
            y = h_curr + mu * (h_curr - h_prev)
            y = np.clip(y, 0.0, 1.0)

        corr = compute_corr_full(y)
        top_lags = np.argsort(-corr)[:K]
        g = smooth_max_grad(y, top_lags, n, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break

        step = 0.1 / gn
        h_new = project_feasible(y - step * g, n)
        c5_new = compute_c5(h_new)

        h_prev = h_curr.copy()
        h_curr = h_new

        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_curr.copy()

    return h_best, c5_best


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def optimize_resolution(h_init, n, time_budget):
    """Full optimization pipeline at a given resolution."""
    h = project_feasible(h_init.copy(), n)
    c5 = compute_c5(h)
    start = time.time()

    # Get top lags
    corr = compute_corr_full(h)
    top_lags = np.argsort(-corr)[:20]

    # Phase A: L-BFGS with moderate smooth max
    for beta in [200, 500, 1500]:
        if time.time() - start > time_budget * 0.25:
            break
        rem_beta = time_budget * 0.25 - (time.time() - start)
        if rem_beta < 3:
            break
        h_new = lbfgs_smooth_max(h, n, top_lags, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            corr = compute_corr_full(h)
            top_lags = np.argsort(-corr)[:20]

    # Phase B: Full gradient descent with adaptive top lags
    rem_time = time_budget - (time.time() - start)
    if rem_time > 5:
        fg_time = min(rem_time * 0.35, 120.0)
        h_fg, c5_fg = full_gradient_descent(h, n, fg_time, beta=1000, K=10)
        if c5_fg < c5:
            h = h_fg
            c5 = c5_fg

    # Phase C: Nesterov for exploration
    rem_time = time_budget - (time.time() - start)
    if rem_time > 5:
        ne_time = min(rem_time * 0.2, 60.0)
        h_ne, c5_ne = nesterov_approach(h, n, ne_time, K=15, beta=200)
        if c5_ne < c5:
            h = h_ne
            c5 = c5_ne

    # Phase D: Final L-BFGS polish with high beta
    rem_time = time_budget - (time.time() - start)
    if rem_time > 3:
        corr = compute_corr_full(h)
        top_lags = np.argsort(-corr)[:8]
        for beta in [5000, 20000, 50000]:
            if time.time() - start > time_budget - 1:
                break
            h_new = lbfgs_smooth_max(h, n, top_lags, beta, maxiter=200)
            c5_new = compute_c5(h_new)
            if c5_new < c5 - 1e-14:
                h = h_new
                c5 = c5_new

    return project_feasible(h, n), c5


def slsqp_refine(h, n, top_lags, beta, maxiter=300):
    """SLSQP refinement for small n or final polish."""
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        v, _ = smooth_max_obj(x, top_lags, n, beta)
        return v

    def grad(x):
        return smooth_max_grad(x, top_lags, n, beta)

    def cfun(x):
        return np.sum(x) - n / 2.0

    def cjac(x):
        return np.ones(n)

    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)

    # Get initial solution
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 512
        best_h = np.ones(best_n) * 0.5
        best_c5 = compute_c5(best_h)

    # Phase 1: Optimize at parent resolution
    n1 = best_n
    h1 = best_h.copy()
    c5_1 = best_c5
    if remaining() > 60:
        t_budget = min(180.0, remaining() * 0.25)
        h1, c5_1 = optimize_resolution(h1, n1, t_budget)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()
            best_n = n1

    # Phase 2: Try different resolutions
    res_list = []
    for ratio in [0.5, 0.75, 1.0, 1.25, 1.5, 2.0]:
        r = int(round(best_n * ratio / 32) * 32)
        if 128 <= r <= 2048 and r % 2 == 0 and r != best_n:
            res_list.append(r)
    res_list = sorted(set(res_list))

    for n_t in res_list[:4]:
        if remaining() < 40:
            break
        h_t = project_feasible(interp_to(best_h, best_n, n_t), n_t)
        c5_t = compute_c5(h_t)
        t_budget = min(100.0, remaining() * 0.2)
        h_t, c5_t = optimize_resolution(h_t, n_t, t_budget)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            best_n = n_t

    # Phase 3: Multi-start perturbations at best resolution
    n3 = best_n
    h3 = best_h.copy()
    if remaining() > 60:
        n_perts = min(5, int(remaining() / 40))
        for pi in range(n_perts):
            if remaining() < 30:
                break
            sigma = 0.01 + 0.005 * pi
            h_pert = project_feasible(h3 + rng.randn(n3) * sigma, n3)
            c5_pert = compute_c5(h_pert)
            if c5_pert > best_c5 + 0.05:
                continue
            t_budget = min(60.0, remaining() * 0.15)
            h_pert, c5_pert = optimize_resolution(h_pert, n3, t_budget)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()
                best_n = n3
                h3 = h_pert.copy()

    # Phase 4: SLSQP refinement at best resolution for small n
    n4 = best_n
    h4 = best_h.copy()
    if n4 <= 512 and remaining() > 20:
        corr = compute_corr_full(h4)
        top_lags = np.argsort(-corr)[:10]
        for beta in [1000, 5000, 20000, 50000, 100000]:
            if remaining() < 5:
                break
            h_new = slsqp_refine(h4, n4, top_lags, beta, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                h4 = h_new.copy()
                corr = compute_corr_full(h4)
                top_lags = np.argsort(-corr)[:10]

    # Phase 5: Final fine-tuning with gradient descent
    if remaining() > 10:
        n5 = best_n
        h5 = best_h.copy()
        fg_time = min(30.0, remaining() * 0.3)
        h5, c5_5 = full_gradient_descent(h5, n5, fg_time, beta=3000, K=6)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
            best_n = n5

    # Phase 6: Ultra-fine SLSQP polish
    if remaining() > 5 and best_n <= 640:
        n6 = best_n
        h6 = best_h.copy()
        corr = compute_corr_full(h6)
        top_lags = np.argsort(-corr)[:6]
        for beta in [100000, 200000]:
            if remaining() < 3:
                break
            h_new = slsqp_refine(h6, n6, top_lags, beta, maxiter=400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Final feasibility enforcement
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    # Verify with evaluate function if available
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        best_c5 = min(best_c5, eval_c5)
    except Exception:
        pass

    # Ensure feasibility
    assert np.all(best_h >= -1e-10) and np.all(best_h <= 1.0 + 1e-10)
    assert abs(np.sum(best_h) - best_n / 2.0) < 1e-6

    return (best_h, best_c5, best_n)
