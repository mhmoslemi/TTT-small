        g_y, _ = softmax_grd(y, n, K, beta)
        gy_n = np.linalg.norm(g_y)
        if gy_n < 1e-15:
            break
        step_size = 0.12 / gy_n
        h_new = project_feasible(y - step_size * g_y, n)
        c5_new = compute_c5(h_new)
        h_prev = h_curr.copy()
        h_curr = h_new
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_curr.copy()
    return h_best, c5_best


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def greedy_refine(h, n, n_trials=3000, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    h_best = h.copy()
    c5_best = compute_c5(h)
    for trial in range(n_trials):
        h_try = h_best.copy()
        n_change = rng.randint(1, 4)
        idxs = rng.choice(n, n_change, replace=False)
        for idx in idxs:
            delta = rng.uniform(0.02, 0.4) * (1.0 if h_try[idx] < 0.5 else -1.0)
            h_try[idx] += delta
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = project_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < c5_best - 1e-13:
            c5_best = c5_try
            h_best = h_try.copy()
    return h_best, c5_best


def active_equalize(h, n, K, maxiter=300):
    """Try to equalize top-K correlations via SLSQP on sum of squares"""
    def obj(x):
        corr = compute_corr(x)
        idx = np.argsort(-corr)[:K]
        c_top = corr[idx]
        return float(np.sum((c_top - c_top.mean()) ** 2))

    def grad(x):
        corr = compute_corr(x)
        idx = np.argsort(-corr)[:K]
        c_top = corr[idx]
        mean = c_top.mean()
        dx = 2.0 / n
        g = np.zeros(n)
        for i in range(K):
            coeff = 2.0 * (c_top[i] - mean) * dx / K
            g += coeff * grad_corr_k(x, idx[i], n)
        return g

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            if compute_c5(h_new) <= compute_c5(h) + 1e-12:
                return h_new
    except Exception:
        pass
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

    # Get initial solution
    try:
        best_h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        best_h = np.zeros(512)
        best_h[:256] = 1.0
    best_n = len(best_h)
    best_c5 = compute_c5(best_h)
    h = best_h.copy()
    n = best_n

    # Phase 1: SLSQP at parent resolution - multiple K and beta
    for K in [8, 12, 15, 20, 25]:
        for beta in [2000, 8000, 20000, 50000]:
            if remaining() < 6:
                break
            h_new = slsqp_run(h, n, K, beta, maxiter=400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 2: Active set equalization
    if remaining() > 20:
        for K in [5, 8, 12]:
            if remaining() < 10:
                break
            h_new = active_equalize(h, n, K, maxiter=200)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 3: Nesterov exploration at parent resolution
    if remaining() > 60:
        sg_time = min(60.0, remaining() * 0.2)
        h_sg, c5_sg = nesterov_subgrad(h, n, sg_time, K=15, beta=80)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 4: Interpolate to n=1024 and optimize
    n_hi = 1024
    h_hi = project_feasible(interp_to(best_h, best_n, n_hi), n_hi)
    c5_hi = compute_c5(h_hi)

    for K in [15, 20, 30]:
        for beta in [3000, 10000, 30000]:
            if remaining() < 8:
                break
            h_hi = slsqp_run(h_hi, n_hi, K, beta, maxiter=350)

    if remaining() > 40:
        h_hi, c5_hi = nesterov_subgrad(h_hi, n_hi, min(50.0, remaining() * 0.25), K=25, beta=80)

    for K in [10, 15, 20]:
        for beta in [20000, 50000]:
            if remaining() < 8:
                break
            h_hi = slsqp_run(h_hi, n_hi, K, beta, maxiter=400)

    c5_hi = compute_c5(h_hi)
    if c5_hi < best_c5:
        best_c5 = c5_hi
        best_h = h_hi.copy()
        best_n = n_hi

    # Phase 5: Try n=2048
    if remaining() > 120:
        n_hi2 = 2048
        h_hi2 = project_feasible(interp_to(best_h, best_n, n_hi2), n_hi2)
        c5_hi2 = compute_c5(h_hi2)

        for K in [20, 30]:
            for beta in [5000, 15000]:
                if remaining() < 10:
                    break
                h_hi2 = slsqp_run(h_hi2, n_hi2, K, beta, maxiter=250)

        if remaining() > 40:
            h_hi2, c5_hi2 = nesterov_subgrad(h_hi2, n_hi2, min(40.0, remaining() * 0.2), K=30, beta=80)

        for beta in [30000, 60000]:
            if remaining() < 8:
                break
            h_hi2 = slsqp_run(h_hi2, n_hi2, 20, beta, maxiter=300)

        c5_hi2 = compute_c5(h_hi2)
        if c5_hi2 < best_c5:
            best_c5 = c5_hi2
            best_h = h_hi2.copy()
            best_n = n_hi2

    # Phase 6: Greedy local search at best resolution
    n = best_n
    if remaining() > 30:
        h_g, c5_g = greedy_refine(best_h, n, n_trials=min(5000, int(remaining() * 8)), rng=rng)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()

    # Phase 7: SLSQP polish with very high beta
    n = best_n
    h = best_h.copy()
    for K in [6, 8, 10, 15]:
        for beta in [100000, 200000, 500000]:
            if remaining() < 5:
                break
            h_new = slsqp_run(h, n, K, beta, maxiter=500)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 8: L-BFGS-B as alternative optimizer
    if remaining() > 30:
        for K in [10, 15, 20]:
            for beta in [5000, 20000]:
                if remaining() < 8:
                    break
                h_new = lfbgbs_run(best_h, best_n, K, beta, maxiter=3000)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h_new.copy()

    # Phase 9: More greedy refinement
    if remaining() > 20:
        h_g, c5_g = greedy_refine(best_h, best_n, n_trials=min(3000, int(remaining() * 6)), rng=rng)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()

    # Phase 10: Final SLSQP with highest beta
    for beta in [200000, 500000]:
        if remaining() < 5:
            break
        h_new = slsqp_run(best_h, best_n, 8, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final projection
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
