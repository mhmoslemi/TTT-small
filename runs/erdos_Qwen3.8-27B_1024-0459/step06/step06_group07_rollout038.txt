    """Compute dcorr[i] = change in correlation vector if h[i] increases by 1"""
    nfull = 2 * n - 1
    k_range = np.arange(nfull)
    dcorr = np.zeros((n, nfull))

        j2 = i - k_range + (n - 1)

        v2 = (j2 >= 0) & (j2 < n)
        dcorr[i, v1] += (1.0 - h[j1[v1]])
        dcorr[i, v2] -= h[j2[v2]]
    return dcorr

def compute_c5(h):

def compute_corr(h):

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mid = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.count_nonzero(mid))
        if cnt > 0:
            h[mid] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def binarize(h, n):
    h_sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[h_sorted_idx[:n // 2]] = 1.0
    return h_bin


def precompute_dcorr(h, n):
    nfull = 2 * n - 1
    k_range = np.arange(nfull)
    dcorr = np.zeros((n, nfull))
    for i in range(n):
        j1 = i + k_range - (n - 1)
        j2 = i - k_range + (n - 1)
        v1 = (j1 >= 0) & (j1 < n)
        v2 = (j2 >= 0) & (j2 < n)
        dcorr[i, v1] += (1.0 - h[j1[v1]])
        dcorr[i, v2] -= h[j2[v2]]
    return dcorr


def _cons_fun(x):
    return np.sum(x) - _N_GLOBAL / 2.0


def _cons_jac(x):
    return np.ones(len(x))


def smooth_max_value(x, beta, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50.0, 50.0)
    sm = m + np.log(np.sum(np.exp(shifted))) / beta
    return sm * (2.0 / n)


def smooth_max_grad(x, beta, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50.0, 50.0)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = w / w_sum
    g = np.zeros(n)
    dx = 2.0 / n
    nfull = 2 * n - 1
    for k in range(nfull):
        wk = w[k] * dx
        if wk < 1e-16:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            g[offset:offset + npairs] += wk * (1.0 - x[:npairs])
            g[:npairs] -= wk * x[offset:offset + npairs]
        else:
            a = -offset
            npairs = n - a
            g[:npairs] += wk * (1.0 - x[a:a + npairs])
            g[a:a + npairs] -= wk * x[:npairs]
    return g


def slsqp_optimize(h, n, beta, maxiter=400):
    global _N_GLOBAL
    _N_GLOBAL = n
    h = h.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]

    def obj(x):
        return smooth_max_value(x, beta, n)

    def grad(x):
        return smooth_max_grad(x, beta, n)

    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None:

    except Exception:
        pass
    return h.copy()


def coord_optimize_scalar(h, n, i, corr):
    """Solve the 1D subproblem for coordinate i using ternary search."""
    nfull = 2 * n - 1
    p = np.arange(nfull)
    j_a = i + p - (n - 1)
    j_b = i - p + (n - 1)

    valid_a = (j_a >= 0) & (j_a < n)
    valid_b = (j_b >= 0) & (j_b < n)

    hi_old = h[i]
    term_a = np.zeros(nfull)
    if np.any(valid_a):
        term_a[valid_a] = hi_old * (1.0 - h[j_a[valid_a]])

    term_b = np.zeros(nfull)
    if np.any(valid_b):
        term_b[valid_b] = h[j_b[valid_b]] * (1.0 - hi_old)

    R = corr - term_a - term_b
    D = np.zeros(nfull)
    if np.any(valid_a):

        D[valid_b] -= h[j_b[valid_b]]

def binary_steepest_descent(h, n, corr, dcorr):
    """Find and apply the best improving binary swap. Returns (made_move, new_h, new_corr, new_c5)."""
    ones_idx = np.where(h == 1)[0]
    zeros_idx = np.where(h == 0)[0]

    if len(ones_idx) == 0 or len(zeros_idx) == 0:
        return False, h, corr, float(np.max(corr)) * (2.0 / n)

    c5_curr = float(np.max(corr)) * (2.0 / n)
    best_new_c5 = c5_curr
    best_i = -1
    best_j = -1

    for i in ones_idx:
        base = corr - dcorr[i]
        new_maxs = np.max(base[None, :] + dcorr[zeros_idx], axis=1)
        best_local = int(np.argmin(new_maxs))
        local_c5 = float(new_maxs[best_local]) * (2.0 / n)
        if local_c5 < best_new_c5 - 1e-14:
            best_new_c5 = local_c5
            best_i = i
            best_j = zeros_idx[best_local]

    if best_i < 0:
        return False, h, corr, c5_curr

    h[best_i] = 0.0
    h[best_j] = 1.0
    corr = compute_corr(h)
    return True, h, corr, best_new_c5

                j_val = zeros_idx[j_pos]
                h[i_val] = 0.0
                h[j_val] = 1.0
                ones_idx = np.delete(ones_idx, i_pos)
                zeros_idx = np.delete(zeros_idx, j_pos)
                ones_idx = np.append(ones_idx, j_val)
                zeros_idx = np.append(zeros_idx, i_val)

            corr = compute_corr(h)
            c5 = compute_c5(h)
            perturbation_size = k_pert

    return h_best, c5_best


def interpolate(h_src, n_src, n_dst):
    xp = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_new = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_new, xp, h_src)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)

    # Phase 0: Initialize
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        best_n = len(parent_h)
        best_h = project_feasible(parent_h, best_n)
    except Exception:
        best_n = 512
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0

    best_c5 = compute_c5(best_h)

    # Phase A: SLSQP to refine continuous solution (15% time)
    n = best_n
    h = best_h.copy()
    t_a = remaining() * 0.15
    for beta in [500, 2000, 10000, 50000]:
        if remaining() < t_a:
            break
        h_new = slsqp_optimize(h.copy(), n, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase B: Coordinate descent on continuous problem (35% time)
    if remaining() > 50:
        t_b = remaining() * 0.35
        h_cd, c5_cd, h_cd_end, _ = coordinate_descent(h.copy(), n, t_b, rng)
        if c5_cd < best_c5 - 1e-13:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h = h_cd.copy()

    # Phase C: Binary local search (40% time)
    if remaining() > 30:
        t_c = remaining() * 0.40
        h_bin = binarize(h, n)
        h_bl, c5_bl = binary_local_search(h_bin, n, t_c, rng)
        if c5_bl < best_c5 - 1e-13:
            best_c5 = c5_bl
            best_h = h_bl.copy()
            h = h_bl.copy()
            best_n = n

    # Phase D: SLSQP polish after binary search (10% time)
    if remaining() > 20:
        for beta in [20000, 100000]:
            if remaining() < 15:
                break
            h_new = slsqp_optimize(h.copy(), n, beta, maxiter=400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase E: Multi-resolution attempts (remaining time)
    resolutions = []
    for mult in [1.5, 2.0, 2.5, 3.0, 0.5]:
        r = int(round(best_n * mult / 64) * 64)
        if 128 <= r <= 2048 and r % 2 == 0 and r != best_n:
            resolutions.append(r)
    resolutions = sorted(set(resolutions))

    for n_t in resolutions:
        if remaining() < 40:
            break

        h_interp = interpolate(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        c5_t = compute_c5(h_feas)

        # Coordinate descent at new resolution
        if remaining() > 30 and n_t <= 1024:
            t_rem = remaining() * 0.5
            h_cd2, c5_cd2, _, _ = coordinate_descent(h_feas.copy(), n_t, t_rem, rng)
            if c5_cd2 < c5_t:
                c5_t = c5_cd2
                h_feas = h_cd2.copy()

        # Binary local search at new resolution
        if remaining() > 20 and n_t <= 1024:
            h_bin2 = binarize(h_feas, n_t)
            h_bl2, c5_bl2 = binary_local_search(h_bin2, n_t, remaining() * 0.4, rng)
            if c5_bl2 < c5_t:
                c5_t = c5_bl2
                h_feas = h_bl2.copy()

        # SLSQP polish
        if remaining() > 15:
            h_fin = slsqp_optimize(h_feas.copy(), n_t, 30000, maxiter=300)
            c5_fin = compute_c5(h_fin)
            if c5_fin < c5_t:
                c5_t = c5_fin
                h_feas = h_fin.copy()

        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_feas.copy()
            best_n = n_t

    # Phase F: Final intensive refinement at best resolution
    n = best_n
    h = best_h.copy()

    if remaining() > 20:
        # Binary local search again with more time
        h_bin_f = binarize(h, n)
        h_bl_f, c5_bl_f = binary_local_search(h_bin_f, n, remaining() * 0.5, rng)
        if c5_bl_f < best_c5 - 1e-13:
            best_c5 = c5_bl_f
            best_h = h_bl_f.copy()
            h = h_bl_f.copy()

    if remaining() > 15:
        # Final SLSQP
        h_fin = slsqp_optimize(h.copy(), n, 100000, maxiter=500)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5 - 1e-13:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    if remaining() > 10:
        # Final binary polish
        h_bin_final = binarize(best_h, best_n)
        h_bp, c5_bp = binary_local_search(h_bin_final, best_n, remaining(), rng)
        if c5_bp < best_c5 - 1e-13:
            best_c5 = c5_bp
            best_h = h_bp.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
