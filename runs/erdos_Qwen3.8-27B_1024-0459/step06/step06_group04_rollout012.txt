def fista_smooth(h, n, beta, time_limit, n_iter_max=50000):
    """FISTA on smooth max with projection onto {0<=h<=1, sum=h_target}"""
    h_curr = h.copy()
    h_next = h.copy()
    t_mom = 1.0
    start = time.time()
    best_h = h_curr.copy()
    best_c5 = compute_c5(h_curr)
    
    for it in range(n_iter_max):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 10:
            break
        
        # Gradient of smooth max
        g = smooth_max_jac(h_curr, beta)
        
        # Step size (for FISTA, need Lipschitz constant of gradient)
        # The gradient of the max correlation has bounded norm
        # For smooth max with beta, the Lipschitz constant is roughly beta * max_weight + something
        # Let's use a safe step size
        L = 4.0 * n / 2.0  # rough upper bound
        step = 1.0 / max(L, 1e-10)
        
        # Gradient step
        h_next = h_curr - step * g
        
        # Proximal step (project onto constraints)
        h_next = project_feasible(h_next, n)
        
        # Update momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        h_temp = h_next + ((t_mom - 1.0) / t_new) * (h_next - h_curr)
        h_curr = h_temp.copy()
        h_curr = project_feasible(h_curr, n)
        t_mom = t_new
        h_next = h_curr.copy()
        
        # Check improvement
        if it % 10 == 0:
            c = compute_c5(h_curr)
            if c < best_c5:
                best_c5 = c
                best_h = h_curr.copy()
    
    # Final evaluation
    c = compute_c5(best_h)
    if c < best_c5:
        best_c5 = c
        best_h = project_feasible(best_h, n)
    
    return best_h, best_c5

        k_max = np.argmax(corr)
        off = k_max - (n - 1)
        n_one = np.where(h_bin == 1)[0]
        n_zero = np.where(h_bin == 0)[0]
        if len(n_one) == 0 or len(n_zero) == 0:
            break
        improved = False
        max_tries = min(300, len(n_one) * len(n_zero))
        for _ in range(max_tries):
            if off >= 0:
                valid_ones = n_one[(n_one < n - off)]
                valid_zeros = n_zero[(n_zero >= off) & (n_zero < n)]
            else:
                ao = -off
                valid_ones = n_one[(n_one < n - ao)]
                valid_zeros = n_zero[(n_zero >= ao)]
            if len(valid_ones) == 0 or len(valid_zeros) == 0:
                valid_ones = n_one
                valid_zeros = n_zero
            i1 = valid_ones[np.random.randint(len(valid_ones))]
            i0 = valid_zeros[np.random.randint(len(valid_zeros))]
            h_try = h_bin.copy()
            h_try[i1] = 0.0
            h_try[i0] = 1.0
            h_try = project_box_plane(h_try, n)
            c_try = compute_c5(h_try)
            if c_try < best_c5 - 1e-14:
                h_bin = h_try
                best_c5 = c_try
                best_h = h_try.copy()
                n_one = np.where(h_bin == 1)[0]
                n_zero = np.where(h_bin == 0)[0]
                improved = True
                break
        if not improved and rnd > 80:
            break
    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    np.random.seed(seed)

    try:
        best_h = np.array(initial_h_values, dtype=float).copy()
        best_n = len(best_h)
    except Exception:
        best_n = 512
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0

    best_h = project_box_plane(best_h, best_n)
    best_c5 = compute_c5(best_h)

    def rem():
        return max(0.0, _BUDGET - (time.time() - _T0) - 10.0)

    # Phase 1: SLSQP smooth max at parent resolution with increasing beta
    if rem() > 30:
        alphas = [500, 2000, 8000, 30000, 100000, 400000, 1000000]
        for beta in alphas:
            if rem() < 10:
                break
            t_alloc = min(60.0, rem() * 0.25)
            h_new = slsqp_optimize(best_h, best_n, beta, maxiter=300)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()

    # Phase 2: FISTA at parent resolution
    if rem() > 20:
        for beta in [50000, 200000, 500000]:
            if rem() < 10:
                break
            t_alloc = min(30.0, rem() * 0.15)
            h_new, c_new = fista_optimize(best_h, best_n, beta, t_alloc)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()

    # Phase 3: Nesterov top-K exact subgradient
    if rem() > 20:
        t_sg = min(60.0, rem() * 0.15)
        h_new, c_new = nesterov_topk(best_h, best_n, t_sg, K=3, beta_w=80.0)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Phase 4: Multi-resolution optimization
    res_list = []
    for target_n in [768, 1024, 1536, 2048]:
        if target_n != best_n and target_n % 2 == 0:
            res_list.append(target_n)

    for rn in res_list:
        if rem() < 30:
            break
        h_interp = interp_to(best_h, best_n, rn)
        h_feas = project_box_plane(h_interp, rn)
        c_int = compute_c5(h_feas)
        # FISTA at this resolution
        t_alloc = min(80.0, rem() * 0.2)
        for beta in [5000, 20000, 80000]:
            if rem() < 10:
                break
            t_sub = min(40.0, t_alloc * 0.4)
            h_new, c_new = fista_optimize(h_feas, rn, beta, t_sub)
            if c_new < c_int:
                c_int = c_new
                h_feas = h_new.copy()
        # SLSQP at this resolution (shorter)
        if rem() > 15:
            h_new = slsqp_optimize(h_feas, rn, 50000, maxiter=100)
            c_new = compute_c5(h_new)
            if c_new < c_int:
                c_int = c_new
                h_feas = h_new.copy()
        if c_int < best_c5:
            best_c5 = c_int
            best_h = h_feas.copy()
            best_n = rn

    # Phase 5: Go back to original resolution and re-optimize
    if best_n != 512 and rem() > 20:
        h_back = interp_to(best_h, best_n, 512)
        h_back = project_box_plane(h_back, 512)
        c_back = compute_c5(h_back)
        if c_back < best_c5:
            best_c5 = c_back
            best_h = h_back.copy()
            best_n = 512
        else:
            h_back = project_box_plane(interpolation_init(best_h, best_n, 512), 512)
            c_back = compute_c5(h_back)
            if rem() > 15:
                for beta in [100000, 500000]:
                    if rem() < 8:
                        break
                    h_new = slsqp_optimize(h_back, 512, beta, maxiter=200)
                    c_new = compute_c5(h_new)
                    if c_new < best_c5:
                        best_c5 = c_new
                        best_h = h_new.copy()
                        best_n = 512

    # Phase 6: Structured binary search
    if rem() > 30:
        t_bs = min(60.0, rem() * 0.12)
        h_bin, c_bin = structured_binary_search(best_h, best_n, t_bs, max_rounds=400)
        if c_bin < best_c5:
            best_c5 = c_bin
            best_h = h_bin.copy()
            # Polish binary with continuous
            if rem() > 20:
                h_pol = slsqp_optimize(best_h, best_n, 200000, maxiter=200)
                c_pol = compute_c5(h_pol)
                if c_pol < best_c5:
                    best_c5 = c_pol
                    best_h = h_pol.copy()

    # Phase 7: More Nesterov with different K
    if rem() > 15:
        t_sg2 = min(40.0, rem() * 0.1)
        h_new, c_new = nesterov_topk(best_h, best_n, t_sg2, K=5, beta_w=30.0)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Phase 8: Final SLSQP polish
    if rem() > 10:
        for beta in [2000000, 5000000]:
            if rem() < 6:
                break
            h_new = slsqp_optimize(best_h, best_n, beta, maxiter=150)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()

    # Phase 9: FISTA final polish
    if rem() > 10:
        h_new, c_new = fista_optimize(best_h, best_n, 1000000, min(30.0, rem() * 0.2))
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Final feasibility
    best_h = project_box_plane(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)


def interpolation_init(h_full, n_full, n_target):
    xp = (np.arange(n_full) + 0.5) * 2.0 / n_full
    x_new = (np.arange(n_target) + 0.5) * 2.0 / n_target
    return np.interp(x_new, xp, h_full)
