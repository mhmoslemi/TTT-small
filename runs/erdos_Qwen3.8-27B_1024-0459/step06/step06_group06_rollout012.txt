    """Given v=1-h_fixed, find h that minimizes max_k corr[k].
    corr[k] = sum_i h[i+s_k] * v[idx]
    """
    n_shifts = 2 * n - 1
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

        s = k - (n - 1)
        row = np.zeros(n + 1)
        row[-1] = -1.0

            row[s:n] += v[:n-s]
        else:
            a = -s
            row[:n-a] += v[a:n]
