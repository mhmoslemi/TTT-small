def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)

        if abs(diff) < 1e-15:

        best_n = len(best_h)
    except Exception:
        best_n = 512
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
    best_c5 = compute_c5(best_h)

    # Phase 1: Optimize at parent resolution with trust-constr
    if rem() > 60:
        t1 = min(250.0, rem() * 0.25)
        h1, c1 = multi_beta_optimize(best_h, best_n, t1, use_tc=True)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1.copy()

    # Phase 2: Nesterov subgradient at parent resolution
    if rem() > 60:
        t2 = min(120.0, rem() * 0.15)
        h2, c2 = nesterov_subgradient(best_h, best_n, t2, K=8, T0_temp=50.0, T1_temp=300.0)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()

    # Phase 3: Adaptive subgradient
    if rem() > 40:
        t3 = min(80.0, rem() * 0.1)
        h3, c3 = adaptive_subgradient(best_h, best_n, t3, K=10, T0_temp=30.0, T1_temp=400.0)
        if c3 < best_c5:
            best_c5 = c3
            best_h = h3.copy()

    # Phase 4: Simulated annealing at parent resolution
    if rem() > 80:
        t4 = min(200.0, rem() * 0.2)
        h4, c4 = simulated_annealing(best_h, best_n, t4, rng, max_iters=80000)
        if c4 < best_c5:
            best_c5 = c4
            best_h = h4.copy()

    # Phase 5: Try higher resolutions
    res_list = []
    for rn in [768, 1024, 1536]:
        if rn != best_n and rem() > 100:
            res_list.append(rn)

    for rn in res_list[:3]:
        if rem() < 100:
            break
        h_interp = interp_to(best_h, best_n, rn)
        h_interp = project_feasible(h_interp, rn)
        t_alloc = min(200.0, rem() * 0.25)
        h_new, c_new = multi_beta_optimize(h_interp, rn, t_alloc, use_tc=True)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
            best_n = rn

    # Phase 6: Mutation restarts at best resolution
    if rem() > 80:
        n_restarts = min(4, int(rem() // 80))
        for m in range(n_restarts):
            if rem() < 80:
                break
            # Structured mutation
            h_mut = best_h.copy()
            freq = rng.randint(2, max(3, best_n // 6))
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.08, 0.2)
            x = np.arange(best_n)
            h_mut += amp * np.sin(2 * np.pi * freq * x / best_n + phase)
            h_mut = project_feasible(h_mut, best_n)

            t_alloc = min(100.0, rem() * 0.3)
            h_opt, c_opt = multi_beta_optimize(h_mut, best_n, t_alloc, use_tc=True)
            if c_opt < best_c5:
                best_c5 = c_opt
                best_h = h_opt.copy()

    # Phase 7: Final SLSQP polish with very high beta
    if rem() > 30:
        for beta in [300000, 800000, 2000000]:
            if rem() < 15:
                break
            h_new = optimize_slsqp(best_h, best_n, beta, maxiter=300)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()

    # Phase 8: Final subgradient pass
    if rem() > 30:
        h_sg, c_sg = nesterov_subgradient(best_h, best_n, min(60.0, rem() * 0.4),
                                          K=6, T0_temp=100.0, T1_temp=800.0)
        if c_sg < best_c5:
            best_c5 = c_sg
            best_h = h_sg.copy()

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
