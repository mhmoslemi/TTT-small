def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)

        free = (h > 1e-15) & (h < 1.0 - 1e-15)
        cnt = np.count_nonzero(free)
        if cnt > 0:
            h[free] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_corr_k_vec(h, k, n):
    """Gradient of corr[k] = sum_i h[i]*(1-h[i+k-(n-1)]) w.r.t. h"""

        g[:np_] -= h[offset:]
    else:
        a = -offset
        np_ = n - a
        g[:np_] += (1.0 - h[a:])
        g[a:] -= h[:np_]
    return g
