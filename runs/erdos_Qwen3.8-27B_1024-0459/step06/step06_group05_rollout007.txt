def slsqp_quad_constraints(h, n, K_subset, maxiter=300):
    """SLSQP with explicit quadratic inequality constraints f_k(h) <= t."""
    # We minimize t with constraints f_k(h) <= t for k in K_subset
    # Expand to n+1 variables: [h, t]
    
    x0 = np.concatenate([h, [compute_c5(h) * 2.0 + 0.01]])
    
    def obj(x):
        return x[n]  # minimize t
    
    def jac(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    # Equality: sum(h) = n/2
    def eq_fun(x):
        return np.sum(x[:n]) - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    # Inequality: f_k(h) - t <= 0 for each k
    def ineq_fun(x):
        h_x = x[:n]
        t = x[n]
        corr = compute_corr(h_x)
        dx = 2.0 / n
        return np.array([corr[k] * dx - t for k in K_subset])
    
    def ineq_jac(x):
        h_x = x[:n]
        J = np.zeros((len(K_subset), n + 1))
        for i, k in enumerate(K_subset):
            g_k = grad_corr_k_vec(h_x, k, n)
            J[i, :n] = g_k
            J[i, n] = -1.0
        return J
    
    cons_list = [{
        'type': 'ineq',
        'fun': lambda x: -(ineq_fun(x)),  # scipy requires f(x) >= 0 for ineq
        'jac': lambda x: -ineq_jac(x)
    }, {
        'type': 'eq',
        'fun': eq_fun,
        'jac': eq_jac
    }]
    
    bounds = [(0, 1)] * n + [(0, None)]
    
    try:
        res = minimize(obj, x0, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons_list,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.success or res.x is not None:
            h_new = project_feasible(res.x[:n], n)
            c5_new = compute_c5(h_new)
            if c5_new < compute_c5(h) + 1e-12:
                return h_new
    except Exception:
        pass
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_corr_k_vec(h, k, n):
    """Vectorized gradient of correlation integral at index k."""
    g = np.zeros(n)
    dx = 2.0 / n
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        if npairs <= 0:
            return g
        j_lo = 0
        j_hi = min(npairs, n)
        g[j_lo:j_hi] -= dx
        lo = offset
        hi = min(offset + npairs, n)
        if lo < hi:
            g[lo:hi] += dx * (1.0 - h[lo - offset:hi - offset])
    else:
        a = -offset
        npairs = n - a
        if npairs <= 0:
            return g
        g[:npairs] += dx * (1.0 - h[a:a + npairs])
        lo = a
        hi = min(a + npairs, n)
        if lo < hi:
            g[lo:hi] -= dx * h[lo - a:hi - a]
    return g

        g_y, c_max = smooth_max_topk_grad(y, n, K, beta)
        gy_n = np.linalg.norm(g_y)
        if gy_n < 1e-15:
            break

        step_size = 0.2 / gy_n
        step_size = min(step_size, 0.5)

        h_new = project_feasible(y - step_size * g_y, n)
        c5_new = compute_c5(h_new)

        h_prev = h_curr.copy()
        h_curr = h_new

        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_curr.copy()

    return h_best, c5_best


def subgrad_top1(h, n, time_limit):
    """Pure subgradient on the top-1 correlation."""
    h_curr = h.copy()
    h_best = h.copy()
    c5_best = compute_c5(h)
    start = time.time()
    n_iter = max(int(time_limit * 80), 100)
    alpha = 0.1

    for it in range(n_iter):
        if time.time() - start > time_limit:
            break
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break

        corr = compute_corr(h_curr)
        k_max = int(np.argmax(corr))
        g = grad_corr_k_vec(h_curr, k_max, n)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break

        step = alpha / (1.0 + 0.1 * it)
        h_new = project_feasible(h_curr - step * g, n)
        c5_new = compute_c5(h_new)
        h_curr = h_new

        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_curr.copy()

    return h_best, c5_best


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 512
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_c5 = compute_c5(best_h)

    n = best_n
    h = best_h.copy()

    # Phase 1: Cutting-plane LP at parent resolution
    if remaining() > 100:
        for K in [10, 20, 40, 60]:
            if remaining() < 25:
                break
            h_cp, c5_cp = cutting_plane_lp(h, n, K, max_iters=400)
            if c5_cp < best_c5:
                best_c5 = c5_cp
                best_h = h_cp.copy()
                h = h_cp.copy()

    # Phase 2: SLSQP sweep
    for K in [10, 15, 20, 30]:
        for beta in [500, 2000, 10000, 50000]:
            if remaining() < 10:
                break
            h_new = slsqp_optimize(h, n, K, beta, maxiter=400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 3: Cutting-plane LP again (refined state)
    if remaining() > 60:
        for K in [20, 35, 50]:
            if remaining() < 20:
                break
            h_cp, c5_cp = cutting_plane_lp(h, n, K, max_iters=500)
            if c5_cp < best_c5:
                best_c5 = c5_cp
                best_h = h_cp.copy()
                h = h_cp.copy()

    # Phase 4: Subgradient top-1 (simple but effective for the exact max)
    if remaining() > 40:
        h_s1, c5_s1 = subgrad_top1(h, n, min(30.0, remaining() * 0.3))
        if c5_s1 < best_c5:
            best_c5 = c5_s1
            best_h = h_s1.copy()
            h = h_s1.copy()

    # Phase 5: Nesterov subgradient for landscape exploration
    if remaining() > 50:
        sg_time = min(60.0, remaining() * 0.2)
        h_sg, c5_sg = nesterov_subgrad(h, n, sg_time, K=20, beta=100)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

        if remaining() > 20:
            for beta in [5000, 20000]:
                if remaining() < 10:
                    break
                h_new = slsqp_optimize(h, n, 15, beta, maxiter=300)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()

    # Phase 6: Multi-resolution exploration
    res_candidates = []
    for ratio in [1.5, 2.0, 2.5, 3.0]:
        r = int(round(best_n * ratio / 64.0) * 64)
        if 256 <= r <= 4096 and r != best_n and r % 2 == 0:
            res_candidates.append(r)
    res_candidates = sorted(set(res_candidates))

    for n_t in res_candidates[:3]:
        if remaining() < 100:
            break

        h_t = project_feasible(interp_to(best_h, best_n, n_t), n_t)
        c5_t = compute_c5(h_t)

        for K in [15, 25, 40]:
            for beta in [2000, 10000]:
                if remaining() < 15:
                    break
                h_t = slsqp_optimize(h_t, n_t, K, beta, maxiter=250)
                c5_t = compute_c5(h_t)

        if remaining() > 50:
            h_t, c5_t = cutting_plane_lp(h_t, n_t, 30, max_iters=200)

        if remaining() > 25:
            for beta in [20000, 80000]:
                if remaining() < 10:
                    break
                h_t = slsqp_optimize(h_t, n_t, 15, beta, maxiter=300)
                c5_t = compute_c5(h_t)

        if remaining() > 15:
            h_t, c5_t = cutting_plane_lp(h_t, n_t, 20, max_iters=200)

        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            best_n = n_t
            n = n_t
            h = h_t.copy()

    # Phase 7: Intensive final refinement at best resolution
    n = best_n
    h = best_h.copy()

    if remaining() > 30:
        h_cp, c5_cp = cutting_plane_lp(h, n, 30, max_iters=500)
        if c5_cp < best_c5:
            best_c5 = c5_cp
            best_h = h_cp.copy()
            h = h_cp.copy()

    for K in [8, 10, 15, 20]:
        for beta in [20000, 80000, 200000]:
            if remaining() < 5:
                break
            h_new = slsqp_optimize(h, n, K, beta, maxiter=500)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 8: Perturbation multi-start
    if remaining() > 30:
        for pert_i in range(4):
            if remaining() < 15:
                break
            sigma = 0.003 + 0.004 * pert_i
            h_pert = project_feasible(h + rng.randn(n) * sigma, n)
            c5_pert = compute_c5(h_pert)
            if c5_pert > best_c5 + 0.003:
                continue
            h_pert = slsqp_optimize(h_pert, n, 15, 10000, maxiter=200)
            if remaining() > 15:
                h_pert, c5_pert = cutting_plane_lp(h_pert, n, 20, max_iters=150)
            c5_pert = compute_c5(h_pert)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()
                h = h_pert.copy()

    # Phase 9: Final cutting-plane + SLSQP polish
    n = best_n
    h = best_h.copy()
    if remaining() > 15:
        h_cp, c5_cp = cutting_plane_lp(h, n, 40, max_iters=300)
        if c5_cp < best_c5:
            best_c5 = c5_cp
            best_h = h_cp.copy()
            h = h_cp.copy()

    if remaining() > 10:
        h_new = slsqp_optimize(h, n, 12, 100000, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
