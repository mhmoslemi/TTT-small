                g[:np_] -= h[off:]

            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    return g


def _eq_fun(x):
    return np.sum(x) - _CUR_N / 2.0


def _eq_jac(x):
    return np.ones(_CUR_N)


def slsqp_opt(h, n, beta, maxiter=500):
    global _CUR_N, _SM_BETA
    _CUR_N = n
    _SM_BETA = beta
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_sm_objective, h0, jac=_sm_gradient, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.fun < _sm_objective(h0) - 1e-14:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def two_swap(h, n, max_pairs=5000, seed=42):
    rng = np.random.RandomState(seed)
    h = project_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    for it in range(max_pairs):
        if remaining_time() < 5:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        hi = h[i]
        hj = h[j]
        t_min = max(-hi, hj - 1.0) + 1e-10
        t_max = min(1.0 - hi, hj) - 1e-10
        if t_min >= t_max:
            continue
        a, b = t_min, t_max
        best_t = 0.0
        best_f = compute_c5(h)
        for _ in range(20):
            m1 = a + (b - a) * 0.381966
            m2 = a + (b - a) * 0.618034
            h[i] = hi + m1
            h[j] = hj - m1
            f1 = compute_c5(h)
            h[i] = hi + m2
            h[j] = hj - m2
            f2 = compute_c5(h)
            if f1 < f2:
                b = m2
                if f1 < best_f:
                    best_f = f1
                    best_t = m1
            else:
                a = m1
                if f2 < best_f:
                    best_f = f2
                    best_t = m2
        h[i] = hi + best_t
        h[j] = hj - best_t
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
        else:
            h[i] = hi
            h[j] = hj
    return h, best_c5


def topk_equalize(h, n, top_k=5, max_iters=500):
    h = project_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining_time() < 3:
            break
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        if corr[idx_top[0]] - corr[idx_top[-1]] < 1e-11:
            break
        g_dir = np.zeros(n)
        for pos, k in enumerate(idx_top):
            w_k = 1.0 if pos == 0 else (-1.0 / (len(idx_top) - 1))
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_dir[off:] += w_k * (1.0 - h[:np_])
                g_dir[:np_] -= w_k * h[off:]
            else:
                a = -off
                np_ = n - a
                g_dir[:np_] += w_k * (1.0 - h[a:])
                g[a:] -= w_k * h[:np_]
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        d = g_dir / gn
        improved = False
        for step in [0.001, 0.003, 0.005, 0.01, 0.02, 0.002, 0.008, 0.015, 0.03]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-13:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            for step in [0.003, 0.005, 0.01, 0.02]:
                ht = project_feasible(h - step * d, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-13:
                    h = ht
                    best_c5 = cn
                    improved = True
                    break
            if not improved:
                break
    return h, best_c5


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def coord_descent(h, n, max_sweeps=20):
    h = project_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if remaining_time() < 10:
            break
        any_improve = False
        for j in range(n):
            old_val = h[j]
            best_v = old_val
            best_obj = compute_c5(h)
            a, b = 0.0, 1.0
            for _ in range(22):
                m1 = a + (b - a) * 0.381966
                m2 = a + (b - a) * 0.618034
                h[j] = m1
                f1 = compute_c5(h)
                h[j] = m2
                f2 = compute_c5(h)
                if f1 < f2:
                    b = m2
                    if f1 < best_obj:
                        best_obj = f1
                        best_v = m1
                else:
                    a = m1
                    if f2 < best_obj:
                        best_obj = f2
                        best_v = m2
            h[j] = best_v
            curr = compute_c5(h)
            if curr < best_c5 - 1e-13:
                best_c5 = curr
                any_improve = True
        h = project_feasible(h, n)
        if not any_improve and sweep > 3:
            break
    return h, best_c5


def adam_subgrad(h, n, max_iters=3000, lr=0.005, top_k=5):
    h = project_feasible(h, n).copy()
    m_acc = np.zeros(n)
    v_acc = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    dx = 2.0 / n
    b1, b2 = 0.9, 0.999
    for it in range(max_iters):
        if remaining_time() < 5:
            break
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        for ki in idx_top:
            off = ki - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:] += (1.0 - h[:np_])
                g[:np_] -= h[off:]
            else:
                a = -off
                np_ = n - a
                g[:np_] += (1.0 - h[a:])
                g[a:] -= h[:np_]
        g *= dx / top_k
        m_acc = b1 * m_acc + (1 - b1) * g
        v_acc = b2 * v_acc + (1 - b2) * g ** 2
        m_hat = m_acc / (1 - b1 ** (it + 1))
        v_hat = v_acc / (1 - b2 ** (it + 1))
        update = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h_new = project_feasible(h - update, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5


def theta_morse_init(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _CUR_N, _SM_BETA
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = np.zeros(640)
        best_h[:320] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick SLSQP at parent resolution
    h_work = best_h.copy()
    for beta in [5000, 50000]:
        if remaining_time() < 60:
            break
        h_work = slsqp_opt(h_work, best_n, beta, maxiter=400)
        c5_w = compute_c5(h_work)
        if c5_w < best_c5:
            best_c5 = c5_w
            best_h = h_work.copy()

    # Phase 2: Upsample to n=1024 and optimize
    n_hi = 1024
    h_hi = interp_to(best_h, best_n, n_hi)
    h_hi = project_feasible(h_hi, n_hi)
    for beta in [2000, 20000, 200000]:
        if remaining_time() < 80:
            break
        h_hi = slsqp_opt(h_hi, n_hi, beta, maxiter=600)
        c5_hi = compute_c5(h_hi)
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi.copy()
            best_n = n_hi

    # Phase 3: Top-k equalization at n=1024
    if remaining_time() > 30:
        h_eq, c5_eq = topk_equalize(h_hi, n_hi, top_k=5, max_iters=800)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            best_n = n_hi
            h_hi = h_eq.copy()

    # Phase 4: 2-swap local search (key innovation)
    if remaining_time() > 120:
        h_sw, c5_sw = two_swap(h_hi.copy(), n_hi, max_pairs=8000, seed=777)
        if c5_sw < best_c5:
            best_c5 = c5_sw
            best_h = h_sw.copy()
            best_n = n_hi
            h_hi = h_sw.copy()

    # Phase 5: SLSQP polish with very high beta
    if remaining_time() > 60:
        h_hi = slsqp_opt(h_hi, n_hi, 1000000, maxiter=800)
        c5_hi = compute_c5(h_hi)
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi.copy()
            best_n = n_hi

    # Phase 6: Adam subgradient with polygonal rounding
    if remaining_time() > 60:
        h_ad, c5_ad = adam_subgrad(best_h.copy(), best_n, max_iters=3000, lr=0.003, top_k=7)
        if c5_ad < best_c5:
            best_c5 = c5_ad
            best_h = h_ad.copy()
        if remaining_time() > 30:
            h_pol = slsqp_opt(best_h, best_n, 500000, maxiter=500)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 7: Try n=1280 for more DOF
    if remaining_time() > 150:
        n2 = 1280
        h_n2 = interp_to(best_h, best_n, n2)
        h_n2 = project_feasible(h_n2, n2)
        for beta in [10000, 100000]:
            if remaining_time() < 60:
                break
            h_n2 = slsqp_opt(h_n2, n2, beta, maxiter=500)
        if remaining_time() > 30:
            h_n2, c5_n2 = topk_equalize(h_n2, n2, top_k=5, max_iters=400)
        else:
            c5_n2 = compute_c5(h_n2)
        if c5_n2 < best_c5:
            best_c5 = c5_n2
            best_h = h_n2.copy()
            best_n = n2
        if remaining_time() > 40:
            h_n2_sw, c5_n2_sw = two_swap(h_n2.copy(), n2, max_pairs=4000, seed=314)
            if c5_n2_sw < best_c5:
                best_c5 = c5_n2_sw
                best_h = h_n2_sw.copy()
                best_n = n2

    # Phase 8: Coordinate descent (limited sweeps for safety)
    if remaining_time() > 60:
        h_cd, c5_cd = coord_descent(best_h.copy(), best_n, max_sweeps=10)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 9: Alternative initialization (Thue-Morse) at n=1024
    if remaining_time() > 100:
        h_tm = theta_morse_init(n_hi)
        for beta in [5000, 50000]:
            if remaining_time() < 40:
                break
            h_tm = slsqp_opt(h_tm, n_hi, beta, maxiter=400)
        h_tm, c5_tm = topk_equalize(h_tm, n_hi, top_k=5, max_iters=300)
        if c5_tm < best_c5:
            best_c5 = c5_tm
            best_h = h_tm.copy()
            best_n = n_hi

    # Phase 10: Final 2-swap
    if remaining_time() > 30:
        h_fsw, c5_fsw = two_swap(best_h.copy(), best_n, max_pairs=5000, seed=555)
        if c5_fsw < best_c5:
            best_c5 = c5_fsw
            best_h = h_fsw.copy()

    # Phase 11: Final SLSQP + topk polish
    if remaining_time() > 20:
        h_fin = slsqp_opt(best_h, best_n, 2000000, maxiter=400)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
    if remaining_time() > 10:
        h_eq2, c5_eq2 = topk_equalize(best_h, best_n, top_k=7, max_iters=300)
        if c5_eq2 < best_c5:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
