            h_b, c5_b = binary_swap_search(h_bin_init, n_bp, time_limit=40, rng=rng)
            # Continuous refinement
            h_bc = h_b.copy()
            for beta in [10000, 100000]:
                if remaining_time() < 10:
                    break
                h_bc = slsqp_opt(h_bc, n_bp, beta, maxiter=400)
            c5_bc = compute_c5(h_bc)
            h_bc, c5_bc = targeted_gradient_search(h_bc, n_bp, time_limit=20, top_k=5)
            
            if c5_bc < best_c5 - 1e-15:
                best_c5 = c5_bc
                best_h = h_bc.copy()
                best_n = n_bp
    
    # Phase 7: Final intensive refinement
    n_final = best_n
    h_final = best_h.copy()
    
    for beta in [30000, 100000, 500000, 2000000]:
        if remaining_time() < 12:
            break
        h_new = slsqp_opt(h_final, n_final, beta, maxiter=1000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_final = h_new.copy()
    
    if remaining_time() > 15:
        h_t, c5_t = targeted_gradient_search(h_final, n_final, time_limit=min(30, int(remaining_time() * 0.5)), top_k=4)
        if c5_t < best_c5 - 1e-15:
            best_c5 = c5_t
            best_h = h_t.copy()
    
    # Last SLSQP push
    if remaining_time() > 10:
        h_pol = slsqp_opt(best_h, n_final, 1000000, maxiter=1500)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5 - 1e-15:
            best_c5 = c5_pol
            best_h = h_pol.copy()
    
    # Ensure feasibility
    best_h = project_feasible(best_h, n_final)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    # Validate with evaluator
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass
    
    return (best_h, best_c5, n_final)
