import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
import time

_T0 = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if mask.any():
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    # Final exact adjustment
    if h.sum() < target:
        deficit = target - h.sum()
        idx = np.where((h > 0) & (h < 1))[0]
        if len(idx) > 0:
            h[idx[0]] = min(1.0, h[idx[0]] + deficit)
    else:
        excess = h.sum() - target
        idx = np.where((h > 0) & (h < 1))[0]
        if len(idx) > 0:
            h[idx[0]] = max(0.0, h[idx[0]] - excess)
    return h

def smooth_max_func(x, beta):

    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    s = np.clip(beta * (corr - m), -500.0, 500.0)
    e = np.exp(s)
    return (m + np.log(e.sum()) / beta) * dx

def smooth_max_jac(x, beta):

        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - x[ao:])
            g[ao:] -= wk * x[:np_]
    return g

def slsqp_optimize(h, n, beta, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_max_func, h, args=(beta,),
                       jac=smooth_max_jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': eq_func,
                                     'jac': eq_jac, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def tc_optimize(h, n, beta, maxiter=200):

                       jac=smooth_max_jac, method='trust-constr',
                       bounds=bnds, constraints=cons,

_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if mask.any():
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    # Exact final fix
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        idx = np.where((h > 0.01) & (h < 0.99))[0]
        if len(idx) > 0:
            h[idx[0]] += diff
            h[idx[0]] = np.clip(h[idx[0]], 0.0, 1.0)

def smooth_max_func(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    s = np.clip(beta * (corr - m), -500.0, 500.0)
    e = np.exp(s)
    return (m + np.log(e.sum()) / beta) * dx


def smooth_max_jac(x, beta):

    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    rng = np.random.RandomState(789)
    per_restart = (time_limit - 5) / n_restarts

    for r in range(n_restarts):
        if time.time() - start > time_limit - 5:
            break
        if _BUDGET - (time.time() - _T0) < 15:
            break
        # Create perturbed start
        scale = 0.03 * (1 + r * 0.5)
        h_pert = project_feasible(h + rng.randn(n) * scale, n)
        c_pert = compute_c5(h_pert)
        # Optimize from perturbed start
        h_new = slsqp_optimize(h_pert, n, beta, maxiter=150)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    return best_h, best_c5


def coordinate_polish(h, n, time_limit, max_rounds=500):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    rng = np.random.RandomState(321)

    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 10:
            break

        # Pick random pair
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue

        h_i = best_h[i]
        h_j = best_h[j]

        # Try moving mass between i and j, or adjusting both
        candidates = []
        for delta in [0.001, 0.005, 0.01, 0.02, 0.05]:
            # Move from i to j
            if h_i > delta and h_j < 1.0:
                candidates.append((i, delta, j, -delta))
            # Move from j to i
            if h_j > delta and h_i < 1.0:
                candidates.append((j, delta, i, -delta))
            # Push both toward 0.5 (reduce variance)
            if h_i > 0.5:
                candidates.append((i, -min(delta, h_i - 0.5), j, min(delta, 0.5 - h_j) if h_j < 0.5 else -min(delta, h_j - 0.45) if h_j > 0.5 else 0))
            if h_j > 0.5:
                candidates.append((j, -min(delta, h_j - 0.5), i, min(delta, 0.5 - h_i) if h_i < 0.5 else -min(delta, h_i - 0.45) if h_i > 0.5 else 0))

        found = False
        for (ai, di, bj, dj) in candidates:
            if di == 0 and dj == 0:
                continue
            h_try = best_h.copy()
            h_try[ai] += di
            h_try[bj] += dj
            if 0.0 <= h_try[ai] <= 1.0 and 0.0 <= h_try[bj] <= 1.0:
                h_try = project_feasible(h_try, n)
                c_try = compute_c5(h_try)
                if c_try < best_c5 - 1e-13:
                    best_h = h_try
                    best_c5 = c_try
                    found = True
                    break
        if not found and rnd > 100:
            break

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    np.random.seed(seed)

    def rem():
        return max(0.0, _BUDGET - (time.time() - _T0) - 8.0)

    # Initialize from parent
    try:
        best_h = np.array(initial_h_values, dtype=float).copy()
        best_n = len(best_h)
    except NameError:
        best_n = 512
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with increasing smoothness at original resolution
    if rem() > 40:
        h1, c1 = multi_beta_slsqp(best_h, best_n, [500, 2000, 10000, 40000], min(200.0, rem() * 0.3))
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1.copy()

    # Phase 2: Trust-constr at original resolution
    if rem() > 30:
        for beta in [20000, 80000, 300000]:
            if rem() < 20:
                break
            h_new = tc_optimize(best_h, best_n, beta, maxiter=150)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()

    # Phase 3: Top-K projected gradient descent
    if rem() > 30:
        t_pg = min(120.0, rem() * 0.15)
        h_new, c_new = topk_projected_grad(best_h, best_n, t_pg, K=12, beta_w=80.0, max_iters=30000)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Phase 4: Random restarts to find better basin
    if rem() > 40:
        t_rr = min(150.0, rem() * 0.15)
        h_new, c_new = random_restart_optimize(best_h, best_n, t_rr, n_restarts=6, beta=15000)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Phase 5: Try different resolutions
    res_candidates = []
    for mult in [0.5, 1.5, 2.0]:
        rn = int(round(best_n * mult / 32) * 32)
        if 128 <= rn <= 2048 and rn != best_n and rn % 2 == 0:
            res_candidates.append(rn)
    res_candidates = sorted(set(res_candidates))

    for rn in res_candidates[:3]:
        if rem() < 50:
            break
        h_interp = interp_to(best_h, best_n, rn)
        h_feas = project_feasible(h_interp, rn)
        c_int = compute_c5(h_feas)

        # Optimize at new resolution
        h_new, c_new = multi_beta_slsqp(h_feas, rn, [1000, 5000, 20000], min(150.0, rem() * 0.2))
        if c_new < c_int:
            c_int = c_new
            h_feas = h_new

        if rem() > 30:
            h_new, c_new = topk_projected_grad(h_feas, rn, min(60.0, rem() * 0.12), K=10, beta_w=60.0)
            if c_new < c_int:
                c_int = c_new
                h_feas = h_new

        if c_int < best_c5:
            best_c5 = c_int
            best_h = h_feas.copy()
            best_n = rn

    # Phase 6: Re-run SLSQP + TC after potential resolution change
    if rem() > 30:
        h_new = slsqp_optimize(best_h, best_n, 100000, maxiter=400)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    if rem() > 20:
        h_new = tc_optimize(best_h, best_n, 500000, maxiter=200)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Phase 7: More top-K gradient with tighter settings
    if rem() > 20:
        t_fine = min(60.0, rem() * 0.1)
        h_new, c_new = topk_projected_grad(best_h, best_n, t_fine, K=8, beta_w=200.0, max_iters=20000)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Phase 8: Coordinate polish
    if rem() > 20:
        t_cd = min(60.0, rem() * 0.1)
        h_new, c_new = coordinate_polish(best_h, best_n, t_cd, max_rounds=1000)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Phase 9: Final SLSQP with very high beta
    if rem() > 10:
        for beta in [500000, 2000000]:
            if rem() < 8:
                break
            h_new = slsqp_optimize(best_h, best_n, beta, maxiter=300)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()

    # Phase 10: One more trust-constr
    if rem() > 10:
        h_new = tc_optimize(best_h, best_n, 1000000, maxiter=100)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Phase 11: Final micro top-K gradient
    if rem() > 8:
        h_new, c_new = topk_projected_grad(best_h, best_n, min(30.0, rem() * 0.4), K=5, beta_w=500.0, max_iters=10000)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            best_c5 = min(best_c5, float(eval_c5))
    except Exception:
        pass

    return (best_h, best_c5, best_n)
