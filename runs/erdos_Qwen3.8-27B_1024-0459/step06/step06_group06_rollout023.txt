```python
from scipy.optimize import minimize
import time

_BUDGET = 1300.0

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_corr_k(h, k, n):
    """Gradient of correlation at index k with respect to h."""
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        g[offset:n] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:n]
    else:
        a = -offset
        npairs = n - a
        g[:npairs] += (1.0 - h[a:n])
        g[a:n] -= h[:npairs]
    return g

def smooth_obj_grad(h, n, beta, K):
    """Smooth max objective and gradient using top-K correlations."""
    corr = compute_corr(h)
    m = float(np.max(corr))
    # Get top K indices
    top_idx = np.argsort(-corr)[:K]
    shifted = corr[top_idx] - m
    clamped = np.clip(beta * shifted, -50.0, 50.0)
    exp_vals = np.exp(clamped)
    log_sum = np.log(np.sum(exp_vals))
    obj = (m + log_sum / beta) * (2.0 / n)
    
    weights = exp_vals / np.sum(exp_vals)
    g = np.zeros(n)
    wc = weights * (2.0 / n)
    for i, k in enumerate(top_idx):
        if wc[i] > 1e-14:
            g += wc[i] * grad_corr_k(h, k, n)
    return obj, g

    """L-BFGS-B with soft constraint on sum."""
    def obj(x):

        s = np.sum(x) - n / 2.0
        return val + 0.5 * penalty * s * s

        s = np.sum(x) - n / 2.0
        g += penalty * s
        return g
    
    bounds = [(0.0, 1.0)] * n

                   bounds=bounds, options={'maxiter': maxiter, 
                   'ftol': 1e-18, 'gtol': 1e-12, 'maxfun': maxiter*2,
                   'maxcor': 50})
    if res.x is not None:

def slsqp_run(h, n, beta, K, maxiter):

    def obj(x):
        return smooth_obj_grad(x, n, beta, K)[0]

        return smooth_obj_grad(x, n, beta, K)[1]
```

```python
def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        np_ = n - offset
        g[offset:] += (1.0 - h[:np_])
        g[:np_] -= h[offset:]
    else:
        a = -offset
        np_ = n - a
        g[:np_] += (1.0 - h[a:])
        g[a:] -= h[:np_]
    return g

def smooth_obj_grad(h, n, beta, K):
    corr = compute_corr(h)
    m = float(np.max(corr))
    top_idx = np.argsort(-corr)[:K]
    shifted = corr[top_idx] - m
    clamped = np.clip(beta * shifted, -60.0, 60.0)
    exp_vals = np.exp(clamped)
    s = np.sum(exp_vals)
    obj = (m + np.log(s) / beta) * (2.0 / n)
    weights = exp_vals / s
    g = np.zeros(n)
    for i in range(K):
        wt = weights[i] * (2.0 / n)
        if wt > 1e-14:
            g += wt * grad_corr_k(h, top_idx[i], n)
    return obj, g

def lbfgsb_optimize(h, n, beta, K, penalty, maxiter):
    def obj(x):
        v, _ = smooth_obj_grad(x, n, beta, K)
        s = np.sum(x) - n / 2.0
        return v + 0.5 * penalty * s * s

    def grad(x):
        _, g = smooth_obj_grad(x, n, beta, K)
        s = np.sum(x) - n / 2.0
        g = g + penalty * s
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-12, 'maxfun': maxiter * 3,
                                'maxcor': 80})
        if res.x is not None and np.all(res.x >= -1e-10):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def slsqp_optimize(h, n, beta, K, maxiter):
    def obj(x):
        return smooth_obj_grad(x, n, beta, K)[0]

    def grad(x):
        return smooth_obj_grad(x, n, beta, K)[1]

    def cfun(x):
        return np.sum(x) - n / 2.0

    def cjac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,

        if res.x is not None:

    except Exception:
        pass
    return h

    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n

def subgrad_polish(h, n, time_limit, rng):

    start = time.time()
    it = 0

        if gn < 1e-14:
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_corr_k(h, k, n):
    """Corrected gradient of correlation at output index k w.r.t. h."""
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        np_ = n - offset
        if np_ <= 0:
            return g
        # c_k = sum_{i=0}^{np_-1} h[i] * (1-h[i+offset])
        # d/dh[m] = (1-h[m+offset])*[m < np_] - h[m-offset]*[m >= offset]
        g[:np_] += (1.0 - h[offset:offset + np_])
        g[offset:offset + np_] -= h[:np_]
    else:
        a = -offset
        np_ = n - a
        if np_ <= 0:
            return g
        # c_k = sum_{i=0}^{np_-1} h[i+a] * (1-h[i])
        # d/dh[m] = (1-h[m-a])*[m >= a] - h[m+a]*[m < np_]
        g[a:a + np_] += (1.0 - h[:np_])
        g[:np_] -= h[a:a + np_]
    return g


def smooth_obj_grad(h, n, beta, K):
    corr = compute_corr(h)
    m = float(np.max(corr))
    top_idx = np.argsort(-corr)[:K]
    shifted = corr[top_idx] - m
    clamped = np.clip(beta * shifted, -60.0, 60.0)
    exp_vals = np.exp(clamped)
    s = float(np.sum(exp_vals))
    obj = (m + np.log(s) / beta) * (2.0 / n)
    weights = exp_vals / s
    g = np.zeros(n)
    for i in range(K):
        wt = weights[i] * (2.0 / n)
        if wt > 1e-15:
            g += wt * grad_corr_k(h, int(top_idx[i]), n)
    return obj, g


def lbfgsb_run(h, n, beta, K, penalty, maxiter):
    def obj(x):
        v, _ = smooth_obj_grad(x, n, beta, K)
        s = np.sum(x) - n / 2.0
        return v + 0.5 * penalty * s * s

    def grad(x):
        _, g = smooth_obj_grad(x, n, beta, K)
        s = np.sum(x) - n / 2.0
        g = g + penalty * s
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-13, 'maxfun': maxiter * 3,
                                'maxcor': 100})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def slsqp_run(h, n, beta, K, maxiter):
    def obj(x):
        return smooth_obj_grad(x, n, beta, K)[0]

    def grad(x):
        return smooth_obj_grad(x, n, beta, K)[1]

    def cfun(x):
        return np.sum(x) - n / 2.0

    def cjac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def interp_to(src, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def remain():
    return max(0.0, _BUDGET - (time.time() - _T0) - 8.0)


def subgrad_polish(h, n, time_limit, rng):
    h = h.copy()
    best = compute_c5(h)
    start = time.time()
    it = 0
    while time.time() - start < time_limit and remain() > 10:
        it += 1
        corr = compute_corr(h)
        K = max(5, min(40, 40 - it // 5))
        top_idx = np.argsort(-corr)[:K]
        g = np.zeros(n)
        for k in top_idx:
            g += grad_corr_k(h, int(k), n)
        g /= K
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        base_step = min(0.06 / (1.0 + 0.06 * it), 0.025)
        improved = False
        for mult in [1.0, 0.5, 0.25, 0.1, 2.0, 0.75]:
            ht = project_feasible(h - base_step * mult * g / gn, n)
            ct = compute_c5(ht)
            if ct < best - 1e-15:
                h = ht
                best = ct
                improved = True
                break
        if not improved and it % 50 == 0:
            h = project_feasible(h + rng.randn(n) * 0.002, n)
    return h, best


def equalize_shifts(h, n, time_limit, K=10):
    h = h.copy()
    best = compute_c5(h)
    start = time.time()
    for it in range(1000):
        if time.time() - start > time_limit or remain() < 10:
            break
        corr = compute_corr(h)
        top = np.argsort(-corr)[:K]
        c_max = float(corr[top[0]])
        c_min = float(corr[top[-1]])
        if c_max - c_min < 1e-10:
            break
        g = np.zeros(n)
        for k in top:
            g += grad_corr_k(h, int(k), n)
        g /= K
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        improved = False
        for s in [0.02, 0.01, 0.005, 0.04, 0.002, 0.08, 0.015, 0.008]:
            ht = project_feasible(h - s * g / gn, n)
            ct = compute_c5(ht)
            if ct < best - 1e-15:
                h = ht
                best = ct
                improved = True
                break
        if not improved:
            for s in [0.01, 0.005, 0.02, 0.003]:
                ht = project_feasible(h + s * g / gn, n)
                ct = compute_c5(ht)
                if ct < best - 1e-15:
                    h = ht
                    best = ct
                    improved = True
                    break
        if not improved:
            break
    return h, best

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        parent_h = None

    best_c5 = float('inf')
    best_h = None
    best_n = 0

    if parent_h is not None:
        parent_n = len(parent_h)
        pc5 = compute_c5(parent_h)
        best_c5 = pc5
        best_h = parent_h.copy()
        best_n = parent_n
    else:
        best_n = 512
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_c5 = compute_c5(best_h)

    # Interpolate to n=1024
    n1 = 1024
    h1 = interp_to(best_h, best_n, n1)
    h1 = project_feasible(h1, n1)
    c5_1 = compute_c5(h1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
        best_n = n1

    h = best_h.copy()
    n = best_n

    # Phase 1: L-BFGS-B with corrected gradient, beta sweep at n=1024
    lbfgs_phases = [
        (200, 40, 50, 1500),
        (800, 30, 200, 1500),
        (3000, 20, 1000, 1200),
        (8000, 12, 3000, 1000),
        (20000, 8, 5000, 800),
    ]
    for beta, K, pen, maxit in lbfgs_phases:
        if remain() < 40:
            break
        t_start = time.time()
        h_new = lbfgsb_run(h, n, beta, K, pen, maxit)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            h = project_feasible(0.5 * h + 0.5 * h_new, n)
            c5_try = compute_c5(h)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h.copy()

    # Phase 2: Subgradient polish at n=1024
    if remain() > 40:
        h_sg, c5_sg = subgrad_polish(h, n, min(40.0, remain() * 0.5), rng)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 3: Equalize shifts
    if remain() > 30:
        h_eq, c5_eq = equalize_shifts(h, n, min(30.0, remain() * 0.5), K=10)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 4: L-BFGS-B at n=2048
    n2 = 2048
    h2 = interp_to(best_h, best_n, n2)
    h2 = project_feasible(h2, n2)
    c5_2 = compute_c5(h2)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    h2 = h2.copy()

    if remain() > 60:
        lbfgs_phases_2 = [
            (500, 30, 200, 800),
            (2000, 20, 1000, 800),
            (8000, 10, 3000, 600),
        ]
        for beta, K, pen, maxit in lbfgs_phases_2:
            if remain() < 40:
                break
            h_new = lbfgsb_run(h2, n2, beta, K, pen, maxit)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h2 = h_new.copy()

    # Phase 5: Subgradient at n=2048
    if remain() > 30:
        h_sg2, c5_sg2 = subgrad_polish(h2, n2, min(30.0, remain() * 0.4), rng)
        if c5_sg2 < best_c5:
            best_c5 = c5_sg2
            best_h = h_sg2.copy()
            h2 = h_sg2.copy()

    # Phase 6: Equalize at n=2048
    if remain() > 25:
        h_eq2, c5_eq2 = equalize_shifts(h2, n2, min(25.0, remain() * 0.4), K=8)
        if c5_eq2 < best_c5:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()
            h2 = h_eq2.copy()

    # Phase 7: Active-set fine polish at n=2048
    if remain() > 20:
        h_af, c5_af = active_set_polish(h2, n2, min(20.0, remain() * 0.5))
        if c5_af < best_c5:
            best_c5 = c5_af
            best_h = h_af.copy()

    # Phase 8: Try SLSQP at n=1024 (smaller problem, can allow more iterations)
    n3 = 1024
    h3 = interp_to(best_h, len(best_h), n3)
    h3 = project_feasible(h3, n3)
    c5_3 = compute_c5(h3)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    else:
        h3 = best_h.copy()
        n3 = len(best_h)

    if remain() > 20 and n3 <= 1024:
        h_sl = slsqp_run(h3, n3, 5000, 8, 300)
        c5_sl = compute_c5(h_sl)
        if c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl.copy()

    # Phase 9: Another round of L-BFGS-B at best resolution with very high beta
    if remain() > 30:
        h_final = best_h.copy()
        n_final = len(best_h)
        for beta, K, pen, maxit in [(20000, 6, 5000, 500), (50000, 4, 10000, 400)]:
            if remain() < 20:
                break
            h_new = lbfgsb_run(h_final, n_final, beta, K, pen, maxit)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_final = h_new.copy()

    # Phase 10: Final active-set polish
    if remain() > 10:
        h_af2, c5_af2 = active_set_polish(best_h, len(best_h), min(15.0, remain() * 0.6))
        if c5_af2 < best_c5:
            best_c5 = c5_af2
            best_h = h_af2.copy()

    # Final feasibility check
    best_h = project_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Try evaluate_erdos_solution if available
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if isinstance(eval_c5, float) and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, len(best_h))
```
