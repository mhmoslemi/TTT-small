    h_c = h.copy()
    h_p = h.copy()

            break
        if (it % 200) == 0 and it > 0:

            if c5_chk < best_c5:
                best_c5 = c5_chk
                best_h = h_c.copy()

        if it > 0:
            gamma = (np.sqrt(1.0 + 4.0 * it * it) - 1.0) / (2.0 * it)
            y = h_c + gamma * (h_c - h_p)
