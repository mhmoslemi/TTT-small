    #                                        - sum_{offset<0} w[off] * ...

        offset = k - (n - 1)

        else:
            a = -offset
            npairs = n - a
            g[:npairs] += wk * (1.0 - h[a:])
            g[a:] -= wk * h[:npairs]
    return g, corr, weights

def lse_obj_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -100.0, 100.0)
    exp_s = np.exp(shifted)
    wsum = exp_s.sum()
    weights = exp_s / wsum
    val = (max_c + np.log(wsum) / beta) * dx
    g = np.zeros(n)
    for k in range(2 * n - 1):

        offset = k - (n - 1)

            g[offset:] += wk * (1.0 - h[:npairs])

        else:
            a = -offset
            npairs = n - a
            g[:npairs] += wk * (1.0 - h[a:])
            g[a:] -= wk * h[:npairs]

def topk_lse_obj_grad(h, n, beta, K):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    m = np.max(c_top)
    sh = np.clip(beta * (c_top - m), -100.0, 100.0)
    ex = np.exp(sh)
    wsum = ex.sum()
    w = ex / wsum
    val = (m + np.log(wsum) / beta) * dx
    g = np.zeros(n)

            return topk_lse_obj_grad(x, n, beta, K)[0]

def nesterov_lse(h, n, tlimit, beta, K, use_full):
    h_curr = h.copy()
    h_best = h.copy()
    c5_best = compute_c5(h)
    h_prev = h.copy()
    start = time.time()

    it = 0
    while time.time() - start < tlimit:
        if _BUDGET - (time.time() - _T0) < 5:
            break
        if it == 0:
            y = h_curr.copy()
        else:
            gamma = (it - 1.0 + np.sqrt(1.0 + 4.0 * (it - 1.0) ** 2)) / (2.0 * it)
            y = h_curr + gamma * (h_curr - h_prev)
            y = np.clip(y, 0.0, 1.0)

        if use_full:
            _, g = lse_obj_grad(y, n, beta)
        else:
            _, g = topk_lse_obj_grad(y, n, beta, K)

        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break

        base_step = 1.0 / (beta * gn)
        improved = False
        for factor in [1.0, 0.5, 0.25, 2.0, 0.1, 4.0, 0.05]:
            step = min(base_step * factor, 0.5)
            h_try = project_feasible(y - step * g, n)
            c_try = compute_c5(h_try)
            if c_try < c5_best - 1e-15:
                c5_best = c_try
                h_best = h_try.copy()
                h_curr = h_try.copy()
                improved = True
                break

        if not improved and it > 100:
            step = base_step * 0.3
            h_curr = project_feasible(y - step * g, n)

        h_prev = h_curr.copy()
        it += 1

    return h_best, c5_best

def topk_fixed_idx_slsqp(h, n, idx, beta, maxiter):
    dx = 2.0 / n

    def fun(x):
        corr_k = np.zeros(len(idx))
        for i, k in enumerate(idx):
            offset = k - (n - 1)
            if offset >= 0:
                npairs = n - offset
                corr_k[i] = np.sum(x[:npairs] * (1.0 - x[offset:]))
            else:
                a = -offset
                npairs = n - a
                corr_k[i] = np.sum(x[:npairs] * (1.0 - x[a:]))
        m = np.max(corr_k)
        sh = np.clip(beta * (corr_k - m), -100.0, 100.0)
        ex = np.exp(sh)
        wsum = ex.sum()
        return (m + np.log(wsum) / beta) * dx

    def grad(x):
        corr_k = np.zeros(len(idx))
        for i, k in enumerate(idx):
            offset = k - (n - 1)
            if offset >= 0:
                npairs = n - offset
                corr_k[i] = np.sum(x[:npairs] * (1.0 - x[offset:]))
            else:
                a = -offset
                npairs = n - a
                corr_k[i] = np.sum(x[:npairs] * (1.0 - x[a:]))
        m = np.max(corr_k)
        sh = np.clip(beta * (corr_k - m), -100.0, 100.0)
        ex = np.exp(sh)
        w = ex / ex.sum()
        g = np.zeros(n)
        for i in range(len(idx)):
            if w[i] < 1e-14:
                continue
            wk = w[i] * dx
            k = idx[i]
            offset = k - (n - 1)
            if offset >= 0:
                npairs = n - offset
                g[offset:] += wk * (1.0 - x[:npairs])
                g[:npairs] -= wk * x[offset:]
            else:
                a = -offset
                npairs = n - a
                g[:npairs] += wk * (1.0 - x[a:])
                g[a:] -= wk * x[:npairs]
        return g

    def cfun(x):
        return np.sum(x) - n / 2.0
    def cjac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(fun, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        best_n = len(parent_h)
        best_h = parent_h.copy()
        best_c5 = compute_c5(best_h)
    except Exception:
        best_n = 512
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_c5 = compute_c5(best_h)

    # Phase 1: Polish at parent resolution with full LSE continuation
    h = best_h.copy()
    n = best_n
    betas_coarse = [20, 50, 100, 300, 1000, 3000]
    h, c5 = continuation_slsqp(h, n, betas_coarse, 300, True, 0)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: Go to n=1024
    n2 = 1024
    if remaining() > 120:
        h2 = interp_to(best_h, best_n, n2)
        h2 = project_feasible(h2, n2)
        c5_2 = compute_c5(h2)

        # L-BFGS-B with penalty for fast initial optimization
        if remaining() > 100:
            h2 = lbfgsb_penalty(h2, n2, beta=100, penalty=50.0, maxiter=300)
            h2 = project_feasible(h2, n2)
            c5_2 = compute_c5(h2)

        # SLSQP continuation at n=1024
        betas_med = [50, 150, 500, 2000]
        if remaining() > 60:
            h2, c5_2 = continuation_slsqp(h2, n2, betas_med, 250, True, 0)

        # Nesterov for exploration
        if remaining() > 40:
            sg_time = min(50.0, remaining() * 0.25)
            h2s, c5_2s = nesterov_lse(h2, n2, sg_time, beta=100, K=50, use_full=False)
            if c5_2s < c5_2:
                h2 = h2s.copy()
                c5_2 = c5_2s

        # Active set optimization
        if remaining() > 30:
            h2 = active_set_optimize(h2, n2, maxiter=200)
            c5_2 = compute_c5(h2)

        # Final SLSQP polish at high beta
        if remaining() > 15:
            h2, c5_2 = continuation_slsqp(h2, n2, [5000, 10000, 30000], 200, True, 0)

        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
            best_n = n2

    # Phase 3: Try n=2048 with L-BFGS-B only (SLSQP too slow)
    n3 = 2048
    if remaining() > 200:
        h3 = interp_to(best_h, best_n, n3)
        h3 = project_feasible(h3, n3)
        c5_3 = compute_c5(h3)

        # L-BFGS-B with increasing penalty
        for pen in [20.0, 100.0, 500.0, 2000.0]:
            if remaining() < 30:
                break
            h3 = lbfgsb_penalty(h3, n3, beta=200, penalty=pen, maxiter=200)
            h3 = project_feasible(h3, n3)
            c5_3 = compute_c5(h3)

        # Nesterov at n=2048
        if remaining() > 60:
            sg_time = min(80.0, remaining() * 0.3)
            h3s, c5_3s = nesterov_lse(h3, n3, sg_time, beta=150, K=80, use_full=False)
            if c5_3s < c5_3:
                h3 = h3s.copy()
                c5_3 = c5_3s

        # SLSQP at n=2048 (fewer iterations since QP is expensive)
        if remaining() > 40:
            h3, c5_3 = continuation_slsqp(h3, n3, [200, 1000, 5000], 100, True, 0)

        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
            best_n = n3

    # Phase 4: Fine-tuning at best resolution
    n = best_n
    h = best_h.copy()

    # Active set refinement
    if remaining() > 20:
        h = active_set_optimize(h, n, maxiter=300)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Final SLSQP with very high beta
    for beta in [10000, 30000, 100000]:
        if remaining() < 8:
            break
        h_new = slsqp_lse(best_h, best_n, beta, 400, True, 0)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 5: Multi-start perturbation
    if remaining() > 30:
        for pert_i in range(3):
            if remaining() < 15:
                break
            sigma = 0.005 + 0.005 * pert_i
            h_p = best_h + rng.randn(best_n) * sigma
            h_p = project_feasible(h_p, best_n)
            c5_p = compute_c5(h_p)
            if c5_p < best_c5 + 0.005:
                h_p, c5_p = continuation_slsqp(h_p, best_n, [500, 2000, 10000], 150, True, 0)
                if c5_p < best_c5 - 1e-14:
                    best_c5 = c5_p
                    best_h = h_p.copy()

    # Phase 6: Very fine gradient descent
    if remaining() > 20:
        fg_time = min(30.0, remaining() * 0.4)
        h_fg, c5_fg = nesterov_lse(best_h, best_n, fg_time, beta=500, K=30, use_full=True)
        if c5_fg < best_c5 - 1e-14:
            best_c5 = c5_fg
            best_h = h_fg.copy()

    # Final SLSQP polish
    if remaining() > 5:
        h_new = slsqp_lse(best_h, best_n, 50000, 300, True, 0)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final feasibility enforcement
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
