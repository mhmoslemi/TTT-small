        _SM_BETA = beta
        x = best_h.copy()
        y = x.copy()
        t = 1.0
        lr = 1.0 / (2.0 * beta)

        for it in range(iters_per_beta):
            if remaining_time() < 15:
                break
            g = _sm_gradient(y)
            x_new = project_feasible(y - lr * g, n)
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y = x_new + ((t - 1.0) / t_new) * (x_new - x)
            x = x_new
            t = t_new

            if it % 200 == 0:
                c5 = compute_c5(x)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = x.copy()

        c5 = compute_c5(x)
        if c5 < best_c5:
            best_c5 = c5
            best_h = x.copy()

    return best_h, best_c5


def projected_descend(h, n, k=5, max_iters=2000, step_decay=0.999):
    """Projected gradient descent using top-k gradient (fast)."""
    global _CUR_N
    _CUR_N = n
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.1

    for it in range(max_iters):
        if remaining_time() < 10:
            break
        g = _topk_gradient(h, k)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        d = -g / gn
        # Try a few step sizes
        improved = False
        for s in [step, step * 0.5, step * 2.0, step * 0.1]:
            ht = project_feasible(h + s * d, n)
            c5 = compute_c5(ht)
            if c5 < best_c5 - 1e-15:
                h = ht
                best_c5 = c5
                best_h = ht.copy()
                improved = True
                break
        if not improved:
            step *= step_decay
            if step < 1e-6:
                break
        # Occasionally increase k
        if it % 500 == 0 and it > 0:
            k = min(k + 1, 10)

    return best_h, best_c5


def topk_equalize(h, n, top_k=5, max_iters=400):
    """Equalize the top-k correlation values."""
    global _CUR_N
    _CUR_N = n
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining_time() < 5:
            break
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        c_max = corr[idx_top[0]]
        c_min_top = corr[idx_top[-1]]
        if c_max - c_min_top < 1e-10:
            break
        g_dir = np.zeros(n)
        for pos, ki in enumerate(idx_top):
            w_k = 1.0 if pos == 0 else (-1.0 / (len(idx_top) - 1))
            off = ki - (n - 1)
            if off >= 0:
                np_ = n - off
                g_dir[off:] += w_k * (1.0 - h[:np_])
                g_dir[:np_] -= w_k * h[off:]
            else:
                a = -off
                np_ = n - a
                g_dir[:np_] += w_k * (1.0 - h[a:])
                g_dir[a:] -= w_k * h[:np_]
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        d = g_dir / gn
        improved = False
        for step in [0.005, 0.01, 0.02, 0.002, 0.05, 0.001, 0.08]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-13:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            for step in [0.005, 0.01, 0.02]:
                ht = project_feasible(h - step * d, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-13:
                    h = ht
                    best_c5 = cn
                    improved = True
                    break
            if not improved:
                break
    return h, best_c5


def adaptive_slsqp_schedule(h, n, betas, iters=None):
    """Run SLSQP with a schedule of betas, returning best result."""
    global _CUR_N
    _CUR_N = n
    best = h.copy()
    best_c5 = compute_c5(h)
    for i, beta in enumerate(betas):
        if remaining_time() < 30:
            break
        if iters is None:
            mi = 500
        else:
            mi = iters[i] if i < len(iters) else 500
        h_new = slsqp_opt(best, n, beta, maxiter=mi)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best = h_new.copy()
    return best, best_c5


def multi_restart(h, n, n_restarts=3, rng=None):
    """Try small perturbations and optimize each."""
    global _CUR_N
    if rng is None:
        rng = np.random.RandomState(0)
    _CUR_N = n
    best = h.copy()
    best_c5 = compute_c5(h)
    for r in range(n_restarts):
        if remaining_time() < 60:
            break
        # Add small noise
        noise = rng.randn(n) * 0.05
        h_pert = project_feasible(h + noise, n)
        # Quick FISTA
        for beta in [100, 500]:
            if remaining_time() < 20:
                break
            h_pert, _ = fista_opt(h_pert, n, betas=[beta], iters_per_beta=200)
        # SLSQP
        h_pert = slsqp_opt(h_pert, n, 5000, maxiter=400)
        c5 = compute_c5(h_pert)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best = h_pert.copy()
    return best, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _CUR_N, _SM_BETA
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    # Get parent
    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = np.zeros(640)
        best_h[:320] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick optimization at parent resolution
    h = best_h.copy()
    h, _ = adaptive_slsqp_schedule(h, best_n, [500, 2000, 10000], iters=[300, 400, 500])
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: Transfer to higher resolution n=1280
    target_n = 1280
    h_hi = interp_to(best_h, best_n, target_n)
    h_hi = project_feasible(h_hi, target_n)
    c5_hi = compute_c5(h_hi)

    # Phase 3: FISTA at higher resolution (fast convergence)
    if remaining_time() > 120:
        h_fista, c5_fista = fista_opt(h_hi, target_n,
                                       betas=[50, 100, 200, 500],
                                       iters_per_beta=400)
        if c5_fista < best_c5:
            best_c5 = c5_fista
            best_h = h_fista.copy()
            best_n = target_n

    # Phase 4: SLSQP at higher resolution with increasing beta
    if best_n != target_n:
        h_opt = project_feasible(interp_to(best_h, best_n, target_n), target_n)
        best_n = target_n
    else:
        h_opt = best_h.copy()
    h_opt, c5_opt = adaptive_slsqp_schedule(h_opt, target_n,
                                             [2000, 10000, 50000, 200000],
                                             iters=[400, 500, 500, 500])
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt.copy()

    # Phase 5: Top-k equalization
    if remaining_time() > 30:
        h_eq, c5_eq = topk_equalize(best_h, best_n, top_k=5, max_iters=500)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Phase 6: Projected gradient with top-k (fast iterations)
    if remaining_time() > 60:
        h_pg, c5_pg = projected_descend(best_h, best_n, k=5, max_iters=3000)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Phase 7: More SLSQP after the above
    if remaining_time() > 40:
        h_opt2, c5_opt2 = adaptive_slsqp_schedule(best_h, best_n,
                                                   [100000, 500000, 2000000],
                                                   iters=[400, 500, 400])
        if c5_opt2 < best_c5:
            best_c5 = c5_opt2
            best_h = h_opt2.copy()

    # Phase 8: Multi-restart to escape local optima
    if remaining_time() > 120:
        h_mr, c5_mr = multi_restart(best_h, best_n, n_restarts=3, rng=rng)
        if c5_mr < best_c5:
            best_c5 = c5_mr
            best_h = h_mr.copy()

    # Phase 9: Final top-k equalization with larger k
    if remaining_time() > 25:
        h_eq2, c5_eq2 = topk_equalize(best_h, best_n, top_k=8, max_iters=300)
        if c5_eq2 < best_c5:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()

    # Phase 10: Try even higher resolution if time permits
    if remaining_time() > 200:
        target_n2 = 1920
        h_hi2 = interp_to(best_h, best_n, target_n2)
        h_hi2 = project_feasible(h_hi2, target_n2)
        h_hi2, c5_hi2 = fista_opt(h_hi2, target_n2, betas=[100, 500], iters_per_beta=300)
        h_hi2 = slsqp_opt(h_hi2, target_n2, 50000, maxiter=400)
        c5_hi2 = compute_c5(h_hi2)
        h_eq3, c5_eq3 = topk_equalize(h_hi2, target_n2, top_k=5, max_iters=200)
        if c5_eq3 < best_c5:
            best_c5 = c5_eq3
            best_h = h_eq3.copy()
            best_n = target_n2

    # Phase 11: Final SLSQP polish
    if remaining_time() > 20:
        h_final = slsqp_opt(best_h, best_n, 5000000, maxiter=400)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
