                break
            base = corr - dcorr[i]  # corr if h[i] removed (set to 0)
            # Adding h[j]=1 (from 0): new_corr[k] = base[k] + dcorr_j_at_zero[k]
            # When h[j] was 0, the contribution was 0. Setting to 1: change = (1-h[j+s])*1[...] - h[j-s]*1[...]
            # But since h[j] was 0, we need the contribution when h[j]=1:
            # Actually dcorr[j, k] is the contribution of h[j] (regardless of its value)
            # If h[j]=0, contribution to corr[k] from position j is:
            #   (1-h[j+s])*0 - h[j-s]*0 ... no that's wrong.
            # Let me recompute. corr[k] = sum_i h[i]*(1-h[i+s])
            # Contribution of position i to corr[k]:
            #   h[i]*(1-h[i+s]) where i is the "first" index
            #   + h[i-s]*(-h[i]) where i is the "second" index (i.e., h[i] appears as h[j+s] with j=i-s)
            # Wait, I defined dcorr differently. Let me just use the Jacobian.
            # dcorr[i, k] = dc[k]/dh[i] (the partial derivative)
            # If h[i] goes from 1 to 0, corr[k] changes by -dcorr[i, k] (approximately, since it's linear for binary)
            # If h[j] goes from 0 to 1, corr[k] changes by +dcorr[j, k]
            # But for binary, it's exact: the change is exactly dcorr[i,k]*(-1) + dcorr[j,k]*(+1)
            # Wait no. dc[k] = sum_i h[i](1-h[i+s]) is bilinear, not linear.
            # For binary h, going h[i] from 1 to 0:
            #   New corr[k] = old_corr[k] - (1-h[i+s])*1[valid] + h[i-s]*1[valid]
            #   = old_corr[k] - dcorr[i, k] (where dcorr[i,k] is the derivative at the current h)
            # Hmm, this is getting confusing. Let me just compute the new corr directly for promising swaps.

            # Better approach: for each candidate j in zeros_idx, compute the new max
            # new_corr[k] = corr[k] - contrib_i_to_corr[k] + contrib_j_to_corr[k]
            # where contrib_i is the actual contribution of h[i]=1 to corr[k]
            # and contrib_j is the contribution h[j] would add if set to 1

            # For binary h, contrib of h[i] to corr[k]:
            # corr[k] = sum_m h[m](1-h[m+s])
            # h[i] contributes: h[i]*(1-h[i+s]) for the "first" role
            #                + (-h[i-s]*h[i]) for the "second" role (when m = i-s)
            # Since h[i]=1: first role gives (1-h[i+s]), second role gives -h[i-s]
            # This is exactly dcorr[i, k]!

            # So: new_corr[k] = corr[k] - dcorr[i, k] + dcorr[j, k] (for binary)
            # But we need to be careful: when h[j] changes from 0 to 1, dcorr[j,k] is the
            # derivative, which for the bilinear form equals the actual change (since the
            # h[i]*h[j] terms are linear in each variable when the other is fixed).

            # Actually for bilinear f(h) = sum h[m](1-h[m+s]):
            # f(h) - f(h - e_i) = (partial f / partial h_i) evaluated at h - e_i? No...
            # f(h) = sum_m h[m] - sum_m h[m]h[m+s]
            # f(h - e_i) = sum_m(h[m]-delta_mi) - sum_m (h[m]-delta_mi)(h[m+s]-delta_{m+s,i})
            # = sum_m h[m] - 1 - sum_m h[m]h[m+s] + h[i]h[i+s] + h[i-s]h[i] - delta stuff
            # This is getting messy. Let me just use: for binary, the change is exact.
            # corr_new[k] = corr[k] - (1-h[i+s])*valid1_i - (-h[i-s])*valid2_i + (1-h[j+s])*valid1_j + (-h[j-s])*valid2_j
            # where valid1_i = 1[0<=i+s<n], valid2_i = 1[0<=i-s<n]
            # (removing h[i]=1, adding h[j]=1)

            # This is: corr[k] - dcorr[i,k] + dcorr[j,k] where dcorr is as I computed it.
            # Let me verify: dcorr[i,k] = (1-h[i+s])*1[0<=i+s<n] - h[i-s]*1[0<=i-s<n]
            # Removing h[i] (was 1, now 0): corr decreases by the contribution of h[i] which is dcorr[i,k]
            # Adding h[j] (was 0, now 1): corr increases by dcorr[j,k]
            # So new_corr[k] = corr[k] - dcorr[i,k] + dcorr[j,k]. YES!

            new_maxs = np.max(corr[None, :] - dcorr[i][None, :] + dcorr[zeros_idx][:, :], axis=1)
            best_pos = int(np.argmin(new_maxs))
            best_new_max = float(new_maxs[best_pos])

            if best_new_max < c_max_val - 1e-10:
                j = zeros_idx[best_pos]
                h[i] = 0.0
                h[j] = 1.0
                h = project_feasible(h, n)
                c5_new = compute_c5(h)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    improved = True
                    break

        if not improved:
            # Try fractional adjustments: nudge one value slightly
            peak_k = int(np.argmax(corr))
            s_peak = peak_k - (n - 1)
            # Find positions where reducing h[i] would reduce corr[peak_k] the most
            # dc[peak_k]/dh[i] < 0 means increasing h[i] reduces corr[peak_k]
            neg_grad = np.where(dcorr[:, peak_k] < -0.01)[0]
            pos_grad = np.where(dcorr[:, peak_k] > 0.01)[0]
            # Try: decrease a high-h[i] with positive gradient, increase a low-h[i] with negative gradient
            for i in neg_grad[:20]:
                if h[i] < 0.1:
                    continue
                for j in pos_grad[:20]:
                    if h[j] > 0.9:
                        continue
                    # Try small swap
                    delta = 0.05
                    h_try = h.copy()
                    h_try[i] -= delta
                    h_try[j] += delta
                    h_try = project_feasible(h_try, n)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-12:
                        h = h_try
                        best_c5 = c5_try
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            break

    return h, best_c5


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    x_new = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(x_new, xp, src_h)


def subgradient_descent(h, n, time_limit=40):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    n_iter = max(30, int(time_limit * 30))

    for it in range(n_iter):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 5:
            break

        progress = it / max(n_iter, 1)
        K = 40 if progress < 0.3 else (20 if progress < 0.7 else 10)
        bw = 50.0 if progress < 0.3 else (100.0 if progress < 0.7 else 200.0)

        corr = compute_corr(h)
        idx = np.argsort(-corr)[:K]
        c_top = corr[idx]
        weights = np.exp(bw * (c_top - c_top.max()))
        w_sum = np.sum(weights)
        if w_sum < 1e-30:
            weights = np.ones(K) / K
        else:
            weights /= w_sum

        g = np.zeros(n)
        nfull = 2 * n - 1
        for wt, k in zip(weights, idx):
            if wt < 1e-14:
                continue
            offset = k - (n - 1)
            if offset >= 0:
                np_ = n - offset
                g[offset:offset + np_] += wt * (1.0 - h[:np_])
                g[:np_] -= wt * h[offset:offset + np_]
            else:
                a = -offset
                np_ = n - a
                g[:np_] += wt * (1.0 - h[a:a + np_])
                g[a:a + np_] -= wt * h[:np_]

        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break

        base_step = 0.03 / (1.0 + 0.03 * it)
        step = min(base_step, 0.15) * n / (gn + 1e-14)

        h_new = project_feasible(h - step * g, n)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            for factor in [0.5, 0.25, 0.1]:
                h_try = project_feasible(h - step * factor * g, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    break

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)

    # Phase 0: Initialize
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        best_n = len(parent_h)
        best_h = project_feasible(parent_h, best_n)
    except Exception:
        best_n = 512
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_h = project_feasible(best_h, best_n)

    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP at parent resolution
    n = best_n
    h = best_h.copy()
    for beta in [300, 1000, 5000, 20000, 80000]:
        if remaining() < 15:
            break
        h_new = slsqp_optimize(h.copy(), n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: LP refinement at parent resolution
    if remaining() > 30:
        h_lp, c5_lp = lp_refine(h.copy(), n, max_rounds=25)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp.copy()
            h = h_lp.copy()

    # Phase 3: Local pair search
    if remaining() > 20:
        h_ls, c5_ls = local_pair_search(h.copy(), n, time_limit=min(40.0, remaining() * 0.3))
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
            h = h_ls.copy()

    # Phase 4: SLSQP + LP after local search
    for beta in [30000, 100000, 300000]:
        if remaining() < 10:
            break
        h_new = slsqp_optimize(h.copy(), n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    if remaining() > 30:
        h_lp2, c5_lp2 = lp_refine(h.copy(), n, max_rounds=20)
        if c5_lp2 < best_c5:
            best_c5 = c5_lp2
            best_h = h_lp2.copy()
            h = h_lp2.copy()

    # Phase 5: Try higher resolutions
    res_candidates = []
    for mult in [1.25, 1.5, 2.0, 2.5, 3.0]:
        r = int(round(best_n * mult / 32)) * 32
        if 96 <= r <= 4096 and r % 2 == 0 and r != best_n:
            res_candidates.append(r)
    res_candidates = sorted(set(res_candidates))

    for n_t in res_candidates:
        if remaining() < 80:
            break

        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        c5_t = compute_c5(h_feas)

        # SLSQP at new resolution
        for beta in [1000, 5000, 30000]:
            if remaining() < 20:
                break
            h_cur = slsqp_optimize(h_feas.copy(), n_t, beta, maxiter=400)
            c5_cur = compute_c5(h_cur)
            if c5_cur < c5_t:
                c5_t = c5_cur
                h_feas = h_cur.copy()

        # LP refinement at new resolution
        if remaining() > 40:
            h_lp_t, c5_lp_t = lp_refine(h_feas.copy(), n_t, max_rounds=15)
            if c5_lp_t < c5_t:
                c5_t = c5_lp_t
                h_feas = h_lp_t.copy()

        # Local search (only for manageable sizes)
        if n_t <= 1200 and remaining() > 25:
            h_ls_t, c5_ls_t = local_pair_search(h_feas.copy(), n_t, time_limit=min(25.0, remaining() * 0.2))
            if c5_ls_t < c5_t:
                c5_t = c5_ls_t
                h_feas = h_ls_t.copy()

        # Final SLSQP polish
        if remaining() > 15:
            h_fin = slsqp_optimize(h_feas.copy(), n_t, 100000, maxiter=600)
            c5_fin = compute_c5(h_fin)
            if c5_fin < c5_t:
                c5_t = c5_fin
                h_feas = h_fin.copy()

        # LP refinement again after all
        if remaining() > 30:
            h_lp_t2, c5_lp_t2 = lp_refine(h_feas.copy(), n_t, max_rounds=10)
            if c5_lp_t2 < c5_t:
                c5_t = c5_lp_t2
                h_feas = h_lp_t2.copy()

        if c5_t < best_c5 - 1e-13:
            best_c5 = c5_t
            best_h = h_feas.copy()
            best_n = n_t
            n = n_t
            h = h_feas.copy()

    # Phase 6: Intensive refinement at best resolution
    if remaining() > 30:
        for beta in [50000, 200000, 500000]:
            if remaining() < 10:
                break
            h_new = slsqp_optimize(best_h.copy(), best_n, beta, maxiter=800)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()

    if remaining() > 25:
        h_lp3, c5_lp3 = lp_refine(best_h.copy(), best_n, max_rounds=30)
        if c5_lp3 < best_c5:
            best_c5 = c5_lp3
            best_h = h_lp3.copy()

    # Phase 7: Random perturbations to escape local optima
    if remaining() > 40:
        for attempt in range(5):
            if remaining() < 30:
                break
            perturbation = rng.randn(best_n) * 0.02
            h_pert = project_feasible(best_h + perturbation, best_n)
            c5_pert = compute_c5(h_pert)
            if c5_pert < best_c5 + 0.001:  # only if not too far
                # Optimize from perturbed point
                h_opt = slsqp_optimize(h_pert.copy(), best_n, 100000, maxiter=600)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                h_lp_r, c5_lp_r = lp_refine(h_opt.copy(), best_n, max_rounds=10)
                if c5_lp_r < best_c5:
                    best_c5 = c5_lp_r
                    best_h = h_lp_r.copy()

    # Phase 8: Final polish
    for beta in [200000, 500000, 1000000]:
        if remaining() < 5:
            break
        h_new = slsqp_optimize(best_h.copy(), best_n, beta, maxiter=800)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final LP with very small steps
    if remaining() > 20:
        h_final_lp, c5_final_lp = lp_refine(best_h.copy(), best_n, max_rounds=20,
                                             eps_list=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
        if c5_final_lp < best_c5:
            best_c5 = c5_final_lp
            best_h = h_final_lp.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
