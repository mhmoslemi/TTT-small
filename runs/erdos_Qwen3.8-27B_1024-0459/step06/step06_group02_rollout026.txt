        if s < 1e-30:
            return np.zeros(n)
        w = ex / s
        g = np.zeros(n)
        # Only compute for lags with significant weight
        threshold = 1e-10
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < threshold:
                continue
            offset = k - (n - 1)
            if offset >= 0:
                m_pairs = n - offset
                if m_pairs > 0:
                    g[offset:offset+m_pairs] += wk * (1.0 - x[:m_pairs])
                    g[:m_pairs] -= wk * x[offset:offset+m_pairs]
            else:
                a = -offset
                m_pairs = n - a
                if m_pairs > 0:
                    g[:m_pairs] += wk * (1.0 - x[a:a+m_pairs])
                    g[a:a+m_pairs] -= wk * x[:m_pairs]
        return g * dx
