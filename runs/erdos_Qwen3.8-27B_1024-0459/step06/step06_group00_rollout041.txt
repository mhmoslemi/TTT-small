import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(v, n):
    v = np.array(v, dtype=float).copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(x)
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(x)
    if abs(diff) > 1e-14:
        mask = (x > 1e-10) & (x < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
        else:
            x[:] += diff / n
            x = np.clip(x, 0.0, 1.0)
    return x


class _OptState:
    def __init__(self, n, beta, penalty):
        self.n = n
        self.beta = beta
        self.penalty = penalty


def _sm_obj(x, st):
    n = st.n
    beta = st.beta
    corr = compute_corr(x)
    c_max = np.max(corr)
    z = np.clip(beta * (corr - c_max), -50.0, 50.0)
    sm = c_max + np.log(np.sum(np.exp(z))) / beta
    s = np.sum(x) - n / 2.0
    return (sm + st.penalty * s * s) * (2.0 / n)


def _sm_grad(x, st):
    n = st.n
    beta = st.beta
    dx = 2.0 / n
    corr = compute_corr(x)
    c_max = np.max(corr)
    z = np.clip(beta * (corr - c_max), -50.0, 50.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        if d >= 0:
            np_ = n - d
            g[d:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[d:]
        else:
            a = -d
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    s = np.sum(x) - n / 2.0
    g += 2.0 * st.penalty * s * dx
    return g


def lbfgs_solve(h, n, beta, penalty, maxiter=200, time_limit=60):
    st = _OptState(n, beta, penalty)
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    start = time.time()
    try:
        res = minimize(_sm_obj, h0, jac=lambda x: _sm_grad(x, st),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-14, 'maxfun': maxiter * 4})
        return project_feasible(res.x, n)
    except Exception:
        return h0


def slsqp_solve(h, n, beta, penalty, maxiter=500, time_limit=60):
    st = _OptState(n, beta, penalty)
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    start = time.time()
    try:
        res = minimize(_sm_obj, h0, jac=lambda x: _sm_grad(x, st),
                       method='SLSQP', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h0


def subgradient_zigzag(h, n, max_iters=5000, time_limit=90):
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    k_prev = None
    stale = 0
    corr_arr = np.zeros(2 * n - 1)

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if remaining_time() < 5:
            break
        corr = compute_corr(h)
        k_curr = int(np.argmax(corr))
        if k_prev is not None and k_curr != k_prev:
            g = 0.5 * (_grad_single(h, k_curr, n) + _grad_single(h, k_prev, n))
            k_prev = k_curr
        else:
            g = _grad_single(h, k_curr, n)
            if k_prev is None:
                k_prev = k_curr
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g = g / gn
        improved = False
        base = max(0.001, 0.03 * (1.0 - 0.5 * it / max_iters))
        for step in [base * 0.1, base * 0.3, base, base * 3.0, base * 10,
                     0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
            ht = project_feasible(h - step * g, n)
            cn = compute_c5(ht)

                improved = True

def _grad_single(h, k, n):
    g = np.zeros(n)
    d = k - (n - 1)
    if d >= 0:
        np_ = n - d
        g[d:] += (1.0 - h[:np_])
        g[:np_] -= h[d:]
    else:
        a = -d
        np_ = n - a
        g[:np_] += (1.0 - h[a:])
        g[a:] -= h[:np_]
    return g


def top_k_dir(h, n, top_k=5, max_iters=500, time_limit=60):

def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def theta_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)


def golden_ratio_seq(n):
    h = np.zeros(n)
    phi = 0.5 * (1.0 + math.sqrt(5.0))
    for i in range(n):
        frac = (i * phi) % 1.0
        h[i] = 1.0 if frac < 0.5 else 0.0
    return project_feasible(h, n)
