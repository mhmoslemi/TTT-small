    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        if interior.any():
            h[interior] += diff / interior.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def corr_gradient_components(h, n, k_idx):
    """Return the gradient of corr[k_idx] * dx w.r.t. h."""
    dx = 2.0 / n
    off = k_idx - (n - 1)
    g = np.zeros(n)
    if off >= 0:
        np_ = n - off
        g[off:] += (1.0 - h[:np_])
        g[:np_] -= h[off:]
    else:
        ao = -off
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:])
        g[ao:] -= h[:np_]
    return g * dx

def sqp_step(h, n, K, max_iter=50, local_time=30.0):

    start = time.time()
    h_curr = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h_curr)

        if time.time() - start > local_time:
            break
        if _BUDGET - (time.time() - _T0) < 15:

            best_h = h_curr.copy()

            A[i] = g_k

            # constraint: corr_k(h_curr)*dx + g_k · (h - h_curr) <= t
            # i.e.: g_k · h <= t - corr_k(h_curr)*dx + g_k · h_curr
            b[i] = -corr[k_idx] * dx + np.dot(g_k, h_curr)

        #      sum(h) = n/2
        #      0 <= h <= 1

        # s.t. A_ub @ [h; t] <= b_ub
        #      A_eq @ [h; t] = b_eq
        #      bounds
        
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0  # minimize t

        b_ub = b
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        
        bounds_lp = [(0, 1)] * n + [(best_c5 - 1e-6, best_c5)]
        # t is bounded: 0 <= t <= best_c5 (we only care if we can go below best_c5)
        
        try:

            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

            if res.success:

                    if c_new < best_c5:
                        pass
                    else:
                        break
            else:
                break
        except Exception:
            break
    
    return best_h, best_c5

        for alpha in [1.0, 0.8, 0.6, 0.5, 0.3, 0.2, 0.1, 0.05, 0.01]:
            h_try = h_curr + alpha * d
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try)
            if c_try < best_c5 - 1e-15:
                h_curr = h_try
                best_c5 = c_try
                best_h = h_curr.copy()
                improved = True
                break
        if not improved:
            no_improve += 50
    return best_h, best_c5


def topk_grad_descent(h, n, time_limit, K=30, beta=100.0):
    start = time.time()
    h_curr = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h_curr)
    for it in range(30000):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 15:
            break
        corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
        c5_val = float(corr.max()) * (2.0 / n)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h_curr.copy()
        idx = np.argsort(-corr)[:K]
        c_top = corr[idx]
        s = np.clip(beta * (c_top - c_top.max()), -200.0, 200.0)
        e = np.exp(s)
        w = e / e.sum()
        g = np.zeros(n)
        for i, k_idx in enumerate(idx):
            wk = w[i]
            if wk < 1e-14:
                continue
            g += wk * corr_grad(h_curr, n, k_idx)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        improved = False
        gap = best_c5 - c5_val
        step0 = max(1e-8, min(0.5, (gap + 1e-10) / (gn * gn + 1e-30)))
        for mult in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05, 0.01]:
            alpha = step0 * mult
            h_try = h_curr - alpha * g
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try)
            if c_try < best_c5 - 1e-15:
                h_curr = h_try
                best_c5 = c_try
                best_h = h_curr.copy()
                improved = True
                break
        if not improved:
            for alpha in [1e-5, 5e-6, 1e-6, 5e-7]:
                h_try = h_curr - alpha * g / gn
                h_try = project_feasible(h_try, n)
                c_try = compute_c5(h_try)
                if c_try < best_c5 - 1e-16:
                    h_curr = h_try
                    best_c5 = c_try
                    best_h = h_curr.copy()
                    improved = True
                    break
            if not improved:
                break
    return best_h, best_c5


def slsqp_smooth(h, n, beta, K, maxiter):
    def obj_sm(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        idx = np.argsort(-corr)[:K]
        c_top = corr[idx]
        s = np.clip(beta * (c_top - c_top.max()), -200.0, 200.0)
        e = np.exp(s)
        dx = 2.0 / n
        return (c_top.max() + np.log(e.sum()) / beta) * dx

    def jac_sm(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        idx = np.argsort(-corr)[:K]
        c_top = corr[idx]
        s = np.clip(beta * (c_top - c_top.max()), -200.0, 200.0)
        e = np.exp(s)
        w = e / e.sum()
        g = np.zeros(n)
        for i, k_idx in enumerate(idx):
            if w[i] < 1e-14:
                continue
            g += w[i] * corr_grad(x, n, k_idx)
        return g

    def eq_fun(x):
        return x.sum() - n / 2.0

    def eq_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_sm, h.copy(), jac=jac_sm, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def tc_optimize(h, n, beta, K, maxiter):
    def obj_sm(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        idx = np.argsort(-corr)[:K]
        c_top = corr[idx]
        s = np.clip(beta * (c_top - c_top.max()), -200.0, 200.0)
        e = np.exp(s)
        dx = 2.0 / n
        return (c_top.max() + np.log(e.sum()) / beta) * dx

    def jac_sm(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        idx = np.argsort(-corr)[:K]
        c_top = corr[idx]
        s = np.clip(beta * (c_top - c_top.max()), -200.0, 200.0)
        e = np.exp(s)
        w = e / e.sum()
        g = np.zeros(n)
        for i, k_idx in enumerate(idx):
            if w[i] < 1e-14:
                continue
            g += w[i] * corr_grad(x, n, k_idx)
        return g

    bnds = Bounds(np.zeros(n), np.ones(n))
    A_eq = np.ones((1, n))
    cons = LinearConstraint(A_eq, n / 2.0, n / 2.0)
    try:
        res = minimize(obj_sm, h.copy(), jac=jac_sm, method='trust-constr',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'gtol': 1e-12,
                                'xtol': 1e-14, 'barrier_tol': 1e-14})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def perturb_h(h, n, rng, scale=0.01):
    h_p = h.copy()
    noise = rng.randn(n) * scale
    h_p = h_p + noise
    return project_feasible(h_p, n)


def targeted_hill_climb(h, n, time_limit, n_rounds=200):
    start = time.time()
    h_curr = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h_curr)
    rng = np.random.RandomState(12345)
    for rnd in range(n_rounds):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 15:
            break
        corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
        k_max = int(np.argmax(corr))
        off = k_max - (n - 1)
        improved = False
        n_one = np.where(h_curr > 0.5)[0]
        n_zero = np.where(h_curr < 0.5)[0]
        n_try = min(300, len(n_one) * len(n_zero))
        if len(n_one) == 0 or len(n_zero) == 0:
            break
        for _ in range(n_try):
            i1 = n_one[rng.randint(len(n_one))]
            i0 = n_zero[rng.randint(len(n_zero))]
            h_try = h_curr.copy()
            delta = 0.3
            h_try[i1] -= delta
            h_try[i0] += delta
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try)
            if c_try < best_c5 - 1e-14:
                h_curr = h_try
                best_c5 = c_try
                best_h = h_try.copy()
                improved = True
                n_one = np.where(h_curr > 0.5)[0]
                n_zero = np.where(h_curr < 0.5)[0]
                break
        if not improved and rnd > 50:
            break
    return best_h, best_c5


def optimize_resolution(h, n, time_limit):
    if time_limit <= 2:
        return h, compute_c5(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_start = time.time()

    t_sqp = min(time_limit * 0.35, 120.0)
    if t_sqp > 10:
        h_sqp, c_sqp = sqp_optimize(best_h, n, t_sqp, K=30)
        if c_sqp < best_c5:
            best_c5 = c_sqp
            best_h = h_sqp
        time_left = time_limit - (time.time() - t_start)
    else:
        time_left = time_limit

    if time_left > 10:
        t_gd = min(time_left * 0.3, 80.0)
        h_gd, c_gd = topk_grad_descent(best_h, n, t_gd, K=40, beta=200.0)
        if c_gd < best_c5:
            best_c5 = c_gd
            best_h = h_gd
        time_left -= t_gd

    if time_left > 10:
        for beta in [50000, 200000, 800000]:
            if time_left < 10:
                break
            h_sl = slsqp_smooth(best_h, n, beta, K=30, maxiter=200)
            c_sl = compute_c5(h_sl)
            if c_sl < best_c5:
                best_c5 = c_sl
                best_h = h_sl
            time_left = time_limit - (time.time() - t_start)

    if time_left > 8:
        h_tc = tc_optimize(best_h, n, 300000, K=30, maxiter=80)
        c_tc = compute_c5(h_tc)
        if c_tc < best_c5:
            best_c5 = c_tc
            best_h = h_tc
        time_left = time_limit - (time.time() - t_start)

    if time_left > 10:
        t_hs = min(time_left * 0.3, 40.0)
        h_hs, c_hs = targeted_hill_climb(best_h, n, t_hs)
        if c_hs < best_c5:
            best_c5 = c_hs
            best_h = h_hs
        time_left = time_limit - (time.time() - t_start)

    if time_left > 5:
        h_gd2, c_gd2 = topk_grad_descent(best_h, n, time_left * 0.5, K=15, beta=500.0)
        if c_gd2 < best_c5:
            best_c5 = c_gd2
            best_h = h_gd2

    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    try:
        best_h = np.array(initial_h_values, dtype=float).copy()
        best_n = len(best_h)
        best_c5 = compute_c5(best_h)
    except Exception:
        best_n = 512
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_c5 = compute_c5(best_h)

    def rem():
        return max(0.0, _BUDGET - (time.time() - _T0) - 10.0)

    # Phase 1: Intensive optimization at parent resolution
    if rem() > 60:
        t1 = min(250.0, rem() * 0.25)
        h1, c1 = optimize_resolution(best_h, best_n, t1)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1.copy()

    # Phase 2: Try n=1024
    if rem() > 80:
        n2 = 1024
        h2 = interp_to(best_h, best_n, n2)
        h2 = project_feasible(h2, n2)
        t2 = min(300.0, rem() * 0.3)
        h2_opt, c2 = optimize_resolution(h2, n2, t2)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2_opt.copy()
            best_n = n2

    # Phase 3: Try n=2048
    if rem() > 100:
        n3 = 2048
        h3 = interp_to(best_h, best_n, n3)
        h3 = project_feasible(h3, n3)
        t3 = min(350.0, rem() * 0.3)
        h3_opt, c3 = optimize_resolution(h3, n3, t3)
        if c3 < best_c5:
            best_c5 = c3
            best_h = h3_opt.copy()
            best_n = n3

    # Phase 4: Final polishing at best resolution
    if rem() > 30:
        t4 = min(200.0, rem() * 0.4)
        h4, c4 = optimize_resolution(best_h, best_n, t4)
        if c4 < best_c5:
            best_c5 = c4
            best_h = h4.copy()

    # Phase 5: Perturbation restarts
    for restart in range(3):
        if rem() < 40:
            break
        h_pert = perturb_h(best_h, best_n, rng, scale=0.005)
        t_r = min(120.0, rem() * 0.2)
        h_r, c_r = optimize_resolution(h_pert, best_n, t_r)
        if c_r < best_c5:
            best_c5 = c_r
            best_h = h_r.copy()

    # Phase 6: Very fine final gradient steps
    if rem() > 10:
        h_fin, c_fin = topk_grad_descent(best_h, best_n, min(60.0, rem() * 0.4),
                                         K=10, beta=1000.0)
        if c_fin < best_c5:
            best_c5 = c_fin
            best_h = h_fin.copy()

    # Final feasibility enforcement
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
