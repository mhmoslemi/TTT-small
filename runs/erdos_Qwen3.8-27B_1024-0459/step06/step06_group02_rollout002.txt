- If d ≥ 0: sum_{i=0}^{n-1-d} h[i] * (1-h[i+d])
- If d < 0: sum_{i=-d}^{n-1} h[i] * (1-h[i+d]) = sum_{i=0}^{n-1+d} h[i+d] * (1-h[i])

And C₅ = max over all d of these sums * dx.

import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(x, n, beta, top_k):
    corr = compute_corr(x)
    if top_k is not None and top_k < len(corr):
        idx = np.argpartition(-corr, top_k)[:top_k]
        cv = corr[idx]
    else:
        idx = np.arange(len(corr))
        cv = corr
    m = np.max(cv)
    sh = beta * (cv - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    s = np.sum(ex)
    w = ex / s
    dx = 2.0 / n
    val = (m + np.log(s) / beta) * dx
    return val, w, idx

def smooth_max_grad(x, n, beta, top_k):
    _, w, idx = smooth_max_obj(x, n, beta, top_k)
    dx = 2.0 / n
    g = np.zeros(n)
    center = n - 1
    for i in range(len(idx)):
        wt = w[i] * dx
        if wt < 1e-15:
            continue
        k = idx[i]
        offset = k - center
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                g[offset:] += wt * (1.0 - x[:npairs])
                g[:npairs] -= wt * x[offset:]
        else:
            a = -offset
            npairs = n - a
            if npairs > 0:
                g[:npairs] += wt * (1.0 - x[a:])
                g[a:] -= wt * x[:npairs]
    return g

def slsqp_solve(h, n, beta, top_k, maxiter, ftol):
    def obj(x):
        return smooth_max_obj(x, n, beta, top_k)[0]
    def jac(x):
        return smooth_max_grad(x, n, beta, top_k)
    def cfun(x):
        return np.sum(x) - n / 2.0
    def cjac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    best_x = h.copy()
    best_f = compute_c5(h)
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        if res.x is not None:
            hx = project_feasible(res.x, n)
            cf = compute_c5(hx)
            if cf < best_f:
                best_x = hx
                best_f = cf
    except Exception:
        pass
    return best_x, best_f

def lbfgsb_solve(h, n, beta, top_k, maxiter, tol):
    pen = 50.0
    def obj(x):
        base = smooth_max_obj(x, n, beta, top_k)[0]
        pen_term = pen * (np.sum(x) - n / 2.0) ** 2
        bound_pen = 0.0
        for xi in x:
            if xi < 0:
                bound_pen += 10.0 * xi * xi
            elif xi > 1:
                bound_pen += 10.0 * (xi - 1.0) ** (2,)
        return base + pen_term + bound_pen
    def jac(x):
        g = smooth_max_grad(x, n, beta, top_k)
        g += 2.0 * pen * (np.sum(x) - n / 2.0) * np.ones(n)
        for i in range(n):
            if x[i] < 0:
                g[i] += 20.0 * x[i]
            elif x[i] > 1:
                g[i] += 20.0 * (x[i] - 1.0)
        return g
    best_x = h.copy()
    best_f = compute_c5(h)
    try:
        res = minimize(obj, h, jac=jac, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'gtol': tol, 'iprint': 0})
        if res.x is not None:
            hx = project_feasible(res.x, n)
            cf = compute_c5(hx)
            if cf < best_f:
                best_x = hx
                best_f = cf
    except Exception:
        pass
    return best_x, best_f

    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    out = np.interp(x_new, xp, src_h)
    return out

    nf = rng.randint(1, 8)

        h_p += scale * 0.4 * np.sin(2 * np.pi * freq * x + phase)
    h_p += scale * 0.2 * rng.randn(n)
    return project_feasible(h_p, n)
