            if npairs > 0:
                g[:npairs] += wk * (1.0 - x[a:a + npairs])
                g[a:] -= wk * x[:npairs]
    if penalty > 0:
        sv = np.sum(x) - n / 2.0
        g += 2.0 * penalty * sv
    return g


def lbfgs_run(h, n, beta, penalty, maxiter=500):
    def obj(x):
        return smooth_max_obj_val(x, n, beta, penalty)

    def grad(x):
        return smooth_max_grad_val(x, n, beta, penalty)

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h.copy(), jac=grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-22,
                                'gtol': 1e-18, 'maxls': 50, 'maxfun': maxiter * 5})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()


def slsqp_run(h, n, beta, maxiter=400):
    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        shifted = beta * (corr - m)
        clipped = np.clip(shifted, -50.0, 50.0)
        ex = np.exp(clipped)
        return (m + np.log(np.sum(ex)) / beta) * (2.0 / n)

    def grad(x):
        return smooth_max_grad_val(x, n, beta, 0.0)

    def con_fun(x):
        return np.sum(x) - n / 2.0

    def con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()


def coord_descent_hill(h, n, steps=50, step_sizes=None):
    """Simple coordinate descent: for each coordinate, try adjusting it to reduce C5."""
    h = h.copy()
    best_c5 = compute_c5(h)
    target = n / 2.0
    if step_sizes is None:
        step_sizes = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]
    for _ in range(steps):
        improved = False
        idx = np.random.permutation(n)
        for i in idx:
            old = h[i]
            for ds in step_sizes:
                for sign in [1, -1]:
                    new_val = old + sign * ds
                    if new_val < 0 or new_val > 1:
                        continue
                    h[i] = new_val
                    # Adjust sum: spread to a random neighbor
                    sv = target - np.sum(h)
                    if abs(sv) < 1e-12:
                        c = compute_c5(h)
                        if c < best_c5 - 1e-12:
                            best_c5 = c
                            improved = True
                            old = h[i]
                            break
                    h[i] = old
                else:
                    continue
                break
        if not improved:
            break
    return project_feasible(h, n), best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def time_left():
        return _BUDGET - (time.time() - _T0)

    # Initialize from parent
    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = np.zeros(1000)
        best_h[:500] = 1.0
    best_n = len(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick L-BFGS-B polish at initial resolution
    for beta in [500, 5000, 50000]:
        if time_left() < 20:
            break
        h_new = lbfgs_run(best_h, best_n, beta, penalty=500, maxiter=300)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    # Phase 2: Multi-resolution L-BFGS-B
    resolutions = [1200, 1600, 2048, 2500, 3000, 4000, 5000]
    for target_n in resolutions:
        if time_left() < 80:
            break
        if target_n <= best_n and target_n < 2048:
            continue

        h_interp = interp_to(best_h, best_n, target_n)
        h_feas = project_feasible(h_interp, target_n)
        c_cur = compute_c5(h_feas)

        # Progressive beta increase
        for beta in [300, 2000, 10000, 50000]:
            if time_left() < 30:
                break
            h_new = lbfgs_run(h_feas, target_n, beta, penalty=500,
                              maxiter=min(500, int(time_left() * 3)))
            c_new = compute_c5(h_new)
            if c_new < c_cur - 1e-12:
                c_cur = c_new
                h_feas = h_new

        # One more pass with very high beta
        if time_left() > 40:
            h_new = lbfgs_run(h_feas, target_n, 200000, penalty=500, maxiter=400)
            c_new = compute_c5(h_new)
            if c_new < c_cur - 1e-12:
                c_cur = c_new
                h_feas = h_new

        if c_cur < best_c5 - 1e-12:
            best_c5 = c_cur
            best_h = h_feas.copy()
            best_n = target_n

    # Phase 3: SLSQP polish at best resolution (constrained, more precise)
    if time_left() > 40 and best_n <= 3000:
        for beta in [30000, 150000]:
            if time_left() < 25:
                break
            h_new = slsqp_run(best_h, best_n, beta, maxiter=400)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new

    # Phase 4: Small perturbations + L-BFGS-B
    for p in range(6):
        if time_left() < 30:
            break
        scale = 0.005 + p * 0.003
        h_pert = best_h + rng.randn(best_n) * scale
        h_pert = project_feasible(h_pert, best_n)
        h_new = lbfgs_run(h_pert, best_n, 20000, penalty=500, maxiter=300)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    # Phase 5: Coordinate descent at final resolution
    if time_left() > 30:
        h_cd, c_cd = coord_descent_hill(best_h, best_n, steps=30)
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd

    # Phase 6: Final SLSQP at higher resolution if we can afford it
    if time_left() > 60 and best_n <= 2500:
        n_up = min(best_n * 2, 5000)
        if n_up % 2 == 1:
            n_up += 1
        h_up = interp_to(best_h, best_n, n_up)
        h_up = project_feasible(h_up, n_up)
        h_up = lbfgs_run(h_up, n_up, 20000, penalty=500, maxiter=400)
        c_up = compute_c5(h_up)
        if c_up < best_c5:
            best_c5 = c_up
            best_h = h_up
            best_n = n_up
        # SLSQP polish
        if time_left() > 20:
            h_up = slsqp_run(h_up, n_up, 100000, maxiter=300)
            c_up = compute_c5(h_up)
            if c_up < best_c5:
                best_c5 = c_up
                best_h = h_up
                best_n = n_up

    # Phase 7: Very final L-BFGS-B with max beta
    if time_left() > 15:
        h_new = lbfgs_run(best_h, best_n, 500000, penalty=1000, maxiter=600)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    # Final feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Try evaluate_erdos_solution for validation
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
