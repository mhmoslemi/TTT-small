import numpy as np

import time

_T0 = 0.0
_BUDGET = 1300.0
_CUR_N = 0
_SM_BETA = 1000.0


def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)


def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break

        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def _sm_objective(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = beta * (corr - m)
    z = np.clip(z, -50.0, 50.0)
    ez = np.exp(z)
    return float((m + np.log(np.sum(ez)) / beta) * dx)


def _sm_gradient(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = beta * (corr - m)
    z = np.clip(z, -50.0, 50.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    return g


def _eq_fun(x):
    return np.sum(x) - _CUR_N / 2.0


def _eq_jac(x):
    return np.ones(_CUR_N)


def slsqp_opt(h, n, beta, maxiter=500):

    _CUR_N = n
    _SM_BETA = beta
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_sm_objective, h0, jac=_sm_gradient, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.fun < _sm_objective(h0) - 1e-14:

    except Exception:

def block_binary_search(h, n, block_sizes=(2, 3, 4), max_sweeps=200):
    """Try all binary patterns for blocks of consecutive positions."""
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        any_improve = False
        n_half = n // 2
        for B in block_sizes:
            for start in range(0, n, B):
                if remaining_time() < 5:
                    break
                end = min(start + B, n)
                B_actual = end - start
                if B_actual < 2:
                    continue
                n_ones = int(np.sum(h))
                orig_block = h[start:end].copy()
                ones_in_block = int(np.sum(h[start:end] > 0.5))
                for mask in range(1, (1 << B_actual)):
                    new_ones = bin(mask).count('1')
                    if new_ones == ones_in_block:
                        continue
                    h[start:end] = ((mask >> np.arange(B_actual)) & 1).astype(float)
                    n_ones_new = n_ones - ones_in_block + new_ones
                    if n_ones_new != n_half:
                        h[start:end] = orig_block
                        continue
                    c5_try = compute_c5(h)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        any_improve = True
                    else:
                        h[start:end] = orig_block
        if not any_improve and sweep > 5:
            break
    return project_feasible(h, n), best_c5


def binary_sa_block(h, n, time_limit, rng, block_size=2):
    """SA with block swaps: move a block of 1s to where 0s are."""
    h = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        sort_idx = np.argsort(-h)
        h = np.zeros(n)
        h[sort_idx[:n_half]] = 1.0
    start = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    current_c5 = best_c5
    T = 0.5
    cool_rate = 0.9997
    no_improve = 0
    max_no_improve = 5000
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    while time.time() - start < time_limit and remaining_time() > 10:
        if len(ones) < block_size or len(zeros) < block_size:
            break
        i_start = ones[rng.randint(len(ones))]
        j_start = zeros[rng.randint(len(zeros))]
        i_idx = np.arange(i_start, i_start + block_size)
        j_idx = np.arange(j_start, j_start + block_size)
        i_idx = i_idx[i_idx < n]
        j_idx = j_idx[j_idx < n]
        if len(i_idx) < block_size or len(j_idx) < block_size:
            i_idx = i_idx[i_idx < n]
            j_idx = j_idx[j_idx < n]
            if len(i_idx) == 0 or len(j_idx) == 0:
                continue
        B = min(len(i_idx), len(j_idx))
        i_idx = i_idx[:B]
        j_idx = j_idx[:B]
        h[i_idx] = 0.0
        h[j_idx] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            current_c5 = new_c5
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i_idx] = 1.0
            h[j_idx] = 0.0
            no_improve += 1
        T *= cool_rate
        if no_improve > max_no_improve:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
    return best_h, best_c5


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def nelder_mead_block(h, n, rng, n_blocks=20, maxiter=80):
    """Nelder-Mead on randomly chosen blocks of consecutive variables."""
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    n_half = n // 2
    for _ in range(n_blocks):
        if remaining_time() < 10:
            break
        B = rng.randint(3, 7)
        start = rng.randint(0, max(1, n - B))
        end = start + B
        block_idx = np.arange(start, end)
        rest_idx = np.setdiff1d(np.arange(n), block_idx)
        x0 = h[block_idx].copy()

        def obj_block(xb):
            h_tmp = h.copy()
            h_tmp[block_idx] = np.clip(xb, 0, 1)
            # Adjust to maintain constraint
            diff = n_half - np.sum(h_tmp)
            if abs(diff) > 1e-6:
                h_tmp[rest_idx] += diff / len(rest_idx)
                h_tmp = np.clip(h_tmp, 0, 1)
            return compute_c5(h_tmp)

        res = fmin(obj_block, x0, maxiter=maxiter, maxfun=2000,
                   disp=False, full_output=True)
        x_opt = res[0]
        h_new = h.copy()
        h_new[block_idx] = np.clip(x_opt, 0, 1)
        diff = n_half - np.sum(h_new)
        if abs(diff) > 1e-6:
            h_new[rest_idx] += diff / len(rest_idx)
            h_new = np.clip(h_new, 0, 1)
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            h = h_new
    return project_feasible(h, n), best_c5


def penalty_rebalance(h, n, top_k=8, max_iters=200):

        if c_max - c_mean < 1e-10:

        # Direction: reduce max, increase mean of top-k
        g_dir = np.zeros(n)

        gn = np.linalg.norm(g_dir)

        d = g_dir / gn

                if cn < best_c5 - 1e-14:

    if remaining_time() > 60:
        h_nm, c5_nm = nelder_mead_block(best_h.copy(), best_n, rng,
                                         n_blocks=30, maxiter=100)
        if c5_nm < best_c5 - 1e-14:
            best_c5 = c5_nm
            best_h = h_nm.copy()
        if remaining_time() > 20:
            h_pol = slsqp_opt(best_h, best_n, 200000, maxiter=400)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5 - 1e-14:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 6: Binary SA with block moves
    if remaining_time() > 120:
        h_bin = (best_h > 0.5).astype(float)
        n_half = best_n // 2
        n_ones = int(np.sum(h_bin))
        if n_ones != n_half:
            sort_idx = np.argsort(-h_bin)
            h_bin = np.zeros(best_n)
            h_bin[sort_idx[:n_half]] = 1.0
        sa_time = min(180, max(30, int(remaining_time() * 0.25)))
        h_b, c5_b = binary_sa_block(h_bin, best_n, sa_time, rng, block_size=2)
        # Continuous refinement of SA result
        h_bc = project_feasible(h_b, best_n)
        for beta in [5000, 50000, 500000]:
            if remaining_time() < 15:
                break
            h_bc = slsqp_opt(h_bc, best_n, beta, maxiter=500)
        c5_bc = compute_c5(h_bc)
        if c5_bc < best_c5 - 1e-14:
            best_c5 = c5_bc
            best_h = h_bc.copy()
        if c5_b < best_c5 - 1e-14:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 7: Multi-resolution
    for n_t in [1280, 960, 320]:
        if remaining_time() < 100:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        c5_int = compute_c5(h_feas)
        h_opt = h_feas.copy()
        for beta in [2000, 20000, 200000]:
            if remaining_time() < 25:
                break
            h_opt = slsqp_opt(h_opt, n_t, beta, maxiter=400)
        h_opt, c5_opt = topk_equalize(h_opt, n_t, top_k=6, max_iters=300)
        if remaining_time() > 20:
            h_opt = slsqp_opt(h_opt, n_t, 1000000, maxiter=500)
            c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5 - 1e-14:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t

    # Phase 8: Second round of block search + top-k
    if remaining_time() > 40:
        h_bb2, c5_bb2 = block_binary_search(best_h.copy(), best_n,
                                            block_sizes=(2, 3, 4), max_sweeps=80)
        if c5_bb2 < best_c5 - 1e-14:
            best_c5 = c5_bb2
            best_h = h_bb2.copy()
        if remaining_time() > 20:
            h_eq2, c5_eq2 = topk_equalize(best_h, best_n, top_k=8, max_iters=500)
            if c5_eq2 < best_c5 - 1e-14:
                best_c5 = c5_eq2
                best_h = h_eq2.copy()

    # Phase 9: High-beta SLSQP final polish
    if remaining_time() > 20:
        h_final = slsqp_opt(best_h, best_n, 5000000, maxiter=800)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-14:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Phase 10: Very fine top-k
    if remaining_time() > 15:
        h_eq3, c5_eq3 = topk_equalize(best_h, best_n, top_k=10, max_iters=300)
        if c5_eq3 < best_c5 - 1e-14:
            best_c5 = c5_eq3
            best_h = h_eq3.copy()

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
