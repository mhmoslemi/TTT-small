corr[k] = np.correlate(h, 1-h)[k]

For a given k (0 to 2n-2), let shift = k - (n-1):
- If shift >= 0: corr[k] = sum_{i=0}^{n-1-shift} h[i+shift] * (1-h[i])
- If shift < 0: corr[k] = sum_{i=-shift}^{n-1} h[i] * (1-h[i+shift])

∂corr[k]/∂h[i]:
- If shift >= 0 and i+shift < n: ∂corr[k]/∂h[i+shift] += (1-h[i]), ∂corr[k]/∂h[i] -= h[i+shift]
  Actually, corr[k] = sum_j h[j+shift]*(1-h[j]) for j=0 to n-1-shift
  = sum_j h[j+shift] - sum_j h[j+shift]*h[j]
  
  ∂corr[k]/∂h[m]:
  - If m = j+shift for some j in [0, n-1-shift]: contributes (1-h[j]) = (1-h[m-shift])
  - If m = j for some j in [0, n-1-shift]: contributes -h[m+shift]
  
  So: ∂corr[k]/∂h[m] = (1-h[m-shift]) * I[shift ≤ m ≤ n-1] - h[m+shift] * I[0 ≤ m ≤ n-1-shift]

This is O(n) per k, and we have O(n) values of k, so the full gradient is O(n²). For n=640, that's 400K operations per gradient evaluation, which is very fast.

Let me also think about a different approach: **direct min-max optimization using the "subgradient" of the max function**.

The subgradient of max_k f_k(x) at x is the convex hull of {∇f_k(x) : f_k(x) = max}.

So the steepest descent direction for min max_k f_k is:
d = -min_{λ ∈ Δ_active} ||∑ λ_k ∇f_k|| 

where Δ_active is the simplex over active constraints.

This is a quadratic program (QP) of size |active|, which is small (typically 5-20).

**Steepest descent for min-max**:
1. Find active set A = {k : f_k(x) > max - ε}
2. Solve: min_{λ ∈ simplex(A)} ||∑_{k∈A} λ_k ∇f_k(x)||²
3. Take step in direction -∑ λ_k ∇f_k(x)
4. Project onto feasible set
5. Line search

def steepest_descent_minmax(h, n, max_iters, eps=0.01, time_limit=60):
    """Steepest descent for min max_k corr[k] with projection."""
    h = project_feasible(h, n).copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(max_iters):
        if time.time() - _T0 > _BUDGET - 10:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        
        # Find active set
        active = np.where(corr > c_max - eps)[0]
        
        # Compute gradients for active set
        grads = np.zeros((len(active), n))
        for i, k in enumerate(active):
            grads[i] = grad_single_shift(h, k, n)
        
        # Solve QP: min ||G^T λ||² s.t. sum(λ) = 1, λ ≥ 0
        # This is equivalent to: min λ^T (GG^T) λ s.t. 1^T λ = 1, λ ≥ 0
        GGt = grads @ grads.T  # (|A| × |A|)
        
        if len(active) == 1:
            lam = np.array([1.0])
        elif len(active) == 2:
            # Analytical solution
            a = GGt[0,0]
            b = GGt[0,1]
            c = GGt[1,1]
            f0 = a  # λ=(1,0)
            f1 = c  # λ=(0,1)
            f01 = 0.25*(a + 2*b + c)  # λ=(0.5,0.5)
            # Minimize on [0,1]: f(λ1) = a(1-λ1)² + 2b(1-λ1)λ1 + cλ1²
            # = a - 2aλ1 + aλ1² + 2bλ1 - 2bλ1² + cλ1²
            # = a + (2b-2a)λ1 + (a-2b+c)λ1²
            # d/dλ1 = (2b-2a) + 2(a-2b+c)λ1 = 0
            # λ1* = (a-b) / (a-2b+c)
            denom = a - 2*b + c
            if abs(denom) > 1e-12:
                lam1 = (a - b) / denom
                lam1 = np.clip(lam1, 0, 1)
            else:
                lam1 = 0.5 if f01 < f0 else (0.0 if f1 < f0 else 0.5)
            lam = np.array([1-lam1, lam1])
        else:
            # Use scipy for the QP
            from scipy.optimize import minimize as min2
            m = len(active)
            def qp_obj(l):
                return l @ GGt @ l
            def qp_grad(l):
                return 2 * GGt @ l
            l0 = np.ones(m) / m
            res = min2(qp_obj, l0, jac=qp_grad, method='SLSQP',
                       bounds=[(0,1)]*m,
                       constraints=[{'type':'eq', 'fun': lambda l: np.sum(l)-1}])
            lam = np.clip(res.x, 0, 1)
            lam /= np.sum(lam)
        
        # Direction
        d = grads.T @ lam  # gradient of the weighted average
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-14:
            break
        d = d / d_norm
        
        # Line search (backtracking)
        improved = False
        for step in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
            h_new = project_feasible(h - step * d, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                h = h_new
                best_c5 = c5_new
                best_h = h_new.copy()
                improved = True
                break
        
        if not improved:
            break
    
    return best_h, best_c5

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    return (m + np.log(np.sum(ez)) / beta) * dx

def _sm_grad(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)

    S = np.sum(ez)
