import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)


def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def coord_pass(h, n, rng):
    """One full pass of coordinate descent. Returns updated h and corr."""
    corr = compute_corr_full(h)
    order = rng.permutation(n)
    d_arr = np.arange(1, n)

    for idx in order:
        h_i = h[idx]
        ipd = idx + d_arr
        jpd = idx - d_arr
        ipd_valid = (ipd < n).astype(float)
        jpd_valid = (jpd >= 0).astype(float)
        ipd_clip = np.clip(ipd, 0, n - 1)
        jpd_clip = np.clip(jpd, 0, n - 1)
        h_ipd = h[ipd_clip]
        h_jpd = h[jpd_clip]

        s_pos = ipd_valid * (1.0 - h_ipd) - jpd_valid * h_jpd
        s_neg = jpd_valid * (1.0 - h_jpd) - ipd_valid * h_ipd

        slopes = np.zeros(2 * n - 1)
        slopes[0:n - 1] = s_pos[::-1]
        slopes[n:2 * n - 1] = s_neg

        # Golden section for h[i] in [0, 1]
        gr = 0.6180339887498949
        a, b = 0.0, 1.0
        c = b - gr * (b - a)
        dd = a + gr * (b - a)

        # Precompute for fast evaluation
        base = corr.copy()
        s0_lin = 1.0 - 2.0 * h_i

        vals_c = base + (c - h_i) * slopes
        vals_c[n - 1] = base[n - 1] + (c - h_i) * s0_lin - (c - h_i) ** 2
        fc = np.max(vals_c)
        vals_d = base + (dd - h_i) * slopes
        vals_d[n - 1] = base[n - 1] + (dd - h_i) * s0_lin - (dd - h_i) ** 2
        fd = np.max(vals_d)

        for _ in range(30):
            if fc < fd:
                b = dd
                dd = c
                fd = fc
                c = b - gr * (b - a)
                vals_c = base + (c - h_i) * slopes
                vals_c[n - 1] = base[n - 1] + (c - h_i) * s0_lin - (c - h_i) ** 2
                fc = np.max(vals_c)
            else:
                a = c
                c = dd
                fc = fd
                dd = a + gr * (b - a)
                vals_d = base + (dd - h_i) * slopes
                vals_d[n - 1] = base[n - 1] + (dd - h_i) * s0_lin - (dd - h_i) ** 2
                fd = np.max(vals_d)

        x_opt = (a + b) / 2.0
        if abs(x_opt - h_i) > 1e-12:
            h[idx] = x_opt
            delta = x_opt - h_i
            corr = base + delta * slopes
            corr[n - 1] = base[n - 1] + delta * s0_lin - delta * delta

    return h, corr


def smooth_max_obj(v):
    h = v[:-1]
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    beta = v[-1]
    K = min(80, len(corr))
    top = np.argsort(-corr)[:K]
    c_top = corr[top]
    sh = c_top - c_top.max()
    cl = np.clip(beta * sh, -60.0, 60.0)
    ex = np.exp(cl)
    val = (c_top.max() + np.log(ex.sum()) / beta) * (2.0 / n)
    return val


def smooth_max_grad(v):
    h = v[:-1]
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    beta = v[-1]
    K = min(80, len(corr))
    top = np.argsort(-corr)[:K]
    c_top = corr[top]
    sh = c_top - c_top.max()
    cl = np.clip(beta * sh, -60.0, 60.0)
    ex = np.exp(cl)
    w = ex / ex.sum()
    g = np.zeros(n)
    for i in range(K):
        if w[i] < 1e-12:
            continue
        k = top[i]
        offset = k - (n - 1)
        wk = w[i] * (2.0 / n)
        if offset >= 0:
            np_ = n - offset
            g[offset:] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[offset:]
        else:
            a2 = -offset
            np_ = n - a2
            g[:np_] += wk * (1.0 - h[a2:])
            g[a2:] -= wk * h[:np_]
    g_beta = np.log(ex.sum()) / beta ** 2
    return np.concatenate([g, [g_beta]])


def sum_eq_fun(v):
    return np.sum(v[:-1]) - len(v[:-1]) / 2.0


def sum_eq_jac(v):
    n = len(v) - 1
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g


def slsqop_smooth(h, n, beta, maxiter):
    global _BUDGET
    v0 = np.concatenate([h, [float(beta)]])
    bounds = [(0.0, 1.0)] * n + [(100.0, 50000.0)]
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}]
    try:
        res = minimize(smooth_max_obj, v0, jac=smooth_max_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x[:n], n)
            return h_new
    except Exception:
        pass
    return h.copy()


def subgrad_step(h, n, K, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    sh = c_top - c_top.max()
    cl = np.clip(beta * sh, -60.0, 60.0)
    ex = np.exp(cl)
    w = ex / ex.sum()
    g = np.zeros(n)
    for i in range(K):
        if w[i] < 1e-14:
            continue
        k = idx[i]
        offset = k - (n - 1)
        wk = w[i] * (2.0 / n)
        if offset >= 0:
            np_ = n - offset
            g[offset:] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[offset:]
        else:
            a2 = -offset
            np_ = n - a2
            g[:np_] += wk * (1.0 - h[a2:])
            g[a2:] -= wk * h[:np_]
    return g

    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)
