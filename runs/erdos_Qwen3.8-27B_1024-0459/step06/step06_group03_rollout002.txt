        g_avg /= K
        g_dir = -g_avg
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        g_dir /= gn
        improved = False
        for step in [0.01, 0.005, 0.02, 0.002, 0.008, 0.001, 0.015, 0.004]:
            ht = project_feasible(h + step * g_dir, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-15:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            # Try negative direction too
            for step in [0.01, 0.005, 0.02, 0.002, 0.008]:
                ht = project_feasible(h - step * g_dir, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-15:
                    h = ht
                    best_c5 = cn
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def tmin_ineq_fun(v):
    h = v[:_global_n]
    t = v[-1]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return _global_tK_val * (_global_n / 2.0) - corr[_global_tK_idx]


def tmin_ineq_jac(v):
    h = v[:_global_n]
    n = _global_n
    K_idx = _global_tK_idx
    J = np.zeros((len(K_idx), n + 1))
    for i, k in enumerate(K_idx):
        offset = k - (n - 1)
        if offset >= 0:
            np_ = n - offset
            J[i, offset:] += (1.0 - h[:np_])
            J[i, :np_] -= h[offset:]
        else:
            a = -offset
            np_ = n - a
            J[i, :np_] += (1.0 - h[a:])
            J[i, a:] -= h[:np_]
        J[i, -1] = -n / 2.0
    return -J


_global_tK_idx = None
_global_tK_val = 1.0


def _obj_t_fun(v):
    return v[-1]


def _obj_t_jac(v):
    g = np.zeros(_global_n + 1)
    g[-1] = 1.0
    return g


def _sum_fun2(v):
    return np.sum(v[:_global_n]) - _global_n / 2.0


def _sum_jac2(v):
    g = np.zeros(_global_n + 1)
    g[:_global_n] = 1.0
    return g


def slsqp_tminmax(h, n, K_list, maxiter):
    global _global_n, _global_tK_idx
    _global_n = n
    _global_tK_idx = np.array(K_list)
    t0 = compute_c5(h)
    v0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'ineq', 'fun': tmin_ineq_fun, 'jac': tmin_ineq_jac},
        {'type': 'eq', 'fun': _sum_fun2, 'jac': _sum_jac2}
    ]
    try:
        res = minimize(_obj_t_fun, v0, jac=_obj_t_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x[:n], n)
            c5_new = compute_c5(h_new)
            if c5_new <= compute_c5(h) + 1e-14:
                return h_new, c5_new
    except Exception:
        pass
    return h.copy(), compute_c5(h)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _global_n, _global_beta, _global_K
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 512
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    h = best_h.copy()
    n = best_n

    # Phase 1: L-BFGS-B with beta annealing (fast, many iterations)
    beta_sched = [10, 30, 100, 300, 1000, 3000, 10000, 30000]
    for beta in beta_sched:
        if remaining() < 15:
            break
        K_use = 60 if beta < 500 else 50
        h, c5_new = lbfgs_optimize(h, n, beta, K_use, maxiter=800)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()

    # Phase 2: SLSQP for differentiability (use where L-BFGS-B might struggle)
    for beta in [500, 2000, 8000, 25000]:
        if remaining() < 12:
            break
        h, c5_new = slsqp_optimize(h, n, beta, K=50, maxiter=400)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()

    # Phase 3: Active-set min-max SLSQP
    if remaining() > 30:
        for K_a in [8, 12, 16, 20, 25, 30, 40]:
            if remaining() < 10:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            idx_a = np.argsort(-corr)[:K_a]
            h, c5_new = slsqp_tminmax(h, n, idx_a, maxiter=300)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                h = h.copy()

    # Phase 4: Equalize top shifts
    if remaining() > 20:
        h, c5_new = equalize_active(h, n, K=10, max_iters=300)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()

    # Phase 5: Multi-start with small perturbations
    if remaining() > 100:
        for attempt in range(6):
            if remaining() < 30:
                break
            noise_scale = 0.01 / (attempt + 1)
            h_pert = best_h + rng.randn(best_n) * noise_scale
            h_pert = project_feasible(h_pert, best_n)
            # Quick L-BFGS-B re-optimization
            for beta in [100, 1000, 10000]:
                if remaining() < 15:
                    break
                h_pert, c5_p = lbfgs_optimize(h_pert, best_n, beta, K=50, maxiter=400)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_pert.copy()
            # Try equalize too
            if remaining() > 20:
                h_pert, c5_p2 = equalize_active(h_pert, best_n, K=8, max_iters=150)
                if c5_p2 < best_c5:
                    best_c5 = c5_p2
                    best_h = h_pert.copy()

    # Phase 6: Try higher resolution n=1024
    n_high = 1024
    if remaining() > 150:
        h_interp = interp_to(best_h, best_n, n_high)
        h_interp = project_feasible(h_interp, n_high)
        c5_interp = compute_c5(h_interp)
        if c5_interp <= best_c5 + 0.002:
            for beta in [50, 200, 1000, 5000, 20000]:
                if remaining() < 20:
                    break
                h_interp, c5_hi = lbfgs_optimize(h_interp, n_high, beta, K=60, maxiter=600)
            if remaining() > 30:
                for beta in [3000, 15000]:
                    if remaining() < 15:
                        break
                    h_interp, c5_hi = slsqp_optimize(h_interp, n_high, beta, K=50, maxiter=300)
            if remaining() > 25:
                h_interp, c5_hi = equalize_active(h_interp, n_high, K=10, max_iters=200)
            if remaining() > 20:
                corr_h = np.correlate(h_interp, 1.0 - h_interp, mode='full')
                for K_a in [10, 20, 30]:
                    if remaining() < 12:
                        break
                    idx_a = np.argsort(-np.correlate(h_interp, 1.0 - h_interp, mode='full'))[:K_a]
                    h_interp, c5_hi = slsqp_tminmax(h_interp, n_high, idx_a, maxiter=250)
            if c5_hi < best_c5:
                best_c5 = c5_hi
                best_h = h_interp.copy()
                best_n = n_high

    # Phase 7: Intensive refinement at best resolution
    n = best_n
    h = best_h.copy()
    for cycle in range(3):
        if remaining() < 25:
            break
        # L-BFGS-B with high beta
        for beta in [5000, 20000, 50000]:
            if remaining() < 10:
                break
            h, c5_c = lbfgs_optimize(h, n, beta, K=60, maxiter=600)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h.copy()
        # SLSQP polish
        if remaining() > 15:
            for beta in [10000, 40000]:
                if remaining() < 10:
                    break
                h, c5_c = slsqp_optimize(h, n, beta, K=50, maxiter=300)
                if c5_c < best_c5:
                    best_c5 = c5_c
                    best_h = h.copy()
        # Equalize
        if remaining() > 15:
            h, c5_c = equalize_active(h, n, K=12, max_iters=300)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h.copy()
        # Active set
        if remaining() > 20:
            corr_c = np.correlate(h, 1.0 - h, mode='full')
            for K_a in [15, 25, 35, 50]:
                if remaining() < 10:
                    break
                idx_a = np.argsort(-np.correlate(h, 1.0 - h, mode='full'))[:K_a]
                h, c5_c = slsqp_tminmax(h, n, idx_a, maxiter=250)
                if c5_c < best_c5:
                    best_c5 = c5_c
                    best_h = h.copy()

    # Phase 8: Very fine final tuning - tiny steps
    if remaining() > 10:
        h_f = best_h.copy()
        c5_f = best_c5
        for it in range(200):
            if remaining() < 3:
                break
            corr_f = np.correlate(h_f, 1.0 - h_f, mode='full')
            idx_f = np.argsort(-corr_f)[:20]
            g_f = np.zeros(best_n)
            for k in idx_f:
                g_f += grad_corr_k(h_f, k, best_n)
            g_f /= len(idx_f)
            gn_f = np.linalg.norm(g_f)
            if gn_f < 1e-14:
                break
            improved = False
            for step in [0.003, 0.001, 0.005, 0.0005, 0.002, 0.0015]:
                h_try = project_feasible(h_f - step * g_f / gn_f, best_n)
                c_try = compute_c5(h_try)
                if c_try < c5_f - 1e-15:
                    c5_f = c_try
                    h_f = h_try.copy()
                    improved = True
                    break
            if not improved:
                for step in [0.003, 0.001, 0.005]:
                    h_try = project_feasible(h_f + step * g_f / gn_f, best_n)
                    c_try = compute_c5(h_try)
                    if c_try < c5_f - 1e-15:
                        c5_f = c_try
                        h_f = h_try.copy()
                        improved = True
                        break
            if not improved and it > 50:
                break
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 9: One final L-BFGS-B with very high beta
    if remaining() > 15:
        h_fin = best_h.copy()
        for beta in [80000, 150000]:
            if remaining() < 8:
                break
            h_fin, c5_fin = lbfgs_optimize(h_fin, best_n, beta, K=80, maxiter=500)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()

    # Final feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
