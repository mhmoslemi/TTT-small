        if c3 < best_c5:
            best_c5 = c3
            best_h = h3.copy()

    # Phase 4: Subgradient again with different params
    if rem() > 30:
        t4 = min(80.0, rem() * 0.1)
        h4, c4 = subgrad_momentum(best_h, best_n, t4, K=3, mu=0.85, lr_init=0.04)
        if c4 < best_c5:
            best_c5 = c4
            best_h = h4.copy()

    # Phase 5: Try n=256 (coarser)
    if rem() > 120:
        n_coarse = 256
        h_coarse = project_feasible(interp_to(best_h, best_n, n_coarse), n_coarse)
        t5 = min(120.0, rem() * 0.15)
        h5, c5 = active_set_loop(h_coarse, n_coarse, t5)
        if c5 < best_c5 - 1e-5:
            if rem() > 30:
                h5s, c5s = subgrad_momentum(h5, n_coarse, min(40.0, rem() * 0.08), K=4)
                if c5s < c5:
                    h5, c5 = h5s, c5s
            if c5 < best_c5:
                best_c5 = c5
                best_h = h5.copy()
                best_n = n_coarse

    # Phase 6: Try n=1024 (finer)
    if rem() > 150:
        n_fine = best_n * 2
        if n_fine <= 2048 and n_fine % 2 == 0:
            h_fine_init = project_feasible(interp_to(best_h, best_n, n_fine), n_fine)
            t6 = min(150.0, rem() * 0.2)
            h6, c6 = active_set_loop(h_fine_init, n_fine, t6)
            if c6 < best_c5 - 1e-5:
                if rem() > 20:
                    h6s, c6s = subgrad_momentum(h6, n_fine, min(40.0, rem() * 0.1), K=5)
                    if c6s < c6:
                        h6, c6 = h6s, c6s
                if c6 < best_c5:
                    best_c5 = c6
                    best_h = h6.copy()
                    best_n = n_fine

    # Phase 7: Pair swap local search
    if rem() > 30:
        t7 = min(80.0, rem() * 0.1)
        h7, c7 = pair_swap_search(best_h, best_n, t7)
        if c7 < best_c5:
            best_c5 = c7
            best_h = h7.copy()

    # Phase 8: Final active set polish (very focused)
    if rem() > 30:
        t8 = min(120.0, rem() * 0.2)
        h8, c8 = active_set_loop(best_h, best_n, t8)
        if c8 < best_c5:
            best_c5 = c8
            best_h = h8.copy()

    # Phase 9: High-beta SLSQP on top-5 active
    if rem() > 20:
        corr = get_corr_scaled(best_h)
        c5_val = corr.max()
        top5 = np.argsort(-corr)[:5]
        for beta in [200000, 1000000]:
            if rem() < 8:
                break
            h9 = slsqp_run(best_h, beta, top5, maxiter=150)
            c9 = compute_c5(h9)
            if c9 < best_c5:
                best_c5 = c9
                best_h = h9.copy()

    # Phase 10: Final tiny subgradient steps
    if rem() > 15:
        h10, c10 = subgrad_momentum(best_h, best_n, min(30.0, rem() * 0.5), K=3, mu=0.7, lr_init=0.01)
        if c10 < best_c5:
            best_c5 = c10
            best_h = h10.copy()

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
