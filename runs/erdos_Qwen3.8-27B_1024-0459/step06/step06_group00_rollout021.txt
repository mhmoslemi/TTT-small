This is linear in h! So we can solve an LP at each iteration.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:

        h = np.clip(h, 0.0, 1.0)

def c5_gradient_d(h, d, n):
    """Gradient of C(d) = (2/n)*sum_{i=0}^{n-1-d} h[i+d]*(1-h[i]) wrt h."""
    dx = 2.0 / n
    g = np.zeros(n)
    # h[j] appears as h[i+d] when i = j-d (if 0 <= j-d <= n-1-d, i.e., d <= j <= n-1)
    # and as h[i] when i = j (if 0 <= j <= n-1-d)
    for j in range(d, n):
        g[j] += dx * (1.0 - h[j - d])
    for j in range(0, n - d):
        g[j] -= dx * h[j + d]
    return g

    return float(np.sum(h[d:] * (1 - h[:n-d])) * 2.0 / n)

def c5_grad_d(h, d, n):
    """Gradient of C(d) wrt h, where C(d) = (2/n)*sum_{i=0}^{n-1-d} h[i+d]*(1-h[i])."""
    dx = 2.0 / n
    g = np.zeros(n)
    if d == 0:
        # C(0) = (2/n)*sum h[i]*(1-h[i]), grad = (2/n)*(1-2h)
        g[:] = dx * (1.0 - 2.0 * h)
    else:
        # h[j] appears with:
        # +dx*(1-h[j-d]) for j=d,...,n-1
        # -dx*h[j+d] for j=0,...,n-1-d
        g[d:] += dx * (1.0 - h[:n-d])
        g[:n-d] -= dx * h[d:]
    return g

def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_corr_full(h):

def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)

def c5_grad_d(h, d, n):
    dx = 2.0 / n
    g = np.zeros(n)
    if d == 0:
        g[:] = dx * (1.0 - 2.0 * h)
    else:
        g[d:] += dx * (1.0 - h[:n - d])
        g[:n - d] -= dx * h[d:]
    return g

def c5_val_d(h, d, n):
    dx = 2.0 / n
    if d == 0:
        return float(np.sum(h * (1.0 - h)) * dx)
    return float(np.sum(h[d:] * (1.0 - h[:n - d])) * dx)

def lp_single(h, n, active_d, dx):
    """Solve one LP subproblem. Returns (h_new, t_new) or (None, None)."""
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    rows = []
    rhs = []
    for d in active_d:
        d_int = int(d)
        g = c5_grad_d(h, d_int, n)
        cv = c5_val_d(h, d_int, n)
        row = np.concatenate([g, [-1.0]])
        r = float(g @ h) - cv
        rows.append(row)
        rhs.append(r)
    
    A_ub = np.array(rows)
    b_ub = np.array(rhs)

def _sm_obj(x):

    dx = 2.0 / n
    corr = compute_corr_full(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    return (m + np.log(np.sum(ez)) / beta) * dx

def _sm_grad(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr_full(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    return g
