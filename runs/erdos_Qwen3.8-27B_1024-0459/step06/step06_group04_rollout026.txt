    best_val = compute_c5(h)
    start = time.time()
    t_nest = 1.0
    for it in range(50000):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 10:
            break
        val_y, g_y = smooth_max_val_grad(y, beta)
        gn = np.linalg.norm(g_y)
        if gn < 1e-15:
            break
        alpha = min(0.05, 0.5 / (gn + 1e-10))
        h_new = project_feasible(y - alpha * g_y, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        mu = (t_new - 1.0) / t_new
        y_new = project_feasible(h_new + momentum * mu * (h_new - prev_y), n)
        prev_y = y.copy()
        y = y_new
        t_nest = t_new
        val_h = smooth_max_val_grad(h_new, beta)[0]
        if val_h < best_val:
            best_val = val_h
            best_h = h_new.copy()
        elif it % 500 == 0:
            exact = compute_c5(h_new)
            if exact < best_val:
                best_val = exact
                best_h = h_new.copy()
    return best_h, best_val


def slsqp_smooth(h, beta, maxiter=200):
    n = len(h)
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return smooth_max_val_grad(x, beta)[0]

    def jac(x):
        return smooth_max_val_grad(x, beta)[1]

    def eq_fun(x):
        return x.sum() - n / 2.0

    def eq_jac(x):
        return np.ones(n)

    try:
        res = minimize(obj, h, jac=jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def topk_subgradient(h, time_limit, K=25):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    for it in range(20000):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 10:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = corr.max() * (2.0 / n)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        idx = np.argsort(-corr)[:K]
        g = np.zeros(n)
        for k_idx in idx:
            off = k_idx - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:] += (1.0 - h[:np_])
                g[:np_] -= h[off:]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += (1.0 - h[ao:])
                g[ao:] -= h[:np_]
        g /= K
        g *= (2.0 / n)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        alpha = 1.0 / (np.sqrt(it + 1.0) * gn + 1e-10)
        h = project_feasible(h - alpha * g, n)
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def rem():
        return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)

    try:
        best_h = np.array(initial_h_values, dtype=float).copy()
        best_n = len(best_h)
        best_c5 = compute_c5(best_h)
    except Exception:
        best_n = 512
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP continuation at parent resolution
    beta_schedule = [100, 500, 2000, 10000, 50000, 200000]
    for beta in beta_schedule:
        if rem() < 40:
            break
        maxiter = min(200, 50 + int(beta / 500))
        h_new = slsqp_smooth(best_h, beta, maxiter=maxiter)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Phase 2: Adam refinement with high beta
    if rem() > 60:
        t_alloc = min(180.0, rem() * 0.2)
        h_new, _ = adam_smooth(best_h, 100000, t_alloc, lr=0.003)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Phase 3: Nesterov with very high beta
    if rem() > 60:
        t_alloc = min(120.0, rem() * 0.15)
        h_new, _ = nesterov_smooth(best_h, 500000, t_alloc)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Phase 4: Diversity - perturb and re-optimize
    for pert_level in [0.02, 0.05, 0.01]:
        if rem() < 50:
            break
        h_pert = project_feasible(best_h + rng.randn(best_n) * pert_level, best_n)
        # Quick SLSQP polish from perturbed start
        h_new = slsqp_smooth(h_pert, 50000, maxiter=100)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Phase 5: Try higher resolution
    for mult in [1.5, 2.0]:
        if rem() < 100:
            break
        rn = int(round(best_n * mult / 64) * 64)
        if rn <= best_n or rn > 2048:
            continue
        h_interp = interp_to(best_h, best_n, rn)
        h_feas = project_feasible(h_interp, rn)
        if rem() < 40:
            break
        # Continuation at higher resolution
        for beta in [5000, 50000, 200000]:
            if rem() < 30:
                break
            maxiter = min(150, 50 + int(beta / 1000))
            h_new = slsqp_smooth(h_feas, beta, maxiter=maxiter)
            c_new = compute_c5(h_new)
            if c_new < compute_c5(h_feas):
                h_feas = h_new.copy()
        c_hr = compute_c5(h_feas)
        if c_hr < best_c5:
            best_c5 = c_hr
            best_h = h_feas.copy()
            best_n = rn

    # Phase 6: Final SLSQP polish
    for beta in [500000, 1000000, 2000000]:
        if rem() < 20:
            break
        h_new = slsqp_smooth(best_h, beta, maxiter=150)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Phase 7: Subgradient refinement
    if rem() > 30:
        h_sg, c_sg = topk_subgradient(best_h, min(80.0, rem() * 0.3), K=30)
        if c_sg < best_c5:
            best_c5 = c_sg
            best_h = h_sg.copy()
            # Polish subgradient result
            if rem() > 20:
                h_new = slsqp_smooth(best_h, 500000, maxiter=100)
                c_new = compute_c5(h_new)
                if c_new < best_c5:
                    best_c5 = c_new
                    best_h = h_new.copy()

    # Phase 8: Final Adam with very small learning rate
    if rem() > 15:
        h_new, _ = adam_smooth(best_h, 1000000, min(20.0, rem() * 0.3), lr=0.001)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
