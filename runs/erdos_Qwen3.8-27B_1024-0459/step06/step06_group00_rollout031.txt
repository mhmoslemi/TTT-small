def compute_c5(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)

def grad_single_shift(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        np_ = n - off
        g[off:] += (1.0 - h[:np_])
        g[:np_] -= h[off:]
    else:
        a = -off
        np_ = n - a
        g[:np_] += (1.0 - h[a:])
        g[a:] -= h[:np_]
    return g

    dx = 2.0 / n

    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -60.0, 60.0)
    ez = np.exp(z)
    return (m + np.log(np.sum(ez)) / beta) * dx


def _sm_grad(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA

    S = np.sum(ez)

        if wk < 1e-15:

        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]

def binary_sa(h, n, time_limit=120, block_size=1, rng=None):

def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def theta_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)


def periodic_pattern(n, period, ones_in_period):

    dists = np.minimum(h, 1.0 - h)
    sorted_idx = np.argsort(-dists)
    free_idx = sorted_idx[:n_free]
    frozen_idx = sorted_idx[n_free:]

    h_work = h.copy()
    h_work[frozen_idx] = np.where(h_work[frozen_idx] > 0.5, 1.0, 0.0)

    frozen_sum = np.sum(h_work[frozen_idx])
    target_free = n / 2.0 - frozen_sum

    if target_free < -0.01 or target_free > len(free_idx) + 0.01:
        return h, compute_c5(h)

    target_free = np.clip(target_free, 0.0, float(len(free_idx)))

    # Optimize free variables with SLSQP
    # Wrap with global variables - need a different approach
    # Just do a simple scipy optimization on the free variables
    from scipy.optimize import minimize as sp_min

    def obj_sub(xf):
        h_temp = h_work.copy()
        h_temp[free_idx] = np.clip(xf, 0.0, 1.0)
        h_temp = project_feasible(h_temp, n)
        return compute_c5(h_temp)

    x0 = h_work[free_idx].copy()
    # Simple Nelder-Mead on the free variables
    try:
        res = sp_min(obj_sub, x0, method='Nelder-Mead',
                     options={'maxiter': maxiter, 'xatol': 1e-8, 'fatol': 1e-12})
        h_result = h_work.copy()
        h_result[free_idx] = np.clip(res.x, 0.0, 1.0)
        h_result = project_feasible(h_result, n)
        c5_result = compute_c5(h_result)
        if c5_result < compute_c5(h):
            return h_result, c5_result
    except Exception:
        pass

    return h, compute_c5(h)


def coordinate_descent(h, n, max_sweeps=50, time_limit=60):
    """Coordinate descent with binary/continuous choices."""
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    start = time.time()

    for sweep in range(max_sweps):
        if time.time() - start > time_limit or remaining_time() < 5:
            break
        improved = False
        for i in range(n):
            old = h[i]
            if 0.1 < old < 0.9:
                cands = [0.0, 1.0, old + 0.1, old - 0.1, old + 0.3, old - 0.3,
                         old + 0.05, old - 0.05]
            elif old <= 0.1:
                cands = [0.0, 1.0, 0.3, 0.5]
            else:
                cands = [1.0, 0.0, 0.7, 0.5]

            best_val = old
            best_c = best_c5
            for c in cands:
                if not (0.0 <= c <= 1.0) or abs(c - old) < 1e-8:
                    continue
                h[i] = c
                h_p = project_feasible(h, n)
                c5_v = compute_c5(h_p)
                if c5_v < best_c - 1e-13:
                    best_c = c5_v
                    best_val = c
                    improved = True
            h[i] = best_val
            h = project_feasible(h, n)
            if best_c < best_c5:
                best_c5 = best_c

        if not improved:
            break

    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _CUR_N
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    # Load parent
    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = theta_morse(640)
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with progressive beta on parent resolution
    h = best_h.copy()
    for beta in [1000, 5000, 50000, 500000, 5000000, 50000000]:
        if remaining_time() < 20:
            break
        h_new = slsqp_opt(h, best_n, beta, maxiter=2000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Multiplicative weights / saddle point
    if remaining_time() > 30:
        h_mw, c5_mw = multiplicative_weights_opt(h, best_n, time_limit=100, eta=3.0)
        if c5_mw < best_c5 - 1e-15:
            best_c5 = c5_mw
            best_h = h_mw.copy()
            h = h_mw.copy()

    # Phase 3: Top-k equalization
    if remaining_time() > 20:
        for tk in [2, 3, 5, 10]:
            if remaining_time() < 15:
                break
            h_eq, c5_eq = top_k_equalize(h, best_n, top_k=tk, max_iters=1000, time_limit=50)
            if c5_eq < best_c5 - 1e-15:
                best_c5 = c5_eq
                best_h = h_eq.copy()
                h = h_eq.copy()

    # Phase 4: Subgradient zigzag
    if remaining_time() > 30:
        h_sg, c5_sg = subgradient_zigzag(h, best_n, max_iters=10000, time_limit=100)
        if c5_sg < best_c5 - 1e-15:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 5: Higher resolution optimization
    for n_t in [best_n * 2, best_n * 3, best_n + 256]:
        if remaining_time() < 60:
            break
        if n_t > 4096:
            continue
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        c5_int = compute_c5(h_feas)

        h_opt = h_feas
        for beta in [10000, 100000, 1000000, 10000000]:
            if remaining_time() < 20:
                break
            h_opt = slsqp_opt(h_opt, n_t, beta, maxiter=1000)
        c5_opt = compute_c5(h_opt)

        if remaining_time() > 20:
            h_opt, c5_opt2 = top_k_equalize(h_opt, n_t, top_k=5, max_iters=500, time_limit=40)
            if c5_opt2 < c5_opt:
                c5_opt = c5_opt2
                h_opt = h_opt.copy()

        if remaining_time() > 20:
            h_opt, c5_opt3 = subgradient_zigzag(h_opt, n_t, max_iters=3000, time_limit=40)
            if c5_opt3 < c5_opt:
                c5_opt = c5_opt3
                h_opt = h_opt.copy()

        if c5_opt < best_c5 - 1e-15:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t
            h = h_opt.copy()

    # Phase 6: More SLSQP at best resolution
    for beta in [1000000, 10000000, 100000000]:
        if remaining_time() < 15:
            break
        h_new = slsqp_opt(h, best_n, beta, maxiter=3000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 7: Binary SA with block moves
    if remaining_time() > 60:
        h_bin = (best_h > 0.5).astype(float)
        sa_time = min(100, int(remaining_time() * 0.25))
        h_b, c5_b = binary_sa(h_bin, best_n, time_limit=sa_time, block_size=1, rng=rng)
        # Refine continuously from binary
        h_b_c = h_b.copy()
        for beta in [50000, 500000, 5000000]:
            if remaining_time() < 15:
                break
            h_b_c = slsqp_opt(h_b_c, best_n, beta, maxiter=1000)
        c5_b_c = compute_c5(h_b_c)
        if remaining_time() > 15:
            h_b_c, c5_b_c2 = top_k_equalize(h_b_c, best_n, top_k=5, max_iters=300, time_limit=30)
            if c5_b_c2 < c5_b_c:
                c5_b_c = c5_b_c2
                h_b_c = h_b_c.copy()

        if c5_b_c < best_c5 - 1e-15:
            best_c5 = c5_b_c
            best_h = h_b_c.copy()
            h = h_b_c.copy()
        if c5_b < best_c5 - 1e-15:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 8: Multiplicative weights again (after all the above)
    if remaining_time() > 40:
        h_mw2, c5_mw2 = multiplicative_weights_opt(h, best_n, time_limit=80, eta=5.0)
        if c5_mw2 < best_c5 - 1e-15:
            best_c5 = c5_mw2
            best_h = h_mw2.copy()
            h = h_mw2.copy()

    # Phase 9: Alternative starting points
    if remaining_time() > 50:
        alt_starts = [theta_morse(best_n)]
        h_rand = np.zeros(best_n)
        h_rand[rng.choice(best_n, best_n // 2, replace=False)] = 1.0
        alt_starts.append(h_rand)
        # Periodic patterns
        for period in [4, 6, 8, 12, 16]:
            if period < best_n:
                ones_count = period // 2
                ones_in_p = list(range(ones_count))
                alt_starts.append(periodic_pattern(best_n, period, ones_in_p))
                if ones_count > 1:
                    alt_starts.append(periodic_pattern(best_n, period, list(range(ones_count, period))))

        for h_alt in alt_starts:
            if remaining_time() < 30:
                break
            h_opt = h_alt
            for beta in [5000, 100000, 1000000]:
                if remaining_time() < 15:
                    break
                h_opt = slsqp_opt(h_opt, best_n, beta, maxiter=800)
            c5_opt = compute_c5(h_opt)
            if remaining_time() > 15:
                h_opt, c5_opt2 = top_k_equalize(h_opt, best_n, top_k=5, max_iters=400, time_limit=25)
                if c5_opt2 < c5_opt:
                    c5_opt = c5_opt2
                    h_opt = h_opt.copy()
            if c5_opt < best_c5 - 1e-15:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                h = h_opt.copy()

    # Phase 10: Coordinate descent
    if remaining_time() > 20:
        h_cd, c5_cd = coordinate_descent(best_h, best_n, max_sweeps=30, time_limit=min(60, int(remaining_time() * 0.3)))
        if c5_cd < best_c5 - 1e-15:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h = h_cd.copy()

    # Phase 11: Final intensive refinement loop
    for iteration in range(5):
        if remaining_time() < 20:
            break

        # SLSQP
        if remaining_time() > 15:
            h_new = slsqp_opt(h, best_n, 100000000, maxiter=2000)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

        # Top-k eq
        if remaining_time() > 15:
            h_eq, c5_eq = top_k_equalize(h, best_n, top_k=3, max_iters=500, time_limit=30)
            if c5_eq < best_c5 - 1e-15:
                best_c5 = c5_eq
                best_h = h_eq.copy()
                h = h_eq.copy()

        # Subgradient
        if remaining_time() > 15:
            h_sg, c5_sg = subgradient_zigzag(h, best_n, max_iters=5000, time_limit=30)
            if c5_sg < best_c5 - 1e-15:
                best_c5 = c5_sg
                best_h = h_sg.copy()
                h = h_sg.copy()

    # Phase 12: Final SLSQP polish with very high beta
    if remaining_time() > 10:
        h_final = slsqp_opt(best_h, best_n, 500000000, maxiter=3000)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-15:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Validate with evaluator
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
