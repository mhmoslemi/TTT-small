    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    return np.correlate(h, 1.0 - h, mode='full')

def corr_gradient(h, k_idx):
    """Compute gradient of corr[k_idx] with respect to h."""
    n = len(h)
    dx = 2.0 / n
    d = k_idx - (n - 1)  # offset
    g = np.zeros(n)
    if d >= 0:
        # c = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])*dx
        # dc/dh[m] = (1-h[m+d])*dx for m in [0, n-1-d]

        for i in range(n - d):

            g[i + d] -= h[i] * dx

        for i in range(n - a):
            g[i + a] += (1.0 - h[i]) * dx
            g[i] -= h[i + a] * dx
    return g

    #   sum(d) = 0  (1 equality constraint)

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t

    # For each active k: grad_k · d - t ≤ -f_k(h)
    # i.e., [grad_k, -1] · [d, t] ≤ -f_k(h)
    A_ub = np.zeros((top_k + 2 * n, n + 1))

        b_ub[row] = h[i]
        row += 1

        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1

import numpy as np
from scipy.optimize import minimize, linprog
import time

_T0 = 0.0
_BUDGET = 1300.0


def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def corr_grad_single(h, k_idx):
    n = len(h)
    dx = 2.0 / n
    d = k_idx - (n - 1)
    g = np.zeros(n)
    if d >= 0:
        for i in range(n - d):
            g[i] += (1.0 - h[i + d]) * dx
            g[i + d] -= h[i] * dx
    else:
        a = -d
        for i in range(n - a):
            g[i + a] += (1.0 - h[i]) * dx
            g[i] -= h[i + a] * dx
    return g


def smooth_max_obj(x, n, beta):
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    z = beta * (corr - m)
    z = np.clip(z, -50.0, 50.0)
    ez = np.exp(z)
    return (m + np.log(np.sum(ez)) / beta) * dx


def smooth_max_grad(x, n, beta):
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    z = beta * (corr - m)
    z = np.clip(z, -50.0, 50.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    return g


def slsqp_run(h, n, beta, maxiter=500):
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n

    def eq_fun(x):
        return np.sum(x) - n / 2.0

    def eq_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]

    def obj(x):
        return smooth_max_obj(x, n, beta)

    def obj_grad(x):
        return smooth_max_grad(x, n, beta)

    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.fun < smooth_max_obj(h0, n, beta) - 1e-15:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def lp_refine(h, n, top_k=10, max_iters=20000, step_scale=0.5):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    corr = compute_corr(h)
    c5_current = best_c5

    for it in range(max_iters):
        if remaining_time() < 3:
            break
        if it % 50 == 0:
            corr = compute_corr(h)
            c5_current = float(np.max(corr) * (2.0 / n))

        idx_top = np.argsort(-corr)[:top_k]
        c_max = corr[idx_top[0]] * (2.0 / n)

        # Build LP
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0

        n_constr = top_k + 2 * n
        A_ub = np.zeros((n_constr, n + 1))
        b_ub = np.zeros(n_constr)

        dx = 2.0 / n
        row = 0
        for ki in idx_top:
            g = corr_grad_single(h, ki)
            A_ub[row, :n] = g
            A_ub[row, n] = -1.0
            b_ub[row] = -corr[ki] * dx
            row += 1

        for i in range(n):
            A_ub[row, i] = -1.0
            b_ub[row] = h[i] * step_scale
            row += 1
            A_ub[row, i] = 1.0
            b_ub[row] = h[i] * step_scale
            row += 1

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          method='highs', options={'presolve': True, 'time_limit': 5.0})
        except Exception:
            break

        if not res.success:
            break

        d = res.x[:n]
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-14:
            break

        h_try = project_feasible(h + d, n)
        c5_try = compute_c5(h_try)

        if c5_try < best_c5 - 1e-15:
            best_c5 = c5_try
            best_h = h_try.copy()
            h = h_try.copy()
            c5_current = c5_try
            top_k = min(top_k + 1, 20) if it % 100 == 0 else top_k
        else:
            # Reduce step
            for frac in [0.7, 0.5, 0.3, 0.2, 0.1]:
                h_try2 = project_feasible(h + frac * d, n)
                c5_try2 = compute_c5(h_try2)
                if c5_try2 < best_c5 - 1e-15:
                    best_c5 = c5_try2
                    best_h = h_try2.copy()
                    h = h_try2.copy()
                    step_scale *= 0.7
                    break
            else:
                if it > 200:
                    break
                step_scale *= 0.5

        if c5_current - best_c5 < 1e-15 and it > 50:
            # Check if we're at a plateau
            if it % 200 == 0:
                top_k = min(top_k + 1, 25)
            if it > 2000:
                break

    return best_h, best_c5


def topk_direct(h, n, top_k=6, max_iters=300):
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining_time() < 3:
            break
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        c_max = corr[idx_top[0]]
        c_min = corr[idx_top[-1]]
        if c_max - c_min < 1e-11:
            break

        # Direction: increase h where it reduces top correlations, decrease where it increases bottom
        g_dir = np.zeros(n)
        for k in idx_top:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_dir[off:] += (1.0 - h[:np_])
                g_dir[:np_] -= h[off:]
            else:
                a = -off
                np_ = n - a
                g_dir[:np_] += (1.0 - h[a:])
                g_dir[a:] -= h[:np_]

        # Normalize
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        d = g_dir / gn

        improved = False
        for step in [0.01, 0.005, 0.02, 0.001, 0.008, 0.03, 0.05]:
            ht = project_feasible(h - step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            for step in [0.01, 0.005, 0.02]:
                ht = project_feasible(h + step * d, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-14:
                    h = ht
                    best_c5 = cn
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def binary_sa(h, n, time_limit, rng):
    h = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        sort_idx = np.argsort(-h)
        h = np.zeros(n)
        h[sort_idx[:n_half]] = 1.0
    start = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    current_c5 = best_c5
    T = 0.3
    cool_rate = 0.9997
    no_improve = 0
    while time.time() - start < time_limit and remaining_time() > 10:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5
        if delta < 0 or (T > 1e-10 and rng.rand() < np.exp(-delta / T)):
            current_c5 = new_c5
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
        T *= cool_rate
        if no_improve > 5000:
            break
    return best_h, best_c5

    h0 = project_feasible(h, n)
    best_c5 = compute_c5(h0)

    beta_seq = [5000, 20000, 100000, 500000, 2000000]

    for bi, b in enumerate(beta_seq):

        try:
