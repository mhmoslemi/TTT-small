        for h in population:
            f = evaluate_candidate(h)
            fitness.append(f)
        population_with_fitness = sorted(zip(fitness, population))

        # Update best solution
        current_best = population_with_fitness[0]
        if current_best[0] < best_c5:
            best_c5 = current_best[0]
            best_h = current_best[1].copy()

    return (best_h, best_c5, n_points)
