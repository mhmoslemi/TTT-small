        result = opt.minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 300, 'ftol': 1e-5}
        )

        if result.success:
            h_opt = np.clip(result.x, 0, 1)
            h_opt_sum = np.sum(h_opt)
            if not np.isclose(h_opt_sum, target_sum, atol=1e-3):
                h_opt = h_opt * target_sum / h_opt_sum
                h_opt = np.clip(h_opt, 0, 1)

            current_c5 = objective(h_opt)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_opt

    if best_h is None:
        best_h = initial_h_values

    best_h = np.clip(best_h, 0, 1)
    best_h_sum = np.sum(best_h)
    if not np.isclose(best_h_sum, target_sum, atol=1e-6):
        best_h = best_h * target_sum / best_h_sum
        best_h = np.clip(best_h, 0, 1)

    return (best_h.tolist(), best_c5, n_points)
