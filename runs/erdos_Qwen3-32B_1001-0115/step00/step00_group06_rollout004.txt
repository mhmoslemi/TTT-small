    best_c5 = np.max(np.correlate(h, 1 - h, mode="full") * dx)
    
    max_iter = 200
    for iter in range(max_iter):
        if time() - start_time > budget_s:
            break
            
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        current_c5 = np.max(overlaps)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        k_max_idx = np.argmax(overlaps)
        k_max = k_max_idx - (n_points - 1)
        
        # Build linear approximation for current worst k
        c = np.zeros(n_points)
        for i in range(n_points):
            j = i + k_max
            if 0 <= j < n_points:
                c[i] += (1 - h[j])
                c[j] -= h[i]
            else:
                c[i] += 1.0
        
        # Formulate QP with trust region
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(c @ h_var + 2e-3 * cp.sum_squares(h_var - h))
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.OSQP, max_iter=2000)
        
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_candidate = h_var.value
            new_c5 = np.max(np.correlate(h_candidate, 1 - h_candidate, mode="full") * dx)
            if new_c5 < best_c5:
                h = h_candidate
                best_c5 = new_c5
    
    return (best_h.tolist(), best_c5, n_points)
