import numpy as np
import scipy.optimize

def run(seed=42, budget_s=1100):
    h0 = initial_h_values.copy()
    n_points = len(h0)
    dx = 2.0 / n_points

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraint_sum = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        correlations = np.correlate(h, 1 - h, mode='full')
        overlaps = correlations * dx
        return np.max(overlaps)

    result = scipy.optimize.minimize(
        objective,
        x0=h0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraint_sum,
        options={'maxiter': 200, 'ftol': 1e-6, 'disp': False}
    )

    if result.success:
        h_opt = result.x
        # Ensure sum constraint
        sum_h = np.sum(h_opt)
        target_sum = n_points / 2
        if not np.isclose(sum_h, target_sum, atol=1e-6):
            scaling_factor = target_sum / sum_h
            h_opt = h_opt * scaling_factor
            h_opt = np.clip(h_opt, 0.0, 1.0)
    else:
        h_opt = h0

    h_vec = h_opt.reshape(n_points)
    correlations = np.correlate(h_vec, 1 - h_vec, mode='full')
    overlaps = correlations * dx
    c5_bound = np.max(overlaps)

    return (h_opt, c5_bound, n_points)
