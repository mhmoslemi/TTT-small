    cons = [{'type': 'eq', 'fun': sum_constraint}]
    bounds = [(0, 1)] * n_points
    x0 = h_initial.copy().ravel()

    result = opt.minimize(
        compute_c5,
        x0=x0,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'maxiter': 50}
    )

    best_h = result.x.reshape(n_points)
    best_h = np.clip(best_h, 0, 1)
    current_sum = np.sum(best_h)
    if abs(current_sum - n_points / 2) > 1e-6 and current_sum > 1e-10:
        scale = (n_points / 2) / current_sum
        best_h = np.clip(best_h * scale, 0, 1)

    elapsed = time.time() - start_time
    if elapsed > budget_s - 60:
        best_h = h_initial
        best_c5 = compute_c5(h_initial)
    else:
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_points
