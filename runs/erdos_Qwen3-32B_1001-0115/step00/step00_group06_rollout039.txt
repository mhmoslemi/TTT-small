import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    try:
        h = initial_h_values.copy()
        n_points = len(h)
    except:
        n_points = 95
        h = np.zeros(n_points)
        half = n_points / 2
        h[:int(half)] = 1.0
        if n_points % 2 == 1:
            h[int(half)] = 1.0
        # Adjust to ensure exact sum(h) == n_points / 2
        current_sum = np.sum(h)
        adjustment = n_points / 2 - current_sum
        h[-1] += adjustment

    dx = 2.0 / n_points

    def compute_c5(h_val):
        overlaps = np.correlate(h_val, 1 - h_val, mode='full') * dx
        k_relevant = overlaps[n_points - 1 : 2 * n_points - 2]
        return np.max(k_relevant)

    best_c5 = compute_c5(h)
    best_h = h.copy()
    start_time = time.time()

    LAMBDA = 1e-2
    max_iter = 300

    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break

        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        k_relevant = overlaps[n_points - 1 : 2 * n_points - 2]
        current_c5 = np.max(k_relevant)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        if len(k_relevant) == 0:
            break

        max_index = np.argmax(k_relevant)
        k_max = max_index + 1

        # Compute gradient efficiently
        term1 = np.zeros_like(h)
        valid_i = np.arange(n_points - k_max)
        term1[valid_i] = 1.0 - h[valid_i + k_max]

        term2 = np.zeros_like(h)
        term2[k_max:] = h[:n_points - k_max]

        gradient = (term1 - term2) * dx

        # Solve QP
        delta = cp.Variable(n_points)
        objective = cp.Minimize(gradient @ delta + 0.5 * LAMBDA * cp.quad_form(delta, np.eye(n_points)))

            h += delta.value
            h = np.clip(h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n_points
