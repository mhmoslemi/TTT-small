def threshold_and_project(h, n_points, target_sum):

    # Threshold h to 0.01 or 0.99
    h_thresholded = np.where(h > 0.5, 0.99, 0.01)

    current_sum = np.sum(h_thresholded)
    desired_sum = target_sum

    while abs(current_sum - desired_sum) > 1e-6:

        # Determine if we need to increase or decrease sum
        delta = desired_sum - current_sum

        if delta == 0:
            break

        if delta > 0:
            # Need to increase sum: find 0.01 entries and flip to 0.99

            idx = indices[0]
