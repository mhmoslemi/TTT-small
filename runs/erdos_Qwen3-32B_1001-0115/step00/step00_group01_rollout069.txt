    n = len(h_candidate)
    current_sum = np.sum(h_candidate)
    delta = target_sum - current_sum
    if np.abs(delta) < 1e-6:  # negligible
        return h_candidate

    indices = list(range(n))

        if delta > 0:

import numpy as np
from time import time
from numpy import max as npmax, correlate, full

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points

    def compute_c5(hvec):
        return npmax(correlate(hvec, 1 - hvec, mode='full') * dx)

    current_best_h = h.copy()
    current_best_c5 = compute_c5(current_best_h)

    start_time = time()

    tabu_size = max(10, int(0.2 * n_points))
    tabu_list = []
    temperature = 1.0
    cooling_rate = 0.995
    noise_level = 0.05
    step_count = max(5, int(0.1 * n_points))

    def project(hvec):
        hvec_clipped = np.clip(hvec, 0.0, 1.0)
        target_sum = n_points / 2
        delta = target_sum - hvec_clipped.sum()
        indices = np.random.permutation(n_points)
        for idx in indices:
            if abs(delta) < 1e-7:
                break
            if delta > 0:
                space = 1.0 - hvec_clipped[idx]
                if space > 0:
                    add = min(delta, space)
                    hvec_clipped[idx] += add
                    delta -= add
            else:
                space = hvec_clipped[idx]
                if space > 0:
                    subtract = min(abs(delta), space)
                    hvec_clipped[idx] -= subtract
                    delta += subtract
        return hvec_clipped

    for _ in range(1000000):
        if time() - start_time > budget_s - 1.0:
            break

        candidate = h.copy()
        perturb_indices = np.random.choice(n_points, size=step_count, replace=False)
        for idx in perturb_indices:
            delta = np.random.uniform(-noise_level, noise_level)
            candidate[idx] += delta
            candidate[idx] = max(0.0, min(1.0, candidate[idx]))

        candidate_proj = project(candidate)
        candidate_tuple = tuple(np.round(candidate_proj, 6))
        if candidate_tuple in tabu_list:
            continue

        c5_candidate = compute_c5(candidate_proj)
        delta_c5 = c5_candidate - current_best_c5

        if delta_c5 < 0:
            current_best_h[:] = candidate_proj[:]
            current_best_c5 = c5_candidate
            h[:] = candidate_proj[:]
            tabu_list.append(candidate_tuple)
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)
        else:
            if np.random.rand() < np.exp(-delta_c5 / temperature):
                h[:] = candidate_proj[:]
                tabu_list.append(candidate_tuple)
                if len(tabu_list) > tabu_size:
                    tabu_list.pop(0)

        temperature *= cooling_rate

    return current_best_h.tolist(), current_best_c5, n_points
