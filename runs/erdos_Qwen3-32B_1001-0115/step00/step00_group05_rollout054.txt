                    if current_c5 < global_best_c5:
                        global_best = particles[i].copy()
                        global_best_c5 = current_c5
            iteration += 1
    
    h_values = global_best
    c5_bound = evaluate_c5(h_values, n_points)
    return h_values, c5_bound, n_points
