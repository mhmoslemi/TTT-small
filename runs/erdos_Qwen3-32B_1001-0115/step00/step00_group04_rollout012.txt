def repair(h, target_sum):
    h = np.array(h)
    current_sum = float(np.sum(h))
    if abs(current_sum - target_sum) < 1e-8:
        return h
    diff = target_sum - current_sum
    while abs(diff) > 1e-8:
        if diff > 0:  # Need to increase
            # Sort elements in ascending order to find those that can be increased
            indices = np.argsort(h)
            for i in indices:
                max_increase = 1.0 - h[i]
                if max_increase <= 0:
                    continue
                to_add = min(diff, max_increase)
                h[i] += to_add
                diff -= to_add
                if diff <= 1e-8:
                    break
        else:  # Need to decrease
            # Sort elements in descending order to find those that can be decreased
            indices = np.argsort(-h)
            for i in indices:
                max_decrease = h[i]
                if max_decrease <= 0:
                    continue
                to_remove = min(-diff, max_decrease)
                h[i] -= to_remove
                diff += to_remove
                if diff >= -1e-8:
                    break
    return h

import numpy as np
import time

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    return np.max(corr) * dx

def repair(h, target_sum):
    h = h.copy()
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    while abs(diff) > 1e-10:
        if diff > 0:
            indices = np.argsort(h)
            for i in indices:
                max_increase = 1.0 - h[i]
                if max_increase <= 0:
                    continue
                to_add = min(diff, max_increase)
                h[i] += to_add
                diff -= to_add
                if diff <= 1e-10:
                    break
        else:
            indices = np.argsort(-h)
            for i in indices:
                max_decrease = h[i]
                if max_decrease <= 0:
                    continue
                to_remove = min(-diff, max_decrease)
                h[i] -= to_remove
                diff += to_remove
                if diff >= -1e-10:
                    break
    return h

def mutate(h, target_sum, mutation_scale=0.1, mutation_rate=0.1, rng=np.random):
    h = h.copy()
    perturbation = rng.normal(0, mutation_scale, len(h))
    perturbation = np.clip(perturbation, -0.5, 0.5)
    h += mutation_rate * perturbation
    h = np.clip(h, 0, 1)
    h = repair(h, target_sum)
    return h

def crossover(parent1, parent2, target_sum, rng=np.random):
