import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    # Retrieve the initial solution
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    desired_sum = n_points / 2.0

    # Normalize the initial solution to satisfy the integral constraint
    if abs(np.sum(h) - desired_sum) > 1e-6:
        h *= desired_sum / np.sum(h)

    best_h = h.copy()
    best_c5 = float('inf')

    # Time tracking
    start_time = time.time()
    max_time = budget_s

    while time.time() - start_time < max_time:
        overlaps = np.zeros(n_points)
        for k in range(n_points):
            shifted = np.roll(h, -k)
            overlaps[k] = dx * np.sum(h * (1.0 - shifted))
        current_c5 = np.max(overlaps)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        k_max = np.argmax(overlaps)
        if current_c5 <= 0.38085:
            break  # Stop early if we meet or beat target C5

        # Compute gradients for k_max
        grad = np.zeros(n_points)
        for i in range(n_points):
            i_plus = (i + k_max) % n_points
            i_minus = (i - k_max) % n_points
            grad[i] = (1.0 - h[i_plus] - h[i_minus]) * dx

        # Convex QP formulation
        h_var = cp.Variable(n_points)
        mu = 1.0  # Regularization strength
        objective = cp.Minimize(grad @ h_var + 0.5 * mu * cp.sum_squares(h_var - h))
        constraints = [

            h_var <= 1.0,
            cp.sum(h_var) == desired_sum

        problem = cp.Problem(objective, constraints)

        if problem.status != 'optimal':

        h_new = h_var.value
