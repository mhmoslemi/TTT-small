import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    h = initial_h_values.copy()
    np.random.seed(seed)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    eta = 0.01
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 10:
        corr = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(corr)
        m_max = np.argmax(corr)
        lag = m_max - (n_points - 1)
        
        # Compute shifted_h
        lag_int = int(lag)
        shifted_h = np.zeros_like(h)
        if lag_int >= 0:
            shifted_h[lag_int:] = h[:-lag_int]
        else:
            lag_int_abs = -lag_int
            shifted_h[:n_points - lag_int_abs] = h[lag_int_abs:]
        
        # Gradient step
        gradient = shifted_h - h
        h -= eta * gradient
        
        # Project to [0,1] and renormalize
        h = np.clip(h, 0, 1)
        current_sum = h.sum()
        if current_sum > 1e-12:
            h *= target_sum / current_sum
        
        # Check improvement
        new_corr = np.correlate(h, 1 - h, mode='full') * dx
        new_c5 = np.max(new_corr)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h[:] = h[:]
            eta = min(eta * 1.1, 0.1)
        else:
            eta = max(eta * 0.5, 1e-5)
    
    return best_h, best_c5, n_points
