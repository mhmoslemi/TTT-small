    alpha = 0.99
    tolerance = 1e-8
    
    while time() - start_time < budget_s and T > tolerance:
        for _ in range(1000):
            if time() - start_time >= budget_s:
                break
            new_h = current_h.copy()
            while True:
                i, j = np.random.choice(n_points, size=2, replace=False)
                if new_h[j] > 1e-8 or new_h[i] < 1 - 1e-8:
                    break
            delta_max = min(1 - new_h[i], new_h[j])
            if delta_max < 1e-8:
                continue
            delta = np.random.rand() * delta_max
            new_h[i] = min(1.0, new_h[i] + delta)
            new_h[j] = max(0.0, new_h[j] - delta)
            new_c5 = compute_c5(new_h)
            delta_c = new_c5 - best_c5
            if delta_c < 0 or np.random.rand() < math.exp(-delta_c / T):
                current_h = new_h
                if delta_c < 0:
                    best_h = current_h.copy()
                    best_c5 = new_c5
        T *= alpha
    
    return (best_h.tolist(), best_c5, n_points)
