import numpy as np
import cvxpy as cp

def run(seed=42, budget_s=1100):
    h_init = initial_h_values
    n_points = len(h_init)
    sum_required = n_points / 2
    dx = 2.0 / n_points

    h_rounded = np.round(h_init).astype(int)
    current_sum = np.sum(h_rounded)
    delta = int(sum_required - current_sum)
    
    if delta != 0:
        idxs_0 = np.where(h_rounded == 0)[0]
        idxs_1 = np.where(h_rounded == 1)[0]
        if delta > 0:
            h_rounded[idxs_0[:delta]] = 1
        else:
            h_rounded[idxs_1[: -delta]] = 0

    h = cp.Variable(n_points, boolean=True)
    C5 = cp.Variable(nonneg=True)
    constraints = [cp.sum(h) == sum_required]

    z_vars = {}
    for i in range(n_points):
        for j in range(i + 1, n_points):
            z = cp.Variable()
            z_vars[(i, j)] = z
            constraints += [z >= h[i] + h[j] - 1,
                            z <= h[i],
                            z <= h[j]]

    for k in range(1, n_points):
        valid_i = n_points - k
        sum_z = 0
        for i in range(valid_i):
            j = i + k
            sum_z += z_vars[(i, j)]
        constraints.append(sum_z >= sum_required - C5 / dx)

    objective = cp.Minimize(C5)
    problem = cp.Problem(objective, constraints)

    try:
        problem.solve(solver=cp.GUROBI, verbose=False, timeLimit=budget_s)
    except Exception as e:
        pass

    if h.value is None:
        h_values = h_rounded
        c5_bound = evaluate_erdos_solution(h_values)
    else:
        h_values = np.round(h.value).astype(int)
        c5_bound = evaluate_erdos_solution(h_values)

    return h_values.tolist(), c5_bound, n_points
