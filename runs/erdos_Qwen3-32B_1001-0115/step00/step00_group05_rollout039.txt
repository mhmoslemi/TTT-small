import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    h = initial_h_values.copy()
    n_points = len(h)
    T = n_points / 2.0
    dx = 2.0 / n_points
    start_time = time.time()

    best_h = h.copy()
    best_c5 = float('inf')

    # Evaluate initial C5
    overlaps = np.correlate(h, 1 - h, mode='full') * dx
    best_c5 = np.max(overlaps)

    while time.time() - start_time < budget_s:
        # Compute overlaps
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(overlaps)
        max_indices = np.where(overlaps == current_c5)[0]

        active_ks = []
        for idx in max_indices:
            k_candidate = idx - (n_points - 1)
            if 1 <= k_candidate <= n_points - 1:
                active_ks.append(k_candidate)
        active_ks = np.unique(active_ks)
        if not active_ks:
            continue  # Skip if no valid k

        # Compute total gradient
        total_gradient = np.zeros(n_points, dtype=np.float64)
        for k in active_ks:
            grad = np.zeros(n_points)
            for i in range(n_points):
                term1 = (1 - h[i + k]) * dx if (i + k) < n_points else 0.0
                term2 = h[i - k] * dx if (i - k) >= 0 else 0.0
                grad[i] += term1 - term2
            total_gradient += grad
        total_gradient /= max(1, len(active_ks))

        # Learning rate schedule
        lr = 0.1 * np.exp(-0.01 * (time.time() - start_time))

        # Update h
        h_candidate = h - lr * total_gradient
        h_clipped = np.clip(h_candidate, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h != 0:
            scale = T / sum_h
            h_clipped = np.clip(h_clipped * scale, 0, 1)

        prev_h = h.copy()
        h[:] = h_clipped

        # Re-evaluate C5 and update best
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(overlaps)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h[:] = h

        # Projected gradient descent with backtracking if needed
        if np.isnan(current_c5):
            h[:] = prev_h

    return best_h, best_c5, n_points
