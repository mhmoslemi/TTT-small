                gradient[i] -= h[x] * dx

        h_update = h - learning_rate * gradient
        h_clipped = np.clip(h_update, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum != 0:
            h = h_clipped * (target_sum / current_sum)
        else:
            h = h.copy()

        new_c5 = evaluate_erdos_solution(h)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            learning_rate *= 1.05
        else:
            h = best_h.copy()
            learning_rate *= 0.5

        learning_rate = np.clip(learning_rate, 1e-4, 1e-1)

    return best_h.tolist(), best_c5, n_points
