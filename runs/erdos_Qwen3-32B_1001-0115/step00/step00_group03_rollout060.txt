    g_best = p_best[np.argmin(p_best_c5)]
    g_best_c5 = min(p_best_c5)
    
    omega_start = 0.7
    omega_end = 0.4
    c1 = 2.0
    c2 = 2.0
    
    no_improvement = 0
    max_no_improvement = 1000

def adjust_sum(h, target_sum):
    h = h.copy()
    current_sum = h.sum()
    if np.isclose(current_sum, target_sum):
        return h
    D = target_sum - current_sum
    if D > 1e-10:
        indices = np.argsort(h)
        for i in indices:
            capacity = 1.0 - h[i]
            add = min(capacity, D)
            h[i] += add
            D -= add
            if D <= 1e-10:
                break
    elif D < -1e-10:
        indices = np.argsort(-h)
        for i in indices:
            capacity = h[i]
            subtract = min(capacity, -D)
            h[i] -= subtract
            D += subtract
            if D >= -1e-10:
                break
    return h
