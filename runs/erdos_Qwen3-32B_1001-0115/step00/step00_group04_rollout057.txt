    # Enforce integral constraint on initial h
    def normalize_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        total = np.sum(h_vec)
        if abs(total - n_points / 2) > 1e-8:
            scale = (n_points / 2) / total
            h_vec = np.clip(h_vec * scale, 0, 1)
        return h_vec

    h = normalize_h(h)
    best_h = h.copy()
    best_c5 = float('inf')

    def compute_c5(h_vec):
        h_vec = h_vec.reshape(n_points)
        return np.max(np.correlate(h_vec, 1 - h_vec, mode='full')) * dx

    def objective(x, penalty=1e6):
        x = x.reshape(n_points)
        x = normalize_h(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx + penalty * (np.sum(x) - n_points / 2)**2

    bounds = [(0, 1) for _ in range(n_points)]
    minimizer_kwargs = {
        'method': 'L-BFGS-B',
        'bounds': bounds,
        'options': {'ftol': 1e-6, 'maxiter': 50}
    }

    # Track time and run basin-hopping if requested
    start_time = time.time()

    # Run basin-hopping if time allows
    if budget_s > 20:
        result = basinhopping(
            objective,
            h.ravel(),
            niter=20,
            minimizer_kwargs=minimizer_kwargs,
            T=1.0,
            stepsize=1.0,
            interval=10,
            take_step=None
        )
        if result is not None:
            h_candidate = result.x.reshape(n_points)
            h_candidate = normalize_h(h_candidate)
            current_c5 = compute_c5(h_candidate)
            if current_c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = current_c5

    # Final normalization and evaluation
    best_h = normalize_h(best_h)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n_points
