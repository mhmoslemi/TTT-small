def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    max_overlap = 0.0
    for k in range(1, n):
        total = 0.0
        for i in range(n - k):
            total += h[i] * (1 - h[i + k])
        max_overlap = max(max_overlap, total * dx)

import numpy as np
import time

def compute_c5_fast(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    relevant = corr[n-1 : 2*n - 2]
    return np.max(relevant) * dx

def run():
    start_time = time.time()
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)
    best_c5 = compute_c5_fast(h)
    last_improvement = start_time
    
    while time.time() - start_time < 1100:
        # Find indices where h is 0 and 1
        zeros = np.where(h < 0.5)[0]
        ones = np.where(h > 0.5)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        # Random swap
        i, j = np.random.choice(zeros), np.random.choice(ones)
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5_fast(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            last_improvement = time.time()
        else:
            # Revert swap if not improving
            h[i], h[j] = h[j], h[i]
        # Early exit if no progress in 30s
        if time.time() - last_improvement > 30 and time.time() - start_time > 600:
            break
    return h, best_c5, n_points
