        for phase in phases:
            if time.time() - start_time >= budget_s:
                return (best_h, best_c5, n_points)
            # Generate swaps for this phase
            if phase == 'pairwise':
                swap_indices = [(i, j) for i in range(n_points) for j in range(i+1, n_points)]
            elif phase == 'neighborhood':
                swap_indices = [(i, i+1) for i in range(n_points - 1)]
            else:  # random
                swap_indices = []
                for _ in range(n_points * 5):
                    i, j = np.random.randint(n_points, size=2)
                    if i == j:
                        continue
                    swap_indices.append((min(i, j), max(i, j)))
                # Remove duplicates
                swap_indices = list(dict.fromkeys(swap_indices))
            
            # Shuffle the indices
            np.random.shuffle(swap_indices)
            
            for i, j in swap_indices:
                if time.time() - start_time >= budget_s:
                    return (best_h, best_c5, n_points)
                # Perform the swap
                h_i = h[i]
                h_j = h[j]
                h[i] = 1 - h_i
                h[j] = 1 - h_j
                
                # Evaluate the new C5
                new_c5 = evaluate_erdos_solution(h)
                
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h[:] = h[:]
                else:
                    # Revert the swap
                    h[i] = h_i
                    h[j] = h_j
        iteration += 1
        if iteration >= max_iterations:
            break
    
    return (best_h, best_c5, n_points)
