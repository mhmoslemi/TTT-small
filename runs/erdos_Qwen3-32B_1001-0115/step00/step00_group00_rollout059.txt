    v = np.zeros(n_points)
    t = 0

    best_h = h.copy()
    best_c5 = float('inf')
    epsilon_smooth = 0.1
    patience = 500
    no_improvement_counter = 0
    decay_rate = 0.5

    while time.time() - start_time < budget_seconds - 1.0:
        t += 1

        # Compute overlaps
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(overlaps)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_counter = 0
        else:
            no_improvement_counter += 1

        if no_improvement_counter >= patience:
            alpha *= decay_rate
            no_improvement_counter = 0
            if alpha < 1e-6:
                break

        # Compute smooth max gradient
        log_sum = np.log(np.sum(np.exp(overlaps / epsilon_smooth))) * epsilon_smooth
        grad_overlaps = np.exp(overlaps / epsilon_smooth) / (np.sum(np.exp(overlaps / epsilon_smooth)) + 1e-12)

        # Compute gradient via cross-correlation
        part1 = np.correlate(grad_overlaps, (1 - h), mode='full')[n_points-1:2*n_points-1]
        part2 = np.correlate(grad_overlaps, -h, mode='full')[n_points-1:2*n_points-1]
        grad_h = (part1 + part2) * dx

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_h
        v = beta2 * v + (1 - beta2) * (grad_h ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = alpha * m_hat / (np.sqrt(v_hat) + epsilon_adam)

        h -= step

        # Projection to feasible region
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        h *= (n_points / 2.0) / (sum_h + 1e-12)

    # Final evaluation
    overlaps_best = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(overlaps_best)

    return best_h, final_c5, n_points
