        max_corr_idx = np.argmax(current_corrs)
        k0 = max_corr_idx - (n_points - 1)

        gradient = np.zeros_like(h)
        if k0 > 0:
            for i in range(n_points - k0):
                j = i + k0
                gradient[i] += (1 - h[j])
                gradient[j] -= h[i]
        elif k0 < 0:
            abs_k = -k0
            for i in range(abs_k, n_points):
                j = i - abs_k
                gradient[i] += (1 - h[j])
                gradient[j] -= h[i]
        else:
            for i in range(n_points):
                gradient[i] += (1 - 2 * h[i])

        h_new = h - alpha * gradient
        h_new = np.clip(h_new, 0.0, 1.0)
        h_sum = np.sum(h_new)

        if not np.isclose(h_sum, target_sum):
            delta = (target_sum - h_sum) / n_points
            h_new += delta
            h_new = np.clip(h_new, 0.0, 1.0)

        h = h_new.copy()
        current_c5 = evaluate_c5(h, dx)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            alpha = 0.1  # Reset learning rate on improvement
        else:
            alpha *= 0.9  # Reduce learning rate otherwise

    return best_h.tolist(), best_c5, n_points
