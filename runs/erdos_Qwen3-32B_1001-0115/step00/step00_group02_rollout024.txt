import numpy as np
import time

def project(h, target_sum):
    h_clipped = np.clip(h, 0, 1)
    current_sum = h_clipped.sum()
    if abs(current_sum - target_sum) < 1e-6:
        return h_clipped
    delta = (target_sum - current_sum) / len(h)
    h_adjusted = h_clipped + delta
    h_adjusted = np.clip(h_adjusted, 0, 1)
    return h_adjusted

def compute_gradient(h, k):
    n = len(h)
    grad = np.zeros(n)
    k_abs = abs(k)
    if k_abs == 0:
        grad = (1 - 2 * h)
        return grad
    if k_abs >= n:
        return grad
    indices1 = np.arange(n - k_abs)
    indices2 = indices1 + k_abs
    grad[indices1] += 1 - h[indices2]
    grad[indices2] -= h[indices1]
    return grad
