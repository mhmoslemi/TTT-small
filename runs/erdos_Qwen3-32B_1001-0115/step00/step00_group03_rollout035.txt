import time

def run():
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        ones_minus_h = 1.0 - h
        conv = np.correlate(h, ones_minus_h, mode='full') * dx
        return np.max(conv)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': lambda h_flat: np.sum(h_flat) - n_points / 2}
    h_flat_initial = h_initial.copy().flatten()
    
    start_time = time.time()
    budget_s = 1100
    
    result = minimize(
        objective,
        h_flat_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=constraint,
        options={'maxiter': 1000, 'disp': False}
    )
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s * 0.9:
        return h_initial, objective(h_initial), n_points

    h_opt = result.x.reshape(n_points)
    h_opt = np.clip(h_opt, 0.0, 1.0)
    total = np.sum(h_opt)
    if not np.isclose(total, n_points / 2, atol=1e-6):
        h_opt = h_opt * (n_points / 2) / total
    
    c5_bound = objective(h_opt)
    return h_opt, c5_bound, n_points
