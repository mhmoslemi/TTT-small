    current_c5 = evaluate_erdos_solution(current_h)
    
    best_h = current_h.copy()
    best_c5 = current_c5
    
    T = 1.0  # Initial temperature
    alpha = 0.999  # Cooling rate
    start_time = time()
    
    while time() - start_time < budget_s - 1:
        # Perturb current solution
        new_h = current_h.copy()
        indices = np.random.choice(n_points, size=2, replace=False)
        i, j = indices
        
        # Determine possible changes
        delta_add = min(1 - new_h[i], new_h[j])
        delta_subtract = min(new_h[i], 1 - new_h[j])
        
        # Apply perturbation
        if np.random.rand() < 0.5:
            if delta_add > 1e-12:
                delta = np.random.uniform(0, delta_add)
                new_h[i] += delta
                new_h[j] -= delta
            else:
                continue  # Skip invalid perturbation
        else:
            if delta_subtract > 1e-12:
                delta = np.random.uniform(0, delta_subtract)
                new_h[i] -= delta
                new_h[j] += delta
            else:
                continue  # Skip invalid perturbation
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(new_h)
        delta_c = new_c5 - current_c5
        
        # Acceptance criterion
        if delta_c < 0 or np.random.rand() < np.exp(-delta_c / T):
            current_h = new_h
            current_c5 = new_c5
        
        # Update best solution
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = current_h.copy() if np.array_equal(new_h, current_h) else new_h.copy()
        
        # Cool down
        T *= alpha
    
    return best_h.tolist(), best_c5, n_points
