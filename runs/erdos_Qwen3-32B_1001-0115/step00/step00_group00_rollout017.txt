def run(seed=42, budget_s=1100):
    # Use initial_h_values
    h = initial_h_values.copy()

    dx = 2.0 / n_points

    # Best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Optimization parameters
    learning_rate = 0.01
    max_iter = 100000
    epsilon_grad = 1e-4
    start_time = time.time()

    for iter in range(max_iter):
        # Check remaining time
        if time.time() - start_time >= budget_s:
            break

        # Compute current overlaps and C5
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(overlaps)

        # Update best solution
        if current_c5 < best_c5:
            best_h[:] = h.copy()
            best_c5 = current_c5

        # Compute gradient with finite differences
        grad = np.zeros(n_points)
        h_flat = h.flatten()
        for i in range(n_points):
            # Perturb h[i] upwards
            h_perturbed = h_flat.copy()
            h_perturbed[i] += epsilon_grad
            overlaps_perturbed = np.correlate(h_perturbed, 1 - h_perturbed, mode='full') * dx
            c_perturbed = np.max(overlaps_perturbed)
            grad[i] = (c_perturbed - current_c5) / epsilon_grad

        # Gradient descent step
        h_flat -= learning_rate * grad

        # Project back into feasible region
        # Step 1: clip to [0,1]
        h_clipped = np.clip(h_flat, 0, 1)
        sum_clipped = h_clipped.sum()
        target_sum = n_points / 2.0

            h_projected = np.clip(h_scaled, 0, 1)
        else:
            # This is an edge case; handle with clipping
            h_projected = h_clipped.copy()

        # Adjust learning rate
        if iter % 100 == 0:
            learning_rate *= 0.95

    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(best_h)
