def project(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = h.sum()
    if abs(current_sum - target_sum) < 1e-8:
        return h
    delta = target_sum - current_sum
    non_corner = (h > 0) & (h < 1)
    if not np.any(non_corner):
        return h
    num_non_corner = non_corner.sum()
    adjustment = delta / num_non_corner
    h_copy = h.copy()
    h_copy[non_corner] += adjustment
    h_copy = np.clip(h_copy, 0, 1)
    return h_copy
