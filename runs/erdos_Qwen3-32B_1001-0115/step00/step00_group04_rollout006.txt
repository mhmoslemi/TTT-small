        if time.time() - start_time > budget_s - 10:
            break

        current_h = best_h.copy()

        # Setup constraints and bounds
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        bounds = [(0, 1) for _ in range(n_points)]

        # Run optimization
        result = scipy.optimize.minimize(
            smooth_objective, current_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 20, 'ftol': 1e-6}
        )
        if result.success:
            optimized_h = result.x
            optimized_h = np.clip(optimized_h, 0, 1)
            optimized_h_sum = np.sum(optimized_h)
            if optimized_h_sum != 0:
                optimized_h *= target_sum / optimized_h_sum
            current_c5 = evaluate_objective(optimized_h, dx)
            if current_c5 < best_c5:
                best_h = optimized_h.copy()
                best_c5 = current_c5

        # Perturbation step
        perturbation = np.random.normal(0, perturbation_scale, size=n_points)
        best_h = np.clip(best_h + perturbation, 0, 1)
        best_h_sum = np.sum(best_h)
        if best_h_sum != 0:
            best_h *= target_sum / best_h_sum

    # Final rescaling to ensure constraints
    best_h_sum = np.sum(best_h)
    if abs(best_h_sum - target_sum) > 1e-6:
        best_h = best_h * (target_sum / best_h_sum)
    best_h = np.clip(best_h, 0, 1)
    c5_bound = evaluate_objective(best_h, dx)

    return (best_h.tolist(), c5_bound, n_points)

def evaluate_objective(h, dx):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)
