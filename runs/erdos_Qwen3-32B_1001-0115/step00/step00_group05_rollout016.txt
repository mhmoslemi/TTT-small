        if time.time() - start_time > budget_s - 0.1:
            break

        overlaps = compute_overlaps(h)
        current_c5 = overlaps.max()

        if current_c5 < best_c5 - 1e-8:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement >= patience:
                break

        active_indices = np.flatnonzero(np.abs(overlaps - current_c5) < 1e-8)
        active_lags = [idx - (n_points - 1) for idx in active_indices]

        grad = np.zeros(n_points)
        for k in active_lags:
            if k >= 0:
                for i in range(n_points - k):
                    j = i + k
                    grad[i] += 1.0 - h[j]
                    grad[j] -= h[i]
            else:
                m = -k
                for i in range(m, n_points):
                    j = i - m
                    grad[i] += 1.0 - h[j]
                    grad[j] -= h[i]

        grad *= dx
        h_candidate = np.clip(h - eta * grad, 0.0, 1.0)
        h_proj = project(h_candidate, target_sum)

        if h_proj is not None:
            h = h_proj.copy()
        else:
            h = np.clip(h, 0.0, 1.0)

        if iteration % 100 == 99:
            eta *= 0.95

    final_overlaps = compute_overlaps(best_h)
    final_c5 = final_overlaps.max()
    return (best_h.tolist(), final_c5, n_points)
