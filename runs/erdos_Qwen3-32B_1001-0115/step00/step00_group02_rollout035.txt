import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n_points = len(h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    
    def adjust_and_clip(h, target_sum):
        np.clip(h, 0, 1, out=h)
        current_sum = np.sum(h)
        delta = target_sum - current_sum
        if abs(delta) < 1e-8:
            return
        mask = (h > 1e-8) & (h < 1 - 1e-8)
        count = np.sum(mask)
        if count == 0:
            return
        h[mask] += delta / count
        np.clip(h, 0, 1, out=h)
    
    def compute_c5(h, dx):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    current_h = h.copy()
    
    T = 1.0
    alpha = 0.999
    delta_step = 0.1
    
    while time.time() - start_time < budget_s:
        h_before = current_h.copy()
        i, j = np.random.choice(n_points, size=2, replace=False)
        current_h[i] += delta_step
        current_h[j] -= delta_step
        adjust_and_clip(current_h, target_sum)
