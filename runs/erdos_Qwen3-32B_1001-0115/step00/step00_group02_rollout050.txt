def reconstruct_initial_h(initial_h_values, n_points):
    # Round to nearest binary while preserving the sum constraint

        h[np.argmin(h)] = 1
        current_sum += 1
    while current_sum > target:
        h[np.argmax(h)] = 0
        current_sum -= 1
