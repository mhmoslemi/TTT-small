import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    # Initial step function
    h = initial_h_values.copy().astype(float)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # sum(h) must equal this

    # Ensure initial h is valid
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if not np.isclose(h_sum, target_sum):
        # Adjust sum if needed
        scale_factor = target_sum / h_sum
        h = np.clip(h * scale_factor, 0, 1)

    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()
    max_iterations = 500
    learning_rate = 1e-3
    tolerance = 1e-5  # for considering overlaps as equal to the maximum

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break

        # Compute overlaps for all k
        overlaps = []
        for k in range(n_points):
            shifted = np.roll(h, -k)
            overlap_val = np.sum(h * (1 - shifted)) * dx
            overlaps.append(overlap_val)
        overlaps = np.array(overlaps)
        current_c5 = np.max(overlaps)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Find max lags
        max_val = current_c5
        max_lags = [k for k in range(n_points) if abs(overlaps[k] - max_val) < tolerance]

        # Compute gradient from max lags
        grad_total = np.zeros(n_points)
        for k in max_lags:
            for i in range(n_points):
                j = (i + k) % n_points
                term1 = (1 - h[j]) - 2 * h[i]
                term2 = -h[i]
                grad_total[i] += term1
                grad_total[j] += term2
        grad_total *= dx  # Multiply by dx after accumulating terms

        # Apply gradient step
        h -= learning_rate * grad_total

        # Clip to [0,1]
        h = np.clip(h, 0, 1)

        # Adjust sum
        current_sum = np.sum(h)
        if not np.isclose(current_sum, target_sum):
            delta = target_sum - current_sum
            non_bounds = np.where((h > 0) & (h < 1))[0]
            if len(non_bounds) > 0:
                correction = delta / len(non_bounds)
                h[non_bounds] += correction
                h = np.clip(h, 0, 1)

    return (best_h.tolist(), best_c5, n_points)
