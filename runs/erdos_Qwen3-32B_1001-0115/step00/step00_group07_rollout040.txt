    best_c5 = compute_c5(best_h, dx)
    start_time = time.time()
    max_time = budget_s

    while time.time() - start_time < max_time:
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        k_star = np.argmax(corr)
        k_shift = k_star - (n_points - 1)
        grad = compute_gradient(h, k_shift, dx)
        h -= 0.01 * grad
        h = np.clip(h, 0, 1)
        current_sum = h.sum()
        if not np.isclose(current_sum, s, atol=1e-8):
            if current_sum != 0:
                h = h * (s / current_sum)
            h = np.clip(h, 0, 1)
            current_sum_new = h.sum()
            if not np.isclose(current_sum_new, s, atol=1e-8):
                delta = s - current_sum_new
                h += delta / n_points
                h = np.clip(h, 0, 1)
    h = best_h
    h = np.clip(h, 0, 1)
    current_sum = h.sum()
    if not np.isclose(current_sum, s, atol=1e-8):
        if current_sum != 0:
            h = h * (s / current_sum)
        h = np.clip(h, 0, 1)
    return (h.tolist(), best_c5, n_points)
