    h_clamped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clamped)
    if current_sum == 0:
        h_clamped = np.ones_like(h_clamped) * (target_sum / h_clamped.size)
    else:
        h_clamped *= (target_sum / current_sum)
    return np.clip(h_clamped, 0, 1)

def mutate(h, target_sum):
    noise_std = 0.1
    h_mut = h + np.random.normal(0, noise_std, h.shape)
    return normalize_h(h_mut, target_sum)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    global initial_h_values
    use_initial = 'initial_h_values' in globals()
    if use_initial:
        h_initial = initial_h_values
        n_points = len(h_initial)
    else:
        n_points = 42
        h_initial = np.random.rand(n_points)
        h_initial = normalize_h(h_initial, n_points / 2)

    target_sum = n_points / 2.0

    # Initialize population
    pop_size = 50
    population = []
    population.append(h_initial.copy())
    for _ in range(pop_size - 1):
        h = h_initial.copy()
        h += np.random.normal(0, 0.1, h.shape)
        h = normalize_h(h, target_sum)
        population.append(h)
    population = np.array(population)

    best_c5 = evaluate_erdos_solution(h_initial)
    best_h = h_initial.copy()

    while time.time() - start_time < budget_s:
        # Evaluate all
        c5_list = [evaluate_erdos_solution(h) for h in population]
        best_index = np.argmin(c5_list)
        current_best_c5 = c5_list[best_index]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[best_index].copy()

        # Selection
        rank = np.argsort(c5_list)
        top_indices = rank[:int(0.2 * pop_size)]
        top_parents = population[top_indices]

        # Generate new population with elitism
        new_pop = [p.copy() for p in top_parents]
        while len(new_pop) < pop_size:
            # Crossover
            p1, p2 = top_parents[np.random.choice(len(top_parents), 2)]
            child1 = np.where(np.random.rand(n_points) < 0.5, p1, p2)
            child2 = np.where(np.random.rand(n_points) < 0.5, p2, p1)

            # Mutate
            child1 = mutate(child1, target_sum)
            child2 = mutate(child2, target_sum)

            new_pop.append(child1)
            new_pop.append(child2)

        population = np.array(new_pop[:pop_size])

    return best_h, best_c5, n_points
