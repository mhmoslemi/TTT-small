from time import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time()
    h = np.array(initial_h_values, dtype=np.float64)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = None

    while time() - start_time < budget_s - 1.0:
        cross_corr = correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(cross_corr)
        if best_c5 is None or current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

        max_index = np.argmax(cross_corr)
        l_shift = max_index - (n_points - 1)

        valid_i = []
        j_values = []
        for i in range(n_points):
            j = i + l_shift
            if 0 <= j < n_points:
                valid_i.append(i)
                j_values.append(j)

        if not valid_i:
            break

        contributions = np.array([h[i] * (1 - h[j]) for i, j in zip(valid_i, j_values)])
        if np.all(contributions <= 1e-6):
            break

        max_c_idx = np.argmax(contributions)
        i_max = valid_i[max_c_idx]
        j_max = j_values[max_c_idx]

        delta = min(h[i_max], 1 - h[j_max])
        if delta > 1e-8:
            h[i_max] -= delta
            h[j_max] += delta
            h[i_max] = np.clip(h[i_max], 0, 1)
            h[j_max] = np.clip(h[j_max], 0, 1)

    cross_corr = correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(cross_corr)
