import numpy as np
import cvxpy as cp
import time

def compute_c5(h, dx):
    overlaps = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(overlaps)

def project_h(h, target_sum):
    n = len(h)
    h_var = cp.Variable(n)
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum]
    objective = cp.Minimize(cp.sum_squares(h_var - h))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status != cp.OPTIMAL:
            raise ValueError
    except:
        h_var = np.clip(np.linspace(0, 1, n), 0, 1).copy()
        h_var *= target_sum / np.sum(h_var)
    return h_var.value if hasattr(h_var, 'value') else h_var

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    learning_rate = 0.1
    decay_rate = 0.95
    decay_steps = 100
    early_stop_threshold = 1e-6
    early_stop_window = 100

    start_time = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    i_decay = 0

    while time.time() - start_time < budget_s:
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(overlaps)
        active_indices = np.where(np.isclose(overlaps, current_c5, atol=1e-8))[0]
        active_k_list = [int(k_idx - (n_points - 1)) for k_idx in active_indices]
        active_k = list(set(active_k_list))

        grad = np.zeros(n_points)
        for k in active_k:
            start_i = max(0, -k)
            end_i = min(n_points - 1, n_points - 1 - k)
            for i in range(start_i, end_i + 1):
                j = i + k
                grad[i] += (1 - h[j])

        h_proposed = h - learning_rate * grad
