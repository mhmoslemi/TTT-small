import time

    n_points = len(h)

        middle = n_points - 1
        positive_part = corr[middle + 1:]
        return np.max(positive_part) * dx

    best_h = h.copy()
    best_c5 = evaluate_c5(h)

    constraints_coeffs = []

    while time.time() - start_time < budget_s - 1:

        middle = n_points - 1

            break

            constraints_coeffs.append(coeffs)

        problem_constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,

            desired_sum = n_points / 2.0
