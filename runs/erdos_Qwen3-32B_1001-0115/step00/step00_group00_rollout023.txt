    np.random.seed(seed)
    n_points = 45

    start_time = time.time()
    while time.time() - start_time < budget_s - 1:
        t += 1

        # Compute overlaps for all shifts k = 1 to n_points-1
        overlaps = correlate(h, 1 - h, mode='full') * dx
        relevant_overlaps = overlaps[n_points:]

        grad = np.zeros_like(h)

        for k_idx in range(len(relevant_overlaps)):
