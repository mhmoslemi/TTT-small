def project(h, target_sum):
    sum_h = h.sum()
    if sum_h == 0:
        return np.zeros_like(h)
    h_scaled = h * (target_sum / sum_h)
    h_clipped = np.clip(h_scaled, 0.0, 1.0)
    sum_clipped = h_clipped.sum()
    if sum_clipped == 0:
        return h_scaled
    h_final = h_clipped * (target_sum / sum_clipped)
    return h_final
