import numpy as np

def run(seed=42, budget_s=1100):

    positions = np.zeros((swarm_size, n_points))

        perturbation = np.random.uniform(-0.05, 0.05, n_points)
        new_h = h0 + perturbation
        new_h = np.clip(new_h, 0, 1)
        s = new_h.sum()
        scale = target_sum / s if s != 0 else 1.0

    velocities = np.random.uniform(-v_max, v_max, (swarm_size, n_points))

    # Initialize personal bests and global best
    p_best_positions = positions.copy()
    p_best_values = np.array([evaluate_erdos_solution(pos) for pos in positions])
    g_best_index = np.argmin(p_best_values)

    start_time = time.time()
