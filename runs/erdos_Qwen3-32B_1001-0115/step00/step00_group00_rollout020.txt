import numpy as np
from time import time

def run(seed=42, budget_s=1100, **kwargs):
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)
    start_time = time()

    learning_rate = 0.01

    while time() - start_time < budget_s:
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_index = np.argmax(overlaps)
        k_shift = max_index - (n_points - 1)
        current_c5 = overlaps[max_index]

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        grad = np.zeros(n_points)
        for j in range(n_points):
            j_plus_k = (j + k_shift) % n_points
            j_minus_k = (j - k_shift) % n_points
            term1 = (1 - h[j_plus_k])
            term2 = h[j_minus_k]
            grad[j] = (term1 - term2) * dx

        h -= learning_rate * grad

        h_clipped = np.clip(h, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            h.fill(0.5)
        else:
            scale_factor = (n_points / 2.0) / current_sum
            h_scaled = h_clipped * scale_factor
            h = np.clip(h_scaled, 0, 1)

    return best_h, best_c5, n_points
