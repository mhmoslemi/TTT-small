import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Use provided initial_h_values
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points

    # Ensure initial h satisfies the integral constraint
    current_sum = np.sum(h_initial)
    if abs(current_sum - n_points / 2) > 1e-8:
        h_initial = h_initial * (n_points / 2) / current_sum

    def objective(h_flat):
        h = h_flat.reshape(-1)
        one_minus_h = 1.0 - h
        correlation = np.correlate(h, one_minus_h, mode='full')
        max_overlap = np.max(correlation) * dx  # Multiply by dx to get C5
        return max_overlap

    # Constraints: sum(h) = n_points / 2
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    # Bounds: 0 <= h_i <= 1
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Optimization settings
    result = minimize(

        x0=h_initial,
        method='SLSQP',
        bounds=bounds,
