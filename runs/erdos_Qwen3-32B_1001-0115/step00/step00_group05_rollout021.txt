import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    desired_sum = n_points / 2
    assert abs(h.sum() - desired_sum) < 1e-6

    start_time = time.time()
    learning_rate = 0.2
    max_iterations = 4000

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break

        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = overlaps.max()
        max_idx = overlaps.argmax()
        k = max_idx - (n_points - 1)

        # Compute gradient
        grad = np.zeros(n_points)
        for j in range(n_points):
            term1 = 0.0
            j_plus_k = j + k
            if 0 <= j_plus_k < n_points:
                term1 = 1.0 - h[j_plus_k]
            term2 = 0.0
            j_minus_k = j - k
            if 0 <= j_minus_k < n_points:
                term2 = -h[j_minus_k]
            grad[j] = (term1 + term2) * dx

        h -= learning_rate * grad

        # Project to [0,1]
        np.clip(h, 0.0, 1.0, out=h)
        # Adjust sum
        current_sum = h.sum()
        if current_sum > 1e-9:
            scale = desired_sum / current_sum
            h *= scale
            np.clip(h, 0.0, 1.0, out=h)
        else:
            h.fill(desired_sum / n_points)

    overlaps = np.correlate(h, 1 - h, mode='full') * dx
    final_c5 = overlaps.max()
    return h.tolist(), final_c5, n_points
