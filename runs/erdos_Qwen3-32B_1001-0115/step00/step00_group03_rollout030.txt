    while time.time() - start_time < budget_s:
        for phase in ['pairwise', 'neighborhood', 'random']:
            if time.time() - start_time >= budget_s:
                break
            if phase == 'pairwise':
                pairs = [(i, j) for i in range(n_points) for j in range(i+1, n_points)]
                np.random.shuffle(pairs)
                for i, j in pairs:
                    if time.time() - start_time >= budget_s:

                    old_i, old_j = h[i], h[j]
                    h[i], h[j] = h[j], h[i]

                        best_c5 = new_c5

                    else:
                        h[i], h[j] = old_i, old_j
            elif phase == 'neighborhood':
                pairs = [(i, i+1) for i in range(n_points - 1)]
                np.random.shuffle(pairs)
                for i, j in pairs:
                    if time.time() - start_time >= budget_s:
                        break
                    old_i, old_j = h[i], h[j]
                    h[i], h[j] = h[j], h[i]
                    new_c5 = evaluate_erdos_solution(h)

                        best_c5 = new_c5
                        best_h = h.copy()

                        h[i], h[j] = old_i, old_j

                for _ in range(100):
