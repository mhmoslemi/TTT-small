def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time

    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points

    start_time = time.time()
    best_h = h.copy()
    best_c5 = float('inf')

    def evaluate_c5(h):
        n = len(h)
        c5_values = []
        for l in range(n):
            sum_val = 0.0
            for x in range(n - l):
                sum_val += h[x] * (1 - h[x + l])
            c5_val = sum_val * dx
            c5_values.append(c5_val)
        return max(c5_values)

    while time.time() - start_time < budget_s:
        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Find l with max c5
        max_c5 = -1.0
        problem_l = 0
        for l in range(n_points):
            sum_val = 0.0
            for x in range(n_points - l):
                sum_val += h[x] * (1 - h[x + l])
            c5_val = sum_val * dx
            if c5_val > max_c5:
                max_c5 = c5_val
                problem_l = l

        # Now find the x with maximum contribution for problem_l
        l = problem_l
        max_contribution = -1.0
        candidate_x = 0
        for x in range(n_points - l):
            contribution = h[x] * (1 - h[x + l])
            if contribution > max_contribution:
                max_contribution = contribution
                candidate_x = x

        # Adjust h at x and x + l
        x = candidate_x
        delta = min(h[x], 1 - h[x + l])
        if delta > 1e-8:
            step = 0.01 * delta  # small step
            h[x] -= step
            h[x + l] += step

        # Clamp to [0,1] in case of numerical errors
        h = np.clip(h, 0, 1)

    # Final evaluation
    current_c5 = evaluate_c5(h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
