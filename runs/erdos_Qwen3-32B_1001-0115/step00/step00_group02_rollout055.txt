import numpy as np
import time
import math

def adjust_and_clip(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    current_sum = np.sum(h)
    delta = target_sum - current_sum
    if abs(delta) < 1e-10:
        return h
    in_bounds = np.isclose(h, 0.0, atol=1e-10) == False
    in_bounds &= np.isclose(h, 1.0, atol=1e-10) == False
    if np.any(in_bounds):
        n = in_bounds.sum()
        h[in_bounds] += delta / n
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100):

    T = 1.0
    alpha = 0.999
    max_iter = int(1e6)
    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        i, j = np.random.choice(n_points, 2, replace=False)
        d = np.random.uniform(-0.05, 0.05)
        h_new = h.copy()
        h_new[i] += d
        h_new[j] -= d
        h_new = adjust_and_clip(h_new, target_sum)
        
        c5_new = evaluate_erdos_solution(h_new)
        c5_old = evaluate_erdos_solution(h)
        
        if c5_new < c5_old:

            c5_old = c5_new

            delta_energy = (best_c5 - c5_new)
