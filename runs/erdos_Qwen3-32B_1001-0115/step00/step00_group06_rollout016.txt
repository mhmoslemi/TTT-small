    best_h = h.copy()
    best_c5 = None

    while time.time() - start_time < budget_s:
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        overlaps_non_negative = overlaps[n_points - 1:]
        current_c5 = overlaps_non_negative.max()
        if best_c5 is None or current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        max_idx = overlaps_non_negative.argmax()
        k = max_idx

        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k
            if j < n_points:
                grad[i] += (1 - h[j]) * dx
                grad[j] -= h[i] * dx

        alpha_step = alpha
        while True:
            new_h = h - alpha_step * grad
            new_h_proj = project_h(new_h)
            overlaps_new = np.correlate(new_h_proj, 1 - new_h_proj, mode='full') * dx
            overlaps_new_non_negative = overlaps_new[n_points - 1:]
            new_c5 = overlaps_new_non_negative.max()
            if new_c5 <= current_c5 - c * alpha_step * np.dot(grad, grad) or alpha_step < 1e-8:
                break
            alpha_step *= beta
        h = new_h_proj

    overlaps = np.correlate(best_h, 1 - best_h, mode='full') * dx
    overlaps_non_negative = overlaps[n_points - 1:]
    c5_bound = overlaps_non_negative.max()
    return best_h, c5_bound, n_points
