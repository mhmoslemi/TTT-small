    n_points = 200  # Adjusted discretization for better approximation

    # Interpolate initial_h to higher resolution
    x_initial = np.linspace(0, 2, n_initial)
    x_new = np.linspace(0, 2, n_points)
    h = np.interp(x_new, x_initial, initial_h)
    
    # Normalize to meet integral constraint
    desired_sum = n_points / 2
    h *= desired_sum / np.sum(h)
    h = np.clip(h, 0, 1)

    dx = 2.0 / n_points
    alpha = 0.05
    epsilon = 0.1
    max_iterations = 500
    perturb = 1e-4
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 5:
            break
        
        # Compute approximate objective using smooth max
        def compute_obj(h_):
            overlaps = np.correlate(h_, 1 - h_, mode='full') * dx
            overlaps_max = np.max(overlaps)
            return epsilon * (np.log(np.sum(np.exp((overlaps - overlaps_max) / epsilon))) + overlaps_max / epsilon)
        
        # Finite difference gradient
        grad = np.zeros_like(h)
        h_flat = h.copy()
        for i in range(n_points):
            h_flat[i] += perturb
            obj_plus = compute_obj(h_flat)
            h_flat[i] -= 2 * perturb
            obj_minus = compute_obj(h_flat)
            h_flat[i] += perturb  # restore
            grad[i] = (obj_plus - obj_minus) / (2 * perturb)
        
        h -= alpha * grad
        
        # Projection onto feasible region
        h = np.clip(h, 0, 1)
        curr_sum = np.sum(h)
        if abs(curr_sum - desired_sum) > 1e-8:
            h *= desired_sum / curr_sum
            h = np.clip(h, 0, 1)
        
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
