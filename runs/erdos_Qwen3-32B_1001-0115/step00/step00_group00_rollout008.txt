def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()

    start_time = time.time()
    alpha = 0.05
    momentum = 0.9
    velocity = np.zeros_like(h)

    while time.time() - start_time < budget_s - 1:
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_idx = np.argmax(overlaps)
        current_c5 = overlaps[max_idx]

        if current_c5 < best_c5:

            best_h = h.copy()
