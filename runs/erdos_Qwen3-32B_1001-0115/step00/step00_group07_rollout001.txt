                    vec[underflow] = 0.0
                    remaining -= np.sum(np.abs(deficit))
                    in_range = idx[vec > 1e-12]
                    step_size = remaining / in_range.size if in_range.size > 0 else 0
                else:
                    return vec.copy()
            return vec.copy()
    
    def objective(vec):
        vec = project(vec)
        cc = np.correlate(vec, 1 - vec, mode='full') * dx
        return np.max(cc)
    
    def gradient(vec):
        vec = project(vec)
        cc = np.correlate(vec, 1 - vec, mode='full')
        max_idx = np.argmax(cc)
        k = max_idx - (n_points - 1)
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Compute contribution from h[i]*(1-h[i+k])
            term1 = 0.0
            if 0 <= i + k < n_points:
                term1 = 1 - vec[i + k]
            # Compute contribution from -(h[i-k]) * h[i]
            term2 = 0.0
            if 0 <= i - k < n_points:
                term2 = -vec[i - k]
            grad[i] = (term1 + term2) * dx
        return grad
    
    start_time = time.time()
    
    # Optimization settings
    bounds = [(0.0, 1.0)] * n_points
    result = minimize(
        fun=objective,
        x0=h_current,
        method='L-BFGS-B',
        jac=gradient,
        bounds=bounds,
        options={
            'maxiter': 1000,
            'ftol': 1e-12,
            'gtol': 1e-10
        }
    )
    
    elapsed = time.time() - start_time
    
    # Update best solution
    h_result = result.x.copy()
    c5_result = objective(project(h_result))
    
    # Final projection and re-evaluation
    h_result = project(h_result)
    c5_result = objective(h_result)
    
    return (h_result, c5_result, n_points)
