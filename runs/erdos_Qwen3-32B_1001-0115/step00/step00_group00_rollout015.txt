        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 200, 'ftol': 1e-6}
    )
    if result.success:
        curr_h = result.x
        curr_c5 = result.fun
        if curr_c5 < best_c5:
            best_h = curr_h.copy()
            best_c5 = curr_c5

    if time.time() - start_time < budget_s * 0.7:
        perturbed = h_initial.copy()
        noise = np.random.normal(0, 0.1, n_points)
        perturbed += noise
        perturbed = np.clip(perturbed, 0, 1)
        if np.abs(np.sum(perturbed) - required_sum) > 1e-6:
            perturbed[0] += (required_sum - np.sum(perturbed))
            perturbed[0] = np.clip(perturbed[0], 0, 1)
        result = opt.minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 200, 'ftol': 1e-6}
        )
        if result.success and result.fun < best_c5:
            best_h = result.x.copy()
            best_c5 = result.fun

    best_h = np.clip(best_h, 0, 1)
    if not np.isclose(np.sum(best_h), required_sum, atol=1e-6):
        diff = required_sum - np.sum(best_h)
        best_h[0] += diff
        best_h[0] = np.clip(best_h[0], 0, 1)

    final_c5 = objective(best_h)
    return best_h.tolist(), final_c5, n_points
