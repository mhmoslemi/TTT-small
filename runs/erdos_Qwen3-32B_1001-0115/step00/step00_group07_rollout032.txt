import numpy as np
import scipy.optimize
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    h_padded = np.pad(h, (0, len(h)), mode='constant')
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def generate_feasible_h(n_points):
    h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum == 0:
        return np.zeros(n_points)
    scale_factor = (n_points / 2) / current_sum
    h = h * scale_factor
    return np.clip(h, 0, 1)

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum == 0:
        return h
    scale_factor = (n_points / 2) / current_sum
    h = h * scale_factor
    return np.clip(h, 0, 1)

def crossover(parent1, parent2):
    alpha = np.random.rand(len(parent1))
    return np.clip(alpha * parent1 + (1 - alpha) * parent2, 0, 1)

def mutate(h, mutation_rate=0.1, noise_scale=0.05):
    h = h.copy()
    perturbation = np.random.uniform(-noise_scale, noise_scale, len(h))
    for i in range(len(h)):
        if np.random.rand() < mutation_rate:
            h[i] += perturbation[i]
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    timeout = budget_s

    initial_n = len(initial_h_values)
    n_points = initial_n
    best_h = project_to_feasible(initial_h_values, n_points)
    best_c5 = evaluate_erdos_solution(best_h)

    population_size = 50
    gens_ga = 30
    mut_rate = 0.1
    pop = [best_h]
    for _ in range(population_size - 1):
        pop.append(generate_feasible_h(n_points))

    for gen in range(gens_ga):
        if time.time() - start_time > timeout - 300:
            break

        fitnesses = []
        for ind in pop:
            c5 = evaluate_erdos_solution(ind)
            fitnesses.append(c5)
            if c5 < best_c5:

        selected = [pop[i] for i in top_idx]

            child = project_to_feasible(child, n_points)
