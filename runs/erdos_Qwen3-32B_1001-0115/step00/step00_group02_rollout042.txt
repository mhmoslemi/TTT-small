import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_c5 = None
    best_h = h.copy()
    alpha = 0.1

    while time.time() - start_time < budget_s:
        # Compute current max overlap and corresponding k
        cross_corr = np.correlate(h, 1 - h, mode='full') * dx
        worst_k_index = int(np.argmax(cross_corr))
        k_actual = worst_k_index - (n_points - 1)
        gradient = np.zeros(n_points, dtype=np.float64)

        if k_actual == 0:
            gradient = (1 - 2 * h) * dx
        elif k_actual > 0:
            max_i = n_points - k_actual
            indices_i = np.arange(max_i)
            indices_j = indices_i + k_actual
            grad_i_part = (1 - h[indices_j]) * dx
            grad_j_part = -h[indices_i] * dx
            gradient[indices_i] += grad_i_part
            gradient[indices_j] += grad_j_part
        else:
            m = -k_actual
            indices_i = np.arange(m, n_points)
            indices_j = indices_i - m
            grad_i_part = (1 - h[indices_j]) * dx
            grad_j_part = -h[indices_i] * dx
            gradient[indices_i] += grad_i_part
            gradient[indices_j] += grad_j_part

        # Gradient descent step
        h -= alpha * gradient

        # Projection: clip to [0, 1]
        h = np.clip(h, 0.0, 1.0)

        # Enforce integral constraint
        current_sum = h.sum()
        if abs(current_sum - target_sum) > 1e-6:
            diff = (target_sum - current_sum) / n_points
            h += diff
            h = np.clip(h, 0.0, 1.0)

        # Evaluate and update best solution
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)
        if best_c5 is None or current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    return best_h.tolist(), best_c5, n_points
