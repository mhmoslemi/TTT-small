import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    S = n_points / 2.0

    n_particles = 50
    max_iter = 1000
    c1 = 2.0
    c2 = 2.0
    omega = 0.7
    omega_min = 0.4
    v_max = 0.1

    # Initialize particles
    particles = [initial_h_values.copy() for _ in range(n_particles)]
    for p in particles:
        noise = np.random.uniform(-0.05, 0.05, size=n_points)
        p += noise
        p = np.clip(p, 0, 1)
        # Rescale to meet sum constraint
        for _ in range(3):
            sum_p = np.sum(p)
            if abs(sum_p - S) < 1e-8:
                break
            if sum_p != 0:
                scaling = S / sum_p
                p = p * scaling
                p = np.clip(p, 0, 1)
            else:
                p = np.full_like(p, 0.5)

    velocities = [np.random.uniform(-v_max, v_max, size=n_points) for _ in range(n_particles)]

    pbest = [p.copy() for p in particles]
    pbest_c5 = [evaluate_erdos_solution(p)['c5_bound'] for p in particles]
    idx = np.argmin(pbest_c5)
    gbest = particles[idx].copy()
    gbest_c5 = pbest_c5[idx]

    start_time = time.time()
    elapsed_time = 0
    iter_count = 0

    while elapsed_time < budget_s and iter_count < max_iter:
        iter_count += 1
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = omega * v + c1 * r1 * (pbest[i] - h) + c2 * r2 * (gbest - h)
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            h_new = h + velocities[i]
            h_new = np.clip(h_new, 0, 1)
            # Projection to feasible set
            for _ in range(3):
                sum_h = np.sum(h_new)
                if abs(sum_h - S) < 1e-8:
                    break
                if sum_h != 0:
                    scale = S / sum_h
                    h_new = h_new * scale
                    h_new = np.clip(h_new, 0, 1)
                else:
                    h_new = np.full_like(h_new, 0.5)
            particles[i] = h_new

            c5 = evaluate_erdos_solution(h_new)['c5_bound']
            if c5 < pbest_c5[i]:
                pbest[i] = h_new.copy()
                pbest_c5[i] = c5

        # Update global best
        idx = np.argmin(pbest_c5)
        if pbest_c5[idx] < gbest_c5:
            gbest = pbest[idx].copy()
            gbest_c5 = pbest_c5[idx]

        # Inertia weight decay
        omega = 0.7 - (0.7 - omega_min) * (iter_count / max_iter)
        elapsed_time = time.time() - start_time

    return (gbest, gbest_c5, n_points)
