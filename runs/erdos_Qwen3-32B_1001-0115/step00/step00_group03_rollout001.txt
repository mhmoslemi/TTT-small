import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    np.random.seed(seed)
    best_h = initial_h.copy()
    dx = 2.0 / n_points
    
    # Evaluate initial C5
    def compute_c5(h):
        overlap = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)
    
    best_c5 = compute_c5(best_h)
    current_h = best_h.copy()
    start_time = time.time()
    max_iter = 1_000_000  # High upper bound; actual iterations depend on time budget

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Compute overlaps and find max shift
        overlap = np.correlate(current_h, 1 - current_h, mode='full') * dx
        max_idx = np.argmax(overlap)
        k_shift = max_idx - (n_points - 1)
        
        # Compute gradient for the worst shift
        grad = np.zeros(n_points)
        for i in range(n_points):
            shifted_idx = i + k_shift
            term1 = (1 - current_h[shifted_idx]) * dx if 0 <= shifted_idx < n_points else 0.0
            i_new = i - k_shift
            term2 = -current_h[i_new] * dx if 0 <= i_new < n_points else 0.0
            grad[i] = term1 + term2
        
        # Gradient descent step with adaptive learning rate
        step_size = 0.01
        current_h -= step_size * grad
        current_h = np.clip(current_h, 0, 1)
        
        # Enforce the integral constraint by scaling
        current_sum = np.sum(current_h)
        target_sum = n_points / 2.0
        if abs(current_sum - target_sum) > 1e-6:
            scale = target_sum / current_sum
            current_h *= scale
            current_h = np.clip(current_h, 0, 1)
        
        # Update best solution
        current_c5 = compute_c5(current_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h[:] = current_h[:]  # Ensure a deep copy
        
    return (best_h, best_c5, n_points)
