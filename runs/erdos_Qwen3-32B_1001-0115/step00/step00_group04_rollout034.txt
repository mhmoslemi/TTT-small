        h_vals = best_h_for_res
        one_minus_h_vals = 1.0 - h_vals
        corrs = np.correlate(h_vals, one_minus_h_vals, mode='full')
        positive_corrs = corrs[-n_points:]
        max_corr = np.max(positive_corrs)
        best_c5_for_res = max_corr * dx

        learning_rate = 0.1
        iterations = 300 if n_points <= 128 else 150
        for iter in range(iterations):
            if time.time() - start_time > budget_s * 0.99:
                break
            h_vals = h
            one_minus_h_vals = 1.0 - h_vals
            corrs = np.correlate(h_vals, one_minus_h_vals, mode='full')
            positive_corrs = corrs[-n_points:]
            k_corr = np.argmax(positive_corrs)
            shift_k = k_corr

            gradient = np.zeros_like(h_vals)
            for i in range(n_points - shift_k):
                gradient[i] = (1.0 - h_vals[i + shift_k]) * dx

            h = h - learning_rate * gradient
            h = np.clip(h, 0.0, 1.0)
            current_sum = h.sum()
            target_sum = n_points / 2.0
            if current_sum != 0.0:
                h *= target_sum / current_sum
                h = np.clip(h, 0.0, 1.0)

            h_vals = h
            one_minus_h_vals = 1.0 - h_vals
            corrs = np.correlate(h_vals, one_minus_h_vals, mode='full')
            positive_corrs = corrs[-n_points:]
            max_corr = np.max(positive_corrs)
            current_c5 = max_corr * dx
            if current_c5 < best_c5_for_res:
                best_c5_for_res = current_c5
                best_h_for_res = h.copy()
            if iter % 25 == 0:
                learning_rate *= 0.5

        if best_c5_for_res < best_c5:
            best_c5 = best_c5_for_res
            best_h = best_h_for_res
            best_n_points = n_points

        if res < resolutions[-1]:
            next_res = resolutions[resolutions.index(res) + 1]
            x_current = np.linspace(0., 2., n_points)
            x_next = np.linspace(0., 2., next_res)
            interpolator = interp1d(x_current, best_h_for_res, kind='cubic', fill_value='extrapolate')
            h_upsampled = interpolator(x_next)
            h_upsampled = np.clip(h_upsampled, 0.0, 1.0)
            current_sum = h_upsampled.sum()
            target_sum = next_res / 2.0
            if current_sum != 0.0:
                h_upsampled *= target_sum / current_sum
                h_upsampled = np.clip(h_upsampled, 0.0, 1.0)
            current_h = h_upsampled

    h_vals = best_h
    one_minus_h_vals = 1.0 - h_vals
    corrs = np.correlate(h_vals, one_minus_h_vals, mode='full')
    n = best_n_points
    positive_corrs = corrs[-n:]
    max_corr = np.max(positive_corrs)
    final_c5 = max_corr * (2.0 / n)
    return (best_h, final_c5, best_n_points)
