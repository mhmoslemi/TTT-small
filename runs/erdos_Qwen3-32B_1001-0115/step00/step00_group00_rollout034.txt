    # Define constraint: sum(h) = n_points / 2
    sum_constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    
    # Define bounds: 0 <= h[i] <= 1
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Ensure initial guess is valid
    h_initial = h_initial.astype(float)
    h_initial *= (n_points / 2) / np.sum(h_initial)  # Normalize if needed
    
    # Time management
    start_time = time.time()
    deadline = start_time + budget_s * 0.9
    
    # Run optimization
    result = minimize(
        fun=objective,
        x0=h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=sum_constraint,
        options={
            'maxiter': 200,
            'ftol': 1e-6,
            'disp': False
        }
    )
    
    # Finish early if needed
    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.3:
