def compute_gradient(h, shift):
    n = len(h)
    grad = np.zeros(n)
    shift_int = int(shift)
    if shift_int > 0:
        if shift_int >= n:
            # invalid, no overlap
            return grad
        grad[:n - shift_int] = 1 - h[shift_int:]
    elif shift_int < 0:

            # invalid, no overlap
            return grad
        grad[shift_int_abs:] = 1 - h[:-shift_int_abs]
    else:
