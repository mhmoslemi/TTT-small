import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    
    try:
        initial_h = initial_h_values.copy()
        n_points = len(initial_h)
    except NameError:
        n_points = 45
        target_sum = n_points / 2
        integer_part = int(target_sum)
        fraction = target_sum - integer_part
        initial_h = np.zeros(n_points)
        initial_h[:integer_part] = 1.0
        if fraction > 1e-10:
            initial_h[integer_part] = fraction

    def objective(h):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    constraints = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    bounds = [(0, 1)] * n_points
    options = {'maxiter': 5000, 'ftol': 1e-8, 'disp': False}

    result = minimize(objective, initial_h, method='SLSQP', bounds=bounds, constraints=constraints, options=options)

    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        raise TimeoutError("Budget exceeded during optimization")

    h_opt = result.x
    dx = 2.0 / n_points
    corr = np.correlate(h_opt, 1 - h_opt, mode='full') * dx
    c5_bound = np.max(corr)

    return h_opt, c5_bound, n_points
