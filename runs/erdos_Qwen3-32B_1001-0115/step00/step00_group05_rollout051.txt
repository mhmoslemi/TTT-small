            if k >= 0:
                valid_indices = np.arange(n_points - k)
            else:
                k_neg = -k
                valid_indices = np.arange(k_neg, n_points)
            i = valid_indices
            j = i + k
            grad[i] += 1.0 - h[j]
            grad[j] += -h[i]
        
        h_update = h - learning_rate * grad
        h_update = np.clip(h_update, 0, 1)
        
        h_proj = cp.Variable(n_points)
        obj = cp.Minimize(cp.sum_squares(h_proj - h_update))
        constraints = [cp.sum(h_proj) == required_sum, 0 <= h_proj, h_proj <= 1]
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.ECOS)
        
        if prob.status != cp.OPTIMAL:
            continue  # Skip if projection fails
        
        h = h_proj.value.copy()
        
        current_c5_new = evaluate_erdos_solution(h)
        if current_c5_new < best_c5:
            best_h = h.copy()
            best_c5 = current_c5_new
        
        learning_rate *= decay_rate
        if iter > 100 and best_c5 - current_c5_new < 1e-6:
            break
    
    return (best_h, best_c5, n_points)
