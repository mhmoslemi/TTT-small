import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum_h = n_points / 2.0
    learning_rate = 0.01
    max_iterations = 10000
    best_h = np.copy(h)
    best_c5 = np.inf
    start_time = time.time()

    for iter in range(max_iterations):
        if time.time() - start_time >= budget_s:
            break
            
        # Compute overlaps
        ones_minus_h = 1.0 - h
        overlaps = np.correlate(h, ones_minus_h, mode='full') * dx
        c5_candidate = np.max(overlaps)
        
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = np.copy(h)
        
        # Identify indices of current maximum overlaps
        max_mask = (np.abs(overlaps - c5_candidate) < 1e-8)
        max_indices = np.where(max_mask)[0]
        
        if len(max_indices) == 0:
            continue
        
        # Compute aggregated gradient
        gradients = []
        n = len(h)
        for m in max_indices:
            k = m - (n - 1)
            h_shifted = np.roll(h, -k)

            grad_k = dx * (1.0 - h_shifted - h_rolled)
            gradients.append(grad_k)
