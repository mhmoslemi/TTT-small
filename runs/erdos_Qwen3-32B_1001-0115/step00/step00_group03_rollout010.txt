            if time.time() - start_time >= max_time:
                break
            i, j = np.random.randint(0, n_points, 2)
            if i != j:
                a, b = h[i], h[j]
                h[i], h[j] = b, a
                new_c5 = evaluate_erdos_solution(h)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h[:] = h[:]
                else:
                    h[i], h[j] = a, b

        if time.time() - start_time >= max_time:
            break

    return best_h.tolist(), best_c5, n_points
