            break
        
        num_elite = int(population_size * elitism_rate)
        elite_indices = np.argsort(fitness)[:num_elite]
        elite = [population[i] for i in elite_indices]
        
        new_population = elite.copy()
        
        while len(new_population) < population_size:
            parent1 = None
            parent2 = None
            while parent1 is None:
                idxs = np.random.choice(len(population), tournament_size, replace=False)
                min_idx = idxs[np.argmin([fitness[i] for i in idxs])]
                parent1 = population[min_idx]
            while parent2 is None:
                idxs = np.random.choice(len(population), tournament_size, replace=False)
                min_idx = idxs[np.argmin([fitness[i] for i in idxs])]
                parent2 = population[min_idx]
            
            child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
            
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, mutation_strength, child.shape)
                child += noise
                child = np.clip(child, 0, 1.0)
                sum_h = np.sum(child)
                if sum_h == 0:
                    child += 1e-8
                    sum_h = np.sum(child)
                target_sum = n_points / 2.0
                child *= target_sum / sum_h
            
            new_population.append(child)
        
        population = new_population[:population_size]
        
        if time.time() - last_improvement_time > 60:  # 60s patience
            break
    
    return (best_h.tolist(), best_c5, n_points)
