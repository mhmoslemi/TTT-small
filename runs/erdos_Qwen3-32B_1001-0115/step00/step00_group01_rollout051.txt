import numpy as np

import copy

def evaluate_c5(h, n_points, dx):
    h = np.array(h)
    max_c5 = 0.0
    for k in range(n_points):
        shifted = np.roll(h, -k)
        overlap = np.sum(h * (1 - shifted))
        c5_val = overlap * dx
        if c5_val > max_c5:
            max_c5 = c5_val
    return max_c5

def evaluate_c5(h, n_points, dx):
    h = np.array(h)
    max_c5 = 0.0
    for k in range(n_points):
        shifted = np.roll(h, -k)
        overlap = np.sum(h * (1 - shifted))
        c5_val = overlap * dx
        if c5_val > max_c5:
            max_c5 = c5_val
    return max_c5

def perturb_with_move(current_h, n_points):

        i = np.random.randint(n_points)
        j = np.random.randint(n_points)
        if i != j:
            break
    current_i = current_h[i]
    current_j = current_h[j]
    max_positive = min(1.0 - current_i, current_j)
    max_negative = min(current_i, 1.0 - current_j)

        return current_h.copy(), (i, j)

    new_h = current_h.copy()
