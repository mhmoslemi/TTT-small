import numpy as np
import time

def perturb_h(h, delta_size=0.05):
    h = h.copy()
    n = len(h)
    i, j = np.random.choice(n, 2, replace=False)
    max_move = min(h[i], 1 - h[j])
    min_move = -min(1 - h[i], h[j])
    delta = (2 * np.random.rand() - 1) * delta_size
    delta = np.clip(delta, min_move, max_move)
    h[i] += delta
    h[j] -= delta
    return h

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    current_sum = np.sum(h)
    if not np.isclose(current_sum, target_sum, atol=1e-6):
        adjustment = (target_sum - current_sum) / n_points
        h += adjustment
        h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5, _ = evaluate_erdos_solution(h)
    
    T = 1.0
    T_min = 1e-4
    alpha = 0.99
    delta_size = 0.05
    
    while time.time() - start_time < budget_s:
        h_new = perturb_h(h, delta_size)
        try:
            current_c5, _ = evaluate_erdos_solution(h_new)
        except:

        delta_c5 = current_c5 - best_c5
        
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / max(T, 1e-8)):
            h = h_new.copy()
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
        
        T *= alpha
        T = max(T, T_min)
