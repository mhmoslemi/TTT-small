import numpy as np
import time
import random

def run(seed=42, budget_s=1100, **kwargs):
    import evaluate_erdos_solution
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    initial_h_values = globals().get('initial_h_values', None)
    if initial_h_values is not None:
        h = np.array(initial_h_values, copy=True)
        n_points = len(h)
    else:
        n_points = 70
        count = n_points // 2
        h = np.array([1.0] * count + [0.0] * (n_points - count), dtype=np.float64)
        np.random.shuffle(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution.evaluate_erdos_solution(best_h)
    
    def compute_phase(phase, swaps):
        nonlocal best_h, best_c5
        current_h = best_h.copy()
        for i, j in swaps:
            if time.time() - start_time >= budget_s:
                return
            # Apply swap
            new_h = current_h.copy()
            new_h[i] = 1 - new_h[i]
            new_h[j] = 1 - new_h[j]
            new_c5 = evaluate_erdos_solution.evaluate_erdos_solution(new_h)
            if new_c5 < best_c5:
                current_h[:] = new_h
                best_c5 = new_c5
                best_h[:] = current_h
    
    # Generate swap indices
    i_indices, j_indices = np.triu_indices(n_points, 1)
    all_swaps = list(zip(i_indices, j_indices))
    neighborhood_swaps = [(i, i+1) for i in range(n_points - 1)]
    
    # Phase-wise optimization
    for _ in range(3):  # Repeat phases to avoid local minima
        if time.time() - start_time >= budget_s:
            break
        
        # Pairwise greedy swaps
        random.shuffle(all_swaps)
        compute_phase("pairwise", all_swaps[:2000])  # Limit swap attempts
        
        # Neighborhood swaps for finer adjustments
        random.shuffle(neighborhood_swaps)
        compute_phase("neighbor", neighborhood_swaps)
        
        # Random uniform swaps
        random_swaps = [(random.randint(0, n_points-1), random.randint(0, n_points-1)) for _ in range(200)]
        random_swaps = [(i, j) for i, j in random_swaps if i != j]
        compute_phase("random", random_swaps)
    
    return best_h.tolist(), best_c5, n_points
