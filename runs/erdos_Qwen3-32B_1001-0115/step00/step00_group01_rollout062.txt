    best_C5 = current_C5
    T = 1.0
    alpha = 0.99
    no_improvement = 0
    
    while time.time() - start_time < budget_s:
        new_h = perturb_h(current_h, S)
        new_C5 = compute_C5(new_h, n_points)
        delta = new_C5 - current_C5
        
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h[:] = new_h[:]
            current_C5 = new_C5
            if new_C5 < best_C5:
                best_h[:] = current_h[:]
                best_C5 = new_C5
                no_improvement = 0
            else:
                no_improvement += 1
        else:
            no_improvement += 1
        
        T *= alpha
        if no_improvement >= 100:
            break
    
    return (best_h.tolist(), best_C5, n_points)
