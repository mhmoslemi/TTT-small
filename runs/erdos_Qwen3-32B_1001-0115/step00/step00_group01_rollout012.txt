    current_h = h.copy()
    current_c5 = best_c5
    T = 1.0
    alpha = 0.995
    start_time = time.time()
    improve_iter = 0

    while time.time() - start_time < budget_s:
        # Perturb and normalize
        candidate_h = perturb_h(current_h)
        # Compute new c5
        new_c5 = compute_C5(candidate_h, 2.0 / n_points)
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0 or np.exp(-delta_c5 / T) > np.random.rand():
            current_h[:] = candidate_h[:]
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h[:] = current_h[:]
                best_c5 = new_c5
                improve_iter = 0
            else:
                improve_iter += 1
        else:
            improve_iter += 1
        T *= alpha
        if improve_iter > 150:
            break
    return (best_h.tolist(), best_c5, n_points)
