        h_initial = h_initial * (n_points / 2.0 / h_initial_sum)
    h_initial = np.clip(h_initial, 0, 1)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0

    bounds = [(0, 1)] * n_points
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    result = minimize(objective, x0=h_initial, method='SLSQP', bounds=bounds, constraints=constraints,
                      options={'maxiter': 100, 'ftol': 1e-6, 'disp': False})

    optimized_h = result.x
    optimized_h = np.clip(optimized_h, 0, 1)
    optimized_sum = np.sum(optimized_h)
    if optimized_sum != 0:
        optimized_h = optimized_h * (n_points / 2.0 / optimized_sum)

    final_c5 = objective(optimized_h)
    elapsed = time.time() - start_time

    if elapsed > budget_s:
        optimized_h = h_initial.copy()
        final_c5 = objective(optimized_h)

    return (optimized_h.tolist(), final_c5, n_points)
