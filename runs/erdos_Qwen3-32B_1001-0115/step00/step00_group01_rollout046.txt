import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return np.max(c * dx)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    h_current = initial_h_values.copy()
    n_points = len(h_current)
    s_target = n_points / 2.0
    dx = 2.0 / n_points
    
    assert np.allclose(np.sum(h_current), s_target, atol=1e-8), "Invalid initial h"
    
    h_best = h_current.copy()
    c5_best = compute_c5(h_best)
    
    T = 1.0
    alpha = 0.99
    sigma = 0.1
    no_improve = 0
    max_no_improve = 100
    
    while time.time() - start_time < budget_s:
        # Perturb
        h_new = h_current.copy()
        n_perturb = max(1, int(0.1 * n_points))
        indices = np.random.choice(n_points, size=n_perturb, replace=False)
        noise = np.random.normal(0, sigma, size=n_perturb)
        h_new[indices] += noise
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize sum
        sum_new = np.sum(h_new)
        if not np.isclose(sum_new, 0):
            scaling = s_target / sum_new
            h_new = h_new * scaling
        h_new = np.clip(h_new, 0, 1)
        
        # Compute new C5
        c5_new = compute_c5(h_new)
        
        # Acceptance
        c5_current = compute_c5(h_current)
        if c5_new < c5_current:
            h_current = h_new.copy()
            no_improve = 0
            if c5_new < c5_best:
                h_best = h_new.copy()
                c5_best = c5_new
        else:
            delta = c5_new - c5_current
            if np.random.rand() < np.exp(-delta / T):
                h_current = h_new.copy()
        
        T *= alpha
        no_improve += 1
        if no_improve > max_no_improve:
            break
    
    return (h_best.tolist(), c5_best, n_points)
