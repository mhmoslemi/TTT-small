def run(seed=42, budget_s=1100, **kwargs):

        t += 1
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        exp_vals = np.exp(overlaps / epsilon_smooth)
        sum_exp = np.sum(exp_vals)
        if sum_exp == 0:
