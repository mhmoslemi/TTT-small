import numpy as np
import time

def run(seed=42, budget_s=1100):
    n_points = 70
    h0 = initial_h_values.copy()
    dx = 2.0 / n_points

    # PSO parameters
    n_particles = 50
    max_vel = 0.5
    c1 = 2.0
    c2 = 2.0
    w_max, w_min = 0.7, 0.4

    np.random.seed(seed)
    particles = []
    velocities = []
    p_best_positions = []
    p_best_scores = []

    # Initialize particles with perturbations around h0
    for _ in range(n_particles):
        perturbation = np.random.uniform(-0.05, 0.05, n_points)
        h = np.clip(h0 + perturbation, 0, 1)
        # Enforce constraint sum(h) = n_points/2
        s = h.sum()
        h = h * (n_points/2.0 / s)
        h = np.clip(h, 0, 1)
        particles.append(h)
        velocities.append(np.random.uniform(-max_vel, max_vel, n_points))
        c5 = evaluate_erdos_solution(h)
        p_best_positions.append(h.copy())
        p_best_scores.append(c5)

    g_best_score = min(p_best_scores)
    g_best_pos = p_best_positions[p_best_scores.index(g_best_score)].copy()

    start_time = time.time()

    while time.time() - start_time < budget_s:
        for i in range(n_particles):
            # Update velocity
            w = w_max - (w_max - w_min) * (time.time() - start_time) / budget_s
            w = max(w_min, w)
            vel = velocities[i] * w
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            vel += c1 * r1 * (p_best_positions[i] - particles[i]) + c2 * r2 * (g_best_pos - particles[i])
            velocities[i] = np.clip(vel, -max_vel, max_vel)

            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            s = particles[i].sum()
            if s < 1e-10:  # Prevent division by near-zero
                particles[i] = 1e-10
                s = 1e-10
            particles[i] *= (n_points/2.0 / s)
            particles[i] = np.clip(particles[i], 0, 1)

            # Evaluate new fitness
            c5 = evaluate_erdos_solution(particles[i])
            if c5 < p_best_scores[i]:
                p_best_scores[i] = c5
                p_best_positions[i] = particles[i].copy()

        # Update global best
        idx = np.argmin(p_best_scores)
        if p_best_scores[idx] < g_best_score:
            g_best_score = p_best_scores[idx]
            g_best_pos = p_best_positions[idx].copy()

    return g_best_pos, g_best_score, n_points
