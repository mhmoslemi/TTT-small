def perturb_h(h, epsilon):
    noise = np.random.normal(0, epsilon, len(h))
    h_perturbed = h + noise
    h_perturbed = np.clip(h_perturbed, 0, 1)
    return normalize_h(h_perturbed, len(h))

def normalize_h(h, n_points):
    total = np.sum(h)
    if np.isclose(total, 0):
        h_norm = h.copy()
    else:
        target_sum = n_points / 2.0
        h_norm = h * (target_sum / total)
        h_norm = np.clip(h_norm, 0, 1)
    return h_norm

def evaluate_C5(h, n_points):
    dx = 2.0 / n_points
    h_1 = 1 - h
    corr = np.correlate(h, h_1, mode='full')
    c5 = np.max(corr * dx)
    return c5

def optimize_step(h_start, n_points, maxiter=20):

        x0=h_start,
        bounds=bounds,
        constraints=[sum_constraint],
        method='SLSQP',
