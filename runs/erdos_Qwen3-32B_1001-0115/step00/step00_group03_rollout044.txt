import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    h_initial = np.array(initial_h_values).copy()

    n_particles = 50

        s = positions[i].sum()

        p_best_c5[i] = c5
        if c5 < g_best_c5:
            g_best_c5 = c5

    start_time = time.time()
