    def objective(h_flat):
        h = h_flat.reshape(n_points)
        correlation = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(correlation)

    def constraint_sum(h):
        return np.sum(h) - target_sum

    def project_feasible(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        delta = target_sum - current_sum
        if abs(delta) < 1e-8:
            return h
        h += delta / n_points
        h = np.clip(h, 0, 1)
        return h

    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0.0, 1.0)] * n_points

    start_time = time.time()
    try:
        result = minimize(
            objective,
            h0,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 1000, 'ftol': 1e-8, 'disp': False}
        )
        if result.success:
            h_opt = project_feasible(result.x)
        else:
            h_opt = project_feasible(h0)
    except:
        h_opt = project_feasible(h0)

    elapsed = time.time() - start_time
    remaining_time = 1100 - elapsed

    if remaining_time > 200:
        try:
            result = minimize(
                objective,
                h_opt,
                method='Nelder-Mead',
                bounds=bounds,
                options={'maxiter': 500, 'xatol': 1e-6, 'disp': False}
            )
            if result.success:
                h_opt = project_feasible(result.x)
        except:
            pass

    c5_bound = evaluate_erdos_solution(h_opt)
    return h_opt, c5_bound, n_points
