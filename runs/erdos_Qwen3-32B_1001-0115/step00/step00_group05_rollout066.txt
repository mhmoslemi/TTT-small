            iter_without_improvement = 0
        else:
            iter_without_improvement += 1
        
        # Adapt learning rate
        if iter_without_improvement > 50:
            eta = max(eta * 0.9, 1e-5)
    
    # Final validation
    assert abs(np.sum(best_h) - target_sum) < 1e-8
    return best_h, best_c5, n_points
