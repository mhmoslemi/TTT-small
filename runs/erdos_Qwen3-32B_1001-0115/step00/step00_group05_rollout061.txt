        n_points = 70
        h = np.zeros(n_points)
        indices = np.random.choice(n_points, size=35, replace=False)
        h[indices] = 1
    n_points = len(h)
    dx = 2.0 / n_points
    
    def compute_c5(h_vec, dx_val):
        return np.max(np.correlate(h_vec, 1 - h_vec, mode='full') * dx_val)
    
    best_h = h.copy()
    current_c5 = compute_c5(best_h, dx)
    current_h = h.copy()
    
    initial_temp = 1.0
    cooling_rate = 0.99
    current_temp = initial_temp
    
    while time.time() - start_time < budget_s:
        for _ in range(100):
            # Generate candidate by swapping 5 ones and 5 zeros
            ones = np.where(current_h == 1)[0]
            zeros = np.where(current_h == 0)[0]
            if len(ones) >= 5 and len(zeros) >= 5:
                swap_ones = np.random.choice(ones, 5, replace=False)
                swap_zeros = np.random.choice(zeros, 5, replace=False)
                candidate = current_h.copy()
                candidate[swap_ones] = 0
                candidate[swap_zeros] = 1
                c5_candidate = compute_c5(candidate, dx)
                delta = c5_candidate - current_c5
                if delta < 0 or np.random.rand() < np.exp(-delta / current_temp):
                    current_h = candidate.copy()
                    current_c5 = c5_candidate
                    if current_c5 < compute_c5(best_h, dx):
                        best_h = current_h.copy()
            if time.time() - start_time >= budget_s:
                break
        current_temp *= cooling_rate
    
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, n_points
