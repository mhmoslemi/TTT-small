            return h_proj
        while abs(delta) > 1e-8 and len(idx_subtract) > 0:
            num_subtract = len(idx_subtract)
            max_decrease = np.min(h_proj[idx_subtract])
            subtract_ = min(delta / num_subtract, max_decrease)
            h_proj[idx_subtract] -= subtract_
            delta -= subtract_ * num_subtract
            idx_subtract = np.where(h_proj > 0)[0]
    return h_proj

def run(seed=42, budget_s=1100, **kwargs):
    h = np.array(initial_h_values)
    n_points = len(h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    best_c5, _, _ = compute_overlap_C5(h, dx)
    best_h = h.copy()
    start_time = time()
    max_iter = 10000
    alpha = 0.1
    for iteration in range(max_iter):
        if time() - start_time > budget_s - 1:
            break
        c5_current, k_shift, _ = compute_overlap_C5(h, dx)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
        grad = compute_gradient(h, k_shift, dx)
        direction = -grad
        alpha_test = alpha
        h_new = h + alpha_test * direction
        h_new = project_feasible(h_new, target_sum)
        c5_new, _, _ = compute_overlap_C5(h_new, dx)
        backtracking_attempts = 0
        while c5_new >= c5_current and backtracking_attempts < 10:
            alpha_test *= 0.5
            h_new = h + alpha_test * direction
            h_new = project_feasible(h_new, target_sum)
            c5_new, _, _ = compute_overlap_C5(h_new, dx)
            backtracking_attempts += 1
        if backtracking_attempts < 10:
            h = h_new
            alpha = min(alpha * 1.2, 0.5) if c5_new < c5_current else max(alpha * 0.5, 1e-6)
    c5_bound, _, _ = compute_overlap_C5(best_h, dx)
    return best_h.tolist(), c5_bound, n_points
