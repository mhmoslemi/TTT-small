import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 45
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    alpha = 0.001  # Learning rate
    beta1 = 0.9  # Adam exponential avg factor
    beta2 = 0.999  # Adam RMS factor
    eps_adam = 1e-8  # Adam epsilon
    epsilon = 0.1  # Temperature for log-sum-exp
    m = np.zeros_like(h)  # Adam first moment
    v = np.zeros_like(h)  # Adam second moment
    t = 0
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    start_time = time.time()

    while time.time() - start_time < budget_s - 0.1:
        t += 1
        # Compute overlaps
        overlaps_raw = np.correlate(h, 1 - h, mode='full')
        overlaps_dx = overlaps_raw * dx
        sum_exp = np.sum(np.exp(overlaps_dx / epsilon))
        weights = np.exp(overlaps_dx / epsilon) / sum_exp

        grad_total = np.zeros_like(h)
        for k in range(-n_points + 1, n_points):
            m_index = k + (n_points - 1)
            w = weights[m_index]
            # Compute gradients for this k
            rolled_neg = np.roll(h, -k)
            rolled_pos = np.roll(h, k)
            grad_k = dx * (1 - rolled_neg - rolled_pos)
            grad_total += w * grad_k

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_total
        v = beta2 * v + (1 - beta2) * (grad_total ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h_update = h - alpha * m_hat / (np.sqrt(v_hat) + eps_adam)

        # Project to feasible region (clip and rescale)
        h_update = np.clip(h_update, 0.0, 1.0)
        current_sum = np.sum(h_update)
        if abs(current_sum) > 1e-10:
            h_update *= (n_points / 2) / current_sum
        h = h_update.copy()

        # Check for improvement and reduce learning rate
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if t % 500 == 0:
            alpha *= 0.5  # Decay learning rate

    return best_h, best_c5, n_points
