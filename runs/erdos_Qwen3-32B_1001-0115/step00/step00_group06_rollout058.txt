    h = initial_h_values.copy()

    S = n_points / 2
    dx = 2.0 / n_points

    start_time = time.time()

    while time.time() - start_time < budget_s:
        current_C5, correl = compute_C5(current_h)
        k_max_index = np.argmax(correl)

        if current_C5 < best_C5:
