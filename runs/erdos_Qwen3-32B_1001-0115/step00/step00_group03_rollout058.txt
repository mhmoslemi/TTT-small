import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    try:
        initial_h = initial_h_values
        n_points = len(initial_h_values)
    except NameError:
        n_points = 70
        initial_h = np.repeat(0.5, n_points)

    dx = 2.0 / n_points
    
    def compute_c5(h_vec):
        h = np.array(h_vec)
        one_h = 1 - h
        conv = np.correlate(h, one_h, mode='full')
        return np.max(conv) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    sum_constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}

    def optimize_once(x0):
        return scipy.optimize.minimize(
            compute_c5, 
            x0=x0, 
            method='SLSQP', 
            bounds=bounds,
            constraints=sum_constraint,
            options={'maxiter': 100, 'ftol': 1e-8}
        )

    best_h = initial_h.copy()
    best_c5 = compute_c5(initial_h)
    start_time = time.time()
    np.random.seed(seed)

    # Optimization with initial guess
    res = optimize_once(initial_h)
    if res.success:
        curr_c5 = compute_c5(res.x)
        if curr_c5 < best_c5:
            best_h = res.x
            best_c5 = curr_c5

    # Random restarts
    for _ in range(3):
        if time.time() - start_time > budget_s * 0.9:
            break
        target_sum = n_points / 2
        mask = np.zeros(n_points)
        n_ones = int(target_sum)
        mask[:n_ones] = 1.0
        if n_points % 2 == 1:
            mask[n_ones] = 1.0  # Added for odd n_points

        # Adjust for exact sum if odd
