import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals():

    else:
        n_points = 42

        h_initial = h_initial / h_initial.sum() * (n_points / 2)

    POP_SIZE = 50
    MUTATION_RATE = 0.1
    MUTATION_SIGMA = 0.1
    ELITE_RATIO = 0.2
    MAX_NO_IMPROVE = 50

    def evaluate(h):
        dx = 2.0 / n_points
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        return np.max(corr) * dx

    population = []
    if 'initial_h_values' in globals():
        population.append(h_initial)
        for _ in range(POP_SIZE - 1):
            h = h_initial.copy()
            h += np.random.normal(0, 0.1, n_points)
            h = np.clip(h, 0, 1)
            h = h / h.sum() * (n_points / 2)
            population.append(h)
    else:
        for _ in range(POP_SIZE):
            h = np.random.rand(n_points)
            h = h / h.sum() * (n_points / 2)
            population.append(h)

    best_h = np.array(population)
    best_c5 = evaluate(best_h)
    best_idx = 0
    no_improve_count = 0

    while time.time() - start_time < budget_s - 0.1 and no_improve_count < MAX_NO_IMPROVE:
        fitnesses = [evaluate(h) for h in population]
        sorted_indices = np.argsort(fitnesses)
        elites = [population[i].copy() for i in sorted_indices[:int(POP_SIZE * ELITE_RATIO)]]
        new_population = elites.copy()

        while len(new_population) < POP_SIZE:
            parent_indices = np.random.choice(sorted_indices[:POP_SIZE // 2], size=2)
            parent1, parent2 = population[parent_indices[0]], population[parent_indices[1]]
            child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
            child = child / child.sum() * (n_points / 2)
            if np.random.rand() < 0.1:
                mask = np.random.rand(n_points) < MUTATION_RATE
                child = child.copy()
                child[mask] += np.random.normal(0, MUTATION_SIGMA, size=np.sum(mask))
                child = np.clip(child, 0, 1)
                child = child / child.sum() * (n_points / 2)
            new_population.append(child)

        population = new_population
        current_best = min(fitnesses)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[sorted_indices[0]].copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

    return (best_h.tolist(), best_c5, n_points)
