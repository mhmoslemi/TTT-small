import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    # Use initial_h_values if available, otherwise create a random valid h
    try:
        initial_h = initial_h_values.copy()
        n_points = len(initial_h)
    except NameError:
        n_points = 70
        initial_h = np.zeros(n_points)
        indices = np.random.choice(n_points, size=n_points // 2, replace=False)
        initial_h[indices] = 1.0
    
    # Ensure the initial_h is feasible
    if abs(initial_h.sum() - n_points / 2) > 1e-8:
        scale = (n_points / 2) / initial_h.sum()
        initial_h = np.clip(initial_h * scale, 0, 1)
    
    current_h = initial_h.copy()
    best_h = initial_h.copy()
    current_c5 = evaluate_erdos_solution(current_h)
    best_c5 = current_c5
    
    T = 1.0
    cooling_rate = 0.99
    iterations_per_temp = 100
    start_time = time.time()
    timeout = budget_s
    
    while time.time() - start_time < timeout:
        for _ in range(iterations_per_temp):
            # Generate candidate by perturbing current_h
            h_candidate = current_h.copy()
            # Apply 5 random pair swaps
            for _ in range(5):
                i, j = np.random.choice(n_points, size=2, replace=False)
                delta_upper = min(1.0 - h_candidate[i], h_candidate[j])
                delta_lower = -min(h_candidate[i], 1.0 - h_candidate[j])
                if delta_upper >= delta_lower:
                    delta = np.random.uniform(delta_lower, delta_upper)
                    h_candidate[i] += delta
                    h_candidate[j] -= delta
            # Ensure values stay in [0,1] due to rounding
            h_candidate = np.clip(h_candidate, 0, 1)
            
            # Compute candidate's C5
            c5_candidate = evaluate_erdos_solution(h_candidate)
            delta_c = c5_candidate - current_c5
            
            # Acceptance criterion
            if delta_c < 0 or np.exp(-delta_c / T) > np.random.rand():
                current_h = h_candidate.copy()
                current_c5 = c5_candidate
                if c5_candidate < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
        
        # Cool down
        T *= cooling_rate
        if T < 1e-8:
            break
    
    return best_h, best_c5, n_points
