import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    # Use the provided initial_h_values
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    # Ensure sum is exactly n_points / 2
    current_sum = np.sum(h_initial)
    if not np.isclose(current_sum, n_points / 2.0):
        scaling = (n_points / 2.0) / current_sum
        h_initial *= scaling
    # Set bounds for h
    bounds = [(0.0, 1.0) for _ in h_initial]

    constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2.0}
    constraints = [constraint]
    # Define objective function
    def objective(x_vec):
        x = np.array(x_vec)
        dx = 2.0 / n_points
        # Mode 'full' correlation computes h * (1-h) for all lags
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr * dx)
    # Initial evaluation for fallback
    best_h = h_initial.copy()
    best_c5 = objective(best_h)
    # Track time
    start_time = time.time()
    max_duration = budget_s * 0.8  # Reserve some time for final eval
    # Run optimization
    result = minimize(
        objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 1000, 'ftol': 1e-8}
    )
    # Update best if optimization succeeded
    if result.success and (time.time() - start_time) < max_duration:

    # Final validation and return
