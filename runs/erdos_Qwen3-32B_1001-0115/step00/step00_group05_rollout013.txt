import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100):
    h0 = initial_h_values
    n_points = len(h0)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    linear_constraint = optimize.LinearConstraint(np.ones(n_points), lb=target_sum, ub=target_sum)
    
    start_time = time.time()
    best_sol = h0
    best_c5 = objective(h0)

    # SLSQP from initial guess
    if time.time() - start_time < budget_s * 0.9:
        result = optimize.minimize(
            objective, x0=h0, method='SLSQP', bounds=bounds,
            constraints=linear_constraint, options={'maxiter': 100}
        )
        if result.success and objective(result.x) < best_c5:
            best_sol = result.x
            best_c5 = objective(result.x)

    # Differential Evolution with time control
    if time.time() - start_time < budget_s * 0.75:
        de_result = optimize.differential_evolution(
            objective, bounds, strategy='best1bin', maxiter=20, popsize=5,
            mutation=(0.5, 1.0), recombination=0.8, constraints=linear_constraint,
            polish=True, workers=1, seed=seed, updating='deferred'
        )
        if de_result.fun < best_c5:
            best_sol = de_result.x
            best_c5 = de_result.fun

    # Final check of progress
    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.95:
        # Local refinement with Nelder-Mead
        nm_bounds = [(0.0, 1.0)] * n_points
        nm_result = optimize.minimize(
            objective, x0=best_sol, method='Nelder-Mead', bounds=nm_bounds,
            options={'maxiter': 50, 'xatol': 1e-4, 'fatol': 1e-4}
        )
        if nm_result.fun < best_c5:
            best_sol = nm_result.x
            best_c5 = nm_result.fun

    return best_sol, best_c5, n_points
