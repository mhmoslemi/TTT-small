    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (initial_h, objective(initial_h), n_points)

    h_values = np.clip(result.x, 0, 1)
    h_values_sum = np.sum(h_values)
    delta = (n_points / 2) - h_values_sum
    if abs(delta) > 1e-10:
        if delta > 0:
            idx = np.where(h_values < 0.999)[0]
        else:
            idx = np.where(h_values > 1e-3)[0]
        if idx.size > 0:
            h_values[idx[0]] += delta
    h_values = np.clip(h_values, 0, 1)

    final_c5 = objective(h_values)

    return (h_values, final_c5, n_points)
