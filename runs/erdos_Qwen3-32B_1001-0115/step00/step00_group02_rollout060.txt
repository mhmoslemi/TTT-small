import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h = initial_h_values.copy().astype(float)
    n_points = len(h)
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        dx = 2.0 / n_points
        zero_lag_index = n_points - 1
        indices = np.delete(np.arange(len(corr)), zero_lag_index)
        values = corr[indices]
        max_value = np.max(values)
        return max_value * dx

    def compute_worst_k(h):
        corr = np.correlate(h, 1 - h, mode='full')
        zero_lag_index = n_points - 1
        indices = np.delete(np.arange(len(corr)), zero_lag_index)
        values = corr[indices]
        max_idx = indices[np.argmax(values)]
        lag = max_idx - (n_points - 1)
        return abs(lag)

    best_c5 = compute_c5(h)
    best_h = h.copy()
    patience_counter = 0

    while time.time() - start_time < budget_s:
        try:
            k = compute_worst_k(h)
            if k <= 0 or k >= n_points:
                continue  # skip if k is invalid
        except:
            break  # no valid k, break early

        def objective(x):
            total = 0.0
            for i in range(n_points - k):
                total += x[i] * (1 - x[i + k])
            return total

        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

        result = scipy.optimize.minimize(
            objective, h, method='SLSQP', bounds=bounds, constraints=constraint
        )

        if result.success:
            h_new = np.clip(result.x, 0.0, 1.0)
            s = np.sum(h_new)
            if not np.isclose(s, target_sum):
                scale_factor = target_sum / s
                h_new = np.clip(h_new * scale_factor, 0.0, 1.0)
            h = h_new
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                patience_counter = 0
            else:
                patience_counter += 1
            if patience_counter > 10:
                break  # early stopping if no improvement
        else:
            # Optionally perturb h if optimization fails
            h += np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0.0, 1.0)
            h = target_sum * h / np.sum(h)

    return best_h.tolist(), best_c5, n_points
