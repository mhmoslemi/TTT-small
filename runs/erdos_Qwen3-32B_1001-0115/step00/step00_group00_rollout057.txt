def compute_approx_c5(overlaps, epsilon):

    max_val = np.max(overlaps)

    exp_vals = np.exp((overlaps - max_val) / epsilon)

    sum_exp = np.sum(exp_vals)

    return max_val + np.log(sum_exp * epsilon)
