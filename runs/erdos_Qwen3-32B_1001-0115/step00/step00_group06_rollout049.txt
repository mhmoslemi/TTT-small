def compute_c5(h, dx):
    overlaps = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(overlaps)

def project_feasible(h, n_points):
    h_clipped = np.clip(h, 0.0, 1.0)
    current_sum = np.sum(h_clipped)
    target_sum = n_points / 2.0
    if np.isclose(current_sum, target_sum):
        return h_clipped
    delta = (target_sum - current_sum) / n_points
    h_clipped += delta
    h_clipped = np.clip(h_clipped, 0.0, 1.0)
    return h_clipped
