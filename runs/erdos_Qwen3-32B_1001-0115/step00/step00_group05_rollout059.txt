import numpy as np
import time
import random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        n_points = len(h)
    else:
        h = np.zeros(n_points)
        idxs = np.random.choice(n_points, size=int(n_points/2), replace=False)
        h[idxs] = 1.0
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    current_h = h.copy()
    current_c5 = best_c5
    
    T = 1.0
    alpha = 0.995
    m = 5  # Number of flip pairs per move
    
    while time.time() - start_time < budget_s:
        iterations_per_temp = 50 if T > 0.1 else 20
        for _ in range(iterations_per_temp):
            # Generate candidate with m swaps
            candidate_h = current_h.copy()
            zeros = np.where(candidate_h == 0)[0]
            ones = np.where(candidate_h == 1)[0]
            
            if len(zeros) < m or len(ones) < m:
                continue
            
            zero_indices = np.random.choice(zeros, size=m, replace=False)
            one_indices = np.random.choice(ones, size=m, replace=False)
            
            candidate_h[zero_indices] = 1.0
            candidate_h[one_indices] = 0.0
            
            candidate_c5 = evaluate_erdos_solution(candidate_h)
            delta_c = candidate_c5 - current_c5
            
            # Acceptance criterion
            if delta_c < 0 or random.random() < np.exp(-delta_c / T):
                current_h[:] = candidate_h[:]
                current_c5 = candidate_c5
                
                if candidate_c5 < best_c5:
                    best_h[:] = current_h[:]
                    best_c5 = current_c5
        
        T *= alpha
        if T < 1e-6:
            break
    
    return (best_h.tolist(), best_c5, n_points)
