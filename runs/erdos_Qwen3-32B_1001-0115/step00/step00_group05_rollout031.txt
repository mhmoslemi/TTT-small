def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    from time import time
    import evaluate_erdos_solution as evaluator

    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    start_time = time()
    best_h = h.copy()
    try:
        best_c5 = evaluator.evaluate_erdos_solution(h)['C5']
    except:
        best_c5 = float('inf')

    max_iter = 100_000
    for iter in range(max_iter):
        if time() - start_time > budget_s - 0.1:
            break

        # Compute current overlaps using np.correlate
        corr = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(corr)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Find worst shift
        worst_index = np.argmax(corr)
        shift = worst_index - (n_points - 1)

        # Compute gradient for the worst shift
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + shift
            if 0 <= j < n_points:
                grad[i] += (1.0 - h[j]) * dx
            j = i - shift
            if 0 <= j < n_points:
                grad[i] -= h[j] * dx

        # Apply gradient step
        learning_rate = 0.01 / (1 + iter * 0.0001)  # decreasing learning rate
        h -= learning_rate * grad

        # Clip to [0,1]
        np.clip(h, 0, 1, out=h)

        # Adjust sum
        current_sum = np.sum(h)
        delta = target_sum - current_sum
        malleable = (h > 1e-8) & (h < 1 - 1e-8)
        if np.any(malleable):
            h[malleable] += delta / np.sum(malleable)
            np.clip(h, 0, 1, out=h)

    # Final evaluation
    try:
        final_c5 = evaluator.evaluate_erdos_solution(best_h)['C5']
    except:
        final_c5 = best_c5
    return (best_h.tolist(), final_c5, n_points)
