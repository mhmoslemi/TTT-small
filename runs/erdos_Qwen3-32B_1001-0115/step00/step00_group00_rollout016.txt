import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h0 = initial_h_values.copy()
    n_points = len(h0)
    dx = 2.0 / n_points

    def objective(h_vec):
        h = h_vec.reshape((n_points,))
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h.reshape(n_points)) - n_points / 2}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = minimize(
        objective,
        h0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 300, 'ftol': 1e-8}
    )

    h_opt = result.x.astype(np.float64)
    corr_final = np.correlate(h_opt.reshape(n_points), 1 - h_opt.reshape(n_points), mode="full") * dx
    c5_bound = np.max(corr_final)

    return h_opt, c5_bound, n_points
