import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # Get initial h
    h = initial_h_values.copy().astype(np.float64)
    n_points = len(h)
    dx = 2.0 / n_points

    # Check sum constraint (should be satisfied)
    target_sum = n_points / 2.0
    assert np.allclose(np.sum(h), target_sum)

    best_h = h.copy()
    best_c5 = None

    alpha = 0.001 # learning rate
    max_iter = 1000
    start_time = time.time()

    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1.0:
            break

        # Compute cross-correlation
        corr = np.correlate(h, 1 - h, mode='full')

        # Compute current C5
        current_c5 = np.max(corr * dx)
        if best_c5 is None or current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Find max indices
        max_val = np.max(corr)
        max_indices = np.where(corr == max_val)[0]

        # Compute gradient
        gradient = np.zeros(n_points)
        for index in max_indices:
            k = index - (n_points - 1)
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    gradient[i] += (1 - h[j])

        # Average the gradient
        avg_gradient = (gradient / len(max_indices)) * dx

        # Update h
        h -= alpha * avg_gradient

        # Clip and normalize
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        delta = target_sum - current_sum
        if delta != 0:
            h += delta / n_points
            h = np.clip(h, 0, 1)

    return best_h, best_c5, n_points
