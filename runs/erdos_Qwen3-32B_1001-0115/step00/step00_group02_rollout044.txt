def project_simplex_with_bounds(h, S, bounds=(0,1)):
    h = np.array(h)
    n = len(h)
    lower = bounds[0]
    upper = bounds[1]
    # Compute the initial sum
    current_sum = np.sum(h)
    if abs(current_sum - S) < 1e-8:
        return h
    # Compute the difference
    delta = S - current_sum

    # If delta is positive, we need to increase the sum
    if delta > 0:
        # Can only increase the elements where h < upper
        indices = np.where(h < upper)[0]
        num_indices = len(indices)
        if num_indices == 0:
            # All elements are at upper bound, cannot increase
            return h  # but this should not happen because current_sum < S
        max_possible_inc = (upper - h[indices]).sum()
        if max_possible_inc < delta:
            # Cannot reach the desired sum
            return h + (upper - h[indices]).ravel() / num_indices  # but this is just an example; not sure
        h[indices] += (delta / num_indices) if num_indices > 0 else 0
    else:
        # Need to decrease the sum
        delta = -delta  # positive now
        indices = np.where(h > lower)[0]
        num_indices = len(indices)
        if num_indices == 0:
            return h
        max_possible_dec = (h[indices] - lower).sum()
        if max_possible_dec < delta:
            return h - (h[indices] - lower).ravel() / num_indices
        h[indices] -= (delta / num_indices) if num_indices > 0 else 0
    # Clip to ensure [lower, upper]
    h = np.clip(h, lower, upper)
    return h
