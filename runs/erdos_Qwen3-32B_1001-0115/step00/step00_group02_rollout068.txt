        new_n = original_n + 1
        x_old = np.linspace(0, 2, original_n)
        x_new = np.linspace(0, 2, new_n)
        h_resampled = np.interp(x_new, x_old, h_vals)
    else:

        h_resampled = h_vals.copy()

            elapsed = time() - start_time
