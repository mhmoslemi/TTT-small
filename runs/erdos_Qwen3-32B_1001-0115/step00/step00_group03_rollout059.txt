import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = float(np.max(np.correlate(h, 1 - h, mode="full")) * dx)
    start_time = time.time()
    
    def compute_c5(h_vec):
        return np.max(np.correlate(h_vec, 1 - h_vec, mode="full")) * dx

    middle_idx = n_points - 1
    while time.time() - start_time < budget_s - 10:
        # Find current max overlap and corresponding shift
        corr = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        k_samples = max_idx - middle_idx
        k_samples = int(np.round(k_samples))
        
        def objective(x):
            h_flat = x
            h_current = h_flat.reshape(n_points)
            if k_samples >= 0:
                valid_i = np.arange(0, n_points - k_samples)
                overlap = np.sum(h_current[valid_i] * (1 - h_current[valid_i + k_samples]))
            else:
                k_abs = -k_samples
                valid_i = np.arange(k_abs, n_points)
                overlap = np.sum(h_current[valid_i - k_abs] * (1 - h_current[valid_i]))
            return overlap * dx

        # Enforce integral constraint and bounds
        constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        bounds = tuple([(0, 1)] * n_points)
        
        # Optimize with SLSQP
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 10, 'ftol': 1e-6}
        )
        
        if result.success:
            new_h = result.x.reshape(n_points)
            new_c5 = compute_c5(new_h)
            if new_c5 < best_c5:
                best_h[:] = new_h
                best_c5 = new_c5
                h[:] = new_h

        # Early stop if no improvement (prevent infinite loops)
        if np.isclose(best_c5, compute_c5(h), atol=1e-6):
            break

    return best_h, best_c5, n_points
