import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h = initial_h_values.copy()
    n_points = len(h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_C5(h_vec):
        one_minus_h = 1.0 - h_vec
        corr = np.correlate(h_vec, one_minus_h, mode='full')
        valid_corr = corr[n_points - 1 : 2 * n_points - 1]
        return np.max(valid_corr) * dx

    best_h = h.copy()
    best_c5 = compute_C5(best_h)
    current_h = h.copy()
    current_c5 = best_c5

    temp = 1.0  # initial temperature
    alpha = 0.99  # cooling rate
    MAX_ATTEMPTS = 10  # for sum adjustment

    while time.time() - start_time < budget_s - 1:
        # Generate candidate
        candidate = current_h.copy()
        n_perturb = max(1, int(0.1 * n_points))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for idx in indices:
            delta = np.random.uniform(-0.1, 0.1)
            candidate[idx] = np.clip(candidate[idx] + delta, 0.0, 1.0)
        # Sum adjustment
        current_sum = np.sum(candidate)
        delta_sum = target_sum - current_sum
        adjusted = False
        for _ in range(MAX_ATTEMPTS):
            adjust_idx = np.random.choice(n_points)
            if 0 <= candidate[adjust_idx] + delta_sum <= 1.0:
                candidate[adjust_idx] += delta_sum
                adjusted = True
                break
        if not adjusted:
            continue  # Skip invalid candidate
        
        # Evaluate candidate
        candidate_c5 = compute_C5(candidate)
        delta_c5 = candidate_c5 - current_c5
        
        # Acceptance
        if delta_c5 < 0:
            current_h = candidate
            current_c5 = candidate_c5
            if candidate_c5 < best_c5:
                best_h = candidate.copy()
                best_c5 = candidate_c5
        else:
            # Simulated annealing acceptance
            if temp > 1e-6 and np.random.rand() < np.exp(-delta_c5 / temp):
                current_h = candidate
                current_c5 = candidate_c5
        
        # Cool down
        temp *= alpha

    return best_h, best_c5, n_points
