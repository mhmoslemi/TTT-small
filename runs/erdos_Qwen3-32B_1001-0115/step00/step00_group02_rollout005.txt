def project(h, target_sum):
    h_proj = np.clip(h, 0, 1)
    current_sum = h_proj.sum()
    if np.isclose(current_sum, target_sum):
        return h_proj
    excess = current_sum - target_sum
    non_fixed = np.where((h_proj > 0) & (h_proj < 1))[0]
    if len(non_fixed) == 0:
        return h_proj
    delta = -excess / len(non_fixed)
    h_proj[non_fixed] += delta
    h_proj = np.clip(h_proj, 0, 1)
    return h_proj
