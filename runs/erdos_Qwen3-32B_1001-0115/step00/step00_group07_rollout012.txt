            return np.max(corr) * dx
        
        result = scipy.optimize.minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=eq_constraint,
            options={'maxiter': 300, 'ftol': 1e-8}
        )
        
        if not result.success:
            continue
        h_opt = result.x
        h_opt = np.clip(h_opt, 0, 1)
        h_opt_sum = h_opt.sum()
        if not np.isclose(h_opt_sum, 40.0, rtol=1e-5, atol=1e-5):
            h_opt = h_opt * (40.0 / h_opt_sum)
        c5 = objective(h_opt)
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5
    
    # Final fallback to initial guess if all optimizations failed
    if best_h is None:
        best_h = initial_guesses[0]
        best_c5 = objective(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
