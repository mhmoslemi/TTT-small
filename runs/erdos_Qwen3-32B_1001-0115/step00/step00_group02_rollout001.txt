        new_c5 = compute_c5(h, dx)
        
        # Acceptance criterion
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / T):
            # Accept the change
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            # Reject, revert
            h[i] = old_a
            h[j] = old_b
        
        # Cool down
        T *= alpha

import time

def compute_c5(h_vals, dx):
    one_minus_h = 1.0 - h_vals
    correlate_vals = np.correlate(h_vals, one_minus_h, mode='full')
    return np.max(correlate_vals) * dx

    start_time = time.time()

    n_points = len(h)
    dx = 2.0 / n_points

    best_h = h.copy()
    best_c5 = current_c5
    
    T = 1.0
    alpha = 0.999
    
    while time.time() - start_time < budget_s:
        i, j = np.random.choice(n_points, size=2, replace=False)
        
        a = h[i]
        b = h[j]

        a_max = min(1.0, s)
        
        if a_min >= a_max:

        delta = np.random.rand() * (a_max - a_min) + a_min
        new_a = delta
        new_b = s - delta
        
        old_a, old_b = h[i], h[j]
        h[i] = new_a
        h[j] = new_b
        
        new_c5 = compute_c5(h, dx)
