            h_scaled = h_clipped * scale
            h = np.clip(h_scaled, 0, 1)
        else:
            h = h_clipped
        
        # Final sum check
        h_sum = np.sum(h)
        if abs(h_sum - desired_sum) > 1e-8:
            # Adjust last element to enforce sum constraint
            h[-1] += (desired_sum - h_sum)
            h = np.clip(h, 0, 1)
    
    return best_h, best_c5, n_points
