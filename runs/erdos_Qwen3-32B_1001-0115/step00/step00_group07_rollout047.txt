        remaining_sum = target_sum - 2.0
        h[2:] = remaining_sum / (n_points - 2)
        h_clipped = np.clip(h, 0, 1)
        starts.append(h_clipped)

    best_c5 = float('inf')
    best_h = None
    start_time = time.time()

    for h_start in starts:
        if time.time() - start_time > budget_s - 5:
            break
        h_start = h_start + 0.01 * np.random.randn(*h_start.shape)
        h_start = np.clip(h_start, 0, 1)
        sum_h = np.sum(h_start)
        if sum_h != target_sum:
            h_start = h_start / sum_h * target_sum
            h_start = np.clip(h_start, 0, 1)
        bounds = [(0, 1)] * n_points
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        try:
            result = opt.minimize(lambda x: evaluate_erdos_solution(n_points, x), h_start, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 300, 'ftol': 1e-6})
        except:
            continue
        if result.success:
            h_proj = np.clip(result.x, 0, 1)
            sum_proj = np.sum(h_proj)
            if sum_proj != 0:
                h_proj *= (target_sum / sum_proj)
            else:
                h_proj = np.full(n_points, 0.5)
            c5 = evaluate_erdos_solution(n_points, h_proj)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_proj
        else:
            c5 = evaluate_erdos_solution(n_points, h_start)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_start.copy()

    if best_h is None:
        best_h = h_start.copy()
        best_c5 = evaluate_erdos_solution(n_points, best_h)

    return (best_h.tolist(), best_c5, n_points)
