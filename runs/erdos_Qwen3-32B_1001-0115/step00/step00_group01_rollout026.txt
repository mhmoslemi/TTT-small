import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

    best_c5 = compute_c5(h, dx)

    alpha = 1e-3  # Learning rate

    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode="full") * dx

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = np.copy(h)
        max_lags = np.where(corr == current_c5)[0]
        if not max_lags.size:

        for k_index in max_lags:

                valid_i = np.arange(n - k_actual)
                j = valid_i + k_actual
