import numpy as np
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h = initial_h_values
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    best_h = h.copy()
    best_c5 = max(np.correlate(h, 1 - h, mode='full')[n_points:n_points * 2 - 1]) * dx
    start_time = time.time()

    step_size = 0.1
    patience = 50
    no_improve = 0
    max_iterations = 10000

    for iter in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        h_np = h.astype(np.float64)

        # Compute overlaps for positive shifts only
        corr = np.correlate(h_np, 1 - h_np, mode='full')
        max_corr = -np.inf
        worst_k = 1

        # Find worst shift
        for k in range(1, n_points):
            idx = n_points - 1 + k
            if idx >= len(corr):
                continue
            current_corr = corr[idx]
            if current_corr > max_corr:
                max_corr = current_corr
                worst_k = k

        # Calculate gradient direction
        direction = np.zeros(n_points)
        for i in range(n_points):
            if i + worst_k < n_points:
                direction[i] += (1 - h_np[i + worst_k])
            if i - worst_k >= 0:
                direction[i] -= h_np[i - worst_k]

        # Take step and project
        h_new = h_np - step_size * direction
        h_new = np.clip(h_new, 0, 1)

        # Adjust for sum constraint
        h_new += (target_sum - h_new.sum()) / n_points
        h_new = np.clip(h_new, 0, 1)

        # Evaluate improvement

            step_size *= 0.5
