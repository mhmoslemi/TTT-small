from scipy.optimize import minimize

def run():
    global initial_h_values
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    required_sum = n_points / 2.0

    initial_overlaps = []

        initial_overlaps.append(overlap)
    c5_initial = max(initial_overlaps) if initial_overlaps else 0.0

    initial_guess = np.concatenate([h_initial, [c5_initial]])

    # Objective function
    def objective(x):
        return x[-1]  # minimize C5

    # Constraints
    def sum_constraint(x):
        return np.sum(x[:-1]) - required_sum

    # Nonlinear overlapping constraints
    class OverlapConstraint:
        def __init__(self, k):
            self.k = k

        def __call__(self, x):
            h = x[:-1]
            c5 = x[-1]
            k = self.k

            end = n_points - k
            if end <= start:

    opt_result = minimize(objective, initial_guess, method='SLSQP', bounds=bounds, constraints=constraints_list, options={'maxiter': 200, 'ftol': 1e-6})  # Adjust maxiter for time constraints
