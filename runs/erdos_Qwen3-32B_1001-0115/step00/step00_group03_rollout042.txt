    def _enforce_sum_constraint(h, target_sum=target_sum):
        h_clipped = np.clip(h, 0, 1.0)
        current_sum = np.sum(h_clipped)
        if abs(current_sum) < 1e-10:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        h_scaled = h_clipped * scale
        h_final = np.clip(h_scaled, 0, 1.0)
        return h_final

    swarm_size = 50

    velocities = np.zeros_like(particles)

    pbest_c5 = np.full(swarm_size, np.inf)
    gbest = np.zeros(n_points, dtype=np.float64)
    gbest_c5 = np.inf

    for i in range(swarm_size):

        particle = initial + perturb
        particle = _enforce_sum_constraint(particle)
        particles[i] = particle
        velocities[i] = 0.01 * np.random.randn(n_points)

    for i in range(swarm_size):
