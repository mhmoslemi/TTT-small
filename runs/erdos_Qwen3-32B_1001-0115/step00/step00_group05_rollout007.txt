        corr_all = np.correlate(h_flat, one_minus_h, mode="full")
        max_corr = np.max(corr_all)
        max_indices = np.where(corr_all == max_corr)[0]
        k_index = max_indices[0]
        k = k_index - (n_points - 1)

        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] += (1.0 - h[j]) * dx
                grad[j] -= h[i] * dx

        h -= learning_rate * grad

        # Project back to feasible region
        h_clipped = np.clip(h, 0.0, 1.0)
        current_sum = np.sum(h_clipped)
        delta = (target_sum - current_sum) / n_points
        h = np.clip(h_clipped + delta, 0.0, 1.0)

        # Update best solution
        curr_c5 = evaluate_erdos_solution(h)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = h.copy()

        # Learning rate decay
        learning_rate *= 0.999

    return (best_h.tolist(), best_c5, n_points)
