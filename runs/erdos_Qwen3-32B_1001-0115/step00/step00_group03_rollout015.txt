    corr = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr * dx)
    sum_diff = h.sum() - target_sum
    penalty = 1e4 * (sum_diff ** 2)
    return c5 + penalty

def run(seed=42, budget_s=1100):
    start_time = time.time()
    
    global initial_h_values
    if 'initial_h_values' in globals():
        h0 = initial_h_values.copy()
        n_points = len(h0)
        target_sum = n_points / 2.0
    else:
        n_points = 70
        h0 = np.zeros(n_points)
        mid = n_points // 2
        h0[mid-1:mid+2] = [0.0, 2.0/3, 2.0/3, 0.0, 0.0]
        h0 = np.clip(h0, 0, 1)
        target_sum = n_points / 2.0

    dx = 2.0 / n_points
    
    h0 = np.clip(h0, 0, 1)
    current_sum = h0.sum()
    if not np.isclose(current_sum, target_sum):
        h0 = h0 * target_sum / current_sum
        h0 = np.clip(h0, 0, 1)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = None
    best_h = None
    
    np.random.seed(seed)
    max_restarts = 3
    
    for restart in range(max_restarts):
        if time.time() - start_time > budget_s - 5:
            break
        
        h_perturbed = h0 + np.random.normal(0, 0.05, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_sum = h_perturbed.sum()
        scaling_factor = target_sum / current_sum
        h_perturbed = h_perturbed * scaling_factor
        h_perturbed = np.clip(h_perturbed, 0, 1)
        
        res = minimize(
            lambda h_vec: penalized_objective(h_vec, n_points, target_sum, dx),
            x0=h_perturbed.flatten(),
            method='Nelder-Mead',
            bounds=bounds,
            options={'maxiter': 200, 'xatol': 1e-4, 'fatol': 1e-4}
        )
        
        if res.success:
            h_opt = res.x.reshape(n_points)
            h_opt = np.clip(h_opt, 0, 1)
            current_sum = h_opt.sum()
            if not np.isclose(current_sum, target_sum):
                h_opt = h_opt * (target_sum / current_sum)
                h_opt = np.clip(h_opt, 0, 1)
            
            c5_value = np.max(np.correlate(h_opt, 1 - h_opt, mode='full') * dx)
            if best_c5 is None or c5_value < best_c5:
                best_c5 = c5_value
                best_h = h_opt
    
    if best_h is None:
        best_h = h0
        best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full') * dx)
    
    return best_h, best_c5, n_points
