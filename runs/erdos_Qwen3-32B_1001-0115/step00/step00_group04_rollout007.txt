import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    global initial_h_values
    h0 = np.array(initial_h_values, dtype=np.float64).copy()
    n_points = len(h0)

    start_t = time.time()

                      constraints=constraints, options=options)

    runtime = time.time() - start_t

        print("Approaching budget limit, returning best found.")

    h_result = result.x.reshape(n_points)
    full_corr = np.correlate(h_result, 1 - h_result, mode='full') * dx
    c_overlaps = full_corr[-n_points:]
    best_c5 = np.max(c_overlaps)
