        initial_set = False

    n_points = len(h0)
    desired_sum = n_points / 2.0
    dx = 2.0 / n_points

    def project_h(h_vec):
        h = clip(h_vec, 0, 1)
        current_sum = np.sum(h)
        delta = desired_sum - current_sum
        if delta == 0:
            return h
        indices = np.where((h > 0.0) & (h < 1.0))[0]
        if len(indices) == 0:
            return h
        add = delta / len(indices)
        h[indices] += add
        h = clip(h, 0, 1)
        return h

    h_initial = project_h(h0)

    class State:
        best_h = h_initial.copy()
        best_c5 = np.inf
        elapsed = 0
        start_time = time.time()

    def objective(h_flat):
        h = h_flat.reshape(-1)
        one_minus_h = 1 - h
        correlations = correlate(h, one_minus_h, mode='full')
        max_corr = np.max(correlations)
        State.elapsed = time.time() - State.start_time
        State.best_h[:] = h.copy()
        State.best_c5 = max_corr * dx
        return max_corr * dx

    def gradient(h_flat):
        h = h_flat.reshape(-1)
        one_minus_h = 1 - h
        correlations = correlate(h, one_minus_h, mode='full')
        max_index = argmax(correlations)
        k = max_index - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] += (1 - h[j])
                grad[j] -= h[i]
        return (grad * dx).flatten()

    def callback(_):
        pass

    bounds = [(0, 1) for _ in range(n_points)]
    result = scipy.optimize.minimize(
        objective,
