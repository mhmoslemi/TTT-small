                new_h = h.copy()
                new_h[i], new_h[j] = new_h[j], new_h[i]
                # Evaluate
                new_c5, _ = evaluate_erdos_solution(new_h)
                if new_c5 < best_c5:
                    h[:] = new_h[:]
                    best_h[:] = new_h[:]
                    best_c5 = new_c5
    return best_h, best_c5, n_points
