import cvxpy as cp

def project(h_current):

    n = len(h_current)

    h = cp.Variable(n)

    objective = cp.Minimize(cp.sum_squares(h - h_current))

    constraints = [h >= 0, h <= 1, cp.sum(h) == n_points / 2]

    prob = cp.Problem(objective, constraints)

    prob.solve(solver=cp.ECOS)

    if prob.status != cp.OPTIMAL:

        raise ValueError("Projection failed")

    return h.value
