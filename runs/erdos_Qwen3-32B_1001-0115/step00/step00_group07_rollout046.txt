import numpy as np
from scipy.optimize import minimize
from time import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h0 = initial_h_values.copy()
    n_points = len(h0)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate diverse initial guesses
    initial_candidates = [h0]
    
    # Random initializations with sum correction
    for _ in range(20):
        h = np.random.rand(n_points)
        h *= target_sum / h.sum()
        h = np.clip(h, 0, 1)
        # Ensure sum constraint
        if abs(h.sum() - target_sum) > 1e-6:
            factor = target_sum / h.sum()
            h *= factor
            h = np.clip(h, 0, 1)
        initial_candidates.append(h)
    
    # Structured patterns
    h_alt = np.zeros(n_points)
    h_alt[::2] = 1.0
    h_half = np.ones(n_points) * 0.5
    h_split = np.zeros(n_points)
    h_split[:int(n_points/2)] = 1.0
    for h in [h_alt, h_half, h_split]:
        # Ensure sum constraint
        scale = target_sum / h.sum()
        h = np.clip(h * scale, 0, 1)
        initial_candidates.append(h)

    best_c5 = np.inf
    best_h = None
    start = time()
    
    for candidate in initial_candidates:
        if time() - start > 0.98 * budget_s:
            break  # Leave time for final corrections

        # Define constraints and bounds
        bounds = [(0, 1)] * n_points
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        
        # Run optimizer
        result = minimize(
            fun=lambda x: evaluate_erdos_solution(np.ravel(x)),
            x0=candidate,
            bounds=bounds,
            constraints=cons,
            method='SLSQP',
            options={'maxiter': 500, 'ftol': 1e-6}
        )
        
        if result.success:
            current_h = result.x
            current_c5 = evaluate_erdos_solution(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
        elif best_h is None:
            best_h = candidate
            best_c5 = evaluate_erdos_solution(best_h)

    # Final feasibility enforcement
    if best_h is None:
        best_h = initial_candidates[0]
    
    best_h = np.clip(best_h, 0, 1)
    s = best_h.sum()
    if abs(s - target_sum) > 1e-6:
        scale = target_sum / s
        best_h = np.clip(best_h * scale, 0, 1)
    
    return (best_h.tolist(), evaluate_erdos_solution(best_h), n_points)
