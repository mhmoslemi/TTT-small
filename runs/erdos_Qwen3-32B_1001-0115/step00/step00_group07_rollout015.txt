import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def project(h):
        h = np.clip(h, 0, 1)
        s = h.sum()
        if s == 0:
            return np.ones_like(h) * 0.5
        return np.clip(h * (n_points / 2 / s), 0, 1)

    def calc_c5(h):
        h = project(h)
        cross_cor = np.correlate(h, 1 - h, mode="full")
        return float(np.max(cross_cor) * dx)

    pop_size = 50
    mut = 0.1
    cross = 0.8
    max_gen = 50

    if 'initial_h_values' in globals():
        pop = [project(initial_h_values)]
    else:
        pop = [project(np.random.rand(n_points)) for _ in range(pop_size)]

    best_h = pop[0].copy()
    best_c5 = calc_c5(best_h)

    while time.time() - start_time < budget_s and len(pop) == pop_size:
        fitness = np.array([calc_c5(ind) for ind in pop])
        idx = np.argmin(fitness)
        if fitness[idx] < best_c5:
            best_h, best_c5 = pop[idx].copy(), fitness[idx]

        parents = [pop[i] for i in np.argsort(fitness)[:pop_size//5]]
        new_pop = parents[:]
        while len(new_pop) < pop_size:
            a, b = np.random.choice(parents, 2)
            chi = a * (np.random.rand(n_points) < cross) + b * (1 - (np.random.rand(n_points) < cross))
            if np.random.rand() < mut:
                chi += np.random.randn(n_points) * 0.1
            new_pop.append(project(chi))
        pop = new_pop

    if time.time() - start_time < budget_s:
        res = minimize(lambda x: calc_c5(x), x0=best_h, method='Nelder-Mead',
                       bounds=[(0, 1)] * n_points, options={'maxiter': 100})
        if res.success:
            final_h = project(res.x)
            final_c5 = calc_c5(final_h)
            if final_c5 < best_c5:
                best_h, best_c5 = final_h, final_c5

    return best_h.tolist(), best_c5, n_points
