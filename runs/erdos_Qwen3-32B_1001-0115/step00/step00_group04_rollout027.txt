def normalize_h(h, n_points):
    target_sum = n_points / 2.0

    child = parent.copy()
    mutation_rate = 0.1

    mask = rng.random(n_points) < mutation_rate
    perturbation = rng.normal(0, mutation_sigma, size=parent.shape)
    child[mask] += perturbation[mask]
    child = normalize_h(child, n_points)
    return child

    probs /= probs.sum()
    # Select indices
    selected = rng.choice(sorted_indices, size=size, replace=True, p=probs)
    return selected

def normalize_h(h, n_points):
    target_sum = n_points / 2.0
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    if current_sum <= 1e-10:
        return np.full(n_points, target_sum / n_points)
    scaling_factor = target_sum / current_sum
    h_scaled = h_clipped * scaling_factor
    h_scaled = np.clip(h_scaled, 0, 1)
    return h_scaled

def mutate(rng, parent, n_points):
    child = parent.copy()
    mutation_rate = 0.1
    mutation_sigma = 0.05
    mask = rng.random(n_points) < mutation_rate
    perturbation = rng.normal(0, mutation_sigma, size=child.shape)
    child[mask] += perturbation[mask]
    child = normalize_h(child, n_points)
    return child

    probs = 1.0 / ranks
    probs /= probs.sum()
    selected = rng.choice(sorted_indices, size=size, replace=True, p=probs)
