            cross_corr = np.correlate(current_h, inv_h, mode='full') * dx
            max_index = np.argmax(cross_corr)
            k = max_index - (n - 1)
            
            grad = np.zeros_like(current_h)
            if k > 0:
                grad[k:] = inv_h[:-k]
            elif k < 0:
                k_abs = -k
                grad[:-k_abs] = inv_h[k_abs:]
            else:
                grad[:] = inv_h
            
            # Gradient step
            lr = learning_rate * (learning_decay ** (i // decay_interval))
            current_h = current_h - lr * grad
            current_h = np.clip(current_h, 0, 1)
            
            # Normalize
            target_sum = n / 2
            current_sum = current_h.sum()
            if current_sum != 0:
                current_h = current_h * (target_sum / current_sum)
                current_h = np.clip(current_h, 0, 1)
            
            # Track best
            current_c5 = evaluate_erdos_solution(current_h, n)
            if current_c5 < best_c5_level:
                best_h_level[:] = current_h.copy()
                best_c5_level = current_c5
        
        if best_c5_level < best_c5:
            best_h = best_h_level.copy()
            best_c5 = best_c5_level
            best_n = n
    
    return (best_h, best_c5, best_n)
