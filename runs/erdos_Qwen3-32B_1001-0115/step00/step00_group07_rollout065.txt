    initial = np.array(initial_h_values, dtype=np.float64).copy()
    n_points = len(initial)
    dx = 2.0 / n_points
    desired_sum = n_points / 2.0

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        omh = 1 - h
        corr = np.correlate(h, omh, mode='full')
        corr *= dx
        return np.max(corr)

    candidates = [initial.copy()]
    for _ in range(50):
        h = np.random.rand(n_points)
        h_sum = h.sum()
        if h_sum > 1e-9:
            h *= desired_sum / h_sum
        h = np.clip(h, 0, 1)
        candidates.append(h)
    for _ in range(10):
        idx = np.random.choice(n_points, int(desired_sum), replace=False)
        h = np.zeros(n_points)
        h[idx] = 1.0
        candidates.append(h)

    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    budget_remaining = budget_s

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - desired_sum}]

    for h_initial in candidates:
        if time.time() - start_time > budget_s - 5:
            break
        h_initial = np.clip(h_initial, 0, 1)
        h_initial_sum = h_initial.sum()
        if not np.isclose(h_initial_sum, desired_sum, atol=1e-6):
            h_initial = np.clip(h_initial * (desired_sum / h_initial_sum), 0, 1)
        try:
            res = scipy.optimize.minimize(
                objective, h_initial, method='SLSQP',
                bounds=bounds, constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-6, 'disp': False}
            )
            if res.success:
                h_opt = np.clip(res.x, 0, 1)
                h_opt_sum = h_opt.sum()
                if not np.isclose(h_opt_sum, desired_sum, atol=1e-6):
                    h_opt = np.clip(h_opt * (desired_sum / h_opt_sum), 0, 1)
                current_c5 = objective(h_opt)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_opt
        except:
            pass

    if best_h is None:
        best_h = initial
        best_c5 = objective(best_h)

    best_h_sum = best_h.sum()
    if not np.isclose(best_h_sum, desired_sum, atol=1e-6):
        best_h = np.clip(best_h * (desired_sum / best_h_sum), 0, 1)

    return (best_h.tolist(), best_c5, n_points)
