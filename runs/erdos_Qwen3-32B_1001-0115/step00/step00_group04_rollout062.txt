import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    import time
    np.random.seed(seed)
    
    n_points = 72  # Chosen for balance between accuracy and speed
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Starting from uniform distribution
    h0 = np.full(n_points, 0.5)

    # Objective function for minimization
    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, (1 - h), mode='full') * dx
        return np.max(corr)

    # Constraints: sum(h) = target_sum
    constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    # Bounds for each h[i]
    bounds = [(0.0, 1.0)] * n_points

    # Set up result storage and timer
    best_h = h0.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Use SLSQP with adaptive iteration limit
    result = opt.minimize(
        objective,
        h0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 1000, 'ftol': 1e-6}
    )

    if result.success:
        h_values = result.x
        h_values = h_values.reshape(n_points)
        # Clamp to constraints in case of numerical errors
        np.clip(h_values, 0.0, 1.0, out=h_values)
        c5_bound = objective(h_values)
    else:
        # Fall back to best unconverged solution
        h_values = best_h
        c5_bound = best_c5

    # Final verification
    assert np.isclose(np.sum(h_values) * dx, 1.0, atol=1e-5), "Integral constraint failed"
    assert np.all(0.0 <= h_values) and np.all(h_values <= 1.0), "Bounds violated"
    
    return (h_values, c5_bound, n_points)
