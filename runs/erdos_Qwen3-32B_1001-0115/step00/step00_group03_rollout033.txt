import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    h0 = initial_h_values.copy().astype(np.float64)
    n_points = len(h0)
    dx = 2.0 / n_points

    def objective(h_vec):
        h = np.clip(h_vec, 0.0, 1.0)
        overlap = np.correlate(h, 1.0 - h, mode='full')
        return np.max(overlap) * dx

    cons = (
        {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}
    )
    bounds = [(0.0, 1.0)] * n_points

    # Initial check of time and constraints
    start_time = time.time()
    result = opt.minimize(
        objective, 
        x0=h0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 300, 'ftol': 1e-8, 'disp': False}
    )

    elapsed = time.time() - start_time
    remaining_time = budget_s - elapsed

    # Spend remaining time on additional iterations if possible
    if remaining_time > 0:
        max_add_iter = int(remaining_time / (elapsed / result.nit)) if result.nit > 0 else 100
        if max_add_iter > 0 and not result.success:
            result = opt.minimize(
                objective,
                x0=result.x,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': max_add_iter, 'ftol': 1e-8, 'disp': False}
            )

    best_h = np.clip(result.x, 0.0, 1.0)
    best_h_sum = np.sum(best_h)
    if best_h_sum > 1e-8:
        scale = (n_points / 2.0) / best_h_sum
        best_h *= scale
        best_h = np.clip(best_h, 0.0, 1.0)

    overlap = np.correlate(best_h, 1.0 - best_h, mode='full')
    c5_bound = np.max(overlap) * dx

    return best_h, c5_bound, n_points
