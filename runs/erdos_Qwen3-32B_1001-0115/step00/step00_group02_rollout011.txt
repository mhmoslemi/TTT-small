    h.value = h_initial.copy()  # Warm start
    constraints = [
        cp.sum(h) == n_points / 2,
        0 <= h,
        h <= 1
    ]
    start_time = time()
    
    while time() - start_time < budget_s - 10:
        # Solve current problem
        objective = cp.Minimize(C5)
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CVXOPT, maxiters=100)
        if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            break
        current_h = h.value
        
        # Compute current C5 and worst shift
        h_np = current_h
        one_minus_h = 1.0 - h_np
        corr = np.correlate(h_np, one_minus_h, mode='full') * dx
        i_max = np.argmax(corr)
        wshift_k = i_max - (n_points - 1)
        
        # Add constraint for worst shift
        valid_is = []
        for i in range(n_points):
            if 0 <= i + wshift_k < n_points:
                valid_is.append(i)
        if valid_is:
            sum_expr = 0
            for i in valid_is:
                sum_expr += h[i] * (1 - h[i + wshift_k])
            constraints.append(sum_expr <= C5 / dx)
    
    # Final solution
    prob = cp.Problem(cp.Minimize(C5), constraints)
    prob.solve(solver=cp.CVXOPT)
    h_opt = np.clip(h.value, 0, 1)
    h_sum = np.sum(h_opt)
    if abs(h_sum - n_points / 2) > 1e-5:
        adjustment = (n_points / 2 - h_sum) / n_points
        h_opt = np.clip(h_opt + adjustment, 0, 1)
    
    h_np = h_opt
    one_minus_h = 1.0 - h_np
    corr = np.correlate(h_np, one_minus_h, mode='full') * dx
    c5_bound = np.max(corr)
    return (h_opt, c5_bound, n_points)
