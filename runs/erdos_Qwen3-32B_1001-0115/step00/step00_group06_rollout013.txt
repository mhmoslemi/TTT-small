import numpy as np
import cvxpy as cp
import time

def run():
    h = np.array(initial_h_values).copy()

    dx = 2.0 / n_points
    best_h = h.copy()

    start_time = time.time()
    budget_seconds = 1100

    threshold = 0.5
    lower_bound = 0.01
    upper_bound = 0.99
    iter_start = time.time()

    while time.time() - start_time < budget_seconds:
        n = n_points
        ks = np.arange(1, n)
        overlaps = []
        for k in ks:
            indices = (np.arange(n) + k) % n
            overlap = np.sum(h * (1 - h[indices])) * dx
            overlaps.append(overlap)
        overlaps = np.array(overlaps)
        if overlaps.size == 0:
            break
        max_k_index = np.argmax(overlaps)
        max_k = ks[max_k_index]

        indices_plus = (np.arange(n) + max_k) % n
        indices_minus = (np.arange(n) - max_k) % n
        grad = dx * (1 - h[indices_plus] - h[indices_minus])

        h_var = cp.Variable(n)
        constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == n / 2]
        objective = cp.Minimize(grad @ h_var)
        problem = cp.Problem(objective, constraints)

        try:
            problem.solve(solver=cp.CVXOPT, verbose=False)
            if problem.status in {cp.OPTIMAL, cp.OPTIMAL_INACCURATE}:
                new_h = np.array(h_var.value)
