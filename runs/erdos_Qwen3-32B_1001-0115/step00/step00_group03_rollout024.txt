import numpy as np
import scipy.optimize as opt
import time

def run():
    start_time = time.time()
    h_start = initial_h_values
    n_points = len(h_start)
    dx = 2.0 / n_points

    def objective(h):
        cc = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(cc)

    def sum_eq(h):
        return np.sum(h) - n_points / 2.0

    cons = [{'type': 'eq', 'fun': sum_eq}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    x0 = h_start.copy()

    result = None
    try:
        result = opt.minimize(
            objective,
            x0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-6, 'disp': False}
        )
    except:
        pass

    elapsed = time.time() - start_time
    h_opt = result.x if result is not None and result.success else x0
    h_opt_sum = np.sum(h_opt)
    if abs(h_opt_sum - n_points / 2) > 1e-6:
        delta = (n_points / 2 - h_opt_sum) / n_points
        h_opt += delta
        h_opt = np.clip(h_opt, 0, 1)

    best_h = h_opt
    best_c5 = objective(best_h)
    if best_c5 > objective(h_start):
        best_h, best_c5 = h_start, objective(h_start)

    return best_h, best_c5, n_points
