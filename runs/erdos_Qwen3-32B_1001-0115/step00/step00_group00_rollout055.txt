        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_val = np.max(overlaps)
        exp_vals = np.exp((overlaps - max_val) / epsilon_smooth)
        obj_val = max_val + epsilon_smooth * np.log(np.sum(exp_vals))
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_minus = h.copy()
            h_minus[i] -= eps
            
            h_plus_clipped = np.clip(h_plus, 0, 1)
            h_minus_clipped = np.clip(h_minus, 0, 1)
            
            overlaps_plus = np.correlate(h_plus_clipped, 1 - h_plus_clipped, mode='full') * dx
            max_val_p = np.max(overlaps_plus)
            exp_vals_p = np.exp((overlaps_plus - max_val_p) / epsilon_smooth)
            obj_p = max_val_p + epsilon_smooth * np.log(np.sum(exp_vals_p))
            
            overlaps_minus = np.correlate(h_minus_clipped, 1 - h_minus_clipped, mode='full') * dx
            max_val_m = np.max(overlaps_minus)
            exp_vals_m = np.exp((overlaps_minus - max_val_m) / epsilon_smooth)
            obj_m = max_val_m + epsilon_smooth * np.log(np.sum(exp_vals_m))
            
            grad[i] = (obj_p - obj_m) / (2 * eps)

        h -= alpha * grad
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        h *= target_sum / current_sum

        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(overlaps)
        if best_c5 is None or current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        if (iter_count + 1) % 100 == 0:
            alpha *= 0.9
            epsilon_smooth *= 0.95

    best_h = np.clip(best_h, 0, 1)
    best_h *= target_sum / np.sum(best_h)
    overlaps = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(overlaps)
    
    return (best_h.tolist(), final_c5, n_points)
