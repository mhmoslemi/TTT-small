import numpy as np
import time

def compute_C5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values
    n_points = len(h)
    target_sum = n_points / 2.0
    assert np.all(h >= 0) and np.all(h <= 1)
    assert np.isclose(np.sum(h), target_sum)

    current_h = h.copy()
    best_h = current_h.copy()
    best_c5 = compute_C5(current_h)

    T = 1.0
    alpha = 0.99
    max_iter = 1000
    sigma = 0.1
    no_improve = 0
    start = time.time()

    for _ in range(max_iter):
        if time.time() - start > budget_s - 1:
            break

        new_h = current_h.copy()
        n_perturb = max(1, int(0.1 * n_points))
        idxs = np.random.choice(n_points, n_perturb, replace=False)
        noise = np.random.normal(0, sigma, size=n_perturb)
        new_h[idxs] = np.clip(new_h[idxs] + noise, 0, 1)
        s = np.sum(new_h)
        delta = (target_sum - s) / n_points
        new_h = np.clip(new_h + delta, 0, 1)

        new_c5 = compute_C5(new_h)
        if new_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - new_c5) / T):
            current_h = new_h
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current_h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        T *= alpha
        if no_improve > 100:
            break

    return (best_h.tolist(), best_c5, n_points)
