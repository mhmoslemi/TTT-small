import numpy as np
import time

def run(seed=42, budget_s=1100):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize_h(h_new, target_sum):
        current_sum = np.sum(h_new)
        if current_sum == 0:
            return h_new
        scale = target_sum / current_sum
        h_new = np.clip(h_new * scale, 0, 1)
        return h_new

    def compute_c5(h, dx):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h_initial.copy()
    current_h = h_initial.copy()
    best_c5 = compute_c5(best_h, dx)
    current_c5 = best_c5

    np.random.seed(seed)
    start_time = time.time()
    temp = 1.0
    alpha = 0.99
    sigma = 0.1
    max_stall = 100
    stall_count = 0

    while time.time() - start_time < budget_s:
        perturb_indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
        h_new = current_h.copy()
        noise = np.random.normal(0, sigma, size=len(perturb_indices))
        h_new[perturb_indices] += noise
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, target_sum)

        new_c5 = compute_c5(h_new, dx)
        if new_c5 < best_c5:
            best_h[:] = h_new
            best_c5 = new_c5
            stall_count = 0
        else:
            stall_count += 1

        delta = new_c5 - current_c5
        if new_c5 < current_c5:
            current_h[:] = h_new
            current_c5 = new_c5
        elif temp > 1e-6 and np.random.rand() < np.exp(-delta / temp):
            current_h[:] = h_new
            current_c5 = new_c5

        temp *= alpha
        if stall_count > max_stall:
            break

    return best_h, best_c5, n_points
