```python
import numpy as np
from scipy.optimize import minimize

def run(budget_s=1100):
    h = initial_h_values.copy()
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def objective(x):
        x = x.reshape(-1)
        ones_h = 1.0 - x
        corr = np.correlate(x, ones_h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    res = minimize(
        fun=objective,
        x0=h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 1000, 'ftol': 1e-10}
    )

    optimized_h = res.x
    c5 = objective(optimized_h)
    return optimized_h, c5, n_points
```
