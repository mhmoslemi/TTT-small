import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h = np.copy(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points

    # Adam parameters
    alpha = 0.001
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    m = np.zeros(n_points)
    v = np.zeros(n_points)
    t = 0

    best_c5 = float('inf')
    best_h = np.copy(h)

    def compute_c5(h_):
        overlaps = np.correlate(h_, 1 - h_, mode='full') * dx
        overlaps_pos = overlaps[n_points - 1 : 2 * n_points - 1]
        return np.max(overlaps_pos)

    # Initial evaluation
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = np.copy(h)

    while time.time() - start_time < budget_s - 1.0:
        t += 1

        # Compute overlaps and find max
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        overlaps_pos = overlaps[n_points - 1 : 2 * n_points - 1]
        max_c5 = np.max(overlaps_pos)
        k_index = np.argmax(overlaps_pos)
        k = k_index  # positive shift

        # Compute gradient for the current k
        grad = np.zeros(n_points)
        for m_idx in range(n_points):
            if m_idx + k < n_points:
                term1 = (1.0 - h[m_idx + k]) * dx
            else:
                term1 = 1.0 * dx

            if m_idx - k >= 0:
                term2 = -h[m_idx - k] * dx
            else:
                term2 = 0.0
            grad[m_idx] = term1 + term2

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps_adam)

        # Update h
        h -= step

        # Clip and project
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)
        h = h_clipped * (n_points / 2) / total

        # Evaluate current
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = np.copy(h)

        # Learning rate decay
        if t % 200 == 0 and current_c5 >= best_c5:
            alpha *= 0.5

    return (best_h.tolist(), best_c5, n_points)
