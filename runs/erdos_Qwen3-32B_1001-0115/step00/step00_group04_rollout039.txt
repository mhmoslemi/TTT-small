    def constraint_sum(h):
        return np.sum(h) - target_sum

    constraints = {'type': 'eq', 'fun': constraint_sum}

    def objective_func(h, beta=20.0):
        h = h.reshape(-1)
        one_minus_h = 1.0 - h
        cross_corr = np.correlate(h, one_minus_h, mode='full') * dx
        exp_values = np.exp(beta * cross_corr)
        return np.log(exp_values.sum()) / beta

    best_h = h_vals.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full') * dx)

    start_time = time.time()
    elapsed_time = 0.0
    beta = 20.0

    while elapsed_time < budget_s - 5:
        result = minimize(
            objective_func,
            x0=best_h,
            args=(beta,),
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        if result.success:
            new_h = result.x
            new_h = np.clip(new_h, 0, 1)
            if not np.isclose(np.sum(new_h), target_sum):
                new_h = new_h / np.sum(new_h) * target_sum
                new_h = np.clip(new_h, 0, 1)
            new_c5 = np.max(np.correlate(new_h, 1 - new_h, mode='full') * dx)
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5
        beta = min(beta * 1.1 + 1.0, 100.0)
        elapsed_time = time.time() - start_time
        if elapsed_time < budget_s - 5 and np.random.rand() < 0.25:
            perturb = np.random.normal(0, 0.05, n_points)
            best_h = (best_h + perturb).clip(0, 1)
            best_h = best_h / np.sum(best_h) * target_sum
            best_h = best_h.clip(0, 1)

    return (best_h, best_c5, n_points)
