def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import scipy.interpolate
    import time

    start_time = time.time()
    np.random.seed(seed)

    # Resolution levels
    resolutions = [32, 64, 128, 256]

    # Initialize
    h = None
    best_h = None
    best_c5 = float('inf')

    # Check if initial_h_values exists
    initial_h_available = 'initial_h_values' in globals()

    for n_points in resolutions:
        if h is None:
            # First resolution
            if initial_h_available:
                # Downsample initial_h_values to current n_points
                x_initial = np.linspace(0, 2, len(initial_h_values))
                x_current = np.linspace(0, 2, n_points)
                f = scipy.interpolate.interp1d(x_initial, initial_h_values, kind='cubic', bounds_error=False, fill_value=(0, 0))
                h_current = f(x_current)
                h_current = np.clip(h_current, 0.0, 1.0)
            else:
                # Initialize uniformly
                h_current = np.full(n_points, 0.5)  # Since sum(h) must be n_points / 2, each is 0.5
        else:

            h_current = f(x_current)

def optimize_resolution(h, n_points, start_time, budget_s):
    import numpy as np
    max_iterations = 500
    alpha = 0.1
    iterations_without_improvement = 0
    best_c5 = float('inf')
    best_h = h.copy()

    for iteration in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            print("Time budget exceeded.")
            break

        # Compute cross-correlate
        correlate_vals = np.correlate(h, 1 - h, mode='full')

        # Find the maximum value and corresponding shift index
        max_idx = np.argmax(correlate_vals)
        max_val = correlate_vals[max_idx]
        k_shift = max_idx - (n_points - 1)

        # Compute gradient for this shift
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = i + k_shift
            if 0 <= j < n_points:
                gradient[i] = 1 - h[j]

        # Gradient descent step
        h_new = h - alpha * gradient

        # Clip to [0,1]
        h_new = np.clip(h_new, 0.0, 1.0)

        # Scale to satisfy sum constraint
        sum_h = np.sum(h_new)
        if sum_h != 0:
            scale = (n_points / 2) / sum_h
            h_new = np.clip(h_new * scale, 0.0, 1.0)

        # Update h
        h = h_new

        # Compute current C5
        current_c5 = max_val * (2.0 / n_points)

        # Check if improved
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            iterations_without_improvement = 0
        else:
            iterations_without_improvement += 1

        # Learning rate decay
        if iteration % 25 == 0:
            alpha *= 0.5

        # Early stopping
        if iterations_without_improvement >= 50:
            break

    return best_h, best_c5

import numpy as np
import scipy.interpolate
import time

def optimize_resolution(h, n_points, start_time, budget_s):
    max_iterations = 500
    alpha = 0.1
    iterations_without_improvement = 0
    best_c5 = float('inf')
    best_h = h.copy()

    for iteration in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            return best_h, best_c5

        # Compute cross-correlate
        correlate_vals = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(correlate_vals)
        current_c5 = max_val * (2.0 / n_points)

        # Check for improvement
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            iterations_without_improvement = 0
        else:
            iterations_without_improvement += 1

        # Find k_shift
        max_idx = np.argmax(correlate_vals)
        k_shift = max_idx - (n_points - 1)

        # Compute gradient
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = i + k_shift
            if 0 <= j < n_points:
                gradient[i] = 1 - h[j]

        # Gradient step
        h_new = h - alpha * gradient
        h_new = np.clip(h_new, 0.0, 1.0)

        # Scale to maintain sum
        sum_h = np.sum(h_new)
        if sum_h != 0:
            scale = (n_points / 2) / sum_h
            h_new = np.clip(h_new * scale, 0.0, 1.0)
        else:
            h_new = np.full(n_points, 0.5)

        h[:] = h_new[:]

        # Decay learning rate
        if iteration % 25 == 0:
            alpha *= 0.5

        # Early stopping
        if iterations_without_improvement >= 50:
            break

    return best_h, best_c5
