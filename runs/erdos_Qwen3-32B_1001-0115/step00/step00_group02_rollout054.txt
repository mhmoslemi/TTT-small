import math
import time
import random

def adjust_sum(h, required_sum):
    current_sum = h.sum()
    if abs(current_sum - required_sum) < 1e-8:
        return h.copy()
    D = required_sum - current_sum
    h += D / len(h)
    np.clip(h, 0.0, 1.0, out=h)
    return h.copy()
