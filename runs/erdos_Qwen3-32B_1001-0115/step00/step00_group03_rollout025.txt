def adjust_h_to_sum_and_bounds(h, target_sum, n_points):
    h_clipped = np.clip(h, 0.0, 1.0)
    current_sum = h_clipped.sum()
    if np.isclose(current_sum, target_sum):

    diff = target_sum - current_sum

    h_clipped = np.clip(h_clipped, 0.0, 1.0)
