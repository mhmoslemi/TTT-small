        result = prob.solve(solver=cp.ECOS)
        h = h_proj.value.copy()
        
        if iter % 100 == 0 and iter > 0:
            eta *= 0.95
    
    corrs = np.correlate(best_h, 1 - best_h, mode="full") * dx
    return (best_h, np.max(corrs), n)
