import numpy as np
import time

def project(h_proj, target_sum):
    h_proj = np.clip(h_proj, 0.0, 1.0)
    current_sum = np.sum(h_proj)
    delta = current_sum - target_sum
    if abs(delta) < 1e-8:
        return h_proj
    if delta > 0:  # Need to decrease
        scaling = (current_sum - delta) / current_sum
        h_proj *= scaling
        h_proj = np.clip(h_proj, 0.0, 1.0)
    else:  # Need to increase
        available = np.sum(1.0 - h_proj)
        if available >= -delta:
            scaling = (-delta) / available
            h_proj += scaling * (1.0 - h_proj)
    return h_proj

def compute_grad_k(h, k_shift):
    N = h.size
    grad = np.zeros(N)
    if k_shift >= 0:
        valid_j = N - k_shift
        for j in range(valid_j):
            grad[j] += 1.0 - h[k_shift + j]
        for j in range(k_shift, N):
            grad[j] -= h[j - k_shift]
    else:
        m = -k_shift
        for j in range(m, N):
            grad[j] += 1.0 - h[j - m]
        for j in range(N - m):
            grad[j] -= h[m + j]
    return grad
