                delta = np.random.uniform(0, max_delta)
                h_new[i] += delta
                h_new[j] -= delta
            else:  # move from i to j
                max_delta = delta_out
                if max_delta <= 0:
                    continue
                delta = np.random.uniform(0, max_delta)
                h_new[i] -= delta
                h_new[j] += delta
            h_new = np.clip(h_new, 0.0, 1.0)
            new_c5 = evaluate(h_new)
            delta_c = new_c5 - current_c5
            if delta_c < 0 or np.random.rand() < np.exp(-delta_c / T):
                h = h_new
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
        T *= alpha
        if T < tolerance:
            break

    return (best_h.tolist(), best_c5, n_points)
