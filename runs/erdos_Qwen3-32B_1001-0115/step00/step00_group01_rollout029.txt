import numpy as np
import time
import random

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    required_sum = n_points / 2.0
    
    # Validate initial_h
    assert abs(np.sum(initial_h) - required_sum) < 1e-6, "Initial h sum is invalid"
    
    current_h = initial_h.copy()
    best_h = current_h.copy()
    best_c5 = np.max(np.correlate(current_h, 1 - current_h, mode="full")) * dx
    
    # Simulated Annealing Parameters
    T = 1.0
    alpha = 0.99
    sigma = 0.1
    no_improvement_threshold = 100
    no_improvement_count = 0
    
    start_time = time.time()
    
    def evaluate(h):
        return np.max(np.correlate(h, 1 - h, mode="full")) * dx
    
    while time.time() - start_time < budget_s - 1:
        # Perturb
        new_h = current_h.copy()
        n_perturb = max(1, int(0.1 * n_points))
        indices = np.random.choice(n_points, size=n_perturb, replace=False)
        noise = np.random.normal(0, sigma, size=n_perturb)
        new_h[indices] += noise
        new_h = np.clip(new_h, 0, 1)
        
        # Normalize sum to required_sum
        current_sum = np.sum(new_h)
        delta = required_sum - current_sum
        new_h += delta / n_points
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h.astype(current_h.dtype)
        
        # Evaluate
        new_c5 = evaluate(new_h)
        
        # Acceptance
        if new_c5 < best_c5:
            current_h = new_h.copy()
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current_h.copy()
                no_improvement_count = 0
        else:
            delta_c5 = new_c5 - best_c5
            if random.random() < np.exp(-delta_c5 / T):
                current_h = new_h.copy()
            else:
                pass
        no_improvement_count += 1
        T *= alpha
        
        # Early stopping
        if no_improvement_count > no_improvement_threshold:
            break
    
    return (best_h.tolist(), best_c5, n_points)
