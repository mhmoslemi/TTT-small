import numpy as np
import numpy.random as npr
from scipy.optimize import minimize
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    h2 = 1 - h
    cross_corr = np.correlate(h, h2, mode='full')
    return np.max(cross_corr) * dx

def generate_random_individual(n_points):
    h = npr.rand(n_points)
    scaling = (n_points / 2) / h.sum()
    h *= scaling
    return np.clip(h, 0, 1)

def project_solution(h, n_points):
    h = np.clip(h, 0, 1)
    target_sum = n_points / 2
    current_sum = h.sum()
    if current_sum == 0:
        return np.zeros_like(h)
    scaling = target_sum / current_sum
    h *= scaling
    return np.clip(h, 0, 1)

def optimize_individual(best_h, n_points):
    def objective(h_flat):
        h = h_flat.reshape((n_points,))
        return compute_c5(h, n_points)
    result = minimize(objective, best_h, method='Nelder-Mead', options={'maxiter': 200})
    if result.success:
        return project_solution(result.x, n_points)
    else:
        return best_h

    population = [h_initial.copy()]
