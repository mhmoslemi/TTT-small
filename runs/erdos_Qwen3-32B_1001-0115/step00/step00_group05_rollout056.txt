def compute_c5(h):

    dx = 2.0 / n_points
    overlaps = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(overlaps)

    n_points = len(h)
    h_clipped = np.clip(h, 0, 1)
    current_sum = h_clipped.sum()
    if np.isclose(current_sum, target_sum):
        return h_clipped

        return h_clipped
    delta = target_sum - current_sum
    h_clipped += delta / adjustable_count * mask.astype(float)
    return np.clip(h_clipped, 0, 1)
