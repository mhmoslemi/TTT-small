    dx = 2.0 / n_pts
    target_sum = n_pts / 2.0

    # Normalize h0 to meet the sum constraint
    current_sum = h0.sum()
    if not np.isclose(current_sum, target_sum, atol=1e-6):
        scale = target_sum / current_sum
        h0 = np.clip(h0 * scale, 0.0, 1.0)
        h_opt_sum = h0.sum()
        delta = target_sum - h_opt_sum
        for i in range(n_pts):
            add = delta / n_pts
            h0[i] += add
            h0[i] = np.clip(h0[i], 0.0, 1.0)
            delta -= add
            if abs(delta) < 1e-9:
                break

    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h, dx)

    # Optimization settings
    beta = 10.0
    start_time = time.time()
    budget_s = 1100

    def objective_smooth(h_flat):
        h = h_flat.reshape((n_pts,))
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        exp_overlaps = np.exp(beta * overlaps)
        return (1.0 / beta) * np.log(exp_overlaps.sum())

    def constraint_sum(h_flat):
        return h_flat.sum() - target_sum

    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_pts)]

    # Iterative optimization
    for iter in range(3):
        if time.time() - start_time > 0.95 * budget_s:
            break
        result = minimize(
            objective_smooth,
            x0=best_h.copy().ravel(),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 50, 'ftol': 1e-6}
        )
        if result.success:
            h_opt = result.x.reshape(n_pts)
            h_opt = np.clip(h_opt, 0.0, 1.0)
            h_sum = h_opt.sum()
            if not np.isclose(h_sum, target_sum, atol=1e-6):
                scale = target_sum / h_sum
                h_opt = np.clip(h_opt * scale, 0.0, 1.0)
                h_sum = h_opt.sum()
                delta = target_sum - h_sum
                for i in range(n_pts):
                    add = delta / n_pts
                    h_opt[i] += add
                    h_opt[i] = np.clip(h_opt[i], 0.0, 1.0)
                    delta -= add
                    if abs(delta) < 1e-9:
                        break
            c5_val = evaluate_c5(h_opt, dx)
            if c5_val < best_c5:
                best_h[:] = h_opt
                best_c5 = c5_val
        beta *= 1.5

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    h_sum = best_h.sum()
    if not np.isclose(h_sum, target_sum, atol=1e-6):
        scale = target_sum / h_sum
        best_h = np.clip(best_h * scale, 0.0, 1.0)
        h_sum = best_h.sum()
        delta = target_sum - h_sum
        for i in range(n_pts):
            add = delta / n_pts
            best_h[i] += add
            best_h[i] = np.clip(best_h[i], 0.0, 1.0)
            delta -= add
            if abs(delta) < 1e-9:
                break

    best_c5 = evaluate_c5(best_h, dx)

    return (best_h.tolist(), best_c5, n_pts)
