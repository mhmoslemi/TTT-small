import numpy as np
import cvxpy as cp
import time

def project(h, sum_h):
    n = len(h)
    h_proj = cp.Variable(n)
    objective = cp.Minimize(cp.sum_squares(h_proj - h))
    constraints = [cp.sum(h_proj) == sum_h, 0 <= h_proj, h_proj <= 1]
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS)
    return h_proj.value
