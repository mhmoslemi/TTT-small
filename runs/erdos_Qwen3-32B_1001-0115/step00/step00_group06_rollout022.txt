import numpy as np
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    # Use the initial_h_values provided
    h = initial_h_values.copy().astype(np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # Ensure sum(h) == n_points / 2

    # Initial C5 evaluation
    overlaps = np.correlate(h, 1 - h, mode='full') * dx
    best_c5 = np.max(overlaps)
    best_h = h.copy()
    best_time = time.time()

    # Prepare optimization variables
    # The main loop
    import time
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Compute current overlaps and find the active k_shift
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        pos_max = np.argmax(overlaps)
        m = pos_max
        k_shift = m - (n_points - 1)  # lag in sample steps

        # Compute the active gradient for this k_shift
        grad = np.zeros(n_points)
        for i in range(n_points):
            i_minus_k = (i - k_shift) % n_points
            i_plus_k = (i + k_shift) % n_points
            grad[i] = dx * (1.0 - (h[i_minus_k] + h[i_plus_k]))

        # Formulate the LP to minimize grad^T * h_new
        h_new_var = cp.Variable(n_points)
        constraints = [0 <= h_new_var, h_new_var <= 1.0, cp.sum(h_new_var) == target_sum]
        objective = cp.Minimize(cp.sum(cp.multiply(grad, h_new_var)))
        problem = cp.Problem(objective, constraints)
        try:
            problem.solve(solver=cp.GLPK, verbose=False)
        except:
            # Fallback in case solver fails
            problem.solve(solver=cp.OSQP, verbose=False)

        if problem.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            break  # Couldn't solve, break loop

        h_candidate = h_new_var.value

        # Check if h_candidate is better
        overlaps_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full') * dx
        c5_candidate = np.max(overlaps_candidate)
        if c5_candidate < best_c5:
            best_h = h_candidate.copy()
            best_c5 = c5_candidate
            best_time = time.time()
        # Take a step towards h_candidate
        # Simple line search: step size alpha
        alpha = 1.0
        h_new = h + alpha * (h_candidate - h)
        # Enforce constraints manually
        h_new = np.clip(h_new, 0.0, 1.0)
        h_sum = h_new.sum()
        if not np.isclose(h_sum, target_sum, atol=1e-6):
            # Adjust the sum to be exact
            # Find one element to adjust
            # For simplicity, adjust the first element
            diff = h_sum - target_sum
            h_new[0] -= diff
        h = h_new

        # Check if improvement
        overlaps_new = np.correlate(h, 1 - h, mode='full') * dx
        c5_new = np.max(overlaps_new)
        if c5_new >= best_c5:
            # No improvement, reduce step size or revert
            h = best_h.copy()
            alpha *= 0.5
            if alpha < 1e-4:
                break

import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy().astype(np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    overlaps = np.correlate(h, 1 - h, mode='full') * dx
    best_c5 = np.max(overlaps)
    best_h = h.copy()

    start_time = time.time()

    while time.time() - start_time < budget_s:
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        pos_max = np.argmax(overlaps)
        m = pos_max
        k_shift = m - (n_points - 1)

        # Vectorized gradient computation
        i_minus_k = np.arange(n_points) - k_shift
        i_minus_k = np.mod(i_minus_k, n_points)
        i_plus_k = np.arange(n_points) + k_shift
        i_plus_k = np.mod(i_plus_k, n_points)
        grad = dx * (1.0 - (h[i_minus_k] + h[i_plus_k]))

        # Form and solve the LP
        h_new_var = cp.Variable(n_points)
        constraints = [0 <= h_new_var, h_new_var <= 1.0, cp.sum(h_new_var) == target_sum]
        objective = cp.Minimize(cp.sum(cp.multiply(grad, h_new_var)))
        problem = cp.Problem(objective, constraints)

        try:
            problem.solve(solver=cp.GLPK, verbose=False)
        except:
            problem.solve(solver=cp.OSQP, verbose=False)

        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_candidate = h_new_var.value
            # Line search
            alpha = 1.0
            h_candidate = np.clip(h_candidate, 0.0, 1.0)
            h = (1 - alpha) * h + alpha * h_candidate
            h_sum = h.sum()
            diff = h_sum - target_sum
            h[0] -= diff
            h = np.clip(h, 0.0, 1.0)

            overlaps_candidate = np.correlate(h, 1 - h, mode='full') * dx
            c5_candidate = np.max(overlaps_candidate)
            if c5_candidate < best_c5:
                best_h = h.copy()
                best_c5 = c5_candidate

    return best_h, best_c5, n_points
