import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    dx = 2.0 / n_points

    def compute_c5(h_curr):
        one_minus_h = 1.0 - h_curr
        corr_result = np.correlate(h_curr, one_minus_h, mode='full')
        return np.max(corr_result) * dx

    def project(h_curr):
        h_clipped = np.clip(h_curr, 0.0, 1.0)
        sum_target = float(n_points) / 2.0
        sum_clipped = h_clipped.sum()
        if sum_clipped == 0.0:
            scaling = 1.0
        else:
            scaling = sum_target / sum_clipped
        h_rescaled = h_clipped * scaling
        return np.clip(h_rescaled, 0.0, 1.0)

    def compute_gradient(h_curr, delta=1e-5):
        n = len(h_curr)
        grad = np.zeros(n)
        c5_base = compute_c5(h_curr)
        for i in range(n):
            h_perturbed = h_curr.copy()
            h_perturbed[i] += delta
            c5_perturbed = compute_c5(h_perturbed)
            grad[i] = (c5_perturbed - c5_base) / delta
        return grad

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    learning_rate = 0.05
    iteration = 0

    while time.time() - start_time < budget_s - 1.0:
        grad = compute_gradient(h)
        h -= learning_rate * grad
        h = project(h)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        if iteration % 100 == 0:
            learning_rate *= 0.95
        iteration += 1

    assert np.all((best_h >= 0) & (best_h <= 1)), "h out of bounds"
    assert abs(best_h.sum() - n_points / 2.0) < 1e-6, "sum constraint violated"

    return best_h.tolist(), best_c5, n_points
