    h = np.clip(h, 0.0, 1.0)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    T = 1.0  # Initial temperature
    alpha = 0.99  # Cooling rate
    delta_max = 0.05  # Perturbation step size

    while time.time() - start_time < budget_s - 1:
        new_h = perturb_h(h, delta_max)
        new_h = adjust_sum_to_target(new_h, n_points, target_sum)
        
        new_c5 = compute_c5(new_h, dx)
        delta_c5 = new_c5 - best_c5

        if delta_c5 < 0:
            h = new_h.copy()
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            prob = np.exp(-delta_c5 / T)
            if np.random.rand() < prob:
                h = new_h.copy()
        
        T *= alpha
        
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final constraint enforcement
    best_h = adjust_sum_to_target(best_h, n_points, target_sum)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h, dx)
    
    return best_h, final_c5, n_points
