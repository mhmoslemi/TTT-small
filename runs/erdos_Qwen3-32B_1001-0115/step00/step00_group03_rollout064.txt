                new_c5 = float(new_c5)
                
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    improved = True
                else:
                    best_h[i], best_h[j] = temp_i, temp_j
            
            if not improved:
                break
        
        if time.time() - start_time >= budget_s:
            break
    
    return best_h.tolist(), best_c5, n_points
