import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # ∑h = n_points / 2

    alpha = 1e-4
    start_time = time.time()
    best_c5 = float('inf')
    best_h = h.copy()

    while time.time() - start_time < budget_s:
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full') * dx

        # Evaluate current max C₅
        current_c5 = np.max(corr)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Find all lags achieving the current maximum
        max_indices = np.where(corr == current_c5)[0]
        gradient = np.zeros(n_points)

        # Compute gradients for all max lags
        for m in max_indices:
            # Shift k corresponding to index m in 'full' mode
            k = m - (n_points - 1)
            # Use circular shift (approximation for zero-padding)
            shifted_h = np.roll(h, shift=k)
            grad = (1 - shifted_h) * dx
            gradient += grad

        if len(max_indices) > 0:
            gradient /= len(max_indices)

        # Update h
        h -= alpha * gradient
        np.clip(h, 0, 1, out=h)
        current_sum = np.sum(h)
        if current_sum != 0:
            h *= (target_sum / current_sum)
            np.clip(h, 0, 1, out=h)

    return best_h, best_c5, n_points
