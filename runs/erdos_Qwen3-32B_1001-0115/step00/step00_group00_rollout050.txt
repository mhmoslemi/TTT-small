import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective_func(h_flat):

        one_minus_h = 1 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx

    constraints = [{'type': 'eq', 'fun': lambda h: h.sum() - target_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    try:
        result = scipy.optimize.minimize(
            fun=objective_func,
            x0=h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 10000, 'ftol': 1e-12}
        )
    except Exception:
        pass  # Fallback to initial in case of failure
    else:
        elapsed = time.time() - start_time
        if elapsed < budget_s:
            h_opt = np.clip(result.x, 0.0, 1.0)
            current_sum = h_opt.sum()
            if abs(current_sum - target_sum) > 1e-6:
                scale_factor = target_sum / current_sum
                h_opt = np.clip(h_opt * scale_factor, 0.0, 1.0)
            one_minus_h = 1 - h_opt
            corr = np.correlate(h_opt, one_minus_h, mode='full')
            c5_bound = np.max(corr) * dx
            return h_opt, c5_bound, n_points

    # Fallback to initial values if optimization fails or time runs out
    h_opt = h_initial.copy()
    h_opt = h_opt / h_opt.sum() * target_sum
    h_opt = np.clip(h_opt, 0.0, 1.0)
    one_minus_h = 1 - h_opt

    c5_bound = np.max(corr) * dx
