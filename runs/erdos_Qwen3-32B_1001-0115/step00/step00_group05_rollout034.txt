    start_time = time.time()
    h = np.array(initial_h_values, copy=True, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    total_sum = n_points / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    step_size = 0.1
    iteration = 0
    decay_steps = 100
    decay_rate = 0.95

    while time.time() - start_time < budget_s:
        iteration += 1
        # Compute current cross-correlation
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = (corr * dx).max()
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h[:] = h.copy()
        # Determine active lags
        max_val = (corr * dx).max()
        active_indices = [
            i for i in range(len(corr))
            if abs((corr[i] * dx) - max_val) < 1e-10
        ]
        active_lags = []
        for idx in active_indices:
            lag = idx - (n_points - 1)
            active_lags.append(lag)
        # Compute gradient
        grad = np.zeros(n_points)
        for lag in active_lags:
            start_i = max(0, -lag)
            end_i = min(n_points - 1, n_points - 1 - lag)
            for i in range(start_i, end_i + 1):
                j = i + lag
                grad[i] += (1 - h[j])
                grad[j] += (-h[i])
        # Gradient step and clipping
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        # Project onto feasible region
        h_proj = cp.Variable(n_points)
        constraints = [0 <= h_proj, h_proj <= 1, cp.sum(h_proj) == total_sum]
        objective = cp.Minimize(cp.sum_squares(h_proj - h_new))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=100, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h[:] = h_proj.value
        except:
            pass
        # Step decay
        if iteration % decay_steps == 0:
            step_size *= decay_rate
    # Final projection to ensure feasibility
    h_proj = cp.Variable(n_points)
    constraints = [0 <= h_proj, h_proj <= 1, cp.sum(h_proj) == total_sum]
    objective = cp.Minimize(cp.sum_squares(h_proj - best_h))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=100, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            best_h[:] = h_proj.value
    except:
        pass
    final_c5 = compute_c5(best_h, dx)
    return best_h.tolist(), final_c5, n_points
