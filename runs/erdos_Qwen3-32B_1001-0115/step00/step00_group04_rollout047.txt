import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    if 'initial_h_values' in globals():
        h0 = initial_h_values.copy()
        n_points = len(h0)
    else:
        n_points = 42
        h0 = np.random.rand(n_points)
        h0 = h0 / h0.sum() * n_points / 2

    # Normalize initial solution and clip to [0, 1]
    h0 = np.clip(h0, 0, 1)
    h0_total = h0.sum()
    h0 = h0 / h0_total * (n_points / 2) if h0_total != 0 else h0

    # Initialize population
    pop_size = 50
    pop = [h0.copy()]
    while len(pop) < pop_size:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        total = h.sum()
        if total != 0:
            h = h / total * (n_points / 2)
        pop.append(h)

    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h, n_points)

    generation = 0
    max_generations = 1000
    patience = 50
    no_improvement = 0

    while time.time() - start_time < budget_s and generation < max_generations and no_improvement < patience:
        # Evaluate all candidates
        fitness = [evaluate_c5(h, n_points) for h in pop]
        current_best_index = np.argmin(fitness)
        current_best_c5 = fitness[current_best_index]

        # Update best solution
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = pop[current_best_index].copy()
            no_improvement = 0
        else:
            no_improvement += 1

        # Sort population by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        num_elites = int(0.2 * pop_size)
        elites = [pop[i] for i in sorted_indices[:num_elites]]
        num_parents = int(0.5 * pop_size)
        parents = [pop[i] for i in sorted_indices[:num_parents]]

        new_pop = elites.copy()
        while len(new_pop) < pop_size:
            # Select parents
            p1_idx = np.random.randint(len(parents))
            p2_idx = np.random.randint(len(parents))
            parent1 = parents[p1_idx].copy()
            parent2 = parents[p2_idx].copy()

            # Crossover
            child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)

            # Mutation
            if np.random.rand() < 0.1:
                mut_indices = np.random.rand(n_points) < 0.1
                noise = np.random.randn(n_points)
                child[mut_indices] += 0.1 * noise[mut_indices]
                child = np.clip(child, 0, 1)

            # Normalize
            total = child.sum()
            if total != 0:
                child = child / total * (n_points / 2)
            else:
                child.fill(0)
                child[0] = n_points / 2

            new_pop.append(child)

        pop = new_pop
        generation += 1

    return best_h, best_c5, n_points

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    ones_minus_h = 1.0 - h
    corr = np.correlate(h, ones_minus_h, mode='full')
    return np.max(corr) * dx
