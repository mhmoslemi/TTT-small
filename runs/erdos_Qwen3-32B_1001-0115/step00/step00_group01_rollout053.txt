        for m in max_indices:
            k = m - (n_points - 1)
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    gradients[i] += (1.0 - h[j]) * dx
                    gradients[j] += (-h[i]) * dx

        h -= learning_rate * gradients
        np.clip(h, 0.0, 1.0, out=h)
        current_sum = np.sum(h)
        delta = target_sum - current_sum
        h += delta / n_points
        np.clip(h, 0.0, 1.0, out=h)

    cross_corr = np.correlate(best_h, 1.0 - best_h, mode='full') * dx
    c5_bound = np.max(cross_corr)

    return (best_h.tolist(), c5_bound, n_points)
