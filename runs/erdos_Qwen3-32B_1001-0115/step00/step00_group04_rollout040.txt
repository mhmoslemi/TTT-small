import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = h / np.sum(h) * target_sum

    def compute_overlap(h):
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return overlaps

    beta = 50.0
    def objective(h_vec):
        h = h_vec.ravel()
        overlaps = compute_overlap(h)
        return logsumexp(beta * overlaps) / beta

    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]

    best_h = h.copy()
    best_C5 = np.max(compute_overlap(best_h))
    start_time = time.time()

    cycles = 0
    while time.time() - start_time < budget_s - 10 and cycles < 10:
        perturbation = np.random.normal(0, 0.05, n_points)
        current_h = np.clip(best_h + perturbation, 0, 1)
        current_h = current_h / np.sum(current_h) * target_sum

        result = minimize(
            objective,
            current_h,
            bounds=bounds,
            constraints=constraints,
            method='L-BFGS-B',
            options={'maxiter': 20, 'ftol': 1e-6}
        )
        if result.success:
            prop_h = result.x
        else:
            prop_h = current_h

        prop_h = np.clip(prop_h, 0, 1)
        s = np.sum(prop_h)
        prop_h = prop_h * target_sum / s

        overlaps = compute_overlap(prop_h)
        curr_C5 = np.max(overlaps)
        if curr_C5 < best_C5:
            best_h = prop_h
            best_C5 = curr_C5

        cycles += 1

    # Final polish
    final_result = minimize(
        objective,
        best_h,
        bounds=bounds,
        constraints=constraints,
        method='L-BFGS-B',
        options={'maxiter': 25, 'ftol': 1e-7}
    )
    if final_result.success:
        best_h = final_result.x

    best_h = np.clip(best_h, 0, 1)
    s = np.sum(best_h)
    if not np.isclose(s, target_sum):
        best_h = best_h * target_sum / s

    overlaps = compute_overlap(best_h)
    final_C5 = np.max(overlaps)

    return best_h, final_C5, n_points
