        result = minimize(
            objective,
            x0=h0,
            bounds=bounds,
            constraints=sum_constraint,
            method="SLSQP",
            options=options,
            callback=callback
        )
        if result.success:
            final_obj = result.fun
            if final_obj < best_c5:
                best_h[:] = result.x
                best_c5 = final_obj
    except RuntimeError:
        pass

    finally:
        return (best_h.tolist(), best_c5, n_points)
