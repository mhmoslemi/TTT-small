import numpy as np
import time

def project(h, target_sum):
    h_clipped = np.clip(h, 0, 1)
    current_sum = h_clipped.sum()
    delta = current_sum - target_sum
    if abs(delta) < 1e-8:
        return h_clipped
    n = len(h)
    if delta > 0:  # need to reduce sum
        idxs_one = h_clipped == 1.0
        if idxs_one.sum() > 0:
            amount = delta / idxs_one.sum()
            h_clipped[idxs_one] -= amount
        else:
            amount = delta / n
            h_clipped -= amount
    else:  # need to increase sum
        idxs_zero = h_clipped == 0.0
        if idxs_zero.sum() > 0:
            amount = -delta / idxs_zero.sum()
            h_clipped[idxs_zero] += amount
        else:
            amount = -delta / n
            h_clipped += amount
    return h_clipped
