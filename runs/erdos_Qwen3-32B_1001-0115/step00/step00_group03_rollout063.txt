import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100):
    # Use pre-imported initial_h_values
    h0 = np.copy(initial_h_values)
    n_points = len(h0)

    np.random.seed(seed)
    swarm_size = 50
    particles = np.zeros((swarm_size, n_points))
    velocities = 0.1 * np.random.randn(swarm_size, n_points)
    particles[0] = h0.copy()
    for i in range(1, swarm_size):
        particles[i] = np.clip(particles[0] + 0.1 * np.random.randn(n_points), 0, 1)
        # Rescale to satisfy sum constraint
        s = particles[i].sum()
        particles[i] *= (n_points / 2) / s

    # Evaluate initial C5 for all particles
    c5_vals = np.array([evaluate_erdos_solution(p) for p in particles])
    p_best = particles.copy()
    p_best_vals = c5_vals.copy()
    g_best_idx = np.argmin(p_best_vals)
    g_best = particles[g_best_idx].copy()
    g_best_c5 = p_best_vals[g_best_idx]

    # PSO parameters
    max_iter = 1000
    omega = 0.7
    omega_min = 0.4
    c1, c2 = 2.0, 2.0
    v_max = 1.0

    # Track time
    start_time = time.time()
    no_improvement = 0
    max_no_improve = 100

    # Main PSO loop
    for iter in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 1.0:
            break

        # Adjust omega
        omega = max(omega_min, omega - (0.7 - 0.4) * (elapsed / budget_s))

        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (omega * velocities[i] +
                            c1 * r1 * (p_best[i] - particles[i]) +
                            c2 * r2 * (g_best - particles[i]))
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)

            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)

            # Enforce sum constraint
            s = particles[i].sum()
            if abs(s) > 1e-10:  # Prevent division by zero
                particles[i] *= (n_points / 2) / s

            # Evaluate
            c5 = evaluate_erdos_solution(particles[i])
            c5_vals[i] = c5

            # Update personal best
            if c5 < p_best_vals[i]:

                p_best_vals[i] = c5
                # Update global best
                if c5 < g_best_c5:
                    g_best = particles[i].copy()
                    g_best_c5 = c5
                    no_improvement = 0

            else:
                no_improvement += 1

        # Early stopping if no improvement
        if no_improvement >= max_no_improve:
            break

    return (g_best.tolist(), g_best_c5, n_points)
