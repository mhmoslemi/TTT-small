from time import time

def run(seed=42, budget_s=1100, **kwargs):
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    
    dx = 2.0 / n_points
    
    # Rescale h to ensure sum is n_points / 2
    sum_h = np.sum(h)
    scaling_factor = (n_points / 2.0) / sum_h
    h *= scaling_factor
    
    # Define objective function
    def objective(h_flat):
        h_reshaped = h_flat.reshape(-1)
        g = 1 - h_reshaped
        corr = np.correlate(h_reshaped, g, mode='full') * dx
        return np.max(corr)
    
    # Constraints and bounds
    cons = ({'type': 'eq', 'fun': lambda h_flat: np.sum(h_flat) - n_points / 2})
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial guess
    h_initial = h.copy()
    
    start_time = time()
    max_time = budget_s * 0.9  # leave some buffer
    
    while time() - start_time < max_time:
        # Run one iteration of optimization
        result = minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 10}
        )
        
        # Update h_initial with result
        h_initial = result.x.copy()
    
    # Final result is the best found during iterations
    best_h = h_initial.copy()
