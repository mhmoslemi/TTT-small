    n_points = len(h)

    dx = 2.0 / n_points

    best_h = h.copy()
    best_C5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    current_h = h.copy()
    current_C5 = best_C5

    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    T = 1.0
    alpha = 0.999
    delta = 0.1
    kill_time = start_time + budget_s

    while time.time() < kill_time:
        i, j = random.sample(range(n_points), 2)
        delta_val = (random.random() - 0.5) * 2 * delta
        h_proposed = current_h.copy()
        h_proposed[i] += delta_val
        h_proposed[j] -= delta_val
        h_proposed = adjust_and_clip(h_proposed, n_points)

        new_C5 = np.max(np.correlate(h_proposed, 1 - h_proposed, mode='full')) * dx
        dC = current_C5 - new_C5

        if dC > 0 or random.random() < np.exp(dC / max(1e-9, T)):
            current_h = h_proposed.copy()
            current_C5 = new_C5
            if new_C5 < best_C5:
                best_h = current_h.copy()
                best_C5 = new_C5
                T = max(1.0, T * 1.5)
            else:
                T *= alpha
        else:
            T *= alpha

    assert np.isclose(np.sum(best_h), n_points / 2.)
    return best_h.tolist(), best_C5, n_points
