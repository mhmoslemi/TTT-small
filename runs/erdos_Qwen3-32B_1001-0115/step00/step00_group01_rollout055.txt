    learning_rate = 1e-3
    iterations = 0
    max_iterations = 10000  # Adjust based on budget

    while iterations < max_iterations and time.time() - start_time < budget_s:
        iterations += 1
        # Compute overlap integrals
        corr = np.correlate(h, 1.0 - h, mode='full')
        corr *= dx
        current_c5 = np.max(corr)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

        # Find indices where current_c5 is achieved
        max_indices = np.where(np.isclose(corr, current_c5, atol=1e-8))[0]
        grads = np.zeros_like(h)

        for m in max_indices:
            k = m - (n_points - 1)
            for j in range(n_points):

                term1 = 1.0
                if j + k < n_points:

        # Apply gradient descent
        h -= learning_rate * grads

        # Project onto constraints
        # Clip
