    while time.time() - start_time < budget_s:
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = w * velocities[i] + c1 * r1 * (p_best[i] - particles[i]) + c2 * r2 * (g_best - particles[i])
            particles[i] = particles[i] + velocities[i]
            particles[i] = project(particles[i], target_sum)

        current_scores = [compute_c5(p) for p in particles]
        current_scores = np.array(current_scores)
        for i in range(n_particles):
            if current_scores[i] < p_best_scores[i]:
                p_best[i] = particles[i].copy()
                p_best_scores[i] = current_scores[i]
        current_g_idx = np.argmin(p_best_scores)
        if p_best_scores[current_g_idx] < g_best_score:
            g_best[...] = p_best[current_g_idx][...]
            g_best_score = p_best_scores[current_g_idx]

        if g_best_score <= 0.38085:
            break

    return (g_best, g_best_score, n_points)
