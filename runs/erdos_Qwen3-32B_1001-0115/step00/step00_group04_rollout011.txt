import numpy as np
import time

def normalize_h(h, desired_sum):
    h_clipped = np.clip(h, 0, 1)
    current_sum = h_clipped.sum()
    if current_sum == 0:
        h_clipped[:] = 0.5
    else:
        scaling = desired_sum / current_sum
        h_clipped *= scaling
        h_clipped = np.clip(h_clipped, 0, 1)
    return h_clipped

def evaluate_individual(h, n_points):
    dx = 2.0 / n_points
    ones_minus_h = 1.0 - h
    corr = np.correlate(h, ones_minus_h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    desired_sum = n_points / 2.0
    population_size = 100
    mutation_prob = 0.1
    mutation_strength = 0.1
    elite_ratio = 0.2
    num_elites = int(population_size * elite_ratio)
    start_time = time.time()

    population = [h_init.copy()]
    for _ in range(population_size - 1):
        h = h_init.copy()
        h += np.random.normal(0, mutation_strength, h.shape)
        h = normalize_h(h, desired_sum)
        population.append(h)
    population = np.array(population)

    best_h = h_init.copy()
    best_c5 = evaluate_individual(best_h, n_points)

    while time.time() - start_time < budget_s:
        c5_values = []
        for indiv in population:
            c5 = evaluate_individual(indiv, n_points)
            c5_values.append(c5)
