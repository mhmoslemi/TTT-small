        if len(indices) == 0:
            return adjusted
        d = delta / len(indices)
        adjusted[indices] += d
        adjusted[indices] = np.clip(adjusted[indices], 0, 1)
    else:
        indices = np.where(adjusted >= eps)[0]
        if len(indices) == 0:
            return adjusted
        d = delta / len(indices)
        adjusted[indices] += d
        adjusted[indices] = np.clip(adjusted[indices], 0, 1)
    return adjusted

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    try:
        initial_h = initial_h_values
        initial_h = np.array(initial_h, dtype=np.float64)
        n_points = len(initial_h)
        target_sum = n_points / 2
        if not np.isclose(np.sum(initial_h), target_sum):
            initial_h = adjust_sum(np.clip(initial_h, 0, 1), target_sum)
    except NameError:
        n_points = 70
        target_sum = n_points / 2
        h = np.random.rand(n_points)
        h = adjust_sum(np.clip(h / h.sum() * target_sum, 0, 1), target_sum)
        initial_h = h
    
    n_particles = 20
    w, c1, c2 = 0.7, 1.5, 1.5
    
    if 'initial_h' in locals():
        particles = np.tile(initial_h, (n_particles, 1)).copy()
    else:
        particles = np.random.rand(n_particles, n_points)
        for i in range(n_particles):
            particles[i] = adjust_sum(np.random.rand(n_points), target_sum)
    
    for i in range(n_particles):
        noise = np.random.normal(0, 0.05, size=n_points)
        particles[i] = adjust_sum(np.clip(particles[i] + noise, 0, 1), target_sum)
    
    velocities = np.random.rand(n_particles, n_points) * 0.1
    
    personal_best = particles.copy()
    personal_best_c5 = np.array([compute_c5(p, 2.0 / n_points) for p in particles])
    global_best_idx = np.argmin(personal_best_c5)
    global_best = personal_best[global_best_idx].copy()
    best_c5 = personal_best_c5[global_best_idx]
    
    max_iterations = 1000
    iteration, patience = 0, 0
    max_patience = 100
    
    while time.time() - start_time < budget_s and iteration < max_iterations and patience < max_patience:
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = w * velocities[i] + c1 * r1 * (personal_best[i] - particles[i]) + c2 * r2 * (global_best - particles[i])
            particles[i] += velocities[i]
            particles[i] = adjust_sum(np.clip(particles[i], 0, 1), target_sum)
            
            current_c5 = compute_c5(particles[i], 2.0 / n_points)
            if current_c5 < personal_best_c5[i]:
                personal_best[i] = particles[i].copy()
                personal_best_c5[i] = current_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    global_best = particles[i].copy()
                    patience = 0
                else:
                    patience += 1
            else:
                patience += 1
        iteration += 1
    
    return global_best.tolist(), best_c5, n_points
