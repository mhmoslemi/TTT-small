import numpy as np
from time import time

def run(seed=42, budget_s=1100, **kwargs):
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    assert np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), target_sum), "Initial h invalid"

    start_time = time()
    learning_rate = 0.1
    decay_rate = 0.998
    best_h = h.copy()
    best_c5 = float('inf')

    def compute_c5(h_val):
        return (np.correlate(h_val, 1 - h_val, mode='full') * dx).max()

    def project(h_val):
        h_clipped = np.clip(h_val, 0, 1)
        current_sum = h_clipped.sum()
        if np.isclose(current_sum, target_sum):
            return h_clipped
        delta = (target_sum - current_sum) / n_points
        h_clipped += delta
        h_clipped = np.clip(h_clipped, 0, 1)
        return h_clipped

    while time() - start_time < budget_s - 0.1:
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        k_values = np.arange(-n_points + 1, n_points)
        max_idx = np.argmax(overlaps[k_values <= n_points - 1])
        k_star = k_values[max_idx]

        grad = np.zeros_like(h)
        if k_star > 0:
            k = k_star
            grad[:n_points - k] += 1 - h[k:]
            grad[k:] -= h[:-k]
        elif k_star < 0:
            m = -k_star
            grad[m:] += 1 - h[:n_points - m]
            grad[:n_points - m] -= h[m:]

        grad /= (np.linalg.norm(grad) + 1e-8)
        h -= learning_rate * grad
        h = project(h)

        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        learning_rate *= decay_rate

    return best_h.tolist(), best_c5, n_points
