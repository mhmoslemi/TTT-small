        delta1 = min(1 - h[i], h[j])
        delta2 = min(1 - h[j], h[i])
        if delta1 <= 0 and delta2 <= 0:
            continue
        if np.random.rand() < 0.5:
            delta = np.random.uniform(0, delta1)
            new_h = h.copy()
            new_h[i] += delta
            new_h[j] -= delta
        else:
            delta = np.random.uniform(0, delta2)
            new_h = h.copy()
            new_h[j] += delta
            new_h[i] -= delta
        return new_h

def run(seed=42, budget_s=1100, **kwargs):
    try:
        h = initial_h_values.copy()
    except NameError:
        n_points = 95
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1.0
        remaining = n_points / 2 - half
        h[half] = remaining
    n_points = len(h)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    np.random.seed(seed)
    T = 1.0
    alpha = 0.99
    time_start = time.time()
    while True:
        current_h = best_h.copy()
        new_h = perturb_h(current_h)
        assert np.all(new_h >= 0) and np.all(new_h <= 1)
        assert np.isclose(np.sum(new_h), n_points / 2)
        new_c5 = evaluate_erdos_solution(new_h)
        delta_c5 = new_c5 - best_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            best_h[:] = new_h[:]
            best_c5 = new_c5
        T *= alpha
        if time.time() - time_start > budget_s - 1:
            break
    return best_h, best_c5, n_points
