import time
import math

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def perturb_h(h):
    n = h.size
    i, j = np.random.choice(n, 2, replace=False)
    delta_ij = min(h[i], 1.0 - h[j])
    delta_ji = min(h[j], 1.0 - h[i])
    if np.random.rand() < 0.5:
        if delta_ij > 0:
            delta = np.random.uniform(0, delta_ij)
            h[i] -= delta
            h[j] += delta
    else:
        if delta_ji > 0:
            delta = np.random.uniform(0, delta_ji)
            h[j] -= delta
            h[i] += delta
