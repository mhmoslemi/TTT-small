                best_c5 = candidate_c5

        best_h = np.clip(best_h + 0.05 * np.random.randn(n_points), 0, 1)
        best_h /= best_h.sum() / target_sum
    
    # Final projection
    optimized_h = np.clip(best_h, 0, 1)
    optimized_h /= optimized_h.sum() / target_sum
    final_c5 = objective(optimized_h)
    
    return optimized_h, final_c5, n_points
