            velocities[i] = np.clip(
                omega * velocities[i] + 
                c1 * r1 * (p_best[i] - particles[i]) + 
                c2 * r2 * (g_best - particles[i]),
                -v_max, v_max
            )
            particles[i] = np.clip(particles[i] + velocities[i], 0, 1)
            # Rescale to maintain sum constraint
            current_sum = particles[i].sum()
            if current_sum != 0:
                particles[i] = np.clip(particles[i] * (target_sum / current_sum), 0, 1)

            # Evaluate and update bests
            new_c5 = evaluate_erdos_solution(particles[i])
            if new_c5 < p_best_c5[i]:
                p_best[i] = particles[i].copy()
                p_best_c5[i] = new_c5
                if new_c5 < g_best_c5:
                    g_best[:] = particles[i].copy()
                    g_best_c5 = new_c5
                    no_improvement = 0
                else:
                    no_improvement += 1
            else:
                no_improvement += 1

        if no_improvement > max_no_improvement:
            break

    final_c5 = evaluate_erdos_solution(g_best)
    return (g_best, final_c5, n_points)
