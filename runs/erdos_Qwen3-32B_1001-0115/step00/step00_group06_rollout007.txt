import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64, copy=True)
    else:
        n_points = 95
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
        h[n_points//2] = 0.5
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    # Simulated Annealing parameters
    T = 1.0
    T_min = 1e-4
    alpha = 0.995
    while (time.time() - start_time) < budget_s and T > T_min:
        for _ in range(100):
            if (time.time() - start_time) >= budget_s:
                break
            h_candidate = h.copy()
            i, j = np.random.choice(n_points, 2, replace=False)
            a, b = h_candidate[i], h_candidate[j]
            direction = np.random.randint(0, 2)
            if direction == 0:
                delta_max = min(1.0 - a, b)
            else:
                delta_max = min(1.0 - b, a)
            if delta_max <= 1e-10:
                continue
            delta = np.random.uniform(0, delta_max)
            if direction == 0:
                h_candidate[i] += delta
                h_candidate[j] -= delta
            else:
                h_candidate[j] += delta
                h_candidate[i] -= delta
            current_c5 = evaluate_erdos_solution(h_candidate)
            delta_c5 = current_c5 - best_c5
            if delta_c5 < 0:
                h = h_candidate.copy()
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_candidate.copy()
            else:
                prob = np.exp(-delta_c5 / T)
                if np.random.rand() < prob:
                    h = h_candidate.copy()
        T *= alpha
    return best_h, best_c5, n_points
