```python
def project_feasible(h, n_points):
    sum_n = n_points / 2
    current_sum = np.sum(h)
    
    # Project to feasible region using least squares
    if current_sum != sum_n:
        scale = sum_n / current_sum
        h *= scale
        h = np.clip(h, 0.0, 1.0)
        h = h * sum_n / np.sum(h)  # Re-normalize
    return h
```
