def initialize_population(initial_h, n_points, pop_size=30):
    if initial_h is not None:
        population = np.tile(initial_h, (pop_size, 1))
        # Add diverse perturbations
        epsilon = 0.15
        perturb = np.random.uniform(-epsilon, epsilon, size=population.shape)
        population = np.clip(population + perturb, 0.0, 1.0)
        # Normalize to ensure sum constraint
        sums = np.sum(population, axis=1, keepdims=True)

def compute_fitness(h, n_points):
    return -compute_c5(h, n_points)

def select_parents(population, fitnesses, tournament_size):
    winner_indices = []
    for _ in range(len(population)//2):
        tournament = np.random.choice(len(population), tournament_size)
        winner_idx = tournament[np.argmax([fitnesses[i] for i in tournament])]
        winner_indices.append(winner_idx)
    return population[winner_indices]

def blend_crossover(h1, h2, n_points):
    alpha = np.random.uniform(0.2, 0.8, n_points)
    return h1 * alpha + h2 * (1 - alpha)

def mutate(h, n_points):
    mask = np.random.random(n_points) < mutation_rate
    multiplicative_noise = np.exp(np.random.normal(0, 0.2, n_points))
    additive_noise = np.random.uniform(0.01, 0.05, n_points)
    h = np.where(mask, h * multiplicative_noise, h)
    h = np.where(mask, h + additive_noise, h)
    return np.clip(h, 0.0, 1.0)
```

def project_feasible(h, n_points):
    sum_h = np.sum(h)
    if sum_h == 0:
        return np.full(n_points, 0.5 / n_points)  # Avoid division by zero
    scale = (n_points / 2.0) / sum_h
    h = np.clip(h * scale, 0.0, 1.0)
    # Second pass in case clipping after scaling broke sum
    sum_h = np.sum(h)
    if sum_h != 0:
        h = h * (n_points / 2.0) / sum_h
    return h
```
