import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initial guess for centers (grid-based layout)
    num_circles = 26
    rows, cols = 5, 6  # A grid that can accommodate 26 circles
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Generate initial centers in a grid layout
    for i in range(rows):
        for j in range(cols):
            idx = i * cols + j
            if idx >= num_circles:
                break
            centers[idx, 0] = j / (cols - 1) if cols > 1 else 0.5
            centers[idx, 1] = i / (rows - 1) if rows > 1 else 0.5

    # Initial radius (to be optimized)
    initial_radius = 0.08
    radii = np.full(num_circles, initial_radius)

    # Objective function: maximize the total area (equivalent to maximizing sum of radii)
    # We will minimize the negative of the total area
    def objective(radii):
        return -np.sum(radii)

    # Constraints
    def constraints(radii):
        # Non-overlapping constraint (between all pairs)
        constraints = []
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                x1, y1 = centers[i]
                x2, y2 = centers[j]
                d = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                constraint = d - (radii[i] + radii[j])
                constraints.append(constraint)

        # Boundary constraint: circles must be within the unit square
        for i in range(num_circles):
            x, y = centers[i]
            r = radii[i]
            constraint = x - r >= -1e-12
            constraints.append(constraint)
            constraints.append(x + r <= 1 + 1e-12)
            constraints.append(y - r >= -1e-12)
            constraints.append(y + r <= 1 + 1e-12)

        return constraints

    # Use scipy.optimize to find the optimal radii
    bounds = [(0, 1) for _ in range(num_circles)]
    result = optimize.minimize(
        objective,
        radii,
        bounds=bounds,
        constraints={"type": "ineq", "fun": constraints},
        method="SLSQP",
        options={"maxiter": 200, "ftol": 1e-9}
    )

    # Extract the final result
    best_radii = result.x
    best_centers = centers.copy()

    # Validate the packing
    valid, msg = validate_packing(best_centers, best_radii)
    if not valid:
        raise ValueError(msg)

    sum_radii = np.sum(best_radii)
    return best_centers, best_radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
