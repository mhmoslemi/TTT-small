import scipy.optimize as opt
import math

def build_initial_hexagonal_layout(num_circles):

                break
        if len(centers) == num_circles:
            break

    return np.array(centers)

def compute_objective(radii, centers):

    return -np.sum(radii)

def compute_constraints(centers, radii):
    """Compute constraints for optimization."""
    n = len(centers)
    constraints = []

    # Boundary constraint: circles must lie within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        constraints.append({
            'type': 'ineq',
            'fun': lambda r, x=x, y=y: x - r,
            'args': (x, y)
        })

            'type': 'ineq',
            'fun': lambda r, x=x, y=y: 1 - x - r,

            'fun': lambda r, x=x, y=y: 1 - y - r,
            'args': (x, y)

    for i in range(n):

    bounds = [(0.0, 0.1) for _ in range(26)]  # Radii bounded between 0 and 0.1

    result = opt.minimize(

        method='SLSQP',
