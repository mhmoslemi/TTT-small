import scipy.optimize as optimize

    x0 = np.concatenate([centers.flatten(), radii])

    def constraint_square(x, i):

        x_i, y_i = centers_opt[i]
        r_i = radii_opt[i]
        return x_i - r_i >= 0, x_i + r_i <= 1, y_i - r_i >= 0, y_i + r_i <= 1

        dy = centers_opt[i, 1] - centers_opt[j, 1]
        dist = np.sqrt(dx**2 + dy**2)
        return dist - radii_opt[i] - radii_opt[j] >= -1e-8

    # Bounds for all variables: x in [0,1], y in [0,1], r >= 0
    bounds = []
