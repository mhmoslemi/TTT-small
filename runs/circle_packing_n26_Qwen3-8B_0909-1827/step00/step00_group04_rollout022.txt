import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing():

    # Grid of rows and columns
    rows = 5
    cols = 5

    dy = np.sqrt(3) / (2 * (rows - 1))  # vertical spacing for hexagonal packing

    offset = dx / 2

    centers = np.zeros((rows * cols, 2))

            row_center = i * dy
            col_center = j * dx

                col_center += offset
            centers[i * cols + j] = [col_center, row_center]

    def objective(x):
        """Objective function to maximize sum of radii."""
        return -np.sum(x)  # We minimize the negative to achieve maximization

    # Optimization constraints
