import numpy as np
import scipy.optimize as optimize
import math

def run_packing():
    # Initial guess for circle centers using a hexagonal grid layout
    num_circles = 26
    radius_initial = 0.05  # Initial estimate of radius
    radius_guess = np.full(num_circles, radius_initial)
    
    # Generate initial centers using a grid layout
    grid_density = 5  # Density of grid points
    grid_x = np.linspace(0.2, 0.8, grid_density)
    grid_y = np.linspace(0.2, 0.8, grid_density)
    centers = np.array([(x, y) for y in grid_y for x in grid_x])
    
    # Additional circles for optimal packing
    additional_centers = np.array([
        [0.4, 0.4],
        [0.6, 0.4],
        [0.4, 0.6],
        [0.6, 0.6],
        [0.5, 0.2],
        [0.5, 0.8],
        [0.2, 0.5],
        [0.8, 0.5]
    ])
    centers = np.vstack([centers, additional_centers])
    
    # Define constraint functions
    def constraint_circle_inside(x, r, idx):
        x_center, y_center = x[idx * 2], x[idx * 2 + 1]
        return x_center - r[idx] >= -1e-12 and x_center + r[idx] <= 1 + 1e-12 and \
               y_center - r[idx] >= -1e-12 and y_center + r[idx] <= 1 + 1e-12
    
    def constraint_no_overlap(x, r, i, j):
        x_i, y_i = x[i * 2], x[i * 2 + 1]
        x_j, y_j = x[j * 2], x[j * 2 + 1]
        dist = math.hypot(x_i - x_j, y_i - y_j)
        return dist >= r[i] + r[j] - 1e-12
    
    # Define the objective function to maximize sum of radii
    def objective(x):
        r = x[num_circles * 2:]
        return -np.sum(r)  # Negative for minimization

    # Define bounds on the variables
    bounds = []
    for i in range(num_circles):
        # Bounds for center coordinates (x and y)
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 1.0))
        # Bounds for radii (non-negative)
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 1.0))
    
    # Initialize x vector: [x0, y0, r0, x1, y1, r1, ...]

        x0[i * 2 + 3] = radius_guess[i]

    # Define constraints as a list
    constraints = []
    for i in range(num_circles):

            'type': 'ineq',

                'fun': lambda x, i=i, j=j: constraint_no_overlap(x, x[num_circles * 2:], i, j)
            })

    # Run optimization
    result = optimize.minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        tol=1e-8

    if not result.success:

    sum_radii = np.sum(radii_opt)
