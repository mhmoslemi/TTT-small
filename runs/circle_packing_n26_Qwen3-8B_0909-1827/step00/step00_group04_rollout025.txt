import numpy as np
import scipy.optimize as opt
import scipy.spatial.distance as distance

def run_packing():
    # Step 1: Define initial circular packing grid (hexagonal arrangement)
    n = 26
    dim = int(np.ceil(np.sqrt(n)))
    x, y = np.linspace(0, 1, dim), np.linspace(0, 1, dim)
    X, Y = np.meshgrid(x, y)
    centers = np.column_stack((X.flatten(), Y.flatten()))
    radii = np.ones(n) * 0.1  # Initial guess for radii

    # Objective function: maximize the sum of radii
    def objective(radii_vec):
        return -np.sum(radii_vec)  # Negative because we use minimization

    # Constraint functions for optimization
    def constraint_non_overlap(radii_vec):
        centers = np.column_stack((X.flatten(), Y.flatten()))
        dists = distance.pdist(centers)
        for i in range(len(radii_vec)):
            for j in range(i + 1, len(radii_vec)):
                dist = dists[i * len(radii_vec) + j]
                if dist < radii_vec[i] + radii_vec[j] - 1e-8:
                    return -1
        return 0

    def constraint_boundary(radii_vec):
        centers = np.column_stack((X.flatten(), Y.flatten()))
        for i in range(len(radii_vec)):
            x, y = centers[i]
            r = radii_vec[i]
            if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
                return -1
        return 0

    # Bounds for radii (non-negative)
    bounds = [(0, 1) for _ in range(n)]

    # Optimization problem
    cons = [
        {'type': 'ineq', 'fun': constraint_non_overlap},
        {'type': 'ineq', 'fun': constraint_boundary}
    ]

    # Run optimization
    result = opt.minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        jac=None,
        tol=1e-8,
        options={'disp': False, 'ftol': 1e-8, 'maxiter': 1000}
    )

    # Extract the result
    centers = np.column_stack((X.flatten(), Y.flatten()))
    radii = result.x
    sum_radii = np.sum(radii)

    return (centers, radii, sum_radii)
