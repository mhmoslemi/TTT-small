import numpy as np
import cvxpy as cp

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26  # number of circles

    # Define variables for optimization
    centers = cp.Variable((n, 2))
    radii = cp.Variable(n)

    # Objective function: maximize sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []

    # Each circle lies within the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        constraints.append(x - r >= 0)
        constraints.append(x + r <= 1)
        constraints.append(y - r >= 0)
        constraints.append(y + r <= 1)

    # No two circles overlap
    for i in range(n):
        for j in range(i + 1, n):
            x_i, y_i = centers[i]
            x_j, y_j = centers[j]
            r_i = radii[i]
            r_j = radii[j]
            dist_squared = (x_i - x_j)**2 + (y_i - y_j)**2
            constraints.append(dist_squared >= (r_i + r_j)**2)

    # Each radius is non-negative
    for i in range(n):
        constraints.append(radii[i] >= 0)

    # Solve the optimization problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.OSQP, max_iter=100000)

    if prob.status != "optimal":
        raise RuntimeError(f"Optimization failed with status: {prob.status}")

    centers_np = centers.value
    radii_np = radii.value
    total_radii = np.sum(radii_np)

    return centers_np, radii_np, total_radii
