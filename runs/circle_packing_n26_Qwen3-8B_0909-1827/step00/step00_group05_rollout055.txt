def run_packing():

    for i in range(n):
        row = i // 5  # 5 rows with 5 or 6 circles, depends on row
        col = i % 5
        centers[i, 0] = col * 0.2 + 0.1  # X-coordinate
        centers[i, 1] = row * 0.2 + 0.1  # Y-coordinate

    # Initial guess for radii (equal radii for simplicity)

        return -np.sum(radii)  # minimize negative sum

            constraints.append({'type': 'ineq', 'fun': lambda r, i=i: x - r})
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i: x + r - 1})
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i: y - r})
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i: y + r - 1})

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
