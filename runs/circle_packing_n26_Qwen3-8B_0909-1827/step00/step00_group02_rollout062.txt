import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, LinearConstraint, NonlinearConstraint
import random

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26

    # Initial grid layout: 5 rows with 5 or 6 circles per row
    rows = 5
    cols = 6
    # Hexagonal close packing layout
    # Height of hexagonal packing: 2r * sqrt(3)/2 = r * sqrt(3)
    # Total height limit = 1
    # So radii are limited by 1 / sqrt(3) ≈ 0.577

    # Use random initialization for better convergence
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Initialize with a grid and some random variation
    # Grid-based pattern for better coverage
    for i in range(n):
        col = i % cols
        row = i // cols
        x = col / (cols - 1)  # x in [0,1]
        y = row / (rows - 1)  # y in [0,1]
        # Small random perturbation to avoid overlap
        x += np.random.uniform(-0.05, 0.05)
        y += np.random.uniform(-0.05, 0.05)
        centers[i, 0] = x
        centers[i, 1] = y
        radii[i] = 0.05  # Initial radius

    def objective_function(radii, centers):
        return -np.sum(radii)  # Minimize negative of sum to maximize

    def constraint_non_overlap(radii, centers):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return dist - radii[i] - radii[j] + 1e-8
        return 0

    def constraint_boundary(radii, centers):
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return (x - r) + (x + r - 1) + (y - r) + (y + r - 1)
        return 0

    # Define bounds
    bounds = Bounds(0.0, 0.577)  # Based on hexagonal packing density

    # Define constraints
    non_overlap_cons = NonlinearConstraint(
        constraint_non_overlap, -np.inf, 0, jac="2-point"
    )
    boundary_cons = NonlinearConstraint(
        constraint_boundary, 0, np.inf, jac="2-point"
    )

    # Initial guess for radii
    initial_guess = np.full(n, 0.05)

    # Optimization
    result = optimize.minimize(

        initial_guess,
        args=(centers,),
        method="SLSQP",
        bounds=bounds,
        constraints=[non_overlap_cons, boundary_cons],

    )

    radii_opt = result.x
    centers_opt = centers
    sum_radii = np.sum(radii_opt)

    return centers_opt, radii_opt, sum_radii
