import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initialize centers with random positions
    centers = np.random.uniform(0, 1, (26, 2))
    radii = np.zeros(26)

    # Define objective function
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize radii sum

    # Define constraint: Each circle must be inside the unit square
    def square_boundary(radii):
        return np.sum((centers - radii[:, np.newaxis]) ** 2 < 1)  # Ensure all circles are inside

    # Define constraint: No two circles overlap
    def non_overlap(radii):
        dists = np.sqrt(np.sum((centers[:, np.newaxis] - centers[np.newaxis, :]) ** 2, axis=2))
        return np.sum(dists < radii[:, np.newaxis] + radii[np.newaxis, :])  # Count overlapping pairs

    # Set up constraints
    cons = [
        {'type': 'ineq', 'fun': lambda r: np.min(r)},  # All radii must be non-negative
        {'type': 'ineq', 'fun': lambda r: square_boundary(r)},  # All circles within square
        {'type': 'ineq', 'fun': lambda r: non_overlap(r)},  # No two circles overlap
    ]

    # Run optimization
    result = opt.minimize(
        objective,
        radii,
        method='SLSQP',
        constraints=cons,
        options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-8}
    )

    # Validate the result (optional, but necessary for correctness)
    valid, msg = validate_packing(result.x.reshape(-1, 1), result.x)
    if not valid:
        raise ValueError(msg)

    return centers, result.x, -result.fun

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
