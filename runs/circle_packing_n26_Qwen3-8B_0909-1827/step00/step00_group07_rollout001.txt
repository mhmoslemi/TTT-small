def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

def initial_hexagonal_grid(n_circles):
    n_rows = int(np.ceil(np.sqrt(n_circles)))
    n_cols = int(np.ceil(n_circles / n_rows))

    grid = []
    for row in range(n_rows):
        for col in range(n_cols):
            x = col / (n_cols - 1)
            y = row / (n_rows - 1)
            if row % 2 == 1:
                x += 0.5 / (n_cols - 1)
            grid.append((x, y))
    return np.array(grid)

def compute_total_area(centers, radii):
    return np.sum(np.pi * radii ** 2)

def compute_total_radius_sum(radii):
    return np.sum(radii)

    bounds = np.array([[(0.0, 1.0)] * 2 for _ in range(n)])

            r_i = x_radii[i]

        constraints.append({'type': 'ineq', 'fun': boundary_constraint})

            x_radii = x[2 * n:]
            x_i, y_i = x_centers[i]
            r_i = x_radii[i]
            return (x_i + r_i) - (1 + 1e-12)
