import numpy as np
import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial guess for centers based on hexagonal packing
    num_circles = 26
    spacing = 0.4  # Adjust based on expected radius
    grid_rows = int(np.ceil(np.sqrt(num_circles)))
    grid_cols = int(np.ceil(num_circles / grid_rows))
    
    # Create hexagonal grid
    centers = np.zeros((num_circles, 2))
    idx = 0
    for row in range(grid_rows):
        for col in range(grid_cols):
            if row % 2 == 0:
                x = col * spacing
            else:
                x = (col + 0.5) * spacing
            y = row * spacing * np.sqrt(3) / 2
            centers[idx] = [x, y]
            idx += 1

    # Initial guess for radii
    radii = np.full(num_circles, 0.1)  # Start with small radius

    # Objective function to maximize the sum of radii
    def objective(radii_vec):
        return -np.sum(radii_vec)  # Minimize the negative sum

    # Constraints
    def constraint_boundary(radii_vec):
        for i in range(num_circles):
            x, y = centers[i]
            r = radii_vec[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
                return -1  # Violation
        return 0

    def constraint_overlap(radii_vec):
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.hypot(dx, dy)
                if dist < radii_vec[i] + radii_vec[j] - 1e-12:
                    return -1  # Violation
        return 0

    # Prepare constraints for optimization (using penalty method)
    def constraint_total(radii_vec):
        # Boundary constraint
        bc = constraint_boundary(radii_vec)
        # Overlap constraint
        oc = constraint_overlap(radii_vec)
        # Return value for optimization
        return bc + oc

    # Optimization with constraints
    bounds = [(0, 1.0) for _ in range(num_circles)]
    result = optimize.minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': constraint_total}]
    )

    # Extract results
    centers_opt = centers
    radii_opt = result.x
    sum_radii = np.sum(radii_opt)

    return centers_opt, radii_opt, sum_radii
