from scipy.optimize import Bounds

def validate_packing(centers, radii):

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

        x, y = centers[i]
        r = radii[i]

            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def create_initial_centers(num_circles):
    # Create a hexagonal grid pattern inside the unit square
    # Adjust based on number of circles and desired spacing
    spacing = 1.0 / np.sqrt(num_circles)
    centers = []
    for row in range(int(np.sqrt(num_circles)) + 1):
        for col in range(int(np.sqrt(num_circles)) + 1):
            x = col * spacing
            y = row * spacing * np.sqrt(3) / 2
            if x + spacing > 1 or y + spacing > 1:
                continue
            centers.append([x, y])
    centers = np.array(centers)
    centers -= np.min(centers, axis=0)
    centers /= np.max(centers, axis=0)
    return centers

def run_packing():

    radii_bounds = Bounds([0], [1])  # bounds for radii

        return -np.sum(radii)  # We minimize the negative sum to maximize the sum
