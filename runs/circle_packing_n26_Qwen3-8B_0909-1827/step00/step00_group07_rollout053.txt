import scipy.optimize as opt
import math

def run_packing():
    """
    Generate 26 circles in a unit square to maximize the sum of radii.
    Returns (centers, radii, sum_radii) as (np.ndarray, np.ndarray, float)
    """
    # Initial guess: hexagonal grid of 26 circles
    n = 26
    r_initial = 0.05

    # Create hexagonal grid layout
    row_count = int(np.ceil(np.sqrt(n)))
    col_count = int(np.ceil(n / row_count))

    # Place the circles in a hexagonal pattern
    centers = []
    for i in range(row_count):
        for j in range(col_count):
            x = j + 0.5 * (i % 2)
            y = i * np.sqrt(3) / 2
            centers.append((x, y))

    # Adjust to enforce the unit square bounds
    centers = np.array(centers)
    centers -= centers.min(axis=0)
    centers /= centers.max(axis=0)
    centers = np.clip(centers, 0, 1)
    centers = np.array(centers)

    # Define radii
    radii = np.full(n, r_initial)

    # Define the objective function to minimize (negative of sum of radii)
    def objective(radii):
        return -np.sum(radii)

    # Define constraints for optimality
    def constraints(radii):
        # No overlap between circles
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                constraints.append(dist - (radii[i] + radii[j]) >= -1e-12)
        return constraints

    # Define bounds for the radii
    bounds = [(0, 1) for _ in range(n)]

    # Optimize
    result = opt.minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        tol=1e-10
    )

    # Get final centers and radii
    final_centers = centers
    final_radii = result.x

    if not valid:

    return final_centers, final_radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
