import scipy.optimize as optimize

    rows = 6
    cols = 5
    radius_initial = 0.05  # Small initial radius

    radii = np.full(num_circles, radius_initial)

            y_center = y[i]
            centers[count] = [x_center, y_center]

    # Remove the last 4
    centers = centers[:26]
    radii = radii[:26]

    def objective(radii, centers):
        return -np.sum(radii)  # Minimize negative sum = maximize sum

    def constraints(radii, centers):
        # Create a list of constraints for optimization
        constraints = []
