import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing():

    n = 26

    # We'll divide the square into a grid and fill it with hexagonal arrangement
    # Approximate hexagonal packing for 26 circles
    rows = 5
    cols = 6
    centers = np.zeros((n, 2))

        row = i // cols

        x = col * (1 + np.sqrt(3)/2) / (cols - 1)

        centers[i, 1] = y

    def objective(radii):
        return -np.sum(radii)  # Negative for minimization

    # Bounds for radii: non-negative
    bounds = Bounds(0, np.inf)

    def constraint_center_inside(x, y, r):
        return np.array([x - r, x + r, y - r, y + r])

    def constraint_overlap(x1, y1, r1, x2, y2, r2):
        dx = x1 - x2
        dy = y1 - y2
        return np.sqrt(dx**2 + dy**2) - r1 - r2

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
