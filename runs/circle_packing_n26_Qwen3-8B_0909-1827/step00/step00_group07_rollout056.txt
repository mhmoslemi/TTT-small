import numpy as np
import scipy.optimize as opt

def run_packing():

    n = 26

    num_rows = int(np.ceil(np.sqrt(n)))

        for j in range(int(np.ceil(n / num_rows))):
            x = j + 0.5 * (i % 2)
            y = i * np.sqrt(3) / 2
            if i % 2 == 1:
                x += 0.5
            centers.append([x, y])

    radii = np.full(n, 1 / n)

        return [x_pos - r >= -1e-12, x_pos + r <= 1 + 1e-12,
                y_pos - r >= -1e-12, y_pos + r <= 1 + 1e-12]

    # Define constraints: no overlaps between circles
    def constraint_overlap(i, j, x):

        r1 = x[n + i]
        x2, y2 = x[2*j], x[2*j + 1]
        r2 = x[n + j]
        dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
        return dist >= r1 + r2 - 1e-12

    x0 = np.concatenate([centers.flatten(), radii])

        bounds.append((0, 1))
        bounds.append((0, 1))
        # Radius constraints
        bounds.append((0, 1))

    constraints = []

        bounds=bounds,

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
