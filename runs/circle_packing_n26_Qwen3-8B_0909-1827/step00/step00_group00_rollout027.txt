import scipy.optimize as optimize

def run_packing():

    N = 6  # number of columns
    M = 5  # number of rows

    vertical_spacing = np.sqrt(3) / 2
    horizontal_spacing = 1.0

    # Generate initial centers for the hexagonal grid
    centers = np.zeros((N*M, 2))
    for i in range(M):
        for j in range(N):
            x = j * horizontal_spacing
            y = i * vertical_spacing
            centers[i*N + j] = [x, y]

    # Find the bounding box of the hexagonal grid
    x_min, x_max = np.min(centers[:, 0]), np.max(centers[:, 0])

    # Scale the grid to [0,1]×[0,1]
    centers = (centers - [x_min, y_min]) / [x_max - x_min, y_max - y_min]
