import scipy.optimize as opt
from typing import Tuple, Any
from numpy.typing import NDArray

def initialize_grid(n_circles: int) -> Tuple[NDArray, NDArray]:
    """Initialize a hexagonal grid of circles."""
    max_rows = int(np.sqrt(n_circles))
    max_cols = int(np.ceil(n_circles / max_rows))
    centers = np.zeros((n_circles, 2))
    r = 0.05  # Initial radius
    for i in range(n_circles):
        row = i // max_cols
        col = i % max_cols

        y = row * (1 + np.sqrt(3)/2) * r
        centers[i] = [x, y]
    return centers, np.full(n_circles, r)

def constraint_no_overlap(x: NDArray, n_circles: int) -> NDArray:
    """Constraint to ensure no overlap between circles."""
    centers = x[:n_circles*2].reshape(n_circles, 2)
    radii = x[n_circles*2:]
    constraints = []
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            cons = dist - (radii[i] + radii[j])
            constraints.append(cons)
    return np.array(constraints)

def constraint_in_square(x: NDArray, n_circles: int) -> NDArray:
    """Constraint to ensure all circles are within the unit square."""
    centers = x[:n_circles*2].reshape(n_circles, 2)

        # y - r >= 0
        constraints.append(y - r)
        # y + r <= 1
        constraints.append(1 - y - r)
    return np.array(constraints)

def objective(x: NDArray, n_circles: int) -> float:

    radii = x[n_circles*2:]
    return -np.sum(radii)  # Use negative for minimization
