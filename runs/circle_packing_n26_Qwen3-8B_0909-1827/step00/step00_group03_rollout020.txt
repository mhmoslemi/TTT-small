import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds
from scipy.optimize import minimize

def run_packing():
    n = 26

    # Initial guess for centers: 5x5 grid with some spacing
    centers = np.array([[x, y] for x in np.linspace(0.1, 0.9, 5) for y in np.linspace(0.1, 0.9, 5)])
    centers = centers[:n]
    radii = np.ones(n) * 0.1  # Initial radius for each circle

    # Objective function to minimize: negative sum of radii (to maximize it)
    def objective(x):
        # x contains flattened centers and radii
        flat_centers = x[:2*n].reshape(n, 2)
        flat_radii = x[2*n:]
        return -np.sum(flat_radii)

    # Constraint function for non-overlapping circles
    def constraint_non_overlap(x):
        flat_centers = x[:2*n].reshape(n, 2)
        flat_radii = x[2*n:]
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((flat_centers[i] - flat_centers[j])**2))
                if dist < flat_radii[i] + flat_radii[j] - 1e-8:
                    return dist - (flat_radii[i] + flat_radii[j] - 1e-8)
        return 0

    # Constraint function for boundary conditions
    def constraint_boundary(x):
        flat_centers = x[:2*n].reshape(n, 2)
        flat_radii = x[2*n:]
        for i in range(n):
            x_c, y_c = flat_centers[i]
            r = flat_radii[i]
            if x_c - r < -1e-8 or x_c + r > 1 + 1e-8 or y_c - r < -1e-8 or y_c + r > 1 + 1e-8:
                return 1e8  # Large penalty for boundary violation
        return 0

    def constraint_non_negative(x):

        for r in flat_radii:
            if r < -1e-8:
                return 1e8  # Large penalty for negative radius
        return 0

    # Constraints
    cons = [
        {'type': 'ineq', 'fun': constraint_non_overlap},
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_non_negative}

    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 1000})

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
