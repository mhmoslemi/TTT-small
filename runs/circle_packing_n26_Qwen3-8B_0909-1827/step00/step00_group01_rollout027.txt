from scipy.optimize import minimize
from scipy.optimize import Bounds
import warnings

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

            if dist < radii[i] + radii[j] - 1e-12:

    return True, "ok"

def run_packing():

    # Initial guess: grid-like distribution
    centers = np.zeros((n_circles, 2))

        row = i // 5

        radii = x[n_circles:]
        return -np.sum(radii)

    # Bounds for radii (must be non-negative)
    radii_bounds = [(0, 1)] * n_circles

    centers_bounds = [(0, 1)] * (2 * n_circles)

    # Constraints: circle boundaries and non-overlapping
    def constraints(x):

            r_i = radii[i]
            constraints.append((x_i - r_i >= -1e-8) & (x_i + r_i <= 1 + 1e-8) &

                constraints.append(dist >= r_i + r_j - 1e-8)

        return constraints

    with warnings.catch_warnings():

        result = minimize(
            objective,
            np.concatenate([centers.flatten(), np.ones(n_circles)]),

            method="SLSQP",
