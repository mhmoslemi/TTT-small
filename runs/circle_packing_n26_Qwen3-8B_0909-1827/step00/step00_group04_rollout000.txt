import scipy.optimize as opt

def validate_packing(centers, radii):

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

    num_circles = 26

        y = row * spacing + 0.5 * spacing
        centers[i] = [x, y]
        radii[i] = 0.01  # Initial small radius for optimization

    # Optimization bounds
    bounds = []
    for _ in range(num_circles):
        bounds.append((0, 1))  # x
        bounds.append((0, 1))  # y
        bounds.append((0, 1))  # radius

    # Define objective function to maximize the sum of radii
    def objective(x):
        centers_opt = x[:2*num_circles].reshape(num_circles, 2)
        radii_opt = x[2*num_circles:].reshape(num_circles)
        return -np.sum(radii_opt)  # Minimize negative sum to maximize sum

        centers_opt = x[:2*num_circles].reshape(num_circles, 2)
