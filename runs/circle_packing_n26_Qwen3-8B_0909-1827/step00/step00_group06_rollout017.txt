import numpy as np
import scipy.optimize as optimize

def distance_constraint(centers, radii, i, j):
    dx = centers[i, 0] - centers[j, 0]
    dy = centers[i, 1] - centers[j, 1]
    dist = np.sqrt(dx * dx + dy * dy)
    return dist - (radii[i] + radii[j])

def boundary_constraint(center, radius):
    x, y = center
    return np.array([
        x - radius,
        x + radius - 1,
        y - radius,
        y + radius - 1
    ])

def objective_function(x, n_circles=26):
    centers = x[:n_circles * 2].reshape(n_circles, 2)
    radii = x[n_circles * 2:]
    return -np.sum(radii)  # Maximize the sum by minimizing -sum

def run_packing():

    # Initial guess: regular grid with one circle in the center
    # Generate 5x5 grid with spacing 0.2, then place a circle at the center
    grid = np.array([
        [i * 0.25, j * 0.25]
        for i in range(5)
        for j in range(5)
    ])
    centers = np.vstack((grid, [0.5, 0.5]))
    radii = np.full(n_circles, 0.1)  # Initial radius guess

    # Flatten the centers and radii for optimization
    x0 = np.hstack((centers.flatten(), radii))

    # Constraints: non-overlapping and within bounds
    constraints = []

    # Add non-overlapping constraints
    for i in range(n_circles):
        for j in range(i + 1, n_circles):

                                          x[n_circles * 2:], i, j)
            constraints.append({'type': 'ineq', 'fun': constraint})

    # Add boundary constraints

            radius = x[n_circles * 2:]

        constraints.append({'type': 'ineq', 'fun': boundary})

    # Optimization: use L-BFGS-B with bounds
    bounds = []
    for i in range(n_circles):
        bounds.append((0.0, 1.0))  # x and y coordinates must be in [0, 1]

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
