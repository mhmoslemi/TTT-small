import numpy as np
import scipy.optimize as opt
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Step 1: Hexagonal Grid Initialization
    # Use a hexagonal lattice to place the centers with some initial spacing
    grid_points = []
    for row in range(4):
        for col in range(6):
            x = col * 0.25
            y = row * (np.sqrt(3) / 2) * 0.25
            grid_points.append([x, y])
    
    centers = np.array(grid_points)
    # Scale to fit in [0,1]^2
    centers = centers / np.max(centers)

    # Initialize radii so that circles just touch each other
    radius_initial = 0.05
    radii[:] = radius_initial

    # Objective function to maximize sum of radii (minimize negative)
    def objective(x):
        return -np.sum(x)

    # Optimization variables: radii and center coordinates
    x_bounds = [(0, 0.3) for _ in range(num_circles * 2)]
    x0 = np.concatenate((centers.flatten(), radii))

    # Constraints: ensure no two circles overlap and are within the square
    def constraint1(x):
        centers_opt = x[:num_circles * 2].reshape((num_circles, 2))
        radii_opt = x[num_circles * 2:]
        # Check boundaries
        for i in range(num_circles):
            cx, cy = centers_opt[i]
            r = radii_opt[i]
            if cx - r < -1e-8 or cx + r > 1 + 1e-8 or cy - r < -1e-8 or cy + r > 1 + 1e-8:
                return -1  # Violation
        # Check overlap
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers_opt[i, 0] - centers_opt[j, 0]
                dy = centers_opt[i, 1] - centers_opt[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii_opt[i] + radii_opt[j] - 1e-8:
                    return -1  # Violation
        return 0

    # Constrained optimization with SLSQP
    result = opt.minimize(
        objective,
        x0=x0,
        method='SLSQP',
        bounds=x_bounds,
        constraints=[{'type': 'ineq', 'fun': constraint1}],
        options={'ftol': 1e-8, 'maxiter': 500}
    )

    if not result.success:
        raise ValueError(f"Optimization failed: {result.message}")

    centers_opt = result.x[:num_circles * 2].reshape((num_circles, 2))
    radii_opt = result.x[num_circles * 2:]

    centers = np.clip(centers_opt, 0, 1)
    radii = np.clip(radii_opt, 0, 0.3)

    # Final validation
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise ValueError(f"Validation failed: {msg}")

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
