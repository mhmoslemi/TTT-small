import numpy as np
import scipy.optimize as opt
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Initial estimate of circle positions and radii
    # Start with a hexagonal grid
    num_circles = 26
    rows = 5
    cols_per_row = 5
    delta_x = 1.0 / (cols_per_row + 1)
    delta_y = np.sqrt(3) / (rows + 1)

    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Place in hexagonal grid
    for i in range(rows):
        for j in range(cols_per_row):
            x = j * delta_x + delta_x / 2
            y = i * delta_y + delta_y / 2
            centers[i * cols_per_row + j] = [x, y]

    # Adjust to fit within square [0,1]×[0,1] and optimize
    # Initial radii based on hexagonal spacing
    initial_radii = np.ones(num_circles) * (delta_x / 2)  # radius roughly half the horizontal spacing

    # Initial guess
    x0 = np.concatenate([centers.flatten(), initial_radii])

    # Optimization objective: maximize sum of radii
    def objective(x):
        centers_flat = x[:num_circles * 2].reshape(num_circles, 2)
        radii_flat = x[num_circles * 2:]

        # Sum of radii
        sum_r = np.sum(radii_flat)

        # Constraints
        constraints = []

        # Constraint: radii >= 0
        for i in range(num_circles):
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[num_circles * 2 + i]})

        # Constraint: circles within [0,1]^2
        for i in range(num_circles):
            x_center, y_center = centers_flat[i]
            r = radii_flat[i]
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i, x_center=x_center, y_center=y_center, r=r:
                               x_center - r})
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i, x_center=x_center, y_center=y_center, r=r:
                               1.0 - x_center - r})
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i, x_center=x_center, y_center=y_center, r=r:
                               y_center - r})
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i, x_center=x_center, y_center=y_center, r=r:
                               1.0 - y_center - r})

        # Constraint: no overlaps
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                x1, y1 = centers_flat[i]
                x2, y2 = centers_flat[j]
                r1 = radii_flat[i]
                r2 = radii_flat[j]
                dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                constraints.append({'type': 'ineq', 'fun': lambda x, i=i, j=j, x1=x1, y1=y1, x2=x2, y2=y2, r1=r1, r2=r2:
                                   dist - r1 - r2})

        return -sum_r, constraints

    # Run optimization
    bounds = [(-1e-6, 1.0 + 1e-6) for _ in range(num_circles * 2)]
    result = opt.minimize(lambda x: objective(x)[0], x0, method='SLSQP', bounds=bounds, constraints=objective)

    # Extract centers and radii

    radii = result.x[num_circles * 2:]

    # Validate
    valid, message = validate_packing(centers, radii)
    if not valid:
        raise ValueError(f"Invalid packing: {message}")

    # Return the result
    return centers, radii, np.sum(radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
