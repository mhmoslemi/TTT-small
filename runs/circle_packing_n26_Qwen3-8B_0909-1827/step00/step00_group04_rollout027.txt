import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial positions: hexagonal packing with starting radius
    num_circles = 26
    radii = np.full(num_circles, 0.1)
    
    # Initial center positions: hexagonal grid
    centers = np.zeros((num_circles, 2))
    row = 0
    col = 0
    for i in range(num_circles):
        # Compute horizontal and vertical steps
        h = radii[i] * np.sqrt(3)
        v = radii[i] * 2
        if i % 2 == 0:
            centers[i, 0] = col * (2 * radii[i] + h)
        else:
            centers[i, 0] = col * (2 * radii[i] + h) + h / 2
        centers[i, 1] = row * v
        col += 1
        if col % 2 == 0:
            row += 1
    
    # Function to compute the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Function to compute the constraints
    def constraint_overlap(radii, centers):
        n = len(radii)
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return -1.0  # Violation
        return 0.0

    def constraint_boundary(radii, centers):
        n = len(radii)
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
                return -1.0  # Violation
        return 0.0

    # Optimization with bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(num_circles)]
    constraints = [
        {'type': 'ineq', 'fun': lambda r: constraint_overlap(r, centers)},
        {'type': 'ineq', 'fun': lambda r: constraint_boundary(r, centers)}
    ]

    # Initial guess for optimization
    initial_guess = radii.copy()

    # Run optimization
    result = opt.minimize(
        objective,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'ftol': 1e-12, 'disp': False}
    )

    # Update radii and centers
    radii = result.x

    sum_radii = np.sum(radii)
