import numpy as np
import scipy.optimize as opt
import scipy.spatial as spatial
from typing import Tuple, List

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Generates a packing of 26 circles in a unit square to maximize the sum of radii.
    Utilizes a combination of hexagonal packing and greedy refinement to achieve a good
    initial configuration and then refines it with optimization.
    """
    # Initial hexagonal packing
    centers = hexagonal_packing(26)
    radii = np.ones(26) * 0.05  # Initial estimate of radii

    # Optimize the packing
    centers, radii = optimize_packing(centers, radii)

    # Final validation
    success, message = validate_packing(centers, radii)
    if not success:
        raise ValueError(f"Packing validation failed: {message}")

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii

def hexagonal_packing(n: int) -> np.ndarray:
    """
    Generates an initial hexagonal packing of n circles.
    """
    if n == 0:
        return np.array([])

    # Calculate the number of rows
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))

    # Generate coordinates
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = j + 0.5 * i
            y = i
            centers.append([x, y])

    # Truncate to n circles
    centers = np.array(centers)[:n]

    # Normalize to unit square
    centers = centers / np.max(centers, axis=0)
    return centers

def optimize_packing(centers: np.ndarray, radii: np.ndarray) -> Tuple[np.ndarray, np.ndarray]:
    """
    Optimizes the packing by adjusting centers and radii to maximize the sum of radii.
    """
    # Objective function to maximize sum of radii
    def objective(x):
        centers_flat = x[:2 * 26].reshape(26, 2)
        radii_flat = x[2 * 26:]
        return -np.sum(radii_flat)  # Negative for minimization

    # Constraints
    def constraints(x):
        centers_flat = x[:2 * 26].reshape(26, 2)
        radii_flat = x[2 * 26:]
        constraints = []

        # Boundary constraint
        for i in range(26):
            x, y = centers_flat[i]
            r = radii_flat[i]
            constraints.append((x - r) - 0.0)
            constraints.append((1.0 - x - r) - 0.0)
            constraints.append((y - r) - 0.0)
            constraints.append((1.0 - y - r) - 0.0)

        # Non-overlap constraint
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                constraints.append(dist - (radii_flat[i] + radii_flat[j]) - 1e-8)

        return constraints

    # Initial guess: centers and radii
    initial_centers = centers.copy()
    initial_radii = radii.copy()
    x0 = np.concatenate([initial_centers.ravel(), initial_radii])

    # Optimization
    bounds = []
    for i in range(26):
        bounds.extend([(0.0, 1.0), (0.0, 1.0)])  # centers
        bounds.extend([(0.0, 1.0)])              # radius

    result = opt.minimize(
        objective,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints={"type": "ineq", "fun": constraints},
        options={"maxiter": 100, "ftol": 1e-8, "eps": 1e-8}
    )

    if not result.success:
        raise ValueError("Optimization failed")

    optimized_centers = result.x[:2 * 26].reshape(26, 2)
    optimized_radii = result.x[2 * 26:]

    return optimized_centers, optimized_radii

def validate_packing(centers: np.ndarray, radii: np.ndarray) -> Tuple[bool, str]:
    """
    Validates the packing constraints: within square, non-overlapping, no NaNs.
    """
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
