import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    radii = np.zeros(n)

        row = i // len(grid)
        col = i % len(grid)

        radii[i] = 0.05  # Initial small radius

        return -np.sum(radii_flat)  # Negative sum of radii to be minimized

        radii_flat = x[2*n:]
        for i in range(n):

            r_i = radii_flat[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
