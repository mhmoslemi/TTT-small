import scipy.optimize as optimize
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:

    num_circles = 26

            centers[i * num_per_row + j] = [x, y]
            radii[i * num_per_row + j] = 0.05

    # Objective function to maximize total radius, minimized as negative
    def objective(x):
        flat_centers = x[:num_circles * 2].reshape((num_circles, 2))
        flat_radii = x[num_circles * 2:]
        return -np.sum(flat_radii)

    # Bounds: centers in [0,1]^2 and radii ≥ 0
    bounds = []
    for _ in range(num_circles):
        bounds += [(0.0, 1.0), (0.0, 1.0)]  # x and y coordinates for each circle
        bounds += [(0.0, 1.0)]              # radius for each circle

    # Constraints: boundary and non-overlapping
    def constraint_boundary(x):
        flat_centers = x[:num_circles * 2].reshape((num_circles, 2))
        flat_radii = x[num_circles * 2:]

            x_pos, y_pos = flat_centers[i]

        return 0.0

                if dist < flat_radii[i] + flat_radii[j] - 1e-12:

    result = optimize.minimize(
