import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    n = 26

    radii = np.zeros(n)

    # We'll optimize the radii and centers in one go

    def objective(radii, centers):
        return -np.sum(radii)  # Negative for minimization
    
    # Constraints: Each circle must be inside the square and not overlap with others
    def constraint_center_inside(radii, centers):
        return np.array([0 <= centers[i, 0] - radii[i] for i in range(n)]) & \
               np.array([0 <= centers[i, 1] - radii[i] for i in range(n)]) & \
               np.array([centers[i, 0] + radii[i] <= 1 for i in range(n)]) & \
               np.array([centers[i, 1] + radii[i] <= 1 for i in range(n)])
    
    def constraint_no_overlap(radii, centers):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

        centers[i, 1] = y + 0.5 * spacing

    # Set initial radii such that they are small, but have potential to grow

        bounds.extend([(0, 1), (0, 1)])

        bounds.append((0, 1))
    
    # Initialize the optimization
    result = opt.minimize(
