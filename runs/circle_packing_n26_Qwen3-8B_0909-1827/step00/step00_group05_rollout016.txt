import numpy as np
import scipy.optimize as opt
from scipy.spatial.distance import pdist, squareform

def run_packing():
    # Initial configuration: hexagonal packing in a unit square
    num_circles = 26
    # Try different grid sizes to fit 26 circles in a square
    # We'll choose a grid of m rows and n columns with staggered rows
    m, n = 5, 6  # 30 circles would fit a 5x6 grid; 26 is close
    grid_rows, grid_cols = m, n

    # Initial radius guess
    initial_radius = 0.06  # A starting radius that should work for hexagonal packing

    # Grid spacing: will be adjusted during optimization
    dx = 1.0 / (grid_cols - 1)
    dy = 1.0 / (grid_rows - 1)

    # Generate initial centers for a hexagonal grid
    centers = np.zeros((num_circles, 2))
    idx = 0
    for i in range(grid_rows):
        for j in range(grid_cols):
            if i % 2 == 0:
                x = j * dx
            else:
                x = (j + 0.5) * dx
            y = i * dy
            if idx < num_circles:
                centers[idx] = [x, y]
                idx += 1
            else:
                break

    # Initial radii
    radii = np.full(num_circles, initial_radius)

    # Define the objective function to maximize the sum of radii
    def objective(radii_vec):
        return -np.sum(radii_vec)  # Minimize the negative sum of radii

    # Define constraints for the optimization
    def constraints(radii_vec, centers):
        # All radii must be non-negative
        return radii_vec >= 0

    def equality_constraints(radii_vec, centers):
        # No two circles can overlap
        dists = squareform(pdist(centers))
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                if dists[i, j] < radii_vec[i] + radii_vec[j] - 1e-12:
                    return np.inf
        return 0

    # Optimization bounds
    bounds = [(0, 1.0) for _ in range(num_circles)]

    # Initial optimization
    # Use SLSQP for constrained optimization
    result = opt.minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': lambda r: constraints(r, centers)},
                     {'type': 'eq', 'fun': lambda r: equality_constraints(r, centers)}],
        tol=1e-6
    )

    # Final radii and centers
    radii = result.x
    sum_radii = np.sum(radii)

    # Validate the packing
    is_valid, msg = validate_packing(centers, radii)
    if not is_valid:
        raise RuntimeError(f"Invalid packing: {msg}")

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
