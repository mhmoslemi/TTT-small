import numpy as np
import scipy.optimize as opt

def run_packing():
    # Initial grid-based placement of 26 circles
    num_circles = 26
    grid_rows = 6
    grid_cols = 5
    
    # For simplicity, we'll use a grid-like initial layout with some spacing
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)
    
    # Create an initial grid-like arrangement
    for i in range(num_circles):
        row = i // grid_cols
        col = i % grid_cols
        x = col / (grid_cols - 1) * 1.0
        y = row / (grid_rows - 1) * 1.0
        centers[i] = [x, y]
        radii[i] = 0.1  # Initial guess for each radius

    # Objective function: maximize sum of radii
    def objective(x):
        # x is a flattened array of [c1x, c1y, r1, c2x, c2y, r2, ..., cnx, cny, rn]
        x_unflattened = x.reshape((num_circles, 3))
        centers = x_unflattened[:, :2]
        radii = x_unflattened[:, 2]
        return -np.sum(radii)  # Minimize negative sum (maximize sum)

    # Constraint functions
    def constraint_bounds(x):
        x_unflattened = x.reshape((num_circles, 3))
        centers = x_unflattened[:, :2]
        radii = x_unflattened[:, 2]
        # Check center boundaries
        for i in range(num_circles):
            x, y = centers[i]
            r = radii[i]
            if (x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1):
                return np.inf
        return 0

    def constraint_overlap(x):
        x_unflattened = x.reshape((num_circles, 3))
        centers = x_unflattened[:, :2]
        radii = x_unflattened[:, 2]
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return np.inf
        return 0

    # Optimization bounds: x and y in [0, 1], radii >= 0
    bounds = []
    for _ in range(num_circles):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 1.0)])  # x, y, r for each circle

    # Optimization options
    options = {
        'ftol': 1e-6,
        'maxiter': 2000,
        'disp': False
    }

    # Optimization
    initial_guess = np.hstack([centers.flatten(), radii])
    result = opt.minimize(
        fun=objective,
        x0=initial_guess,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint_bounds},
            {'type': 'ineq', 'fun': constraint_overlap}
        ],
        method='SLSQP',
        options=options
    )

    if not result.success:
        raise RuntimeError("Optimization failed: " + result.message)

    # Extract final centers and radii
    final_centers = result.x[:num_circles*2].reshape((num_circles, 2))
    final_radii = result.x[num_circles*2:].reshape((num_circles,))

    sum_radii = np.sum(final_radii)

    return final_centers, final_radii, sum_radii
