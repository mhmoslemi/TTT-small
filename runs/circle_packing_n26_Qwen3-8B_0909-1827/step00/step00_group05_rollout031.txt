import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Initial hexagonal close packing (HCP) layout
    num_circles = 26
    radius = 0.05  # Initial radius
    centers = np.zeros((num_circles, 2))

    # Create a hexagonal grid with 5 rows
    row_height = np.sqrt(3) * radius
    col_width = 2 * radius

    for row in range(5):
        for col in range(5):
            if (row % 2 == 0):
                x = col * col_width
            else:
                x = (col + 0.5) * col_width
            y = row * row_height
            idx = row * 5 + col
            if idx < num_circles:
                centers[idx] = [x, y]

    # Scale the initial layout to fit in the unit square
    scale_factor = 1.0 / np.max(centers)
    centers *= scale_factor
    radius = radius * scale_factor

    bounds = [(0.0, 1.0) for _ in range(2 * num_circles)]
    x0 = np.concatenate((centers.flatten(), np.full(num_circles, radius)))

    def objective(x: np.ndarray) -> float:
        centers_opt = x[:2*num_circles].reshape(num_circles, 2)
        radii_opt = x[2*num_circles:]
        return -np.sum(radii_opt)  # Minimize negative sum to maximize sum

    # Constraints: distance between centers >= sum of radii
    def constraint_dist(x: np.ndarray) -> np.ndarray:
        centers_opt = x[:2*num_circles].reshape(num_circles, 2)
        radii_opt = x[2*num_circles:]
        dists = squareform(pdist(centers_opt))
        r_sum = radii_opt[:, np.newaxis] + radii_opt[np.newaxis, :]
        return dists - r_sum + 1e-12

        return x[2*num_circles:]

    # Create the optimization problem
    constraints = [

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
