import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    # Initial grid-based packing
    n = 26
    grid_size = int(np.sqrt(n))
    spacing = 1.0 / (grid_size + 1)
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    for i in range(grid_size):
        for j in range(grid_size):
            idx = i * grid_size + j
            if idx >= n:
                break
            x = (i + 1) * spacing
            y = (j + 1) * spacing
            centers[idx] = [x, y]
            radii[idx] = 0.01  # Initial small radius

    # Objective: maximize sum of radii -> minimize -sum(radii)
    def objective(radii_vec):
        return -np.sum(radii_vec)

    # Constraints: must not overlap and stay within [0,1]^2
    def constraint_overlap(radii_vec):
        centers = np.zeros((n, 2))
        for i in range(n):
            x = (i // grid_size + 1) * spacing
            y = (i % grid_size + 1) * spacing
            centers[i] = [x, y]
        # Update centers based on radii (assuming radii affect placement)
        # For this example, we assume centers are fixed for now
        dist_matrix = cdist(centers, centers)
        for i in range(n):
            for j in range(i + 1, n):
                if dist_matrix[i, j] < radii_vec[i] + radii_vec[j] - 1e-12:
                    return -1
        return 0

    # Initial guess for radii
    initial_guess = np.full(n, 0.01)

    # Optimization
    bounds = [(0, 0.5) for _ in range(n)]  # Radii cannot exceed half the unit square

    result = minimize(
        objective,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': constraint_overlap},
        options={'maxiter': 500, 'ftol': 1e-8, 'disp': False}
    )

    # Extract optimal radii and centers
    optimal_radii = result.x
    optimal_centers = np.zeros((n, 2))
    for i in range(n):
        x = (i // grid_size + 1) * spacing
        y = (i % grid_size + 1) * spacing
        optimal_centers[i] = [x, y]

    sum_radii = np.sum(optimal_radii)

    # Validate the packing
    valid, msg = validate_packing(optimal_centers, optimal_radii)
    if not valid:
        raise ValueError(msg)

    return optimal_centers, optimal_radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
