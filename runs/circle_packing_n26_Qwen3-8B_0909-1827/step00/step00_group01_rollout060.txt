import numpy as np
import scipy.optimize as opt
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Initial guess: place 26 circles evenly in a grid
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)
    
    # Create a grid-like initial layout
    grid = np.array([
        [0.1, 0.1], [0.1, 0.3], [0.1, 0.5], [0.1, 0.7], [0.1, 0.9],
        [0.3, 0.1], [0.3, 0.3], [0.3, 0.5], [0.3, 0.7], [0.3, 0.9],
        [0.5, 0.1], [0.5, 0.3], [0.5, 0.5], [0.5, 0.7], [0.5, 0.9],

        [0.9, 0.1], [0.9, 0.3], [0.9, 0.5], [0.9, 0.7], [0.9, 0.9],
    ])
    
    centers[:] = grid
    radii[:] = 0.03  # Small initial radius to allow optimization

    # Define objective function to maximize sum of radii
    def objective(x):
        return -np.sum(x)  # Negative for minimization

    # Define constraints
    def constraint_center(x, i):
        return x[2*i] - 1e-6  # Ensure center within square

    def constraint_center2(x, i):
        return 1 - x[2*i] - 1e-6  # Ensure center within square

    def constraint_center3(x, i):
        return x[2*i + 1] - 1e-6  # Ensure center within square

    def constraint_center4(x, i):
        return 1 - x[2*i + 1] - 1e-6  # Ensure center within square

    def constraint_overlap(x, i, j):
        dx = x[2*i] - x[2*j]
        dy = x[2*i + 1] - x[2*j + 1]
        dist = np.sqrt(dx*dx + dy*dy)
        return dist - x[2*num_circles + i] - x[2*num_circles + j] + 1e-6

    # Prepare variables for optimization
    x_bounds = [(0, 1) for _ in range(2*num_circles)] + [(0, 0.1) for _ in range(num_circles)]
    x0 = np.concatenate([centers.flatten(), radii])

    # Create constraints
    constraints = []
    for i in range(num_circles):
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: constraint_center(x, i)})

        for j in range(i + 1, num_circles):
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: constraint_overlap(x, i, j)})

    # Run optimization
    result = opt.minimize(

        x0,

        constraints=constraints,
