from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    n_circles = 26
    grid_spacing = 1 / np.sqrt(2)  # Spacing based on hexagonal packing
    rows = int(np.sqrt(n_circles))
    cols = int(np.ceil(n_circles / rows))
    
    centers = np.zeros((n_circles, 2))
    index = 0

            if index >= n_circles:
                break

            index += 1

    initial_radius = 0.05
    radii = np.full(n_circles, initial_radius)

        return -np.sum(radii_vec)  # Minimize negative sum

    # Constraint: all circles are within the square
    def square_bounds(radii_vec):
        return np.array([np.clip(r, 0, 1 - r) for r in radii_vec])

        }

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
