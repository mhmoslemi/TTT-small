import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial grid-based placement with 5x5 grid (25 cells)
    grid_size = 5
    grid = np.linspace(0, 1, grid_size + 1)
    centers = np.array([[x, y] for y in grid for x in grid])
    radii = np.full(len(centers), 1 / (grid_size + 1))
    
    # Optimization: Maximize sum of radii while respecting constraints
    # Objective function to maximize: sum of radii
    def objective(radii):
        return -np.sum(radii)  # Negative for minimization

    # Constraints: 
    # 1. Each circle must be within the unit square
    # 2. Circles must not overlap
    def constraints(radii):
        # Constraint 1: Within unit square
        constraints = []
        for i in range(len(radii)):
            def within_square(radii):
                return np.array([radii[i] <= 1 - centers[i, 0], 
                                radii[i] <= 1 - centers[i, 1], 
                                radii[i] >= centers[i, 0], 
                                radii[i] >= centers[i, 1]])
            constraints.append({'type': 'ineq', 'fun': within_square})
        
        # Constraint 2: Non-overlapping circles
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                def no_overlap(radii):
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx**2 + dy**2)
                    return dist - (radii[i] + radii[j])
                constraints.append({'type': 'ineq', 'fun': no_overlap})
        
        return constraints

    # Optimization
    bounds = [(0, 1) for _ in range(len(radii))]
    result = opt.minimize(objective, radii, method='SLSQP', 
                          constraints=constraints(radii), bounds=bounds)

    # Extract the result
    optimized_radii = result.x
    sum_radii = np.sum(optimized_radii)
    return centers, optimized_radii, sum_radii
