import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds
from typing import Tuple, List

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Function to generate the optimal circle packing in a unit square.
    Returns (centers, radii, sum_radii) satisfying all constraints.
    """
    # Constants
    n = 26
    max_radius = 1.0
    radius_guess = 0.1  # Initial guess for radius
    num_iterations = 100  # Number of optimization steps
    tolerance = 1e-5  # Optimization tolerance

    # Initial positions: Hexagonal grid inside a unit square
    # Based on https://www.mathpages.com/home/kmath503/kmath503.htm
    # For 26 circles, we'll use 5 rows with 5, 6, 5, 6, and 5 circles respectively

    # Compute initial positions
    centers = np.zeros((n, 2))
    index = 0
    for row in range(5):
        for col in range(5 + (row % 2)):
            x = col * (max_radius * np.sqrt(3) * 1.5) + 0.5 * max_radius
            y = row * (max_radius * 2.0) + max_radius
            if x > 1 - max_radius or y > 1 - max_radius:
                continue
            centers[index] = [x, y]
            index += 1

    # Since we have only 25 circles in the grid, we add one more at bottom center
    centers[index] = [0.5, 1.0 - max_radius]

    # Initial radius guess
    radii = np.full(n, radius_guess)

    # Objective function to minimize (negative of sum of radii)
    def objective(radii_vec):
        return -np.sum(radii_vec)

    # Constraints for optimization
    # 1. Each circle must be within the unit square
    # 2. No two circles must overlap
    # 3. Radii must be non-negative

    # Optimization bounds: radius between 0 and max_radius
    bounds = Bounds(0.0, max_radius)

    # Optimization parameters
    method = 'L-BFGS-B'

    # Run optimization
    result = optimize.minimize(
        objective,
        radii,
        method=method,
        bounds=bounds,
        jac=None,
        options={'ftol': tolerance, 'maxiter': num_iterations}
    )

    # Get optimized radii and centers
    optimized_radii = result.x
    centers = np.array(centers)  # Reassign centers

    if not result.success:
        raise RuntimeError(f"Optimization failed: {result.message}")

    sum_radii = np.sum(optimized_radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
