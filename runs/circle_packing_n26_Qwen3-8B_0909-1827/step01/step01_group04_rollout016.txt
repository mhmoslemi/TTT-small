import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    num_circles = 26

    def generate_hexagonal_centers(num_circles, spacing):

                y = row * spacing * 1.5 + spacing * 0.75

        return np.array(centers)

    spacing = 0.2

    centers = centers / centers.max()

    # Initial guess for radii
    radii = np.full(num_circles, spacing / 2)

    # Define the cost function to minimize (negative of sum of radii to maximize it)
    def objective(x):
        # x contains all the values: first half is centers, second half is radii
        flat_centers = x[:num_circles * 2].reshape(num_circles, 2)
        flat_radii = x[num_circles * 2:].reshape(num_circles)
        return -np.sum(flat_radii)

    # Define constraints
    constraints = []

    # Boundary constraints
    for i in range(num_circles):
        def boundary_constraint(x):
            flat_centers = x[:num_circles * 2].reshape(num_circles, 2)
            flat_radii = x[num_circles * 2:].reshape(num_circles)
            center = flat_centers[i]

            return np.array([center[0] - r * 1.0, 1.0 - center[0] - r * 1.0, center[1] - r * 1.0, 1.0 - center[1] - r * 1.0])

    # Non-overlap constraints
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            def overlap_constraint(x):
                flat_centers = x[:num_circles * 2].reshape(num_circles, 2)

                return dist - r_i - r_j
            constraints.append({'type': 'ineq', 'fun': overlap_constraint})

    bounds = Bounds(np.zeros(num_circles * 2 + num_circles), np.ones(num_circles * 2 + num_circles))

    # Extract centers and radii

    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
