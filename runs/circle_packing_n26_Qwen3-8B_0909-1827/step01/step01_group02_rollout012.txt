import scipy.optimize as opt
import matplotlib.pyplot as plt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    n = 26

    for i in range(n):
        row = i // 4
        col = i % 4
        x = col / 4.0 + 0.25

        centers[i] = [x, y]
        radii[i] = 0.05

    # Define the objective function to maximize the sum of radii
    def objective(radii_vec):
        return -np.sum(radii_vec)  # Negative for minimization

    # Define constraints: circles must not overlap, remain within the unit square
    def constraints_func(x):

        # Boundary constraints
        for i in range(n):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            constraints.append((x_i - r_i >= -1e-12))
            constraints.append((x_i + r_i <= 1 + 1e-12))

                dist = np.sqrt(dx**2 + dy**2)
