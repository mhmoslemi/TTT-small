import numpy as np
import scipy.optimize as optimize
import matplotlib.pyplot as plt

def run_packing():
    # Step 1: Define initial guess with a hexagonal-like grid
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)
    grid_rows = 6
    grid_cols = 5

    # Step 2: Place initial grid positions (hexagonal grid)
    for i in range(grid_rows):
        for j in range(grid_cols):
            x = j + 0.5 * i / grid_cols
            y = i / grid_rows
            centers[i * grid_cols + j] = [x, y]

    # Step 3: Define the function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Since we want to maximize, we minimize the negative sum

    # Step 4: Define constraints
    def constraints(centers, radii):
        # Bounds constraint (each circle must be within the unit square)
        bounds = [(0.0, 1.0) for _ in range(2 * num_circles)]  # bounds for centers x and y

        # Circle radius non-negativity constraint
        for i in range(num_circles):
            if radii[i] < 0:
                return False, f"Circle {i} has negative radius {radii[i]}"

        # Circle boundary constraint
        for i in range(num_circles):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
                return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

        # All constraints satisfied
        return True, "ok"

    def optimize_radii(centers, radii):

            radii,
