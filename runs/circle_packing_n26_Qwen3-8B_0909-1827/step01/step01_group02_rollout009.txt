import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Step 1: Initialize centers using a grid layout
    num_circles = 26
    grid_size = int(np.ceil(np.sqrt(num_circles)))
    centers = np.zeros((num_circles, 2))
    for i in range(num_circles):
        row = i // grid_size
        col = i % grid_size
        centers[i, 0] = col / (grid_size - 1)
        centers[i, 1] = row / (grid_size - 1)

    # Step 2: Initial radii based on spacing between centers
    initial_radii = np.zeros(num_circles)
    for i in range(num_circles):
        min_dist = np.inf
        for j in range(num_circles):
            if i != j:
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < min_dist:
                    min_dist = dist
        initial_radii[i] = min_dist / 2

    # Step 3: Define optimization variables and parameters
    x = np.concatenate([centers.flatten(), initial_radii])

    # Step 4: Define the objective function (maximize sum of radii)
    def objective(x):
        return -x[-num_circles:]  # Negative for minimization

    # Step 5: Define constraints
    def constraints(x):
        centers = x[:-num_circles].reshape(num_circles, 2)
        radii = x[-num_circles:]
        constraints = []

        # Boundary constraints
        for i in range(num_circles):
            x_i, y_i = centers[i]
            r_i = radii[i]
            constraints.append((x_i - r_i >= -1e-12))
            constraints.append((x_i + r_i <= 1 + 1e-12))
            constraints.append((y_i - r_i >= -1e-12))
            constraints.append((y_i + r_i <= 1 + 1e-12))

        # Non-overlapping constraints
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                constraints.append((dist - (radii[i] + radii[j]) >= -1e-12))

        return constraints

    # Step 6: Run optimization
    bounds = []
    for i in range(num_circles):
        bounds.append((0.0, 1.0))  # x and y coordinates between 0 and 1
    for i in range(num_circles):
        bounds.append((0.0, 1.0))  # radii between 0 and 1

    result = opt.minimize(
        objective,
        x,
        bounds=bounds,
        constraints={"type": "ineq", "fun": constraints},
        method="SLSQP",
        options={"ftol": 1e-9, "maxiter": 1000}
    )

    # Step 7: Extract results
    optimal_centers = result.x[:-num_circles].reshape(num_circles, 2)
    optimal_radii = result.x[-num_circles:]
    sum_radii = np.sum(optimal_radii)

    # Step 8: Validate results
    valid, message = validate_packing(optimal_centers, optimal_radii)
    if not valid:
        raise ValueError(f"Invalid packing found: {message}")

    return optimal_centers, optimal_radii, sum_radii


def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
