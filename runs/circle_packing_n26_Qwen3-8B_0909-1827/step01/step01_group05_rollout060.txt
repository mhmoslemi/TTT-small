import scipy.optimize as opt
from scipy.optimize import Bounds, NonlinearConstraint

def run_packing():

    n_circles = 26

    # Initial placement using a hexagonal grid
    # We'll create concentric layers of circles
    radius_initial = 0.05
    positions = []
    for layer in range(4):
        angle = 2 * np.pi / 6
        for i in range(6):
            x = radius_initial * (layer + 1) * np.cos(i * angle)
            y = radius_initial * (layer + 1) * np.sin(i * angle)
            positions.append([x, y])
        # Add hexagon center
        if layer == 0:
            positions.append([0, 0])
        else:
            positions.append([0, radius_initial * (layer + 1)])
    
    # Add extra circles to reach 26
    for i in range(len(positions), n_circles):
        # Use random perturbation to make initial configuration diverse
        x = np.random.uniform(0, 1)
        y = np.random.uniform(0, 1)
        positions.append([x, y])
    
    # Convert to numpy arrays
    centers = np.array(positions)
    # Initial radii: same for all circles
    radii = np.full(n_circles, 0.05)

    # Objective: maximize the sum of radii
    def objective(radii_flat):
        return -np.sum(radii_flat)

    # Bounds for radii: all radii must be non-negative
    bounds = Bounds(0, np.inf)

    # Constraints for circle packing
    def constraint_overlap(x):
        # x is a flattened array: [c1x, c1y, r1, c2x, c2y, r2, ...]
        # Extract centers and radii
        centers_flat = x[::4]
        radii_flat = x[2::4]
        centers = centers_flat.reshape((-1, 2))
        radii = radii_flat

        # Check for overlaps
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-8:
                    return dist - (radii[i] + radii[j] - 1e-8)
        return 0.0

    # To handle non-overlap as a constraint, we'll need a list of constraints
    nl_constraints = []
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            def constraint_overlap_single(x, i=i, j=j):

                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))

    def constraint_boundary(x):

    x0 = np.concatenate([centers.flatten(), radii])

    options = {

        'gtol': 1e-8,
        'maxiter': 500,
        'disp': False,

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
