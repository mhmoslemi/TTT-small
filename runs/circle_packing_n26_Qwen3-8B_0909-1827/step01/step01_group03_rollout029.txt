import numpy as np
import cvxpy as cp

def run_packing():
    # Initial layout: a 5x5 grid of circles with spacing
    # We will start with an initial guess of circle centers and equal radii
    # This is a simple, even layout that leaves room for optimization

    # Define the initial centers as a grid
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    for i in range(num_circles):
        row = i // 5
        col = i % 5
        centers[i, 0] = (col + 0.5) / 5.0
        centers[i, 1] = (row + 0.5) / 5.0

    # Define the radii variable
    radii = cp.Variable(num_circles, nonneg=True)

    # Objective: Maximize the sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []

    # Each circle must be inside the unit square
    for i in range(num_circles):
        x, y = centers[i]
        constraints.append(x - radii[i] >= 0)
        constraints.append(x + radii[i] <= 1)
        constraints.append(y - radii[i] >= 0)
        constraints.append(y + radii[i] <= 1)

    # No two circles may overlap
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            x1, y1 = centers[i]
            x2, y2 = centers[j]
            dist = cp.sqrt((x1 - x2)**2 + (y1 - y2)**2)
            constraints.append(dist >= radii[i] + radii[j])

    # Solve the optimization problem
    problem = cp.Problem(objective, constraints)
    problem.solve()

    # Extract the results
    centers_opt = centers

    sum_radii = np.sum(radii_opt)
