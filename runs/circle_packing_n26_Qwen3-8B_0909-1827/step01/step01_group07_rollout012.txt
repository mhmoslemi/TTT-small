import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    # Let's create 26 circles initially with a hexagonal arrangement
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # For simplicity, I'll use a basic grid first then optimize it.

    # Step 1: Generate an initial guess using a grid-like pattern
    # with an opportunity to fill space

    cols = int(np.ceil(n / rows))

    # Use a hexagonal-like pattern to start
    for i in range(n):
        r = int(i // cols)
        c = i % cols
        x = c / (cols - 1)
        y = r / (rows - 1) + (r % 2) * 0.25  # Slight offset for hexagonal-like
        centers[i] = [x, y]

    # Step 2: Initial radius guess is based on spacing
    initial_radius = 0.15
    radii[:] = initial_radius

    # Function to compute the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Constraint: All circles are within the square
    def in_square(radii):
        return np.array([r <= 1 - 1e-12 for r in radii], dtype=bool)

    # Constraint: Circle i is not overlapping with circle j
    def no_overlap(radii):
        dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
        return dist >= radii[i] + radii[j] - 1e-12

    # Define optimization variables
    bounds = [(0, 1) for _ in range(n)]

    # Optimization step
    result = opt.minimize(
        fun=objective,
        x0=radii,
        method='SLSQP',
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': in_square},
            {'type': 'ineq', 'fun': lambda r: np.array([no_overlap(r)[i, j] for i in range(n) for j in range(i + 1, n)])}
        ]
    )

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
