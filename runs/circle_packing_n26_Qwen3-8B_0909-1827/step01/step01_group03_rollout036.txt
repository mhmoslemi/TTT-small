import numpy as np
import scipy.optimize as opt
from typing import Tuple, Optional

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    # Initial grid-based placement
    rows, cols = 5, 6  # 5 rows, 6 columns (total 30 circles, we use 26)
    x = np.linspace(0.1, 0.9, cols)
    y = np.linspace(0.1, 0.9, rows)
    centers = np.array([[(xi, yi) for xi in x] for yi in y])[:n]
    radii = np.ones(n) * 0.1  # Initial guess for radii

    def objective(radii_vec):
        return -np.sum(radii_vec)  # Maximize sum of radii

    def constraint_overlap(i, j):
        def cons(radii_vec):
            r_i = radii_vec[i]
            r_j = radii_vec[j]
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = np.sqrt(dx**2 + dy**2)
            return dist - (radii_vec[i] + radii_vec[j]) + 1e-12
        return cons

    def constraint_boundary(i):
        def cons(radii_vec):
            r = radii_vec[i]
            x, y = centers[i][0], centers[i][1]
            return (x - r) - (-1e-12)  # Ensure x - r >= -1e-12
        return cons

    def constraint_boundary_right(i):
        def cons(radii_vec):
            r = radii_vec[i]
            x, y = centers[i][0], centers[i][1]
            return (x + r) - (1 + 1e-12)  # Ensure x + r <= 1 + 1e-12
        return cons

    def constraint_boundary_top(i):
        def cons(radii_vec):
            r = radii_vec[i]
            x, y = centers[i][0], centers[i][1]
            return (y + r) - (1 + 1e-12)  # Ensure y + r <= 1 + 1e-12
        return cons

    def constraint_boundary_bottom(i):
        def cons(radii_vec):
            r = radii_vec[i]
            x, y = centers[i][0], centers[i][1]
            return (y - r) - (-1e-12)  # Ensure y - r >= -1e-12
        return cons

    # Define constraints
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': constraint_overlap(i, j)})
        constraints.append({'type': 'ineq', 'fun': constraint_boundary(i)})
        constraints.append({'type': 'ineq', 'fun': constraint_boundary_right(i)})
        constraints.append({'type': 'ineq', 'fun': constraint_boundary_top(i)})
        constraints.append({'type': 'ineq', 'fun': constraint_boundary_bottom(i)})

    # Define bounds for radii (non-negative)
    bounds = [(0.0, 1.0) for _ in range(n)]

    # Optimization
    result = opt.minimize(
        fun=objective,
        x0=radii,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 200, 'ftol': 1e-8, 'disp': False}

    # Extract result
    radii_opt = result.x
    centers_opt = centers
    sum_radii = np.sum(radii_opt)

    # Validate result
    is_valid, message = validate_packing(centers_opt, radii_opt)
    if not is_valid:
        raise ValueError(f"Invalid packing: {message}")

    return centers_opt, radii_opt, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
