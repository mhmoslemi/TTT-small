import numpy as np
import scipy.optimize
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    def is_within_bounds(centers, radii):
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            if not (0 - 1e-12 <= x - r <= 1 + 1e-12 and 0 - 1e-12 <= y - r <= 1 + 1e-12):
                return False
        return True

    # Helper function to check if circles are non-overlapping
    def is_non_overlapping(centers, radii):
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = math.hypot(dx, dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    n = 26
    grid_size = math.ceil(math.sqrt(n))
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

        row = i // grid_size
        col = i % grid_size
        x = col / (grid_size - 1) if grid_size > 1 else 0.5
        y = row / (grid_size - 1) if grid_size > 1 else 0.5
        centers[i] = np.array([x, y])

    def objective(radii):

    optimized_radii = result.x
