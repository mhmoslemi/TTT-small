import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initial guess: hexagonal packing as a starting point
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Start with a hexagonal grid layout
    # This is a simplified approximation for a hexagonal grid in a square
    # Compute the number of circles per row and column
    rows = int(np.ceil(np.sqrt(num_circles)))
    cols = int(np.ceil(num_circles / rows))
    
    # Define a hexagonal lattice with appropriate spacing
    spacing = 1.0 / (cols - 1)
    radius_init = spacing / 2.5  # Initial homogeneous radius
    
    radii[:] = radius_init
    for i in range(num_circles):
        row = i // cols
        col = i % cols
        x = spacing * col + spacing / 2
        y = spacing * row + spacing / 2
        centers[i, 0] = x
        centers[i, 1] = y

    # Define objective function to maximize sum of radii
    def objective(x):
        # x is a vector of radii: x[0], x[1], ..., x[25]
        # We will maximize the sum of radii, so minimize -sum(radii)
        return -np.sum(x)

    # Define constraints
    def constraint_boundary(x):
        # Check that circles are within the unit square
        for i in range(num_circles):
            x_center = centers[i, 0]
            y_center = centers[i, 1]
            r = x[i]
            if x_center - r < 0 or x_center + r > 1 or y_center - r < 0 or y_center + r > 1:
                return -1
        return 0

    def constraint_overlap(x):
        # Check that all pairs of circles do not overlap
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                distance = np.sqrt(dx**2 + dy**2)
                if distance < x[i] + x[j] - 1e-8:
                    return -1
        return 0

    # Set up optimization problem
    bounds = [(0.0, 1.0) for _ in range(num_circles)]
    constraints = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]

    # Run optimization with SLSQP method
    result = optimize.minimize(
        objective,
        x0=radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
    )

    # Extract results
    radii_opt = result.x
    centers_opt = centers.copy()
    sum_radii = np.sum(radii_opt)

    # Final validation
    valid, msg = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise ValueError(f"Validation failed: {msg}")

    return centers_opt, radii_opt, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
