import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    # Initial guess: uniform distribution of centers

    # Initial radius: small value

    # Objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def boundary_constraint(radii, centers):

                constraints.append(dist - (radii[i] + radii[j]))
        return np.array(constraints)

    def constraints(radii):

        b = boundary_constraint(radii, centers)
        if not b.all():
            raise ValueError("Boundary constraint violated")

        return np.concatenate((b, o))
