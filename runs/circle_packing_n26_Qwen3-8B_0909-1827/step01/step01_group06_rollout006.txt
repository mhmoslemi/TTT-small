import cvxpy as cp

    centers = np.zeros((n, 2))

        centers[i] = [x, y]

    for i in range(1):

        centers[20 + i] = [x, y]

    # Define variables
    radii = cp.Variable(n)

    # Objective: maximize sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []

    # Each circle must be within the square
    for i in range(n):
        x, y = centers[i]
        constraints.append(x - radii[i] >= -1e-12)

    # Solve the optimization problem
    prob = cp.Problem(objective, constraints)
