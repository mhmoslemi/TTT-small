import numpy as np
import scipy.optimize as optimize
import cvxpy as cp

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Optimal packing using a hexagonal grid refinement with optimization
    n_circles = 26
    num_rows = 5  # For hexagonal arrangement
    num_cols = 5
    
    # Initial grid spacing calculation
    # Aim to fit about 5 rows and 5 columns with some spacing
    grid_spacing = 0.25  # Initial spacing
    initial_centers = np.zeros((n_circles, 2))
    initial_radii = np.ones(n_circles) * 0.05  # Small initial radius

    # Generate hexagonal grid
    idx = 0
    for row in range(num_rows):
        for col in range(num_cols):
            x = col * grid_spacing
            y = row * grid_spacing * np.sqrt(3) / 2
            if row % 2 == 1:  # Shift for staggered rows
                x += grid_spacing / 2
            initial_centers[idx, 0] = x
            initial_centers[idx, 1] = y
            idx += 1

    # Define the problem: maximize sum of radii under constraints
    centers = cp.Variable((n_circles, 2))
    radii = cp.Variable(n_circles, nonneg=True)

    # Objective: maximize sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []

    # Boundary constraints
    for i in range(n_circles):
        x = centers[i, 0]
        y = centers[i, 1]
        r = radii[i]
        constraints += [
            x >= r,
            x <= 1 - r,
            y >= r,
            y <= 1 - r
        ]

    # Non-overlap constraints
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = cp.sqrt(dx**2 + dy**2)
            constraints += [dist >= radii[i] + radii[j]]

    # Initial guess from the grid
    initial_centers_guess = initial_centers.astype(np.float64)
    initial_radii_guess = initial_radii.astype(np.float64)

    # Convert to convex form
    prob = cp.Problem(objective, constraints)

    # Solve the optimization problem
    result = prob.solve(solver=cp.SCS, verbose=False, warm_start=True,
                        x0=initial_centers_guess, y0=initial_radii_guess)

    if result is None or result < 0:
        raise RuntimeError("Optimization failed")

    centers = centers.value
    radii = radii.value
    sum_radii = np.sum(radii)

    # Validate packing
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise RuntimeError(f"Invalid packing: {msg}")

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
