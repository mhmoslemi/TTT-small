import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    num_circles = 26

    # Generate an initial configuration based on hexagonal packing
    # This is a heuristic to get a starting point

    radii = np.zeros(num_circles)

        row = i // cols

        x = col + 0.5 * (row % 2)
        y = row * np.sqrt(3) / 2
        centers[i, 0] = x
        centers[i, 1] = y

    # Scale centers to fit into the unit square
    max_x = np.max(centers[:, 0])
    max_y = np.max(centers[:, 1])
    centers = centers / np.max([max_x, max_y])

    def objective(radii):
        return -np.sum(radii)

    def constraint_non_overlap(x):
        centers = x[:num_circles, :2]
        radii = x[num_circles:]

            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0

    def constraint_boundary(x):
        centers = x[:num_circles, :2]
        radii = x[num_circles:]

            x_center, y_center = centers[i]
            r = radii[i]
            if x_center - r < 0 or x_center + r > 1 or y_center - r < 0 or y_center + r > 1:
                return -1
        return 0

    # Combine constraints
    def constraint_total(x):
        return constraint_non_overlap(x) + constraint_boundary(x)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
