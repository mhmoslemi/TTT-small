import numpy as np
import scipy.optimize as opt

def run_packing():
    # Initial packing using a hexagonal close packing (HCP) configuration
    num_circles = 26
    rows = 5
    cols = 6

    # Initial estimate for each circle's radius
    radius = 0.045  # Estimate based on area and square size

    # Initial grid points in hexagonal packing
    centers = np.zeros((num_circles, 2))
    idx = 0
    for row in range(rows):
        for col in range(cols):
            # Odd rows are offset to simulate hexagonal packing
            if row % 2 == 1:
                x = col * (2 * radius) + radius
            else:
                x = col * (2 * radius) + radius
            y = row * (np.sqrt(3) * radius) + radius
            centers[idx, 0] = x
            centers[idx, 1] = y
            idx += 1

    # Ensure that we only have 26 circles
    if idx > num_circles:
        centers = centers[:num_circles, :]
    elif idx < num_circles:
        # Add additional circles near the center if needed
        for i in range(idx, num_circles):
            x = 0.5
            y = 0.5
            centers = np.vstack((centers, [x, y]))

    # Define the objective function to maximize (we minimize negative of sum)
    def objective(radii):
        return -np.sum(radii)

    # Define the constraints: circles must not overlap and be within bounds
    def constraint_overlap(i, j):
        def cons(x):
            x_i = x[i]
            x_j = x[j]
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            return dist - (x_i + x_j) + 1e-12
        return cons

    def constraint_boundary(i):
        def cons(x):
            x_i = x[i]
            return (x_i - 1e-12) - 1.0  # Ensure not exceeding 1.0
        return cons

    # Prepare constraints
    constraints = []

    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            cons = constraint_overlap(i, j)
            constraints.append({'type': 'ineq', 'fun': cons})
        cons = constraint_boundary(i)
        constraints.append({'type': 'ineq', 'fun': cons})

    # Initial guess for radii
    radii_initial = np.full(num_circles, radius)

    # Set bounds for radii: must be non-negative
    bounds = [(0.0, 1.0) for _ in range(num_circles)]

    # Run optimization
    result = opt.minimize(
        objective,
        radii_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        tol=1e-7,
        options={'maxiter': 1000}
    )

    # Extract the optimal centers and radii
    centers = np.array(centers)
    radii = result.x
    sum_radii = np.sum(radii)

    # Validate the packing
    is_valid, msg = validate_packing(centers, radii)
    if not is_valid:
        raise ValueError(f"Packing is invalid: {msg}")

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
