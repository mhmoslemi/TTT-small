import scipy.optimize as optimize

def run_packing():

    centers = np.random.rand(26, 2)
    radii = np.random.rand(26) * 0.1  # Initial small radii

    def objective(radii_vector):
        return -np.sum(radii_vector)  # Minimize negative sum to maximize sum

    # Define constraints
    def constraint_boundary(radii_vector):
        return np.sum((centers[:, 0] - radii_vector) < 0) + np.sum((centers[:, 0] + radii_vector) > 1) + \

    def constraint_overlap(radii_vector):

                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii_vector[i] + radii_vector[j] - 1e-12:
                    return -1
        return 0

    # Define bounds for radii
    bounds = [(0, 1) for _ in range(26)]

    # Set up optimization problem with constraints
    constraints = [

            'fun': lambda r: constraint_boundary(r)
        },
        {
            'type': 'ineq',
            'fun': constraint_overlap
        }
    ]

    # Run the optimization
    result = optimize.minimize(

        radii,
        method='SLSQP',
        bounds=bounds,
