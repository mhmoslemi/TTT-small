```python
import numpy as np
import scipy.optimize as optimize
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Start with a hexagonal packing of 26 circles
    # In a hexagonal packing, the number of circles in a 6-layer setup is
    # 1 + 6 + 12 + 18 + 24 + 30 = 81 (greater than 26), so we pick a smaller setup
    # For simplicity, we'll use a hexagonal packing where:
    # - Each row has 5 circles
    # - Total of 5 rows with 5 circles each = 25, plus 1 in the center = 26

    # Set up a grid of centers
    # Start with a hexagonal packing
    num_circles = 26
    rows = 5
    cols_per_row = 5
    spacing = 0.1  # Initial spacing (to be adjusted)
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    for i in range(rows):
        for j in range(cols_per_row):
            row_y = i * spacing * math.sqrt(3)
            col_x = j * spacing
            centers[i * cols_per_row + j] = [col_x, row_y]
            radii[i * cols_per_row + j] = spacing / 2  # Initial estimate

    # Center circle (row 2, col 2)
    centers[12] = [0.5, 0.5]
    radii[12] = spacing / 2

    # Try a quadratic programming approach to adjust radii and positions to maximize sum
    # This is a non-convex problem, so we'll use a local optimizer with good initial guess

    # Define variables
    x = np.concatenate((centers.flatten(), radii))

    # Objective function to maximize: sum(radii)
    def objective(x):
        centers_new = x[:2 * num_circles].reshape(num_circles, 2)
        radii_new = x[2 * num_circles:]
        return -np.sum(radii_new)  # Minimize negative of sum

    # Bounds: positions in [0, 1] and radii >= 0
    bounds = []
    for i in range(num_circles):
        bounds.append((0, 1))  # x
        bounds.append((0, 1))  # y
        bounds.append((0, 1))  # radius

    # Constraints: distance between centers >= sum of radii
    def constraint_func(x):
        centers_new = x[:2 * num_circles].reshape(num_circles, 2)
        radii_new = x[2 * num_circles:]
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers_new[i, 0] - centers_new[j, 0]
                dy = centers_new[i, 1] - centers_new[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii_new[i] + radii_new[j] - 1e-12:
                    return dist - (radii_new[i] + radii_new[j])
        return 0

    cons = {'type': 'ineq', 'fun': constraint_func}

    # Optimization
    result = optimize.minimize(
        objective,
        x,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-6
    )

    # Extract results
    centers_opt = result.x[:2 * num_circles].reshape(num_circles, 2)
    radii_opt = result.x[2 * num_circles:]

    # We must validate the packing
    valid, msg = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise RuntimeError(msg)

    sum_radii = np.sum(radii_opt)
    return centers_opt, radii_opt, sum_radii
```

```python
def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
```
