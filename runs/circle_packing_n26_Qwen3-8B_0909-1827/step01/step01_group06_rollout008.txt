import scipy.optimize as opt
from scipy.optimize import Bounds

def validate_packing(centers, radii):

        return False, "NaN values present"

        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def objective_function(x, num_circles=26):
    # Flatten the input for easy access to centers and radii
    x_flat = x.reshape(num_circles, 3)
    centers = x_flat[:, :2]
    radii = x_flat[:, 2]
    return -np.sum(radii)  # Minimize the negative sum to maximize the sum

def constraint_boundary(x, num_circles=26):
    x_flat = x.reshape(num_circles, 3)

def constraint_overlap(x, num_circles=26):
