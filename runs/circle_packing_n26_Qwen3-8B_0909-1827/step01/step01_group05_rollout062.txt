import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds

def run_packing():

    n = 26
    # Create a grid with 5x5 cells (25 circles) and an additional circle in the center
    grid = np.array([(i / 4, j / 4) for i in range(5) for j in range(5)])
    centers = np.vstack((grid, [0.5, 0.5]))
    # Start with uniform radii
    radii = np.full(n, 0.1)

    def objective(radii, centers):
        return -np.sum(radii)  # Negative for minimization

    def constraint_centers(radii, centers):
        # Boundary constraint
        for i in range(n):
            x, y = centers[i]
            r = radii[i]

                return False
        return True

        # Pairwise distance constraint
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    # Bounds for radii (non-negative)
    bounds = Bounds(0, np.inf)

    result = optimize.minimize(
        fun=lambda r: objective(r, centers),

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"
