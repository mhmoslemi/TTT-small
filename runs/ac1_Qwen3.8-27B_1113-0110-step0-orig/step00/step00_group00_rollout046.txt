import numpy as np
import time

def fft_convolve(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(fa * fa, fft_size)[:size]
    return conv

def score(a):
    """Compute the score of a non-negative sequence."""
    conv = fft_convolve(a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return 2 * len(a) * max_conv / (sum_a * sum_a)

For a sequence of length n, the score is determined by the maximum of 2n-1 convolution values. The "active" lag (where the max occurs) is the one we most want to reduce.

**Local search strategy:**
1. Compute conv, find the active lag k*.
2. The gradient of conv[k*] with respect to a[j] is (a[k*-j] + a[j]) (with bounds).
3. Move in the negative gradient direction (with projection to non-negative).
4. Use adaptive step size.

But this is just gradient descent on one of the convolution values, which might increase others.

**Better: subgradient descent on the max.**
The subgradient of max(conv) is the gradient of the active component. If multiple lags are at the max, we can use the average of their gradients.

Let me implement this:

```python
def subgradient_step(a, eta, rng):
    conv = fft_convolve(a)
    max_val = np.max(conv)
    # Find all lags at (or near) the maximum
    threshold = max_val * 0.999  # Slightly below max to get multiple active lags
    active_lags = np.where(conv >= threshold)[0]
    
    # Compute subgradient (average of active gradients)
    n = len(a)
    grad = np.zeros(n)
    for k in active_lags:
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                if k == 2 * j:
                    grad[j] += 2 * a[j]
                else:
                    grad[j] += a[kj] + a[j]
            # Actually, let me reconsider
            # conv[k] = sum_i a[i]*a[k-i]
            # d(conv[k])/d(a[j]) = a[k-j] (if valid) + a[j] (if k-j valid)
            # But if k-j == j (k=2j), it's 2*a[j]
    grad /= len(active_lags)
    
    # Update
    a_new = a - eta * grad
    return np.clip(a_new, 0, None)

def compute_subgradient(a, active_lags):
    n = len(a)
    grad = np.zeros(n)
    for k in active_lags:
        # For each j, gradient contribution is 2*a[k-j] if 0 <= k-j < n
        # k-j ranges from k-(n-1) to k as j ranges from n-1 to 0
        # Valid j: max(0, k-n+1) <= j <= min(n-1, k)
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        # For j in [j_min, j_max]: grad[j] += 2 * a[k-j]
        # k-j ranges from k-j_min to k-j_max
        idx = k - np.arange(j_min, j_max + 1)
        grad[j_min:j_max+1] += 2 * a[idx]
    grad /= len(active_lags)
    return grad
```

import numpy as np

def _fft_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(fa * fa, fft_size)[:size]
    # Clip small negative values from numerical error
    np.maximum(conv, 0, out=conv)
    return conv


def _score(a):
    """Compute the evaluation score for a non-negative sequence."""
    n = len(a)
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    conv = _fft_convolve(a)
    max_conv = conv.max()
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient(a, active_lags):
    """Compute the subgradient of max(conv) at the given active lags."""
    n = len(a)
    grad = np.zeros(n, dtype=np.float64)
    for k in active_lags:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min > j_max:

        idx = k - np.arange(j_min, j_max + 1)
        grad[j_min:j_max+1] += 2.0 * a[idx]

    return grad

    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = _fft_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def _lp_step(a, M):
    """Solve the LP: max sum(b) s.t. b >= 0, conv(a,b)[k] <= M for all k."""
    import scipy.optimize as opt
    n = len(a)
    
    # conv(a, b)[k] = sum_i a[i] * b[k-i] <= M
    # This is: for each k, sum_i a[i] * b[k-i] <= M
    # Which is a linear constraint in b.
    
    # The constraint matrix A has shape (2n-1, n)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    
    # Objective: maximize sum(b) → minimize -sum(b)
    c = -np.ones(n)
    
    # Constraints: A @ b <= M (for each k)
    # A[k, j] = a[k - j] if 0 <= k-j < n
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                A[k, j] = a[kj]
    
    # Also b >= 0 is handled by bounds
    bounds = [(0, None)] * n
    constraints = [{'type': 'ineq', 'fun': lambda b, A=A, M=M: M - A @ b}]
    
    # Use scipy's linprog
    result = opt.linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    return None

def _lp_blend_step(a, t=0.1):
    """One iteration of the LP-based method."""
    from scipy.optimize import linprog
    n = len(a)
    sum_a = a.sum()
    if sum_a < 0.01:
        return a
    
    # Compute M = max(a * a)
    conv_aa = _fft_convolve(a)
    M = conv_aa.max()
    
    # Solve LP: max sum(g) s.t. A@g <= M, g >= 0
    # A[k, j] = a[k-j] if 0 <= k-j < n
    # To save memory, we can use a sparse matrix
    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                if a[kj] > 0:  # Only add non-zero entries
                    rows.append(k)
                    cols.append(j)
                    vals.append(a[kj])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # LP: max 1^T g s.t. A@g <= M, g >= 0
    result = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M), 
                     bounds=[(0, None)]*n, method='highs')
    
    if not result.success:
        return a
    
    g = result.x
    sum_g = g.sum()
    if sum_g < 0.01:
        return a
    
    # Rescale g to match sum_a
    g = g * (sum_a / sum_g)
    
    # Blend
    new_a = (1 - t) * a + t * g
    new_a = np.clip(new_a, 0, 1000)
    
    return new_a
