        n_fallback = 256
        candidate = np.ones(n_fallback) / np.sqrt(2 * n_fallback)
        candidate_sequence = candidate.tolist()
        print("[First-stage fallback] Initialized with self-normalized probability simplex of size %i" % n_fallback)
    else:
        parent = np.asarray(kwargs['height_sequence_1'], dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            candidate_sequence = [0.0] + [42.0]*parent.size + [0.0]  # Fallback simple
            print("[First-stagemsg] Provided sequence invalid, returning fallback of %i zeros and %i constants" % (2, parent.size))
        else:
            candidate_sequence = parent.tolist()
            # Normalize to the simplex if we need to
            n_val = parent.size
            s_val = float(np.sqrt(2 * n_val))
            sum_par = np.sum(parent)
            candidate_normalized = np.clip(parent / max(s_val, sum_par), 0, 1000.0)
            candidate_sequence = candidate_normalized.tolist()
            print("[First-stage] Using provided sequence of size %i" % n_val)
            n_val = parent.size
    
    # Parameter container
    param_save = {
        'sequence': candidate_sequence[:],
        'adaptive_iterations': 0,
        'budget_remaining': budget_s,
        'current_objective': float(evaluate_sequence(candidate_sequence)),
        'faintness_shutoff': 2.0 if n_val < 100 else (2.5 if n_val < 512 else 1.5),  # Progressive failure threshold
        'time_remaining': budget_s
    }
    
    # Record time for remaining operations
    first_eval_time = time.time() - start_time
    param_save['budget_remaining'] -= first_eval_time
    
    # --- Setup --- Possibly iterate over n with length scaling ---
    # n_val is fixed as the length we started with (either from provided parent, randomly 256, or fallback)
    # Two phases: simplex projection, local search, and length increase
    
    # Adaptive execution plan
    def adaptive_phase(time_left, candidate=current_candidate):
        # Discard too bad candidates
        if param_save['current_objective'] >= param_save['faintness_shutoff']:
            print("[Early abandonment]: Current objective %.5f exceeds threshold %.3f, switching to fallback" % (param_save['current_objective'], param_save['faintness_shutoff']))
            return 'fallback'
        
        # Avoid poisonously mis-scaled sequences
        scale_components = param_save['current_objective'] >= 0.01  # It should always be true for normalized simplex candidates
        if not scale_components:
            print("[Fallbacktrigger]: Objective constraint violation, falling back to probability simplex")
            return 'fallback'
            
        # Decide on the active search set
        if time_left >= 180:
            return 'convolution_simplex'  # Full exploration
        elif time_left < 120:
            return 'diffusive'  # Sweep strategy
        else:
            return 'localized'  # Targeted updates
    
    # --- Second-stage procedures ---
    def simplex_proj(candidate):
        # New representation: Local norm deformation
        simplex_factor = param_save['faintness_shutoff'] / (param_save['current_objective'] + 1e-3)
        deformed_sequence = [x * simplex_factor for x in candidate]
        total_deformed = sum(deformed_sequence)
        normalized_deformed = [x / total_deformed if total_deformed else x for x in deformed_sequence]
        # Project the sequence to the simplex with adjustments for bounds
        if any(x < 0 for x in normalized_deformed) or any(x > 1000 for x in normalized_deformed):
            normalized_deformed = [float(x) for x in np.clip(np.array(normalized_deformed) * (sum(normalized_deformed)-1) * (1000/sum(normalized_deformed)), 0, 1000.0).tolist()]
        param_save['current_objective'] = float(evaluate_sequence(normalized_deformed))
        print("[SimplexProject] New norm factor %.2f -> current objective %.5f" % (simplex_factor, param_save['current_objective']))
        return normalized_deformed
        
    def length_double(candidate):
        # Gently increase length for potential improvements
        n_curr = len(candidate)
        # Subjective heuristic: allow growth when current objective is borderline
        if param_save['current_objective'] > param_save['faintness_shutoff'] * 0.9:
            # Double length with zeros
            new_len = min(4000, n_curr*2)
            zeros_to_add = new_len - n_curr
            double_candidate = candidate + [0.0] * zeros_to_add
            # Re-normalize
            sum_par = np.sum(candidate)
            new_sum = sum_par + sum([0]*zeros_to_add)
            current_norm = max(0.01, sum_par)
            scaling = new_sum / current_norm
            if scaling < 500: # Defensive scaling
                scaling = 500.0
            normalized_candidate = [x * scaling for x in candidate] + [0.0] * zeros_to_add
            param_save['current_objective'] = float(evaluate_sequence(normalized_candidate))
            print("[LengthDouble] From %i to %i length, objective now %.5f" % (n_curr, new_len, param_save['current_objective']))
            return normalized_candidate
        else:
            return candidate
            
    def gradient_step(candidate):
        # Compute generalized gradient projection for full convolution minimization
        # Formulate as: minimize maximize_conv term
        # We will solve a linear program approximately for gradient reversal
        n_val = len(candidate)
        # Normalization for gradient stability
        s0 = np.sqrt(2*n_val)
        norm_sum = np.sum(candidate)
        scaling = s0 / max(0.01, norm_sum)
        scaled_candidate = np.asarray([x * scaling for x in candidate])
        
        # Now the objective (after normalization) is max(scaled_candidate) and should be small.
        # Instead, use a target T=param_save['faintness_shutoff'] and do iterative refinement.
        T = param_save['faintness_shutoff']
        current_best = scaled_candidate
        current_index = np.argmax(scaled_candidate * 1.0 if False else np.convolve(scaled_candidate, scaled_candidate)[:])
        
        # We can only maximize the convex combination via the unit simplex
        # Instead, we use a heuristic: if the maximum is at a single index, try to reduce it by distributing non-negative masses
        
        # For large n, direct optimization is O(n^3). We need to be sparse.
        try:
            # Attempt to explicit gradient descent on the active index (if there is one) for local minimum
            current_max_val = np.max(np.convolve(candidate, candidate))
            potential_improvement = -scaling * current_max_val  # Negative because we want to minimize
            # Backward step?
            present_val = candidate[current_index]
            if present_val >= 1e-3 * scaling:
                target_reduction = 0.01 * param_save['current_objective']  # Only 1% reduction
                candidate_adjusted = np.where(current_index == current_index, candidate - target_reduction, candidate)
                # Random center perturbation
                center_perturb = np.random.randint(0, n_val)
                candidate_adjusted[center_perturb] = max(candidate_adjusted[center_perturb], 1e-7)
                
                # Provide future max to current state gradient estimate by reuse
                param_save['current_objective'] = potential_improvement
                print("[GradientStep] Attempting reduction at index %i" % current_index)
                return candidate_adjusted
        except:
            # Fallback to simplex-based if heavy gradient descent fails or is not applicable
            pass
        
        # Fallback: better use simplex method
        print("[FallbackGradient] Using simplex projection")
        return simplex_proj(candidate)
        
    def evaluate_sequence(seq):
        """ Helper evaluates sequence with bounds checking """
        if not isinstance(seq, (list, tuple)) or any not (isinstance(x, (int, float)) and np.isfinite(x) for x in seq):
            print("[evaluate_sequence] Invalid input")
            np.inf
            
        # Now compute
        n_len = len(seq)
        # Convert to seconds remaining for the whole budget
        sequence_array = np.clip(np.asarray(seq, dtype=np.float64), 0, 1000.0).tolist()
        if n_len < 2:  # Handle edge cases
            print("[evaluate_sequence] Edge case: sequence length %i, returning %f" % (n_len, np.inf))
            return np.inf
        # Compute max of convolution
        conv_result = np.convolve(sequence_array, sequence_array)
        if len(conv_result) == 0:
            print("[evaluate_sequence] Zero-length convolution, returning nothing")
            return np.inf
            
        max_conv = np.max(conv_result)
        total_sum = np.sum(seq)
        
        # Check if sum is sufficient
        if total_sum < 0.01:
            print("[evaluate_sequence] Sum too small: %.5f < 0.01" % total_sum)
            return np.inf
            
        return 2.0 * n_len * max_conv / (total_sum**2)
    
    # --- Core sections ---
    def next_step():
        candidate_global = candidate_sequence
        adapter = adaptive_phase(param_save['budget_remaining'], candidate=candidate_sequence[:])
        
        if adapter == 'fallback':
            # Straightreturn to the starting point with scaling
            best_estimate = probability_simplex_initial()
            print("[Finalization] Early fallback triggered")
            return best_estimate
            
        elif adapter == 'convolution_simplex':
            # Full convolution-based optimization is too heavy; use sparse representation
            pass
            
        elif adapter == 'localized':
            # Try to do better than slow mixer
            print("[Localized] Trying gradient-controlled updates")
            candidate_global = gradient_step(candidate_sequence)
        
        def probability_simplex_initial():
            n_val = param_save.get('n_val', 0)
            if n_val == 0:
                fallback = 512
            else:
                fallback = n_val
            simplex = np.ones(fallback) / np.sqrt(2 * fallback)
            return np.clip(simplex, 0, 1000.0).tolist()
            
        print("[AdaptivePhase] Switched strategy based on %.2f budget remaining" % param_save['budget_remaining'])
        
        # We must choose an akin: here only probability simplex or local searching
        if param_save['budget_remaining'] >= 500:
            return candidate_sequence
            
    def evaluate_adaptive(cand, time_left):
        if time_left < 60:
            return True  # Decisive move
        else:
            return False  # Iterative fine-tuning
        
    # --- To iterate without falling below fallback hardness ---
    candidates = []
    t_known_cur = np.inf
    n_initial = 0
    best_candidate_so_far = candidate_sequence
    
    # With only the given budget_s=1800 seconds, but model complex evolution
    # This will consider up to 4 generations of adaptive learning
    
    # params
    budget_safe = 200.0  # Time in seconds for fallback safety
    budget_stop = 1200.0  # Max runtime budget for the main phase
    
    # Outer adaptive loop
    iteration_count = 0
    time_budget = budget_s - 300.0  # Reserve some time for recovery
    
    while time_budget > 0.0 and iteration_count < 4:
        iteration_count += 1
        
        # First, simplex projection if the candidate is unreasonably scaled
        if evaluate_adaptive(best_candidate_so_far, time_budget):
            proj_candidate = simplex_proj(best_candidate_so_far)
        else:
            proj_candidate = best_candidate_so_far
            
        # Second, if the objective is above threshold, try length doubling
        if evaluate_sequence(proj_candidate) > 0.99 * budget_stop:
            expand_cand = length_double(proj_candidate)
        else:
            expand_cand = None
            
        # Third, only at this point we can determine the improvement
        # But also consider that we can compare previous and present objective without recalc
        candidates.append(expand_cand if expand_cand else proj_candidate)
        print("[AdaptiveIter]: Iteration %i," % iteration_count)
        
        if iteration_count == 4:
            break
            
        # Time decay
        time_budget -= budget_safe
    
    # Final candidate is the best of the lot
    # But note, in case time is exhausted, we have a fallback design
    best_candidate = None
    for cand in candidates:
        cand_val = float(evaluate_sequence(cand))
        current_val = float(evaluate_sequence(candidate_sequence))
        best_candidate_so_far = cand if cand_val < current_val else best_candidate_so_far
        
        if np.isnan(cand_val) or np.isinf(cand_val):
            cand_val = np.inf
        
        # Update if better violation is found
        if cand_val < current_val:
            # We haven't written to `evaluate_sequence` directly, but ensure consistency
            current_val = cand_val
            # Also, we need to check if the candidate time_windows is pruned but we have adaptive_phase done that
            
    # Fallback mechanism if no candidate found
    if best_candidate is None or best_candidate_val >= np.inf:
        print("[LTimout] Fallback sequence activated")
        n_used = param_save.get('n_val', 256)
        fallback_simplex = np.ones(n_used) / np.sqrt(2*n_used)
        fallback = [float(x) for x in fallback_simplex] + [0.0]*(2)
        best_candidate = fallback
        best_candidate_so_far = fallback
        
    # Make sure the best candidate is evaluated with the same objective function
    best_score = float(evaluate_sequence(best_candidate_so_far))
    
    # And then return it
    final_sequence = [float(x) for x in best_candidate_so_far]
    return final_sequence
