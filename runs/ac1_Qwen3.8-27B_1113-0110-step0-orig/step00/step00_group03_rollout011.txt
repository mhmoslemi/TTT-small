import numpy as np
import time
from scipy.optimize import linprog


def build_conv_matrix(f):
    """Build matrix A such that A @ g = full_convolution(f, g)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            A[k, k - i] = f[i]
    return A


def lp_direction(f, M):
    """Solve max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    n = len(f)
    try:
        A = build_conv_matrix(f)
        b = np.full(2 * n - 1, M)
        obj = -np.ones(n)
        res = linprog(obj, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n, method='highs')
        if res.success:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-12:
                return g
    except Exception:
        pass
    return None


def make_perturbation(seq, rng):
    """Random multiplicative block perturbation."""
    n = len(seq)
    cand = seq.copy()
    n_blocks = int(rng.integers(1, 4))
    for _ in range(n_blocks):
        w = max(1, int(n * rng.uniform(0.003, 0.04)))
        s = int(rng.integers(0, max(1, n - w)))
        cand[s:s + w] *= float(np.exp(rng.normal(0, 0.05, 1)))
    if rng.random() < 0.12:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def normalize_sum(a, target):
    """Scale a to have the desired sum, keeping non-negativity."""
    s = a.sum()
    if s < 1e-15:
        return a
    return a * (target / s)
