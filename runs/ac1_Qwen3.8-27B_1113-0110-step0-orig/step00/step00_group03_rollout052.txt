"""
Minimizes score = 2*n*max(convolve(a,a))/sum(a)^2 for non-negative sequence a.
Uses LP-based iterative direction finding with line search, combined with
targeted coordinate descent for fine-tuning. Multiple lengths and initializations.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fft_autocorr(a):
    """Compute linear autocorrelation of a (length n) via FFT, result length 2n-1."""
    n = a.size
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    ac = np.fft.irfft(fa * fa, size)
    return ac[:2 * n - 1]


def fast_score(a):
    """Compute 2*n*max(c)/S^2 where c=autocorr(a), S=sum(a)."""
    n = a.size
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_autocorr(a)
    return 2.0 * n * float(c.max()) / (S * S)


def build_lp_indices(n):
    """Precompute (row, col) indices for the LP constraint matrix A[k,j] = a[k-j]."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int64), np.array(cols, dtype=np.int64)


def solve_lp_direction(a, M, rows, cols, n):
    """Solve max sum(b) s.t. b>=0, (a*b)(k)<=M for all k. Returns b or None."""
    if a.size != n:
        return None
    vals = a[rows - cols]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={"presolve": True, "time_limit": max(1.0, 0.8 * (1800 - 0))})
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0.0)
            if b.sum() > 1e-12:
                return b
    except Exception:
        pass
    return None


def line_search_score(a, b, max_evals=14):
    """Find t in [0,1] minimizing fast_score((1-t)*a + t*b). Returns t."""
    diff = b - a
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = fast_score(a + c_pt * diff)
    fd = fast_score(a + d_pt * diff)
    for _ in range(max_evals - 2):
        if fc <= fd:
            hi = d_pt
            d_pt = c_pt
            f_old = fc
            fc = fast_score(a + c_pt * diff)
            fd = f_old
        else:
            lo = c_pt
            c_pt = d_pt
            f_old = fd
            fd = fast_score(a + d_pt * diff)
            fc = f_old
    return (lo + hi) / 2.0

        if M < 1e-15:
            break
        b = solve_lp_direction(a, M, rows, cols, n)
        if b is None:
            stale += 1
            if stale > 5:
                break
            b = solve_lp_direction(a, M * 1.5, rows, cols, n)
            if b is None:
                break
        Sb = b.sum()
        if Sb < 1e-12:
            break
        b = b / Sb
        if np.max(np.abs(b - a)) < 1e-10:
            stale += 1
            if stale > 8:
                break
            a = a + 0.05 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            continue
        t = line_search_score(a, b, max_evals=12)
        a_new = a + t * (b - a)
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        s_new = fast_score(a_new)
        if s_new < best_score - 1e-10:
            best = a_new.copy()
            best_score = s_new
            stale = 0
            if it % 5 == 0:
                print(f"    LP it={it}, score={s_new:.6f}, t={t:.3f}")
        else:
            stale += 1
            if stale > 15:
                break
        a = a_new
        it += 1
    return best


def coord_shave(a, time_budget):
    """Targeted coordinate descent: move mass from high-gradient to low-gradient elements."""
    n = a.size
    a = a.copy().astype(np.float64)
    S = a.sum()
    if S < 1e-15:
        return a
    a = a / S
    best = a.copy()
    best_score = fast_score(a)
    deadline = time.monotonic() + time_budget
    rng = np.random.default_rng(2024)
    it = 0
    no_improve = 0
    while time.monotonic() < deadline:
        c = fft_autocorr(a)
        k_star = int(np.argmax(c))
        # Gradient: grad[j] = 2*a[k_star - j] if 0 <= k_star-j < n, else 0
        j_valid = np.arange(n)
        partner_idx = k_star - j_valid
        grad = np.zeros(n)
        mask = (partner_idx >= 0) & (partner_idx < n)
        grad[mask] = 2.0 * a[partner_idx[mask]]
        grad_max = grad.max()
        if grad_max < 1e-15:
            break
        # Find source (high grad) and target (low grad)
        i_src = int(np.argmax(grad))
        i_dst_candidates = np.where(grad < grad_max * 0.25)[0]
        if i_dst_candidates.size == 0:
            i_dst_candidates = np.where(grad == grad.min())[0]
        i_dst = int(rng.choice(i_dst_candidates))
        if i_src == i_dst:
            i_dst = int(rng.choice(np.delete(np.arange(n), i_src)))
        # Step size
        delta = a[i_src] * 0.15
        if delta < 1e-15:
            delta = 1e-8
        a_new = a.copy()
        a_new[i_src] -= delta
        a_new[i_dst] += delta
        if a_new[i_src] < 0:
            a_new[i_src] = 0.0
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        s_new = fast_score(a_new)
        if s_new < best_score - 1e-10:
            best = a_new.copy()
            best_score = s_new
            a = a_new
            no_improve = 0
            if it % 1000 == 0 and it > 0:
                print(f"    CD it={it}, score={s_new:.6f}")
        else:
            # Try smaller step or random perturbation
            improved = False
            for frac in [0.05, 0.02, 0.3]:
                delta2 = a[i_src] * frac
                if delta2 < 1e-15:
                    continue
                a_try = a.copy()
                a_try[i_src] -= delta2
                a_try[i_dst] += delta2
                if a_try[i_src] < 0:
                    a_try[i_src] = 0.0
                S_try = a_try.sum()
                if S_try < 1e-12:
                    continue
                a_try = a_try / S_try
                s_try = fast_score(a_try)
                if s_try < best_score - 1e-10:
                    best = a_try.copy()
                    best_score = s_try
                    a = a_try
                    no_improve = 0
                    improved = True
                    break
            if not improved:
                no_improve += 1
                if no_improve > 200:
                    # Random restart perturbation
                    a_reshuff = a * (1.0 + 0.1 * rng.standard_normal(n))
                    a_reshuff = np.maximum(a_reshuff, 0.0)
                    if a_reshuff.sum() > 1e-10:
                        a_reshuff = a_reshuff / a_reshuff.sum()
                        s_r = fast_score(a_reshuff)
                        if s_r < best_score:
                            best = a_reshuff.copy()
                            best_score = s_r
                            a = a_reshuff
                        else:
                            a = a_reshuff
                    no_improve = 0
        it += 1
    return best


def make_inits(n, rng):
    """Generate diverse initializations of length n (normalized to sum=1)."""
    inits = []
    x = np.linspace(-1.0, 1.0, n)
    # Uniform
    inits.append(np.ones(n, dtype=np.float64) / n)
    # Tent
    t = np.maximum(1.0 - np.abs(x), 0.05)
    inits.append(t / t.sum())
    # U-shape
    u = 1.0 + 4.0 * x * x
    inits.append(u / u.sum())
    # Two peaks
    tp = np.exp(-30.0 * (x - 0.4) ** 2) + np.exp(-30.0 * (x + 0.4) ** 2) + 0.05
    inits.append(tp / tp.sum())
    # Cosine window
    cw = 0.5 + 0.5 * np.cos(np.pi * x)
    cw = np.maximum(cw, 0.01)
    inits.append(cw / cw.sum())
    # Random
    r = rng.random(n) + 0.05
    inits.append(r / r.sum())
    # Alternating
    alt = np.ones(n)
    alt[::2] = 1.0
    alt[1::2] = 0.6
    inits.append(alt / alt.sum())
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Combines LP-based direction finding, line search, and coordinate descent.
    """
    total_deadline = time.monotonic() + max(10.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(seed)

    # Get parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        if p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p
    except (NameError, TypeError, ValueError, AttributeError):
        parent = None

    # Candidate lengths
    if parent is not None:
        plen = parent.size
        lengths = sorted(set([plen, 300, 500, 800, 1000, 1200, 1500, 2000, 2500]))
    else:
        lengths = [300, 500, 800, 1000, 1200, 1500, 2000]
    lengths = [l for l in lengths if l <= 3000 and l >= 64]

    print(f"Trying lengths: {lengths}")
    if parent is not None:
        pn = parent / parent.sum()
        print(f"Parent: n={parent.size}, score={fast_score(pn):.6f}")

    best_seq = None
    best_score = np.inf

    # Phase 1: Quick initialization search
    print("\nPhase 1: Initialization search")
    for n in lengths:
        if time.monotonic() > total_deadline - 400:
            break
        inits = make_inits(n, rng)
        # Add parent (resized) if available
        if parent is not None:
            if parent.size == n:
                inits.append(parent / parent.sum())
            else:
                xo = np.linspace(0, 1, parent.size)
                xn = np.linspace(0, 1, n)
                pr = np.interp(xn, xo, parent)
                pr = np.maximum(pr, 0.0)
                if pr.sum() > 1e-10:
                    inits.append(pr / pr.sum())
        for init in inits:
            if time.monotonic() > total_deadline - 350:
                break
            s = fast_score(init)
            if s < best_score:
                best_score = s
                best_seq = init.copy()
    print(f"  Best init: n={best_seq.size}, score={best_score:.6f}")

    # Phase 2: LP refinement on the best sequence
    remaining = total_deadline - time.monotonic()
    lp_budget = remaining * 0.50
    print(f"\nPhase 2: LP refinement (budget={lp_budget:.0f}s)")
    seq_lp = lp_refine(best_seq, lp_budget)
    s_lp = fast_score(seq_lp)
    if s_lp < best_score:
        best_score = s_lp
        best_seq = seq_lp
        print(f"  LP result: score={s_lp:.6f}")
    else:
        print(f"  LP no improvement: {s_lp:.6f} vs {best_score:.6f}")

    # Phase 3: Coordinate descent
    remaining = total_deadline - time.monotonic()
    cd_budget = remaining * 0.55
    print(f"\nPhase 3: Coordinate descent (budget={cd_budget:.0f}s)")
    seq_cd = coord_shave(best_seq, cd_budget)
    s_cd = fast_score(seq_cd)
    if s_cd < best_score:
        best_score = s_cd
        best_seq = seq_cd
        print(f"  CD result: score={s_cd:.6f}")
    else:
        print(f"  CD no improvement: {s_cd:.6f} vs {best_score:.6f}")

    # Phase 4: Try a few other lengths with LP
    remaining = total_deadline - time.monotonic()
    if remaining > 80:
        print(f"\nPhase 4: Other lengths (remaining={remaining:.0f}s)")
        for n in [400, 600, 900, 1200, 1800]:
            if time.monotonic() > total_deadline - 40:
                break
            if n == best_seq.size:
                continue
            # Reshape best pattern
            xo = np.linspace(0, 1, best_seq.size)
            xn = np.linspace(0, 1, n)
            cand = np.interp(xn, xo, best_seq)
            cand = np.maximum(cand, 0.0)
            if cand.sum() < 1e-10:
                continue
            cand = cand / cand.sum()
            t_avail = min((total_deadline - time.monotonic() - 40) / 3.0, 100.0)
            if t_avail > 15:
                cand = lp_refine(cand, t_avail)
            s = fast_score(cand)
            if s < best_score:
                best_score = s
                best_seq = cand.copy()
                print(f"  n={n}: score={s:.6f} (improvement!)")

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"\n=== FINAL: n={best_seq.size}, fast={fast_score(best_seq):.6f}, eval={final_score:.6f} ===")

    return [float(x) for x in best_seq]
