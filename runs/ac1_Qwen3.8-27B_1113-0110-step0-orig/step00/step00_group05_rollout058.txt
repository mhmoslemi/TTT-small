Smooth objective:
```
def smooth_max_grad(a, tau_rel=0.05):
    b=np.convolve(a,a)
    M=b.max()
    if not finite or M<=0: return M, 2*np.ones_like(a)
    tau=max(1e-15, tau_rel*M)
    shifted=b-M
    expb=np.exp(shifted/tau)
    w=expb/expb.sum()
    g=2.0*np.correlate(w,a,mode='valid')
    S=M+tau*np.log(expb.sum())
    return S,g,b,M,w
```

"""
Search for low-score nonnegative step sequences.

The optimizer combines several ideas:
- Fast analytic / structured initial guesses, especially the central-binomial
  profile whose early convolution coefficients are exactly flat.
- A projected p-norm / smooth-max gradient line search in coefficient space.
- Periodic stochastic block perturbations to escape local minima.
- Optional small-scale linear-program updates inspired by the previous SOTA:
  given f and M = max(f*f), maximize sum(g) subject to g >= 0 and f*g <= M,
  then mix g back into f and accept only score-improving moves.
- A final polish pass.

Only one top-level function propose_candidate is required; all helpers are
top-level and use no closures or lambdas.
"""

import time
import numpy as np

try:
    from scipy.optimize import linprog
    HAVE_SCIPY_OPT = True
except Exception:
    HAVE_SCIPY_OPT = False


def normalize_sequence_array(a, target=1.0):
    """Make a 1D nonnegative finite array with sum exactly target."""
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0:
        return np.ones(1, dtype=np.float64) * target
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 1e-15:
        arr = np.ones_like(arr)
        s = float(arr.size)
    arr = arr * (float(target) / s)
    return arr


def convolution_full(a):
    """Full convolution a*a, using FFT for longer arrays."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    if n <= 200:
        return np.convolve(a, a)
    need = 2 * n - 1
    L = 1
    while L < need:
        L <<= 1
    fa = np.fft.rfft(a, L)
    out = np.fft.irfft(fa * fa, L)
    return out[:need]


def exact_score_array(a):
    """Fast objective: 2*n*max(conv(a,a))/(sum(a)^2)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return float("inf")
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-8:
        return float("inf")
    b = convolution_full(a)
    M = float(np.max(b))
    if not np.isfinite(M) or M < 0.0:
        return float("inf")
    return float(2.0 * n * M / (s * s))


def central_binom(n):
    """Coefficients of the first n terms of (1-z)^(-1/2)."""
    c = np.ones(n, dtype=np.float64)
    for k in range(1, n):
        c[k] = c[k - 1] * ((2 * k - 1) / (2.0 * k))
    return c


def generate_candidates(n, seed):
    """Generate a diversity of simple normalized nonnegative candidates."""
    xs = np.arange(n, dtype=np.float64)
    den = max(1, n - 1)
    left = 1.0 / np.sqrt(xs + 1.0)
    right = 1.0 / np.sqrt(n - xs)
    right = np.maximum(right, 1e-6)

    raw = []
    raw.append(central_binom(n))
    raw.append(left)
    raw.append(right)
    raw.append(left ** 0.5)
    raw.append(right ** 0.5)
    raw.append(left + 0.5 * right)
    raw.append(left + 0.1 * right)
    raw.append(left * (1.0 + 0.5 * xs / den))
    raw.append(left * (1.0 - 0.3 * xs / den))
    raw.append(np.exp(-xs / max(1.0, n / 5.0)))
    raw.append(np.exp(-(xs - 0.5 * n) ** 2 / max(1.0, n / 4.0)))
    raw.append(0.5 + 0.5 * np.cos(2.0 * np.pi * xs / den))
    raw.append(1.0 - xs / (den + 1.0))
    raw.append(1.0 / (xs + 1.0)**0.1 + 0.01)

    rng = np.random.default_rng(seed)
    raw.append(rng.random(n) + 0.01)
    raw.append(np.maximum(rng.normal(0.0, 1.0, n), 0.0) + 0.01)

    blk = max(2, n // 10)
    base = rng.random(n // blk + 1) + 0.01
    raw.append(np.repeat(base, blk)[:n])

    cands = []
    seen = set()
    for arr in raw:
        a = normalize_sequence_array(arr, 1.0)
        if a.size != n:
            continue
        if not np.all(np.isfinite(a)):
            continue
        key = bytes(a.tobytes())
        if key in seen:
            continue
        seen.add(key)
        cands.append(a)
    return cands


def make_pnorm_gradient(a, p):
    """
    Directional gradient of the p-norm surrogate for max(conv(a,a)).

    Only the direction matters because line search rescales the step.
    """
    n = a.size
    b = convolution_full(a)
    bm = float(np.max(b))
    if not np.isfinite(bm) or bm <= 0.0:
        g = np.zeros(n, dtype=np.float64)
        g[np.argmax(a)] = 1.0
        return g

    scaled = np.maximum(b / bm, 0.0)
    try:
        bp = np.power(scaled, p - 1.0)
    except Exception:
        idx = int(np.argmax(b))
        w = np.zeros_like(b, dtype=np.float64)
        w[idx] = 1.0
        return 2.0 * np.correlate(w, a, mode="valid")

    s = float(bp.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros_like(b, dtype=np.float64)
        idx = int(np.argmax(b))
        w[idx] = 1.0
    else:
        w = bp / s

    g = 2.0 * np.correlate(w, a, mode="valid")
    if not np.all(np.isfinite(g)) or float(np.max(np.abs(g))) <= 0.0:
        idx = int(np.argmax(a))
        g = np.zeros(n, dtype=np.float64)
        g[idx] = 1.0
    return g


def line_search_step(a, g, n_factors=5):
    """Try several scaled projected negative-gradient steps."""
    cur_score = exact_score_array(a)
    best_a = a.copy()
    best_score = cur_score

    ng = float(np.max(np.abs(g)))
    if not np.isfinite(ng) or ng <= 1e-300:
        return best_a, best_score

    mean_a = float(np.mean(a))
    base = 0.25 * max(mean_a, 1e-12) / ng
    factors = (4.0, 1.0, 0.5, 0.25, 0.125, 0.0625)
    factors = factors[:max(1, int(n_factors))]

    for f in factors:
        cand = a - base * f * g
        cand = np.maximum(cand, 0.0)
        s = float(cand.sum())
        if not np.isfinite(s) or s < 1e-12:
            continue
        cand = cand / s
        sc = exact_score_array(cand)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_a = cand
    return best_a, best_score


def stochastic_block_perturb(a, rng, use_mirror_prob=0.3):
    """One random block-cauchy perturbation, returned normalized."""
    n = a.size
    cand = a.copy()
    if n <= 1:
        return cand
    lo = max(2, n // 200)
    hi = max(lo + 1, n // 5)
    width = int(rng.integers(lo, hi + 1))
    width = min(width, n)
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    factor = float(np.exp(rng.normal(0.0, 0.30)))
    cand[start:stop] *= factor
    if rng.random() < use_mirror_prob:
        cand = 0.5 * (cand + cand[::-1])
    cand = normalize_sequence_array(cand, 1.0)
    return cand


def gradient_search(a_init, budget, seed=0):
    """
    Local p-norm gradient line search with stochastic restarts.
    """
    a0 = normalize_sequence_array(a_init, 1.0)
    n = a0.size
    if n == 0:
        return a0, float("inf")

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.05, float(budget))

    best_a = a0.copy()
    best_score = exact_score_array(best_a)
    cur = best_a.copy()
    cur_score = best_score

    if best_score > 1.85:
        p_min, p_max = 4.0, 10.0
    elif best_score > 1.58:
        p_min, p_max = 6.0, 24.0
    else:
        p_min, p_max = 10.0, 48.0

    n_factors = 3 if n > 180 else 5
    stall_limit = 50 if n > 180 else 140
    no_improve = 0
    it = 0
    max_iters = 200000

    while it < max_iters:
        if time.monotonic() >= deadline:
            break
        it += 1
        prog = min(1.0, float(it) / 2500.0)
        p = p_min * ((p_max / p_min) ** prog)

        g = make_pnorm_gradient(cur, p)
        new_a, new_score = line_search_step(cur, g, n_factors)

        if np.isfinite(new_score) and new_score < cur_score - 1e-12:
            cur = new_a
            cur_score = new_score
            no_improve = 0
        else:
            no_improve += n_factors

        if np.isfinite(cur_score) and cur_score < best_score:
            best_score = cur_score
            best_a = cur.copy()
            if best_score < 1.02:
                break

        if no_improve > stall_limit:
            no_improve = 0
            r = rng.random()
            if r < 0.4:
                pert = stochastic_block_perturb(best_a, rng)
            elif r < 0.7:
                pert = stochastic_block_perturb(cur, rng)
            else:
                pert = best_a + rng.normal(0.0, 0.05 * float(np.mean(best_a)), n)
                pert = normalize_sequence_array(pert, 1.0)

            ps = exact_score_array(pert)
            if np.isfinite(ps) and ps < best_score:
                best_a = pert.copy()
                best_score = ps
                cur = best_a.copy()
                cur_score = best_score
            else:
                cur = pert
                cur_score = ps

    final_sc = exact_score_array(best_a)
    if np.isfinite(final_sc) and final_sc < best_score:
        best_score = final_sc
    return best_a, best_score


def block_polish(a_init, budget, seed=0):
    """Simple stochastic block multiplicative hill-climbing polish."""
    a0 = normalize_sequence_array(a_init, 1.0)
    best_a = a0.copy()
    best_score = exact_score_array(best_a)
    n = best_a.size
    if n == 0:
        return best_a, float("inf")

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.05, float(budget))
    cur = best_a.copy()
    cur_score = best_score
    stall = 0
    stall_limit = 120
    max_iters = 200000

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        cand = stochastic_block_perturb(cur if rng.random() < 0.7 else best_a, rng)
        sc = exact_score_array(cand)
        if np.isfinite(sc) and sc < cur_score - 1e-12:
            cur = cand
            cur_score = sc
            stall = 0
        else:
            stall += 1
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = cand.copy()
            if best_score < 1.02:
                break
        if stall > stall_limit:
            stall = 0
            if rng.random() < 0.5:
                cur = stochastic_block_perturb(best_a, rng, 0.5)
            else:
                cur = best_a.copy()

    final_sc = exact_score_array(best_a)
    if np.isfinite(final_sc) and final_sc < best_score:
        best_score = final_sc
    return best_a, best_score


def lp_build_convolution_matrix(a):
    """Dense matrix A such that A@g = convolve(a, g)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        sl = max(0, k - n + 1)
        hi = min(n - 1, k)
        if sl <= hi:
            vals = a[k - hi:k - sl + 1][::-1]
            A[k, sl:hi + 1] = vals
    return A


def lp_solve_direction(a, M, time_limit_s=5.0):
    """
    Solve max sum(g) s.t. g >= 0 and convolve(a, g) <= M.
    Returns None on failure or if too large.
    """
    if not HAVE_SCIPY_OPT:
        return None
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n < 4 or n > 220:
        return None
    if not np.isfinite(M) or M <= 0.0:
        return None

    A = lp_build_convolution_matrix(a)
    A = A / M
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.ones(2 * n - 1, dtype=np.float64)
    bounds = [(0.0, None)] * n

    tl = float(max(0.5, min(20.0, time_limit_s)))
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"time_limit": tl, "presolve": True},
        )
    except Exception:
        try:
            res = linprog(
                c,
                A_ub=A,
                b_ub=b_ub,
                bounds=bounds,
                method="highs",
            )
        except Exception:
            return None

    if res is None or getattr(res, "status", 1) != 0:
        x = getattr(res, "x", None) if res is not None else None
        if x is None:
            return None
    else:
        x = getattr(res, "x", None)
        if x is None:
            return None

    g = np.asarray(x, dtype=np.float64).ravel()
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    if float(g.sum()) <= 0.0:
        return None
    return g


def lp_update(a_init, budget, seed=0):
    """
    Iterate the SOTA linear-programming direction update.
    """
    if not HAVE_SCIPY_OPT:
        return normalize_sequence_array(a_init, 1.0), exact_score_array(
            normalize_sequence_array(a_init, 1.0))

    a0 = normalize_sequence_array(a_init, 1.0)
    n = a0.size
    if n < 6 or n > 220:
        return a0, exact_score_array(a0)

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.05, float(budget))

    best_a = a0.copy()
    best_score = exact_score_array(best_a)
    a = best_a.copy()
    last_improve = time.monotonic()
    max_rounds = 12

    for _ in range(max_rounds):
        if time.monotonic() >= deadline:
            break
        b = convolution_full(a)
        M = float(np.max(b)) if b.size else 0.0
        if not np.isfinite(M) or M <= 0.0:
            break

        relaxes = (1.0, 1.03)
        improved_this_round = False

        for relax in relaxes:
            if time.monotonic() >= deadline:
                break
            Mrel = M * relax
            rem = max(0.5, deadline - time.monotonic())
            g = lp_solve_direction(a, Mrel, time_limit_s=min(12.0, rem * 0.5))
            if g is None:
                continue
            g = normalize_sequence_array(g, 1.0)

            for t in (0.1, 0.25, 0.5, 0.75, 1.0):
                h = (1.0 - t) * a + t * g
                h = normalize_sequence_array(h, 1.0)
                sc = exact_score_array(h)
                if np.isfinite(sc) and sc < best_score:
                    improved_this_round = True
                    best_a = h
                    best_score = sc
                    a = h
                    if t < 1.0 and rng.random() < 0.1:
                        a = stochastic_block_perturb(h, rng)
                        a = normalize_sequence_array(a, 1.0)

        if improved_this_round:
            last_improve = time.monotonic()

        if not improved_this_round:
            if time.monotonic() - last_improve > max(2.0, budget * 0.5):
                break
            g = lp_solve_direction(a, M * 1.05, time_limit_s=min(8.0, max(1.0, deadline - time.monotonic())))
            if g is None:
                break
            g = normalize_sequence_array(g, 1.0)
            sc = exact_score_array(g)
            if np.isfinite(sc) and sc < best_score:
                best_a = g
                best_score = sc
                a = g
            else:
                break

    return best_a, best_score


def try_add_trial(a, score, best_a, best_score, best_for_n, n):
    """Update best variables using tuple return values."""
    if a is None or score is None:
        return best_a, best_score, best_for_n
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0 or not np.all(np.isfinite(a)):
        return best_a, best_score, best_for_n
    if not np.isfinite(score):
        sc = exact_score_array(a)
    else:
        sc = float(score)
    if not np.isfinite(sc):
        return best_a, best_score, best_for_n

    a = normalize_sequence_array(a, 1.0)
    sc = exact_score_array(a)
    if not np.isfinite(sc):
        return best_a, best_score, best_for_n

    if n in best_for_n:
        old_a, old_s = best_for_n[n]
        if sc < old_s:
            best_for_n[n] = (a.copy(), sc)
    else:
        best_for_n[n] = (a.copy(), sc)

    if sc < best_score:
        best_a = a.copy()
        best_score = sc
    return best_a, best_score, best_for_n


def propose_candidate(seed=42, budget_s=1800, **kwargs):

    Run the multi-start optimizer and return the best nonnegative sequence.
    """

    # Seed from any provided prior construction.
    try:
        hs = normalize_sequence_array(height_sequence_1, 1.0)
        if hs.size >= 1:
            hs_score = exact_score_array(hs)
            best_a, best_score, _ = try_add_trial(hs, hs_score, best_a, best_score, best_for_n, hs.size)
    except Exception:
        pass

    quick_ns = [
        6, 7, 8, 9, 10, 12, 14, 16, 18, 20, 24, 28, 32, 36, 40,
        45, 50, 56, 64, 72, 80, 90, 100, 120, 150

    # Phase 1: quick structured scan.
    for n in quick_ns:
        if time.monotonic() >= deadline - 60.0:
            break
        cands = generate_candidates(n, seed + 1000 * n)
        # Evaluate at most 12 candidates per n.

            if time.monotonic() >= deadline - 60.0:
                break
            sc = exact_score_array(a)
            best_a, best_score, best_for_n = try_add_trial(a, sc, best_a, best_score, best_for_n, n)

    print(f"quick scan best={best_score:.9f} n={best_a.size} t={time.monotonic()-start_t:.1f}s", flush=True)

    # Prioritize n values by current per-n scores.
    prio = [(s, n, a) for (n, (a, s)) in best_for_n.items()]

    max_gradient_rounds = 6
    round = 0
    while round < max_gradient_rounds:

        did_work = False
        for score_val, n_val, a0 in prio:

            if time.monotonic() < deadline - 30.0 and best_score < 1.52:
                a2, s2 = gradient_search(a1, max(2.0, alloc * 0.5), seed=seed + 57 * n_val + round + 1)
                best_a, best_score, best_for_n = try_add_trial(a2, s2, best_a, best_score, best_for_n, n_val)

            did_work = True
            print(
                f"grad n={n_val} score={best_score:.9f} t={time.monotonic()-start_t:.1f}s",
                flush=True
            )

        round += 1
        if not did_work:
            break

    # Refresh priority with new per-n bests.
    prio2 = [(s, n, a) for (n, (a, s)) in best_for_n.items()]
    prio2.sort()

    # Phase 3: LP-inspired updates for smaller/moderate n.
    if HAVE_SCIPY_OPT:
        lp_ns = [8, 10, 12, 14, 16, 18, 20, 24, 28, 32, 36, 40, 48, 56, 64]
        for n_val in lp_ns:
            if time.monotonic() >= deadline - 45.0:
                break
            if n_val not in best_for_n:
                cands = generate_candidates(n_val, seed + 999 * n_val)
                if not cands:
                    continue
                best_a, best_score, best_for_n = try_add_trial(
                    cands[0], None, best_a, best_score, best_for_n, n_val

            )

    # Phase 4: final polish on the global best and a few top per-n candidates.
    remaining = deadline - time.monotonic()
    if remaining > 8.0:
        top = prio2[:4]
        polished_best = (None, None)
        for score_val, n_val, a0 in [(best_score, best_a.size, best_a)] + top:
            if time.monotonic() >= deadline - 6.0:
                break
            a1, s1 = block_polish(a0, min(8.0, deadline - time.monotonic() - 6.0),
                                  seed=seed + 10000 + n_val)
            best_a, best_score, best_for_n = try_add_trial(a1, s1, best_a, best_score, best_for_n, n_val)

        remaining = deadline - time.monotonic()
        if remaining > 8.0 and best_a.size <= 350:
            a1, s1 = gradient_search(best_a, max(2.0, remaining * 0.55), seed=seed + 424242)
            best_a, best_score, _ = try_add_trial(a1, s1, best_a, best_score, best_for_n, best_a.size)

        remaining = deadline - time.monotonic()
        if remaining > 5.0:
            a1, s1 = block_polish(best_a, remaining * 0.8, seed=seed + 535353)
            best_a, best_score, _ = try_add_trial(a1, s1, best_a, best_score, best_for_n, best_a.size)

    # Final safety checks.
    best_a = normalize_sequence_array(best_a, 1.0)
    fl = [float(max(0.0, min(1000.0, x))) for x in best_a]
    last_score = exact_score_array(np.asarray(fl, dtype=np.float64))
    print(f"final internal score={last_score:.9f} n={len(fl)}", flush=True)

    try:
        ev = evaluate_sequence(fl)
        if not np.isfinite(ev):
            fallback = [1.0] * 64
            return fallback
        # The evaluator is authoritative for the exact returned list.
        print(f"final evaluate_sequence={float(ev):.9f}", flush=True)
    except Exception:
        pass

    return fl
