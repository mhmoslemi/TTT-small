def fft_convolve(a):
    """Compute linear autocorrelation a * a via FFT."""
    n = a.size
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def fast_score(a):
    """Compute score = 2*n*max(c)/S^2 where c = convolve(a,a), S = sum(a)."""
    n = a.size
    if n == 0:
        return np.inf
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = fft_convolve(a)
    return 2.0 * n * c.max() / (S * S)


def build_lp_pattern(n):
    """Precompute sparsity pattern for LP constraint matrix."""
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            rows_list.append(k)
            cols_list.append(j)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_lp_step(f, M, rows, cols, n):
    """Solve the LP: max sum(b) s.t. b>=0, (f*b)[k] <= M for all k."""
    # Entries of A_ub: A_ub[k,j] = f[k-j]
    k_idx = rows  # these are the k values
    j_idx = cols   # these are the j values
    vals = f[k_idx - j_idx]
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def line_search(a, b, max_evals=20):
    """Find t in [0,1] minimizing fast_score((1-t)*a + t*b)."""
    diff = b - a
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    f_c = fast_score(a + c_pt * diff)
    f_d = fast_score(a + d_pt * diff)
    evals = 2
    for _ in range(max_evals - 2):
        if f_c <= f_d:
            hi = d_pt
            d_pt = c_pt
            f_d = f_c
            c_pt = hi - gr * (hi - lo)
            f_c = fast_score(a + c_pt * diff)
        else:
            lo = c_pt
            c_pt = d_pt
            f_c = f_d
            d_pt = lo + gr * (hi - lo)
            f_d = fast_score(a + d_pt * diff)
        evals += 1
    t_best = (lo + hi) / 2.0
    return t_best

3. Pick the change that gives the most improvement.

This is O(n) per step (after the O(n log n) convolution).

Actually, this is getting complex. Let me just use a simpler approach: for the element with the largest contribution to the peak, decrease it and increase a random other element. This is O(n) per step.

"""
Minimizes score = 2*n*max(convolve(a,a))/sum(a)^2 for non-negative a.
Strategy: LP-based direction finding + line search + targeted coordinate descent.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fft_autocorr(a):
    """Linear autocorrelation via FFT (full mode)."""
    n = a.size
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    ac = np.fft.irfft(fa * fa, size)
    return ac[:2 * n - 1]


def fast_score(a):
    """Compute score without calling evaluate_sequence."""
    n = a.size
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_autocorr(a)
    return 2.0 * n * float(c.max()) / (S * S)


def build_lp_indices(n):
    """Precompute row/col indices for the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int64), np.array(cols, dtype=np.int64)


def solve_lp(a, M, rows, cols, n):
    """Solve: max sum(b) s.t. b >= 0, (a*b)[k] <= M for all k."""
    vals = a[rows - cols]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None and res.x.sum() > 1e-15:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def line_search_step(a, b):
    """Find best t in [0,1]: minimize fast_score(a + t*(b-a))."""
    diff = b - a
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = fast_score(a + c_pt * diff)
    fd = fast_score(a + d_pt * diff)
    for _ in range(12):
        if fc <= fd:
            hi, d_pt, fd = d_pt, c_pt, fc
            c_pt = hi - gr * (hi - lo)
            fc = fast_score(a + c_pt * diff)
        else:
            lo, c_pt, fc = c_pt, d_pt, fd
            d_pt = lo + gr * (hi - lo)
            fd = fast_score(a + d_pt * diff)
    return (lo + hi) / 2.0


def run_lp_iters(a, budget_s):
    """Iterative LP-based improvement."""
    n = a.size
    a = a.copy()
    S = a.sum()
    if S < 1e-15:
        return a
    a = a / S
    best = a.copy()
    best_score = fast_score(a)
    deadline = time.monotonic() + budget_s
    rows, cols = build_lp_indices(n)
    it = 0
    stale = 0
    while time.monotonic() < deadline:
        c = fft_autocorr(a)
        M = float(c.max())
        b = solve_lp(a, M, rows, cols, n)
        if b is None:
            stale += 1
            if stale > 5:
                break
            b = solve_lp(a, M * 2.0, rows, cols, n)
            if b is None:
                break
        Sb = b.sum()
        if Sb < 1e-15:
            break
        b = b / Sb  # normalize to sum=1
        if np.allclose(b, a, atol=1e-10):
            stale += 1
            if stale > 5:
                break
            # Perturb a slightly
            a += 0.01 * np.random.randn(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            continue
        t = line_search_step(a, b)
        a_new = a + t * (b - a)
        a_new = np.maximum(a_new, 0.0)
        s_new = fast_score(a_new)
        if s_new < best_score - 1e-9:
            best = a_new.copy()
            best_score = s_new
            stale = 0
        else:
            stale += 1
        a = a_new
        it += 1
    return best


def coord_shave(a, budget_s):
    """Targeted coordinate descent: reduce the peak by adjusting contributing elements."""
    n = a.size
    a = a.copy()
    S = a.sum()
    if S < 1e-15:
        return a
    a = a / S
    best = a.copy()
    best_score = fast_score(a)
    deadline = time.monotonic() + budget_s
    rng = np.random.default_rng(99)
    it = 0
    while time.monotonic() < deadline:
        c = fft_autocorr(a)
        k_star = int(c.argmax())
        # Find contributing elements
        j_lo = max(0, k_star - n + 1)
        j_hi = min(k_star, n - 1)
        # Gradient of c[k_star] w.r.t. a[j]:
        # For each j in [j_lo, j_hi], the contribution is a[j] * a[k_star - j]
        # If j != k_star - j, gradient is 2*a[k_star - j]
        # If j == k_star - j (k_star even, j = k_star/2), gradient is a[j]
        jmid = (k_star + 0)  # for the pair
        # Find the j with largest gradient (most impactful)
        if k_star % 2 == 0 and k_star // 2 >= j_lo and k_star // 2 <= j_hi:
            mid = k_star // 2
            grad = 2.0 * a[k_star - np.arange(j_lo, j_hi + 1)]
            grad[mid - j_lo] = a[mid]  # self-term
        else:
            grad = 2.0 * a[k_star - np.arange(j_lo, j_hi + 1)]
        
        if grad.max() <= 0:
            break
        
        # Decrease the most impactful element
        idx_max = j_lo + int(np.argmax(grad))
        # Increase a random element that doesn't contribute (or contributes less)
        candidates = np.arange(n)
        # Prefer elements with zero gradient
        zero_grad = np.where((candidates < j_lo) | (candidates > j_hi))[0]
        if zero_grad.size > 0:
            idx_inc = zero_grad[rng.integers(zero_grad.size)]
        else:
            # Pick from low-gradient elements
            all_grad = np.zeros(n)
            all_grad[j_lo:j_hi+1] = grad
            idx_inc = rng.choice(np.where(all_grad == all_grad.min())[0])
        
        # Try several step sizes
        improved = False
        for frac in [0.05, 0.1, 0.2, 0.02, 0.5]:
            delta = a[idx_max] * frac
            if delta < 1e-15:
                continue
            a_new = a.copy()
            a_new[idx_max] -= delta
            a_new[idx_inc] += delta
            a_new = np.maximum(a_new, 0)
            if a_new.sum() < 1e-10:
                continue
            a_new = a_new / a_new.sum()
            s = fast_score(a_new)
            if s < best_score - 1e-10:
                best = a_new.copy()
                best_score = s
                a = a_new
                improved = True
                break
        
        if not improved:
            # Try random perturbation
            i = rng.integers(n)
            a_new = a.copy()
            a_new[i] *= (1.0 + rng.normal(0, 0.1))
            a_new = np.maximum(a_new, 0)
            if a_new.sum() > 1e-10:
                a_new = a_new / a_new.sum()
                s = fast_score(a_new)
                if s < best_score - 1e-10:
                    best = a_new.copy()
                    best_score = s
                    a = a_new
        
        it += 1
    
    return best
