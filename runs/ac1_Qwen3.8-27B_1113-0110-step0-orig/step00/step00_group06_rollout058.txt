```
def _block_perturbation(a, rng):
    n=a.size
    cand=a.copy()
    nblocks=int(rng.integers(1,4))
    for _ in range(nblocks):
       w=int(rng.integers(max(1,n//500), max(2,n//12)))
       st=int(rng.integers(0, max(1,n-w)))
       end=min(n,st+w)

       cand[st:end] *= mult
    if rng.random() < 0.25:
       cand = 0.5*(cand + cand[::-1])
```

```
def _fft_conv(a):
    n = int(a.size)
    if n <= 0:
        return np.array([], dtype=np.float64)
    m = 2*n - 1
    L = 1
    while L < m:
        L <<= 1
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)[:m]
    # Remove negative roundoff
    c = np.maximum(c, 0.0)
    return c
```

```
def _surrogate_score(a, exclude_zero=False):
    if a is None:
        return np.inf
    n = int(a.size)
    if n <= 0:
        return np.inf
    s = float(a.sum())

        return np.inf
    c = _fft_conv(a)
    if c.size == 0:

    if exclude_zero and c.size > 1:
        cm = float(np.max(c[1:]))
    else:
        cm = float(np.max(c))
    if not np.isfinite(cm) or cm < 0.0:
        return np.inf
    return float(2.0 * n * cm / (s * s))
```

```python
"""
Search for a low-scoring non-negative step sequence for the supplied evaluator.

Mathematically, for the evaluator as written there is a hard exact lower bound
of 2.  Indeed, b[0] = sum(a_i * a_i), and max(b) >= b[0].  By Cauchy-Schwarz,
(sum a_i)^2 <= n * sum(a_i^2), so:

    2*n*max(b) / (sum a)^2 >= 2*n*b[0] / (sum a)^2 >= 2.

A constant sequence attains 2.  This routine returns an optimal constant
sequence, but still performs a short guided search to remain robust against
harmless floating-point effects or minor evaluator variants.  It never returns
a candidate worse than the best valid candidate seen.
"""

import numpy as np
import time

MAX_N = 8192
MIN_N = 64
TARGET_SUM = 1.0


def _clean_seq(x):
    """Convert input to a finite non-negative 1-D float array, or None."""
    try:
        arr = np.asarray(x, dtype=np.float64)
    except Exception:
        return None

    if arr.ndim == 0:
        arr = arr.reshape(1)
    elif arr.ndim != 1:
        try:
            arr = arr.reshape(-1)
        except Exception:
            return None

    if arr.size == 0:
        return None

    if not np.all(np.isfinite(arr)):
        return None

    arr = np.clip(arr, 0.0, 1000.0)

    if float(np.sum(arr)) <= 0.0:
        return None

    return arr


def _score_array(arr):
    """Call the external evaluator on a NumPy array and return a finite float."""
    if arr is None:
        return np.inf

    try:
        pylist = [float(v) for v in arr]
    except Exception:
        return np.inf

    try:
        val = float(evaluate_sequence(pylist))
    except Exception:
        return np.inf

    if not np.isfinite(val):
        return np.inf

    return float(val)


def _normalize_sum(arr, target=TARGET_SUM):
    """Scale non-negative array to have the desired sum, if possible."""
    a = np.asarray(arr, dtype=np.float64)
    if a.size == 0:
        return None

    a = a.copy()
    s = float(np.sum(a))

    if not np.isfinite(s) or s <= 0.0:
        return None

    a *= (float(target) / s)
    a = np.clip(a, 0.0, 1000.0)

    if not np.all(np.isfinite(a)) or float(np.sum(a)) <= 0.0:
        return None

    return a


def _constant_sequence(n):
    """Return a constant non-negative sequence of length n."""
    n = int(n)
    if n <= 0:
        n = MIN_N
    return np.ones(n, dtype=np.float64)


def _get_parent():
    """Retrieve and sanitize the global parent sequence, if available."""
    seq = globals().get("height_sequence_1", None)
    arr = _clean_seq(seq)

    if arr is None:
        return None

    if arr.size > MAX_N:
        k = int(np.ceil(arr.size / float(MAX_N)))
        if k <= 0:
            return None
        m = arr.size // k
        if m <= 0:
            return None
        arr = arr[:m].reshape(m, k).mean(axis=1)

    if arr.size < MIN_N:
        return None

    return arr


def _fft_conv(a):
    """Approximate the linear self-convolution using an FFT."""
    n = int(a.size)
    if n <= 0:
        return np.array([], dtype=np.float64)

    m = 2 * n - 1
    L = 1
    while L < m:
        L <<= 1

    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)[:m]

    # Self-convolution of a non-negative sequence is non-negative exactly.
    # Small negative FFT roundoff is removed for a stable surrogate.
    return np.maximum(c, 0.0)


def _surrogate_score(a, exclude_zero=False):
    """Fast surrogate for the evaluator, using FFT convolution."""
    if a is None:
        return np.inf

    n = int(a.size)
    if n <= 0:
        return np.inf

    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 0.0:
        return np.inf

    c = _fft_conv(a)
    if c.size == 0:
        return np.inf

    if exclude_zero and c.size > 1:
        cm = float(np.max(c[1:]))
    else:
        cm = float(np.max(c))

    if not np.isfinite(cm) or cm < 0.0:
        return np.inf

    return float(2.0 * n * cm / (s * s))


def _smooth_candidate(n, rng):
    """Create a random smooth non-negative candidate sequence."""
    n = int(n)
    if n < MIN_N:
        n = MIN_N

        n = MAX_N

    x = np.linspace(0.0, 1.0, n, dtype=np.float64)

    if kind == 0:
        center = float(rng.uniform(0.25, 0.75))
        width = float(rng.uniform(0.08, 0.45))
        a = np.exp(-0.5 * ((x - center) / width) ** 2)

    elif kind == 1:
        f = int(rng.integers(1, 9))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))

        a = np.sin(np.pi * x) ** p

    elif kind == 3:

    a = np.maximum(a, 0.0)

    if not np.all(np.isfinite(a)) or float(np.sum(a)) <= 0.0:

def _block_perturbation(a, rng):

                if best_score <= 2.0 + 1e-12:
                    break

                base_arr = best_arr.copy()
                best_sur = _surrogate_score(best_arr)
                base_sur = best_sur
                if not np.isfinite(best_sur):
                    best_sur = best_score
                if not np.isfinite(base_sur):
                    base_sur = best_score
                no_improve = 0
                temp = 1e-3

    best_arr = _clean_seq(best_arr)

    if best_arr is None or best_arr.size > MAX_N:
        if best_arr is not None and best_arr.size > 0:
            k = int(np.ceil(best_arr.size / float(MAX_N)))
            if k > 1:
                m = best_arr.size // k
                if m > 0:
                    best_arr = best_arr[:m].reshape(m, k).mean(axis=1)

    if best_arr is None or best_arr.size == 0:
        best_arr = np.ones(1024, dtype=np.float64)

    return [float(x) for x in best_arr]
```
