def fast_score(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2 using FFT."""
    n = a.size

    b = np.fft.irfft(fa * fa, size)
    max_b = b[:2*n-1].max()
    return float(2.0 * n * max_b / (s * s))


def fast_conv(a):

    a = np.ones(n, dtype=np.float64)

            candidate += noise

        candidate = np.clip(candidate, 0.0, 1000.0)
        s = candidate.sum()
        if s < 0.01:
            continue

        # Evaluate candidate
        cand_b = fast_conv(candidate)
        cand_max = float(cand_b.max())
        cand_score = float(2.0 * n * cand_max / (s * s))

        # Accept/reject with annealing
        delta = cand_score - best_score
        if delta < 0 or (anneal_temp > 0 and rng.random() < math.exp(-abs(delta) / anneal_temp)):
            current = candidate
            current_b = cand_b
            current_max = cand_max
            current_kstar = int(np.argmax(cand_b))
            if delta < 0:
                best_score = cand_score
                best_a = candidate.copy()
                no_improve = 0
                anneal_temp = max(0.0001, anneal_temp * 0.999)
                if int(eval_count) % 200 == 0:
                    print(f"  iter {eval_count}: score={best_score:.8f}, k*={current_kstar}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Cool down
        if no_improve > 500:
            anneal_temp = max(0.0001, anneal_temp * 0.95)
            no_improve = 0

        eval_count += 1

        # Periodic verification with official evaluator
        if eval_count % 500 == 0:
            v = evaluate_sequence(best_a.tolist())
            if np.isfinite(v):
                if v < best_score:
                    best_score = v
                # Re-sync
                s_check = best_a.sum()
                if s_check > 0.01:
                    best_score_check = float(2.0 * n * fast_conv(best_a).max() / (s_check * s_check))
            if time.monotonic() > deadline:
                break

    print(f"Phase 2 complete: best_score={best_score:.8f}, evals={eval_count}")

    # ===== Phase 3: Fine-grained polish =====
    # Small coordinate descent on a subset of elements
    step = 0.1
    for _ in range(200):
        if time.monotonic() > deadline:
            break
        # Pick a random subset of ~50 elements and try small adjustments
        idx = rng.choice(n, size=min(50, n), replace=False)
        cand = best_a.copy()
        for i in idx:
            delta = rng.uniform(-step, step)
            cand[i] = max(0.0, cand[i] + delta)
        cand = np.clip(cand, 0, 1000)
        s = cand.sum()
        if s < 0.01:
            continue
        b = fast_conv(cand)
        sc = float(2.0 * n * b.max() / (s * s))
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()

    # Final verification
    final_score = evaluate_sequence(best_a.tolist())
    if np.isfinite(final_score):
        print(f"Final verified score: {final_score:.8f}")
    else:
        print(f"Final verification failed, using FFT score: {best_score:.8f}")
        final_score = best_score

    return [float(x) for x in best_a]
