    # Compute convolution using FFT
    def convolve_fft(a):
        n = len(a)
        size = 2*n - 1
        # Use next power of 2 for FFT efficiency
        fft_size = 1
        while fft_size < size:
            fft_size *= 2
        A = np.fft.rfft(a, fft_size)
        C = A * A
        c = np.fft.irfft(C, fft_size)[:size]
        return c
    
    # Project onto simplex {x >= 0, sum(x) = 1}
    def project_simplex(v):
        n = len(v)
        if v.sum() <= 0:
            return np.ones(n) / n
        # Sort in descending order
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u)
        rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
        theta = (cssv[rho] - 1) / (rho + 1.0)
        return np.maximum(v - theta, 0)
    
    # Subgradient descent
    eta0 = 0.1  # Initial step size
    for it in range(100000):
        if time.monotonic() > deadline:
            break

        k_star = np.argmax(c)
