import numpy as np
import time
import math

"""
Minimize J(a) = 2n * max(convolve(a,a)) / (sum(a))^2 using:
1. Multiple structured initializations (two-level, geometric, etc.)
2. Multiplicative subgradient descent with FFT-based autocorrelation
3. Adaptive learning rate and periodic perturbations
4. Coordinate descent refinement on the best candidate
"""


def _fft_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()  # next power of 2 >= 2n-1
    fa = np.fft.rfft(a, size=size)
    ac = np.fft.irfft(fa * fa, size=size)[:2 * n - 1]
    return ac


def _fast_score(a):
    """Fast internal score computation."""
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    ac = _fft_autocorr(a)
    M = float(np.max(ac))
    return float(2 * n * M / (S * S))


def _compute_pressure(a, peak_indices):
    """Compute pressure vector from peak indices of autocorrelation."""
    n = len(a)
    pressure = np.zeros(n)
    for k in peak_indices:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            vals = a[k - idx]
            pressure[lo:hi + 1] += vals
    pressure /= max(1, len(peak_indices))
    return pressure


def _make_two_level(n, frac, level):
    """Create a two-level sequence: first frac*n entries = 1, rest = level."""
    m = max(1, int(n * frac))
    a = np.ones(n)
    a[m:] = level
    return a


def _make_geometric(n, r):
    """Create a geometric sequence."""
    a = r ** np.arange(n, dtype=np.float64)
    return a


def _make_diamond(n, alpha):
    """Create a 'diamond' shape: rises then falls, front-loaded."""
    t = np.arange(n, dtype=np.float64)
    center = n * 0.4  # front-loaded center
    a = np.exp(-alpha * (t - center) ** 2 / n)
    return a


def _optimize_sequence(a, deadline, rng, max_iters=50000):
    """
    Optimize a sequence using multiplicative subgradient descent.
    Returns the best sequence and score found.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.clip(a, 1e-10, 1000.0)
    
    S0 = float(np.sum(a))
    a = a * (math.sqrt(2.0 * n) / S0)  # normalize to sum = sqrt(2n)
    
    best = a.copy()
    best_score = _fast_score(a)
    
    lr = 0.15
    patience = 0
    max_patience = 300
    iter_count = 0
    
    while time.monotonic() < deadline and iter_count < max_iters:
        ac = _fft_autocorr(a)
        M = float(np.max(ac))
        S = float(np.sum(a))
        score = 2.0 * n * M / (S * S)
        
        if score < best_score:
            best = a.copy()
            best_score = score
            patience = 0
        else:
            patience += 1
        
        if patience > max_patience:
            # Restart with perturbation
            noise_scale = 0.3 * patience / max_patience
            a = best * np.exp(rng.normal(0, noise_scale, size=n))
            a = np.clip(a, 1e-10, 1000.0)
            patience = 0
            lr = 0.15
            continue
        
        # Find peaks (within 1% of max)
        threshold = M * 0.995
        peak_mask = ac >= threshold
        peak_indices = np.where(peak_mask)[0]
        if len(peak_indices) == 0:
            peak_indices = np.array([int(np.argmax(ac))])
        # Cap number of peaks for efficiency
        if len(peak_indices) > 10:
            # Keep the largest 10
            peak_vals = ac[peak_indices]
            top_idx = np.argsort(peak_vals)[-10:]
            peak_indices = peak_indices[top_idx]
        
        # Compute pressure
        pressure = _compute_pressure(a, peak_indices)
        
        # Subgradient: dJ/da[i] proportional to (pressure[i] * S - M) / S^2
        # We want to decrease a[i] where pressure[i] * S > M (contributes to peak)
        # and increase where pressure[i] * S < M
        ratio = pressure * S / (M + 1e-15)
        log_update = -lr * (ratio - 1.0)
        log_update = np.clip(log_update, -2.0, 2.0)
        
        a = a * np.exp(log_update)
        a = np.clip(a, 1e-10, 1000.0)
        
        # Adaptive learning rate
        new_score = _fast_score(a)
        if new_score < score:
            lr = min(lr * 1.02, 0.4)
        else:
            lr = max(lr * 0.85, 0.01)
        
        iter_count += 1
        if iter_count % 500 == 0:
            print(f"  iter {iter_count}, score={best_score:.6f}, lr={lr:.4f}, patience={patience}")
    
    return best, best_score


def _coordinate_refinement(a, deadline, rng, max_rounds=20):
    """
    Coordinate descent refinement: try adjusting individual coordinates.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    best = a.copy()
    best_score = _fast_score(a)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        # Process coordinates in random order
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            # Try several candidate values
            candidates = [0.0, old_val * 0.5, old_val * 0.75, old_val * 1.25, old_val * 1.5,
                          old_val * 0.25, old_val * 2.0]
            # Also try values near the pressure-optimal
            best_trial = old_val
            best_trial_score = _fast_score(a)
            
            for cv in candidates:
                cv = max(0.0, min(1000.0, cv))
                a[i] = cv
                s = _fast_score(a)
                if s < best_trial_score:
                    best_trial_score = s
                    best_trial = cv
            
            a[i] = best_trial
            if best_trial_score < best_score:
                best = a.copy()
                best_score = best_trial_score
                improved = True
        
        if not improved:
            break
    
    return best, best_score
