def compute_objective(a):
    """Compute 2*n*max(convolve(a,a)) / sum(a)^2"""
    n = len(a)
    c = np.convolve(a, a)
    return 2.0 * n * np.max(c) / (np.sum(a)**2)

def optimize_sequence(a, deadline, rng):

    a = np.clip(a, 0, 1000)
    if np.sum(a) < 0.01:
        a[:] = 1.0
    # Normalize to sum = 1
    a = a / np.sum(a)

    best_score = compute_objective(a)
    
    # Subgradient descent
    step = 1.0  # will be adapted
    patience = 0
    max_patience = 200
    
    while time.monotonic() < deadline:
        c = np.convolve(a, a)
        max_c = np.max(c)
        score = 2.0 * n * max_c  # since sum = 1
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > max_patience:
            # Try a random restart
            a = a + rng.normal(0, 0.1/n, n)
            a = np.clip(a, 0, None)
            a = a / np.sum(a)
            patience = 0
            continue
        
        # Find active constraints (top indices of c)
        # Use the single max for simplicity
        k_star = np.argmax(c)
        
        # Compute subgradient
        grad = np.zeros(n)
        lo = max(0, k_star - (n - 1))
        hi = min(n - 1, k_star)
        if lo <= hi:
            grad[lo:hi+1] = 2.0 * a[k_star - lo : k_star - hi : -1]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Step
        a_new = a - step * grad / grad_norm
        a_new = np.clip(a_new, 0, None)
        s = np.sum(a_new)
        if s > 0:
            a_new = a_new / s
        else:
            a_new = a.copy()
            a_new[:] = 1.0/n
        
        # Check if improvement
        c_new = np.convolve(a_new, a_new)
        score_new = 2.0 * n * np.max(c_new)
        
        if score_new < score:
            a = a_new
        else:
            # Try smaller step
            step *= 0.9
    
    return best_a, 2.0 * n * best_score  # wait, best_score already includes the 2n factor if sum=1

def _subgradient_at(a, k):
    """Subgradient of c[k] = convolve(a,a)[k] with respect to a."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    if lo <= hi:
        grad[lo:hi + 1] = 2.0 * a[k - lo : k - hi - 1 : -1] if hi > lo else 2.0 * a[k - lo : k - lo + 1]
    return grad
