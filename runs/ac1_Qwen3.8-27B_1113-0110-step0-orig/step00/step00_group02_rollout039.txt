def smooth_max_subgradient(x, T=0.01):
    """Compute smooth subgradient of max(conv(x,x)) using log-sum-exp."""
    n = len(x)
    p = np.convolve(x, x, mode='full')
    # softmax weights
    p_max = p.max()
    w = np.exp((p - p_max) / T)
    w /= w.sum()
    # subgradient: g_j = 2 * sum_k w_k * x[k-j] (with bounds)
    # This is 2 * (w * x_reversed) appropriately
    # Actually, sum_k w_k * x[k-j] for 0≤k-j<n

    # Let i = k-j, so k = i+j, i ranges over 0..n-1
    # = sum_i w[i+j] * x[i] (where w is defined on 0..2n-2, 0 elsewhere)
    # This is a cross-correlation of w and x
    g = np.zeros(n)

import time
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import scipy.sparse as sp

def build_conv_matrix(x):
    """Build sparse matrix A such that A @ b = convolve(x, b, 'full')."""
    n = len(x)
    m = 2 * n - 1
    A = lil_matrix((m, n))
    for k in range(m):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for i in range(lo, hi + 1):
            A[k, i] = x[k - i]
    return A.tocsc()

        max_p = p_new.max()

import time
from scipy.optimize import linprog
import scipy.sparse as sp

def build_conv_matrix(x):
    """Build sparse matrix A such that A @ b = convolve(x, b, 'full')."""
    n = len(x)
    m = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(m):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for i in range(lo, hi + 1):
            rows.append(k)
            cols.append(i)
            vals.append(x[k - i])
    A = sp.csc_matrix((vals, (rows, cols)), shape=(m, n))
    return A

def lp_optimize(n, time_limit=600, max_iter=2000, seed=42):
    """LP-based iterative optimization."""
    x = np.ones(n) / n
    best_x = x.copy()
    best_score = 2.0
    
    start = time.monotonic()
    for it in range(max_iter):
        if time.monotonic() - start > time_limit:
            break
        
        p = np.convolve(x, x, mode='full')
        M = p.max()
        
        A = build_conv_matrix(x)
        c = -np.ones(n)
        res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), 
                      bounds=[(0, None)]*n, method='highs')
        
        if res.status != 0 or res.x is None:
            break
        
        b = res.x
        b_sum = b.sum()
        if b_sum < 1e-10:
            break
        
        b = b / b_sum
        
        # Try different mixing parameters
        t = max(0.01, 0.5 / (it + 1))
        x_new = (1 - t) * x + t * b
        
        p_new = np.convolve(x_new, x_new, mode='full')
        score_new = 2 * n * p_new.max()
        
        if score_new < best_score:
            best_score = score_new
            best_x = x_new.copy()
            x = x_new
        else:
            # Don't update x, just continue
            pass
        
        if it % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"  LP iter {it}: best_score={best_score:.6f}, elapsed={elapsed:.1f}s")
    
    return best_x, best_score

def subgradient_optimize(x, time_limit=300, T_init=1.0, T_final=0.01, max_iter=10000, seed=42):
    """Smoothed subgradient optimization."""
    n = len(x)
    x = x.copy()
    x = x / x.sum()  # normalize
    best_x = x.copy()
    p = np.convolve(x, x, mode='full')
    best_score = 2 * n * p.max()
    
    start = time.monotonic()
    alpha = 0.1  # initial step size
    
    for it in range(max_iter):
        if time.monotonic() - start > time_limit:
            break
        
        # Anneal temperature
        T = T_init * (T_final / T_init) ** (it / max_iter)
        
        p = np.convolve(x, x, mode='full')
        p_max = p.max()
        
        # Softmax weights
        w = np.exp((p - p_max) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(p)
            w[np.argmax(p)] = 1.0
        else:
            w /= w_sum
        
        # Subgradient via cross-correlation: g_j = 2 * sum_k w_k * x[k-j] (with bounds)
        # Use FFT for efficiency
        fft_size = 1
        while fft_size < 2*n:
            fft_size *= 2
        X = np.fft.rfft(x, fft_size)
        W = np.fft.rfft(w, fft_size)
        # g_j = 2 * sum_{i=0}^{n-1} x_i * w_{i+j}
        # This is the cross-correlation of x and w
        # crosscorr = ifft(conj(FFT(x)) * FFT(w)) gives sum_i x_i * w_{i+j}
        g = np.fft.irfft(np.conj(X) * W, fft_size)[:n] * 2
        
        # Step
        x_new = x - alpha * g
        x_new = np.maximum(x_new, 0)
        x_sum = x_new.sum()
        if x_sum < 1e-10:
            break
        x_new = x_new / x_sum
        
        p_new = np.convolve(x_new, x_new, mode='full')
        score_new = 2 * n * p_new.max()
        
        if score_new < best_score:
            best_score = score_new
            best_x = x_new.copy()
            x = x_new
            alpha = min(alpha * 1.05, 1.0)  # increase step size slightly
        else:
            alpha *= 0.9  # decrease step size
        
        if it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"  Subgrad iter {it}: best_score={best_score:.6f}, T={T:.4f}, alpha={alpha:.4f}, elapsed={elapsed:.1f}s")
    
    return best_x, best_score

import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp

def build_conv_matrix_sparse(x):
    """Build sparse matrix A such that A @ b = convolve(x, b, 'full')."""
    n = len(x)
    m = 2 * n - 1
    total_nz = n * n
    rows = np.empty(total_nz, dtype=np.int32)
    cols = np.empty(total_nz, dtype=np.int32)
    vals = np.empty(total_nz, dtype=np.float64)
    idx = 0
    for k in range(m):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        count = hi - lo + 1
        cols_arr = np.arange(lo, hi + 1, dtype=np.int32)
        rows[idx:idx + count] = k
        cols[idx:idx + count] = cols_arr
        vals[idx:idx + count] = x[k - cols_arr]
        idx += count
    return sp.csc_matrix((vals, (rows, cols)), shape=(m, n))

def lp_solve_one(x, M):
    """Solve LP: max sum(b) s.t. A@b <= M, b >= 0. Returns normalized b or None."""
    n = len(x)
    A = build_conv_matrix_sparse(x)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    s = res.x.sum()
    if s < 1e-10:
        return None
    return res.x / s
