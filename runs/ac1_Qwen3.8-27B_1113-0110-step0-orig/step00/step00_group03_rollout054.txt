import numpy as np
from scipy.optimize import linprog
from scipy import sparse

def solve_direction_lp(a):
    """Given fixed a, find b that minimizes max(a*b) subject to sum(b)=1, b>=0."""
    n = len(a)
    
    # Build constraint matrix for (a*b)[k] <= M
    # (a*b)[k] = sum_j a[j]*b[k-j] for valid j, k-j
    # So the k-th row has a[j] in column (k-j) for valid j
    
    # Variables: [M, b[0], b[1], ..., b[n-1]]
    # Objective: minimize M -> c = [1, 0, 0, ..., 0]
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2*n - 1):
        # (a*b)[k] - M <= 0
        # terms: -1 * M + sum_j a[j] * b[k-j] <= 0
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        # M term
        rows.append(k)
        cols.append(0)  # M is variable 0
        vals.append(-1.0)
        # b terms
        for j in range(j_start, j_end + 1):
            # b[j] appears with coefficient a[k-j]
            rows.append(k)
            cols.append(j + 1)  # b[j] is variable j+1
            vals.append(a[k-j])
    
    A_ub = sparse.csc_matrix((vals, (rows, cols)), shape=(2*n-1, n+1))
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([1.0])
    
    c = np.zeros(n+1)
    c[0] = 1.0  # minimize M
    
    bounds = [(0, None)] + [(0, None)] * n  # M >= 0, b >= 0
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.status == 0:
        return res.x[1:]  # return b
    return None

from scipy.optimize import linprog

def build_conv_matrix(n):
    """Build the sparse structure for convolution constraints.
    Returns lists for rows, cols of the (2n-1) x n matrix where
    entry [k, j] = 1 if max(0,k-n+1) <= j <= min(n-1,k), else 0.
    We'll multiply by a or b coefficients later.
    """
    row_idx = []
    col_idx = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            row_idx.append(k)
            col_idx.append(j)
    return row_idx, col_idx

def lp_solve_direction(a, n, row_idx, col_idx):
    """Given fixed a (sum=1), find b >= 0, sum(b)=1, minimizing max(a*b)."""
    # Variables: [M, b0, b1, ..., b_{n-1}]
    # Constraints: (a*b)[k] - M <= 0 for all k
    #             sum(b) = 1
    #             M >= 0, b >= 0
    
    # (a*b)[k] = sum_j a[j]*b[k-j] (where k-j is in [0,n-1])
    # In terms of b: (a*b)[k] = sum_j B[k,j] * b[j] where B[k,j] = a[k-j] if valid
    # So the constraint matrix for (a*b)[k] - M <= 0 is:
    #   -1 * M + sum_j B[k,j] * b[j] <= 0
    
    # Build sparse matrix
    rows = []
    cols = []
    vals = []
    
    for k in range(2*n - 1):
        # M term
        rows.append(k)
        cols.append(0)
        vals.append(-1.0)
        # b terms: b[j] has coefficient a[k-j] where k-j is in [0,n-1]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):

        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)

    n = len(parent)

    # Normalize to sum=1 for computation
    parent_norm = parent / parent.sum()

    # Evaluate initial score
    def eval_norm(a):
        """Evaluate a normalized (sum=1) sequence directly."""
        seq = [float(x) for x in a]
        s = float(evaluate_sequence(seq))
        return s

    best = parent_norm.copy()
    best_score = eval_norm(best)
    print(f"Initial: n={n}, score={best_score:.6f}")

    # Generate alternative initializations
    init_list = [parent_norm]

    # Ramp
    for exp in [0.5, 1.0, 2.0]:
        init_list.append(make_ramp_init(n, rng, exponent=exp))

    # Binary block
    init_list.append(make_binary_block_init(n, rng))

    # Staircase
    for steps in [3, 5, 8]:
        init_list.append(make_staircase_init(n, rng, steps=steps))

    # Strongly perturbed parent
    for _ in range(3):
        p = parent_norm * np.exp(rng.normal(0, 0.5, n))
        p = np.maximum(1e-4, p)
        p = p / p.sum()
        init_list.append(p)

    # Evaluate all initializations
    for i, init in enumerate(init_list):
        s = eval_norm(init)
        if np.isfinite(s) and s < best_score:
            best_score = s
            best = init.copy()
            print(f"  Init {i}: score={best_score:.6f}")

    # Precompute convolution indices for LP
    indices = precompute_conv_indices(n)

    # === Phase 1: LP-based alternating optimization ===
    a = best.copy()
    t_blend = 0.6
    lp_count = 0
    stall = 0

    print(f"\n--- LP Phase (n={n}) ---")
    while time.monotonic() < total_deadline - 400:
        t_start = time.monotonic()
        b = solve_lp_direction(a, n, indices, time_limit=min(60, max(10, total_deadline - time.monotonic() - 400)))
        lp_time = time.monotonic() - t_start

        if b is None:
            print(f"  LP failed at iter {lp_count}, switching to local search")
            break

        # Blend
        a_new = (1.0 - t_blend) * a + t_blend * b
        a_new = np.maximum(0.0, a_new)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        score = eval_norm(a_new)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best = a_new.copy()
            a = a_new.copy()
            lp_count += 1
            stall = 0
            t_blend = min(0.8, t_blend + 0.05)
            if lp_count % 3 == 0:
                print(f"  LP {lp_count}: score={best_score:.6f} (t={t_blend:.2f}, {lp_time:.1f}s)")
        else:
            stall += 1
            # Still update a partially to explore
            a = 0.7 * a + 0.3 * a_new
            a = np.maximum(0, a)
            a = a / max(a.sum(), 1e-15)
            t_blend = max(0.15, t_blend * 0.8)

        if stall > 30:
            print(f"  LP stalled after {stall} iters (count={lp_count}), switching to local search")
            break

    # === Phase 2: Local search ===
    print(f"\n--- Local Search Phase (best={best_score:.6f}) ---")
    a = best.copy()
    lr = 0.15
    ls_count = 0
    ls_stall = 0

    while time.monotonic() < total_deadline:
        # Occasionally do an LP step
        if ls_count > 0 and ls_count % 5000 == 0 and time.monotonic() < total_deadline - 80:
            b = solve_lp_direction(a, n, indices, time_limit=30)
            if b is not None:
                a_new = 0.4 * a + 0.6 * b
                a_new = np.maximum(0, a_new) / max(a_new.sum(), 1e-15)
                a = a_new
                continue

        # Choose perturbation
        r = rng.random()
        if r < 0.35:
            a_cand = block_perturb(a, rng, scale=lr)
        elif r < 0.55:
            a_cand = antisym_perturb(a, rng, scale=lr)
        elif r < 0.75:
            a_cand = wide_block_antisym(a, rng, scale=lr)
        elif r < 0.85:
            # Paired coordinate swap (break symmetry)
            a_cand = a.copy()
            i1 = int(rng.integers(0, n // 2))
            i2 = int(rng.integers(n // 2, n))
            swap_amt = min(a[i1], a[i2]) * lr * float(rng.uniform(0.5, 2.0))
            a_cand[i1] = a[i1] + swap_amt
            a_cand[i2] = max(0.0, a[i2] - swap_amt)
        else:
            # Random reset of a small block to new values
            a_cand = a.copy()
            w = max(1, int(rng.integers(1, max(2, n // 80))))
            st = int(rng.integers(0, max(1, n - w + 1)))
            a_cand[st:st + w] = a[st:st + w] * float(np.exp(rng.normal(0, lr * 2, w)))
            a_cand = np.maximum(0, a_cand)

        s = a_cand.sum()
        if s < 1e-10:
            continue
        a_cand = a_cand / s

        score = eval_norm(a_cand)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best = a_cand.copy()
            a = a_cand.copy()
            ls_count += 1
            ls_stall = 0
            if ls_count % 2000 == 0:
                print(f"  LS {ls_count}: score={best_score:.6f}, lr={lr:.4f}")
        else:
            ls_stall += 1
            if ls_stall % 500 == 0 and lr > 0.02:
                lr *= 0.9

        # Periodic restart: try a fresh perturbation of current best
        if ls_count % 10000 == 0 and ls_count > 0:
            a_restart = best * np.exp(rng.normal(0, 0.1, n))
            a_restart = np.maximum(1e-6, a_restart)
            a_restart = a_restart / a_restart.sum()
            a = a_restart

    print(f"\nFinal result: n={n}, score={best_score:.6f}")
    print(f"  LP iterations: {lp_count}")
    print(f"  LS improvements: {ls_count}")

    return [float(x) for x in best]
