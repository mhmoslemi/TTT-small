    w = np.exp(shifted)

def _compute_conv(p):
    n = len(p)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    F = np.fft.rfft(p, N)
    return np.fft.irfft(F * F, N)[:2 * n - 1]


def _compute_grad(p, conv, tau):
    n = len(p)
    shifted = tau * (conv - conv.max())
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return np.zeros(n)

    N = 1
    while N < 3 * n - 2:

    W = np.fft.fft(w, N)
