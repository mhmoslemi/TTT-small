def build_convolution_matrix(f):
    """Build the matrix A such that A @ g = fftconvolve(f, g, 'full')."""
    n = len(f)
    # Full convolution has length 2n-1
    # (f*g)[k] = sum_j f[j] * g[k-j] for valid j

    # A[k, j] = f[k-j] if 0 <= k-j <= n-1, else 0
    # But we need 0 <= j <= n-1 as well.
    
    # Let me build it row by row
    prev = 2*n - 1  # length of full convolution
    A = np.zeros((prev, n))
    for k in range(prev):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:

    return A

Minimizes 2*n*max(convolve(x,x))/(sum(x))^2 using LP-based iteration 
with simulated annealing restarts and multiple starting points.
"""

from scipy.sparse import lil_matrix, csr_matrix

def build_conv_matrix(f):
    """Build sparse matrix A where A @ g = full_convolution(f, g)."""
    n = len(f)
    m = 2 * n - 1
    A = lil_matrix((m, n))
    for k in range(m):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            A[k, j] = f[k - j]
    return A.tocsr()

def max_conv(x):
    """Compute max of full self-convolution."""
    conv = np.convolve(x, x)
    return np.max(conv)

def objective(x):
    n = len(x)
    s = np.sum(x)
    if s < 0.01:
        return np.inf
    return 2.0 * n * np.max(np.convolve(x, x)) / (s * s)

def lp_step(f, M):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)_k <= M for all k."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0)
    return None

            M_new = max_conv(f_new)
            if M_new < M:
                f = f_new
                M = M_new
                if M < best_M:
                    best_M = M
                    best_f = f.copy()
                improved = True
                break

                    M = M_new

                        best_f = f.copy()
                    improved = True

        if it % 10 == 0:
            print(f"  LP iter {it}: M={M:.6f}, obj={2*n*M/(np.sum(f)**2):.6f}")
    
    return best_f, best_M

def simulated_annealing(f, time_budget=60, T_start=0.01, T_end=0.0001):
    """Simulated annealing to minimize max convolution with fixed sum."""
    f = f.copy().astype(np.float64)
    n = len(f)
    s = np.sum(f)
    f = f / s  # normalize to sum = 1
    M = max_conv(f)
    best_M = M
    best_f = f.copy()
    
    start = time.monotonic()
    it = 0
    while time.monotonic() - start < time_budget:
        # Cooling schedule
        progress = min(1.0, it / (time_budget * 1000))  # rough estimate
        T = T_start * (T_end / T_start) ** progress

        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        amount = np.random.uniform(0, 0.5) * f[i]
        if amount > f[i]:
            amount = f[i]
        f_new = f.copy()
        f_new[i] -= amount
        f_new[j] += amount

            f = f_new
            M = M_new
            if M < best_M:
                best_M = M
                best_f = f.copy()
        it += 1
    
    # Rescale to original sum
    return best_f * s, best_M

"""
Minimizes 2*n*max(convolve(x,x))/(sum(x))^2 using LP-based direction finding
with simulated annealing fine-tuning and multiple diverse starting points.
Key insight: non-symmetric sequences can break the obj=2 barrier; the LP finds
optimal blending directions that flatten the autocorrelation peak.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import lil_matrix


def build_A(f):
    """Build sparse matrix A where A @ g = full_convolution(f, g)."""
    n = len(f)
    m = 2 * n - 1
    A = lil_matrix((m, n))
    for k in range(m):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        # A[k, j] = f[k-j] for j in [lo, hi]
        vals = f[k - hi : k - lo + 1][::-1]
        A[k, lo:hi + 1] = vals
    return A.tocsr()


def max_conv(x):
    """Compute max of full self-convolution using FFT for speed."""
    conv = np.convolve(x, x)
    return np.max(conv)


def get_obj(x):
    """Compute objective 2*n*max_b/sum_a^2."""
    n = len(x)
    s = np.sum(x)
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_conv(x) / (s * s)


def lp_find_direction(f, M, timeout=20):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for all k. Returns g or None."""
    n = len(f)
    A = build_A(f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                      method='highs', options={'time_limit': timeout, 'presolve': True})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0)
            if np.sum(g) > 1e-10:
                return g
    except Exception:
        pass
    return None


def lp_optimize(f, time_budget=300):
    """Run LP-based iteration to reduce max convolution."""
    f = f.astype(np.float64).copy()
    n = len(f)
    M = max_conv(f)

    it = 0
    stall = 0

        M_work = max_conv(f_work)
        g = lp_find_direction(f_work, M_work, timeout=15 if n <= 500 else 10)
        if g is None:
            stall += 1
            if stall > 5:
                break
            f_work = f + np.random.default_rng(it + 100).random(n) * 0.01 * np.max(f)

            g = lp_find_direction(f_work, M_work, timeout=10)

            stall += 1
            if stall > 8:
                break
            continue

        g = g / np.sum(g) * np.sum(f)
        found = False

def simulated_annealing(f, time_budget=120, seed=0):
    """SA to minimize max_conv with fixed sum, non-negative constraint."""
    rng = np.random.default_rng(seed)
    f = f.astype(np.float64).copy()
    n = len(f)
    s = np.sum(f)
    fn = f / s  # normalize to sum=1
    M = max_conv(fn)
    best_M = M
    best_fn = fn.copy()
    start = time.monotonic()
    it = 0
    T = s * 0.01

    while time.monotonic() - start < time_budget:
        # Cooling
        T *= 0.9995
        # Perturbation: move mass between two or a few positions
        n_pert = int(rng.integers(1, 4))
        idx = rng.integers(0, n, size=2 * n_pert)
        fn_new = fn.copy()
        for p in range(n_pert):
            i, j = idx[2 * p], idx[2 * p + 1]
            if i == j:
                continue
            amt = rng.uniform(0, 0.3) * fn[i]
            if amt > fn[i]:
                amt = fn[i]
            fn_new[i] -= amt
            fn_new[j] += amt
        fn_new = np.maximum(fn_new, 0)
        M_new = max_conv(fn_new)
        d = M_new - M
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-15)):
            fn = fn_new
            M = M_new
            if M < best_M:
                best_M = M
                best_fn = fn.copy()
        it += 1

    return best_fn * s, best_M


def block_perturbation(f, rng, sigma=0.1):
    """Block-based perturbation for local search."""
    n = len(f)
    f_new = f.copy()
    n_blocks = int(rng.integers(1, max(2, n // 100) + 1))
    for _ in range(n_blocks):
        width = int(rng.integers(max(2, n // 50), max(3, n // 8)))
        start = int(rng.integers(0, max(1, n - width)))
        f_new[start:start + width] *= np.exp(rng.normal(0, sigma, width))
    f_new = np.maximum(f_new, 0)
    # Renormalize to keep sum constant
    if np.sum(f_new) > 0:
        f_new = f_new * np.sum(f) / np.sum(f_new)
    return f_new

            print(f"    -> Best so far from LP: {obj_res:.6f}")

    # Phase 2: SA on current best
    print("\n=== Phase 2: Simulated Annealing ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        sa_budget = min(remaining * 0.35, 300)
        print(f"  SA budget: {sa_budget:.0f}s")
        f_sa, M_sa = simulated_annealing(best_seq, time_budget=sa_budget, seed=42)
        obj_sa = get_obj(f_sa)
        print(f"  SA result: obj={obj_sa:.6f}")
        if obj_sa < best_obj:
            best_obj = obj
            best_seq = f_sa.copy()
            print(f"    -> New best: {best_obj:.6f}")

    # Phase 3: More LP on refined best
    print("\n=== Phase 3: Additional LP ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        lp_budget = min(remaining * 0.4, 400)
        print(f"  LP budget: {lp_budget:.0f}s")
        f_lp, obj_lp = lp_optimize(best_seq, time_budget=lp_budget)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_seq = f_lp.copy()
            print(f"    -> New best: {best_obj:.6f}")

    # Phase 4: Local search with block perturbations
    print("\n=== Phase 4: Local Search ===")
    remaining = deadline - time.monotonic()
    if remaining > 10:
        rng = np.random.default_rng(99)
        n = len(best_seq)
        s0 = np.sum(best_seq)
        it = 0
        while time.monotonic() < deadline - 5:
            f_pert = block_perturbation(best_seq, rng, sigma=0.08)
            obj_pert = get_obj(f_pert)
            if obj_pert < best_obj:
                best_obj = obj_pert
                best_seq = f_pert.copy()
                if it % 100 == 0:
                    print(f"    it={it}, new best={best_obj:.6f}")
            it += 1
        print(f"  Local search done, iterations={it}")

    # Phase 5: Final LP polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 5: Final LP Polish ({remaining:.0f}s) ===")
        f_final, obj_final = lp_optimize(best_seq, time_budget=min(remaining - 5, 200))
        if obj_final < best_obj:
            best_obj = obj_final
            best_seq = f_final.copy()
            print(f"  -> New best: {best_obj:.6f}")

    # Ensure validity
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if np.sum(best_seq) < 0.01:
        best_seq = np.ones(200)

    final_score = float(evaluate_sequence([float(x) for x in best_seq]))
    print(f"\n=== FINAL RESULT ===")
    print(f"  n={len(best_seq)}, best_found={best_obj:.6f}, verified={final_score:.6f}")
    print(f"  sum={np.sum(best_seq):.4f}, max_conv={max_conv(best_seq):.6f}")

    return [float(x) for x in best_seq]
