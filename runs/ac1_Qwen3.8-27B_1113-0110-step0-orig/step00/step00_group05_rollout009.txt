import time

def make_step_sequence(n, m, r):
    """Create a two-level step sequence: [1]*m + [r]*(n-m)."""
    seq = np.ones(n)
    seq[m:] = r
    return seq

def fast_objective(seq):
    """Compute the objective without using evaluate_sequence."""
    n = len(seq)
    conv = np.convolve(seq, seq)
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)

    rng = np.random.default_rng(seed)
