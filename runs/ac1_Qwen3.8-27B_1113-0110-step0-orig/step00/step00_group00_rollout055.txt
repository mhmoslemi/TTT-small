"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences a.
Uses log-sum-exp smoothed gradient descent with Adam optimizer,
projected onto the probability simplex. Multiple restarts with
various initializations and a final exact-subgradient polish phase.
"""

import numpy as np
import time


def _next_pow2(x):
    """Next power of 2 >= x."""
    return 1 << (x - 1).bit_length() if x > 1 else 2


def _fft_cross(w, a, N):
    """Compute grad[k] = sum_j w[j] * a[j-k] for k=0..n-1 using FFT.
    Here w has length 2n-1 and a has length n."""
    n = len(a)
    rev_a = a[::-1]
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(rev_a, N)
    conv = np.fft.irfft(W * A, N)
    return conv[n-1:2*n-1]


def _smooth_gradient(a, e, alpha, N):
    """Compute gradient of log-sum-exp(alpha * conv(a,a))."""
    n = len(a)  # a has length n, e has length 2n-1
    emax = e.max()
    log_w = alpha * (e - emax)
    w = np.exp(log_w)
    w /= w.sum()
    grad = _fft_cross(w, a, N)
    return 2.0 * grad  # factor of 2 from d/dx of x*y


def _exact_subgradient(a, e, N, rel_thresh=0.01):
    """Compute subgradient of max(conv(a,a)) using exact peak positions."""
    n = len(a)
    M = e.max()
    threshold = M * (1.0 - rel_thresh)
    mask = (e >= threshold).astype(np.float64)
    if mask.sum() == 0:
        mask[e.argmax()] = 1.0
    grad = _fft_cross(mask, a, N)
    return 2.0 * grad / mask.sum()


def _project_simplex(a):
    """Project onto {x >= 0, sum(x) = 1} via clip and normalize."""
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _optimize(a_init, rng, time_budget, use_smooth=True, smooth_first=0.6):
    """Main optimization loop."""
    n = len(a_init)
    a = np.clip(a_init.astype(np.float64), 0.0, None)
    if a.sum() < 0.01:
        a = np.ones(n)
    a = a / a.sum()
    
    N = _next_pow2(3 * n - 2)
    
    best = a.copy()
    e = np.convolve(a, a)
    best_score = 2.0 * n * e.max()
    
    deadline = time.monotonic() + time_budget
    phase_deadline = time.monotonic() + time_budget * smooth_first if use_smooth else 0
    
    # Adam parameters
    lr = 0.1
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    no_improve = 0
    alpha = 50.0
    
    while time.monotonic() < deadline:
        t += 1
        e = np.convolve(a, a)
        M = e.max()
        score = 2.0 * n * M
        
        if score < best_score * (1.0 - 1e-9):
            best_score = score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Choose gradient type
        use_exact = (not use_smooth) or (time.monotonic() > phase_deadline)
        if use_exact:
            grad = _exact_subgradient(a, e, N)
        else:
            grad = _smooth_gradient(a, e, alpha, N)
        
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        
        # Project
        a = _project_simplex(a)
        
        # Adaptive schedule
        if t % 500 == 0:
            if not use_exact:
                alpha = min(alpha * 1.4, 500.0)
            lr = max(lr * 0.95, 0.005)
        
        # Restart
        if no_improve > 2000:
            sigma = min(0.05, 0.01 * np.sqrt(t / 1000.0))
            a = a + sigma * rng.standard_normal(n)
            a = _project_simplex(a)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            if not use_exact:
                alpha = max(20.0, alpha * 0.5)
            lr = 0.1
            no_improve = 0
    
    # Final polish with exact subgradient
    remaining = deadline - time.monotonic()
    if remaining > 5.0:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences a.
Uses log-sum-exp smoothed gradient (transitioning to exact subgradient)
with Adam optimizer projected onto the probability simplex.
FFT-based convolution and gradient computation for speed.
Multiple restarts with various initializations and sequence lengths.

import numpy as np
import time


def _next_pow2(x):
    """Next power of 2 >= x."""
    if x <= 1:
        return 2
    return 1 << (x - 1).bit_length()


def _convolve_fft(a, N):
    """Linear self-convolution using FFT. Returns length 2n-1."""
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:2 * len(a) - 1]


def _fft_cross(w, a, N):
    """Compute grad[k] = sum_j w[j]*a[j-k] for k=0..n-1.
    w has length 2n-1, a has length n. Uses FFT."""
    n = len(a)
    rev_a = a[::-1]
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(rev_a, N)
    conv = np.fft.irfft(W * A, N)
    return conv[n - 1:2 * n - 1]


def _smooth_gradient(a, e, alpha, N):
    """Gradient of log-sum-exp(alpha * conv(a,a))."""
    emax = e.max()
    log_w = alpha * (e - emax)
    w = np.exp(log_w)
    w /= w.sum()
    return 2.0 * _fft_cross(w, a, N)


def _exact_subgradient(a, e, N, rel_thresh=0.005):
    """Subgradient of max(conv(a,a)) at near-peak positions."""
    M = e.max()
    threshold = M * (1.0 - rel_thresh)
    mask = (e >= threshold).astype(np.float64)
    s = mask.sum()
    if s == 0:
        mask[e.argmax()] = 1.0
        s = 1.0
    return 2.0 * _fft_cross(mask, a, N) / s


def _project_simplex(a):
    """Project onto {x >= 0, sum(x)=1}."""
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s
