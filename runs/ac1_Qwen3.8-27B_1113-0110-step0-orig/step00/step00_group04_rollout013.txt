"""
Return an exact optimizer (with a small safe probe step) for the stated
nonnegative evaluator.

For any finite nonnegative sequence a of length n,
    max(a*a) >= (a*a)[n-1] = sum_i a_i**2 >= (sum_i a_i)**2 / n
by Cauchy-Schwarz.  Therefore the displayed evaluation function satisfies
    score(a) >= 2.
Equality is attained by any positive constant sequence.

The constant sequence is therefore globally optimal for the evaluator as
written.  A few additional candidates are evaluated only as guards: if the
runtime evaluator differs and one of them is finite and better, it is
returned; otherwise the constant optimizer is returned.
"""


def _optimal_length():
    """Choose a reasonable length, preferring the supplied construction."""
    try:
        ref = height_sequence_1

        return 3809

        n = int(len(ref))

        return 3809

    # Keep direct convolution cheap.  The optimum is length-invariant.
    if 2 <= n <= 5000:
        return n
    return 3809


def _to_float_sequence(obj, limit_n=5000):
    """Convert an iterable of numbers to a finite, nonnegative, scaled list."""
    try:
        if len(obj) > limit_n:
            return None
    except TypeError:
        pass

    seq = []
    total = 0.0

    for x in obj:
        try:
            y = float(x)
        except Exception:
            return None

        if y != y or y == float("inf") or y == float("-inf"):
            return None
        if y < 0.0:
            y = 0.0
        elif y > 1000.0:
            y = 1000.0

        seq.append(y)
        total += y

        if len(seq) > limit_n:
            return None

    if not seq or total <= 0.0:
        return None

    inv = 1.0 / total
    if inv != inv or inv == float("inf"):
        return None

    return [float(x * inv) for x in seq]


def _global_candidate(limit_n=5000):

def _candidate_pool(n):
    """Gather simple robust candidates.  Only the best finite one is kept."""
    candidates = [[1.0] * n]

    if n > 1:

        # variants where the centered peak is not the limiting term.

    for h in (1e-3, 0.1, 0.3, 0.7, 10.0):

    return candidates


def _select_best(n):
