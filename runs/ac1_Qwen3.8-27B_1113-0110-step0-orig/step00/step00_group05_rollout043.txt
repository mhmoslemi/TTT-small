    """

    deadline = time.monotonic() + budget_s - 5.0
    
    best_score = np.inf

    inits = []

    if 'height_sequence_1' in globals():

            inits.append(given / np.sum(given))

    for n in [2048, 4096, 8192]:

    for n in [2048, 4096]:
        h = np.hanning(n)
        # Make it slightly asymmetric
        h = h * (1 + 0.1 * np.linspace(-1, 1, n))

        inits.append(h / h.sum())

    for n in [2048, 4096, 8192]:
        a = np.exp(-np.linspace(0, 3, n))

import numpy as np
import time

# Helper functions at top level

def fft_convolve(a, b):
    """FFT-based linear convolution."""
    n = len(a) + len(b) - 1
    N = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n]

def convolve_seq(a):
    """Compute a*a (autocorrelation) efficiently."""
    n = len(a)
    if n <= 8192:
        return np.convolve(a, a)
    else:
        return fft_convolve(a, a)

def project_simplex(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1) / rho_val
    return np.maximum(v - theta, 0)

def compute_grad_lse(a, tau):
    """
    Compute gradient of tau * log(sum exp(ac/tau)) where ac = a*a.
    Returns (gradient, lse_value, ac).
    """
    n = len(a)
    ac = convolve_seq(a)

    ac_max = ac.max()

    # S[i] = sum_{j=0}^{n-1} probs[i+j]

    S = np.zeros(n)
