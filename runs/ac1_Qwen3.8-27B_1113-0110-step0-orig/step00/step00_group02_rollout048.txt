"""
Minimizes 2n*max(conv(x,x))/sum(x)^2 using gradient descent on a
softmax-smoothed peak objective. Uses FFT for O(n log n) convolution
and cross-correlation. Employs softmax (log-space) parameterization for
automatic constraint handling, momentum for acceleration, adaptive
temperature, and multiple diverse initializations.
"""

import numpy as np
import time


def _fft_convolve(a, b):
    """Compute full linear convolution of two real arrays using FFT."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]


def _cross_correlate(p, x):
    """Compute (p star x)[k] = sum_j p[j+k]*x[j] for k=0..len(x)-1.
    p has length 2n-1, x has length n."""
    n = len(x)
    rev_x = x[::-1]
    conv = _fft_convolve(p, rev_x)
    return conv[n - 1:2 * n - 1]


def _compute_grad_and_c(x, soft_beta):
    """Compute gradient of soft-max smoothed objective and the convolution."""
    n = len(x)
    c = _fft_convolve(x, x)
    c_max = np.max(c)
    exp_c = np.exp(soft_beta * (c - c_max))
    p = exp_c / np.sum(exp_c)
    r = _cross_correlate(p, x)
    g = 2.0 * r
    diag = 2 * np.arange(n)
    valid = diag < len(p)
    g[valid] -= p[diag[valid]] * x[valid]
    return g, c


def _optimize_logspace(x_init, deadline, rng):

        g_y = x * (g_x - xg)

        y = y - lr * momentum
