def compute_convolution(p):
    """Compute the full convolution of p with itself using FFT."""
    n = len(p)
    # Use FFT for efficiency
    size = 2 * n - 1
    # Next power of 2 for FFT efficiency
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    F = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(F * F, fft_size)[:size]
    return conv
