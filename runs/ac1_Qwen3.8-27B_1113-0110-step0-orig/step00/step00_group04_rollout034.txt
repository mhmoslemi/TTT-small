    sum_a = a.sum()
    if sum_a < 0.01:

def subgradient(a, active_lags):
    """Compute subgradient of max_k c[k] at a."""
    n = len(a)
    g = np.zeros(n)
    for k in active_lags:
        # c[k] = sum_i a[i]*a[k-i], valid i: max(0, k-n+1) <= i <= min(n-1, k)

            if 0 <= idx < n:

    g /= len(active_lags)
    return g

def project_simplex(v, s):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    # Binary search for threshold
    lo, hi = 0.0, v.max()
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.maximum(v - mid, 0)
        s_proj = proj.sum()
        if s_proj > s:
            lo = mid
        else:
            hi = mid
    return np.maximum(v - (lo + hi) / 2, 0)

def optimize_sequence(a_init, n_iter=5000, eta0=0.01):
    """Subgradient descent to minimize max_k c[k] with sum(a)=n, a>=0."""
    n = len(a_init)
    a = np.array(a_init, dtype=np.float64)
    # Normalize to sum = n
    a = a / a.sum() * n
    best_a = a.copy()
    best_score = objective(a)

        max_c = c.max()
        active = np.where(c == max_c)[0]

"""
Minimizes the evaluation function 2*n*max(convolve(a,a))/(sum(a))^2
using subgradient descent with simplex projection, multiple restarts,
and adaptive step sizes.
"""

import numpy as np
import time

def compute_score(a):
    """Fast computation of the objective without calling evaluate_sequence."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    c = np.convolve(a, a)
    max_c = c.max()
    return 2.0 * n * max_c / (sum_a ** 2)

def get_active_lags(c, tol=1e-6):
    """Get indices where c is close to its maximum."""
    max_c = c.max()
    if max_c == 0:
        return [0]
    threshold = max_c * (1 - tol)
    return list(np.where(c >= threshold)[0])

def compute_subgradient(a, active_lags):
    """Compute subgradient of max_k c[k] at a, using FFT-based convolution."""
    n = len(a)
    if len(active_lags) == 0:
        return np.zeros(n)
    h = np.zeros(2 * n - 1)
    for k in active_lags:
        h[k] = 1.0
    a_rev = a[::-1]
    conv = np.convolve(h, a_rev, mode='full')
    g = conv[n-1 : 2*n - 1] * (2.0 / len(active_lags))
    return g

def project_onto_simplex(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    # Use the algorithm from Wang & Carota (2009)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def subgradient_descent(a_init, n_iters, eta0, cooling=0.999):
    """Run subgradient descent to minimize max_k c[k] with sum(a)=n, a>=0."""
    n = len(a_init)
    a = np.array(a_init, dtype=np.float64).copy()
    
    # Ensure non-negative
    a = np.maximum(a, 0)
    if a.sum() < 1e-10:
        a = np.ones(n)
    
    # Normalize to sum = n
    a = a / a.sum() * n
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    eta = eta0
    for t in range(n_iters):
        c = np.convolve(a, a)
        max_c = c.max()
        active = get_active_lags(c, tol=1e-4)
        g = compute_subgradient(a, active)
        
        # Update
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new, n)
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        eta *= cooling
    
    return best_a, best_score

    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

        if time.monotonic() > deadline:

        # Multiple initializations for this n
        inits = []
        
        # Constant
        inits.append(np.ones(n))
        
        # Raised cosine (zero at edges)
        x = np.linspace(0, np.pi, n)
        inits.append(np.sin(x) ** 2)  # zero at both ends
        
        # Cosine bump
        x = np.linspace(-1, 1, n)
        inits.append(np.cos(np.pi * x / 2) ** 2)  # positive everywhere
        
        # Exponential

        # Linear ramp (asymmetric)
