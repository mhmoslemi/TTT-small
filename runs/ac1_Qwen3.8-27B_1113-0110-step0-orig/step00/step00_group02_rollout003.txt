    g = cp.Variable(n, nonneg=True)

Step function optimization via projected subgradient descent with LP polishing.
Minimizes 2*n*max(conv(x,x))/sum(x)^2 by normalizing sum(x)=1 and
minimizing max(conv(x,x)) using subgradient steps and periodic LP refinements.
"""

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.sparse import csr_matrix

def full_convolve(x, y=None):
    """Full convolution using FFT."""
    if y is None:
        y = x
    n = len(x)
    m = len(y)
    size = n + m - 1
    X = fft(x, size)
    Y = fft(y, size)
    return np.real(ifft(X * Y, size))

def lp_polish(f, M, n, solver=cp.CBC):
    """Solve LP: max sum(g) s.t. g >= 0, conv(f,g)_k <= M for all k."""
    import cvxpy as cp
    
    g_var = cp.Variable(n, nonneg=True)
    
    # Build convolution matrix
    # C[k, j] = f[k-j] for 0 <= k-j < n
    row_idx = np.repeat(np.arange(n), n) + np.tile(np.arange(n), n)
    col_idx = np.repeat(np.arange(n), n)
    vals = np.tile(f, n)
    
    # Filter valid entries (row_idx must be < 2n-1)
    valid = row_idx < (2 * n - 1)
    row_idx = row_idx[valid]
    col_idx = col_idx[valid]
    vals = vals[valid]
    
    C = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    
    objective = cp.Maximize(cp.sum(g_var))
    constraints = [C @ g_var <= M]
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=solver)
        if g_var.value is not None:
            return np.asarray(g_var.value, dtype=np.float64)
    except Exception as e:
        pass
    
    # Fallback: uniform
    return np.ones(n) / n
