"""
Step-function search that minimizes 2*n*max(f*f)/(sum(f)**2) for a
non-negative height sequence.

Main idea: alternate between (a) measuring the current convolution peak
M = max(f*f) and (b) solving the linear program
        max sum(g)  s.t.  g >= 0,  (f * g)[k] <= M  for all k,
where f is held fixed so the constraints are linear in g.  The LP-driven
direction 'fills in' the valleys of the autocorrelation, flattening it and
lowering the peak.  We blend f <- (1-t)f + t*g (normalized) with a short
line search and repeat.  Multiple starts break the LP degeneracy that
occurs at the perfectly uniform vector.  A Vecchia-style sparse convolution
matrix keeps each LP solve tractable for n ~ several thousand.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt


# ---------------------------------------------------------------------------
# helpers (all top-level, no closures)
# ---------------------------------------------------------------------------

def build_conv_matrix(f):
    """Sparse (2n-1) x n matrix A with (A @ g)[k] = (f * g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        js = k - n + 1
        if js < 0:
            js = 0
        je = k + 1
        if je > n:
            je = n
        m = je - js
        if m > 0:
            i_start = k - (je - 1)
            rows.append(np.full(m, k, dtype=np.int64))
            cols.append(np.arange(js, je, dtype=np.int64))
            vals.append(np.asarray(f[i_start:i_start + m], dtype=np.float64))
    if not rows:
        return sp.csr_matrix((2 * n - 1, n))
    return sp.csr_matrix((np.concatenate(vals),
                          (np.concatenate(rows), np.concatenate(cols))),
                         shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve max sum(g) s.t. A@g <= M, g >= 0.  Return normalized g or None."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n, dtype=np.float64)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    try:
        res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                          method='highs',
                          options={"presolve": True})
    except Exception:
        return None
    if (not res.success) or res.x is None:
        return None
    g = np.maximum(res.x, 0.0)
    gs = g.sum()
    if not np.isfinite(gs) or gs < 1e-14:
        return None
    return g / gs


def objective(f, n):
    """2*n*max(f*f)/(sum(f))**2 ; returns inf on degenerate input."""
    s = f.sum()
    if not np.isfinite(s) or s <= 1e-14:
        return np.inf
    conv = np.convolve(f, f)
    mb = conv.max()
    if not np.isfinite(mb):
        return np.inf
    return 2.0 * n * mb / (s * s)


def line_search(f, g, base_score, n):
    """Pick the best blend (1-t)f + t g over a few t's."""
    best_f = f
    best_s = base_score
    for t in (1.0, 0.7, 0.5, 0.4, 0.3, 0.2, 0.1):
        fn = (1.0 - t) * f + t * g
        np.clip(fn, 0.0, None, out=fn)
        if fn.sum() < 1e-14:
            continue
        fn = fn / fn.sum()
        s = objective(fn, n)
        if np.isfinite(s) and s < best_s - 1e-12:
            best_s = s
            best_f = fn.copy()
    return best_f, best_s


def block_perturb(base, rng):
    """Cheap local polish: multiply a random block by a small factor."""
    cand = base.copy()
    n = cand.size
    if n < 4:
        return cand
    width = int(rng.integers(max(2, n // 400), max(3, n // 40)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.05)))
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def lp_refine(f0, n, deadline, tag):
    """Run the LP iteration until the deadline or convergence."""
    f = f0.copy()
    best_f = f.copy()
    best_s = objective(f, n)
    print(f"[{tag}] start score = {best_s:.6f}")
    no_improve = 0
    for it in range(400):
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        conv = np.convolve(f, f)
        M = conv.max()
        s = objective(f, n)
        if np.isfinite(s) and s < best_s:
            best_s = s
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        if it % 10 == 0:
            print(f"[{tag}] it={it} score={best_s:.6f} M={M:.3e} "
                  f"last={t0 and (time.monotonic()-t0):.2f}s")
        g = solve_lp_direction(f, M)
        if g is None:
            # Fallback: gentle block perturbation.
            g = block_perturb(f, np.random.default_rng(it))
            gs = g.sum()
            if gs < 1e-14:
                break
            g = g / gs
        nf, ns = line_search(f, g, s, n)
        if ns >= s - 1e-12:
            no_improve += 1
            if ns < best_s:
                best_s = ns
                best_f = nf.copy()
            if no_improve >= 4:
                break
            f = block_perturb(f, np.random.default_rng(10**6 + it))
            continue
        f = nf
        no_improve = 0
    if objective(best_f, n) > best_s + 1e-9:
        best_s = objective(best_f, n)
    return best_f, best_s
