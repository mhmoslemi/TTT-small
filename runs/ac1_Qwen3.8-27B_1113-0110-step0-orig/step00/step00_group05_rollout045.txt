def linear_ramp(n, t):
    """Generate a linear ramp sequence."""
    if n == 1:
        return np.ones(1)
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + t * (i / (n - 1) - 0.5)

    a = np.maximum(a, 0.0)
    return a

def exp_ramp(n, alpha):
    """Generate an exponential ramp sequence."""
