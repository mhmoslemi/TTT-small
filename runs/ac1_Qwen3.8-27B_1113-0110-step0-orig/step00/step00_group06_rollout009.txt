import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def compute_score(a):
    n = len(a)
    c = np.convolve(a, a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    return float(2 * n * np.max(c) / (s ** 2))

def solve_lp_direction(a, M, sum_target=1.0):
    """Find b >= 0, sum(b) = sum_target, maximizing sum(b) s.t. conv(a,b) <= M."""
    n = len(a)
    # Actually, let's reformulate: minimize t s.t. b>=0, sum(b)=1, conv(a,b)<=t
    # This directly gives the flattest correlation.
    # Or: maximize sum(b) s.t. b>=0, conv(a,b)<=M
    # The second is what SOTA uses.
    
    # Constraint matrix: row j is a shifted version of a
    # A[j, i] = a[i] if 0 <= j-i < n
    # This gives conv(a,b)[j] = A[j] @ b
    
    # Build sparse matrix
    rows = []
    cols = []
    vals = []
    for j in range(2 * n - 1):
        i_start = max(0, j - (n - 1))
        i_end = min(n - 1, j)
        for i in range(i_start, i_end + 1):
            rows.append(j)
            cols.append(i)
            vals.append(a[i])
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    
    c_obj = -np.ones(n)  # maximize sum(b)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, 
                  bounds=[(0, None)] * n, method='highs')

    return None
