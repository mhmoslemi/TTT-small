with LP-based direction finding. Uses FFT for fast convolution.
Tries multiple sequence lengths, initializations, and optimization phases.
"""

import numpy as np

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    s = v.sum()
    if s <= 1e-15:
        return np.ones(n) / n
    v = v / s
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_conv(f):
    """Compute linear convolution f*f using FFT. Returns array of length 2n-1."""
    n = len(f)
    size = 2 * n - 1
    F = np.fft.rfft(f, n=size)
    return np.fft.irfft(F * F, n=size)

    n = len(f)

    if S < 1e-10:
        return 1e10
    h = compute_conv(f)
    return 2.0 * n * h.max() / (S * S)

            n_gd = 20000
            
            print(f"n={n}, init={init_idx}, start_score={fast_score(f):.6f}")
            t0 = time.monotonic()
            
            best_f, best_score = try_optimize(f.copy(), deadline, use_lp, n_gd)
            
            elapsed = time.monotonic() - t0
            print(f"  result: score={best_score:.6f}, time={elapsed:.1f}s")
            
            if best_score < global_best_score:
                global_best_score = best_score
                global_best_seq = best_f.copy()
                print(f"  *** NEW GLOBAL BEST: {global_best_score:.6f} ***")
            
            # Verify with exact evaluator
            if global_best_seq is not None:
                exact = evaluate_sequence(global_best_seq.tolist())
                if np.isfinite(exact) and exact < global_best_score:
                    global_best_score = exact
                    print(f"  verified exact score: {exact:.6f}")
    
    # If we still have time, do a final polish on the best sequence
    if global_best_seq is not None and time.monotonic() < deadline - 30:
        print("Polishing best sequence...")
        n = len(global_best_seq)
        polished_f, polished_score = run_gd_phase(
            global_best_seq.copy(), deadline, tau_init=1.0/(n*10.0), lr_init=0.5, max_iters=50000
        )
        if polished_score < global_best_score:
            global_best_score = polished_score
            global_best_seq = polished_f.copy()
            print(f"  polished score: {global_best_score:.6f}")
    
    if global_best_seq is None:
        global_best_seq = np.ones(100) / 100
    
    final_score = evaluate_sequence(global_best_seq.tolist())
    print(f"FINAL: score={final_score:.6f}, length={len(global_best_seq)}")
    
    return [float(max(0.0, min(1000.0, x))) for x in global_best_seq]
