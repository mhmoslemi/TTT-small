import time
from scipy.optimize import linprog

def compute_autocorr_peak(f):
    """Compute max of f*f (full convolution)."""
    n = len(f)
    # For efficiency, use FFT for large n
    if n > 500:
        size = 2 * n - 1
        fft_size = 1
        while fft_size < size:
            fft_size *= 2
        F = np.fft.rfft(f, fft_size)
        conv = np.fft.irfft(F * F, fft_size)[:size]
    else:
        conv = np.convolve(f, f)
    return np.max(conv), conv

def solve_direction_lp(f, M):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)_k <= M for all k."""

    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)

    # So A_ub[k, j] = f[k-j] if 0 <= k-j <= n-1, else 0
