```python
def frank_wolfe(n, max_iter, tol):
    a = np.ones(n) / n  # start at center of simplex
    for it in range(max_iter):
        conv = np.convolve(a, a)
        j_max = np.argmax(conv)
        # subgradient
        g = np.zeros(n)
        for k in range(n):
            idx = j_max - k
            if 0 <= idx < n:
                g[k] = 2 * a[idx]
        # linear minimization over simplex
        k_min = np.argmin(g)
        e_k = np.zeros(n)
        e_k[k_min] = 1.0
        # step size (use L2R: γ = 2/(it+2))
        gamma = 2.0 / (it + 2)
        a_new = (1 - gamma) * a + gamma * e_k
        a_new = np.maximum(a_new, 0)  # projection to non-negative (should be redundant since we're on simplex)
        if np.linalg.norm(a_new - a) < tol:
            break
        a = a_new
    return a
```

```python
def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    N = 1 << (2*n - 1 - 1).bit_length()  # next power of 2 >= 2n-1
    fa = fft(a, N)
    result = np.real(ifft(fa * fa))
    return result[:2*n-1]

def compute_gradient(a, w):
    """Compute gradient g[k] = 2*sum_i a[i]*w[i+k] using FFT."""
    n = len(a)
    w_rev = w[::-1].copy()
    N = 1 << (3*n - 2 - 1).bit_length()  # next power of 2 >= 3n-2
    fa = fft(a, N)
    fw_rev = fft(w_rev, N)
    conv = np.real(ifft(fa * fw_rev))
    # g[k] = conv[2n-2-k]
    g = conv[2*n-2 : n-1 : -1] * 2  # indices 2n-2, 2n-3, ..., n-1
    return g
```

The projection onto the simplex `{x >= 0, sum x = 1}`:

```python
def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    # Algorithm from Duchi et al. 2008
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
    """Compute f(x) = smoothed max of conv(softmax(x), softmax(x)) and its gradient."""
    a = np.exp(x - x.max())  # softmax without dividing (we'll normalize)
    a /= a.sum()
    
    conv = np.convolve(a, a)
    # Smoothed max
    shifted = conv / mu
    shifted -= shifted.max()
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    obj = mu * (np.log(exp_shifted.sum()) + shifted.max())  # = mu * logsumexp(conv/mu)

    # = 2 * cross_correlation(w, a) at appropriate lags
    # w has length 2n-1, a has length n

    # Let i = j-k: grad_k = 2 * sum_{i: 0<=i<n, 0<=i+k<2n-1} w[i+k] * a[i]
    # This is the same as before!
    # grad_k = 2 * sum_i a[i] * w[i+k]
    
    # Using FFT:
    w_rev = w[::-1].copy()
    N = 1 << (3*n - 2 - 1).bit_length()
    fa = fft(a, N)
    fw = fft(w_rev, N)
    conv_au = np.real(ifft(fa * fw))
    grad = 2 * conv_au[2*n-2 : n-1 : -1]
    
    # Gradient w.r.t. x
    dot_grad_a = np.dot(grad, a)
    grad_x = a * (grad - dot_grad_a)
    
    return obj, grad_x
```

```python
Uses softmax parameterization and L-BFGS with multi-scale smoothing.
"""
import numpy as np

from scipy.fft import fft, ifft
from scipy.optimize import minimize

def _fft_convolve(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2*n - 1
    N = 1 << (size - 1).bit_length()
    fa = fft(a, N)
    result = np.real(ifft(fa * fa))
    return result[:size]

def _compute_grad_a(a, w):
    """
    Compute gradient of g(a)=sum_j w_j*(a*a)_j w.r.t. a.
    grad_k = 2 * sum_j w_j * a[j-k] (for valid j-k)
    Uses FFT. Returns array of length n.
    """
    n = len(a)
    w_rev = w[::-1].copy()
    size = n + (2*n-1) - 1  # = 3n-2
    N = 1 << (size - 1).bit_length()
    fa = fft(a, N)
    fw = fft(w_rev, N)
    conv = np.real(ifft(fa * fw))
    # grad_k = 2 * conv[(2n-2) - k]
    grad = 2.0 * conv[2*n-2 : n-1 : -1]
    return grad

def _smoothed_max_obj_and_grad(x, mu):
    """
    Compute smoothed max of conv(softmax(x), softmax(x)) and gradient w.r.t. x.
    x: unconstrained parameter vector of length n
    mu: smoothing parameter
    Returns (objective, gradient).
    """
    n = len(x)
    # Softmax
    a = np.exp(x - x.max())
    a /= a.sum()
    
    # Convolution
    conv = _fft_convolve(a)
    
    # Smoothed max
    s = conv / mu
    s_max = s.max()
    exp_s = np.exp(s - s_max)
    log_sum = np.log(exp_s.sum()) + s_max
    obj = mu * log_sum
    
    # Weights
    w = exp_s / exp_s.sum()
    
    # Gradient w.r.t. a: grad_a[k] = 2 * sum_j w[j] * a[j-k]
    grad_a = _compute_grad_a(a, w)
    
    # Gradient w.r.t. x: grad_x[i] = a[i] * (grad_a[i] - dot(grad_a, a))
    dot_grad_a = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot_grad_a)
    
    return obj, grad_x

def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """
    Projected gradient descent on the simplex.
    Minimizes smoothed max of conv(a,a).
    """
    a = a.copy().astype(np.float64)

    a /= a.sum()

    best_conv_max = np.max(_fft_convolve(a))

        conv = _fft_convolve(a)
        cur_max = np.max(conv)
        if cur_max < best_conv_max:

            best_a = a.copy()

        s = conv / mu
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()

        a = _project_simplex(a_new)
        
        # Adjust step size

    return best_a, best_conv_max
```

2. Fix λ, find the optimal a that minimizes sum_j λ_j (a*a)_j s.t. a >= 0, sum a = 1

Step 2 is a QP: min a^T Q a s.t. a >= 0, sum a = 1 where Q is the matrix corresponding to sum_j λ_j M_j.

This QP can be solved efficiently using the KKT conditions. The optimal a satisfies:
- 2 Q a + μ * 1 + ν = 0 (KKT)
- a_i >= 0, ν_i <= 0, a_i * ν_i = 0
- sum a_i = 1

Where μ is the multiplier for the sum constraint and ν is the vector of multipliers for a >= 0.

On the support S = {i : a_i > 0}:
- 2 (Q a)_i + μ = 0 for i ∈ S
- 2 (Q a)_i + μ >= 0 for i ∉ S

1. Assume a support S
2. Solve the linear system: 2 Q_{S,S} a_S + μ * 1_S = 0, sum_S a_S = 1
3. Check if a_S > 0 and the dual constraints
4. Update S and repeat

```python
"""
Convex optimization approach to minimize 2n*max(conv(a,a))/(sum(a))^2.
Uses softmax parameterization with L-BFGS multi-scale smoothing,
projected gradient descent, and coordinate descent polish.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def _fft_convolve_same(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = fft(a, N)
    result = np.real(ifft(fa * fa))
    return result[:size]


def _compute_grad_a(a, w):
    """
    Compute grad_k = 2 * sum_j w[j] * a[j-k] for k=0..n-1 using FFT.
    a: length n, w: length 2n-1.
    """
    n = len(a)
    w_rev = w[::-1].copy()
    size = n + (2 * n - 1) - 1  # 3n-2
    N = 1 << (size - 1).bit_length()
    fa = fft(a, N)
    fw = fft(w_rev, N)
    conv = np.real(ifft(fa * fw))
    grad = 2.0 * conv[2 * n - 2 : n - 1 : -1]
    return grad


def _smoothed_max(x, n, mu):
    """
    Return (objective, gradient) for smoothed max of conv(softmax(x), softmax(x)).
    x: unconstrained vector of length n.
    """
    a = np.exp(x - x.max())
    a /= a.sum()
    conv = _fft_convolve_same(a)
    s = conv / mu
    s_max = s.max()
    exp_s = np.exp(s - s_max)
    logsum = np.log(exp_s.sum()) + s_max
    obj = mu * logsum
    w = exp_s / exp_s.sum()
    grad_a = _compute_grad_a(a, w)
    dot_ga = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot_ga)
    return obj, grad_x


def _project_simplex(v):
    """Project v onto probability simplex {x>=0, sum x = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    mask = u * idx > (cssv - 1)
    if not mask.any():
        return np.ones(n) / n
    rho = np.nonzero(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _pgd_simplex(a_init, mu_start, max_iter, deadline):
    """Projected gradient descent on simplex for smoothed max of conv(a,a)."""
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    best_a = a.copy()
    best_val = np.max(_fft_convolve_same(a))
    eta = 1.0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        conv = _fft_convolve_same(a)
        cur = np.max(conv)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
        mu = mu_start / (1.0 + it * 0.01)
        s = conv / mu
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        grad = _compute_grad_a(a, w)
        a = _project_simplex(a - eta * grad)
        if it % 200 == 199:
            eta *= 0.98
    return best_a, best_val


def _lbfgs_optimize(a_init, n, deadlines):
    """
    Multi-scale L-BFGS optimization on softmax parameterization.
    Returns best sequence found.
    """
    best_a = a_init.copy()
    best_val = np.max(_fft_convolve_same(a_init))
    x0 = np.log(a_init + 1e-15)

    for mu in [1.0, 0.1, 0.01, 0.001, 0.0001]:
        if time.monotonic() > deadlines:
            break
        try:
            res = minimize(
                _smoothed_max, x0, args=(n, mu),
                method='L-BFGS-B', jac=True,
                options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-10, 'iprint': 0}
            )
            x_opt = res.x
            a_opt = np.exp(x_opt - x_opt.max())
            a_opt /= a_opt.sum()
            a_opt = np.maximum(a_opt, 0.0)
            a_opt /= a_opt.sum()
            val = np.max(_fft_convolve_same(a_opt))
            if val < best_val:
                best_val = val
                best_a = a_opt.copy()
            x0 = x_opt
        except Exception:
            pass
    return best_a

    ns_to_try = [256, 512, 1024, 2048]

    def make_init(n, init_type):
        t = np.arange(n) - (n - 1) / 2.0
        x = np.abs(t) / max(1.0, (n - 1) / 2.0)
        if init_type == 'uniform':
            a = np.ones(n)
        elif init_type == 'gaussian':
            a = np.exp(-3.0 * x ** 2)
        elif init_type == 'raised_cosine':
            a = 0.5 * (1.0 + np.cos(np.pi * np.clip(2.0 * x - 1.0, -1, 1)))
            a = np.clip(a, 0, None)
        elif init_type == 'parabolic':
            a = np.clip(1.0 - x ** 2, 0, None)
        elif init_type == 'bimodal':
            a = np.exp(-5.0 * (x - 0.5) ** 2) + np.exp(-5.0 * (x + 0.5) ** 2)
        elif init_type == 'plateau':
            a = np.ones(n)
            w = int(n * 0.3)
            a[:w] *= 0.7
            a[-w:] *= 0.7
        else:
            a = np.ones(n)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.ones(n) / n
        return a

    init_types = ['uniform', 'gaussian', 'raised_cosine', 'parabolic', 'bimodal', 'plateau']

    for n in ns_to_try:
        if time.monotonic() > deadline - 45:
            break

        for itype in init_types:
            if time.monotonic() > deadline - 30:
                break

            a_init = make_init(n, itype)
            label = f"n{n}_{itype}"

            # L-BFGS multi-scale
            if time.monotonic() < deadline - 30:
                a_opt = _lbfgs_optimize(a_init, n, deadline - 15)
                try_sequence(a_opt, label + "_LBFGS")
                # Polish
                if time.monotonic() < deadline - 15:
                    a_pol = _coord_descent_polish(a_opt, n, deadline - 5, max_sweeps=10)
                    try_sequence(a_pol, label + "_polish")

            # PGD
            if time.monotonic() < deadline - 15:
                a_pgd, _ = _pgd_simplex(a_init, mu_start=1.0, max_iter=50000, deadline=deadline - 5)
                try_sequence(a_pgd, label + "_PGD")

        # A couple random inits for larger n
        if n >= 512 and time.monotonic() < deadline - 20:
            for trial in range(2):
                if time.monotonic() > deadline - 10:
                    break
                a_rand = np.abs(rng.standard_normal(n)) + 0.1
                a_rand /= a_rand.sum()
                a_opt = _lbfgs_optimize(a_rand, n, deadline - 10)
                try_sequence(a_opt, f"n{n}_rand{trial}")

    # Final: try asymmetric inits (shift the bump)
    for n in [512, 1024]:
        if time.monotonic() > deadline - 20:
            break
        for shift in [0.3, 0.4, 0.35, 0.45]:
            if time.monotonic() > deadline - 10:
                break
            t = np.arange(n) - (n - 1) * shift
            a = np.exp(-4.0 * (t / (n * 0.25)) ** 2)
            a = np.clip(a, 0, None)
            a /= a.sum()
            a_opt = _lbfgs_optimize(a, n, deadline - 5)
            try_sequence(a_opt, f"n{n}_shift{shift}")

    if best_seq is None:
        best_seq = [1.0] * 128

    print(f"\nFinal best score: {best_score:.6f}, length={len(best_seq)}")
    return best_seq
```
