Minimizes 2*n*max(convolve(x,x))/sum(x)^2 using FFT-accelerated gradient
descent with power-mean smoothing, momentum, and multi-phase optimization.

def compute_smooth_gradient(x, c, p, N_fft):
    """Compute gradient of sum(c^p) w.r.t. x using FFT."""
    c_power = c ** (p - 1)
    F_x = np.fft.rfft(x, N_fft)
    F_c = np.fft.rfft(c_power, N_fft)
    R = np.fft.irfft(np.conj(F_x) * F_c, N_fft)
    return p * 2.0 * R[:len(x)]
