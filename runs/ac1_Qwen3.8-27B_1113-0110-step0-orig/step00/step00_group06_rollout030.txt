def safe_eval(a):
    """Safely compute the score."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0, 1000)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    max_b = np.max(conv)
    return float(2 * n * max_b / (sum_a ** 2))

def smooth_max_val(conv, tau):
    m = np.max(conv)
    return float(m + np.log(np.sum(np.exp(tau * (conv - m)))) / tau)

def smooth_max_w(conv, tau):
    m = np.max(conv)
    e = np.exp(tau * (conv - m))
    return e / np.sum(e)

    """Gradient of smooth_max(conv(a)) w.r.t. a."""
    n = len(a)
    L = 1
    while L < 4 * n:

    print("Phase 2: Projected gradient descent")
    n_values = [20, 30, 40, 50, 75, 100, 150, 200, 300, 400, 500]
    
    for n in n_values:
        if time.monotonic() > deadline:
            break
        
        remaining = deadline - time.monotonic()
        n_remaining = len([x for x in n_values if x >= n])
        time_per_n = remaining / max(1, n_remaining)
        
        # Multiple initializations
        inits = [
            ("uniform", np.ones(n) / n),
            ("decreasing", np.linspace(1.0, 0.1, n)),
            ("increasing", np.linspace(0.1, 1.0, n)),
            ("cosine", np.cos(np.pi * np.arange(n) / (2.0 * n)) ** 2),
            ("bimodal", np.concatenate([np.ones(n//4), np.zeros(n-n//2), np.ones(n//4)])),
        ]
        
        for name, init in inits:
            if time.monotonic() > deadline:
                break
            t_budget = time_per_n / len(inits)
            if t_budget < 2:
                t_budget = 2
            a_opt, score = optimize_pgdescent(n, init, time_budget=t_budget, seed=hash(name) % 10000)
            update_best(a_opt, score)
        
        # Also try from the best found so far (if same length)
        if best_seq is not None and len(best_seq) == n:
            a_opt, score = optimize_pgdescent(n, best_seq, time_budget=time_per_n/3, seed=42)
            update_best(a_opt, score)
