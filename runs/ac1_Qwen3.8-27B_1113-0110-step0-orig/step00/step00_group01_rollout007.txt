But the issue is: the max is at k_star, and we're only reducing conv[k_star]. After the update, a different k might become the new max. This is like "noisy gradient descent" on the nonsmooth max function.

To handle this better, I could use a "smoothed" version: instead of max, use a soft-max (log-sum-exp) of the conv values.

F_β(a) = (1/β) * log(sum_k exp(β * conv[k]))

This is smooth and its gradient is:
g[j] = (1/β) * sum_k (β * p[k]) * d(conv[k])/d(a[j])
where p[k] = exp(β*conv[k]) / sum_j exp(β*conv[j])

This averages the gradients weighted by how "active" each constraint is. Much smoother!

Let me implement this:

```python
def smooth_objective_and_grad(a, beta):
    n = len(a)
    conv = np.convolve(a, a)
    # Soft max
    max_conv = np.max(conv)
    exp_conv = np.exp(beta * (conv - max_conv))
    p = exp_conv / np.sum(exp_conv)
    smooth_val = (max_conv + np.log(np.sum(exp_conv))) / beta
    
    # Gradient
    g = np.zeros(n)
    for k in range(2*n-1):
        if p[k] < 1e-10:
            continue
        for j in range(max(0, k-n+1), min(n-1, k)+1):
            g[j] += p[k] * 2.0 * a[k-j]
    g /= beta
    return smooth_val, g
```

    """Compute soft-max of conv(a,a) and its gradient w.r.t. a."""
    n = len(a)
    conv = np.convolve(a, a)  # length 2n-1

    exp_shifted = np.exp(shifted)

    if s > 0:

    else:

"""
Search for low-peak convolution sequences using smoothed mirror descent
with LP-based best response updates and multiple starting points.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse

def normalize_nonneg(seq):
    seq = np.maximum(seq, 0.0)
    s = seq.sum()
    if s > 1e-30:
        return seq / s
    return seq

def exact_objective(a):
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (s * s)

def gradient_of_soft_max(a, beta):
    """Compute (2/beta) * grad of soft-max of conv(a,a)."""
    n = len(a)
    conv = np.convolve(a, a)
    mc = conv.max()
    ex = np.exp(beta * (conv - mc))
    p = ex / ex.sum()
    # g[j] = (2/beta) * sum_{m=0}^{n-1} a[m] * p[m+j]
    # = (2/beta) * correlate(p, a, 'full')[:n]
    corr = np.correlate(p, a, mode='full')
    g = (2.0 / beta) * corr[:n]
    return g, conv, p

def mirror_step(a, g, eta):
    a_new = a * np.exp(-eta * g)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-30:
        a_new /= s
    else:
        a_new = a.copy()
    return a_new

def lp_best_response(a, M, top_k=100):
    """Solve reduced LP for best response."""
    n = len(a)
    conv_aa = np.convolve(a, a)
    top_idx = np.argsort(conv_aa)[-top_k:]
    
    rows, cols, vals, rhs = [], [], [], []
    for idx, k in enumerate(top_idx):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            j = k - i
            if a[i] > 1e-15:
                rows.append(idx)
                cols.append(j)
                vals.append(a[i])
        rhs.append(M)
    
    if not rows:
        return None
    A = sparse.csc_matrix((vals, (np.array(rows), np.array(cols))), shape=(len(top_idx), n))
    res = linprog(-np.ones(n), A_ub=A, b=np.array(rhs),
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.status == 0:
        return res.x
    return None

def run_mirror_descent(a0, beta, eta, n_steps, rng, deadline):
    a = a0.copy()
    best_local = a.copy()
    best_score = exact_objective(a)
    for it in range(n_steps):
        if time.monotonic() > deadline:
            break
        g, conv, p = gradient_of_soft_max(a, beta)
        a = mirror_step(a, g, eta)
        if it % 10 == 0:
            sc = exact_objective(a)
            if sc < best_score:
                best_score = sc
                best_local = a.copy()
        # Add noise periodically
        if it % 30 == 0 and it > 0:
            a = a * np.exp(rng.normal(0, 0.03, len(a)))
            a = normalize_nonneg(a)
    return best_local, best_score
