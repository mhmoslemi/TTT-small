import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def solve_lp_for_q(p, n):
    """Solve: min t s.t. (p*q)[k] <= t for all k, sum(q)=1, q>=0."""
    # Variables: q_0, ..., q_{n-1}, t (total n+1 variables)
    # c (objective): minimize t
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Inequality constraints: (p*q)[k] - t <= 0 for k = 0, ..., 2n-2
    # Build sparse matrix A_ub of shape (2n-1, n+1)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        for i in range(max(0, k - n + 1), min(n, k + 1)):
            j = k - i
            if 0 <= j < n:
                rows.append(k)
                cols.append(j)
                vals.append(p[i])
        # -t column
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n+1))
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(q) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: q_i >= 0, t >= 0
    bounds = [(0, None)] * n + [(0, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    
    if result.success:
        return result.x[:n], result.x[n]
    return None, None

"""
Minimize 2*n*max(f*f)/(sum(f))^2 using alternating linear programming
and local search. Reduces to minimizing max_k (p*p)[k] over probability
vectors p of length n. Uses LP-based coordinate descent (many-body style)
followed by perturbation-based local search.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix, csr_matrix


def _build_lp_indices(n):
    """Precompute (row, col) pairs for the (2n-1) x n cross-correlation matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            rows.append(k)
            cols.append(k - i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def _lp_min_cross_peak(fixed, n, row_idx, col_idx):
    """
    Solve: min t  s.t.  (fixed * x)[k] <= t  for k=0..2n-2,
                        sum(x) = 1, x >= 0.
    Returns (x, t_peak) or (None, None).
    """
    m = 2 * n - 1
    # A_ub is (2n-1) x (n+1): last column is -1 for t
    # Nonzeros in first n columns: fixed[row - col]
    vals_main = fixed[row_idx - col_idx]
    
    # Add -1 entries for t column
    all_rows = np.concatenate([row_idx, np.arange(m, dtype=np.int32)])
    all_cols = np.concatenate([col_idx, np.full(m, n, dtype=np.int32)])
    all_vals = np.concatenate([vals_main, -np.ones(m)])
    
    A_ub = coo_matrix((all_vals, (all_rows, all_cols)), shape=(m, n + 1)).tocsr()
    b_ub = np.zeros(m)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    bounds = [(0.0, None)] * (n + 1)
    
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success:
        return res.x[:n].copy(), float(res.x[n])
    return None, None


def _eval_peak(p):
    """Peak of autocorrelation of p (assumes p >= 0)."""
    return float(np.max(np.convolve(p, p)))


def _alternating_lp(p_init, n, row_idx, col_idx, time_budget, seed=0):
    """Run alternating LP iterations. Returns best sequence found."""
    rng = np.random.default_rng(seed)
    best = p_init.copy()
    best_peak = _eval_peak(best)
    deadline = time.monotonic() + time_budget
    
    p = p_init.copy()
    stall = 0
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        # LP: find q minimizing max(p * q)
        q, t_val = _lp_min_cross_peak(p, n, row_idx, col_idx)
        if q is None:
            break
        q = np.clip(q, 0.0, None)
        s = q.sum()
        if s < 1e-14:
            break
        q /= s
        
        peak_q = _eval_peak(q)
        if peak_q < best_peak - 1e-14:
            best_peak = peak_q
            best = q.copy()
            stall = 0
        else:
            stall += 1
        
        # LP: find p_new minimizing max(q * p_new)
        p_new, t_val2 = _lp_min_cross_peak(q, n, row_idx, col_idx)
        if p_new is None:
            p = q
            continue
        p_new = np.clip(p_new, 0.0, None)
        s = p_new.sum()
        if s < 1e-14:
            p = q
            continue
        p_new /= s
        
        peak_p = _eval_peak(p_new)
        if peak_p < best_peak - 1e-14:
            best_peak = peak_p
            best = p_new.copy()
            stall = 0
        
        # Update: use the LP solution as new "fixed" for next round
        p = p_new.copy()
        
        # Periodic random restart to escape local minima
        if stall > 40 and it % 50 == 0:
            # Perturb best and retry
            pert = best + rng.normal(0, 0.02, n)
            pert = np.clip(pert, 0, None)
            ps = pert.sum()
            if ps > 1e-10:
                pert /= ps
                if _eval_peak(pert) < best_peak:
                    best = pert
                    best_peak = _eval_peak(pert)
                p = pert.copy()
                stall = 0
        
        if it % 50 == 0:
            print(f"    LP iter {it}: best_peak={best_peak:.12f}, obj={2*n*best_peak:.6f}, "
                  f"stall={stall}, t={time.monotonic() - (deadline - time_budget):.0f}s")
    
    return best, best_peak


def _local_search(p, n, time_budget, seed=0):
    """Perturbation-based local search to polish the solution."""
    rng = np.random.default_rng(seed)
    best = p.copy()
    best_peak = _eval_peak(best)
    deadline = time.monotonic() + time_budget
    step = 0.08
    stall = 0
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        c = best.copy()
        r = rng.random()
        
        if r < 0.25:
            # Block scaling
            w = int(rng.integers(max(1, n // 100), max(2, n // 10)))
            s0 = int(rng.integers(0, n - w))
            c[s0:s0 + w] *= np.exp(rng.normal(0, 0.15))
        elif r < 0.5:
            # Smooth sinusoidal perturbation
            freq = rng.uniform(0.3, 8.0)
            amp = step * 0.3
            c += amp * np.sin(2 * np.pi * freq * np.arange(n) / n + rng.uniform(0, 6.28))
        elif r < 0.7:
            # Multi-scale smoothing
            alpha = rng.uniform(0.6, 0.9)
            c = alpha * c + (1 - alpha) * 0.5 * (c + c[::-1])
        elif r < 0.85:
            # Targeted reduction at peak lag
            conv = np.convolve(c, c)
            k_peak = int(np.argmax(conv))
            # Reduce elements contributing most to peak
            for i in range(n):
                j = k_peak - i
                if 0 <= j < n:
                    c[i] *= np.exp(-0.05)
                    c[j] *= np.exp(-0.05)
        else:
            # Random walk with reflection
            idx = rng.choice(n, size=max(1, n // 20), replace=False)
            c[idx] *= np.exp(rng.normal(0, 0.2, size=len(idx)))
        
        c = np.clip(c, 0.0, None)
        s = c.sum()
        if s < 1e-12:
            continue
        c /= s
        
        peak = _eval_peak(c)
        if peak < best_peak:
            best = c
            best_peak = peak
            step = max(0.005, step * 0.98)
            stall = 0
        else:
            stall += 1
            if stall % 500 == 0:
                step = max(0.001, step * 0.9)
    
    return best, best_peak
