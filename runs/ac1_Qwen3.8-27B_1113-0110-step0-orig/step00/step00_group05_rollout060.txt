"""
Hybrid FFT-based search for non-negative step-function heights.

The score is 2*n*max(conv(a,a))/sum(a)^2.  The routine uses:
  - token low-resolution (piecewise-constant) initialization,
  - mirror-descent steps biased toward the lag(s) where the autocorrelation
    is maximal,
  - random/value-swapping/block perturbations to escape local minima,
  - periodic optional length doubling near the end.
"""

import math
import time

import numpy as np


def next_pow2(x):
    p = 1
    x = int(max(1, x))
    while p < x:
        p <<= 1
    return p


def fast_normalize(a):
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 0.0:
        return a
    return a / s


def safe_normalize(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a) & (a > 0.0), a, 0.0)
    s = float(np.sum(a))
    if s <= 0.0:
        return a
    return a / s


def conv_fft_size(a, size):
    f = np.fft.rfft(a, size)
    return np.fft.irfft(f * f, size)


def score_norm_size(p, n, size):
    c = conv_fft_size(p, size)
    if c.size < 2 * n - 1:
        return float("inf")
    m = float(np.max(c[:2 * n - 1]))
    if not np.isfinite(m):
        return float("inf")
    if m < 0.0:
        m = 0.0
    return float(2.0 * n * m)


def cross_corr_weights_size(w, p, size, idx):
    wrev = w[::-1]
    fw = np.fft.rfft(wrev, size)
    fp = np.fft.rfft(p, size)
    conv = np.fft.irfft(fw * fp, size)
    return conv[idx]


def profile_exp_poly(t, a, b, c):
    u = t - 0.5
    z = a * t + b * (u * u) + c * (u * u * u)
    zmax = float(np.max(z))
    if not np.isfinite(zmax):
        return safe_normalize(np.ones_like(t, dtype=np.float64))
    z = z - zmax
    p = np.exp(np.clip(z, -740.0, 0.0))
    return safe_normalize(p)


def profile_tanh(t, eps, alpha):
    x = np.tanh(float(alpha) * (t - 0.5))
    p = 1.0 + float(eps) * x
    p = np.clip(p, 1e-12, None)
    return safe_normalize(p)


def profile_logit(t, eps, k):
    z = float(k) * (t - 0.5)
    z = np.clip(z, -50.0, 50.0)
    g = 1.0 / (1.0 + np.exp(-z))
    p = 1.0 + float(eps) * (g - 0.5)
    p = np.clip(p, 1e-12, None)
    return safe_normalize(p)


def profile_two_level(n, frac, ratio):
    p = np.ones(n, dtype=np.float64)
    k = int(np.clip(round(float(frac) * n), 1, n - 1))
    p[:k] = float(ratio)
    return safe_normalize(p)


def make_tile(q, m, n):
    r = n // m
    if r <= 0:
        return safe_normalize(np.ones(n, dtype=np.float64))
    q = np.asarray(q, dtype=np.float64)
    z = q - float(np.max(q))
    vals = np.exp(np.clip(z, -740.0, 0.0))
    p = np.repeat(vals, r)
    p = p[:n]
    if p.size < n:
        pad = np.zeros(n - p.size, dtype=np.float64)
        p = np.concatenate((p, pad))
    return safe_normalize(p)


def interpolate_to_n(p, n):
    p = np.asarray(p, dtype=np.float64)
    if p.size == n:
        return safe_normalize(p)
    if p.size < 2:
        return safe_normalize(np.ones(n, dtype=np.float64))

    q = np.log(np.maximum(p, 1e-300))
    xo = np.linspace(0.0, 1.0, p.size)
    xn = np.linspace(0.0, 1.0, n)
    qn = np.interp(xn, xo, q)
    qn -= float(np.max(qn))
    a = np.exp(np.clip(qn, -740.0, 0.0))
    return safe_normalize(a)


def gradient_direction(p, gamma, n, grad_size, conv_size, cross_idx):
    c = conv_fft_size(p, conv_size)[:2 * n - 1]
    m = float(np.max(c))
    if not np.isfinite(m) or m <= 0.0:
        return None

    d = n * c
    z = d / (m + 1e-300)
    z = np.maximum(z, 0.0)

    w = np.power(z, float(gamma))
    sw = float(np.sum(w))
    if not np.isfinite(sw) or sw <= 0.0:
        return None
    w /= sw

    h = 2.0 * cross_corr_weights_size(w, p, grad_size, cross_idx)

    q = np.log(p + 1e-300)
    with np.errstate(divide="ignore", invalid="ignore"):
        rel = h / (p + 1e-300)
    rel = np.where(np.isfinite(rel), rel, 0.0)

    absr = np.abs(rel)
    mask = absr > 1e-12
    if np.any(mask):
        scale = float(np.median(absr[mask]))
    else:
        scale = 1.0
    if not np.isfinite(scale) or scale <= 1e-300:
        scale = 1.0

    rel = rel / (scale + 1e-12)
    rel = np.clip(rel, -12.0, 12.0)
    rel -= float(np.mean(rel))

    return q, rel, float(2.0 * n * m)


def candidate_from_direction(q, rel, eta):
    q2 = q - float(eta) * rel
    q2 = q2 - float(np.max(q2))
    a = np.exp(np.clip(q2, -740.0, 0.0))
    return safe_normalize(a)


def swap_candidate(p, k, rng):
    n = p.size
    cand = p.copy()
    for _ in range(int(k)):
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i != j:
            tmp = cand[i]
            cand[i] = cand[j]
            cand[j] = tmp
    return cand


def block_candidate(p, sigma, rng):
    n = p.size
    q = np.log(p + 1e-300)
    minw = max(8, n // 64)
    maxw = max(16, n // 8)
    if minw >= maxw:
        w = minw
    else:
        w = int(rng.integers(minw, maxw))
    if w > n:
        w = n
    start = int(rng.integers(0, n - w + 1))
    q[start:start + w] += rng.normal(0.0, float(sigma), size=w)
    q -= float(np.max(q))
    a = np.exp(np.clip(q, -740.0, 0.0))
    return safe_normalize(a)


def sine_candidate(p, amp, rng, idx):
    n = p.size
    q = np.log(p + 1e-300)
    f = float(rng.uniform(0.5, 5.0))
    ph = float(rng.uniform(0.0, 2.0 * np.pi))
    q += float(amp) * np.sin(2.0 * np.pi * f * idx / float(max(1, n)) + ph)
    q -= float(np.max(q))
    a = np.exp(np.clip(q, -740.0, 0.0))
    return safe_normalize(a)


def polarization_candidate(p, beta, rng, conv_size):
    n = p.size
    c = conv_fft_size(p, conv_size)[:2 * n - 1]
    m = float(np.max(c))
    cand = p.copy()
    if not np.isfinite(m) or m <= 0.0:
        return cand

    idx = np.where(c >= 0.75 * m)[0]
    if idx.size == 0:
        return safe_normalize(cand)

    count = min(4, int(idx.size))
    for _ in range(count):
        k = int(idx[int(rng.integers(idx.size))])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi < lo:
            continue

        i = int(rng.integers(lo, hi + 1))
        j = k - i
        if i == j:
            nbr = i + (1 if rng.random() < 0.5 else -1)
            if 0 <= nbr < n:
                delta = float(beta) * cand[i]
                if delta > 0.0 and delta < cand[i]:
                    cand[i] -= delta
                    cand[nbr] += delta
        else:
            a = cand[i]
            b = cand[j]
            small = min(a, b)
            delta = float(beta) * small
            if delta > 0.0 and delta < small - 1e-300:
                if a < b:
                    cand[i] -= delta
                    cand[j] += delta
                else:
                    cand[j] -= delta
                    cand[i] += delta

    return safe_normalize(cand)


def tile_random_init(n, time_limit, rng):
    start = time.monotonic()
    deadline = start + float(time_limit)
    conv_size = next_pow2(2 * n - 1)
    t = np.linspace(0.0, 1.0, n, endpoint=True)

    best_p = None
    best_score = float("inf")

    tile_best_q = None
    tile_best_m = None
    tile_best_score = float("inf")

    # Fixed useful starts.
    fixed_profiles = []
    fixed_profiles.append(safe_normalize(np.ones(n, dtype=np.float64)))
    fixed_profiles.append(profile_exp_poly(t, 1.0, 0.0, 0.0))
    fixed_profiles.append(profile_exp_poly(t, 1.2, 0.0, 0.0))
    fixed_profiles.append(profile_exp_poly(t, 1.5, 0.0, 0.0))
    fixed_profiles.append(profile_exp_poly(t, 1.2, 0.6, 0.0))
    fixed_profiles.append(profile_exp_poly(t, 1.2, -0.6, 0.0))
    fixed_profiles.append(profile_exp_poly(t, 0.9, 0.0, 1.2))
    fixed_profiles.append(profile_exp_poly(t, 0.9, 0.0, -1.2))
    fixed_profiles.append(profile_tanh(t, 0.25, 2.0))
    fixed_profiles.append(profile_tanh(t, 0.35, 3.0))
    fixed_profiles.append(profile_tanh(t, 0.45, 5.0))
    fixed_profiles.append(profile_logit(t, 0.35, 6.0))
    fixed_profiles.append(profile_logit(t, 0.45, 12.0))
    fixed_profiles.append(profile_two_level(n, 0.45, 1.9))
    fixed_profiles.append(profile_two_level(n, 0.5, 2.0))
    fixed_profiles.append(profile_two_level(n, 0.35, 2.6))

    for p in fixed_profiles:
        sc = score_norm_size(p, n, conv_size)
        if sc < best_score:
            best_score = sc
            best_p = p.copy()

    # Short wide random-profile search.
    random_deadline = min(deadline, start + max(20.0, 0.15 * float(time_limit)))
    while time.monotonic() < random_deadline:
        r = float(rng.random())
        if r < 0.50:
            a = float(rng.uniform(-4.0, 4.0))
            b = float(rng.uniform(-3.0, 3.0))
            c = float(rng.uniform(-3.0, 3.0))
            p = profile_exp_poly(t, a, b, c)
        elif r < 0.70:
            eps = float(rng.uniform(0.08, 1.0))
            alpha = float(rng.uniform(0.5, 10.0))
            p = profile_tanh(t, eps, alpha)
        elif r < 0.85:
            eps = float(rng.uniform(0.08, 1.0))
            k = float(rng.uniform(2.0, 30.0))
            p = profile_logit(t, eps, k)
        else:
            frac = float(rng.uniform(0.2, 0.8))
            ratio = float(rng.uniform(1.2, 4.0))
            p = profile_two_level(n, frac, ratio)

        sc = score_norm_size(p, n, conv_size)
        if sc < best_score:
            best_score = sc
            best_p = p.copy()

    # Low-resolution piecewise-constant coordinate/annealing search.
    for m in (8, 16, 32):
        if n % m != 0:
            continue
        sub_deadline = min(deadline, time.monotonic() + max(20.0, (deadline - start) / 3.0))

        restarts = [
            np.linspace(0.0, -0.9, m, dtype=np.float64),
            np.linspace(0.0, 0.9, m, dtype=np.float64),
            np.linspace(0.0, -0.45, m, dtype=np.float64),
            rng.normal(0.0, 0.35, m).astype(np.float64),
        ]

        for q0 in restarts:
            q = q0.copy()
            p = make_tile(q, m, n)
            cur_score = score_norm_size(p, n, conv_size)

            if cur_score < best_score:
                best_score = cur_score
                best_p = p.copy()
            if cur_score < tile_best_score:
                tile_best_score = cur_score
                tile_best_q = q.copy()
                tile_best_m = m
                best_p = p.copy()
                best_score = cur_score

            sigma = 0.6
            no_imp = 0
            temp = max(1e-4, 0.008 * cur_score)

            while time.monotonic() < sub_deadline and no_imp < 700:
                for _ in range(8):
                    j = int(rng.integers(m))
                    delta = float(rng.normal(0.0, sigma))
                    q2 = q.copy()
                    q2[j] = float(np.clip(q2[j] + delta, -8.0, 8.0))

                    p2 = make_tile(q2, m, n)
                    sc2 = score_norm_size(p2, n, conv_size)

                    accept = False
                    if sc2 <= cur_score:
                        accept = True
                    else:
                        diff = sc2 - cur_score
                        if diff < temp:
                            logp = -diff / temp
                            if logp > -50.0 and rng.random() < math.exp(logp):
                                accept = True

                    if accept:
                        q = q2
                        cur_score = sc2
                        no_imp = 0
                        sigma = min(1.4, sigma * 1.04)
                        if cur_score < best_score:
                            best_score = cur_score
                            best_p = p2.copy()
                        if cur_score < tile_best_score:
                            tile_best_score = cur_score
                            tile_best_q = q2.copy()
                            tile_best_m = m
                    else:
                        no_imp += 1
                        if no_imp % 12 == 0:
                            sigma = max(0.08, sigma * 0.92)
                            temp = max(1e-5, temp * 0.97)

                if no_imp > 500 and time.monotonic() >= start:
                    break

    return best_p, tile_best_q, tile_best_m


def refine_sequence(p0, time_limit, n, rng):
    start = time.monotonic()
    deadline = start + float(time_limit)

    conv_size = next_pow2(2 * n - 1)
    grad_size = next_pow2(3 * n - 2)
    cross_idx = (2 * n - 2) - np.arange(n, dtype=np.int64)
    idx_float = np.arange(n, dtype=np.float64)

    cur = safe_normalize(np.asarray(p0, dtype=np.float64).copy())
    cur_score = score_norm_size(cur, n, conv_size)

    if not np.isfinite(cur_score):
        cur = safe_normalize(np.ones(n, dtype=np.float64))
        cur_score = score_norm_size(cur, n, conv_size)

    best = cur.copy()
    best_score = cur_score

    eta = 0.28
    gamma = 5.0
    sigma = 0.45
    amp = 0.28
    beta = 0.12
    temp = max(1e-4, 0.0025 * best_score)

    no_imp = 0
    evals = 0
    last_print = start

    while time.monotonic() < deadline:
        if no_imp > 4500:
            q = np.log(best + 1e-300)
            q += rng.normal(0.0, 0.55, size=n)
            cur = safe_normalize(np.exp(np.clip(q - float(np.max(q)), -740.0, 0.0)))
            cur_score = score_norm_size(cur, n, conv_size)
            no_imp = 0
            eta = max(eta, 0.25)
            gamma = 5.0
            temp = max(1e-4, 0.003 * best_score)

        if no_imp > 1200:
            temp = max(1e-5, temp * 0.97)

        cycle = evals % 10
        cand = None
        score2 = float("inf")

        if cycle < 5:
            dir = gradient_direction(cur, gamma, n, grad_size, conv_size, cross_idx)
            if dir is not None:
                q, rel, dir_score = dir
                if np.isfinite(dir_score):
                    cur_score = dir_score
                cand = candidate_from_direction(q, rel, eta)
                score2 = score_norm_size(cand, n, conv_size)
            else:
                cand = block_candidate(cur, sigma, rng)
                score2 = score_norm_size(cand, n, conv_size)

        elif cycle < 7:
            k = 1 if rng.random() < 0.7 else 2
            cand = swap_candidate(cur, k, rng)
            score2 = score_norm_size(cand, n, conv_size)

        elif cycle < 8:
            cand = block_candidate(cur, sigma, rng)
            score2 = score_norm_size(cand, n, conv_size)

        elif cycle < 9:
            cand = sine_candidate(cur, amp, rng, idx_float)
            score2 = score_norm_size(cand, n, conv_size)

        else:
            if best_score > 1.75:
                cand = polarization_candidate(cur, beta, rng, conv_size)
            else:
                cand = block_candidate(cur, sigma * 0.65, rng)
            score2 = score_norm_size(cand, n, conv_size)

        if cand is None or not np.isfinite(score2):
            cand = block_candidate(cur, sigma, rng)
            score2 = score_norm_size(cand, n, conv_size)

        evals += 1
        improved = np.isfinite(score2) and score2 < best_score - 1e-13

        accept = False
        if np.isfinite(score2) and score2 <= cur_score * (1.0 + 1e-12):
            accept = True
        else:
            diff = score2 - cur_score
            if np.isfinite(diff) and diff < temp:
                logp = -diff / temp
                if logp > -50.0 and rng.random() < math.exp(logp):
                    accept = True

        if accept:
            cur = cand
            cur_score = score2
            if improved:
                best = cur.copy()
                best_score = score2
                no_imp = 0
                eta = min(1.2, eta * 1.07)
                sigma = min(1.5, sigma * 1.03)
                amp = min(0.8, amp * 1.04)
                temp = max(1e-4, 0.0022 * best_score)
            else:
                no_imp += 1
        else:
            no_imp += 1
            eta = max(0.04, eta * 0.92)
            sigma = max(0.10, sigma * 0.97)

        if evals % 1000 == 0:
            if no_imp > 350:
                gamma = min(60.0, gamma * 1.25)
            elif no_imp < 60:
                gamma = max(2.0, gamma * 0.85)

        if time.monotonic() - last_print > 30.0:
            print(
                f"refine n={n} evals={evals} best={best_score:.7f} cur={cur_score:.7f} "
                f"eta={eta:.3f} gamma={gamma:.1f} no_imp={no_imp}",
                flush=True,
            )
            last_print = time.monotonic()

    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the given score."""
    budget_s = float(max(60.0, float(budget_s)))
    final_deadline = time.monotonic() + budget_s - 6.0
    rng = np.random.default_rng(int(seed))

    if budget_s < 900.0:
        n_final = 2048
        n_init = 512
        reserve = 60.0
    else:
        n_final = 4096
        n_init = 1024
        reserve = 180.0

    init_time = min(220.0, max(60.0, 0.12 * budget_s))
    print(f"initial n={n_init} final n={n_final} init_time={init_time:.0f}s", flush=True)

    p_init, tile_q, tile_m = tile_random_init(n_init, init_time, rng)

    conv_final = next_pow2(2 * n_final - 1)
    t_final = np.linspace(0.0, 1.0, n_final, endpoint=True)

    best_start = None
    best_start_score = float("inf")

    def consider_candidate(p):
        nonlocal best_start, best_start_score
        sc = score_norm_size(p, n_final, conv_final)
        if np.isfinite(sc) and sc < best_start_score:
            best_start_score = sc
            best_start = p.copy()

    # Built from the init stage.
    consider_candidate(safe_normalize(np.ones(n_final, dtype=np.float64)))
    if p_init is not None:
        consider_candidate(interpolate_to_n(p_init, n_final))
    if tile_q is not None and tile_m is not None:
        consider_candidate(make_tile(tile_q, tile_m, n_final))

    # A few deterministic finals at target length.
    consider_candidate(profile_exp_poly(t_final, 1.0, 0.0, 0.0))
    consider_candidate(profile_exp_poly(t_final, 1.2, 0.0, 0.0))
    consider_candidate(profile_exp_poly(t_final, 1.2, 0.5, 0.0))
    consider_candidate(profile_exp_poly(t_final, 1.2, -0.5, 0.0))
    consider_candidate(profile_exp_poly(t_final, 1.0, 0.0, 1.0))
    consider_candidate(profile_exp_poly(t_final, 1.0, 0.0, -1.0))
    consider_candidate(profile_tanh(t_final, 0.35, 3.0))
    consider_candidate(profile_logit(t_final, 0.40, 8.0))
    consider_candidate(profile_two_level(n_final, 0.5, 2.0))

    # Small random polish of the starting point at final length.
    for _ in range(160):
        a = float(rng.uniform(-4.0, 4.0))
        b = float(rng.uniform(-3.0, 3.0))
        c = float(rng.uniform(-3.0, 3.0))
        consider_candidate(profile_exp_poly(t_final, a, b, c))

    if best_start is None:
        best_start = safe_normalize(np.ones(n_final, dtype=np.float64))
        best_start_score = score_norm_size(best_start, n_final, conv_final)

    print(f"start score={best_start_score:.7f}", flush=True)

    remaining = final_deadline - time.monotonic()
    main_budget = max(60.0, remaining - reserve)
    best_p, best_score = refine_sequence(best_start, main_budget, n_final, rng)

    remaining = final_deadline - time.monotonic()
    if remaining > 40.0:
        extra_budget = max(20.0, remaining - 10.0)
        if best_score < 1.70 and n_final < 8192:
            n2 = n_final * 2
            p2 = interpolate_to_n(best_p, n2)
            sc2 = score_norm_size(p2, n2, next_pow2(2 * n2 - 1))
            if np.isfinite(sc2):
                if sc2 < best_score:
                    best_p = p2
                    best_score = sc2
                best_p, best_score = refine_sequence(best_p, extra_budget, n2, rng)
        else:
            best_p, best_score = refine_sequence(best_p, extra_budget, n_final, rng)

    # Final validation / fallback.
    out = [float(max(0.0, min(1000.0, x))) for x in np.asarray(best_p, dtype=np.float64)]
    s = sum(out)
    if not np.isfinite(s) or s <= 0.0 or len(out) == 0:
        out = [1.0] * max(1, len(out))
    if s < 0.01:
        out = [max(0.0, min(1000.0, 1.0 / max(1, len(out)))) for _ in out]

    try:
        final_note = evaluate_sequence(out)
        print(f"internal_best={best_score:.7f} evaluator={final_note:.7f}", flush=True)
    except Exception:
        try:
            final_note = float("inf")
        except Exception:
            final_note = float("inf")
        print(f"internal_best={best_score:.7f} evaluator failed", flush=True)

    if not np.isfinite(final_note):
        out = [1.0 / max(1, len(out))] * max(1, len(out))

    return out
