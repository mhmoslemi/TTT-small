def make_profile_prob(n, t, eps, gamma=0.0):
    n=int(max(2,n))
    t=float(max(0.0,t)) if np.isfinite(t) else 0.0

def conv_full_fft(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0: return np.empty(0, dtype=np.float64)
    need = 2*n - 1
    N = next_pow2(need)
    fa = np.fft.rfft(a, N)
    out = np.fft.irfft(fa * fa, N)
    return out[:need]

def cross_corr_p_weights(p, w):
   p=np.asarray(p,float); w=np.asarray(w,float)
   n=p.size
   if w.size < 2*n-1:
       w=np.pad(w, (0,(2*n-1)-w.size))
   else:
       w=w[:2*n-1]
   if n==1:
       return np.array([float(p[0]*w[0])])
   t=w[::-1]
   need=n + (2*n-1) - 1
   N=next_pow2(need)
   fp=np.fft.rfft(p,N); ft=np.fft.rfft(t,N)
   conv=np.fft.irfft(fp*ft,N)
   idx=(2*n-2)-np.arange(n,dtype=np.int64)
   h=conv[idx].copy()
   h[h<0]=0.0
   return h

"""
Asymmetric left-singular + uniform profile constructor with active-constraint
convolution smoothing and block-perturbation refinement.
"""

import numpy as np
import time


def next_pow2(m):
    """Smallest power of two >= m."""
    p = 1
    while p < m:
        p *= 2
    return p


def conv_full_fft(a):
    """Full linear convolution via real FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    need = 2 * n - 1
    N = next_pow2(need)
    fa = np.fft.rfft(a, N)
    out = np.fft.irfft(fa * fa, N)
    return out[:need]


def fast_score_array(a):
    """
    Fast FFT-based proxy for evaluate_sequence:
        score = 2*n*max(a*a)/(sum(a)^2)
    with the same nonnegative clipping behavior.
    """
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    n = a.size
    if n == 0:
        return np.inf
    if not np.all(np.isfinite(a)):
        a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    b = a / s
    q = conv_full_fft(b)
    mb = float(np.max(q)) if q.size else 0.0
    if mb < 0.0:
        mb = 0.0
    return float(2.0 * float(n) * mb)


def make_profile_prob(n, t, eps, gamma=0.0):
    """
    Probability-normalized profile:
        p_i propto 1/sqrt(i+eps) + t/sqrt(n) + gamma*(i+1)/(n*sqrt(n)).
    The first term is a left endpoint singularity; the second is a uniform
    floor. The continuous optimum uses t = pi - 2, eps ~ 0.5.
    """
    n = int(max(2, n))
    t = float(t) if np.isfinite(t) else 0.0
    t = max(0.0, t)
    eps = float(eps) if np.isfinite(eps) else 0.5
    eps = max(1e-3, eps)
    gamma = float(gamma) if np.isfinite(gamma) else 0.0

    i = np.arange(n, dtype=np.float64)
    sqrt_n = np.sqrt(float(n))
    a = 1.0 / np.sqrt(i + eps) + t / sqrt_n
    if gamma != 0.0:
        a += gamma * (i + 1.0) / (float(n) * sqrt_n)

    a = np.clip(a, 0.0, None)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    return a / s


def cross_corr_p_weights(p, w):
    """
    Computes h_i = sum_j p[j] * w[i+j] for i=0..n-1,
    where w has length 2n-1.
    """
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    needed = 2 * n - 1
    if w.size < needed:
        w = np.pad(w, (0, needed - w.size), mode="constant")
    else:
        w = w[:needed]

    if n == 1:
        return np.array([float(max(0.0, p[0] * w[0]))], dtype=np.float64)

    t = w[::-1]
    need = n + (2 * n - 1) - 1
    N = next_pow2(need)
    fp = np.fft.rfft(p, N)
    ft = np.fft.rfft(t, N)
    conv = np.fft.irfft(fp * ft, N)

    idx = (2 * n - 2) - np.arange(n, dtype=np.int64)
    h = conv[idx].copy()
    h[h < 0.0] = 0.0
    return h


def smooth_sequence(a, time_limit, seed=0, frac=0.1, eta0=0.25):
    """
    Multinomial-style smoothing step:
    repeatedly reduce entries contributing most to the largest convolution
    values, accepting only improvements.
    """
    start = time.monotonic()
    if time_limit <= 0:
        return a, np.inf

    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 0:
        return a, np.inf
    a = a / s

    best = a.copy()
    best_s = fast_score_array(a)
    if not np.isfinite(best_s):
        return best, best_s

    loga = np.log(np.maximum(a, 1e-300))
    eta = float(eta0)
    frac_cur = float(np.clip(frac, 0.01, 0.5))
    no_improve = 0

    while time.monotonic() - start < time_limit:
        q = conv_full_fft(a)
        if q.size == 0:
            break

        thr = float(np.quantile(q, 1.0 - frac_cur))
        if thr < 0.0:
            thr = 0.0

        w = q - thr
        w[w < 0.0] = 0.0
        wmax = float(np.max(w))
        if not np.isfinite(wmax) or wmax <= 1e-18:
            break

        w = w / wmax
        h = cross_corr_p_weights(a, w)
        grad = a * h
        gmax = float(np.max(grad))
        if not np.isfinite(gmax) or gmax <= 1e-30:
            break

        grad = grad / gmax
        loga_trial = loga - eta * grad
        mx = float(np.max(loga_trial))
        if not np.isfinite(mx):
            break
        loga_trial = loga_trial - mx

        newa = np.exp(loga_trial)
        floor = 1e-300
        newa = np.where(newa > floor, newa, floor)
        ns = float(np.sum(newa))
        if not np.isfinite(ns) or ns <= 0.0:
            break
        newa = newa / ns

        sc = fast_score_array(newa)
        if np.isfinite(sc) and sc < best_s:
            best = newa.copy()
            best_s = sc
            a = newa
            loga = np.log(np.maximum(a, 1e-300))
            no_improve = 0
            eta = min(eta * 1.04, 2.0)
        else:
            no_improve += 1
            eta *= 0.75
            if no_improve % 7 == 0:
                frac_cur *= 1.25
                if frac_cur > 0.5:
                    frac_cur = 0.1
            if no_improve > 120 or eta < 1e-4:
                break

    return best, best_s


def perturb_search(a, time_limit, seed=0):
    """
    Block-perturbation hill climbing using the FFT score.
    """
    start = time.monotonic()
    if time_limit <= 0:
        return a, np.inf

    rng = np.random.default_rng(seed)
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 0:
        return a, np.inf
    a = a / s

    best = a.copy()
    best_s = fast_score_array(a)
    n = int(a.size)

    sigma = 0.06
    point_sigma = 0.18
    stuck = 0

    while time.monotonic() - start < time_limit:
        trial = a.copy()
        nblocks = int(rng.integers(1, 4))

        min_w = 1
        max_w = max(min_w + 1, min(n, int(n * 0.05)))

        for _ in range(nblocks):
            width = int(rng.integers(min_w, max_w + 1))
            hi = max(1, n - width)
            st = int(rng.integers(0, hi))
            factor = float(np.exp(rng.normal(0.0, sigma)))
            trial[st:st + width] *= factor

        if rng.random() < 0.25:
            idx = int(rng.integers(0, n))
            trial[idx] *= float(np.exp(rng.normal(0.0, point_sigma)))

        trial = np.clip(trial, 0.0, 1000.0)
        ts = float(np.sum(trial))
        if not np.isfinite(ts) or ts <= 0.0:
            continue
        trial = trial / ts

        sc = fast_score_array(trial)
        if np.isfinite(sc) and sc < best_s:
            best = trial.copy()
            best_s = sc
            a = trial
            stuck = 0
            sigma = min(sigma * 1.08, 0.55)
        else:
            stuck += 1
            if stuck % 12 == 0:
                sigma = max(0.012, sigma * 0.88)
            if stuck > 400 or sigma < 0.013:
                break

    return best, best_s


def pick_top_indices(candidates, count):
    """Pick indices of the lowest tuple scores without using lambda."""
    chosen = []
    n = len(candidates)
    for _ in range(min(count, n)):
        best_idx = -1
        best_val = np.inf
        for i in range(n):
            if i in chosen:
                continue
            val = candidates[i][0]
            if val < best_val:
                best_val = val
                best_idx = i
        if best_idx != -1:
            chosen.append(best_idx)
    return chosen


def exact_score(a_list):
    """Call the authoritative evaluator safely."""
    try:
        val = evaluate_sequence(a_list)
    except Exception:
        return np.inf
    try:
        return float(val)
    except Exception:
        return np.inf


def optimize_length(n, budget_s, seed=0):
    """
    Optimize a sequence of fixed length n:
    1) coarse/fine parametric scan of the left-singular + uniform profile,
    2) small tilt scan,
    3) reverse-mix scan,
    4) smoothing,
    5) block perturbations.
    """
    start = time.monotonic()
    n = int(max(16, n))
    budget_s = float(max(1e-6, budget_s))

    t0 = float(np.pi - 2.0)
    e0 = 0.5
    best_a = make_profile_prob(n, t0, e0, 0.0)
    best_s = fast_score_array(best_a)
    best_t = t0
    best_e = e0
    best_g = 0.0

    # Coarse parametric scan.
    limit1 = start + min(0.30 * budget_s, 60.0)
    eps_vals = (0.25, 0.5, 0.75, 1.0, 1.5, 2.0)
    t_vals = np.linspace(0.0, 3.0, 26)
    for eps in eps_vals:
        for t in t_vals:
            if time.monotonic() > limit1:
                break
            a = make_profile_prob(n, float(t), float(eps), 0.0)
            sc = fast_score_array(a)
            if np.isfinite(sc) and sc < best_s:
                best_s = sc
                best_a = a
                best_t = float(t)
                best_e = float(eps)
                best_g = 0.0
        if time.monotonic() > limit1:
            break

    # Fine parametric scan around the best found point.
    limit2 = max(limit1, start + min(0.50 * budget_s, 150.0))
    for _ in range(3):
        t_scan = np.linspace(max(0.0, best_t - 0.25), best_t + 0.25, 21)
        e_scan = np.linspace(max(0.05, best_e - 0.35), best_e + 0.35, 15)
        for eps in e_scan:
            for t in t_scan:
                if time.monotonic() > limit2:
                    break
                a = make_profile_prob(n, float(t), float(eps), 0.0)
                sc = fast_score_array(a)
                if np.isfinite(sc) and sc < best_s:
                    best_s = sc
                    best_a = a
                    best_t = float(t)
                    best_e = float(eps)
                    best_g = 0.0
            if time.monotonic() > limit2:
                break

    # Small linear tilt scan.
    limit3 = max(limit2, start + min(0.60 * budget_s, 210.0))
    for g in (-0.2, -0.1, 0.0, 0.1, 0.2, 0.4):
        if time.monotonic() > limit3:
            break
        a = make_profile_prob(n, best_t, best_e, float(g))
        sc = fast_score_array(a)
        if np.isfinite(sc) and sc < best_s:
            best_s = sc
            best_a = a
            best_g = float(g)

    # Moderate reverse mixing can occasionally help finite-length effects.
    limit4 = max(limit3, start + min(0.65 * budget_s, 240.0))
    for rho in (0.03, 0.07, 0.12):
        if time.monotonic() > limit4:
            break
        a = (1.0 - rho) * best_a + rho * best_a[::-1]
        a = np.clip(a, 0.0, None)
        s = float(np.sum(a))
        if s > 0:
            a = a / s
            sc = fast_score_array(a)
            if np.isfinite(sc) and sc < best_s:
                best_s = sc
                best_a = a

    # Smoothing passes.
    elapsed = time.monotonic() - start
    if budget_s - elapsed > 10.0:
        sb = min(0.35 * budget_s, budget_s - elapsed - 5.0)
        if sb > 5.0:
            a2, s2 = smooth_sequence(best_a, sb, seed=seed, frac=0.1, eta0=0.25)
            if np.isfinite(s2) and s2 < best_s:
                best_s = s2
                best_a = a2

            elapsed = time.monotonic() - start
            if budget_s - elapsed > 10.0:
                sb2 = min(0.20 * budget_s, budget_s - elapsed - 5.0)
                if sb2 > 5.0:
                    a3, s3 = smooth_sequence(best_a, sb2, seed=seed + 123, frac=0.05, eta0=0.15)
                    if np.isfinite(s3) and s3 < best_s:
                        best_s = s3
                        best_a = a3

    # Block-perturbation refinement.
    elapsed = time.monotonic() - start
    if budget_s - elapsed > 10.0:
        pb = min(0.25 * budget_s, budget_s - elapsed - 5.0)
        if pb > 5.0:
            a4, s4 = perturb_search(best_a, pb, seed=seed + 999)
            if np.isfinite(s4) and s4 < best_s:
                best_s = s4
                best_a = a4

    return best_a, best_s


def propose_candidate(budget_s=1800, **kwargs):
    """
    Main entry point.  Builds and refines asymmetric profiles of lengths
    1024/2048/4096/8192, validates the best ones exactly when possible,
    and returns the best sequence found.
    """
    start = time.monotonic()
    try:
        tb = float(budget_s)
    except Exception:
        tb = 1800.0
    if not np.isfinite(tb) or tb <= 0.0:
        tb = 1800.0

    if tb < 10.0:
        deadline = start + max(0.2, tb)
    else:
        deadline = start + max(10.0, tb - 5.0)

    # A strong fallback is available immediately.
    fb_n = 4096
    fb_a = make_profile_prob(fb_n, float(np.pi - 2.0), 0.5, 0.0)
    fb_s = fast_score_array(fb_a)
    candidates = [(fb_s, fb_n, fb_a)]
    print("initial profile len", fb_n, "fft score", fb_s)

    # Length sweep.  4096 is the main target; 8192 is optional.
    lengths = [1024, 2048, 4096, 8192]
    budgets = [20.0, 60.0, 180.0, 120.0]

    for n, bud in zip(lengths, budgets):
        if time.monotonic() > deadline - 30.0:
            break
        rem = deadline - time.monotonic()
        bt = min(float(bud), max(10.0, rem - 30.0))
        if bt < 20.0:
            continue
        try:
            a, s = optimize_length(n, bt, seed=1000 + int(n))
        except Exception as e:
            print("optimize_length failed", n, e)
            continue
        if np.isfinite(s):
            candidates.append((s, n, a))
            print("len", n, "fft score", s)

    # Exact validation of the best FFT candidates.
    top = pick_top_indices(candidates, 4)
    exact_valid = []

    for idx in top:
        sff, n, a = candidates[idx]
        if not np.isfinite(sff):
            continue
        if n <= 4096:
            if time.monotonic() > deadline - 25.0:
                break
            sc = exact_score(a.tolist())
            print("exact len", n, "score", sc, "fft", sff)
            if np.isfinite(sc):
                exact_valid.append((sc, n, a))
        elif n <= 8192 and tb > 1200.0:
            if time.monotonic() > deadline - 240.0:
                break
            sc = exact_score(a.tolist())
            print("exact large len", n, "score", sc, "fft", sff)
            if np.isfinite(sc):
                exact_valid.append((sc, n, a))

    # Choose final candidate, preferring slightly smaller lengths when the
    # score difference is negligible.
    chosen_a = None
    chosen_s = np.inf
    chosen_n = 0

    if exact_valid:
        for sc, n, a in exact_valid:
            if not np.isfinite(sc):
                continue
            if sc < chosen_s - 1e-12:
                chosen_s = sc
                chosen_a = a
                chosen_n = n
            elif abs(sc - chosen_s) <= 0.005 and n < chosen_n:
                chosen_s = sc
                chosen_a = a
                chosen_n = n
    else:
        for idx in pick_top_indices(candidates, len(candidates)):
            sc, n, a = candidates[idx]
            if not np.isfinite(sc):
                continue
            if sc < chosen_s - 1e-12:
                chosen_s = sc
                chosen_a = a
                chosen_n = n
            elif abs(sc - chosen_s) <= 0.01 and n < chosen_n:
                chosen_s = sc
                chosen_a = a
                chosen_n = n

    if chosen_a is None:
        chosen_a = fb_a
        chosen_s = fb_s
        chosen_n = fb_n

    # Final refinement on the selected sequence, if time remains.
    rem = deadline - time.monotonic()
    if rem > 30.0:
        refine_budget = min(0.5 * rem, 180.0)
        a2, s2 = smooth_sequence(chosen_a, refine_budget, seed=555, frac=0.1, eta0=0.20)
        if np.isfinite(s2) and s2 < chosen_s:
            chosen_a = a2
            chosen_s = s2

        rem = deadline - time.monotonic()
        if rem > 20.0:
            p_budget = min(0.3 * rem, 120.0)
            a3, s3 = perturb_search(chosen_a, p_budget, seed=777)
            if np.isfinite(s3) and s3 < chosen_s:
                chosen_a = a3
                chosen_s = s3

    # If the final sequence is not too long, revalidate exactly before returning.
    if chosen_n <= 4096:
        rem = deadline - time.monotonic()
        if rem > 30.0:
            sc = exact_score(chosen_a.tolist())
            print("final exact len", chosen_n, "score", sc)
            if np.isfinite(sc):
                chosen_s = sc
        elif rem > 10.0 and chosen_n <= 2048:
            sc = exact_score(chosen_a.tolist())
            print("final exact short len", chosen_n, "score", sc)
            if np.isfinite(sc):
                chosen_s = sc

    print("chosen len", chosen_n, "return score estimate", chosen_s)
    return [float(x) for x in np.asarray(chosen_a, dtype=np.float64)]
