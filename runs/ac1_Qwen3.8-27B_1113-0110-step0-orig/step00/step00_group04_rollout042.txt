```python
import numpy as np
import time
from scipy.optimize import linprog

def lp_iteration(f, M):
    """Solve the LP to find g that maximizes sum(g) subject to convolve(f,g) <= M, g >= 0."""
    n = len(f)
    # Variables: g[0], ..., g[n-1]
    # Constraints: for each k from 0 to 2n-2: sum_i f[i]*g[k-i] <= M
    #             g[i] >= 0 (handled by bounds)
    # Objective: maximize sum(g) => minimize -sum(g)
    
    c = -np.ones(n)  # minimize -sum(g)
    
    # Build constraint matrix A such that A @ g <= M * ones
    # A[k, i] = f[i] if 0 <= k-i <= n-1, else 0
    # This is a banded matrix with bandwidth n
    
    # For efficiency, we can use the structure
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        A[k, i_min:i_max+1] = f[i_min:i_max+1]
    
    # This is O(n^2) memory and time to build, which is fine for n=3809
    # But for larger n, this could be an issue
    
    res = linprog(c, A_ub=A, b_ub=M*np.ones(2*n-1), bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    else:
        return None
```

import numpy as np

from scipy.sparse import lil_matrix, csr_matrix


def full_conv(a):
    """Compute full convolution a*a using FFT."""
    return np.real(fftconvolve(a, a, mode='full'))


def score_fn(a):
    """Compute the actual score (for tracking, not for optimization)."""
    n = len(a)
    c = full_conv(a)
    max_b = c.max()
    S = a.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * max_b / (S * S)


def soft_max_gradient(a, temperature):
    """Compute gradient of soft-max of convolution with respect to a."""
    n = len(a)
    c = full_conv(a)
    c_max = c.max()
    c_shifted = c - c_max
    w = np.exp(c_shifted / temperature)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones_like(c) / len(c)
    else:
        w = w / w_sum
    # g[j] = 2 * sum_m w[j+m] * a[m] = 2 * correlate(w, a, 'valid')[j]
    g = 2.0 * np.correlate(w, a, mode='valid')
    return g, c_max, w

def subgradient_method(a_init, max_time, alpha=0.15, top_k=3):
    """Subgradient method: reduce the top-k peaks of the convolution."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / S

    best_score = float(evaluate_sequence(a.tolist()))
    best_a = a.copy()
    start_time = time.monotonic()
    it = 0

    # Running average for convergence
    avg_a = a.copy()
    T = 0

    while time.monotonic() - start_time < max_time:
        c = full_conv(a)
        # Get top-k positions
        k_indices = np.argsort(c)[::-1][:top_k]
        k_weights = c[k_indices]
        k_weights = k_weights / k_weights.sum()

        # Combined subgradient
        g = np.zeros(n)
        for wi, k in zip(k_weights, k_indices):
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            if j_max >= j_min:
                idx = k - np.arange(j_min, j_max + 1)
                g[j_min:j_max + 1] += wi * 2.0 * a[idx]

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        step = alpha * g / g_norm
        a_new = a - step
        a_new = np.clip(a_new, 1e-6, 1000.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        a = a_new

        it += 1
        T += 1
        avg_a += (a - avg_a) / T

        if it % 50 == 0:
            # Check both current and average
            for candidate in [a, avg_a / avg_a.sum()]:
                sc = float(evaluate_sequence(candidate.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = candidate.copy()
                    print(f"  SG it={it}, score={best_score:.6f}")

    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_a = a.copy()

    avg_candidate = avg_a / avg_a.sum()
    score = float(evaluate_sequence(avg_candidate.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_a = avg_candidate.copy()

    return best_a, best_score


def lp_refine(f, max_time, max_iters=30):
    """LP-based Frank-Wolfe refinement."""
    n = len(f)
    if n > 5000:
        return f.copy(), float(evaluate_sequence(f.tolist()))

    f = f.copy().astype(np.float64)
    S = f.sum()
    f = f / S

    best_a = f.copy()
    best_score = float(evaluate_sequence(f.tolist()))
    start_time = time.monotonic()

    for iteration in range(max_iters):
        if time.monotonic() - start_time > max_time:
            break

        c = full_conv(f)
        M = c.max()

        # Build constraint matrix (sparse)
        A = lil_matrix((2 * n - 1, n), dtype=np.float64)
        for k in range(2 * n - 1):
            i_min = max(0, k - n + 1)
            i_max = min(k, n - 1)
            if i_max >= i_min:
                A[k, i_min:i_max + 1] = f[i_min:i_max + 1]
        A = A.tocsr()

        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n, method='highs')
        except Exception:
            break

        if not res.success:
            break

        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-10:
            break

        g = g / g_sum  # normalize to sum 1

        # Frank-Wolfe step
        t_step = 2.0 / (iteration + 3)
        f_new = (1 - t_step) * f + t_step * g
        f_new = np.clip(f_new, 1e-8, 1000.0)
        f_new = f_new / f_new.sum()

        score = float(evaluate_sequence(f_new.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = f_new.copy()
            print(f"  LP iter={iteration}, score={best_score:.6f}")

        f = f_new

    return best_a, best_score


def make_ramp(n, direction=1, exponent=1.0, floor=0.0):
    """Create a ramp-like starting sequence."""
    t = np.linspace(0, 1, n)
    if direction == 1:
        a = t ** exponent
    elif direction == -1:
        a = (1 - t) ** exponent
    else:
        a = 1.0 + (2 * t - 1) ** 2
    a = np.maximum(a, floor)
    s = a.sum()
    if s > 0:
        a = a / s
    return a


def make_random_seq(n, rng):
    """Create a random non-negative sequence."""
    a = rng.random(n) ** 2  # skew toward small values
    s = a.sum()
    if s > 0:
        a = a / s
    return a


def make_banded(n, rng, n_bands=5):
    """Create a sequence with a few bands of different heights."""
    a = np.zeros(n)
    band_size = n // n_bands
    for i in range(n_bands):
        h = rng.uniform(0.1, 3.0)
        a[i * band_size:(i + 1) * band_size] = h
    # Add some randomness
    a *= (1.0 + 0.3 * rng.standard_normal(n))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a = a / s
    return a


def make_asymmetric_gaussian(n, center_frac=0.3, width=0.2):
    """Gaussian centered off-center."""
    t = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((t - center_frac) / width) ** 2)
    a += 0.1 * np.ones(n)  # avoid zeros
    s = a.sum()
    a = a / s
    return a

        if time.monotonic() >= deadline - 200:

        t_alloc = min(gd_per, deadline - time.monotonic() - 200)
        if t_alloc < 15:
            break
        a_best, sc2 = gradient_descent(a_init, t_alloc, temperature=3.0, lr=0.4)
        if sc2 < best_score:
            best_score = sc2
            best_seq = a_best.copy()
            print(f"  GD refine: {sc2:.6f}")

    # Phase 3: Try smaller n values with fresh starts
    print(f"Phase 3: Smaller sequence lengths")
    for n_alt in n_values[1:]:
        if time.monotonic() >= deadline - 150:
            break
        t_remaining = deadline - time.monotonic() - 150
        if t_remaining < 60:
            break

        # Try a ramp and a random for this length
        for init_name, a_init in [('ramp', make_ramp(n_alt, 1, 1.0)),
                                  ('g', make_asymmetric_gaussian(n_alt, 0.35, 0.2)),
                                  ('rand', make_random_seq(n_alt, rng))]:
            if time.monotonic() >= deadline - 150:
                break
            t_alloc = min(60.0, (deadline - time.monotonic() - 150) / 3)
            if t_alloc < 10:
                break
            a_best, sc = gradient_descent(a_init, t_alloc, temperature=2.0, lr=0.4)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_best.copy()
                print(f"  [n={n_alt}, {init_name}] GD: {sc:.6f}")

    # Phase 4: LP refinement on the best solution
    print(f"Phase 4: LP refinement")
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_lp = len(best_seq)
        if n_lp <= 5000:
            t_lp = min(300.0, deadline - time.monotonic() - 30)
            a_best, sc = lp_refine(best_seq, t_lp, max_iters=20)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_best.copy()
                print(f"  LP final: {sc:.6f}")

    # Phase 5: Final local search (small perturbations)
    print(f"Phase 5: Local refinement")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a = best_seq.copy()
        n = len(a)
        t_remaining = deadline - time.monotonic() - 10
        it = 0
        local_per = min(3.0, t_remaining / 1000)

        while time.monotonic() < deadline - 10:
            # Random block perturbation
            width = int(rng.integers(max(1, n // 100), max(2, n // 10)))
            start = int(rng.integers(0, max(1, n - width)))
            stop = min(n, start + width)
            a_candidate = a.copy()
            a_candidate[start:stop] *= np.exp(rng.normal(0, 0.05, stop - start))
            a_candidate = np.clip(a_candidate, 0, 1000)
            s = a_candidate.sum()
            if s > 0.01:
                a_candidate = a_candidate / s
                sc = float(evaluate_sequence(a_candidate.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_candidate.copy()
                    a = a_candidate.copy()
                    print(f"  Local it={it}: {best_score:.6f}")
            it += 1

    # Ensure we return a valid sequence
    if best_seq is None:
        # Fallback: constant sequence
        best_seq = np.ones(n_main) / n_main
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final validation
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if not np.isfinite(final_score) or final_score > 2.0:
        # Use the constant sequence as safe fallback
        best_seq = np.ones(n_main) / n_main
        final_score = 2.0

    print(f"Final score: {final_score:.6f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
