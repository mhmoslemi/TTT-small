def build_conv_matrix(f, n):
    # C[k,j] = f[k-j] if 0<=k-j<n else 0

    # Actually: row k has f[0], f[1], ..., f[k] in columns k, k-1, ..., 0 (for k<n)
    #           and f[k-n+1], ..., f[n-1] in columns n-1, ..., k-n+1 (for k>=n)

    i = np.arange(2*n-1)[:, None]  # column vector (2n-1, 1)
    j = np.arange(n)[None, :]       # row vector (1, n)
    idx = i - j  # (2n-1, n), idx[k,j] = k-j
    valid = (idx >= 0) & (idx < n)
    C = np.where(valid, f[np.minimum(np.maximum(idx, 0), n-1)], 0.0)
    return C

from scipy.optimize import linprog

def compute_score_fast(a):
    """Compute the score without calling evaluate_sequence."""
    n = len(a)

    if S < 0.01:

    c = np.convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def build_conv_matrix(f, n):
    """Build the (2n-1) x n matrix C where (C @ b)[k] = (f*b)[k]."""
    # C[k,j] = f[k-j] if 0 <= k-j < n, else 0
    rows = np.arange(2*n-1)[:, None]  # (2n-1, 1)
    cols = np.arange(n)[None, :]       # (1, n)
    idx = rows - cols                  # (2n-1, n)
    C = np.where((idx >= 0) & (idx < n), f[idx.clip(0, n-1)], 0.0)
    return C

def solve_lp_step(f, M, n):
    """
    Solve: max sum(b) s.t. (f*b)[k] <= M for all k, b >= 0, b <= 2*max(f)
    Returns b (normalized to sum 1) or None.
    """
    C = build_conv_matrix(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    upper = 2.0 * np.max(f)
    bounds = [(0.0, upper)] * n
    
    res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status != 0:
        return None
    b = res.x
    s = np.sum(b)
    if s < 1e-15:
        return None
    b /= s
    return b
