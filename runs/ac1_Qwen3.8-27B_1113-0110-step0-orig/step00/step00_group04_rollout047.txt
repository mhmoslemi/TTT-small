import time

    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):

        j_hi = min(n - 1, k)

                      method='highs', options={'time_limit': timeout, 'presolve': True})

    conv = fftconvolve(a, a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (S * S)


def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
