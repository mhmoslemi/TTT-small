def compute_conv_fft(a, n_fft):
    """Compute linear convolution of a with itself using FFT."""
    A = np.fft.rfft(a, n_fft)
    C = np.fft.irfft(A * A, n_fft)
    return C[:2*len(a)-1]

def compute_gradient(a, p, n_fft):
    """
    Compute gradient of smoothed max.
    grad[i] = 2 * sum_j a[j] * p[i+j] for i=0,...,n-1
    = 2 * conv(a_flipped, p)[i+n-1]
    where a_flipped = a[::-1], p has length 2n-1
    """
    n = len(a)
    a_flipped = a[::-1]
    # conv(a_flipped, p) has length n + (2n-1) - 1 = 3n-2
    # We need elements at indices n-1 to 2n-2
    N = n_fft  # should be >= 3n-2
    A = np.fft.rfft(a_flipped, N)
    P = np.fft.rfft(p, N)
    conv_result = np.fft.irfft(A * P, N)
    return 2.0 * conv_result[n-1:2*n-1]

def smoothed_gradient_step(a, beta, lr):
    """One step of smoothed gradient descent. Returns (new_a, info)."""
    n = len(a)
    n_fft = 1
    while n_fft < 2*n - 1:
        n_fft *= 2
    
    # Compute C = conv(a, a)
    A = np.fft.rfft(a, n_fft)
    C = np.fft.irfft(A * A, n_fft)[:2*n-1]
    
    # Compute p = softmax(beta * C)
    C_shifted = C - np.max(C)
    exp_C = np.exp(beta * C_shifted)
    p = exp_C / np.sum(exp_C)
    
    # Compute gradient
    N_grad = 1
    while N_grad < 3*n - 2:
        N_grad *= 2
    a_flipped = a[::-1]
    A_f = np.fft.rfft(a_flipped, N_grad)
    P = np.fft.rfft(p, N_grad)
    conv_result = np.fft.irfft(A_f * P, N_grad)
    grad = 2.0 * conv_result[n-1:2*n-1]
    
    # Update

    a_new = np.maximum(a_new, 0.0)

    s = np.sum(a_new)
    if s < 1e-15:
        a_new = np.ones(n) / n
    else:
        a_new = a_new / s
    
    return a_new, C
