import numpy as np
import time


# ---------------------------------------------------------------------------
#  Optimization of a non-negative step function to minimize
#      score = 2 * n * max_k (a*a)[k] / (sum(a))^2
#  Strategy:
#    - Work on the probability simplex (scale invariance of the score).
#    - Use an FFT-based *smoothed* gradient (log-sum-exp of the convolution)
#      for a smooth search direction, blended with the hard subgradient at
#      the argmax lag for sharpness.
#    - Momentum + adaptive (per-batch) step size, simplex projection each step.
#    - Multiple structured starting points and periodic random kicks to escape
#      local minima.  The exact evaluate_sequence() is used to confirm/update
#      the best valid answer.
# ---------------------------------------------------------------------------


def next_pow2(x):
    n = 1
    while n < x:
        n *= 2
    return n


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} (Euclidean projection)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho_arr = np.nonzero(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n, dtype=np.float64) / n
    rho = int(rho_arr[-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def auto_convolve(p, N):
    """Full convolution p*p (length 2n-1) via FFT of size N (N >= 2n-1)."""
    P = np.fft.rfft(p, N)
    return np.fft.irfft(P * P, N)[:2 * len(p) - 1]


def fast_score(p, N):
    """Fast version of the score assuming we keep sum(p)=1: 2n*max(conv)."""
    n = len(p)
    s = p.sum()
    if s < 1e-15:
        return np.inf
    c = auto_convolve(p, N)
    return 2.0 * n * c.max() / (s * s)


def hard_subgradient(p, k_star):
    """Subgradient of (p*p)[k_star] w.r.t. p."""
    n = len(p)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    j = np.arange(lo, hi + 1)
    g = np.zeros(n, dtype=np.float64)
    g[j] = p[k_star - j] + p[j]
    return g


def smoothed_gradient(p, tau, N):
    """
    Gradient of f_tau(p) = (1/tau)*log(sum_k exp(tau*(p*p)[k])).
    Returns (grad, M, k_star) where M = max conv and k_star = argmax lag.
    Uses O(N log N) FFTs only.
    """
    n = len(p)
    c = auto_convolve(p, N)
    k_star = int(np.argmax(c))
    c_max = c[k_star]
    shifted = np.clip(tau * (c - c_max), -500.0, 500.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w = np.zeros(2 * n - 1, dtype=np.float64)
        w[k_star] = 1.0
        w_sum = 1.0
    w /= w_sum

    # T[j] = sum_i p[i]*w[i+j]  ==  (flip(p) * w)[n-1+j]
    PF = np.fft.rfft(p[::-1], N)
    WF = np.fft.rfft(w, N)
    T_full = np.fft.irfft(PF * WF, N)
    T = T_full[n - 1:2 * n - 1]

    # W[j] = sum_{k=j}^{min(2n-2, j+n-1)} w[k]  via prefix sums
    prefix = np.concatenate([[0.0], np.cumsum(w)])
    j_arr = np.arange(n)
    hi_idx = np.minimum(2 * n - 1, j_arr + n)
    W_vec = prefix[hi_idx] - prefix[j_arr]

    grad = T + p * W_vec
    return grad, c_max, k_star


def make_starts(n, parent):
    """Build a list of candidate starting points (each on the simplex)."""
    starts = []
    if parent is not None:
        pp = np.asarray(parent, dtype=np.float64)
        pp = np.clip(pp, 0.0, None)
        if pp.size == n and pp.sum() > 0:
            starts.append(pp / pp.sum())

    x = np.linspace(-1.0, 1.0, n)
    for a in (0.15, 0.35, 0.6, 0.85):
        p = np.clip(1.0 + a * x, 0.0, None)
        if p.sum() > 0:
            starts.append(p / p.sum())
    for a in (0.4, 0.8):
        p = 1.0 + a * x * x
        starts.append(p / p.sum())
    # Gaussian pair
    c1, c2 = 0.30, 0.70
    sig = 0.12
    t = np.linspace(0.0, 1.0, n)
    p = np.exp(-((t - c1) ** 2) / (2 * sig * sig)) + np.exp(-((t - c2) ** 2) / (2 * sig * sig))
    starts.append(p / p.sum())
    # uniform
    starts.append(np.ones(n, dtype=np.float64) / n)
    return starts


def run_optimizer(p0, N, tau_lo, tau_hi, eta, beta, rng, time_budget):
    """
    Inner optimizer on one starting point.
    tau is annealed from tau_lo (smooth) to tau_hi (sharp).
    Returns (best_p, best_score, prev_tau, prev_eta) for warm restart.
    """
    n = len(p0)
    p = np.asarray(p0, dtype=np.float64).copy()
    p = project_simplex(p)

    best_p = p.copy()
    best_score = fast_score(p, N)

    m = np.zeros(n, dtype=np.float64)
    deadline = time.monotonic() + time_budget

    n_batches_est = max(1, int(time_budget / 0.35))  # ~ one batch per 0.35s
    B = 1500
    batch = 0
    no_improve = 0

    while time.monotonic() < deadline:
        batch += 1
        frac = min(1.0, batch / float(n_batches_est + 1))
        # log-linear tau anneal
        tau = tau_lo * (tau_hi / tau_lo) ** frac

        # blend coefficient toward hard subgradient grows with tau sharpness
        for _ in range(B):
            if time.monotonic() > deadline:
                break
            g, c_max, k_star = smoothed_gradient(p, tau, N)
            # mix with hard subgradient for extra sharpness later on
            if tau * c_max > 40.0:
                gh = hard_subgradient(p, k_star)
                mix = min(1.0, (tau * c_max - 40.0) / 60.0)
                g = (1.0 - mix) * g + mix * gh
            m = beta * m - eta * g
            p = project_simplex(p + m)

        # occasional random kick to escape local minima
        if batch % 4 == 0:
            kick = np.exp(rng.normal(0.0, 0.35, size=n))
            p = project_simplex(0.5 * p + 0.5 * kick / kick.sum())
            m *= 0.3

        sc = fast_score(p, N)
        if sc < best_score - 1e-12:
            best_score = sc
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 2:
                eta = max(0.02, eta * 0.7)
                m *= 0.2
                no_improve = 0

    return best_p, best_score
