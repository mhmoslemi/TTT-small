    # 2. Cosine bump (concentrated in middle)
    for n_try in [4096, 8192, 16384, 32768]:
        j = np.arange(n_try)
        f = 0.5 * (1.0 + np.cos(2.0 * np.pi * j / n_try))
        f = np.maximum(f, 0.0)
        if f.sum() > 0:
            inits.append(f / f.sum())
            labels.append(f"cos_bump_{n_try}")

    # 3. U-shape (concentrated at edges)
    for n_try in [4096, 8192, 16384]:
        j = np.arange(n_try)
        f = 0.5 * (1.0 - np.cos(2.0 * np.pi * j / n_try))
        f = np.maximum(f, 0.0)
        if f.sum() > 0:
            inits.append(f / f.sum())
            labels.append(f"u_shape_{n_try}")

    # 4. Random (Dirichlet with various concentrations)
    for n_try, conc in [(4096, 5.0), (8192, 5.0), (4096, 0.5), (8192, 50.0)]:
        f = rng.dirichlet(np.full(n_try, conc))
        inits.append(f)
        labels.append(f"random_{n_try}_{int(conc)}")

    # 5. Power law decay
    for n_try in [4096, 8192]:
        j = np.arange(1, n_try + 1)
        f = j ** (-1.2)
        inits.append(f / f.sum())
        labels.append(f"power_{n_try}")

    # 6. Two-hump
    for n_try in [4096, 8192]:
        j = np.arange(n_try)
        f = np.exp(-((j - n_try * 0.25) / (n_try * 0.05)) ** 2) + \
            np.exp(-((j - n_try * 0.75) / (n_try * 0.05)) ** 2)
        f = np.maximum(f, 0.0)
        if f.sum() > 0:
            inits.append(f / f.sum())
            labels.append(f"two_hump_{n_try}")

    # 7. Linear ramp
    for n_try in [4096, 8192]:
        j = np.arange(n_try, dtype=np.float64)
        f = j + 1.0
        inits.append(f / f.sum())
        labels.append(f"ramp_{n_try}")

    print(f"Total initializations: {len(inits)}")
    print(f"Budget: {budget_s}s, deadline in {deadline - time.monotonic():.0f}s")

    # Allocate time: give more time to promising initializations
    # Sort by a quick heuristic score
    init_scores = []
    for f_init in inits:
        init_scores.append(_evaluate_fast(f_init))
    sorted_indices = np.argsort(init_scores)

    # Time allocation: proportional to remaining time, with minimum guarantees
    total_time = budget_s - 60
    min_per_init = 20
    max_per_init = 400

    for rank, idx in enumerate(sorted_indices):
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        # Earlier (better heuristic) gets more time
        frac = max(0.5, 1.0 - rank / (len(inits) * 0.5))
        alloc = min(max_per_init, max(min_per_init, total_time * frac / max(1, len(inits) - rank)))
        alloc = min(alloc, remaining - 30)  # leave time for later inits + polish
        if alloc < 10:
            break

        t0 = time.monotonic()
        best_f, best_score = _run_pressure_search(inits[idx], alloc, rng)
        elapsed = time.monotonic() - t0

        if best_score < best_overall_score:
            best_overall_score = best_score
            best_overall = best_f.copy()

        print(f"[{labels[idx]}] n={len(inits[idx])}, score={best_score:.6f}, "
              f"elapsed={elapsed:.1f}s, best_overall={best_overall_score:.6f}")

    # LP polish on the best result
    if best_overall is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            polish_time = min(remaining - 10, 300)
            print(f"\nStarting LP polish with {polish_time:.0f}s budget...")
            t0 = time.monotonic()
            best_f2, best_score2 = _lp_polish(best_overall, polish_time)
            elapsed = time.monotonic() - t0
            if best_score2 < best_overall_score:
                best_overall_score = best_score2
                best_overall = best_f2
                print(f"[LP polish] Improved! score={best_score2:.6f} in {elapsed:.1f}s")
            else:
                print(f"[LP polish] No improvement. score={best_score2:.6f} in {elapsed:.1f}s")

    # If we still have time, do another round of short searches from the best
    if best_overall is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            # Try perturbing the best and running a short search
            f_pert = best_overall + 0.01 * rng.standard_normal(len(best_overall))
            f_pert = np.maximum(f_pert, 0)
            f_pert /= f_pert.sum()
            best_f3, best_score3 = _run_pressure_search(f_pert, min(remaining - 5, 60), rng)
            if best_score3 < best_overall_score:
                best_overall_score = best_score3
                best_overall = best_f3
                print(f"[Final perturbation] Improved! score={best_score3:.6f}")

    # Fallback
    if best_overall is None:
        best_overall = np.ones(4096) / 4096

    # Ensure non-negativity and finite values
    best_overall = np.clip(best_overall, 0.0, 1000.0)
    final = [float(x) for x in best_overall]

    # Verify
    final_score = float(evaluate_sequence(final))
    print(f"\nFinal verified score: {final_score:.6f}")
    print(f"Sequence length: {len(final)}, sum: {np.sum(best_overall):.4f}")

    return final
