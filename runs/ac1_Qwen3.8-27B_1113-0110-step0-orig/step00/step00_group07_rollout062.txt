"""
Optimizes nonnegative sequence heights for

    2 * n * max(convolve(a, a)) / (sum(a)^2)

using:
- fast FFT-based convolutions,
- smooth-max gradient ascent in log-space,
- several random structured restarts,
- an LP-inspired mixing polish for moderate lengths.
"""

import numpy as np
import time
import math

try:
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    HAVE_LINPROG = True
except Exception:
    HAVE_LINPROG = False


def next_pow2(m):
    n = 1
    m = max(1, int(m))
    while n < m:
        n <<= 1
    return n


def conv_self_fft(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    L = 2 * n - 1
    N = next_pow2(L)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:L]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    return c


def normalize_positive(a):
    a = np.asarray(a, dtype=np.float64).copy()
    if a.size == 0:
        return a
    a = np.where(np.isfinite(a) & (a > 0.0), a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    a /= s
    return a


def fast_score_normalized(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.inf
    c = conv_self_fft(a)
    m = float(c.max()) if c.size else 0.0
    if not np.isfinite(m) or m < 0.0:
        return np.inf
    return 2.0 * float(n) * m


def exact_score_array(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.inf
    a = np.where(np.isfinite(a) & (a > 0.0), a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.inf
    c = np.convolve(a, a)
    m = float(c.max()) if c.size else 0.0
    if not np.isfinite(m) or m < 0.0:
        return np.inf
    return 2.0 * float(a.size) * m / (s * s)


def central_sequence(n):
    n = int(n)
    if n <= 0:
        return np.ones(1, dtype=np.float64)
    b = np.ones(n, dtype=np.float64)
    for k in range(1, n):
        b[k] = b[k - 1] * ((2.0 * k - 1.0) / (2.0 * k))
    s = b.sum()
    if s > 0:
        b /= s
    else:
        b[:] = 1.0 / n
    return b


def smooth_max_grad(a, beta):
    """Return smooth max, gradient wrt a, and exact max of self-convolution."""
    n = a.size
    L = 2 * n - 1
    c = conv_self_fft(a)
    cmax = float(c.max()) if c.size else 0.0
    if not np.isfinite(cmax):
        cmax = 0.0
        c[:] = 0.0

    if beta <= 0.0 or cmax <= 0.0:
        w = np.zeros(L, dtype=np.float64)
        if c.size:
            w[int(np.argmax(c))] = 1.0
        val = cmax
    else:
        z = beta * (c - cmax)
        expz = np.exp(z)
        expz = np.where(np.isfinite(expz), expz, 0.0)
        sumexp = float(expz.sum())
        if sumexp <= 0.0 or not np.isfinite(sumexp):
            w = np.zeros(L, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
            val = cmax
        else:
            w = expz / sumexp
            val = cmax + math.log(sumexp) / beta

    if n == 0:
        return val, np.zeros(0, dtype=np.float64), cmax

    # P[m] = sum_j a[j] * w[m + j]
    Nc = next_pow2(n + L - 1)
    ar = a[::-1]
    conv = np.fft.irfft(np.fft.rfft(ar, Nc) * np.fft.rfft(w, Nc), Nc)
    P = conv[n - 1:n - 1 + n]
    P = np.where(np.isfinite(P), P, 0.0)

    # d(conv_k)/d(a_m) = 2 a_{k-m} if valid
    grad = 2.0 * P
    return float(val), grad, cmax


def adam_optimize(a0, time_limit, beta_start=0.0, beta_end=0.0,
                  lr_start=0.0, lr_end=0.015, seed=0):
    a = normalize_positive(a0)
    n = a.size
    if n == 0 or time_limit <= 0.0:
        c = conv_self_fft(a)
        s = 2.0 * n * float(c.max()) if c.size else np.inf
        return a, s

    end_time = time.time() + float(time_limit)

    if beta_start <= 0.0:
        beta_start = max(80.0, min(30000.0, 8.0 * float(n)))
    if beta_end <= 0.0:
        beta_end = max(2000.0, min(60000.0, beta_start * 6.0))
    if lr_start <= 0.0:
        lr_start = max(0.10, min(0.45, 0.15 + 120.0 / float(n)))

    x = np.log(np.maximum(a, 1e-300))
    x -= x.max()
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    pow_b1 = 1.0
    pow_b2 = 1.0
    beta = float(beta_start)
    t = 0
    stagnant = 0

    c0 = conv_self_fft(a)
    best_score = 2.0 * float(n) * float(c0.max())
    best = a.copy()

    rng = np.random.default_rng(int(seed))
    eval_every = 4

    while time.time() < end_time - 0.2:
        t += 1
        x = x - x.max()
        a_work = np.exp(x)
        a_sum = float(a_work.sum())
        if (not np.isfinite(a_sum)) or a_sum <= 1e-300:
            x = rng.normal(size=n)
            x -= x.max()
            continue
        a_work = a_work / a_sum

        val, grad, cmax = smooth_max_grad(a_work, beta)
        if (not np.isfinite(val)) or (not np.all(np.isfinite(grad))):
            x = rng.normal(size=n)
            x -= x.max()
            continue

        sdot = float(np.dot(a_work, grad))
        gx = a_work * (grad - sdot)

        if not np.all(np.isfinite(gx)):
            gx = np.where(np.isfinite(gx), gx, 0.0)

        g_peak = float(np.max(np.abs(gx))) if gx.size else 0.0
        if g_peak < 1e-18:
            break

        pow_b1 *= 0.9
        pow_b2 *= 0.999
        one_minus_b1 = 1.0 - pow_b1
        one_minus_b2 = 1.0 - pow_b2
        m = 0.9 * m + 0.1 * gx
        v = 0.999 * v + 0.001 * (gx * gx)
        mhat = m / one_minus_b1
        vhat = v / one_minus_b2
        step_lr = max(float(lr_end), float(lr_start) * math.exp(-t / 9000.0))
        x -= step_lr * mhat / (np.sqrt(vhat) + 1e-12)

        if t % eval_every == 0 or t < 16:
            score = 2.0 * float(n) * cmax
            if np.isfinite(score) and score < best_score - 1e-12:
                best_score = float(score)
                best = a_work.copy()
                stagnant = 0
            else:
                stagnant += 1

        if t % 2000 == 0:
            beta = min(float(beta_end), beta * 1.35)
            if stagnant > 8000:
                x += rng.normal(0.0, 0.18, n)
                x -= x.max()
                stagnant = 0

    a_best = normalize_positive(best)
    c_best = conv_self_fft(a_best)
    score_best = 2.0 * float(n) * float(c_best.max()) if c_best.size else np.inf
    return a_best, float(score_best)


def generate_random_starts(n, rng, count, base):
    base = normalize_positive(base)
    starts = [base.copy()]
    idx = np.arange(n, dtype=np.float64)
    two_pi = 2.0 * math.pi

    for i in range(int(count)):
        kind = i % 4
        if kind == 0:
            alpha = float(rng.uniform(0.20, 0.70))
            a = (idx + 1.0) ** (-alpha)
        elif kind == 1:
            amp = float(rng.uniform(0.0, 1.6))
            ph = float(rng.uniform(0.0, two_pi))
            a = base * (1.0 + amp * np.sin(two_pi * idx / float(n) + ph))
        elif kind == 2:
            a = base.copy()
            center = int(np.floor(float(n) * float(rng.uniform(0.42, 0.95))))
            width = max(1.0, float(n) * float(rng.uniform(0.04, 0.25)))
            amp = float(rng.uniform(0.15, 2.2))
            if 0 <= center < n:
                t = idx - float(center)
                a = a + amp * np.exp(-(t * t) / (2.0 * width * width))
        else:
            c1 = float(rng.uniform(-0.55, 0.55))
            c2 = float(rng.uniform(-0.55, 0.55))
            p1 = float(rng.uniform(0.0, two_pi))
            p2 = float(rng.uniform(0.0, two_pi))
            alpha = float(rng.uniform(0.20, 0.65))
            a_log = -alpha * np.log(idx + 1.0)
            a_log += c1 * np.cos(two_pi * idx / float(n) + p1)
            a_log += c2 * np.cos(4.0 * math.pi * idx / float(n) + p2)
            a = np.exp(a_log - np.max(a_log))

        a = np.where(np.isfinite(a) & (a > 0.0), a, 0.0)
        s = float(a.sum())
        if not np.isfinite(s) or s <= 1e-300:
            a = base.copy()
        else:
            a = a / s
        starts.append(a)

    return starts


def perturb_sequence(a, rng, sigma):
    a = normalize_positive(a)
    x = np.log(np.maximum(a, 1e-300))
    x -= x.max()
    x += rng.normal(0.0, float(sigma), a.size)
    x -= x.max()
    b = np.exp(x)
    s = float(b.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return a
    b /= s
    return b


def downsample_sequence(seq, n):
    seq = np.asarray(seq, dtype=np.float64)
    seq = np.where(np.isfinite(seq) & (seq > 0.0), seq, 0.0)
    if seq.size == 0:
        return np.ones(n, dtype=np.float64)
    if seq.size == n:
        return normalize_positive(seq)
    if seq.size < n:
        if seq.size == 1:
            return normalize_positive(np.full(n, float(seq[0])))
        xs = np.linspace(0.0, float(seq.size - 1), n)
        arr = np.interp(xs, np.arange(seq.size, dtype=np.float64), seq)
    else:
        arr = np.zeros(n, dtype=np.float64)
        edges = np.linspace(0, seq.size, n + 1, dtype=int)
        for i in range(n):
            chunk = seq[edges[i]:edges[i + 1]]
            arr[i] = float(chunk.mean()) if chunk.size else 0.0
    return normalize_positive(arr)


def build_conv_matrix(f):
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    offsets = np.arange(n, dtype=np.int64)
    cols = np.tile(offsets, n)
    rows = cols + np.repeat(offsets, n)
    data = np.tile(f, n)
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def lp_polish(a0, time_limit):
    a = normalize_positive(a0)
    n = a.size
    c0 = conv_self_fft(a)
    best_score = 2.0 * float(n) * float(c0.max()) if c0.size else np.inf
    best = a.copy()

    if (not HAVE_LINPROG) or n == 0 or time_limit <= 0.0:
        return best, best_score

    end_time = time.time() + float(time_limit)
    unit = np.ones(2 * n - 1, dtype=np.float64)
    cobj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    mixes = (0.05, 0.10, 0.15, 0.25, 0.40, 0.60, 0.85, 1.00)

    it = 0
    while it < 30 and time.time() < end_time - 1.0:
        c = conv_self_fft(a)
        M = float(c.max()) if c.size else 0.0
        if not np.isfinite(M) or M <= 1e-18:
            break

        try:
            A = build_conv_matrix(a / M)
            A = A.tocsr()
            res = linprog(
                cobj,
                A_ub=A,
                b_ub=unit,
                bounds=bounds,
                method="highs"
            )
        except Exception:
            break

        if res is None or not res.success:
            break

        g = np.asarray(res.x, dtype=np.float64)
        g = np.where(np.isfinite(g) & (g > 0.0), g, 0.0)
        gs = float(g.sum())
        if gs <= 1e-18:
            break
        g /= gs

        cand_best = a.copy()
        score_best = best_score
        found = False

        for t in mixes:
            cand = (1.0 - t) * a + t * g
            cand = np.where(np.isfinite(cand) & (cand > 0.0), cand, 0.0)
            cs = float(cand.sum())
            if cs <= 1e-18:
                continue
            cand /= cs
            cc = conv_self_fft(cand)
            sc = 2.0 * float(n) * float(cc.max()) if cc.size else np.inf
            if np.isfinite(sc) and sc < score_best - 1e-12:
                score_best = float(sc)
                cand_best = cand.copy()
                found = True

        if not found:
            break

        a = cand_best
        best = a.copy()
        best_score = score_best
        it += 1

    return best, float(best_score)


def update_best_record(best_seq, best_score, best_n,
                       small_seq, small_score,
                       cand, score):
    if cand is None or cand.size == 0 or (not np.isfinite(score)):
        return best_seq, best_score, best_n, small_seq, small_score

    cand = np.asarray(cand, dtype=np.float64)
    score = float(score)

    if score < best_score:
        best_seq = cand.copy()
        best_score = score
        best_n = int(best_seq.size)

    if small_seq is None:
        small_seq = cand.copy()
        small_score = score
    elif score < small_score and cand.size <= 520:
        small_seq = cand.copy()
        small_score = score

    return best_seq, best_score, best_n, small_seq, small_score


def add_top_candidate(top, score, seq, idx):
    if seq is None or seq.size == 0 or (not np.isfinite(score)):
        return top
    top.append((float(score), int(idx), np.asarray(seq, dtype=np.float64).copy()))
    top.sort()
    if len(top) > 14:
        del top[-1]
    return top

            best_seq, best_score, best_n, small_seq, small_score, res_seq, res_score
        )
        top_candidates = add_top_candidate(top_candidates, res_score, res_seq, top_counter)
        top_counter += 1

        processed += 1
        if processed % 2 == 0:
            print(f"pass1 processed={processed} best={best_score:.10f} n={best_n}")
            print(flush=True)

    # Second pass: longer search from the current best and a perturbed copy.
    if best_seq is not None:
        rem = deadline - time.time()
        if rem > 260.0:
            budget = min(420.0, rem / 3.0)
            res_seq, res_score = adam_optimize(best_seq, budget, seed=seed + 999)
            best_seq, best_score, best_n, small_seq, small_score = update_best_record(
                best_seq, best_score, best_n, small_seq, small_score, res_seq, res_score
            )
            top_candidates = add_top_candidate(top_candidates, res_score, res_seq, top_counter)
            top_counter += 1

        rem = deadline - time.time()
        if rem > 220.0:
            a_pert = perturb_sequence(best_seq, rng, 0.12)
            budget = min(360.0, rem / 3.0)
            res_seq, res_score = adam_optimize(a_pert, budget, seed=seed + 1234)
            best_seq, best_score, best_n, small_seq, small_score = update_best_record(
                best_seq, best_score, best_n, small_seq, small_score, res_seq, res_score
            )
            top_candidates = add_top_candidate(top_candidates, res_score, res_seq, top_counter)
            top_counter += 1

    # LP-inspired polish on moderate-length candidates.
    lp_candidates = []
    if small_seq is not None and small_seq.size <= 520:
        lp_candidates.append(small_seq)
    if best_seq is not None and best_seq.size <= 520:
        dup = False
        for cand in lp_candidates:
            if cand.size == best_seq.size:
                dup = True
                break
        if not dup:
            lp_candidates.append(best_seq)

    for cand in lp_candidates:
        rem = deadline - time.time()
        if rem < 140.0 or not HAVE_LINPROG:
            break
        budget = min(360.0, rem - 70.0)
        if budget < 60.0:
            break
        res_seq, res_score = lp_polish(cand, budget)
        best_seq, best_score, best_n, small_seq, small_score = update_best_record(
            best_seq, best_score, best_n, small_seq, small_score, res_seq, res_score
        )
        top_candidates = add_top_candidate(top_candidates, res_score, res_seq, top_counter)
        top_counter += 1

    print(f"before exact final: best={best_score:.10f} n={best_n}")
    print(flush=True)

    # Final exact fallback: verify and choose the best valid candidate.
    final_seq = None
    final_score = np.inf

    # Include best explicitly.
    if best_seq is not None:
        top_candidates = add_top_candidate(top_candidates, best_score, best_seq, top_counter)
        top_counter += 1
    if small_seq is not None:
        top_candidates = add_top_candidate(top_candidates, small_score, small_seq, top_counter)
        top_counter += 1

    top_candidates.sort()
    evaluated = 0
    for sc, idxs, seq in top_candidates:
        rem = deadline - time.time()
        if rem < 2.0:
            break
        if evaluated >= 10:
            break
        if seq is None or seq.size == 0:
            continue

        # For very large sequences, exact np.convolve can be unnecessary slow.
        if seq.size <= 6500:
            true_sc = exact_score_array(seq)
        else:
            true_sc = fast_score_normalized(normalize_positive(seq))

        if np.isfinite(true_sc) and true_sc < final_score:
            final_score = float(true_sc)
            final_seq = seq.copy()

        evaluated += 1

    if final_seq is None:
        final_seq = best_seq if best_seq is not None else np.ones(64, dtype=np.float64)

    final_seq = normalize_positive(final_seq)

    # Authoritative check, if available.
    try:
        ev = evaluate_sequence(final_seq.tolist())
        if np.isfinite(ev):
            final_score = float(ev)
    except Exception:
        pass

    if best_seq is not None:
        try:
            ev2 = evaluate_sequence(best_seq.tolist())
            if np.isfinite(ev2) and ev2 < final_score:
                final_score = float(ev2)
                final_seq = normalize_positive(best_seq)
        except Exception:
            pass

    final_seq = np.where(np.isfinite(final_seq) & (final_seq > 0.0), final_seq, 0.0)
    fs = float(final_seq.sum())
    if not np.isfinite(fs) or fs <= 0.0:
        final_seq = np.ones(64, dtype=np.float64)
        fs = float(final_seq.sum())
    final_seq /= fs

    try:
        ev_final = evaluate_sequence(final_seq.tolist())
        if np.isfinite(ev_final):
            final_score = float(ev_final)
    except Exception:
        pass

    print(f"final selected score={final_score:.10f} n={final_seq.size}")
    print(flush=True)

    return [float(x) for x in final_seq]
