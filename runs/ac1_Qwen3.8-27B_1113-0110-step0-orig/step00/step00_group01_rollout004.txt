        # For now, minimize C_smooth directly with constraint sum(a)=1

        a_new = np.clip(a_new, 0, None)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-10:
            a_new = a  # fallback

            c_new = autocorr(a_new)
            if c_new.max() < C:  # check actual improvement
                a = a_new

    def auto(x):
        """Compute autocorrelation using FFT."""
        N = 1 << (2 * len(x) - 1).bit_length()
        F = fft(x, N)
        return ifft(F * F.conj()).real[:2*len(x)-1]
    
    # Main optimization loop
    tau = 0.01
    for outer in range(500):

            break

        c = auto(a)
        C = c.max()
        kstar = c.argmax()

        c_scaled = c / tau
        c_scaled -= c_scaled.max()
        w = np.exp(c_scaled)
        w /= w.sum()
        
        # Compute weighted gradient: grad[j] = 2 * sum_k w[k] * a[k-j]
        # = 2 * (correlation of a and w)[j]
        # Compute via FFT
        N = 1 << (3 * n - 1).bit_length()
        F_a = fft(a, N)
        F_w = fft(w, N)
        # cross[j] = sum_i a[i] * w[i+j] = (reverse(a) * w)[n-1+j]
        F_reva = fft(a[::-1], N)
        cross_full = ifft(F_reva * F_w).real
        cross = cross_full[n-1 : 2*n-1]
        grad = 2 * cross
        
        # Gradient descent step
        lr = 0.01
        a_new = a - lr * grad
        a_new = np.clip(a_new, 0, None)
        s = a_new.sum()
        if s > 1e-10:
            a_new /= s
            c_new = auto(a_new)
            score_new = 2 * n * c_new.max()
            if score_new < 2 * n * C:
                a = a_new
                C = c_new.max()
        
        # Periodic re-evaluation
        if outer % 50 == 0:
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_a = a.copy()
                best_score = score
                print(f"Iter {outer}: score = {best_score:.6f}")
        
        # Anneal tau
        tau *= 0.99
    
    # Final evaluation
    score = evaluate_sequence(a.tolist())
    if score < best_score:
        best_a = a.copy()
        best_score = score
    
    # Try random restarts with remaining time
    while time.monotonic() < deadline - 30:

            if s > 1e-10:
