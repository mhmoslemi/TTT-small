def build_constraint_matrix(a):
    """Build the constraint matrix for the LP.
    
    Row m: A[m, i] = a[m-i] if 0 <= m-i < n, else 0.
    """
    n = len(a)
    rows = 2*n - 1
    cols = n
    
    # Build as sparse matrix
    row_indices = []
    col_indices = []
    values = []
    
    for m in range(rows):
        for i in range(cols):
            j = m - i
            if 0 <= j < n:
                row_indices.append(m)
                col_indices.append(i)
                values.append(a[j])

    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='hihps')

"""
LP-based iterative optimization for minimum peak full-convolution sequences.
Minimizes 2n*max(conv(a,a))/(sum(a))^2 by alternating between computing the
current peak, solving an LP for the optimal update direction (maximizing sum
of new sequence subject to bounded cross-convolution), and blending.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def _precompute_indices(n):
    """Precompute sparse matrix indices for convolution constraint a*b."""
    row_idx = []
    col_idx = []
    val_idx = []
    for m in range(2 * n - 1):
        i_start = max(0, m - n + 1)
        i_end = min(m, n - 1)
        for i in range(i_start, i_end + 1):
            row_idx.append(m)
            col_idx.append(i)
            val_idx.append(m - i)
    return (np.array(row_idx, dtype=np.int32),
            np.array(col_idx, dtype=np.int32),
            np.array(val_idx, dtype=np.int32))

def _solve_lp(a, M, idx_data):
    """Solve: max 1'b s.t. A(a) b <= M, b >= 0."""
    n = len(a)
    row_idx, col_idx, val_idx = idx_data
    vals = a[val_idx]
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True, 'dual_feasibility_tolerance': 1e-8})
    if res.success:
        return res.x
    return None

def _quick_score(a):
    """Compute score without security checks (assumes valid non-negative a)."""
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (s * s)

def _lp_iteration(a, idx_data, t_step, rng):
    """One LP-based iteration. Returns new a."""
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return rng.random(n)
    a_norm = a / s
    
    c = np.convolve(a_norm, a_norm)
    M = c.max()
    
    b = _solve_lp(a_norm, M, idx_data)
    if b is None or b.sum() < 1e-14:
        # Fallback: just return a with small perturbation
        a_new = a_norm + 0.01 * rng.standard_normal(n)
        a_new = np.clip(a_new, 0, None)
        return a_new
    
    b = b / b.sum()
    
    a_new = (1.0 - t_step) * a_norm + t_step * b
    a_new = np.clip(a_new, 0, None)
    s_new = a_new.sum()
    if s_new < 1e-12:
        a_new = a_norm + 0.01 * rng.random(n)
        a_new = np.clip(a_new, 0, None)
    return a_new / a_new.sum()

def propose_candidate(seed=42, budget_s=1800, **kwargs):

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 20.0
    
    # Start with given sequence
    a = np.asarray(height_sequence_1, dtype=np.float64)

    print(f"Init: n={n}, score={best_score:.6f}")
    
    # Precompute LP structure
    idx_data = _precompute_indices(n)
    nnz = idx_data[0].shape[0]
    print(f"LP: {n} vars, {2*n-1} constr, {nnz} nnz")
    
    t_step = 0.5
    stagnant = 0
    iteration = 0
    start_time = time.monotonic()
    
    while time.monotonic() < deadline:
        iteration += 1
        
        a_new = _lp_iteration(a, idx_data, t_step, rng)
        
        score = _quick_score(a_new)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_new.copy()
            stagnant = 0
            t_step = min(0.8, t_step * 1.03)
        else:
            stagnant += 1
            t_step = max(0.1, t_step * 0.92)
        
        a = a_new
        
        # Progress reporting
        if iteration % 20 == 1:
            elapsed = time.monotonic() - start_time
            print(f"Iter {iteration}: best={best_score:.6f}, curr={_quick_score(a):.6f}, t={t_step:.3f}, elapsed={elapsed:.0f}s")
        
        # Restart if stuck
        if stagnant >= 40:
            print(f"  Restart at iter {iteration}")
            a = best_a + 0.03 * rng.standard_normal(n)
            a = np.clip(a, 0, None)
            a = a / a.sum()
            stagnant = 0
            t_step = 0.5
    
    # Final verification
    final_score = float(evaluate_sequence(best_a.tolist()))
    print(f"Final: best={final_score:.6f} (verified), iterations={iteration}, time={time.monotonic()-start_time:.0f}s")
    
    return [float(x) for x in best_a]
