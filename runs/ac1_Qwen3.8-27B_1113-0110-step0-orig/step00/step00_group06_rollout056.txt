                init = np.ones(n) / n
            elif trial == 1:
                rng_t = np.random.default_rng(seed + n * 7 + trial)
                init = np.abs(rng_t.standard_normal(n))
                init /= np.sum(init)
            else:
                # Smooth window (raised cosine)
                x = np.linspace(0, 1, n)
                init = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
                init = np.maximum(init, 1e-10)
                init /= np.sum(init)

            a, max_c = optimize_sequence(n, init, sub_deadline, seed=seed + n + trial)
            score = 2.0 * n * max_c
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"n={n} trial={trial}: score={best_score:.6f}")

        remaining_time = deadline - time.monotonic()
        # Reduce total weight for remaining
        total_w = sum(weights[idx + 1:]) if idx + 1 < n_items else 0
        if total_w == 0:
            break

    # Final: try to polish the best solution found with a short gradient run
    if best_a is not None and time.monotonic() < deadline - 10:
        n_best = len(best_a)
        polish_deadline = deadline - 2.0
        alloc = min(60.0, polish_deadline - time.monotonic())
        a_pol, max_c_pol = optimize_sequence(n_best, best_a, time.monotonic() + alloc,
                                              seed=seed + 999)
        score_pol = 2.0 * n_best * max_c_pol
        if score_pol < best_score:
            best_score = score_pol
            best_a = a_pol.copy()
            print(f"Polished: score={best_score:.6f}")

    if best_a is None:
        best_a = np.ones(1)

    print(f"Final best score: {best_score:.6f}, n={len(best_a)}")
    return [float(x) for x in best_a]
