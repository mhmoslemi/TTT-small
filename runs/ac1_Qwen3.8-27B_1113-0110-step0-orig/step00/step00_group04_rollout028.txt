```python
import numpy as np
import time
from scipy import fftpack  # or use numpy's fft

def convolve_full(a, b):
    """Full linear convolution using FFT."""
    n = len(a) + len(b) - 1
    N = 1
    while N < n:
        N *= 2
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    return np.fft.irfft(A * B, N)[:n]

def correlate_full(a, b):
    """Full linear correlation: result[k] = sum_i a[i+k] * b[i]."""
    N = len(a) + len(b) - 1
    Nfft = 1
    while Nfft < N:
        Nfft *= 2
    A = np.fft.rfft(a, Nfft)
    B = np.fft.rfft(b, Nfft)
    return np.fft.irfft(A * np.conj(B), Nfft)[:N]

def project_simplex(v):
    """Project v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1)).sum()
    theta = (cssv[rho-1] - 1) / rho if rho > 0 else 0
    return np.maximum(v - theta, 0)
```

```python
    n = len(p0)
    p = p0.copy()
```

```python
    no_improve = 0

        if time.monotonic() > deadline:
            break
        beta = beta_start * (beta_end / beta_start) ** (it / max(1, n_iter))

        z = beta * C
        z_max = z.max()
        if z_max > 700:  # prevent overflow
            z = np.clip(z, z_max - 700, z_max)
        Z = np.exp(z - z_max)
        Z_sum = Z.sum()
        if Z_sum == 0:

            Z[C.argmax()] = 1.0
```

The standard approach for the minimum peak sidelobe problem is:
1. In the time domain: minimize max(C_k) for k ≠ 0 (or all k)
2. In the frequency domain: this is equivalent to finding a spectrum that's "flat" (so that the inverse transform has low peaks)

For our problem (all k, including k=0), the approach is similar.

1. Compute C = autocorrelation of p
2. Find peaks in C
3. In the frequency domain, reduce the power at frequencies that contribute to those peaks
4. Transform back, project onto simplex
5. Repeat

This is essentially the "projected gradient" approach but in the frequency domain. It might converge faster for some initializations.
