    c_shifted = c - c.max()  # for numerical stability

import numpy as np
import time
import math

def next_fast_len(n):
    """Next power of 2 >= n."""
    return 1 << (n - 1).bit_length()

def soft_max_gradient(a, c, temperature):
    """Compute gradient of soft-max of c with respect to a.
    Uses FFT for efficiency."""
    n = len(a)
    c_max = c.max()
    # Soft max weights (numerically stable)
    exp_c = np.exp((c - c_max) / temperature)
    w = exp_c / exp_c.sum()
    
    # Gradient: g[i] = (2/temperature) * sum_k w[k] * a[k-i]
    # = (2/temperature) * (w cross-correlate a)[i]
    # Compute via convolution: cross_corr[i] = sum_k w[k] * a[k-i]
    # = convolve(w, a_reversed)[n-1+i]
    a_rev = a[::-1]
    N = next_fast_len(len(w) + n - 1)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a_rev, N)
    conv = np.fft.irfft(W * A, N)
    # cross_corr[i] = conv[n-1+i] for i = 0,...,n-1
    cross = conv[n-1 : 2*n-1]
    return (2.0 / temperature) * cross

def make_init_uniform(n, endpoint_reduction=0.0):
    """Uniform with optional endpoint reduction."""
    a = np.ones(n)
    if endpoint_reduction > 0:
        edge = max(1, int(n * endpoint_reduction))
        # Linearly ramp down at edges
        ramp = np.linspace(0, 1, edge)
        a[:edge] = ramp
        a[-edge:] = ramp[::-1]
    return a / a.sum()

def make_init_sinusoidal(n, freq=1, phase=0, amplitude=0.3):
    """1 + amplitude * sin(2*pi*freq*i/n + phase), clipped to non-negative."""
    i = np.arange(n)
    a = 1 + amplitude * np.sin(2 * np.pi * freq * i / n + phase)
    a = np.clip(a, 0, None)
    if a.sum() < 1e-10:
        a = np.ones(n)
    return a / a.sum()
