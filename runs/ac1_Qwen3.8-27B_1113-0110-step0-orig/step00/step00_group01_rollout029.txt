    deadline = time.monotonic() + budget_s - 5.0

    n = len(a)

    best_score = evaluate_sequence(best.tolist())
    
    while time.monotonic() < deadline:
        # Compute current convolution
        conv = np.convolve(a, a)
        M = np.max(conv)
        s = np.sum(a)
        
        # Solve LP: max sum(b) s.t. b >= 0, conv(a,b) <= M
        # Variables: b[0], ..., b[n-1]
        # Objective: -sum(b) (minimize negative)
        c = -np.ones(n)
        
        # Constraints: for each k in 0..2n-2, sum_j a[j]*b[k-j] <= M
        # This is A @ b <= M * ones(2n-1)
        # A[k, m] = a[m] if 0 <= k-m <= n-1, else 0
        #           + a[k-m] if 0 <= k-m <= n-1, else 0
        # Wait, conv(a,b)[k] = sum_j a[j]*b[k-j]
        # So for a fixed k, the coefficient of b[m] is a[k-m] (if 0 <= k-m <= n-1)
        
        # A is (2n-1) x n, where A[k, m] = a[k-m] if 0 <= k-m <= n-1, else 0
        # But this is a sparse matrix! We can use scipy.sparse.
        
        from scipy.sparse import csr_matrix
        
        # Build A
        rows = []
        cols = []
        vals = []
        for m in range(n):
            # b[m] appears in conv(a,b)[k] for k = m, m+1, ..., m+n-1
            # with coefficient a[k-m]
            for k in range(m, min(2*n-1, m+n)):
                rows.append(k)
                cols.append(m)
                vals.append(a[k-m])
        
        A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
        
        # Constraints: A @ b <= M, b >= 0
        res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')

            s_b = np.sum(b)
            if s_b > 0:

                # Take step

                a_new = np.clip(a_new, 0, 1000)
                score = evaluate_sequence(a_new.tolist())
                if score < best_score:
                    a = a_new
                    best = a_new.copy()
                    best_score = score
                    print(f"Improvement: {best_score:.6f}")
    
    return best.tolist()

from scipy.optimize import linprog
import time

def solve_lp(a, M):
    """Solve the LP: max sum(b) s.t. b >= 0, conv(a,b) <= M."""
    n = len(a)

    rows = []
    cols = []
    vals = []
    
    for m in range(n):
        # b[m] contributes to conv(a,b)[k] for k in [m, m+n-1] with coefficient a[k-m]

$= a[k^* - m] + a[m]$ (if both $m$ and $k^*-m$ are in range)

For $m = k^* - m$ (i.e., $2m = k^*$):
$= 2a[m]$ (if $m$ is in range)

    # Each column m has nonzeros at rows m..min(m+n-1, 2n-2)

    nnz_per_col = []
    for m in range(n):

        nnz_per_col.append(k_max - m + 1)
    total = sum(nnz_per_col)

    cols = np.empty(total, dtype=np.int64)

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequence a.
Strategy: LP-based iterative smoothing (from SoA paper) with line search,
multiple initializations, and local polishing.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def build_conv_matrix(a):
    """Sparse matrix A: A[k,m] = a[k-m] if 0<=k-m<=n-1, else 0. Shape (2n-1, n)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for m in range(n):
        k_lo = m
        k_hi = min(m + n - 1, 2 * n - 2)
        if k_lo > k_hi:
            continue
        ks = np.arange(k_lo, k_hi + 1)
        rows.append(ks)
        cols.append(np.full(len(ks), m, dtype=np.int32))
        vals.append(a[:len(ks)])
    r = np.concatenate(rows)
    c = np.concatenate(cols)
    v = np.concatenate(vals)
    return csr_matrix((v, (r, c)), shape=(2 * n - 1, n))


def lp_direction(a, M, timeout=25):
    """LP: max 1'b s.t. Ab<=M, b>=0, b_j <= cap. Returns b or None."""
    n = len(a)
    A = build_conv_matrix(a)
    cap = 4.0 * np.sum(a) / n  # Prevent degenerate solutions
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, cap)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout})
    if res.success and res.x is not None:
        b = np.maximum(res.x, 0.0)
        if np.sum(b) > 1e-12:
            return b
    return None


def do_line_search(a, b, ref_score):
    """Try several convex combinations, return best (alpha, score)."""
    best_a = 0.0
    best_s = ref_score
    for alpha in (0.15, 0.25, 0.35, 0.5, 0.65, 0.8, 1.0):
        cand = (1.0 - alpha) * a + alpha * b
        cand = np.clip(cand, 0.0, 1000.0)
        if cand.sum() < 0.01:
            continue
        sc = evaluate_sequence(cand.tolist())
        if sc < best_s:
            best_s = sc
            best_a = alpha
    return best_a, best_s


def polish(a, best_score, rng, t_end):
    """Local search with small random multiplicative perturbations."""
    cur = a.copy()
    n = len(cur)
    while time.monotonic() < t_end:
        w = int(rng.integers(1, max(2, n // 25)))
        st = int(rng.integers(0, n - w))
        cand = cur.copy()
        cand[st:st+w] *= np.exp(rng.normal(0.0, 0.05, w))
        np.clip(cand, 0.0, 1000.0, out=cand)
        sc = evaluate_sequence(cand.tolist())
        if sc < best_score:
            cur = cand
            best_score = sc
    return cur, best_score

    # For column m (i from m*n to (m+1)*n - 1): indices = m, m+1, ..., m+n-1
