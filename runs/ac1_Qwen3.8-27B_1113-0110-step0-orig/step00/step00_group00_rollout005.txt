def build_conv_matrix(f, n):
    """Build sparse matrix A where A[k,i] = f[k-i] for valid indices."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:

            rows.extend(np.full(len(idx), k).tolist())
            cols.extend(idx.tolist())
