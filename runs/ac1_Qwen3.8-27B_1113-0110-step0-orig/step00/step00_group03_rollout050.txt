    F_full = fft(f, n=size)
    P_full = fft(p, n=size)
    cross_corr = ifft(np.conj(F_full) * P_full).real
    grad = 2.0 * cross_corr[:n]
    return smooth_val, grad, C


def pgd_optimize(f_init, beta_start, beta_end, n_iter, deadline, seed):
    """Projected gradient descent with beta annealing on smooth max objective."""
    rng = np.random.default_rng(seed)
    n = len(f_init)
    f = f_init.copy()
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_f = f.copy()
    best_smooth = np.inf

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        t = it / max(1, n_iter - 1)
        beta = beta_start * (beta_end / beta_start) ** t

        smooth_val, grad, C = compute_smooth_max_grad(f, beta)

        if smooth_val < best_smooth:
            best_smooth = smooth_val
            best_f = f.copy()

        # Step size: 1/L where L ~ beta (Lipschitz of gradient)
        # Use conservative step with occasional backtracking
        step = 2.0 / (beta + 1.0)

        # Backtracking line search (limited to 5 steps)
        for _ in range(5):
            f_trial = f - step * grad
            f_trial = project_onto_simplex(f_trial, 1.0)
            smooth_trial, _, _ = compute_smooth_max_grad(f_trial, beta)
            if smooth_trial < smooth_val:
                f = f_trial
                break
            step *= 0.5
        else:
            # All backtracking failed, keep previous
            pass

    return best_f


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max(conv)/sum^2.
    Uses FFT-based projected gradient descent with smooth max objective,
    diverse asymmetric initializations, beta annealing, and local refinement.
    """
    start_time = time.monotonic()
    total_budget = float(budget_s)
    deadline = start_time + total_budget - 15.0

    n = 2535
    rng = np.random.default_rng(seed)

    # --- Generate diverse initializations ---
    inits = []

    # Try parent
    parent = None
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.size == 0 or not np.isfinite(parent).all():
            parent = None
    except Exception:
        parent = None

    if parent is not None and np.sum(parent) > 0.01:
        p = np.maximum(parent, 0.0)
        inits.append(p / np.sum(p))

    # Uniform
    inits.append(np.ones(n) / n)

    # Linearly increasing
    lin = np.arange(1, n + 1, dtype=np.float64)
    inits.append(lin / np.sum(lin))

    # Linearly decreasing
    inits.append(lin[::-1] / np.sum(lin))

    # U-shaped (parabola)
    x = np.linspace(0, 1, n)
    u = 1.0 + 2.0 * (2.0 * x - 1.0) ** 2
    inits.append(u / np.sum(u))

    # Two-peak (high at both edges, low middle)
    two_peak = np.ones(n) * 0.1
    two_peak[:n // 4] = 1.0
    two_peak[-n // 4:] = 1.0
    inits.append(two_peak / np.sum(two_peak))

    # Exponential decay
    ed = np.exp(-np.linspace(0, 4, n))
    inits.append(ed / np.sum(ed))

    # Exponential growth
    eg = np.exp(-np.linspace(-4, 0, n))
    inits.append(eg / np.sum(eg))

    # Sine modulation
    sine = 1.0 + 0.8 * np.sin(2.0 * np.pi * x)
    sine = np.maximum(sine, 0.01)
    inits.append(sine / np.sum(sine))

    # Asymmetric peak (peak at 30%)
    asym = np.zeros(n)
    pk = int(n * 0.3)
    asym[:pk] = np.linspace(0.1, 1.0, pk)
    asym[pk:] = np.linspace(1.0, 0.1, n - pk)
    inits.append(asym / np.sum(asym))

    # Step function (low left, high right)
    step_f = np.zeros(n)
    step_f[:n // 3] = 0.3
    step_f[n // 3:] = 1.0
    inits.append(step_f / np.sum(step_f))

    # Random sequences
    for _ in range(4):
        r = rng.random(n) * 0.8 + 0.2
        inits.append(r / np.sum(r))

    # Power law
    pw = np.power(np.linspace(0.01, 1.0, n), 0.5)
    inits.append(pw / np.sum(pw))
    pw2 = np.power(np.linspace(1.0, 0.01, n), 0.5)
    inits.append(pw2 / np.sum(pw2))

    # --- Phase 0: Quick evaluation of all initializations ---
    scores = []
    for i, fi in enumerate(inits):
        sc = float(evaluate_sequence(fi.tolist()))
        scores.append(sc)
        if np.isfinite(sc):
            pass

    # Sort by score, keep top 8 for optimization
    order = np.argsort(scores)
    top_inits = [inits[order[i]] for i in range(min(8, len(inits)))]
    print(f"Phase 0: Best initial score = {scores[order[0]]:.6f}")

    best_f = inits[order[0]].copy()
    best_score = float(scores[order[0]])

    # --- Phase 1: Projected gradient descent on top initializations ---
    beta_start = 3.0
    beta_end = 3000.0

    # Allocate time proportionally
    n_top = len(top_inits)
    time_per_init = (deadline - time.monotonic()) / max(1, n_top) - 2.0
    time_per_init = max(30.0, time_per_init)

    for idx, f_init in enumerate(top_inits):
        if time.monotonic() > deadline:
            break

        init_deadline = time.monotonic() + time_per_init
        n_iter = 80000

        f_opt = pgd_optimize(f_init, beta_start, beta_end, n_iter, init_deadline,
                             seed + idx * 1000)

        sc = float(evaluate_sequence(f_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_f = f_opt.copy()
            print(f"PGD init {idx}: {best_score:.6f} at t={time.monotonic()-start_time:.1f}s")

    print(f"After PGD: best = {best_score:.6f} at t={time.monotonic()-start_time:.1f}s")

    # --- Phase 2: Local refinement via block perturbations ---
    if time.monotonic() < deadline:
        f_ref = best_f.copy()
        sc_ref = best_score
        n_trials = 0
        for _ in range(200000):
            if time.monotonic() > deadline:
                break
            n_trials += 1

            width = int(rng.integers(3, max(5, n // 30)))
            start = int(rng.integers(0, n - width))
            f_cand = f_ref.copy()
            f_cand[start:start + width] *= np.exp(rng.normal(0, 0.15, size=width))
            f_cand = np.maximum(f_cand, 0.0)
            s = np.sum(f_cand)
            if s < 1e-15:
                continue
            f_cand /= s

            sc = float(evaluate_sequence(f_cand.tolist()))
            if np.isfinite(sc) and sc < sc_ref:
                sc_ref = sc
                f_ref = f_cand
                if sc < best_score:
                    best_score = sc
                    best_f = f_ref.copy()
                    print(f"Refine: {best_score:.6f} at t={time.monotonic()-start_time:.1f}s")

    # --- Phase 3: Fine local search with small random perturbations ---
    if time.monotonic() < deadline:
        f_ref = best_f.copy()
        sc_ref = best_score
        for _ in range(500000):
            if time.monotonic() > deadline:
                break

            n_pert = int(rng.integers(1, 15))
            idx_p = rng.choice(n, size=n_pert, replace=False)
            f_cand = f_ref.copy()
            f_cand[idx_p] *= np.exp(rng.normal(0, 0.04, size=n_pert))
            f_cand = np.maximum(f_cand, 0.0)
            s = np.sum(f_cand)
            if s < 1e-15:
                continue
            f_cand /= s

            sc = float(evaluate_sequence(f_cand.tolist()))
            if np.isfinite(sc) and sc < sc_ref:
                sc_ref = sc
                f_ref = f_cand
                if sc < best_score:
                    best_score = sc
                    best_f = f_ref.copy()
                    print(f"Fine: {best_score:.6f} at t={time.monotonic()-start_time:.1f}s")

    # --- Phase 4: One more PGD round from the best found, with higher beta ---
    if time.monotonic() < deadline and best_score > 1.5:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            f_opt = pgd_optimize(best_f, 50.0, 5000.0, 60000, deadline, seed + 99999)
            sc = float(evaluate_sequence(f_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_f = f_opt.copy()
                print(f"Final PGD: {best_score:.6f} at t={time.monotonic()-start_time:.1f}s")

    result = [float(x) for x in best_f]
    final_score = float(evaluate_sequence(result))
    print(f"FINAL SCORE: {final_score:.6f} | time: {time.monotonic()-start_time:.1f}s")
    return result
