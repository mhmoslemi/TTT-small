import numpy as np
import time

def _compute_conv_fft(a, fft_size):
    """Compute full linear convolution a*a using FFT."""
    A = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(A * A, fft_size)
    return conv[:2*len(a)-1]

def _project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    # Algorithm from "Euclidean projection to the simplex" (Duchi et al. 2012)
    if np.sum(v) <= 1.0 and np.all(v >= 0):
        return v / np.sum(v)  # normalize
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _run_gradient_descent(a0, rng, max_time, seed_offset=0):
    """Run projected gradient descent to minimize the objective."""
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s  # normalize to sum = 1
    
    fft_size = 1
    while fft_size < 2 * n - 1:

def _next_power_of_2(x):
    """Return the next power of 2 >= x."""
    p = 1
    while p < x:
        p <<= 1
    return p

    """Compute the full linear autocorrelation a*a[0..2n-2] via FFT."""
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:2 * len(a) - 1]

    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return np.ones_like(a) / len(a)
    return a / s

def _smooth_max_grad(conv, T, a, n):
    """Compute gradient of (1/T)*log(sum(exp(T*conv))) w.r.t. a."""
    # conv has length 2n-1
    # To avoid overflow, subtract the max
    shifted = T * conv - T * np.max(conv)
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    # Gradient is sum_k weights[k] * d(conv[k])/da
    # d(conv[k])/da[i] = a[k-i] + a[i] if 0 <= k-i < n, else 0
    grad = np.zeros(n)
    for k in range(2*n - 1):
        if weights[k] < 1e-10:
            continue
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        ii = np.arange(i_lo, i_hi + 1)
        grad[ii] += weights[k] * (a[k - ii] + a[ii])
    return grad

This is O(n²) in the worst case (when many weights are significant). For practical purposes, most weights will be concentrated on a few k values, so it's closer to O(n).

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using projected gradient descent

multiple starting points, and a final local-search polish phase.

import numpy as np

def _next_pow2(x):
    p = 1
    while p < x:
        p <<= 1
    return p


def _autocorr_fft(a, fft_size):
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[: 2 * len(a) - 1]


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return np.ones_like(a) / len(a)
    return a / s


def _optimize(a0, rng, max_time, lr_init=0.02, mu=0.9):
    n = len(a0)
    a = _normalize(a0.astype(np.float64).copy())
    fft_size = _next_pow2(2 * n - 1)

    best_score = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    v = np.zeros(n)

    start = time.monotonic()

        conv = _autocorr_fft(a, fft_size)
        max_b = float(np.max(conv))
        k_star = int(np.argmax(conv))
        score = 2.0 * n * max_b

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        i_lo = max(0, k_star - n + 1)
        i_hi = min(n - 1, k_star)
        grad = np.zeros(n)
        if i_lo <= i_hi:
            ii = np.arange(i_lo, i_hi + 1)
            grad[ii] = a[k_star - ii] + a[ii]
