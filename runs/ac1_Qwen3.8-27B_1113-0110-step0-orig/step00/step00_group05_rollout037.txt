```python
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csc_matrix
import sys

def compute_score(a):
    """Compute the evaluation score (without calling evaluate_sequence)."""
    n = len(a)
    b = np.convolve(a, a)
    max_b = max(b)
    sum_a = sum(a)
    if sum_a < 0.01:
        return float('inf')
    return 2 * n * max_b / (sum_a ** 2)

def generate_ramp(n, p=1.0):
    """Generate a power-law ramp: a[j] = (j+1)^p."""
    return np.array([(j+1)**p for j in range(n)])

def solve_lp(f, M):
    """Solve the LP: max sum(g) s.t. (f*g) <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)  # minimize -sum(g)
    
    # Build constraint matrix A_ub
    # A_ub has shape (2n-1, n)
    # A_ub[k, j] = f[k-j] if 0 <= k-j <= n-1, else 0
    # b_ub = M * ones(2n-1)
    
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            h = k - j  # h is the index into f
            rows.append(k)
            cols.append(j)
            vals.append(f[h])
    
    A_ub = csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', 
                     options={'presolve': True})
    
    if result.status == 0:
        return result.x
    else:
        return None

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    start_time = time.monotonic()
    deadline = start_time + budget_s - 5
    
    # Start with a ramp
    n = 1000
    a = generate_ramp(n, p=1.0)
    a = a / sum(a)  # normalize to sum = 1
    
    best_a = a.copy()
    best_score = compute_score(best_a)
    print(f"Initial ramp (n={n}): score = {best_score:.6f}")
    
    # Iterative LP refinement
    t = 0.1  # step size
    iteration = 0
    while time.monotonic() < deadline:
        f = a.copy()
        M = max(np.convolve(f, f))
        
        g = solve_lp(f, M)
        if g is None:
            # Try a larger M
            g = f.copy()
        else:
            # Rescale g to have same sum as f
            sum_g = sum(g)
            if sum_g < 1e-10:
                g = f.copy()
            else:
                g = g * (sum(f) / sum_g)
        
        # Update
        a = (1 - t) * f + t * g
        a = np.clip(a, 0, None)
        # Normalize
        if sum(a) > 0:
            a = a / sum(a)
        
        score = compute_score(a)
        if score < best_score:
            best_a = a.copy()
            best_score = score
            print(f"Iter {iteration}: score = {best_score:.6f}, M = {M:.6f}")
        
        iteration += 1
    
    # Convert to list and ensure non-negative
    result = np.clip(best_a, 0, 1000)
    # Scale up to use the full range (the score is scale-invariant, but let's keep values reasonable)
    result = result * 100  # arbitrary scale
    
    return [float(x) for x in result]
```
