    # A has shape (2n-1, n)

    #   A[k, i] = f[k-i]

Alternative approach: Multiplicative Weights / Mirror Descent

We want to minimize max_k (a*a)[k] subject to a >= 0, sum(a) = 1.

Mirror descent with entropy regularization:
- Update: a_new[j] = a[j] * exp(-eta * sg[j]) / Z
- Where sg is the subgradient

This maintains a > 0 automatically and is O(n) per step.

def mirror_descent_step(a, eta, num_peak=5):
    n = len(a)
    conv = np.convolve(a, a)
    # Use top-k peaks for a smoothed gradient
    top_k = np.argsort(conv)[-num_peak:]
    sg = np.zeros(n)
    for k in top_k:
        km = k - np.arange(n)
        valid = (km >= 0) & (km < n)
        sg[valid] += 2.0 * a[km[valid]]
    a_new = a * np.exp(-eta * sg)
    a_new = a_new / np.sum(a_new)
    return a_new

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    result = np.fft.irfft(fa * fb, size)
    return result[:n+m-1]

"""
Minimize 2*n*max(self-convolution)/sum(a)^2 using smooth mirror descent.
Uses log-sum-exp smoothing of the max operator, exponential weights for
non-negativity constraint, FFT-based convolutions for O(n log n) per step,
and multiple diverse initializations with random restarts.

import numpy as np
import time


def fft_convolve(a, b):
    """FFT-based convolution for speed on long sequences."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    result = np.fft.irfft(fa * fb, size)
    return result[:n + m - 1]


def compute_grad(a, weights):
    """
    Compute grad[m] = 2 * sum_k weights[k] * a[k-m]
    using FFT-based cross-correlation.
    weights: length 2n-1, a: length n
    """
    n = len(a)
    a_rev = a[::-1]
    conv = fft_convolve(weights, a_rev)
    # g[m] = conv[n-1+m]
    g = conv[n - 1:n - 1 + n]
    return 2.0 * g


def smooth_objective_grad(a, tau):
    """
    Compute value and gradient of smooth max:
    F(a) = (1/tau) * log(sum_k exp(tau * (a*a)[k]))
    Gradient: 2 * correlation(softmax(tau*(a*a)), a)
    """
    conv = fft_convolve(a, a)
    # Stable log-sum-exp
    cmax = conv.max()
    shifted = tau * (conv - cmax)
    exp_shifted = np.exp(shifted)
    log_sum_exp = cmax + np.log(exp_shifted.sum()) / tau
    # Softmax weights
    weights = exp_shifted / exp_shifted.sum()
    # Gradient
    grad = compute_grad(a, weights)
    return log_sum_exp, grad


def md_update(a, grad, eta):
    """Mirror descent (exponential weights) update on simplex."""
    # Clamp gradient for numerical safety
    grad_max = np.max(np.abs(grad))
    if grad_max > 1e-15:
        scale = min(eta, 2.0 / grad_max)
    else:
        scale = eta
    log_a = np.log(np.maximum(a, 1e-300)) - scale * grad
    log_a -= log_a.max()
    a_new = np.exp(log_a)
    s = a_new.sum()
    if s > 1e-300:
        a_new /= s
    else:
        a_new = np.ones_like(a) / len(a)
    return a_new


def true_score(a):
    """Compute the actual objective: 2n * max(conv) / sum(a)^2"""
    n = len(a)
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


def polish_subgradient(a, num_iters=5000, base_eta=0.05):
    """Polish with subgradient steps targeting top peaks directly."""
    n = len(a)
    idx = np.arange(n)
    for it in range(num_iters):
        conv = np.convolve(a, a)
        # Use top-5 peaks for subgradient
        top_k = np.argpartition(conv, -5)[-5:]
        sg = np.zeros(n)
        for k in top_k:
            km = k - idx
            valid = (km >= 0) & (km < n)

        log_a -= log_a.max()
        a_new = np.exp(log_a)
