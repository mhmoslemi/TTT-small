import numpy as np

def fast_convolve(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:size]

def fast_cross_corr_e_a(e, a):
    """
    Compute r[i] = sum_{j=0}^{n-1} e[i+j] * a[j] for i = 0, ..., n-1.
    e has length 2n-1, a has length n.
    """
    n = len(a)
    m = len(e)  # 2n-1
    N = 1 << (m + n - 2 - 1).bit_length()  # 3n-2
    A = np.fft.rfft(a, N)
    E = np.fft.rfft(e, N)
    R = np.fft.irfft(np.conj(A) * E, N)
    # r[i] = R[(N-i) % N]
    # r[0] = R[0], r[1] = R[N-1], ..., r[n-1] = R[N-n+1]
    r = np.empty(n, dtype=np.float64)
    r[0] = R[0]
    if n > 1:
        r[1:] = R[N-1:N-n:-1]
    return r

def compute_gradient(a, T):
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.zeros(n), 0.0
    
    c = fast_convolve(a)
    c_max = np.max(c)
    # For numerical stability, subtract c_max
    shifted = (c - c_max) / T
    # Clip to avoid overflow

    exp_c = np.exp(shifted)
    w_sum = np.sum(exp_c)
    SM = c_max + T * np.log(w_sum)
    
    corr = fast_cross_corr_e_a(exp_c, a)
    grad_SM = (2.0 / T) * corr / w_sum
    
    grad = (2.0 * n) / (S * S) * grad_SM - (4.0 * n * SM) / (S**3)
    
    return grad, SM

"""
Multi-phase optimizer for step-function coefficient sequences.
Phase 1: Try several parametric starting shapes, pick best.
Phase 2: Adam gradient descent with soft-max smooth approximation of max(convolve(a,a)),
         using FFT for O(n log n) convolution and cross-correlation.
Phase 3: Block-perturbation local search for final refinement.
"""

import numpy as np
import time


def fast_convolve(a):
    """FFT-based linear convolution of a with itself. Returns array of length 2*n-1."""
    n = a.size
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:size]


def fast_weighted_corr(e, a):
    """
    Compute r[i] = sum_{j=0}^{n-1} e[i+j] * a[j] for i = 0, ..., n-1.
    e has length 2*n-1, a has length n.
    Uses FFT cross-correlation: r[i] = (a star e)[-i].
    """
    n = a.size
    m = e.size
    N = 1 << (m + n - 2 - 1).bit_length()
    A = np.fft.rfft(a, N)
    E = np.fft.rfft(e, N)
    R = np.fft.irfft(np.conj(A) * E, N)
    r = np.empty(n, dtype=np.float64)
    r[0] = R[0]
    if n > 1:
        r[1:] = R[N - 1 : N - n : -1]
    return r


def compute_gradient(a, T):
    """
    Gradient of f(a) = 2n * SM / S^2 where SM = softmax(convolve(a,a), T).
    Returns (grad_vector, SM_value).
    """
    n = a.size
    S = a.sum()
    if S < 1e-10:
        return np.zeros(n, dtype=np.float64), 0.0

    c = fast_convolve(a)
    c_max_val = c.max()
    shifted = np.clip((c - c_max_val) / T, -400.0, 400.0)
    exp_c = np.exp(shifted)
    w_sum = exp_c.sum()
    SM = c_max_val + T * np.log(max(w_sum, 1e-300))

    corr = fast_weighted_corr(exp_c, a)
    grad_SM = (2.0 / T) * corr / max(w_sum, 1e-300)

    grad = (2.0 * n) / (S * S) * grad_SM - (4.0 * n * SM) / (S ** 3)
    return grad, SM


def block_perturb(base, rng, sigma=0.1):
    """Random block multiplicative perturbation, preserves non-negativity."""
    cand = base.copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 300), max(2, n // 10)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= np.exp(rng.normal(0.0, sigma, size=stop - start))
    if rng.random() < 0.3:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand
