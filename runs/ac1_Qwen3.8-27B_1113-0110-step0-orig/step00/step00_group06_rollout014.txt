```python
"""
Search algorithm:
- Use one-sided power-law profiles a_i proportional to (1 + i/beta)^(-alpha).
- These profiles give very flat anti-diagonal sums of the outer product a_i a_j
  and empirically place the maximum of np.convolve(a, a) well below the uniform
  baseline.
- Optimize alpha and beta with a fast FFT-based surrogate, then validate a small
  set of candidates with the exact evaluate_sequence function before returning.
"""

import numpy as np
import time

INF = float("inf")


def get_search_length():
    """Choose a sensible length, preferring the provided construction length."""
    n = 5582
    seq = globals().get("height_sequence_1")
    if seq is not None:
        try:
            L = len(seq)
            if isinstance(L, int) and L > 0:
                if 16 <= L <= 12000:
                    n = L
                elif L > 12000:
                    n = 5582
        except Exception:
            pass
    if n < 128:
        n = 1024
    if n > 12000:
        n = 12000
    return int(n)


def make_power_array(n, alpha, beta):
    """
    Build a normalized nonnegative power-law sequence.

    Returns a numpy array with sum 1, score-invariant scaling.
    """
    n = int(n)
    if n <= 0:
        return np.array([], dtype=np.float64)

    alpha = float(alpha)
    beta = float(beta)
    if not np.isfinite(alpha):
        alpha = 0.2
    if not np.isfinite(beta) or beta <= 1e-8:
        beta = 1.0

    i = np.arange(n, dtype=np.float64)
    arg = 1.0 + i / beta

    vals = np.power(arg, -alpha)
    if not np.all(np.isfinite(vals)):
        vals = np.nan_to_num(vals, nan=0.0, posinf=0.0, neginf=0.0)
    vals = np.maximum(vals, 0.0)

    s = float(np.sum(vals))
    if s > 0.0 and np.isfinite(s):
        vals = vals / s
    else:
        vals = np.ones(n, dtype=np.float64) / float(n)

    return vals


def fft_score(arr):
    """
    Fast surrogate for the exact score.

    The sequence is normalized to sum 1, so the score is 2*n*max(conv(a,a)).
    """
    arr = np.asarray(arr, dtype=np.float64)
    n = arr.size
    if n <= 0:
        return INF

    s = float(np.sum(arr))
    if not np.isfinite(s) or s <= 0.0:
        return INF

    a = arr / s
    target = 2 * n - 1
    m = 1
    while m < target:
        m <<= 1

    try:
        fa = np.fft.rfft(a, m)
        c = np.fft.irfft(fa * fa, m)[:target]
        mx = float(np.max(c))
        if not np.isfinite(mx):
            return INF
        if mx < 0.0:
            mx = 0.0
        return 2.0 * float(n) * mx
    except Exception:
        return INF


def exact_score_array(arr):
    """Use the external authoritative evaluator."""
    try:
        seq = [float(x) for x in arr]
        val = evaluate_sequence(seq)
        return float(val)
    except Exception:
        return INF


def local_refine_parameters(n, alpha, beta, best_score, deadline):
    """Small coordinate/greedy refinement around the best alpha/beta."""
    a = float(alpha)
    b = float(beta)
    bs = float(best_score)

    alpha_steps = (0.035, 0.022, 0.014, 0.008, 0.004, 0.002, 0.001, 0.0005)
    beta_factors = (0.80, 0.90, 0.95, 1.00, 1.05, 1.10, 1.25)

    for step in alpha_steps:
        if time.monotonic() > deadline:
            break
        for da in (-step, 0.0, step):
            a2 = a + da
            if a2 < 0.02 or a2 > 0.48:
                continue
            for bf in beta_factors:
                b2 = b * bf
                if b2 < 0.05 or b2 > 12.0:
                    continue
                arr = make_power_array(n, a2, b2)
                sc = fft_score(arr)
                if np.isfinite(sc) and sc < bs:
                    bs = sc
                    a = a2
                    b = b2
    return a, b, bs


def optimize_parameters(n, deadline, init_arr):
    """Search over the power-law family using the FFT surrogate."""
    best_arr = np.asarray(init_arr, dtype=np.float64).copy()
    best_alpha = 0.20
    best_beta = 1.00
    best_score = fft_score(best_arr)

    alpha_list = np.array([
        0.05, 0.08, 0.10, 0.12, 0.14, 0.16, 0.17, 0.18, 0.19, 0.20,
        0.21, 0.22, 0.24, 0.26, 0.28, 0.30, 0.33, 0.36, 0.39, 0.42, 0.45
    ], dtype=np.float64)

    beta_list = np.array([
        0.20, 0.30, 0.40, 0.50, 0.60, 0.75, 0.90, 1.00, 1.10,
        1.25, 1.50, 2.00, 3.00
    ], dtype=np.float64)

    top_candidates = []

    for alpha in alpha_list:
        if time.monotonic() > deadline:
            break
        for beta in beta_list:
            arr = make_power_array(n, float(alpha), float(beta))
            sc = fft_score(arr)
            if np.isfinite(sc):
                top_candidates.append((sc, float(alpha), float(beta)))
                if sc < best_score:
                    best_score = sc
                    best_alpha = float(alpha)
                    best_beta = float(beta)
                    best_arr = arr

    top_candidates.sort()
    if top_candidates:
        sc0, a0, b0 = top_candidates[0]
        if np.isfinite(sc0) and sc0 < best_score:
            best_score = sc0
            best_alpha = a0
            best_beta = b0
            best_arr = make_power_array(n, a0, b0)

    if time.monotonic() < deadline - 1.0:
        best_alpha, best_beta, best_score = local_refine_parameters(
            n, best_alpha, best_beta, best_score, deadline
        )
        best_arr = make_power_array(n, best_alpha, best_beta)

    return best_arr, best_score, best_alpha, best_beta, top_candidates[:30]


def validate_candidates(n, deadline, alpha, beta, top_candidates, best_seq, best_score):
    """Validate promising power-law candidates with the exact evaluator."""
    candidates = []
    candidates.append((float(alpha), float(beta)))

    for _sc, a, b in top_candidates[:15]:
        candidates.append((float(a), float(b)))

    # Very small exact local grid around the FFT best point.
    for da in (-0.012, -0.006, 0.0, 0.006, 0.012):
        for bf in (0.92, 1.00, 1.08):
            candidates.append((float(alpha) + da, float(beta) * bf))

    unique = []
    for a, b in candidates:
        if not (np.isfinite(a) and np.isfinite(b)):
            continue
        if a < 0.02 or a > 0.48:
            continue
        if b < 0.05 or b > 12.0:
            continue
        duplicate = False
        for ua, ub in unique:
            if abs(ua - a) < 1e-12 and abs(ub - b) < 1e-12:
                duplicate = True
                break
        if not duplicate:
            unique.append((a, b))

    max_exact_evals = 8 if n > 8000 else 28
    for idx, (a, b) in enumerate(unique):
        if idx >= max_exact_evals:
            break
        if time.monotonic() > deadline:
            break
        arr = make_power_array(n, a, b)
        sc = exact_score_array(arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr
            print("exact improvement:", sc, "alpha=", a, "beta=", b)

    return best_seq, best_score


def propose_candidate(seed=123, budget_s=1800, **kwargs):
    """
    Return the best sequence found within the time budget.
    """
    start = time.monotonic()
    budget = max(0.0, float(budget_s))
    deadline = start + max(0.0, budget - 5.0)

    n = get_search_length()

    # A very safe fallback profile.
    fallback_alpha = 0.20
    fallback_beta = 1.00
    fallback = make_power_array(n, fallback_alpha, fallback_beta)

    try:
        print("search length:", n)

        init_arr = fallback
        best_arr, best_fast, best_alpha, best_beta, top = optimize_parameters(

        best_seq = np.asarray(best_arr, dtype=np.float64).copy()
        best_exact = exact_score_array(best_seq)
        print("exact score after validation:", best_exact)

        if time.monotonic() < deadline:
            best_seq, best_exact = validate_candidates(
                n, deadline, best_alpha, best_beta, top, best_seq, best_exact
            )
            print("validated exact score:", best_exact)

        fallback_exact = exact_score_array(fallback)
        if (not np.isfinite(best_exact)) or np.isfinite(fallback_exact):
            if (
                not np.isfinite(best_exact)
                or np.isfinite(fallback_exact)
                and fallback_exact < best_exact
            ):
                best_seq = fallback
                best_exact = fallback_exact
                print("using fallback exact score:", best_exact)

        return [float(x) for x in best_seq]

    except Exception:
        # In any unexpected situation, return a valid nonnegative power law.
        return [float(x) for x in fallback]
```
