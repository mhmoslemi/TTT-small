import numpy as np
import time

"""
Algorithm: Soft-max gradient descent on the simplex to minimize the peak
autocorrelation ratio. We work with sequences normalized to sum=1 (the
objective is scale-invariant). The soft-max of the convolution provides a
smooth surrogate for the max, giving a dense gradient that avoids stagnation
at symmetric points. We use projected gradient steps with backtracking,
adaptive softness (beta), and random perturbations to escape local minima.
"""


def proj_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    k = ind[cond].max()
    theta = cssv[k - 1] / k
    return np.maximum(v - theta, 0)


def compute_soft_gradient(a, beta):
    """Compute gradient of beta*log(sum(exp(conv(a,a)/beta))) w.r.t. a.
    Uses FFT-free O(n^2) convolution for the cross-correlation."""
    n = a.size
    conv = np.convolve(a, a)
    peak = conv.max()
    # Numerically stable softmax
    shifted = (conv - peak) / beta
    # Clip to avoid underflow issues
    shifted = np.clip(shifted, -700, 0)
    weights = np.exp(shifted)
    total = weights.sum()
    if total < 1e-300:
        # Fallback: uniform over top few
        top_k = np.argsort(conv)[-n:]
        weights = np.zeros_like(conv)
        weights[top_k] = 1.0 / n
    else:
        weights /= total
    # grad[i] = 2 * sum_k weights[k] * a[k-i]
    # = 2 * cross_correlation[weights, a][i]
    # Computed as convolution with reversed a
    a_rev = a[::-1]
    corr = np.convolve(weights, a_rev, mode='full')
    # corr[m] = sum_k weights[k]*a_rev[m-k] = sum_k weights[k]*a[n-1-m+k]
    # We want sum_k weights[k]*a[k-i], so n-1-m+k = k-i => m = n-1+i
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, peak


def exp_gradient_step(a, grad, eta):
    """Exponentiated gradient (mirror descent) step on the simplex."""
    a_new = a * np.exp(-eta * grad)
    s = a_new.sum()
    if s < 1e-300:
        return a
    return a_new / s

    S = a.sum()
    if S < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max() / (S * S)
