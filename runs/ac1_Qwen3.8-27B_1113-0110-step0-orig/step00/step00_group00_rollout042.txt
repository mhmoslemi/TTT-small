    a = a / s

    c = _fft_convolve(a)
    best_score = 2.0 * n * float(np.max(c))
    best_a = a.copy()

    eta = 0.15
    no_improve = 0
    momentum = np.zeros(n)
    mu = 0.7
    deadline = time.monotonic() + time_budget
    it = 0
    last_print = time.monotonic()

    while time.monotonic() < deadline:
        c = _fft_convolve(a)
        k_max = int(np.argmax(c))
        score = 2.0 * n * c[k_max]

        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            if time.monotonic() - last_print > 10.0:
                print(f"    [opt] it={it}, score={score:.6f}, n={n}")
                last_print = time.monotonic()
        else:
            no_improve += 1

        if no_improve > 800:
            # Restart from best with multiplicative noise
            a = best_a * np.exp(rng.normal(0, 0.08, n))
            a = np.maximum(a, 1e-20)
            a = a / a.sum()
            no_improve = 0
            eta = 0.15
            momentum = np.zeros(n)
            it += 1
            continue

        # Compute subgradient with momentum smoothing
        g = _subgradient_vec(a, k_max)
        g_max = g.max()
        if g_max < 1e-15:
            break
        g_norm = g / g_max
        momentum = mu * momentum + (1.0 - mu) * g_norm

        # Multiplicative update: reduce a[j] more where gradient is high
        a = a * np.exp(-eta * momentum)
        a = np.maximum(a, 1e-20)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            eta *= 0.5
            it += 1
            continue
        a = a / s

        # Adaptive step size
        if no_improve > 400:
            eta *= 0.97
        elif no_improve > 100:
            eta *= 0.99
        elif no_improve < 30:
            eta = min(eta * 1.003, 0.4)

        # Occasional additive perturbation for diversity
        if it % 200 == 100 and rng.random() < 0.2:
            block = max(1, n // 50)
            start = int(rng.integers(0, n - block))
            a[start:start + block] *= np.exp(rng.normal(0, 0.15))
            a = np.maximum(a, 1e-20)
            a = a / a.sum()

        it += 1

    return best_a, best_score


def _optimize_additive(a0, time_budget, rng):
    """Optimize using additive subgradient descent with projection."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, np.inf
    a = a / s

    c = _fft_convolve(a)
    best_score = 2.0 * n * float(np.max(c))
    best_a = a.copy()

    eta = 0.05
    no_improve = 0
    momentum = np.zeros(n)
    mu = 0.8
    deadline = time.monotonic() + time_budget
    it = 0
    last_print = time.monotonic()

    while time.monotonic() < deadline:
        c = _fft_convolve(a)
        k_max = int(np.argmax(c))
        score = 2.0 * n * c[k_max]

        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            if time.monotonic() - last_print > 10.0:
                print(f"    [add] it={it}, score={score:.6f}, n={n}")
                last_print = time.monotonic()
        else:
            no_improve += 1

        if no_improve > 600:
            a = best_a.copy()
            a += rng.normal(0, 0.03, n)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            else:
                a = best_a.copy()
            no_improve = 0
            eta = 0.05
            momentum = np.zeros(n)
            it += 1
            continue

        g = _subgradient_vec(a, k_max)
        g_max = g.max()
        if g_max < 1e-15:
            break
        g_norm = g / g_max
        momentum = mu * momentum + (1.0 - mu) * g_norm

        a = a - eta * momentum
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            eta *= 0.5
            a = best_a.copy()
            it += 1
            continue
        a = a / s

        if no_improve > 300:
            eta *= 0.96
        elif no_improve < 50:
            eta = min(eta * 1.01, 0.3)

        it += 1

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Minimizes 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
    
    Strategy: FFT-accelerated subgradient descent (both multiplicative and additive
    variants) with momentum, adaptive step sizes, and random restarts. Tries multiple
    initializations (ramp, off-center Gaussian, V-shaped, block, and provided sequence)
    across several sequence lengths to find the global best.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 15.0  # 15s safety margin

    best_overall = None
    best_overall_score = np.inf

    # Build candidate initializations
    candidates = []  # list of (n, a0_array)

    # 1. Provided sequence (if available)
    try:
        hs = np.array(height_sequence_1, dtype=np.float64)
        hs = np.clip(hs, 0, None)
        if hs.size > 100 and hs.sum() > 0.01:
            candidates.append(("provided", hs.copy()))
    except NameError:
        pass
    except Exception:
        pass

    # 2. Ramp sequences (linearly increasing)
    for n in [1500, 2000, 3000, 4000, 5000]:
        a = np.arange(1, n + 1, dtype=np.float64)
        candidates.append((f"ramp_{n}", a))

    # 3. Quadratic ramp (more aggressive asymmetry)
    for n in [2000, 4000]:
        x = np.arange(n, dtype=np.float64)
        a = (x + 1.0) ** 2
        candidates.append((f"quad_{n}", a))

    # 4. Off-center Gaussian
    for n in [2000, 4000]:
        x = np.arange(n, dtype=np.float64)
        for center_frac in [0.25, 0.35]:
            center = n * center_frac
            sigma = n * 0.12
            a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
            candidates.append((f"gauss_{n}_{int(center_frac*100)}", a))

    # 5. Block sequence (high first half, low second half)
    for n in [2000, 4000, 5000]:
        a = np.ones(n, dtype=np.float64)
        a[:n * 2 // 3] = 2.0
        a[n * 2 // 3:] = 0.3
        candidates.append((f"block_{n}", a))

    # 6. Smooth asymmetric (exponential decay from one end)
    for n in [2000, 4000]:
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-0.8 * x / n)
        a = a + 0.1  # avoid zeros
        candidates.append((f"decay_{n}", a))

    # 7. Sinusoidal modulation of uniform
    for n in [2000, 4000]:
        x = np.arange(n, dtype=np.float64)
        a = 1.0 + 0.4 * np.sin(2 * np.pi * x / n + np.pi / 4)
        a = np.maximum(a, 0.05)
        candidates.append((f"sin_{n}", a))

    print(f"Total candidates: {len(candidates)}")

    # Quick initial evaluation to rank candidates
    scored = []
    for name, a0 in candidates:
        a0 = np.maximum(a0, 0)
        s = a0.sum()
        if s < 1e-15:
            continue
        a0 = a0 / s
        c0 = _fft_convolve(a0)
        init_score = 2.0 * len(a0) * float(np.max(c0))
        scored.append((init_score, name, a0))

    scored.sort(key=lambda x: x[0])
    print("Initial scores (sorted):")
    for i, (sc, name, _) in enumerate(scored[:8]):
        print(f"  {i+1}. {name}: {sc:.6f}")

    # Allocate time to top candidates
    # Give more time to better initial candidates
    n_top = min(len(scored), 8)
    remaining_time = budget_s - 20.0
    # Geometric allocation: top gets most time
    weights = [max(1.0, n_top - i) for i in range(n_top)]
    total_w = sum(weights)
    times = [remaining_time * w / total_w for w in weights]

    for i in range(n_top):
        if time.monotonic() > deadline:
            break

        init_score, name, a0 = scored[i]
        t_alloc = min(times[i], max(5.0, deadline - time.monotonic()))

        print(f"\nOptimizing '{name}' (n={len(a0)}, init_score={init_score:.6f}, time={t_alloc:.0f}s)")

        # Try both multiplicative and additive, split time
        t_half = t_alloc / 2.0

        # Multiplicative subgradient
        a1, s1 = _optimize_subgrad(a0, t_half, rng)
        eval1 = evaluate_sequence(a1.tolist())
        print(f"  Multiplicative: internal={s1:.6f}, eval={eval1:.6f}")

        if np.isfinite(eval1) and eval1 < best_overall_score:
            best_overall_score = eval1
            best_overall = a1.copy()
            print(f"  *** NEW BEST: {eval1:.6f} ***")

        # Additive subgradient (from best so far or original)
        start2 = a1 if np.isfinite(s1) and s1 < init_score else a0
        a2, s2 = _optimize_additive(start2, t_half, rng)
        eval2 = evaluate_sequence(a2.tolist())
        print(f"  Additive: internal={s2:.6f}, eval={eval2:.6f}")

        if np.isfinite(eval2) and eval2 < best_overall_score:
            best_overall_score = eval2
            best_overall = a2.copy()
            print(f"  *** NEW BEST: {eval2:.6f} ***")

    # If time remains, do final polishing from the best solution
    remaining = deadline - time.monotonic()
    if best_overall is not None and remaining > 30:
        print(f"\nFinal polishing with {remaining:.0f}s remaining...")
        a_polished, sp = _optimize_subgrad(best_overall, remaining * 0.6, rng)
        eval_p = evaluate_sequence(a_polished.tolist())
        print(f"  Polished (mult): {eval_p:.6f}")
        if np.isfinite(eval_p) and eval_p < best_overall_score:
            best_overall_score = eval_p
            best_overall = a_polished.copy()

        remaining2 = deadline - time.monotonic()
        if remaining2 > 10:
            a_polished2, sp2 = _optimize_additive(best_overall, remaining2, rng)
            eval_p2 = evaluate_sequence(a_polished2.tolist())
            print(f"  Polished (add): {eval_p2:.6f}")
            if np.isfinite(eval_p2) and eval_p2 < best_overall_score:
                best_overall_score = eval_p2
                best_overall = a_polished2.copy()

    # Final verification
    if best_overall is not None:
        final_score = evaluate_sequence(best_overall.tolist())
        elapsed = time.monotonic() - total_start
        print(f"\nFinal result: score={final_score:.6f}, n={len(best_overall)}, elapsed={elapsed:.1f}s")
        return [float(x) for x in best_overall]
    else:
        # Fallback
        a = np.ones(1000, dtype=np.float64)
        print("Falling back to uniform sequence")
        return a.tolist()
