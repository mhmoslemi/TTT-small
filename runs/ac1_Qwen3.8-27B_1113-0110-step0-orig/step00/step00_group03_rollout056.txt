        for r in [1.0, 2.0, 3.0, 5.0, 8.0]:
            cands.append(np.exp(-r * xn))
        for p in [0.5, 1.0, 2.0, 3.0]:
            cands.append(np.maximum(0.0, 1.0 - xn) ** p)
        cands.append(1.0 / (1.0 + 2.0 * xn))
        cands.append(1.0 / (1.0 + 5.0 * xn))
        cands.append(0.5 + 0.5 * np.cos(np.pi * xn))
        cands.append(1.0 - xn ** 2)
        cands.append(xn)
        cands.append(xn ** 0.5)
        if n >= 2:
            cands.append((x // 2 + 1).astype(np.float64))
        cands.append(0.05 + 0.95 * np.exp(-4.0 * xn))
        cands.append(np.exp(-2.0 * (xn - 0.3) ** 2) + 0.5 * np.exp(-2.0 * (xn - 0.7) ** 2))
        cands.append(1.0 / (1.0 + xn * n * 0.1))

        for c in cands:
            c = np.clip(c, 0.0, 1000.0)
            if np.sum(c) < 0.01:
                continue
            s = safe_evaluate(c)
            if np.isfinite(s) and s < best_score:
                best = c.copy()
                best_score = s
                param_starts.append(c.copy())
                if len(param_starts) > 30:
                    param_starts.pop(0)
                print(f"  n={n}, score={s:.6f}")

    # --- Phase 2: LP refinement on diverse starting points ---
    print(f"[phase2] LP refinement (best so far={best_score:.6f})...")

    lp_starts = []
    if best is not None:
        lp_starts.append(best.copy())
        n_b = len(best)
        x_b = np.arange(n_b, dtype=np.float64) / max(1, n_b - 1)
        for taper_fn in [lambda x: 1.0 - 0.3 * x,
                         lambda x: 1.0 - 0.5 * x,
                         lambda x: np.exp(-0.5 * x),
                         lambda x: np.exp(-1.0 * x),
                         lambda x: 0.5 + 0.5 * np.cos(np.pi * x)]:
            t = best * taper_fn(x_b)
            t = np.clip(t, 0.0, 1000.0)
            if np.sum(t) >= 0.01:
                lp_starts.append(t)

    # Add some parametric starts that scored well
    for ps in param_starts[:10]:
        if ps is not None and np.sum(ps) >= 0.01:
            lp_starts.append(ps.copy())

    # Deduplicate by length
    seen_lengths = set()
    unique_starts = []
    for s_ in lp_starts:
        if len(s_) not in seen_lengths or safe_evaluate(s_) < 1.9:
            seen_lengths.add(len(s_))
            unique_starts.append(s_)

    for idx, start_f in enumerate(unique_starts):
        if time.monotonic() > time_budget_global - 100:
            break
        if len(start_f) < 2:
            continue

        remaining = time_budget_global - time.monotonic()
        alloc = min(120.0, remaining * 0.5 / max(1, len(unique_starts) - idx))
        if alloc < 5.0:
            break

        res_f, res_s, improved = lp_refine(start_f, max_iters=150, time_limit=alloc)

        if np.isfinite(res_s) and res_s < best_score:
            best = res_f.copy()
            best_score = res_s
            print(f"  LP[{idx}] n={len(res_f)}: {best_score:.6f} (improved={improved})")

    # --- Phase 3: Continued LP on the single best ---
    if best is not None and len(best) <= 6000 and time.monotonic() < time_budget_global - 200:
        print(f"[phase3] Focused LP on best (score={best_score:.6f})...")
        res_f, res_s, _ = lp_refine(best, max_iters=500,
                                    time_limit=time_budget_global - time.monotonic() - 300)
        if np.isfinite(res_s) and res_s < best_score:
            best = res_f.copy()
            best_score = res_s
            print(f"  Focused LP: {best_score:.6f}")

    # --- Phase 4: Random local search ---
    print(f"[phase4] Local search (best={best_score:.6f})...")
    if best is not None:
        f = best.copy().astype(np.float64)
        n = len(f)
        step = 0
        stall = 0

        while time.monotonic() < time_budget_global:
            step += 1

            # Adaptive perturbation
            sigma = 0.05 + 0.05 * min(stall / 20.0, 1.0)
            candidate = f.copy()
            n_blocks = int(rng.integers(1, max(2, n // 500 + 1)))
            for _ in range(n_blocks):
                min_w = max(1, n // 150)
                max_w = max(min_w + 1, n // 8)
                w = int(rng.integers(min_w, max_w))
                st = int(rng.integers(0, max(1, n - w + 1)))
                en = min(n, st + w)
                candidate[st:en] *= np.exp(rng.normal(0.0, sigma, size=en - st))

            candidate = np.clip(candidate, 0.0, 1000.0)
            if np.sum(candidate) < 0.01:
                continue

            s = safe_evaluate(candidate)
            if np.isfinite(s) and s < best_score:
                best = candidate.copy()
                best_score = s
                f = candidate.copy()
                stall = 0
                if step % 10 == 1:
                    print(f"  local[{step}]: {best_score:.6f}")
            else:
                stall += 1

            # Periodic larger jumps
            if stall > 50 and step % 25 == 0:
                candidate = f * np.exp(rng.normal(0, 0.2, size=n))
                candidate = np.clip(candidate, 0.0, 1000.0)
                if np.sum(candidate) >= 0.01:
                    s2 = safe_evaluate(candidate)
                    if np.isfinite(s2) and s2 < best_score:
                        best = candidate.copy()
                        best_score = s2
                        f = candidate.copy()
                        stall = 0
                        print(f"  jump[{step}]: {best_score:.6f}")

            # Occasional fresh parametric trial
            if step % 200 == 0:
                n_try = int(rng.integers(3, min(500, n + 1)))
                xt = np.arange(n_try, dtype=np.float64) / max(1, n_try - 1)
                shape = int(rng.integers(0, 5))
                if shape == 0:
                    cand2 = np.exp(-float(rng.uniform(0.5, 6.0)) * xt)
                elif shape == 1:
                    cand2 = np.maximum(0.0, 1.0 - float(rng.uniform(0.3, 1.0)) * xt)
                elif shape == 2:
                    cand2 = 1.0 / (1.0 + float(rng.uniform(0.5, 10.0)) * xt)
                elif shape == 3:
                    cand2 = (np.maximum(0.0, 1.0 - xt)) ** float(rng.uniform(0.5, 4.0))
                else:
                    cand2 = 0.5 + 0.5 * np.cos(np.pi * rng.uniform(0.0, 1.5) * xt)
                cand2 = np.clip(cand2, 0.0, 1000.0)
                if np.sum(cand2) >= 0.01:
                    s3 = safe_evaluate(cand2)
                    if np.isfinite(s3) and s3 < best_score:
                        best = cand2.copy()
                        best_score = s3
                        f = cand2.copy()
                        n = n_try
                        stall = 0
                        print(f"  param[{step}] n={n_try}: {best_score:.6f}")

    # --- Final output ---
    if best is None:
        best = np.ones(4, dtype=np.float64)
    best = np.clip(np.asarray(best, dtype=np.float64), 0.0, 1000.0)
    if not np.all(np.isfinite(best)) or np.sum(best) < 0.01:
        best = np.ones(4, dtype=np.float64)

    final_score = safe_evaluate(best)
    print(f"[FINAL] n={len(best)}, score={final_score:.6f}")
    return [float(x) for x in best]
