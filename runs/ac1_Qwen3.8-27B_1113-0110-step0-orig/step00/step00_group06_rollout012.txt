import numpy as np
import time
from scipy.optimize import linprog
from scipy.linalg import toeplitz

def compute_score(s):
    """Compute the score without calling evaluate_sequence (for speed)."""
    s = np.maximum(0, s)
    s = np.minimum(1000, s)
    n = len(s)
    conv = np.convolve(s, s)
    max_b = np.max(conv)
    sum_a = np.sum(s)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)

def make_step_sequence(n, frac, ratio):
    """Create a step sequence: first frac*n elements are high, rest are low."""
    a = int(frac * n)
    b = n - a
    H, L = ratio, 1.0
    s = np.zeros(n)
    s[:a] = H
    s[a:] = L
    return s

def make_ramp_sequence(n, ratio):
    """Create a linear ramp from ratio to 1."""
    s = np.linspace(ratio, 1.0, n)
    return s

def make_exp_sequence(n, alpha):
    """Create an exponentially decaying sequence."""
    s = np.exp(-alpha * np.arange(n) / n)
    return s

def local_search(s, rng, n_steps, base_score):
    """Perform local search with various perturbations."""
    best = s.copy()
    best_score = base_score
    n = len(best)
    
    for step in range(n_steps):
        # Choose perturbation type
        r = rng.random()
        
        if r < 0.3:
            # Block perturbation
            width = int(rng.integers(max(1, n//100), max(2, n//10)))
            start = int(rng.integers(0, max(1, n - width)))
            end = min(n, start + width)
            candidate = best.copy()
            candidate[start:end] *= np.exp(rng.normal(0, 0.2, end-start))
        elif r < 0.5:
            # Element perturbation
            indices = rng.integers(0, n, size=max(1, n//50))
            candidate = best.copy()
            candidate[indices] *= np.exp(rng.normal(0, 0.3, len(indices)))
        elif r < 0.7:
            # Peak reduction: reduce elements contributing to max conv
            conv = np.convolve(best, best)
            max_pos = np.argmax(conv)
            # Elements that contribute to conv[max_pos]
            indices = list(range(max(0, max_pos-n+1), min(n, max_pos+1)))
            candidate = best.copy()
            for i in indices:
                if candidate[i] > 0:
                    candidate[i] *= (1 - rng.uniform(0.01, 0.1))
        elif r < 0.85:
            # Valley filling: increase small elements
            candidate = best.copy()
            small_indices = np.where(candidate < np.mean(candidate) * 0.5)[0]
            if len(small_indices) > 0:
                selected = rng.choice(small_indices, size=min(len(small_indices), max(1, n//50)), replace=False)
                candidate[selected] *= np.exp(rng.uniform(0.1, 0.5, len(selected)))
        else:
            # Mirror perturbation
            candidate = best.copy()
            candidate = 0.5 * (candidate + candidate[::-1] * rng.uniform(0.5, 1.5))
        
        candidate = np.clip(candidate, 0, 1000)
        score = compute_score(candidate)
        if score < best_score:
            best = candidate
            best_score = score
    
    return best, best_score

def lp_refinement(s, n_iters, deadline):

    n = len(s)
    s = s.copy()
    # Normalize
    s = s / np.sum(s) * np.sqrt(2*n)

        M = np.max(np.convolve(s, s))

        # Actually: conv(s, b)[k] = sum_j s[j] * b[k-j] for valid j

        s_new = np.maximum(s_new, 0.0)
        # Renormalize
        if np.sum(s_new) > 1e-10:
            s_new = s_new / np.sum(s_new) * sum_s
        
        score = _compute_score(s_new)
        if score < best_score:
            s = s_new
            best_score = score
            lr = min(lr * 1.05, 0.1)
        else:
            lr *= 0.9
            T = max(T_min, T * 0.95)
        
        # Skip ahead if too slow
        if time.monotonic() > deadline:
            break
    
    return s, best_score

def _local_search(s, rng, budget_s):
    """Adaptive local search with multiple perturbation strategies."""
    n = len(s)
    best = s.copy()
    best_score = _compute_score(best)
    deadline = time.monotonic() + budget_s
    no_improve = 0
    step_scale = 0.15
    
    while time.monotonic() < deadline:
        r = rng.random()
        
        if r < 0.25:
            # Block multiplicative perturbation
            width = int(rng.integers(max(1, n // 200), max(2, n // 8)))
            start = int(rng.integers(0, max(1, n - width)))
            end = min(n, start + width)
            cand = best.copy()
            cand[start:end] *= np.exp(rng.normal(0, step_scale, end - start))
            
        elif r < 0.40:
            # Targeted peak reduction
            conv = np.convolve(best, best)
            # Find top few peaks
            top_k = np.argsort(conv)[-min(10, len(conv)//10):]
            cand = best.copy()
            for pos in top_k:
                # Elements contributing to conv[pos]
                j_min = max(0, pos - n + 1)
                j_max = min(n - 1, pos)
                if j_min <= j_max:
                    # Slightly reduce contributing elements
                    for j in range(j_min, j_max + 1):
                        if cand[j] > 1e-6:
                            cand[j] *= (1.0 - rng.uniform(0.005, 0.05))
            
        elif r < 0.55:
            # Increase "valley" elements (those not contributing to peaks)
            conv = np.convolve(best, best)
            peak_val = np.max(conv)
            # Elements with small contribution
            mean_val = np.mean(best)
            low_idx = np.where(best < mean_val * 0.7)[0]
            if len(low_idx) > 0:
                n_sel = min(len(low_idx), max(1, n // 30))
                sel = rng.choice(low_idx, n_sel, replace=False)
                cand = best.copy()
                cand[sel] *= np.exp(rng.uniform(0.05, 0.3, n_sel))
            else:
                cand = best * np.exp(rng.normal(0, step_scale * 0.5, n))
                
        elif r < 0.70:
            # Asymmetric scaling: scale first half up, second half down (or vice versa)
            mid = n // 2
            cand = best.copy()
            direction = rng.choice([-1, 1])
            s1 = rng.uniform(0.02, 0.15)
            if direction > 0:
                cand[:mid] *= (1 + s1)
                cand[mid:] *= (1 - s1 * 0.7)
            else:
                cand[:mid] *= (1 - s1 * 0.7)
                cand[mid:] *= (1 + s1)
                
        elif r < 0.80:
            # Smooth perturbation (low-frequency)
            k = int(rng.integers(1, max(2, n // 20)))
            phase = rng.uniform(0, 2 * np.pi)
            pert = 1.0 + 0.1 * np.sin(2 * np.pi * k * np.arange(n) / n + phase)
            cand = best * pert
            
        elif r < 0.90:
            # Random element perturbations
            n_mod = max(1, int(n * rng.uniform(0.05, 0.2)))
            idx = rng.integers(0, n, n_mod)
            cand = best.copy()
            cand[idx] *= np.exp(rng.normal(0, step_scale, n_mod))
            
        else:
            # Mirror blend
            cand = (1 - 0.3) * best + 0.3 * best[::-1]
            cand *= np.exp(rng.normal(0, 0.05, n))
        
        cand = np.clip(cand, 0, 1000)
        
        t0 = time.monotonic()
        score = _compute_score(cand)
        
        if score < best_score:
            best = cand
            best_score = score
            no_improve = 0
            step_scale = min(step_scale * 1.1, 0.5)
        else:
            no_improve += 1
            if no_improve > 100:
                step_scale *= 0.8
                no_improve = 0
    
    return best, best_score

def _lp_refinement(s, budget_s):
    """LP-based refinement for smaller sequences."""
    n = len(s)
    if n > 3000:
        # Too large for comfortable LP, downsample
        # Actually just skip
        print("LP: n too large, skipping")
        return s, _compute_score(s)
    
    best = s.copy()
    best_score = _compute_score(best)
    n = len(best)
    deadline = time.monotonic() + budget_s
    
    # Build structural indices for the conv matrix (structure doesn't change)
    row_indices = []
    col_indices = []
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        ii = np.arange(i_min, i_max + 1)
        row_indices.extend([k] * len(ii))
        col_indices.extend(ii.tolist())
    row_indices = np.array(row_indices)
    col_indices = np.array(col_indices)
    
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    
    t_step = 0.4
    for it in range(500):
        if time.monotonic() > deadline:
            break
        
        M = np.max(np.convolve(best, best))
        
        # Build sparse matrix with current s values
        data = best[col_indices]
        A = csr_matrix((data, (row_indices, col_indices)), shape=(2 * n - 1, n))
        
        t0 = time.monotonic()
        res = linprog(c_obj, A_ub=A, b_ub=np.full(2 * n - 1, M), bounds=bounds,
                      method='highs', options={'time_limit': max(1, (deadline - time.monotonic()) * 0.5)})
        
        if not res.success:
            t_step *= 0.7
            continue
        
        b = res.x
        sum_b = np.sum(b)
        if sum_b < 1e-10:
            t_step *= 0.7
            continue
        
        sum_s = np.sum(best)
        g = b / sum_b * sum_s
        
        # Try different step sizes
        improved = False
        for t in [t_step, t_step * 0.5, t_step * 0.2]:
            s_new = (1 - t) * best + t * g
            s_new = np.maximum(s_new, 0)
            sc = _compute_score(s_new)
            if sc < best_score:
                best = s_new
                best_score = sc
                improved = True
                break
        
        if not improved:
            t_step *= 0.7
        
        elapsed = time.monotonic() - t0
        
    return best, best_score

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main search function combining multiple optimization phases."""
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(1.0, float(budget_s) - 2.0)
    
    # Phase 1: Initialize
    print("=== Phase 1: Initialization ===")
    best, best_score = _phase1_initialize(rng)
    print(f"Best after init: {best_score:.6f} (n={len(best)})")
    
    # Phase 2: Gradient descent (short, for warm-up)
    t_remaining = total_deadline - time.monotonic()
    if t_remaining > 20:
        print("=== Phase 2: Gradient Descent ===")
        gd_budget = min(60.0, t_remaining * 0.1)
        s_gd, score_gd = _gradient_descent(best, gd_budget)
        if score_gd < best_score:
            best, best_score = s_gd, score_gd
            print(f"Best after GD: {best_score:.6f}")
    
    # Phase 3: Local search (main workhorse)
    t_remaining = total_deadline - time.monotonic()
    if t_remaining > 10:
        print("=== Phase 3: Local Search ===")
        ls_budget = min(t_remaining * 0.5, 600.0)
        best, best_score = _local_search(best, rng, ls_budget)
        print(f"Best after local search: {best_score:.6f} (n={len(best)})")
    
    # Phase 4: Try different n values from the best structure
    t_remaining = total_deadline - time.monotonic()
    if t_remaining > 30 and len(best) > 500:
        print("=== Phase 4: Length adaptation ===")
        for target_n in [max(100, len(best) // 2), max(200, len(best) * 3 // 2)]:
            if time.monotonic() > total_deadline:
                break
            # Resample
            x_old = np.linspace(0, 1, len(best))
            x_new = np.linspace(0, 1, target_n)
            s_resampled = np.interp(x_new, x_old, best)
            s_resampled = np.clip(s_resampled, 0, 1000)
            
            # Quick local search
            ls_b = min(30.0, (total_deadline - time.monotonic()) * 0.3)
            s_ls, sc_ls = _local_search(s_resampled, rng, ls_b)
            if sc_ls < best_score:
                best, best_score = s_ls, sc_ls
                print(f"  n={target_n}: score={best_score:.6f} (improved!)")
            else:
                print(f"  n={target_n}: score={sc_ls:.6f} (no improvement)")
    
    # Phase 5: LP refinement (if feasible)
    t_remaining = total_deadline - time.monotonic()
    if t_remaining > 30 and len(best) <= 3000:
        print("=== Phase 5: LP Refinement ===")
        lp_budget = min(120.0, t_remaining * 0.4)
        best, best_score = _lp_refinement(best, lp_budget)
        print(f"Best after LP: {best_score:.6f}")
    
    # Phase 6: Final fine-grained local search
    t_remaining = total_deadline - time.monotonic()
    if t_remaining > 5:
        print("=== Phase 6: Final Local Search ===")
        best, best_score = _local_search(best, rng, t_remaining - 1.0)
        print(f"Final best: {best_score:.6f} (n={len(best)})")
    
    # Verify with actual evaluator
    final_score = float(evaluate_sequence([float(x) for x in best]))
    if not np.isfinite(final_score) or final_score > best_score:
        # Fallback: use all-ones (score = 2)
        best = np.ones(256)
        final_score = float(evaluate_sequence([float(x) for x in best]))
    
    print(f"=== FINAL: score={final_score:.6f}, n={len(best)} ===")
    return [float(x) for x in best]
