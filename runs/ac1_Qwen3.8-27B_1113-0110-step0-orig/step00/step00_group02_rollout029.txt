import time

def compute_objective(a):
    """Compute the objective value (without calling evaluate_sequence)."""
    n = len(a)
    c = np.convolve(a, a)
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    return 2 * n * M / (S * S)

def run_subgradient(a, n_iterations, eta_schedule, rng):
    """Run subgradient descent with multiplicative updates."""
    a = a.copy()
    a = a / np.sum(a)
    n = len(a)
    best_a = a.copy()
    best_score = compute_objective(a)
    
    for it in range(n_iterations):
        c = np.convolve(a, a)
        M = np.max(c)
        if M < 1e-15:
            break
        
        # Soft active set with temperature
        tau = 0.02 * M
        w = np.exp((c - M) / tau)
        w_sum = np.sum(w)
        if w_sum < 1e-15:
            break
        w /= w_sum
        
        # Compute gradient g_j = sum_k w_k * a_{k-j}
        # Using FFT
        N = 1
        while N < 3 * n:
            N *= 2
        A = np.fft.rfft(a, N)
        W = np.fft.rfft(w, N)
        R = np.fft.irfft(np.conj(A) * W, N)
        # g[j] = R[(-j) % N]
        g = R[(-np.arange(n)) % N]
        
        # Step size
        eta = eta_schedule(it)
        
        # Multiplicative update
        log_a = np.log(np.maximum(a, 1e-30))
        log_a -= eta * g / M
        log_a -= np.max(log_a)  # for numerical stability
        a_new = np.exp(log_a)
        a_new = np.clip(a_new, 0, 1e10)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new /= s
        
        score = compute_objective(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score
