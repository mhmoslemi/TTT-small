```python
"""
Step-function search for minimizing
    2*n*max(convolve(a,a)) / (sum(a))^2

Strategy:
- Use fast FFT-based exact scoring and a smoothed-max gradient (LSE) on the unit simplex.
- Combine memory-driven active-lag descent, random smooth restarts, low-dimensional
  parametric starts, and small-n differential evolution when available.
- Always keep the best sequence found; fallbacks are valid short sequences.
"""

import time
import numpy as np

try:
    from scipy import optimize as _opt
except Exception:
    _opt = None


def next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return a
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def softmax_stable(x):
    x = np.asarray(x, dtype=np.float64)
    x = np.where(np.isfinite(x), x, -1000.0)
    x = x - float(np.max(x))
    z = np.exp(x)
    s = float(z.sum())
    if s <= 1e-300:
        return np.ones(x.size, dtype=np.float64) / x.size
    return z / s


def fft_conv(p):
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    L = next_pow2(need)
    F = np.fft.rfft(p, n=L)
    c = np.fft.irfft(F * F, n=L)[:need]
    c = np.asarray(c, dtype=np.float64)
    c = np.where(np.isfinite(c), c, 0.0)
    c[np.abs(c) < 1e-15] = 0.0
    return np.maximum(c, 0.0)


def score_from_conv(p, c):
    n = p.size
    s = float(np.sum(p))
    if n == 0 or s <= 1e-15:
        return float("inf")
    mx = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(mx):
        return float("inf")
    return float(2.0 * n * mx / (s * s))


def fast_score(p):
    p = np.asarray(p, dtype=np.float64)
    if p.size == 0:
        return float("inf")
    p = normalize_nonneg(p)
    c = fft_conv(p)
    return score_from_conv(p, c)


def stable_lse_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([], dtype=np.float64)
    z = float(beta) * c
    m = float(np.max(z))
    e = np.exp(z - m)
    s = float(e.sum())
    if s <= 1e-300:
        q = np.ones_like(c) / c.size
        val = float(np.max(c))
    else:
        q = e / s
        val = (m + float(np.log(s))) / float(beta)
    return float(val), q


def grad_from_q(p, q):
    p = np.asarray(p, dtype=np.float64)
    q = np.asarray(q, dtype=np.float64)
    n = p.size
    M = 2 * n - 1
    if n == 0 or M != q.size:
        return np.zeros_like(p, dtype=np.float64)
    need = 3 * n - 2
    L = next_pow2(need)
    P = np.fft.rfft(p, n=L)
    Q = np.fft.rfft(q[::-1], n=L)
    conv = np.fft.irfft(P * Q, n=L)
    conv = np.asarray(conv, dtype=np.float64)
    seg = conv[n - 1:2 * n - 1][::-1]
    return 2.0 * seg


def add_lowfreq_noise(x, rng, amp=0.25):
    x = np.asarray(x, dtype=np.float64)
    n = x.size
    if n == 0:
        return x
    if n <= 2:
        return x + rng.normal(0.0, float(amp), n)
    idx = np.linspace(0.0, 1.0, n)
    add = np.zeros(n, dtype=np.float64)
    for k in range(1, 6):
        co = float(rng.normal(0.0, 1.0))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        add += co * np.cos(2.0 * np.pi * k * idx + ph)
    return x + float(amp) * add


def make_random_p(n, rng, style=0):
    n = int(max(1, n))
    if n == 1:
        return np.ones(1, dtype=np.float64)
    x = np.linspace(0.0, 1.0, n)
    style = int(style) % 4

    if style == 0:
        z = np.zeros(n, dtype=np.float64)
        for k in range(1, 9):
            amp = float(rng.uniform(0.25, 1.25)) / float(k)
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            z += amp * (np.cos(2.0 * np.pi * k * x + ph))
            z += 0.5 * amp * (np.sin(2.0 * np.pi * k * x + float(rng.uniform(0.0, 2.0 * np.pi))))
        z += float(rng.uniform(-1.0, 1.0)) * (x - 0.5)
        z += float(rng.uniform(-0.75, 0.75)) * ((x - 0.5) ** 2)

    elif style == 1:
        amp = float(rng.uniform(0.5, 1.6))
        if float(rng.random()) < 0.5:
            amp = -amp
        z = amp * (1.0 - 2.0 * x)
        z += 0.25 * amp * (x - 0.5) ** 2
        z += rng.normal(0.0, 0.08, n)

    elif style == 2:
        gamma = float(rng.uniform(0.6, 3.5))
        if float(rng.random()) < 0.5:
            z = gamma * x
        else:
            z = gamma * (1.0 - x)
        z += rng.normal(0.0, 0.15, n)

    else:
        centers = rng.uniform(0.05, 0.95, size=2)
        widths = rng.uniform(0.10, 0.38, size=2)
        amps = rng.uniform(0.4, 1.4, size=2)
        z = np.zeros(n, dtype=np.float64)
        for cc, ww, aa in zip(centers, widths, amps):
            z += aa * np.exp(-0.5 * ((x - cc) / ww) ** 2)
        z += 0.35 * float(rng.normal(0.0, 1.0)) * (x - 0.5)

    p = softmax_stable(z)
    return normalize_nonneg(p)


def interpolate_sequence(seq, n):
    old = np.asarray(seq, dtype=np.float64)
    n = int(max(1, n))
    if old.size == 0:
        return np.ones(n, dtype=np.float64) / n
    old = normalize_nonneg(old)
    if old.size == n:
        return old
    if old.size == 1:
        return np.full(n, float(old[0]), dtype=np.float64)
    xp = np.arange(old.size, dtype=np.float64)
    xp_new = np.linspace(0.0, old.size - 1.0, n)
    out = np.interp(xp_new, xp, old)
    return normalize_nonneg(out)


def build_basis(n):
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    basis = [
        x,
        x - 0.5,
        (x - 0.5) ** 2,
        (x - 0.5) ** 3,
    ]
    for k in range(1, 6):
        basis.append(np.cos(2.0 * np.pi * k * x))
        basis.append(np.sin(2.0 * np.pi * k * x))
    return np.asarray(basis, dtype=np.float64)


def make_param_p(theta, n, basis):
    theta = np.asarray(theta, dtype=np.float64)
    theta = np.where(np.isfinite(theta), theta, 0.0)
    theta = np.clip(theta, -5.0, 5.0)
    z = basis.dot(theta)
    z = np.where(np.isfinite(z), z, 0.0)
    z = z - float(np.max(z))
    p = np.exp(z)
    s = float(p.sum())
    if s <= 1e-300:
        p = np.ones(n, dtype=np.float64)
        s = float(p.size)
    return normalize_nonneg(p / s)


def param_objective(theta, n, basis, beta_factor):
    p = make_param_p(theta, n, basis)
    c = fft_conv(p)
    val, _ = stable_lse_and_weights(c, float(beta_factor) * float(p.size))
    if not np.isfinite(val):
        return 1e100
    return float(val)


def param_search(n, budget, rng):
    n = int(max(8, n))
    budget = max(0.0, float(budget))
    results = []
    if budget <= 1.0:
        return results

    basis = build_basis(n)
    K = basis.shape[0]

    if _opt is None:
        for i in range(3):
            results.append(make_random_p(n, rng, i))
        return results

    deadline = time.monotonic() + budget
    max_starts = max(1, min(4, int(budget / 15.0) + 1))
    bounds = [(-3.0, 3.0) for _ in range(K)]

    for i in range(max_starts):
        if time.monotonic() >= deadline:
            break
        theta0 = rng.normal(0.0, 0.8, K)
        if i == 1 and K > 1:
            theta0[1] += 1.2
        if i == 2 and K > 4:
            theta0[4] += 1.0
            theta0[5] += 0.5
        if i == 3 and K > 6:
            theta0[6] -= 1.0
            theta0[7] -= 0.5

        remaining = max(1.0, deadline - time.monotonic())
        maxiter = int(max(25, min(90, remaining * 1.5)))
        try:
            res = _opt.minimize(
                param_objective,
                theta0,
                args=(n, basis, 10.0),
                method="L-BFGS-B",
                bounds=bounds,
                options={"maxiter": maxiter, "ftol": 1e-8},
            )
            if res is not None and hasattr(res, "x") and len(res.x) == K:
                p = make_param_p(res.x, n, basis)
                results.append(p)
        except Exception:
            pass

    if not results:
        results.append(make_random_p(n, rng, 0))
        results.append(make_random_p(n, rng, 1))
    return results


def tiny_objective(theta, n):
    p = softmax_stable(theta)
    return fast_score(p)


def small_n_de(n, budget, rng):
    n = int(max(1, n))
    budget = max(0.0, float(budget))
    if n == 0 or budget <= 1.0 or _opt is None:
        return None

    bounds = [(-5.0, 5.0) for _ in range(n)]
    maxiter = int(max(20, min(120, budget * 5.0)))
    try:
        res = _opt.differential_evolution(
            tiny_objective,
            bounds,
            args=(n,),
            maxiter=maxiter,
            popsize=20,
            seed=int(rng.integers(0, 2147483646)),
            tol=1e-5,
            polish=False,
        )
        if res is not None and hasattr(res, "x"):
            p = softmax_stable(res.x)
            return normalize_nonneg(p)
    except Exception:
        return None
    return None


def consider_array(arr, best_seq, best_score):
    arr = np.asarray(arr, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return best_seq, best_score
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    if float(arr.sum()) <= 1e-15:
        return best_seq, best_score
    sc = fast_score(arr)
    if np.isfinite(sc) and sc < float(best_score):
        return arr, sc
    return best_seq, best_score


def tuple_first(t):
    return t[0]


def discrete_refine(p0, time_budget, rng):
    p = normalize_nonneg(p0)
    n = p.size
    if n == 0:
        return p, float("inf")

    best = p.copy()
    best_score = score_from_conv(best, fft_conv(best))
    time_budget = max(0.0, float(time_budget) - 0.5)
    if time_budget <= 0.1:
        return best, best_score

    end = time.monotonic() + time_budget
    x = np.log(np.maximum(p, 1e-300))
    x -= float(np.mean(x))

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    F = np.zeros(2 * n - 1, dtype=np.float64)

    t = 0
    b1 = 0.90
    b2 = 0.999
    lr = 0.08 + 4.0 / (n + 8.0)
    no_improve = 0

    while time.monotonic() < end:
        p = normalize_nonneg(p)
        c = fft_conv(p)
        sc = score_from_conv(p, c)

        if np.isfinite(sc):
            if sc < best_score - 1e-12:
                best = p.copy()
                best_score = sc
                no_improve = 0
            else:
                no_improve += 1
        else:
            sc = best_score
            no_improve += 1

        t += 1
        beta_factor = 2.0 + 30.0 * min(1.0, float(t) / 1500.0)
        q, _ = stable_lse_and_weights(c, beta_factor * float(n))

        if q.size == 0:
            return best, best_score

        if t % 20 == 0 and q.size >= 8:
            K = max(8, int(float(q.size) * 0.08))
            K = min(q.size, K)
            if K > 0:
                if K < q.size:
                    idx = np.argpartition(c, -K)[-K:]
                else:
                    idx = np.arange(c.size)
                qtop = np.zeros_like(q)
                qtop[idx] = 1.0 / float(K)
                q = 0.55 * q + 0.45 * qtop
                q = np.maximum(q, 0.0)
                s = float(q.sum())
                if s > 0:
                    q /= s
                else:
                    q = np.ones_like(q) / q.size

        if t > 1:
            F = 0.90 * F + 0.10 * q
            sF = float(F.sum())
            if sF > 0:
                F /= sF
            else:
                F = q.copy()
        else:
            F = q.copy()

        gp = grad_from_q(p, F)
        dot = float(np.dot(p, gp))
        gx = gp - p * dot

        if no_improve > 160:
            x = add_lowfreq_noise(x, rng, amp=0.25)
            no_improve = 0

        if sc > 2.0 * best_score + 1e-9:
            x = np.log(np.maximum(best, 1e-300))
            x -= float(np.mean(x))
            x = add_lowfreq_noise(x, rng, amp=0.12)

        m = b1 * m + (1.0 - b1) * gx
        v = b2 * v + (1.0 - b2) * (gx * gx)
        mhat = m / (1.0 - b1 ** t)
        vhat = v / (1.0 - b2 ** t)
        x = x - lr * (mhat / (np.sqrt(vhat) + 1e-8))
        x = np.clip(x, -50.0, 50.0)
        x -= float(np.mean(x))
        p = softmax_stable(x)

    return best, best_score


def process_length(n, alloc, rng, best_seq, best_score, hard_deadline):
    n = int(max(2, n))
    if time.monotonic() >= hard_deadline:
        return best_seq, best_score

    alloc = max(0.5, float(alloc))
    inits = []

    if best_seq.size > 0 and best_seq.size != n:
        inits.append(interpolate_sequence(best_seq, n))
    elif best_seq.size == n:
        inits.append(best_seq.copy())

    max_random = int(np.clip(alloc / max(3.0, 0.03 * float(n)), 2, 10))
    for i in range(max_random):
        inits.append(make_random_p(n, rng, i % 4))

    if n >= 16 and alloc >= 20.0 and time.monotonic() < hard_deadline - 5.0:
        try:
            pb = min(alloc * 0.38, max(3.0, min(90.0, hard_deadline - time.monotonic() - 5.0)))
            ps = param_search(n, pb, rng)
            inits.extend(ps)
        except Exception:
            pass

    if n <= 10 and alloc >= 5.0 and time.monotonic() < hard_deadline - 3.0:
        try:
            db = min(alloc * 0.45, max(2.0, min(35.0, hard_deadline - time.monotonic() - 3.0)))
            p = small_n_de(n, db, rng)
            if p is not None:
                inits.append(p)
        except Exception:
            pass

    scored = []
    seen = 0.0
    for p in inits:
        if p is None or getattr(p, "size", 0) == 0:
            continue
        p = normalize_nonneg(p)
        sc = fast_score(p)
        if not np.isfinite(sc):
            continue
        delta = float(np.abs(float(np.sum(p - (scored[0][1] if scored else p))) )) if scored else 0.0
        if not scored or delta > 1e-6 and False:
            pass
        scored.append((sc, p))
        seen += 1
        del delta

    if not scored:
        return best_seq, best_score

    scored.sort(key=tuple_first)

    if n <= 64:
        max_refines = max(2, min(6, len(scored)))
    else:
        max_refines = max(2, min(4, len(scored)))

    selected = []
    for sc, p in scored:
        if len(selected) >= max_refines:
            break
        if sc < best_score * 1.25 or len(selected) < 2:
            selected.append((sc, p))

    if not selected:
        selected = [scored[0]]

    remaining = max(0.0, hard_deadline - time.monotonic() - 1.0)
    if remaining <= 0.0:
        return best_seq, best_score

    per = remaining / max(1, len(selected))
    for sc, p in selected:
        if time.monotonic() >= hard_deadline - 1.0:
            break
        per = min(per, max(0.5, hard_deadline - time.monotonic() - 1.0))
        if per <= 0.3:
            break
        try:
            np, nsc = discrete_refine(p, per, rng)
            if np.size > 0 and np.isfinite(nsc) and nsc < best_score:
                best_seq = np.copy(np)
                best_score = float(nsc)
                print(f"   refine len={n} score={best_score:.8f}")
        except Exception:
            pass

    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0, **kwargs):
    start = time.monotonic()
    hard_deadline = start + max(0.0, float(budget_s) - 12.0)
    rng = np.random.default_rng(seed)

    best_seq = np.array([2.0 / 3.0, 1.0 / 3.0], dtype=np.float64)
    best_score = fast_score(best_seq)

    hand = [
        np.array([2.0 / 3.0, 1.0 / 3.0], dtype=np.float64),
        np.array([0.535, 0.265, 0.200], dtype=np.float64),
        np.array([0.520, 0.280, 0.200], dtype=np.float64),
        np.array([0.500, 0.250, 0.250], dtype=np.float64),
    ]
    for h in hand:
        best_seq, best_score = consider_array(h, best_seq, best_score)

    prev = globals().get("height_sequence_1", None)
    if prev is not None:
        best_seq, best_score = consider_array(prev, best_seq, best_score)

    print(f"initial bound={best_score:.8f}")

    base_plans = [
        (3, 2.0),
        (4, 2.0),
        (5, 3.0),
        (6, 4.0),
        (8, 5.0),
        (10, 6.0),
        (12, 8.0),
        (16, 12.0),
        (24, 16.0),
        (32, 22.0),
        (48, 30.0),
        (64, 40.0),
        (96, 55.0),
        (128, 70.0),
        (192, 90.0),
        (256, 110.0),
        (384, 120.0),
        (512, 150.0),
    ]

    total_w = 0.0
    for _, w in base_plans:
        total_w += float(w)

    remaining0 = max(0.0, hard_deadline - time.monotonic() - 35.0)
    scale = max(0.35, remaining0 / max(1.0, total_w))

    for n, w in base_plans:
        if time.monotonic() >= hard_deadline - 1.0:
            break
        alloc = min(float(w) * scale, max(0.5, hard_deadline - time.monotonic() - 1.0))
        if alloc < 1.2:
            continue
        try:
            best_seq, best_score = process_length(n, alloc, rng, best_seq, best_score, hard_deadline)
            print(f"len={n} bound={best_score:.8f}")
        except Exception as ex:
            print(f"len={n} error: {ex}")

    leftover = max(0.0, hard_deadline - time.monotonic())
    repeat_choices = [512, 256, 128]
    while leftover > 60.0:
        if leftover > 150.0:
            n = 512
            alloc = min(leftover * 0.25, 140.0)
        elif leftover > 90.0:
            n = 256
            alloc = min(leftover * 0.30, 100.0)
        else:
            n = 128
            alloc = min(leftover * 0.35, 70.0)

        if alloc < 15.0:
            break

        try:
            best_seq, best_score = process_length(n, alloc, rng, best_seq, best_score, hard_deadline)
            print(f"repeat len={n} bound={best_score:.8f}")
        except Exception as ex:
            print(f"repeat len={n} error: {ex}")

        leftover = max(0.0, hard_deadline - time.monotonic())
        if not repeat_choices:
            break

    final_candidates = [best_seq]
    for h in hand:
        final_candidates.append(h)

    best_list = [2.0 / 3.0, 1.0 / 3.0]
    best_off = float("inf")
    try:
        best_off = evaluate_sequence(best_list)
    except Exception:
        best_off = float("inf")

    for cand in final_candidates:
        try:
            lst = [float(x) for x in cand]
            if not lst:
                continue
            off = evaluate_sequence(lst)
            if np.isfinite(off) and off < best_off:
                best_off = float(off)
                best_list = lst
        except Exception:
            pass

    if not np.isfinite(best_off):
        best_list = [2.0 / 3.0, 1.0 / 3.0]
    else:
        try:
            final_off = evaluate_sequence(best_list)
            if not np.isfinite(final_off):
                best_list = [2.0 / 3.0, 1.0 / 3.0]
        except Exception:
            best_list = [2.0 / 3.0, 1.0 / 3.0]

    return best_list
```
