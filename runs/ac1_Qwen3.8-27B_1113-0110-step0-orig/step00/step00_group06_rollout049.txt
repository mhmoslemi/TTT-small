Minimizes 2*n*max(convolve(a,a))/sum(a)^2 over non-negative sequences.
Uses subgradient descent initialized from monotone (ramp/power-law) sequences,

import numpy as np

def _subgradient_at(a, k):
    """Approximate subgradient of c[k]=conv(a,a)[k] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    if lo <= hi:
        idx = np.arange(hi, lo - 1, -1)
        grad[lo:hi + 1] = 2.0 * a[idx]
    return grad

def _make_ramp(n, power=1.0):
    """Create a power-law ramp of length n."""
    return np.power(np.arange(1, n + 1, dtype=np.float64), power)
