    eta = eta_start
    step = 0
    stall = 0
    recompute_interval = 15

    while time.monotonic() < deadline:
        # Take a batch of MW steps
        batch_size = 30
        for _ in range(batch_size):
            if step % recompute_interval == 0:
                g = grad_direction(a, k_star)
            a = mw_step(a, g, eta)
            step += 1

        # Recompute peak
        M, k_star = compute_peak_info(a)
        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Adapt step size
        if stall > 80:
            eta *= 0.6
            stall = 0
        elif stall == 0 and eta < 3.0:
            eta = min(eta * 1.02, 3.0)

        # Random perturbation if really stuck
        if stall > 300:
            width = max(1, int(n * rng.uniform(0.02, 0.15)))
            start = int(rng.integers(0, max(1, n - width)))
            a[start:start + width] *= np.exp(rng.normal(0.0, 0.8, width))
            a = np.maximum(a, 0.0)
            S = a.sum()
            if S >= 0.01:
                a = a / S
            else:
                a = best_a.copy()
            stall = 0
            eta = max(eta * 0.5, 0.1)

        # If very close to convergence, reduce eta
        if step > 100000:
            eta = min(eta, 0.1)
        if step > 300000:
            eta = min(eta, 0.03)

    return best_a, best_M


def build_constraint_matrix_sparse(a):
    """Build sparse constraint matrix for LP refinement."""
    from scipy.sparse import lil_matrix, csr_matrix
    n = len(a)
    size = 2 * n - 1
    A = lil_matrix((size, n))
    for i in range(n):
        end = min(i + n, size)
        length = end - i
        if length > 0:
            A[i:end, i] = a[:length]
    return csr_matrix(A)


def lp_refine(a, M, max_iters=3, deadline=None):
    """
    LP-based refinement from SOTA. Given current a and its peak M,
    solve LP to find b maximizing sum(b) subject to (a*b)_k <= M, b>=0.
    Blend and iterate.
    """
    n = len(a)
    a = a.copy()
    size = 2 * n - 1

    try:
        from scipy.optimize import linprog
    except ImportError:
        return a

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        # Build constraint matrix
        A_mat = build_constraint_matrix_sparse(a)

        # LP: maximize sum(b) => minimize -sum(b)
        # subject to: A_mat @ b <= M (vector), b >= 0
        c_obj = -np.ones(n)
        b_ub = np.full(size, M)
        bounds = [(0, None)] * n

        t0 = time.monotonic()
        res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': 120, 'maxiter': 50000})

        if not res.success:
            break

        b_opt = np.maximum(res.x, 0.0)
        S_b = b_opt.sum()
        if S_b < 1e-10:
            break

        # Blend: a' = (1-t)*a + t*b_opt, then renormalize
        t_blend = 0.3
        a_new = (1.0 - t_blend) * a + t_blend * b_opt
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 0.01:
            break
        a_new = a_new / S_new

        # Check if peak decreased
        M_new, _ = compute_peak_info(a_new)
        if M_new < M * 0.999:
            a = a_new
            M = M_new
            print(f"    LP iter {it}: M reduced to {M:.8f}")
            t_blend = 0.4  # be more aggressive
        else:
            print(f"    LP iter {it}: no improvement (M_new={M_new:.8f})")
            break

    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Uses FFT-accelerated multiplicative-weights
    gradient descent with diverse initializations, adaptive step size,
    random restarts, and LP-based refinement in the final phase.
    """
    rng = np.random.default_rng(seed)
    run_start = time.monotonic()
    hard_deadline = run_start + budget_s - 30.0

    # Get base sequence
    try:
        init_base = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000).ravel()
        n = len(init_base)
    except Exception:
        n = 512
        init_base = np.ones(n)

    print(f"n={n}, budget={budget_s}s")

    # Build initializations
    inits = make_inits(n, rng)

    # Add provided sequence
    S_base = init_base.sum()
    if S_base >= 0.01:
        inits.insert(0, init_base / S_base)

    # Quick-rank all inits using fast FFT evaluation
    init_scores = []
    for a_init in inits:
        M, _ = compute_peak_info(a_init)
        score = 2.0 * n * M
        init_scores.append(score)

    sorted_idx = np.argsort(init_scores)
    n_top = min(6, len(inits))
    top_inits = [inits[i] for i in sorted_idx[:n_top]]
    top_scores = [init_scores[i] for i in sorted_idx[:n_top]]

    print(f"Top {n_top} initial scores: {[f'{s:.6f}' for s in top_scores]}")

    # Verify best with actual evaluator
    best = top_inits[0].copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"Verified initial: {best_score:.6f}")

    # Phase 1: MW optimization from top initializations
    # Allocate time: give best init more time
    total_phase1_time = budget_s * 0.55
    time_per_init = total_phase1_time / n_top

    results = []
    for idx, a_init in enumerate(top_inits):
        t_start = time.monotonic()
        t_end = min(hard_deadline, t_start + time_per_init * (1.4 if idx == 0 else 1.0))

        if time.monotonic() > hard_deadline:
            break

        a_opt, local_M = run_mw_phase(a_init, t_end, n, rng,
                                      eta_start=0.8 if idx == 0 else 0.5,
                                      label=f"init{idx}")

        score = float(evaluate_sequence(a_opt.tolist()))
        results.append((score, a_opt))
        if np.isfinite(score) and score < best_score:
            best = a_opt.copy()
            best_score = score
            print(f"  Init {idx}: new best = {best_score:.6f}")

    print(f"After Phase 1 (MW): best = {best_score:.6f}")

    # Phase 2: LP refinement (if time permits)
    lp_deadline = min(hard_deadline, run_start + budget_s * 0.85)
    if time.monotonic() < lp_deadline:
        try:
            M_cur, _ = compute_peak_info(best)
            best_lp = lp_refine(best, M_cur, max_iters=5, deadline=lp_deadline)
            score_lp = float(evaluate_sequence(best_lp.tolist()))
            if np.isfinite(score_lp) and score_lp < best_score:
                best = best_lp.copy()
                best_score = score_lp
                print(f"After LP refinement: {best_score:.6f}")
        except Exception as e:
            print(f"LP refinement failed: {e}")

    # Phase 3: Final polish with very small MW steps
    polish_deadline = hard_deadline
    if time.monotonic() < polish_deadline:
        a = best.copy()
        eta = 0.01
        M, k_star = compute_peak_info(a)
        best_polish_M = M
        best_polish_a = a.copy()
        stall = 0

        while time.monotonic() < polish_deadline:
            g = grad_direction(a, k_star)
            a = mw_step(a, g, eta)

            if M % 200 == 0:
                pass  # placeholder for step counting

            # Recompute periodically
            M, k_star = compute_peak_info(a)
            if M < best_polish_M - 1e-12:
                best_polish_M = M
                best_polish_a = a.copy()
                stall = 0
            else:
                stall += 1

            if stall > 200:
                # Small random perturbation
                width = max(1, int(n * 0.02))
                start = int(rng.integers(0, max(1, n - width)))
                a[start:start + width] *= np.exp(rng.normal(0.0, 0.3, width))
                a = np.maximum(a, 0.0)
                S = a.sum()
                if S >= 0.01:
                    a = a / S
                stall = 0
                eta = min(eta * 1.5, 0.05)

            # Evaluate
            if stall % 500 == 0 or stall == 0:
                score = float(evaluate_sequence(a.tolist()))
                if np.isfinite(score) and score < best_score:
                    best = a.copy()
                    best_score = score
                    print(f"  Polish: {best_score:.6f}")

        # Use the best polish result
        M_final, _ = compute_peak_info(best_polish_a)
        score_check = 2.0 * n * M_final
        if score_check < best_score:
            best = best_polish_a.copy()
            best_score = float(evaluate_sequence(best.tolist()))

    # Final output
    final_score = float(evaluate_sequence(best.tolist()))
    elapsed = time.monotonic() - run_start
    print(f"FINAL: score = {final_score:.6f}, time = {elapsed:.1f}s, n = {n}")

    return [float(x) for x in best]
