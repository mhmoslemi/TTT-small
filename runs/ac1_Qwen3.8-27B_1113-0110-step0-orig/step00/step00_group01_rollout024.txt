"""
Subgradient descent with multi-restart and jerk-based escape for minimizing
J(a) = 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Key insights:
- The objective is scale-invariant, so we work on the softmax-smoothed version
- The uniform sequence (J=2) is a local minimum; we use structured perturbations + random kicks to escape
- FFT-based convolution gives O(n log n) per evaluation
- Adaptive step size with backtracking ensures convergence
"""

import numpy as np
import time


def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    nfft = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, nfft)
    result = np.fft.irfft(A * A, nfft)[:size]
    return np.maximum(result, 0.0)  # numerical safety


def softmax_max(conv, T):
    """Softmax approximation of max for smoothness."""
    shifted = conv - conv.max()
    exp_vals = np.exp(T * shifted)
    result = conv.max() + np.log(exp_vals.sum()) / T
    return result, exp_vals  # return both value and weights


def compute_objective(a):
    """Compute the smoothed objective and its gradient."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf, np.zeros(n)
    conv = fast_convolve(a)
    max_conv = conv.max()
    # Use temperature for softness
    # For the hard problem, we just use the argmax
    return 2.0 * n * max_conv / (S * S), conv


def subgradient_step(a, conv, step_frac):
    """
    One subgradient descent step.
    Returns new a and the k_star used.
    """
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return a.copy(), -1
    
    k_star = int(np.argmax(conv))
    max_conv = conv[k_star]
    
    # Gradient: dJ/da[i] proportional to a[k_star - i] - max_conv/S
    # (times positive constant 4n/S^2, which we absorb into step size)
    grad = np.zeros(n)
    i_low = max(0, k_star - (n - 1))
    i_high = min(n - 1, k_star)
    if i_low <= i_high:
        i_idx = np.arange(i_low, i_high + 1)
        k_minus_i = k_star - i_idx
        grad[i_idx] = a[k_minus_i] - max_conv / S
    
    # Normalize and apply step
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return a.copy(), k_star
    
    # Step size: proportional to S (scale invariance) times a fraction
    step = step_frac * S / gnorm
    a_new = a - step * grad
    a_new = np.clip(a_new, 0.0, 1000.0)
    
    return a_new, k_star


def generate_initialization(n, rng, init_type):
    """Generate various types of initial sequences."""
    if init_type == 0:
        # Uniform (baseline)
        return np.ones(n)
    elif init_type == 1:
        # Random uniform
        return rng.uniform(0.1, 1.0, n)
    elif init_type == 2:
        # U-shaped (high at edges)
        t = np.linspace(-1, 1, n)
        return 0.5 + 0.5 * (1 + t**2) / (1 + 1)  # normalized to [0.25, 1]
    elif init_type == 3:
        # V-shaped (high in middle)
        t = np.linspace(0, 1, n)
        center = np.abs(t - 0.5) * 2
        return 0.3 + 0.7 * (1 - center)
    elif init_type == 4:
        # Periodic modulation
        k = rng.integers(1, max(2, n // 10))
        t = np.arange(n)
        return 0.5 + 0.4 * np.cos(2 * np.pi * k * t / n)
    elif init_type == 5:
        # Two-scale random
        base = rng.uniform(0, 1, n // 100)
        extended = np.interp(np.linspace(0, n // 100 - 1, n), 
                             np.arange(n // 100), base)
        noise = rng.normal(0, 0.1, n)
        return np.clip(extended + noise, 0.01, 2.0)
    elif init_type == 6:
        # Logistic bell
        t = np.linspace(0, 1, n)
        sigma = rng.uniform(0.1, 0.4)
        return np.exp(-0.5 * ((t - 0.5) / sigma)**2) + 0.1
    elif init_type == 7:
        # Random with correlations (AR(1))
        phi = rng.uniform(0.7, 0.99)
        a = np.zeros(n)
        a[0] = rng.uniform(0.5, 1.5)
        for i in range(1, n):
            a[i] = phi * a[i-1] + rng.normal(0, 0.3)
        return np.clip(a, 0.01, None)
    elif init_type == 8:
        # Sinusoidal mixture
        n_freqs = rng.integers(2, 6)
        result = np.zeros(n)
        t = np.arange(n, dtype=np.float64)
        for _ in range(n_freqs):
            freq = rng.uniform(0.01, 5.0)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.2, 0.5)
            result += amp * np.sin(2 * np.pi * freq * t / n + phase)
        return np.abs(result) + 0.2
    else:

        return np.abs(rng.normal(0.5, 0.3, n)) + 0.05


def run_gradient_descent(a, n, rng, time_limit, initial_step_frac=0.1):
    """
    Run subgradient descent until time_limit.
    Returns best sequence and score found.
    """
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max() / (a.sum()**2)
    
    step_frac = initial_step_frac
    no_improve_count = 0
    max_no_improve = 50
    adj_no_improve = 0
    
    while time.monotonic() < time_limit:
        conv = fast_convolve(a)
        S = a.sum()
        if S < 0.01:
            break
        
        current_score = 2.0 * n * conv.max() / (S * S)
        
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()
            no_improve_count = 0
            adj_no_improve = 0
        else:
            no_improve_count += 1
        
        # Adaptive step size: decrease if no improvement
        if no_improve_count > max_no_improve:
            step_frac *= 0.9
            no_improve_count = 0
            if step_frac < 1e-6:
                break
        
        # Take a step
        a_new, k_star = subgradient_step(a, conv, step_frac)
        
        # Check if step was valid
        new_S = a_new.sum()
        if new_S < 0.01:
            step_frac *= 0.5
            continue
        
        # Check improvement with backtracking
        new_conv = fast_convolve(a_new)
        new_score = 2.0 * n * new_conv.max() / (new_S * new_S)
        
        if new_score < current_score * (1 + 1e-10):
            a = a_new
            adj_no_improve = 0
        else:
            # Backtrack
            step_frac *= 0.7
            adj_no_improve += 1
            if adj_no_improve > 10:
                # Try a random perturbation to escape
                a = a + rng.normal(0, 0.01 * a.mean(), n)
                a = np.clip(a, 0.0, 1000.0)
                adj_no_improve = 0
        
        # Periodic jitter to escape local minima
        if no_improve_count % 200 == 199 and no_improve_count > 0:
            # Small random perturbation
            jitter_scale = 0.005 * a.mean()
            a = a + rng.normal(0, jitter_scale, n)
            a = np.clip(a, 0.0, 1000.0)
            step_frac = max(step_frac, 0.05)
    
    return best_a, best_score


def run_jerk_escaping(a, n, rng, time_limit):
    """
    Enhanced gradient descent with structured escape mechanisms.
    Uses the insight that we need to break the symmetry of the uniform solution.
    """
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max() / (a.sum()**2)
    
    step_frac = 0.08
    iteration = 0
    max_iter_no_improve = 100
    
    while time.monotonic() < time_limit:
        iteration += 1
        conv = fast_convolve(a)
        S = a.sum()
        if S < 0.01:
            break
        
        current_score = 2.0 * n * conv.max() / (S * S)
        
        improved = False
        if current_score < best_score - 1e-10:
            best_score = current_score
            best_a = a.copy()
            max_iter_no_improve = 100
            step_frac = min(step_frac * 1.02, 0.3)
            improved = True
        
        if not improved:
            max_iter_no_improve -= 1
            if max_iter_no_improve < 30:
                step_frac *= 0.95
            if max_iter_no_improve <= 0:
                # Major escape: structured perturbation
                # Try different perturbation strategies
                strategy = rng.integers(0, 4)
                if strategy == 0:
                    # Mirror-breaking: perturb first half differently from second
                    half = n // 2
                    pert = rng.normal(0, 0.02 * a.mean(), half)
                    a[:half] += pert
                    a[half:] += -pert[::-1]  # anti-symmetric perturbation
                elif strategy == 1:
                    # Block perturbation
                    block_size = rng.integers(n // 50, n // 5)
                    start = rng.integers(0, n - block_size)
                    a[start:start+block_size] *= np.exp(rng.normal(0, 0.1))
                elif strategy == 2:
                    # Frequency-domain perturbation
                    A = np.fft.rfft(a)
                    A += rng.normal(0, 0.01 * np.abs(A).mean(), len(A))
                    a = np.fft.irfft(A, n * 2)[:n]
                    a = np.clip(a, 0, None)
                else:
                    # Blend with a random sequence
                    other = generate_initialization(n, rng, int(rng.integers(0, 9)))
                    other = other / other.sum() * a.sum()
                    t = rng.uniform(0.05, 0.2)
                    a = (1 - t) * a + t * other
                
                a = np.clip(a, 0.0, 1000.0)
                max_iter_no_improve = 50
                step_frac = 0.08
        
        # Subgradient step
        k_star = int(np.argmax(conv))
        max_conv = conv[k_star]
        grad = np.zeros(n)
        i_low = max(0, k_star - (n - 1))
        i_high = min(n - 1, k_star)
        if i_low <= i_high:
            i_idx = np.arange(i_low, i_high + 1)
            k_minus_i = k_star - i_idx
            grad[i_idx] = a[k_minus_i] - max_conv / S
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            # At a symmetric point; kick out
            a += rng.normal(0, 0.01 * max(a.mean(), 0.1), n)
            a = np.clip(a, 0, None)
            continue
        
        step = step_frac * max(S, 1.0) / gnorm
        a_new = a - step * grad
        a_new = np.clip(a_new, 0.0, 1000.0)
        
        # Validate step
        new_S = a_new.sum()
        if new_S < 0.01:
            a_new = a * 0.99 + 0.01 * np.ones(n)  # fallback
        else:
            new_conv = fast_convolve(a_new)
            new_score = 2.0 * n * new_conv.max() / (new_S * new_S)
            if new_score > current_score * 1.01:
                # Step too large, shrink
                step_frac *= 0.8
                # Try half step
                a_new = a - 0.5 * step * grad
                a_new = np.clip(a_new, 0.0, 1000.0)
        
        a = a_new
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy search for minimizing J(a) = 2n * max(conv(a,a)) / (sum(a))^2.
    
    Uses subgradient descent with multiple initialization strategies and
    structured escape mechanisms to find sequences with J < 2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 10.0
    
    # Get n from available sequence
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = parent.size
    except:
        n = 6121
        parent = np.ones(n)
    
    # Phase 1: Quick evaluation of several initializations (use ~10% of time)
    phase1_time = deadline - (budget_s * 0.9)
    best_a = parent.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))
    print(f"Starting: n={n}, initial score={best_score:.6f}")
    
    # Evaluate several initializations
    init_types = list(range(9))
    rng.shuffle(init_types)
    
    for it in init_types:
        if time.monotonic() > phase1_time:
            break
        a_try = generate_initialization(n, rng, it)
        # Normalize to same scale as parent
        a_try = a_try / a_try.sum() * parent.sum()
        score = 2.0 * n * fast_convolve(a_try).max() / (a_try.sum()**2)
        if score < best_score:
            best_score = score
            best_a = a_try.copy()
            print(f"  Init type {it}: score={score:.6f}")
    
    # Also try perturbations of the parent
    for _ in range(5):
        if time.monotonic() > phase1_time:
            break
        a_try = parent * np.exp(rng.normal(0, 0.1, n))
        score = 2.0 * n * fast_convolve(a_try.max(0)).max() / (a_try.sum()**2)
        a_try = np.clip(a_try, 0, None)
        if a_try.sum() > 0.01:
            score = 2.0 * n * fast_convolve(a_try).max() / (a_try.sum()**2)
            if score < best_score:
                best_score = score
                best_a = a_try.copy()
    
    print(f"After Phase 1: best score={best_score:.6f}")
    
    # Phase 2: Main optimization with multiple restarts
    restart_count = 0
    min_restart_time = 30  # minimum seconds per restart
    best_a_overall = best_a.copy()
    
    while time.monotonic() < deadline - 5:
        restart_count += 1
        time_left = deadline - time.monotonic()
        if time_left < min_restart_time:
            break
        
        # Allocate time for this restart (decreasing)
        restart_time = min(time_left - 5, max(30, 120 / restart_count * 3))
        restart_deadline = time.monotonic() + restart_time
        
        # Choose starting point for this restart
        if restart_count == 1:
            a_start = best_a.copy()
        elif restart_count == 2:
            a_start = best_a * np.exp(rng.normal(0, 0.3, n))
            a_start = np.clip(a_start, 0, None)
        elif restart_count % 4 == 0:
            # Try a completely different initialization
            a_start = generate_initialization(n, rng, int(rng.integers(0, 9)))
            a_start = a_start / max(a_start.sum(), 1e-10) * best_a.sum()
            a_start = np.clip(a_start, 0, None)
        else:
            # Perturb the best found so far
            scale = 0.2 + 0.3 * (restart_count % 5) / 5.0
            a_start = best_a_overall * np.exp(rng.normal(0, scale, n))
            a_start = np.clip(a_start, 0, None)
            if a_start.sum() < 0.01:
                a_start = best_a_overall.copy()
        
        # Run gradient descent
        a_result, score_result = run_jerk_escaping(a_start, n, rng, restart_deadline)
        
        if score_result < best_score:
            best_score = score_result
            best_a = a_result.copy()
            best_a_overall = a_result.copy()
            print(f"  Restart {restart_count}: NEW BEST score={best_score:.6f}")
        
        # Verify with actual evaluator periodically
        if restart_count % 3 == 0 or score_result < best_score + 0.01:
            actual_score = float(evaluate_sequence(best_a.tolist()))
            if actual_score < float(evaluate_sequence(best_a_overall.tolist())):
                best_a_overall = best_a.copy()
            if np.isfinite(actual_score) and actual_score < best_score:
                best_score = actual_score
                best_a_overall = best_a.copy()
                print(f"  Verified: actual score={actual_score:.6f}")
    
    # Final: verify and return the best
    final_score = float(evaluate_sequence(best_a_overall.tolist()))
    if final_score < best_score:
        best_a_overall = best_a.copy()
        best_score = final_score
    
    # Also verify best_a
    score_a = float(evaluate_sequence(best_a.tolist()))
    if score_a < best_score:
        best_a_overall = best_a.copy()
        best_score = score_a
    
    print(f"Final best score: {best_score:.6f}")
    print(f"Length: {len(best_a_overall)}, Sum: {best_a_overall.sum():.2f}, Max: {best_a_overall.max():.2f}")
    
    return [float(x) for x in best_a_overall]
