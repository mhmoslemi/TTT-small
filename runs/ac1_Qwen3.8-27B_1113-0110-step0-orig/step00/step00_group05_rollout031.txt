- LP-based best-response refinement
- Multi-start initialization to escape symmetry

import time
from scipy.signal import fftconvolve
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _normalize(f):
    s = f.sum()
    if s < 1e-300:
        return f
    return f / s


def _mwd_update(f, k_star, eta):
    n = len(f)
    idx = k_star - np.arange(n)
    valid = (idx >= 0) & (idx < n)
    sg = np.zeros(n)
    sg[valid] = 2.0 * f[idx[valid]]
    f_new = f * np.exp(-eta * sg)
    f_new = np.maximum(f_new, 0.0)
    return _normalize(f_new)


def _run_mwd(f, n, deadline, rng, max_iter=100000, eta_init=0.5):
    f = f.copy()
    eta = eta_init
    best_peak = np.inf
    best_f = f.copy()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = fftconvolve(f, f, mode='full')
        k_star = int(np.argmax(c))
        peak = c[k_star]
        
        if peak < best_peak:
            best_peak = peak
            best_f = f.copy()
            if it % 5000 == 0:
                print(f"  MWD iter {it}: F={2*n*peak:.6f}, k*={k_star}, eta={eta:.4f}")
        
        f = _mwd_update(f, k_star, eta)

            if c_check.max() > peak * 1.05:
                eta *= 0.9
            elif c_check.max() < peak * 0.95:
                eta = min(eta * 1.1, 2.0)
    
    return best_f


def _build_conv_matrix(f):
    n = len(f)
    nnz = n * min(n, 2*n-1)  # each column has at most n nonzeros
    row_ind = np.empty(nnz, dtype=np.int32)
    col_ind = np.empty(nnz, dtype=np.int32)
    data = np.empty(nnz, dtype=np.float64)
    
    pos = 0
    for j in range(n):
        start = j
        end = min(j + n, 2 * n - 1)
        length = end - start
        row_ind[pos:pos+length] = np.arange(start, end)
        col_ind[pos:pos+length] = j
        data[pos:pos+length] = f[:length]
        pos += length
    
    C = csr_matrix((data[:pos], (row_ind[:pos], col_ind[:pos])), shape=(2*n-1, n))
    return C

    t = 0.3  # blending parameter

                         bounds=[(0, None)]*n, method='highs',
                         options={'presolve': True, 'time_limit': 20})

            break

        g = np.maximum(g, 0.0)

1. Multi-start FFT-accelerated multiplicative weight descent
2. LP-based best-response refinement (solve max sum(g) s.t. f*g <= M)
3. Iterative re-optimization

import numpy as np

def _normalize(f):
    s = f.sum()
    if s < 1e-300:
        return np.ones_like(f) / len(f)
    return f / s


def _mwd_update(f, k_star, eta):
    n = len(f)
    idx = k_star - np.arange(n)
    valid = (idx >= 0) & (idx < n)
    sg = np.zeros(n)
    sg[valid] = 2.0 * f[idx[valid]]
    f_new = f * np.exp(-eta * sg)
    f_new = np.maximum(f_new, 0.0)
    return _normalize(f_new)
