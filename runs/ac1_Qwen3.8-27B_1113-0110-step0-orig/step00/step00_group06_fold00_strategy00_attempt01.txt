    except:
        fallback_used = False
        if np.isinf(final_score):
            fallback_used = True
        if fallback_used:
            fallback = np.ones(256)
            fallback = 0.5 * fallback / np.sum(fallback)
            fallback_score = evaluate_sequence(fallback.tolist())
            print("Fallback needed with score: ", fallback_score)
            final_sequence = fallback
        else:
            pass

    # Return the best sequence found within the time
    if elapsed_time since start > deadline:
        return candidate_save.tolist()  # Or potentially best_sequence found earlier
    else:
        return final_sequence.tolist()
