def _pow2(n):
    if n <=1: return 1
    return 1 << (int(n)-1).bit_length()

def _normalize(a):
    v = np.asarray(a, dtype=float)
    v = np.maximum(v,0)
    s = v.sum()
    if not np.isfinite(s) or s <= 1e-15:
       return np.full(v.size, 1/v.size)
    return v/s

def _project_simplex(v):
    v = np.asarray(v, dtype=float)
    n=v.size
    if n==0: return v
    # use exact projection
    x=np.sort(v)[::-1]
    css=np.cumsum(x)
    idx=np.arange(1,n+1,dtype=float)
    # condition x > (css -1)/idx
    #为了保护 from all equal? use np.where
    cond = x > (css - 1.0)/idx
    if not np.any(cond):
       # fallback: uniform (shouldn't happen often)
       return np.full(n, 1.0/n)
    r = int(np.where(cond)[0][-1])
    tau = (css[r]-1.0)/(r+1.0)
    out = v - tau
    np.maximum(out, 0.0, out=out)
    # correct sum due to roundoff
    s=out.sum()
    if s <= 0 or not np.isfinite(s):
       return np.full(n,1.0/n)
    out *= 1.0/s
    return out

"""
Search for a non-negative step-function height sequence minimizing
    2 * n * max(convolve(a, a)) / (sum(a))**2.

Strategy:
  * Build strong non-uniform initial candidates (one-sided 1/sqrt-like shapes).
  * Parameterize the positive simplex by log-ratios to a fixed reference entry.
  * Polish with L-BFGS-B on smooth surrogates:
      - ||a*a||_2^2,
      - ||a*a||_3^p,
      - a smoothed maximum / log-sum-exp objective.
  * Finish with random multiplicative block local search.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize as _minimize
except Exception:
    _minimize = None


def _pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _clean_nonnegative(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    return np.maximum(v, 0.0)


def _normalize(v):
    v = _clean_nonnegative(v)
    s = float(v.sum())
    if not np.isfinite(s) or s <= 1e-15:
        n = v.size
        if n == 0:
            return v
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    return v / s


def _project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    v = _clean_nonnegative(v)
    n = v.size
    if n == 0:
        return v.copy()

    x = np.sort(v)[::-1]
    css = np.cumsum(x)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = x > (css - 1.0) / idx

    if not np.any(cond):
        return np.full(n, 1.0 / float(n), dtype=np.float64)

    r = int(np.nonzero(cond)[0][-1])
    tau = (css[r] - 1.0) / float(r + 1)
    out = v - tau
    np.maximum(out, 0.0, out=out)

    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.full(n, 1.0 / float(n), dtype=np.float64)

    out *= 1.0 / s
    return out


def _score_vector(a, L=None):
    """Fast exact score using FFT-based linear convolution."""
    v = _clean_nonnegative(a)
    n = v.size
    if n == 0:
        return np.inf

    s = float(v.sum())
    if s < 1e-12:
        return np.inf
    v = v / s

    if L is None:
        L = _pow2(2 * n - 1)

    A = np.fft.rfft(v, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]

    mx = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(mx):
        return np.inf
    if mx < 0.0:
        mx = 0.0
    return 2.0 * float(n) * mx


def _power_sequence(n, p=0.5, offset=1.0):
    j = np.arange(n, dtype=np.float64)
    x = j + float(offset)
    x = np.maximum(x, 1e-8)
    return _project_simplex(np.power(x, -float(p)))


def _power_window_sequence(n, p=0.5, offset=1.0, boost=0.0, gamma=1.0):
    j = np.arange(n, dtype=np.float64)
    x = np.maximum(j + float(offset), 1e-8)
    v = np.power(x, -float(p))
    if float(boost) != 0.0:
        t = j / max(1.0, float(n - 1))
        v = v * (1.0 + float(boost) * np.power(t, float(gamma)))
    return _project_simplex(_clean_nonnegative(v))


def _two_tail_sequence(n, p=0.5, offset=1.0, tail_weight=0.0):
    j = np.arange(n, dtype=np.float64)
    base = np.power(np.maximum(j + float(offset), 1e-8), -float(p))
    if float(tail_weight):
        rev = np.maximum((n - 1.0 - j) + float(offset), 1e-8)
        base = base + float(tail_weight) * np.power(rev, -float(p))
    return _project_simplex(_clean_nonnegative(base))

def _ratios_from_a(a):
    """
    Represent positive simplex point a by log(a_i / a_last), i < last.
    Adds a tiny epsilon if the reference entry is zero.
    """
    a = _clean_nonnegative(np.asarray(a, dtype=np.float64)).copy()
    n = a.size
    if n == 0:
        return a
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        a = np.full(n, 1.0 / float(n), dtype=np.float64)
        s = 1.0
    else:
        a = a / s

    if n == 1:
        return a

    ref = float(a[-1])
    if ref <= 1e-200:
        a[-1] = 1e-12
        a = a / float(a.sum())
        ref = float(a[-1])

    x = np.log(np.maximum(a[:-1], 1e-300) / max(ref, 1e-300))
    return np.clip(x, -100.0, 100.0)


def _softmax_from_ratios(x):
    """
    Given log(a_i / a_last), i < last, reconstruct positive simplex a.
    """
    x = np.asarray(x, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    n = x.size + 1
    if n == 1:
        return np.ones(1, dtype=np.float64)

    m = 0.0
    if x.size:
        mx = float(np.max(x))
        if mx > m:
            m = mx

    e = np.exp(np.clip(x - m, -745.0, 709.0))
    e_last = float(np.exp(-m))
    s = float(e.sum()) + e_last

    if not np.isfinite(s) or s <= 0.0:
        return np.full(n, 1.0 / float(n), dtype=np.float64)

    a = np.empty(n, dtype=np.float64)
    a[:-1] = e / s
    a[-1] = e_last / s
    return a


def _q_objective_grad(x, q=2.0):
    """
    Objective Q_q(a) = sum_k (a*a)_k^q, gradient wrt log-ratio variables.
    """
    a = _softmax_from_ratios(x)
    n = a.size
    if n <= 1:
        return 0.0, np.zeros_like(x, dtype=np.float64)

    L = _pow2(3 * n - 2)
    A = np.fft.rfft(a, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    c = np.maximum(c, 0.0)

    qq = float(q)
    if qq == 2.0:
        f = float(np.dot(c, c))
        w = c
    else:
        f = float(np.sum(np.power(c, qq)))
        w = np.power(c, qq - 1.0)

    W = np.fft.rfft(w, n=L)
    ga = (2.0 * qq) * np.fft.irfft(W * np.conj(A), n=L)[:n]

    dot = float(np.dot(a, ga))
    gfull = a * (ga - dot)
    gy = gfull[:-1]

    if not np.all(np.isfinite(gy)):
        gy = np.zeros_like(x, dtype=np.float64)
    if not np.isfinite(f):
        f = 1e300

    return f, gy


def _soft_objective_grad(x, eps=1e-5):
    """
    Smoothed max objective:
      eps * logsumexp(c/eps), c = a*a.
    Gradient wrt log-ratio variables.
    """
    a = _softmax_from_ratios(x)
    n = a.size
    if n <= 1:
        return 0.0, np.zeros_like(x, dtype=np.float64)

    L = _pow2(3 * n - 2)
    A = np.fft.rfft(a, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    c = np.maximum(c, 0.0)

    cmax = float(c.max())
    if cmax <= 1e-300:
        return 0.0, np.zeros_like(x, dtype=np.float64)

    eps = float(eps)
    if eps <= 1e-300:
        eps = 1e-12

    z = (c - cmax) / eps
    z = np.clip(z, -745.0, 0.0)
    e = np.exp(z)
    sume = float(e.sum())

    if not np.isfinite(sume) or sume <= 0.0:
        return 1e300, np.zeros_like(x, dtype=np.float64)

    f = cmax + eps * float(np.log(sume))
    w = e / sume

    W = np.fft.rfft(w, n=L)
    ga = 2.0 * np.fft.irfft(W * np.conj(A), n=L)[:n]

    dot = float(np.dot(a, ga))
    gfull = a * (ga - dot)
    gy = gfull[:-1]

    if not np.all(np.isfinite(gy)):
        gy = np.zeros_like(x, dtype=np.float64)

    return f, gy


def _run_minimize_once(fun, x, args, maxiter, bounds=None):
    """One bounded L-BFGS-B call. Returns (x, status)."""
    if _minimize is None:
        return np.asarray(x, dtype=np.float64), 99

    x = np.asarray(x, dtype=np.float64)
    if x.size == 0:
        return x, 1

    if bounds is None:
        bounds = [(-100.0, 100.0)] * int(x.size)

    options = {
        "maxiter": int(maxiter),
        "disp": 0,
        "maxfun": int(max(200, 12 * maxiter)),
        "ftol": 1e-18,
        "gtol": 1e-12,
    }

    try:
        res = _minimize(
            fun,
            x,
            args=args,
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options=options,
        )
    except Exception:
        return x, 99

    x_new = np.asarray(res.x, dtype=np.float64)
    x_new = np.where(np.isfinite(x_new), x_new, 0.0)
    x_new = np.clip(x_new, -100.0, 100.0)

    status = 0
    try:
        status = int(res.status)
    except Exception:
        status = 0

    return x_new, status


def _projected_gradient_search(a0, end_time, seed):
    """Fallback projected gradient/Adam-style search if SciPy is unavailable."""
    a = _project_simplex(a0)
    n = a.size
    L = _pow2(3 * n - 2)
    Lscore = _pow2(2 * n - 1)

    best = a.copy()
    best_score = _score_vector(best, Lscore)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    lr = 1e-3
    lr_min = 1e-7
    lr_max = 1e-2
    rng = np.random.default_rng(seed)

    start = time.monotonic()
    total = max(1e-6, end_time - start)
    last_best = start
    beta1 = 0.9
    beta2 = 0.999

    while time.monotonic() < end_time:
        elapsed = max(0.0, time.monotonic() - start)
        frac = min(1.0, elapsed / total)
        q = 2.0 + 3.0 * frac

        A = np.fft.rfft(a, n=L)
        C = A * A
        c = np.fft.irfft(C, n=L)[:2 * n - 1]
        c = np.maximum(c, 0.0)
        mx = float(c.max())

        if mx <= 1e-300:
            break

        score = 2.0 * float(n) * mx
        now = time.monotonic()

        if score < best_score - 1e-12:
            best_score = score
            best = a.copy()
            last_best = now
            lr = min(lr_max, lr * 1.05)

        if now - last_best > 6.0:
            lr = max(lr_min, lr * 0.75)
            if now - last_best > 15.0:
                a = _project_simplex(best * np.exp(rng.normal(0.0, 0.05, size=n)))
                m.fill(0.0)
                v.fill(0.0)
                t = 0

        w = np.power(c / mx, q - 1.0)
        w = 0.7 * w + 0.3 * (c / mx)

        W = np.fft.rfft(w, n=L)
        g = np.fft.irfft(W * np.conj(A), n=L)[:n]
        g -= float(g.mean())
        rms = float(np.sqrt(np.mean(g * g))) + 1e-12
        g = g / rms

        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        mh = m / (1.0 - beta1 ** t)
        vh = v / (1.0 - beta2 ** t)

        step = lr * mh / (np.sqrt(vh) + 1e-8)
        a_new = _project_simplex(a - step)

        if not np.all(np.isfinite(a_new)):
            a = best.copy()
            m = np.zeros_like(a)
            v = np.zeros_like(a)
            t = 0
        else:
            a = a_new

    return best, best_score


def _local_search(a0, end_time, seed):
    """Random multiplicative block perturbations, accepting only exact score gains."""
    a = _project_simplex(a0)
    n = a.size
    Lscore = _pow2(2 * n - 1)

    best = a.copy()
    best_score = _score_vector(best, Lscore)

    rng = np.random.default_rng(seed)
    sigma = 0.12
    last_best = time.monotonic()
    start = last_best

    while time.monotonic() < end_time:
        cand = best.copy()
        nblocks = 1 + int(rng.integers(0, 2))

        for _ in range(nblocks):
            maxw = max(3, n // 8)
            w = int(np.exp(rng.uniform(np.log(2.0), np.log(float(maxw)))))
            w = max(1, min(w, n))
            s = int(rng.integers(0, n - w + 1))
            cand[s:s + w] *= float(np.exp(rng.normal(0.0, sigma)))

        cand = _project_simplex(cand)
        sc = _score_vector(cand, Lscore)

        if np.isfinite(sc) and sc < best_score - 1e-12:
            best = cand
            best_score = sc
            last_best = time.monotonic()
            sigma = min(0.35, sigma * 1.07)
        else:
            if time.monotonic() - last_best > 5.0:
                sigma = max(0.02, sigma * 0.75)
                if time.monotonic() - last_best > 20.0 and time.monotonic() - start > 30.0:
                    # Occasionally re-center around the best known sequence.
                    cand = _project_simplex(best * np.exp(rng.normal(0.0, 0.03, size=n)))
                    cand_sc = _score_vector(cand, Lscore)
                    if np.isfinite(cand_sc) and cand_sc < best_score:
                        best = cand
                        best_score = cand_sc
                        last_best = time.monotonic()

    return best, best_score


def _refine_start(a0, end_time, seed):
    """
    Refine one starting point for the allotted time.
    Uses L-BFGS if available, otherwise falls back to projected gradient.
    """
    a = _project_simplex(a0)
    n = a.size
    Lscore = _pow2(2 * n - 1)

    best = a.copy()
    best_score = _score_vector(best, Lscore)

    start = time.monotonic()
    total = max(1e-6, end_time - start)
    if total < 2.0:
        return best, best_score

    rng = np.random.default_rng(seed)
    x = _ratios_from_a(a)
    bounds = [(-100.0, 100.0)] * max(0, n - 1)

    # If SciPy is unavailable, use the generic projected gradient search.
    if _minimize is None:
        local_frac = 0.20
        grad_end = start + total * (1.0 - local_frac)
        best, best_score = _projected_gradient_search(best, grad_end, seed)
        if time.monotonic() < end_time:
            best, sc = _local_search(best, end_time, seed + 991)
            if sc < best_score:
                best_score = sc
        return best, best_score

    # Stage 1: smooth L2 surrogate.
    q2_end = start + 0.18 * total
    while time.monotonic() < q2_end:
        x, status = _run_minimize_once(_q_objective_grad, x, (2.0,), 80, bounds)
        if status == 1:
            break

    a = _softmax_from_ratios(x)
    sc = _score_vector(a, Lscore)
    if np.isfinite(sc) and sc < best_score:
        best = a.copy()
        best_score = sc

    # Stage 2: slightly peak-focused L3 surrogate.
    q3_end = start + 0.34 * total
    if time.monotonic() < q3_end:
        while time.monotonic() < q3_end:
            x, status = _run_minimize_once(_q_objective_grad, x, (3.0,), 80, bounds)
            if status == 1:
                break
        a = _softmax_from_ratios(x)
        sc = _score_vector(a, Lscore)
        if np.isfinite(sc) and sc < best_score:
            best = a.copy()
            best_score = sc

    # Stage 3: smoothed maximum with decreasing temperature.
    local_time = max(30.0, 0.08 * total)
    soft_end = end_time - local_time

    if time.monotonic() < soft_end:
        eps = max(1e-7, 0.10 * (best_score / (2.0 * float(n))))
        eps_min = 1e-8
        no_improve_chunks = 0
        last_improve = time.monotonic()

        while time.monotonic() < soft_end:
            remaining = soft_end - time.monotonic()
            chunk = min(remaining, max(8.0, 0.18 * remaining))
            chunk_end = time.monotonic() + chunk

            while time.monotonic() < chunk_end:
                x, status = _run_minimize_once(_soft_objective_grad, x, (eps,), 80, bounds)
                if status == 1:
                    break

            a = _softmax_from_ratios(x)
            sc = _score_vector(a, Lscore)
            now = time.monotonic()

            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best = a.copy()
                last_improve = now
                no_improve_chunks = 0
                eps = max(eps_min, eps * 0.82)
            else:
                no_improve_chunks += 1
                if no_improve_chunks >= 2:
                    x = np.clip(x + rng.normal(0.0, 0.035, size=max(0, x.size)), -100.0, 100.0)
                if eps <= eps_min * 1.01:
                    if no_improve_chunks >= 3:
                        break
                else:
                    eps = max(eps_min, eps * 0.55)

    # Final local search inside this start's budget.
    if time.monotonic() < end_time:
        a, sc = _local_search(best, end_time, seed + 997)
        if np.isfinite(sc) and sc < best_score:
            best = a
            best_score = sc

    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Return the best sequence found before the time budget.
    """
    t0 = time.monotonic()
    budget = max(0.0, float(budget_s) - 8.0)
    deadline = t0 + budget

    # Prefer the provided construction length if it is reasonable.
    n = 3809
    try:
        h = height_sequence_1
        if isinstance(h, (list, np.ndarray)):
            arr = np.asarray(h, dtype=np.float64)
            if arr.ndim == 1 and 100 <= arr.size <= 20000:
                n = int(arr.size)
    except Exception:
        pass

    if n < 32:
        n = 3809

    # Very short fallback: just use a strong one-sided sequence.
    if deadline <= t0 + 5.0:
        a = _power_sequence(n, 0.5, 1.0)
        return [float(x) for x in a]

    cands = _generate_candidates(n)
    scored = _sorted_candidates(cands)

    if not scored:
        a = np.full(n, 1.0 / float(n), dtype=np.float64)
        return [float(x) for x in a]

    global_best_a = np.asarray(scored[0][2], dtype=np.float64).copy()
    global_best_score = float(scored[0][0])

    top = scored[:4]

    # Allocate time to the best starting points.
    alloc = [
        (0.58, 0),
        (0.27, 1),
        (0.15, 2),
    ]

    used = 0.0
    for frac, idx in alloc:
        cur = t0 + used * budget
        alloc_end = min(deadline, t0 + (used + frac) * budget)
        if idx < len(top) and alloc_end > cur + 3.0:
            a = np.asarray(top[idx][2], dtype=np.float64).copy()
            a, sc = _refine_start(a, alloc_end, int(seed) + 137 * (idx + 1))
            if np.isfinite(sc) and sc < global_best_score:
                global_best_score = sc
                global_best_a = np.asarray(a, dtype=np.float64).copy()
        used += frac

    # Final polish on the global best.
    if time.monotonic() < deadline:
        a, sc = _local_search(global_best_a, deadline, int(seed) + 10009)
        if np.isfinite(sc) and sc < global_best_score:
            global_best_score = sc
            global_best_a = np.asarray(a, dtype=np.float64).copy()

    global_best_a = _project_simplex(global_best_a)

    # Final sanity check against the official evaluator if available.
    out = [float(x) for x in global_best_a]
    try:
        val = evaluate_sequence(out)
        if not np.isfinite(val):
            # Fallback to a known-valid simple sequence.
            out = [float(x) for x in _power_sequence(len(out), 0.5, 1.0)]
    except Exception:
        pass

    return out
