def build_A(f, n):
    """Build the constraint matrix A where (f*g)[k] = sum_j A[k,j]*g[j]."""
    # A[k,j] = f[k-j] if 0 <= k-j < n, else 0
    k_idx = np.arange(2*n-1)

    counts = hi - lo + 1

    col_idx = np.empty(counts.sum(), dtype=int)
