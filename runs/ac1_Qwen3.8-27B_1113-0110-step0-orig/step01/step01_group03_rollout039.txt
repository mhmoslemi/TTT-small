def fft_self_conv(a):
    """Compute full self-convolution using FFT for speed."""
    n = len(a)
    if n <= 1:
        return a * a
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:2 * n - 1]

    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        js = np.arange(j_min, j_max + 1)
        rows.extend([k] * len(js))
        cols.extend(js.tolist())
        vals.extend(f[k - js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_direction(f, M, timeout=30.0):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns optimal g vector or None if failed.
    """
    import cvxpy as cp
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    A = build_conv_matrix(f)
    objective = cp.Maximize(cp.sum(g))
    constraints = [A @ g <= M]
    prob = cp.Problem(objective, constraints)
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and np.all(np.isfinite(g.value)):
                return np.maximum(0.0, g.value)
        except Exception:
            continue
    return None


def subgradient_step(a, c, n_peaks=3):
    """Compute subgradient of the score and return update direction."""
    n = len(a)
    S = a.sum()
    M = c.max()
    if M < 1e-15:
        return np.zeros(n)

    # Find top peaks
    peak_thresh = M * 0.999
    peak_indices = np.where(c >= peak_thresh)[0]
    if len(peak_indices) == 0:
        k_stars = [int(np.argmax(c))]
    else:
        sorted_idx = peak_indices[np.argsort(-c[peak_indices])]
        k_stars = []
        for idx in sorted_idx:
            if all(abs(int(idx) - k) > 2 for k in k_stars):
                k_stars.append(int(idx))
            if len(k_stars) >= n_peaks:
                break
        if not k_stars:
            k_stars = [int(np.argmax(c))]

    # Compute subgradient of max(c) averaged over peaks
    dc_da = np.zeros(n)
    for k in k_stars:
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        js = np.arange(j_min, j_max + 1)
        # d(c[k])/d(a[j]) = a[k-j] + a[j-k] (with zero-padding)
        dc_da[js] += a[k - js]
        sym = js - k
        valid = (sym >= 0) & (sym < n)
        dc_da[js[valid]] += a[sym[valid]]
    dc_da /= len(k_stars)

    # Gradient of score = 2n*M/S^2:
    # d(score)/d(a[j]) = 2n*(dc_da[j]*S - 2*M)/S^3
    grad = 2.0 * n * (dc_da * S - 2.0 * M) / (S ** 3)
    return grad


def lp_phase(f, n, deadline, max_iters=200):
    """LP-based refinement phase. Returns (f, score, iterations_done)."""
    c = fft_self_conv(f)
    M = c.max()
    score = M  # score = M when sum(f) = sqrt(2n)
    best_M = M
    best_f = f.copy()
    it = 0
    no_improve = 0

    while time.monotonic() < deadline - 120 and it < max_iters and no_improve < 60:
        t0 = time.monotonic()

        g = lp_direction(f, M)
        solve_time = time.monotonic() - t0

        if g is None or g.sum() < 1e-10:
            no_improve += 3
            it += 1
            continue

        g_norm = g / g.sum() * np.sqrt(2.0 * n)

        improved = False
        for t in [0.15, 0.08, 0.04, 0.02, 0.01]:
            f_new = (1.0 - t) * f + t * g_norm
            f_new = np.maximum(f_new, 0.0)
            S_new = f_new.sum()
            if S_new < 1e-10:
                continue
            f_new = f_new / S_new * np.sqrt(2.0 * n)
            c_new = fft_self_conv(f_new)
            M_new = c_new.max()

            if M_new < best_M - 1e-12:
                f = f_new
                M = M_new
                best_M = M_new
                best_f = f_new.copy()
                score = M_new
                no_improve = 0
                it += 1
                improved = True
                if it % 3 == 0:
                    print(f"  LP it={it}, t={t}: score={M_new:.8f} ({solve_time:.2f}s)")
                break

        if not improved:
            no_improve += 1

    return best_f, best_M, it


def gradient_phase(a, n, deadline, n_peaks=3):
    """Gradient-based fine-tuning phase. Returns (a, best_score)."""
    best_a = a.copy()
    best_score = fast_score(a)
    momentum = np.zeros(n)
    beta = 0.85
    lr = 0.06
    it = 0
    eval_every = 30
    no_imp = 0

    while time.monotonic() < deadline - 30:
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 0.01:
            a = best_a.copy()
            S = a.sum()
        a = a / S * n

        c = fft_self_conv(a)
        grad = subgradient_step(a, c, n_peaks)

        momentum = beta * momentum - lr * grad
        a = a + momentum
        a = np.maximum(a, 0.0)
        S2 = a.sum()
        if S2 > 1e-10:
            a = a / S2 * n

        it += 1
        if it % eval_every == 0:
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < best_score:
                best_a = a.copy()
                best_score = score
                no_imp = 0
                if it % (eval_every * 20) == 0 or (no_imp == 0 and it < eval_every * 10):
                    print(f"  Grad it={it}: score={score:.8f}, lr={lr:.6f}")
            else:
                no_imp += 1

            # Adaptive learning rate
            if no_imp >= 30:
                lr *= 0.7
                momentum *= 0.3
                no_imp = 0
            if lr < 1e-6:
                rng = np.random.default_rng(it)
                perturb = np.exp(rng.normal(0, 0.04, n))
                a = best_a * perturb
                a = normalize_sum(a, n)
                lr = 0.03
                momentum = np.zeros(n)

    return best_a, best_score


def parametric_init(n, rng):
    """Generate diverse parametric initializations."""
    t = np.arange(n, dtype=np.float64)
    x = 2.0 * t / max(1, n - 1) - 1.0
    candidates = []

    # Uniform
    candidates.append(np.ones(n))
    # Linear taper
    candidates.append(np.maximum(1.0 - 0.8 * x, 0.1))
    # Quadratic dome (high center)
    candidates.append(np.maximum(1.0 - 0.7 * x ** 2, 0.05))
    # Quadratic valley (high edges)
    candidates.append(np.maximum(1.0 + 0.7 * (x ** 2 - 1.0 / 3), 0.05))
    # Cosine single period
    candidates.append(np.maximum(1.0 + 0.5 * np.cos(2 * np.pi * t / n), 0.05))
    # Two-period cosine
    candidates.append(np.maximum(1.0 + 0.4 * np.cos(4 * np.pi * t / n), 0.05))
    # Sine
    candidates.append(np.maximum(1.0 + 0.4 * np.sin(2 * np.pi * t / n), 0.05))
    # Spline-like
    candidates.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x), 0.05))
    # Edge-concentrated
    candidates.append(np.maximum(1.0 - np.exp(-((x ** 2 - 1) ** 2) / 0.1), 0.05))

    # Random smooth
    k = rng.integers(3, 8)
    coeffs = rng.normal(0, 0.3, k)
    phis = rng.uniform(0, 2 * np.pi, k)
    wave = np.ones(n)
    for kk in range(1, k + 1):
        wave += coeffs[kk - 1] * np.cos(2 * np.pi * kk * t / n + phis[kk - 1])
    candidates.append(np.maximum(wave, 0.05))

    return candidates


def propose_candidate(seed=42, budget_s=2350, **kwargs):
    """
    Search for the best sequence using multi-phase optimization:
    1. Parametric sweep for initialization
    2. LP-based directional refinement
    3. Gradient-based fine-tuning
    4. Multi-length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    # Get parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        parent = np.ones(512, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    if parent.size == 0:
        parent = np.ones(512, dtype=np.float64)

    n_parent = len(parent)
    best_score = np.inf
    best_a = None

    # === Phase 0: Parametric sweep for good initialization ===
    t = np.arange(n_parent, dtype=np.float64)
    x = 2.0 * t / max(1, n_parent - 1) - 1.0

    families = parametric_init(n_parent, rng)
    # Also add parent as a candidate
    families.append(parent / parent.sum())

    for fam in families:
        a_try = np.maximum(fam, 0.0)
        S = a_try.sum()
        if S < 0.01:
            continue
        a_try = a_try / S * n_parent
        sc = fast_score(a_try)
        if sc < best_score:
            exact_sc = float(evaluate_sequence(a_try.tolist()))
            if np.isfinite(exact_sc) and exact_sc < best_score:
                best_score = exact_sc
                best_a = a_try.copy()

    print(f"After parametric sweep: score={best_score:.8f}, n={n_parent}")

    if best_a is None:
        best_a = np.ones(n_parent)
        best_score = 2.0

    # === Phase 1: LP-based refinement ===
    lp_deadline = deadline - 800
    f = best_a / best_a.sum() * np.sqrt(2.0 * n_parent)
    f, M_lp, lp_iters = lp_phase(f, n_parent, lp_deadline)

    # Evaluate LP result
    a_lp = f / f.sum() * n_parent
    sc_lp = float(evaluate_sequence(a_lp.tolist()))
    if np.isfinite(sc_lp) and sc_lp < best_score:
        best_score = sc_lp
        best_a = a_lp.copy()
    else:
        # LP didn't help; use the LP-score as fallback
        if M_lp < best_score:
            best_score = M_lp
            best_a = a_lp.copy()

    print(f"After LP: score={best_score:.8f}, iters={lp_iters}")

    # === Phase 2: Gradient fine-tuning ===
    grad_deadline = deadline - 60
    a_grad, score_grad = gradient_phase(best_a.copy(), n_parent, grad_deadline, n_peaks=3)
    if score_grad < best_score:
        best_score = score_grad
        best_a = a_grad.copy()

    # === Phase 3: Try a different sequence length ===
    if time.monotonic() < deadline - 120:
        for n_try in [int(n_parent * 0.8), int(n_parent * 1.2)]:
            if n_try < 100 or n_try > 20000:
                continue
            # Initialize by resampling
            a_resample = np.interp(np.linspace(0, n_parent - 1, n_try),
                                   np.arange(n_parent), best_a)
            a_resample = np.maximum(a_resample, 0.0)
            S = a_resample.sum()
            if S < 0.01:
                continue
            a_resample = a_resample / S * n_try

            sc = fast_score(a_resample)
            if sc < best_score + 0.1:
                # Worth trying
                f_try = a_resample / a_resample.sum() * np.sqrt(2.0 * n_try)
                sub_deadline = min(deadline - 60, time.monotonic() + 300)
                f_try, M_t, _ = lp_phase(f_try, n_try, sub_deadline)
                a_t = f_try / f_try.sum() * n_try
                sc_t = float(evaluate_sequence(a_t.tolist()))
                if np.isfinite(sc_t) and sc_t < best_score:
                    best_score = sc_t
                    best_a = a_t.copy()
                    print(f"  n={n_try}: improved to {sc_t:.8f}")

    # Final validation
    if best_a is not None:
        final = normalize_sum(best_a, len(best_a))
        fs = float(evaluate_sequence(final.tolist()))
        if np.isfinite(fs):
            best_score = fs
            best_a = final
        else:
            # Fallback: use parent
            best_a = normalize_sum(parent, n_parent)
            best_score = float(evaluate_sequence(best_a.tolist()))

    print(f"FINAL: score={best_score:.8f}, n={len(best_a)}")
    return [float(x) for x in best_a]
