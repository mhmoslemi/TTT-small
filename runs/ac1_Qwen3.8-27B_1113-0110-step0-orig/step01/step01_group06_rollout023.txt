import numpy as np
import time


def _next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _fft_convolve(a, b=None):
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    L = n + m - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N).real[:L]


def _fft_square(a):
    return _fft_convolve(a, a)


def _score(a, b=None):
    if b is None:
        b = _fft_square(a)
    s = float(a.sum())
    if s < 1e-300:
        return float("inf"), b
    mb = float(b.max())
    sc = 2.0 * len(a) * mb / (s * s)
    return sc, b

    grad b_k with respect to a: (grad)_i = 2*a_{k-i} (out of range = 0).

def _greedy_anti_init(n, rng):
    m = n // 2
    # Greedy: select positions to place "mass" to minimize overlap
    selected = np.zeros(m, dtype=bool)
    counts = np.zeros(2 * n - 1, dtype=np.int64)
    t = 0
    order = rng.permutation(m)
    for idx in order:
        l = int(idx)
        r = n - 1 - int(idx)
        if t == 0:
            y = l
        else:
            sel = np.where(selected[:t])[0]
            if len(sel) > 0:
                ml = int(np.max(counts[sel + l]))
                mr = int(np.max(counts[sel + r]))
                sl = int(counts[2 * l])
                sr = int(counts[2 * r])
                y = l if (ml + sl) <= (mr + sr) else r
            else:
                y = l
        if t > 0:
            sel = np.where(selected[:t])[0]
            counts[sel + y] += 2
        counts[2 * y] += 1
        selected[t] = True
        t += 1
    x = np.full(m, 0.05, dtype=np.float64)
    x[selected] = 0.95
    return x

        if max_g < 1e-18:
            break
        grad_c = grad_c / max_g

        # Backtracking line search
        best_new_score = score
        best_new_a = a
        best_new_b = b
        best_new_eta = eta

        for attempt in range(6):
            trial_eta = eta * (0.7 ** attempt)
            a_new = a * np.exp(-trial_eta * grad_c)
            a_new = np.maximum(a_new, 1e-300)
            a_new = _normalize(a_new)
            b_new = _fft_square(a_new)
            sc_new = float(_score(a_new, b_new)[0])

            if sc_new < best_new_score:
                best_new_score = sc_new
                best_new_a = a_new
                best_new_b = b_new
                best_new_eta = trial_eta

            if sc_new < score * (1.0 - 1e-5):
                break

        if np.isfinite(best_new_score) and best_new_score < score:
            a = best_new_a
            b = best_new_b
            score = best_new_score
            eta = min(eta * 1.08, 10.0)
        else:
            # No improvement: try a random perturbation
            if eta < 0.1:
                a = a * np.exp(rng.normal(0.0, 0.08, size=n))
                a = np.maximum(a, 1e-15)
                a = _normalize(a)
                b = _fft_square(a)
                score = float(_score(a, b)[0])
                eta = 0.5
            else:
                eta *= 0.6

        focus = min(focus * 1.03, 1000.0)
        it += 1

    return best_a, float(best_score)


def _optimize_anti(x0, time_limit, rng, label="", verbose=False):
    """Anti-palindromic optimizer in x-space."""
    x = np.asarray(x0, dtype=np.float64).copy()
    m = len(x)
    n = 2 * m
    x = np.clip(x, 1e-8, 1.0 - 1e-8)

    a = _anti_palindromic_a(x, n)
    b = _fft_square(a)
    score = float(_score(a, b)[0])

    best_x = x.copy()
    best_score = score

    eta = 0.3
    focus = 5.0
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    max_no = 1500
    resets = 0
    max_resets = 20

    while time.monotonic() - t0 < time_limit:
        if score < best_score - 1e-12:
            best_score = score
            best_x = x.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and it % 5000 == 0:
            print(f"  [{label}] it={it} t={time.monotonic()-t0:.1f}s "
                  f"score={score:.6f} best={best_score:.6f} eta={eta:.3f}")

        if no_improve > max_no:
            resets += 1
            if resets > max_resets:
                break
            no_improve = 0
            x = np.clip(best_x + rng.normal(0.0, 0.2, size=m), 1e-8, 1.0 - 1e-8)
            a = _anti_palindromic_a(x, n)
            b = _fft_square(a)
            score = float(_score(a, b)[0])
            eta = 0.3
            focus = float(rng.choice([5.0, 20.0, 50.0, 100.0]))
            continue

        # Gradient in x-space
        w = _softmax_weights(b, focus)
        ga = _grad_weighted(a, b, w)
        # For anti-palindromic: a_i = x_i/m, a_{n-1-i} = (1-x_i)/m
        # d(score)/dx_i = d(score)/da_i - d(score)/da_{n-1-i} / 1
        # (since da_i/dx_i = 1/m and da_{n-1-i}/dx_i = -1/m)
        # So grad_x = (ga[:m] - ga[m:][::-1]) * m  (the m cancels with normalization)
        gx = ga[:m] - ga[m:][::-1]

        # Center for normalization invariance
        gx = gx - float(np.mean(gx))
        max_g = float(np.max(np.abs(gx)))
        if max_g < 1e-18:
            break
        gx = gx / max_g

        # Line search
        best_new_score = score
        best_new_x = x
        for attempt in range(5):
            te = eta * (0.7 ** attempt)
            x_new = np.clip(x - te * gx, 1e-8, 1.0 - 1e-8)
            a_new = _anti_palindromic_a(x_new, n)
            b_new = _fft_square(a_new)
            sc_new = float(_score(a_new, b_new)[0])
            if sc_new < best_new_score:
                best_new_score = sc_new
                best_new_x = x_new
            if sc_new < score * (1.0 - 1e-5):
                break

        if best_new_score < score:
            x = best_new_x
            a = _anti_palindromic_a(x, n)
            b = _fft_square(a)
            score = best_new_score
            eta = min(eta * 1.06, 5.0)
        else:
            eta *= 0.7
            if eta < 0.05:
                x = np.clip(x + rng.normal(0.0, 0.1, size=m), 1e-8, 1.0 - 1e-8)
                a = _anti_palindromic_a(x, n)
                b = _fft_square(a)
                score = float(_score(a, b)[0])
                eta = 0.3

        focus = min(focus * 1.02, 500.0)
        it += 1

    return best_x, float(best_score)


def _top_k_bottleneck_refine(a, time_limit, rng, label=""):
    """Refine by directly minimizing the top-K b_k values using coordinate changes."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = _normalize(a)
    n = len(a)
    best_a = a.copy()
    b = _fft_square(a)
    best_score = float(_score(a, b)[0])

    t0 = time.monotonic()
    it = 0
    top_k_fixed = 8

    while time.monotonic() - t0 < time_limit:
        # Find top-k active constraints
        idxs = np.argpartition(b, -top_k_fixed)[-top_k_fixed:]
        active_max = float(b[idxs].min())  # the k-th largest

        # For each active k, the gradient is 2*a[k-i]
        # We want to decrease these. Try: for a random pair (i, j),
        # transfer mass from i to j in a way that decreases the active b_k.
        
        # Simpler: use the bottleneck gradient with exact top-k
        grad, _ = _bottleneck_grad(a, b, top_k=top_k_fixed)
        grad_c = grad - float(np.mean(grad))
        max_g = float(np.max(np.abs(grad_c)))
        if max_g < 1e-18:
            break
        grad_c /= max_g

        # Try multiple step sizes
        best_sc = score if False else float(_score(a)[0])
        b_cur = _fft_square(a)
        best_sc = float(_score(a, b_cur)[0])
        best_trial = a.copy()

        for eta_try in [0.3, 0.15, 0.075, 0.04, 0.02]:
            a_try = a * np.exp(-eta_try * grad_c)
            a_try = np.maximum(a_try, 1e-15)
            a_try = _normalize(a_try)
            b_try = _fft_square(a_try)
            sc_try = float(_score(a_try, b_try)[0])
            if sc_try < best_sc:
                best_sc = sc_try
                best_trial = a_try

        if best_sc < best_score - 1e-12:
            best_score = best_sc
            best_a = best_trial.copy()
            a = best_trial
            b = _fft_square(a)
        elif best_sc < best_score:
            a = best_trial
            b = _fft_square(a)
        else:
            # Perturb
            a = a * np.exp(rng.normal(0.0, 0.05, size=n))
            a = _normalize(a)
            b = _fft_square(a)

        it += 1

    return best_a, float(best_score)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    seed = int(seed)
    total_budget = max(100.0, float(budget_s))
    t_start = time.monotonic()
    deadline = t_start + total_budget - 40.0  # Leave 40s margin

    print("Starting convolution optimizer v2")
    print(f"Budget: {total_budget}s, deadline in {deadline - t_start:.1f}s")

    # Get initial sequence
    init_a = None
    init_n = 0
    try:
        gs = globals().get("height_sequence_1")
        if gs is not None:
            ga = np.asarray(gs, dtype=np.float64).ravel()
            ga = np.nan_to_num(ga, nan=0.0, posinf=1000.0, neginf=0.0)
            ga = np.clip(ga, 0.0, 1000.0)
            s = float(ga.sum())
            if s > 0.01 and ga.size > 50:
                ga = ga / s
                sc, _ = _score(ga)
                print(f"Given sequence: n={ga.size}, score={sc:.6f}")
                init_a = ga.copy()
                init_n = ga.size
    except Exception as e:
        print(f"Failed to load given sequence: {e}")

    best_a = init_a.copy() if init_a is not None else None
    best_score = float(_score(best_a)[0]) if best_a is not None else float("inf")

    # Time allocation
    # Phase 1: Optimize given sequence (full vector) - most promising
    # Phase 2: Fresh anti-palindromic with larger n
    # Phase 3: Final refinement

    tasks = []
    if init_a is not None:
        # Give most time to the given sequence optimization
        tasks.append(("full_given", init_a.size, 900))
    tasks.append(("full_given2", 8192, 300))  # Slight perturbation of given
    tasks.append(("anti_16k", 16384, 400))
    tasks.append(("anti_8k", 8192, 200))
    tasks.append(("full_8k_fresh", 8192, 200))

    task_idx = 0
    for task_name, n, alloc in tasks:
        now = time.monotonic()
        remaining = deadline - now
        if remaining < 60:
            print(f"Not enough time for {task_name}, stopping tasks")
            break

        actual_alloc = min(alloc, remaining - 100)  # Leave 100s for refinement
        if actual_alloc < 60:
            print(f"Insufficient time for {task_name} ({actual_alloc:.0f}s), skipping")
            break

        print(f"\n--- Task: {task_name}, n={n}, alloc={actual_alloc:.0f}s, "
              f"current best={best_score:.6f} ---")
        t0 = time.monotonic()
        rng = np.random.default_rng(seed * 7919 + task_idx * 104729)

        try:
            if task_name == "full_given" and init_a is not None:
                cand_a, cand_sc = _optimize_full(
                    init_a, actual_alloc, rng, label="given",
                    top_k=15, focus_start=5.0, verbose=True
                )
            elif task_name == "full_given2" and init_a is not None:
                # Perturbed version of given
                a0 = init_a * np.exp(rng.normal(0.0, 0.3, size=init_n))
                a0 = np.maximum(a0, 1e-15)
                a0 = _normalize(a0)
                cand_a, cand_sc = _optimize_full(
                    a0, actual_alloc, rng, label="given2",
                    top_k=12, focus_start=10.0, verbose=False
                )
            elif task_name == "anti_16k":
                x0 = _greedy_anti_init(n, rng)
                x_best, cand_sc = _optimize_anti(
                    x0, actual_alloc, rng, label=f"anti{n}", verbose=True
                )
                cand_a = _anti_palindromic_a(x_best, n)
            elif task_name == "anti_8k":
                x0 = _greedy_anti_init(n, rng)
                # Smooth perturbation
                x0 = x0 + rng.normal(0.0, 0.1, size=n // 2)
                x0 = np.clip(x0, 1e-8, 1.0 - 1e-8)
                x_best, cand_sc = _optimize_anti(
                    x0, actual_alloc, rng, label=f"anti{n}", verbose=False
                )
                cand_a = _anti_palindromic_a(x_best, n)
            elif task_name == "full_8k_fresh":
                a0 = rng.dirichlet(np.ones(n) * 0.5)
                a0 = np.maximum(a0, 1e-15)
                a0 = _normalize(a0)
                cand_a, cand_sc = _optimize_full(
                    a0, actual_alloc, rng, label="fresh8k",
                    top_k=15, focus_start=8.0, verbose=False
                )
            else:
                continue

            if cand_a is not None and np.isfinite(cand_sc):
                cand_a = np.maximum(cand_a, 0.0)
                cand_a = _normalize(cand_a)
                verify_sc, _ = _score(cand_a)
                cand_sc = min(cand_sc, verify_sc)
                if cand_sc < best_score:
                    best_score = cand_sc
                    best_a = cand_a.copy()
                    print(f"  NEW BEST: {best_score:.6f} from {task_name}")
                else:
                    print(f"  {task_name} best: {cand_sc:.6f} (not better than {best_score:.6f})")

        except Exception as e:
            print(f"  Task {task_name} failed: {repr(e)}")

        print(f"  Task {task_name} used {time.monotonic()-t0:.1f}s")
        task_idx += 1

    # Phase: Final refinement of best candidate
    if best_a is not None:
        now = time.monotonic()
        remaining = deadline - now
        if remaining > 80:
            refine_time = min(remaining - 50, 300)
            print(f"\n--- Final refinement: {refine_time:.0f}s ---")
            rng_ref = np.random.default_rng(seed + 12345)
            try:
                refined_a, refined_sc = _top_k_bottleneck_refine(
                    best_a, refine_time, rng_ref, label="refine"
                )
                if refined_sc < best_score:
                    best_a = refined_a
                    best_score = refined_sc
                    print(f"  Refinement improved to {best_score:.6f}")
                else:
                    print(f"  Refinement: {refined_sc:.6f} (no improvement over {best_score:.6f})")
            except Exception as e:
                print(f"  Refinement failed: {repr(e)}")

            # One more pass with exponentiated gradient
            now = time.monotonic()
            remaining = deadline - now
            if remaining > 60:
                t2 = min(remaining - 40, 200)
                print(f"  Second refinement: {t2:.0f}s")
                try:
                    r2_a, r2_sc = _optimize_full(
                        best_a, t2, rng_ref, label="refine2",
                        top_k=20, focus_start=50.0, verbose=False
                    )
                    if r2_sc < best_score:
                        best_a = r2_a
                        best_score = r2_sc
                        print(f"  Second refinement improved to {best_score:.6f}")
                except Exception as e:
                    print(f"  Second refinement failed: {repr(e)}")

    # Final safety check
    if best_a is None:
        best_a = np.ones(1024, dtype=np.float64) / 1024.0

    best_a = np.nan_to_num(best_a, nan=0.0, posinf=1000.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    ss = float(best_a.sum())
    if ss < 0.01:
        best_a = np.ones(1024, dtype=np.float64) / 1024.0
        ss = 1.0
    else:
        best_a = best_a / ss

    final_seq = [float(x) for x in best_a]

    # Verify with exact evaluation if possible
    if best_a.size <= 20000:
        try:
            ev = globals().get("evaluate_sequence")
            if ev is not None and callable(ev):
                exact = float(ev(final_seq))
                print(f"\nFinal: n={best_a.size}, internal={best_score:.6f}, exact={exact:.6f}")
                if not np.isfinite(exact):
                    print("WARNING: exact evaluation failed/inf")
            else:
                print(f"\nFinal: n={best_a.size}, internal={best_score:.6f}")
        except Exception as e:
            print(f"\nFinal: n={best_a.size}, internal={best_score:.6f} (eval error: {e})")
    else:
        print(f"\nFinal: n={best_a.size}, internal={best_score:.6f}")

    print(f"Total time: {time.monotonic() - t_start:.1f}s")
    return final_seq
