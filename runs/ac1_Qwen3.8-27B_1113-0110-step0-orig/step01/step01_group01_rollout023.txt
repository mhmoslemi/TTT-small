def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def normalize_positive(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, float(floor), None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def softmax_seq(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = z - float(np.max(z))
    z = np.clip(z, -60.0, 0.0)
    e = np.exp(z)
    e = np.maximum(e, 1e-300)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(z.size, 1.0 / float(z.size), dtype=np.float64)
    return e / s


def convolve_seq(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if N is None:
        N = next_pow2(2 * n - 1)
    else:
        need = next_pow2(2 * n - 1)
        if int(N) < int(need):
            N = need
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[: 2 * n - 1]


def internal_score(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.inf
    a = a / s
    c = convolve_seq(a, N)
    return float(2.0 * n * float(np.max(c)))


def binary_score(b, n, m):
    """Score for binary sequence b of length n with m ones."""
    c = convolve_seq(b, next_pow2(2 * n - 1))
    cmax = float(np.max(c))
    return float(2.0 * n * cmax) / (float(m) * float(m))


def compute_weights(c, cmax, tau, floor=0.02):
    c = np.asarray(c, dtype=np.float64).ravel()
    L = c.size
    if L == 0 or cmax <= 0.0 or not np.isfinite(cmax) or tau <= 0.0:
        return np.full(L, 1.0 / float(L), dtype=np.float64)
    z = (c - float(cmax)) / float(tau)
    z = np.clip(z, -50.0, 0.0)
    ez = np.exp(z)
    s = float(ez.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(L, 1.0 / float(L), dtype=np.float64)
    w = ez / s
    if floor > 0.0:
        w = (1.0 - floor) * w + (floor / float(L))
    return w


def quadrangle_residue_seq(n, prime=None):
    """Generate a binary sequence based on quadratic residues modulo a prime."""
    if prime is None:
        # Find largest prime <= n that is 1 mod 4 or 3 mod 4
        p = n
        while p > 4:
            if is_prime(p):
                prime = p
                break
            p -= 1
        if prime is None or prime < 8:
            prime = n
    if prime >= n:
        # Pad: place QR( prime ) in first `prime` positions
        p = min(prime, n)
        residues = quadratic_residues(p)
        b = np.zeros(n, dtype=np.float64)
        for r in residues:
            if r < n:
                b[r] = 1.0
        return b
    else:
        # Tile or repeat
        p = prime
        residues = quadratic_residues(p)
        b = np.zeros(n, dtype=np.float64)
        for i in range(n):
            if (i % p) in residues:
                b[i] = 1.0
        return b


def is_prime(x):
    if x < 2:
        return False
    if x < 4:
        return True
    if x % 2 == 0 or x % 3 == 0:
        return False
    i = 5
    while i * i <= x:
        if x % i == 0 or x % (i + 2) == 0:
            return False
        i += 6
    return True


def quadratic_residues(p):
    """Return set of quadratic residues mod p (excluding 0)."""
    if p <= 1:
        return set()
    res = set()
    for x in range(1, p):
        res.add((x * x) % p)
    if 0 in res:
        res.discard(0)
    return res


def make_binary_from_density(n, m, seed, style=0):
    """Create initial binary sequence with m ones out of n positions."""
    rng = np.random.default_rng(seed)
    b = np.zeros(n, dtype=np.float64)
    if style == 0:
        # Random
        idx = rng.choice(n, size=m, replace=False)
        b[idx] = 1.0
    elif style == 1:
        # Quadratic residues
        b = quadrangle_residue_seq(n)
        cur = int(b.sum())
        if cur > m:
            idx = np.nonzero(b > 0.5)[0]
            rng.shuffle(idx)
            b[idx[: cur - m]] = 0.0
        elif cur < m:
            idx = np.nonzero(b < 0.5)[0]
            rng.shuffle(idx)
            b[idx[: m - cur]] = 1.0
    elif style == 2:
        # Arithmetic progression based
        step = max(1, n // m)
        for i in range(m):
            pos = (i * step + rng.integers(0, max(1, step // 3))) % n
            b[pos] = 1.0
    elif style == 3:
        # Low-discrepancy (van der Corput-like)
        for i in range(m):
            v = i + 1
            # Van der Corput in base 2
            x = 0.0
            d = 1.0 / (n + 1)
            v = int(v)
            while v > 0:
                d /= 2.0
                x += (v & 1) * d
                v >>= 1
            pos = int(x * n) % n
            b[pos] = 1.0
    elif style == 4:
        # Fibonacci word based
        golden = 0.6180339887
        for i in range(n):
            frac = (i * golden) % 1.0
            if frac < m / float(n):
                b[i] = 1.0
    # Ensure exactly m ones
    cur = int(np.sum(b > 0.5))
    if cur > m:
        idx = np.nonzero(b > 0.5)[0]
        rng.shuffle(idx)
        for i in idx[: cur - m]:
            b[i] = 0.0
    elif cur < m:
        idx = np.nonzero(b < 0.5)[0]
        rng.shuffle(idx)
        for i in idx[: m - cur]:
            b[i] = 1.0
    return b


def optimize_binary_sa(b, n, m, deadline, seed=0):

    cmax = float(np.max(c))
    score = 2.0 * n * cmax / (m * m)

    best_score = score
    best_cmax = cmax

    rng = np.random.default_rng(seed)
    temp = 0.15
    cooling = 0.997
    min_temp = 0.0005

    one_idx = np.nonzero(b > 0.5)[0]
    zero_idx = np.nonzero(b < 0.5)[0]

    stalled = 0
    it = 0

    while time.monotonic() < deadline and temp > min_temp:
        it += 1
        # Choose move size
        if it % 5000 < 100:
            block_size = int(rng.integers(2, max(3, n // 50) + 1))
        else:
            block_size = 1

        if len(one_idx) == 0 or len(zero_idx) == 0:
            break

        if block_size == 1:
            # Single swap
            i = int(one_idx[rng.integers(len(one_idx))])
            j = int(zero_idx[rng.integers(len(zero_idx))])
            if i == j:
                continue

            old_b = b.copy()
            b[i] = 0.0
            b[j] = 1.0

            # Update convolution incrementally
            two_b = 2.0 * b
            c[i:i + n] -= two_b
            c[2 * i] -= 1.0
            c[j:j + n] += two_b
            c[2 * j] += 1.0

            new_cmax = float(np.max(c))
            new_score = 2.0 * n * new_cmax / (m * m)

            if new_score < score * (1.0 - 1e-12):
                score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_cmax = new_cmax
                    best_b = b.copy()
                one_idx = np.nonzero(b > 0.5)[0]
                zero_idx = np.nonzero(b < 0.5)[0]
                stalled = 0
            else:
                delta = new_score - score
                if delta <= 0 or rng.random() < np.exp(-delta / max(temp, 1e-15)):
                    score = new_score
                    one_idx = np.nonzero(b > 0.5)[0]
                    zero_idx = np.nonzero(b < 0.5)[0]
                    stalled = 0
                else:
                    # Revert
                    b[i] = 1.0
                    b[j] = 0.0
                    c[i:i + n] += two_b
                    c[2 * i] += 1.0
                    c[j:j + n] -= two_b
                    c[2 * j] -= 1.0
                    stalled += 1
        else:
            # Block swap: move a block of ones to new positions
            if len(one_idx) < block_size:
                block_size = max(1, len(one_idx) // 2)
            if block_size < 1:
                continue
            sel_ones = rng.choice(one_idx, size=min(block_size, len(one_idx)), replace=False)
            sel_zeros = rng.choice(zero_idx, size=min(block_size, len(zero_idx)), replace=False)
            if len(sel_ones) != len(sel_zeros):
                mn = min(len(sel_ones), len(sel_zeros))
                sel_ones = sel_ones[:mn]
                sel_zeros = sel_zeros[:mn]
            if mn == 0:
                continue

            old_b = b.copy()
            for i in sel_ones:
                b[i] = 0.0
            for j in sel_zeros:
                b[j] = 1.0

            # Recompute convolution (expensive but block move makes incremental hard)
            c_new = np.rint(convolve_seq(b, N)).astype(np.float64)
            new_cmax = float(np.max(c_new))
            new_score = 2.0 * n * new_cmax / (m * m)

            if new_score < score * (1.0 - 1e-12):
                score = new_score
                c = c_new
                if new_score < best_score:
                    best_score = new_score
                    best_cmax = new_cmax
                    best_b = b.copy()
                one_idx = np.nonzero(b > 0.5)[0]
                zero_idx = np.nonzero(b < 0.5)[0]
                stalled = 0
            else:
                delta = new_score - score
                if delta <= 0 or rng.random() < np.exp(-delta / max(temp, 1e-15)):
                    score = new_score
                    c = c_new
                    one_idx = np.nonzero(b > 0.5)[0]
                    zero_idx = np.nonzero(b < 0.5)[0]
                    stalled = 0
                else:
                    b = old_b.copy()
                    c = np.rint(convolve_seq(b, N)).astype(np.float64)
                    stalled += 1

        # Update temperature
        temp = max(min_temp, temp * cooling)
        if stalled > 2000:
            temp = max(min_temp, temp * 0.5)
            if temp <= min_temp * 1.1:
                break
            stalled = 0

        if it % 20000 == 0:
            print("  binary_sa n={} m={} it={} score={:.8f} best={:.8f} temp={:.5f}".format(
                n, m, it, score, best_score, temp), flush=True)

    return best_b, best_score

    while improved and time.monotonic() < deadline_greedy:
        improved = False
        one_idx = np.nonzero(b > 0.5)[0]
        zero_idx = np.nonzero(b < 0.5)[0]
        if len(one_idx) == 0 or len(zero_idx) == 0:
            break

        best_delta = 0
        best_i = -1
        best_j = -1

        # Sample a subset of swaps to evaluate (full would be O(m*(n-m)))
        sample_size = min(500, len(one_idx) * len(zero_idx))
        for _ in range(sample_size):
            i = int(one_idx[rng.integers(len(one_idx))])
            j = int(zero_idx[rng.integers(len(zero_idx))])
            if i == j:
                continue

            # Removing 1 at i: c changes by -2*b shifted, plus -1 at 2i
            # Adding 1 at j: c changes by +2*b shifted, plus +1 at 2j
            # This is approximate; just compute the effect on max
            two_b = 2.0 * b
            delta_c = np.zeros(2 * n - 1, dtype=np.float64)
            delta_c[i:i + n] -= two_b
            delta_c[2 * i] -= 1.0
            delta_c[j:j + n] += two_b
            delta_c[2 * j] += 1.0
            new_c = c + delta_c
            new_cmax = float(np.max(new_c))
            new_score = 2.0 * n * new_cmax / (m * m)
            delta = new_score - score
            if delta < best_delta:
                best_delta = delta
                best_i = i
                best_j = j

        if best_i >= 0 and best_j >= 0:
            b[best_i] = 0.0
            b[best_j] = 1.0
            two_b = 2.0 * b
            c[best_i:best_i + n] -= two_b

            c[best_j:best_j + n] += two_b

            cmax = float(np.max(c))
            score = 2.0 * n * cmax / (m * m)
            if score < best_score:
                best_score = score
                best_b = b.copy()
            improved = True

    deadline_sa = min(deadline, time.monotonic() + max(1.0, (deadline - time.monotonic())))

def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def normalize_positive(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, float(floor), None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s
