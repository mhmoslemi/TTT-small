import numpy as np
import time

def clean_array(arr):
    a = np.asarray(arr, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    np.clip(a, 0.0, 1000.0, out=a)
    return a

def normalize_array(a):
    a = clean_array(a)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s

def compute_score(p):
    """Compute 2*n*max(conv) for a normalized sequence."""
    p = normalize_array(p)
    n = p.size
    if n == 0:
        return float('inf')
    c = np.convolve(p, p)
    mx = float(np.max(c))
    return 2.0 * n * mx

def score_from_conv(p, c):
    """Compute score from already-computed convolution."""
    n = p.size
    mx = float(np.max(c))
    return 2.0 * n * mx

def make_initial(n, strategy, rng):

def expand_to_length(proto, target_len):
    """Expand a prototype sequence to the target length by block replication."""
    target_len = int(target_len)
    p = normalize_array(proto)
    R = p.size
    if R <= 0 or target_len <= 0:
        return np.ones(max(1, target_len), dtype=np.float64)
    if R == target_len:
        return p.copy()
    if R > target_len:
        # Downsample by averaging
        out = np.zeros(target_len)
        for j in range(target_len):
            lo = int(j * R / target_len)
            hi = max(lo + 1, int((j + 1) * R / target_len))
            out[j] = np.mean(p[lo:hi])
        return normalize_array(out)
    # Upsample by block replication
    base = target_len // R
    rem = target_len % R
    lengths = np.full(R, base, dtype=np.int64)
    lengths[:rem] += 1
    out = np.zeros(target_len)
    idx = 0
    for i in range(R):
        L = int(lengths[i])
        if L > 0:
            out[idx:idx + L] = p[i] / L
            idx += L
    return normalize_array(out)

Looking at this problem, I need to minimize `2*n*max(convolve(a,a))/(sum(a))^2` for non-negative sequences. The current best is 1.555 with target 1.5027.

Key insights I'll exploit:
1. The problem is scale-invariant, so I normalize to sum=1
2. Coordinate descent is effective but gets stuck in local minima - I'll add subgradient-based escape
3. Multi-scale optimization (short → long) finds good basin-of-attraction
4. The paired anti-correlation structure (a[i] vs a[n-1-i]) is crucial
5. I can efficiently update the convolution when changing one coordinate using the formula: `conv_new[k] = (conv[k] + (x-old)*2*a[k-i] - (x²-old²)*[k==2i]) / S²`

"""
Hybrid optimizer for min 2*n*max(convolve(a,a))/(sum(a))^2, a>=0.
Uses multi-scale coordinate descent with subgradient escape, diverse
initializations, and efficient O(n) per-coordinate convolution updates.

import numpy as np

def _clean(arr):
    a = np.asarray(arr, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    np.clip(a, 0.0, 1000.0, out=a)
    return a


def _norm(a):
    a = _clean(a)
    s = float(a.sum())
    if s <= 0 or not np.isfinite(s):
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def _score(p, c=None):
    n = p.size
    if n == 0:
        return float('inf')
    if c is None:
        c = np.convolve(p, p)
    return 2.0 * n * float(np.max(c))


def _init(n, strat, rng):

def _expand(proto, tgt):
    p = _norm(proto)
    R = p.size
    tgt = int(tgt)
    if R == tgt:
        return p.copy()
    if R > tgt:
        out = np.zeros(tgt)
        for j in range(tgt):
            lo, hi = int(j*R/tgt), max(int(j*R/tgt)+1, int((j+1)*R/tgt))
            out[j] = np.mean(p[lo:hi])
        return _norm(out)
    base = tgt // R
    rem = tgt % R
    lens = np.full(R, base, dtype=np.int64)
    lens[:rem] += 1
    out = np.zeros(tgt)
    idx = 0
    for i in range(R):
        L = int(lens[i])
        if L > 0:
            out[idx:idx+L] = p[i]/L
            idx += L
    return _norm(out)

    if i > 0:
        m = float(np.max(c[:i]))
        if m > max_unaff:
            max_unaff = m

        m = float(np.max(c[i_end:]))
        if m > max_unaff:
            max_unaff = m
    
    A_k = c[aff_k] - 2.0*old*p[partner] + old*old*is_c
    B_k = 2.0*p[partner]
    C_k = is_c
    
    xs = grid[(grid >= 0) & (grid <= 1e3)]
    if xs.size == 0:
        return p, c, 2.0*n*float(np.max(c)), False

    S2 = S*S
    num = A_k[:, None] + B_k[:, None]*xs[None, :] - C_k[:, None]*(xs[None,:]**2)
    max_num = np.max(num, axis=0)
    max_c = np.maximum(max_unaff, max_num)
    scores = np.full(xs.size, float('inf'))
    v = S2 > 1e-14
    scores[v] = 2.0*n*max_c[v]/S2[v]
    
    cur = 2.0*n*float(np.max(c))

    best_x, best_sc = float(xs[idx]), float(scores[idx])

        hi = float(xs[idx])*5.0 if idx == xs.size-1 else float(xs[idx+1])
        if best_x > 0:
            loc = best_x*np.logspace(-1.5, 1.5, refine_n)
        else:
            loc = np.linspace(0, max(hi, 1e-8), refine_n)
        loc = np.unique(np.concatenate([loc, [lo, best_x, hi]]))
        loc = loc[(loc >= 0) & (loc <= 1e3)]
        if loc.size > 1:
            S_l = D + loc
            S2_l = S_l*S_l
            num_l = A_k[:, None] + B_k[:, None]*loc[None, :] - C_k[:, None]*(loc[None,:]**2)
            max_c_l = np.maximum(max_unaff, np.max(num_l, axis=0))
            sc_l = np.full(loc.size, float('inf'))
            vl = S2_l > 1e-14
            sc_l[vl] = 2.0*n*max_c_l[vl]/S2_l[vl]
            i2 = int(np.argmin(sc_l))
            if sc_l[i2] < best_sc:
                best_x, best_sc = float(loc[i2]), float(sc_l[i2])
    
    if best_sc >= cur - 1e-12:
        return p, c, cur, False
    
    S = D + best_x
    S2 = S*S
    p_new = p / S
    p_new[i] = best_x / S
    c_new = c.copy()
    c_new[aff_k] = (A_k + B_k*best_x - C_k*best_x*best_x) / S2
    c_new[:i_start] /= S2
    if i_end < 2*n-1:
        c_new[i_end:] /= S2
    
    sn = float(p_new.sum())
    if sn <= 0 or not np.isfinite(sn):
        return p, c, cur, False
    p_new /= sn
    c_new /= (sn*sn)
    np.clip(p_new, 0, None, out=p_new)

def _subgrad_step(p, c, lr, rng):
    n = p.size
    M = float(np.max(c))
    active = np.where(c > M * 0.85)[0]
    if active.size == 0:
        active = np.where(c == M)[0]
    
    subgrad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            subgrad[lo:hi+1] += 2.0 * p[k-hi:k-lo+1][::-1]
    subgrad /= active.size
    p_new = p - lr * subgrad
    p_new = np.clip(p_new, 0, None)
    s = float(p_new.sum())
    if s <= 0 or not np.isfinite(s):
        return p.copy()
    return p_new / s


def _coord_opt(p, c, score, deadline, rng, grid, refine_n, max_escapes=5):

    stag, esc = 0, 0
    noise_lrs = [0.5, 0.3, 0.15, 0.07, 0.03]
    
    while time.monotonic() < deadline - 0.2:
        order = rng.permutation(n)
        improved = False
        for i in order:
            if time.monotonic() > deadline - 0.15:

            p, c, score, imp = _coord_step(p, c, int(i), grid, refine_n)
            if imp:
                improved = True
        
        # Periodic recomputation to fix drift
        c = np.convolve(p, p)
        np.clip(c, 0, None, out=c)
        score = 2.0 * n * float(np.max(c))
        
        if score < best_sc - 1e-12:
            best_sc, best_p, best_c = score, p.copy(), c.copy()
            stag = 0
        else:
            stag += 1
        
        if not improved and stag > 1:
            if esc < max_escapes:
                lr = noise_lrs[min(esc, len(noise_lrs)-1)]
                p = _subgrad_step(best_p, best_c, lr, rng)
                p = best_p * np.exp(rng.normal(0, lr, n))

                sn = float(p.sum())

                    p /= sn
                    c = np.convolve(p, p)

                    score = 2.0*n*float(np.max(c))
                    if score < best_sc - 1e-10:

                        stag = 0
                esc += 1
                stag = 0
            elif stag > 5:
                break
        if improved:
            stag = 0
    
    return best_p, best_c, best_sc

    full_candidates = []
    if best_full is not None:
        full_candidates.append((best_full_sc, best_full.copy()))
    
    # Expand best medium
    for sc, pm in med_candidates[:3]:
        exp = _expand(pm, FINAL_LEN)
        full_candidates.append((_score(exp), exp))
    
    # Expand best short
    for sc, np_, pp in proto_results[:3]:
        exp = _expand(pp, FINAL_LEN)
        full_candidates.append((_score(exp), exp))
    
    # Direct full starts
    for strat in [6, 9, 1, 7]:
        pf = _init(FINAL_LEN, strat, rng)
        full_candidates.append((_score(pf), pf))
    
    full_candidates.sort(key=lambda x: x[0])
    print(f"Full candidates: best={full_candidates[0][0]:.6f}")
    
    # Optimize best full candidates
    gc_f = 16
    rc_f = 5
    grid_f = np.concatenate(([0.0], np.logspace(-9.0, 2.5, gc_f-1)))
    
    opt_deadline = deadline - 25.0
    n_opt = min(3, len(full_candidates))
    per_cand = max(15.0, (opt_deadline - time.monotonic()) / max(1, n_opt))
    
    for idx in range(n_opt):
        if time.monotonic() > opt_deadline:
            break
        sc, pf = full_candidates[idx]
        c = np.convolve(pf, pf)
        np.clip(c, 0, None, out=c)
        td = min(opt_deadline, time.monotonic() + per_cand)
        pf, c, sc = _coord_opt(pf, c, sc, td, rng, grid_f, rc_f)
        update_best(pf, sc)
        print(f"  Full opt {idx}: {sc:.6f}")
    
    # Phase 4: Final polish on absolute best
    if time.monotonic() < deadline - 15:
        if best_full is not None:
            p_fin = best_full.copy()
        elif best_p is not None:
            p_fin = _expand(best_p, FINAL_LEN)
        else:
            p_fin = base.copy()
        
        c_fin = np.convolve(p_fin, p_fin)
        np.clip(c_fin, 0, None, out=c_fin)
        sc_fin = _score(p_fin, c_fin)
        print(f"Final polish start: {sc_fin:.6f}")
        
        # Use finer grid for final polish
        gc_fin = 18
        rc_fin = 6
        grid_fin = np.concatenate(([0.0], np.logspace(-10.0, 2.5, gc_fin-1)))
        p_fin, c_fin, sc_fin = _coord_opt(p_fin, c_fin, sc_fin, deadline - 10.0, rng, grid_fin, rc_fin, max_escapes=3)
        update_best(p_fin, sc_fin)
        print(f"Final polish done: {sc_fin:.6f}")
    
    # Choose final answer
    chosen = best_full if best_full is not None else (best_p if best_p is not None else base)
    chosen = _norm(chosen)
    
    # Verify with external evaluator
    try:
        final_score = evaluate_sequence([float(x) for x in chosen])
        print(f"External eval: {final_score:.6f}")
    except Exception:
        pass
    
    return [float(x) for x in chosen]
