```python
def subgradient_descent_v2(a_init, deadline, seed=0):
    """
    Sharp subgradient descent with momentum and adaptive step size.
    Minimizes 2*n*max(c) subject to a >= 0, sum(a) = 1.
    """
    n = len(a_init)
    a = project_simplex(a_init.copy())
    rng = np.random.default_rng(seed)
    
    c = convolve_full(a)
    cmax = float(np.max(c))
    best_a = a.copy()
    best_cmax = cmax
    
    # Parameters
    eta = 0.005  # step size
    momentum = 0.7
    m = np.zeros(n)  # momentum buffer
    k = 5  # number of hot spots to consider
    
    stale = 0
    it = 0
    
    while time.monotonic() < deadline:
        # Update k adaptively
        if cmax < 1.0 / n * 1.1:  # close to optimal
            k = 1
        elif cmax < 1.0 / n * 1.5:
            k = 3
        else:
            k = 5
        
        # Compute subgradient from top-k positions
        g, c = subgradient_topk(a, k)
        cmax = float(np.max(c))
        
        # Center and normalize
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        
        # Momentum
        m = momentum * m + g
        mn = np.linalg.norm(m)
        if mn < 1e-15:
            break
        m /= mn
        
        # Step
        a_new = project_simplex(a - eta * m)
        
        # Evaluate
        c_new = convolve_full(a_new)
        cmax_new = float(np.max(c_new))
        
        if cmax_new < cmax:
            a = a_new
            c = c_new
            cmax = cmax_new
            if cmax < best_cmax:
                best_cmax = cmax
                best_a = a.copy()
            stale = 0
            eta = min(0.05, eta * 1.02)
        else:
            stale += 1
            eta *= 0.8
            if stale > 100:
                # Perturbation
                scale = 0.02 + 0.03 * (stale / 500.0)
                a = project_simplex(a + rng.normal(0, scale, n))
                c = convolve_full(a)
                cmax = float(np.max(c))
                if cmax < best_cmax:
                    best_cmax = cmax
                    best_a = a.copy()
                stale = 0
                eta = 0.003
            if stale > 1000:
                break
        
        it += 1
        if it % 2000 == 0:
            print("subgrad2 n={} it={} cmax={:.6f} score={:.9f}".format(n, it, cmax, 2.0*n*cmax), flush=True)
    
    return best_a, 2.0 * n * best_cmax
```

```python
def smoothed_subgradient(a_init, deadline, seed=0):
    n = len(a_init)
    a = project_simplex(a_init.copy())
    rng = np.random.default_rng(seed)
    
    best_a = a.copy()
    best_cmax = float(np.max(convolve_full(a)))
    
    # Smoothing parameter schedule: start large, decrease
    tau = 0.5  # initial smoothing (relative to cmax)
    eta = 0.01  # step size
    
    it = 0

        c = convolve_full(a)
        cmax = float(np.max(c))
        
        # Soft max weights
        tau_eff = tau * cmax
        if tau_eff < 1e-10:
            tau_eff = 1e-10
        z = (c - cmax) / tau_eff
        z = np.clip(z, -50, 0)
        w = np.exp(z)
        w /= w.sum()
        
        # Gradient
        g = compute_gradient_from_weights(w, a, n)
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn

        a_new = project_simplex(a - eta * g)
```

"""
Search for nonnegative step-function heights minimizing
    2*n*max(convolve(a,a)) / (sum(a))^2

Uses adaptive subgradient descent with top-k hotspot focus,
Nesterov-style momentum, simplex projection, and multi-scale
restart strategy. Key insight: the max autocorrelation is
typically at the middle lag, and the subgradient there is
proportional to the reversed sequence, guiding the search.
"""

import time
import numpy as np


def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} using exact algorithm."""
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return v
    if n == 1:
        return np.array([max(0.0, float(v[0]))], dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def normalize_sum1(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 1e-30:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def convolve_full(a):
    n = len(a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return c


def internal_score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    a_n = a / s
    c = convolve_full(a_n)
    return float(2.0 * n * np.max(c))


def compute_gradient_weights(w, a, n):
    """Compute gradient: g_i = 2 * sum_j w_j * a[j-i] for valid j-i."""
    g = np.zeros(n, dtype=np.float64)
    for j in range(len(w)):
        wj = float(w[j])
        if wj < 1e-18:
            continue
        lo = max(0, j - n + 1)
        hi = min(n - 1, j)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            g[lo:hi + 1] += wj * 2.0 * a[j - idx]
    return g


def gradient_topk(a, k):
    """Compute subgradient using top-k convolution positions."""
    n = len(a)
    c = convolve_full(a)
    cmax = float(np.max(c))
    L = len(c)
    if k < L:
        top_k = np.argsort(c)[-k:]
    else:
        top_k = np.arange(L)
    g = np.zeros(n, dtype=np.float64)
    for j in top_k:
        j = int(j)
        lo = max(0, j - n + 1)
        hi = min(n - 1, j)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            g[lo:hi + 1] += 2.0 * a[j - idx]
    g /= float(len(top_k))
    return g, c, cmax


def subgrad_descent(a_init, deadline, seed=0, k_init=3, eta_init=0.008,
                    momentum_init=0.7, verbose=True):
    """
    Adaptive subgradient descent with momentum and top-k focus.
    Minimizes 2*n*max(c) subject to a >= 0, sum(a) = 1.
    """
    n = len(a_init)
    a = project_simplex(a_init.copy())
    rng = np.random.default_rng(seed)

    c = convolve_full(a)
    cmax = float(np.max(c))
    best_a = a.copy()
    best_cmax = cmax
    best_score = 2.0 * n * best_cmax

    eta = eta_init
    momentum = momentum_init
    m = np.zeros(n, dtype=np.float64)
    k = k_init

    stale = 0
    it = 0
    accept_worse_prob = 0.05
    T = 0.1  # "temperature" for simulated annealing

    while time.monotonic() < deadline:
        g, c, cmax = gradient_topk(a, k)

        # Center gradient (direction on simplex)
        g -= g.mean()
        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            break
        g /= gn

        # Momentum
        m = momentum * m + g
        mn = float(np.linalg.norm(m))
        if mn < 1e-15:
            m = g
            mn = gn
        m /= mn

        # Step
        a_new = project_simplex(a - eta * m)

        # Evaluate
        c_new = convolve_full(a_new)
        cmax_new = float(np.max(c_new))

        improved = cmax_new < cmax * (1.0 - 1e-12)
        accepted = False

        if improved:
            accepted = True
        else:
            # Simulated annealing: accept worse with probability
            delta = (cmax_new - cmax) / max(cmax, 1e-15)
            if delta < T and rng.random() < accept_worse_prob:
                accepted = True

        if accepted:
            a = a_new
            c = c_new
            cmax = cmax_new
            if cmax < best_cmax:
                best_cmax = cmax
                best_a = a.copy()
                best_score = 2.0 * n * best_cmax
            stale = 0
            if improved:
                eta = min(0.08, eta * 1.04)
                momentum = min(0.92, momentum * 1.002)
                # Adapt k: decrease when making good progress
                if k > 1 and cmax < best_cmax * 1.01:
                    k = max(1, k - 1 if it % 200 == 0 else k)
            else:
                eta = max(0.002, eta * 0.9)
        else:
            stale += 1
            eta = max(0.0005, eta * 0.75)
            if stale > 50:
                # Random perturbation to escape
                scale = 0.01 + 0.04 * min(1.0, stale / 300.0)
                a_pert = project_simplex(a + rng.normal(0, scale, n))
                c_pert = convolve_full(a_pert)
                cmax_pert = float(np.max(c_pert))
                if cmax_pert < cmax:
                    a = a_pert
                    c = c_pert
                    cmax = cmax_pert
                    if cmax < best_cmax:
                        best_cmax = cmax
                        best_a = a.copy()
                        best_score = 2.0 * n * best_cmax
                elif cmax_pert < best_cmax:
                    pass  # not better than current, skip
                stale = 0
                eta = 0.005
                momentum = 0.7
                m *= 0.1
                # Cool down
                T = max(0.005, T * 0.8)
            if stale > 800:
                # Bigger restart
                a = project_simplex(a + rng.normal(0, 0.1, n))
                c = convolve_full(a)
                cmax = float(np.max(c))
                m = np.zeros(n)
                eta = 0.004
                stale = 0
                T = 0.05
            if stale > 2000:
                break

        it += 1
        if verbose and it % 3000 == 0:
            print("subgrad n={} it={} cmax={:.6f} score={:.9f} k={} eta={:.5f}".format(
                n, it, cmax, 2.0 * n * cmax, k, eta), flush=True)

    return best_a, best_score


def lowdim_optimize(n, deadline, seed=0, a_init=None):
    """Low-dimensional basis optimization using scipy L-BFGS-B."""
    try:
        import scipy.optimize as _scipy_optimize
    except Exception:
        _scipy_optimize = None

    if _scipy_optimize is None:
        return None, np.inf

    # Build basis
    i = np.arange(n, dtype=np.float64)
    x = (i - 0.5 * (n - 1)) / max(1.0, 0.5 * n)
    cols = []
    for m in range(1, 6):
        cols.append(np.sin(np.pi * float(m) * x))
    for m in range(1, 5):
        t = 2.0 * np.pi * float(m) / n
        cols.append(np.sin(t))
        cols.append(np.cos(t))
    cols.append(x - 0.5)
    cols.append(x * x - np.mean(x * x))
    cols.append(np.sin(4.0 * np.pi * x))
    cols.append(np.cos(4.0 * np.pi * x))

    B = np.column_stack(cols)
    B -= B.mean(axis=0)
    norms = np.linalg.norm(B, axis=0)
    norms[norms <= 1e-14] = 1.0
    B = B / norms
    D = B.shape[1]
    N = next_pow2(3 * n - 2)

    # Initial theta
    theta0 = np.zeros(D, dtype=np.float64)
    if a_init is not None and len(a_init) == n:
        z = np.log(np.maximum(a_init, 1e-15))
        z -= z.mean()
        theta0, _, _, _ = np.linalg.lstsq(B, z, rcond=None)
        theta0 = np.asarray(theta0, dtype=np.float64).ravel()

    def objective(theta):
        z = B @ theta
        z = z - np.max(z)
        z = np.clip(z, -60, 0)
        e = np.exp(z)
        a = e / e.sum()
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
        cmax = float(np.max(c))
        return 2.0 * n * cmax

    def grad(theta):
        z = B @ theta
        z = z - np.max(z)
        z = np.clip(z, -60, 0)
        e = np.exp(z)
        a = e / e.sum()
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
        cmax = float(np.max(c))
        # Soft weight with small tau
        tau = 0.05 * cmax
        zc = (c - cmax) / tau
        zc = np.clip(zc, -50, 0)
        w = np.exp(zc)
        w /= w.sum()
        # Gradient of max(c) w.r.t. a
        ga = np.zeros(n)
        for j in range(len(w)):
            if w[j] < 1e-18:
                continue
            lo = max(0, j - n + 1)
            hi = min(n - 1, j)
            if lo <= hi:
                idx = np.arange(lo, hi + 1)
                ga[lo:hi + 1] += w[j] * 2.0 * a[j - idx]
        # Chain rule: d(theta->a)
        ga_centered = ga - a @ ga
        return 2.0 * n * (B.T @ ga_centered)

    best_theta = theta0.copy()
    best_score = objective(theta0)
    rng = np.random.default_rng(seed)

    max_starts = 4
    starts = [theta0.copy()]
    for std in (0.5, 1.0, 1.5):
        starts.append(theta0 + rng.normal(0, std, D))
    starts = starts[:max_starts]

    for th in starts:
        if time.monotonic() > deadline:
            break
        try:
            res = _scipy_optimize.minimize(
                objective, th, method='L-BFGS-B', jac=grad,
                options={'maxiter': 30, 'ftol': 1e-14, 'gtol': 1e-8, 'maxls': 20}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                sc = objective(res.x)
                if sc < best_score:
                    best_score = sc
                    best_theta = res.x.copy()
        except Exception:
            pass

    z = B @ best_theta
    z = z - np.max(z)
    a = np.exp(np.clip(z, -60, 0))
    a = a / a.sum()
    a = np.maximum(a, 0.0)
    return a, best_score
