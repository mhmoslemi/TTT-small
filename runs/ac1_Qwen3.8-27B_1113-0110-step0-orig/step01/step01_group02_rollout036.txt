import numpy as np
import time
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = rfft(a, fft_size)
    ca = irfft(fa * fa, fft_size)
    return ca[:size]

def _score(a):
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    c = _fft_convolve(a_norm)
    return 2.0 * len(a_norm) * float(np.max(c))

def _sinusoidal_seq(n, amplitudes, phases):
    x = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    for k, (A, phi) in enumerate(zip(amplitudes, phases), 1):
        a += A * np.cos(2 * np.pi * k * x / n + phi)
    return np.maximum(a, 0.0)

def _build_conv_matrix(a):

    # This is a banded Toeplitz matrix

    # But j must be in [0, n-1]

    A = lil_matrix((m, n), dtype=np.float64)

        lo = max(0, k - n + 1)
        hi = min(n - 1, k)

def _lp_project(a, M):
    """Solve LP: max sum(b) s.t. b>=0, (a*b)[k] <= M for all k."""
    n = len(a)

def _subgradient_vec(a, k_max):
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k_max - n + 1)
    hi = min(n - 1, k_max)
    if lo <= hi:
        j_range = np.arange(lo, hi + 1)
        indices = k_max - j_range
        g[lo:hi + 1] = 2.0 * a[indices]
    return g
