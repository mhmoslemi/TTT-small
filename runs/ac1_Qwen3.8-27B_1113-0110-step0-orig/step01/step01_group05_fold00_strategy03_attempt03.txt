def clean_array(arr):
    arr = np.asarray(arr, dtype=np.float64)
    if arr.ndim != 1:
        return np.ones(1, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    return arr.clip(0, None)

def score_array(seq):
    n = len(seq)
    scores = np.zeros(n, dtype=np.float64)
    shifted = np.zeros(n, dtype=np.float64)
    try:
        convolve_scores = np.convolve(seq, seq, mode="full")
        scores = convolve_scores
    except Exception:
        pass
    
    max_conv = np.max(scores) if n>0 else 0.0
    total = np.sum(seq)
    score_val = float(2.0 * n * max_conv) / (total * total) if total > 0 else float('inf')
    return score_val

def score_with_normalization(seq):
    if not (np.isfinite(seq).all()):
        return float('inf')
    normalized_sequence = clean_array(seq)
    s = float(np.sum(normalized_sequence))
    if s <= 0:
        return float('inf')
    # Score is defined using normalized sum
    convolved = np.convolve(normalized_sequence, normalized_sequence, mode="full")
    max_conv = np.max(convolved)
    score_val = float(2.0 * len(normalized_sequence) * max_conv) / (s * s)
    return score_val
