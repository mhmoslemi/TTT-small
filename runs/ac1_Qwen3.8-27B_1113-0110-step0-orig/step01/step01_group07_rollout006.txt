Multi-initialization, multi-phase Adam optimizer for low-peak-autocorrelation
non-negative sequences. Uses FFT-based fast scoring, smooth-max gradient,
and periodic restarts to escape local minima.
"""

import time
import numpy as np


def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def fast_score(a):
    """Compute 2n * max(convolution) for normalized non-negative sequence."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n <= 0:
        return np.inf
    s = float(a.sum())
    if s <= 1e-300:
        return np.inf
    a = a / s
    L = next_pow2(2 * n - 1)

"""
Multi-initialization Adam optimizer with smooth-max gradient for finding
non-negative sequences minimizing 2n*max(conv)/sum^2.
Uses FFT-based fast evaluation, multiple starting points (linear ramp, cosine,
low-freq random, previous best), multi-phase optimization with decreasing
smoothing, and periodic restarts to escape local minima.
"""

import time
import numpy as np


def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def fast_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n <= 0:
        return np.inf
    s = float(a.sum())
    if s <= 1e-300:
        return np.inf
    a = a / s
    L = next_pow2(2 * n - 1)
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0:
        return np.inf
    return float(2.0 * n * cm)


def smooth_max_grad(a, tau, L):
    n = a.size
    P = np.fft.rfft(a, L)
    c = np.fft.irfft(P * P, L)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0:
        return np.inf, np.zeros(n)
    z = (c - cm) / tau
    z = np.clip(z, -50.0, 0.0)
    expz = np.exp(z)
    expsum = float(expz.sum())
    if not np.isfinite(expsum) or expsum <= 0:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
    else:
        w = expz / expsum
    W_fft = np.fft.rfft(w, L)
    A_fft = np.fft.rfft(a, L)
    h = np.fft.irfft(np.conj(W_fft) * A_fft, L)[:n]
    h = np.where(np.isfinite(h), h, 0.0)
    grad = 2.0 * h
    return cm, grad


def normalize_prob(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        return None
    return a / s


def make_init_ramp(n, alpha=0.85):
    x = np.linspace(0.0, 1.0, n)
    a = 1.0 - alpha * x
    return normalize_prob(a)


def make_init_cosine(n, alpha=0.7):
    x = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * np.cos(np.pi * x)
    return normalize_prob(a)


def make_init_bump(n, alpha=0.6):
    x = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * np.sin(np.pi * x) ** 2
    return normalize_prob(a)


def make_init_lowfreq(n, rng, num_modes=6, amp=0.3):
    phases = rng.uniform(0, 2 * np.pi, num_modes)
    freqs = rng.choice(np.arange(2, n // 4 + 1), size=num_modes, replace=False)
    amps = rng.uniform(0.1, amp, num_modes)
    x = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for f, p, am in zip(freqs, phases, amps):
        a += am * np.cos(2 * np.pi * f * x + p)
    return normalize_prob(a)

    tilt = 1.0 - alpha * x
    freq = rng.integers(2, max(3, n // 20))

                  lr_start=0.03, lr_min=0.002, restart_every=4000,
                  noise_scale=0.15):
    a = a_init.copy()
    n = a.size
    L = next_pow2(2 * n - 1)
    deadline = time.monotonic() + max(0.0, duration_s)

    z = np.log(np.maximum(a, 1e-300))
    z = np.where(np.isfinite(z), z, -50.0)
    z -= z.mean()
    z = np.clip(z, -60.0, 60.0)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step_count = 0

    best_a = a.copy()
    best_score = fast_score(a)
    no_improve = 0
    tau = tau_start
    lr = lr_start

    while time.monotonic() < deadline:
        # Adaptive tau: decrease geometrically
        tau = max(tau_min, tau * 0.9995)
        lr = max(lr_min, lr * 0.9998)

        cm, grad = smooth_max_grad(a, tau, L)
        if not np.isfinite(cm):
            break

        # Gradient w.r.t. log-parameters: a * grad, minus mean for simplex
        lg_grad = a * grad
        mean_lg = float(np.dot(a, lg_grad))
        lg_grad -= mean_lg

        # Clip gradient magnitude
        mg = float(np.max(np.abs(lg_grad)))
        if mg > 1e-2:
            lg_grad *= (1e-2 / mg)

        # Adam update
        step_count += 1
        m_buf = beta1 * m_buf + (1 - beta1) * lg_grad
        v_buf = beta2 * v_buf + (1 - beta2) * (lg_grad * lg_grad)
        b1 = 1.0 - beta1 ** step_count
        b2 = 1.0 - beta2 ** step_count
        z -= lr * (m_buf / b1) / (np.sqrt(v_buf / b2) + eps)
        z = np.where(np.isfinite(z), z, 0.0)
        z -= z.mean()
        z = np.clip(z, -60.0, 60.0)

        z_shift = z - z.max()
        e = np.exp(z_shift)
        s = float(e.sum())
        if not np.isfinite(s) or s <= 0:
            a = np.ones(n) / n
        else:
            a = e / s

        true_s = 2.0 * n * cm

            best_score = true_s
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Periodic restart
        if no_improve > restart_every:
            zz = np.log(np.maximum(best_a, 1e-300))
            zz = np.where(np.isfinite(zz), zz, -50.0)
            zz += rng.normal(0, noise_scale, n)
            zz -= zz.mean()
            z = np.clip(zz, -60.0, 60.0)
            m_buf *= 0.1
            v_buf *= 0.1
            no_improve = 0
            tau = max(tau_start * 0.5, tau * 1.5)
            lr = max(lr_start * 0.5, lr * 1.3)
            noise_scale *= 0.85
            if step_count % 20000 == 0:

        if step_count % 10000 == 0:
            print(f"  step {step_count} tau={tau:.5f} lr={lr:.5f} "
                  f"true={true_s:.6f} best={best_score:.6f} noimp={no_improve}")

def lofti_optimize(a_init, duration_s, rng):
    """L-infinity optimization via iterative projection (Frank-Wolfe style)."""
    a = a_init.copy()
    n = a.size
    L = next_pow2(2 * n - 1)
    deadline = time.monotonic() + max(0.0, duration_s)
    best_a = a.copy()
    best_score = fast_score(a)
    stall = 0

    for it in range(100000):
        if time.monotonic() > deadline:
            break
        # Find the maximizing lag
        P = np.fft.rfft(a, L)
        c = np.fft.irfft(P * P, L)[:2 * n - 1]
        c = np.where(np.isfinite(c), c, 0.0)
        c = np.maximum(c, 0.0)
        cm = float(c.max())
        k_max = int(np.argmax(c))

        # Subgradient direction: 2 * (delta_k * a)
        # delta_k is one-hot at k_max, so (delta_k * a)[i] = a[k_max - i] (valid)
        sg = np.zeros(n)
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        for i in range(lo, hi + 1):
            sg[i] = a[k_max - i]
        sg *= 2.0

        # Frank-Wolfe: move toward minimizing sg . a subject to a >= 0, sum a = 1
        # Linear minimizer: put all mass on argmin sg
        # But that's too extreme. Use step size.
        mean_sg = float(np.mean(sg))
        # Projected gradient step
        step = 0.01 / max(1.0, float(np.max(np.abs(sg))))
        new_a = a - step * (sg - mean_sg)
        new_a = np.maximum(new_a, 0)
        sn = float(new_a.sum())
        if sn > 1e-12:
            new_a /= sn
        else:
            new_a = a

        # Also try exponential tilting
        tilt = np.exp(-0.05 * sg / max(cm, 1e-12))
        tilted = a * tilt
        st = float(tilted.sum())
        if st > 1e-12:
            tilted /= st

        # Pick best of the two
        s1 = fast_score(new_a)
        s2 = fast_score(tilted)
        cur_s = fast_score(a)
        cand = min([(s1, new_a), (s2, tilted), (cur_s, a)], key=lambda x: x[0])
        if cand[0] < best_score - 1e-10:
            best_score = cand[0]
            best_a = cand[1].copy()
        a = cand[1]
        if cand[0] >= best_score - 1e-10:
            stall += 1
        else:
            stall = 0
        if stall > 500:
            a = normalize_prob(best_a * np.exp(rng.normal(0, 0.1, n)))
            if a is None:
                a = best_a.copy()
            stall = 0

        if (it + 1) % 20000 == 0:
            print(f"  lofti iter {it+1} best {best_score:.6f}")

    return best_a, best_score

            p4, s4 = lofti_optimize(best_local, d4, rng)
            try_update(p4, f"n={n} lofti")

    # Final: try a few targeted perturbations of the best
    rem = deadline - time.monotonic()
    if rem > 30 and best_seq is not None:
        n_best = best_seq.size
        for attempt in range(3):
            if time.monotonic() > deadline:
                break
            noise = np.exp(rng.normal(0, 0.05 + 0.03 * attempt, n_best))
            perturbed = normalize_prob(best_seq * noise)
            if perturbed is not None:
                dur = min(30.0, (deadline - time.monotonic()) * 0.4)
                pp, sp = adam_optimize(perturbed, dur, rng,
                                       tau_start=0.005, tau_min=0.001,
                                       lr_start=0.005, lr_min=0.001,
                                       restart_every=5000, noise_scale=0.05)
                try_update(pp, f"final pert #{attempt}")

    # Validate with actual evaluator
    final_seq = None
    final_score = np.inf
    candidates = []
    if best_seq is not None:
        candidates.append(best_seq)
    # Fallback
    candidates.append(np.ones(1024) / 1024.0)

    for cand in candidates:
        lst = [float(max(0.0, min(1000.0, x))) for x in np.asarray(cand, dtype=np.float64)]
        if not lst:
            continue
        try:
            sc = evaluate_sequence(lst)
        except Exception:
            sc = np.inf
        if np.isfinite(sc) and sc < final_score:
            final_score = float(sc)
            final_seq = lst
            print(f"Validated candidate: score={sc:.6f} (n={len(lst)})")

    if final_seq is None:
        final_seq = [1.0 / 1024.0] * 1024

    print(f"\nFinal best score: {final_score:.6f}")
    print(f"Total time used: {time.monotonic() - t0:.1f}s")
    return final_seq
