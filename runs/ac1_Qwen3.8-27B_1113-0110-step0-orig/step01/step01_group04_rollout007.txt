def _pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _clean(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    return np.maximum(v, 0.0)


def _project_simplex(v):
    v = _clean(v)
    n = v.size
    if n == 0:
        return v.copy()
    x = np.sort(v)[::-1]
    css = np.cumsum(x)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = x > (css - 1.0) / idx
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    r = int(np.nonzero(cond)[0][-1])
    tau = (css[r] - 1.0) / float(r + 1)
    out = v - tau
    np.maximum(out, 0.0, out=out)
    s = out.sum()
    if not np.isfinite(s) or s <= 1e-15:
        return np.full(n, 1.0 / n)
    return out / s


def _score(a):
    v = _clean(a)
    n = v.size
    if n == 0:
        return np.inf
    s = v.sum()
    if s < 1e-12:
        return np.inf
    v = v / s
    L = _pow2(2 * n - 1)
    A = np.fft.rfft(v, n=L)
    c = np.fft.irfft(A * A, n=L)[:2 * n - 1]
    mx = max(0.0, float(c.max()))
    return 2.0 * n * mx


def _conv(a):

    A = np.fft.rfft(a, n=L)
    c = np.fft.irfft(A * A, n=L)[:2 * n - 1]

def _sigmoid_bump(n, center, width, steepness):

def _ratios(a):
    a = _clean(a).astype(np.float64)
    n = a.size
    s = a.sum()
    if s <= 0 or not np.isfinite(s):
        a = np.full(n, 1.0 / n)
        s = 1.0
    else:
        a = a / s
    if n == 1:
        return np.zeros(0)
    ref = a[-1]
    if ref <= 1e-200:
        a[-1] = 1e-12
        a /= a.sum()
        ref = a[-1]
    x = np.log(np.maximum(a[:-1], 1e-300) / max(ref, 1e-300))
    return np.clip(x, -100.0, 100.0)


def _from_ratios(x):
    x = np.asarray(x, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    n = x.size + 1
    if n == 1:
        return np.ones(1)
    m = max(0.0, float(x.max())) if x.size else 0.0
    e = np.exp(np.clip(x - m, -745.0, 709.0))
    e_last = float(np.exp(-m))
    s = float(e.sum()) + e_last
    if not np.isfinite(s) or s <= 0:
        return np.full(n, 1.0 / n)
    a = np.empty(n)
    a[:-1] = e / s
    a[-1] = e_last / s
    return a


def _q_obj_grad(x, q=2.0):
    a = _from_ratios(x)
    n = a.size
    if n <= 1:
        return 0.0, np.zeros_like(x)
    L = _pow2(3 * n - 2)
    A = np.fft.rfft(a, n=L)
    c = np.maximum(np.fft.irfft(A * A, n=L)[:2 * n - 1], 0.0)
    qq = float(q)
    if qq == 2.0:
        f = float(np.dot(c, c))
        w = c.copy()
    else:
        c_safe = np.maximum(c, 1e-300)
        f = float(np.sum(np.power(c_safe, qq)))
        w = np.power(c_safe, qq - 1.0)
    W = np.fft.rfft(w, n=L)
    ga = (2.0 * qq) * np.fft.irfft(W * np.conj(A), n=L)[:n]
    dot = float(np.dot(a, ga))
    gy = a * (ga - dot)
    gy = gy[:-1]
    if not np.all(np.isfinite(gy)):
        gy = np.zeros_like(x)
    if not np.isfinite(f):
        f = 1e300
    return f, gy


def _soft_obj_grad(x, eps=1e-5):
    a = _from_ratios(x)
    n = a.size
    if n <= 1:
        return 0.0, np.zeros_like(x)
    L = _pow2(3 * n - 2)
    A = np.fft.rfft(a, n=L)
    c = np.maximum(np.fft.irfft(A * A, n=L)[:2 * n - 1], 0.0)
    cmax = float(c.max())
    if cmax <= 1e-300:
        return 0.0, np.zeros_like(x)
    eps = max(float(eps), 1e-12)
    z = np.clip((c - cmax) / eps, -745.0, 0.0)
    e = np.exp(z)
    sume = float(e.sum())
    if not np.isfinite(sume) or sume <= 0:
        return 1e300, np.zeros_like(x)
    f = cmax + eps * float(np.log(sume))
    w = e / sume
    W = np.fft.rfft(w, n=L)
    ga = 2.0 * np.fft.irfft(W * np.conj(A), n=L)[:n]
    dot = float(np.dot(a, ga))
    gy = np.maximum(a * (ga - dot), -1e10)[:-1]
    if not np.all(np.isfinite(gy)):
        gy = np.zeros_like(x)
    return f, gy


def _lbfgs_step(fun, x, args, maxiter, bounds=None):
    if _minimize is None:
        return x, 99
    x = np.asarray(x, dtype=np.float64)
    if x.size == 0:
        return x, 1
    if bounds is None:
        bounds = [(-100.0, 100.0)] * x.size
    try:
        res = _minimize(fun, x, args=args, method="L-BFGS-B", jac=True,
                        bounds=bounds,
                        options={"maxiter": int(maxiter), "disp": 0,
                                 "maxfun": int(max(300, 15 * maxiter)),
                                 "ftol": 1e-18, "gtol": 1e-12})
        xn = np.clip(np.where(np.isfinite(res.x), res.x, 0.0), -100, 100)
        return xn, int(getattr(res, "status", 0))
    except Exception:
        return x, 99


def _lp_direction(a, M, n):
    """
    Solve: max sum(g) s.t. A@g <= M*1, g >= 0.
    A[k,i] = a[k-i] for 0<=k-i<n.
    Returns normalized g or None on failure.
    """
    if _linprog is None:
        return None
    # Build constraint matrix A_ub (2n-1) x n
    # A_ub[k, i] = a[k-i] if 0 <= k-i <= n-1 else 0
    # This is a Toeplitz-like banded matrix.
    try:
        A_ub = np.zeros((2 * n - 1, n), dtype=np.float64)
        for i in range(n):
            a_slice = a  # a[i] contributes to positions k = i, i+1, ..., i+n-1
            A_ub[i:i + n, i] = a
        b_ub = np.full(2 * n - 1, M)
        c_obj = -np.ones(n)  # minimize -sum(g)
        bounds = [(0.0, None)] * n
        res = _linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            g = np.maximum(_clean(res.x), 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
        return None
    except Exception:
        return None


def _subgradient_step(a, step_size, n):
    """
    One subgradient step on max(c). Find k*=argmax c, then push a down
    at positions i where a[k*-i] is large.
    """
    c = _conv(a)
    k_star = int(np.argmax(c))
    # Subgradient of c[k*] w.r.t. a[i] is a[k*-i] (if 0<=k*-i<n)
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j <= n - 1:
            grad[i] = a[j]
    # Center and scale
    grad = grad - grad.mean()
    rms = np.sqrt(np.mean(grad * grad)) + 1e-12
    grad /= rms
    a_new = _project_simplex(a - step_size * grad)
    return a_new

        alloc_end = t0 + (sum(fracs[:idx + 1])) * budget
        alloc_end = min(alloc_end, deadline - 30.0)  # leave 30s for final polish
        if alloc_end - cur < 10.0:
            break

        a0 = np.asarray(scored[idx][2], dtype=np.float64).copy()
        print(f"[{time.monotonic()-t0:.1f}s] Refining candidate {idx} (sc={scored[idx][0]:.6f}), "
              f"time_until={alloc_end-time.monotonic():.0f}s")
        a, sc = _refine_start(a0, alloc_end, int(seed) + 100 * (idx + 1))
        if np.isfinite(sc) and sc < global_best_sc:
            global_best_sc = sc
            global_best_a = a.copy()
            print(f"  -> improved to {sc:.6f}")

    # Final global polish
    if time.monotonic() < deadline:
        print(f"[{time.monotonic()-t0:.1f}s] Final local search, budget={deadline-time.monotonic():.0f}s")
        a, sc = _local_search(global_best_a, deadline - 5.0, int(seed) + 50000)
        if np.isfinite(sc) and sc < global_best_sc:
            global_best_sc = sc
            global_best_a = a
            print(f"  -> final polish improved to {sc:.6f}")

    global_best_a = _project_simplex(global_best_a)
    out = [float(x) for x in global_best_a]

    try:
        val = evaluate_sequence(out)
        print(f"[{time.monotonic()-t0:.1f}s] Final evaluate_sequence = {val}")
        if not np.isfinite(val):
            out = [float(x) for x in _power_seq(n, 0.5, 1.0)]
    except Exception:
        pass

    return out
