Search for nonnegative step-function heights minimizing
    2*n*max(convolve(a,a)) / (sum(a))^2
using full-sequence L-BFGS with smooth-max surrogate,
progressive temperature reduction, and multi-start diversity.
"""

import numpy as np
import scipy.optimize as sci_opt

def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()

def normalize_positive(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, float(floor), None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s

def convolve_seq(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if N is None:
        N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[: 2 * n - 1]

    n = a.size
    if n == 0:

    s = float(a.sum())

"""
Search for nonnegative step-function heights minimizing
    2*n*max(convolve(a,a)) / (sum(a))^2
using full-sequence L-BFGS with smooth-max surrogate,
progressive temperature reduction, diverse multi-start,
and multi-resolution (n=2048, n=4096) optimization.
"""

import time
import numpy as np
import scipy.optimize as sci_opt

def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()

def normalize_positive(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, float(floor), None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s

def convolve_seq(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if N is None:
        N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[: 2 * n - 1]

def internal_score(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.inf
    a = a / s
    c = convolve_seq(a, N)
    return float(2.0 * n * float(np.max(c)))

def obj_and_grad(z, n, N, tau):
    """Objective and gradient for f(a)=2n*smoothmax(conv(a,a)) with a=softmax(z)."""
    z_shifted = z - float(np.max(z))
    a = np.exp(np.clip(z_shifted, -700.0, 0.0))
    s = float(a.sum())
    if s <= 0 or not np.isfinite(s):
        return 1e10, np.zeros(n)
    a = a / s

    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[: 2 * n - 1]
    cmax = float(np.max(c))
    zc = (c - cmax) / tau
    zc = np.clip(zc, -50.0, 0.0)
    ez = np.exp(zc)
    s_exp = float(ez.sum())
    if not np.isfinite(s_exp) or s_exp <= 0:
        w = np.zeros(c.size)
        w[int(np.argmax(c))] = 1.0
        smax = cmax
    else:
        w = ez / s_exp
        smax = cmax + tau * np.log(s_exp)

    fval = 2.0 * n * smax

    # Gradient: df/da[i] = 4n * sum_k w[k]*a[k-i] = 4n * d[i]
    # where d[i] = sum_{m} a[m]*w[i+m]
    # Using FFT: conv[n-1+i] = d[i]
    w_rev = w[::-1]
    fw = np.fft.rfft(w_rev, N)
    conv = np.fft.irfft(fa * fw, N)
    d = conv[n - 1: 2 * n - 1]
    grad_da = 4.0 * n * d

    dot = float(np.dot(grad_da, a))
    grad_dz = grad_da * a - a * dot

    return float(fval), grad_dz

def run_lbfgs(a_init, deadline, n_tau=8, max_iter=None, verbose=False):
    """Run L-BFGS with progressive tau reduction."""
    a = normalize_positive(a_init, floor=1e-15)
    n = a.size
    N = next_pow2(2 * n - 1)

    c = convolve_seq(a, N)
    cmax = float(np.max(c))
    if cmax <= 0 or not np.isfinite(cmax):
        return a.copy(), internal_score(a, N)

    if max_iter is None:
        max_iter = 60

    tau0 = cmax * 0.25
    tau_ratios = [1.0, 0.6, 0.35, 0.2, 0.12, 0.07, 0.04, 0.02, 0.01, 0.005][:n_tau]

    z = np.log(a + 1e-20)
    z -= z.mean()

    best_a = a.copy()
    best_score = internal_score(a, N)

    for ti, ratio in enumerate(tau_ratios):
        if time.monotonic() > deadline:
            break
        tau = tau0 * ratio
        if tau < 1e-12:
            tau = 1e-12
        try:
            res = sci_opt.minimize(
                obj_and_grad, z, method='L-BFGS-B', jac=True,
                args=(n, N, tau),
                options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10, 'maxls': 40}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                z_new = res.x - float(np.max(res.x))
                a_new = np.exp(np.clip(z_new, -700.0, 0.0))
                a_new = a_new / float(a_new.sum())
                a_new = normalize_positive(a_new, floor=0.0)
                sc = internal_score(a_new, N)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    if verbose:
                        print("    tau={:.2e} score={:.12f}".format(tau, best_score), flush=True)
                z = res.x - float(np.max(res.x))
        except Exception:
            pass

    return best_a, best_score

def generate_diverse_starts(a_base, n, rng, count=30):
    """Generate diverse starting points around a_base."""
    starts = []
    a_resized = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, a_base.size), a_base)
    a_resized = normalize_positive(a_resized, floor=1e-15)
    starts.append(a_resized.copy())

    log_a = np.log(a_resized + 1e-20)
    log_centered = log_a - log_a.mean()
    x = np.arange(n, dtype=np.float64) / n

    # Log-space Gaussian perturbations at various scales
    for scale in (0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0):
        pert = rng.normal(0, scale, n)
        z = log_centered + pert
        a_try = np.exp(z - z.max())
        a_try = a_try / a_try.sum()
        starts.append(normalize_positive(a_try, floor=1e-15))

    # Wave perturbations
    for freq in (1, 2, 3, 4, 5, 7, 11, 17, 29, 47):
        for amp in (0.05, 0.1, 0.2):
            phase = rng.uniform(0, 2 * np.pi)
            wave = amp * np.sin(2 * np.pi * freq * x + phase)
            z = log_centered + wave
            a_try = np.exp(z - z.max())
            a_try = a_try / a_try.sum()
            starts.append(normalize_positive(a_try, floor=1e-15))

    # Polynomial perturbations
    for deg in (2, 3, 4, 5, 6):
        coeffs = rng.normal(0, 0.15, deg + 1)
        poly = np.polyval(coeffs, x)
        z = log_centered + poly
        a_try = np.exp(z - z.max())

    for _ in range(10):

        print("Phase 5: Final precision", flush=True)
        n_f = best_a.size
        c_f = convolve_seq(best_a, next_pow2(2 * n_f - 1))
        cmax_f = float(np.max(c_f))
        tau_final = cmax_f * 0.01
        a_f, sc_f = run_lbfgs(best_a, deadline - 5, n_tau=3, max_iter=100)
        if sc_f < best_score:
            best_score = sc_f
            best_a = a_f.copy()
            print("Phase 5: score={:.12f}".format(best_score), flush=True)

    # Final
    best_a = normalize_positive(best_a, floor=0.0)
    out_list = [float(x) for x in best_a]
    try:
        final_eval = float(evaluate_sequence(out_list))
    except Exception:
        final_eval = float('inf')

    if not np.isfinite(final_eval):
        best_a = np.ones(1024, dtype=np.float64) / 1024.0
        out_list = [float(x) for x in best_a]
        final_eval = 2.0

    elapsed = time.monotonic() - start_time
    print("FINAL: internal={:.12f}, evaluated={:.12f}, n={}, time={:.1f}s".format(
        best_score, final_eval, len(out_list), elapsed), flush=True)
    return out_list
