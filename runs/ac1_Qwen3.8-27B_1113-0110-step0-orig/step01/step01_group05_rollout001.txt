"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative a.

Uses a combination of:
  - Projected subgradient / mirror descent for global progress
  - Coordinate descent with log-space grids for local polishing
  - Multiple diverse starting points
  - FFT-based autocorrelation for efficient peak tracking
  - Periodic perturbation to escape local minima
"""

import numpy as np
import time

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, None)
    s = float(np.sum(a))
    if s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / max(1, a.size)
    return a / s

def _score(a):
    a = _normalize(a)
    n = a.size
    r = np.convolve(a, a)
    mx = float(np.max(r))
    return float(2.0 * n * mx)

def _autocorr(a):
    return np.convolve(a, a)

def _subgradient_step(a, r, step):
    """One subgradient + projection step. Returns updated a."""
    n = a.size
    k_star = int(np.argmax(r))
    # grad[j] = 2 * a[k_star - j] with zero-padding
    grad = np.zeros(n, dtype=np.float64)
    idx = k_star - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    grad[mask] = 2.0 * a[idx[mask]]
    
    # Mirror descent (exponentiated gradient)
    # a_new[j] = a[j] * exp(-step * grad[j]) / Z
    log_val = np.log(np.maximum(a, 1e-300)) - step * grad
    log_val -= np.max(log_val)  # for numerical stability
    w = np.exp(log_val)
    sw = float(np.sum(w))
    if sw <= 1e-300:
        return a.copy()
    return w / sw

def _subgradient_descent(a, max_iters=200, deadline=None, verbose=False):
    """Run subgradient descent with mirror descent."""
    a = a.copy()
    n = a.size
    best_a = a.copy()
    best_score = _score(a)
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        r = _autocorr(a)
        score = float(2.0 * n * np.max(r))
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Adaptive step size
        grad_norm = _grad_norm(r, a)
        if grad_norm < 1e-300:
            break
        step = 1.0 / (np.sqrt(it + 1) * grad_norm)
        step = min(step, 0.5)  # cap the step
        
        a_new = _subgradient_step(a, r, step)
        
        # Check if the step improved things
        score_new = _score(a_new)
        if score_new < score:
            a = a_new
        else:
            # Try smaller step
            for factor in [0.5, 0.25, 0.1]:
                a_try = _subgradient_step(a, r, step * factor)
                sc_try = _score(a_try)
                if sc_try < score:
                    a = a_try
                    break
            else:
                # Accept anyway to keep moving (with small step)
                a = _subgradient_step(a, r, step * 0.01)
        
        if verbose and it % 20 == 0:
            print(f"  Subgrad iter {it}: score={score:.6f}, best={best_score:.6f}")
    
    return best_a, best_score

def _grad_norm(r, a):
    n = a.size
    k_star = int(np.argmax(r))
    idx = k_star - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    grad = np.zeros(n, dtype=np.float64)
    grad[mask] = 2.0 * a[idx[mask]]
    return float(np.linalg.norm(grad))

def _coordinate_descent(a, r, score, deadline, grid, refine, max_perturb=3, rng=None):
    """Coordinate descent polishing. Similar to existing code but streamlined."""
    if rng is None:
        rng = np.random.default_rng(42)
    n = a.size
    best_a = a.copy()
    best_r = r.copy()
    best_score = score
    
    stagnant = 0
    perts = 0
    noise_scales = [0.5, 0.25, 0.1, 0.05]
    
    while time.monotonic() < deadline - 0.5:
        order = rng.permutation(n)
        improved = False
        for i in order:
            if time.monotonic() > deadline - 0.3:
                break
            a, r, score, imp = _coord_update(a, r, score, int(i), grid, refine)
            if imp:
                improved = True
        
        # Renormalize and recompute
        s = float(np.sum(a))
        if s <= 1e-15:
            break
        a = a / s
        r = _autocorr(a)
        np.clip(r, 0.0, None, out=r)
        score = float(2.0 * n * np.max(r))
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_r = r.copy()
            stagnant = 0
        else:
            stagnant += 1
            if score > best_score * 1.1 + 1e-8:
                a = best_a.copy()
                r = best_r.copy()
                score = best_score
        
        if not improved:
            stagnant += 1
            if stagnant >= 3 and perts < max_perturb:
                scale = noise_scales[min(perts, len(noise_scales)-1)]
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.clip(a, 0, None)
                s = np.sum(a)
                if s > 1e-15:
                    a /= s
                    r = _autocorr(a)
                    score = float(2.0 * n * np.max(r))
                perts += 1
                stagnant = 0
            elif stagnant >= 8:
                break
    
    return best_a, best_r, best_score

def _coord_update(a, r, score, i, grid, refine):
    """Update coordinate i using log-space grid."""
    n = a.size
    old = float(a[i])
    
    # Convolution entries affected by changing a[i]
    # r[k] = sum_j a[j] * a[k-j]
    # Changing a[i] affects r[k] for k where 0<=i<n and 0<=k-i<n
    # i.e., k in [i, i+n-1]
    
    coeff = 2.0 * a.copy()
    const = -2.0 * old * a.copy()
    # Special case: k = 2i (where a[i]*a[i] term)
    mid = 2 * i  # not used directly, handled below
    
    # For each k, the new r[k] = r[k] + 2*x*a[k-i] - 2*old*a[k-i] + (x^2 - old^2 * ...) 
    # Actually: r[k] involves a[i] in terms where i appears.
    # r_new[k] = r_old[k] + (x - old) * (a[k-i] + a[i-k]) + (x^2 - old^2) * [k==2i]
    # where a[m]=0 for m<0 or m>=n
    
    ks_lo = max(0, i)
    ks_hi = min(2*n-2, i+n-1)
    k_range = np.arange(ks_lo, ks_hi+1)
    
    # Compute the "affine" part: for each k, the coefficient of x and the constant
    # r_new[k] = r[k] + (x - old) * (a[k-i] + a[i-k]) + [k==2i](x^2 - old^2)
    #          = r[k] - old*(a[k-i]+a[i-k]) + x*(a[k-i]+a[i-k]) + [k==2i](x^2 - old^2)
    
    a_kmi = np.zeros(len(k_range), dtype=np.float64)
    for idx_k, k in enumerate(k_range):
        ki = k - i
        ik = i - k
        val = 0.0
        if 0 <= ki < n: val += a[ki]
        if 0 <= ik < n: val += a[ik]
        a_kmi[idx_k] = val
    
    # Find max of r outside the affected range
    max_unaff = 0.0
    if ks_lo > 0:
        m = float(np.max(r[:ks_lo]))
        if m > max_unaff: max_unaff = m
    if ks_hi < 2*n-2:
        m = float(np.max(r[ks_hi+1:]))
        if m > max_unaff: max_unaff = m
    
    r_k = r[ks_lo:ks_hi+1].copy()
    
    # Evaluate on grid
    is_mid = (k_range == 2*i)
    
    xs = grid
    # Affine update: r_new[k] = r[k] - old*a_kmi[k] + x*a_kmi[k] + is_mid*(x^2 - old^2)
    base = r_k - old * a_kmi  # constant part
    lin = a_kmi  # coefficient of x
    quad = is_mid.astype(np.float64)  # coefficient of x^2
    
    aff = base[None, :] + lin[None, :] * xs[:, None] + quad[None, :] * (xs[:, None] ** 2)
    max_aff = np.max(aff, axis=1)
    maxc = np.maximum(max_unaff, max_aff)
    
    S = float(np.sum(a))  # current sum
    S_new = S + xs - old
    
    vals = np.full(xs.size, float('inf'), dtype=np.float64)
    valid = S_new > 1e-15
    vals[valid] = (2.0 * n) * maxc[valid] / (S_new[valid] ** 2)
    
    idx_best = int(np.argmin(vals))
    best_x = float(xs[idx_best])
    best_val = float(vals[idx_best])
    
    # Refine
    if refine > 0 and xs.size > 1:
        left = 0.0 if idx_best == 0 else float(xs[idx_best - 1])
        right = float(xs[idx_best]) * 10.0 if idx_best == xs.size - 1 else float(xs[idx_best + 1])
        lo = max(0.0, left * 0.5)
        hi = right * 1.5
        if hi <= lo: hi = lo + 1e-6
        local = np.logspace(np.log10(max(lo, 1e-12)), np.log10(hi), refine)
        local = np.unique(np.concatenate(([lo, best_x, hi], local)))
        local = local[(local >= 0) & (local <= 1e4)]
        
        if local.size > 1:
            aff2 = base[None, :] + lin[None, :] * local[:, None] + quad[None, :] * (local[:, None] ** 2)
            maxc2 = np.maximum(max_unaff, np.max(aff2, axis=1))
            S_new2 = S + local - old
            vals2 = np.full(local.size, float('inf'), dtype=np.float64)
            valid2 = S_new2 > 1e-15
            vals2[valid2] = (2.0 * n) * maxc2[valid2] / (S_new2[valid2] ** 2)
            idx2 = int(np.argmin(vals2))
            if vals2[idx2] < best_val:
                best_x = float(local[idx2])
                best_val = float(vals2[idx2])
    
    tol = 1e-12 * max(1.0, score)
    if not np.isfinite(best_val) or best_val >= score - tol:
        return a, r, score, False
    
    # Apply update
    a_new = a.copy()
    a_new[i] = best_x
    S_new = S + best_x - old
    if S_new <= 1e-15:
        return a, r, score, False
    a_new /= S_new
    
    r_new = _autocorr(a_new)
    np.clip(r_new, 0.0, None, out=r_new)
    score_new = float(2.0 * n * np.max(r_new))
    
    if not np.isfinite(score_new) or score_new >= score - tol:
        return a, r, score, False
    
    return a_new, r_new, score_new, True

def _make_starts(n, rng):
    """Generate diverse starting points."""
    starts = []
    
    # Uniform
    starts.append(np.ones(n) / n)
    
    # Two-level half block (2:1 ratio)
    a = np.zeros(n)
    m = n // 2
    a[:m] = 2.0 / (3.0 * m)
    a[m:] = 1.0 / (3.0 * (n - m))
    starts.append(a / a.sum())
    
    # Two-level 3:1
    a = np.zeros(n)
    a[:m] = 3.0 / (4.0 * m)
    a[m:] = 1.0 / (4.0 * (n - m))
    starts.append(a / a.sum())
    
    # Two-level 5:1
    a = np.zeros(n)
    a[:m] = 5.0 / (6.0 * m)
    a[m:] = 1.0 / (6.0 * (n - m))
    starts.append(a / a.sum())
    
    # Triangular (peak in middle)
    x = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 - np.abs(x - mid) / (mid + 1)
    a = np.maximum(a, 0.01)
    starts.append(a / a.sum())
    
    # Raised cosine
    a = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    a = np.maximum(a, 0.01)
    starts.append(a / a.sum())
    
    # Exponential decay
    a = np.exp(-3.0 * x / n)
    starts.append(a / a.sum())
    
    # Linear ramp
    a = x + 1.0
    starts.append(a / a.sum())
    
    # Dirichlet (several)
    for alpha in [1.0, 0.5, 0.3]:
        try:
            a = rng.dirichlet(np.full(n, alpha))
            starts.append(a)
        except:
            pass
    
    # Random (Gaussian, clipped)
    for _ in range(3):
        a = np.abs(rng.normal(0, 1, n)) + 0.1
        starts.append(a / a.sum())
    
    # Mirror-paired
    m2 = (n + 1) // 2
    w = rng.dirichlet(np.full(m2, 1.0))
    a = np.zeros(n)
    for j in range(m2):
        a[j] = w[j] * 0.7
        if n - 1 - j != j:
            a[n-1-j] = w[j] * 0.3
        else:
            a[j] = w[j]
    starts.append(a / a.sum())
    
    # Multi-level (4 levels)
    a = np.zeros(n)
    q = n // 4
    a[:q] = 0.4 / q
    a[q:2*q] = 0.3 / q
    a[2*q:3*q] = 0.2 / (n - 3*q + 1) if n > 3*q else 0.2 / max(1, q)
    a[3*q:] = 0.1 / max(1, n - 3*q)
    starts.append(a / a.sum())
    
    # Three-level with gaps
    a = np.zeros(n)
    a[::3] = 2.0 / (n/3)
    a[1::3] = 1.0 / (n/3)
    # leave a[2::3] = 0
    if a.sum() > 0:
        starts.append(a / a.sum())
    
    return starts

def _expand_to_length(proto, target_len):
    """Expand a prototype sequence to target length by block replication."""
    p = _normalize(proto)
    R = p.size
    if R == target_len:
        return p.copy()
    base = target_len // R
    rem = target_len % R
    lengths = np.full(R, base, dtype=np.int64)
    if rem > 0:
        lengths[:rem] += 1
    out = np.zeros(target_len, dtype=np.float64)
    idx = 0
    for i in range(R):
        L = int(lengths[i])
        if L > 0:
            out[idx:idx+L] = p[i] / L
            idx += L
    return _normalize(out)

def _fft_autocorr(a):
    """Compute autocorrelation using FFT."""
    n = a.size
    N = 1
    while N < 2*n - 1:
        N *= 2
    A = np.fft.rfft(a, N)
    S = A * np.conj(A)
    r = np.fft.irfft(S, N)
    return r[:2*n-1]
