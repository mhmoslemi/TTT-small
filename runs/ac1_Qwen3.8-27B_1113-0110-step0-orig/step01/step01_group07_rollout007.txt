"""
Hybrid LP Frank-Wolfe + Adam optimization for non-negative coefficient sequences.
Minimizes 2*n*max(conv(a,a))/sum(a)^2 via:
1. Structured polynomial initialization
2. LP-based Frank-Wolfe at moderate N
3. Log-domain Adam refinement
4. Multi-resolution finalization
"""

import time
import numpy as np

try:
    from scipy.optimize import linprog
    HAVE_LINPROG = True
except Exception:
    HAVE_LINPROG = False


def _np2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _norm_pos(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s


def _fscore(a):
    """Fast score: 2n * max(conv(p,p)) where p = a/sum(a)."""
    p = _norm_pos(a)
    if p is None:
        return np.inf
    n = p.size
    if n < 2:
        return np.inf
    L = _np2(2 * n - 1)
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0:
        return np.inf
    return float(2.0 * n * cm)


def _convolve_full(p, L=None):
    """Full linear convolution via FFT. Returns array of length 2n-1."""
    n = p.size
    if L is None:
        L = _np2(2 * n - 1)
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    return np.where(np.isfinite(c), c, 0.0)


def _resize(seq, m):
    p = _norm_pos(seq)
    m = int(m)
    if p is None:
        return np.ones(m) / m
    n = p.size
    if n == m:
        return p.copy()
    if n == 1:
        return np.full(m, 1.0 / m)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, m)
    out = np.interp(x_new, x_old, p)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 0:
        return np.ones(m) / m
    return out / s


def _cheb_basis(n, D):
    x = np.linspace(0.0, 1.0, n)
    t = 2.0 * x - 1.0
    t = np.clip(t, -1.0, 1.0)
    ang = np.arccos(t)
    out = np.empty((n, D), dtype=np.float64)
    for m in range(1, D + 1):
        out[:, m - 1] = np.cos(m * ang)
    return out


def _poly_shape(theta, B, mode=0, base=0.12):
    theta = np.asarray(theta, dtype=np.float64)
    val = B @ theta
    if mode == 0:
        h = 1.0 + base + val
    else:
        h = np.exp(np.clip(val, -8.0, 8.0))
    p = _norm_pos(h)
    return p

    """Build A_ub (2n-1) x (n+1) for the LP.
    Row k: sum_i f[k-i]*g[i] - M <= 0
    Last column is -M.

    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            A[k, i] = f[k - i]

"""
LP Frank-Wolfe + log-domain Adam for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses structured initialization, LP-based Frank-Wolfe at moderate N, 
and Adam refinement for final polish.
"""

import time
import numpy as np

try:
    from scipy.optimize import linprog
    HAVE_LP = True
except Exception:
    HAVE_LP = False


def _np2(x):
    x = max(1, int(x))
    return 1 << (x - 1).bit_length()


def _norm(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.array([1.0])
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-300:
        return np.ones_like(a) / a.size
    return a / s


def _score(a):
    p = _norm(a)
    n = p.size
    if n < 2:
        return np.inf
    L = _np2(2 * n - 1)
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    c = np.clip(c, 0, None)
    return float(2.0 * n * c.max())


def _conv(p):
    n = p.size
    L = _np2(2 * n - 1)
    F = np.fft.rfft(p, L)
    return np.fft.irfft(F * F, L)[:2 * n - 1]


def _resize(a, m):
    p = _norm(a)
    n = p.size
    m = int(m)
    if n == m:
        return p.copy()
    if n == 1:
        return np.full(m, 1.0 / m)
    xo = np.linspace(0, 1, n)
    xn = np.linspace(0, 1, m)
    out = np.interp(xn, xo, p)
    return _norm(out)


def _cheb(n, D=8):
    x = np.linspace(0, 1, n)
    t = np.clip(2 * x - 1, -1, 1)
    a = np.arccos(t)
    B = np.zeros((n, D))
    for k in range(1, D + 1):
        B[:, k - 1] = np.cos(k * a)
    return B


def _poly(theta, B, base=0.1, exp_mode=False):
    v = B @ np.asarray(theta, dtype=np.float64)
    if exp_mode:
        h = np.exp(np.clip(v, -10, 10))
    else:
        h = 1 + base + v
    return _norm(h)


def _lp_matrix(f, n):
    A = np.zeros((2 * n - 1, n + 1))
    A[:, n] = -1.0
    for i in range(n):
        hi = min(2 * n - 1, i + n)
        A[i:hi, i] = f[:hi - i]
    return A


def _fw_lp(f, deadline, n=2048, verbose=True):

        return _resize(f, n), _score(_resize(f, n))
    f = _resize(f, n).copy()
    best = f.copy()
    best_s = _score(f)
    n_items = n
    c_obj = np.zeros(n_items + 1)
    c_obj[n_items] = 1.0
    A_eq = np.zeros((1, n_items + 1))
    A_eq[0, :n_items] = 1.0
    b_eq = np.ones(1)
    bounds = [(0.0, None)] * n_items + [(None, None)]
    no_imp = 0
    it = 0
    while time.monotonic() < deadline:
        A_ub = _lp_matrix(f, n_items)
        b_ub = np.zeros(2 * n_items - 1)
        try:

                        bounds=bounds, method='highs',
                        options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,

        except Exception:
            break
        if r is None or not r.success or r.x is None:

        g = np.clip(np.asarray(r.x[:n_items]), 0, None)

        if sg < 1e-12:

        g /= sg
        improved = False
        # Line search

        s_best = best_s

            nf = np.clip(nf, 0, None)

            if s_best < best_s - 1e-12:

            improved = True

    z = np.clip(np.where(np.isfinite(z), z, 0), -50, 50)

    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    step = 0

    while time.monotonic() < dl:

        c = np.clip(c, 0, None)

        ez = np.exp(np
