                    pass
        
        # Standardize the sequence representation
        try:
            if len(best_seq) != len(initial_seq):
                best_seq = list(initial_seq)
        except:
            # Fallback to initial_seq if representation fails
            best_seq = initial_seq.copy()
            best_score_val = evaluate_sequence(best_seq)
    
        # Monitor improvement over time
        improvement_counter = 0
        for iter_num in range(min(50, int(self.deadline))):  # Max 50 iterations or time expiration
            try:
                self.learning_rate *= (1.0 + 0.1*(iter_num%5))  # Dynamic learning adjustment
                
                # Select an exploratory direction
                directions = ['linear', 'exponential_decay', 'oscillatory']  # We exclude 'gaussian_blur' initially
                
                if iter_num % 10 == 0:
                    self.current_direction_index = (self.current_direction_index + 1) % len(directions)
                    
                direction = directions[self.current_direction_index]
                
                # Bayesian acquisition function selection
                new_seq = self.local_search(best_seq, direction=direction)
                score_val = evaluate_sequence(new_seq)
                
                # Evaluate the improvement
                delta = score_val - best_score_val
                if delta < 0:
                    improvement_counter += 1
                    if improvement_counter >= 5:  # Consecutive 5 improvements signal to continue
                        pass
                    else:
                        best_score_val = score_val
                        best_seq = new_seq.copy()
                        improvement_counter = 0
                        continue
                        
                best_score_val = score_val
                best_seq = new_seq.copy()
                improvement_counter = 0
                
                # Print progress if time is low
                if time.time() - self.start_time > self.deadline - 5:
                    print(f"Iteration {iter_num+1}:score={score_val:.6f},Usingdirection={direction}")
            except Exception:
                pass  # Fallback without panic
            
            # Exit condition if time expires
            if time.time() - self.start_time > self.deadline - 10:
                break
                
        # Finally, accept the best candidate
        if best_score_val is not np.inf:
            best_sequence = best_seq
            best_score = best_score_val

        # Return normalized candidate
        try:
            if len(best_sequence) != self.length:
                best_sequence = list(best_sequence[:self.length].flatten()[:self.length])
        except:
            pass

        if np.sum(best_sequence) < 0.01:
            print("Normalization failed: insufficient sum")
            # Try different length alternative?
            # But we are optimizing for current length
            return initial_seq, np.inf
            
        return best_sequence, best_score

    def local_search(self, candidate: List[float], direction: Optional[str] = None) -> List[float]:
        """Perform gradient-based local search with half-gaussian updates."""
        # Multiple restarts and directional pushing
        if direction is None:
            # Random initialization
            if 'parent' in locals():
                new_candidate = copy.deepcopy(parent)
            else:
                # Create a standard hill-start
                new_candidate = []
                for idx in range(len(candidate)):
                    if idx == 0:
                        val = 1.0 + 0.5*np.random.randn()
                    else:
                        # Create a structure: delay the first element, then drop off towards exponentials
                        val = 0.5 * np.exp(-0.3*idx)
                    new_candidate.append(val)

        # Compute the analytical gradient using finite differences
        try:
            gradient = _score_derivative(candidate)
        except:
            gradient = None

        # Alternative numerical gradient if analytic gradient is infeasible
        if gradient is None:
            h = 0.001  # Step size for numerical gradient
            # Use centered difference
            grad_numerical = np.zeros_like(candidate)
            for i, x_val in enumerate(candidate):
                candidate_plus = candidate.copy()
                candidate_plus[i] = x_val + h
                candidate_minus = candidate.copy()
                candidate_minus[i] = x_val - h

                conv_plus = full_convolve(candidate_plus)
                conv_minus = full_convolve(candidate_minus)

                if conv_plus is None or conv_minus is None:
                    continue
                g_val_plus = evaluate_sequence(candidate_plus)
                g_val_minus = evaluate_sequence(candidate_minus)

                if g_val_plus == np.inf or g_val_minus == np.inf:
                    g_val_plus = np.inf
                    grad_numerical[i] = 0
                else:
                    grad_numerical[i] = (g_val_plus - g_val_minus) / (2*h)

            gradient = grad_numerical.tolist()

        # Gradient descent update
        new_seq = candidate.copy()
        new_seq = np.array(new_seq) - self.learning_rate * np.array(gradient)
        new_seq = np.maximum(new_seq, 0.0)
        new_seq = np.minimum(new_seq, 1000.0)

        # Normalize to maintain the length
        sum_val = np.sum(new_seq)
        if sum_val < 0.01:
            new_seq = new_seq / sum_val * 0.01  # Scale to meet the minimum sum requirement

        # Add multiplicative noise from Gaussian distribution
        # Optionally: multiply by a static factor
        new_seq = new_seq + 0.5 * np.random.randn(len(new_seq)) * np.sqrt(self.hessian_smoothing)

        return list(new_seq)

def bayes_opt_fallback(score_obj, dimensions, n_evals, bounds, deadline):
    """Define a Bayesian fallback optimization if the chief method is unavailable for any reason."""
    # Create a Bayesian optimization similar to the chief one
    # But using a different method
    est = TreeParzenEstimator()
    est.gridpoints = []
    est.y = []
    est.bm = np.inf
    for trial in range(n_evals):
        x = [a * (bound[1]-bound[0]) + bound[0] for a, bound in zip(np.random.randn(len(dimensions)), bounds]
        y = score_obj(x)
        est.y.append(y)
        est.x.append(x)
        est.gridpoints.append(x)
    # Then, select the minima and continue until deadline
    return est.min_point()
