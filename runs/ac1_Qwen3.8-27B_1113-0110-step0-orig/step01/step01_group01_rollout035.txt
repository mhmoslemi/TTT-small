"""

Uses multi-scale mirror descent with adaptive parameters, continuous mass-transfer
local search, peak-targeted moves, and binary refinement. Starts from the best

import numpy as np

    import scipy.optimize as _scipy_optimize

def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def normalize_positive(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, float(floor), None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def convolve_seq(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if N is None:
        N = next_pow2(2 * n - 1)
    else:
        need = next_pow2(2 * n - 1)
        if int(N) < int(need):
            N = need
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]


def convolve_and_fft(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if N is None:
        N = next_pow2(3 * n - 2)
    else:
        need = next_pow2(3 * n - 2)
        if int(N) < int(need):
            N = need
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return c, fa


def internal_score(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.inf
    a = a / s
    c = convolve_seq(a, N)
    return float(2.0 * n * float(np.max(c)))


def make_uniform(n):
    return np.ones(max(1, int(n)), dtype=np.float64) / float(max(1, int(n)))


def resample_seq(a, n):
    a = normalize_positive(a, floor=0.0)
    old_n = a.size
    n = max(2, int(n))
    if old_n == 1:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    if old_n == n:
        return a.astype(np.float64, copy=True)
    old_x = np.linspace(0.0, 1.0, old_n, dtype=np.float64)
    new_x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    b = np.interp(new_x, old_x, a)
    return normalize_positive(b, floor=1e-12)

def compute_weights(c, cmax, tau, floor=0.0):
    """Compute softmax weights emphasizing peaks above cmax."""
    c = np.asarray(c, dtype=np.float64).ravel()
    L = c.size
    if L == 0 or cmax <= 0.0 or not np.isfinite(cmax) or tau <= 0.0:
        return np.full(L, 1.0 / float(L), dtype=np.float64)
    z = (c - float(cmax)) / float(tau)
    z = np.clip(z, -60.0, 0.0)
    ez = np.exp(z)
    s = float(ez.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(L, 1.0 / float(L), dtype=np.float64)
    w = ez / s
    if floor > 0.0:
        w = (1.0 - floor) * w + (floor / float(L))
        w = w / w.sum()
    return w


def grad_from_weights(w, fa, N, n):
    """Compute gradient of weighted convolution sum."""
    w = np.asarray(w, dtype=np.float64).ravel()
    w_rev = w[::-1]
    fw = np.fft.rfft(w_rev, N)
    conv = np.fft.irfft(fa * fw, N)
    return 2.0 * conv[n - 1:2 * n - 1][::-1]


def mirror_descent(a, deadline, verbose=False):
    """Mirror descent with adaptive parameters and early stopping."""
    a = normalize_positive(a, floor=1e-15)
    n = a.size
    N = next_pow2(3 * n - 2)

    best_a = a.copy()
    best_score = internal_score(a, N)

    eta = 0.5
    beta = 0.12
    w_floor = 0.03
    stagnant = 0
    it = 0
    last_improve = best_score

    while time.monotonic() < deadline:
        c, fa = convolve_and_fft(a, N)
        cmax = float(np.max(c))
        score = float(2.0 * n * cmax)
        if np.isfinite(score) and score < best_score * (1.0 - 1e-12):
            best_score = score
            best_a = a.copy()

        if cmax <= 0.0 or not np.isfinite(cmax):
            break

        tau = beta * cmax
        w = compute_weights(c, cmax, tau, floor=w_floor)
        g = grad_from_weights(w, fa, N, n)
        g = g - g.mean()
        scale = float(np.abs(g).max())
        if not np.isfinite(scale) or scale <= 1e-15:
            break
        g = g / scale

        loga = np.log(a + 1e-20)
        accepted = False

        for e in (eta, 0.5 * eta, 0.25 * eta, 0.1 * eta, 0.05 * eta):
            if e <= 1e-9:
                continue
            loga_try = loga - e * g
            loga_try = loga_try - float(np.max(loga_try))
            a_try = np.exp(np.clip(loga_try, -60.0, 0.0))
            a_try = np.maximum(a_try, 1e-300)
            a_try = a_try / float(a_try.sum())
            a_try = normalize_positive(a_try, floor=1e-15)

            c_try, _ = convolve_and_fft(a_try, N)
            cmax_try = float(np.max(c_try))
            score_try = float(2.0 * n * cmax_try)

            if np.isfinite(score_try) and score_try < score * (1.0 - 1e-10):
                accepted = True
                a = a_try
                score = score_try
                break

        if accepted:
            stagnant = 0
            eta = min(1.5, eta * 1.03)
        else:
            stagnant += 1
            eta = max(1e-4, eta * 0.6)
            if stagnant % 100 == 0:
                beta = min(0.3, beta * 1.5)
                w_floor = min(0.06, w_floor + 0.005)
            if eta < 1e-5 or stagnant > 500:
                # Try a random perturbation to escape
                rng_esc = np.random.default_rng(it + 7)
                pert = 1.0 + 0.1 * rng_esc.normal(size=n)
                pert = np.clip(pert, 0.3, 3.0)
                a_try = normalize_positive(a * pert, floor=1e-15)
                sc_try = internal_score(a_try, N)
                if np.isfinite(sc_try) and sc_try < best_score:
                    a = a_try
                    best_a = a_try.copy()
                    best_score = sc_try
                stagnant = 0
                eta = 0.5
                beta = 0.12
                if it > 5000:
                    break

        it += 1
        if verbose and it % 500 == 0:
            print("  mirror n={} iter={} best={:.8f}".format(n, it, best_score), flush=True)

    return best_a, best_score


def mass_transfer_search(a, deadline, seed=0, verbose=False):
    """
    Continuous mass-transfer local search.
    Pick two positions and transfer mass between them to reduce the max convolution.
    """
    a = normalize_positive(a, floor=1e-15)
    n = a.size
    N = next_pow2(3 * n - 2)

    best_a = a.copy()
    best_score = internal_score(a, N)

    rng = np.random.default_rng(int(seed))
    c, fa = convolve_and_fft(a, N)
    cmax = float(np.max(c))
    score = float(2.0 * n * cmax)

    stale = 0
    it = 0
    transfer_frac = 0.05

    while time.monotonic() < deadline:
        if it % 50 == 0:
            c, fa = convolve_and_fft(a, N)
            cmax = float(np.max(c))
            score = float(2.0 * n * cmax)
            if np.isfinite(score) and score < best_score * (1.0 - 1e-12):
                best_score = score
                best_a = a.copy()

        # Identify "hot" positions (those contributing most to peaks)
        tau = 0.1 * cmax
        w = compute_weights(c, cmax, tau, floor=0.02)
        g = grad_from_weights(w, fa, N, n)

        # Pick source (high g = high pressure) and target (low g)
        g_sorted = np.argsort(g)
        # Source: from top 30% by pressure
        n_top = max(1, n // 3)
        top_indices = g_sorted[-n_top:]
        # Target: from bottom 30% by pressure
        bot_indices = g_sorted[:n_top]

        i = int(rng.choice(top_indices))
        j = int(rng.choice(bot_indices))
        if i == j or a[i] <= 1e-10:
            continue

        # Transfer a fraction of a[i] to a[j]
        amt = transfer_frac * a[i]
        a[i] -= amt
        a[j] += amt
        a = normalize_positive(a, floor=1e-15)

        # Compute new score (approximate using incremental update)
        c_new, _ = convolve_and_fft(a, N)
        cmax_new = float(np.max(c_new))
        score_new = float(2.0 * n * cmax_new)

        if np.isfinite(score_new) and score_new < score * (1.0 - 1e-9):
            c = c_new
            cmax = cmax_new
            score = score_new
            stale = 0
            transfer_frac = min(0.15, transfer_frac * 1.05)
        else:
            # Revert
            a[i] += amt
            a[j] -= amt
            a = normalize_positive(a, floor=1e-15)
            stale += 1
            transfer_frac = max(0.01, transfer_frac * 0.92)

            if stale > 200:
                # Restart with perturbation
                pert = 1.0 + 0.05 * rng.normal(size=n)
                pert = np.clip(pert, 0.5, 2.0)
                a = normalize_positive(a * pert, floor=1e-15)
                c, fa = convolve_and_fft(a, N)
                cmax = float(np.max(c))
                score = float(2.0 * n * cmax)
                stale = 0
                transfer_frac = 0.05

        it += 1
        if verbose and it % 1000 == 0:
            print("  mass_transfer n={} iter={} best={:.8f}".format(n, it, best_score), flush=True)

    return best_a, best_score


def binary_refine(a0, deadline, seed=0):
    """Binary (0/1) refinement search."""
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = max(16, int(a0.size))
    m = max(2, min(n - 2, int(round(0.5 * n))))

    # Initialize binary from a0
    order = np.argsort(a0, kind="stable")
    b = np.zeros(n, dtype=np.float64)
    b[order[-m:]] = 1.0

    N = next_pow2(3 * n - 2)
    c = np.rint(convolve_seq(b, next_pow2(2 * n - 1))).astype(np.float64)

    best_b = b.copy()
    best_max = float(np.max(c))
    best_score = float(2.0 * n * best_max / (float(m) * float(m)))

    rng = np.random.default_rng(int(seed))
    beta = 0.5
    temp = 0.04
    stale = 0
    two_b = 2.0 * b
    it = 0

    while time.monotonic() < deadline:
        cmax = float(np.max(c))
        if cmax <= 0.0 or not np.isfinite(cmax):
            break
        score = float(2.0 * n * cmax / (float(m) * float(m)))
        if score < best_score * (1.0 - 1e-12):
            best_score = score
            best_max = cmax
            best_b = b.copy()

        w = compute_weights(c, cmax, beta * cmax, floor=0.05)
        fa = np.fft.rfft(b, N)
        g = grad_from_weights(w, fa, N, n)

        one_idx = np.nonzero(b > 0.5)[0]
        zero_idx = np.nonzero(b < 0.5)[0]
        if one_idx.size == 0 or zero_idx.size == 0:
            break

        # Weighted selection
        g_one = g[one_idx]
        g_zero = g[zero_idx]

        # Pick high-pressure 1 to turn off
        rng1 = rng.random()
        thresh1 = np.percentile(g_one, min(95, 50 + 50 * (1 - temp / 0.2)))
        candidates1 = one_idx[g_one >= thresh1]
        if candidates1.size == 0:
            candidates1 = one_idx
        i = int(rng.choice(candidates1))

        # Pick low-pressure 0 to turn on
        thresh0 = np.percentile(g_zero, max(5, 50 - 50 * (1 - temp / 0.2)))
        candidates0 = zero_idx[g_zero <= thresh0]
        if candidates0.size == 0:
            candidates0 = zero_idx
        j = int(rng.choice(candidates0))

        if i == j:
            continue

        # Swap
        c[i:i + n] -= two_b
        c[2 * i] -= 1.0
        c[j:j + n] += two_b
        c[2 * j] += 1.0
        b[i] = 0.0
        b[j] = 1.0
        two_b = 2.0 * b

        new_max = float(np.max(c))
        new_score = float(2.0 * n * new_max / (float(m) * float(m)))

        accept = False
        if new_score < score * (1.0 - 1e-12):
            accept = True
        else:
            delta = new_score - score
            if delta <= 1e-12:
                if rng.random() < 0.4:
                    accept = True
            elif delta < 0.5:
                if rng.random() < np.exp(-delta / max(temp, 1e-12)):
                    accept = True

        if accept:
            if new_score < best_score:
                best_score = new_score
                best_max = new_max
                best_b = b.copy()
            stale = 0
            temp = max(0.005, temp * 0.997)
        else:
            # Revert
            c[j:j + n] -= two_b
            c[2 * j] -= 1.0
            c[i:i + n] += two_b
            c[2 * i] += 1.0
            b[j] = 0.0
            b[i] = 1.0
            two_b = 2.0 * b
            stale += 1
            temp = max(0.003, temp * 0.95)
            if stale % 200 == 0:
                beta = min(1.0, beta * 1.4)
                temp = min(0.1, temp * 1.7)
                stale = 0

        it += 1

    return normalize_positive(best_b, floor=0.0), best_score


def multi_start_optimize(a_init, n, deadline, seed, n_starts=3):
    """Run multiple optimization passes from perturbed starts."""
    best_a = a_init.copy()
    best_score = internal_score(a_init)
    N = next_pow2(3 * n - 2)

    rng = np.random.default_rng(int(seed))

    for s in range(n_starts):
        if time.monotonic() > deadline:
            break

        # Create perturbed start
        if s == 0:
            a_cur = a_init.copy()
        else:
            pert_scale = 0.05 + 0.1 * s
            pert = 1.0 + pert_scale * rng.normal(size=n)
            pert = np.clip(pert, 0.2, 5.0)
            a_cur = normalize_positive(a_init * pert, floor=1e-15)

        sc_cur = internal_score(a_cur, N)
        if not np.isfinite(sc_cur):
            continue

        # Phase 1: Mirror descent
        t1_end = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.4)
        a1, sc1 = mirror_descent(a_cur, t1_end, verbose=False)
        if np.isfinite(sc1) and sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()

        # Phase 2: Mass transfer
        t2_end = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.35)
        a2, sc2 = mass_transfer_search(a1, t2_end, seed=seed + 100 * s + 1)
        if np.isfinite(sc2) and sc2 < best_score:
            best_score = sc2
            best_a = a2.copy()

        # Phase 3: Another mirror pass
        t3_end = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.5)
        a3, sc3 = mirror_descent(a2, t3_end, verbose=False)
        if np.isfinite(sc3) and sc3 < best_score:
            best_score = sc3
            best_a = a3.copy()

    return best_a, best_score


def generate_ramp_sequence(n, direction=1):
    """Generate a ramp sequence (increasing or decreasing)."""
    n = max(4, int(n))
    t = np.arange(n, dtype=np.float64)
    if direction == 1:
        vals = t + 1.0
    else:
        vals = (n - t)
    return normalize_positive(vals, floor=1e-12)


def generate_parabola_sequence(n, center=0.5, width=0.3):
    """Generate a parabolic/bump sequence."""
    n = max(4, int(n))
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    vals = np.exp(-((x - center) ** 2) / (2.0 * width ** 2))
    return normalize_positive(vals, floor=1e-12)


def generate_sine_sequence(n, k=1, phase=0.0):
    """Generate a sine-modulated sequence."""
    n = max(4, int(n))
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    vals = 1.0 + 0.5 * np.sin(2.0 * np.pi * k * x + phase)
    return normalize_positive(vals, floor=1e-12)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    start = time.monotonic()
    total = max(120.0, float(budget_s) - 30.0)
    deadline = start + total

    best_a = make_uniform(1024)
    best_score = internal_score(best_a)
    print("Initial uniform score = {:.9f}".format(best_score), flush=True)

    # Load given sequence
    given = get_given_sequence()
    if given is not None:
        sc = internal_score(given)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = given.copy()
            print("Given sequence score = {:.9f}, n={}".format(best_score, best_a.size), flush=True)

    # Try various initial shapes at different sizes
    init_seqs = [
        ("ramp_up_512", generate_ramp_sequence(512, 1)),
        ("ramp_dn_512", generate_ramp_sequence(512, -1)),
        ("ramp_up_1024", generate_ramp_sequence(1024, 1)),
        ("ramp_dn_1024", generate_ramp_sequence(1024, -1)),
        ("para_1024", generate_parabola_sequence(1024, 0.4, 0.25)),
        ("sine_1024", generate_sine_sequence(1024, 1, 0.0)),
        ("sine_1024_p", generate_sine_sequence(1024, 2, np.pi / 3)),
        ("ramp_up_2048", generate_ramp_sequence(2048, 1)),
        ("ramp_dn_2048", generate_ramp_sequence(2048, -1)),
    ]
    for name, seq in init_seqs:
        sc = internal_score(seq)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = seq.copy()
            print("Init {} score = {:.9f}".format(name, best_score), flush=True)

    # Multi-scale optimization
    # Allocate time: 60% to main optimization, 20% to binary, 20% to final polish
    t_main_end = start + total * 0.55
    t_bin_end = start + total * 0.80
    t_final_end = deadline

    # Try n = 1024, 2048, 4096
    sizes = [1024, 2048, 4096]
    for idx, n in enumerate(sizes):
        if time.monotonic() > t_main_end:
            break

        # Determine time allocation for this size
        remaining_sizes = len(sizes) - idx
        time_per_size = min(
            (t_main_end - time.monotonic()) / remaining_sizes,
            total * 0.2
        )
        if time_per_size < 20.0:
            break

        # Start from best (resampled)
        a_start = resample_seq(best_a, n)
        # Also try ramp as alternative start
        a_ramp = generate_ramp_sequence(n, direction=1 if idx % 2 == 0 else -1)

        # Optimize from best-based start
        sc_start = internal_score(a_start)
        sc_ramp = internal_score(a_ramp)

        if sc_start <= sc_ramp:
            a_opt, sc_opt = multi_start_optimize(a_start, n, time.monotonic() + time_per_size * 0.7, seed=seed + 1000 * n)
        else:
            a_opt, sc_opt = multi_start_optimize(a_ramp, n, time.monotonic() + time_per_size * 0.7, seed=seed + 1000 * n)

        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print("Size n={} optimized score = {:.9f}".format(n, best_score), flush=True)

        # If we have time, try the other start too
        if time.monotonic() < t_main_end - 15.0:
            alt = a_ramp if sc_start <= sc_ramp else a_start
            sc_alt_start = sc_ramp if sc_start <= sc_ramp else sc_start
            a_opt2, sc_opt2 = multi_start_optimize(alt, n, t_main_end - 5.0, seed=seed + 2000 * n, n_starts=2)
            if np.isfinite(sc_opt2) and sc_opt2 < best_score:
                best_score = sc_opt2
                best_a = a_opt2.copy()
                print("Size n={} alt start score = {:.9f}".format(n, best_score), flush=True)

    # Binary refinement
    if time.monotonic() < t_bin_end - 30.0:
        # Try binary on the best at its current size
        n_bin = best_a.size
        if n_bin >= 512:
            a_bin, sc_bin = binary_refine(best_a, t_bin_end - 10.0, seed=seed + 50000)
            if np.isfinite(sc_bin) and sc_bin < best_score:
                best_score = sc_bin
                best_a = a_bin.copy()
                print("Binary n={} score = {:.9f}".format(n_bin, best_score), flush=True)

        # Also try binary at n=2048 if best is different size
        if n_bin != 2048:
            a_2048 = resample_seq(best_a, 2048)
            a_bin2, sc_bin2 = binary_refine(a_2048, t_bin_end - 10.0, seed=seed + 60000)
            if np.isfinite(sc_bin2) and sc_bin2 < best_score:
                best_score = sc_bin2
                best_a = a_bin2.copy()
                print("Binary n=2048 score = {:.9f}".format(best_score), flush=True)

    # Final mirror descent polish
    if time.monotonic() < deadline - 15.0:
        a_final, sc_final = mirror_descent(best_a, deadline - 5.0, verbose=True)
        if np.isfinite(sc_final) and sc_final < best_score:
            best_score = sc_final
            best_a = a_final.copy()
            print("Final polish score = {:.9f}".format(best_score), flush=True)

    # Ensure valid output
    best_a = normalize_positive(best_a, floor=0.0)
    out_list = [float(x) for x in best_a]

    try:
        final_eval = float(evaluate_sequence(out_list))
    except Exception:
        final_eval = float("inf")

    if not np.isfinite(final_eval) or final_eval > 2.5:
        best_a = make_uniform(1024)
        out_list = [float(x) for x in best_a]
        final_eval = 2.0

    elapsed = time.monotonic() - start
    print("Final score = {:.12f}, length = {}, elapsed = {:.1f}s".format(
        final_eval, len(out_list), elapsed), flush=True)
    return out_list
