```python
from scipy.sparse import csr_matrix
from scipy.optimize import linprog

def solve_lp(a, M):
    n = len(a)
    # Build constraint matrix A where A[k,j] = a[k-j] if 0 <= k-j < n
    # A is (2n-1) x n
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            idx = k - j
            if 0 <= idx < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[idx])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Maximize 1^T g subject to A @ g <= M, g >= 0
    c_obj = -np.ones(n)  # minimize -sum(g)
    res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return None
```

```python
Optimize nonnegative step-function heights minimizing 2*n*max(conv(a,a))/sum(a)^2
using LP-based iterative projection, mirror descent, and binary search.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
try:
    import scipy.optimize as _scipy_optimize
except Exception:
    _scipy_optimize = None

def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()

def normalize_positive(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1)
    a = np.clip(a, float(floor), None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size) / float(a.size)
    return a / s

def convolve_seq(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if N is None:
        N = next_pow2(2 * n - 1)
    else:
        need = next_pow2(2 * n - 1)
        if int(N) < int(need):
            N = need
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[: 2 * n - 1]

def internal_score(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.inf
    a = a / s
    c = convolve_seq(a, N)
    return float(2.0 * n * float(np.max(c)))

def make_uniform(n):
    return np.ones(max(1, int(n)), dtype=np.float64) / float(max(1, int(n)))
```

"""
Optimize nonnegative step-function heights minimizing 2*n*max(conv(a,a))/sum(a)^2
using LP-based cross-correlation minimization, mirror descent, and binary search.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import scipy.sparse as sp

def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()

def normalize_positive(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1)
    a = np.clip(a, float(floor), None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size) / float(a.size)
    return a / s

def softmax_seq(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = z - float(np.max(z))
    z = np.clip(z, -60.0, 0.0)
    e = np.exp(z)
    e = np.maximum(e, 1e-300)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(z.size, 1.0 / float(z.size))
    return e / s

def convolve_seq(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if N is None:
        N = next_pow2(2 * n - 1)
    else:
        need = next_pow2(2 * n - 1)
        if int(N) < int(need):
            N = need
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[: 2 * n - 1]

def internal_score(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.inf
    a = a / s
    c = convolve_seq(a, N)
    return float(2.0 * n * float(np.max(c)))

def make_uniform(n):
    return np.ones(max(1, int(n)), dtype=np.float64) / float(max(1, int(n)))

def resample_seq(a, n):

    old_n = a.size
    n = max(2, int(n))
    if old_n == 1:
        return np.full(n, 1.0 / float(n))
    if old_n == n:

    old_x = np.linspace(0.0, 1.0, old_n)

    b = np.interp(new_x, old_x, a)

def build_conv_matrix(a):

def lp_optimize(a, deadline, max_iters=500):
    """
    LP-based iterative method:
    At each step, solve min M s.t. (a*g)[k] <= M, sum(g) = sum(a), g >= 0.
    Then update a <- (1-t)*a + t*g.
    """
    a = normalize_positive(a, floor=1e-15)
    n = a.size
    S = float(a.sum())
    N = next_pow2(3 * n - 2)
    
    best_a = a.copy()
    best_score = internal_score(a, N)
    
    A_conv = build_conv_matrix(a)
    
    t = 0.1  # mixing parameter
    stagnant = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Build LP problem
        # Variables: [g_0, ..., g_{n-1}, M]
        # Minimize: M (last variable)
        # Subject to:
        #   A_conv @ g - M <= 0  (2n-1 constraints)
        #   sum(g) = S
        #   g >= 0, M >= 0
        
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0  # minimize M
        
        # Inequality constraints: A_conv @ g - M <= 0
        # Each row: [A_conv[k,0], ..., A_conv[k,n-1], -1]
        # Build sparse A_ub
        # A_ub has (2n-1) rows and (n+1) columns
        # The first n columns are A_conv
        # The last column is -1 for all rows
        
        minus_ones = sp.csr_matrix(-np.ones((2 * n - 1, 1)))
        A_conv_sparse = sp.csr_matrix(A_conv)
        A_ub = sp.hstack([A_conv_sparse, minus_ones], format='csr')
        b_ub = np.zeros(2 * n - 1)
        
        # Equality constraint: sum(g) = S
        A_eq = sp.csr_matrix(np.array([[np.ones(n), 0.0]]))
        b_eq = np.array([S])
        
        bounds = [(0, None)] * n + [(0, None)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={"disp": False, "time_limit": max(5.0, deadline - time.monotonic() - 2)})
        except Exception as e:
            print("LP error:", str(e)[:100], flush=True)
            break
        
        if not res.success or res.x is None:
            break
        
        g = res.x[:n]
        M_new = res.x[n]
        
        # Ensure g is non-negative and has the right sum
        g = np.maximum(g, 0)
        s_g = float(g.sum())
        if s_g <= 1e-15:
            break
        g = g / s_g * S  # rescale
        
        # Update
        a_new = (1 - t) * a + t * g
        a_new = normalize_positive(a_new, floor=1e-15)
        
        score_new = internal_score(a_new, N)
        
        if np.isfinite(score_new) and score_new < best_score * (1.0 - 1e-9):
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            stagnant = 0
            # Rebuild matrix for new a
            A_conv = build_conv_matrix(a)
            if (it + 1) % 50 == 0:
                print("LP iter={} n={} score={:.8f} t={:.4f}".format(it + 1, n, best_score, t), flush=True)
        else:
            stagnant += 1
            if stagnant % 10 == 0:
                t = max(0.01, t * 0.7)
            if stagnant > 50:
                # Try a different approach: restart with current best
                a = normalize_positive(best_a.astype(np.float64), floor=1e-15)
                A_conv = build_conv_matrix(a)
                t = 0.05
                stagnant = 0
        
        # Adapt t
        if stagnant == 0 and t < 0.3:
            t = min(0.3, t * 1.05)
    
    return best_a, best_score

def compute_weights(c, cmax, tau, floor=0.02):
    c = np.asarray(c, dtype=np.float64).ravel()
    L = c.size
    if L == 0 or cmax <= 0.0 or not np.isfinite(cmax) or tau <= 0.0:
        return np.full(L, 1.0 / float(L))
    z = (c - float(cmax)) / float(tau)
    z = np.clip(z, -50.0, 0.0)
    ez = np.exp(z)
    s = float(ez.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(L, 1.0 / float(L))
    w = ez / s
    if floor > 0.0:
        w = (1.0 - floor) * w + (floor / float(L))
    return w

def grad_from_weights(w, fa, N, n):
    w = np.asarray(w, dtype=np.float64).ravel()
    w_rev = w[::-1]
    fw = np.fft.rfft(w_rev, N)
    conv = np.fft.irfft(fa * fw, N)
    return 2.0 * conv[n - 1:2 * n - 1][::-1]

def smooth_value(c, cmax, tau):
    c = np.asarray(c, dtype=np.float64).ravel()
    if c.size == 0 or cmax <= 0.0 or not np.isfinite(cmax):
        return 0.0
    if tau <= 0.0:
        return cmax
    z = (c - cmax) / tau
    z = np.clip(z, -50.0, 0.0)
    s = float(np.exp(z).sum())
    if not np.isfinite(s) or s <= 0.0:
        return cmax
    return float(cmax + tau * np.log(s))

def optimize_mirror(a, deadline, verbose=False):
    a = normalize_positive(a, floor=1e-15)
    n = a.size
    N = next_pow2(3 * n - 2)
    
    best_a = a.copy()
    best_score = internal_score(a, N)
    
    eta = 0.4
    beta = 0.15
    w_floor = 0.04
    stagnant = 0
    it = 0
    
    while time.monotonic() < deadline:
        c = convolve_seq(a, N)
        cmax = float(np.max(c))
        score = float(2.0 * n * cmax)
        if np.isfinite(score) and score < best_score * (1 - 1e-12):
            best_score = score
            best_a = a.copy()
        
        if cmax <= 0.0 or not np.isfinite(cmax):
            break
        
        tau = beta * cmax
        w = compute_weights(c, cmax, tau, floor=w_floor)
        fa = np.fft.rfft(a, N)
        g = grad_from_weights(w, fa, N, n)
        g = g - g.mean()
        scale = float(np.abs(g).max())
        if not np.isfinite(scale) or scale <= 1e-15:
            break
        g = g / scale
        
        loga = np.log(a + 1e-20)
        J_cur = smooth_value(c, cmax, tau)
        accepted = False
        
        for e in (eta, 0.5 * eta, 0.25 * eta, 0.125 * eta):
            if e <= 1e-9:
                continue
            loga_try = loga - e * g
            loga_try = loga_try - float(np.max(loga_try))
            a_try = np.exp(np.clip(loga_try, -60.0, 0.0))
            a_try = np.maximum(a_try, 1e-300)
            a_try = a_try / float(a_try.sum())
            a_try = normalize_positive(a_try, floor=1e-15)
            
            c_try = convolve_seq(a_try, N)
            cmax_try = float(np.max(c_try))
            score_try = float(2.0 * n * cmax_try)
            J_try = smooth_value(c_try, cmax_try, tau)
            
            if np.isfinite(score_try):
                if score_try < score * (1.0 - 1e-10):
                    accepted = True
                    a = a_try
                    score = score_try
                    break
                if np.isfinite(J_try) and J_try < J_cur * (1.0 - 1e-7) and score_try < score * 1.05:
                    accepted = True
                    a = a_try
                    score = score_try
                    break
        
        if accepted:
            stagnant = 0
            eta = min(1.0, eta * 1.02)
            beta = max(0.005, beta * 0.999)
        else:
            stagnant += 1
            eta = max(1e-4, eta * 0.65)
            beta = max(0.003, beta * 0.85)
            if stagnant % 60 == 0:
                beta = min(0.3, beta * 1.8)
                stagnant = 0
            if eta < 1e-5 or stagnant > 500:
                break
        
        it += 1
        if verbose and it % 200 == 0:
            print("mirror n={} iter={} best={:.8f}".format(n, it, best_score), flush=True)
    
    return best_a, best_score

def binary_from_values(a0, m):
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    m = int(max(0, min(n, m)))
    b = np.zeros(n, dtype=np.float64)
    if m <= 0:
        return b
    if m >= n:
        b.fill(1.0)
        return b
    order = np.argsort(a0, kind='stable')
    b[order[-m:]] = 1.0
    return b

def choose_by_pressure(g, rng, pick_high, temp=0.12):
    g = np.asarray(g, dtype=np.float64).ravel()
    m = g.size
    if m == 0:
        return -1
    if m == 1:
        return 0
    if not np.all(np.isfinite(g)):
        g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
    vmin = float(np.min(g))
    vmax = float(np.max(g))
    range_g = vmax - vmin
    if range_g <= 1e-14:
        return int(rng.integers(0, m))
    denom = max(float(temp), 1e-12) * range_g
    if pick_high:
        z = (g - vmax) / denom
    else:
        z = (vmin - g) / denom
    z = np.clip(z, -50.0, 0.0)
    p = np.exp(z)
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        return int(rng.integers(0, m))
    p = p / s
    return int(rng.choice(m, p=p))
