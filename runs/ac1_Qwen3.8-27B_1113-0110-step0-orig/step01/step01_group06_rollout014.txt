"""
Optimize a non-negative step function sequence to minimize
    2 * n * max(convolve(a, a)) / (sum(a))**2.

Key improvements over previous approach:
1. Adam optimizer (momentum + adaptive per-parameter learning rates) for faster convergence
2. Anti-palindromic parameterization at n=16384 (double the previous best size)
3. Simulated annealing acceptance criterion for better exploration
4. Multi-scale peak focusing with temperature annealing
5. Greedy binary support construction for initialization
6. Cross-pollination between different-sized candidates
"""

import numpy as np
import time


def _pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _fft_square(a):
    n = int(a.size)
    L = 2 * n - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, n=N)
    b = np.fft.irfft(fa * fa, n=N).real[:L]
    b = np.maximum(b, 0.0)
    return b


def _fft_grad(weights, a):
    """Compute gradient of dot(weights, convolve(a,a)) with respect to a."""
    n = int(a.size)
    L = int(weights.size)
    N = _pow2(L + n - 1)
    fw = np.fft.rfft(weights, n=N)
    fa = np.fft.rfft(a, n=N)
    c = np.fft.irfft(fw * np.conj(fa), n=N)[:n]
    return 2.0 * c


def _internal_score(a, b=None):
    if b is None:
        b = _fft_square(a)
    s = float(a.sum())
    if s <= 1e-300:
        return float("inf"), b, 0.0
    maxb = float(b.max())
    score = float(2.0 * a.size * maxb / (s * s))
    return score, b, maxb


def _make_weights(b, focus, eps=0.01):
    L = int(b.size)
    maxb = float(b.max())
    if maxb <= 1e-300:
        return np.full(L, 1.0 / L, dtype=np.float64)
    z = (maxb - b) / maxb
    p = np.exp(-z * float(focus))
    s = float(p.sum())
    if s <= 1e-12:
        p = np.zeros(L, dtype=np.float64)
        p[int(np.argmax(b))] = 1.0
    else:
        p = (1.0 - eps) * (p / s) + eps / L
    return p


def _a_from_x(x, n=None):
    x = np.asarray(x, dtype=np.float64).ravel()
    m = x.size
    if n is None:
        n = 2 * m
    t = 1.0 / m
    a = np.empty(n, dtype=np.float64)
    a[:m] = t * x
    a[m:] = t * (1.0 - x[::-1])
    return a


def _greedy_support(n, rng, shuffle=True):
    n = int(n)
    m = n // 2
    counts = np.zeros(2 * n - 1, dtype=np.int64)
    selected = np.empty(m, dtype=np.int64)
    order = np.arange(m, dtype=np.int64)
    if shuffle:
        rng.shuffle(order)
    t = 0
    for idx_val in order:
        idx = int(idx_val)
        l = idx
        r = n - 1 - idx
        if t == 0:
            y = l
        else:
            s = selected[:t]
            affected_l = int(np.max(counts[s + l])) if t > 0 else 0
            self_l = int(counts[2 * l])
            metric_l = max(affected_l + 2, self_l + 1)
            affected_r = int(np.max(counts[s + r])) if t > 0 else 0
            self_r = int(counts[2 * r])
            metric_r = max(affected_r + 2, self_r + 1)
            y = l if metric_l <= metric_r else r
        if t > 0:
            counts[selected[:t] + y] += 2
        counts[2 * y] += 1
        selected[t] = y
        t += 1
    return selected


def _greedy_x(n, rng, shuffle=True):
    sel = _greedy_support(n, rng, shuffle=shuffle)
    m = n // 2
    x = np.full(m, 0.05, dtype=np.float64)
    left = sel < m
    if np.any(left):
        x[sel[left]] = 0.95
    return x


def _anti_log_init(n, rng, amp=3.0, smooth=False):
    n = int(n)
    m = n // 2
    if m < 1:
        return np.ones(n, dtype=np.float64) / max(1, n)
    x = np.arange(m, dtype=np.float64)
    if smooth:
        v = np.zeros(m, dtype=np.float64)
        for f in (1, 3, 5, 7, 11, 13, 17):
            v += (rng.normal(0.0, 1.0) / max(1, f)) * np.sin(
                2.0 * np.pi * f * x / m + rng.uniform(0.0, 2.0 * np.pi)
            )
        v += 0.1 * rng.normal(0.0, 1.0, m)
    else:
        v = rng.normal(0.0, 1.0, m)
    v = v - float(v.mean())
    std = float(v.std())
    if std > 1e-12:
        v = (float(amp) * v) / std
    z = np.zeros(n, dtype=np.float64)
    z[:m] = v
    if n % 2 == 0:
        z[m:] = -v[::-1]
    else:
        z[m + 1:] = -v[::-1]
        z[m] = 0.0
    z = z - z.max()
    a = np.exp(z)
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    return a


def _random_positive(n, rng, amp=3.0):
    z = rng.normal(0.0, amp, size=n)
    z -= z.max()
    a = np.exp(z)
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    return a


def _adam_optimize_pair(x0, time_limit, rng, verbose=False,
                        focus_start=5.0, max_focus=5000.0,
                        lr=0.1, label=""):
    """Adam optimizer for anti-palindromic parameterization."""
    x = np.asarray(x0, dtype=np.float64).copy()
    m = int(x.size)
    n = 2 * m
    x_eps = 1e-10
    x = np.clip(x, x_eps, 1.0 - x_eps)

    a = _a_from_x(x, n)
    b = None
    score = float("inf")

    best_score = float("inf")
    best_x = x.copy()

    # Adam state
    m_adam = np.zeros(m, dtype=np.float64)
    v_adam = np.zeros(m, dtype=np.float64)
    t_adam = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    focus = float(focus_start)
    focus_inc = 1.015
    no_improve = 0
    max_no = 800
    resets = 0
    max_resets = 40
    temp = 0.02  # annealing temperature for acceptance
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < float(time_limit):
        if b is None:
            b = _fft_square(a)
            score, _, _ = _internal_score(a, b)

        if score < best_score - 1e-12:
            best_score = score
            best_x = x.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and (it % 2000 == 0):
            print(f"[{label}] it={it} t={time.monotonic()-t0:.1f}s "
                  f"score={score:.6f} best={best_score:.6f} "
                  f"focus={focus:.1f} lr={lr:.4f} resets={resets}")

        if no_improve > max_no:
            resets += 1
            if resets > max_resets:
                break
            no_improve = 0
            max_no = int(max_no * 1.4)
            # Restart: interpolate between best and random
            r = rng.random()
            if r < 0.4:
                x = np.clip(best_x + rng.normal(0.0, 0.15, size=m), x_eps, 1.0 - x_eps)
            elif r < 0.7:
                fresh = _greedy_x(n, rng, shuffle=True)
                x = np.clip(0.6 * best_x + 0.4 * fresh, x_eps, 1.0 - x_eps)
            else:
                x = np.clip(rng.uniform(0.1, 0.9, size=m), x_eps, 1.0 - x_eps)
            a = _a_from_x(x, n)
            b = None
            score = float("inf")
            focus = float(rng.choice([5.0, 20.0, 50.0, 100.0, 500.0]))
            lr = 0.1
            m_adam = np.zeros(m, dtype=np.float64)
            v_adam = np.zeros(m, dtype=np.float64)
            t_adam = 0
            continue

        p = _make_weights(b, focus)
        ga = _fft_grad(p, a)
        gx = ga[:m] - ga[m:][::-1]

        # Adam update
        t_adam += 1
        m_adam = beta1 * m_adam + (1 - beta1) * gx
        v_adam = beta2 * v_adam + (1 - beta2) * gx * gx
        m_hat = m_adam / (1 - beta1 ** t_adam)
        v_hat = v_adam / (1 - beta2 ** t_adam)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        x_new = x - step
        x_new = np.clip(x_new, x_eps, 1.0 - x_eps)

        a_new = _a_from_x(x_new, n)
        b_new = _fft_square(a_new)
        score_new, _, _ = _internal_score(a_new, b_new)

        # Acceptance: accept if better, or with probability based on improvement
        accept = False
        if score_new < best_score - 1e-12:
            accept = True
        elif score_new < score - 1e-12:
            accept = True
        else:
            # Simulated annealing: accept worse with decreasing probability
            delta = score_new - score
            if delta < 0.5 * temp:
                accept = np.exp(-delta / max(temp, 1e-10)) > rng.random()

        if accept:
            x = x_new
            a = a_new
            b = b_new
            score = score_new
            lr = min(lr * 1.05, 2.0)
        else:
            lr = max(lr * 0.7, 0.005)
            if lr < 0.02 and no_improve > 100:
                x = np.clip(x + rng.normal(0.0, 0.03, size=m), x_eps, 1.0 - x_eps)
                a = _a_from_x(x, n)
                b = None
                score = float("inf")
                m_adam = np.zeros(m, dtype=np.float64)
                v_adam = np.zeros(m, dtype=np.float64)
                t_adam = 0

        focus = min(focus * focus_inc, float(max_focus))
        temp = max(temp * 0.995, 0.001)
        it += 1

    return _a_from_x(best_x, n), float(best_score)


def _adam_optimize_full(a0, time_limit, rng, verbose=False,
                        focus_start=5.0, max_focus=5000.0,
                        lr=0.1, label=""):

    n = int(a.size)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-300:
        a = np.ones(n, dtype=np.float64) / max(1, n)
    else:
        a /= s
    a = np.maximum(a, 1e-15)
    a /= a.sum()

    b = None
    score = float("inf")

    best_score = float("inf")
    best_a = a.copy()

    m_adam = np.zeros(n, dtype=np.float64)
    v_adam = np.zeros(n, dtype=np.float64)
    t_adam = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    focus = float(focus_start)
    focus_inc = 1.015
    no_improve = 0
    max_no = 800
    resets = 0
    max_resets = 40
    temp = 0.02
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < float(time_limit):
        if b is None:
            b = _fft_square(a)
            score, _, _ = _internal_score(a, b)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and (it % 2000 == 0):
            print(f"[{label}] it={it} t={time.monotonic()-t0:.1f}s "
                  f"score={score:.6f} best={best_score:.6f} "
                  f"focus={focus:.1f} lr={lr:.4f} resets={resets}")

        if no_improve > max_no:
            resets += 1
            if resets > max_resets:
                break
            no_improve = 0
            max_no = int(max_no * 1.4)
            r = rng.random()
            if r < 0.3:
                a = best_a * np.exp(rng.normal(0.0, 0.2, size=n))
            elif r < 0.6:
                fresh = _anti_log_init(n, rng, amp=3.0, smooth=(resets % 2 == 0))
                a = 0.7 * best_a + 0.3 * fresh
            else:
                a = _random_positive(n, rng, amp=3.0)
            a = np.maximum(a, 1e-15)
            a /= a.sum()
            b = None
            score = float("inf")
            focus = float(rng.choice([5.0, 20.0, 50.0, 100.0, 500.0]))
            lr = 0.1
            m_adam = np.zeros(n, dtype=np.float64)
            v_adam = np.zeros(n, dtype=np.float64)
            t_adam = 0
            continue

        p = _make_weights(b, focus)
        grad = _fft_grad(p, a)
        # Project gradient to preserve sum (remove mean component)
        g_mean = float(np.dot(a, grad)) / max(1e-300, float(a.sum()))
        gd = grad - g_mean
        gd = gd / max(1e-300, float(a.sum()))

        # Multiplicative Adam (for positive variables)
        t_adam += 1
        log_grad = gd / (a + 1e-300)
        m_adam = beta1 * m_adam + (1 - beta1) * log_grad
        v_adam = beta2 * v_adam + (1 - beta2) * log_grad * log_grad
        m_hat = m_adam / (1 - beta1 ** t_adam)
        v_hat = v_adam / (1 - beta2 ** t_adam)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a * np.exp(-step)
        a_new = np.maximum(a_new, 1e-300)
        ss = float(a_new.sum())
        if ss <= 1e-300:
            break
        a_new /= ss

        b_new = _fft_square(a_new)
        score_new, _, _ = _internal_score(a_new, b_new)

        accept = False
        if score_new < best_score - 1e-12:
            accept = True
        elif score_new < score - 1e-12:
            accept = True
        else:
            delta = score_new - score
            if delta < 0.5 * temp:
                accept = np.exp(-delta / max(temp, 1e-10)) > rng.random()

        if accept:
            a = a_new
            b = b_new
            score = score_new
            lr = min(lr * 1.05, 2.0)
        else:
            lr = max(lr * 0.7, 0.005)
            if lr < 0.02 and no_improve > 100:
                a = a * np.exp(rng.normal(0.0, 0.03, size=n))
                a = np.maximum(a, 1e-15)
                a /= a.sum()
                b = None
                score = float("inf")
                m_adam = np.zeros(n, dtype=np.float64)
                v_adam = np.zeros(n, dtype=np.float64)
                t_adam = 0

        focus = min(focus * focus_inc, float(max_focus))
        temp = max(temp * 0.995, 0.001)
        it += 1

    return best_a, float(best_score)


def _coarse_to_fine(coarse_a, n_fine, rng):
    """Upsample a coarse sequence to a finer one by repeating and perturbing."""
    n_coarse = int(coarse_a.size)
    n_fine = int(n_fine)
    n_coarse = min(n_coarse, n_fine)
    # Tile the coarse sequence
    reps = n_fine // n_coarse
    fine = np.tile(coarse_a, reps)
    fine = fine[:n_fine]
    # Add small perturbation
    fine = fine * np.exp(rng.normal(0.0, 0.1, size=n_fine))
    fine = np.maximum(fine, 1e-15)
    fine /= fine.sum()
    return fine


def _safe_evaluate(seq):
    ev = globals().get("evaluate_sequence")
    if ev is None or not callable(ev):
        return float("inf")
    try:
        return float(ev(seq))
    except Exception:
        return float("inf")


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main search function. Returns the best sequence found."""
    seed = int(seed)
    total = max(60.0, float(budget_s))

    now0 = time.monotonic()
    final_deadline = now0 + total - 20.0
    task_deadline = now0 + max(60.0, total - 180.0)

    print("starting improved step-function convolution optimizer")
    best_a = None
    best_score = float("inf")

    # Load given sequence
    given_a = None
    try:
        given = globals().get("height_sequence_1")
        if given is not None:
            g = np.asarray(given, dtype=np.float64).ravel()
            g = np.nan_to_num(g, nan=0.0, posinf=1000.0, neginf=0.0)
            g = np.clip(g, 0.0, 1000.0)
            s = float(g.sum())
            if s > 0.01 and g.size > 10:
                gn = g / s
                sc, _, _ = _internal_score(gn)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = gn.copy()
                if g.size <= 16384:
                    given_a = gn
                print(f"given internal score: {sc:.6f}, n={gn.size}")
    except Exception:
        given_a = None

    tasks = []

    # Phase 1: Optimize given sequence (quick, high-value)
    if given_a is not None:
        tasks.append((given_a.size, "full", 0, 120.0, "given"))

    # Phase 2: Anti-palindromic at n=8192 (refinement of previous best)
    tasks.append((8192, "pair", 0, 240.0, "p8192a"))
    tasks.append((8192, "pair", 1, 240.0, "p8192b"))

    # Phase 3: Anti-palindromic at n=16384 (new, more DOF)
    tasks.append((16384, "pair", 0, 360.0, "p16384a"))
    tasks.append((16384, "pair", 1, 300.0, "p16384b"))
    tasks.append((16384, "pair", 2, 240.0, "p16384c"))

    # Phase 4: Full vector at n=16384 (anti-palindromic init)
    tasks.append((16384, "full", 0, 180.0, "f16384"))

    total_alloc = sum(t[3] for t in tasks)
    print(f"total planned allocation: {total_alloc:.0f}s")

    for idx, (n, kind, task_seed, max_alloc, label) in enumerate(tasks):
        now = time.monotonic()
        if now >= task_deadline:
            break
        alloc = min(max_alloc, task_deadline - now)
        if alloc < 30.0:
            break

        ng_seed = (seed * 1000003 + idx * 7919 + task_seed * 131) % (2**32)
        task_rng = np.random.default_rng(ng_seed)

        print(f"start task idx={idx} kind={kind} n={n} seed={task_seed} "
              f"alloc={alloc:.1f}s best={best_score:.6f} label={label}")
        t0 = time.monotonic()
        cand_a = None
        cand_score = float("inf")

        try:
            if kind == "pair":
                # Anti-palindromic optimization
                if task_seed == 0:
                    x0 = _greedy_x(n, task_rng, shuffle=False)
                elif task_seed == 1:
                    x0 = _greedy_x(n, task_rng, shuffle=True)
                else:
                    # Start from a smoothed anti-palindromic init
                    a_init = _anti_log_init(n, task_rng, amp=4.0, smooth=True)
                    m = n // 2
                    x0 = a_init[:m] * m
                    x0 = np.clip(x0, 1e-6, 1.0 - 1e-6)

                cand_a, cand_score = _adam_optimize_pair(
                    x0, alloc, task_rng,
                    verbose=(n == 16384 and task_seed == 0),
                    focus_start=10.0,
                    max_focus=5000.0,
                    lr=0.1,
                    label=label,
                )
            elif kind == "full":
                if label == "given" and given_a is not None:
                    a0 = given_a.copy()
                else:
                    a0 = _anti_log_init(n, task_rng, amp=3.0, smooth=True)
                cand_a, cand_score = _adam_optimize_full(
                    a0, alloc, task_rng,
                    verbose=False,
                    focus_start=8.0,
                    max_focus=5000.0,
                    lr=0.1,
                    label=label,
                )
            else:
                a0 = _anti_log_init(n, task_rng, amp=3.0)
                cand_a, cand_score = _adam_optimize_full(
                    a0, alloc, task_rng,
                    verbose=False,
                    focus_start=5.0,
                    label=label,
                )

            if cand_a is not None:
                cand_a = np.nan_to_num(cand_a, nan=0.0, posinf=0.0, neginf=0.0)
                cand_a = np.clip(cand_a, 0.0, 1000.0)
                cc = float(cand_a.sum())
                if cc > 0.01:
                    cand_a /= cc
                    internal, _, _ = _internal_score(cand_a)
                    if np.isfinite(internal) and internal < cand_score:
                        cand_score = internal
                    if np.isfinite(cand_score) and cand_score < best_score:
                        best_score = cand_score
                        best_a = cand_a.copy()
                        print(f"  new best! score={best_score:.6f} n={cand_a.size}")
        except Exception as exc:
            print(f"  task {label} failed: {repr(exc)}")

        print(f"end task idx={idx} label={label} used={time.monotonic()-t0:.1f}s "
              f"candidate={cand_score:.6f} best={best_score:.6f}")

        if best_score <= 1.51:
            print("very strong score found, moving to final refinement")
            break

    # Final refinement phase
    if best_a is not None:
        remaining = final_deadline - time.monotonic()
        if remaining > 60.0:
            refine_time = min(140.0, remaining - 30.0)
            if refine_time > 30.0:
                task_rng = np.random.default_rng((seed + 999999) % (2**32))
                print(f"final refine: time={refine_time:.1f}s, n={best_a.size}")
                # Try full-vector refinement
                refined_a, refined_score = _adam_optimize_full(
                    best_a, refine_time * 0.6, task_rng,
                    verbose=True,
                    focus_start=50.0,
                    max_focus=10000.0,
                    lr=0.05,
                    label="final_full",
                )
                if refined_a is not None and np.isfinite(refined_score):
                    if refined_score < best_score:
                        best_a = refined_a
                        best_score = refined_score
                        print(f"  full refinement improved to {best_score:.6f}")

                # Try anti-palindromic refinement if n is even
                if best_a.size % 2 == 0 and best_a.size >= 1024:
                    remaining2 = final_deadline - time.monotonic()
                    if remaining2 > 40.0:
                        m = best_a.size // 2
                        x0 = best_a[:m] * m
                        x0 = np.clip(x0, 1e-10, 1.0 - 1e-10)
                        task_rng2 = np.random.default_rng((seed + 1234567) % (2**32))
                        print(f"anti-palindromic refine: time={remaining2-20:.1f}s")
                        ra, rs = _adam_optimize_pair(
                            x0, remaining2 - 30, task_rng2,
                            verbose=False,
                            focus_start=50.0,
                            max_focus=10000.0,
                            lr=0.05,
                            label="final_pair",
                        )
                        if ra is not None and np.isfinite(rs) and rs < best_score:
                            best_a = ra
                            best_score = rs
                            print(f"  pair refinement improved to {best_score:.6f}")

    if best_a is None:
        best_a = np.ones(1024, dtype=np.float64) / 1024.0

    best_a = np.nan_to_num(best_a, nan=0.0, posinf=1000.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    ss = float(best_a.sum())
    if ss <= 0.01:
        best_a = np.ones(1024, dtype=np.float64) / 1024.0
        ss = 1.0
    else:
        best_a /= ss

    final_seq = [float(x) for x in best_a]
    print(f"final: n={best_a.size}, internal_score={best_score:.6f}")

    # Verify with exact evaluation if possible
    if best_a.size <= 20000 and time.monotonic() < final_deadline - 5.0:
        exact = _safe_evaluate(final_seq)
        if np.isfinite(exact):
            print(f"exact evaluation: {exact:.6f}")
        else:
            print("exact evaluation returned inf (unexpected)")

    return final_seq
