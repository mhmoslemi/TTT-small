where Δ is the simplex. The inner max is over probability vectors p on the lags.

By minimax theorem (under appropriate conditions):

The inner min is: min_a sum_k p[k] * c[k] = min_a sum_k p[k] * sum_i a[i]*a[k-i]

subject to sum(a) = 1, a >= 0.

Where P = sum_k p[k] Q_k. If p is supported on a single k, P = Q_k. If p is uniform, P = sum_k Q_k / (2n-1).

The inner problem is: min a^T P a s.t. sum(a) = 1, a >= 0.

If P is PSD, this is a QP. If P is not PSD, it's non-convex.

    """Reduce peaks by targeting the contributing elements."""
    n = len(a)
    S = float(np.sum(a))
    best_a = a.copy()
    best_score = _score(a)
    
    for it in range(n_iters):
        c = _fft_convolve(a)
        max_c = float(np.max(c))
        score = 2.0 * n * max_c / S / S
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Find top peak lags
        top_k = np.argpartition(c, -n_peaks)[-n_peaks:]
        
        # For each peak, compute sensitivity
        s = np.zeros(n)
        for k in top_k:
            k = int(k)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    s[i] += a[j]
        
        # Reduce elements with high sensitivity, increase those with low
        s_mean = np.mean(s)
        direction = s - s_mean  # positive: reduce, negative: increase
        
        # Step
        lr = 0.05 * max_c / max(np.max(s), 1e-10)
        a_new = a - lr * direction
        a_new = np.maximum(a_new, 0.0)
        S_new = float(np.sum(a_new))
        if S_new < 1e-10:
            break
        a_new = a_new * S / S_new
        a = a_new
    
    return best_a, best_score

            return result.x

        pass

    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = _score(a)
    blend = initial_blend
    stall_count = 0
    stall_threshold = 50
    
    for it in range(max_iters):

"""
Minimize J(a) = 2*n*max(conv(a,a))/sum(a)^2 using alternating LP,
smooth gradient descent, and peak-targeted refinement.
"""
import numpy as np
import time
from scipy.optimize import linprog

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]

def _score(a):
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)

def _build_conv_matrix(f):
    n = len(f)
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = f
    return C

def _lp_solve(f, M, tight_lags=None, tight_factor=0.999):
    n = len(f)
    C = _build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    if tight_lags is not None and len(tight_lags) > 0:
        for k in tight_lags:
            b_ub[k] = M * tight_factor
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.status == 0:
            return result.x
    except Exception:
        pass
    # Fallback without tight constraints
    if tight_lags is not None:
        try:
            b_ub2 = M * np.ones(2 * n - 1)
            result = linprog(c_obj, A_ub=C, b_ub=b_ub2, bounds=bounds,
                             method='highs', options={'presolve': True})
            if result.status == 0:
                return result.x
        except Exception:
            pass
    return None

def _interpolate_sequence(src, n_target):
    n_src = len(src)
    if n_src == n_target:
        return src.copy()
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    out = np.interp(x_tgt, x_src, src)
    out = np.maximum(out, 0.0)
    s = float(np.sum(out))
    if s > 1e-10:
        out = out / s
    return out

    s = float(np.sum(a))

def _alternating_lp(a_init, max_iters, deadline, initial_blend=0.8):
    a = a_init.copy().astype(np.float64)
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = _score(a)
    blend = initial_blend
    stall_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        
        c = _fft_convolve(a)
        M = float(np.max(c))
        
        # Get peak lags for tighter constraints
        n_peaks = min(5, len(c))
        peak_lags = np.where(c >= M * 0.999)[0].tolist()
        
        g = _lp_solve(a, M, tight_lags=peak_lags if len(peak_lags) < 20 else None)
        if g is None:
            g = _lp_solve(a, M)
        if g is None:
            break
        sum_g = float(np.sum(g))
        if sum_g < 1e-12:
            break
        g = g / sum_g
        
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)

            break

        else:
            stall_count += 1
        
        a = a_new
        blend = max(0.05, blend * 0.997)
        
        if stall_count > 200:
            break
    
    return best_a, best_score

def _smooth_grad_descent(a_init, max_iters, deadline):
    """Minimize sum(c[k]^2)/S^2 using gradient descent."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    
    best_a = a.copy()
    best_score = _score(a)
    best_smooth = float(np.sum(_fft_convolve(a) ** 2))
    
    lr = 0.1
    decay = 0.999
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        S = float(np.sum(a))
        smooth_obj = float(np.sum(c ** 2)) / (S * S)
        score = _score(a)
        
        if score < best_score - 1e-9:
            best_score = score
            best_a = a.copy()
            best_smooth = smooth_obj
            if it % 100 == 0:
                print(f"    SGrad iter {it}: score={best_score:.6f}, smooth={best_smooth:.6e}")
        
        # Gradient of smooth_obj = sum(c^2)/S^2
        # d/d a[i] = 4 * sum_k c[k]*a[k-i] / S^2 - 2*sum(c^2) / S^3
        # Compute sum_k c[k]*a[k-i] efficiently
        # This is the cross-correlation of c and a (reversed)
        ca = np.zeros(n)
        for i in range(n):
            k_min = max(0, i)
            k_max = min(2*n-1, i + n - 1)
            idx = slice(k_min, k_max + 1)
            a_idx = slice(0, k_max - k_min + 1)
            ca[i] = float(np.dot(c[idx], a[a_idx]))
        
        grad = 4.0 * ca / (S * S) - 2.0 * float(np.sum(c**2)) / (S**3)
        
        # Normalize gradient
        gnorm = float(np.sqrt(np.sum(grad**2)))
        if gnorm > 1e-15:
            grad = grad / gnorm
            a = a - lr * grad
            a = np.maximum(a, 0.0)
            S_new = float(np.sum(a))
            if S_new > 1e-10:
                a = a / S_new
        
        lr *= decay
    
    print(f"    SGrad final: score={best_score:.6f}")
    return best_a, best_score

def _subgradient(a_init, max_iters, deadline, lr_init=0.3, n_peaks=10):
    """Subgradient method targeting top peaks."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n

    lr = lr_init

            break
        
        c = _fft_convolve(a)
        max_c = float(np.max(c))
        score = 2.0 * n * max_c / (S * S)
        
        if score < best_score - 1e-9:
            best_score = score
            best_a = a.copy()
            if it % 200 == 0:
                print(f"    Subgrad iter {it}: score={best_score:.6f}")
        
        # Find top peak lags
        if n_peaks >= len(c):
            top_k = np.arange(len(c))
        else:
            top_k = np.argpartition(c, -n_peaks)[-n_peaks:]
        
        # Sensitivity vector
        s = np.zeros(n)
        for k in top_k:
            k = int(k)
            # s[i] += a[k-i] for valid k-i
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            if i_min <= i_max:
                s[i_min:i_max+1] += a[k - i_min:k - i_max + 1][::-1]
        
        # Normalize and update
        s_max = float(np.max(s))
        if s_max > 1e-15:
            s = s / s_max
            a = a - lr * s
            a = np.maximum(a, 0.0)
            S_new = float(np.sum(a))
            if S_new > 1e-10:
                a = a * S / S_new
        
        lr *= 0.9999
    
    return best_a, best_score
