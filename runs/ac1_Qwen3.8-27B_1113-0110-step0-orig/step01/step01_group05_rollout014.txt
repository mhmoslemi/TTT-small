"""
Search for nonnegative coefficient sequences minimizing
    2 * n * max(conv(a, a)) / (sum(a))^2
using a hybrid of:
  * FFT-accelerated smooth-max / log-sum-exp multiplicative updates,
  * L-BFGS on a softmax parametrization for medium lengths,
  * state-of-the-art LP direction polishes,
  * exact one-coordinate line searches,
  * prototype down/upsampling and final-length polishing.
"""

import numpy as np
import time
import math
from functools import partial


# -----------------------------
# Basic utilities
# -----------------------------

def _pool_key(item):
    return item[0]


def _clean_array(arr):
    a = np.asarray(arr, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    np.clip(a, 0.0, 1000.0, out=a)
    return a


def _normalize(a):
    a = _clean_array(a)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(max(1, a.size), dtype=np.float64)
    return a / s


def _softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return np.ones(1, dtype=np.float64)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(z.size, dtype=np.float64) / float(z.size)
    return e / s


def _convolve(x, y=None):
    x = np.asarray(x, dtype=np.float64).ravel()
    if y is None:
        y = x
    else:
        y = np.asarray(y, dtype=np.float64).ravel()

    nx = int(x.size)
    ny = int(y.size)
    if nx == 0 or ny == 0:
        return np.ones(1, dtype=np.float64)

    need = nx + ny - 1

    # Direct convolution is usually faster for small sizes and avoids roundoff.
    if nx * ny <= 200000:
        out = np.convolve(x, y, mode="full")
        out = np.where(np.isfinite(out), out, 0.0)
        np.maximum(out, 0.0, out=out)
        return out

    L = 1
    while L < need:
        L <<= 1

    X = np.fft.rfft(x, L)
    if y is x:
        Y = X
    else:
        Y = np.fft.rfft(y, L)

    out = np.fft.irfft(X * Y, L)[:need]
    out = np.where(np.isfinite(out), out, 0.0)
    np.maximum(out, 0.0, out=out)
    return out


def _score(a):
    a = _normalize(a)
    n = int(a.size)
    q = _convolve(a, a)
    if q.size == 0:
        return float("inf")
    mx = float(np.max(q))
    if not np.isfinite(mx):
        return float("inf")
    if mx < 0.0:
        mx = 0.0
    return float(2.0 * n * mx)


def _softmax_vals(v, beta):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = int(v.size)
    if n == 0:
        return np.ones(1, dtype=np.float64)
    if beta <= 0.0 or not np.isfinite(beta):
        return np.ones(n, dtype=np.float64) / float(n)
    z = beta * v
    m = float(np.max(z))
    if not np.isfinite(m):
        return np.ones(n, dtype=np.float64) / float(n)
    e = np.exp(z - m)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(n, dtype=np.float64) / float(n)
    return e / s


def _lse(v, beta):
    v = np.asarray(v, dtype=np.float64).ravel()
    if v.size == 0:
        return 0.0
    if beta <= 0.0 or not np.isfinite(beta):
        return 0.0
    z = beta * v
    m = float(np.max(z))
    if not np.isfinite(m):
        return float("inf")
    e = np.exp(z - m)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        return float("inf")
    return float(m + math.log(s))


def _toeplitz_matvec(p, a):
    """
    Returns v with
        v_i = sum_j a_j p_{i+j}
    for i = 0,...,len(a)-1, where p has length 2*len(a)-1.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    p = np.asarray(p, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return np.ones(0, dtype=np.float64)
    if p.size != 2 * n - 1:
        # Fallback: standard convolution/correlation not aligned; zero padding.
        p = np.zeros(2 * n - 1, dtype=np.float64)
        m = min(p.size, int(p.size))
        p[:m] = np.asarray(p, dtype=np.float64)[:m]

    prep = p[::-1]
    out = _convolve(a, prep)
    maxidx = prep.size - 1
    idx = maxidx - np.arange(n, dtype=np.int64)
    out = out[idx]
    out = np.where(np.isfinite(out), out, 0.0)
    np.maximum(out, 0.0, out=out)
    return out

def _two_level(n):
    n = int(n)
    if n <= 1:
        return np.ones(max(1, n), dtype=np.float64)
    p = np.zeros(n, dtype=np.float64)
    m = n // 2
    if m > 0:
        p[:m] = 0.65 / float(m)
    rest = n - m
    if rest > 0:
        p[m:] = 0.35 / float(rest)
    return _normalize(p)


def _paired_initial(n, rng, alpha_w=1.0, split_alpha=0.4):
    n = int(n)
    if n <= 1:
        return np.ones(max(1, n), dtype=np.float64)
    m = (n + 1) // 2
    weights = rng.dirichlet(np.full(m, float(alpha_w), dtype=np.float64))
    p = np.zeros(n, dtype=np.float64)
    sa = max(0.05, float(split_alpha))
    for r in range(m):
        a = r
        b = n - 1 - r
        if a == b:
            p[a] = weights[r]
        else:
            if rng.random() < 0.75:
                # Put all mass on one mirror side.
                if rng.random() < 0.5:
                    q0, q1 = 1.0, 0.0
                else:
                    q0, q1 = 0.0, 1.0
            else:
                q = rng.dirichlet(np.full(2, sa, dtype=np.float64))
                q0, q1 = float(q[0]), float(q[1])
            p[a] = weights[r] * q0
            p[b] = weights[r] * q1
    return _normalize(p)


def _sin_initial(n, rng):
    n = int(n)
    if n <= 1:
        return np.ones(max(1, n), dtype=np.float64)
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    base = 0.15 + 0.85 * np.power(np.sin(math.pi * x), 0.8)
    amp = float(rng.uniform(0.45, 0.95))
    k = int(rng.integers(1, 4))
    p = base * (1.0 + amp * np.sin(2.0 * math.pi * k * x))
    p *= 1.0 + 0.08 * rng.random(n)
    p = np.where(np.isfinite(p), p, 0.0)
    np.maximum(p, 0.0, out=p)
    return _normalize(p)


def _initial(n, code, rng):
    n = int(n)
    code = int(code) % 10
    if n <= 1:
        return np.ones(max(1, n), dtype=np.float64)

    if code == 0:
        p = np.ones(n, dtype=np.float64)
    elif code == 1:
        p = rng.dirichlet(np.ones(n, dtype=np.float64))
    elif code == 2:
        p = np.exp(rng.normal(0.0, 0.8, n))
    elif code == 3:
        p = _paired_initial(n, rng, alpha_w=1.0, split_alpha=0.5)
    elif code == 4:
        p = _two_level(n)
    elif code == 5:
        p = _sin_initial(n, rng)
    elif code == 6:
        K = int(rng.integers(4, max(5, n // 8 + 1)))
        K = min(K, n)
        levels = rng.dirichlet(np.full(K, float(rng.uniform(0.25, 2.0)), dtype=np.float64))
        blocks = (np.arange(n) * K) // n
        p = levels[blocks]
    elif code == 7:
        p = (rng.random(n) > 0.5).astype(np.float64)
        p += 0.08 * rng.random(n)
    elif code == 8:
        p = _sin_initial(n, rng)
        p *= rng.dirichlet(np.ones(n, dtype=np.float64)) * 1.0
        p = p + 0.05 * _sin_initial(n, rng)
    else:
        p = _paired_initial(n, rng, alpha_w=0.6, split_alpha=0.9)

    p = np.where(np.isfinite(p), p, 0.0)
    np.maximum(p, 0.0, out=p)
    return _normalize(p)

def _downsample(a, m):
    a = _clean_array(a)
    m = int(m)
    if m <= 0:
        return np.ones(1, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.ones(m, dtype=np.float64)
    if n == m:
        return _normalize(a)
    if n < m:
        return _expand_to_length(a, m)

    edges = np.linspace(0, n, m + 1).astype(np.int64)
    out = np.zeros(m, dtype=np.float64)
    for i in range(m):
        lo = int(edges[i])
        hi = int(edges[i + 1])
        if hi > lo:
            out[i] = float(np.mean(a[lo:hi]))
        elif lo < n:
            out[i] = float(a[lo])
    return _normalize(out)


def _expand_to_length(p, target_len):
    target_len = int(target_len)
    p = _normalize(p)
    R = int(p.size)
    if R <= 0 or target_len <= 0:
        return np.ones(1, dtype=np.float64)
    if R == target_len:
        return p.copy()
    if R > target_len:
        return _downsample(p, target_len)

    base = target_len // R
    rem = target_len % R
    lengths = np.full(R, base, dtype=np.int64)
    if rem > 0:
        lengths[:rem] += 1

    out = np.zeros(target_len, dtype=np.float64)
    idx = 0
    for i in range(R):
        L = int(lengths[i])
        if L > 0:
            out[idx:idx + L] = p[i] / float(L)
            idx += L
    return _normalize(out)

def _pool_add(pool, seq, score, limit=40):
    if pool is None:
        return
    if seq is None:
        return
    if not np.isfinite(score):
        return
    arr = np.asarray(seq, dtype=np.float64).ravel()
    if arr.size == 0 or not np.all(np.isfinite(arr)):
        return
    arr = _normalize(arr)
    pool.append([float(score), int(arr.size), arr])
    pool.sort(key=_pool_key)
    while len(pool) > int(limit):
        pool.pop()

def _smooth_step(a, beta, deadline, rng, etas=None):
    a = _normalize(a).copy()
    n = int(a.size)
    if n <= 0:
        return a, _score(a)
    if etas is None:
        etas = (0.05, 0.15, 0.35, 0.8)
    etas = list(etas)

    best_a = a.copy()
    q = _convolve(a, a)
    best_sc = float(2.0 * n * max(0.0, float(np.max(q)))) if q.size else float("inf")

    no_best = 0
    it = 0

    while time.monotonic() < deadline - 0.2:
        q = _convolve(a, a)
        cur_sc = float(2.0 * n * max(0.0, float(np.max(q)))) if q.size else float("inf")
        if np.isfinite(cur_sc) and cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            no_best = 0
        else:
            no_best += 1

        if no_best > 18:
            # Perturb from best and try again.
            scale = 0.35 if no_best < 24 else 0.18
            pert = best_a * np.exp(rng.normal(0.0, scale, n))
            pert = _normalize(pert)
            a = pert
            no_best = 0

        p = _softmax_vals(q, beta)
        r = 2.0 * _toeplitz_matvec(p, a)
        r = np.maximum(r, 1e-15)

        log_a = np.log(np.maximum(a, 1e-15))
        log_r = np.log(r)
        mean_lr = float(np.mean(log_r))

        cur_lse = _lse(q, beta)

        accepted = False
        for eta in etas:
            la = log_a - float(eta) * (log_r - mean_lr)
            la = la - float(np.max(la))
            e = np.exp(la)
            s = float(np.sum(e))
            if not np.isfinite(s) or s <= 0.0:
                continue
            cand = e / s
            cq = _convolve(cand, cand)
            csc = float(2.0 * n * max(0.0, float(np.max(cq)))) if cq.size else float("inf")
            if not np.isfinite(csc):
                continue

            if csc < best_sc - 1e-12:
                a = cand.copy()
                best_a = cand.copy()
                best_sc = csc
                no_best = 0
                accepted = True
                break

            # Allow surrogate descent if the true penalty does not worsen much.
            clse = _lse(cq, beta)
            if cur_lse > 1e-9 and clse < cur_lse - 1e-6 and csc < best_sc + 0.03:
                a = cand.copy()
                no_best = 0
                accepted = True
                break

        if not accepted:
            # If no update, rotate etas.
            etas = etas[1:] + etas[:1]
            if no_best >= 12:
                break

        it += 1
        if it > 100000:
            break

    return best_a, best_sc

def _lbfgs_objective(z, beta, n, ridge):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n:
        return 1e100, np.zeros(max(1, int(n)), dtype=np.float64)
    if not np.all(np.isfinite(z)):
        return 1e100, np.zeros(n, dtype=np.float64)

    a = _softmax_from_z(z)
    q = _convolve(a, a)
    if q.size != 2 * n - 1:
        q = np.ones(2 * n - 1, dtype=np.float64) / (2.0 * n - 1.0)

    zbeta = beta * q
    m = float(np.max(zbeta))
    if not np.isfinite(m):
        return 1e100, np.zeros(n, dtype=np.float64)
    ez = np.exp(zbeta - m)
    sz = float(np.sum(ez))
    if not np.isfinite(sz) or sz <= 0.0:
        return 1e100, np.zeros(n, dtype=np.float64)

    lse = m + math.log(sz)

    p = ez / sz
    G = 2.0 * _toeplitz_matvec(p, a)
    dot = float(np.dot(G, a))
    grad = G - a * dot
    grad += 2.0 * ridge * z

    val = lse + ridge * float(np.dot(z, z))
    if not np.isfinite(val):
        val = 1e100
    grad = np.where(np.isfinite(grad), grad, 0.0)

    return float(val), grad


def _lbfgs_optimize(a0, beta, deadline, maxiter=80):
    a0 = _normalize(a0).copy()
    n = int(a0.size)
    sc0 = _score(a0)
    if n < 16 or n > 256:
        return a0, sc0
    if time.monotonic() > deadline - 5.0:
        return a0, sc0

    try:
        from scipy.optimize import minimize
    except Exception:
        return a0, sc0

    z0 = np.log(np.maximum(a0, 1e-15))
    z0 = z0 - float(np.mean(z0))

    try:
        fun = partial(_lbfgs_objective, beta=float(beta), n=n, ridge=1e-9)
        res = minimize(
            fun,
            z0,
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": int(maxiter),
                "maxfun": int(4 * maxiter),
                "ftol": 1e-8,
                "gtol": 1e-6,
            }
        )
        if hasattr(res, "x") and np.all(np.isfinite(np.asarray(res.x, dtype=np.float64))):
            a1 = _softmax_from_z(np.asarray(res.x, dtype=np.float64))
            sc1 = _score(a1)
            if np.isfinite(sc1) and sc1 < sc0:
                return a1, sc1
    except Exception:
        pass

    return a0, sc0

def _lp_polish(a, deadline, max_iters=2, max_n=512):
    a = _normalize(a).copy()
    n = int(a.size)
    best = a.copy()
    sc_best = _score(a)

    if n < 2 or n > int(max_n):
        return best, sc_best
    if time.monotonic() > deadline - 5.0:
        return best, sc_best

    try:
        from scipy.optimize import linprog
        from scipy.sparse import coo_matrix
    except Exception:
        return best, sc_best

    for _ in range(int(max_iters)):
        if time.monotonic() > deadline - 5.0:
            break

        ca = _convolve(best, best)
        M = float(np.max(ca)) if ca.size else 0.0
        if not np.isfinite(M) or M <= 0.0:
            break

        # A @ b = (best * b)
        cols = np.repeat(np.arange(n), n)
        rows = cols + np.tile(np.arange(n), n)
        vals = np.tile(best, n)
        A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()

        try:
            res = linprog(
                c=-np.ones(n, dtype=np.float64),
                A_ub=A,
                b_ub=np.full(2 * n - 1, M, dtype=np.float64),
                bounds=[(0.0, None)] * n,
                method="highs",
                options={"presolve": True}
            )
        except Exception:
            break

        if res is None or not getattr(res, "success", False):
            break

        g = np.asarray(getattr(res, "x", np.empty(n)), dtype=np.float64)
        g = np.where(np.isfinite(g), g, 0.0)
        np.maximum(g, 0.0, out=g)
        gs = float(np.sum(g))
        if not np.isfinite(gs) or gs <= 1e-6:
            break
        g = g / gs

        cg = _convolve(g, g)
        cr = _convolve(best, g)

        cur = float(2.0 * n * max(0.0, float(np.max(ca))))
        best_t = 0.0
        best_val = cur

        for t in np.linspace(0.0, 1.0, 65, dtype=np.float64):
            t = float(t)
            a1 = 1.0 - t
            b1 = t
            ct = a1 * a1 * ca + 2.0 * a1 * b1 * cr + b1 * b1 * cg
            val = float(2.0 * n * max(0.0, float(np.max(ct))))
            if np.isfinite(val) and val < best_val:
                best_val = val
                best_t = t

        if best_t <= 1e-6 or not np.isfinite(best_val) or best_val >= cur - 1e-11:
            break

        # Refine around best_t.
        lo = max(0.0, best_t - 0.25)
        hi = min(1.0, best_t + 0.25)
        if hi > lo:
            for t in np.linspace(lo, hi, 49, dtype=np.float64):
                t = float(t)
                a1 = 1.0 - t
                b1 = t
                ct = a1 * a1 * ca + 2.0 * a1 * b1 * cr + b1 * b1 * cg
                val = float(2.0 * n * max(0.0, float(np.max(ct))))
                if np.isfinite(val) and val < best_val:
                    best_val = val
                    best_t = t

        if best_t <= 1e-6 or not np.isfinite(best_val) or best_val >= cur - 1e-11:
            break

        a1 = best.copy()
        b1 = g.copy()
        nd = (1.0 - best_t) * a1 + best_t * b1
        nd = _normalize(nd)
        sc = _score(nd)
        if np.isfinite(sc) and sc < sc_best - 1e-12:
            sc_best = sc
            best = nd
        else:
            break

    return best, sc_best

def _coord_params(n):
    n = int(n)
    if n <= 64:
        cnt, ref, xmax = 40, 18, 1.0
    elif n <= 128:
        cnt, ref, xmax = 34, 14, 1.0
    elif n <= 256:
        cnt, ref, xmax = 26, 10, 1.0
    elif n <= 1024:
        cnt, ref, xmax = 18, 6, 0.5
    elif n <= 2048:
        cnt, ref, xmax = 15, 5, 0.35
    else:
        cnt, ref, xmax = 14, 4, 0.25

    xmax = min(1.0, float(xmax))
    hi_log = math.log10(max(1e-9, xmax))
    grid = np.logspace(-9.0, hi_log, max(2, int(cnt)), dtype=np.float64)
    grid = np.concatenate(([0.0], grid))
    if grid.size == 0 or grid[-1] < xmax:
        grid = np.concatenate((grid, [xmax]))
    grid = np.unique(grid[np.isfinite(grid) & (grid >= 0.0) & (grid <= 1.0)])
    return grid, int(ref)


def _coord_evaluate(xs, c_k, coeff, const, mid, unaff, n, S0, old):
    xs = np.asarray(xs, dtype=np.float64).ravel()
    xs = xs[np.isfinite(xs) & (xs >= 0.0) & (xs <= 1.0 + 1e-12)]
    if xs.size == 0:
        return np.empty(0, dtype=np.float64), np.empty(0, dtype=np.float64)

    S = S0 + xs - old
    aff = c_k[None, :] + coeff[None, :] * xs[:, None] + const[None, :]
    aff[:, mid] += xs * xs
    max_aff = np.max(aff, axis=1)
    maxc = np.maximum(max_aff, unaff)

    vals = np.full(xs.size, float("inf"), dtype=np.float64)
    valid = S > 1e-14
    if np.any(valid):
        vals[valid] = (2.0 * int(n)) * maxc[valid] / (S[valid] * S[valid])
    return xs, vals


def _coordinate_update(p, c, score, i, grid, refine_count):
    n = int(p.size)
    p = np.asarray(p, dtype=np.float64).ravel()
    c = np.asarray(c, dtype=np.float64).ravel()
    if c.size != 2 * n - 1:
        c = _convolve(p, p)

    S0 = float(np.sum(p))
    if not np.isfinite(S0) or S0 <= 1e-14:
        return p, c, score, False

    old = float(p[i])
    if not np.isfinite(old) or old < 0.0:
        old = 0.0

    ks = slice(i, i + n)
    c_k = c[ks]

    coeff = 2.0 * p.copy()
    const = -2.0 * old * p.copy()
    mid = int(i)
    if 0 <= mid < n:
        coeff[mid] = 0.0
        const[mid] = -old * old

    max_unaff = 0.0
    if i > 0:
        m1 = c[:i]
        if m1.size > 0:
            max_unaff = max(max_unaff, float(np.max(m1)))
    end = i + n
    if end < 2 * n - 1:
        m2 = c[end:]
        if m2.size > 0:
            max_unaff = max(max_unaff, float(np.max(m2)))
    if not np.isfinite(max_unaff):
        max_unaff = 0.0

    xs_arr = np.asarray(grid, dtype=np.float64).ravel()
    xs_arr = np.concatenate((xs_arr, [old, 0.0]))
    xs_arr = np.unique(xs_arr[np.isfinite(xs_arr) & (xs_arr >= 0.0) & (xs_arr <= 1.0)])
    if xs_arr.size == 0:
        return p, c, score, False

    xs_arr, vals = _coord_evaluate(
        xs_arr, c_k, coeff, const, mid, max_unaff, n, S0, old
    )
    if xs_arr.size == 0 or not np.any(np.isfinite(vals)):
        return p, c, score, False

    finite_vals = vals.copy()
    finite_vals[~np.isfinite(finite_vals)] = float("inf")
    idx = int(np.argmin(finite_vals))
    best_x = float(xs_arr[idx])
    best_val = float(finite_vals[idx])

    tol = 1e-12 * max(1.0, float(score))
    if not np.isfinite(best_val) or best_val >= score - tol:
        return p, c, score, False

    if int(refine_count) > 0 and xs_arr.size >= 2:
        if idx == 0:
            left = 0.0
            right = float(xs_arr[1]) if xs_arr.size > 1 else 0.02
            hi = max(right, best_x, 1e-6)
            pts = np.linspace(0.0, min(1.0, hi), int(refine_count), dtype=np.float64)
        elif idx == int(xs_arr.size - 1):
            left = float(xs_arr[idx - 1])
            hi = min(1.0, max(best_x * 5.0, left + 1e-6, 0.01))
            pts = np.linspace(max(0.0, left), hi, int(refine_count), dtype=np.float64)
        else:
            left = float(xs_arr[idx - 1])
            right = float(xs_arr[idx + 1])
            lo = min(left, right)
            hi = max(left, right)
            if hi <= lo:
                pts = np.array([best_x], dtype=np.float64)
            else:
                if lo > 1e-10:
                    pts = np.logspace(math.log10(lo), math.log10(hi), int(refine_count), dtype=np.float64)
                else:
                    pts = np.linspace(lo, hi, int(refine_count), dtype=np.float64)

        local = np.concatenate((xs_arr, [old, best_x, 0.0, best_x * 10.0, best_x / 10.0]))
        local = np.concatenate((local, pts))
        local = np.unique(local[np.isfinite(local) & (local >= 0.0) & (local <= 1.0)])

        if local.size > 0:
            xs_loc, vals_loc = _coord_evaluate(
                local, c_k, coeff, const, mid, max_unaff, n, S0, old
            )
            if xs_loc.size > 0 and np.any(np.isfinite(vals_loc)):
                fv = vals_loc.copy()
                fv[~np.isfinite(fv)] = float("inf")
                j2 = int(np.argmin(fv))
                if float(fv[j2]) < best_val:
                    best_x = float(xs_loc[j2])
                    best_val = float(fv[j2])

    if not np.isfinite(best_val) or best_val >= score - tol:
        return p, c, score, False

    S = S0 + best_x - old
    if not np.isfinite(S) or S <= 1e-14:
        return p, c, score, False
    S2 = S * S

    p_new = p / S
    p_new[i] = best_x / S

    c_new = c / S2
    aff_best = c_k + coeff * best_x + const
    aff_best[mid] += best_x * best_x
    c_new[ks] = aff_best / S2

    np.clip(p_new, 0.0, None, out=p_new)
    np.clip(c_new, 0.0, None, out=c_new)

    snew = float(np.sum(p_new))
    if not np.isfinite(snew) or snew <= 0.0:
        return p, c, score, False
    p_new = p_new / snew
    c_new = c_new / (snew * snew)

    score_new = float(2.0 * n * max(0.0, float(np.max(c_new)))) if c_new.size else float("inf")
    if not np.isfinite(score_new) or score_new >= score - tol:
        return p, c, score, False

    return p_new, c_new, score_new, True


def _coordinate_optimize(p, deadline, rng, grid, refine_count, max_perturb=3):
    p = _normalize(p).copy()
    n = int(p.size)
    c = _convolve(p, p)
    score = float(2.0 * n * max(0.0, float(np.max(c)))) if c.size else float("inf")

    best_p = p.copy()
    best_c = c.copy()
    best_score = score

    stagnant = 0
    cycles_since_best = 0
    perts = 0
    noise_scales = (0.45, 0.25, 0.12, 0.06)

    while time.monotonic() < deadline - 0.4:
        if n <= 20000:
            order = rng.permutation(n)
        else:
            order = np.arange(n, dtype=np.int64)

        improved = False
        for i in order:
            if time.monotonic() > deadline - 0.2:
                break
            p, c, score, imp = _coordinate_update(p, c, score, int(i), grid, refine_count)
            if imp:
                improved = True

        # Re-sync exact convolution after a pass.
        c = _convolve(p, p)
        np.clip(c, 0.0, None, out=c)
        score = float(2.0 * n * max(0.0, float(np.max(c)))) if c.size else float("inf")

        tol = 1e-12 * max(1.0, best_score)
        if np.isfinite(score) and score < best_score - tol:
            best_score = score
            best_p = p.copy()
            best_c = c.copy()
            cycles_since_best = 0
        else:
            cycles_since_best += 1

        if improved:
            stagnant = 0
        else:
            stagnant += 1
            if score > best_score * 1.25 + 1e-8:
                p = best_p.copy()
                c = best_c.copy()
                score = best_score

        if cycles_since_best >= 5 and score > best_score * 1.02:
            p = best_p.copy()
            c = best_c.copy()
            score = best_score
            cycles_since_best = 0

        if stagnant >= 2 and perts < int(max_perturb):
            scale = noise_scales[min(perts, len(noise_scales) - 1)]
            p = best_p * np.exp(rng.normal(0.0, float(scale), n))
            p = _normalize(p)
            c = _convolve(p, p)
            np.clip(c, 0.0, None, out=c)
            score = float(2.0 * n * max(0.0, float(np.max(c)))) if c.size else float("inf")
            if np.isfinite(score) and score < best_score - 1e-10:
                best_score = score
                best_p = p.copy()
                best_c = c.copy()
                cycles_since_best = 0
            perts += 1
            stagnant = 0
        elif stagnant >= 8:
            break

    return best_p, best_c, best_score

# Composite refiner

def _refine_sequence(a, end_time, rng, smooth=True, use_lp=True, coord=True):
    a = _normalize(a).copy()
    n = int(a.size)
    best = a.copy()
    sc = _score(a)
    now = time.monotonic()
    total = max(0.0, float(end_time) - now)

    if smooth and total > 25.0:
        smooth_end = min(float(end_time), now + max(15.0, total * 0.45))

        if n <= 128:
            gammas = (5.0, 20.0, 80.0, 200.0)
        elif n <= 512:
            gammas = (5.0, 20.0, 80.0)
        else:
            gammas = (5.0, 20.0)

        for k, g in enumerate(gammas):
            if time.monotonic() >= smooth_end:
                break
            remain = max(1.0, float(smooth_end) - time.monotonic())
            stage_end = min(float(smooth_end), time.monotonic() + remain / max(1, len(gammas) - k))
            stage_end = max(stage_end, time.monotonic() + 3.0)
            beta = float(g) * float(n)
            a2, sc2 = _smooth_step(best, beta, stage_end, rng)
            if np.isfinite(sc2) and sc2 < sc:
                best = np.asarray(a2, dtype=np.float64).copy()
                sc = float(sc2)

    # L-BFGS for short/medium lengths.
    if n <= 192 and time.monotonic() < float(end_time) - 35.0:
        lb_end = min(float(end_time), time.monotonic() + min(60.0, 0.12 * total))
        beta_l = 20.0 * float(n)
        a2, sc2 = _lbfgs_optimize(best, beta_l, lb_end, maxiter=70)
        if np.isfinite(sc2) and sc2 < sc:
            best = np.asarray(a2, dtype=np.float64).copy()
            sc = float(sc2)

    # LP polish for sizes where the LP is tractable.
    if use_lp and n <= 512 and time.monotonic() < float(end_time) - 25.0:
        lp_end = min(float(end_time), time.monotonic() + min(75.0, 0.2 * total))
        a2, sc2 = _lp_polish(best, lp_end, max_iters=2, max_n=512)
        if np.isfinite(sc2) and sc2 < sc:
            best = np.asarray(a2, dtype=np.float64).copy()
            sc = float(sc2)

    # Coordinate polish with remaining time.
    if coord and time.monotonic() < float(end_time) - 5.0:
        grid, ref = _coord_params(n)
        max_perturb = 2 if n <= 512 else 1
        a2, _, sc2 = _coordinate_optimize(best, float(end_time), rng, grid, ref, max_perturb=max_perturb)
        if np.isfinite(sc2) and sc2 < sc:
            best = np.asarray(a2, dtype=np.float64).copy()
            sc = float(sc2)

    return best, sc

# Final evaluation / selection
# -----------------------------

def _external_score(seq):
    try:
        ev = globals().get("evaluate_sequence")
        if ev is None:
            return float("inf")
        return float(ev([float(x) for x in np.asarray(seq, dtype=np.float64).ravel()]))
    except Exception:
        return float("inf")


def _finalize_choice(pool):
    if not pool:
        return [1.0]

    pool = sorted(pool, key=_pool_key)
    best_seq = None
    best_sc = float("inf")
    best_len = 0
    seen = set()

    # Prefer authoritative evaluator on the best candidates, with tie-breaking
    # toward longer sequences when scores are essentially equal.
    for item in pool[:18]:
        seq = item[2]
        L = int(item[1])
        if L in seen:
            continue
        seen.add(L)
        sc = _external_score(seq)
        if not np.isfinite(sc):
            sc = float(item[0])
        if not np.isfinite(sc):
            continue
        if sc < best_sc - 1e-12:
            best_sc = float(sc)
            best_seq = np.asarray(seq, dtype=np.float64).copy()
            best_len = L
        elif sc <= best_sc + 1e-9 and L > best_len:
            best_seq = np.asarray(seq, dtype=np.float64).copy()
            best_len = L

    if best_seq is None:
        best_seq = np.asarray(pool[0][2], dtype=np.float64).copy()

    out = _normalize(best_seq)
    return [float(x) for x in out]

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Returns the best nonnegative sequence found under the requested time budget.
    """
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    budget = max(5.0, budget)

    deadline = time.monotonic() + max(1.0, budget - 30.0)
    rng = np.random.default_rng(int(seed))

    pool = []
    limit_pool = 40

    # Seed from previously found sequence if present.
    gseq = globals().get("height_sequence_1", None)
    has_global = False
    global_arr = None
    if gseq is not None:
        try:
            global_arr = _clean_array(gseq)
            sg = _score(global_arr)
            if np.isfinite(sg):
                _pool_add(pool, global_arr, sg, limit_pool)
                has_global = True
        except Exception:
            pass

    # Guarantee a valid candidate.
    if not pool:
        p = np.ones(64, dtype=np.float64)
        _pool_add(pool, p, _score(p), limit_pool)

    print(f"[initialize] pool best score = {pool[0][0]:.10f} len={pool[0][1]}", flush=True)

    # Build initial prototype pool.
    proto_lengths = (64, 128, 256, 512, 1024, 2048)
    init_reserve = max(120.0, 0.35 * (deadline - time.monotonic()))

    for n in proto_lengths:
        if time.monotonic() >= deadline - init_reserve:
            break
        if has_global and global_arr is not None:
            try:
                d = _downsample(global_arr, n)
                sd = _score(d)
                if np.isfinite(sd):
                    _pool_add(pool, d, sd, limit_pool)
            except Exception:
                pass

        codes = (5, 3, 1, 0, 8, 2)
        for code in codes[:2 if n >= 1024 else 3]:
            if time.monotonic() >= deadline - init_reserve:
                break
            try:
                p = _initial(n, code, rng)
                sp = _score(p)
                if np.isfinite(sp):
                    _pool_add(pool, p, sp, limit_pool)
            except Exception:
                pass

    print(f"[init done] best score = {pool[0][0]:.10f} len={pool[0][1]}", flush=True)

    # Staged refinement of prototypes.
    stage_plan = (
        # length, count, slice_seconds
        (64, 2, 25),
        (128, 3, 40),
        (256, 3, 70),
        (512, 3, 100),
        (1024, 2, 130),
        (2048, 2, 160),
    )

    refine_reserve = max(260.0, 0.28 * (deadline - time.monotonic()))

    for n, count, slice_s in stage_plan:
        if time.monotonic() >= deadline - refine_reserve:
            break
        cands = [item for item in pool if int(item[1]) == int(n)]
        if not cands:
            continue
        cands.sort(key=_pool_key)

        for item in cands[:int(count)]:
            if time.monotonic() >= deadline - refine_reserve:
                break
            end = min(deadline - refine_reserve, time.monotonic() + float(slice_s))
            if end <= time.monotonic() + 10.0:
                continue
            try:
                a2, sc2 = _refine_sequence(
                    item[2], end, rng,
                    smooth=True,
                    use_lp=(int(n) <= 512),
                    coord=True
                )
                if np.isfinite(sc2):
                    _pool_add(pool, a2, sc2, limit_pool)
                    print(f"[stage n={n}] {pool[0][0]:.10f} len={pool[0][1]}", flush=True)
            except Exception as ex:
                print(f"[stage error n={n}] {ex}", flush=True)

    # Create final-length candidates.
    final_lengths = (3809, 4096)

    def _add_final_start(p):
        try:
            sc = _score(p)
            if np.isfinite(sc):
                _pool_add(pool, p, sc, limit_pool)
        except Exception:
            pass

    for FL in final_lengths:
        if time.monotonic() >= deadline - 120.0:
            break
        # From global if available.
        if has_global and global_arr is not None:
            _add_final_start(_downsample(global_arr, FL))

        # From best prototypes.
        for item in pool[:8]:
            if time.monotonic() >= deadline - 120.0:
                break
            if int(item[1]) <= FL:
                try:
                    _add_final_start(_expand_to_length(item[2], FL))
                except Exception:
                    pass

    # Polish the best final candidate(s).
    finals = [item for item in pool if int(item[1]) in final_lengths]
    finals.sort(key=_pool_key)
    for item in finals[:1]:
        if time.monotonic() >= deadline - 60.0:
            break
        remaining_final = max(60.0, float(deadline) - time.monotonic() - 30.0)
        end = min(deadline - 30.0, time.monotonic() + remaining_final * 0.7)
        try:
            a2, sc2 = _refine_sequence(
                item[2], end, rng,
                smooth=True,
                use_lp=False,
                coord=True
            )
            if np.isfinite(sc2):
                _pool_add(pool, a2, sc2, limit_pool)
                print(f"[final polish] {pool[0][0]:.10f} len={pool[0][1]}", flush=True)
        except Exception as ex:
            print(f"[final polish error] {ex}", flush=True)

    # Iterative polishing / diversification until close to deadline.
    stagnant = 0
    while time.monotonic() < deadline - 35.0:
        best_before = pool[0][0]
        improved = False

        # Coordinate polish top candidates with small slices.
        for item in pool[:5]:
            if time.monotonic() >= deadline - 35.0:
                break
            n = int(item[1])
            if n > 3000:
                slice_s = 55.0
            elif n > 1000:
                slice_s = 38.0
            elif n > 200:
                slice_s = 25.0
            else:
                slice_s = 18.0
            end = min(deadline - 35.0, time.monotonic() + slice_s)
            grid, ref = _coord_params(n)
            try:
                a2, _, sc2 = _coordinate_optimize(item[2], end, rng, grid, ref, max_perturb=1)
                if np.isfinite(sc2):
                    _pool_add(pool, a2, sc2, limit_pool)
                    if sc2 < item[0] - 1e-11:
                        improved = True
            except Exception:
                pass

        if improved and pool[0][0] < best_before - 1e-12:
            stagnant = 0
            print(f"[loop improved] {pool[0][0]:.10f} len={pool[0][1]}", flush=True)
        else:
            stagnant += 1

            # Try an LP polish on a short candidate.
            if stagnant >= 2 and time.monotonic() < deadline - 80.0:
                for item in pool[:8]:
                    if int(item[1]) <= 512:
                        end = min(deadline - 80.0, time.monotonic() + 45.0)
                        try:
                            a2, sc2 = _lp_polish(item[2], end, max_iters=2, max_n=512)
                            if np.isfinite(sc2):
                                _pool_add(pool, a2, sc2, limit_pool)
                                if sc2 < item[0] - 1e-11:
                                    improved = True
                        except Exception:
                            pass
                        break

            # Try a fresh smooth step on a medium candidate.
            if not improved and time.monotonic() < deadline - 140.0:
                nwant = int(rng.choice((128, 256, 512)))
                src = None
                for item in pool:
                    if int(item[1]) <= 2048:
                        src = item[2]
                        break
                if src is not None:
                    a0 = _downsample(src, nwant)
                    sc0 = _score(a0)
                    end = min(deadline - 140.0, time.monotonic() + 90.0)
                    try:
                        a2, sc2 = _smooth_step(a0, 100.0 * float(nwant), end, rng)
                        if np.isfinite(sc2):
                            _pool_add(pool, a2, sc2, limit_pool)
                            if sc2 < best_before - 1e-11:
                                improved = True
                    except Exception:
                        pass

            # Perturb top candidates to escape plateaus.
            if stagnant >= 3:
                for item in pool[:4]:
                    n = int(item[1])
                    try:
                        pert = np.asarray(item[2], dtype=np.float64) * np.exp(
                            rng.normal(0.0, 0.12, n)
                        )
                        sp = _score(pert)
                        if np.isfinite(sp):
                            _pool_add(pool, pert, sp, limit_pool)
                    except Exception:
                        pass
                stagnant = 0

            if stagnant >= 10:
                break

    out = _finalize_choice(pool)
    # Final sanity: ensure list of floats, nonnegative finite values.
    out_arr = np.asarray(out, dtype=np.float64)
    out_arr = np.where(np.isfinite(out_arr), out_arr, 0.0)
    np.maximum(out_arr, 0.0, out=out_arr)
    s = float(np.sum(out_arr))
    if not np.isfinite(s) or s <= 0.0:
        out_arr = np.ones(64, dtype=np.float64)
        s = float(np.sum(out_arr))
    out_arr = out_arr / s
    return [float(x) for x in out_arr]
