def binary_swap_update(c, b, i, j, n):
    """Apply swap: b[i] 1->0, b[j] 0->1. Returns delta applied to c."""
    # Effect of removing 1 at position i:
    # c[k] -= 2*b[k-i] for k in [i, i+n-1] (with bounds)
    # c[2*i] -= 1 (self-pair)
    lo1 = max(0, i)
    hi1 = min(2*n-1, i+n-1)
    for k in range(lo1, hi1+1):
        c[k] -= 2.0 * b[k-i]
    c[2*i] -= 1.0  # Wait, this is already accounted for in the sum above?
    
    # Actually, let me be more careful.
    # c[k] = sum_{a=0}^{n-1} b[a]*b[k-a]
    # When b[i] changes from 1 to 0:
    # new_c[k] = sum_{a != i} b[a]*b[k-a]
    # = c[k] - b[i]*b[k-i] - b[k-i]*b[i] + (b[i]*b[k-i] if k-i == i else 0)
    # = c[k] - 2*b[k-i] + (1 if k == 2*i else 0)
    # (assuming b[i] was 1, so b[i]*b[k-i] = b[k-i])

    #     c[2*i] += 1 (to correct the double-counting of the self-pair)
    
    # Similarly for adding 1 at j:
    # c[k] += 2*b[k-j] for k where 0 <= k-j <= n-1
    # c[2*j] += 1
    
    # But wait, the order matters! If we first remove i, then add j,
    # the b used in the "add j" step should be the one after removing i.
    # Since i != j, this means b[i]=0, b[j]=1 in the "add" step.
    # The formula c[k] += 2*b_after[k-j] uses b_after where b[i]=0, b[j]=1.
    # But b_after[k-j] = b_before[k-j] except when k-j == i (then it's 0) or k-j == j (then it's 1).
    # Since k-j == i means k = i+j, and k-j == j means k = 2j.
    # So the correction is: at k = i+j, we use 0 instead of b_before[i] = 1, so we subtract 1 extra.
    # At k = 2j, we use 1 instead of b_before[j] = 0, so we add 1 extra.
    # But c[2j] += 1 is already there, so the net at k=2j is +2 (which is correct: we're adding the self-pair).
    # At k = i+j: we'd normally add 2*b_before[i+j-j] = 2*b_before[i] = 2, but since b[i] is now 0,
    # we should only add 0. So we need to subtract 2 at k = i+j.

    # Compute the total delta as:
    # delta_c = 2*(b_after * delta_j) - 2*(b_before * delta_i)

    # b_after = b_before - delta_i + delta_j (i.e., b[i]=0, b[j]=1)
    
    # Actually, the exact formula is:
    # new_c = (b_before - delta_i + delta_j) * (b_before - delta_i + delta_j)
    # = b_before*b_before - 2*b_before*delta_i + delta_i*delta_i + 2*b_before*delta_j - 2*delta_i*delta_j + delta_j*delta_j
    # = c - 2*(b_before shifted by i) + delta_{2i} + 2*(b_before shifted by j) - 2*delta_{i+j} + delta_{2j}
    
    # Where "b_before shifted by i" means the vector v where v[k] = b_before[k-i] (0 if out of bounds).
    
    # So: delta_c[k] = -2*b_before[k-i] + 2*b_before[k-j] + (1 if k==2i) - (2 if k==i+j) + (1 if k==2j)
    # (with b_before[k-*] = 0 if k-* not in [0, n-1])

def optimize_binary_v2(a0, deadline, seed=0):
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = int(a0.size)
    m = max(2, min(n - 2, int(round(0.5 * n))))
    
    # Initialize
    order = np.argsort(a0, kind="stable")
    b = np.zeros(n)
    b[order[-m:]] = 1.0
    
    # Initial convolution
    N_cv = next_pow2(2 * n - 1)
    c = np.fft.irfft(np.fft.rfft(b, N_cv) ** 2, N_cv)[:2*n-1]
    c = np.rint(c).astype(np.float64)
    
    best_b = b.copy()
    best_max = float(np.max(c))
    best_score = 2.0 * n * best_max / (m * m)
    
    rng = np.random.default_rng(seed)
    temp = 0.05
    stale = 0
    one_set = set(np.nonzero(b > 0.5)[0].tolist())
    
    while time.monotonic() < deadline:
        cmax = float(np.max(c))
        score = 2.0 * n * cmax / (m * m)
        if score < best_score - 1e-12:
            best_score = score
            best_max = cmax
            best_b = b.copy()
        
        if cmax <= float(m):  # Optimal for this weight
            break
        
        if len(one_set) == 0 or len(one_set) == n:
            break
        
        i = int(rng.choice(list(one_set)))
        j = int(rng.integers(0, n))
        attempts = 0
        while j in one_set and attempts < 10:
            j = int(rng.integers(0, n))
            attempts += 1
        if j in one_set:
            continue
        
        # Apply swap using incremental update
        # Save old values in affected range
        lo = min(i, j)
        hi = max(i, j) + n - 1
        lo = max(0, lo)
        hi = min(2*n-2, hi)
        
        # Compute delta for k in [lo, hi]
        # delta[k] = -2*b[k-i] + 2*b[k-j] + (1 if k==2i) - (2 if k==i+j) + (1 if k==2j)
        # where b[k-*] = 0 if k-* not in [0, n-1]
        
        old_c_segment = c[lo:hi+1].copy()
        
        for k in range(lo, hi+1):
            delta = 0.0
            ki = k - i
            kj = k - j
            if 0 <= ki < n:
                delta -= 2.0 * b[ki]
            if 0 <= kj < n:
                delta += 2.0 * b[kj]
            if k == 2 * i:
                delta += 1.0
            if k == i + j:
                delta -= 2.0
            if k == 2 * j:
                delta += 1.0
            c[k] += delta
        
        b[i] = 0.0
        b[j] = 1.0
        one_set.remove(i)
        one_set.add(j)
        
        new_cmax = float(np.max(c))
        new_score = 2.0 * n * new_cmax / (m * m)
        
        accept = False
        if new_score < score - 1e-12:
            accept = True
        elif new_score - score < 1.0:
            if rng.random() < np.exp(-(new_score - score) / max(temp, 1e-10)):
                accept = True
        
        if accept:
            stale = 0
            temp = max(0.001, temp * 0.998)
        else:
            # Revert
            c[lo:hi+1] = old_c_segment
            b[i] = 1.0
            b[j] = 0.0
            one_set.add(i)
            one_set.discard(j)
            stale += 1
            temp = max(0.0005, temp * 0.97)
            if stale % 200 == 0:
                temp = min(0.1, temp * 2.0)
                stale = 0
            if stale > 500:
                break
    
    return normalize_positive(best_b, floor=0.0), best_score

Let me now think about the continuous optimization more carefully.

**New continuous optimization approach: "Peak shaving" with subgradient**

The idea:
1. Compute c = a*a
2. Find the K largest entries of c
3. For each peak at position k, compute the "contribution" of each a[i] to c[k]:
   - a[i] contributes a[i]*a[k-i] to c[k]
   - So the "pressure" on a[i] from peak k is a[k-i]
4. Total pressure on a[i] = sum_{k in peaks} a[k-i] * (c[k] - threshold)
5. Update: reduce a[i] where pressure is high, increase where it's low
6. Renormalize

This is essentially what the mirror descent does, but I'll make it more aggressive:
- Use a "hard" threshold (only the top K peaks contribute)
- Use a larger step size
- Use momentum

Let me also think about a completely different approach: **alternating minimization over the active set**.

At the optimum, the KKT conditions say:
- For the active constraints (where c[k] = max_c): the gradient of c[k] is in the span of the constraint gradients
- For inactive constraints: c[k] < max_c

So the algorithm:
1. Fix the active set S = {k : c[k] >= max_c - epsilon}
2. Solve: minimize t subject to c[k] = t for k in S, a >= 0, sum(a) = 1
   - This is still non-convex, but we can use Newton's method or gradient descent
3. Update S
4. Repeat

For step 2, the constraint c[k] = t for k in S means:
sum_i a[i]*a[k-i] = t for each k in S.

This is a system of quadratic equations. It's hard to solve exactly, but we can use iterative methods.

Actually, let me try a different approach that might be more practical:

**Approach: "Water-filling" on the peaks**

1. Compute c = a*a, find max_c
2. Set target t = max_c - delta (where delta is a small amount)
3. For each k where c[k] > t:
   - We need to reduce c[k] by (c[k] - t)
   - c[k] = sum_i a[i]*a[k-i]
   - To reduce c[k], we can reduce a[i] for the i that contribute most
   - The contribution of a[i] to c[k] is a[i]*a[k-i]
   - So the "most costly" i's are those where a[i]*a[k-i] is large
   - Reduce a at those positions
4. Renormalize
5. Repeat

This is a greedy approach that might work well in practice.

Multi-stage optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses vectorized binary cost-array search, subgradient continuous optimization
with momentum, and adaptive peak-shaving.

def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def normalize_positive(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, floor, None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def convolve_seq(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if N is None:
        N = next_pow2(2 * n - 1)
    else:
        need = next_pow2(2 * n - 1)
        if N < need:
            N = need
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]


def internal_score(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.inf
    a = a / s
    c = convolve_seq(a, N)
    return float(2.0 * n * float(np.max(c)))

def resample_seq(a, n):
    a = normalize_positive(a, floor=0.0)
    old_n = a.size
    n = max(2, int(n))
    if old_n == n:
        return a.copy()
    if old_n == 1:
        return np.full(n, 1.0 / n)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, n)
    b = np.interp(new_x, old_x, a)
    return normalize_positive(b, floor=1e-12)


def make_sine_sequence(n, alpha):
    n = max(2, int(n))
    x = (np.arange(n, dtype=np.float64) - 0.5 * (n - 1)) / max(1.0, 0.5 * n)
    base = np.ones(n, dtype=np.float64) + float(alpha) * np.sin(np.pi * x)
    return normalize_positive(base, floor=1e-12)


def make_gaussian_sequence(n, mu, sigma):
    n = max(2, int(n))
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    sigma = max(1e-3, float(sigma))
    return normalize_positive(np.exp(-((x - mu) ** 2) / (2 * sigma * sigma)), floor=1e-12)


def generate_initial_sequences(n, seed, max_count=10):
    n = max(8, int(n))
    rng = np.random.default_rng(seed)
    seqs = [make_uniform(n)]
    for alpha in (0.3, -0.3, 0.5, -0.5, 0.7, 0.9):
        seqs.append(make_sine_sequence(n, alpha))
    for mu in (0.3, 0.5, 0.7):
        for sig in (0.15, 0.25, 0.35):
            seqs.append(make_gaussian_sequence(n, mu, sig))
    for _ in range(5):

        vals = np.ones(n)
        for m in range(1, 8):
            amp = rng.uniform(0.05, 0.4) / (m ** 0.6)
            vals += amp * np.sin(2 * np.pi * m * x + rng.uniform(0, 2 * np.pi))

        rng.shuffle(rem)
        keep.extend(rem[:max(0, max_count - len(keep))])
        seqs = [seqs[i] for i in keep]
    return seqs


def optimize_binary_v2(a0, deadline, seed=0, weight_ratio=0.5):
    """Binary cost-array search with vectorized incremental updates."""
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = int(a0.size)
    m = max(4, min(n - 4, int(round(weight_ratio * n))))

    order = np.argsort(a0, kind="stable")
    b = np.zeros(n, dtype=np.float64)
    b[order[-m:]] = 1.0

    N_cv = next_pow2(2 * n - 1)
    c = np.fft.irfft(np.fft.rfft(b, N_cv) ** 2, N_cv)[:2 * n - 1]
    c = np.rint(c).astype(np.float64)

    best_b = b.copy()
    best_max = float(np.max(c))
    best_score = 2.0 * n * best_max / (m * m)

    rng = np.random.default_rng(seed)
    temp = 0.06
    stale = 0
    it = 0

    while time.monotonic() < deadline:
        cmax = float(np.max(c))
        score = 2.0 * n * cmax / (m * m)
        if score < best_score - 1e-12:
            best_score = score
            best_max = cmax
            best_b = b.copy()

        if cmax <= float(m):
            break

        one_idx = np.nonzero(b > 0.5)[0]
        zero_idx = np.nonzero(b < 0.5)[0]
        if one_idx.size == 0 or zero_idx.size == 0:
            break

        i = int(one_idx[rng.integers(0, one_idx.size)])
        j = int(zero_idx[rng.integers(0, zero_idx.size)])
        if i == j:
            continue

        delta = np.zeros(2 * n - 1, dtype=np.float64)
        delta[i:i + n] += -2.0 * b
        delta[j:j + n] += 2.0 * b
        delta[2 * i] += 1.0
        delta[i + j] += -2.0
        delta[2 * j] += 1.0

        old_segment = c.copy()
        c += delta
        b[i] = 0.0
        b[j] = 1.0

        new_max = float(np.max(c))
        new_score = 2.0 * n * new_max / (m * m)

        accept = False
        if new_score < score - 1e-12:
            accept = True
        else:
            d = new_score - score
            if d < 2.0:
                if rng.random() < np.exp(-d / max(temp, 1e-10)):
                    accept = True

        if accept:
            stale = 0
            temp = max(0.002, temp * 0.999)
        else:
            c[:] = old_segment
            b[i] = 1.0
            b[j] = 0.0
            stale += 1
            temp = max(0.001, temp * 0.98)
            if stale % 150 == 0:
                temp = min(0.12, temp * 2.5)
                stale = 0
            if stale > 400:
                break
        it += 1

    return normalize_positive(best_b, floor=0.0), best_score

def optimize_mirror_v2(a, deadline, seed=0):

    it = 0

            break

        K = max(3, n // 50)
        top_k = np.argpartition(c, -K)[-K:]
        c_top = c[top_k]
        cmax_top = float(np.max(c_top))
        tau = beta * cmax_top
        z = (c_top - cmax_top) / max(tau, 1e-10)
        z = np.clip(z, -40.0, 0.0)
        w_top = np.exp(z) / float(np.exp(z).sum())

def local_polish(a, deadline, seed=1):
    """Fine-grained local polish with small block perturbations."""
    a = normalize_positive(a, floor=1e-15)
    n = a.size
    N_cv = next_pow2(3 * n - 2)
    best_a = a.copy()
    best_score = internal_score(a, N_cv)

    loga = np.log(a + 1e-20)
    loga -= float(np.max(loga))
    rng = np.random.default_rng(seed)
    sigma = 0.12
    stale = 0

    while time.monotonic() < deadline:
        width = int(rng.integers(max(2, n // 2000), max(3, n // 15) + 1))
        width = min(width, n // 2)
        start = int(rng.integers(0, n - width + 1))

        t = np.linspace(0.0, 1.0, width)
        env = np.exp(-4.0 * (t - 0.5) ** 2)
        d = sigma * env * rng.normal(size=width)
        d -= d.mean()
        d = np.clip(d, -2.0, 2.0)

        old = loga[start:start + width].copy()
        loga[start:start + width] += d
        a_try = np.exp(np.clip(loga - float(np.max(loga)), -60.0, 0.0))
        a_try = np.maximum(a_try, 1e-300)
        a_try = a_try / float(a_try.sum())
        a_try = normalize_positive(a_try, floor=1e-15)

        score_try = internal_score(a_try, N_cv)
        if np.isfinite(score_try) and score_try < best_score - 1e-12:
            best_a = a_try.copy()
            best_score = score_try
            loga = np.log(a_try + 1e-20)
            loga -= float(np.max(loga))
            stale = 0
            sigma = min(0.3, sigma * 1.03)
        else:
            loga[start:start + width] = old
            stale += 1
            sigma = max(0.02, sigma * 0.97)
            if stale > 300:
                idx = int(rng.integers(0, n))
                old_v = loga[idx]
                loga[idx] += 1.5 * sigma * rng.choice([-1, 1])
                a_try = np.exp(np.clip(loga - float(np.max(loga)), -60.0, 0.0))
                a_try = np.maximum(a_try, 1e-300)
                a_try = a_try / float(a_try.sum())

                score_try = internal_score(a_try, N_cv)
                if np.isfinite(score_try) and score_try < best_score:
                    best_a = a_try.copy()
                    best_score = score_try
                    loga = np.log(a_try + 1e-20)
                    loga -= float(np.max(loga))
                else:
                    loga[idx] = old_v
                stale = 0
                sigma = 0.12

def propose_candidate(seed=42, budget_s=2370, **kwargs):
    start = time.monotonic()
    deadline = start + max(120.0, float(budget_s))

    best_a = make_uniform(2048)
    best_score = internal_score(best_a)
    print("initial uniform score = {:.9f}".format(best_score), flush=True)

    # Try given sequence
    given = get_given_sequence()
    if given is not None:
        sc = internal_score(given)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = given.copy()
            print("given sequence score = {:.9f}".format(best_score), flush=True)

    # Quick analytical inits
    for n in (1024, 2048, 4096):
        if time.monotonic() > deadline - 2200:
            break
        inits = generate_initial_sequences(n, seed + n, max_count=6)
        for a0 in inits:
            sc = internal_score(a0)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a0.copy()
                print("analytic init n={} score={:.9f}".format(n, best_score), flush=True)

    # --- Stage 1: Binary search at multiple sizes ---
    print("=== Stage 1: Binary search ===", flush=True)
    binary_configs = [
        (2048, 0.5, 3),
        (2048, 0.45, 2),
        (4096, 0.5, 3),
        (4096, 0.45, 2),
        (8192, 0.5, 2),
    ]

    for n, wr, n_starts in binary_configs:
        if time.monotonic() > deadline - 1200:
            break
        time_per_start = min(120.0, (deadline - time.monotonic() - 1200) / max(1, n_starts))
        if time_per_start < 30:
            continue

        a_init = resample_seq(best_a, n)
        for rs in range(n_starts):
            if time.monotonic() > deadline - 1150:
                break
            t_dead = time.monotonic() + time_per_start
            if rs == 0:
                a0 = a_init
            elif rs == 1:
                a0 = np.random.default_rng(seed + n + rs).random(n)
            else:
                # Structured init: quadratic residue or arithmetic progression
                arr = np.zeros(n)
                step = rs + 2
                for ix in range(n):
                    arr[(ix * step) % n] = 1.0
                a0 = arr
            a1, sc1 = optimize_binary_v2(a0, t_dead, seed=seed + 10000 + n * 100 + rs, weight_ratio=wr)
            if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
                best_score = sc1
                best_a = a1.copy()
                print("binary n={} wr={:.2f} rs={} score={:.9f}".format(n, wr, rs, best_score), flush=True)

    # --- Stage 2: Continuous optimization ---
    print("=== Stage 2: Continuous optimization ===", flush=True)
    for n in (2048, 4096):
        if time.monotonic() > deadline - 600:
            break
        allow = min(400.0, deadline - time.monotonic() - 500)
        if allow < 60:
            continue
        a0 = resample_seq(best_a, n)
        # Subgradient
        t_dead = time.monotonic() + allow * 0.5
        a1, sc1 = optimize_subgradient(a0, t_dead, seed=seed + 20000 + n, verbose=True)
        if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()
            print("subgrad n={} score={:.9f}".format(n, best_score), flush=True)

        # Mirror descent
        if time.monotonic() < deadline - 500:
            t_dead2 = min(deadline - 480, time.monotonic() + allow * 0.5)
            a1, sc1 = optimize_mirror_v2(best_a, t_dead2, seed=seed + 30000 + n)
            if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
                best_score = sc1
                best_a = a1.copy()
                print("mirror n={} score={:.9f}".format(n, best_score), flush=True)

    # --- Stage 3: Second round binary from continuous solution ---
    print("=== Stage 3: Refine binary from continuous ===", flush=True)
    for n in (2048, 4096):
        if time.monotonic() > deadline - 300:
            break
        allow = min(200.0, deadline - time.monotonic() - 250)
        if allow < 40:
            continue
        a0 = resample_seq(best_a, n)
        a1, sc1 = optimize_binary_v2(a0, time.monotonic() + allow, seed=seed + 40000 + n, weight_ratio=0.5)
        if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()
            print("binary2 n={} score={:.9f}".format(n, best_score), flush=True)

    # --- Stage 4: Final continuous optimization ---
    print("=== Stage 4: Final continuous ===", flush=True)
    if time.monotonic() < deadline - 60:
        n_final = best_a.size
        a1, sc1 = optimize_subgradient(best_a, deadline - 40, seed=seed + 50000, verbose=True)
        if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()
            print("final subgrad score={:.9f}".format(best_score), flush=True)

    # --- Stage 5: Polish ---
    if time.monotonic() < deadline - 20:
        a1, sc1 = local_polish(best_a, deadline - 10, seed=99999)
        if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()
            print("polish score={:.9f}".format(best_score), flush=True)

    # Finalize
    best_a = normalize_positive(best_a, floor=0.0)
    out_list = [float(x) for x in best_a]
    try:
        final_eval = float(evaluate_sequence(out_list))
    except Exception:
        final_eval = float("inf")

    if not np.isfinite(final_eval):
        best_a = make_uniform(1024)
        out_list = [float(x) for x in best_a]
        final_eval = 2.0

    print("FINAL: internal={:.12f}, evaluated={:.12f}, n={}".format(
        best_score, final_eval, len(out_list)), flush=True)
    return out_list
