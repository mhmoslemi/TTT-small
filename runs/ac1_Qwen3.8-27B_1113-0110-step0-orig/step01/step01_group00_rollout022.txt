"""
Step function optimization: minimize J(a) = 2*n*max(conv(a,a))/sum(a)^2.
Uses enhanced alternating LP with smart blending, diverse initializations,
random restarts, and coordinate descent polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _fft_convolve(a):
    """Fast full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]


def _score(a):
    """Compute J(a) = 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)


def _build_conv_matrix(f):
    """Build convolution matrix C where (C @ g)[k] = (f*g)[k]."""
    n = len(f)
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = f
    return C


def _lp_solve(f, M, C=None):
    """Solve max sum(g) s.t. C_f @ g <= M, g >= 0."""
    n = len(f)
    if C is None:
        C = _build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 30})
        if result.status == 0:
            return result.x
    except Exception:
        pass
    return None


def _alternating_lp(a_init, max_iters, deadline, initial_blend=1.0, blend_decay=0.999,
                    restart_every=50, restart_noise=0.05):
    """Enhanced alternating LP with random restarts."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = _score(a)
    blend = initial_blend
    no_improve_count = 0
    rng = np.random.default_rng(hash(a.tobytes()) % 2**31)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        M = float(np.max(c))
        g = _lp_solve(a, M)
        if g is None:
            break
        sum_g = float(np.sum(g))
        if sum_g < 1e-12:
            break
        g = g / sum_g
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        sum_new = float(np.sum(a_new))
        if sum_new < 1e-12:
            break
        a_new = a_new / sum_new
        score = _score(a_new)
        if score < best_score - 1e-10:
            best_score = score
            best_a = a_new.copy()
            no_improve_count = 0
            if it % 50 == 0:
                print(f"      LP iter {it}: score={best_score:.8f}")
        else:
            no_improve_count += 1
        
        a = a_new
        blend = max(0.05, blend * blend_decay)
        
        # Random restart if stuck
        if no_improve_count >= restart_every:
            no_improve_count = 0
            noise = rng.normal(0, restart_noise, n)
            a = a + noise
            a = np.maximum(a, 0.0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            if it % 50 == 0:
                print(f"      LP iter {it}: restart (score={best_score:.8f})")

    return best_a, best_score


def _sensitivity(a, k):
    """Sensitivity vector for peak at lag k."""
    n = len(a)
    s = np.zeros(n)
    idx = k - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    s[mask] = 2.0 * a[idx[mask]]
    return s


def _subgradient(a_init, max_iters, n_peaks, deadline):
    """Subgradient with Polyak step size."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    best_a = a.copy()
    best_score = _score(a)
    # Upper bound for Polyak: use current score as gap estimate
    A = 0.0  # cumulative step size
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        max_c = float(np.max(c))
        score = 2.0 * n * max_c / S * S  # = 2n*max_c/S^2
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            if it % 100 == 0:
                print(f"    Subgrad iter {it}: score={best_score:.8f}")
        
        # Find top peaks
        if n_peaks >= len(c):
            top_k = np.arange(len(c))
        else:
            top_k = np.argpartition(c, -n_peaks)[-n_peaks:]
        
        # Compute subgradient
        s = np.zeros(n)
        for k in top_k:
            s += _sensitivity(a, int(k))
        s /= len(top_k)
        
        # Polyak step size
        gap = best_score - 1.0  # optimistic lower bound
        if gap < 0.001:
            gap = 0.001
        A += 1
        lr = gap / (np.dot(s, s) + 1e-10)
        lr = min(lr, 1.0)  # cap step size
        
        a = a - lr * s
        a = np.maximum(a, 0.0)
        S_new = float(np.sum(a))
        if S_new > 1e-10:
            a = a * S / S_new
        # Keep S constant
        S = float(np.sum(a))  # Should be same as before

    return best_a, best_score


def _coordinate_descent(a_init, max_sweeps, deadline):
    """Coordinate descent: optimize one coordinate at a time."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    best_score = _score(a)
    
    for sweep in range(max_sweps):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            if old_val < 1e-15:
                # Try setting to small values
                for factor in [0.001, 0.01, 0.05]:
                    a[i] = factor * np.mean(a)
                    s = _score(a)
                    if s < best_score:
                        best_score = s
                        improved = True
                a[i] = 0.0
            else:
                # Try scaling
                for factor in [0.5, 0.8, 1.2, 1.5, 2.0]:
                    a[i] = old_val * factor
                    s = _score(a)
                    if s < best_score:
                        best_score = s
                        improved = True
                # If none improved, revert
                current_s = _score(a)
                a[i] = old_val
        
        if not improved and sweep > 0:
            break
    
    return a, best_score

    s = float(np.sum(out))
    if s > 1e-10:

    return out

max sum(g) s.t. conv(f, g) <= M, g >= 0

But the convolution matrix is (2n-1) x n. For n=1000, that's 1999 x 1000 matrix. Building this matrix each time is expensive. Let me cache it.

Actually, I want to reconsider the problem from scratch.

We want to minimize J(a) = 2n * max_k (a*a)[k] / S^2 where (a*a)[k] = sum_j a[j]*a[k-j].

Key observation: the problem is homogeneous of degree 0 (scaling a doesn't change J). So we can fix S = 1 and minimize max_k (a*a)[k].

With S = 1: minimize max_k (a*a)[k] subject to sum(a) = 1, a >= 0.

This is non-convex because (a*a)[k] is quadratic. But we can use the following trick:

(a*a)[k] = sum_j a[j]*a[k-j] <= t

This is equivalent to: the vector a lies in the intersection of ellipsoidal half-spaces. Each constraint defines a non-convex set (a quadratic inequality).

1. Start with a feasible point
2. At each iteration, linearize the quadratic constraints around the current point
3. Solve the resulting LP
4. Update
