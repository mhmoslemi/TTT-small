```python
"""
Search for low-peak non-negative convolution sequences.

Strategy:
1. Start from the supplied previous construction if available.
2. Evaluate several explicit low-dimensional logarithmic shapes:
   - one-parameter phase-sine shapes,
   - centered antisymmetric sine bases,
   - low-degree log-polynomial shapes.
   These shapes are optimized by coarse grids and L-BFGS-B / DE polishing.
3. Use the strongest shapes as starts for a short alternating LP polish:
   fix f, solve max sum(g) subject to f*g <= max(f*f), blend, repeat.
4. Return the best finite sequence found.
"""

import numpy as np
import time
import math
from scipy.optimize import linprog, differential_evolution, minimize


def _fft_convolve(a):
    """Fast full linear convolution of a with itself."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]


def _score(a):
    """
    Homogeneous score used by evaluate_sequence:
        2 * n * max(conv(a, a)) / sum(a)^2
    We normalize internally to sum 1.
    """
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float("inf")
    a = np.clip(a, 0.0, 1000.0)
    a = a[np.isfinite(a)]
    if a.size == 0:
        return float("inf")
    s = float(a.sum())
    if s < 1e-12:
        return float("inf")
    a = a / s
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    return float(2.0 * a.size * max_c)


def _interpolate_sequence(src, n_target):
    """Interpolate a normalized non-negative sequence to a new length."""
    src = np.asarray(src, dtype=np.float64)
    src = np.clip(src, 0.0, None)
    src = src[np.isfinite(src)]
    n_src = src.size
    if n_target <= 0:
        return None
    if n_src == 0:
        out = np.ones(n_target) / n_target
        return out
    if n_src == 1:
        out = np.full(n_target, src[0] / n_target)
        s = out.sum()
        return out / s if s > 0 else out
    x_src = np.linspace(0.0, 1.0, n_src)
    x_tgt = np.linspace(0.0, 1.0, n_target)
    out = np.interp(x_tgt, x_src, src)
    out = np.maximum(out, 0.0)
    s = out.sum()
    if s > 1e-12:
        out = out / s
    else:
        out = np.ones(n_target) / n_target
    return out


def _make_p_phase1(params, n):
    """
    p_i proportional to exp(a * sin(2*pi*i/n + phi)).
    This is a simple skewed periodic log-density.
    """
    a = float(params[0]) if len(params) > 0 else 0.0
    phi = float(params[1]) if len(params) > 1 else 0.0
    t = np.arange(n, dtype=np.float64) / float(n)
    h = a * np.sin(2.0 * np.pi * t + phi)
    h -= h.mean()
    if h.size:
        hm = float(np.max(np.abs(h)))
        if hm > 20.0:
            h *= 20.0 / hm
    p = np.exp(h)
    s = p.sum()
    if not np.isfinite(s) or s <= 0:
        p = np.ones(n) / n
    else:
        p = p / s
    return p


def _make_p_center_sin(amps, n):
    """
    Centered antisymmetric sine basis:
        h_i = sum_k a_k sin(2*pi*k*(i - (n-1)/2)/n)
    Hence h_{n-1-i} = -h_i exactly, making the zero-lag/center convolution
    locally very small.
    """
    amps = np.asarray(amps, dtype=np.float64)
    harmonics = (1, 2, 3, 4)
    t = (np.arange(n, dtype=np.float64) - 0.5 * float(n - 1)) / float(n)
    h = np.zeros(n, dtype=np.float64)
    for idx, k in enumerate(harmonics):
        if idx < amps.size:
            h += float(amps[idx]) * np.sin(2.0 * np.pi * float(k) * t)
    h -= h.mean()
    if h.size:
        hm = float(np.max(np.abs(h)))
        if hm > 20.0:
            h *= 20.0 / hm
    p = np.exp(h)
    s = p.sum()
    if not np.isfinite(s) or s <= 0:
        p = np.ones(n) / n
    else:
        p = p / s
    return p


def _make_p_polylog(coeffs, n):
    """
    p_i proportional to exp(polynomial in centered x).
    Approximates skewed/nonperiodic smooth shapes, including exponentials.
    """
    c0 = float(coeffs[0]) if len(coeffs) > 0 else 0.0
    c1 = float(coeffs[1]) if len(coeffs) > 1 else 0.0
    c2 = float(coeffs[2]) if len(coeffs) > 2 else 0.0
    c3 = float(coeffs[3]) if len(coeffs) > 3 else 0.0
    x = np.arange(n, dtype=np.float64) / float(n)
    y = x - 0.5
    v = c0 + c1 * y + c2 * (y * y - 0.25) + c3 * (y * y * y - 0.4375 * y)
    v -= v.mean()
    if v.size:
        vm = float(np.max(np.abs(v)))
        if vm > 20.0:
            v *= 20.0 / vm
    p = np.exp(v)
    s = p.sum()
    if not np.isfinite(s) or s <= 0:
        p = np.ones(n) / n
    else:
        p = p / s
    return p


def _make_p_from_params(x, n, mode):
    x = np.asarray(x, dtype=np.float64)
    if mode == "phase1":
        if x.size < 2:
            x = np.array([0.8, 0.0])
        return _make_p_phase1(x, n)
    if mode == "center4":
        if x.size < 4:
            x2 = np.zeros(4, dtype=np.float64)
            x2[:min(4, x.size)] = x[:min(4, x.size)]
            x = x2
        return _make_p_center_sin(x, n)
    if mode == "poly3":
        if x.size < 4:
            x2 = np.zeros(4, dtype=np.float64)
            x2[:min(4, x.size)] = x[:min(4, x.size)]
            x = x2
        return _make_p_polylog(x, n)
    return _make_p_center_sin(x, n)


def _param_score(x, n, mode):
    """Objective for parametric candidate generation."""
    try:
        p = _make_p_from_params(x, n, mode)
        return _score(p)
    except Exception:
        return float("inf")


def _param_bounds(mode):
    if mode == "phase1":
        return 2, [(0.0, 2.0), (0.0, 2.0 * math.pi)]
    if mode == "center4":
        return 4, [(-1.8, 1.8)] * 4
    if mode == "poly3":
        return 4, [(-2.2, 2.2)] * 4
    return 0, []


def _polish_params(n, mode, x0):
    """Local L-BFGS-B polish of parametric candidate."""
    dim, bounds = _param_bounds(mode)
    if dim == 0:
        return None, float("inf")
    x0 = np.asarray(x0, dtype=np.float64)
    if x0.size != dim:
        y = np.zeros(dim, dtype=np.float64)
        m = min(dim, x0.size)
        y[:m] = x0[:m]
        x0 = y
    lo = np.array([b[0] for b in bounds], dtype=np.float64)
    hi = np.array([b[1] for b in bounds], dtype=np.float64)
    x0 = np.clip(x0, lo, hi)
    try:
        res = minimize(
            _param_score,
            x0,
            args=(n, mode),
            method="L-BFGS-B",
            bounds=bounds,
            options={"maxiter": 90, "maxls": 40, "ftol": 1e-9},
        )
        x1 = np.asarray(res.x, dtype=np.float64)
        s1 = _param_score(x1, n, mode)
        if np.isfinite(s1) and s1 < _param_score(x0, n, mode):
            return _make_p_from_params(x1, n, mode), float(s1)
    except Exception:
        pass
    p0 = _make_p_from_params(x0, n, mode)
    s0 = _score(p0)
    return p0, s0


def _optimize_phase1(n):
    """Coarse 2D search for exp(a sin(2*pi*x+phi))."""
    best_s = float("inf")
    best_a = 0.8
    best_phi = 0.0

    a_grid = np.linspace(0.0, 2.0, 11)
    phi_grid = np.linspace(0.0, 2.0 * math.pi, 18, endpoint=False)
    for a in a_grid:
        for phi in phi_grid:
            p = _make_p_phase1(np.array([a, phi]), n)
            s = _score(p)
            if np.isfinite(s) and s < best_s:
                best_s = float(s)
                best_a = float(a)
                best_phi = float(phi)

    # local fine grid
    for a in np.linspace(max(0.0, best_a - 0.35), best_a + 0.35, 8):
        for off in np.linspace(-0.8, 0.8, 9):
            phi = best_phi + off
            p = _make_p_phase1(np.array([a, phi]), n)
            s = _score(p)
            if np.isfinite(s) and s < best_s:
                best_s = float(s)
                best_a = float(a)
                best_phi = float(phi)

    p, s2 = _polish_params(n, "phase1", np.array([best_a, best_phi]))
    if p is not None and np.isfinite(s2) and s2 < best_s:
        best_s = float(s2)
        p = np.asarray(p, dtype=np.float64)
    else:
        p = _make_p_phase1(np.array([best_a, best_phi]), n)
        best_s = _score(p)
    return p, float(best_s)


def _optimize_centered1(n):
    """Coarse 1D search over one centered antisymmetric sine."""
    best_s = float("inf")
    best_a = 0.8
    grid = np.linspace(0.0, 2.2, 13)
    for a in grid:
        p = _make_p_center_sin(np.array([a, 0.0, 0.0, 0.0]), n)
        s = _score(p)
        if np.isfinite(s) and s < best_s:
            best_s = float(s)
            best_a = float(a)
    for a in np.linspace(max(0.0, best_a - 0.4), best_a + 0.4, 9):
        p = _make_p_center_sin(np.array([a, 0.0, 0.0, 0.0]), n)
        s = _score(p)
        if np.isfinite(s) and s < best_s:
            best_s = float(s)
            best_a = float(a)
    p = _make_p_center_sin(np.array([best_a, 0.0, 0.0, 0.0]), n)
    return p, float(best_s)


def _optimize_de(n, mode, maxiter, popsize):
    """Global DE search in a low-dimensional parametric family."""
    dim, bounds = _param_bounds(mode)
    if dim == 0:
        return None, float("inf")
    try:
        res = differential_evolution(
            _param_score,
            bounds,
            args=(n, mode),
            maxiter=maxiter,
            popsize=popsize,
            tol=1e-6,
            seed=11,
            mutation=(0.5, 1.5),
            recombination=0.7,
            polish=False,
        )
        x = np.asarray(res.x, dtype=np.float64)
        s = float(res.fun)
        if not np.isfinite(s):
            p = _make_p_from_params(x, n, mode)
            s = _score(p)
        p2, s2 = _polish_params(n, mode, x)
        if p2 is not None and np.isfinite(s2) and s2 < s:
            p = p2
            s = float(s2)
        else:
            p = _make_p_from_params(x, n, mode)
            s = _score(p)
        if not np.isfinite(s):
            p = _make_p_center_sin(np.array([0.8, 0.0, 0.0, 0.0]), n)
            s = _score(p)
        return p, float(s)
    except Exception:
        return None, float("inf")


def _random_centered_seq(n, seed):
    """Random centered antisymmetric sine start."""
    rng = np.random.default_rng(seed)
    amps = rng.normal(0.0, 0.65, 4)
    return _make_p_center_sin(amps, n)


def _build_conv_matrix(f):
    """
    Dense matrix C with (C @ g)[k] = (f*g)[k] for full linear convolution
    indices k=0..2n-2.
    """
    n = f.size
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = f
    return C


def _lp_solve(f, M):
    """
    LP:
        maximize sum(g)
        subject to f*g <= M, 0 <= g <= 10.
    The upper bound avoids numerical/unbounded degeneracies when f has
    very small entries.
    """
    n = f.size
    C = _build_conv_matrix(f)
    c_obj = -np.ones(n, dtype=np.float64)
    b_ub = M * np.ones(2 * n - 1, dtype=np.float64)
    bounds = [(0.0, 10.0)] * n
    try:
        res = linprog(
            c_obj,
            A_ub=C,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if res.status == 0 and res.x is not None:
            x = np.asarray(res.x, dtype=np.float64)
            x = np.clip(x, 0.0, None)
            x = x[np.isfinite(x)]
            if x.size == n and x.sum() > 0:
                return x
    except Exception:
        pass
    return None


def _alternating_lp(a_init, max_iters, max_time, initial_blend=0.55):
    """
    Alternating LP polish.
    Fix a, solve LP for g with a*g <= M(max a*a), blend, repeat.
    """
    a = np.asarray(a_init, dtype=np.float64)
    a = np.clip(a, 0.0, None)
    a = a[np.isfinite(a)]
    if a.size == 0:
        a = np.ones(512) / 512
    s = float(a.sum())
    if s < 1e-12:
        a = np.ones(a.size) / a.size
        s = 1.0
    else:
        a = a / s

    best_a = a.copy()
    best_score = _score(a)
    deadline = time.monotonic() + float(max_time)
    blend = float(initial_blend)
    last_improve_it = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        M = float(np.max(c))
        if not np.isfinite(M) or M <= 1e-15:
            break

        g = _lp_solve(a, M)
        if g is None:
            break
        sg = float(g.sum())
        if not np.isfinite(sg) or sg < 1e-12:
            break
        g = g / sg

        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        sn = float(a_new.sum())
        if not np.isfinite(sn) or sn < 1e-12:
            break
        a_new = a_new / sn

        score_new = _score(a_new)
        if np.isfinite(score_new) and score_new < best_score:
            best_score = float(score_new)
            best_a = a_new.copy()
            last_improve_it = it
            if it % 15 == 0:
                print(f"      LP it {it}: score={best_score:.6f}")
        a = a_new
        blend = max(0.08, blend * 0.985)

        if it > last_improve_it + 35:
            break

    return best_a, float(best_score)


def propose_candidate():
    """
    Main search entry point.
    Returns a list of non-negative floats minimizing the evaluator.
    """
    start_wall = time.monotonic()
    deadline = start_wall + 2350.0

    best_a = None
    best_score = float("inf")
    starts = []

    # ----------------------------------------------------------
    # 1) Parent / previous construction.
    # ----------------------------------------------------------
    parent_raw = None
    try:
        parent_raw = height_sequence_1
    except Exception:
        parent_raw = None

    if parent_raw is not None:
        try:
            parent = np.asarray(parent_raw, dtype=np.float64)
            parent = parent.reshape(-1)
            parent = parent[np.isfinite(parent)]
            parent = np.clip(parent, 0.0, 1000.0)
            if parent.size > 0:
                sp = _score(parent)
                if np.isfinite(sp):
                    best_a = parent.copy()
                    best_score = float(sp)
                    if parent.size <= 4096:
                        starts.append(("parent", parent.copy(), float(sp)))
                    print(f"Parent: n={parent.size}, score={sp:.6f}")
        except Exception:
            pass

    # ----------------------------------------------------------
    # 2) Low-dimensional parametric search.
    # ----------------------------------------------------------
    target_lengths = [500, 512, 640, 768, 1024]
    for n in target_lengths:
        if time.monotonic() > deadline - 760.0:
            break

        # centered antisymmetric one sine
        try:
            p1, s1 = _optimize_centered1(n)
            if p1 is not None and np.isfinite(s1):
                print(f"centered1 n={n}: score={s1:.6f}")
                starts.append((f"centered1_{n}", p1.copy(), float(s1)))
                if s1 < best_score:
                    best_score = float(s1)
                    best_a = p1.copy()
                    print(f"  ** new best centered1: {best_score:.6f}")
        except Exception:
            pass

        # phase-sine shape
        try:
            p2, s2 = _optimize_phase1(n)
            if p2 is not None and np.isfinite(s2):
                print(f"phase1 n={n}: score={s2:.6f}")
                starts.append((f"phase1_{n}", p2.copy(), float(s2)))
                if s2 < best_score:
                    best_score = float(s2)
                    best_a = p2.copy()
                    print(f"  ** new best phase1: {best_score:.6f}")
        except Exception:
            pass

    # Lower-dimensional global DE refinements, if time is available.
    de_runs = [
        (512, "center4", 30, 10),
        (640, "center4", 25, 10),
        (512, "poly3", 25, 10),
    ]
    for n, mode, mit, pop in de_runs:
        if time.monotonic() > deadline - 420.0:
            break
        try:
            pd, sd = _optimize_de(n, mode, maxiter=mit, popsize=pop)
            if pd is not None and np.isfinite(sd):
                print(f"DE {mode} n={n}: score={sd:.6f}")
                starts.append((f"DE_{mode}_{n}", pd.copy(), float(sd)))
                if sd < best_score:
                    best_score = float(sd)
                    best_a = pd.copy()
                    print(f"  ** new best DE {mode}: {best_score:.6f}")
        except Exception:
            pass

    # Random centered starts for LP diversity.
    try:
        for seed in (1, 2):
            pr = _random_centered_seq(512, seed)
            sr = _score(pr)
            if np.isfinite(sr):
                starts.append((f"rand{seed}", pr, float(sr)))
                if sr < best_score:
                    best_score = float(sr)
                    best_a = pr.copy()
    except Exception:
        pass

    # Interpolate current best to a few sensible lengths.
    if best_a is not None:
        for n in (480, 500, 512, 640, 768):
            if time.monotonic() > deadline - 360.0:
                break
            try:
                seq = _interpolate_sequence(best_a, n)
                sc = _score(seq)
                if np.isfinite(sc):
                    starts.append((f"interp_{n}", seq, float(sc)))
                    if sc < best_score:
                        best_score = float(sc)
                        best_a = seq.copy()
                        print(f"  ** interpolation improved to {best_score:.6f}, n={n}")
            except Exception:
                pass

    # ----------------------------------------------------------
    # 3) Alternating LP polish from the best starts.
    # ----------------------------------------------------------
    order = []
    for idx, (_label, _seq, sc) in enumerate(starts):
        if np.isfinite(sc):
            order.append((sc, idx))
    order.sort()

    used_starts = 0
    for _rank, (sc_order, idx_order) in enumerate(order):
        if used_starts >= 3:
            break
        if time.monotonic() > deadline - 180.0:
            break
        _label, seq, sc = starts[idx_order]
        if not np.isfinite(sc) or sc > best_score + 0.35:
            continue

        # Keep LP lengths moderate.
        n_seq = seq.size
        if n_seq < 256:
            seq_lp = _interpolate_sequence(seq, 384)
        elif n_seq > 768:
            seq_lp = _interpolate_sequence(seq, 512)
        else:
            seq_lp = seq.copy()

        # Time allocation:
        #   first start: up to 220s, second: 140s, third: 90s
        remaining = max(0.0, deadline - time.monotonic())
        if used_starts == 0:
            budget = min(220.0, remaining - 120.0)
        elif used_starts == 1:
            budget = min(140.0, remaining - 90.0)
        else:
            budget = min(90.0, remaining - 60.0)
        if budget < 25.0:
            break

        print(f"LP start '{_label}' n={seq_lp.size}, pre={sc:.6f}, budget={budget:.1f}s")
        try:
            lp_a, lp_s = _alternating_lp(
                seq_lp,
                max_iters=160,
                max_time=budget,
                initial_blend=0.55,
            )
            if lp_a is not None and np.isfinite(lp_s):
                starts.append((f"LP_{_label}", lp_a.copy(), float(lp_s)))
                if lp_s < best_score:
                    best_score = float(lp_s)
                    best_a = lp_a.copy()
                    print(f"  ** LP improved to {best_score:.6f}, n={best_a.size}")
            used_starts += 1
        except Exception as e:
            print("LP failure:", e)

    # If we still have generous time, a second LP on the current best can help.
    if best_a is not None and time.monotonic() < deadline - 80.0:
        remaining = max(0.0, deadline - time.monotonic())
        budget = min(120.0, remaining - 40.0)
        if budget > 25.0:
            n_best = best_a.size
            if n_best < 256:
                seq2 = _interpolate_sequence(best_a, 384)
            elif n_best > 768:
                seq2 = _interpolate_sequence(best_a, 512)
            else:
                seq2 = best_a.copy()
            print(f"Second LP polish n={seq2.size}, pre={best_score:.6f}, budget={budget:.1f}s")
            try:
                lp2_a, lp2_s = _alternating_lp(
                    seq2,
                    max_iters=160,
                    max_time=budget,
                    initial_blend=0.35,
                )
                if lp2_a is not None and np.isfinite(lp2_s) and lp2_s < best_score:
                    best_score = float(lp2_s)
                    best_a = lp2_a.copy()
                    print(f"  ** second LP improved to {best_score:.6f}, n={best_a.size}")
            except Exception as e:
                print("Second LP failure:", e)

    # ----------------------------------------------------------
    # 4) Fallback and final cleanup.
    # ----------------------------------------------------------
    if best_a is None or (not best_a.size) or (not np.all(np.isfinite(best_a))):
        try:
            best_a = _make_p_center_sin(np.array([0.85, 0.0, 0.0, 0.0]), 512)
        except Exception:
            best_a = np.ones(512) / 512.0

    arr = np.asarray(best_a, dtype=np.float64)
    arr = arr.reshape(-1)
    arr = np.clip(arr, 0.0, 1000.0)
    arr = arr[np.isfinite(arr)]
    if arr.size == 0:
        arr = np.ones(512) / 512.0

    s = float(arr.sum())
    if s < 0.01:
        arr = np.ones(512)
        s = float(arr.sum())
    if s > 0:
        arr = arr / s

    result = [float(x) for x in arr]

    # Optional validation; do not rely on this for correctness inside search.
    try:
        val = float(evaluate_sequence(result))
        if np.isfinite(val):
            print(f"Final validated score: {val:.6f}, n={len(result)}")
    except Exception:
        pass

    return result
```
