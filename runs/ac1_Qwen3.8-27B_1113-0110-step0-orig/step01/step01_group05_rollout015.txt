```python
import numpy as np
import time

def _clean(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    np.clip(a, 0.0, 1000.0, out=a)
    return a

def _norm(a):
    a = _clean(a)
    s = a.sum()
    if s <= 1e-30:
        return np.ones(max(1, a.size)) / max(1, a.size)
    return a / s

def _obj(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2 for normalized a (sum=1)."""
    a = _norm(a)
    n = a.size
    c = np.convolve(a, a)
    return 2.0 * n * np.max(c)

def _obj_from_conv(a, c):
    """Fast score from precomputed convolution (a must be normalized, sum=1)."""
    n = a.size
    return 2.0 * n * np.max(c)

def _flat_conv(n):
    """Generate sequence with flat left-side convolution."""
    b = np.zeros(n, dtype=np.float64)
    b[0] = 1.0
    for k in range(1, n):
        s = np.dot(b[1:k], b[k-1:0:-1]) if k > 1 else 0.0
        b[k] = (1.0 - s) / 2.0
        if b[k] < 0:
            b[k] = 0.0
    return _norm(b)

def _two_level(n):
    p = np.zeros(n, dtype=np.float64)
    m = n // 2
    r = n - m
    p[:m] = 2.0 / (3.0 * m) if m > 0 else 0
    p[m:] = 1.0 / (3.0 * r) if r > 0 else 0
    return _norm(p)

def _cosine(n):
    x = np.linspace(0, np.pi, n)
    p = 0.5 * (1 + np.cos(x))
    return _norm(p)

def _gaussian(n, sigma_frac=0.3):
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    p = np.exp(-0.5 * ((x - center) / (sigma_frac * n)) ** 2)
    return _norm(p)

def _ramp(n):
    p = np.linspace(1.0, 0.1, n)
    return _norm(p)

def _dirichlet(n, alpha, rng):
    return _norm(rng.dirichlet(np.full(n, alpha)))

def _paired_random(n, rng, alpha=1.0):
    """Random sequence with mirror symmetry."""
    m = (n + 1) // 2
    w = rng.dirichlet(np.full(m, alpha))
    p = np.zeros(n)
    for i in range(m):
        j = n - 1 - i
        if i == j:
            p[i] = w[i]
        else:
            frac = rng.beta(2, 2)  # random split
            p[i] = w[i] * frac
            p[j] = w[i] * (1 - frac)
    return _norm(p)

def _set_pair(n, rng, alpha=1.0, binary_frac=0.7):
    """Random sequence where pairs sum to their Dirichlet weight."""
    m = (n + 1) // 2
    w = rng.dirichlet(np.full(m, alpha))
    p = np.zeros(n)
    for i in range(m):
        j = n - 1 - i
        if i == j:
            p[i] = w[i]
        else:
            if rng.random() < binary_frac:
                p[i] = w[i] if rng.random() < 0.5 else 0
                p[j] = w[i] - p[i]
            else:
                frac = rng.beta(1.5, 1.5)
                p[i] = w[i] * frac
                p[j] = w[i] * (1 - frac)
    return _norm(p)
```

```python
    ks = np.arange(k_lo, k_hi + 1)
    offs = ks - j
    
    c[k_lo:k_hi+1] += 2.0 * delta * a[offs]
    
    # Fix self term
    if k_lo <= 2*j <= k_hi:
        c[2*j] -= 2.0 * delta * old
        c[2*j] += new_val * new_val - old * old
    
    a[j] = new_val
    return c
```
