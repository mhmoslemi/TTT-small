    2*n*max(convolve(a,a)) / (sum(a))^2

import time

def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()

def normalize_positive(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, float(floor), None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s

def convolve_seq(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if N is None:
        N = next_pow2(2 * n - 1)
    else:
        need = next_pow2(2 * n - 1)
        if int(N) < int(need):
            N = need
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]

def score_of(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.inf
    c = convolve_seq(a)
    return float(2.0 * n * float(np.max(c)) / (s * s))

def get_sequence():
    try:
        seq = globals().get("height_sequence_1", None)

def pressure_vector(a, k, N):
    """Compute the gradient contribution of C[k] to positions i."""
    n = a.size
    # C[k] = sum_i a[i]*a[k-i], dC[k]/da[i] = 2*a[k-i] (if valid)
    # a[k-i] is valid when 0 <= k-i <= n-1, i.e., k-(n-1) <= i <= k
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    p = np.zeros(n, dtype=np.float64)
    valid = np.arange(lo, hi + 1)
    p[valid] = 2.0 * a[k - valid]
    return p
