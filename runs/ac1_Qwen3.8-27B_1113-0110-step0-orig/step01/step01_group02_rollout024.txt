"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
1. LP-based direction finding (from SOTA idea)
2. Nesterov-accelerated multiplicative subgradient with multi-peak gradients
3. Coordinate descent for local polishing

import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]

def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a)
    return 2.0 * n * float(np.max(c)) / (s * s)

def _subgrad_at_k(a, k):
    """Subgradient of (a*a)[k] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        j_range = np.arange(lo, hi + 1)
        indices = k - j_range
        g[lo:hi+1] = 2.0 * a[indices]
    return g

def _multi_peak_subgrad(a, k_list, weights=None):
    """Weighted sum of subgradients at multiple peaks."""
    n = len(a)
    g = np.zeros(n)
    if weights is None:
        weights = np.ones(len(k_list)) / len(k_list)
    for k, w in zip(k_list, weights):
        g += w * _subgrad_at_k(a, int(k))
    return g

def _optimize_nesterov_mult(a0, time_budget, rng, top_k=5, verbose=False):
    """Nesterov-accelerated multiplicative subgradient with multi-peak gradients."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-10)
    a = a / a.sum()
    
    c = _fft_convolve(a)
    best_score = 2.0 * n * float(np.max(c))
    best_a = a.copy()
    
    eta = 0.8
    no_improve = 0
    deadline = time.monotonic() + time_budget
    it = 0
    last_log = time.monotonic()
    
    # Nesterov state
    y = a.copy()
    beta_t = 1.0
    prev_a = a.copy()
    
    while time.monotonic() < deadline:
        c = _fft_convolve(y)
        max_val = float(np.max(c))
        score = 2.0 * n * max_val
        
        if score < best_score:
            best_score = score
            best_a = y.copy()
            no_improve = 0
            if verbose and time.monotonic() - last_log > 10:
                print(f"    [nesterov] it={it}, score={score:.6f}, n={n}")
                last_log = time.monotonic()
        
        # Find top-k peaks
        sorted_idx = np.argsort(c)[::-1]
        top_indices = sorted_idx[:top_k]
        # Weight by value (higher peaks get more weight)
        peak_vals = c[top_indices]
        weights = peak_vals / peak_vals.sum()
        
        # Compute multi-peak subgradient
        g = _multi_peak_subgrad(y, top_indices, weights)
        g_max = g.max()
        if g_max < 1e-15:
            break
        g = g / g_max  # normalize
        
        # Nesterov update
        beta_t = (1.0 + np.sqrt(1.0 + 4.0 * beta_t * beta_t)) / 2.0
        momentum = (beta_t - 1.0) / beta_t
        
        # Multiplicative update
        y_new = y * np.exp(-eta * g)
        y_new = np.maximum(y_new, 1e-10)
        y_new = y_new / y_new.sum()
        
        # Nesterov extrapolation
        y = y_new + momentum * (y_new - a)
        y = np.maximum(y, 1e-10)
        y = y / y.sum()
        a = y_new
        
        no_improve += 1
        if no_improve > 1000:
            # Restart with noise
            a = best_a * np.exp(rng.normal(0, 0.06, n))
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            y = a.copy()
            beta_t = 1.0
            eta = 0.8
            no_improve = 0
        
        # Adaptive step size
        if no_improve > 500:
            eta *= 0.97
        elif no_improve > 200:
            eta *= 0.99
        elif no_improve < 50:
            eta = min(eta * 1.005, 1.5)
        
        it += 1
    
    return best_a, best_score

def _lp_direction(f, M, time_limit=60):
    """Solve LP: max sum(b) s.t. (f*b)[k] <= M for all k, b >= 0.

    # So for fixed k, j ranges from max(0, k-n+1) to min(n-1, k)
    # And A[k, j] = f[j]
    # Let me redo:
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi+1] = f[lo:hi+1]
    
    # Objective: maximize sum(b) = 1^T b
    # In linprog form: minimize -1^T b
    c_obj = -np.ones(n)
    
    # Constraints: A @ b <= M
    b_ub = np.full(2*n-1, M)
    
    # bounds: b >= 0
    bounds = [(0, None)] * n

        if result.success:

            s = b_opt.sum()
            if s > 1e-10:

        return None
    except Exception as e:

        return None

def _optimize_lp_based(a0, time_budget, rng, verbose=False):

        if g is None:
            break
        
        # Interpolate: a_new = (1-t)*a + t*g
        a_new = (1 - t_interp) * a + t_interp * g
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / a_new.sum()
        
        c_new = _fft_convolve(a_new)
        M_new = float(np.max(c_new))
        score_new = 2.0 * n * M_new

            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            M = M_new
            if verbose:
                print(f"    [LP] score={score_new:.6f}, M={M_new:.6f}")
        else:
            # Try smaller interpolation
            t_interp *= 0.5
            if t_interp < 0.001:
                # Restart

def _coordinate_descent(a0, time_budget, rng, verbose=False):
    """Cycle through elements, optimizing each one via line search."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-10).copy()
    a = a / a.sum()
    
    c = _fft_convolve(a)
    best_score = 2.0 * n * float(np.max(c))
    best_a = a.copy()
    
    deadline = time.monotonic() + time_budget
    it = 0
    last_log = time.monotonic()
    
    while time.monotonic() < deadline:
        # Iterate through a random permutation of indices
        perm = rng.permutation(n)
        for i in perm:
            if time.monotonic() > deadline:
                    break
            
            # Try a few values for a[i]
            current_val = a[i]
            # Try: current * factor for various factors
            best_local = current_val
            best_local_score = _score(a)
            
            for factor in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                a[i] = current_val * factor
                s = a.sum()
                if s < 1e-15:
                    a[i] = current_val
                    continue
                a_norm = a / s
                c_tmp = _fft_convolve(a_norm)
                sc = 2.0 * n * float(np.max(c_tmp))
                if sc < best_local_score:
                    best_local_score = sc
                    best_local = current_val * factor
            
            a[i] = best_local
            a = a / a.sum()
        
        # Check improvement
        c = _fft_convolve(a)
        score = 2.0 * n * float(np.max(c))
        if score < best_score:
            best_score = score
            best_a = a.copy()
            if verbose and time.monotonic() - last_log > 10:
                print(f"    [coord] it={it}, score={score:.6f}")
                last_log = time.monotonic()
        
        it += 1
    
    return best_a, best_score

        a = np.maximum(a, 0.01)
        candidates.append((f"cos_{n}", a))
    
    # 7. "U-shaped" (high at both ends, low in middle)
    for n in [2000]:
        x = np.arange(n, dtype=np.float64)
        t = x / n
        a = 1.0 + 0.5 * (2*t - 1)**2  # U-shaped
        candidates.append((f"u_shape_{n}", a))
    
    # 8. "Inverted U" (like a parabola)
    for n in [2000]:
        x = np.arange(n, dtype=np.float64)
        t = x / n
        a = 1.0 - 0.3 * (2*t - 1)**2  # inverted U
        a = np.maximum(a, 0.01)
        candidates.append((f"inv_u_{n}", a))
    
    print(f"Total candidates: {len(candidates)}")
    
    # Quick initial evaluation
    scored = []
    for name, a0 in candidates:
        a0 = np.maximum(a0, 0)
        s = a0.sum()
        if s < 1e-15:
            continue
        a0 = a0 / s
        c0 = _fft_convolve(a0)
        init_score = 2.0 * len(a0) * float(np.max(c0))
        scored.append((init_score, name, a0))
    
    scored.sort(key=lambda x: x[0])
    print("Initial scores (top 10):")
    for i, (sc, name, _) in enumerate(scored[:10]):
        print(f"  {i+1}. {name}: {sc:.6f}")
    
    # Strategy: Allocate time to top candidates, try multiple methods
    n_top = min(len(scored), 10)
    remaining_time = budget_s - 30.0
    
    # Geometric allocation
    weights = [max(1.0, n_top - i) ** 1.5 for i in range(n_top)]
    total_w = sum(weights)
    times = [remaining_time * w / total_w for w in weights]
    
    for i in range(n_top):
        if time.monotonic() > deadline:
            break
        
        init_score, name, a0 = scored[i]
        t_alloc = min(times[i], max(5.0, deadline - time.monotonic()))
        n = len(a0)
        
        print(f"\nOptimizing '{name}' (n={n}, init={init_score:.6f}, time={t_alloc:.0f}s)")
        
        t_quarter = t_alloc / 4.0
        
        # Method 1: Basic multiplicative subgradient
        a1, s1 = _optimize_basic_mult(a0, t_quarter, rng, verbose=True)
        eval1 = evaluate_sequence(a1.tolist())
        print(f"  Basic mult: {s1:.6f} / eval={eval1:.6f}")
        if np.isfinite(eval1) and eval1 < best_overall_score:
            best_overall_score = eval1
            best_overall = a1.copy()
            print(f"  *** NEW BEST: {eval1:.6f} ***")
        
        # Method 2: Nesterov multiplicative
        start2 = a1 if np.isfinite(s1) and s1 < init_score else a0
        a2, s2 = _optimize_nesterov_mult(start2, t_quarter, rng, top_k=5, verbose=True)
        eval2 = evaluate_sequence(a2.tolist())
        print(f"  Nesterov mult: {s2:.6f} / eval={eval2:.6f}")
        if np.isfinite(eval2) and eval2 < best_overall_score:
            best_overall_score = eval2
            best_overall = a2.copy()
            print(f"  *** NEW BEST: {eval2:.6f} ***")
        
        # Method 3: LP-based (if time allows and n is manageable)
        if n <= 2500 and t_quarter > 30:
            start3 = best_overall if best_overall is not None and len(best_overall) == n else start2
            a3, s3 = _optimize_lp_based(start3, t_quarter, rng, verbose=True)
            eval3 = evaluate_sequence(a3.tolist())
            print(f"  LP-based: {s3:.6f} / eval={eval3:.6f}")
            if np.isfinite(eval3) and eval3 < best_overall_score:
                best_overall_score = eval3
                best_overall = a3.copy()
                print(f"  *** NEW BEST: {eval3:.6f} ***")
        
        # Method 4: Coordinate descent (polishing)
        if best_overall is not None and len(best_overall) == n and t_quarter > 20:
            start4 = best_overall.copy()
            a4, s4 = _coordinate_descent(start4, t_quarter, rng, verbose=True)
            eval4 = evaluate_sequence(a4.tolist())
            print(f"  Coord desc: {s4:.6f} / eval={eval4:.6f}")
            if np.isfinite(eval4) and eval4 < best_overall_score:
                best_overall_score = eval4
                best_overall = a4.copy()
                print(f"  *** NEW BEST: {eval4:.6f} ***")
    
    # Final polishing
    remaining = deadline - time.monotonic()
    if best_overall is not None and remaining > 30:
        print(f"\nFinal polishing with {remaining:.0f}s remaining...")
        t_polish = remaining * 0.5
        a_pol, sp = _optimize_basic_mult(best_overall, t_polish, rng, verbose=True)
        eval_p = evaluate_sequence(a_pol.tolist())
        print(f"  Polish (mult): {eval_p:.6f}")
        if np.isfinite(eval_p) and eval_p < best_overall_score:
            best_overall_score = eval_p
