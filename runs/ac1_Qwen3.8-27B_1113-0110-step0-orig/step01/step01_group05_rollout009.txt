def _project_simplex(v, target=1.0):
    """Project onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    if n == 0:
        return v

    cond = css - u * np.arange(1, n + 1) < target
    if not cond.any():
        return np.full(n, target / n)
    rho = int(np.max(np.where(cond)) + 1)
    theta = (css[rho - 1] - target) / rho
    out = np.maximum(v - theta, 0.0)
    # Ensure exact sum
    s = out.sum()
    if s > 1e-15:

    length = 2 * n - 1
    S = 1 << (length - 1).bit_length()
    fa = np.fft.rfft(a, S)
    result = np.fft.irfft(fa * fa, S)
    return result[:length]


def _normalize_sum1(a):
    """Normalize array to be non-negative with sum exactly 1."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-15:
        return np.ones_like(a) / a.size
    return a / s


def _true_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-10:
        return float('inf')
    b = _fft_self_conv(a)
    mx = float(np.max(b))
    if not np.isfinite(mx):
        return float('inf')
    return 2.0 * n * max(0.0, mx) / (s * s)


def _project_simplex(v, target=1.0):
    """Project onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    cond = css - u * np.arange(1, n + 1) < target
    if not cond.any():
        return np.full(n, target / n, dtype=np.float64)
    rho = int(np.max(np.where(cond))) + 1
    theta = (css[rho - 1] - target) / float(rho)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 1e-15:
        out *= target / s
    else:
        out = np.full(n, target / n, dtype=np.float64)
    return out


def _compute_grad(a, p, S):
    """Gradient of sum(b_k^p) where b = convolve(a,a)."""
    n = a.size
    b = _fft_self_conv(a)
    b_clipped = np.maximum(b, 1e-300)
    c = np.power(b_clipped, p - 1)
    fa = np.fft.rfft(a, S)
    fc = np.fft.rfft(c, S)
    R = np.fft.irfft(np.conj(fa) * fc, S)
    return 2.0 * p * R[:n]


def _smooth_obj(a, p):
    """Smooth objective sum(b_k^p)."""
    b = _fft_self_conv(a)
    b = np.maximum(b, 0.0)
    return float(np.sum(np.power(b, p)))


def _make_initial(n, type_idx, rng):
    """Generate an initial sequence of length n based on type_idx."""
    n = int(n)
    if type_idx == 0:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif type_idx == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-2.0 * ((x - 0.5) / 0.25) ** 2)
    elif type_idx == 2:
        m = n // 2
        a = np.ones(n)
        a[:m] = 1.0
        a[m:] = 0.4
    elif type_idx == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-((x - 0.2) / 0.1) ** 2) + np.exp(-((x - 0.8) / 0.1) ** 2)
    elif type_idx == 4:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 / (1.0 + np.exp(-8.0 * (x - 0.5)))
    elif type_idx == 5:
        a = rng.dirichlet(np.full(n, 0.5))
    elif type_idx == 6:
        a = rng.dirichlet(np.full(n, 1.0))
    elif type_idx == 7:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - x) ** 0.5 + 0.1
    elif type_idx == 8:
        # Bimodal wide
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-((x - 0.15) / 0.15) ** 2) + np.exp(-((x - 0.85) / 0.15) ** 2) + 0.1
    else:
        a = np.ones(n)
    a = a + 0.01 * rng.random(n)
    return _normalize_sum1(a)


def _gradient_descent(a, deadline, p_schedule, lr_init, use_momentum, max_perts):
    """Main gradient descent loop with adaptive p and backtracking."""
    a = _normalize_sum1(a)
    n = a.size
    S = 1 << (3 * n - 3).bit_length()

    best_a = a.copy()
    best_true = _true_score(a)
    lr = lr_init
    v = np.zeros(n, dtype=np.float64)  # momentum
    beta1 = 0.9
    beta2 = 0.999
    m_t = np.zeros(n, dtype=np.float64)
    v_t = np.zeros(n, dtype=np.float64)
    t_step = 0
    it = 0
    stagnant = 0
    perts = 0
    noise_scales = [0.15, 0.08, 0.04, 0.02, 0.01]

    while time.monotonic() < deadline - 1.0:
        # Determine p
        frac = min(1.0, it / 3000.0)
        if len(p_schedule) <= 1:
            p = p_schedule[0]
        else:
            idx = int(frac * (len(p_schedule) - 1))
            p = p_schedule[min(idx, len(p_schedule) - 1)]

        # Compute gradient
        try:
            grad = _compute_grad(a, p, S)
        except Exception:
            break

        grad_norm = float(np.linalg.norm(grad))
        if not np.isfinite(grad_norm) or grad_norm < 1e-30:
            break

        # Adam update
        t_step += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad * grad)
        m_hat = m_t / (1.0 - beta1 ** t_step)
        v_hat = v_t / (1.0 - beta2 ** t_step)
        step_dir = m_hat / (np.sqrt(v_hat) + 1e-8)
        step_norm = float(np.linalg.norm(step_dir))
        if not np.isfinite(step_norm) or step_norm < 1e-30:
            step_dir = -grad / grad_norm
            step_norm = 1.0

        # Backtracking on normalized step
        F_cur = _smooth_obj(a, p)
        a_new = None
        for bt in range(8):
            alpha = lr / (2.0 ** bt)
            candidate = a - alpha * step_dir / step_norm
            candidate = _project_simplex(candidate, 1.0)
            F_new = _smooth_obj(candidate, p)
            if F_new < F_cur * (1.0 - 1e-10):
                a_new = candidate
                break
        if a_new is None:
            lr *= 0.7
            if lr < 1e-12:
                lr = lr_init * 0.1
            a_new = a - lr * step_dir / step_norm
            a_new = _project_simplex(a_new, 1.0)
        else:
            lr = min(lr * 1.03, 0.5)

        a = a_new
        it += 1

        # Periodic true-objective evaluation
        if it % 30 == 0 or it <= 5:
            ts = _true_score(a)
            if np.isfinite(ts) and ts < best_true:
                best_true = ts
                best_a = a.copy()
                stagnant = 0
            else:
                stagnant += 1

            if stagnant >= 12:
                if perts < max_perts:
                    rng_p = np.random.default_rng(it * 7919 + 13)
                    scale = noise_scales[min(perts, len(noise_scales) - 1)]
                    pert = best_a * np.exp(scale * rng_p.normal(0, 1, n))
                    pert = _normalize_sum1(pert)
                    a = pert
                    stagnant = 0
                    perts += 1
                else:
                    # Fresh random restart
                    rng_r = np.random.default_rng(it * 31 + 7)
                    a = rng_r.dirichlet(np.full(n, 0.3))
                    stagnant = 0
                    perts = 0
                    lr = lr_init

    return best_a, best_true


def _coord_descent_light(a, deadline, max_sweeps=2, grid_size=10):
    """Light coordinate descent for final polish. Uses FFT for score."""
    a = _normalize_sum1(a)
    n = a.size
    b = _fft_self_conv(a)
    score = _true_score(a)
    best_a = a.copy()
    best_score = score

    grid = np.concatenate(([0.0], np.logspace(-5.0, 1.0, grid_size)))

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline - 0.5:
            break
        improved_sweep = 0
        for i in range(n):
            if time.monotonic() > deadline - 0.5:
                break
            old = float(a[i])
            D = 1.0 - old
            if D < 0:
                D = 0.0
            # Affected convolution positions: k in [i, i+n-1]
            ks = slice(i, i + n)
            c_k = b[ks]
            # Unaffected max
            max_unaff = 0.0
            if i > 0:
                mu = float(np.max(b[:i]))
                if mu > max_unaff:
                    max_unaff = mu
            end = i + n
            if end < 2 * n - 1:
                mu = float(np.max(b[end:]))
                if mu > max_unaff:
                    max_unaff = mu

            best_x = old
            best_val = score
            mid_k = 2 * i - i  # position of i in the affected slice (k=2i means index i within c_k)
            # Actually mid index within c_k: k - i where k = 2i, so index = i
            mid = i

            for xv in grid:
                xv = float(xv)
                Sx = D + xv
                if Sx < 1e-12:
                    continue
                # New affected convolution values:
                # b_new[k] = b[k] + 2*(xv - old)*a[k-i] for k != 2i
                # b_new[2i] = b[2i] + (xv-old)*(xv+old)
                other = np.array([a[j] for j in range(n) if j != i], dtype=np.float64)
                # More efficiently: affected[j] = c_k[j] + 2*(xv-old)*a[i+j]... 
                # a[k-i] where k = i+j, so a[k-i] = a[j] for j=0..n-1
                affected = c_k + 2.0 * (xv - old) * a
                affected[mid] = c_k[mid] + (xv - old) * (xv + old)
                max_aff = float(np.max(affected))
                maxc = max(max_unaff, max_aff)
                val = 2.0 * n * max(0.0, maxc) / (Sx * Sx)
                if val < best_val:
                    best_val = val
                    best_x = xv

            if best_x != old and best_val < score - 1e-12:
                Sx = D + best_x
                if Sx > 1e-12:
                    affected = c_k + 2.0 * (best_x - old) * a
                    affected[mid] = c_k[mid] + (best_x - old) * (best_x + old)
                    b_new = b.copy()
                    b_new[ks] = affected
                    b_new = b_new / (Sx * Sx)
                    # Normalize a
                    a_new = a / Sx
                    a_new[i] = best_x / Sx
                    s_new = a_new.sum()
                    if s_new > 1e-12:
                        a = a_new / s_new
                        b = b_new / (s_new * s_new)
                        np.clip(b, 0.0, None, out=b)
                        score = 2.0 * n * float(np.max(b))
                        if score < best_score:
                            best_score = score
                            best_a = a.copy()
                        improved_sweep += 1
                        continue
        # Recompute after sweep
        b = _fft_self_conv(a)
        np.clip(b, 0.0, None, out=b)
        score = _true_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        if improved_sweep == 0:
            break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Return the best nonnegative coefficient sequence found within budget_s."""
    budget = max(1.0, float(budget_s))
    deadline = time.monotonic() + max(0.0, budget - 8.0)
    rng = np.random.default_rng(int(seed))
    final_len = 3809

    print(f"Starting search with budget={budget:.0f}s, deadline in {budget-8:.0f}s")

    # Load parent sequence if available
    parent = None
    try:
        hseq = np.asarray(height_sequence_1, dtype=np.float64)
        if hseq.size > 0:
            parent = _normalize_sum1(hseq)
            pscore = _true_score(parent)
            print(f"Parent sequence loaded: length={hseq.size}, score={pscore:.6f}")
    except Exception:
        pass

    best_seq = None
    best_score = float('inf')

    def _try_update(seq, sc):
        nonlocal best_seq, best_score
        if seq is None or not np.isfinite(sc):
            return
        if sc < best_score:
            best_score = float(sc)
            best_seq = _normalize_sum1(seq).copy()
            print(f"  New best: {best_score:.6f} (len={best_seq.size})")

    # Initial candidates
    candidates = []
    if parent is not None:
        candidates.append(parent)
        _try_update(parent, _true_score(parent))

    # Generate diverse initial sequences
    types = [0, 1, 2, 3, 4, 5, 7, 8]
    for t in types:
        if time.monotonic() > deadline - 60.0:
            break
        a_init = _make_initial(final_len, t, rng)
        candidates.append(a_init)
        sc = _true_score(a_init)
        _try_update(a_init, sc)

    # Sort by score
    scored = []
    for a_c in candidates:
        sc = _true_score(a_c)
        if np.isfinite(sc):
            scored.append((sc, a_c))
    scored.sort(key=lambda x: x[0])
    print(f"Initial candidates scored: best={scored[0][0]:.6f}" if scored else "No valid candidates")

    # Phase 1: Gradient descent on top candidates (main workhorse)
    p_schedule = [4, 8, 16, 32, 64]
    lr_init = 0.03
    max_perts = 6

    top_k = min(4, len(scored))
    phase1_deadline = deadline - 90.0
    per_trial_time = max(20.0, (phase1_deadline - time.monotonic()) / max(1, top_k))

    for idx in range(top_k):
        if time.monotonic() > phase1_deadline:
            break
        sc, a_c = scored[idx]
        trial_deadline = min(phase1_deadline, time.monotonic() + per_trial_time)
        print(f"  Gradient trial {idx+1}/{top_k}: start score={sc:.6f}, time_left={trial_deadline-time.monotonic():.0f}s")
        a_out, sc_out = _gradient_descent(a_c, trial_deadline, p_schedule, lr_init, True, max_perts)
        _try_update(a_out, sc_out)

    # Phase 2: If we have time, do a longer refinement on the current best
    if best_seq is not None and time.monotonic() < deadline - 40.0:
        print(f"  Phase 2: Refinement on best (score={best_score:.6f})")
        refine_deadline = deadline - 30.0
        p_schedule2 = [16, 32, 64, 128]
        a_out, sc_out = _gradient_descent(best_seq, refine_deadline, p_schedule2, 0.01, True, 3)
        _try_update(a_out, sc_out)

    # Phase 3: Light coordinate descent polish
    if best_seq is not None and time.monotonic() < deadline - 10.0:
        print(f"  Phase 3: Coordinate descent polish (score={best_score:.6f})")
        # Reduce grid for speed
        a_out, sc_out = _coord_descent_light(best_seq, deadline - 5.0, max_sweeps=2, grid_size=8)
        _try_update(a_out, sc_out)

    # Final validation
    if best_seq is None:
        best_seq = _make_initial(final_len, 2, rng)
        best_score = _true_score(best_seq)

    # Ensure valid output
    out = _normalize_sum1(best_seq)
    out = np.clip(out, 0.0, 1000.0)
    out_sum = out.sum()
    if out_sum < 0.01:
        out = np.ones(final_len) / final_len
        out_sum = 1.0

    # Scale to have sum = sqrt(2n) for good normalization in evaluator
    target_sum = np.sqrt(2.0 * final_len)
    out = out * (target_sum / out_sum)

    final_score = _true_score(out)
    print(f"Final: length={out.size}, score={final_score:.6f}, sum={out.sum():.4f}")

    return [float(x) for x in out]
