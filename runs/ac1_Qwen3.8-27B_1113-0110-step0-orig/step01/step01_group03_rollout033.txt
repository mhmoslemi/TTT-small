```python
"""
Minimizes 2n*max(convolve(a,a))/sum(a)^2 using:
1. LP-based optimization (state-of-the-art approach)
2. Multi-peak subgradient descent with line search
3. Random perturbations for escaping local minima

import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog

def fft_self_conv(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    if n <= 1:
        return a * a
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:2 * n - 1]

def normalize_sum(a, target):
    """Normalize array to sum=target, keeping non-negativity."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-10:
        return np.full_like(a, target / len(a))
    return a / S * target

def build_conv_matrix(f):
    """Build sparse convolution matrix A such that (A @ g) = f * g (full convolution)."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def lp_solve(f, M):
    """
    Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns optimal g or None.
    """
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.status == 0:
            return result.x
        else:
            print(f"LP status: {result.status}, message: {result.message}")
            return None
    except Exception as e:
        print(f"LP error: {e}")
        return None

def compute_multi_peak_subgradient(a, c, threshold_frac=0.99):
    """
    Compute subgradient using all peaks above threshold_frac * max.
    Returns gradient vector.
    """
    n = len(a)
    M = c.max()
    k_list = np.where(c >= threshold_frac * M)[0]
    if len(k_list) == 0:
        k_list = np.array([int(np.argmax(c))])
    
    grad = np.zeros(n)
    for k in k_list:
        k = int(k)
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        if j_max >= j_min:
            s_start = k - j_max
            s_end = k - j_min + 1
            grad[j_min:j_max + 1] += a[s_start:s_end][::-1]
    grad /= len(k_list)
    
    # Score = 2n*M/S^2 with S=n (normalized)
    # d Score / d a_j = (2n/S^2) * dM/da_j - 2n*M/S^3 * dS/da_j
    # With S = n fixed: d Score / d a_j = (2n/n^2) * dM/da_j = 2/n * dM/da_j
    # dM/da_j = a[k-j] for the peak k (averaged over peaks in subgradient)
    # So grad = 2/n * (averaged a[k-j])
    # But we also need to account for the constraint that sum stays = n.
    # Using Lagrange multiplier: grad = 2/n * (averaged a[k-j]) - lambda
    # lambda = 2/n * mean(averaged a[k-j]) = 2/n * (1/n) * sum_j (averaged a[k-j])
    # sum_j a[k-j] over valid j = (f*f)[k] = M (for the peak)
    # So lambda = 2/n * M/n = 2M/n^2
    
    grad = (2.0 / n) * grad - (2.0 * M / (n * n))
    return grad
```

```python
def fft_self_conv(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    if n <= 1:
        return a * a
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:2 * n - 1]


def normalize_sum(a, target):
    """Normalize array to sum=target, keeping non-negativity."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-10:
        return np.full_like(a, target / len(a))
    return a / S * target


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A @ g)[k] = (f*g)[k] for full convolution."""
    n = len(f)
    row_idx = []
    col_idx = []
    data = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            row_idx.append(k)
            col_idx.append(j)
            data.append(f[k - j])
    return csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))

    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns optimal g array or None if failed/slow.

                         method='highs', options={'time_limit': time_limit})

            return None
        if result.status == 0:
            return result.x
        else:

def multi_peak_gradient(a, c, thresh_frac=0.998):
    """
    Compute subgradient of score w.r.t. a using all peaks above threshold.
    Assumes sum(a) = n.
    Score = 2*n*M / n^2 = 2*M/n where M = max(c).
    d(score)/d(a_j) = (2/n) * mean_k[a[k-j]] - (2*M)/(n*n)
    where mean is over active peaks k.
    """
    n = len(a)
    M = c.max()
    k_list = np.where(c >= thresh_frac * M)[0]
    if len(k_list) == 0:
        k_list = np.array([int(np.argmax(c))])
    
    grad = np.zeros(n)
    for k in k_list:
        k = int(k)
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        if j_max >= j_min:
            s_start = k - j_max
            s_end = k - j_min + 1
            grad[j_min:j_max + 1] += a[s_start:s_end][::-1]
    grad /= len(k_list)
    
    grad = (2.0 / n) * grad - (2.0 * M / (n * n))
    return grad

            peak_grad[j_min:j_max + 1] = a[s_start:s_end][::-1]
        
        grad = (2.0 / n) * peak_grad - (2.0 * M / (n * n))
        
        lr = 0.002 * max(0.1, 200.0 / (it + 1.0))
        a = np.maximum(a - lr * grad, 0.0)
        a = normalize_sum(a, n)
        
        if it % 50 == 0:
            sc = float(evaluate_sequence(a.tolist()))
            if np.isfinite(sc) and sc < fine_tune_score:
                fine_tune_score = sc
                best_a = a.copy()
                best_score = sc
                print(f"  Fine-tune #{it}: score={sc:.8f}")
            else:
                a = best_a.copy()
    
    # Final validation
    final_a = normalize_sum(best_a, n)
    final_score = float(evaluate_sequence(final_a.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score
        best_a = final_a.copy()
    else:
        # Keep original
        final_score = best_score
    
    print(f"Final: score={best_score:.8f}, n={n}")
    return [float(x) for x in best_a]
```
