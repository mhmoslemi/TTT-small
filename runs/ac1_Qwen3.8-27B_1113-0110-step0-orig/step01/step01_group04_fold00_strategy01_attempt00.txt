# Fallback functions from previous code
def _clean_nonnegative(v):
    """Ensure non-negativity and finite values."""
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    return np.maximum(v, 0.0)

def _normalize(v):
    """Normalize to the simplex or uniform fallback."""
    v = _clean_nonnegative(v)
    s = np.sum(v)
    if not np.isfinite(s) or s <= 0.0:
        n = v.size
        if n == 0:
            return v
        if n == 1:
            return np.ones(1, dtype=np.float64) * 0.01
        return np.full(n, 1.0/float(n), dtype=np.float64)
    return v / s

def _project_simplex(v):
    """Euclidean projection onto non-negative simplex with sum constraint."""
    v = _clean_nonnegative(v)
    n = v.size
    if n == 0:
        return v.copy()
    x = np.sort(v)[::-1]
    css = np.cumsum(x)
    tau = 1.0
    r = 0
    for i in range(1, n+1):
        if css[i-1] <= 1.0:
            tau = 1.0 / i
            r = i
            break
    if r == 0:
        return np.full(n, 1.0/float(n), dtype=np.float64)
    if np.any(v > x[r]) or not np.isfinite(v).all():
        v = v - tau * (np.maximum(v, x[r]) - v) if np.size(v) > 0 else np.zeros(n)
    return v / np.sum(v)

def _pow2(m):
    """Round m to power of2"""
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m-1).bit_length()

def _score_vector(a, L=None):
    """Fast exact score using FFT-based linear convolution."""
    v = _clean_nonnegative(a)
    n = v.size
    if n == 0:
        return np.inf
    total_sum = np.sum(v)
    if not np.isfinite(total_sum) or total_sum <= 0.0:
        return np.inf
        
    if L is None:
        L = _pow2(3*n-1)
    try:
        A = np.fft.rfft(v, n=L)
        C = A * A
        c = np.fft.irfft(C, n=L)[:2*n-1]
        if np.any(c < 0):
            c = np.maximum(c, 0.0)
            c[~np.isfinite(c)] = 0.0
        max_b_val = float(c.max())
        return 2.0 * float(n) * max_b_val
    except Exception as e:
        print("FFT failed with", e, "using L=", L)
        # Fallback to direct convolution
        x = np.arange(0, 2*n-1)
        if n < 10:
            conv_result = []
            for i in range(2*n-1):
                s_val = 0.0
                for j in range(max(0, i-n+1), min(i+1, n)):
                    s_val += v[j] * v[i-j]
                conv_result.append(s_val)
            max_b_val = float(max(conv_result))
            return 2.0 * float(n) * max_b_val if np.isfinite(max_b_val) else 10**6
        # Otherwise fall back to safe method
        try:
            w = np.zeros(2*n-1)
            for i in range(0, n):
                for j in range(0, n):
                    w[i+j] += v[i] * v[j]
            max_b_val = float(np.max(w))
            return 2.0 * float(n) * max_b_val
        except Exception:
            return np.inf

def _q_objective_grad_part(x, q_val, simplex_lengths):
    """Smoothed max q-objective helper function for gradient calculation."""
    simplex = _softmax_from_ratios(x)
    n = simplex.size
    try:
        A = np.fft.rfft(simplex, n=simplex_lengths)
    except:
        return np.inf
    
    try:
        if q_val == 2.0:
            conv = np.fft.irfft(A * A, n=simplex_lengths)[:max(0, n)]
            return np.sum(conv * conv) / (2.0 * np.sum(simplex)**2)
        else:
            conv = np.fft.irfft(A * A, n=simplex_lengths)[:n]
            conv_power = np.power(conv, q_val)
            return np.sum(conv_power) / (np.sum(simplex)**2 * 2.0)
    except Exception:
        return np.inf

def _local_search_block(a, block_size, score_func):
    """Local search perturbation handling block."""
    try:
        best = a.copy()

        cand_amp = np.exp(np.random.randn(block_size) * 0.5)
        cand_shift = np.random.rand(block_size)
        for i in range(block_size):
            # Generate candidate with exponential length distribution
            cand1_amp = np.exp(np.random.randn(block_size) * 0.1)
            cand1_shift = cand_amp[i] + np.random.rand()
            cand1 = cand1_amp / cand1_shift
            # Generate another candidate with similar design
            cand2_amp = np.exp(np.random.randn(block_size) * 0.1)
            cand2_shift = cand_amp[1] + np.random.rand() if len(cand_amp)>1 else cand_shift
            cand2 = cand2_amp / cand2_shift
            
            # Compute scores for comparison
            try:
                score_rand1 = score_func(cand1)
                score_rand2 = score_func(cand2)
            except:
                
                continue
                
            # Compare with best candidate
            if score_rand1 < best_score:
                best = cand1
                
                best_score = score_rand1
                
            if score_rand2 < best_score:
                best = cand2
                
                best_score = score_rand2
                
        return best, best_score
    except Exception as e:
        print("Local search error:", e)
        return a, score_func(a)

def _softmax_from_ratios(x):
    """Convert ratios to a simplex representation."""
    x = np.asarray(x, dtype=np.float64)
    n = x.size + 1
    x = np.exp(x)  # this is where the code might have gone wrong due to dimension mismatch
    s = np.sum(x)
    if not np.isfinite(s) or s==0:
        return np.ones(n)/n
    return np.exp(x) / s

def _power_sequence(n, p=0.5, offset=0.5):
    """Generate a power sequence then normalize."""
    x = np.arange(n, dtype=np.float64) + offset
    return _normalize(x**(-p))

def _power_window_sequence(n, p=0.5, offset=0.5, boost=0.5, gamma=0.5):
    """Generate a window boosted power sequence then normalize."""
    x = np.arange(n, dtype=np.float64) + offset
    win = max(1, min(int(n/10), _pow2(np.ceil(5/p)+1)))
    window_factors = np.exp((x[:win] - n/2)*gamma*(-1))
    return _normalize(x**(-p)*window_factors)

def _two_tail_sequence(n, p=0.5, offset=0.5, tail_weight=0.5):
    """Generate two-tail sequence (roughly) then normalize."""
    x_rev = np.arange(n, dtype=np.float64)[::-1] + offset
    tail_amp = min(3, max(1, _pow2(tail_weight)/4))
    return _normalize(x + x_rev * tail_amp)
