import numpy as np
import time

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]

def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a)
    return 2.0 * n * float(np.max(c)) / (s * s)

def _subgrad_topk(a, k_indices):
    """Compute subgradient as weighted sum at given peak indices."""
    n = len(a)
    g = np.zeros(n)
    c = _fft_convolve(a)
    weights = c[k_indices]
    if weights.sum() < 1e-15:
        return g
    weights = weights / weights.sum()
    for w, k in zip(weights, k_indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            j = np.arange(lo, hi + 1)
            g[j] += w * 2.0 * a[k - j]
    return g

def _optimize(a0, time_budget, rng, top_k=5, eta_init=0.1, momentum_init=0.85):
    """Subgradient descent with momentum and adaptive step size."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    
    c = _fft_convolve(a)
    best_score = 2.0 * n * float(np.max(c))
    best_a = a.copy()
    
    eta = eta_init
    momentum = momentum_init
    g_mom = np.zeros(n)
    no_improve = 0
    deadline = time.monotonic() + time_budget
    it = 0
    last_print = time.monotonic()
    
    while time.monotonic() < deadline:
        c = _fft_convolve(a)
        # Get top-K indices
        if top_k >= len(c):
            k_idx = np.arange(len(c))
        else:
            k_idx = np.argpartition(c, -top_k)[-top_k:]
            k_idx = k_idx[np.argsort(c[k_idx])[::-1]]
        
        k_max = int(k_idx[0])
        score = 2.0 * n * c[k_max]
        
        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            if it % 50 == 0:
                if time.monotonic() - last_print > 5.0:
                    print(f"      it={it}, score={score:.8f}")
                    last_print = time.monotonic()
        else:

        if no_improve > 500:
            a = best_a * np.exp(rng.normal(0, 0.05, n))
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            no_improve = 0
            eta = eta_init
            g_mom = np.zeros(n)
            it += 1
            continue

        g = _subgrad_topk(a, k_idx)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        
        # Momentum update
        g_mom = momentum * g_mom + (1 - momentum) * g
        
        # Adaptive step size
        if no_improve > 300:
            eta *= 0.95
        elif no_improve > 100:
            eta *= 0.98
        elif no_improve < 20:
            eta = min(eta * 1.01, 0.3)

        a = a * np.exp(-eta * g_mom)
        a = np.maximum(a, 1e-15)

        if s > 1e-15:
            a = a / s
        else:
            a = best_a.copy()
            eta *= 0.5

import numpy as np
import time

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses: (1) multiple initializations, (2) subgradient descent with momentum 
and top-K smoothing, (3) coordinate descent polishing with incremental 
convolution updates.
"""

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]

def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a)
    return 2.0 * n * float(np.max(c)) / (s * s)

def _subgrad_topk(a, c, top_k=5):
    """Compute smoothed subgradient using top-K peaks."""
    n = len(a)
    k_idx = np.argpartition(c, -top_k)[-top_k:]
    weights = c[k_idx].astype(np.float64)
    if weights.sum() < 1e-30:
        return np.zeros(n)
    weights = weights / weights.sum()
    g = np.zeros(n)
    for w, k in zip(weights, k_idx):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            j = np.arange(lo, hi + 1)
            g[j] += w * 2.0 * a[k - j]
    return g

def _sgd_optimize(a0, time_budget, rng, top_k=5, eta_init=0.12, mom_init=0.82):
    """Subgradient descent with multiplicative updates, momentum, and adaptive step size."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    
    c = _fft_convolve(a)
    best_score = 2.0 * n * float(np.max(c))
    best_a = a.copy()
    
    eta = eta_init
    mom = mom_init
    g_mom = np.zeros(n)
    no_improve = 0
    deadline = time.monotonic() + time_budget
    it = 0
    last_print = time.monotonic()
    
    while time.monotonic() < deadline:
        c = _fft_convolve(a)
        k_max = int(np.argmax(c))
        score = 2.0 * n * c[k_max]
        
        if score < best_score - 1e-9:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            if time.monotonic() - last_print > 3.0:
                print(f"        it={it}, score={score:.8f}")
                last_print = time.monotonic()
        else:
            no_improve += 1
        
        if no_improve > 400:
            # Restart
            noise_scale = 0.03 + 0.02 * min(no_improve / 2000.0, 1.0)
            a = best_a * np.exp(rng.normal(0, noise_scale, n))
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            no_improve = 0
            eta = eta_init
            g_mom = np.zeros(n)
            it += 1
            continue
        
        g = _subgrad_topk(a, c, top_k)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        
        g_mom = mom * g_mom + (1.0 - mom) * g
        
        # Adaptive step size
        if no_improve > 250:
            eta *= 0.96
        elif no_improve > 80:
            eta *= 0.985
        elif no_improve < 30:
            eta = min(eta * 1.005, 0.25)

        a = a * update
        a = np.maximum(a, 1e-15)

        if s < 1e-10:
            a = best_a.copy()
            eta = max(eta * 0.5, 0.01)
        else:

                # Note: a[k-j] is valid for k-j in [0, n-1], i.e., k in [j, j+n-1]
                
                # For k outside [lo, hi]: no change
                # For k inside: c_new[k] = c[k] + 2*delta*a[k-j] + delta^2*(k==2j)
                
                # We only need to check the max over [lo, hi] and the rest
                # max(c_new) = max(max(c[lo:hi+1] + changes), max(c outside [lo,hi]))
                
                k_range = np.arange(lo, hi + 1)
                changes = 2.0 * delta * a[k_range - j]
                if 2 * j >= lo and 2 * j <= hi:
                    changes[2 * j - lo] += delta * delta
                
                c_new_range = c[lo:hi+1] + changes
                new_max = max(c_new_range.max(), 
                             max(c[:lo].max() if lo > 0 else 0, 
                                 c[hi+1:].max() if hi + 1 < len(c) else 0))
                
                if new_max < best_delta_score - 1e-12:
                    best_delta_score = new_max
                    best_mult = mult
            
            if best_mult != 1.0:
                # Apply the best update
                delta = aj * (best_mult - 1.0)
                lo = j
                hi = min(j + n - 1, 2 * n - 2)
                k_range = np.arange(lo, hi + 1)
                c[lo:hi+1] += 2.0 * delta * a[k_range - j]
                if 2 * j >= lo and 2 * j <= hi:
                    c[2 * j] += delta * delta
                a[j] *= best_mult
                improved = True
        
        # Renormalize
        s = a.sum()
        if s > 1e-10:
            # Scaling a scales c by s^2... wait no.
            # If a -> a/s, then c -> c/s^2
            # But we want to keep c consistent with a
            # Actually, let's just renormalize a and recompute c
            a = a / s
            c = _fft_convolve(a).astype(np.float64)

        if score < best_score - 1e-9:
            best_score = score
            best_a = a.copy()
        
        if not improved:
            break
        
        if time.monotonic() - last_print > 5.0:
            print(f"      CD sweep {sweep}, score={score:.8f}")
            last_print = time.monotonic()
        sweep += 1
    
    return best_a, best_score

Minimize `t` subject to:
- `sum_i a[i] * a[k-i] <= t` for all `k`
- `a[i] >= 0`
- `sum(a) = 1` (normalization)

This is a convex optimization problem (quadratic constraints, linear constraints)! I can use scipy or cvxpy to solve it.

But the number of constraints is `2n-1` which is ~4000 for n=2000. And the number of variables is `n` = 2000. This is a QCQP (quadratically constrained quadratic program) with ~4000 constraints.

Actually, it's not quite convex because the constraints `sum_i a[i]*a[k-i] <= t` are quadratic in `a`. But for fixed `t`, the feasible set is... hmm, the set `{a: (a*a)[k] <= t for all k, a >= 0}` is an intersection of ellipsoidal regions, which is non-convex in general.

Wait, `(a*a)[k] = sum_i a[i]*a[k-i]`. For a fixed `k`, this is a quadratic form: `a^T Q_k a` where `Q_k` is a symmetric matrix with `Q_k[i, k-i] = Q_k[k-i, i] = 1/2` (and `Q_k[i,i] = 1` if `2i=k`). So we can't directly use a convex solver. But we can use it as a subproblem in an iterative method.

Minimizes 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences a.

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]

def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a)
    return 2.0 * n * float(np.max(c)) / (s * s)

def _subgrad(a, c, k_idx):
    """Compute (unnormalized) subgradient at specified peak indices, weighted by peak values."""
    n = len(a)
    weights = c[k_idx].copy()
    wsum = weights.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    weights /= wsum
    g = np.zeros(n)
    for w, k in zip(weights, k_idx):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            j = np.arange(lo, hi + 1)
            g[j] += w * 2.0 * a[k - j]
    return g

def _sgd(a0, t_budget, rng, top_k=3, eta0=0.1, mom0=0.8):
    n = len(a0)
    a = np.maximum(a0, 1e-15).astype(np.float64)
    a /= a.sum()
    c = _fft_convolve(a)
    best = 2.0 * n * c.max()
    best_a = a.copy()
    eta = eta0
    mom = mom0
    gm = np.zeros(n)
    ni = 0
    dl = time.monotonic() + t_budget
    it = 0
    lp = time.monotonic()
    
    while time.monotonic() < dl:
        c = _fft_convolve(a)
        score = 2.0 * n * c.max()
        if score < best - 1e-9:
            best = score
            best_a = a.copy()
            ni = 0
            if time.monotonic() - lp > 5.0:
                print(f"        it={it} best={best:.8f}")
                lp = time.monotonic()
        else:
            ni += 1
        
        if ni > 500:
            ns = 0.02 + 0.03 * min(ni / 3000.0, 1.0)
            a = best_a * np.exp(rng.normal(0, ns, n))
            a = np.maximum(a, 1e-15)
            a /= a.sum()
            ni = 0
            eta = eta0
            gm = np.zeros(n)
            it += 1
            continue
        
        tk = min(top_k, 2 * n - 1)
        ki = np.argpartition(c, -tk)[-tk:]
        g = _subgrad(a, c, ki)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        gm = mom * gm + (1.0 - mom) * g
        
        if ni > 300:
            eta *= 0.95
        elif ni > 100:
            eta *= 0.98
        elif ni < 20:
            eta = min(eta * 1.003, 0.3)
        
        a *= np.exp(-eta * gm)
        a = np.maximum(a, 1e-15)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            eta = max(eta * 0.5, 0.01)
        else:
            a /= s
        it += 1
    
    return best_a, best

                break
            aj = a[j]
            if aj < 1e-14:
                continue
            cm = c.max()
            bm = cm
            bmu = 1.0

                d = aj * (mu - 1.0)
                lo = j
