def _project_simplex(v):

    n = v.size
    if n == 0:

    css = np.cumsum(x)
    idx = np.arange(1, n+1, dtype=np.float64)
    cond = x > (css - 1.0) / idx
    if not np.any(cond):
        return np.full(n, 1.0/n)
    r = int(np.nonzero(cond)[0][-1])
    tau = (css[r] - 1.0) / float(r + 1)
    out = np.maximum(v - tau, 0.0)
    s = float(out.sum())
    if s <= 1e-15:
        return np.full(n, 1.0/n)
    return out / s

    v = np.asarray(v, dtype=np.float64)

    return np.maximum(v, 0.0)


def _normalize(v):
    v = _clean(v)
    s = float(v.sum())
    n = v.size
    if s <= 1e-15 or n == 0:
        return np.full(n, 1.0 / n) if n > 0 else v
    return v / s


def _project_simplex(v):
    v = _clean(v)
    n = v.size
    if n == 0:
        return v.copy()
    x = np.sort(v)[::-1]
    css = np.cumsum(x)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = x > (css - 1.0) / idx
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    r = int(np.nonzero(cond)[0][-1])
    tau = (css[r] - 1.0) / float(r + 1)
    out = np.maximum(v - tau, 0.0)
    s = float(out.sum())
    if s <= 1e-15:
        return np.full(n, 1.0 / n)
    return out / s


def _next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _autocorr(a, L=None):
    """Compute linear autocorrelation via FFT."""
    v = _normalize(a)
    n = v.size
    if L is None:
        L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(v, n=L)
    c = np.fft.irfft(A * A, n=L)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score(a, L=None):
    c = _autocorr(a, L)
    n = a.size
    mx = float(c.max())
    return 2.0 * float(n) * mx


def _subgradient(a, c, top_frac=0.05, gamma=8.0):
    """
    Compute subgradient direction for max(c_k).
    Returns gradient vector of length n (tangential to simplex).
    """
    n = a.size
    mx = float(c.max())
    if mx <= 1e-15:
        return np.zeros(n)

    # Weight active lags with power decay
    ratio = c / mx
    w = np.power(ratio, gamma)
    w_sum = float(w.sum())
    if w_sum < 1e-15:
        # Fallback: uniform over top_frac
        thresh = np.quantile(c, 1.0 - top_frac)
        w = (c >= thresh).astype(np.float64)
        w_sum = float(w.sum())
        if w_sum < 1:
            idx = np.argmax(c)
            w = np.zeros(2 * n - 1)
            w[idx] = 1.0
            w_sum = 1.0
    w /= w_sum

    # Gradient: g_j = sum_k w_k * (a_{j+k} + a_{j-k})
    # Using FFT: g = 2 * ifft(rfft(w, L) * conj(rfft(a, L)))[:n]
    L = _next_pow2(3 * n)
    W = np.fft.rfft(w, n=L)
    A = np.fft.rfft(a, n=L)
    g = 2.0 * np.fft.irfft(W * np.conj(A), n=L)[:n]

    # Make tangential to simplex
    g -= float(np.mean(g))
    return g


def _nesterov_subgrad(a0, end_time, seed=0):
    """Nesterov-accelerated subgradient descent on exact max autocorrelation."""
    a = _project_simplex(a0.copy())
    n = a.size
    Lscore = _next_pow2(2 * n - 1)

    best = a.copy()
    best_score = _score(best, Lscore)
    print(f"  [Nesterov] init score={best_score:.6f}", flush=True)

    rng = np.random.default_rng(seed)
    t = 1
    lr_base = 0.008
    last_improve = time.monotonic()
    best_streak = 0

    # Adaptive parameters
    gamma = 12.0  # Sharpness of active set weighting
    top_frac = 0.03

    while time.monotonic() < end_time:
        elapsed = time.monotonic() - last_improve
        # Widen active set and smooth if stuck
        if elapsed > 8.0:
            gamma = max(2.0, gamma * 0.85)
            top_frac = min(0.2, top_frac * 1.15)
        else:
            gamma = min(20.0, gamma * 1.01)
            top_frac = max(0.02, top_frac * 0.99)

        c = _autocorr(a, Lscore)
        g = _subgradient(a, c, top_frac, gamma)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break

        # Nesterov learning rate: lr_t = lr_base / sqrt(t)
        lr = lr_base / (np.sqrt(t) + 1.0)

        # Step
        a_new = _project_simplex(a - lr * g)

        # Nesterov momentum update
        beta_t = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum_factor = (beta_t - 1.0) / beta_t
        y = a_new + momentum_factor * (a_new - a)
        a = _project_simplex(y)
        t += 1

        sc = _score(a, Lscore)
        if sc < best_score - 1e-10:
            best_score = sc
            best = a.copy()
            last_improve = time.monotonic()
            best_streak += 1
            if best_streak % 50 == 0:
                print(f"  [Nesterov] t={t} score={best_score:.6f}", flush=True)
        else:
            if best_streak > 0:
                best_streak = 0

        # Periodic escape: if stuck for a long time
        if elapsed > 20.0 and t > 200:
            # Multiplicative perturbation
            noise = rng.normal(0.0, 0.08, size=n)
            a = _project_simplex(best * np.exp(noise))
            sc_check = _score(a, Lscore)
            if sc_check > best_score * 1.05:
                a = best.copy()
            last_improve = time.monotonic()
            t = max(1, t // 3)  # Reset momentum partially
            lr_base = min(0.02, lr_base * 1.3)

    return best, best_score


def _freq_flatten_step(a, strength=0.3, n_iters=5):
    """
    Frequency-domain flattening: compress dynamic range of |A|^2.
    This tends to flatten the autocorrelation.
    """
    a = _normalize(a.copy())
    n = a.size
    L = _next_pow2(2 * n)

    for _ in range(n_iters):
        A = np.fft.rfft(a, n=L)
        mag = np.abs(A)
        phase = np.angle(A)

        # Compress magnitude: mag_new = mag^beta where beta < 1
        mag_max = float(mag.max())
        if mag_max < 1e-15:
            break
        norm_mag = mag / mag_max
        beta = 1.0 - strength  # e.g., 0.7
        new_mag = np.power(norm_mag, beta) * mag_max

        # Blend: keep some of original
        new_mag = (1.0 - strength * 0.5) * mag + strength * 0.5 * new_mag

        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, n=L)[:n]

        # Project
        a = _project_simplex(a_new)

    return a


def _lgbfgs_descent(a0, end_time, seed=0):
    """L-BFGS-B on log-sum-exp smoothed max."""
    a = _project_simplex(a0.copy())
    n = a.size
    Lscore = _next_pow2(2 * n - 1)
    best = a.copy()
    best_score = _score(best, Lscore)

    if _scipy_minimize is None:
        return best, best_score

    # Work in log-ratio parameterization
    ref = float(a[-1])
    if ref < 1e-20:
        a[-1] = 1e-12
        a = _normalize(a)
        ref = float(a[-1])
    x0 = np.log(np.maximum(a[:-1], 1e-300) / ref)
    x0 = np.clip(x0, -80.0, 80.0)
    bounds = [(-80.0, 80.0)] * (n - 1)

    def fun_and_grad(x, eps_val):
        x = np.where(np.isfinite(x), x, 0.0)
        x = np.clip(x, -80.0, 80.0)
        m = float(np.max(x)) if x.size else 0.0
        e = np.exp(np.clip(x - m, -700.0, 0.0))
        e_last = np.exp(-m)
        s = float(e.sum()) + e_last
        if s < 1e-300:
            return 1e100, np.zeros_like(x)
        a_rec = np.empty(n)
        a_rec[:-1] = e / s
        a_rec[-1] = e_last / s

        L = _next_pow2(3 * n)
        A = np.fft.rfft(a_rec, n=L)
        c = np.fft.irfft(A * A, n=L)[:2 * n - 1]
        c = np.maximum(c, 0.0)
        cmax = float(c.max())
        if cmax < 1e-30:
            return 0.0, np.zeros_like(x)

        eps = eps_val
        z = (c - cmax) / eps
        z = np.clip(z, -700.0, 0.0)
        expz = np.exp(z)
        sume = float(expz.sum())
        if sume < 1e-300:
            return cmax, np.zeros_like(x)

        f = cmax + eps * np.log(sume)
        w = expz / sume

        W = np.fft.rfft(w, n=L)
        ga = 2.0 * np.fft.irfft(W * np.conj(A), n=L)[:n]
        dot = float(np.dot(a_rec, ga))
        gfull = a_rec * (ga - dot)
        gy = gfull[:-1]
        if not np.all(np.isfinite(gy)):
            gy = np.zeros_like(x)
        return f, gy

    # Anneal epsilon from moderate to very small
    eps_schedule = [0.05, 0.01, 0.003, 0.001, 0.0003, 0.0001, max(1e-6, 0.0001)]
    x = x0.copy()
    rng = np.random.default_rng(seed)

    for eps_val in eps_schedule:
        if time.monotonic() >= end_time:
            break
        eps_end = min(end_time, time.monotonic() + max(5.0, (end_time - time.monotonic()) / len(eps_schedule)))
        while time.monotonic() < eps_end:
            try:
                res = _scipy_minimize(
                    lambda xx: fun_and_grad(xx, eps_val),
                    x,
                    method='L-BFGS-B',
                    jac=True,
                    bounds=bounds,
                    options={'maxiter': 100, 'maxfun': 500, 'ftol': 1e-16, 'gtol': 1e-10, 'disp': 0},
                )
                x = np.clip(np.where(np.isfinite(res.x), res.x, x), -80.0, 80.0)
            except Exception:
                break

        # Evaluate
        a_rec = _ratios_to_simplex(x, ref)
        sc = _score(a_rec, Lscore)
        if sc < best_score - 1e-12:
            best_score = sc
            best = a_rec.copy()
            print(f"  [L-BFGS] eps={eps_val:.6f} score={best_score:.6f}", flush=True)
        else:
            # Add small perturbation to escape
            x = np.clip(x + rng.normal(0, 0.02, size=x.size), -80.0, 80.0)

    return best, best_score


def _ratios_to_simplex(x, ref):
    x = np.clip(np.where(np.isfinite(x), x, 0.0), -80.0, 80.0)
    n = x.size + 1
    m = float(np.max(x)) if x.size else 0.0
    e = np.exp(np.clip(x - m, -700.0, 0.0))
    e_last = np.exp(-m)
    s = float(e.sum()) + e_last
    if s < 1e-300:
        return np.full(n, 1.0 / n)
    a = np.empty(n)
    a[:-1] = e / s
    a[-1] = e_last / s
    return a


def _local_search(a0, end_time, seed=0):
    """Block-based local search with diverse perturbation patterns."""
    a = _project_simplex(a0.copy())
    n = a.size
    Lscore = _next_pow2(2 * n - 1)
    best = a.copy()
    best_score = _score(best, Lscore)
    print(f"  [Local] init score={best_score:.6f}", flush=True)

    rng = np.random.default_rng(seed)
    sigma = 0.10
    last_improve = time.monotonic()
    start_time = time.monotonic()

    while time.monotonic() < end_time:
        elapsed_since_best = time.monotonic() - last_improve

        # Choose perturbation strategy
        strat = rng.integers(0, 4)
        cand = best.copy()

        if strat == 0:
            # Block multiplicative
            nblocks = 1 + int(rng.integers(0, 3))
            for _ in range(nblocks):
                w = int(np.exp(rng.uniform(np.log(2.0), np.log(n // 4 + 1.0))))
                w = max(1, min(w, n))
                s = int(rng.integers(0, max(1, n - w + 1)))
                cand[s:s + w] *= np.exp(rng.normal(0.0, sigma))
        elif strat == 1:
            # Paired swap (move mass from one region to another)
            m = int(min(n // 5, 200))
            s1 = int(rng.integers(0, n - m))
            s2 = int(rng.integers(0, n - m))
            if s1 != s2:
                amt = float(cand[s1:s1 + m].sum()) * rng.uniform(0.05, 0.4)
                ratio = amt / max(float(cand[s1:s1 + m].sum()), 1e-15)
                cand[s1:s1 + m] *= (1.0 - ratio)
                extra = amt / max(float(cand[s2:s2 + m].sum()), 1e-15)
                cand[s2:s2 + m] *= (1.0 + extra)
        elif strat == 2:
            # Frequency-domain flattening
            cand = _freq_flatten_step(best, strength=rng.uniform(0.1, 0.4), n_iters=3)
        else:
            # Symmetry operation + noise
            cand = 0.5 * best + 0.5 * best[::-1]
            cand *= np.exp(rng.normal(0.0, sigma * 0.5, size=n))

        cand = _project_simplex(cand)
        sc = _score(cand, Lscore)

        if sc < best_score - 1e-12:
            best = cand
            best_score = sc
            last_improve = time.monotonic()
            sigma = min(0.30, sigma * 1.04)
            if best_score < 1.52:
                print(f"  [Local] score={best_score:.6f} (sigma={sigma:.3f})", flush=True)
        else:
            if elapsed_since_best > 10.0:
                sigma = max(0.015, sigma * 0.8)
                if elapsed_since_best > 30.0 and time.monotonic() - start_time > 60.0:
                    # Reset from best with larger perturbation
                    cand = _project_simplex(best * np.exp(rng.normal(0.0, 0.15, size=n)))
                    sc2 = _score(cand, Lscore)
                    if sc2 < best_score:
                        best = cand
                        best_score = sc2
                        last_improve = time.monotonic()

    return best, best_score


def _generate_diverse_candidates(n, base_seq):
    """Generate diverse starting candidates including the base sequence."""
    cands = []
    base = _normalize(base_seq)
    cands.append(base.copy())

    # Reversed
    cands.append(base[::-1].copy())

    # Symmetrized
    cands.append(0.5 * base + 0.5 * base[::-1])

    # Power-law variants
    for p in (0.4, 0.45, 0.5, 0.55, 0.6, 0.7):
        j = np.arange(n, dtype=np.float64)
        v = np.power(j + 1.0, -p)
        cands.append(_project_simplex(v))

    # Two-sided exponential
    mid = n / 2.0
    for rate in (0.5, 1.0, 1.5, 2.0, 3.0):
        j = np.arange(n, dtype=np.float64)
        v = np.exp(-rate * np.abs(j - mid) / mid)
        cands.append(_project_simplex(v))

    # One-sided exponential
    for rate in (0.5, 1.0, 1.5, 2.0):
        j = np.arange(n, dtype=np.float64)
        v = np.exp(-rate * j / n)
        cands.append(_project_simplex(v))

    # Cosine window
    j = np.arange(n, dtype=np.float64)
    v = np.cos(np.pi * j / (2.0 * (n - 1.0))) ** 2
    cands.append(_project_simplex(v))

    # Random (seeded)
    for seed in (7, 13, 42, 99):
        rng = np.random.default_rng(seed)
        v = np.exp(rng.normal(0, 0.8, size=n))
        cands.append(_project_simplex(v))

    return cands


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Returns best non-negative sequence found.
    """
    t0 = time.monotonic()
    total_budget = max(0.0, float(budget_s) - 15.0)
    deadline = t0 + total_budget

    # Determine sequence length
    n = 3809
    try:
        h = height_sequence_1
        if isinstance(h, (list, np.ndarray)):
            arr = np.asarray(h, dtype=np.float64)
            if arr.ndim == 1 and 100 <= arr.size <= 50000:
                n = int(arr.size)
    except Exception:
        pass

    if n < 32:
        n = 3809

    print(f"[Main] n={n}, budget={total_budget:.1f}s", flush=True)

    # If very little time, return a good default
    if total_budget < 30.0:
        try:
            h = np.asarray(height_sequence_1, dtype=np.float64)
            h = _clean(h)
            if h.size == n and h.sum() > 0.01:
                return [float(x) for x in h]
        except Exception:
            pass
        j = np.arange(n, dtype=np.float64)
        a = _project_simplex(np.power(j + 1.0, -0.5))
        return [float(x) for x in a]

    # Generate candidates and find best initial
    cands = _generate_diverse_candidates(n, np.asarray(height_sequence_1, dtype=np.float64) if 'height_sequence_1' in dir() else np.ones(n))

    # Also try to load height_sequence_1
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = _clean(h)
        if h.size == n and h.sum() > 0.01:
            cands.insert(0, h)
    except Exception:
        pass

    # Score all candidates
    scored = []
    for i, c in enumerate(cands):
        sc = _score(c)
        if np.isfinite(sc):
            scored.append((sc, i, c))
    scored.sort()

    print(f"[Main] Best initial: score={scored[0][0]:.6f}", flush=True)
    global_best = scored[0][2].copy()
    global_best_score = scored[0][0]

    # Time allocation
    # Phase 1: Nesterov subgradient (45% of time)
    # Phase 2: L-BFGS-B log-sum-exp (30% of time)
    # Phase 3: Local search (25% of time)
    p1_end = t0 + 0.45 * total_budget
    p2_end = t0 + 0.75 * total_budget
    p3_end = deadline

    # Phase 1: Nesterov subgradient on top-2 candidates
    for idx in range(min(2, len(scored))):
        if time.monotonic() >= p1_end:
            break
        a_start = scored[idx][2].copy()
        sub_end = min(p1_end, t0 + (0.45 * total_budget) * (0.6 if idx == 0 else 0.4))
        a, sc = _nesterov_subgrad(a_start, sub_end, seed=int(seed) + 100 * (idx + 1))
        if sc < global_best_score:
            global_best_score = sc
            global_best = a.copy()
            print(f"[Main] Phase1 idx={idx}: {sc:.6f}", flush=True)

    # Phase 2: L-BFGS-B
    if time.monotonic() < p2_end:
        # Start from global best and top initial
        starts = [global_best.copy()]
        if len(scored) > 0:
            starts.append(scored[0][2].copy())

        for idx, a_start in enumerate(starts):
            sub_end = p2_end
            a, sc = _lgbfgs_descent(a_start, sub_end, seed=int(seed) + 500 * (idx + 1))
            if sc < global_best_score:
                global_best_score = sc
                global_best = a.copy()
                print(f"[Main] Phase2 idx={idx}: {sc:.6f}", flush=True)

    # Phase 3: Local search
    if time.monotonic() < p3_end:
        a, sc = _local_search(global_best, p3_end, seed=int(seed) + 9000)
        if sc < global_best_score:
            global_best_score = sc
            global_best = a.copy()
            print(f"[Main] Phase3: {sc:.6f}", flush=True)

    # Final: one more short L-BFGS-B polish
    if time.monotonic() < deadline - 2.0:
        remaining = deadline - time.monotonic() - 2.0
        if remaining > 10.0:
            a, sc = _lgbfgs_descent(global_best, deadline - 2.0, seed=int(seed) + 7777)
            if sc < global_best_score:
                global_best_score = sc
                global_best = a.copy()
                print(f"[Main] Final polish: {sc:.6f}", flush=True)

    global_best = _project_simplex(global_best)
    final_score = _score(global_best)
    print(f"[Main] FINAL: {final_score:.6f}", flush=True)

    # Validate with evaluate_sequence
    out = [float(x) for x in global_best]
    try:
        val = evaluate_sequence(out)
        if not np.isfinite(val):
            j = np.arange(n, dtype=np.float64)
            out = [float(x) for x in _project_simplex(np.power(j + 1.0, -0.5))]
            print(f"[Main] Fallback used, evaluate would give inf", flush=True)
        else:
            print(f"[Main] evaluate_sequence confirms: {val:.6f}", flush=True)
    except Exception as e:
        print(f"[Main] evaluate check failed: {e}", flush=True)

    return out
