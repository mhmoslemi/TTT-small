"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using mirror descent with smoothed max,
projected gradient descent, and LP-based refinement with constraint generation.
"""

import numpy as np
import time
import math

try:
    from scipy.optimize import linprog as _linprog
except ImportError:
    _linprog = None


def _pow2(m):
    """Smallest power of 2 >= m."""
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _clean(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    return np.maximum(v, 0.0)


def _normalize(v):
    v = _clean(v)
    s = float(v.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.full(v.size, 1.0 / max(1, v.size), dtype=np.float64)
    return v / s


def _project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    v = _clean(v).copy()
    n = v.size
    if n == 0:
        return v
    x = np.sort(v)[::-1]
    css = np.cumsum(x)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = x > (css - 1.0) / idx
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    r = int(np.nonzero(cond)[0][-1])
    tau = (css[r] - 1.0) / (r + 1)
    out = v - tau
    np.maximum(out, 0.0, out=out)
    s = out.sum()
    if s <= 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def _autocorr(a, L=None):
    """Compute full autocorrelation a*a of length 2n-1."""
    n = a.size
    if L is None:
        L = _pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score(a):
    """Compute 2*n*max(a*a) assuming sum(a)=1."""
    a = _clean(a)
    s = a.sum()
    if s <= 1e-12:
        return np.inf
    a = a / s
    c = _autocorr(a)
    return 2.0 * a.size * c.max()


def _grad_smoothed_max(a, t, L=None):
    """
    Gradient of t * logsumexp(autocorr(a) / t) w.r.t. a.
    Returns gradient vector of length n.
    """
    n = a.size
    if L is None:
        L = _pow2(2 * n - 1)
    
    A = np.fft.rfft(a, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    
    cmax = c.max()
    if cmax <= 1e-300:
        return np.zeros(n, dtype=np.float64)
    
    # Softmax: p_k = exp(c_k/t) / sum(exp(c/t))
    z = (c - cmax) / t
    z = np.clip(z, -745.0, 0.0)
    e = np.exp(z)
    se = e.sum()
    if se <= 0:
        return np.zeros(n, dtype=np.float64)
    p = e / se
    
    # Gradient: g = 2 * ifft(FFT(p) * conj(FFT(a)))[:n]
    P = np.fft.rfft(p, n=L)
    g_full = np.fft.irfft(P * np.conj(A), n=L)[:n]
    g = 2.0 * g_full
    
    # Ensure no NaN/Inf
    g = np.where(np.isfinite(g), g, 0.0)
    return g


def _mirror_descent(a0, end_time, seed, t_init=None, t_min=1e-6, eta_init=1.0):
    """
    Mirror descent (multiplicative weights) on the smooth max objective.
    """
    a = _normalize(a0)
    n = a.size
    L = _pow2(2 * n - 1)
    
    best = a.copy()
    best_score = _score(best)
    
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    total = max(1.0, end_time - start)
    
    # Initial temperature
    c0 = _autocorr(a, L)
    cmax0 = c0.max()
    if t_init is None:
        t_init = cmax0 / 10.0
    t = t_init
    
    # Score-based: we want t to decrease so that t * logsumexp(c/t) -> max(c)
    # The "effective" t in the score 2*n*max(c) corresponds to t -> 0
    # Let's parameterize by score: t = alpha * cmax * (something)
    # Actually, let's just use a geometric schedule based on fraction of time
    
    last_improve = start
    last_improve_score = best_score
    patience = 30.0  # seconds without improvement before reducing eta
    
    eta = eta_init
    iters = 0
    
    while time.monotonic() < end_time:

        frac = min(1.0, elapsed / total)

        t = t_init * (t_min / t_init) ** frac

        weights = np.exp(log_step)
        a_new = a * weights
        s = a_new.sum()
        if s <= 0 or not np.isfinite(s):
            a_new = a.copy()
        else:
            a_new = a_new / s
            a_new = _clean(a_new)
            s2 = a_new.sum()
            if s2 > 0:
                a_new = a_new / s2
        
        a = a_new
        iters += 1
        
        # Check score (every few iterations to save time)

            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best = a.copy()
                last_improve = time.monotonic()
                # Slightly increase eta on improvement

def _clean(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    return np.maximum(v, 0.0)


def _normalize(v):
    v = _clean(v)
    s = float(v.sum())
    if not np.isfinite(s) or s <= 1e-15:
        n = v.size
        return np.full(n, 1.0 / max(1, n), dtype=np.float64)
    return v / s


def _project_simplex(v):
    v = _clean(v).copy()
    n = v.size
    if n == 0:
        return v
    x = np.sort(v)[::-1]
    css = np.cumsum(x)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = x > (css - 1.0) / idx
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    r = int(np.nonzero(cond)[0][-1])
    tau = (css[r] - 1.0) / (r + 1)
    out = np.maximum(v - tau, 0.0)
    s = out.sum()
    if s <= 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def _autocorr(a, L=None):
    n = a.size
    if L is None:
        L = _pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    c = np.fft.irfft(A * A, n=L)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score_norm(a, L=None):
    """Score assuming sum(a)=1."""
    n = a.size
    if L is None:
        L = _pow2(2 * n - 1)
    c = _autocorr(a, L)
    return 2.0 * n * float(c.max())


def _score(a):
    a = _clean(a)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return _score_norm(a / s)


def _md_step(a, t, L=None):
    """
    One mirror descent step. Returns (new_a, cmax, gmax).
    Computes gradient of t*logsumexp(c/t) where c = a*a.
    """
    n = a.size
    if L is None:
        L = _pow2(2 * n - 1)

    A = np.fft.rfft(a, n=L)
    c = np.fft.irfft(A * A, n=L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    cmax = float(c.max())

    if cmax <= 1e-300:
        return a.copy(), 0.0, 0.0

    # Softmax
    z = (c - cmax) / t
    z = np.clip(z, -700.0, 0.0)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0:
        return a.copy(), cmax, 0.0
    p = e / se

    # Gradient: 2 * ifft(FFT(p) * conj(FFT(a)))[:n]
    P = np.fft.rfft(p, n=L)
    g = 2.0 * np.fft.irfft(P * np.conj(A), n=L)[:n]
    g = np.where(np.isfinite(g), g, 0.0)
    gmax = float(np.abs(g).max())

    if gmax <= 1e-12:
        return a.copy(), cmax, 0.0

    # Adaptive learning rate: keep step size moderate
    eta = 0.4 / gmax

    # Multiplicative weights update
    log_step = np.clip(-eta * g, -40.0, 40.0)
    w = np.exp(log_step)
    a_new = a * w
    s = a_new.sum()
    if s <= 0 or not np.isfinite(s):
        a_new = a.copy()
    else:
        a_new = a_new / s
        a_new = _clean(a_new)
        s2 = a_new.sum()
        if s2 > 0:
            a_new = a_new / s2

    return a_new, cmax, gmax


def _mirror_descent(a0, end_time, seed, K_start=12.0, K_end=5000.0):
    """Mirror descent with smoothed max, adaptive temperature."""
    a = _normalize(a0)
    n = a.size
    L = _pow2(2 * n - 1)

    best = a.copy()
    best_score = 2.0 * n * float(_autocorr(best, L).max())

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    total = max(1.0, end_time - start)

    a_avg = a.copy()
    avg_alpha = 0.08
    last_improve = start
    patience = 20.0
    iters = 0

    while time.monotonic() < end_time:
        elapsed = max(0.0, time.monotonic() - start)
        frac = min(1.0, elapsed / total)

        # Temperature: K = K_start * (K_end/K_start)^frac
        # t = cmax / K
        A = np.fft.rfft(a, n=L)
        c = np.fft.irfft(A * A, n=L)[:2 * n - 1]
        c = np.maximum(c, 0.0)
        cmax = float(c.max())

        if cmax <= 1e-300:
            break

        K = K_start * (K_end / K_start) ** frac
        t = max(cmax / K, 1e-10)

        # Softmax
        z = (c - cmax) / t
        z = np.clip(z, -700.0, 0.0)
        e = np.exp(z)
        se = float(e.sum())
        if se <= 0:
            break
        p = e / se

        # Gradient
        P = np.fft.rfft(p, n=L)
        g = 2.0 * np.fft.irfft(P * np.conj(A), n=L)[:n]
        g = np.where(np.isfinite(g), g, 0.0)
        gmax = float(np.abs(g).max())

        if gmax <= 1e-12:
            break

        # Learning rate
        eta = 0.4 / gmax

        # Mirror descent step
        log_step = np.clip(-eta * g, -40.0, 40.0)
        w = np.exp(log_step)
        a_new = a * w
        s = a_new.sum()
        if s <= 0 or not np.isfinite(s):
            a_new = a.copy()
        else:
            a_new = a_new / s
            a_new = _clean(a_new)
            s2 = a_new.sum()
            if s2 > 0:
                a_new = a_new / s2

        a = a_new
        a_avg = (1.0 - avg_alpha) * a_avg + avg_alpha * a
        iters += 1

        # Score check every 5 iterations
        if iters % 5 == 0:
            c_new = _autocorr(a, L)
            sc = 2.0 * n * float(c_new.max())

            c_avg = _autocorr(a_avg, L)
            sc_avg = 2.0 * n * float(c_avg.max())

            improved = False
            if sc < best_score - 1e-10:
                best_score = sc
                best = a.copy()
                improved = True
            if sc_avg < best_score - 1e-10:
                best_score = sc_avg
                best = _normalize(a_avg)
                improved = True

            if improved:
                last_improve = time.monotonic()

        # Stagnation
        if time.monotonic() - last_improve > patience:
            # Check average
            sc_avg = 2.0 * n * float(_autocorr(a_avg, L).max())
            if sc_avg < best_score:
                best = _normalize(a_avg)
                best_score = sc_avg
                last_improve = time.monotonic()

            # Perturb
            noise_std = min(0.1 + 0.05 * (patience / 20.0), 0.4)
            noise = np.exp(rng.normal(0, noise_std, n))
            a = _normalize(a * noise)
            a_avg = a.copy()
            patience = min(patience * 1.5, 80.0)

    return best, best_score


def _proj_grad(a0, end_time, seed):
    """Projected gradient descent on the smoothed max."""
    a = _normalize(a0)
    n = a.size
    L = _pow2(2 * n - 1)

    best = a.copy()
    best_score = 2.0 * n * float(_autocorr(best, L).max())

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    total = max(1.0, end_time - start)

    lr = 0.01
    last_improve = start
    patience = 15.0
    iters = 0

    while time.monotonic() < end_time:
        elapsed = max(0.0, time.monotonic() - start)
        frac = min(1.0, elapsed / total)

        A = np.fft.rfft(a, n=L)
        c = np.fft.irfft(A * A, n=L)[:2 * n - 1]
        c = np.maximum(c, 0.0)
        cmax = float(c.max())

        if cmax <= 1e-300:
            break

        score = 2.0 * n * cmax
        if score < best_score - 1e-10:
            best_score = score
            best = a.copy()
            last_improve = time.monotonic()
            lr = min(0.05, lr * 1.05)

        # Temperature for smoothed max
        K = 10.0 * (500.0 / 10.0) ** frac
        t = max(cmax / K, 1e-10)

        z = (c - cmax) / t
        z = np.clip(z, -700.0, 0.0)
        e = np.exp(z)
        se = float(e.sum())
        p = e / se

        P = np.fft.rfft(p, n=L)
        g = 2.0 * np.fft.irfft(P * np.conj(A), n=L)[:n]
        g = np.where(np.isfinite(g), g, 0.0)

        # Project gradient onto tangent space of simplex
        g = g - g.mean()

        a_new = _project_simplex(a - lr * g)
        a = a_new
        iters += 1

        if time.monotonic() - last_improve > patience:
            lr *= 0.6
            if time.monotonic() - last_improve > patience * 3:
                noise = np.exp(rng.normal(0, 0.1, n))
                a = _project_simplex(a * noise)
                patience = 15.0

    return best, best_score


def _lp_direction(a, max_constr=3000, max_iters=15):
    """
    LP: max sum(b) s.t. conv(a,b)(k) <= max(a*a) for all k, b >= 0.
    Uses constraint generation. Returns normalized b or None.
    """
    if not _HAS_SCIPY:
        return None

    n = a.size
    M = float(_autocorr(a).max())

    if M <= 0:
        return None

    # Initial active constraints: top M values
    c_a = _autocorr(a)
    active_k = np.argsort(c_a)[::-1][:min(max_constr, 2000)].tolist()

    for _ in range(max_iters):
        ak = np.array(active_k, dtype=np.int64)
        # Build sparse constraint matrix
        j_starts = np.maximum(0, ak - n + 1)
        j_ends = np.minimum(ak, n - 1)
        lengths = j_ends - j_starts + 1
        total_nnz = int(lengths.sum())

        if total_nnz == 0:
            return None

        row_idx = np.repeat(np.arange(len(ak)), lengths)
        col_idx = np.concatenate([np.arange(s, e + 1) for s, e in zip(j_starts, j_ends)])
        val_idx = np.repeat(ak, lengths) - col_idx
        vals = a[val_idx]

        T = _csr_matrix((vals, (row_idx, col_idx)), shape=(len(ak), n))

        # LP: max 1^T b s.t. T b <= M, b >= 0
        try:
            res = _linprog(
                -np.ones(n),
                A_ub=T,
                b_ub=np.full(len(ak), M),
                bounds=[(0, None)] * n,
                method='highs',
                options={'time_limit': 30}
            )
        except Exception:
            return None

        if res is None or not res.success:
            return None

        b = np.maximum(res.x, 0.0)
        b_sum = float(b.sum())
        if b_sum <= 1e-12:
            return None

        # Check all constraints
        L = _pow2(2 * n - 1)
        A_fft = np.fft.rfft(a, n=L)
        B_fft = np.fft.rfft(b, n=L)
        cross = np.fft.irfft(A_fft * B_fft, n=L)[:2 * n - 1]
        cross = np.maximum(cross, 0.0)

        violated = np.where(cross > M * 1.0001)[0]
        if len(violated) == 0:
            return b / b_sum

        # Add most violated
        new_k = violated[np.argsort(cross[violated])[::-1][:500]]
        for k in new_k:
            if k not in active_k:
                active_k.append(k)

        if len(active_k) > max_constr:
            # Keep top by violation
            cross_act = cross[np.array(active_k, dtype=np.int64)]
            top = np.argsort(cross_act)[::-1][:max_constr]
            active_k = [active_k[i] for i in top.tolist()]

    return None


def _lp_refine(a0, end_time, seed, blend_rate=0.3):
    """Iteratively apply LP direction finding."""
    a = _normalize(a0)
    n = a.size

    best = a.copy()
    best_score = 2.0 * n * float(_autocorr(best).max())

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    iters = 0

    while time.monotonic() < end_time - 5.0:
        b = _lp_direction(a)
        if b is None:
            break

        # Blend
        a_new = _normalize((1.0 - blend_rate) * a + blend_rate * b)
        sc = _score_norm(a_new)

        iters += 1
        if sc < best_score - 1e-10:
            best_score = sc
            best = a_new.copy()
            a = a_new
            # Decrease blend rate over time
            blend_rate = max(0.05, blend_rate * 0.95)
        else:
            # Try smaller blend
            for bt in (0.1, 0.05, 0.02):
                a_try = _normalize((1.0 - bt) * a + bt * b)
                sc_try = _score_norm(a_try)
                if sc_try < best_score - 1e-10:
                    best_score = sc_try
                    best = a_try.copy()
                    a = a_try
                    break
            else:
                # No improvement, perturb
                noise = np.exp(rng.normal(0, 0.05, n))
                a = _normalize(a * noise)
                if iters % 5 == 0:
                    break

    return best, best_score


def _local_search(a0, end_time, seed):
    """Random multiplicative block perturbations."""
    a = _normalize(a0)
    n = a.size
    L = _pow2(2 * n - 1)

    best = a.copy()
    best_score = 2.0 * n * float(_autocorr(best, L).max())

    rng = np.random.default_rng(seed)
    sigma = 0.10
    last_best = time.monotonic()
    start = last_best

    while time.monotonic() < end_time:
        cand = best.copy()
        nblocks = 1 + int(rng.integers(0, 3))

        for _ in range(nblocks):
            maxw = max(3, n // 6)
            w = int(np.exp(rng.uniform(np.log(2.0), np.log(float(maxw)))))
            w = max(1, min(w, n))
            s = int(rng.integers(0, n - w + 1))
            cand[s:s + w] *= float(np.exp(rng.normal(0.0, sigma)))

        cand = _normalize(cand)
        c = _autocorr(cand, L)
        sc = 2.0 * n * float(c.max())

        if sc < best_score - 1e-12:
            best = cand
            best_score = sc
            last_best = time.monotonic()
            sigma = min(0.3, sigma * 1.08)
        else:
            if time.monotonic() - last_best > 8.0:
                sigma = max(0.02, sigma * 0.7)
                if time.monotonic() - last_best > 30.0 and time.monotonic() - start > 60.0:
                    cand = _normalize(best * np.exp(rng.normal(0.0, 0.04, n)))
                    c2 = _autocorr(cand, L)
                    sc2 = 2.0 * n * float(c2.max())
                    if sc2 < best_score:
                        best = cand
                        best_score = sc2
                        last_best = time.monotonic()

    return best, best_score


def _power_seq(n, p=0.5, offset=1.0):
    j = np.arange(n, dtype=np.float64)
    x = np.maximum(j + offset, 1e-8)
    return _normalize(np.power(x, -p))
