sum_{k in S} Q_k is a matrix where (sum Q_k)_{i,j} = |{k in S: i+j=k}|. This counts how many active peaks "see" the pair (i,j). Since the entries are non-negative, the matrix is entrywise non-negative. But is it PSD?

A symmetric matrix with non-negative entries is not necessarily PSD. For example, [[0, 1], [1, 0]] has eigenvalues 1 and -1.

However, in our case, the matrix (sum_{k in S} Q_k) has a special structure: (sum Q_k)_{i,j} = |S ∩ {i+j}|. This is the number of elements of S that equal i+j. Since i+j is a single value, |S ∩ {i+j}| is either 0 or 1 (if i+j ∈ S). So (sum Q_k)_{i,j} = 1_{i+j ∈ S}.

This is a "Hankel-like" matrix (constant along anti-diagonals). Is it PSD?

"""
Multi-phase optimizer for minimizing J(a) = 2n*max(conv(a,a))/sum(a)^2.
Uses alternating LP as the primary engine, with subgradient polish and
perturbation restarts for refinement. Explores multiple sequence lengths
and diverse initializations including asymmetric multi-peak shapes.

import numpy as np
import time
from scipy.optimize import linprog


def _fft_convolve(a):
    """Fast full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]


def _score(a):
    """Compute J(a) = 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)


def _build_conv_matrix(f):
    """Build the convolution matrix C where (C @ g)[k] = (f*g)[k]."""
    n = len(f)
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = f
    return C


def _lp_solve(f, M, matrix=None):
    """Solve max sum(g) s.t. C_f @ g <= M, g >= 0. Returns g or None."""
    n = len(f)
    if matrix is None:
        matrix = _build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c_obj, A_ub=matrix, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True,
                                                   'time_limit': 30})
        if result.status == 0:
            return result.x
    except Exception:
        pass
    return None


def _alternating_lp(a_init, max_iters, deadline, initial_blend=0.7, m_shrink=0.0):
    """Run alternating LP with optional M shrinking for escape from local optima."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = _score(a)
    blend = initial_blend
    stagnant = 0
    last_improve_score = best_score

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        M = float(np.max(c))
        
        # If m_shrink > 0 and we're stagnant, try a tighter M
        M_eff = M
        if m_shrink > 0 and stagnant > 50:
            M_eff = M * (1.0 - m_shrink * min(stagnant, 200) / 200.0)
        
        matrix = _build_conv_matrix(a)
        g = _lp_solve(a, M_eff, matrix)
        if g is None:
            break
        sum_g = float(np.sum(g))
        if sum_g < 1e-12:
            break
        g = g / sum_g
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        sum_new = float(np.sum(a_new))
        if sum_new < 1e-12:
            break
        a_new = a_new / sum_new
        score = _score(a_new)
        
        if score < last_improve_score - 1e-8:
            last_improve_score = score
            stagnant = 0
        else:
            stagnant += 1
        
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            if it % 50 == 0:
                print(f"      LP iter {it}: score={best_score:.6f} (M={M:.6e})")
        a = a_new
        blend = max(0.1, blend * 0.98)

    return best_a, best_score


def _subgradient_polish(a_init, max_iters, lr_init, lr_decay, n_peaks, deadline, momentum=0.9):
    """Subgradient optimization with momentum for flat autocorrelation."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    best_a = a.copy()
    best_score = _score(a)
    lr = lr_init
    v = np.zeros(n)  # momentum
    print(f"    Subgrad start: score={best_score:.6f}")

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        max_c = float(np.max(c))
        score = 2.0 * n * max_c / S * S

        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Get top-k peaks
        if n_peaks >= len(c):
            top_k = np.arange(len(c))
        else:
            top_k = np.argpartition(c, -n_peaks)[-n_peaks:]

        # Compute subgradient
        s = np.zeros(n)
        for k in top_k:
            k = int(k)
            for i in range(n):
                idx = k - i
                if 0 <= idx < n:
                    s[i] += a[idx]
        s *= 2.0 / len(top_k)
        
        # Normalize subgradient
        max_s = float(np.max(s))
        if max_s > 1e-15:
            s = s / max_s
            
            # Momentum update
            v = momentum * v + (1 - momentum) * s
            a = a - lr * v
            a = np.maximum(a, 0.0)
            S_new = float(np.sum(a))
            if S_new > 1e-10:
                a = a * S / S_new
        
        lr *= lr_decay

    print(f"    Subgrad final: score={best_score:.6f}")
    return best_a, best_score


def _fast_subgradient_polish(a_init, max_iters, lr_init, lr_decay, n_peaks, deadline):
    """Vectorized subgradient for speed."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    best_a = a.copy()
    best_score = _score(a)
    lr = lr_init
    v = np.zeros(n)
    print(f"    FastSubgrad start: score={best_score:.6f}")
    
    j = np.arange(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        max_c = float(np.max(c))
        score = 2.0 * n * max_c

        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Get top-k peaks (vectorized sensitivity)
        if n_peaks >= len(c):
            top_k = np.arange(len(c))
        else:
            top_k = np.argpartition(c, -n_peaks)[-n_peaks:]
        
        s = np.zeros(n)
        for k in top_k:
            k = int(k)
            idx = k - j
            mask = (idx >= 0) & (idx < n)
            s += a[idx[mask]] * mask
            # Also the "other" direction
            idx2 = j - k
            mask2 = (idx2 >= 0) & (idx2 < n)
            s += a[idx2[mask2]] * mask2
        
        s /= len(top_k)
        
        max_s = float(np.max(s))
        if max_s > 1e-15:
            s = s / max_s
            v = 0.9 * v + 0.1 * s
            a = a - lr * v
            a = np.maximum(a, 0.0)
            S_new = float(np.sum(a))
            if S_new > 1e-10:
                a = a * S / S_new
        
        lr *= lr_decay

    print(f"    FastSubgrad final: score={best_score:.6f}")
    return best_a, best_score

def _interpolate_sequence(src, n_target):
    """Interpolate a sequence to a new length."""
    n_src = len(src)
    if n_src == n_target:
        return src.copy()
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    out = np.interp(x_tgt, x_src, src)
    out = np.maximum(out, 0.0)
    s = float(np.sum(out))
    if s > 1e-10:
        out = out / s
    return out


def _perturb(a, rng, frac=0.1, scale=0.1):
    """Randomly perturb a fraction of elements."""
    n = len(a)
    a_new = a.copy()
    n_perturb = max(1, int(frac * n))
    indices = rng.choice(n, size=n_perturb, replace=False)
    a_new[indices] *= (1.0 + scale * rng.normal(0, 1, n_perturb))
    a_new = np.maximum(a_new, 0.0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new
