def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def normalize_positive(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s


def fast_score(a):
    p = normalize_positive(a)
    if p is None:
        return np.inf
    n = p.size
    if n <= 0:
        return np.inf
    L = next_pow2(2 * n - 1)
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0.0:
        return np.inf
    return float(2.0 * n * cm)


def softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = np.where(np.isfinite(z), z, -50.0)
    z = z - z.max()
    z = np.clip(z, -50.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        e = np.ones_like(z)
        s = float(e.size)
    return e / s


def smooth_logit_grad(p, tau, L):
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    P = np.fft.rfft(p, L)

    c = np.fft.irfft(P * P, L)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)

    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0.0:
        return np.inf, np.inf, np.zeros(n, dtype=np.float64)

    # Smooth maximum: tau * log(sum exp(c / tau)), shifted for stability.
    z = (c - cm) / tau
    expz = np.exp(z)
    expsum = float(expz.sum())
    if not np.isfinite(expsum) or expsum <= 0.0:
        expz = np.zeros_like(c)
        expz[int(np.argmax(c))] = 1.0
        expsum = 1.0

    value_coeff = cm + tau * float(np.log(expsum))

    # d/dp_i of the smooth coefficient objective.
    Wtau = tau * expz / expsum
    Wref = Wtau[::-1]
    H = np.fft.irfft(P * np.fft.rfft(Wref, L), L)
    g_p = 2.0 * H[n - 1:2 * n - 1][::-1]
    g_p = np.where(np.isfinite(g_p), g_p, 0.0)

    mean_g = float(np.dot(p, g_p))
    grad = g_p - p * mean_g

    max_g = float(np.max(np.abs(grad)))
    if max_g > 1e-2:
        grad *= (1e-2 / max_g)

    true_score = 2.0 * n * cm
    smooth_score = 2.0 * n * value_coeff
    return float(true_score), float(smooth_score), grad


def cheb_basis(n, D):
    x = np.linspace(0.0, 1.0, n)
    t = 2.0 * x - 1.0
    t = np.clip(t, -1.0, 1.0)
    ang = np.arccos(t)
    out = np.empty((n, D), dtype=np.float64)
    for m in range(1, D + 1):
        out[:, m - 1] = np.cos(m * ang)
    return out


def poly_parent_from_theta(theta, B, mode, eps=0.12):

def resize_sequence(seq, m):
    p = normalize_positive(seq)
    m = int(m)
    if p is None:
        return np.ones(m, dtype=np.float64) / float(m)
    n = p.size
    if n == m:
        return p.copy()
    if n == 1:
        return np.full(m, 1.0 / float(m), dtype=np.float64)

    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, m)
    out = np.interp(x_new, x_old, p)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(m, dtype=np.float64) / float(m)
    return out / s


def random_poly_scan(n, count, deadline, rng, B, D=8):

def build_conv_matrix_scaled(f, scale):
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    r = (f / scale)[::-1]
    A = np.zeros((2 * n - 1, n), dtype=np.float64)

    for k in range(2 * n - 1):
        s = max(0, k - n + 1)
        e = min(n - 1, k)
        length = e - s + 1
        start = n - 1 - k + s
        A[k, s:e + 1] = r[start:start + length]
    return A

def adam_run(parent, duration_s, rng, tau_ratio=0.05, lr0=0.02):

            lr = max(0.003, lr * 0.7)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

def compare_update(p, s, best_p, best_score):
