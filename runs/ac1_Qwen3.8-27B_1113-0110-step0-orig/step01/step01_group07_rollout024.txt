"""
Multi-scale gradient optimization for non-negative coefficient sequences.
Uses FFT-based convolution, adaptive smooth-max gradient, and progressive
resolution refinement to minimize 2*n*max(conv)/(sum)^2.
"""

import time
import numpy as np


def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def normalize_positive(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s


def fast_score(a):
    p = normalize_positive(a)
    if p is None:
        return np.inf
    n = p.size
    if n <= 0:
        return np.inf
    L = next_pow2(2 * n - 1)
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0.0:
        return np.inf
    return float(2.0 * n * cm)


def compute_conv(p, L):
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:2 * L - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    return c


def smooth_max_grad(p, tau, L):

    c = compute_conv(p, L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0.0:
        return np.inf, np.zeros(n)
    z = (c - cm) / tau
    z = np.clip(z, -50.0, 0.0)
    expz = np.exp(z)
    expsum = float(expz.sum())
    if not np.isfinite(expsum) or expsum <= 0.0:
        expz = np.zeros_like(c)
        expz[int(np.argmax(c))] = 1.0
        expsum = 1.0
    w = expz / expsum

    # = 2 * (w_rev * p)_j where * is cross-correlation
    # Equivalently: 2 * (w * p)_j where * is convolution, but w has length 2n-1
    # Let's compute: grad_j = 2 * sum_k w_k * p[k-j] for valid k-j
    # This is 2 * cross-correlation of w and p, evaluated at lag j

    # Which equals convolution of w_rev and p at position len(w)-1-j

    w_rev = w[::-1]
    Lw = next_pow2(2 * n - 1 + n - 1)  # enough for linear conv
    hw = np.fft.rfft(w_rev, Lw)
    hp = np.fft.rfft(p, Lw)
    hc = np.fft.irfft(hw * hp, Lw)
    # hc[m] = sum_k w_rev[k] * p[m-k] = sum_k w[2n-2-k] * p[m-k]
    # We want grad_j = sum_k w[k] * p[k-j]
    # Let k' = 2n-2-k, then k = 2n-2-k', and p[k-j] = p[2n-2-k'-j]
    # hc[m] = sum_{k'} w[k'] * p[m-k']... hmm this is getting confusing.
    # Let me just do it directly for small n, or use the correlation trick.
    # Actually: cross-correlation cross(j) = sum_k w[k] * p[k-j]
    # This equals the convolution (w * p_flip)[2n-1-j]... 
    # Let me just compute it as: for each j, sum over k of w[k]*p[k-j]
    # Efficient way: cross(j) = sum_k w[k] * p[k-j] = conv(w, p)[j + n - 1]... 
    # No. Let's use: cross(j) = sum_k w[k] * p[k-j]
    # Define q = p reversed: q[i] = p[n-1-i]
    # Then cross(j) = sum_k w[k] * p[k-j]
    # If we define r[i] = p[n-1-i], then p[k-j] = r[n-1-k+j]
    # cross(j) = sum_k w[k] * r[n-1-k+j]
    # Let l = n-1-k+j, k = n-1+j-l

    # This is getting too complex. Let me just use the FFT approach directly.
    
    # The gradient of b_k w.r.t. p is: 2 * p_{k-j} (for valid k-j in [0,n-1])
    # So d(smooth_max)/dp_j = sum_k w[k] * 2 * p[k-j]
    # = 2 * sum_k w[k] * p[k-j]
    # Define the "effective kernel" as: for lag j, the contribution is sum_k w[k] * p[k-j]
    # This is the cross-correlation of w and p at lag j.
    # cross-correlation can be computed via FFT:
    # cross(j) = sum_k w[k] * conj(p[k-j]) (for real: just p[k-j])
    # = sum_k w[k] * p[k-j]
    # = (w * p_flipped)[j + len(p) - 1] / ...

    # Pad both to size Lc

    # For our range (j in [0, n-1], k in [0, 2n-2]), this works if Lc is large enough
    cross = np.fft.irfft(W * np.conj(P), Lc)
    # cross[j] = sum_k w[k] * p[k-j] (circular, but Lc is large enough to avoid wraparound)
    grad_raw = 2.0 * cross[:n]
    grad_raw = np.where(np.isfinite(grad_raw), grad_raw, 0.0)
    
    # Project to tangent space of simplex in log-parameterization:
    # grad_z_j = p_j * (grad_raw_j - sum_l p_l * grad_raw_l)
    mean_g = float(np.dot(p, grad_raw))
    grad = p * (grad_raw - mean_g)
    return cm, grad


def topK_grad(p, K, L):
    n = p.size
    c = compute_conv(p, L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0.0:
        return np.inf, np.zeros(n)
    # Get top K indices
    K = min(K, len(c))
    if K == 1:
        idx = int(np.argmax(c))
        # grad_j = 2 * p[idx - j] for valid idx-j
        # This is 2 * reversed and shifted p
        lo = max(0, idx - n + 1)
        hi = min(n - 1, idx)
        length = hi - lo + 1
        if length > 0:
            start = idx - hi  # = idx - min(n-1, idx)
            if idx < n:
                # p[idx-j] valid for j=0..idx
                grad_raw = np.zeros(n)
                grad_raw[:idx+1] = 2.0 * p[idx::-1]
            else:
                # p[idx-j] valid for j=idx-n+1..n-1
                grad_raw = np.zeros(n)
                s = idx - n + 1
                grad_raw[s:] = 2.0 * p[n-1-s::-1][:(n-s)]
                # Actually: grad_raw[j] = 2*p[idx-j] for j in [idx-n+1, n-1]
                # Let j' = j - (idx-n+1), then idx-j = idx-(idx-n+1+j') = n-1-j'
                # So grad_raw[idx-n+1+j'] = 2*p[n-1-j']
                grad_raw[s:] = 2.0 * p[::-1][:(n-s)]
        else:
            grad_raw = np.zeros(n)
        grads = grad_raw
    else:
        top_idx = np.argpartition(c, -K)[-K:]
        grads = np.zeros(n)
        for idx in top_idx:
            if idx < n:
                grads[:idx+1] += 2.0 * p[idx::-1]
            else:
                s = idx - n + 1
                # grad_raw[j] = 2*p[idx-j] for j in [s, n-1]
                # idx-j ranges from idx-s=n-1 down to idx-(n-1)=s... 
                # At j=s: idx-s = idx-(idx-n+1) = n-1
                # At j=n-1: idx-(n-1) = idx-n+1 = s
                # So p[idx-j] for j=s..n-1 gives p[n-1], p[n-2], ..., p[s]
                grads[s:] += 2.0 * p[n-1:n-1-(n-s):-1]
        grads /= K
    
    mean_g = float(np.dot(p, grads))
    grad = p * (grads - mean_g)
    max_g = float(np.max(np.abs(grad)))
    if max_g > 1e-2:
        grad *= (1e-2 / max_g)
    return cm, grad


def logistic_grad(p, tau, L):
    """Gradient using soft-max weighting over convolution values."""
    n = p.size
    c = compute_conv(p, L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0.0:
        return np.inf, np.zeros(n)
    
    z = (c - cm) / tau
    z = np.clip(z, -50.0, 0.0)
    expz = np.exp(z)
    expsum = float(expz.sum())
    if not np.isfinite(expsum) or expsum <= 0.0:
        expz = np.zeros_like(c)
        expz[int(np.argmax(c))] = 1.0
        expsum = 1.0
    w = expz / expsum
    
    # Compute gradient: grad_j = 2 * sum_k w[k] * p[k-j]
    # Use FFT cross-correlation
    Lc = next_pow2(2 * n - 1 + n)
    W = np.fft.rfft(w, Lc)
    P = np.fft.rfft(p, Lc)
    cross = np.fft.irfft(W * np.conj(P), Lc)
    grad_raw = 2.0 * cross[:n]
    grad_raw = np.where(np.isfinite(grad_raw), grad_raw, 0.0)
    
    mean_g = float(np.dot(p, grad_raw))
    grad = p * (grad_raw - mean_g)
    max_g = float(np.max(np.abs(grad)))
    if max_g > 1e-2:
        grad *= (1e-2 / max_g)
    return cm, grad

    if p is None:
        return None, np.inf
    n = p.size
    L = next_pow2(2 * n - 1)
    deadline = time.monotonic() + max(0.0, float(duration))
    
    z = np.log(np.maximum(p, 1e-300))
    z = np.where(np.isfinite(z), z, -50.0)
    z -= z.mean()
    z = np.clip(z, -50.0, 50.0)
    
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step = 0
    best_p = p.copy()
    best_score = fast_score(p)
    no_improve = 0

    decay_steps = max(10, int(total_steps_target * 0.8))

        # Adaptive tau: cosine decay from tau_init to tau_end
        progress = min(1.0, step / max(1, decay_steps))
        tau = tau_end + (tau_init - tau_end) * 0.5 * (1.0 + np.cos(np.pi * progress))
        tau = max(1e-4, tau / n)

            no_improve = 0
        else:
            no_improve += 1

            zz += rng.normal(0.0, noise_scale, n)
            zz -= zz.mean()
            z = np.clip(zz, -50.0, 50.0)
            m *= 0.3
            v *= 0.3
            no_improve = 0
        
        # Learning rate: inverse sqrt with warmup
        warmup = min(1.0, (step + 1) / 500.0)
        lr = lr0 * warmup / np.sqrt(step + 1)
        lr = max(0.001, min(lr, 0.1))
        
        step += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        b1 = 1.0 - beta1 ** step
        b2 = 1.0 - beta2 ** step
        z = z - lr * (m / b1) / (np.sqrt(v / b2) + eps)
        z = np.where(np.isfinite(z), z, 0.0)
        z -= z.mean()
        z = np.clip(z, -50.0, 50.0)
        p = np.exp(z - z.max())
        s_p = float(p.sum())
        if s_p <= 0 or not np.isfinite(s_p):
            p = best_p.copy()
            z = np.log(np.maximum(p, 1e-300))
            z -= z.mean()
            continue
        p /= s_p
        
        if step % 10000 == 0:
            print(f"  opt step {step}: score={true_s:.6f} best={best_score:.6f} tau={tau:.6f}")
    
    return best_p, best_score


def upsample(p, new_n):
    """Upsample a sequence to a new length using linear interpolation."""
    p = normalize_positive(p)
    if p is None:
        return np.ones(new_n, dtype=np.float64) / new_n
    n = p.size
    if n == new_n:
        return p.copy()
    if n == 1:
        return np.full(new_n, 1.0 / new_n, dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    out = np.interp(x_new, x_old, p)
    out = np.maximum(out, 1e-10)
    s = float(out.sum())
    if s <= 0 or not np.isfinite(s):
        return np.ones(new_n, dtype=np.float64) / new_n
    return out / s


def downsample(p, new_n):
    """Downsample by averaging blocks."""
    p = normalize_positive(p)
    if p is None or new_n <= 0:
        return np.ones(max(1, new_n), dtype=np.float64) / max(1, new_n)
    n = p.size
    if n <= new_n:
        return upsample(p, new_n)
    # Average blocks
    block_size = n // new_n
    remainder = n - block_size * new_n
    out = np.zeros(new_n, dtype=np.float64)
    start = 0
    for i in range(new_n):
        end = start + block_size + (1 if i >= new_n - remainder else 0)
        out[i] = p[start:end].mean()
        start = end
    s = float(out.sum())
    if s <= 0 or not np.isfinite(s):
        return np.ones(new_n, dtype=np.float64) / new_n
    return out / s


def make_poly_nine(n, a0, a1, a2, a3, eps=1.0):
    """Create a sequence from a cubic polynomial shape."""
    x = np.linspace(0.0, 1.0, n)
    t = 2.0 * x - 1.0
    h = 1.0 + eps * (a0 + a1 * t + a2 * t**2 + a3 * t**3)
    h = np.maximum(h, 0.0)
    s = float(h.sum())
    if s <= 0:
        return np.ones(n, dtype=np.float64) / n
    return h / s


def propose_candidate(seed=42, budget_s=1800.0, **kwargs):
    """Multi-scale optimization to find low-score sequences."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    raw_budget = max(0.0, float(budget_s))
    deadline = start_time + max(0.0, raw_budget - 30.0)
    
    print(f"Budget: {raw_budget}s, deadline in {raw_budget - 30:.0f}s")
    
    best_p = None
    best_s = np.inf
    
    # Try to use previous best
    try:
        hs = np.asarray(height_sequence_1, dtype=np.float64)
        hs = np.where(np.isfinite(hs), hs, 0.0)
        hs = np.maximum(hs, 0.0)
        if hs.size > 0 and float(hs.sum()) > 1e-9:
            hs_n = hs / hs.sum()
            s0 = fast_score(hs_n)
            if np.isfinite(s0):
                best_p = hs_n
                best_s = s0
                print(f"Previous best: n={hs_n.size}, score={s0:.6f}")
    except Exception:
        pass
    
    # Also try some parametric shapes at medium resolution for initialization
    N_init = 2048
    candidates_init = []
    
    # Linear shapes
    for asym in [0.3, 0.5, 0.7, 0.9, 1.0, 1.2]:
        x = np.linspace(0, 1, N_init)
        h = 1.0 + asym * (2 * x - 1.0)
        h = np.maximum(h, 0.0)
        s = h.sum()
        if s > 0:
            candidates_init.append(h / s)
    
    # Quadratic shapes
    for a1 in [0.3, 0.5, 0.7, 1.0]:
        for a2 in [-0.3, -0.5, 0.3, 0.5]:
            x = np.linspace(0, 1, N_init)
            t = 2 * x - 1
            h = 1.0 + a1 * t + a2 * t**2
            h = np.maximum(h, 0.0)
            s = h.sum()
            if s > 0:
                candidates_init.append(h / s)
    
    # Cosine shapes
    for freq in [1, 2, 3]:
        for amp in [0.2, 0.4, 0.6]:
            x = np.linspace(0, 1, N_init)
            h = 1.0 + amp * np.cos(2 * np.pi * freq * x + rng.uniform(0, 2 * np.pi))
            h = np.maximum(h, 0.0)
            s = h.sum()
            if s > 0:
                candidates_init.append(h / s)
    
    # Evaluate all initial candidates
    for cand in candidates_init:
        sc = fast_score(cand)
        if np.isfinite(sc) and sc < best_s:
            best_s = sc
            best_p = cand
            print(f"  Init candidate: score={sc:.6f}")
    
    if best_p is None:
        best_p = np.ones(N_init, dtype=np.float64) / N_init
        best_s = 2.0
    
    # Phase 1: Optimize at N_init = 2048
    print(f"\nPhase 1: Optimize at n={N_init}, current best={best_s:.6f}")
    rem = deadline - time.monotonic()
    if rem > 60:
        dur = min(rem * 0.30, 300.0)
        p1, s1 = adam_optimize(best_p, dur, rng, tau_init=0.15, tau_end=0.01, lr0=0.04, K_top=30)
        if p1 is not None and s1 < best_s:
            best_p, best_s = p1, s1
            print(f"Phase 1 result: score={s1:.6f}")
    
    # Phase 2: Upsample to 4096 and optimize
    N2 = 4096
    print(f"\nPhase 2: Upsample to n={N2}, current best={best_s:.6f}")
    rem = deadline - time.monotonic()
    if rem > 60:
        p2_start = upsample(best_p, N2)
        s2_start = fast_score(p2_start)
        print(f"  After upsample: score={s2_start:.6f}")
        dur = min(rem * 0.30, 400.0)
        p2, s2 = adam_optimize(p2_start, dur, rng, tau_init=0.10, tau_end=0.005, lr0=0.03, K_top=50)
        if p2 is not None:
            s2_final = fast_score(p2)
            if s2_final < best_s:
                best_p, best_s = p2, s2_final
            print(f"Phase 2 result: score={s2_final:.6f}")
    
    # Phase 3: Upsample to 8192 and optimize (main phase)
    N3 = 8192
    print(f"\nPhase 3: Upsample to n={N3}, current best={best_s:.6f}")
    rem = deadline - time.monotonic()
    if rem > 60:
        p3_start = upsample(best_p, N3)
        s3_start = fast_score(p3_start)
        print(f"  After upsample: score={s3_start:.6f}")
        dur = min(rem * 0.50, 800.0)
        p3, s3 = adam_optimize(p3_start, dur, rng, tau_init=0.08, tau_end=0.002, lr0=0.025, K_top=80)
        if p3 is not None:
            s3_final = fast_score(p3)
            if s3_final < best_s:
                best_p, best_s = p3, s3_final
            print(f"Phase 3 result: score={s3_final:.6f}")
    
    # Phase 4: Final fine-tuning with very small tau
    rem = deadline - time.monotonic()
    if rem > 30:
        print(f"\nPhase 4: Fine-tuning, remaining={rem:.0f}s")
        p4_start = upsample(best_p, N3) if best_p.size != N3 else best_p.copy()
        dur = min(rem * 0.85, 400.0)
        p4, s4 = adam_optimize(p4_start, dur, rng, tau_init=0.02, tau_end=0.001, lr0=0.012, K_top=100)
        if p4 is not None:
            s4_final = fast_score(p4)
            if s4_final < best_s:
                best_p, best_s = p4, s4_final
            print(f"Phase 4 result: score={s4_final:.6f}")
    
    # Phase 5: Try a random restart from the best
    rem = deadline - time.monotonic()
    if rem > 60:
        print(f"\nPhase 5: Random restart, remaining={rem:.0f}s")
        noise = np.exp(rng.normal(0.0, 0.08, N3))
        p5_start = normalize_positive(best_p * noise)
        if p5_start is None:
            p5_start = best_p.copy()
        dur = min(rem * 0.80, 300.0)
        p5, s5 = adam_optimize(p5_start, dur, rng, tau_init=0.04, tau_end=0.001, lr0=0.015, K_top=60)
        if p5 is not None:
            s5_final = fast_score(p5)
            if s5_final < best_s:
                best_p, best_s = p5, s5_final
            print(f"Phase 5 result: score={s5_final:.6f}")
    
    # Final validation
    print(f"\nFinal best score: {best_s:.6f}")
    final_seq = None
    final_score = np.inf
    
    candidates = []
    if best_p is not None:
        candidates.append(best_p)
    candidates.append(np.ones(4096, dtype=np.float64) / 4096.0)
    
    for cand in candidates:
        lst = [float(x) for x in np.clip(np.asarray(cand, dtype=np.float64), 0.0, 1000.0)]
        if not lst:
            continue
        try:
            sc = evaluate_sequence(lst)
        except Exception:
            sc = np.inf
        if np.isfinite(sc) and sc < final_score:
            final_score = float(sc)
            final_seq = lst
        print(f"  Validated n={len(lst)}: score={sc:.6f}")
    
    if final_seq is None:
        final_seq = [1.0 / 4096.0] * 4096
    
    print(f"  Final validated score: {final_score:.6f}")
    return final_seq
