Minimizes 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses FFT-accelerated multi-peak subgradient descent with adaptive step sizes,
random restarts, and diverse initializations including the current best.
Targets the top several autocorrelation peaks simultaneously for faster convergence.

import numpy as np
import time
import math


def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]


def _subgradient_vec(a, k_max):
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k_max - n + 1)
    hi = min(n - 1, k_max)
    if lo <= hi:
        j_range = np.arange(lo, hi + 1)
        indices = k_max - j_range
        g[lo:hi + 1] = a[indices]
    return g


def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a)
    return 2.0 * n * float(np.max(c)) / (s * s)

        momentum = mu * momentum + (1.0 - mu) * g_norm

        a = a - eta * momentum
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            eta *= 0.5
            a = best_a.copy()
            it += 1
            continue
        a = a / s

        if no_improve > 250:
            eta *= 0.95
        elif no_improve < 50:
            eta = min(eta * 1.01, 0.2)

        it += 1

    return best_a, best_score


def _gaussian_process_init(n, rng, length_scale=0.2, mean=1.0, std=0.3):
    """Generate a random smooth non-negative sequence using a Gaussian process."""
    x = np.arange(n, dtype=np.float64) / n
    # Simple GP with RBF kernel (approximate via random Fourier features)
    n_features = max(50, n // 20)
    freqs = rng.normal(0, 1.0 / length_scale, n_features)
    phases = rng.uniform(0, 2 * np.pi, n_features)
    coefs = rng.normal(0, std / np.sqrt(n_features), n_features)
    f = mean + np.sum(coefs * np.cos(2 * np.pi * freqs[:, None] * x[None, :] + phases[:, None]), axis=0)
    f = np.maximum(f, 0.01)
    return f


def _asymmetric_shape(n, rng):
    """Generate various asymmetric shapes."""
    choice = rng.integers(10)
    x = np.arange(n, dtype=np.float64) / n

    if choice == 0:
        # Sinusoidal with random phase and amplitude
        c = rng.uniform(0.2, 0.7)
        phi = rng.uniform(0, 2 * np.pi)
        a = 1.0 + c * np.sin(2 * np.pi * x + phi)
    elif choice == 1:
        # Two-harmonic
        c1 = rng.uniform(0.1, 0.5)
        c2 = rng.uniform(0.05, 0.3)
        a = 1.0 + c1 * np.sin(2 * np.pi * x + rng.uniform(0, 2*np.pi)) + \
            c2 * np.sin(4 * np.pi * x + rng.uniform(0, 2*np.pi))
    elif choice == 2:
        # Asymmetric Gaussian
        center = rng.uniform(0.1, 0.4)
        sigma = rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2) + 0.05
    elif choice == 3:
        # Power law decay
        alpha = rng.uniform(1.0, 4.0)
        a = (1 - x) ** alpha + 0.05
    elif choice == 4:
        # S-curve (sigmoid)
        steep = rng.uniform(5, 15)
        mid = rng.uniform(0.3, 0.6)
        a = 1.0 / (1.0 + np.exp(steep * (x - mid))) + 0.05
    elif choice == 5:
        # Bump then tail
        a = np.exp(-20 * (x - 0.3) ** 2) + 0.2 * np.exp(-3 * (x - 0.8) ** 2) + 0.05
    elif choice == 6:
        # Raised cosine window (asymmetric)
        a = 0.5 * (1 + np.cos(np.pi * x / 0.7)) * (x < 0.7).astype(float) + 0.1
    elif choice == 7:
        # Gaussian process sample
        a = _gaussian_process_init(n, rng)
    elif choice == 8:
        # Periodic with decay
        freq = rng.uniform(2, 5)
        decay = rng.uniform(1, 3)
        a = np.exp(-decay * x) * (1 + 0.3 * np.cos(2 * np.pi * freq * x)) + 0.05
    else:
        # Pure random smooth
        a = np.ones(n)
        for _ in range(20):
            center = rng.integers(0, n)
            width = rng.integers(n // 20, n // 5)
            amp = rng.normal(0, 0.1)
            mask = np.abs(np.arange(n) - center) < width
            a[mask] += amp * np.exp(-0.5 * ((np.arange(n)[mask] - center) / width) ** 2)
        a = np.maximum(a, 0.05)

    return np.maximum(a, 0.01)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Minimizes 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
    Uses multi-peak subgradient descent with diverse initializations,
    adaptive step sizes, and random restarts.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 20.0  # 20s safety margin

    best_overall = None
    best_overall_score = np.inf

    candidates = []  # list of (name, a0_array)

    # 1. Load current best
    try:
        hs = np.array(height_sequence_1, dtype=np.float64)
        hs = np.clip(hs, 0, None)
        if hs.size > 100 and hs.sum() > 0.01:
            candidates.append(("current_best", hs.copy()))
    except NameError:
        pass
    except Exception:
        pass

    # 2. Diverse initializations across multiple n values
    n_values = [2000, 2500, 3000, 4000]
    for n in n_values:
        # Sin-based (known good family)
        for phase in [np.pi / 4, np.pi / 3, np.pi / 5, 2 * np.pi / 3]:
            for amp in [0.3, 0.4, 0.5]:
                x = np.arange(n, dtype=np.float64)
                a = 1.0 + amp * np.sin(2 * np.pi * x / n + phase)
                a = np.maximum(a, 0.05)
                candidates.append((f"sin_n{n}_p{phase:.2f}_a{amp}", a))

    # 3. Random asymmetric shapes
    for n in [2000, 3000, 4000]:
        for i in range(4):
            a = _asymmetric_shape(n, rng)
            candidates.append((f"asym_n{n}_{i}", a))

    # 4. Gaussian process samples
    for n in [2000, 3000]:
        for ls in [0.1, 0.2, 0.3]:
            a = _gaussian_process_init(n, rng, length_scale=ls)
            candidates.append((f"gp_n{n}_ls{ls}", a))

    print(f"Total candidates: {len(candidates)}")

    # Quick evaluation and ranking
    scored = []
    for name, a0 in candidates:
        a0 = np.maximum(a0, 0)
        s = a0.sum()
        if s < 1e-15:
            continue
        a0_n = a0 / s
        c0 = _fft_convolve(a0_n)
        init_score = 2.0 * len(a0_n) * float(np.max(c0))
        scored.append((init_score, name, a0_n))

    scored.sort(key=lambda x: x[0])
    print("Initial scores (top 12):")
    for i, (sc, name, _) in enumerate(scored[:12]):
        print(f"  {i+1}. {name}: {sc:.6f} (n={len(_)})")

    # Allocate time to top candidates
    # Use more time for better initializations
    n_top = min(len(scored), 10)
    remaining_time = budget_s - 40.0

    # Geometric weights
    weights = [max(1.0, float(n_top - i) ** 1.5) for i in range(n_top)]
    total_w = sum(weights)
    times = [remaining_time * w / total_w for w in weights]

    # Cap individual times
    max_single = remaining_time * 0.25
    for i in range(n_top):
        if times[i] > max_single:
            times[i] = max_single

    for i in range(n_top):
        if time.monotonic() > deadline:
            break

        init_score, name, a0 = scored[i]
        t_alloc = min(times[i], max(10.0, deadline - time.monotonic()))

        print(f"\n[{i+1}/{n_top}] Optimizing '{name}' (n={len(a0)}, score={init_score:.6f}, t={t_alloc:.0f}s)")

        # Split time among methods
        t_third = t_alloc / 3.0

        # Method 1: Multi-peak subgradient
        a1, s1 = _optimize_multi_peak(a0, t_third, rng, n_top_peaks=3, eta0=0.2, mu0=0.75)
        eval1 = evaluate_sequence(a1.tolist())
        print(f"  Multi-peak: {eval1:.6f}")
        if np.isfinite(eval1) and eval1 < best_overall_score:
            best_overall_score = eval1
            best_overall = a1.copy()
            print(f"  *** NEW BEST: {eval1:.6f} ***")

        # Method 2: Single-peak (from best of method 1)
        start2 = a1 if np.isfinite(s1) and s1 < init_score else a0
        a2, s2 = _optimize_single_peak(start2, t_third, rng, eta0=0.15, mu0=0.7)
        eval2 = evaluate_sequence(a2.tolist())
        print(f"  Single-peak: {eval2:.6f}")
        if np.isfinite(eval2) and eval2 < best_overall_score:
            best_overall_score = eval2
            best_overall = a2.copy()
            print(f"  *** NEW BEST: {eval2:.6f} ***")

        # Method 3: Additive projection
        start3 = a2 if np.isfinite(s2) and s2 < (s1 if np.isfinite(s1) else init_score) else a0
        a3, s3 = _optimize_additive_proj(start3, t_third, rng)
        eval3 = evaluate_sequence(a3.tolist())
        print(f"  AddProj: {eval3:.6f}")
        if np.isfinite(eval3) and eval3 < best_overall_score:
            best_overall_score = eval3
            best_overall = a3.copy()
            print(f"  *** NEW BEST: {eval3:.6f} ***")

    # Final polishing phase
    remaining = deadline - time.monotonic()
    if best_overall is not None and remaining > 60:
        print(f"\n=== Final polishing ({remaining:.0f}s remaining) ===")

        # Multiple polishing rounds with different methods
        n_rounds = max(2, int(remaining / 200))
        for r in range(n_rounds):
            t_round = remaining / (n_rounds - r)
            if t_round < 20:
                break

            method_choice = r % 3
            if method_choice == 0:
                a_p, sp = _optimize_multi_peak(best_overall, t_round, rng, n_top_peaks=4, eta0=0.1, mu0=0.8)
            elif method_choice == 1:
                a_p, sp = _optimize_single_peak(best_overall, t_round, rng, eta0=0.1, mu0=0.7)
            else:
                a_p, sp = _optimize_additive_proj(best_overall, t_round, rng)

            eval_p = evaluate_sequence(a_p.tolist())
            print(f"  Round {r+1}: {eval_p:.6f}")
            if np.isfinite(eval_p) and eval_p < best_overall_score:
                best_overall_score = eval_p
                best_overall = a_p.copy()
                print(f"  *** NEW BEST: {eval_p:.6f} ***")

            remaining = deadline - time.monotonic()

    # Final verification
    if best_overall is not None:
        final_score = evaluate_sequence(best_overall.tolist())
        elapsed = time.monotonic() - total_start
        print(f"\nFinal result: score={final_score:.6f}, n={len(best_overall)}, elapsed={elapsed:.1f}s")
        return [float(x) for x in best_overall]
    else:
        a = np.ones(1000, dtype=np.float64)
        print("Falling back to uniform sequence")
        return a.tolist()
