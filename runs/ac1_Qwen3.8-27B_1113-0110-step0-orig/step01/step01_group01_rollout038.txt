def normalize_positive(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).ravel()

def convolve_seq(a, N=None):

    fa = np.fft.rfft(a, N)

def swap_delta_fast(c, b, i, j, n):
    """Fast swap: b[i]=1 -> 0, b[j]=0 -> 1. Updates c in place."""
    # Remove contribution of position i (which had value 1)
    # b[k] = sum_{l=0}^{n-1} b[l]*b[k-l] (for valid indices)
    # Removing b[i]=1:
    #   For l in [0, n-1]: k = i+l, b[k] loses b[l] (from term b[i]*b[l])
    #   AND: for l in [0, n-1]: k = l+i, b[k] loses b[l] (from term b[l]*b[i])
    #   These give the same k, so total loss at k=i+l is 2*b[l], except when l=i (k=2i) where it's just b[i]=1
    # So: c[i:i+n] -= 2*b[0:n], then c[2*i] += 1 (fix the double count)
    
    end_i = min(i + n, 2 * n - 1)
    c[i:end_i] -= 2.0 * b[:end_i - i]
    if 2 * i < 2 * n - 1:
        c[2 * i] += 1.0
    
    # Add contribution of position j (which had value 0, now 1)
    end_j = min(j + n, 2 * n - 1)
    c[j:end_j] += 2.0 * b[:end_j - j]
    if 2 * j < 2 * n - 1:
        c[2 * j] -= 1.0
    
    return c

        b = np.asarray(init_b, dtype=np.float64).ravel().copy()
        m = int(b.sum())
    
    c = compute_conv_binary(b, N)
    best_max = float(np.max(c))
    best_b = b.copy()
    best_score = 2.0 * n * best_max / (m * m)
    
    print(f"  ILS n={n} m={m} initial score={best_score:.8f}", flush=True)
    
    one_idx = np.nonzero(b > 0.5)[0]
    zero_idx = np.nonzero(b < 0.5)[0]
    
    it = 0
    temp = 0.3
    stagnant = 0
    
    while time.monotonic() < deadline:
        # Check time periodically
        if it % 1000 == 0 and time.monotonic() >= deadline:
            break
        
        cmax = float(np.max(c))
        score = 2.0 * n * cmax / (m * m)
        
        if score < best_score * (1.0 - 1e-12):
            best_score = score
            best_max = cmax
            best_b = b.copy()
        
        # Find positions near the peak for gradient-guided selection
        # Peak positions
        peak_thresh = cmax * 0.95
        peak_pos = np.nonzero(c >= peak_thresh)[0]
        
        # Compute a "pressure" for each position based on how much it affects the peak
        # Position i affects c[k] for k in [i, i+n-1] and [0, 2n-2-i] (approximately)
        # We want to move 1s away from positions that contribute to the peak
        # and 0s toward positions that reduce the peak
        
        # Simple approach: compute gradient-like pressure
        # For each 1-position p, pressure[p] = sum of c[k] for k in [p, p+n-1] weighted by b
        # This is expensive, so use a simpler heuristic:
        # pressure[p] ≈ c[p] + c[2n-2-p] + c[n-1] (crude)
        
        # Better: use the actual effect. Moving a 1 from i to j changes c by:
        # delta_c[k] = -2*b[k-i] + 2*b[k-j] (approximately, ignoring corrections)
        # We want delta_c to reduce the max, so we want 2*b[k-j] < 2*b[k-i] at peak positions.
        
        # For efficiency, just use random selection with slight bias
        si = rng.integers(0, one_idx.size)
        sj = rng.integers(0, zero_idx.size)
        i = int(one_idx[si])
        j = int(zero_idx[sj])
        
        # Perform swap
        old_max = cmax
        c = swap_delta_fast_correct(c, b, i, j, n)
        new_max = float(np.max(c))
        new_score = 2.0 * n * new_max / (m * m)
        
        # Accept/reject
        accept = False
        delta = new_score - score
        if delta < 0:
            accept = True
        elif delta < 0.01:
            if rng.random() < np.exp(-delta / max(temp, 1e-10)):
                accept = True
        elif delta < 0.05 and rng.random() < 0.05:
            # Occasionally accept larger increases to escape local optima
            accept = True
        
        if accept:
            # Update index sets
            np.delete(one_idx, si)  # This doesn't work in-place, let's use a different approach
            # Actually let's just rebuild periodically
            if it % 50 == 0:
                one_idx = np.nonzero(b > 0.5)[0]
                zero_idx = np.nonzero(b < 0.5)[0]
            else:
                one_idx[si] = j
                zero_idx[sj] = i
                
            stagnant = 0
            temp = max(0.001, temp * 0.9997)
        else:
            # Revert
            c = swap_delta_fast_correct(c, b, j, i, n)
            b[j], b[i] = 0.0, 1.0  # This is already handled by the swap function
            # Wait, swap_delta_fast_correct already modifies b. Let me check...
            # It sets b[i]=0 and b[j]=1. To revert, we need to set b[i]=1 and b[j]=0.
            # But the function already did that. So after revert (calling swap again with j,i):
            # It sets b[j]=0 and b[i]=1. That's correct!
            
            stagnant += 1
            temp = max(0.001, temp * 0.999)
            
            if stagnant > 500:
                # Restart with perturbation
                n_perturb = max(3, n // 50)
                for _ in range(n_perturb):
                    si2 = rng.integers(0, one_idx.size)
                    sj2 = rng.integers(0, zero_idx.size)
                    i2 = int(one_idx[si2])
                    j2 = int(zero_idx[sj2])
                    c = swap_delta_fast_correct(c, b, i2, j2, n)
                    one_idx[si2] = j2
                    zero_idx[sj2] = i2
                temp = min(0.5, temp * 3.0)
                stagnant = 0
        
        it += 1
        if it % 5000 == 0:
            print(f"    ILS iter={it} best={best_score:.8f} temp={temp:.4f}", flush=True)
    
    return best_b, best_score, int(best_b.sum())


def optimize_binary_greedy_ils(n, m, deadline, seed, init_b=None):
    """
    More aggressive binary optimizer using greedy + ILS.
    Uses batch evaluation of multiple candidate swaps.
    """
    n = int(n)
    m = int(m)
    N = next_pow2(2 * n - 1)
    rng = np.random.default_rng(seed)
    
    if init_b is None:
        b = np.zeros(n, dtype=np.float64)
        idx = rng.choice(n, size=m, replace=False)
        b[idx] = 1.0
    else:
        b = np.asarray(init_b, dtype=np.float64).ravel().copy()
        m = int(b.sum())
    
    c = compute_conv_binary(b, N)
    cmax = float(np.max(c))
    best_max = cmax
    best_b = b.copy()
    best_score = 2.0 * n * best_max / (m * m)
    
    print(f"  Greedy-ILS n={n} m={m} initial score={best_score:.8f}", flush=True)
    
    it = 0
    stagnant = 0
    batch_size = min(200, n // 10)
    
    while time.monotonic() < deadline:
        if it % 200 == 0 and time.monotonic() >= deadline:
            break
        
        cmax = float(np.max(c))
        score = 2.0 * n * cmax / (m * m)
        
        if score < best_score * (1.0 - 1e-12):
            best_score = score
            best_max = cmax
            best_b = b.copy()
        
        # Try a batch of random swaps, pick the best improving one
        one_positions = np.nonzero(b > 0.5)[0]
        zero_positions = np.nonzero(b < 0.5)[0]
        
        if one_positions.size == 0 or zero_positions.size == 0:
            break
        
        best_delta = 0.0
        best_i, best_j = -1, -1
        
        # Sample candidate swaps
        n_samples = min(batch_size, one_positions.size * zero_positions.size)
        n_samples = min(n_samples, 500)
        
        for _ in range(n_samples):
            i = int(one_positions[rng.integers(0, one_positions.size)])
            j = int(zero_positions[rng.integers(0, zero_positions.size)])
            
            # Estimate the change in max without full update
            # The change at position k: c[k] changes by approximately -2*b[k-i] + 2*b[k-j]
            # (ignoring corrections)
            # The max can only decrease if the current peak positions are reduced
            # Quick check: compute the delta at the current peak positions
            peak_ks = np.nonzero(c >= cmax * 0.99)[0]
            if peak_ks.size == 0:
                continue
            
            # For peak positions, what's the delta?
            # delta[k] ≈ -2*b[k-i mod n] + 2*b[k-j mod n] (rough)
            # This is a rough estimate; for exact, we'd need the full update
            # Let's just check a few peak positions
            ks_check = peak_ks[:min(10, peak_ks.size)]
            deltas = np.zeros(ks_check.size)
            for idx_k, k in enumerate(ks_check):
                d = 0.0
                l1 = k - i
                if 0 <= l1 < n:
                    d -= 2.0 * b[l1]
                l2 = k - j
                if 0 <= l2 < n:
                    d += 2.0 * b[l2]
                deltas[idx_k] = d
            
            if np.all(deltas < 0):
                best_delta = min(best_delta, float(np.max(deltas)))
                best_i, best_j = i, j
        
        if best_i >= 0 and best_delta < 0:
            # Accept the best swap
            c = swap_delta_fast_correct(c, b, best_i, best_j, n)
            stagnant = 0
        else:
            # No improving move found; do a random perturbation
            n_perturb = max(2, n // 100)
            for _ in range(n_perturb):
                i = int(one_positions[rng.integers(0, one_positions.size)])
                j = int(zero_positions[rng.integers(0, zero_positions.size)])
                c = swap_delta_fast_correct(c, b, i, j, n)
            stagnant += 1
        
        it += 1
        if it % 2000 == 0:
            print(f"    Greedy-ILS iter={it} best={best_score:.8f} stag={stagnant}", flush=True)
    
    return best_b, best_score, int(best_b.sum())


def mirror_descend(a, deadline, max_iter=2000):
    """Mirror descent on the simplex to reduce the score."""
    a = normalize_positive(a, floor=1e-15)
    n = a.size
    N = next_pow2(2 * n - 1)
    
    best_a = a.copy()
    best_score = internal_score(a)
    
    eta = 0.5
    stagnant = 0
    it = 0
    
    while time.monotonic() < deadline and it < max_iter:
        c = convolve_seq(a, N)
        cmax = float(np.max(c))
        score = 2.0 * n * cmax / (float(a.sum()) ** 2)
        
        if score < best_score * (1.0 - 1e-12):
            best_score = score
            best_a = a.copy()
        
        # Compute weights focused on the peak
        tau = 0.1 * cmax
        z = (c - cmax) / tau
        z = np.clip(z, -40.0, 0.0)
        w = np.exp(z)
        w = w / w.sum()
        
        # Gradient of weighted convolution
        fw = np.fft.rfft(w[::-1], N)
        fa = np.fft.rfft(a, N)
        conv = np.fft.irfft(fa * fw, N)
        g = 2.0 * conv[n - 1:2 * n - 1][::-1]
        g = g - g.mean()
        g_scale = float(np.abs(g).max())
        if g_scale < 1e-15:
            break
        g = g / g_scale
        
        # Mirror descent step
        loga = np.log(a + 1e-20)
        accepted = False
        for lr in (eta, 0.5 * eta, 0.25 * eta):
            loga_new = loga - lr * g
            loga_new -= loga_new.max()
            a_new = np.exp(np.clip(loga_new, -50.0, 0.0))
            a_new = a_new / a_new.sum()
            
            score_new = internal_score(a_new, )
            if np.isfinite(score_new) and score_new < score:
                a = a_new
                loga = np.log(a + 1e-20)
                accepted = True
                break
        
        if accepted:
            stagnant = 0
            eta = min(1.0, eta * 1.02)
        else:
            stagnant += 1
            eta = max(0.01, eta * 0.7)
            if stagnant > 200:
                break
        
        it += 1
    
    return best_a, best_score


def propose_candidate(budget_s=2350, **kwargs):
    start_time = time.monotonic()
    total_budget = min(float(budget_s), 2350.0)
    deadline = start_time + total_budget
    
    best_score = 2.0
    best_a = np.ones(1024, dtype=np.float64) / 1024.0
    
    # Check given sequence
    given = get_given_sequence()
    if given is not None:
        sc = internal_score(given)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = given.copy()
            print(f"Given sequence: n={given.size}, score={sc:.9f}", flush=True)
    
    # Phase 1: Quick binary search at moderate size (20% of time)
    t1_deadline = start_time + 0.20 * total_budget
    
    n1 = 4096
    m1 = n1 // 2
    if given is not None:
        init1 = binary_from_continuous(resample_seq(given, n1), m1)
    else:
        init1 = None
    
    b1, sc1, m1_actual = optimize_binary_ils(n1, m1, t1_deadline, seed=42, init_b=init1)
    if np.isfinite(sc1) and sc1 < best_score:
        best_score = sc1
        best_a = b1.copy()
        print(f"Phase 1 best: n={n1}, score={sc1:.9f}", flush=True)
    
    # Phase 2: Binary search at larger size (40% of time)
    t2_deadline = start_time + 0.60 * total_budget
    
    n2 = 8192
    m2 = n2 // 2
    
    # Try multiple starting points
    starts2 = []
    if given is not None:
        starts2.append(binary_from_continuous(resample_seq(given, n2), m2))
    
    rng = np.random.default_rng(123)
    for _ in range(2):
        b_rand = np.zeros(n2, dtype=np.float64)
        b_rand[rng.choice(n2, size=m2, replace=False)] = 1.0
        starts2.append(b_rand)
    
    # Quadratic residue based start
    qr = set()
    for x in range(2, n2):
        qr.add((x * x) % n2)
    b_qr = np.zeros(n2, dtype=np.float64)
    qr_list = sorted(qr)
    if len(qr_list) > m2:
        b_qr[qr_list[:m2]] = 1.0
    else:
        idx = rng.choice(n2, size=m2 - len(qr_list), replace=False) if len(qr_list) < m2 else np.array([], dtype=int)
        b_qr[qr_list] = 1.0
        if len(idx) > 0:
            b_qr[idx] = 1.0
    starts2.append(b_qr)
    
    per_start = (t2_deadline - time.monotonic()) / len(starts2)
    for si, s_b in enumerate(starts2):
        if time.monotonic() > t2_deadline:
            break
        allow = min(per_start, t2_deadline - time.monotonic())
        if allow < 10:
            break
        b2, sc2, m2a = optimize_binary_ils(n2, m2, time.monotonic() + allow, seed=100 + si, init_b=s_b)
        if np.isfinite(sc2) and sc2 < best_score:
            best_score = sc2
            best_a = b2.copy()
            print(f"Phase 2 best: n={n2}, start={si}, score={sc2:.9f}", flush=True)
    
    # Phase 3: Binary search at even larger size (25% of time)
    t3_deadline = start_time + 0.85 * total_budget
    
    n3 = 16384
    m3 = n3 // 2
    
    if given is not None:
        init3 = binary_from_continuous(resample_seq(best_a, n3), m3)
    else:
        init3 = None
    
    b3, sc3, m3a = optimize_binary_ils(n3, m3, t3_deadline, seed=777, init_b=init3)
    if np.isfinite(sc3) and sc3 < best_score:
        best_score = sc3
        best_a = b3.copy()
        print(f"Phase 3 best: n={n3}, score={sc3:.9f}", flush=True)
    
    # Phase 4: Continuous polishing (15% of time)
    t4_deadline = deadline
    
    if time.monotonic() < t4_deadline - 30:
        a_cont = resample_seq(best_a, min(len(best_a), 8192))
        a4, sc4 = mirror_descend(a_cont, t4_deadline, max_iter=500)
        if np.isfinite(sc4) and sc4 < best_score:
            best_score = sc4
            best_a = a4.copy()
            print(f"Phase 4 best (continuous): n={a4.size}, score={sc4:.9f}", flush=True)
    
    # Final validation
    best_a = normalize_positive(best_a, floor=0.0)
    out_list = [float(x) for x in best_a]
    
    try:
        final_eval = float(evaluate_sequence(out_list))
    except Exception:
        final_eval = float("inf")
    
    if not np.isfinite(final_eval) or final_eval >= best_score:
        # Fallback to given sequence if it was better
        if given is not None:
            out_list = [float(x) for x in given]
            try:
                final_eval = float(evaluate_sequence(out_list))
            except Exception:
                out_list = [1.0 / 1024] * 1024
    
    print(f"FINAL: score={final_eval:.12f}, length={len(out_list)}", flush=True)
    return out_list
