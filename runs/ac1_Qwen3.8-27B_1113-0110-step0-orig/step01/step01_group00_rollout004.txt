import time
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]

def _score(a):
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)

def _fft_convolve(a):

                continue
            idx1 = int(k) - j_arr
            idx2 = j_arr - int(k)
            m1 = (idx1 >= 0) & (idx1 < n)
            if np.any(m1):
                grad[m1] += w[k] * a[idx1[m1]]
            m2 = (idx2 >= 0) & (idx2 < n)
            if np.any(m2):
                grad[m2] += w[k] * a[idx2[m2]]
        a = a - lr * grad
        a = np.maximum(a, 0.0)
        S_new = float(np.sum(a))
        if S_new > 1e-10:
            a = a * S / S_new
        score = _score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            if it % 500 == 0:
                print(f"    Smoothed iter {it}: score={best_score:.6f}, tau={tau:.6f}")
        tau *= tau_decay
        lr *= 0.9998
    return best_a, best_score


def _subgradient_topk(a_init, deadline, n_peaks=40, lr_init=0.04, lr_decay=0.99997):
    """Subgradient using top-k autocorrelation peaks with stagnation restart."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    best_a = a.copy()
    best_score = _score(a)
    lr = lr_init
    rng = np.random.default_rng(789)
    stalled = 0
    j_arr = np.arange(n)

    for it in range(200000):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        max_c = float(np.max(c))
        score = 2.0 * n * max_c / S / S
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stalled = 0
            if it % 1000 == 0:
                print(f"    Subgrad iter {it}: score={best_score:.6f}")
        else:
            stalled += 1
        if n_peaks < len(c):
            top_k = np.argpartition(c, -n_peaks)[-n_peaks:]
        else:
            top_k = np.arange(len(c))
        s = np.zeros(n)
        for k in top_k:
            idx1 = int(k) - j_arr
            idx2 = j_arr - int(k)
            m1 = (idx1 >= 0) & (idx1 < n)
            if np.any(m1):
                s[m1] += a[idx1[m1]]
            m2 = (idx2 >= 0) & (idx2 < n)
            if np.any(m2):
                s[m2] += a[idx2[m2]]
        s /= len(top_k)
        max_s = float(np.max(s))
        if max_s > 1e-15:
            s = s / max_s
            a = a - lr * s
            a = np.maximum(a, 0.0)
            S_new = float(np.sum(a))
            if S_new > 1e-10:
                a = a * S / S_new
        lr *= lr_decay
        if stalled > 500:
            noise = rng.normal(0, 0.04, n)
            a = a + noise
            a = np.maximum(a, 0.0)
            S_p = float(np.sum(a))
            if S_p > 1e-10:
                a = a / S_p
            stalled = 0
            lr = lr_init * 0.7
    return best_a, best_score


def _interpolate_sequence(src, n_target):
    """Interpolate a sequence to a new length, preserving non-negativity."""
    n_src = len(src)
    if n_src == n_target:
        return src.copy()
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    out = np.interp(x_tgt, x_src, src)
    out = np.maximum(out, 0.0)
    s = float(np.sum(out))
    if s > 1e-10:
        out = out / s
    return out


def _make_diverse_init(n, variant, rng):
    """Create diverse initial non-negative sequences for LP warm-start."""
    t = np.linspace(0, 1, n)
    if variant == 0:
        a = 1.0 - t
    elif variant == 1:
        a = np.exp(-3.0 * t)
    elif variant == 2:
        a = (1.0 - t) * (1.0 + 0.5 * np.cos(2.0 * np.pi * 3.0 * t))
        a = np.maximum(a, 0.0)
    elif variant == 3:
        a = np.exp(-((t - 0.3) ** 2) / 0.02) + 0.5 * np.exp(-((t - 0.7) ** 2) / 0.05)
    elif variant == 4:
        a = (1.0 - t) ** 1.5 * (1.0 + 0.4 * np.cos(2.0 * np.pi * 2.0 * t + 1.0))
        a = np.maximum(a, 0.0)
    elif variant == 5:
        a = t ** 2 * (3.0 - 2.0 * t)
    elif variant == 6:
        a = np.abs(rng.normal(0, 1, n))
    elif variant == 7:
        a = (j / (n - 1.0) - 0.5) ** 2 + 0.3 for j in range(n)
        a = np.array(a)
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 0.0)
    elif variant == 8:
        f1, f2 = 3.0, 7.0
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * f1 * t) + 0.2 * np.cos(2.0 * np.pi * f2 * t + 0.5)
        a = np.maximum(a, 0.0)
    elif variant == 9:
        a = np.exp(-2.0 * t) * (1.0 + 0.5 * np.sin(2.0 * np.pi * 4.0 * t))
        a = np.maximum(a, 0.0)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def propose_candidate():
    """
    Multi-phase optimizer minimizing J(a)=2n*max(conv(a,a))/sum(a)^2.
    Phase 1: Sparse alternating LP at n=200 with 10 diverse restarts.
    Phase 2: LP at n=300 and n=400.
    Phase 3: Smoothed log-sum-exp gradient descent at n=200.
    Phase 4: Subgradient with top-40 peaks and stagnation restart.
    Phase 5: Final LP polish on the global best.
    """
    total_deadline = time.monotonic() + 2350.0
    rng = np.random.default_rng(42)
    best_a = None
    best_score = float('inf')

    # Load parent sequence if available
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            parent_score = _score(parent)
            print(f"Parent: n={parent.size}, score={parent_score:.6f}")
            best_a = parent.copy()
            best_score = parent_score
    except Exception:
        pass

    if best_a is None:
        best_a = np.ones(200) / 200.0
        best_score = _score(best_a)
        print(f"Default: score={best_score:.6f}")

    # ---- Phase 1: LP at n=200, multiple diverse restarts ----
    print("\n=== Phase 1: LP n=200 (10 restarts) ===")
    n1 = 200
    phase_dl = min(total_deadline, time.monotonic() + 900.0)
    inits = [_interpolate_sequence(best_a, n1)]
    for v in [0, 1, 2, 3, 4, 5, 6, 8, 9]:
        inits.append(_make_diverse_init(n1, v, rng))
    for ri, a_init in enumerate(inits):
        if time.monotonic() > phase_dl:
            break
        elapsed_so_far = time.monotonic() - (total_deadline - 900.0)
        budget = max(30.0, 900.0 - elapsed_so_far - 20.0)
        n_it = int(budget / 0.5)  # ~0.5s per LP iter at n=200
        n_it = max(100, min(n_it, 400))
        blend = 0.65 if ri == 0 else 0.5
        pmag = 0.06 if ri == 0 else 0.12
        print(f"  r{ri} (n_it={n_it}):", end=" ", flush=True)
        a_opt, sc = _alternating_lp(a_init, n_it, phase_dl,
                                     blend_init=blend, perturb_every=30,
                                     perturb_mag=pmag, verbose=(ri == 0))
        print(f"score={sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  ** New best: {best_score:.6f}")

    print(f"After Phase 1: best={best_score:.6f}")

    # ---- Phase 2: LP at n=300, n=400 ----
    print("\n=== Phase 2: LP n=300, n=400 ===")
    for n_p in [300, 400]:
        if time.monotonic() > total_deadline - 600:
            break
        phase_dl = min(total_deadline, time.monotonic() + 350.0)
        a_start = _interpolate_sequence(best_a, n_p)
        for ri in range(3):
            if time.monotonic() > phase_dl:
                break
            if ri == 0:
                a_init = a_start.copy()
            else:
                a_init = _interpolate_sequence(best_a, n_p)
                mask = rng.random(n_p) < 0.3
                cnt = max(1, int(mask.sum()))
                a_init[mask[:cnt]] *= rng.uniform(0.4, 2.5, cnt)
                a_init = np.maximum(a_init, 0.0)
                sp = float(np.sum(a_init))
                if sp > 1e-10:
                    a_init = a_init / sp
            elapsed = time.monотonic() if False else time.monotonic()
            budget = max(20.0, phase_dl - elapsed - 10.0)
            n_it = max(50, int(budget / 1.5))
            n_it = min(n_it, 200)
            print(f"  n={n_p} r{ri} (n_it={n_it}):", end=" ", flush=True)
            a_opt, sc = _alternating_lp(a_init, n_it, phase_dl,
                                         blend_init=0.5, perturb_every=25,
                                         perturb_mag=0.1, verbose=False)
            print(f"score={sc:.6f}")
            if sc < best_score:
                best_score = sc
                best_a = a_opt.copy()
                print(f"  ** New best: {best_score:.6f}")

    # ---- Phase 3: Smoothed gradient at n=200 ----
    print("\n=== Phase 3: Smoothed gradient n=200 ===")
    if time.monotonic() < total_deadline - 400:
        phase_dl = min(total_deadline, time.monotonic() + 350.0)
        a_start = _interpolate_sequence(best_a, 200)
        a_opt, sc = _smoothed_gradient(a_start, phase_dl,
                                        n_steps=15000, tau_init=0.02,
                                        tau_decay=0.9994)
        print(f"  Smoothed: {sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  ** New best: {best_score:.6f}")
        # Try a second smoothed run from a perturbed start
        if time.monotonic() < phase_dl - 60:
            a_pert = a_start + rng.normal(0, 0.05, 200)
            a_pert = np.maximum(a_pert, 0.0)
            sp = float(np.sum(a_pert))
            if sp > 1e-10:
                a_pert = a_pert / sp
            phase_dl2 = min(total_deadline, time.monotonic() + 250.0)
            a_opt2, sc2 = _smoothed_gradient(a_pert, phase_dl2,
                                              n_steps=10000, tau_init=0.03,
                                              tau_decay=0.9993)
            if sc2 < best_score:
                best_score = sc2
                best_a = a_opt2.copy()
                print(f"  ** New best (pert): {best_score:.6f}")

    # ---- Phase 4: Subgradient top-k at n=200 ----
    print("\n=== Phase 4: Subgradient top-40 n=200 ===")
    if time.monotonic() < total_deadline - 300:
        phase_dl = min(total_deadline, time.monotonic() + 350.0)
        a_start = _interpolate_sequence(best_a, 200)
        a_opt, sc = _subgradient_topk(a_start, phase_dl,
                                       n_peaks=40, lr_init=0.035,
                                       lr_decay=0.99996)
        print(f"  Subgrad: {sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  ** New best: {best_score:.6f}")

    # ---- Phase 5: Final LP polish ----
    print("\n=== Phase 5: LP polish ===")
    if time.monotonic() < total_deadline - 20:
        polish_dl = total_deadline - 5.0
        n_best = len(best_a)
        n_pol = 200
        a_ds = _interpolate_sequence(best_a, n_pol) if n_best != n_pol else best_a.copy()
        # Two polish runs with different blends
        for bl in [0.3, 0.15]:
            if time.monotonic() > polish_dl - 30:
                break
            a_opt, sc = _alternating_lp(a_ds, 300, polish_dl,
                                         blend_init=bl, perturb_every=60,
                                         perturb_mag=0.02, verbose=True)
            if n_best != n_pol:
                a_up = _interpolate_sequence(a_opt, n_best)
                sc = _score(a_up)
            else:
                a_up = a_opt
            print(f"  Polish (blend={bl}): {sc:.6f}")
            if sc < best_score:
                best_score = sc
                best_a = a_up.copy()
                print(f"  ** New best: {best_score:.6f}")

    print(f"\n=== FINAL: score={best_score:.6f}, n={len(best_a)} ===")
    result = [float(x) for x in best_a]
    try:
        verified = float(evaluate_sequence(result))
        print(f"Verified via evaluate_sequence: {verified:.6f}")
    except Exception:
        verified = best_score
    return result
