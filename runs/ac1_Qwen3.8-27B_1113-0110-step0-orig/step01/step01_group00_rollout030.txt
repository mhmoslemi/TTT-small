import time
from scipy.optimize import linprog, minimize_scalar
import math

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]

def _score(a):
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)

def _build_conv_matrix(f):
    n = len(f)
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        C[j:j+n, j] = f
    return C

def _lp_maxsum(f, M):
    """Maximize sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    n = len(f)
    C = _build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.status == 0:
            return result.x
    except Exception:
        pass
    return None

```python
"""
Hybrid optimizer for minimizing J(a) = 2n*max(conv(a,a))/sum(a)^2.
Uses alternating LP (both max-sum and min-t formulations) combined with
coordinate descent polish and random restarts.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize_scalar


def _fft_convolve(a):
    """Fast full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]


def _score(a):
    """Compute J(a) = 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)


def _build_conv_matrix(f):
    """Build convolution matrix C where (C @ g)[k] = (f*g)[k]."""
    n = len(f)
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = f
    return C


def _lp_maxsum(f, M):
    """Maximize sum(g) s.t. (f*g)[k] <= M for all k, g >= 0. Returns g or None."""
    n = len(f)
    C = _build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.status == 0:
            return result.x
    except Exception:
        pass
    return None


def _lp_mint(f, S):
    """Minimize t s.t. (f*g)[k] <= t for all k, sum(g)=S, g>=0. Returns (g, t) or (None, None)."""
    n = len(f)
    C = _build_conv_matrix(f)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = C
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs', options={'presolve': True})
        if result.status == 0:
            return result.x[:n].copy(), float(result.x[-1])
    except Exception:
        pass
    return None, None

                break
            sum_g = float(np.sum(g))
            if sum_g < 1e-12:
                break
            g = g / sum_g
        else:
            g, t_opt = _lp_mint(a, S)
            if g is None:
                break
            sum_g = float(np.sum(g))
            if sum_g < 1e-12:
                break
            g = g / sum_g
        
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        sum_new = float(np.sum(a_new))
        if sum_new < 1e-12:
            break
        a_new = a_new / sum_new
        score = _score(a_new)
        if score < best_score - 1e-10:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 50 and blend <= 0.06:
                break
        a = a_new
        blend = max(0.05, blend * 0.95)
    
    return best_a, best_score


def _coord_1d_obj(x, D, F, two_i, limit):
    """Objective for 1D coordinate descent: max_k {D[k] + 2xF[k] + x^2 (k==two_i)}."""
    if x < 0.0:
        x = 0.0
    vals = D + 2.0 * x * F
    result = float(np.max(vals))
    if 0 <= two_i < limit:
        val_2i = D[two_i] + x * x
        if val_2i > result:
            result = val_2i
    return result


def _coordinate_descent(a_init, max_sweeps, deadline):
    """Coordinate descent to minimize max(conv(a,a)) with a >= 0."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = _score(a)
    limit = 2 * n - 1
    
    c = _fft_convolve(a)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        improved = 0
        for i in range(n):
            if time.monotonic() > deadline:
                break
            a_i = a[i]
            if a_i < 1e-14:
                continue
            
            k_lo = i  # max(0, i) = i since i >= 0
            k_hi = min(limit - 1, i + n - 1)
            
            # F[k] = a[k-i] for k in [k_lo, k_hi], 0 elsewhere
            F = np.zeros(limit)
            F[k_lo:k_hi + 1] = a[k_lo - i:k_hi - i + 1]
            
            # D[k] = c[k] - 2*a_i*F[k] - a_i^2*(k==2i)
            D = c - 2.0 * a_i * F
            two_i = 2 * i
            if 0 <= two_i < limit:
                D[two_i] -= a_i * a_i
            
            # Solve 1D problem
            upper = max(2.0 * a_i, 0.05)
            res = minimize_scalar(_coord_1d_obj, args=(D, F, two_i, limit),
                                  bounds=(0.0, upper), method='bounded',
                                  options={'maxiter': 40, 'xatol': 1e-12})
            x_star = max(0.0, float(res.x))
            
            delta = x_star - a_i
            if abs(delta) > 1e-14:
                c[k_lo:k_hi + 1] += 2.0 * delta * F[k_lo:k_hi + 1]
                if 0 <= two_i < limit:
                    c[two_i] += delta * delta
                a[i] = x_star
                improved += 1
        
        # Normalize
        s = float(np.sum(a))
        if s < 1e-10:
            break
        a = a / s
        c = _fft_convolve(a)
        
        score = _score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if improved == 0:
            break
    
    return best_a, best_score


def _interpolate_sequence(src, n_target):
    """Interpolate a sequence to a new length."""
    n_src = len(src)
    if n_src == n_target:
        return src.copy()
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    out = np.interp(x_tgt, x_src, src)
    out = np.maximum(out, 0.0)
    s = float(np.sum(out))
    if s > 1e-10:
        out = out / s
    else:
        out = np.ones(n_target) / n_target
    return out
