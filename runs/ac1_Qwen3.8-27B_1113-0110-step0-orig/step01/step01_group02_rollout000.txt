"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using Mitchell's LP-based algorithm
combined with multi-peak subgradient descent.
"""
import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]

def _score_from_conv(c, n):
    return 2.0 * n * np.max(c)

def _mitchell_lp_step(a, M, max_time_per_lp=5.0):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0.
    Returns normalized b or None if LP fails."""
    n = len(a)
    size = 2 * n - 1
    
    # Build constraint matrix A where (A @ b)[k] = conv(a,b)[k]
    A = np.zeros((size, n))
    for k in range(size):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            js = np.arange(lo, hi + 1)
            A[k, js] = a[k - js]
    
    # LP: max 1^T b s.t. A b <= M, b >= 0
    c_obj = -np.ones(n)  # minimize -sum = maximize sum
    res = linprog(c_obj, A_ub=A, b_ub=np.full(size, M),
                  bounds=[(0, None)] * n,
                  method='highs', options={'time_limit': max_time_per_lp})
    
    if res.status != 0:
        return None
    
    b = res.x
    s = b.sum()
    if s < 1e-15:
        return None
    return b / s

def _subgradient_step(a, c, top_k=5, eta=0.1, momentum=None, mu=0.7):
    """Multi-peak subgradient step."""
    n = len(a)
    size = len(c)
    
    # Get top-k peaks
    sorted_idx = np.argsort(c)[::-1]
    peaks = sorted_idx[:min(top_k, size)]
    peak_vals = c[peaks]
    peak/max_val = peak_vals.max()

    if momentum is None:
        momentum = g
    else:
        momentum = mu * momentum + (1 - mu) * g
    
    return g, momentum

def _make_sin(n, phase, amp):
    x = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.sin(2.0 * np.pi * x / n + phase)
    return np.maximum(a, 0.05)


def _make_decay(n, rate):
    x = np.arange(n, dtype=np.float64)
    return np.exp(-rate * x / n) + 0.1


def _make_poly(n, c1, c2):
    x = np.linspace(0.0, 1.0, n)
    return np.maximum(1.0 + c1 * x + c2 * x * x, 0.05)


def _make_two_bump(n, c1, s1, c2, s2):
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((x - n * c1) / (n * s1)) ** 2) + np.exp(-0.5 * ((x - n * c2) / (n * s2)) ** 2)
    return a + 0.1


def propose_candidate(seed=42, budget_s=2350, **kwargs):
    """
    Main search: Mitchell LP + multi-peak subgradient with diverse initializations.
    Tries multiple n values and sequence shapes, allocates time adaptively.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 25.0

    best_overall = None
    best_overall_score = np.inf

    # Build diverse candidates
    candidates = []

    # Previous best
    try:
        hs = np.array(height_sequence_1, dtype=np.float64)
        hs = np.clip(hs, 0, None)
        if hs.size > 100 and hs.sum() > 0.01:
            candidates.append(("prev", hs.copy()))
    except (NameError, Exception):
        pass

    # Sinusoidal variants (proven effective)
    for n in [1800, 2000, 2200, 2500, 3000]:
        for phase, amp in [(np.pi / 4, 0.4), (np.pi / 3, 0.5), (np.pi / 6, 0.3)]:
            candidates.append((f"sin{n}_p{int(phase*100)}_a{int(amp*100)}", _make_sin(n, phase, amp)))

    # Exponential decay
    for n in [2000, 2500, 3000]:
        for rate in [0.6, 1.0, 1.5]:
            candidates.append((f"dec{n}_r{rate}", _make_decay(n, rate)))

    # Polynomial
    for n in [2000, 2500]:
        for c1, c2 in [(0.5, -0.3), (-0.4, 0.6), (0.8, -0.5)]:
            candidates.append((f"poly{n}_{c1}_{c2}", _make_poly(n, c1, c2)))

    # Two-bump
    for n in [2000, 3000]:
        a = _make_two_bump(n, 0.25, 0.12, 0.75, 0.15)
        candidates.append((f"2b{n}", a))

    # Cosine window (rising then falling)
    for n in [2000, 2500]:
        x = np.arange(n, dtype=np.float64)
        a = 0.3 + 0.7 * np.cos(np.pi * x / n + 0.3) ** 2
        a = np.maximum(a, 0.05)
        candidates.append((f"cosw{n}", a))

    print(f"Total candidates: {len(candidates)}")

    # Score all candidates
    scored = []
    for name, a0 in candidates:
        a0 = np.maximum(a0, 0)
        s = a0.sum()
        if s < 1e-12:
            continue
        a0 = a0 / s
        c0 = _fft_convolve(a0)
        sc = _score(c0, len(a0))
        scored.append((sc, name, a0))

    scored.sort(key=lambda x: x[0])
    print("Top initial scores:")
    for i in range(min(12, len(scored))):
        print(f"  {i+1}. {scored[i][1]}: {scored[i][0]:.6f}")

    # Allocate time to top candidates with geometric weighting
    n_top = min(len(scored), 8)
    remaining = deadline - time.monotonic()
    weights = [max(1.0, (n_top - i) ** 1.3) for i in range(n_top)]
    total_w = sum(weights)
    times = [remaining * w / total_w for w in weights]

    for i in range(n_top):
        if time.monotonic() > deadline:
            break

        init_sc, name, a0 = scored[i]
        t_alloc = min(times[i], max(15.0, deadline - time.monotonic()))

        print(f"\n>>> Optimizing '{name}' (n={len(a0)}, init={init_sc:.6f}, t={t_alloc:.0f}s)")

        a_opt, s_opt = _optimize_combined(a0, t_alloc, rng, use_lp=True,
                                           lp_ratio=0.3, n_cons=min(800, 2 * len(a0)))
        ev = evaluate_sequence(a_opt.tolist())
        print(f"    Result: internal={s_opt:.6f}, eval={ev:.6f}")

        if np.isfinite(ev) and ev < best_overall_score:
            best_overall_score = ev
            best_overall = a_opt.copy()
            print(f"    *** NEW BEST: {ev:.6f} ***")

    # If time remains, do extended polishing from the global best
    remaining = deadline - time.monotonic()
    if best_overall is not None and remaining > 20:
        print(f"\n=== Final polishing ({remaining:.0f}s) ===")
        a_p, s_p = _optimize_combined(best_overall, remaining * 0.7, rng,
                                       use_lp=True, n_cons=min(1500, 2 * len(best_overall)))
        ev_p = evaluate_sequence(a_p.tolist())
        print(f"  Polished: {ev_p:.6f}")
        if np.isfinite(ev_p) and ev_p < best_overall_score:
            best_overall_score = ev_p
            best_overall = a_p.copy()

        remaining2 = deadline - time.monotonic()
        if remaining2 > 10:
            a_p2, s_p2 = _optimize_combined(best_overall, remaining2, rng,
                                             use_lp=False, n_cons=500)
            ev_p2 = evaluate_sequence(a_p2.tolist())
            print(f"  Polished (SG only): {ev_p2:.6f}")
            if np.isfinite(ev_p2) and ev_p2 < best_overall_score:
                best_overall_score = ev_p2
                best_overall = a_p2.copy()

    # Final check
    if best_overall is not None:
        final_sc = evaluate_sequence(best_overall.tolist())
        elapsed = time.monotonic() - total_start
        print(f"\n=== FINAL: score={final_sc:.6f}, n={len(best_overall)}, elapsed={elapsed:.0f}s ===")
        return [float(x) for x in best_overall]

    # Fallback
    print("Falling back to uniform")
    return [1.0] * 1000
