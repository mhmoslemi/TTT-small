    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.

    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:

    # Check each element in the list for validity
    for x in sequence:

        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

def _pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _fft_square(a):
    n = int(a.size)
    L = 2 * n - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, n=N)
    b = np.fft.irfft(fa * fa, n=N).real[:L]
    b = np.maximum(b, 0.0)
    return b


def _fft_grad(weights, a):
    n = int(a.size)
    L = int(weights.size)
    N = _pow2(L + n - 1)
    fw = np.fft.rfft(weights, n=N)
    fa = np.fft.rfft(a, n=N)
    c = np.fft.irfft(fw * np.conj(fa), n=N)[:n]
    return 2.0 * c


def _internal_score(a, b=None):
    if b is None:
        b = _fft_square(a)
    s = float(a.sum())
    if s <= 1e-300:
        return float("inf"), b, 0.0
    maxb = float(b.max())
    score = float(2.0 * a.size * maxb / (s * s))
    return score, b, maxb


def _make_weights(b, focus, eps=0.02):
    L = int(b.size)
    maxb = float(b.max())
    if maxb <= 1e-300:
        return np.full(L, 1.0 / L, dtype=np.float64), maxb
    with np.errstate(over="ignore", under="ignore"):
        z = (maxb - b) / maxb
        p = np.exp(-z * float(focus))
    s = float(p.sum())
    if s <= 1e-12:
        p = np.zeros(L, dtype=np.float64)
        p[int(np.argmax(b))] = 1.0
    else:
        p = (1.0 - eps) * (p / s) + eps / L
    return p, maxb


def _random_positive(n, rng, amp=3.0):
    n = int(n)
    z = rng.normal(0.0, amp, size=n)
    z -= z.max()
    a = np.exp(z)
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    return a


def _anti_log_init(n, rng, amp=3.0, smooth=False):
    n = int(n)
    m = n // 2
    if m < 1:
        return np.ones(n, dtype=np.float64) / max(1, n)
    x = np.arange(m, dtype=np.float64)
    if smooth:
        v = np.zeros(m, dtype=np.float64)
        for f in (1, 3, 5, 7, 11):
            v += (rng.normal(0.0, 1.0) / max(1, f)) * np.sin(
                2.0 * np.pi * f * x / m + rng.uniform(0.0, 2.0 * np.pi)
            )
        v += 0.1 * rng.normal(0.0, 1.0, m)
    else:
        v = rng.normal(0.0, 1.0, m)
    v = v - float(v.mean())
    std = float(v.std())
    if std > 1e-12:
        v = (float(amp) * v) / std

    z = np.zeros(n, dtype=np.float64)
    z[:m] = v
    if n % 2 == 0:
        z[m:] = -v[::-1]
    else:
        z[m + 1:] = -v[::-1]
        z[m] = 0.0

    z = z - z.max()
    a = np.exp(z)
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    return a


def _greedy_support(n, rng, shuffle=True):
    n = int(n)
    m = n // 2
    counts = np.zeros(2 * n - 1, dtype=np.int64)
    selected = np.empty(m, dtype=np.int64)
    order = np.arange(m, dtype=np.int64)
    if shuffle:
        rng.shuffle(order)

    t = 0
    for idx_val in order:
        idx = int(idx_val)
        l = idx
        r = n - 1 - idx
        if t == 0:
            y = l
        else:
            s = selected[:t]
            affected_l = int(np.max(counts[s + l]))
            self_l = int(counts[2 * l])
            metric_l = max(affected_l + 2, self_l + 1)

            affected_r = int(np.max(counts[s + r]))
            self_r = int(counts[2 * r])
            metric_r = max(affected_r + 2, self_r + 1)

            y = l if metric_l <= metric_r else r

        if t > 0:
            counts[selected[:t] + y] += 2
        counts[2 * y] += 1
        selected[t] = y
        t += 1

    return selected


def _greedy_x(n, rng, shuffle=True):
    sel = _greedy_support(n, rng, shuffle=shuffle)
    m = n // 2
    x = np.full(m, 1e-6, dtype=np.float64)
    left = sel < m
    if np.any(left):
        x[sel[left]] = 1.0 - 1e-6
    return x


def _a_from_x(x, n=None):
    x = np.asarray(x, dtype=np.float64).ravel()

    if n is None:

    t = 1.0 / m
    a = np.empty(n, dtype=np.float64)
    a[:m] = t * x
    a[m:] = t * (1.0 - x[::-1])
    return a
