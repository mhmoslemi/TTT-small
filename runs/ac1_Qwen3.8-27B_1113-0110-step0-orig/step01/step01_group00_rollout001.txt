"""
Multi-phase optimizer for minimizing J(a) = 2n * max(conv(a,a)) / sum(a)^2.
Primary engine: alternating LP with adaptive blend, multiple n, diverse inits.
Secondary: subgradient on larger n, LP polish, random restarts.
Key improvements: wider n sweep, more inits, shrinking M trick, distillation.
"""
import numpy as np
import time
import math
from scipy.optimize import linprog


def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]


def _score(a):
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)


def _build_conv_matrix(f):
    n = len(f)
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = f
    return C


def _lp_solve(f, M, timeout=None):
    n = len(f)
    C = _build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': timeout} if timeout else {'presolve': True})
        if result.status == 0:
            return result.x
    except Exception:
        pass
    return None


def _alternating_lp(a_init, max_iters, deadline, initial_blend=0.8, shrink=1.0, verbose=False):
    a = a_init.copy().astype(np.float64)
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = _score(a)
    blend = initial_blend

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        M = float(np.max(c))
        # Try shrinking M slightly to push for improvement
        M_try = M * shrink
        g = _lp_solve(a, M_try)
        if g is None and shrink < 1.0:
            g = _lp_solve(a, M)
        if g is None:
            break
        sum_g = float(np.sum(g))
        if sum_g < 1e-12:
            break
        g = g / sum_g
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        sum_new = float(np.sum(a_new))
        if sum_new < 1e-12:
            break
        a_new = a_new / sum_new
        score = _score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            if verbose and it % 50 == 0:
                print(f"      iter {it}: score={best_score:.6f}")
        a = a_new
        blend = max(0.05, blend * 0.98)

    return best_a, best_score


def _subgradient(a_init, max_iters, lr_init, lr_decay, n_peaks, deadline):
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    best_a = a.copy()
    best_score = _score(a)
    lr = lr_init

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        max_c = float(np.max(c))
        score = 2.0 * n * max_c / (S * S)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if n_peaks >= len(c):
            top_k = np.arange(len(c))
        else:
            top_k = np.argpartition(c, -n_peaks)[-n_peaks:]

        n = len(a)
        s = np.zeros(n)
        for k in top_k:
            ki = int(k)
            for i in range(n):
                di = ki - i
                if 0 <= di < n:
                    s[i] += 2.0 * a[di]
        s /= len(top_k)
        max_s = float(np.max(s))
        if max_s > 1e-15:
            s = s / max_s
            a = a - lr * s
            a = np.maximum(a, 0.0)
            S_new = float(np.sum(a))
            if S_new > 1e-10:
                a = a * S / S_new
        lr *= lr_decay

    return best_a, best_score
