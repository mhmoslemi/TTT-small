import time

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]

def _smooth_objective_and_grad(a, beta):
    """Compute smooth (logsumexp) objective and gradient."""
    n = len(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return 1e10, np.zeros(n)
    a_n = a / s
    c = _fft_convolve(a_n)
    # logsumexp
    bc = beta * c
    bc_max = bc.max()
    logsumexp = bc_max + np.log(np.sum(np.exp(bc - bc_max)))
    score = 2.0 * n * logsumexp / beta
    # weights
    w = np.exp(bc - logsumexp)
    # gradient: 2n/beta * correlate(w, a_n, 'valid')
    grad = (2.0 * n / beta) * np.correlate(w, a_n, mode='valid')
    return score, grad

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    """Return the next power of 2 >= n."""
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _fft_convolve(a):
    """Fast linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]


def _smooth_obj_grad(a, beta):
    """Compute smooth (logsumexp-based) objective and gradient.
    
    Returns (score, gradient) where score = 2n * logsumexp(beta*c) / beta
    and gradient is w.r.t. a (assuming a is already normalized to sum=1).
    """
    n = len(a)
    c = _fft_convolve(a)
    bc = beta * c
    bc_max = bc.max()
    logsumexp_val = bc_max + np.log(np.sum(np.exp(bc - bc_max)))
    score = 2.0 * n * logsumexp_val / beta
    # Weights for logsumexp
    w = np.exp(bc - logsumexp_val)
    # Gradient: (2n/beta) * correlate(w, a, 'valid')
    grad = (2.0 * n / beta) * np.correlate(w, a, mode='valid')
    return score, grad

def _optimize_lbfgs(a0, beta=100.0, max_iter=80):
    """Use L-BFGS-B on the smoothed objective for fine-tuning."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, np.inf
    a = a / s

    def func(x):
        x_clip = np.maximum(x, 0.0)
        sc, _ = _smooth_obj_grad(x_clip, beta)
        return sc

    def grad(x):
        x_clip = np.maximum(x, 0.0)
        _, g = _smooth_obj_grad(x_clip, beta)
        return g

    try:
        result = minimize(func, a, method='L-BFGS-B', jac=grad,
                          bounds=[(0, None)] * n,
                          options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 20})
        a_opt = np.maximum(result.x, 0.0)
        s = a_opt.sum()
        if s < 1e-15:

        a_opt = a_opt / s

        score = 2.0 * n * float(np.max(c))
        return a_opt, score
    except Exception as e:
        print(f"    L-BFGS error: {e}")
        return a0, np.inf


def _multi_top_k_subgrad(a0, time_budget, rng, top_k=3, beta=20.0):

    mu = 0.75

        score = 2.0 * n * float(np.max(c))

            no_improve = 0
        else:

            no_improve = 0

        # Compute weighted subgradient using top-k
        # Use softmax weights over the top-k values
        c_sorted_idx = np.argsort(c)[::-1]

        if g_norm < 1e-15:
            break
        g = g / g_norm

        momentum = mu * momentum + (1.0 - mu) * g
        a = a - eta * momentum
        a = np.maximum(a, 1e-20)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            eta *= 0.5
            it += 1
            continue
        a = a / s

        if no_improve > 200:
            eta *= 0.95
        elif no_improve < 30:
            eta = min(eta * 1.005, 0.2)

        it += 1

    return best_a, best_score


def _structured_perturb(a, rng, strength=1.0):
    """Add structured perturbations to explore the landscape."""
    n = len(a)
    a_pert = a.copy()
    x = np.arange(n, dtype=np.float64)
    n_perts = int(rng.integers(2, 6))

    for _ in range(n_perts):
        pert_type = int(rng.integers(0, 5))
        if pert_type == 0:
            # Sinusoid at random frequency
            freq = rng.uniform(0.5, 4.0)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.02, 0.08) * strength
            a_pert += amp * np.sin(2 * np.pi * freq * x / n + phase)
        elif pert_type == 1:
            # Chirp
            f1 = rng.uniform(0.5, 3.0)
            f2 = rng.uniform(3.0, 8.0)
            amp = rng.uniform(0.01, 0.05) * strength
            phase = 2 * np.pi * (f1 * x / n + (f2 - f1) * x**2 / (2 * n**2))
            a_pert += amp * np.cos(phase)
        elif pert_type == 2:
            # Gaussian bump
            center = rng.uniform(n * 0.1, n * 0.9)
            width = rng.uniform(n / 30.0, n / 8.0)
            amp = rng.uniform(0.02, 0.08) * strength
            a_pert += amp * np.exp(-0.5 * ((x - center) / width) ** 2)
        elif pert_type == 3:
            # Block modulation
            block = max(1, int(n / 30))
            start = int(rng.integers(0, n - block))
            a_pert[start:start + block] *= np.exp(rng.normal(0, 0.1, block) * strength)
        else:
            # High-frequency small noise
            a_pert += rng.normal(0, 0.01 * strength, n)

    a_pert = np.maximum(a_pert, 0.0)
    s = a_pert.sum()
    if s > 1e-15:
        a_pert = a_pert / s
    return a_pert


def _try_candidate(a0, time_budget, rng, label=""):
    """Run a full optimization pipeline on a candidate. Returns (best_a, best_score)."""
    n = len(a0)
    a0 = np.maximum(a0.astype(np.float64), 0.0)
    s = a0.sum()
    if s < 1e-15:
        return a0, np.inf
    a0 = a0 / s

    c0 = _fft_convolve(a0)
    init_score = 2.0 * n * float(np.max(c0))

    best_a = a0.copy()
    best_score = init_score

    # Phase 1: Smooth subgradient
    t1 = time_budget * 0.30
    a1, s1 = _optimize_smooth(a0, t1, rng, beta=40.0)
    eval1 = evaluate_sequence(a1.tolist())
    if np.isfinite(eval1) and eval1 < best_score:
        best_score = eval1
        best_a = a1.copy()

    # Phase 2: Multi top-k subgradient
    t2 = time_budget * 0.20
    a2, s2 = _multi_top_k_subgrad(best_a, t2, rng, top_k=3, beta=15.0)
    eval2 = evaluate_sequence(a2.tolist())
    if np.isfinite(eval2) and eval2 < best_score:
        best_score = eval2
        best_a = a2.copy()

    # Phase 3: L-BFGS polish
    t3_start = time.monotonic()
    a3, s3 = _optimize_lbfgs(best_a, beta=150.0, max_iter=60)
    eval3 = evaluate_sequence(a3.tolist())
    t3_elapsed = time.monotonic() - t3_start
    if np.isfinite(eval3) and eval3 < best_score:
        best_score = eval3
        best_a = a3.copy()

    # Phase 4: Sharp subgradient
    t4 = max(2.0, time_budget * 0.30 - (time.monotonic() - t3_start))
    a4, s4 = _optimize_sharp(best_a, t4, rng)
    eval4 = evaluate_sequence(a4.tolist())
    if np.isfinite(eval4) and eval4 < best_score:
        best_score = eval4
        best_a = a4.copy()

    # Phase 5: Perturbation + quick re-optimize
    remaining = time_budget - (time.monotonic() - (total_start_time)) if 'total_start_time' in dir() else time_budget * 0.1
    for attempt in range(3):
        if time.monotonic() > (deadline_global - 5) if 'deadline_global' in dir() else False:
            break
        a_p = _structured_perturb(best_a, rng, strength=0.8)
        a_p2, s_p2 = _optimize_smooth(a_p, min(10.0, max(2.0, time_budget * 0.08)), rng, beta=30.0)
        eval_p = evaluate_sequence(a_p2.tolist())
        if np.isfinite(eval_p) and eval_p < best_score:
            best_score = eval_p
            best_a = a_p2.copy()

    return best_a, best_score


# Global variables for time tracking in _try_candidate
deadline_global = 0.0
total_start_time = 0.0


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Minimizes 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
    
    Strategy: Multi-phase optimization combining:
    - Smooth subgradient descent (logsumexp) for global exploration
    - Multi top-k subgradient for balancing multiple peaks
    - L-BFGS-B on smoothed objective for local refinement
    - Sharp subgradient for final polishing
    - Structured perturbations (sinusoids, chirps, bumps) for diversity
    - Multiple diverse initializations (parametric families + provided sequence)
    """
    global deadline_global, total_start_time
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline_global = total_start + budget_s - 25.0
    total_start_time = total_start

    best_overall = None
    best_overall_score = np.inf

    # Build candidate initializations
    candidates = []

    # 1. Provided sequence (current best)
    try:
        hs = np.array(height_sequence_1, dtype=np.float64)
        hs = np.clip(hs, 0, None)
        if hs.size > 100 and hs.sum() > 0.01:
            candidates.append(("provided", hs.copy()))
    except (NameError, Exception):
        pass

    # 2. Sinusoidal family (previous best was in this family)
    for n in [1800, 2000, 2500]:
        x = np.arange(n, dtype=np.float64)
        for A in [0.35, 0.4, 0.5, 0.6]:
            for phi_deg in [45, 60, 90]:
                phi = np.pi * phi_deg / 180.0
                a = 1.0 + A * np.sin(2 * np.pi * x / n + phi)
                a = np.maximum(a, 0.01)
                candidates.append((f"sin_{n}_{A}_{phi_deg}", a))

    # 3. Multi-harmonic
    for n in [2000, 2500]:
        x = np.arange(n, dtype=np.float64)
        # Two harmonics with phase cancellation
        a = 1.0 + 0.35 * np.sin(2 * np.pi * x / n + np.pi / 4) + 0.12 * np.cos(4 * np.pi * x / n + np.pi / 6)
        a = np.maximum(a, 0.01)
        candidates.append((f"2har_{n}", a))
        # Different combination
        a2 = 1.0 + 0.4 * np.cos(2 * np.pi * x / n + 0.3) + 0.1 * np.sin(6 * np.pi * x / n)
        a2 = np.maximum(a2, 0.01)
        candidates.append((f"2har2_{n}", a2))
        # Three harmonics
        a3 = 1.0 + 0.3 * np.sin(2 * np.pi * x / n + np.pi / 3) + 0.1 * np.cos(4 * np.pi * x / n) + 0.05 * np.sin(6 * np.pi * x / n + 1.0)
        a3 = np.maximum(a3, 0.01)
        candidates.append((f"3har_{n}", a3))

    # 4. Power-law edge tapering
    for n in [2000, 2500]:
        x = np.arange(n, dtype=np.float64)
        t = (x + 0.5) / n
        for alpha in [0.4, 0.7, 1.0, 1.5]:
            a = t**alpha * (1.0 - t)**alpha + 0.05
            candidates.append((f"pow_{n}_{alpha}", a))

    # 5. Gaussian envelope modulated sine
    for n in [2000, 2500]:
        x = np.arange(n, dtype=np.float64)
        t = x / n
        for gs in [0.35, 0.5]:
            env = np.exp(-0.5 * ((t - 0.5) / gs) ** 2)
            a = env * (1.0 + 0.5 * np.sin(2 * np.pi * x / n + np.pi / 4)) + 0.08
            candidates.append((f"genv_{n}_{gs}", a))

    # 6. Exponential decay (previous good init)
    for n in [2000, 3000]:
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-0.8 * x / n) + 0.1
        candidates.append((f"decay_{n}", a))

    # 7. Cosine window (Blackman-like)
    for n in [2000, 2500]:
        x = np.arange(n, dtype=np.float64)
        t = x / (n - 1)
        a = 0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t)
        a = np.maximum(a, 0.01)
        candidates.append((f"blackman_{n}", a))

    # 8. Random structured (low-frequency random)
    for n in [2000, 2500]:
        x = np.arange(n, dtype=np.float64)
        a = np.ones(n)
        for freq in [1, 2, 3, 5]:
            amp = rng.uniform(0.05, 0.2)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.sin(2 * np.pi * freq * x / n + phase)
        a = np.maximum(a, 0.05)
        candidates.append((f"rand_{n}", a))

    print(f"Total candidates: {len(candidates)}")

    # Quick initial evaluation
    scored = []
    for name, a0 in candidates:
        a0 = np.maximum(a0, 0)
        s = a0.sum()
        if s < 1e-15:
            continue
        a0 = a0 / s
        c0 = _fft_convolve(a0)
        init_score = 2.0 * len(a0) * float(np.max(c0))
        scored.append((init_score, name, a0))

    scored.sort(key=lambda x: x[0])
    print("Initial scores (top 12):")
    for i in range(min(12, len(scored))):
        sc, name, arr = scored[i]
        print(f"  {i+1}. {name}: {sc:.6f} (n={len(arr)})")

    # Select top candidates
    n_top = min(len(scored), 12)
    remaining_time = deadline_global - time.monotonic()

    # Time allocation: weighted by rank
    weights = []
    for i in range(n_top):
        w = max(0.5, (n_top - i) ** 1.3)
        weights.append(w)
    total_w = sum(weights)
    times = [remaining_time * w / total_w for w in weights]

    for i in range(n_top):
        if time.monotonic() > deadline_global:
            break

        init_score, name, a0 = scored[i]
        t_alloc = min(times[i], max(3.0, deadline_global - time.monotonic()))

        print(f"\n[{name}] n={len(a0)}, init={init_score:.6f}, budget={t_alloc:.0f}s")

        a_cand, s_cand = _try_candidate(a0, t_alloc, rng, label=name)
        eval_cand = evaluate_sequence(a_cand.tolist())
        print(f"  Result: {eval_cand:.6f}")

        if np.isfinite(eval_cand) and eval_cand < best_overall_score:
            best_overall_score = eval_cand
            best_overall = a_cand.copy()
            print(f"  *** NEW BEST: {eval_cand:.6f} ***")

    # If we have a good solution, do additional polishing with different strategies
    remaining = deadline_global - time.monotonic()
    if best_overall is not None and remaining > 20:
        print(f"\n=== Final polishing phase ({remaining:.0f}s) ===")
        n_best = len(best_overall)

        # Try different beta values for smooth optimization
        for beta_try in [20.0, 80.0, 200.0]:
            if time.monotonic() > deadline_global - 10:
                break
            t_sub = min(15.0, (deadline_global - time.monotonic()) * 0.3)
            a_p, s_p = _optimize_smooth(best_overall, t_sub, rng, beta=beta_try)
            eval_p = evaluate_sequence(a_p.tolist())
            print(f"  [beta={beta_try:.0f}] {eval_p:.6f}")
            if np.isfinite(eval_p) and eval_p < best_overall_score:
                best_overall_score = eval_p
                best_overall = a_p.copy()

        # L-BFGS with higher beta (sharper)
        if time.monotonic() < deadline_global - 10:
            a_f, s_f = _optimize_lbfgs(best_overall, beta=300.0, max_iter=80)
            eval_f = evaluate_sequence(a_f.tolist())
            print(f"  [L-BFGS sharp] {eval_f:.6f}")
            if np.isfinite(eval_f) and eval_f < best_overall_score:
                best_overall_score = eval_f
                best_overall = a_f.copy()

        # Sharp subgradient final push
        remaining2 = deadline_global - time.monotonic()
        if remaining2 > 8:
            a_f2, s_f2 = _optimize_sharp(best_overall, remaining2 * 0.8, rng)
            eval_f2 = evaluate_sequence(a_f2.tolist())
            print(f"  [sharp final] {eval_f2:.6f}")
            if np.isfinite(eval_f2) and eval_f2 < best_overall_score:
                best_overall_score = eval_f2
                best_overall = a_f2.copy()

        # Try a few more perturbations
        remaining3 = deadline_global - time.monotonic()
        if remaining3 > 15:
            for _ in range(4):
                if time.monotonic() > deadline_global - 5:
                    break
                a_pert = _structured_perturb(best_overall, rng, strength=1.2)
                a_pp, s_pp = _optimize_smooth(a_pert, min(8.0, (deadline_global - time.monotonic()) / 2), rng, beta=30.0)
                eval_pp = evaluate_sequence(a_pp.tolist())
                if np.isfinite(eval_pp) and eval_pp < best_overall_score:
                    best_overall_score = eval_pp
                    best_overall = a_pp.copy()
                    print(f"  [perturb] NEW BEST: {eval_pp:.6f}")

    # Final verification
    if best_overall is not None:
        final_score = evaluate_sequence(best_overall.tolist())
        elapsed = time.monotonic() - total_start
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(best_overall)}, time={elapsed:.1f}s ===")
        return [float(x) for x in best_overall]
    else:
        a = np.ones(1000, dtype=np.float64)
        print("FALLBACK: uniform sequence")
        return a.tolist()
