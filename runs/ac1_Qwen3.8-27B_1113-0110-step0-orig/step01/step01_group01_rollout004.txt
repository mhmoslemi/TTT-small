"""
Search for non-negative step-function heights minimizing
    2 * n * max(convolve(a,a)) / (sum(a))^2

Strategy:
  1. Fast initialization with structured sequences at multiple lengths.
  2. Low-dimensional basis optimization (L-BFGS on smooth max).
  3. Multi-temperature mirror descent with top-K lag tracking (main phase).
  4. Binary sequence simulated annealing.
  5. Final local polish with coordinate descent.

Key differences from prior approach:
  - Exact top-K lag gradient instead of pure smooth max
  - Multi-phase temperature annealing with re-heating
  - Diverse population management
  - Adaptive n selection based on best-per-n tracking
"""

import time
import numpy as np

try:
    import scipy.optimize as _scipy_optimize
except Exception:
    _scipy_optimize = None


def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def normalize_positive(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, float(floor), None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def softmax_seq(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = z - float(np.max(z))
    z = np.clip(z, -60.0, 0.0)
    e = np.exp(z)
    e = np.maximum(e, 1e-300)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(z.size, 1.0 / float(z.size), dtype=np.float64)
    return e / s


def convolve_fft(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if N is None:
        N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]


def score_from_c(c, n):
    cmax = float(np.max(c))
    if cmax <= 0.0 or not np.isfinite(cmax):
        return 0.0
    return float(2.0 * n * cmax)


def grad_topk(c, a, fa, N, n, K=20):
    """Gradient using top-K lags with soft weighting."""
    L = c.size
    if L == 0 or n == 0:
        return np.zeros(n, dtype=np.float64)
    cmax = float(np.max(c))
    if cmax <= 0.0 or not np.isfinite(cmax):
        return np.zeros(n, dtype=np.float64)
    # Top-K lags
    Keff = min(K, L)
    top_idx = np.argpartition(c, -Keff)[-Keff:]
    top_vals = c[top_idx]
    top_max = float(np.max(top_vals))
    tau = max(0.01 * top_max, 1e-12)
    z = (top_vals - top_max) / tau
    z = np.clip(z, -50.0, 0.0)
    w_top = np.exp(z)
    w_top = w_top / float(w_top.sum())
    # Build full weight vector
    w = np.zeros(L, dtype=np.float64)
    for idx, wt in zip(top_idx, w_top):
        w[idx] = wt
    # Compute gradient via FFT
    w_rev = w[::-1]
    fw = np.fft.rfft(w_rev, N)
    conv = np.fft.irfft(fa * fw, N)
    g = 2.0 * conv[n - 1:2 * n - 1][::-1]
    return g


def smooth_score(c, n, tau):
    cmax = float(np.max(c))
    if cmax <= 0.0 or not np.isfinite(cmax) or tau <= 0.0:
        return 0.0
    z = (c - cmax) / tau
    z = np.clip(z, -50.0, 0.0)
    s = float(np.exp(z).sum())
    if not np.isfinite(s) or s <= 0.0:
        return cmax
    return float(cmax + tau * np.log(s))


def internal_score(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.inf
    a = a / s
    c = convolve_fft(a, N)
    return score_from_c(c, n)


def make_uniform(n):
    return np.ones(max(1, int(n)), dtype=np.float64) / float(max(1, int(n)))


def make_antisymmetric_perturb(n, eps, seed=0):
    rng = np.random.default_rng(seed)
    a = np.ones(n, dtype=np.float64)
    # Antisymmetric perturbation: a[i] * a[n-1-i] product reduced
    i = np.arange(n, dtype=np.float64)
    center = 0.5 * (n - 1)
    f = np.sin(np.pi * (i - center) / n)
    a = a + eps * f
    return normalize_positive(a, floor=1e-12)


def make_periodic_block(n, block_len, pattern, seed=0):
    rng = np.random.default_rng(seed)
    a = np.zeros(n, dtype=np.float64)
    for start in range(0, n, block_len):
        end = min(start + block_len, n)
        a[start:end] = pattern[:end - start]
    return normalize_positive(a, floor=1e-12)


def build_basis(n, n_modes=24):
    i = np.arange(n, dtype=np.float64)
    x = (i - 0.5 * (n - 1)) / max(1.0, 0.5 * n)
    cols = []
    # Antisymmetric odd modes
    for m in range(1, 8):
        cols.append(np.sin(np.pi * m * x))
    # Symmetric even modes
    for m in range(0, 5):
        cols.append(np.cos(np.pi * m * x))
    # Periodic modes
    for m in range(1, 7):
        t = 2.0 * np.pi * m * i / n
        cols.append(np.sin(t) - np.sin(t).mean())
        cols.append(np.cos(t) - np.cos(t).mean())
    # Polynomial
    cols.append(x - x.mean())
    cols.append(x**2 - (x**2).mean())
    B = np.column_stack(cols[:n_modes])
    B -= B.mean(axis=0)
    norms = np.linalg.norm(B, axis=0)
    norms[norms <= 1e-14] = 1.0
    B = B / norms
    return B


def theta_fn_smooth(theta, n, B, tau, N):
    a = softmax_seq(B @ theta)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    cmax = float(np.max(c))
    if cmax <= 0.0 or not np.isfinite(cmax):
        return 0.0
    val = smooth_score(c, n, tau)
    return float(2.0 * n * val)


def theta_grad_smooth(theta, n, B, tau, N):
    a = softmax_seq(B @ theta)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    cmax = float(np.max(c))
    D = B.shape[1]
    if cmax <= 0.0 or not np.isfinite(cmax):
        return np.zeros(D, dtype=np.float64)
    tau_eff = max(tau, 1e-12)
    z = (c - cmax) / tau_eff
    z = np.clip(z, -50.0, 0.0)
    ez = np.exp(z)
    s = float(ez.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.zeros(D, dtype=np.float64)
    w = ez / s
    w_rev = w[::-1]
    fw = np.fft.rfft(w_rev, N)
    conv = np.fft.irfft(fa * fw, N)
    ga = 2.0 * conv[n - 1:2 * n - 1][::-1]
    total = float(np.dot(a, ga))
    aB = a @ B
    agaB = (a * ga) @ B
    gtheta = agaB - total * aB
    return (2.0 * n) * gtheta


def optimize_lowdim(n, deadline, seed, initial_a=None, n_modes=24):

    B = build_basis(n, n_modes)
    D = B.shape[1]
    N = next_pow2(3 * n - 2)
    rng = np.random.default_rng(seed)

    starts = []
    if initial_a is not None:
        a0 = normalize_positive(np.asarray(initial_a, dtype=np.float64).ravel(), floor=1e-12)
        if a0.size == n:
            z = np.log(np.maximum(a0, 1e-12))
            z -= z.mean()
            theta_proj, _, _, _ = np.linalg.lstsq(B, z, rcond=None)
            starts.append(theta_proj.copy())

    for std in (0.5, 1.0, 2.0):
        starts.append(rng.normal(0.0, std, size=D))
    starts.append(np.zeros(D, dtype=np.float64))

    best_a = make_uniform(n)
    best_score = internal_score(best_a, N)

    for theta0 in starts:
        if time.monotonic() > deadline:
            break
        theta = theta0.copy()
        a = softmax_seq(B @ theta)
        sc = internal_score(a, N)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if _scipy_optimize is None:
            break

        for fac in (0.25 / n, 0.10 / n, 0.04 / n, 0.015 / n):
            if time.monotonic() > deadline:
                break
            try:
                res = _scipy_optimize.minimize(
                    theta_fn_smooth, theta,
                    method='L-BFGS-B', jac=theta_grad_smooth,
                    args=(n, B, fac, N),
                    options={'maxiter': 30, 'ftol': 1e-13, 'gtol': 1e-7, 'maxls': 10},
                )
                if res.x is not None and np.all(np.isfinite(res.x)):
                    theta = res.x
            except Exception:
                pass
            a = softmax_seq(B @ theta)
            sc = internal_score(a, N)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                print("lowdim n={} score={:.9f}".format(n, best_score), flush=True)

    return best_a, best_score

                stagnant = 0
                last_improve = it

            if cmax <= 0.0 or not np.isfinite(cmax):
                break

            fa = np.fft.rfft(a, N)
            K = min(top_k + phase * 10, c.size)
            g = grad_topk(c, a, fa, N, n, K=K)
            g -= g.mean()
            gscale = float(np.abs(g).max())
            if not np.isfinite(gscale) or gscale <= 1e-15:
                break
            g = g / gscale

            # Mirror descent step with backtracking
            loga = np.log(a + 1e-20)
            loga -= float(np.max(loga))
            accepted = False

            for e in (eta, 0.5 * eta, 0.25 * eta, 0.125 * eta, 0.0625 * eta):
                if e < 1e-10:
                    continue
                loga_try = loga - e * g
                loga_try -= float(np.max(loga_try))
                a_try = np.exp(np.clip(loga_try, -60.0, 0.0))
                a_try = np.maximum(a_try, 1e-300)
                a_try = a_try / float(a_try.sum())
                a_try = normalize_positive(a_try, floor=1e-15)

                c_try = convolve_fft(a_try, N)
                score_try = score_from_c(c_try, n)

                if np.isfinite(score_try) and score_try < score * (1 - 1e-10):
                    a = a_try
                    score = score_try
                    accepted = True
                    break

            if accepted:
                eta = min(1.5, eta * 1.02)
            else:
                stagnant += 1
                eta = max(1e-4, eta * 0.6)
                if stagnant > 300:
                    # Re-heat: random perturbation
                    pert = np.exp(rng.normal(0, 0.1, n))
                    a_try = a * pert
                    a_try = normalize_positive(a_try, floor=1e-15)
                    sc_try = internal_score(a_try, N)
                    if np.isfinite(sc_try) and sc_try < best_score * 1.1:
                        a = a_try
                        score = sc_try
                    eta = 0.8
                    stagnant = 0

            it += 1

        # Phase transition: slight randomization
        if phase < n_anneal - 1 and time.monotonic() < deadline:
            pert = np.exp(rng.normal(0, 0.03, n))
            a_try = a * pert
            a_try = normalize_positive(a_try, floor=1e-15)
            sc_try = internal_score(a_try, N)
            if np.isfinite(sc_try) and sc_try < best_score:
                a = a_try
                best_a = a.copy()
                best_score = sc_try

    return best_a, best_score


def optimize_binary_v2(a0, deadline, seed=0):
    """Binary sequence SA with fast incremental convolution updates."""
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    m = max(4, n // 2)

    # Initialize binary
    if np.all((a0 <= 0.5) | (a0 >= 0.5)):
        b = (a0 > 0.5).astype(np.float64)
    else:
        order = np.argsort(a0, kind='stable')
        b = np.zeros(n, dtype=np.float64)
        b[order[-m:]] = 1.0

    # Compute initial convolution
    c = np.rint(convolve_fft(b, next_pow2(2 * n - 1))).astype(np.float64)
    cmax = float(np.max(c))
    best_cmax = cmax
    best_b = b.copy()
    best_score = float(2.0 * n * cmax / (m * m))

    rng = np.random.default_rng(seed)
    temp = 0.06
    stale = 0
    two_b = 2.0 * b

    while time.monotonic() < deadline:
        cmax = float(np.max(c))
        score = 2.0 * n * cmax / (m * m)

        if score < best_score * (1 - 1e-12):
            best_score = score
            best_cmax = cmax
            best_b = b.copy()

        # Find top lags for gradient
        K = min(50, c.size)
        top_idx = np.argpartition(c, -K)[-K:]
        top_c = c[top_idx]
        tau = max(0.05 * cmax, 0.5)
        z = (top_c - cmax) / tau
        z = np.clip(z, -40.0, 0.0)
        w_top = np.exp(z)
        w_top /= w_top.sum()

        # Build pressure for 1s and 0s
        w = np.zeros(c.size, dtype=np.float64)
        w[top_idx] = w_top
        fa = np.fft.rfft(b, next_pow2(3 * n - 2))
        w_rev = w[::-1]
        N = next_pow2(3 * n - 2)
        fw = np.fft.rfft(w_rev, N)
        conv = np.fft.irfft(fa * fw, N)
        g = 2.0 * conv[n - 1:2 * n - 1][::-1]

        one_idx = np.nonzero(b > 0.5)[0]
        zero_idx = np.nonzero(b < 0.5)[0]
        if one_idx.size == 0 or zero_idx.size == 0:
            break

        # Pressure-based selection
        g_one = g[one_idx]
        g_zero = g[zero_idx]
        max_g = max(abs(g_one).max(), abs(g_zero).max(), 1e-10)
        temp_eff = max(temp, 0.01) * max_g

        p_remove = np.exp(-(g_one - g_one.min()) / temp_eff)
        p_remove /= p_remove.sum()
        p_add = np.exp(-(g_zero.min() - g_zero) / temp_eff)
        p_add /= p_add.sum()

        si = int(rng.choice(one_idx.size, p=p_remove))
        sj = int(rng.choice(zero_idx.size, p=p_add))
        i = int(one_idx[si])
        j = int(zero_idx[sj])
        if i == j:
            continue

        # Swap: remove 1 at i, add 1 at j
        old_two_b = two_b.copy()
        c[i:i + n] -= old_two_b
        c[2 * i] -= 1.0
        c[j:j + n] += old_two_b
        c[2 * j] += 1.0
        b[i] = 0.0
        b[j] = 1.0
        two_b = 2.0 * b

        new_cmax = float(np.max(c))
        new_score = 2.0 * n * new_cmax / (m * m)

        accept = False
        if new_score < score * (1 - 1e-12):
            accept = True
        else:
            delta = (new_score - score) * m * m / (2.0 * n)
            if delta < 0.01:
                accept = rng.random() < 0.4
            elif delta < 2.0:
                accept = rng.random() < np.exp(-delta / max(temp, 1e-6))

        if accept:
            if new_score < best_score:
                best_score = new_score
                best_cmax = new_cmax
                best_b = b.copy()
            stale = 0
            temp = max(0.005, temp * 0.9985)
        else:
            c[j:j + n] -= two_b
            c[2 * j] -= 1.0
            c[i:i + n] += two_b
            c[2 * i] += 1.0
            b[j] = 0.0
            b[i] = 1.0
            two_b = 2.0 * b
            stale += 1
            temp = max(0.003, temp * 0.97)
            if stale % 500 == 0:
                temp = min(0.12, temp * 2.0)
                stale = 0

    return best_b, best_score


def local_polish_v2(a, deadline, seed=1):
    """Local polish with adaptive step sizes and coordinate descent."""
    a = normalize_positive(a, floor=1e-15)
    n = a.size
    N = next_pow2(3 * n - 2)

    best_a = a.copy()
    best_score = internal_score(a, N)

    loga = np.log(a + 1e-20)
    loga -= float(np.max(loga))
    rng = np.random.default_rng(seed)
    sigma = 0.15
    stale = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        # Random block perturbation
        width = int(rng.integers(max(2, n // 5000), max(3, n // 25)))
        width = min(width, max(1, n // 2))
        start = int(rng.integers(0, n - width + 1))

        t = np.linspace(0.0, 1.0, width)
        env = np.exp(-3.0 * ((t - 0.5) ** 2))
        d = sigma * (0.5 + 0.5 * env) * rng.normal(size=width)
        d -= d.mean()

        old = loga[start:start + width].copy()
        loga[start:start + width] += d
        a_try = np.exp(np.clip(loga - float(np.max(loga)), -60.0, 0.0))
        a_try = np.maximum(a_try, 1e-300)
        a_try /= a_try.sum()
        a_try = normalize_positive(a_try, floor=1e-15)

        score_try = internal_score(a_try, N)
        if np.isfinite(score_try) and score_try < best_score * (1 - 1e-10):
            best_a = a_try.copy()
            best_score = score_try
            loga = np.log(a_try + 1e-20)
            loga -= float(np.max(loga))
            stale = 0
            sigma = min(0.35, sigma * 1.03)
        else:
            loga[start:start + width] = old
            stale += 1
            sigma = max(0.02, sigma * 0.97)
            if stale > 200:
                # Single coordinate perturbation
                idx = int(rng.integers(0, n))
                old_v = loga[idx]
                loga[idx] += 2.0 * sigma * rng.normal()
                a_try = np.exp(np.clip(loga - float(np.max(loga)), -60.0, 0.0))
                a_try = np.maximum(a_try, 1e-300)
                a_try /= a_try.sum()
                a_try = normalize_positive(a_try, floor=1e-15)
                score_try = internal_score(a_try, N)
                if np.isfinite(score_try) and score_try < best_score:
                    best_a = a_try.copy()
                    best_score = score_try
                    loga = np.log(a_try + 1e-20)
                    loga -= float(np.max(loga))
                else:
                    loga[idx] = old_v
                stale = 0

        if best_score < 1.505:
            break

    return best_a, best_score


def resample_seq(a, n):
    a = normalize_positive(a, floor=0.0)
    old_n = a.size
    n = max(2, int(n))
    if old_n == 1:
        return np.full(n, 1.0 / float(n))
    if old_n == n:
        return a.astype(np.float64, copy=True)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, n)
    b = np.interp(new_x, old_x, a)
    return normalize_positive(b, floor=1e-12)


def get_given_sequence():
    try:
        seq = globals().get("height_sequence_1", None)
        if seq is None:
            return None
        a = np.asarray(seq, dtype=np.float64).ravel()
        if a.size == 0:
            return None
        a = normalize_positive(a, floor=0.0)
        if not np.all(np.isfinite(a)):
            return None
        return a
    except Exception:
        return None


def propose_candidate(seed=42, budget_s=1800):
    start = time.monotonic()
    total_budget = max(120.0, float(budget_s) - 30.0)
    deadline = start + total_budget

    best_a = make_uniform(2048)
    best_score = internal_score(best_a)
    print("initial fallback score = {:.9f}".format(best_score), flush=True)

    # Try given sequence
    given = get_given_sequence()
    if given is not None:
        sc = internal_score(given)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = given.copy()
            print("given sequence score = {:.9f}".format(best_score), flush=True)

    # Phase 1: Quick structured initializations (30s)
    phase1_deadline = start + 30.0
    for n_init in (1024, 2048, 4096):
        if time.monotonic() > phase1_deadline:
            break
        candidates = [make_uniform(n_init)]
        candidates.append(make_antisymmetric_perturb(n_init, 0.15, seed))
        candidates.append(make_antisymmetric_perturb(n_init, 0.3, seed + 1))
        candidates.append(make_antisymmetric_perturb(n_init, 0.5, seed + 2))
        # Random smooth sequences
        rng_init = np.random.default_rng(seed + n_init)
        for _ in range(4):
            x = np.linspace(0, 1, n_init)
            vals = np.ones(n_init)
            for m in range(1, 8):
                amp = rng_init.uniform(0.05, 0.4) / m**0.6
                phase = rng_init.uniform(0, 2 * np.pi)
                vals += amp * np.sin(2 * np.pi * m * x + phase)
                vals += amp * np.cos(2 * np.pi * m * x + phase)
            candidates.append(normalize_positive(vals, floor=1e-12))
        # Periodic patterns
        for bl in (4, 6, 8, 12, 16):
            if n_init % bl == 0:
                pat = np.random.default_rng(seed + bl).uniform(0.5, 1.5, bl)
                candidates.append(make_periodic_block(n_init, bl, pat))

        for c0 in candidates:
            sc = internal_score(c0)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = c0.copy()

    print("after init: n={} score={:.9f}".format(len(best_a), best_score), flush=True)

    # Phase 2: Low-dimensional optimization (150s)
    phase2_deadline = min(start + 180.0, deadline - 600.0)
    for n_ld in (1024, 2048, 4096):
        if time.monotonic() > phase2_deadline:
            break
        allow = min(50.0, phase2_deadline - time.monotonic())
        if allow < 10.0:
            break
        a_init = resample_seq(best_a, n_ld)
        a1, sc1 = optimize_lowdim(n_ld, time.monotonic() + allow, seed=seed + 100 + n_ld,
                                  initial_a=a_init, n_modes=24)
        if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()
            print("lowdim n={} score={:.9f}".format(n_ld, best_score), flush=True)

    # Phase 3: Main mirror descent optimization (large budget)
    # Try multiple n values with the best candidate resampled
    phase3_deadline = min(start + 1400.0, deadline - 400.0)
    n_mirror_candidates = [1024, 2048, 4096]

    # Sort by promising-ness: prefer current best length
    n_current = len(best_a)
    n_mirror_candidates.sort(key=lambda x: abs(x - n_current))

    for n_md in n_mirror_candidates:
        if time.monotonic() > phase3_deadline:
            break
        remain = phase3_deadline - time.monotonic()
        if remain < 60.0:
            break
        allow = min(350.0, remain)
        a_init = resample_seq(best_a, n_md)
        a1, sc1 = optimize_mirror_v2(a_init, time.monotonic() + allow,
                                     seed=seed + 2000 + n_md, top_k=30, n_anneal=3)
        if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()
            print("mirror n={} score={:.9f}".format(n_md, best_score), flush=True)

    # Second round of mirror with different params
    phase3b_deadline = min(start + 1650.0, deadline - 250.0)
    for n_md in [2048, 4096]:
        if time.monotonic() > phase3b_deadline:
            break
        remain = phase3b_deadline - time.monotonic()
        if remain < 60.0:
            break
        allow = min(200.0, remain)
        a_init = resample_seq(best_a, n_md)
        a1, sc1 = optimize_mirror_v2(a_init, time.monotonic() + allow,
                                     seed=seed + 5000 + n_md, top_k=50, n_anneal=4)
        if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()
            print("mirror2 n={} score={:.9f}".format(n_md, best_score), flush=True)

    # Phase 4: Binary SA (if time allows)
    phase4_deadline = min(start + 2050.0, deadline - 100.0)
    if time.monotonic() < phase4_deadline - 80.0:
        n_bin = len(best_a)
        if n_bin not in (2048, 4096):
            n_bin = 2048
        a_bin_init = resample_seq(best_a, n_bin)
        b1, sc1 = optimize_binary_v2(a_bin_init,
                                     min(phase4_deadline, time.monotonic() + 300.0),
                                     seed=seed + 7000)
        if b1 is not None and np.isfinite(sc1) and sc1 < best_score:
            best_score = sc1
            best_a = b1.copy()
            print("binary n={} score={:.9f}".format(n_bin, best_score), flush=True)

        # Second binary at different n
        if time.monotonic() < phase4_deadline - 80.0:
            n_bin2 = 4096 if n_bin == 2048 else 2048
            a_bin_init2 = resample_seq(best_a, n_bin2)
            b2, sc2 = optimize_binary_v2(a_bin_init2,
                                         min(phase4_deadline, time.monotonic() + 200.0),
                                         seed=seed + 8000)
            if b2 is not None and np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_a = b2.copy()
                print("binary2 n={} score={:.9f}".format(n_bin2, best_score), flush=True)

    # Phase 5: Final polish
    if time.monotonic() < deadline - 15.0:
        a1, sc1 = local_polish_v2(best_a, deadline - 5.0, seed=99999)
        if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()
            print("polish score={:.9f}".format(best_score), flush=True)

    # Final verification
    best_a = normalize_positive(best_a, floor=0.0)
    out_list = [float(x) for x in best_a]
    try:
        final_eval = float(evaluate_sequence(out_list))
    except Exception:
        final_eval = float("inf")

    if not np.isfinite(final_eval):
        best_a = make_uniform(2048)
        out_list = [float(x) for x in best_a]
        final_eval = 2.0

    print("final internal score = {:.12f}".format(best_score), flush=True)
    print("final evaluated score = {:.12f}, length = {}".format(final_eval, len(out_list)), flush=True)
    return out_list
