import time

def _clean(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    np.clip(a, 0.0, 1000.0, out=a)
    if a.size == 0:

    return a

def _normalize(a):
    a = _clean(a)
    s = a.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones(a.size) / a.size
    return a / s

def _score(a):
    n = a.size
    b = np.convolve(a, a)
    mx = b.max()
    s = a.sum()
    return 2.0 * n * mx / (s * s)

def _score_from_b(a, b):
    n = a.size
    mx = max(b.max(), 0.0)
    s = a.sum()
    if s <= 0:
        return float('inf')
    return 2.0 * n * mx / (s * s)

  The term j=k-i gives a[k-i]*a[i] = a[k-i] * old

           = b[k] + 2*(x-old)*a[k-i]

  else: b_new[k] = b[k] + 2*(x-old)*a[k-i]

And b_new[k] = b[k] for k outside [i, i+n-1].

Now, the "unaffected" entries (k outside [i, i+n-1]) remain the same. Their max is `max_unaff`.

The score is:
2n * max(max_unaff, max_{k in [i,i+n-1]} b_new[k]) / (S_new)²

where S_new = S_old - old + x.
