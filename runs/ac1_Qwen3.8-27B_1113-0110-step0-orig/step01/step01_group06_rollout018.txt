import numpy as np

def _pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _fft_square(a):
    n = int(a.size)
    L = 2 * n - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, n=N)
    b = np.fft.irfft(fa * fa, n=N).real[:L]
    b = np.maximum(b, 0.0)
    return b


def _fft_grad(weights, a):
    n = int(a.size)
    L = int(weights.size)
    N = _pow2(L + n - 1)
    fw = np.fft.rfft(weights, n=N)
    fa = np.fft.rfft(a, n=N)
    c = np.fft.irfft(fw * np.conj(fa), n=N)[:n]
    return 2.0 * c


def _internal_score(a, b=None):
    if b is None:
        b = _fft_square(a)
    s = float(a.sum())
    if s <= 1e-300:
        return float("inf"), b, 0.0
    maxb = float(b.max())
    score = float(2.0 * a.size * maxb / (s * s))
    return score, b, maxb


def _make_weights_adaptive(b, focus, target_k=15.0, eps=0.01):
    """Create soft weights with adaptive focus to target ~target_k active constraints."""
    L = int(b.size)
    maxb = float(b.max())
    if maxb <= 1e-300:
        return np.full(L, 1.0 / L), maxb, focus
    with np.errstate(over="ignore", under="ignore"):
        z = (maxb - b) / maxb
        p = np.exp(-z * focus)
    eff_k = float(p.sum())
    s = eff_k
    if s <= 1e-12:
        p = np.zeros(L)
        p[int(np.argmax(b))] = 1.0
    else:
        p = (1.0 - eps) * (p / s) + eps / L
    return p, maxb, focus


def _adjust_focus(focus, eff_k, target_k=15.0):
    """Adjust focus to keep effective number of active constraints near target."""
    if eff_k > target_k * 2.0:

def _greedy_support(n, rng, shuffle=True):
    n = int(n)
    m = n // 2
    counts = np.zeros(2 * n - 1, dtype=np.int64)
    selected = np.empty(m, dtype=np.int64)
    order = np.arange(m, dtype=np.int64)
    if shuffle:
        rng.shuffle(order)
    t = 0
    for idx_val in order:
        idx = int(idx_val)
        l = idx
        r = n - 1 - idx
        if t == 0:
            y = l
        else:
            s = selected[:t]
            affected_l = int(np.max(counts[s + l])) if t > 0 else 0
            self_l = int(counts[2 * l])
            metric_l = max(affected_l + 2, self_l + 1)
            affected_r = int(np.max(counts[s + r])) if t > 0 else 0
            self_r = int(counts[2 * r])
            metric_r = max(affected_r + 2, self_r + 1)
            y = l if metric_l <= metric_r else r
        if t > 0:
            counts[selected[:t] + y] += 2
        counts[2 * y] += 1
        selected[t] = y
        t += 1
    return selected


def _greedy_x(n, rng, shuffle=True):

    m = n // 2
    x = np.full(m, 1e-6, dtype=np.float64)
    left = sel < m
    if np.any(left):
        x[sel[left]] = 1.0 - 1e-6
    return x


def _sine_x(n, rng, freq=None, amp=0.4):

    v += 0.1 * rng.normal(0, 1, m)
    x_new = 0.5 + v
    return np.clip(x_new, 1e-6, 1.0 - 1e-6)


def _anti_log_init(n, rng, amp=3.0, smooth=False):
    m = n // 2
    if m < 1:
        return np.ones(n) / max(1, n)
    x = np.arange(m, dtype=np.float64)
    if smooth:
        v = np.zeros(m)
        for f in (1, 2, 3, 5, 7, 11, 13):
            v += (rng.normal(0, 1) / f) * np.sin(
                2 * np.pi * f * x / m + rng.uniform(0, 2 * np.pi)
            )
        v += 0.1 * rng.normal(0, 1, m)
    else:
        v = rng.normal(0, 1, m)
    v -= float(v.mean())
    std = float(v.std())
    if std > 1e-12:
        v = (amp * v) / std
    z = np.zeros(n)
    z[:m] = v
    if n % 2 == 0:

    else:
        z[m + 1:] = -v[::-1]
        z[m] = 0.0
    z -= z.max()
    a = np.exp(z)
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    return a

            print(f"task deadline reached at task {idx}")
            break
        alloc = min(max_alloc, task_deadline - now)
        if alloc < 30.0:
            break

        ng_seed = (seed * 1000003 + idx * 7919 + variant * 131) % (2**32)
        task_rng = np.random.default_rng(ng_seed)

        print(f"\n=== Task {idx}: n={n} kind={kind} variant={variant} alloc={alloc:.0f}s ===")
        print(f"    current best: {best_score:.8f}")
        t0 = time.monotonic()
        cand_a = None
        cand_score = float("inf")

        try:
            if kind == "given_refine":
                a0 = given_a.copy() if given_a is not None else _random_positive(n, task_rng)
                # Add small perturbation for diversity
                if variant > 0:
                    a0 = a0 * np.exp(task_rng.normal(0, 0.1 * variant, size=n))
                    a0 = np.maximum(a0, 1e-15)
                    a0 /= a0.sum()
                cand_a, cand_score = _optimize_from_a_v2(
                    a0, alloc, task_rng, target_k=15.0,
                    focus_start=float(3.0 + 2.0 * variant)
                )

            elif kind == "pair_v2":
                if variant == 0:
                    x0 = _greedy_x(n, task_rng, shuffle=True)
                elif variant == 1:
                    x0 = _sine_x(n, task_rng, freq=2, amp=0.3)
                else:
                    x0 = 0.5 + 0.2 * task_rng.normal(0, 1, size=n // 2)
                    x0 = np.clip(x0, 1e-6, 1 - 1e-6)
                cand_a, cand_score = _optimize_pair_v2(
                    x0, alloc, task_rng, target_k=15.0
                )

            elif kind == "pair_long":
                x0 = _greedy_x(n, task_rng, shuffle=(variant == 0))
                cand_a, cand_score = _optimize_pair_long(
                    x0, alloc, task_rng, target_k=10.0
                )

            elif kind == "general":
                if variant == 0 and given_a is not None and given_a.size == n:
                    a0 = given_a.copy()
                else:
                    a0 = _anti_log_init(n, task_rng, amp=3.0, smooth=(variant % 2 == 0))
                cand_a, cand_score = _optimize_from_a_v2(
                    a0, alloc, task_rng, target_k=20.0,
                    focus_start=4.0
                )

            elif kind == "sine":
                # Try sine-wave anti-palindromic initialization
                m = n // 2
                x = np.arange(m, dtype=np.float64)
                best_x_sine = None
                for freq in [1, 2, 3, 5]:
                    for amp in [0.2, 0.4, 0.6]:
                        for phase in [0, np.pi / 2, np.pi]:
                            v = amp * np.sin(2 * np.pi * freq * x / m + phase)
                            xv = np.clip(0.5 + v, 1e-6, 1 - 1e-6)
                            ax = _a_from_x(xv, n)
                            sc, _, _ = _internal_score(ax)
                            if best_x_sine is None or sc < cand_score:
                                cand_score = sc
                                best_x_sine = xv.copy()
                if best_x_sine is not None:
                    print(f"    best sine init score: {cand_score:.8f}")
                    cand_a, cand_score = _optimize_pair_v2(
                        best_x_sine, alloc, task_rng, target_k=12.0
                    )

            else:
                a0 = _random_positive(n, task_rng)
                cand_a, cand_score = _optimize_from_a_v2(
                    a0, alloc, task_rng, target_k=15.0
                )

            if cand_a is not None:
                cand_a = np.nan_to_num(cand_a, nan=0.0, posinf=0.0, neginf=0.0)
                cand_a = np.clip(cand_a, 0.0, 1000.0)
                cc = float(cand_a.sum())
                if cc > 0.01:
                    cand_a /= cc
                    internal, _, _ = _internal_score(cand_a)
                    if np.isfinite(internal) and internal < cand_score:
                        cand_score = internal
                    if np.isfinite(cand_score) and cand_score < best_score:
                        best_score = cand_score
                        best_a = cand_a.copy()
                        print(f"    NEW BEST: {best_score:.8f} (n={cand_a.size})")

        except Exception as exc:
            print(f"    task failed: {repr(exc)}")

        elapsed = time.monotonic() - t0
        print(f"    done: candidate={cand_score:.8f} best={best_score:.8f} used={elapsed:.1f}s")

        if best_score <= 1.51:
            print("Strong score found!")
            break

    # Final refinement phase
    if best_a is not None:
        remaining = final_deadline - time.monotonic()
        if remaining > 100.0:
            refine_time = min(200.0, remaining - 60.0)
            if refine_time > 30.0:
                task_rng = np.random.default_rng((seed + 777777) % (2**32))
                print(f"\n=== Final refinement: time={refine_time:.0f}s ===")

                # Try both pair and general refinement
                refine_n = best_a.size
                m = refine_n // 2
                if refine_n % 2 == 0 and m >= 2:
                    # Convert to pair parameterization if close to anti-palindromic
                    a_check = best_a
                    anti_diff = np.abs(a_check[:m] + a_check[m:][::-1] - 2.0 / refine_n * (refine_n / m))
                    if float(anti_diff.max()) < 0.05 * float(a_check.max()):
                        # Extract x from a
                        x_init = a_check[:m] * m
                        x_init = np.clip(x_init, 1e-7, 1 - 1e-7)
                        print("  Using pair refinement")
                        refined_a, refined_score = _optimize_pair_long(
                            x_init, refine_time * 0.6, task_rng, target_k=8.0
                        )
                        if refined_a is not None and np.isfinite(refined_score):
                            if refined_score < best_score:
                                best_a = refined_a
                                best_score = refined_score
                                print(f"  Improved to {best_score:.8f}")
                        # Remaining time for general
                        rem = refine_time * 0.4
                        if rem > 20:
                            refined_a2, refined_score2 = _optimize_from_a_v2(
                                best_a, rem, task_rng, target_k=10.0, focus_start=20.0
                            )
                            if refined_a2 is not None and np.isfinite(refined_score2):
                                if refined_score2 < best_score:
                                    best_a = refined_a2
                                    best_score = refined_score2
                                    print(f"  Improved to {best_score:.8f}")
                    else:
                        print("  Using general refinement")
                        refined_a, refined_score = _optimize_from_a_v2(
                            best_a, refine_time, task_rng, target_k=10.0,
                            focus_start=10.0
                        )
                        if refined_a is not None and np.isfinite(refined_score):
                            if refined_score < best_score:
                                best_a = refined_a
                                best_score = refined_score
                                print(f"  Improved to {best_score:.8f}")
                else:
                    refined_a, refined_score = _optimize_from_a_v2(
                        best_a, refine_time, task_rng, target_k=10.0,
                        focus_start=10.0
                    )
                    if refined_a is not None and np.isfinite(refined_score):
                        if refined_score < best_score:
                            best_a = refined_a
                            best_score = refined_score
                            print(f"  Improved to {best_score:.8f}")

    # Fallback
    if best_a is None:
        best_a = np.ones(1024, dtype=np.float64) / 1024.0

    # Final cleanup
    best_a = np.nan_to_num(best_a, nan=0.0, posinf=1000.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    ss = float(best_a.sum())
    if ss <= 0.01:
        best_a = np.ones(1024, dtype=np.float64) / 1024.0
        ss = 1.0
    else:
        best_a /= ss

    final_seq = [float(x) for x in best_a]

    # Validate with exact evaluator if possible
    if best_a.size <= 20000 and time.monotonic() < final_deadline - 5.0:
        exact = _safe_evaluate(final_seq)
        if np.isfinite(exact):
            print(f"\nFINAL: internal={best_score:.8f} exact={exact:.8f} n={best_a.size}")
            if exact < best_score:
                best_score = exact
        else:
            print(f"\nFINAL: internal={best_score:.8f} exact=N/A n={best_a.size}")
    else:
        print(f"\nFINAL: internal={best_score:.8f} n={best_a.size}")

    return final_seq
