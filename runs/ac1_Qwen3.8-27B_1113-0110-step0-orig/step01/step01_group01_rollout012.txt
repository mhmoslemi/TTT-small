"""
Search for non-negative step-function heights minimizing

    2*n*max(convolve(a,a)) / (sum(a))^2

using:
  - analytic singular-seed initialization,
  - low-dimensional smooth/log-multiplicative L-BFGS refinement,
  - high-dimensional log-sum-exp mirror search,
  - binary support simulated annealing,
  - final length-capped validated output.
"""

import time
import numpy as np

try:
    import scipy.optimize as _scipy_optimize
except Exception:
    _scipy_optimize = None


def next_pow2(m: int) -> int:
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def normalize_positive(a, floor: float = 0.0) -> np.ndarray:
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    if floor > 0.0:
        a = np.maximum(a, float(floor))
    else:
        a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(a.size, 1.0 / float(a.size), dtype=np.float64)
    return a / s


def make_uniform(n: int) -> np.ndarray:
    n = max(1, int(n))
    return np.ones(n, dtype=np.float64) / float(n)


def make_power_sequence(
    n: int,
    gamma: float = 0.5,
    left_u: float = 1.0,
    right_gamma: float = 0.0,
    right_weight: float = 0.0,
    right_u: float = 1.0,
    tilt: float = 0.0,
) -> np.ndarray:
    n = max(8, int(n))
    i = np.arange(n, dtype=np.float64)
    x = i / float(n - 1)

    if gamma == 0.0:
        left = np.ones(n, dtype=np.float64)
    else:
        left = np.power(i + float(left_u), -float(gamma))

    if tilt != 0.0:
        factor = 1.0 + float(tilt) * x
        left = left * np.maximum(factor, 1e-6)

    if right_weight > 1e-12 and right_gamma > 0.0:
        right = float(right_weight) * np.power(n - i + float(right_u), -float(right_gamma))
        a = left + right
    else:
        a = left

    return normalize_positive(a, floor=1e-14)


def resample_seq(a, n: int) -> np.ndarray:
    a = normalize_positive(a, floor=0.0)
    old_n = a.size
    n = max(2, int(n))
    if old_n == 1:
        return make_uniform(n)
    if old_n == n:
        return a.astype(np.float64, copy=True)

    old_x = np.linspace(0.0, 1.0, old_n, dtype=np.float64)
    new_x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    b = np.interp(new_x, old_x, a)
    return normalize_positive(b, floor=1e-14)


def get_given_sequence():
    try:
        seq = globals().get("height_sequence_1", None)
        if seq is None:
            return None
        a = np.asarray(seq, dtype=np.float64).ravel()
        if a.size == 0:
            return None
        a = normalize_positive(a, floor=0.0)
        if not np.all(np.isfinite(a)):
            return None
        return a
    except Exception:
        return None


def fft_size(n: int) -> int:
    n = max(1, int(n))
    return next_pow2(3 * n - 2)


def convolve_fft(a, N: int = None) -> np.ndarray:
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if N is None:
        N = fft_size(n)
    else:
        N = max(int(N), next_pow2(2 * n - 1))

    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[: 2 * n - 1]
    c = np.real(c).astype(np.float64, copy=False)
    c[np.where((c < 0.0) & (c > -1e-12))[0]] = 0.0
    return c


def grad_weights(a, w, N: int = None) -> np.ndarray:
    a = np.asarray(a, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = a.size
    L = 2 * n - 1
    if w.size != L:
        if w.size > L:
            w = w[:L]
        else:
            pad = np.zeros(L - w.size, dtype=np.float64)
            w = np.concatenate((w, pad))

    if N is None:
        N = fft_size(n)
    else:
        N = max(int(N), next_pow2(3 * n - 2))

    wr = w[::-1]
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(wr, N)
    conv = np.fft.irfft(fa * fw, N)
    g = 2.0 * conv[n - 1 : 2 * n - 1][::-1]
    return np.real(g).astype(np.float64, copy=False)


def softmax_weights(b, cmax: float, tau: float, floor: float = 0.0) -> np.ndarray:
    b = np.asarray(b, dtype=np.float64).ravel()
    L = b.size
    if L == 0:
        return b
    if tau <= 0.0 or cmax <= 0.0 or not np.isfinite(cmax):
        return np.full(L, 1.0 / float(L), dtype=np.float64)

    z = (b - float(cmax)) / float(tau)
    z = np.clip(z, -50.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(L, 1.0 / float(L), dtype=np.float64)
    w = e / s
    if floor > 0.0:
        w = (1.0 - float(floor)) * w + (float(floor) / float(L))
    return w


def top_weights(b, frac: float = 0.02, min_k: int = 16) -> np.ndarray:
    b = np.asarray(b, dtype=np.float64).ravel()
    L = b.size
    if L == 0:
        return b
    k = int(min(max(min_k, int(round(frac * L))), L))
    if k >= L:
        return np.full(L, 1.0 / float(L), dtype=np.float64)
    idx = np.argpartition(b, -k)[-k:]
    w = np.zeros(L, dtype=np.float64)
    w[idx] = 1.0 / float(k)
    return w


def smooth_volume(b, cmax: float, tau: float) -> float:
    cmax = float(cmax)
    if cmax <= 0.0 or not np.isfinite(cmax):
        return 0.0
    if tau <= 0.0:
        return cmax
    z = (np.asarray(b, dtype=np.float64).ravel() - cmax) / float(tau)
    z = np.clip(z, -50.0, 0.0)
    s = float(np.exp(z).sum())
    if not np.isfinite(s) or s <= 0.0:
        return cmax
    return float(cmax + float(tau) * np.log(s))


def score_smooth_b(a, N: int = None, tau: float = 0.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if N is None:
        N = fft_size(a.size)
    n = a.size
    b = convolve_fft(a, N)
    if b.size == 0:
        return float("inf"), float("inf"), b, 0.0
    cmax = float(np.max(b))
    score = float(2.0 * n * cmax)
    smooth = float(2.0 * n * smooth_volume(b, cmax, tau))
    return score, smooth, b, cmax


def internal_score(a, N: int = None) -> float:
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return float("inf")
    a = normalize_positive(a, floor=0.0)
    if N is None:
        N = fft_size(a.size)
    sc, _, _, _ = score_smooth_b(a, N, 0.0)
    if not np.isfinite(sc):
        return float("inf")
    return float(sc)


def generate_quick_seeds(n: int, reference=None, seed: int = 0, max_count: int = 20):
    n = max(32, int(n))
    seqs = []

    if reference is not None:
        try:
            seqs.append(resample_seq(reference, n))
        except Exception:
            pass

    for gamma in (0.45, 0.48, 0.5, 0.52, 0.55):
        for u in (0.75, 1.0, 1.5):
            seqs.append(make_power_sequence(n, gamma=gamma, left_u=u))

    for tilt in (-0.4, -0.2, 0.0, 0.3, 0.7):
        seqs.append(make_power_sequence(n, gamma=0.5, left_u=1.0, tilt=tilt))

    for rw in (0.04, 0.12, 0.30, 0.75):
        seqs.append(
            make_power_sequence(
                n, gamma=0.5, left_u=1.0,
                right_gamma=0.5, right_weight=rw, right_u=1.0,
            )
        )

    for slope in (-1.6, -0.8, 0.8, 1.6):
        x = np.linspace(0.0, 1.0, n, dtype=np.float64)
        vals = np.exp(float(slope) * (2.0 * x - 1.0))
        seqs.append(normalize_positive(vals, floor=1e-14))

    i = np.arange(n, dtype=np.float64)
    rng = np.random.default_rng(int(seed))
    for _ in range(4):
        z = -0.5 * np.log(i + 1.0)
        for m in range(1, 5):
            t = 2.0 * np.pi * float(m) * i / float(n)
            amp = rng.uniform(0.02, 0.12)
            z += amp * np.sin(t + rng.uniform(0.0, 2.0 * np.pi))
            z += amp * np.cos(t + rng.uniform(0.0, 2.0 * np.pi))
        z -= float(np.max(z))
        seqs.append(normalize_positive(np.exp(np.clip(z, -60.0, 0.0)), floor=1e-14))

    if max_count is not None and len(seqs) > int(max_count):
        seqs = seqs[: int(max_count)]
    return seqs


def build_param_basis(n: int, n_modes: int = 4):
    n = max(16, int(n))
    i0 = np.arange(n, dtype=np.float64)
    x = i0 / float(max(1, n - 1))

    cols = []
    cols.append(-0.5 * np.log(i0 + 1.0))
    cols.append(np.log(i0 + 1.0))
    cols.append(-np.log(n - i0 + 1.0))
    cols.append(np.ones(n, dtype=np.float64))

    cols.append(x - x.mean())
    c = x * x - (x * x).mean(); cols.append(c)
    c = x * x * x - (x * x * x).mean(); cols.append(c)

    for sg in (0.08, 0.22):
        c = np.exp(-x / sg); c -= c.mean(); cols.append(c)
        c = np.exp(-(1.0 - x) / sg); c -= c.mean(); cols.append(c)

    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    for m in range(1, int(n_modes) + 1):
        t = 2.0 * np.pi * float(m) * x
        c = np.sin(t); c -= c.mean(); cols.append(c)
        c = np.cos(t); c -= c.mean(); cols.append(c)

    B_raw = np.array(cols, dtype=np.float64)
    norms = np.linalg.norm(B_raw, axis=1)
    norms[~np.isfinite(norms) | (norms <= 1e-14)] = 1.0
    B = B_raw / norms
    idx_base = 0
    idx_left_log = 1
    idx_right_log = 2
    return B, norms, idx_base, idx_left_log, idx_right_log


def param_a(theta, B) -> np.ndarray:
    theta = np.asarray(theta, dtype=np.float64).ravel()
    if B.size == 0:
        return make_uniform(8)
    loga = theta @ B
    if not np.all(np.isfinite(loga)):
        return make_uniform(B.shape[1])
    loga = loga - float(np.max(loga))
    loga = np.clip(loga, -60.0, 0.0)
    a = np.exp(loga)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return make_uniform(B.shape[1])
    return a / s


def param_smooth_value(theta, B, n: int, tau: float, N: int) -> float:
    try:
        a = param_a(theta, B)
        b = convolve_fft(a, N)
        if b.size == 0:
            return float("inf")
        cmax = float(np.max(b))
        if cmax <= 0.0 or not np.isfinite(cmax):
            return 0.0
        if tau <= 0.0:
            vals = cmax
        else:
            vals = smooth_volume(b, cmax, tau)
        val = float(2.0 * int(n) * vals)
        if not np.isfinite(val):
            return float("inf")
        return val
    except Exception:
        return float("inf")


def param_grad(theta, B, n: int, tau: float, N: int) -> np.ndarray:
    try:
        a = param_a(theta, B)
        b = convolve_fft(a, N)
        cmax = float(np.max(b))
        D = B.shape[0]
        if cmax <= 0.0 or not np.isfinite(cmax):
            return np.zeros(D, dtype=np.float64)

        if tau <= 0.0:
            w = top_weights(b, frac=0.02, min_k=16)
        else:
            w = softmax_weights(b, cmax, tau, floor=0.0)

        ga = grad_weights(a, w, N)
        total = float(np.dot(a, ga))
        aB = a @ B
        agaB = (a * ga) @ B
        g = (agaB - total * aB) * (2.0 * float(n))
        g = np.where(np.isfinite(g), g, 0.0)
        return g
    except Exception:
        return np.zeros(B.shape[0], dtype=np.float64)


def param_simple_fd(theta0, B, n: int, tau: float, N: int, deadline, seed: int = 0):
    theta = np.asarray(theta0, dtype=np.float64).ravel().copy()
    best_theta = theta.copy()
    best_score = internal_score(param_a(theta, B), N)
    lr = 0.6
    rng = np.random.default_rng(int(seed))

    for _ in range(1200):
        if time.monotonic() > deadline:
            break
        g = param_grad(theta, B, n, tau, N)
        gn = float(np.linalg.norm(g))
        if not np.isfinite(gn) or gn <= 1e-14:
            break

        accepted = False
        for step in (lr, 0.5 * lr, 0.25 * lr):
            trial = theta - step * g / gn
            a = param_a(trial, B)
            sc = internal_score(a, N)
            if np.isfinite(sc) and sc < best_score * (1.0 - 1e-10):
                theta = trial
                best_theta = theta.copy()
                best_score = sc
                accepted = True
                break
        if accepted:
            lr = min(2.5, lr * 1.03)
        else:
            lr = max(0.02, lr * 0.65)
            old = theta.copy()
            theta += 0.03 * rng.normal(size=theta.size)
            a = param_a(theta, B)
            sc = internal_score(a, N)
            if np.isfinite(sc) and sc < best_score:
                best_theta = theta.copy()
                best_score = sc
            else:
                theta = old
                if lr < 0.05:
                    break
    return best_theta


def run_param_optimize(
    n: int,
    deadline,
    seed: int = 0,
    reference=None,
    max_starts: int = 5,
    verbose: bool = False,
):
    n = max(32, int(n))
    B, norms, ib, il, ir = build_param_basis(n)
    D = B.shape[0]
    N = fft_size(n)
    rng = np.random.default_rng(int(seed))

    theta_base = np.zeros(D, dtype=np.float64)
    theta_base[ib] = norms[ib]

    starts = [theta_base.copy()]

    th = theta_base.copy(); th[ir] += 0.12 * norms[ir]; starts.append(th)
    th = theta_base.copy(); th[il] += 0.06 * norms[il]; starts.append(th)
    th = theta_base.copy(); th[il] -= 0.05 * norms[il]; starts.append(th)

    for std in (0.04, 0.09, 0.16):
        th = theta_base + std * norms * rng.normal(size=D)
        starts.append(th)

    if reference is not None:
        try:
            a0 = normalize_positive(np.asarray(reference, dtype=np.float64).ravel(), floor=1e-14)
            if a0.size == n:
                logz = np.log(np.maximum(a0, 1e-12))
                logz -= float(np.max(logz))
                theta_proj, _, _, _ = np.linalg.lstsq(B, logz, rcond=None)
                theta_proj = np.asarray(theta_proj, dtype=np.float64).ravel()
                if np.all(np.isfinite(theta_proj)):
                    starts.insert(0, theta_proj)
        except Exception:
            pass

    starts = starts[: max(1, int(max_starts))]

    best_a = None
    best_score = float("inf")

    tau_fracs = (0.28, 0.14, 0.07, 0.035, 0.016, 0.007, 0.003)

    for st in starts:
        theta = np.asarray(st, dtype=np.float64).ravel().copy()

        for tf in tau_fracs:
            if time.monotonic() > deadline:
                break
            tau = max(1e-12, tf / float(n))

            if _scipy_optimize is not None:
                maxiter = 32 if n >= 4096 else 48
                try:
                    res = _scipy_optimize.minimize(
                        param_smooth_value,
                        theta,
                        method="L-BFGS-B",
                        jac=param_grad,
                        args=(B, n, tau, N),
                        options={"maxiter": maxiter, "ftol": 1e-12, "gtol": 1e-5, "maxls": 10},
                    )
                    if res is not None and res.x is not None and np.all(np.isfinite(res.x)):
                        theta = np.asarray(res.x, dtype=np.float64).ravel()
                except Exception:
                    pass
            else:
                theta = param_simple_fd(theta, B, n, tau, N, deadline, seed=seed + int(n))

            a = param_a(theta, B)
            sc = internal_score(a, N)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                if verbose:
                    print("param lowdim n={} score={:.9f}".format(n, sc), flush=True)

        if time.monotonic() > deadline:
            break

    if best_a is None:
        best_a = make_power_sequence(n, gamma=0.5, left_u=1.0)
        best_score = internal_score(best_a, N)

    return best_a, float(best_score)


def add_noise_sequence(a, sigma: float, seed: int = 0) -> np.ndarray:
    a = normalize_positive(a, floor=1e-14)
    rng = np.random.default_rng(int(seed))
    loga = np.log(a + 1e-18)
    loga += float(sigma) * rng.normal(size=a.size)
    loga -= float(np.max(loga))
    b = np.exp(np.clip(loga, -60.0, 0.0))
    return normalize_positive(b, floor=1e-14)


def polish_logsum(a, deadline, seed: int = 0, verbose: bool = False):
    a = normalize_positive(np.asarray(a, dtype=np.float64).ravel(), floor=1e-14)
    n = a.size
    N = fft_size(n)

    best_a = a.copy()
    best_score = internal_score(a, N)

    loga = np.log(a + 1e-18)
    loga -= float(np.max(loga))

    score, smooth, b, cmax = score_smooth_b(a, N, 1e-3)

    eta = 0.35
    eta_base = 0.35
    rng = np.random.default_rng(int(seed))

    tau_fracs = (0.28, 0.15, 0.08, 0.04, 0.02, 0.009, 0.004)
    top_mixes = (0.0, 0.0, 0.05, 0.10, 0.18, 0.28, 0.38)

    for tf, mix in zip(tau_fracs, top_mixes):
        if time.monotonic() > deadline:
            break
        tau = max(1e-12, tf / float(n))
        stagn = 0
        it = 0

        while time.monotonic() < deadline and stagn < 420:
            if np.isfinite(score) and score < best_score * (1.0 - 1e-12):
                best_score = float(score)
                best_a = a.copy()

            if cmax <= 0.0 or not np.isfinite(cmax):
                break

            w_soft = softmax_weights(b, cmax, tau, floor=0.02)
            if mix > 0.0:
                w_top = top_weights(b, frac=0.015, min_k=24)
                w = float(mix) * w_top + (1.0 - float(mix)) * w_soft
            else:
                w = w_soft

            g = grad_weights(a, w, N)
            scale = float(np.max(np.abs(g)))
            if not np.isfinite(scale) or scale <= 1e-15:
                break
            gu = g / scale

            smooth_cur = smooth_volume(b, cmax, tau)
            accepted = False

            noise_sigma = 0.0
            if stagn > 50:
                noise_sigma = min(0.25, 0.004 * (stagn // 10))

            for e in (eta, 0.55 * eta, 0.30 * eta):
                if e <= 1e-5:
                    continue
                log_try = loga - float(e) * gu
                if noise_sigma > 0.0:
                    log_try = log_try + noise_sigma * rng.normal(size=n)
                diff = log_try - loga
                log_try = loga + np.clip(diff, -1.6, 1.6)
                log_try = log_try - float(np.max(log_try))
                a_try = np.exp(np.clip(log_try, -60.0, 0.0))
                a_try = normalize_positive(a_try, floor=0.0)

                sc_try, sm_try, b_try, cmax_try = score_smooth_b(a_try, N, tau)
                if not np.isfinite(sc_try):

                local_improve = sc_try < score * (1.0 - 1e-10)
                smooth_improve = (
                    np.isfinite(sm_try)

                    and sm_try < smooth_cur * 0.999
                    and sc_try < score * 1.002
                    and sc_try < best_score * 1.05
                )

                if local_improve or smooth_improve:
                    a = a_try
                    b = np.asarray(b_try, dtype=np.float64).ravel().copy()
                    cmax = float(cmax_try)
                    score = float(sc_try)
                    loga = np.log(a + 1e-18)
                    loga -= float(np.max(loga))
                    eta = min(1.6, eta * 1.025)
                    accepted = True
                    break

            if accepted:
                stagn = 0
                it += 1
                if verbose and it % 800 == 0:

            else:

                stagn += 1
                if stagn > 120:
                    sig = min(0.75, 0.05 + 0.018 * (stagn // 60))
                    loga = loga + sig * rng.normal(size=n)
                    loga -= float(np.max(loga))
                    a = normalize_positive(np.exp(np.clip(loga, -60.0, 0.0)), floor=0.0)
                    score, smooth, b, cmax = score_smooth_b(a, N, tau)
                    if np.isfinite(score) and score < best_score:
                        best_score = float(score)
                        best_a = a.copy()
                    eta = eta_base
                    stagn = 0

        if time.monotonic() > deadline:
            break

    return best_a, float(best_score)


def binary_from_values(a0, m: int) -> np.ndarray:
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    m = int(max(1, min(n - 1, m)))
    order = np.argsort(a0, kind="stable")
    b = np.zeros(n, dtype=np.int64)
    b[order[-m:]] = 1
    return b


def choose_by_pressure(vals, rng, pick_high: bool, temp: float = 0.12) -> int:
    vals = np.asarray(vals, dtype=np.float64).ravel()
    m = vals.size
    if m == 0:
        return -1
    if m == 1:
        return 0
    if not np.all(np.isfinite(vals)):
        vals = np.nan_to_num(vals, nan=0.0, posinf=0.0, neginf=0.0)
    vmin = float(np.min(vals))
    vmax = float(np.max(vals))
    res = vmax - vmin
    if res <= 1e-13:
        return int(rng.integers(0, m))

    denom = max(float(temp), 1e-12) * res
    if pick_high:
        z = (vals - vmax) / denom
    else:
        z = (vmin - vals) / denom
    z = np.clip(z, -50.0, 0.0)
    p = np.exp(z)
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        return int(rng.integers(0, m))
    p = p / s
    return int(rng.choice(m, p=p))


def anneal_binary(b, deadline, seed: int = 0):
    b = np.asarray(b, dtype=np.int64).ravel().copy()
    n = b.size
    m = int(b.sum())
    if m <= 0 or m >= n:
        return b.astype(np.float64), internal_score(b)

    N = fft_size(n)
    c_float = convolve_fft(b.astype(np.float64), N)
    c = np.rint(c_float).astype(np.int64)

    best_b = b.astype(np.float64, copy=True)
    best_cmax = int(np.max(c))
    best_score = float(2.0 * n * best_cmax / (float(m) * float(m)))

    rng = np.random.default_rng(int(seed))
    beta = 0.25
    temp = 0.035
    stale = 0
    it = 0

    while time.monotonic() < deadline:
        cmax = int(np.max(c))
        if cmax <= 0:
            break
        score = float(2.0 * n * cmax / (float(m) * float(m)))
        if np.isfinite(score) and score < best_score * (1.0 - 1e-12):
            best_score = float(score)
            best_cmax = cmax
            best_b = b.astype(np.float64, copy=True)

        weights = softmax_weights(c.astype(np.float64), float(cmax), beta * float(cmax), floor=0.04)
        g = grad_weights(b.astype(np.float64), weights, N)

        one_idx = np.nonzero(b == 1)[0]
        zero_idx = np.nonzero(b == 0)[0]
        if one_idx.size == 0 or zero_idx.size == 0:
            break

        si = choose_by_pressure(g[one_idx], rng, True, temp=0.12)
        sj = choose_by_pressure(g[zero_idx], rng, False, temp=0.12)
        if si < 0 or sj < 0:
            break
        i = int(one_idx[si])
        j = int(zero_idx[sj])
        if i == j:
            continue

        # Apply swap: 1 at i -> 0, 0 at j -> 1.
        c[i:i + n] -= 2 * b
        c[j:j + n] += 2 * b
        c[2 * i] += 1
        c[2 * j] += 1
        c[i + j] -= 2
        b[i] = 0
        b[j] = 1

        new_cmax = int(np.max(c))
        new_score = float(2.0 * n * new_cmax / (float(m) * float(m)))
        delta = new_score - score

        accept = False
        if new_score < score * (1.0 - 1e-12):
            accept = True
        elif delta <= 1e-12:
            if rng.random() < 0.45:
                accept = True
        elif delta < 0.25:
            if rng.random() < np.exp(-delta / max(temp, 1e-12)):
                accept = True

        if accept:
            if new_score < best_score:
                best_score = float(new_score)
                best_cmax = new_cmax
                best_b = b.astype(np.float64, copy=True)
            stale = 0
            temp = max(0.0015, temp * 0.996)
            beta = min(1.2, beta * 1.002)
        else:
            # Revert swap using current b (now has 1 at j, 0 at i).
            c[j:j + n] -= 2 * b
            c[i:i + n] += 2 * b
            c[2 * j] += 1
            c[2 * i] += 1
            c[i + j] -= 2
            b[j] = 0
            b[i] = 1

            stale += 1
            temp = max(0.0007, temp * 0.965)
            if stale % 220 == 0:
                beta = min(1.8, beta * 1.7)
                temp = min(0.09, temp * 1.8)
                stale = 0
            if stale > 1400:
                break

        it += 1
        if it % 20000 == 0:
            c = np.rint(c.astype(np.float64)).astype(np.int64)

    return best_b, float(best_score)


def run_binary_phase(a0, deadline, seed: int = 0):
    a0 = normalize_positive(np.asarray(a0, dtype=np.float64).ravel(), floor=0.0)
    n = a0.size
    if n < 512:
        a0 = resample_seq(a0, 1024)
        n = 1024
    elif n > 4096:
        a0 = resample_seq(a0, 4096)
        n = 4096

    fracs = (0.22, 0.35, 0.50)
    best_b = None
    best_score = float("inf")

    per = max(20.0, (float(deadline) - time.monotonic()) / float(len(fracs) + 0.25))

    for frac in fracs:
        if time.monotonic() > deadline - 20.0:
            break
        m = int(round(float(n) * float(frac)))
        b0 = binary_from_values(a0, m)
        b1, sc1 = anneal_binary(b0, time.monotonic() + per, seed=seed + int(frac * 1000) + int(n))
        if b1 is not None and np.isfinite(sc1) and sc1 < best_score:
            best_score = float(sc1)
            best_b = b1
            print("binary fraction n={} score={:.9f}".format(n, best_score), flush=True)

    if best_b is None:
        b0 = binary_from_values(a0, max(32, n // 3))
        best_b, best_score = anneal_binary(b0, deadline, seed=seed + 9999)

    return np.asarray(best_b, dtype=np.float64), float(best_score)


def evaluate_best_final(out_list):
    try:
        ev = globals().get("evaluate_sequence", None)
        if ev is None:
            return float("inf")
        val = float(ev(out_list))
        return val
    except Exception:
        return float("inf")


def propose_candidate(seed: int = 42, budget_s: float = 2370.0, **kwargs):
    start = time.monotonic()
    total = max(180.0, min(2370.0, float(budget_s) - 45.0))
    deadline = start + total

    best_a = make_power_sequence(2048, gamma=0.5, left_u=1.0)
    best_score = internal_score(best_a)
    print("initial power score = {:.9f}".format(best_score), flush=True)

    given = get_given_sequence()
    if given is not None:
        sc = internal_score(given)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_a = given.copy()
            print("given sequence score = {:.9f}".format(best_score), flush=True)

    # Quick raw seeds at 2048.
    if time.monotonic() < deadline - 1800.0:
        for seq in generate_quick_seeds(2048, reference=best_a, seed=int(seed) + 17, max_count=18):
            sc = internal_score(seq)
            if np.isfinite(sc) and sc < best_score:
                best_score = float(sc)
                best_a = seq.copy()
                print("seed 2048 score = {:.9f}".format(best_score), flush=True)

    # Low-dimensional smooth acclimation.
    lowdim_plans = (
        (512, 35.0),
        (1024, 90.0),
        (2048, 210.0),
        (4096, 330.0),
    )
    for n, allow in lowdim_plans:
        used = time.monotonic() - start
        reserve = 900.0 if n <= 2048 else 450.0
        if used > total - reserve:
            break
        allow_now = min(float(allow), max(0.0, deadline - time.monotonic() - reserve))
        if allow_now < 18.0:
            break

        a_init = resample_seq(best_a, n)
        a1, sc1 = run_param_optimize(
            n,
            time.monotonic() + allow_now,
            seed=int(seed) + 1000 + int(n),
            reference=a_init,
            max_starts=5,
            verbose=False,
        )
        if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
            best_score = float(sc1)
            best_a = a1.copy()
            print("lowdim selected n={} score={:.9f}".format(n, best_score), flush=True)

    # Main 2048 high-dimensional polish.
    if time.monotonic() < deadline - 1300.0:
        allow = min(400.0, max(0.0, deadline - time.monotonic() - 1200.0))
        if allow > 40.0:
            a0 = resample_seq(best_a, 2048)
            a1, sc1 = polish_logsum(a0, time.monotonic() + allow, seed=int(seed) + 2000)
            if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
                best_score = float(sc1)
                best_a = a1.copy()
                print("polish 2048 score={:.9f}".format(best_score), flush=True)

    # Move to 4096 and polish there.
    if time.monotonic() < deadline - 700.0:
        a0 = resample_seq(best_a, 4096)

        allow_param = min(160.0, max(0.0, deadline - time.monotonic() - 620.0))
        if allow_param > 30.0:
            a1, sc1 = run_param_optimize(
                4096,
                time.monotonic() + allow_param,
                seed=int(seed) + 3000,
                reference=a0,
                max_starts=5,
                verbose=False,
            )
            if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
                best_score = float(sc1)
                best_a = a1.copy()
                print("lowdim 4096 selected score={:.9f}".format(best_score), flush=True)
            a0 = resample_seq(best_a, 4096)

        allow Polish = min(650.0, max(0.0, deadline - time.monotonic() - 320.0))
        if allow_polish > 90.0:
            a1, sc1 = polish_logsum(a0, time.monotonic() + allow_polish, seed=int(seed) + 3100, verbose=True)
            if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
                best_score = float(sc1)
                best_a = a1.copy()
                print("polish 4096 score={:.9f}".format(best_score), flush=True)

        # A shorter restarted polish from a perturbed point.
        if time.monotonic() < deadline - 360.0:
            allow2 = min(240.0, max(0.0, deadline - time.monotonic() - 260.0))
            if allow2 > 45.0:
                a2 = add_noise_sequence(resample_seq(best_a, 4096), sigma=0.035, seed=int(seed) + 3200)
                a1, sc1 = polish_logsum(a2, time.monotonic() + allow2, seed=int(seed) + 3250)
                if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
                    best_score = float(sc1)
                    best_a = a1.copy()
                    print("restart polish 4096 score={:.9f}".format(best_score), flush=True)

    # Binary-support search, then optional continuous lifting of a good binary.
    if time.monotonic() < deadline - 520.0:
        allow_bin = min(300.0, max(0.0, deadline - time.monotonic() - 350.0))
        if allow_bin > 60.0:
            a0 = resample_seq(best_a, 2048)
            b0, sc0 = run_binary_phase(a0, time.monotonic() + allow_bin, seed=int(seed) + 4000)
            if b0 is not None and np.isfinite(sc0) and sc0 < best_score:
                best_score = float(sc0)
                best_a = np.asarray(b0, dtype=np.float64)
                print("binary best score={:.9f}".format(best_score), flush=True)

    if time.monotonic() < deadline - 300.0:
        allow_bin2 = min(260.0, max(0.0, deadline - time.monotonic() - 220.0))
        if allow_bin2 > 60.0:
            a0 = resample_seq(best_a, 4096)
            b0, sc0 = run_binary_phase(a0, time.monotonic() + allow_bin2, seed=int(seed) + 4100)
            if b0 is not None and np.isfinite(sc0) and sc0 < best_score:
                best_score = float(sc0)
                best_a = np.asarray(b0, dtype=np.float64)
                print("binary 4096 best score={:.9f}".format(best_score), flush=True)

    # If a binary candidate is current, try to continuously lift it briefly.
    if time.monotonic() < deadline - 180.0 and np.max(np.abs(best_a - np.rint(best_a))) < 1e-6:
        lifted = np.asarray(best_a, dtype=np.float64)
        lifted = lifted + 0.035
        lifted = normalize_positive(lifted, floor=0.0)
        allow_lift = min(120.0, max(0.0, deadline - time.monotonic() - 100.0))
        if allow_lift > 35.0:
            a1, sc1 = polish_logsum(lifted, time.monotonic() + allow_lift, seed=int(seed) + 4500)
            if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
                best_score = float(sc1)
                best_a = a1.copy()
                print("binary lift score={:.9f}".format(best_score), flush=True)

    # Short final polish on the current best length, capping length to 4096.
    if best_a.size > 4096:
        best_a = resample_seq(best_a, 4096)
        best_score = internal_score(best_a)

    if best_a.size < 1024:
        best_a = resample_seq(best_a, 2048)
        best_score = internal_score(best_a)

    if time.monotonic() < deadline - 25.0:
        final_allow = min(90.0, max(0.0, deadline - time.monotonic() - 20.0))
        if final_allow > 20.0:
            a1, sc1 = polish_logsum(best_a, time.monotonic() + final_allow, seed=int(seed) + 9100)
            if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
                best_score = float(sc1)
                best_a = a1.copy()
                print("final polish score={:.9f}".format(best_score), flush=True)

    best_a = normalize_positive(np.asarray(best_a, dtype=np.float64).ravel(), floor=0.0)
    out_list = [float(x) for x in best_a]

    final_eval = evaluate_best_final(out_list)
    if not np.isfinite(final_eval) or final_eval > 2.5:
        fallback = make_uniform(2048)
        out_list = [float(x) for x in fallback]
        final_eval = evaluate_best_final(out_list)

    if not np.all(np.isfinite(out_list)):
        fallback = make_uniform(1024)
        out_list = [float(x) for x in fallback]
        final_eval = evaluate_best_final(out_list)

    print("final internal score = {:.12f}".format(best_score), flush=True)
    print("final evaluated score = {:.12f}, length = {}".format(final_eval, len(out_list)), flush=True)
    return out_list
