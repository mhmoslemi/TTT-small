```python
"""
Search for non-negative sequences minimizing
    2*n*max(convolve(a,a)) / sum(a)^2.

Implementation notes:
- The objective is scale invariant, so points are optimized as probabilities.
- A correct FFT/backprop gradient is used for the full auto-convolution.
- Optimization is done in logit/softmax coordinates with Adam + line search.
- Coarse-to-fine resizing, diverse monotone/oscillatory seeds, and periodic
  restarts are used to avoid local minima.
"""

import time
import math
import numpy as np


def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def normalize_positive(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s


def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64)
    if z.size == 0:
        return z
    z = np.where(np.isfinite(z), z, -50.0)
    z = z - float(np.max(z))
    z = np.clip(z, -50.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(z.size, dtype=np.float64) / float(z.size)
    return e / s


def score_normalized_fast(p, L=None):
    """Score assuming p is already normalized to sum 1."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n <= 0:
        return np.inf
    p = np.where(np.isfinite(p), p, 0.0)
    if L is None:
        L = next_pow2(2 * n - 1)
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0.0:
        return np.inf
    return float(2.0 * n * cm)


def fast_score(a):
    p = normalize_positive(a)
    if p is None:
        return np.inf
    n = p.size
    if n <= 0:
        return np.inf
    L = next_pow2(2 * n - 1)
    return score_normalized_fast(p, L)


def make_weights(c, cm, tau, top_K=None, bg=0.0):
    n = len(c)
    if n == 0 or not np.isfinite(cm) or cm <= 0.0:
        return np.ones(n, dtype=np.float64) / max(1, n)
    tau = max(1e-300, float(tau))

    if top_K is None:
        z = (c - cm) / tau
        z = np.clip(z, -80.0, 0.0)
        e = np.exp(z)
        s = float(e.sum())
        if not np.isfinite(s) or s <= 0.0:
            e = np.zeros(n, dtype=np.float64)
            e[int(np.argmax(c))] = 1.0
            s = 1.0
        return e / s

    k = int(top_K)
    if k <= 0:
        return np.ones(n, dtype=np.float64) / float(n)
    if k >= n:
        z = (c - cm) / tau
        z = np.clip(z, -80.0, 0.0)
        e = np.exp(z)
        s = float(e.sum())
        if not np.isfinite(s) or s <= 0.0:
            e = np.zeros(n, dtype=np.float64)
            e[int(np.argmax(c))] = 1.0
            s = 1.0
        return e / s

    idx = np.argpartition(c, -k)[-k:]
    z = (c[idx] - cm) / tau
    z = np.clip(z, -80.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())

    w = np.zeros(n, dtype=np.float64)
    if not np.isfinite(s) or s <= 0.0:
        w[idx] = 1.0 / float(idx.size)
        return w

    bgv = 0.0 if bg is None else float(bg)
    if bgv <= 0.0:
        w[idx] = e / s
        return w
    if bgv >= 1.0:
        bgv = 1.0

    w[:] = bgv / float(n)
    scale = (1.0 - bgv) / s
    w[idx] += e * scale
    tw = float(w.sum())
    if not np.isfinite(tw) or tw <= 0.0:
        w[idx] = e / s
        return w
    return w / tw


def gradient_from_weights(p, w, L):
    n = p.size
    Wf = np.fft.rfft(w, L)
    pref = np.fft.rfft(p[::-1], L)
    H = np.fft.irfft(Wf * pref, L)
    grad = 2.0 * H[n - 1:2 * n - 1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return grad


def resize_sequence(seq, m):
    p = normalize_positive(seq)
    m = int(m)
    if m <= 1:
        return np.ones(max(1, m), dtype=np.float64) / float(max(1, m))
    if p is None:
        return np.ones(m, dtype=np.float64) / float(m)
    n = p.size
    if n == m:
        return p.copy()
    if n == 1:
        return np.ones(m, dtype=np.float64) / float(m)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, m)
    out = np.interp(x_new, x_old, p)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(m, dtype=np.float64) / float(m)
    return out / s


def power_seed(n, a, b, reverse=False, eps=1e-6):
    if n <= 1:
        return np.ones(n, dtype=np.float64)
    x = np.linspace(0.0, 1.0, n)
    p = (x + eps) ** a * ((1.0 - x) + eps) ** b
    if reverse:
        p = p[::-1]
    return normalize_positive(p)


def exp_seed(n, beta, reverse=False):
    if n <= 1:
        return np.ones(n, dtype=np.float64)
    x = np.linspace(0.0, 1.0, n)
    p = np.exp(beta * (x - 0.5))
    if reverse:
        p = p[::-1]
    return normalize_positive(p)


def random_smooth_parent(n, rng):
    if n <= 1:
        return np.ones(n, dtype=np.float64)
    x = np.linspace(0.0, 1.0, n)
    z = np.zeros(n, dtype=np.float64)

    side = 1.0 if rng.random() < 0.5 else -1.0
    z += side * rng.uniform(0.5, 4.0) * x

    curv_sign = 1.0 if rng.random() < 0.5 else -1.0
    z += curv_sign * rng.uniform(0.3, 1.8) * (x - 0.5) ** 2

    for k in range(1, 6):
        amp = rng.uniform(0.0, 1.2 / (k ** 0.75))
        phase = rng.uniform(0.0, 2.0 * math.pi)
        z += amp * np.cos(2.0 * math.pi * k * x + phase)

    z -= float(np.mean(z))
    return softmax_from_z(z)


def consider_candidate(p, arrays, scores, Lscore):
    q = normalize_positive(p)
    if q is None:
        return
    s = score_normalized_fast(q, Lscore)
    arrays.append(q)
    scores.append(float(s))


def generate_seed_candidates(n, rng, parent=None, n_random=16, max_keep=4):
    arrays = []
    scores = []
    Lscore = next_pow2(2 * n - 1)

    if parent is not None:
        consider_candidate(resize_sequence(parent, n), arrays, scores, Lscore)

    consider_candidate(np.ones(n, dtype=np.float64), arrays, scores, Lscore)

    a_vals = (0.0, 0.25, 0.4, 0.5, 0.6, 0.75, 1.0, 1.5, 2.0)
    b_vals = (0.0, 0.1, 0.25, 0.5)

    for a in a_vals:
        for b in b_vals:
            if a == 0.0 and b == 0.0:
                continue
            consider_candidate(power_seed(n, a, b, False), arrays, scores, Lscore)
            consider_candidate(power_seed(n, a, b, True), arrays, scores, Lscore)

    for beta in (-4.0, -2.0, -1.0, 0.0, 1.0, 2.0, 4.0):
        consider_candidate(exp_seed(n, beta, False), arrays, scores, Lscore)
        consider_candidate(exp_seed(n, beta, True), arrays, scores, Lscore)

    n_random = int(max(0, n_random))
    for _ in range(n_random):
        consider_candidate(random_smooth_parent(n, rng), arrays, scores, Lscore)

    if not arrays:
        arrays.append(np.ones(n, dtype=np.float64) / float(n))
        scores.append(np.inf)

    scores_arr = np.asarray(scores, dtype=np.float64)
    order = np.argsort(scores_arr)
    top = []
    for idx in order:
        idx = int(idx)
        if not np.isfinite(scores_arr[idx]):
            continue
        top.append(arrays[idx])
        if len(top) >= max_keep:
            break
    return top


def make_restart_parent(best_p, n, rng):
    if best_p is None:
        return random_smooth_parent(n, rng)
    if rng.random() < 0.35:
        return best_p
    rp = random_smooth_parent(n, rng)
    t = rng.uniform(0.25, 0.75)
    mix = (1.0 - t) * best_p + t * rp
    return normalize_positive(mix)


def update_bests(p, s, bparent, bscore, b8192, s8192, b16384, s16384, max_len=16384):
    if p is None or not np.isfinite(s):
        return bparent, bscore, b8192, s8192, b16384, s16384
    try:
        arr = np.asarray(p, dtype=np.float64)
    except Exception:
        return bparent, bscore, b8192, s8192, b16384, s16384
    if arr.size == 0:
        return bparent, bscore, b8192, s8192, b16384, s16384

    if arr.size > max_len:
        arr = resize_sequence(arr, max_len)
        s = fast_score(arr)
        if not np.isfinite(s):
            return bparent, bscore, b8192, s8192, b16384, s16384

    arr = normalize_positive(arr)
    if arr is None:
        return bparent, bscore, b8192, s8192, b16384, s16384

    if s < bscore:
        bparent = arr.copy()
        bscore = float(s)
    if arr.size == 8192 and s < s8192:
        b8192 = arr.copy()
        s8192 = float(s)
    if arr.size == 16384 and s < s16384:
        b16384 = arr.copy()
        s16384 = float(s)
    return bparent, bscore, b8192, s8192, b16384, s16384


def run_local_optimizer(p0, duration_s, rng, tag="opt", lr0=None):
    p0 = normalize_positive(p0)
    if p0 is None or p0.size == 0:
        return None, np.inf
    n = p0.size
    if n <= 1:
        return p0, 2.0
    duration_s = max(0.0, float(duration_s))
    start = time.monotonic()
    deadline = start + duration_s
    s0 = fast_score(p0)
    if duration_s <= 0.0:
        return p0, s0

    if lr0 is None:
        if n <= 256:
            lr0 = 0.055
        elif n <= 1024:
            lr0 = 0.040
        elif n <= 2048:
            lr0 = 0.030
        elif n <= 4096:
            lr0 = 0.022
        else:
            lr0 = 0.018

    Lgrad = next_pow2(3 * n - 2)
    Lscore = next_pow2(2 * n - 1)

    z = np.log(np.maximum(p0, 1e-300))
    z = np.where(np.isfinite(z), z, -50.0)
    z -= float(np.mean(z))
    z = np.clip(z, -50.0, 50.0)
    p = softmax_from_z(z)

    best_p = p.copy()
    current_s = score_normalized_fast(p, Lscore)
    best_s = current_s
    if not np.isfinite(best_s):
        best_s = 2.0

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    bias = 0
    beta1 = 0.90
    beta2 = 0.999
    adam_eps = 1e-8

    if n <= 64:
        patience = 80
    elif n <= 256:
        patience = 250
    elif n <= 1024:
        patience = 800
    elif n <= 2048:
        patience = 1500
    elif n <= 4096:
        patience = 3000
    else:
        patience = 5000

    if n <= 128:
        base_top = None
    elif n <= 512:
        base_top = 24
    elif n <= 1024:
        base_top = 48
    elif n <= 2048:
        base_top = 80
    elif n <= 8192:
        base_top = 128
    else:
        base_top = 256

    initial_dur = max(1e-12, deadline - start)
    no_improve = 0
    step = 0
    last_print = start
    eps_improve = 1e-10

    while time.monotonic() < deadline:
        rem_prop = max(0.0, min(1.0, (deadline - time.monotonic()) / initial_dur))
        if rem_prop <= 0.0:
            break

        F = np.fft.rfft(p, Lgrad)
        c = np.fft.irfft(F * F, Lgrad)[:2 * n - 1]
        c = np.where(np.isfinite(c), c, 0.0)
        c = np.maximum(c, 0.0)
        cm = float(c.max())
        current_s = 2.0 * n * cm
        if not np.isfinite(current_s):
            break

        if current_s < best_s - eps_improve:
            best_s = current_s
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > patience:
            rp = make_restart_parent(best_p, n, rng)
            noise = rng.uniform(0.08, 0.45)
            zz = np.log(np.maximum(rp, 1e-300)) + rng.normal(0.0, noise, n)
            zz = np.where(np.isfinite(zz), zz, -50.0)
            zz -= float(np.mean(zz))
            zz = np.clip(zz, -50.0, 50.0)
            z = zz
            p = softmax_from_z(z)
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            bias = 0
            no_improve = 0
            continue

        if no_improve > max(50, patience // 4) and current_s > best_s * (1.0 + 1e-4) + 1e-9:
            zz = np.log(np.maximum(best_p, 1e-300))
            zz = np.where(np.isfinite(zz), zz, -50.0)
            zz -= float(np.mean(zz))
            zz = np.clip(zz, -50.0, 50.0)
            z = zz
            p = softmax_from_z(z)
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            bias = 0
            no_improve = 0
            continue

        if n <= 512:
            tau_target = 0.012 + (0.20 - 0.012) * rem_prop
        else:
            tau_target = 0.012 + (0.12 - 0.012) * rem_prop
        if no_improve > max(50, patience * 0.3):
            tau_target = min(0.30, tau_target * 1.6)
        tau = tau_target * cm

        if base_top is None:
            top_K = None
        else:
            top_K = max(8, int(base_top * (0.35 + 0.65 * rem_prop)))
            if rem_prop > 0.65:
                top_K = None

        bg = 0.01 + 0.05 * rem_prop
        w = make_weights(c, cm, tau, top_K=top_K, bg=bg)
        grad = gradient_from_weights(p, w, Lgrad)

        gp = float(np.dot(p, grad))
        grad = grad - p * gp
        grad = np.where(np.isfinite(grad), grad, 0.0)
        gmax = float(np.max(np.abs(grad))) if grad.size else 0.0

        if gmax <= 1e-18:
            small = rng.uniform(1e-6, 1e-4)
            z = z + rng.normal(0.0, small, n)
            z = np.where(np.isfinite(z), z, 0.0)
            z -= float(np.mean(z))
            z = np.clip(z, -50.0, 50.0)
            p = softmax_from_z(z)
            continue

        bias += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        b1 = 1.0 - beta1 ** bias
        b2 = 1.0 - beta2 ** bias
        mhat = m / b1
        vhat = v / b2
        upd = mhat / (np.sqrt(vhat) + adam_eps)
        upd = np.where(np.isfinite(upd), upd, 0.0)
        umax = float(np.max(np.abs(upd))) if upd.size else 0.0
        if umax > 8.0:
            upd *= 8.0 / umax

        lr_eff = float(lr0) * (0.45 + 0.55 * rem_prop)

        best_trial_s = np.inf
        best_trial_z = None
        best_trial_p = None

        for mult in (1.0, 0.5, 2.0):
            zt = z - lr_eff * mult * upd
            zt = np.where(np.isfinite(zt), zt, z)
            zt -= float(np.mean(zt))
            zt = np.clip(zt, -50.0, 50.0)
            pt = softmax_from_z(zt)
            st = score_normalized_fast(pt, Lscore)
            if np.isfinite(st) and st < best_trial_s:
                best_trial_s = st
                best_trial_z = zt
                best_trial_p = pt

        accept = False
        if best_trial_z is not None:
            if best_trial_s < current_s - eps_improve:
                accept = True
            elif no_improve > 120 and best_trial_s < current_s + 1e-9:
                accept = True

        if accept:
            z = best_trial_z
            p = best_trial_p
            current_s = best_trial_s

        step += 1
        if time.monotonic() - last_print > 12.0:
            print(f"{tag} n={n} step={step} best={best_s:.6f} cur={current_s:.6f}")
            last_print = time.monotonic()

    if np.isfinite(current_s) and current_s < best_s - eps_improve:
        best_s = current_s
        best_p = p.copy()
    return best_p, best_s


def propose_candidate(seed=42, budget_s=2400.0, **kwargs):
    """Search for a good non-negative sequence under the given time budget."""
    if "seed" not in kwargs:
        pass
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not np.isfinite(budget) or budget <= 5:
        return [1.0 / 1024.0] * 1024

    deadline = start + max(0.0, budget - 45.0)
    MAX_SAFE_LEN = 16384

    best_parent = None
    best_score = np.inf
    best8192 = None
    score8192 = np.inf
    best16384 = None
    score16384 = np.inf

    hs_parent = None
    try:
        hs_arr = np.asarray(height_sequence_1, dtype=np.float64)
        hs = normalize_positive(hs_arr)
        if hs is not None:
            hs_parent = hs.copy()
            s = fast_score(hs)
            best_parent, best_score, best8192, score8192, best16384, score16384 = update_bests(
                hs, s, best_parent, best_score, best8192, score8192, best16384, score16384, MAX_SAFE_LEN
            )
            if hs.size != 8192:
                r8 = resize_sequence(hs, 8192)
                s8 = fast_score(r8)
                best_parent, best_score, best8192, score8192, best16384, score16384 = update_bests(
                    r8, s8, best_parent, best_score, best8192, score8192, best16384, score16384, MAX_SAFE_LEN
                )
            if hs.size != 16384:
                r16 = resize_sequence(hs, 16384)
                s16 = fast_score(r16)
                best_parent, best_score, best8192, score8192, best16384, score16384 = update_bests(
                    r16, s16, best_parent, best_score, best8192, score8192, best16384, score16384, MAX_SAFE_LEN
                )
    except Exception:
        pass

    base_stages = [
        (32, 10.0),
        (64, 20.0),
        (128, 30.0),
        (256, 45.0),
        (512, 65.0),
        (1024, 100.0),
        (2048, 180.0),
        (4096, 330.0),
        (8192, 950.0),
        (16384, 170.0),
    ]
    scale = max(0.18, min(1.0, budget / 2400.0))

    for n, base_dur in base_stages:
        rem = deadline - time.monotonic()
        if rem < 45.0:
            break
        dur = base_dur * scale
        dur = min(dur, max(0.0, rem - 35.0))
        if dur < 12.0:
            continue

        if best_parent is None:
            seeds = generate_seed_candidates(n, rng, None, n_random=30, max_keep=4)
            pstart = seeds[0] if seeds else (np.ones(n, dtype=np.float64) / float(n))
            Lscore = next_pow2(2 * n - 1)
            for cand in seeds:
                sc = score_normalized_fast(cand, Lscore)
                best_parent, best_score, best8192, score8192, best16384, score16384 = update_bests(
                    cand, sc, best_parent, best_score, best8192, score8192, best16384, score16384, MAX_SAFE_LEN
                )
        else:
            if n <= 512:
                n_rand = 12
            elif n <= 2048:
                n_rand = 22
            else:
                n_rand = 32
            seeds = generate_seed_candidates(n, rng, best_parent, n_random=n_rand, max_keep=4)
            Lscore = next_pow2(2 * n - 1)
            for cand in seeds:
                sc = score_normalized_fast(cand, Lscore)
                best_parent, best_score, best8192, score8192, best16384, score16384 = update_bests(
                    cand, sc, best_parent, best_score, best8192, score8192, best16384, score16384, MAX_SAFE_LEN
                )
            pstart = seeds[0] if seeds else resize_sequence(best_parent, n)

        popt, sopt = run_local_optimizer(pstart, dur, rng, tag=f"stage{n}")
        if popt is not None and np.isfinite(sopt):
            best_parent, best_score, best8192, score8192, best16384, score16384 = update_bests(
                popt, sopt, best_parent, best_score, best8192, score8192, best16384, score16384, MAX_SAFE_LEN
            )

    rem = deadline - time.monotonic()
    if rem > 70.0:
        if best8192 is None and best_parent is not None:
            p8 = resize_sequence(best_parent, 8192)
            s8 = fast_score(p8)
            best_parent, best_score, best8192, score8192, best16384, score16384 = update_bests(
                p8, s8, best_parent, best_score, best8192, score8192, best16384, score16384, MAX_SAFE_LEN
            )

        if best8192 is not None:
            pstart = best8192
            dur = min(max(0.0, rem - 30.0) * 0.70, 320.0)
            if dur > 25.0:
                popt, sopt = run_local_optimizer(pstart, dur, rng, tag="pass8192A")
                best_parent, best_score, best8192, score8192, best16384, score16384 = update_bests(
                    popt, sopt, best_parent, best_score, best8192, score8192, best16384, score16384, MAX_SAFE_LEN
                )

        rem = deadline - time.monotonic()
        if rem > 50.0:
            alt_seeds = generate_seed_candidates(8192, rng, best_parent, n_random=24, max_keep=1)
            if alt_seeds:
                pstart = alt_seeds[0]
                dur = min(max(0.0, rem - 25.0) * 0.65, 180.0)
                if dur > 20.0:
                    popt, sopt = run_local_optimizer(pstart, dur, rng, tag="pass8192B")
                    best_parent, best_score, best8192, score8192, best16384, score16384 = update_bests(
                        popt, sopt, best_parent, best_score, best8192, score8192, best16384, score16384, MAX_SAFE_LEN
                    )

        rem = deadline - time.monotonic()
        if rem > 90.0:
            if best16384 is not None:
                pstart = best16384
            elif best_parent is not None:
                pstart = resize_sequence(best_parent, 16384)
            else:
                pstart = np.ones(16384, dtype=np.float64) / 16384.0
            dur = min(max(0.0, rem - 35.0), 220.0)
            if dur > 30.0:
                popt, sopt = run_local_optimizer(pstart, dur, rng, tag="pass16384")
                best_parent, best_score, best8192, score8192, best16384, score16384 = update_bests(
                    popt, sopt, best_parent, best_score, best8192, score8192, best16384, score16384, MAX_SAFE_LEN
                )

    final_list = []
    for cand in (best_parent, best8192, best16384, hs_parent):
        if cand is None:
            continue
        p = normalize_positive(cand)
        if p is None:
            continue
        if p.size > MAX_SAFE_LEN:
            p = resize_sequence(p, 8192)
        p = np.asarray(p, dtype=np.float64)
        final_list.append(p)
        if p.size != 8192:
            final_list.append(resize_sequence(p, 8192))
        if p.size < 16384:
            final_list.append(resize_sequence(p, 16384))

    final_arr = None
    final_score = np.inf
    for p in final_list:
        p = normalize_positive(p)
        if p is None:
            continue
        if p.size > MAX_SAFE_LEN:
            p = resize_sequence(p, 8192)
        s = fast_score(p)
        if np.isfinite(s) and s < final_score:
            final_score = s
            final_arr = p.copy()

    if final_arr is None:
        final_arr = np.ones(1024, dtype=np.float64) / 1024.0
        final_score = 2.0

    if final_arr.size > MAX_SAFE_LEN:
        final_arr = resize_sequence(final_arr, 8192)
        final_score = fast_score(final_arr)

    if final_arr.size == 16384:
        r8 = resize_sequence(final_arr, 8192)
        s8 = fast_score(r8)
        if np.isfinite(s8) and s8 <= final_score + 0.001:
            final_arr = r8
            final_score = s8

    out = np.asarray(final_arr, dtype=np.float64)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 0.0:
        out = np.ones(1024, dtype=np.float64) / 1024.0
    elif s < 0.01:
        scale = 10.0 / s
        out = np.clip(out * scale, 0.0, 1000.0)

    try:
        return [float(x) for x in out]
    except Exception:
        return [1.0 / 1024.0] * 1024
```
