- Frank-Wolfe LP direction for moderate lengths
- Multi-scale exploration with diverse initializations
- Nesterov momentum and adaptive step sizing
"""

import time
import numpy as np

try:
    from scipy.optimize import linprog
    HAVE_LINPROG = True
except Exception:
    linprog = None
    HAVE_LINPROG = False


def next_pow2(x):
    x = int(max(x, 1))
    return 1 << (x - 1).bit_length()


def normalize_positive(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s


def fast_score(a, L=None):
    p = normalize_positive(a)
    if p is None:
        return np.inf
    n = p.size
    if n <= 0:
        return np.inf
    if L is None:
        L = next_pow2(2 * n - 1)
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0.0:
        return np.inf
    return float(2.0 * n * cm)


def compute_conv(p, L):
    n = p.size
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    return np.where(np.isfinite(c), c, 0.0)


def gradient_topk(p, c, K=5, L=None):
    """Compute gradient of sum of top-K convolution values w.r.t. logit p."""
    n = p.size
    if K >= 2 * n - 1:
        K = 2 * n - 1
    top_idx = np.argpartition(c, -K)[-K:]
    top_vals = c[top_idx]
    w = top_vals / top_vals.sum()
    
    # Gradient: 2 * sum_k w_k * p[k-i] for valid k-i
    # This is 2 * (w_conv with p_reversed), taking first n entries
    p_flipped = p[::-1]
    L2 = next_pow2(2 * n - 1 + n - 1)  # 3n-2
    W = np.fft.rfft(np.concatenate([w, np.zeros(L2 - 2 * n + 1)]), L2)
    Pf = np.fft.rfft(p_flipped, L2)
    h_full = np.fft.irfft(W * Pf, L2)
    h = h_full[:n]
    
    grad = 2.0 * h
    mean_g = float(np.dot(p, grad))
    grad_z = grad - p * mean_g
    return grad_z, float(c.max())


def gradient_l2(p, c, L=None):
    """Gradient of L2 norm squared of convolution w.r.t. logit p."""
    n = p.size
    # d(sum c^2)/dp_i = 4 * sum_k c_k * p_{k-i}
    p_flipped = p[::-1]
    L2 = next_pow2(3 * n - 2)
    C = np.fft.rfft(np.concatenate([c, np.zeros(L2 - 2 * n + 1)]), L2)
    Pf = np.fft.rfft(p_flipped, L2)
    h_full = np.fft.irfft(C * Pf, L2)
    h = h_full[:n]
    
    grad = 4.0 * h
    mean_g = float(np.dot(p, grad))
    grad_z = grad - p * mean_g
    return grad_z


def gradient_hybrid(p, c, tau=0.01, L=None):
    """Hybrid gradient: smooth max + L2 component."""
    n = p.size
    cm = float(c.max())
    z = (c - cm) / tau
    np.clip(z, -50, 0, out=z)
    expz = np.exp(z)
    expsum = float(expz.sum())
    if expsum <= 0:
        expz = np.zeros_like(c)
        expz[int(np.argmax(c))] = 1.0
        expsum = 1.0
    w = expz / expsum
    
    # Smooth max gradient component
    p_flipped = p[::-1]
    L2 = next_pow2(3 * n - 2)
    W = np.fft.rfft(np.concatenate([w, np.zeros(L2 - 2 * n + 1)]), L2)
    Pf = np.fft.rfft(p_flipped, L2)
    h_sm = np.fft.irfft(W * Pf, L2)[:n]
    
    # L2 gradient component
    C = np.fft.rfft(np.concatenate([c, np.zeros(L2 - 2 * n + 1)]), L2)
    h_l2 = np.fft.irfft(C * Pf, L2)[:n]
    
    grad = 2.0 * h_sm + 0.5 * h_l2
    mean_g = float(np.dot(p, grad))
    grad_z = grad - p * mean_g
    
    # Clip gradient for stability
    max_g = float(np.max(np.abs(grad_z)))
    if max_g > 0.05:
        grad_z *= 0.05 / max_g
    
    return grad_z, cm


def softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = np.where(np.isfinite(z), z, -50.0)
    z = z - z.max()
    z = np.clip(z, -50.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0:
        e = np.ones_like(z)
        s = float(e.size)
    return e / s


def cheb_basis(n, D):
    x = np.linspace(0.0, 1.0, n)
    t = 2.0 * x - 1.0
    t = np.clip(t, -1.0, 1.0)
    ang = np.arccos(t)
    out = np.empty((n, D), dtype=np.float64)
    for m in range(1, D + 1):
        out[:, m - 1] = np.cos(m * ang)
    return out


def poly_parent(B, theta, mode=0, eps=0.12):
    val = B.dot(theta)
    if mode == 0:
        h = 1.0 + eps + val
    else:
        h = np.exp(np.clip(val, -10.0, 10.0))
    return normalize_positive(h)


def resize_sequence(seq, m):
    p = normalize_positive(seq)
    m = int(m)
    if p is None:
        return np.ones(m, dtype=np.float64) / m
    n = p.size
    if n == m:
        return p.copy()
    if n == 1:
        return np.full(m, 1.0 / m, dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, m)
    out = np.interp(x_new, x_old, p)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 0:
        return np.ones(m, dtype=np.float64) / m
    return out / s


def build_conv_matrix(f, scale):
    """Build the convolution matrix A such that A @ q = convolve(f/scale, q)."""
    n = f.size
    f_s = f / scale
    r = f_s[::-1]
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        s = max(0, k - n + 1)
        e = min(n - 1, k)
        start = n - 1 - k + s
        A[k, s:e + 1] = r[start:start + 2 * n - 2 - k + 1 - (e - s)]
        # Simpler: A[k, j] = f_s[n-1 - (k - j)] for valid j
    # Redo properly
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = f_s[idx]
    return A


def lp_direction(f, rng, n_lp):
    """Frank-Wolfe: find best q in simplex with convolve(f, q) <= max(f*f)."""
    if not HAVE_LINPROG:
        return None
    f = resize_sequence(f, n_lp)
    f = f / f.sum()
    n = f.size
    maxf = float(np.max(compute_conv(f, next_pow2(2*n-1))))
    if maxf <= 0:
        return None
    A = build_conv_matrix(f, maxf)
    obj = -np.ones(n, dtype=np.float64)
    obj += rng.uniform(0, 1e-8, n)
    try:
        res = linprog(obj, A_ub=A, b_ub=np.ones(2 * n - 1),
                      bounds=[(0.0, None)] * n, method='highs')
    except Exception:
        return None
    if res is None or not getattr(res, 'success', False) or res.x is None:
        return None
    q = np.maximum(res.x, 0)
    sq = q.sum()
    if sq <= 0:
        return None
    return q / sq


def adam_optimizer(p_init, duration, rng, tau0=0.04, lr0=0.015, K=5):
    """Adam optimizer with hybrid gradient and periodic strategy switching."""
    p = normalize_positive(p_init)
    if p is None:
        return None, np.inf
    n = p.size
    L = next_pow2(2 * n - 1)
    deadline = time.monotonic() + max(0.0, duration)
    
    z = np.log(np.maximum(p, 1e-30))
    z = np.where(np.isfinite(z), z, -50.0)
    z -= z.mean()
    z = np.clip(z, -50.0, 50.0)
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    bias_step = 0
    lr = lr0
    tau = tau0
    best_p = p.copy()
    best_score = fast_score(p, L)
    no_improve = 0
    use_topk = True
    step = 0
    
    while time.monotonic() < deadline:
        c = compute_conv(p, L)
        true_s = float(2.0 * n * c.max())
        
        if true_s < best_score - 1e-10:
            best_score = true_s
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive strategy
        if step % 2000 == 0:
            if no_improve > 500:
                use_topk = not use_topk
                tau = max(0.005, tau * 0.9)
                lr = max(0.001, lr * 0.9)
            elif no_improve > 3000:
                # Restart with noise
                z = np.log(np.maximum(best_p, 1e-300))
                z = z + rng.normal(0, 0.1, n)
                z -= z.mean()
                z = np.clip(z, -50, 50)
                m *= 0.5
                v *= 0.5
                no_improve = 0
                lr = max(0.003, lr0 * 0.5)
                tau = max(0.008, tau0 * 0.7)
        
        # Compute gradient
        if use_topk:
            grad_z, _ = gradient_topk(p, c, K=K, L=L)
        else:
            grad_z, _ = gradient_hybrid(p, c, tau=tau, L=L)
        
        # Adam update
        bias_step += 1
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z * grad_z)
        m_hat = m / (1 - beta1 ** bias_step)
        v_hat = v / (1 - beta2 ** bias_step)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        z = np.where(np.isfinite(z), z, 0.0)
        z -= z.mean()
        z = np.clip(z, -50.0, 50.0)
        p = softmax(z)
        step += 1
        
        if step % 10000 == 0:
            print(f"  adam step {step} score {true_s:.6f} best {best_score:.6f} topk={use_topk}")
    
    return best_p, best_score


def waterfilling_optimizer(p_init, duration, rng, K=10):
    """Direct subgradient (water-filling) optimizer targeting top-K peaks."""
    p = normalize_positive(p_init)
    if p is None:
        return None, np.inf
    n = p.size
    L = next_pow2(2 * n - 1)
    deadline = time.monotonic() + max(0.0, duration)
    
    best_p = p.copy()
    best_score = fast_score(p, L)
    momentum = np.zeros(n)
    mu = 0.9  # momentum coefficient
    base_lr = 0.02
    
    step = 0
    while time.monotonic() < deadline:
        c = compute_conv(p, L)
        true_s = float(2.0 * n * c.max())
        
        if true_s < best_score - 1e-10:
            best_score = true_s
            best_p = p.copy()
        
        # Top-K subgradient
        minK = min(K, 2 * n - 1)
        top_idx = np.argpartition(c, -minK)[-minK:]
        top_c = c[top_idx]
        w = top_c / top_c.sum()
        
        # Gradient direction in p-space
        p_flipped = p[::-1]
        L2 = next_pow2(3 * n - 2)
        w_pad = np.zeros(L2)
        w_pad[:2 * n - 1] = w
        W = np.fft.rfft(w_pad)
        Pf = np.fft.rfft(p_flipped)
        h_full = np.fft.irfft(W * Pf, L2)
        h = h_full[:n]
        
        grad_p = 2.0 * h
        # Project to sum=0 direction
        grad_p = grad_p - p * float(np.dot(p, grad_p))
        
        # Nesterov momentum
        momentum = mu * momentum + grad_p
        eff_grad = momentum
        
        # Adaptive step size
        gnorm = float(np.linalg.norm(eff_grad))
        if gnorm < 1e-15:
            break
        lr = base_lr / (gnorm + 1e-10)
        lr = min(lr, 0.1)
        
        # Update in logit space for stability
        z = np.log(np.maximum(p, 1e-300))
        z -= lr * eff_grad
        z -= z.mean()
        z = np.clip(z, -50, 50)
        p = softmax(z)
        step += 1
        
        if step % 5000 == 0:
            print(f"  wf step {step} score {true_s:.6f} best {best_score:.6f}")
        
        # Periodic restart
        if step % 10000 == 0 and true_s > best_score * 1.01:
            p = normalize_positive(best_p * np.exp(rng.normal(0, 0.05, n)))
            if p is None:
                p = best_p.copy()
            momentum *= 0.1
    
    return best_p, best_score


def spectral_init(n, rng):
    """Initialize with random spectral phase, projected to non-negative."""
    L = next_pow2(2 * n - 1)
    P = np.zeros(L // 2 + 1, dtype=np.complex128)
    P[0] = 1.0
    for k in range(1, L // 2 + 1):
        P[k] = np.exp(1j * rng.uniform(0, 2 * np.pi))
    p = np.fft.irfft(P, L)[:n]
    p = np.maximum(p, 0)
    s = p.sum()
    if s <= 0:
        return np.ones(n) / n
    return p / s


def diverse_init(n, rng, D=12):
    """Generate diverse initialization candidates."""
    B = cheb_basis(n, D)
    candidates = []
    
    # Uniform
    candidates.append(np.ones(n) / n)
    
    # Linear ramp (antisymmetric around center)
    x = np.linspace(0, 1, n)
    candidates.append(normalize_positive(1 + 0.5 * (2 * x - 1)))
    candidates.append(normalize_positive(1 - 0.5 * (2 * x - 1)))
    
    # Quadratic
    candidates.append(normalize_positive(1 + 0.3 * (2 * x - 1)**2))
    candidates.append(normalize_positive(1 - 0.2 * (2 * x - 1)**2))
    
    # Cosine-based (antisymmetric)
    for freq in [1, 2, 3, 5, 8]:
        theta = np.zeros(D)
        theta[min(freq - 1, D - 1)] = 0.8 * (1 if freq % 2 == 1 else -1)
        candidates.append(poly_parent(B, theta, 0, 0.15))
        theta2 = np.zeros(D)
        theta2[min(freq - 1, D - 1)] = -0.8 * (1 if freq % 2 == 1 else -1)
        candidates.append(poly_parent(B, theta2, 0, 0.15))
    
    # Random Chebyshev
    for _ in range(20):
        theta = rng.normal(0, 0.4, D)
        theta[0] = rng.choice([-1, 1]) * rng.uniform(0.3, 1.0)
        for j in range(2, D):
            theta[j] *= 0.7
        candidates.append(poly_parent(B, theta, 0, 0.12))
        candidates.append(poly_parent(B, theta * 0.5, 1, 0.12))
    
    # Spectral
    for _ in range(5):
        candidates.append(spectral_init(n, rng))
    
    # Filter valid
    valid = []
    for c in candidates:
        if c is not None and c.size == n and float(c.sum()) > 0:
            valid.append(normalize_positive(c))
    return valid


def compare_update(p, s, best_p, best_s):
    if p is None or not np.isfinite(s):
        return best_p, best_s
    if best_p is None or s < best_s:
        return p, float(s)
    return best_p, best_s


def propose_candidate(seed=42, budget_s=1800.0, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()

    # Choose primary n
    if float(budget_s) < 300:
        N = 2048
    elif float(budget_s) < 900:
        N = 4096
    else:
        N = 8192
    
    best_p = None
    best_s = np.inf

    try:
        hs = np.asarray(height_sequence_1, dtype=np.float64)

        hs = np.maximum(hs, 0.0)
        if hs.size > 0 and hs.sum() > 1e-9:
            pN = resize_sequence(hs, N)
            sN = fast_score(pN)
            best_p, best_s = compare_update(pN, sN, best_p, best_s)
            print(f"Previous best resized: {sN:.6f}")
    except Exception:
        pass
    
    # Generate diverse initializations
    inits = diverse_init(N, rng)
    for init in inits:
        s = fast_score(init)
        best_p, best_s = compare_update(init, s, best_p, best_s)

        best_p = np.ones(N) / N
        best_s = 2.0

    N_small = min(512, N)
    
    # Phase 1: Water-filling (fast, aggressive)

    # Phase 2: Adam refinement

    if rem > 120:

        best_p, best_s = compare_update(p_adam, s_adam, best_p, best_s)

    # Phase 3: LP-based Frank-Wolfe on smaller n, then upscale

    if HAVE_LINPROG and rem > 180:

        lp_deadline = time.monotonic() + dur
        n_lp = 256

        n = n_lp

        while time.monotonic() < lp_deadline and no_improve < 10:

            try:
                res = linprog(obj, A_ub=A, b_ub=np.ones(2 * n - 1),
                              bounds=[(0, None)] * n, method='highs')
            except Exception:
                break
            if not res.success or res.x is None:
                break

            sq = q.sum()
            if sq <= 0:
                break

            d = 0.6 * q + 0.4 * f_lp

    print(f"Final score: {final_score:.6f}, length={len(final_seq)}")
    return final_seq
