def fft_conv(a, b=None):

    n = len(a)
    w_len = len(weights)  # 2n-1
    
    a_rev = a[::-1]
    w_rev = weights[::-1]
    
    size = 1

    W = np.fft.rfft(weights, size)

def fft_conv(a, b=None):
    if b is None:
        b = a
    n1, n2 = len(a), len(b)
    size = 1
    while size < n1 + n2 - 1:
        size <<= 1
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    return np.fft.irfft(A * B, size)[:n1 + n2 - 1]

def softmax_score_grad(a, tau):
    n = len(a)
    c = fft_conv(a)

    exp_c = np.exp(c_s - c_max)
