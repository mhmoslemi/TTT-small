```python
import numpy as np
import time
from scipy.optimize import minimize as _minimize

def _pow2(m):
    m = int(max(1, m))
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()

def _clean(v):
    v = np.asarray(v, dtype=np.float64)
    return np.maximum(np.where(np.isfinite(v), v, 0.0), 0.0)

def _project_simplex(v):
    v = _clean(v)
    n = v.size
    if n == 0:
        return v.copy()
    x = np.sort(v)[::-1]
    css = np.cumsum(x)
    idx = np.arange(1, n+1, dtype=np.float64)
    cond = x > (css - 1.0) / idx
    if not np.any(cond):
        return np.full(n, 1.0/n)
    r = int(np.nonzero(cond)[0][-1])
    tau = (css[r] - 1.0) / (r + 1)
    out = v - tau
    np.maximum(out, 0.0, out=out)
    s = out.sum()
    if s <= 0 or not np.isfinite(s):
        return np.full(n, 1.0/n)
    return out / s

def _score(a, L=None):
    v = _clean(a)
    n = v.size
    if n == 0:
        return np.inf
    s = v.sum()
    if s < 1e-12:
        return np.inf
    v = v / s
    if L is None:
        L = _pow2(2*n - 1)
    A = np.fft.rfft(v, L)
    c = np.fft.irfft(A*A, L)[:2*n-1]
    mx = c.max()
    return 2.0 * n * mx

def _conv(a, L=None):

    n = v.size
    if L is None:
        L = _pow2(2*n - 1)
    A = np.fft.rfft(v, L)
    c = np.fft.irfft(A*A, L)[:2*n-1]
    return np.maximum(c, 0.0)
```

But wait, I need the gradient w.r.t. the log-ratio parameters, not w.r.t. $a$ directly. The chain rule: if $a = \phi(x)$ where $x$ is the log-ratio vector, then:

$\frac{\partial f}{\partial x_i} = \sum_j \frac{\partial f}{\partial a_j} \frac{\partial a_j}{\partial x_i}$

The Jacobian of $\phi$ is:
- $a_j = \frac{e^{x_j}}{\sum_{j'} e^{x_{j'}} + 1}$ for $j < n-1$
- $a_{n-1} = \frac{1}{\sum_{j'} e^{x_{j'}} + 1}$

$\frac{\partial a_j}{\partial x_i} = a_j (\delta_{ij} - a_i)$ for $i, j < n-1$
$\frac{\partial a_{n-1}}{\partial x_i} = -a_{n-1} a_i$ for $i < n-1$

So: $\frac{\partial f}{\partial x_i} = \sum_{j=0}^{n-2} g_j a_j (\delta_{ij} - a_i) + g_{n-1}(-a_{n-1} a_i)$
$= a_i (g_i - \sum_j g_j a_j)$ for $i < n-1$

where $g_j = \partial f / \partial a_j$.

This is the standard simplex gradient projection. Good.

Hmm, the subgradient step is a bit tricky. Let me think about this more carefully.

We want to minimize $f(a) = \max_k c_k$ subject to $\sum a = 1, a \geq 0$.

The subgradient at the current point (where $c_{k^*}$ is the unique max) is $g = \nabla c_{k^*} = 2 S_{k^*} a$.

We take a step: $a_{new} = \text{proj}_{\text{simplex}}(a - \eta g)$.

The step size $\eta$ should be scaled appropriately. A natural choice is $\eta = f(a) / \|g\|^2$ (from the subgradient method analysis).

Actually, let me simplify. The key idea is: reduce the elements of $a$ that contribute most to the peak $c_{k^*}$. The peak $c_{k^*} = \sum_i a_i a_{k^*-i}$. To reduce it, we should reduce pairs $(a_i, a_{k^*-i})$ with large products.

A simpler heuristic:
- Find the top $t$ values of $a_i a_{k^*-i}$ for $i = 0, ..., n-1$.
- Scale down those $a_i$ (and $a_{k^*-i}$) slightly.
- Redistribute the mass to elements that don't contribute to the peak.

Let me implement this as `_peak_reduction`:

```python
def _peak_reduction(a, n_reduce=20, frac=0.1):
    """Reduce the peak of the convolution by scaling down contributing elements."""
    n = a.size
    L = _pow2(2 * n - 1)
    c = _convolve_full(a, L)
    k_star = int(c.argmax())
    
    # Products a_i * a[k_star - i] for valid i
    valid = np.arange(max(0, k_star - n + 1), min(k_star, n - 1) + 1)
    if len(valid) == 0:
        return a
    products = a[valid] * a[k_star - valid]
    
    # Find top contributors
    top_idx = np.argpartition(-products, min(n_reduce, len(valid)))[:n_reduce]
    top_j = valid[top_idx]
    
    # Scale down those elements
    a_new = a.copy()
    mass_removed = 0.0
    for j in top_j:
        old = a_new[j]
        a_new[j] *= (1.0 - frac)
        mass_removed += old - a_new[j]
    
    # Also scale down a[k_star - j] if different
    for j in top_j:
        j2 = k_star - j
        if 0 <= j2 < n and j2 != j:
            old = a_new[j2]
            a_new[j2] *= (1.0 - frac * 0.5)
            mass_removed += old - a_new[j2]
    
    # Redistribute mass to elements with smallest contribution
    # (or just normalize)
    a_new = _project_simplex(a_new)
    return a_new
```

```python
    n = v.size

    idx = np.arange(1, n + 1, dtype=np.float64)
```

```python
"""
Step-function height sequence optimizer.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 via multi-stage gradient optimization.

Stages:
  1. Diverse candidate generation (power-law, Gaussian, sinusoidal).
  2. L-BFGS polishing with L2 then top-peaks softmax objectives.
  3. Convolution-aware local search with multi-scale perturbations.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize as _spm
except Exception:
    _spm = None


# --- utilities ---

def _p2(m):
    """Next power of 2 >= m."""
    m = max(1, int(m))
    return 1 if m <= 1 else 1 << (m - 1).bit_length()


def _cl(v):
    """Clean: finite, non-negative."""
    v = np.asarray(v, dtype=np.float64).copy()
    v[~np.isfinite(v)] = 0.0
    return np.maximum(v, 0.0)


def _pj(v):
    """Project onto probability simplex."""
    v = _cl(v)
    n = v.size
    if n == 0:
        return v
    x = np.sort(v)[::-1]
    cs = np.cumsum(x)
    ix = np.arange(1, n + 1, dtype=np.float64)
    cn = x > (cs - 1.0) / ix
    if not cn.any():
        return np.full(n, 1.0 / n)
    r = int(np.nonzero(cn)[0][-1])
    tau = (cs[r] - 1.0) / (r + 1)
    o = np.maximum(v - tau, 0.0)
    s = o.sum()
    if s < 1e-15:
        return np.full(n, 1.0 / n)
    return o / s


def _score(a, L=None):
    v = _cl(a)
    n = v.size
    s = v.sum()
    if n == 0 or s < 1e-12:
        return np.inf
    v /= s
    if L is None:
        L = _p2(2 * n - 1)
    A = np.fft.rfft(v, L)
    c = np.fft.irfft(A * A, L)[:2 * n - 1]
    mx = float(c.max())
    return 2.0 * n * mx if mx > 0 else 0.0


def _cv(a, L=None):
    v = _cl(a)
    n = v.size
    if L is None:
        L = _p2(2 * n - 1)
    A = np.fft.rfft(v, L)
    return np.maximum(np.fft.irfft(A * A, L)[:2 * n - 1], 0.0)


def _tp(a):
    """a -> log-ratio params."""
    a = _cl(a).copy()
    n = a.size
    s = a.sum()
    if s < 1e-300:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    r = a[-1]
    if r < 1e-300:
        a[-1] = 1e-12
        a /= a.sum()
        r = a[-1]
    return np.clip(np.log(np.maximum(a[:-1], 1e-300) / r), -80.0, 80.0)


def _fp(x):
    """log-ratio params -> a."""
    x = np.where(np.isfinite(x), x, 0.0)
    n = x.size + 1
    mx = x.max() if x.size else 0.0
    e = np.exp(np.clip(x - mx, -700.0, 700.0))
    el = np.exp(-mx)
    s = e.sum() + el
    if s < 1e-300:
        return np.full(n, 1.0 / n)
    a = np.empty(n)
    a[:-1] = e / s
    a[-1] = el / s
    return a


# --- objectives (return (f, grad) for L-BFGS) ---

def _og_l2(x):
    a = _fp(x)
    n = a.size
    L = _p2(3 * n - 2)
    A = np.fft.rfft(a, L)
    c = np.maximum(np.fft.irfft(A * A, L)[:2 * n - 1], 0.0)
    f = float(c @ c)
    W = np.fft.rfft(c, L)
    ga = 4.0 * np.fft.irfft(W * np.conj(A), L)[:n]
    gy = a * (ga - float(a @ ga))
    gy = gy[:-1]
    if not np.all(np.isfinite(gy)):
        gy = np.zeros(n - 1)
    return (f if np.isfinite(f) else 1e300), gy


def _og_peaks(x, m, beta):
    a = _fp(x)
    n = a.size
    c = _cv(a)
    m = min(m, len(c))
    idx = np.argpartition(c, -m)[-m:] if m < len(c) else np.arange(len(c))
    ct = c[idx]
    cm = float(ct.max())
    z = np.clip((ct - cm) * beta, -700.0, 0.0)
    e = np.exp(z)
    w = e / e.sum()
    f = cm + float(np.log(e.sum())) / beta
    
    ga = np.zeros(n)
    for k, wk in zip(idx, w):
        k = int(k)
        j0 = max(0, k - n + 1)
        j1 = min(k, n - 1)
        if j1 >= j0:
            j = np.arange(j0, j1 + 1)
            ga[j] += 2.0 * wk * a[k - j]
    
    gy = a * (ga - float(a @ ga))
    gy = gy[:-1]
    if not np.all(np.isfinite(gy)):
        gy = np.zeros(n - 1)
    return (f if np.isfinite(f) else 1e300), gy


def _og_soft(x, eps):
    a = _fp(x)
    n = a.size
    L = _p2(3 * n - 2)
    A = np.fft.rfft(a, L)
    c = np.maximum(np.fft.irfft(A * A, L)[:2 * n - 1], 0.0)
    cm = float(c.max())
    if cm < 1e-300:
        return 0.0, np.zeros(n - 1)
    z = np.clip((c - cm) / eps, -700.0, 0.0)
    e = np.exp(z)
    w = e / e.sum()
    f = cm + eps * float(np.log(e.sum()))
    W = np.fft.rfft(w, L)
    ga = 2.0 * np.fft.irfft(W * np.conj(A), L)[:n]
    gy = a * (ga - float(a @ ga))
    gy = gy[:-1]
    if not np.all(np.isfinite(gy)):
        gy = np.zeros(n - 1)
    return f, gy
```

Actually, I want to reconsider the objective more carefully. The goal is to minimize $\max_k c_k$. The log-sum-exp over all positions is a good smooth approximation, but it gives equal weight to positions that are far below the max. That's wasteful.

A better approximation: $\max_k c_k \approx c_{k^*} + \max_{k \neq k^*} (c_k - c_{k^*})$ where the second term is bounded. Or even simpler: just minimize $c_{k^*}$ where $k^* = \arg\max_k c_k$. This is the subgradient approach.

The issue with just minimizing $c_{k^*}$: after reducing $c_{k^*}$, a different position becomes the max, and we need to shift focus. This is exactly what the outer loop (recomputing $k^*$) does.

So the algorithm is:
1. Compute $c = a*a$, find $k^*$.
2. Minimize $c_{k^*}$ using L-BFGS (with the constraint that $\sum a = 1, a \geq 0$).
3. Recompute $c$, find new $k^*$.
4. Repeat.

This is a sequence of constrained QPs (minimize a quadratic subject to linear constraints). L-BFGS handles the smooth approximation.

The gradient of $c_{k^*}$ is $2 S_{k^*} a$, which I compute directly.

This is even simpler than the top-m approach! Let me use this.

```python
    # c[k] = sum_i a_i a[k-i]
    j0 = max(0, k - n + 1)
    j1 = min(k, n - 1)
    if j1 >= j0:
        j = np.arange(j0, j1 + 1)
        f = float(a[j] @ a[k - j])
        # Gradient: 2 * a[k-j] for j in [j0, j1]
        ga = np.zeros(n)
        ga[j] = 2.0 * a[k - j]
    else:
        f = 0.0
        ga = np.zeros(n)
    
    gy = a * (ga - float(a @ ga))
    gy = gy[:-1]
    if not np.all(np.isfinite(gy)):
        gy = np.zeros(n - 1)
    return f, gy
```
