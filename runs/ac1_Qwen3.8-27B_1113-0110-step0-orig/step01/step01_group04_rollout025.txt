def _pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _clean(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    return np.maximum(v, 0.0)


def _project_simplex(v):
    v = _clean(v)
    n = v.size
    if n == 0:
        return v.copy()
    x = np.sort(v)[::-1]
    css = np.cumsum(x)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = x > (css - 1.0) / idx
    if not np.any(cond):
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    r = int(np.nonzero(cond)[0][-1])
    tau = (css[r] - 1.0) / float(r + 1)
    out = v - tau
    np.maximum(out, 0.0, out=out)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    out *= 1.0 / s
    return out


def _score(a, L=None):
    v = _clean(a)
    n = v.size
    if n == 0:
        return np.inf
    s = float(v.sum())
    if s < 1e-12:
        return np.inf
    v = v / s
    if L is None:
        L = _pow2(2 * n - 1)
    A = np.fft.rfft(v, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    mx = float(c.max()) if c.size else 0.0

def _ratios_from_a(a):

def _lp_obj_grad(x, p, L):
    """Objective: sum(c^p) where c = a*a; gradient wrt log-ratio params."""
    a = _from_ratios(x)
    n = a.size

    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    pp = float(p)
    if pp == 2.0:
        f = float(np.dot(c, c))
        w = c
    else:
        csafe = np.maximum(c, 1e-300)
        f = float(np.sum(np.power(csafe, pp)))
        w = np.power(csafe, pp - 1.0)
    W = np.fft.rfft(w, n=L)
    ga = (2.0 * pp) * np.fft.irfft(W * np.conj(A), n=L)[:n]

def _softmax_obj_grad(x, eps, L):
    """Smoothed max: eps*logsumexp(c/eps) - cmax + cmax = cmax + eps*logsumexp((c-cmax)/eps)."""
    a = _from_ratios(x)
    n = a.size
    if n <= 1:
        return 0.0, np.zeros_like(x)
    A = np.fft.rfft(a, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    cmax = float(c.max())
    if cmax <= 1e-300:
        return 0.0, np.zeros_like(x)
    eps = max(float(eps), 1e-10)
    z = (c - cmax) / eps
    z = np.clip(z, -745.0, 0.0)
    e = np.exp(z)
    sume = float(e.sum())
    if not np.isfinite(sume) or sume <= 0.0:
        return 1e300, np.zeros_like(x)
    f = cmax + eps * float(np.log(sume))
    w = e / sume
    W = np.fft.rfft(w, n=L)
    ga = 2.0 * np.fft.irfft(W * np.conj(A), n=L)[:n]
    dot = float(np.dot(a, ga))
    gfull = a * (ga - dot)
    gy = gfull[:-1]
    if not np.all(np.isfinite(gy)):
        gy = np.zeros_like(x)
    return f, gy


def _run_lbfgs(fun, x, args, maxiter, bounds, L):
    """Run one L-BFGS-B call."""
    if _minimize is None:
        return x, 99
    x = np.asarray(x, dtype=np.float64)
    if x.size == 0:
        return x, 1
    if bounds is None:
        bounds = [(-100.0, 100.0)] * int(x.size)

    def wrapped(xw, *a):
        return fun(xw, *a, L)

    options = {
        "maxiter": int(maxiter),
        "disp": 0,
        "maxfun": int(max(300, 15 * maxiter)),
        "ftol": 1e-20,
        "gtol": 1e-14,
    }
    try:
        res = _minimize(wrapped, x, args=args, method="L-BFGS-B",
                        jac=True, bounds=bounds, options=options)
    except Exception:
        return x, 99
    x_new = np.asarray(res.x, dtype=np.float64)
    x_new = np.where(np.isfinite(x_new), x_new, 0.0)
    x_new = np.clip(x_new, -100.0, 100.0)
    status = 0
    try:
        status = int(res.status)
    except Exception:
        status = 0
    return x_new, status

        mx = float(c.max())
        # Find top-K active lags
        topk = min(20, len(c))
        top_idx = np.argpartition(c, -topk)[-topk:]
        # Build a "pressure" vector: how much each a[j] contributes to the peak
        pressure = np.zeros(n, dtype=np.float64)
        for k in top_idx:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                pressure[lo:hi + 1] += c[k] * best[k - lo:k - lo + (hi - lo + 1)]

        # Normalize pressure
        pmax = pressure.max()
        if pmax > 0:
            pressure = pressure / pmax

        # Strategy 1: directional move (reduce high-pressure, increase low-pressure)
        direction = pressure - np.mean(pressure)
        # Random block perturbation mixed with direction
        move = rng.normal(0.0, sigma, size=n) + 0.5 * sigma * direction
        cand = _project_simplex(best * np.exp(move))
        sc = _score(cand, L)

        if np.isfinite(sc) and sc < best_score - 1e-14:
            best = cand
            best_score = sc
            last_best = time.monotonic()
            stale = 0
            sigma = min(0.30, sigma * 1.04)
        else:
            # Strategy 2: pure random block
            cand2 = best.copy()
            w = int(np.exp(rng.uniform(np.log(2.0), np.log(max(3, n // 6)))))
            w = max(1, min(w, n))
            s = int(rng.integers(0, n - w + 1))
            cand2[s:s + w] *= float(np.exp(rng.normal(0.0, sigma * 0.7)))
            cand2 = _project_simplex(cand2)
            sc2 = _score(cand2, L)
            if np.isfinite(sc2) and sc2 < best_score - 1e-14:
                best = cand2
                best_score = sc2
                last_best = time.monotonic()
                stale = 0
            else:
                stale += 1
                if stale % 50 == 0:
                    sigma = max(0.01, sigma * 0.85)
                if time.monotonic() - last_best > 30.0:
                    # Re-center
                    cand3 = _project_simplex(best * np.exp(rng.normal(0.0, 0.02, size=n)))
                    sc3 = _score(cand3, L)
                    if np.isfinite(sc3) and sc3 < best_score:
                        best = cand3
                        best_score = sc3
                        last_best = time.monotonic()
                        stale = 0
                    sigma = max(0.015, sigma * 0.7)

    return best, best_score


def _subgradient_polish(a0, end_time, seed):
    """
    Projected subgradient / Frank-Wolfe hybrid for the non-smooth max objective.
    Uses the active set (top lags) to construct a "feature" vector and does
    multiplicative updates.
    """
    a = _project_simplex(a0)
    n = a.size
    L = _pow2(2 * n - 1)
    best = a.copy()
    best_score = _score(best, L)
    rng = np.random.default_rng(seed)
    lr = 0.05
    last_best = time.monotonic()

    while time.monotonic() < end_time:
        c, _ = _convolve_norm(a, L)
        mx = float(c.max())
        if mx < 1e-15:
            break
        # Top active lags
        k_active = int(np.argmax(c))
        # Subgradient direction
        lo = max(0, k_active - n + 1)
        hi = min(n - 1, k_active)
        sg = np.zeros(n, dtype=np.float64)
        sg[lo:hi + 1] = 2.0 * a[k_active - lo:k_active - lo + (hi - lo + 1)]
        # Center to preserve sum
        sg -= float(np.dot(a, sg))
        # Multiplicative update
        step = lr * sg / (np.abs(sg).max() + 1e-8)
        step = np.clip(step, -0.5, 0.5)
        a_new = _project_simplex(a * np.exp(step))
        sc = _score(a_new, L)
        if np.isfinite(sc) and sc < best_score - 1e-14:
            best = a_new.copy()
            best_score = sc
            a = a_new
            last_best = time.monotonic()
            lr = min(0.15, lr * 1.03)
        else:
            lr = max(0.005, lr * 0.95)
            if time.monotonic() - last_best > 15.0:
                # Try a random perturbation
                a = _project_simplex(best * np.exp(rng.normal(0.0, 0.03, size=n)))
                lr = 0.05

    return best, best_score


def _multi_start_refine(a0, seed, end_time):
    """
    Full multi-stage refinement of a starting point.
    """
    a = _project_simplex(a0)
    n = a.size
    L = _pow2(2 * n - 1)
    Lobj = _pow2(3 * n - 2)
    best = a.copy()
    best_score = _score(best, L)

    start = time.monotonic()
    total = max(1.0, end_time - start)

    if total < 5.0:
        return best, best_score

    x = _ratios_from_a(a)
    bounds = [(-100.0, 100.0)] * max(0, n - 1)
    rng = np.random.default_rng(seed)

    if _minimize is None:
        # Fallback: just local search
        return _peak_shave_local_search(a, end_time, seed)

    # Stage 1: L2 (very short)
    t_frac = 0.06
    stage_end = start + t_frac * total
    while time.monotonic() < stage_end:
        x, status = _run_lbfgs(_lp_obj_grad, x, (2.0,), 100, bounds, Lobj)
        if status == 1:
            break
    a = _from_ratios(x)
    sc = _score(a, L)
    if np.isfinite(sc) and sc < best_score:
        best = a.copy()
        best_score = sc

    # Stage 2: L4
    t_frac2 = 0.10
    stage_end = start + t_frac2 * total
    if time.monotonic() < stage_end:
        while time.monotonic() < stage_end:
            x, status = _run_lbfgs(_lp_obj_grad, x, (4.0,), 100, bounds, Lobj)
            if status == 1:
                break
        a = _from_ratios(x)
        sc = _score(a, L)
        if np.isfinite(sc) and sc < best_score:
            best = a.copy()
            best_score = sc

    # Stage 3: L8
    t_frac3 = 0.12
    stage_end = start + t_frac3 * total
    if time.monotonic() < stage_end:
        while time.monotonic() < stage_end:
            x, status = _run_lbfgs(_lp_obj_grad, x, (8.0,), 80, bounds, Lobj)
            if status == 1:
                break
        a = _from_ratios(x)
        sc = _score(a, L)
        if np.isfinite(sc) and sc < best_score:
            best = a.copy()
            best_score = sc

    # Stage 4: L16
    t_frac4 = 0.12
    stage_end = start + t_frac4 * total
    if time.monotonic() < stage_end:
        while time.monotonic() < stage_end:
            x, status = _run_lbfgs(_lp_obj_grad, x, (16.0,), 60, bounds, Lobj)
            if status == 1:
                break
        a = _from_ratios(x)
        sc = _score(a, L)
        if np.isfinite(sc) and sc < best_score:
            best = a.copy()
            best_score = sc

    # Stage 5: Soft-max with adaptive eps
    # Reserve time for stages 6,7
    local_time = max(20.0, 0.20 * total)
    subgrad_time = max(10.0, 0.10 * total)
    soft_end = end_time - local_time - subgrad_time

    if time.monotonic() < soft_end:
        # Initialize eps based on current "spread" of top conv values
        c, _ = _convolve_norm(best, L)
        c_sorted = np.sort(c)[::-1]
        top10 = c_sorted[:min(10, len(c_sorted))]
        spread = float(top10[0] - top10[-1]) if len(top10) > 1 else c_sorted[0]
        eps = max(1e-6, spread * 0.3 / float(n))
        eps_min = 1e-9
        no_improve = 0
        last_improve = time.monotonic()

        while time.monotonic() < soft_end:
            # Run L-BFGS on soft-max
            chunk_iters = 50 + int(30 * (eps / max(eps, 1e-9)))
            x, status = _run_lbfgs(_softmax_obj_grad, x, (eps,), chunk_iters, bounds, Lobj)
            a = _from_ratios(x)
            sc = _score(a, L)
            now = time.monotonic()

            if np.isfinite(sc) and sc < best_score - 1e-14:
                best_score = sc
                best = a.copy()
                last_improve = now
                no_improve = 0
                # Decrease eps more aggressively
                eps = max(eps_min, eps * 0.75)
            else:
                no_improve += 1
                if no_improve >= 3:
                    # Perturb in log-ratio space
                    x = np.clip(x + rng.normal(0.0, 0.02, size=max(1, x.size)), -100.0, 100.0)
                    no_improve = 0
                eps = max(eps_min, eps * 0.5)

            if eps <= eps_min * 1.001 and no_improve >= 4:
                break

    # Stage 6: Subgradient polish
    if time.monotonic() < end_time - local_time:
        sub_end = end_time - local_time
        a, sc = _subgradient_polish(best, sub_end, seed + 7777)
        if np.isfinite(sc) and sc < best_score:
            best = a.copy()
            best_score = sc

    # Stage 7: Peak-shaving local search
    if time.monotonic() < end_time:
        a, sc = _peak_shave_local_search(best, end_time, seed + 8888)
        if np.isfinite(sc) and sc < best_score:
            best = a.copy()
            best_score = sc

    return best, best_score


def _power_seq(n, p, offset):
    j = np.arange(n, dtype=np.float64)
    x = np.maximum(j + float(offset), 1e-10)
    return _project_simplex(np.power(x, -float(p)))


def _generate_starts(n, seed):
    """Generate diverse starting candidates."""
    cands = []
    rng = np.random.default_rng(seed)

    # Try to use the parent solution
    try:
        h = height_sequence_1
        if isinstance(h, (list, np.ndarray)):
            arr = np.asarray(h, dtype=np.float64)
            arr = _clean(arr)
            if arr.size == n and arr.sum() > 0:
                cands.append(_project_simplex(arr))
                # Slight perturbations of parent
                for scale in (0.01, 0.03, 0.05, 0.10):
                    cands.append(_project_simplex(arr * np.exp(rng.normal(0, scale, n))))
    except Exception:
        pass

    # Pure power laws
    for p in (0.48, 0.49, 0.50, 0.51, 0.52, 0.55):
        cands.append(_power_seq(n, p, 1.0))
    cands.append(_power_seq(n, 0.5, 0.5))
    cands.append(_power_seq(n, 0.5, 1.5))

    # Two-sided hybrid (mostly left-heavy)
    j = np.arange(n, dtype=np.float64)
    left = np.power(np.maximum(j + 1.0, 1e-10), -0.5)
    right = np.power(np.maximum(n - 1.0 - j + 1.0, 1e-10), -0.5)
    for mix in (0.01, 0.03, 0.05, 0.10):
        cands.append(_project_simplex(left + mix * right))

    # Exponential decay
    for tau in (0.1, 0.2, 0.3, 0.5):
        v = np.exp(-j / (tau * n))
        cands.append(_project_simplex(v))

    # Power * exponential
    for p in (0.3, 0.4, 0.5):
        for tau in (0.3, 0.5, 1.0):
            v = np.power(np.maximum(j + 1.0, 1e-10), -p) * np.exp(-j / (tau * n))
            cands.append(_project_simplex(v))

    return cands


def propose_candidate(seed=42, budget_s=2392, **kwargs):
    """
    Main entry: search for best sequence within time budget.
    Returns a list of non-negative floats.
    """
    t0 = time.monotonic()
    deadline = t0 + max(10.0, float(budget_s) - 8.0)

    # Determine n
    n = 3809
    try:
        h = height_sequence_1
        if isinstance(h, (list, np.ndarray)):
            arr = np.asarray(h, dtype=np.float64)
            if arr.ndim == 1 and 100 <= arr.size <= 20000:
                n = int(arr.size)
    except Exception:
        pass
    if n < 32:
        n = 3809

    # Very short time: return a decent power law
    if deadline - t0 < 10.0:
        a = _power_seq(n, 0.5, 1.0)
        return [float(x) for x in a]

    print(f"n={n}, budget={deadline - t0:.1f}s")

    # Generate and score candidates
    cands = _generate_starts(n, int(seed))
    scored = []
    L = _pow2(2 * n - 1)
    for i, c in enumerate(cands):
        sc = _score(c, L)
        if np.isfinite(sc):
            scored.append((sc, i, c))
    scored.sort()

    if not scored:
        a = np.full(n, 1.0 / n, dtype=np.float64)
        return [float(x) for x in a]

    global_best_a = np.asarray(scored[0][2], dtype=np.float64).copy()
    global_best_s = float(scored[0][0])
    print(f"Initial best score: {global_best_s:.6f}")

    # Allocate time to top starting points
    top_k = min(3, len(scored))
    # Give more time to the best start
    fracs = [0.60, 0.25, 0.15][:top_k]

    used_frac = 0.0
    for idx in range(top_k):
        cur_time = t0 + used_frac * (deadline - t0)
        alloc_end = t0 + (used_frac + fracs[idx]) * (deadline - t0)
        if alloc_end > cur_time + 5.0:
            a0 = np.asarray(scored[idx][2], dtype=np.float64).copy()
            print(f"  Refining start {idx} (score {scored[idx][0]:.6f}), "
                  f"time {cur_time - t0:.0f}s to {alloc_end - t0:.0f}s")
            a, sc = _multi_start_refine(a0, int(seed) + 100 * (idx + 1), alloc_end)
            if np.isfinite(sc) and sc < global_best_s:
                global_best_s = sc
                global_best_a = np.asarray(a, dtype=np.float64).copy()
                print(f"  -> improved to {sc:.6f}")
            else:
                print(f"  -> no improvement (got {sc:.6f})")
        used_frac += fracs[idx]

    # Final global polish with remaining time
    if time.monotonic() < deadline:
        remaining_end = deadline
        print(f"  Final polish from {time.monotonic() - t0:.0f}s to {deadline - t0:.0f}s")
        a, sc = _peak_shave_local_search(global_best_a, remaining_end, int(seed) + 99999)
        if np.isfinite(sc) and sc < global_best_s:
            global_best_s = sc
            global_best_a = np.asarray(a, dtype=np.float64).copy()
            print(f"  -> final polish improved to {sc:.6f}")

    global_best_a = _project_simplex(global_best_a)

    # Sanity check
    out = [float(x) for x in global_best_a]
    try:
        val = evaluate_sequence(out)
        print(f"  Final evaluated score: {val:.6f}")
        if not np.isfinite(val):
            out = [float(x) for x in _power_seq(n, 0.5, 1.0)]
    except Exception:
        pass

    return out
