def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def normalize_positive(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, float(floor), None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:

    z = np.log(np.maximum(best_a, 1e-15))
    z -= z.mean()

    for p in p_list:
        if time.monotonic() > deadline:
            break
        try:
            res = sp_opt.minimize(
                power_mean_val,
                z.copy(),
                method='L-BFGS-B',
                jac=power_mean_grad,
                args=(n, float(p), N),
                options={'maxiter': max_iter_per_p, 'ftol': 1e-14, 'gtol': 1e-8, 'maxls': 50}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                a_try = softmax_seq(res.x)
                sc = score_of(a_try, N)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_try.copy()
                    z = res.x.copy()
                    print("  PM-LBFGS p={} score={:.9f}".format(p, best_score), flush=True)
        except Exception:
            pass
    return best_a, best_score


def optimize_mirror_v2(a0, deadline, seed=0):
    """Improved mirror descent with adaptive restarts and zero enforcement."""
    a = normalize_positive(a0, floor=1e-15)
    n = a.size
    N = next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = score_of(a, N)

    rng = np.random.default_rng(seed)
    eta = 0.5
    beta = 0.15
    w_floor = 0.05
    stagnant = 0
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        b, fa, N2 = convolve_fft(a, N)
        cmax = float(np.max(b))
        score = float(2.0 * n * cmax)
        if score < best_score * (1.0 - 1e-10):
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if cmax <= 0.0 or not np.isfinite(cmax):
            break

        tau = beta * cmax
        z = (b - cmax) / tau
        z = np.clip(z, -50.0, 0.0)
        ez = np.exp(z)
        s_ez = float(ez.sum())
        if not np.isfinite(s_ez) or s_ez <= 0.0:
            break
        w = ez / s_ez
        if w_floor > 0:
            w = (1.0 - w_floor) * w + w_floor / (2 * n - 1)

        g = grad_conv_weighted(a, w, N2)
        g = g - float(np.mean(g))
        scale = float(np.abs(g).max())
        if not np.isfinite(scale) or scale <= 1e-15:
            break
        g = g / scale

        loga = np.log(a + 1e-20)
        accepted = False
        for e in (eta, 0.5 * eta, 0.25 * eta, 0.1 * eta):
            if e <= 1e-9:
                continue
            loga_try = loga - e * g
            loga_try -= float(np.max(loga_try))
            a_try = np.exp(np.clip(loga_try, -70.0, 0.0))
            a_try = np.maximum(a_try, 1e-300)
            a_try = a_try / a_try.sum()
            sc_try = score_of(a_try, N)
            if np.isfinite(sc_try) and sc_try < score * (1.0 - 1e-10):
                accepted = True
                a = a_try
                loga = np.log(a + 1e-20)
                break

        if accepted:
            stagnant = 0
            eta = min(1.5, eta * 1.03)
            beta = max(0.005, beta * 0.999)
        else:
            stagnant += 1
            eta = max(1e-4, eta * 0.6)
            beta = max(0.003, beta * 0.8)

        if stagnant % 60 == 0:
            # Adaptive restart: partially randomize
            noise = rng.normal(0.0, 0.15, size=n)
            a_try = a * np.exp(noise)
            a_try = normalize_positive(a_try, floor=1e-15)
            sc_try = score_of(a_try, N)
            if np.isfinite(sc_try) and sc_try < best_score * 1.1:
                a = a_try
            stagnant = 0
            eta = 0.4
            beta = 0.12

        # Zero enforcement heuristic: try setting small values to exactly 0
        if it % 200 == 100:
            threshold = float(np.percentile(a, 5))
            a_zero = a.copy()
            a_zero[a_zero < threshold] = 0.0
            a_zero = normalize_positive(a_zero, floor=0.0)
            sc_zero = score_of(a_zero, N)
            if np.isfinite(sc_zero) and sc_zero < best_score:
                best_score = sc_zero
                best_a = a_zero.copy()
                a = a_zero.copy()

        if no_improve > 2000 and stagnant > 500:
            break

        it += 1
        if it % 500 == 0:
            print("  mirror2 iter={} score={:.9f}".format(it, best_score), flush=True)

    return best_a, best_score


def optimize_direct_subgrad(a0, deadline, seed=1):
    """Direct subgradient descent on the max function with adaptive steps."""
    a = normalize_positive(a0, floor=1e-15)
    n = a.size
    N = next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = score_of(a, N)

    rng = np.random.default_rng(seed)
    step = 0.1 / n
    decay = 0.9997
    it = 0
    no_imp = 0

    while time.monotonic() < deadline:
        b, fa, N2 = convolve_fft(a, N)
        cmax = float(np.max(b))
        score = float(2.0 * n * cmax)
        if score < best_score * (1.0 - 1e-10):
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Subgradient: indicator of argmax positions
        peak_idx = np.nonzero(np.abs(b - cmax) < 1e-10 * cmax)[0]
        u = np.zeros(2 * n - 1, dtype=np.float64)
        u[peak_idx] = 1.0 / len(peak_idx)

        g = grad_conv_weighted(a, u, N2)
        g = g - float(np.mean(g))
        gscale = float(np.linalg.norm(g))
        if gscale < 1e-15:
            break
        g = g / gscale

        # Mirror step with non-negativity
        a_try = a - step * g
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            a_try = a.copy()
        else:
            a_try = a_try / s

        # Check if improved
        sc_try = score_of(a_try, N)
        if np.isfinite(sc_try) and sc_try < score:
            a = a_try
            step = min(0.5, step * 1.02)
        else:
            step = max(1e-6 / n, step * 0.7)
            # Occasional random kick
            if no_imp % 100 == 0:
                kick = rng.normal(0.0, 0.05, size=n)
                a_try2 = a * np.exp(kick)
                a_try2 = normalize_positive(a_try2, floor=1e-15)
                sc2 = score_of(a_try2, N)
                if np.isfinite(sc2) and sc2 < best_score:
                    a = a_try2
                    best_score = sc2
                    best_a = a.copy()

        if no_imp > 5000:
            break

        it += 1
        if it % 1000 == 0:
            print("  subgrad iter={} score={:.9f}".format(it, best_score), flush=True)

    return best_a, best_score
