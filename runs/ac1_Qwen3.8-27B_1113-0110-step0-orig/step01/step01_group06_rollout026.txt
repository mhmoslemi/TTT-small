import numpy as np

def _pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _fft_conv(a, b=None):
    if b is None:
        b = a
    L = int(a.size) + int(b.size) - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    return np.fft.irfft(fa * fb, n=N).real[:L]


def _grad_weighted(weights, a):
    """Gradient of dot(weights, conv(a,a)) w.r.t. a."""
    n = int(a.size)
    L = int(weights.size)
    N = _pow2(n + L - 1)
    fw = np.fft.rfft(weights, n=N)
    fa = np.fft.rfft(a, n=N)
    corr = np.fft.irfft(np.conj(fa) * fw, n=N)
    return 2.0 * corr[:n]


def _x_to_a(x, n):
    m = n // 2
    t = 1.0 / m
    a = np.empty(n, dtype=np.float64)
    a[:m] = t * x
    a[m:] = t * (1.0 - x[::-1])
    return a

def _make_topk_weights(b, k, temperature=0.5):
    """Create weights that heavily focus on top-k convolution elements."""
    L = int(b.size)
    maxb = float(b.max())
    if maxb < 1e-300:

    k = min(k, L)

    top_k = sorted_idx[:k]
    
    # Give more weight to the very top elements
    vals = b[top_k]
    if vals.max() - vals.min() < 1e-15:
        p[top_k] = 1.0 / k
    else:
        t = (vals - vals.min()) / (vals.max() - vals.min())
        w = np.exp(t * temperature)
        w /= w.sum()
        p[top_k] = w
    
    # Small uniform background
    eps = 0.01
    p = (1.0 - eps) * p + eps / L
    return p


def _adam_step(m, v, grad, lr, t, beta1=0.9, beta2=0.999, eps=1e-8):
    m[:] = beta1 * m + (1 - beta1) * grad
    v[:] = beta2 * v + (1 - beta2) * (grad * grad)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    return lr * m_hat / (np.sqrt(v_hat) + eps)


def _greedy_init(n, rng):

def _smooth_init(n, rng, order=3):

def _project_to_antipal(given_a):

    # x[i] = a[i] * n / 2

                    rng4 = np.random.default_rng((seed * 999 + 500) % (2**32))
                    try:
                        bx, bs, ba = _coordinate_descent(ba, n_fin, alloc4, rng4, label="coord")
                        if bs < best_score:
                            best_score = bs
                            best_a = ba.copy()
                            print(f"  Coordinate descent improved: {bs:.6f}")
                    except Exception as e:
                        print(f"  Error: {e}")
                        
                        # Also try short anti-pal optimization
                        remaining4 = final_deadline - time.monotonic() - 30
                        if remaining4 > 20:
                            try:
                                bx, bs, ba = _optimize_antipal(x_fin, n_fin, remaining4, rng4, label="anti-final")
                                if bs < best_score:
                                    best_score = bs
                                    best_a = ba.copy()
                                    print(f"  Final anti-pal improved: {bs:.6f}")
                            except Exception as e2:
                                print(f"  Error: {e2}")
            else:
                alloc4 = min(remaining3 - 30, 150.0)
                if alloc4 > 20:
                    rng4 = np.random.default_rng((seed * 999 + 500) % (2**32))
                    try:
                        ca, cs = _optimize_general(best_a, alloc4, rng4, label="gen-final")
                        if cs < best_score:
                            best_score = cs
                            best_a = ca.copy()
                            print(f"  Final general improved: {cs:.6f}")
                    except Exception as e:
                        print(f"  Error: {e}")
    
    # --- Finalize ---
    if best_a is None:
        best_a = np.ones(8192, dtype=np.float64) / 8192.0
        best_score = 2.0
    
    best_a = np.nan_to_num(best_a, nan=0.0, posinf=1000.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    ss = float(best_a.sum())
    if ss < 0.01:
        best_a = np.ones(8192, dtype=np.float64) / 8192.0
        ss = 1.0
    else:
        best_a = best_a / ss
    
    final_seq = [float(x) for x in best_a]
    
    # Verify with actual evaluator if possible
    exact = _safe_evaluate(final_seq)
    print(f"\nFinal: internal={best_score:.6f}, exact={exact:.6f}, n={best_a.size}")
    
    return final_seq
