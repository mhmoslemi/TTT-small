```python
"""
Search for a non-negative step-function height sequence minimizing
    2*n*max(convolve(a,a)) / (sum(a))**2.

The code builds a large pool of analytic and algebraically inspired seeds:
  * power/tapered power sequences,
  * center-weighted and two-tailed sequences,
  * central-binomial "square-root-of-geometry" sequences,
  * polynomial square-root sequences constructed from chosen first-half
    convolution profiles,
  * random multiplicative perturbations.

The best seeds are then polished with:
  * L-BFGS-B on smooth surrogates (quoted sums and log-sum-exp),
  * mirror-descent / hot-coefficient multiplicative local search.

The implementation uses FFT-based exact convolution for internal scoring and
falls back to mirror search if scipy is unavailable.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize as _scipy_minimize
except Exception:
    _scipy_minimize = None

_L_CACHE = {}


def _pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def _get_fft_len(n):
    n = int(n)
    if n <= 0:
        return 1
    val = _L_CACHE.get(n)
    if val is None:
        val = _pow2(2 * n - 1)
        _L_CACHE[n] = val
    return int(val)


def _clean_nonnegative(v):
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        v = np.ravel(v)
    v = np.where(np.isfinite(v), v, 0.0)
    return np.maximum(v, 0.0)


def _normalize(v, target=1.0):
    v = _clean_nonnegative(v)
    n = v.size
    if n == 0:
        return v
    s = float(v.sum())
    if not np.isfinite(s) or s <= 1e-15:
        mx = float(v.max()) if v.size else 0.0
        if mx > 0.0 and np.isfinite(mx):
            v = v / mx
            s = float(v.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.full(n, float(target) / float(n), dtype=np.float64)
    return (v / s) * float(target)


def _convolution_and_transform(a, L=None):
    v = _clean_nonnegative(a)
    n = v.size
    if n == 0:
        return np.zeros(0, dtype=np.float64), np.zeros(0, dtype=np.float64)
    if L is None or int(L) < 2 * n - 1:
        L = _get_fft_len(n)
    A = np.fft.rfft(v, n=int(L))
    c = np.fft.irfft(A * A, n=int(L))[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    return A, c


def _score_normalized(a, L=None):
    v = _clean_nonnegative(a)
    n = v.size
    if n == 0:
        return np.inf
    s = float(v.sum())
    if not np.isfinite(s) or s <= 1e-15:
        v = np.full(n, 1.0 / float(n), dtype=np.float64)
    else:
        v = v / s
    _, c = _convolution_and_transform(v, L)
    if c.size == 0:
        return 0.0
    mx = float(c.max())
    if not np.isfinite(mx) or mx < 0.0:
        mx = 0.0
    return float(2.0 * n * mx)


def _score(a):
    v = _clean_nonnegative(a)
    n = v.size
    if n == 0:
        return np.inf
    s = float(v.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    _, c = _convolution_and_transform(v)
    if c.size == 0:
        return 0.0
    mx = float(c.max())
    if not np.isfinite(mx) or mx < 0.0:
        mx = 0.0
    return float(2.0 * n * mx / (s * s))


def _a_from_x(x):
    x = np.asarray(x, dtype=np.float64)
    if x.ndim != 1:
        x = np.ravel(x)
    x = np.where(np.isfinite(x), x, 0.0)
    n = x.size + 1
    if n == 1:
        return np.ones(1, dtype=np.float64)
    m = 0.0
    if x.size:
        mm = float(np.max(x))
        if np.isfinite(mm) and mm > 0.0:
            m = mm
    e = np.exp(np.clip(x - m, -745.0, 709.0))
    e_last = float(np.exp(-m))
    s = float(e.sum()) + e_last
    if not np.isfinite(s) or s <= 0.0:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    a = np.empty(n, dtype=np.float64)
    a[:-1] = e / s
    a[-1] = e_last / s
    return a


def _x_from_a(a):
    a = _clean_nonnegative(a).astype(np.float64)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    a = _normalize(a, 1.0)
    if n == 1:
        return np.zeros(0, dtype=np.float64)

    mx = float(a.max())
    if not np.isfinite(mx) or mx <= 0.0:
        minval = 1e-18
    else:
        minval = max(1e-18, 1e-14 * mx)
    a = np.maximum(a, minval)
    a = _normalize(a, 1.0)

    ref = float(a[-1])
    if ref <= 1e-300:
        a[-1] = 1e-14
        a = _normalize(a, 1.0)
        ref = float(a[-1])

    x = np.log(np.maximum(a[:-1], 1e-300) / max(ref, 1e-300))
    x = np.where(np.isfinite(x), x, 0.0)
    return np.clip(x, -50.0, 50.0)


def _fun_q(x, p, ref):
    if x.size == 0:
        return 0.0, np.zeros(0, dtype=np.float64)
    a = _a_from_x(x)
    n = a.size
    L = _get_fft_len(n)
    A, b = _convolution_and_transform(a, L)
    if b.size == 0:
        return 0.0, np.zeros(max(0, n - 1), dtype=np.float64)
    if not np.isfinite(ref) or ref <= 1e-300:
        ref = 1.0
    c = np.maximum(b, 0.0) / float(ref)
    c = np.clip(c, 0.0, 1e4)
    p = float(p)
    f = float(np.sum(np.power(c, p)))
    if p > 1.0:
        w = np.power(c, p - 1.0)
    else:
        w = np.ones_like(c)
    w = np.clip(w, 0.0, 1e6)
    W = np.fft.rfft(w, n=L)
    ga = float(2.0 * p / float(ref)) * np.fft.irfft(W * np.conj(A), n=L)[:n]
    ga = np.where(np.isfinite(ga), ga, 0.0)
    dot = float(np.dot(a, ga))
    g = a * (ga - dot)
    g = g[:-1]
    if not np.all(np.isfinite(g)):
        g = np.zeros(max(0, n - 1), dtype=np.float64)
    if not np.isfinite(f):
        f = 1e100
    return f, g


def _fun_soft(x, temp, ref):
    if x.size == 0:
        return 0.0, np.zeros(0, dtype=np.float64)
    a = _a_from_x(x)
    n = a.size
    L = _get_fft_len(n)
    A, b = _convolution_and_transform(a, L)
    if b.size == 0:
        return 0.0, np.zeros(max(0, n - 1), dtype=np.float64)
    if not np.isfinite(ref) or ref <= 1e-300:
        ref = 1.0
    if not np.isfinite(temp) or temp <= 1e-15:
        temp = 1e-15
    c = np.maximum(b, 0.0) / float(ref)
    cmax = float(c.max())
    if not np.isfinite(cmax) or cmax <= 0.0:
        return 0.0, np.zeros(max(0, n - 1), dtype=np.float64)
    z = (c - cmax) / float(temp)
    z = np.clip(z, -745.0, 0.0)
    e = np.exp(z)
    sume = float(e.sum())
    if not np.isfinite(sume) or sume <= 0.0:
        f = 1e100
        g = np.zeros(max(0, n - 1), dtype=np.float64)
        return f, g
    f = float(cmax + float(temp) * np.log(sume))
    w = e / sume
    W = np.fft.rfft(w, n=L)
    ga = float(2.0 / float(ref)) * np.fft.irfft(W * np.conj(A), n=L)[:n]
    ga = np.where(np.isfinite(ga), ga, 0.0)
    dot = float(np.dot(a, ga))
    g = a * (ga - dot)
    g = g[:-1]
    if not np.all(np.isfinite(g)):
        g = np.zeros(max(0, n - 1), dtype=np.float64)
    if not np.isfinite(f):
        f = 1e100
    return f, g


def _lbfgs_chunk(fun, x, args, maxiter, bounds):
    if _scipy_minimize is None:
        return np.asarray(x, dtype=np.float64), 999
    x = np.asarray(x, dtype=np.float64)
    if x.size == 0:
        return x, 1
    options = {
        "maxiter": int(maxiter),
        "disp": 0,
        "maxfun": int(max(300, 20 * int(maxiter))),
        "ftol": 1e-18,
        "gtol": 1e-7,
        "maxls": 20,
        "maxcor": 16,
    }
    try:
        res = _scipy_minimize(
            fun,
            x,
            args=args,
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options=options,
        )
    except Exception:
        return x, 999
    x_new = np.asarray(res.x, dtype=np.float64)
    x_new = np.where(np.isfinite(x_new), x_new, 0.0)
    x_new = np.clip(x_new, -50.0, 50.0)
    try:
        status = int(res.status)
    except Exception:
        status = 1
    return x_new, status


def _binomial_sequence(n):
    n = int(n)
    if n <= 0:
        return np.zeros(0, dtype=np.float64)
    d = np.empty(n, dtype=np.float64)
    d[0] = 1.0
    for i in range(n - 1):
        d[i + 1] = d[i] * float(2 * i + 1) / float(2 * (i + 1))
    return _normalize(d, 1.0)


def _power_candidate(n, p, offset, boost=0.0, gamma=1.0):
    n = int(n)
    if n <= 0:
        return np.zeros(0, dtype=np.float64)
    j = np.arange(n, dtype=np.float64)
    x = j + max(float(offset), 1e-8)
    v = np.power(np.maximum(x, 1e-12), -float(p))
    if abs(float(boost)) > 1e-15 and n > 1:
        t = j / float(n - 1)
        v = v * (1.0 + float(boost) * np.power(t, float(gamma)))
    return _clean_nonnegative(v)


def _two_tail_candidate(n, p, offset, back_weight=0.0):
    n = int(n)
    if n <= 0:
        return np.zeros(0, dtype=np.float64)
    j = np.arange(n, dtype=np.float64)
    front = np.power(j + max(float(offset), 1e-8), -float(p))
    if abs(float(back_weight)) > 1e-15:
        back = np.power((n - 1.0 - j) + max(float(offset), 1e-8), -float(p))
        front = front + float(back_weight) * back
    return _clean_nonnegative(front)


def _add_candidate(cands, v):
    try:
        a = _normalize(v, 1.0)
        if a.size > 0 and np.all(np.isfinite(a)):
            cands.append(a)
    except Exception:
        pass


def _recursive_sqrt_from_profile(n, b_profile):
    n = int(n)
    b_profile = np.asarray(b_profile, dtype=np.float64)
    if n <= 0 or b_profile.size != n:
        return None
    if not np.all(np.isfinite(b_profile)) or np.any(b_profile < 0):
        return None
    a0 = float(np.sqrt(float(b_profile[0])))
    if not np.isfinite(a0) or a0 <= 1e-15:
        return None
    a = np.zeros(n, dtype=np.float64)
    a[0] = a0
    scale = max(1.0, abs(float(b_profile[0])))
    for k in range(1, n):
        if k > 1:
            s = float(np.dot(a[1:k], a[k - 1:0:-1]))
        else:
            s = 0.0
        val = float(b_profile[k]) - s
        if val < -1e-7 * scale:
            return None
        if val < 0.0:
            val = 0.0
        a[k] = val / (2.0 * a0)
        if not np.isfinite(a[k]):
            return None
    return a


def _add_profile_candidates(cands, n):
    n = int(n)
    if n < 200 or n > 4500:
        return
    j = np.arange(n, dtype=np.float64)
    t = j / float(n - 1)
    exp_profile = np.exp(-9.0 * (t - 0.5) * (t - 0.5))
    for mode in range(4):
        for alpha in (-0.2, -0.1, 0.1, 0.2):
            b = np.ones(n, dtype=np.float64)
            if mode == 0:
                b *= 1.0 + alpha * (t - 0.5)
            elif mode == 1:
                b *= 1.0 + alpha * np.cos(2.0 * np.pi * t)
            elif mode == 2:
                b *= 1.0 + alpha * np.cos(4.0 * np.pi * t)
            else:
                b *= 1.0 - alpha * exp_profile
            if not np.all(np.isfinite(b)) or np.any(b <= 0.0):
                continue
            a = _recursive_sqrt_from_profile(n, b)
            if a is not None:
                _add_candidate(cands, a)


def _quick_candidates(n, seed=98765, rich=True):
    n = int(n)
    cands = []
    if n <= 0:
        return cands

    ones = np.ones(n, dtype=np.float64)
    _add_candidate(cands, ones)

    if rich:
        p_list = (0.35, 0.40, 0.45, 0.50, 0.55, 0.60)
        off_list = (0.05, 0.5, 1.0, 2.0)
    else:
        p_list = (0.45, 0.50, 0.60)
        off_list = (0.5, 1.0)

    for p in p_list:
        for off in off_list:
            _add_candidate(cands, _power_candidate(n, p, off))

    if rich:
        boost_list = (-0.9, -0.7, -0.4, -0.1, 0.1, 0.4, 0.8, 1.5)
    else:
        boost_list = (-0.5, 0.0, 0.5, 1.0)
    for p in (0.45, 0.50, 0.55):
        for off in (0.05, 0.5, 1.0):
            for boost in boost_list:
                for gamma in (1.0, 2.0):
                    _add_candidate(
                        cands,
                        _power_candidate(n, p, off, boost, gamma),
                    )

    for p in (0.45, 0.50, 0.55):
        for wb in (0.0, 0.05, 0.15, 0.40):
            _add_candidate(cands, _two_tail_candidate(n, p, 0.5, wb))

    d = _binomial_sequence(n)
    _add_candidate(cands, d)

    j = np.arange(n, dtype=np.float64)
    t = (j + 1.0) / float(n)
    cos1 = np.cos(2.0 * np.pi * j / float(n))
    cos2 = np.cos(4.0 * np.pi * j / float(n))

    for alpha in (-0.4, -0.2, -0.1, 0.1, 0.3):
        _add_candidate(cands, d * (1.0 + alpha * cos1))
        _add_candidate(cands, d * (1.0 + alpha * cos2))

    for alpha in (0.2, 0.5, 0.8):
        _add_candidate(cands, d * (1.0 - alpha * t))
        _add_candidate(cands, d * (1.0 + alpha * t))

    for beta in (0.2, 1.0, 5.0):
        _add_candidate(cands, d + beta * ones)

    rng = np.random.default_rng(seed + int(n))
    nrnd = 8 if rich else 4
    basep = _power_candidate(n, 0.5, 1.0)
    for _ in range(nrnd):
        noise = np.exp(rng.normal(0.0, 0.25, size=n))
        _add_candidate(cands, d * noise)
        _add_candidate(cands, basep * noise)

    _add_profile_candidates(cands, n)
    return cands


def _get_base_n(default=3809):
    n = int(default)
    try:
        h = height_sequence_1  # noqa: F821
        if isinstance(h, (list, np.ndarray)):
            arr = np.asarray(h, dtype=np.float64)
            if arr.ndim == 1 and 100 <= arr.size <= 30000:
                n = int(arr.size)
    except Exception:
        pass
    if n < 64:
        n = int(default)
    return n


def _gradient_refine(a0, end_time, seed, initial_score=None):
    a = _clean_nonnegative(a0)
    n = a.size
    if n == 0:
        return a, np.inf
    a = _normalize(a, 1.0)

    if initial_score is None or not np.isfinite(initial_score):
        best_score = _score_normalized(a)
    else:
        best_score = float(initial_score)
    best = a.copy()
    best_score = float(best_score)

    now = time.monotonic()
    total = float(end_time) - now
    if total <= 3.0 or not np.isfinite(best_score):
        return best, best_score

    if _scipy_minimize is None:
        return _mirror_polish(a, float(end_time), int(seed))

    _, b0 = _convolution_and_transform(a)
    ref = float(b0.max()) if b0.size else 0.0
    if not np.isfinite(ref) or ref <= 1e-15:
        ref = 1.0

    x = _x_from_a(a)
    best_x = x.copy()
    bounds = [(-50.0, 50.0)] * max(0, n - 1)
    rng = np.random.default_rng(int(seed))

    def _stage_q(p, stage_end):
        nonlocal best, best_score, best_x, x
        no_improve = 0
        sigma = 0.02
        while time.monotonic() < stage_end:
            x, status = _lbfgs_chunk(_fun_q, x, (float(p), ref), 180, bounds)
            a_new = _a_from_x(x)
            sc = _score_normalized(a_new)
            if np.isfinite(sc) and sc < best_score - 1e-13:
                best = a_new.copy()
                best_score = float(sc)
                best_x = x.copy()
                no_improve = 0
            else:
                if status == 1:
                    no_improve += 1
                no_improve += 1
                if no_improve >= 2:
                    sigma = min(0.12, sigma * 1.6)
                    x = np.clip(
                        best_x + rng.normal(0.0, sigma, size=max(0, x.size)),
                        -50.0,
                        50.0,
                    )
                    no_improve = 0

    # Stage 1: broad L2 surrogate.
    stage_end = now + 0.18 * total
    if stage_end > now + 2.0:
        _stage_q(2.0, stage_end)

    # Stage 2: peak-biased L3 surrogate.
    stage_end2 = now + 0.38 * total
    if stage_end2 > time.monotonic() + 2.0:
        _stage_q(3.0, stage_end2)

    # Stage 3: smoothed maximum.
    temp = 0.16
    min_temp = 1e-5
    no_improve = 0
    sigma = 0.02
    while time.monotonic() < float(end_time):
        x, status = _lbfgs_chunk(_fun_soft, x, (temp, ref), 180, bounds)
        a_new = _a_from_x(x)
        sc = _score_normalized(a_new)
        if np.isfinite(sc) and sc < best_score - 1e-13:
            best = a_new.copy()
            best_score = float(sc)
            best_x = x.copy()
            no_improve = 0
            temp = max(min_temp, temp * 0.84)
            sigma = max(0.005, sigma * 0.8)
        else:
            if status == 1:
                no_improve += 1
            no_improve += 1
            if no_improve >= 2:
                sigma = min(0.16, sigma * 1.7)
                x = np.clip(
                    best_x + rng.normal(0.0, sigma, size=max(0, x.size)),
                    -50.0,
                    50.0,
                )
                no_improve = 0
                if temp <= min_temp * 1.2:
                    temp = min(0.12, temp * 2.0)
                else:
                    temp = min(0.25, temp * 1.25)
            elif temp <= min_temp * 1.05 and no_improve >= 6:
                break

    return best, best_score


def _mirror_polish(a0, end_time, seed):
    a = _clean_nonnegative(a0)
    n = a.size
    if n == 0:
        return a, np.inf
    a = _normalize(a, 1.0)
    L = _get_fft_len(n)

    best = a.copy()
    best_score = _score_normalized(best, L)
    if not np.isfinite(best_score):
        best_score = 1e100

    rng = np.random.default_rng(int(seed))
    eta = 0.045
    alpha = 0.12
    Ks = (6, 12, 24, 48, 96, 192, 384, 768)
    last_improve = time.monotonic()
    start = last_improve
    it = 0

    while time.monotonic() < float(end_time):
        it += 1
        A, b = _convolution_and_transform(a, L)
        if b.size == 0:
            break
        bmax = float(b.max())
        if not np.isfinite(bmax) or bmax <= 1e-300:
            break
        cur = float(2.0 * n * bmax)
        now = time.monotonic()

        if cur < best_score - 1e-13:
            best = a.copy()
            best_score = cur
            last_improve = now
            eta = min(1.2, eta * 1.08)
            alpha = max(0.006, alpha * 0.94)
        else:
            if now - last_improve > 3.5:
                eta = max(1e-4, eta * 0.78)
            if now - last_improve > 15.0:
                a = best.copy()
                if rng.random() < 0.55:
                    a = _normalize(best * np.exp(rng.normal(0.0, 0.025, size=n)), 1.0)
                eta = 0.05
                alpha = 0.10

        use_top = (it % 4 == 0)
        if use_top:
            kk = Ks[(it // 4) % len(Ks)]
            kk = min(int(kk), b.size)
            if kk <= 0:
                kk = 1
            if kk >= b.size:
                idx = np.arange(b.size)
            else:
                idx = np.argpartition(b, -kk)[-kk:]
            w = np.zeros_like(b)
            w[idx] = 1.0
        else:
            z = (b - bmax) / (alpha * bmax)
            z = np.clip(z, -745.0, 0.0)
            w = np.exp(z)

        ws = float(w.sum())
        if not np.isfinite(ws) or ws <= 0.0:
            w = np.ones_like(b)
            ws = float(b.size)
        w = w / ws
        w = 0.5 * (w + w[::-1])
        ws = float(w.sum())
        if not np.isfinite(ws) or ws <= 0.0:
            w = np.ones_like(b) / float(b.size)
        else:
            w = w / ws

        W = np.fft.rfft(w, n=L)
        H = np.fft.irfft(W * np.conj(A), n=L)[:n]
        H = np.where(np.isfinite(H), H, 0.0)
        g = H - float(H.mean())
        rms = float(np.sqrt(np.mean(g * g)))

        accepted = False
        if rms > 1e-300:
            direction = -g / rms
            for mult in (1.0, 0.45):
                step = eta * mult
                ex = np.clip(step * direction, -30.0, 30.0)
                trial = _normalize(a * np.exp(ex), 1.0)
                sc = _score_normalized(trial, L)
                if np.isfinite(sc) and sc < best_score - 1e-14:
                    a = trial
                    best = trial.copy()
                    best_score = float(sc)
                    last_improve = time.monotonic()
                    accepted = True
                    break

        if accepted:
            eta = min(1.3, eta * 1.06)
            continue

        if it % 120 == 0:
            trial = _normalize(a * np.exp(rng.normal(0.0, 0.008, size=n)), 1.0)
            sc = _score_normalized(trial, L)
            if np.isfinite(sc) and sc < best_score - 1e-14:
                a = trial
                best = trial.copy()
                best_score = float(sc)
                last_improve = time.monotonic()
                continue

        eta = max(1e-4, eta * 0.72)

    return best, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t0 = time.monotonic()
    try:
        budget = max(0.0, float(budget_s) - 25.0)
    except Exception:
        budget = 2375.0
    deadline = t0 + max(0.0, budget)

    base_n = _get_base_n(3809)

    # Construct a small set of sensible lengths.
    raw_ns = [base_n]
    if base_n <= 12000:
        raw_ns.append(min(20000, max(2048, int(round(2.0 * base_n)))))
    if base_n >= 2048:
        raw_ns.append(max(1024, int(round(0.5 * base_n))))
    if base_n < 9000:
        raw_ns.append(min(20000, max(1500, int(round(1.5 * base_n)))))

    n_list = []
    for nn in raw_ns:
        nn = int(nn)
        if 800 <= nn <= 22000 and nn not in n_list:
            n_list.append(nn)

    if not n_list:
        n_list = [max(800, min(22000, base_n))]

    states = []
    counter = 0
    global_score = np.inf
    global_a = None
    global_n = int(n_list[0])

    # Quick generation and scoring.
    for idx, n in enumerate(n_list):
        rich = n <= 9000
        cands = _quick_candidates(n, seed=100003 + 17 * idx, rich=rich)
        # Include the provided state-of-the-art sequence if available and matching n.
        try:
            h = height_sequence_1  # noqa: F821
            if isinstance(h, (list, np.ndarray)):
                arr = np.asarray(h, dtype=np.float64)
                if arr.ndim == 1 and arr.size == n:
                    _add_candidate(cands, arr)
        except Exception:
            pass

        best_state_a = None
        best_state_s = np.inf
        for a in cands:
            sc = _score_normalized(a)
            if np.isfinite(sc) and sc < best_state_s:
                best_state_s = float(sc)
                best_state_a = a.astype(np.float64, copy=True)

        if best_state_a is not None and np.isfinite(best_state_s):
            states.append((float(best_state_s), int(n), int(counter), best_state_a))
            counter += 1
            if best_state_s < global_score:
                global_score = float(best_state_s)
                global_a = best_state_a.copy()
                global_n = int(n)

    if global_a is None:
        a = _binomial_sequence(base_n)
        return [float(x) for x in _normalize(a, 1.0)]

    states.sort()
    print(f"quick-best n={global_n} score={global_score:.10f}", flush=True)

    now = time.monotonic()
    remaining = deadline - now

    # Short low-budget fallback: return best quick candidate.
    if remaining < 120.0:
        out = [float(x) for x in global_a]
        try:
            if "evaluate_sequence" in globals():
                val = evaluate_sequence(out)  # noqa: F821
                if not np.isfinite(val):
                    out = [float(x) for x in _binomial_sequence(len(out))]
        except Exception:
            pass
        return out

    # Very brief preliminary polish of a few promising states.
    if remaining > 360.0 and _scipy_minimize is not None:
        quick_budget = min(75.0, remaining / 7.0)
        for st in states[:3]:
            if deadline - now < quick_budget + 5.0:
                break
            if st[0] > global_score + 0.08:
                break
            end = min(deadline - 5.0, now + quick_budget)
            if end - now < 15.0:
                break
            ra, rs = _gradient_refine(st[3], end, seed + 7919 + st[1], st[0])
            if np.isfinite(rs) and rs < global_score:
                global_score = float(rs)
                global_a = ra.copy()
                global_n = int(st[1])
            now = time.monotonic()

    now = time.monotonic()
    remaining = deadline - now

    # Main polish.
    if remaining > 90.0:
        if _scipy_minimize is not None:
            local_budget = min(420.0, max(60.0, 0.22 * remaining))
            main_end = deadline - local_budget
            if main_end - now > 45.0:
                global_a, global_score = _gradient_refine(
                    global_a,
                    main_end,
                    seed + 104729,
                    global_score,
                )
                print(
                    f"after-gradient n={global_n} score={global_score:.10f}",
                    flush=True,
                )

            if deadline - now > 25.0:
                global_a, global_score = _mirror_polish(
                    global_a,
                    deadline - 5.0,
                    seed + 555667,
                )
                print(
                    f"after-mirror n={global_n} score={global_score:.10f}",
                    flush=True,
                )
        else:
            global_a, global_score = _mirror_polish(
                global_a,
                deadline - 5.0,
                seed + 31337,
            )
            print(
                f"after-mirror-noscipy n={global_n} score={global_score:.10f}",
                flush=True,
            )

    out = [float(x) for x in _normalize(global_a, 1.0)]

    # Final validity check against the official evaluator if available.
    try:
        if "evaluate_sequence" in globals():
            val = evaluate_sequence(out)  # noqa: F821
            if not np.isfinite(val):
                out = [float(x) for x in _binomial_sequence(max(64, len(out)))]
    except Exception:
        pass

    return out
```
