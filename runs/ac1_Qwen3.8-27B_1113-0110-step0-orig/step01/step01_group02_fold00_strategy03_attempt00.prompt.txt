    """

    """
    # --- Security Checks ---

    # Verify that the input is a list

        return np.inf

    # Check each element in the list for validity

        if np.isnan(x) or np.isinf(x):
            return np.inf

    sequence = [float(x) for x in sequence]

    # Protect against negative numbers

    """Return the next power of 2 >= n."""
    if n <= 1:

def _fft_convolve(a):
    """Fast linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]


def _subgradient_vec(a, k_max):
    """Compute subgradient of (a*a)[k_max] w.r.t. each a[j].

    n = len(a)

    if lo <= hi:
        j_range = np.arange(lo, hi + 1)
        indices = k_max - j_range
        g[lo:hi + 1] = 2.0 * a[indices]
