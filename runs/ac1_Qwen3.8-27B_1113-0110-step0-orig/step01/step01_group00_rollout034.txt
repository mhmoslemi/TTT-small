"""
Multi-strategy optimizer for minimizing J(a) = 2n*max(conv(a,a))/sum(a)^2.
Uses soft-max smooth optimization with L-BFGS-B, alternating LP, and subgradient methods
across multiple sequence lengths. Key novelty: FFT-based soft-max gradients with adaptive
temperature scheduling and elite-based local search.

import numpy as np
import time
import math
from scipy.optimize import linprog, minimize


def _fft_convolve(a):
    """Fast full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]


def _fft_convolve_ab(a, b):
    """Fast full linear convolution of a with b using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    c = np.fft.irfft(A * B, fft_size)
    return c[:size]


def _score(a):
    """Compute J(a) = 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)


def _softmax_grad(a, T):
    """
    Compute gradient of soft_max(T*c)/T with respect to a, where c = conv(a,a).
    Uses FFT for O(n log n) computation.
    Returns (loss_value, gradient).
    """
    n = len(a)
    c = _fft_convolve(a)
    # Soft-max computation (numerically stable)
    Tc = T * c
    Tc_max = np.max(Tc)
    exp_Tc = np.exp(Tc - Tc_max)
    sum_exp = np.sum(exp_Tc)
    loss = (Tc_max + np.log(sum_exp)) / T
    # Weights
    w = exp_Tc / sum_exp  # softmax weights over c values
    # Gradient: dL/da[m] = 2 * sum_k w[k] * a[k-m] (for valid k-m in [0,n-1])
    # = 2 * conv(w, a_reversed)[n-1 : 2n-1]
    a_rev = a[::-1]
    conv_w_ar = _fft_convolve_ab(w, a_rev)
    grad = 2.0 * conv_w_ar[n - 1: 2 * n - 1]
    return loss, grad


def _softmax_objective(a, T):
    """Objective function wrapper for L-BFGS-B."""
    loss, grad = _softmax_grad(a, T)
    return loss, grad


def _soft_max_optimize(a_init, T_values, max_iters_per_T, deadline):
    """
    Soft-max optimization with adaptive temperature.
    """
    a = a_init.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(len(a)) / len(a)
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = _score(a)

    for T in T_values:
        if time.monotonic() > deadline:
            break
        # Compute appropriate T scaling based on current max_c
        c = _fft_convolve(a)
        max_c = float(np.max(c))
        # Scale T so that T * max_c is in a good range [5, 50]
        T_scaled = T * max_c
        T_scaled = max(5.0, min(50.0, T_scaled))

        def obj(z):
            loss, g = _softmax_grad(z, T_scaled)
            return loss, g

        try:
            result = minimize(obj, a, jac=True, method='L-BFGS-B',
                              bounds=[(0.0, None)] * len(a),
                              options={'maxiter': max_iters_per_T,
                                       'ftol': 1e-12, 'gtol': 1e-8})
            a_new = np.maximum(result.x, 0.0)
            s_new = float(np.sum(a_new))

                score_new = _score(a_new)

                    best_a = a_new.copy()
                    a = a_new
        except Exception:

    return best_a, best_score


def _subgradient_smooth(a_init, max_iters, deadline, n_peaks=20, lr_init=0.5, lr_decay=0.9995):
    """
    Subgradient method focusing on top n_peaks of the convolution.
    Uses averaging for convergence.
    """
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    best_a = a.copy()
    best_score = _score(a)
    lr = lr_init

    # Polyak step size tracking
    best_lb = 1.0 / (2.0 * n)  # Lower bound on max_c (when S=1)
    best_score_lb = 2.0 * n / (2.0 * n - 1)

    a_avg = a.copy()
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        max_c = float(np.max(c))
        score = 2.0 * n * max_c  # since S=1
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Get top n_peaks
        if n_peaks >= len(c):
            top_k = np.arange(len(c))
        else:
            top_k = np.argpartition(c, -n_peaks)[-n_peaks:]

        # Compute subgradient: 2 * sum_{k in top_k} a[k-m]
        # = 2 * conv(w_one_hot, a_rev)[n-1:2n-1] where w_one_hot has 1 at top_k
        w = np.zeros(len(c))
        w[top_k] = 1.0 / len(top_k)
        a_rev = a[::-1]
        conv_w_ar = _fft_convolve_ab(w, a_rev)
        grad = 2.0 * conv_w_ar[n - 1: 2 * n - 1]

        # Normalize gradient
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm > 1e-15:
            grad = grad / grad_norm

        # Polyak-like step size
        lr_polyak = (score - best_score_lb) / max(grad_norm * grad_norm, 1e-15)
        lr_use = min(lr, lr_polyak)

        a = a - lr_use * grad
        a = np.maximum(a, 0.0)
        S_new = float(np.sum(a))
        if S_new > 1e-10:
            a = a / S_new
        a_avg = a_avg + a
        lr *= lr_decay

    # Also check the averaged solution
    a_avg = a_avg / (it + 1)
    S_avg = float(np.sum(a_avg))
    if S_avg > 1e-10:
        a_avg = a_avg / S_avg
        score_avg = _score(a_avg)
        if score_avg < best_score:
            best_score = score_avg
            best_a = a_avg.copy()

    return best_a, best_score


def _build_conv_matrix(f):
    """Build the convolution matrix C where (C @ g)[k] = (f*g)[k]."""
    n = len(f)
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = f
    return C


def _lp_solve(f, M):
    """Solve max sum(g) s.t. C_f @ g <= M, g >= 0. Returns g or None."""
    n = len(f)
    C = _build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.status == 0:
            return result.x
    except Exception:
        pass
    return None


def _alternating_lp(a_init, max_iters, deadline, initial_blend=0.8):
    """Run alternating LP: fix f, solve LP for g, blend, repeat."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = _score(a)
    blend = initial_blend

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        M = float(np.max(c))
        g = _lp_solve(a, M)
        if g is None:
            break
        sum_g = float(np.sum(g))
        if sum_g < 1e-12:
            break
        g = g / sum_g
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        sum_new = float(np.sum(a_new))
        if sum_new < 1e-12:
            break
        a_new = a_new / sum_new
        score = _score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        blend = max(0.05, blend * 0.98)

    return best_a, best_score

def _interpolate_sequence(src, n_target):
    """Interpolate a sequence to a new length."""
    n_src = len(src)
    if n_src == n_target:
        return src.copy()
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    out = np.interp(x_tgt, x_src, src)
    out = np.maximum(out, 0.0)
    s = float(np.sum(out))
    if s > 1e-10:
        out = out / s
    return out

        # Try perturbing the best solution
        for perturb_scale in [0.01, 0.05, 0.1]:
            if time.monotonic() > total_deadline - 15:
                break
            a_perturbed = best_a + rng.normal(0, perturb_scale / n_best, n_best)
            a_perturbed = np.maximum(a_perturbed, 0.0)
            s_p = float(np.sum(a_perturbed))
            if s_p > 1e-10:
                a_perturbed = a_perturbed / s_p
            # Quick soft-max optimize from perturbed start
            a_opt, score = _soft_max_optimize(a_perturbed, [2.0, 5.0], 100, total_deadline - 15)
            print(f"  perturb={perturb_scale}: score={score:.6f}", flush=True)
            if score < best_score:
                best_score = score
                best_a = a_opt.copy()
                print(f"    ** New best: {best_score:.6f}", flush=True)

    # Final verification
    print(f"\n=== FINAL: score={best_score:.6f}, n={len(best_a) if best_a is not None else 0} ===", flush=True)
    if best_a is not None:
        result = [float(x) for x in best_a]
        verified = float(evaluate_sequence(result))
        print(f"Verified score: {verified:.6f}", flush=True)
        return result

    # Fallback
    return [1.0] * 100
