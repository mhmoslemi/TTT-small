    ca=np.fft.irfft(fa*fa, N)[:size]

"""
Search for non-negative sequences minimizing
    2 * n * max(conv(a, a)) / sum(a)^2
by combining:
1) FFT-accelerated soft-max / top-active multiplicative subgradient descent,
   with simulated-annealing acceptance and random Fourier/roll restarts.
2) Local linearized trust-region LP steps (first-order balancing of active
   convexification constraints).
3) Occasional use of the previous SOTA LP construction:
   max sum(b) s.t. a*b <= M, b>=0, then line-search mixture.

The program schedules multiple initializations, polishes the global best with
local LP steps, and finally applies more gradient polishing.
"""

import numpy as np
import time
import math

try:
    from scipy.optimize import linprog as _linprog
    _HAS_LINPROG = True
except Exception:
    _HAS_LINPROG = False


def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (int(n) - 1).bit_length()


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if (not np.isfinite(s)) or s <= 1e-15:
        n = max(1, len(a))
        return np.ones(n, dtype=np.float64) / float(n)
    a /= s
    return a


def _fft_convolve(a):
    n = len(a)
    if n <= 0:
        return np.array([], dtype=np.float64)
    if n == 1:
        v = float(a[0])
        return np.array([v * v], dtype=np.float64)

    size = 2 * n - 1
    N = _next_pow2(size)
    fa = np.fft.rfft(a, N)
    ca = np.fft.irfft(fa * fa, N)[:size]
    ca = np.ascontiguousarray(ca)
    ca[ca < 0.0] = 0.0
    return ca


def _score_norm(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    if n <= 0:
        return np.inf
    s = float(a.sum())
    if s <= 1e-15:
        return np.inf
    if abs(s - 1.0) > 1e-12:
        a = a / s
    c = _fft_convolve(a)
    return float(2.0 * n * np.max(c))


def _weighted_corr(a, w, N, idx):
    """
    h_j = sum_m a_m * w_{j+m}, computed by circular correlation with a large
    enough FFT so that no wrap-around contaminates j=0..len(a)-1.
    """
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    res = np.fft.irfft(A * np.conjugate(W), N)
    return res[idx]


def _softmax_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    m = float(np.max(c))
    if m <= 1e-300:
        w = np.zeros_like(c)
        if len(w):
            w[int(np.argmax(c))] = 1.0
        return w
    z = c / m - 1.0
    ex = np.exp(beta * z)
    s = float(ex.sum())
    if s <= 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return ex / s


def _add_noise(a, amp, rng):
    n = len(a)
    b = a * np.exp(rng.normal(0.0, amp, n))
    return _normalize(np.maximum(b, 1e-30))


def _fourier_perturb(base, amp, rng, harmonics=4):
    n = len(base)
    x = np.linspace(0.0, 1.0, n, endpoint=False)
    p = np.zeros(n, dtype=np.float64)
    for m in range(1, harmonics + 1):
        p += float(amp) * float(rng.normal()) * np.cos(2.0 * math.pi * m * x)
        p += float(amp) * float(rng.normal()) * np.sin(2.0 * math.pi * m * x)
    b = base * (1.0 + p)
    return _normalize(np.maximum(b, 1e-30))


def _mix_seqs(a, b, alpha):
    a = _normalize(a)
    b = _normalize(b)
    a = np.maximum(a, 1e-30)
    b = np.maximum(b, 1e-30)
    c = np.exp((1.0 - alpha) * np.log(a) + alpha * np.log(b))
    return _normalize(c)


def _make_inits(n, rng, provided=None):
    inits = []

    if provided is not None:
        p = np.asarray(provided, dtype=np.float64).ravel()
        if len(p) != n:
            if len(p) >= 2:
                old_x = np.linspace(0.0, 1.0, len(p), endpoint=False)
                new_x = np.linspace(0.0, 1.0, n, endpoint=False)
                p = np.interp(new_x, old_x, np.maximum(p, 0.0))
            else:
                p = np.maximum(p, 0.0)
        p = np.maximum(p, 0.0)
        s = float(p.sum())
        if s > 1e-12:
            inits.append(p / s)

    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)

    # Left in-square-root edge singular family.
    for pe in (0.25, 0.35, 0.45, 0.50, 0.55, 0.65):
        f = x ** (-pe)
        inits.append(_normalize(f))

    # Two-sided edge singular family.
    for pe in (0.35, 0.45, 0.50, 0.60):
        f = x ** (-pe) + (1.0 - x) ** (-pe)
        inits.append(_normalize(f))

    # U-shaped endpoint families.
    for amp in (0.5, 1.0, 1.8, 3.0):
        f = 1.0 + amp * (2.0 * x - 1.0) ** 2
        inits.append(_normalize(f))

    # Center bump.
    for amp in (0.8, 1.5, 2.5):
        f = 1.0 + amp * (np.sin(math.pi * x) ** 2)
        inits.append(_normalize(f))

    # Sinusoidal perturbations of a constant baseline.
    for amp in (0.25, 0.40, 0.55, 0.70):
        for _ in range(2):
            phase = float(rng.uniform(0.0, 2.0 * math.pi))
            f = 1.0 + amp * np.sin(2.0 * math.pi * x + phase)
            f = np.maximum(f, 0.03)
            inits.append(_normalize(f))

    # Low-order random Fourier shapes.
    for _ in range(4):
        f = np.ones(n, dtype=np.float64)
        for m in range(1, 6):
            f += 0.22 * float(rng.normal()) * np.cos(2.0 * math.pi * m * x)
            f += 0.22 * float(rng.normal()) * np.sin(2.0 * math.pi * m * x)
        f = np.maximum(f, 0.03)
        inits.append(_normalize(f))

    return inits


def _cand_score_key(item):
    return float(item[0])


def _optimize_steep(a0, budget, rng):
    """
    Soft-max / top-active multiplicative subgradient descent with adaptive
    step size and temperature-controlled acceptance.
    """
    start = time.monotonic()
    a = _normalize(a0)
    n = len(a)
    if n < 20 or budget < 1.0:
        return a, _score_norm(a)

    deadline = start + float(budget)

    Ncorr = _next_pow2(3 * n + 5)
    idx_corr = np.arange(n, dtype=np.int64)
    idx_corr = np.where(idx_corr == 0, 0, Ncorr - idx_corr)

    c = _fft_convolve(a)
    cur_score = float(2.0 * n * np.max(c))
    best_score = cur_score
    best_a = a.copy()

    base_eta = 0.14
    beta = 6.0
    T = 0.006
    no_improve = 0
    it = 0
    last_print = start
    p_ema = np.zeros(n, dtype=np.float64)

    top_cycle = (1, 2, 3, 5, 8, 13, 21, 34, 55)
    floor = 1e-18
    n_candidates = 3 if n <= 2500 else 2
    alphas = (0.45, 0.90, 1.80) if n_candidates == 3 else (0.50, 1.10)

    while time.monotonic() < deadline:
        c = _fft_convolve(a)
        cur_score = float(2.0 * n * np.max(c))

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            if time.monotonic() - last_print > 5.0:
                print(f"    [steep] score={best_score:.6f} n={n}")
                last_print = time.monotonic()
        else:
            no_improve += 1

        if cur_score > best_score + 0.035:
            a = _add_noise(best_a, 0.08, rng)
            cur_score = _score_norm(a)
            p_ema = np.zeros(n, dtype=np.float64)
            beta = 5.0
            base_eta = 0.16
            T = 0.009
            no_improve = 0
            continue

        if no_improve > 650:
            r = float(rng.random())
            if r < 0.30:
                a = _add_noise(best_a, 0.11, rng)
            elif r < 0.60:
                m = max(1, n // 6)
                shift = int(rng.integers(-m, m))
                a = np.roll(best_a, shift)
                a = _normalize(a)
            else:
                a = _fourier_perturb(best_a, 0.09, rng, 4)
            cur_score = _score_norm(a)
            p_ema = np.zeros(n, dtype=np.float64)
            beta = max(3.0, beta * 0.55)
            base_eta = 0.17
            T = 0.012
            no_improve = 0
            continue

        # Choose active constraints: either smoothed soft-max or top-K uniform.
        if (it % 4) < 2:
            if no_improve > 220:
                beta = min(beta * 1.045, 95.0)
            elif (it % 25) == 0:
                beta = max(beta * 0.99, 3.0)
            w = _softmax_weights(c, beta)
        else:
            K = top_cycle[it % len(top_cycle)]
            L = len(c)
            K = min(K, L)
            if K <= 0:
                w = _softmax_weights(c, beta)
            elif K == L:
                idxw = np.arange(L)
            else:
                idxw = np.argpartition(c, -K)[-K:]
            w = np.zeros_like(c, dtype=np.float64)
            w[idxw] = 1.0 / float(len(idxw))

        h = _weighted_corr(a, w, Ncorr, idx_corr)
        h = np.maximum(h, 0.0)
        hmax = float(np.max(h))
        if hmax <= 1e-15:
            it += 1
            continue

        p = h / hmax
        p_ema = 0.62 * p_ema + 0.38 * p
        pmax = float(np.max(p_ema))
        if pmax <= 1e-15:
            it += 1
            continue
        p_norm = p_ema / pmax

        best_cand = None
        best_cand_score = np.inf

        for al in alphas:
            a_cand = a * np.exp(-base_eta * al * p_norm)
            a_cand = np.maximum(a_cand, floor)
            ss = float(a_cand.sum())
            if ss <= 1e-15:
                continue
            a_cand /= ss
            cc = _fft_convolve(a_cand)
            sc = float(2.0 * n * np.max(cc))
            if sc < best_cand_score:
                best_cand_score = sc
                best_cand = a_cand

        if best_cand is None:
            it += 1
            continue

        accept = False
        if best_cand_score <= cur_score + T:
            accept = True
        elif best_cand_score < best_score - 1e-12:
            accept = True
        else:
            diff = best_cand_score - cur_score
            if diff < 50.0 * T:
                accept = (rng.random() < math.exp(-diff / max(T, 1e-12)))

        if accept:
            a = best_cand
            cur_score = best_cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                no_improve = 0
                base_eta = min(0.36, base_eta * 1.025)
                T = max(0.00035, T * 0.988)
                if time.monotonic() - last_print > 5.0:
                    print(f"    [steep] score={best_score:.6f} n={n}")
                    last_print = time.monotonic()
        else:
            base_eta = max(0.028, base_eta * 0.965)

        if no_improve > 180:
            base_eta = max(0.030, base_eta * 0.985)
            T = min(0.022, T * 1.015)

        it += 1

    return best_a, best_score


def _lp_step(a, rng, maxM=None):
    """
    Local first-order LP trust-region step.
    Minimize t subject to:
       t >= c0_k + J_k d   for selected k
       sum(d) = 0
       box bounds on d
    then line-search on a + theta*d.
    """
    if not _HAS_LINPROG:
        return _normalize(a), _score_norm(a)

    a = _normalize(a)
    n = len(a)
    if n < 120:
        return a, _score_norm(a)

    c = _fft_convolve(a)
    cur_score = float(2.0 * n * np.max(c))

    L = 2 * n - 1
    if maxM is None:
        maxM = min(900, max(50, int(1400000 // max(1, n))))
    maxM = min(maxM, L)

    M = int(maxM)
    M = max(25, int(M * (0.65 + 0.70 * float(rng.random()))))
    M = min(M, maxM, L)

    if M * (n + 1) > 2800000:
        M = max(25, int(2800000 // (n + 1)))
        if M * (n + 1) > 3500000:
            return a, cur_score

    order = np.argsort(c)[::-1]
    pool = min(L, max(100, 3 * M))
    top_pool = order[:pool]
    top_count = max(1, int(0.7 * M))
    top_count = min(top_count, pool)

    if pool <= M:
        idx = top_pool.copy()
    else:
        extra = M - top_count
        extra = min(max(0, extra), pool - top_count)
        if extra > 0:
            extra_idx = rng.choice(top_pool, size=extra, replace=False)
            idx = np.concatenate((top_pool[:top_count], extra_idx))
        else:
            idx = top_pool[:top_count]

    if len(idx) < M:
        fill = rng.choice(order, size=(M - len(idx)), replace=False)
        idx = np.concatenate((idx, fill))

    idx = idx[:M].astype(np.int64)
    rng.shuffle(idx)

    A = np.zeros((M, n + 1), dtype=np.float64)
    b_ub = np.empty(M, dtype=np.float64)

    for r in range(M):
        k = int(idx[r])
        lo = k - (n - 1)
        if lo < 0:
            lo = 0
        hi = k if k < n else n - 1

        A[r, -1] = -1.0
        if lo <= hi:
            start = k - hi
            end = k - lo + 1
            A[r, lo:hi + 1] = 2.0 * a[start:end][::-1]
        b_ub[r] = c[k]

    cobj = np.zeros(n + 1, dtype=np.float64)
    cobj[-1] = 1.0

    Aeq = np.zeros((1, n + 1), dtype=np.float64)
    Aeq[0, :n] = 1.0

    add_cap = float(rng.uniform(1.0, 4.5)) / float(n)
    lb = -0.5 * a
    ub = np.minimum(add_cap + 0.1 * a, 0.025)

    lb_full = np.empty(n + 1, dtype=np.float64)
    ub_full = np.empty(n + 1, dtype=np.float64)
    lb_full[:n] = lb
    ub_full[:n] = ub
    lb_full[n] = 0.0
    ub_full[n] = cur_score + 1.0

    try:
        res = _linprog(
            cobj,
            A_ub=A,
            b_ub=b_ub,
            A_eq=Aeq,
            b_eq=[0.0],
            bounds=(lb_full, ub_full),
            method="highs",
            options={"presolve": True}
        )
        if (not res.success) or res.x is None:
            return a, cur_score
        d = np.asarray(res.x[:n], dtype=np.float64)
        d -= d.mean()
    except Exception:
        return a, cur_score

    best_cand = a
    best_sc = cur_score
    thetas = (0.05, 0.10, 0.18, 0.30, 0.45, 0.65, 0.90, 1.00)

    for th in thetas:
        cand = a + float(th) * d
        cand = np.maximum(cand, 0.0)
        ss = float(cand.sum())
        if ss <= 1e-15:
            continue
        cand /= ss
        cc = _fft_convolve(cand)
        sc = float(2.0 * n * np.max(cc))
        if sc < best_sc:
            best_sc = sc
            best_cand = cand

    return best_cand, best_sc


def _sota_lp_step(a, rng):
    """
    Previous SOTA convex LP:
        max sum(b) s.t. a*b <= M, b >= 0,
    where M = max(a*a). Then line-search a mixed toward b.
    """
    if not _HAS_LINPROG:
        return _normalize(a), _score_norm(a)

    a = _normalize(a)
    n = len(a)
    if n > 2400:
        return a, _score_norm(a)

    c = _fft_convolve(a)
    M = float(np.max(c))
    cur_score = float(2.0 * n * M)
    if M <= 1e-300:
        return a, cur_score

    L = 2 * n - 1
    try:
        A = np.zeros((L, n), dtype=np.float64)
        for k in range(L):
            lo = k - (n - 1)
            if lo < 0:
                lo = 0
            hi = k if k < n else n - 1
            if lo <= hi:
                start = k - hi
                end = k - lo + 1
                A[k, lo:hi + 1] = a[start:end][::-1]

        cobj = -np.ones(n, dtype=np.float64)
        b_ub = np.full(L, M, dtype=np.float64)
        bounds = [(0.0, None) for _ in range(n)]

        res = _linprog(
            cobj,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True}
        )
        if (not res.success) or res.x is None:
            return a, cur_score
        b = np.asarray(res.x, dtype=np.float64)
        b = np.maximum(b, 0.0)
        sb = float(b.sum())
        if sb <= 1e-15:
            return a, cur_score
        b /= sb
    except Exception:
        return a, cur_score

    best_cand = a
    best_sc = cur_score
    for t in (0.04, 0.08, 0.15, 0.25, 0.38, 0.55, 0.75, 1.00):
        cand = (1.0 - float(t)) * a + float(t) * b
        cand = _normalize(cand)
        cc = _fft_convolve(cand)
        sc = float(2.0 * n * np.max(cc))
        if sc < best_sc:
            best_sc = sc
            best_cand = cand

    return best_cand, best_sc


def _lp_polish(a0, budget, rng):
    if not _HAS_LINPROG or budget < 30.0:
        a = _normalize(a0)
        return a, _score_norm(a)

    deadline = time.monotonic() + float(budget)
    a = _normalize(a0)
    a_score = _score_norm(a)
    best_a = a.copy()
    best_score = a_score

    it = 0
    last_print = time.monotonic()
    last_steep = time.monotonic()
    no_progress = 0
    m_cycle = (80, 160, 280, 450, 650)

    while time.monotonic() < deadline:
        if a_score > best_score + 0.025:
            if rng.random() < 0.55:
                a = _add_noise(best_a, 0.10, rng)
            else:
                a = _fourier_perturb(best_a, 0.12, rng, 4)
            a_score = _score_norm(a)

        if time.monotonic() > deadline - 6.0:
            break

        M = int(m_cycle[it % len(m_cycle)])
        cand, sc = _lp_step(a, rng, M)

        if np.isfinite(sc):
            if sc < best_score:
                best_a = cand
                best_score = sc
                no_progress = 0
                if time.monotonic() - last_print > 5.0:
                    print(f"    [lp] score={best_score:.6f} n={len(best_a)}")
                    last_print = time.monotonic()
                a = best_a.copy()
                a_score = best_score
            elif sc < a_score + 0.004:
                a = cand
                a_score = sc
            else:
                no_progress += 1
                if no_progress > 6 or a_score > best_score + 0.012:
                    a = _mix_seqs(best_a, _fourier_perturb(best_a, 0.22, rng, 5), 0.55)
                    a_score = _score_norm(a)
                    no_progress = 0

        # Periodic gradient polishing to refine sequences found by LP.
        if time.monotonic() - last_steep > 30.0 and time.monotonic() < deadline - 30.0:
            t_budget = min(18.0, max(5.0, deadline - time.monotonic()))
            a, sc = _optimize_steep(a, t_budget, rng)
            if np.isfinite(sc):
                a_score = sc
                if sc < best_score:
                    best_a = a
                    best_score = sc
                    no_progress = 0
                    last_print = time.monotonic()

        last_steep = time.monotonic()
        it += 1

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2350, **kwargs):
    """
    Main search entry point. Returns a list of non-negative floats.
    """
    total_start = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2350.0

    budget = max(120.0, min(budget - 25.0, 2380.0))
    deadline = total_start + budget

    rng = np.random.default_rng(seed)

    # Retrieve any previous best sequence if available.
    prov = globals().get("height_sequence_1")
    prov_norm = None
    if prov is not None:
        try:
            prov_arr = np.asarray(prov, dtype=np.float64).ravel()
            prov_arr = np.maximum(prov_arr, 0.0)
            if prov_arr.size > 1 and float(prov_arr.sum()) > 1e-12:
                prov_norm = _normalize(prov_arr)
        except Exception:
            prov_norm = None

    if prov_norm is not None:
        main_n = int(len(prov_norm))
    else:
        main_n = 2000

    Ns = [main_n, 2000, 3000, 4000]
    if main_n <= 1800:
        Ns.append(3000)
    if main_n > 4200:
        Ns.append(5000)

    Ns = []
    for n in Ns:
        n = int(n)
        if 150 <= n <= 8000 and n not in Ns:
            Ns.append(n)

    print(f"Candidate lengths: {Ns}")

    candidates = []
    for n in Ns:
        provided_for_n = None
        if prov_norm is not None and len(prov_norm) == n:
            provided_for_n = prov_norm

        inits = _make_inits(n, rng, provided_for_n)
        # Limit memory/time for very long lengths.
        if n > 3500 and len(inits) > 16:
            inits = inits[:16]

        for a0 in inits:
            sc = _score_norm(a0)
            if np.isfinite(sc):
                candidates.append((sc, n, a0))

    if not candidates:
        a = np.ones(2000, dtype=np.float64)
        print("No candidates generated; falling back to uniform.")
        return [float(v) for v in a]

    candidates.sort(key=_cand_score_key)
    print(f"Generated {len(candidates)} candidates. Top internal scores:")
    for i in range(min(8, len(candidates))):
        print(f"  {i+1}. n={candidates[i][1]}, score={candidates[i][0]:.6f}")

    best_a = candidates[0][2].copy()
    best_score = np.inf
    try:
        ev = evaluate_sequence([float(x) for x in best_a])
        if np.isfinite(ev):
            best_score = float(ev)
        else:
            best_score = float(candidates[0][0])
    except Exception:
        best_score = float(candidates[0][0])

    print(f"Initial global best: {best_score:.6f} (n={len(best_a)})")

    elapsed = time.monotonic() - total_start
    remaining = max(0.0, deadline - time.monotonic())

    if _HAS_LINPROG:
        steep_time = remaining * 0.50
    else:
        steep_time = remaining * 0.82

    if steep_time < 60.0:
        steep_time = remaining * 0.90

    top_count = min(8, len(candidates))
    weights = []
    for i in range(top_count):
        n_cand = candidates[i][1]
        w = float(max(1, top_count - i)) * ((2000.0 / float(max(1, n_cand))) ** 0.35)
        weights.append(w)
    total_w = sum(weights)

    for i in range(top_count):
        now = time.monotonic()
        if now > deadline - 45.0:
            break
        sc0, n0, a0 = candidates[i]
        t_alloc = steep_time * weights[i] / total_w
        t_alloc = min(t_alloc, max(5.0, deadline - now - 30.0))
        if t_alloc < 8.0:
            continue

        print(f"\nOptimizing candidate i={i}, n={n0}, init={sc0:.6f}, time={t_alloc:.0f}s")
        res_a, res_sc = _optimize_steep(a0, t_alloc, rng)

        if res_a is not None and len(res_a) > 10:
            try:
                ev = evaluate_sequence([float(x) for x in res_a])
            except Exception:
                ev = res_sc
            if np.isfinite(ev) and float(ev) < best_score:
                best_score = float(ev)
                best_a = res_a.copy()
                print(f"  *** NEW BEST: {best_score:.6f} (n={len(best_a)}) ***")

    # SOTA LP from previous state of art, cautiously.
    now = time.monotonic()
    if _HAS_LINPROG and (deadline - now) > 260.0 and len(best_a) > 0:
        nn = len(best_a)
        if nn <= 2050:
            sota_calls = 2
        elif nn <= 2500:
            sota_calls = 1
        else:
            sota_calls = 0

        for _ in range(sota_calls):
            now = time.monotonic()
            if now > deadline - 80.0:
                break
            print("\nSOTA LP step...")
            sa, ss = _sota_lp_step(best_a, rng)
            if sa is not None and len(sa) > 10:
                try:
                    ev = evaluate_sequence([float(x) for x in sa])
                except Exception:
                    ev = ss
                if np.isfinite(ev) and float(ev) < best_score:
                    best_score = float(ev)
                    best_a = sa.copy()
                    print(f"  *** NEW BEST after SOTA LP: {best_score:.6f} ***")

    # Local LP polishing.
    now = time.monotonic()
    if _HAS_LINPROG and (deadline - now) > 120.0 and best_a is not None:
        lp_budget = min((deadline - now) * 0.58, 1300.0)
        lp_budget = max(60.0, lp_budget)
        print(f"\nLocal LP polish: budget={lp_budget:.0f}s")
        pa, ps = _lp_polish(best_a, lp_budget, rng)
        if pa is not None and len(pa) > 10:
            try:
                ev = evaluate_sequence([float(x) for x in pa])
            except Exception:
                ev = ps
            if np.isfinite(ev) and float(ev) < best_score:
                best_score = float(ev)
                best_a = pa.copy()
                print(f"  *** NEW BEST after LP polish: {best_score:.6f} ***")

    # Final gradient polish.
    now = time.monotonic()
    final_budget = max(0.0, deadline - now - 5.0)
    if best_a is not None and final_budget > 25.0:
        print(f"\nFinal gradient polish: {final_budget:.0f}s")
        fa, fs = _optimize_steep(best_a, final_budget * 0.92, rng)
        if fa is not None and len(fa) > 10:
            try:
                ev = evaluate_sequence([float(x) for x in fa])
            except Exception:
                ev = fs
            if np.isfinite(ev) and float(ev) < best_score:
                best_score = float(ev)
                best_a = fa.copy()
                print(f"  *** FINAL NEW BEST: {best_score:.6f} ***")

    if best_a is None or len(best_a) < 20:
        a = np.ones(2000, dtype=np.float64)
        print("Fallback to uniform sequence.")
        return [float(x) for x in a]

    # Final sanitization: ensure list of Python floats, finite, non-negative.
    final = np.asarray(best_a, dtype=np.float64).ravel()
    final = np.maximum(final, 0.0)
    final[~np.isfinite(final)] = 0.0
    s = float(final.sum())
    if s <= 1e-12:
        final = np.ones(2000, dtype=np.float64)
    else:
        final /= s

    try:
        final_ev = evaluate_sequence([float(x) for x in final])
        print(f"\nReturned sequence score: {final_ev:.6f}, length={len(final)}")
    except Exception as e:
        print(f"Final evaluation failed: {e}")

    return [float(x) for x in final]
