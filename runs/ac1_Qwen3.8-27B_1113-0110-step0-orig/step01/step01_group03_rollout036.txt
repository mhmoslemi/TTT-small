2. Nesterov momentum with adaptive learning rate
3. Structured perturbation restarts (sinusoidal, random low-rank)

import numpy as np
import time


def fft_self_conv(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    if n <= 1:
        return a * a
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:2 * n - 1]


def fast_score(a):
    """Fast score computation."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_self_conv(a)
    M = c.max()
    return 2.0 * n * M / (S * S)


def normalize_sum(a, target):
    """Normalize to target sum, keep non-negative."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-10:
        return np.full_like(a, target / len(a))
    return a / S * target


def smooth_gradient(a, c, T):
    """
    Compute gradient of 2n*max(c)/S^2 using smooth max approximation.
    Uses FFT-based cross-correlation for efficiency.
    """
    n = len(a)
    S = a.sum()
    M = c.max()

    # Compute exponential weights
    w = np.exp((c - M) / T)
    w_sum = w.sum()

    # Cross-correlation: cross[j] = sum_k w[k] * a[k-j] (valid range)
    # Using FFT: cross = ifft(FFT(w) * conj(FFT(a)))
    L = 1
    while L < 3 * n + 4:
        L *= 2
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)
    cross = np.fft.irfft(W * np.conj(A), L)
    cross = cross[:n]

    # Gradient: d(score)/d(a_j) = 4n*cross[j]/(S*w_sum) - 4n*M/S^2
    grad = 4.0 * n * cross / (S * w_sum) - 4.0 * n * M / (S * S)
    return grad

def find_top_peaks(c, count=5, min_gap=3):
    """Find top peaks with minimum gap between them."""
    n_c = len(c)
    thresh = c.max() * 0.95
    candidates = np.where(c >= thresh)[0]
    if len(candidates) == 0:
        return [int(np.argmax(c))]
    sorted_idx = candidates[np.argsort(-c[candidates])]
    peaks = []
    for idx in sorted_idx:
        if all(abs(int(idx) - k) > min_gap for k in peaks):
            peaks.append(int(idx))
        if len(peaks) >= count:
            break
    return peaks

        perturbation = np.sin(2 * np.pi * freq * t / n + phase)
        a_new = a * (1.0 + amp * perturbation)
    elif mode == 'block':

        start = rng.integers(0, n - block_len)

        a_new = a.copy()
        a_new[start:start + block_len] *= factor

            p = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.02, 0.15)
            perturb += amp * np.sin(2 * np.pi * f * t / n + p)
        a_new = a * (1.0 + perturb)
    else:
        a_new = a * np.exp(rng.normal(0, 0.1, n))

                    # Restart with structured perturbation
                    mode = rng.choice(['sine', 'block', 'lowrank', 'random'])
                    a = structured_perturb(best_a, rng, mode)
                    a = normalize_sum(a, n)
                    lr = lr_init * 0.5
                    momentum = np.zeros(n)

        if time.monotonic() > deadline - 10:
            break

    return best_a, best_score


def coordinate_descent(a, budget_s, rng):
    """
    Coordinate descent: randomly pick small groups of indices and adjust them.
    """
    n = len(a)
    best_a = a.copy()
    best_score = fast_score(a)
    sc = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(sc) and sc < best_score:
        best_score = sc

    deadline = time.monotonic() + budget_s
    group_size = max(3, n // 200)
    it = 0

    while time.monotonic() < deadline:
        a = normalize_sum(best_a, n).copy()
        S = a.sum()

        # Pick a random group of indices
        idx = rng.integers(0, n, size=group_size)
        pert = rng.normal(1.0, 0.15, size=group_size)
        a_new = a.copy()
        a_new[idx] *= pert
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_sum(a_new, n)

        sc_new = fast_score(a_new)
        if sc_new < best_score - 1e-10:
            sc_exact = float(evaluate_sequence(a_new.tolist()))
            if np.isfinite(sc_exact) and sc_exact < best_score:
                best_a = a_new.copy()
                best_score = sc_exact

        it += 1
        if it % 500 == 0:
            print(f"  CD it={it}: score={best_score:.8f}")

    return best_a, best_score


def try_length(parent, target_n, rng):
    """Resample parent to target length."""
    n_old = len(parent)
    if target_n == n_old:
        return parent.copy()
    # Linear interpolation
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, target_n)
    resampled = np.interp(x_new, x_old, parent)
    return np.maximum(resampled, 0.0)


def propose_candidate(seed=42, **kwargs):
    """
    Multi-phase optimization:
    - Phase 0: Parametric initialization + length exploration
    - Phase 1: Smooth-max gradient descent with Nesterov momentum
    - Phase 2: Multi-restart with structured perturbations
    - Phase 3: Coordinate descent fine-tuning
    """
    start_time = time.monotonic()
    total_budget = 2340.0  # safety margin

    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        parent = np.ones(512, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    if parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(512, dtype=np.float64)

    n = len(parent)
    S_parent = parent.sum()
    if S_parent < 0.01:
        parent = np.ones(n)
        S_parent = n

    # === Phase 0: Initialization ===
    print(f"Phase 0: Initialization, n={n}, parent_score={fast_score(parent):.6f}")

    # Normalize parent
    p = normalize_sum(parent, n)
    best_a = p.copy()
    best_score = fast_score(p)
    sc_p = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(sc_p):
        best_score = sc_p
        best_a = p.copy()

    # Try different lengths
    t = np.arange(n, dtype=np.float64)
    x_unit = 2.0 * t / max(1, n - 1) - 1.0

    lengths_to_try = [n]
    # Try a few nearby lengths
    for dl in [-50, 50, -100, 100, 200]:
        ln = n + dl
        if 100 <= ln <= 10000:
            lengths_to_try.append(ln)

    for test_n in lengths_to_try[:5]:
        if test_n == n:
            continue
        a_try = try_length(parent, test_n, rng)
        a_try = normalize_sum(a_try, test_n)
        sc = fast_score(a_try)
        if sc < best_score - 0.01:
            sc_exact = float(evaluate_sequence(a_try.tolist()))
            if np.isfinite(sc_exact) and sc_exact < best_score:
                best_score = sc_exact
                best_a = a_try.copy()
                n = test_n
                print(f"  Better length found: n={n}, score={sc_exact:.6f}")

    # Try parametric families at current best length
    n_cur = len(best_a)
    t_c = np.arange(n_cur, dtype=np.float64)
    x_c = 2.0 * t_c / max(1, n_cur - 1) - 1.0

    families = []
    for alpha in [0.2, 0.4, 0.6, 0.8]:
        families.append(1.0 + alpha * np.sin(2 * np.pi * t_c / n_cur))
        families.append(1.0 + alpha * np.cos(2 * np.pi * t_c / n_cur))
        families.append(1.0 + alpha * np.sin(4 * np.pi * t_c / n_cur + 0.3))
        families.append(1.0 - alpha * x_c ** 2)
        families.append(1.0 + alpha * x_c ** 2)

    for fam in families:
        a_try = np.maximum(fam, 0.0)
        a_try = normalize_sum(a_try, n_cur)
        sc = fast_score(a_try)
        if sc < best_score - 0.005:
            sc_exact = float(evaluate_sequence(a_try.tolist()))
            if np.isfinite(sc_exact) and sc_exact < best_score:
                best_score = sc_exact
                best_a = a_try.copy()

    print(f"Phase 0 done: best_score={best_score:.8f}, n={n_cur}")

    # === Phase 1: Main gradient descent ===
    elapsed = time.monotonic() - start_time
    budget1 = min(1200.0, total_budget - elapsed - 200.0)
    print(f"Phase 1: Gradient descent, budget={budget1:.0f}s")

    best_a, best_score = run_gradient_phase(
        best_a, budget1, lr_init=0.08, T_init=1.0, rng=rng, label="P1"
    )
    elapsed = time.monotonic() - start_time
    print(f"Phase 1 done: score={best_score:.8f}, elapsed={elapsed:.0f}s")

    # === Phase 2: Multi-restart ===
    elapsed = time.monotonic() - start_time
    budget2 = min(800.0, total_budget - elapsed - 100.0)
    n_restarts = max(1, int(budget2 / 120))
    print(f"Phase 2: {n_restarts} restarts, budget={budget2:.0f}s")

    for r in range(n_restarts):
        if time.monotonic() - start_time > total_budget - 60:
            break
        mode = ['sine', 'block', 'lowrank', 'random'][r % 4]
        a_pert = structured_perturb(best_a, rng, mode)
        a_pert = normalize_sum(a_pert, len(a_pert))

        # Short gradient descent from perturbed point
        sub_budget = (total_budget - (time.monotonic() - start_time) - 80.0) / (n_restarts - r)
        sub_budget = min(sub_budget, 150.0)
        if sub_budget < 10:
            break
        a_try, sc_try = run_gradient_phase(
            a_pert, sub_budget, lr_init=0.04, T_init=1.0, rng=rng, label=f"P2r{r}"
        )
        if sc_try < best_score:
            best_a = a_try.copy()
            best_score = sc_try
            print(f"  Restart {r} improved: {sc_try:.8f}")

    print(f"Phase 2 done: score={best_score:.8f}")

    # === Phase 3: Fine-tuning ===
    elapsed = time.monotonic() - start_time
    budget3 = max(30.0, total_budget - elapsed - 30.0)
    print(f"Phase 3: Fine-tuning, budget={budget3:.0f}s")

    # First: one more gradient phase with small lr
    budget3a = budget3 * 0.5
    best_a, best_score = run_gradient_phase(
        best_a, budget3a, lr_init=0.01, T_init=0.5, rng=rng, label="P3g"
    )

    # Then: coordinate descent
    budget3b = budget3 * 0.5 - 20
    if budget3b > 30:
        best_a, best_score = coordinate_descent(best_a, budget3b, rng)

    # Final exact evaluation
    best_a = normalize_sum(best_a, len(best_a))
    final_score = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        # Fallback to previous best if final is invalid
        best_a = normalize_sum(parent, len(parent))
        best_score = float(evaluate_sequence(best_a.tolist()))

    print(f"FINAL: score={best_score:.8f}, n={len(best_a)}, total_time={time.monotonic()-start_time:.0f}s")
    return [float(x) for x in best_a]
