```
def initialize_bit_state(s,n):
    L=n//2
    left=np.arange(L, dtype=np.int64)
    right=n-1-left
    s=np.asarray(s,dtype=np.int8)
    one_pos = np.empty(L,dtype=np.int64)
    zero_pos = np.empty(L,dtype=np.int64)
    pos_mask = s <= 0
    one_pos[pos_mask]=right[pos_mask]
    zero_pos[pos_mask]=left[pos_mask]
    one_pos[~pos_mask]=left[~pos_mask]
    zero_pos[~pos_mask]=right[~pos_mask]
    b=np.zeros(n,dtype=np.int8)
    b[one_pos]=1
    return b, one_pos, zero_pos
```

```
def compute_weights_floor(c,cmax,tau,floor=0.0):

    if not finite or s<=0: return full
    w=ez/s
    if floor>0: w=(1-floor)*w + floor/L
```

```python
"""
Search for nonnegative step-function heights minimizing

    2*n*max(convolve(a, a)) / (sum(a))^2.

Main new idea: search a highly structured binary family where exactly one
of each mirror pair {i, n-1-i} is nonzero. For n = 2L this gives a sequence
with L ones of weight 1/L and a very small coordinate-search space. Each
orientation flip moves a single 1 from u to n-1-u, and the full integer
convolution can be updated in O(n). The flips are selected by an exact
surrogate gradient for weighted convolution peaks, with simulated-annealing
acceptance and top-K move trials to escape plateaus.

Continuous mirror-descent polishing is also run from the best binary/given
candidate as a final refinement stage.
"""

import time
import math
import numpy as np


def next_pow2(m: int) -> int:
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def normalize_positive(a, floor: float = 0.0) -> np.ndarray:
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, float(floor), None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def make_uniform(n: int) -> np.ndarray:
    return np.ones(max(1, int(n)), dtype=np.float64) / float(max(1, int(n)))


def internal_score(a, N=None) -> float:
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.inf
    a = a / s
    if N is None:
        N = next_pow2(2 * n - 1)
    N = max(int(N), next_pow2(2 * n - 1))
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return float(2.0 * n * np.max(c))


def resample_seq(a, n: int) -> np.ndarray:
    a = normalize_positive(a, 0.0)
    n = max(2, int(n))
    old = a.size
    if old == n:
        return a.astype(np.float64, copy=True)
    if old == 1:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    old_x = np.linspace(0.0, 1.0, old, dtype=np.float64)
    new_x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    b = np.interp(new_x, old_x, a)
    return normalize_positive(b, 1e-12)


def get_given_sequence():
    try:
        seq = globals().get("height_sequence_1", None)
        if seq is None:
            return None
        a = np.asarray(seq, dtype=np.float64).ravel()
        if a.size == 0:
            return None
        a = normalize_positive(a, floor=0.0)
        if not np.all(np.isfinite(a)):
            return None
        return a
    except Exception:
        return None

# Binary mirror-pair search

def binary_convolution_fft(b) -> np.ndarray:
    b = np.asarray(b, dtype=np.float64).ravel()
    n = b.size
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.rint(c).astype(np.int64)


def initialize_bit_state(s, n: int):
    n = int(n)
    L = n // 2
    left = np.arange(L, dtype=np.int64)
    right = n - 1 - left
    s = np.asarray(s, dtype=np.int8).ravel()
    if s.size != L:
        s = np.resize(s, L).astype(np.int8)

    one_pos = np.empty(L, dtype=np.int64)
    zero_pos = np.empty(L, dtype=np.int64)

    is_left = s > 0
    one_pos[is_left] = left[is_left]
    zero_pos[is_left] = right[is_left]
    one_pos[~is_left] = right[~is_left]
    zero_pos[~is_left] = left[~is_left]

    b = np.zeros(n, dtype=np.int8)
    b[one_pos] = 1
    return b, one_pos, zero_pos


def apply_bit_move(n: int, r: np.ndarray, b: np.ndarray, u: int, v: int) -> None:
    """
    Move the 1 from u to v, with u currently 1 and v currently 0.
    Updates the integer convolution r in place.
    """
    u = int(u)
    v = int(v)
    r[u:u + n] -= 2 * b[:n]
    r[v:v + n] += 2 * b[:n]
    r[2 * u] += 1
    r[u + v] -= 2
    r[2 * v] += 1
    b[u] = 0
    b[v] = 1


def make_bit_weights(r: np.ndarray, M: int, tau_frac: float, alpha: float) -> np.ndarray:
    tau = max(1.0, float(tau_frac) * float(M))
    z = -(float(M) - r.astype(np.float64)) / tau
    z = np.clip(z, -60.0, 0.0)
    w = np.exp(z)
    s = float(w.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full_like(w, 1.0 / float(w.size), dtype=np.float64)
    w /= s
    if alpha > 0.0:
        w = (1.0 - float(alpha)) * w + float(alpha) / float(w.size)
    return w


def bit_start_random(L: int, seed: int = 0) -> np.ndarray:
    L = int(L)
    rng = np.random.default_rng(int(seed))
    return (rng.integers(0, 2, size=L, dtype=np.int64) * 2 - 1).astype(np.int8)


def bit_start_cosine(L: int, freq: float, phase: float = 0.0) -> np.ndarray:
    L = int(L)
    x = np.arange(L, dtype=np.float64)
    vals = np.cos(2.0 * np.pi * float(freq) * x / float(max(1, L)) + float(phase))
    vals[vals == 0.0] = 1.0
    return np.sign(vals).astype(np.int8)


def bit_start_thue(L: int) -> np.ndarray:
    L = int(L)
    arr = np.array([1], dtype=np.int8)
    while arr.size < L:
        arr = np.concatenate((arr, -arr))
    return arr[:L].astype(np.int8)


def bit_start_legendre(L: int, p: int = 2039) -> np.ndarray:
    L = int(L)
    p = int(p)
    out = np.ones(L, dtype=np.int8)
    h = (p - 1) // 2
    for i in range(L):
        x = (i + 1) % p
        if x == 0:
            continue
        val = pow(x, h, p)
        if val == 1:
            out[i] = 1
        elif val == p - 1:
            out[i] = -1
        else:
            # Composite-modulo fallback: still gives a reasonable deterministic sign.
            out[i] = 1 if (i & 1) == 0 else -1
    return out


def bit_start_from_sequence(a, n: int):
    if a is None:
        return None
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = resample_seq(a, int(n))
    n = int(a.size)
    L = n // 2
    left = a[:L]
    right = a[n - 1:n - L - 1:-1]
    s = np.ones(L, dtype=np.int8)
    mask = right > left
    s[mask] = -1
    return s


def generate_bit_starts(n: int, seed: int, given_a) -> list:
    n = int(n)
    L = n // 2
    starts = []

    if given_a is not None:
        s = bit_start_from_sequence(given_a, n)
        if s is not None:
            starts.append(s)

    starts.append(bit_start_random(L, seed + 1))
    starts.append(bit_start_random(L, seed + 99))
    starts.append(bit_start_thue(L))
    starts.append(bit_start_cosine(L, 1, 0.0))
    starts.append(bit_start_cosine(L, 2, 0.0))
    starts.append(bit_start_cosine(L, 5, 0.7))
    starts.append(bit_start_cosine(L, 11, 2.1))
    if L % 2 == 0:
        starts.append(bit_start_cosine(L, L // 2, 0.0))
    p = 2039 if L <= 2048 else 4099
    starts.append(bit_start_legendre(L, p))

    # Keep a modest number of starts. The first one (given-projection if present)
    # is deliberately kept.
    return starts[:8]


def optimize_orientation_bits(n: int, s_init, deadline: float, seed: int = 0, verbose: bool = True):

    best_b = b.copy()

    elif score > 1.62:
        tau_frac = 0.08
        alpha = 0.06
        temp_sa = 0.020
    else:
        tau_frac = 0.030
        alpha = 0.025
        temp_sa = 0.007

    it = 0
    stale = 0
    no_improve = 0
    best_prints = 0

            break

            best_b = b.copy()
            no_improve = 0
            if verbose and (best_prints < 8 or it - last_print_it > 10000):
                print("bit n={} M={} score={:.9f} it={}".format(n, M, best_score, it), flush=True)
                best_prints += 1
                last_print_it = it

        if no_improve > 180000 and it > 20000:
            break

        w = make_bit_weights(r, M, tau_frac, alpha)
        wrev = w[::-1].astype(np.float64)
        fb = np.fft.rfft(b.astype(np.float64), N)
        fw = np.fft.rfft(wrev, N)
        y = np.fft.irfft(fb * fw, N)
        A = y[n - 1:2 * n - 1][::-1]

        # Exact change in ∑ w_t r_t for moving the 1 in pair idx from one_pos to zero_pos.
        Av = A[zero_pos]
        Au = A[one_pos]
        quad = w[2 * one_pos] + w[2 * zero_pos] - 2.0 * w[n - 1]
        delta = 2.0 * (Av - Au) + quad

        if not np.all(np.isfinite(delta)):
            delta = np.where(np.isfinite(delta), delta, 1e6)

        # Top-K candidate moves. K grows with stagnation; random jumps help escape.
        K = 1
        if stale > 50:
            K = 4
        if stale > 200:
            K = 12
        if stale > 700:
            K = 36
        K = min(K, L)

        if stale > 3500 and rng.random() < 0.45:
            cand = [int(rng.integers(0, L))]
        else:
            if K == 1:
                part = np.array([int(np.argmin(delta))], dtype=np.int64)
            else:
                part = np.argpartition(delta, K - 1)[:K]
                part = part[np.argsort(delta[part]).astype(np.int64)]
            cand = [int(x) for x in part]

        accepted = False
        for idx in cand:
            u = int(one_pos[idx])
            v = int(zero_pos[idx])

            apply_bit_move(n, r, b, u, v)
            new_M = int(np.max(r))
            new_score = 4.0 * float(new_M) / float(L)

            acc = False
            if new_score < score - 1e-12:
                acc = True
            else:
                ds = new_score - score
                if abs(ds) <= 1e-12:
                    acc = rng.random() < 0.09
                elif ds < 1.0:
                    acc = rng.random() < math.exp(-ds / max(float(temp_sa), 1e-12))

            if acc:
                one_pos[idx], zero_pos[idx] = v, u
                if new_score < best_score - 1e-12:
                    best_score = new_score
                    best_b = b.copy()
                    no_improve = 0
                    if verbose and (best_prints < 12 or it - last_print_it > 5000):
                        print("bit n={} M={} score={:.9f} it={}".format(n, new_M, best_score, it), flush=True)
                        best_prints += 1
                        last_print_it = it
                else:
                    no_improve += 1

                if new_score <= score + 1e-12:
                    temp_sa = max(1e-5, temp_sa * 0.98)
                    tau_frac = max(0.006, tau_frac * 0.996)
                    alpha = max(0.0, alpha * 0.996)
                else:
                    temp_sa = min(0.18, temp_sa * 1.12)
                stale = 0
                accepted = True
                break
            else:
                # Revert move.
                apply_bit_move(n, r, b, v, u)

        if not accepted:
            stale += 1
            no_improve += 1
            temp_sa = max(1e-5, temp_sa * 0.93)

            if stale > 120:
                tau_frac = min(0.28, tau_frac * 1.55)
                alpha = min(0.30, alpha + 0.02)
                stale = 0

        it += 1

    return best_b, best_score


def search_bits(n: int, starts: list, deadline: float, seed: int, current_best_score: float):
    if not starts:
        return None, np.inf

    n = int(n)
    best_b = None
    best_score = float(current_best_score)
    num = len(starts)

    for idx, s in enumerate(starts):
        if time.monotonic() > deadline - 15.0:
            break
        remaining = deadline - time.monotonic()
        left = max(1, num - idx)

        if best_score <= 1.5020:
            per = min(110.0, remaining / float(left))
        else:
            per = min(360.0, remaining / float(left))

        if per < 30.0:
            break

        sub_deadline = min(deadline, time.monotonic() + per)
        b, sc = optimize_orientation_bits(n, s, sub_deadline, seed=seed + 1000 * n + 31 * idx)
        if b is not None and np.isfinite(sc) and sc < best_score:
            best_b = b.copy()
            best_score = sc
            print("bit search n={} start={} improved to {:.9f}".format(n, idx, best_score), flush=True)

    return best_b, best_score

# Lightweight continuous mirror-descent polishing
# ----------------------------------------------------------------------

def convolve_seq(a, N=None) -> np.ndarray:
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if N is None:
        N = next_pow2(2 * n - 1)
    N = max(int(N), next_pow2(2 * n - 1))
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]


def convolve_and_fft(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if N is None:
        N = next_pow2(3 * n - 2)
    N = max(int(N), next_pow2(3 * n - 2))
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return c, fa


def grad_from_weights(w, fa, N, n: int) -> np.ndarray:
    w = np.asarray(w, dtype=np.float64).ravel()
    w_rev = w[::-1]
    fw = np.fft.rfft(w_rev, N)
    conv = np.fft.irfft(fa * fw, N)
    return 2.0 * conv[n - 1:2 * n - 1][::-1]


def compute_weights_floor(c, cmax, tau, floor: float = 0.0) -> np.ndarray:
    c = np.asarray(c, dtype=np.float64).ravel()
    L = c.size
    if L == 0:
        return c
    cmax = float(cmax)
    if cmax <= 0.0 or not np.isfinite(cmax) or tau <= 0.0:
        return np.full(L, 1.0 / float(L), dtype=np.float64)
    z = (c - cmax) / float(tau)
    z = np.clip(z, -50.0, 0.0)
    ez = np.exp(z)
    s = float(ez.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(L, 1.0 / float(L), dtype=np.float64)
    w = ez / s
    if floor > 0.0:
        w = (1.0 - float(floor)) * w + float(floor) / float(L)
    return w


def optimize_mirror(a, deadline: float, seed: int = 123, verbose: bool = False):
    a = normalize_positive(a, floor=1e-15)
    n = a.size
    if n == 0:
        return a, np.inf

    N = next_pow2(3 * n - 2)
    best_a = a.copy()
    best_score = internal_score(a, N)

    loga = np.log(a + 1e-20).astype(np.float64)
    loga -= float(np.max(loga))

    eta = 0.35
    tau_frac = 0.05
    w_floor = 0.025
    temp_sa = 0.020
    rng = np.random.default_rng(int(seed))

    it = 0
    stale = 0
    last_print_it = -100000

    while time.monotonic() < deadline:
        c, fa = convolve_and_fft(a, N)
        cmax = float(np.max(c))
        score = float(2.0 * n * cmax)

        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            if verbose and (it - last_print_it > 250):
                print("mirror n={} score={:.9f} it={}".format(n, best_score, it), flush=True)
                last_print_it = it

        if cmax <= 1e-300:
            break

        tau = max(1e-12, tau_frac * cmax)
        w = compute_weights_floor(c, cmax, tau, floor=w_floor)
        g = grad_from_weights(w, fa, N, n)
        g = g - float(np.mean(g))
        scale = float(np.max(np.abs(g)))
        if not np.isfinite(scale) or scale <= 1e-14:
            break
        g = g / scale

        loga_try = loga - eta * g
        loga_try -= float(np.max(loga_try))
        a_try = np.exp(np.clip(loga_try, -60.0, 0.0))
        a_try = np.maximum(a_try, 1e-300)
        a_try = a_try / float(a_try.sum())
        a_try = normalize_positive(a_try, floor=1e-15)

        c_try, _ = convolve_and_fft(a_try, N)
        score_try = float(2.0 * n * np.max(c_try))

        acc = False
        if np.isfinite(score_try) and score_try < score * (1.0 - 1e-10):
            acc = True
        elif np.isfinite(score_try):
            ds = score_try - score
            if ds < 0.35:
                acc = rng.random() < math.exp(-max(0.0, ds) / max(float(temp_sa), 1e-12))

        if acc:
            a = a_try
            loga = np.log(a + 1e-20).astype(np.float64)
            loga -= float(np.max(loga))

            if score_try <= score + 1e-12:
                eta = min(1.15, eta * 1.05)
                tau_frac = max(0.006, tau_frac * 0.995)
                w_floor = max(0.0, w_floor * 0.99)
                temp_sa = max(1e-5, temp_sa * 0.97)
            else:
                temp_sa = min(0.15, temp_sa * 1.15)
            stale = 0
        else:
            eta = max(1e-4, eta * 0.62)
            stale += 1
            if stale % 220 == 0:
                noise = np.exp(rng.normal(0.0, 0.06, size=n))
                kick = normalize_positive(a * noise, floor=1e-15)
                sc_k = internal_score(kick, N)
                if np.isfinite(sc_k) and sc_k < best_score:
                    a = kick.copy()
                    loga = np.log(a + 1e-20).astype(np.float64)
                    loga -= float(np.max(loga))
                tau_frac = min(0.22, tau_frac * 1.5)
                w_floor = min(0.08, w_floor + 0.005)
                temp_sa = min(0.10, temp_sa * 1.5)
                eta = max(eta, 0.25)
                stale = 0

            if eta < 1e-5 or stale > 5000:
                break

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Candidate bookkeeping
# ----------------------------------------------------------------------

def add_candidate(candidates: list, a) -> None:
    if a is None:
        return
    try:
        arr = normalize_positive(a, floor=0.0)
        if arr.size == 0 or not np.all(np.isfinite(arr)):
            return
        sc = internal_score(arr)
        if np.isfinite(sc):
            candidates.append((sc, arr))
    except Exception:
        return


def get_best(candidates: list):
    best_a = None
    best_sc = np.inf
    for sc, a in candidates:
        if sc < best_sc:
            best_sc = sc
            best_a = a
    return best_a, best_sc


def binary_to_heights(b, n: int) -> np.ndarray:
    L = n // 2
    return normalize_positive(np.asarray(b, dtype=np.float64) / float(L), floor=0.0)

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t0 = time.monotonic()
    end = t0 + max(120.0, float(budget_s) - 60.0)

    candidates = []
    given = get_given_sequence()

    add_candidate(candidates, given)
    add_candidate(candidates, make_uniform(2048))

    best_a, best_score = get_best(candidates)
    if best_a is None:
        best_a = make_uniform(2048)
        best_score = internal_score(best_a)
    print("initial best = {:.9f}".format(best_score), flush=True)

    # ------------------------------------------------------------------
    # Main structured binary search at n = 2048.
    # ------------------------------------------------------------------
    n1 = 2048
    remaining = end - time.monotonic()
    if remaining > 700.0:
        bit1_deadline = min(end - 420.0, time.monotonic() + 1500.0)
    else:
        bit1_deadline = min(end - 20.0, time.monotonic() + max(60.0, remaining - 30.0))

    if bit1_deadline > time.monotonic() + 45.0:
        starts1 = generate_bit_starts(n1, int(seed), given)
        b1, sc1 = search_bits(n1, starts1, bit1_deadline, int(seed), best_score)
        if b1 is not None:
            add_candidate(candidates, binary_to_heights(b1, n1))
            new_a, new_sc = get_best(candidates)
            if new_a is not None and new_sc < best_score:
                print("n=2048 binary search updated best to {:.9f}".format(new_sc), flush=True)
            best_a, best_score = new_a, new_sc

    # ------------------------------------------------------------------
    # If 2048 did badly or there is still ample time, try n = 4096.
    # ------------------------------------------------------------------
    n2 = 4096
    remaining = end - time.monotonic()
    remaining_cont_reserved = 220.0
    if remaining > remaining_cont_reserved + 220.0:
        bit2_deadline = min(end - remaining_cont_reserved, time.monotonic() + 330.0)
        if bit2_deadline > time.monotonic() + 90.0:
            starts2 = generate_bit_starts(n2, int(seed) + 500, given)
            b2, sc2 = search_bits(n2, starts2[:4], bit2_deadline, int(seed) + 9100, best_score)
            if b2 is not None:
                add_candidate(candidates, binary_to_heights(b2, n2))
                new_a, new_sc = get_best(candidates)
                if new_a is not None and new_sc < best_score:
                    print("n=4096 binary search updated best to {:.9f}".format(new_sc), flush=True)
                best_a, best_score = new_a, new_sc

    # ------------------------------------------------------------------
    # Continuous polishing from the best known structure.
    # ------------------------------------------------------------------
    remaining = end - time.monotonic()
    reserve = 45.0
    if best_a is not None and remaining > reserve + 90.0:
        cont_allow = min(360.0, remaining - reserve)
        if cont_allow > 80.0:
            cont_deadline = time.monotonic() + cont_allow
            c1, sc_cont = optimize_mirror(best_a, cont_deadline, seed=int(seed) + 12345, verbose=True)
            if c1 is not None:
                add_candidate(candidates, c1)

            new_a, new_sc = get_best(candidates)
            if new_a is not None and new_sc < best_score:
                print("continuous polish updated best to {:.9f}".format(new_sc), flush=True)
            best_a, best_score = new_a, new_sc

    # If still above target and the given sequence has a different length/score,
    # try one more continuous polish from it.
    remaining = end - time.monotonic()
    if given is not None and best_score > 1.5020 and remaining > 120.0:
        cont_deadline = min(end - reserve, time.monotonic() + 150.0)
        c2, sc_cont2 = optimize_mirror(given, cont_deadline, seed=int(seed) + 77777, verbose=False)
        if c2 is not None:
            add_candidate(candidates, c2)

        new_a, new_sc = get_best(candidates)
        if new_a is not None and new_sc < best_score:
            print("continuous polish from given updated best to {:.9f}".format(new_sc), flush=True)
        best_a, best_score = new_a, new_sc

    # ------------------------------------------------------------------
    # Finalization.
    # ------------------------------------------------------------------
    if best_a is None:
        best_a = make_uniform(2048)

    best_a = normalize_positive(best_a, floor=0.0)
    out_list = [float(x) for x in best_a]

    try:
        final_eval = float(evaluate_sequence(out_list))
    except Exception:
        final_eval = best_score

    if not np.isfinite(final_eval):
        best_a = make_uniform(2048)
        out_list = [float(x) for x in best_a]
        final_eval = 2.0

    print("final internal score = {:.12f}".format(best_score), flush=True)
    print("final evaluated score = {:.12f}, length = {}".format(final_eval, len(out_list)), flush=True)
    return out_list
```
