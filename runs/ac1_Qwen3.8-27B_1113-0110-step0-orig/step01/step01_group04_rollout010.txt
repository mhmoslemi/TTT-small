def _fft_convolution(a, L=None):
    n=a.size
    if L is None: L=_pow2(2*n-1)
    A=np.fft.rfft(a,n=L)
    C=A*A
    c=np.fft.irfft(C,n=L)[:2*n-1]
    # clean small negatives
    c = np.where(c>0, c, 0.0)
    return A,c

"""
Search for non-negative step-function heights minimizing
    2 * n * max(np.convolve(a, a)) / (sum(a))^2.

Strategy:
  * Build strong analytic candidates: one-sided half-power tails, beta/tail
    windows, mirrored mixtures, and random multiplicative perturbations.
  * Polish with bounded L-BFGS-B on a log-sum-exp smoothed maximum in
    log-ratio coordinates on the probability simplex.
  * Finish with a gradient-informed multiplicative local search that moves
    away from the currently dominant convolution coefficients.
"""

import time
import numpy as np

try:
    from scipy.optimize import minimize as _minimize
except Exception:
    _minimize = None


def _pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _clean_nonnegative(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    return np.maximum(v, 0.0)


def _normalize_sum(v):
    v = _clean_nonnegative(v)
    n = v.size
    if n == 0:
        return v
    s = float(v.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    return v / s


def _fft_convolution(a, L=None):
    a = _clean_nonnegative(a)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64), np.zeros(0, dtype=np.float64)
    if L is None:
        L = _pow2(2 * n - 1)
    if L < 2 * n - 1:
        L = _pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    return A, c


def _score_L(a, L=None):
    v = _normalize_sum(a)
    n = v.size
    if n == 0:
        return np.inf
    L = _pow2(2 * n - 1) if L is None else L
    if L < 2 * n - 1:
        L = _pow2(2 * n - 1)
    _, c = _fft_convolution(v, L)
    if c.size == 0:
        return np.inf
    mx = float(c.max())
    if not np.isfinite(mx) or mx < 0.0:
        return np.inf
    return 2.0 * float(n) * mx


def _score(a):
    return _score_L(a, None)


def _power_sequence(n, p=0.5, offset=0.5):
    j = np.arange(n, dtype=np.float64)
    x = np.maximum(j + float(offset), 1e-12)
    v = np.power(x, -float(p))
    return _normalize_sum(v)


def _beta_sequence(n, alpha=0.5, beta=0.0, eta=0.5):
    j = np.arange(n, dtype=np.float64)
    left = np.maximum(j + float(eta), 1e-12)
    right = np.maximum(float(n + eta - 1) - j, 1e-12)
    v = np.power(left, -float(alpha)) * np.power(right, -float(beta))
    return _normalize_sum(v)


def _mirror_mixture(n, alpha=0.5, u=0.0, eta=0.5):
    u = float(min(max(u, 0.0), 1.0))
    j = np.arange(n, dtype=np.float64)
    left = np.maximum(j + float(eta), 1e-12)
    right = np.maximum(float(n + eta - 1) - j, 1e-12)
    p1 = np.power(left, -float(alpha))
    p2 = np.power(right, -float(alpha))
    v = (1.0 - u) * p1 + u * p2
    return _normalize_sum(v)


def _generate_candidates(n):
    cands = []
    cands.append(np.full(n, 1.0 / float(n), dtype=np.float64))

    pvals = (0.40, 0.425, 0.45, 0.475, 0.50, 0.525, 0.55, 0.575, 0.60)
    offs = (0.25, 0.35, 0.5, 0.75, 1.0, 1.5, 2.0)
    for p in pvals:
        for off in offs:
            cands.append(_power_sequence(n, p, off))

    alphas = (0.30, 0.35, 0.40, 0.45, 0.48, 0.50, 0.52, 0.55)
    betas = (0.0, 0.02, 0.05, 0.10, 0.20)
    etas = (0.5, 1.0)
    for a in alphas:
        for b in betas:
            for e in etas:
                cands.append(_beta_sequence(n, a, b, e))

    for a in (0.45, 0.50, 0.55):
        for u in (0.0, 0.05, 0.10, 0.20, 0.35, 0.50, 0.70, 0.90, 1.0):
            cands.append(_mirror_mixture(n, a, u, 0.5))

    rng = np.random.default_rng(20250101)
    base = _power_sequence(n, 0.5, 0.5)
    for k in range(64):
        sig = 0.10 + 0.22 * float((k % 8) / 7.0)
        v = base * np.exp(rng.normal(0.0, sig, size=n))
        cands.append(_normalize_sum(v))

    return cands


def _sorted_candidates(cands, L=None):
    out = []
    for i, a in enumerate(cands):
        sc = _score_L(a, L)
        if np.isfinite(sc):
            out.append((float(sc), int(i), np.asarray(a, dtype=np.float64)))
    out.sort()
    return out


def _ratios_from_a(a):
    v = _normalize_sum(a)
    n = v.size
    if n <= 1:
        return np.zeros(max(0, n - 1), dtype=np.float64)
    if v[-1] < 1e-300:
        v[-1] = 1e-300
        v = v / float(v.sum())
    x = np.log(np.maximum(v[:-1], 1e-300) / max(float(v[-1]), 1e-300))
    x = np.where(np.isfinite(x), x, 0.0)
    return np.clip(x, -80.0, 80.0)


def _softmax_from_ratios(x):
    x = np.asarray(x, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    n = int(x.size) + 1
    if n <= 1:
        return np.ones(max(1, n), dtype=np.float64)
    m = 0.0
    if x.size:
        mx = float(np.max(x))
        if np.isfinite(mx) and mx > m:
            m = mx
    e = np.exp(np.clip(x - m, -745.0, 709.0))
    e_last = float(np.exp(-m))
    s = float(e.sum()) + e_last
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    a = np.empty(n, dtype=np.float64)
    a[:-1] = e / s
    a[-1] = e_last / s
    return a


def _lse_objective(x, eps):
    a = _softmax_from_ratios(x)
    n = a.size
    if n <= 1:
        return 0.0, np.zeros(max(0, n - 1), dtype=np.float64)

    L = _pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)

    cmax = float(c.max())
    if not np.isfinite(cmax) or cmax <= 1e-300:
        return 0.0, np.zeros(n - 1, dtype=np.float64)

    eps = float(max(eps, 1e-300))
    z = (c - cmax) / eps
    z = np.clip(z, -745.0, 0.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if not np.isfinite(sum_e) or sum_e <= 1e-300:
        return 1e300, np.zeros(n - 1, dtype=np.float64)

    f = cmax + eps * float(np.log(sum_e))
    w = e / sum_e
    W = np.fft.rfft(w, n=L)
    ga = 2.0 * np.fft.irfft(W * np.conj(A), n=L)[:n]
    ga = np.where(np.isfinite(ga), ga, 0.0)

    dot = float(np.dot(a, ga))
    g = a * (ga - dot)
    g = g[:-1]
    g = np.where(np.isfinite(g), g, 0.0)

    if not np.isfinite(f):
        f = 1e300
    return float(f), g


def _run_lbfgs(fun, x, args, bounds, maxiter, maxfun):
    x = np.asarray(x, dtype=np.float64)
    if _minimize is None:
        return x, 99
    if x.size == 0:
        return x, 1

    x = np.clip(x, -80.0, 80.0)
    nvar = int(x.size)
    if nvar == 0:
        return x, 1
    if bounds is None:
        bounds = [(-80.0, 80.0)] * nvar

    options = {
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "disp": 0,
        "ftol": 1e-20,
        "gtol": 1e-16,
        "maxls": 50,
    }

    try:
        res = _minimize(
            fun,
            x,
            args=args,
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options=options,
        )
    except Exception:
        return x, 99

    x_new = np.asarray(res.x, dtype=np.float64)
    x_new = np.where(np.isfinite(x_new), x_new, 0.0)
    x_new = np.clip(x_new, -80.0, 80.0)
    try:
        status = int(res.status)
    except Exception:
        status = 0
    return x_new, status


def _refine_lbfgs(a0, end_time, seed):
    a = _normalize_sum(a0)
    n = a.size
    if n <= 1:
        sc = _score(a)
        return a, sc

    Lscore = _pow2(2 * n - 1)
    best = a.copy()
    best_score = _score_L(best, Lscore)
    if not np.isfinite(best_score):
        best = _power_sequence(n, 0.5, 0.5)
        best_score = _score_L(best, Lscore)

    rng = np.random.default_rng(int(seed))
    work_x = _ratios_from_a(best)
    bounds = [(-80.0, 80.0)] * int(max(0, work_x.size))

    cmax0 = best_score / (2.0 * float(n)) if best_score > 0 else 1e-4
    eps = float(max(1e-10, 0.25 * cmax0))
    eps_min = max(1e-12, 1e-9)

    fail_count = 0
    stale = 0

    while time.monotonic() < end_time:
        remaining = max(0.0, end_time - time.monotonic())
        if remaining < 1.0:
            break
        chunk = min(max(10.0, 0.10 * remaining), 70.0)
        chunk_end = time.monotonic() + chunk
        improved = False

        while time.monotonic() < chunk_end:
            work_x, status = _run_lbfgs(
                _lse_objective,
                work_x,
                (eps,),
                bounds,
                maxiter=220,
                maxfun=4200,
            )

            if status == 99:
                fail_count += 1
                if fail_count > 4:
                    return best, best_score
                continue
            fail_count = 0

            a_try = _normalize_sum(_softmax_from_ratios(work_x))
            sc_try = _score_L(a_try, Lscore)

            if np.isfinite(sc_try) and sc_try < best_score - 1e-14:
                best = a_try.copy()
                best_score = float(sc_try)
                improved = True

            if not np.isfinite(sc_try) or sc_try > best_score + 1e-9:
                work_x = _ratios_from_a(best)

            if status == 1:
                break

        if improved:
            eps = max(eps_min, eps * 0.84)
            stale = 0
        else:
            stale += 1
            base_x = _ratios_from_a(best)
            if stale == 1:
                work_x = np.clip(base_x + rng.normal(0.0, 0.020, size=max(0, base_x.size)), -80.0, 80.0)
            elif stale == 2:
                eps = max(eps_min, eps * 0.72)

                stale = 0

        if time.monotonic() >= end_time:
            break

    a = _normalize_sum(_softmax_from_ratios(_ratios_from_a(best)))
    sc = _score_L(a, Lscore)
    if np.isfinite(sc) and sc < best_score:
        best = a
        best_score = sc
    return best, best_score


def _adam_refine(a0, end_time, seed):
    a = _normalize_sum(a0)
    n = a.size
    if n <= 1:
        return a, _score(a)

    Lscore = _pow2(2 * n - 1)
    best = a.copy()
    best_score = _score_L(best, Lscore)
    x = _ratios_from_a(best)
    m = np.zeros(x.size, dtype=np.float64)
    v = np.zeros(x.size, dtype=np.float64)
    t = 0
    lr = 0.02
    b1 = 0.9
    b2 = 0.999
    rng = np.random.default_rng(int(seed))

    cmax0 = best_score / (2.0 * float(n)) if best_score > 0 else 1e-4
    eps = float(max(1e-10, 0.20 * cmax0))
    eps_min = max(1e-12, 1e-9)
    last_best = time.monotonic()
    no_improve = 0

    while time.monotonic() < end_time:
        f, g = _lse_objective(x, eps)
        g = np.where(np.isfinite(g), g, 0.0)
        rms = float(np.sqrt(np.mean(g * g)))
        if not np.isfinite(rms) or rms <= 1e-18:
            eps = max(eps_min, eps * 0.7)
            x = _ratios_from_a(best)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve += 1
            if no_improve > 30:
                x = np.clip(_ratios_from_a(best) + rng.normal(0.0, 0.02, size=max(0, x.size)), -80.0, 80.0)
                no_improve = 0
            continue
        g = g / rms

        t += 1
        m = b1 * m + (1.0 - b1) * g
        v = b2 * v + (1.0 - b2) * (g * g)
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)
        step = lr * mh / (np.sqrt(vh) + 1e-8)
        step = np.where(np.isfinite(step), step, 0.0)
        step = np.clip(step, -1.0, 1.0)

        x_new = np.clip(x - step, -80.0, 80.0)
        a_new = _normalize_sum(_softmax_from_ratios(x_new))
        sc_new = _score_L(a_new, Lscore)

        if np.isfinite(sc_new):
            x = x_new
            if sc_new < best_score - 1e-14:
                best = a_new.copy()
                best_score = float(sc_new)
                last_best = time.monotonic()
                no_improve = 0
                lr = min(0.10, lr * 1.05)
                if time.monotonic() - last_best > 2.0:
                    eps = max(eps_min, eps * 0.92)
        else:
            m.fill(0.0)
            v.fill(0.0)
            t = 0

        now = time.monotonic()
        if now - last_best > 8.0:
            no_improve += 1
            lr = max(0.002, lr * 0.85)
            if no_improve > 24 or now - last_best > 30.0:
                x = np.clip(_ratios_from_a(best) + rng.normal(0.0, 0.025, size=max(0, x.size)), -80.0, 80.0)
                m.fill(0.0)
                v.fill(0.0)
                t = 0
                no_improve = 0
                eps = max(eps_min, eps * 0.75)

    return best, best_score


def _local_search(a0, end_time, seed):
    a = _normalize_sum(a0)
    n = a.size
    if n <= 1:
        return a, _score(a)

    Lscore = _pow2(2 * n - 1)
    best = a.copy()
    best_score = _score_L(best, Lscore)
    rng = np.random.default_rng(int(seed))

    step = 0.045
    sigma = 0.018
    step_min = 0.004
    sigma_min = 0.003
    last_best = time.monotonic()

    while time.monotonic() < end_time:
        A, c = _fft_convolution(best, Lscore)
        if c.size == 0:
            break

        K = max(4, min(c.size, c.size // 200 + 8))
        if K < c.size:
            idx = np.argpartition(-c, K)[:K]
        else:
            idx = np.arange(c.size)

        w = np.zeros_like(c)
        w[idx] = 1.0 / float(max(1, len(idx)))
        W = np.fft.rfft(w, n=Lscore)
        ga = 2.0 * np.fft.irfft(W * np.conj(A), n=Lscore)[:n]
        ga = np.where(np.isfinite(ga), ga, 0.0)

        dot = float(np.dot(best, ga))
        d = -(ga - dot)
        d = np.where(np.isfinite(d), d, 0.0)
        normd = float(np.sqrt(np.mean(d * d)))
        if np.isfinite(normd) and normd > 1e-18:
            d = d / normd
            d = np.where(np.isfinite(d), d, 0.0)
        else:
            d = np.zeros(n, dtype=np.float64)

        improved = False
        attempts = 0
        max_attempts = 2500

        while time.monotonic() < end_time and attempts < max_attempts:
            attempts += 1
            u = step * d + rng.normal(0.0, sigma, size=n)
            u = np.clip(u, -5.0, 5.0)
            cand = best * np.exp(u)
            cand = _normalize_sum(cand)
            sc = _score_L(cand, Lscore)

            if np.isfinite(sc) and sc < best_score - 1e-14:
                best = cand
                best_score = float(sc)
                last_best = time.monotonic()
                improved = True
                step = min(0.13, step * 1.07)
                sigma = min(0.09, sigma * 1.04)
                break

            if time.monotonic() - last_best > 10.0:
                step = max(step_min, step * 0.88)
                sigma = max(sigma_min, sigma * 0.93)

        if not improved:
            step = max(step_min, step * 0.82)
            sigma = max(sigma_min, sigma * 0.82)
            pert = _normalize_sum(best * np.exp(rng.normal(0.0, 0.015, size=n)))
            scp = _score_L(pert, Lscore)
            if np.isfinite(scp) and scp < best_score - 1e-14:
                best = pert
                best_score = float(scp)
                last_best = time.monotonic()

    return best, best_score


def _append_unique(lst, a, limit=12):
    if a is None:
        return
    v = _normalize_sum(a)
    if v.size == 0:
        return
    for old in lst:
        oldv = _normalize_sum(old)
        if oldv.size != v.size:
            continue
        diff = float(np.linalg.norm(oldv - v))
        if diff < 1e-12:
            return
    lst.append(v)
    if len(lst) > limit:
        del lst[0]


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Return the best non-negative height sequence found within budget_s."""
    t0 = time.monotonic()
    try:
        total = float(budget_s)
    except Exception:
        total = 1800.0
    total = max(20.0, total - 20.0)
    deadline = t0 + total

    seed = int(seed) if seed is not None else 42

    n = 3809
    height_arr = None
    try:
        h = height_sequence_1
        if isinstance(h, (list, tuple, np.ndarray)):
            arr = np.asarray(h, dtype=np.float64).ravel()
            if 80 <= arr.size <= 12000:
                cleaned = _clean_nonnegative(arr)
                if float(cleaned.sum()) > 1e-9:
                    height_arr = _normalize_sum(cleaned)
                    n = int(arr.size)
                else:
                    height_arr = None
            else:
                height_arr = None
        else:
            height_arr = None
    except Exception:
        height_arr = None

    if n < 64 or n > 12000:
        n = 3809
        height_arr = None

    if deadline <= t0 + 4.0:
        a = _power_sequence(n, 0.5, 0.5)
        return [float(x) for x in a]

    cands = []
    if height_arr is not None and height_arr.size == n:
        cands.append(height_arr)
        cands.append(height_arr[::-1].copy())

    cands.extend(_generate_candidates(n))

    Lgen = _pow2(2 * n - 1)
    scored = _sorted_candidates(cands, Lgen)

    if not scored:
        a = _power_sequence(n, 0.5, 0.5)
        return [float(x) for x in a]

    best_a = np.asarray(scored[0][2], dtype=np.float64).copy()
    best_score = float(scored[0][0])

    print(f"n={n}, initial best score={best_score:.10f}, candidates={len(scored)}", flush=True)

    starts = []
    for i in range(min(3, len(scored))):
        starts.append(np.asarray(scored[i][2], dtype=np.float64).copy())
    if not starts:
        starts.append(best_a)

    candidates_for_official = []
    _append_unique(candidates_for_official, best_a)

    # Give a larger share to the best start, smaller shares to alternates.
    if len(starts) >= 3:
        alloc = ((0.68, 0), (0.18, 1), (0.10, 2))
    elif len(starts) == 2:
        alloc = ((0.80, 0), (0.18, 1))
    else:
        alloc = ((0.92, 0),)

    used = 0.0
    for frac, idx in alloc:
        cur = t0 + used * total
        seg_end = min(deadline - 30.0, t0 + (used + frac) * total)
        if idx < len(starts) and seg_end > cur + 5.0:
            a0 = starts[idx]
            if _minimize is None:
                a, sc = _adam_refine(a0, seg_end, seed + 101 * (idx + 1))
            else:
                a, sc = _refine_lbfgs(a0, seg_end, seed + 101 * (idx + 1))
            a = _normalize_sum(a)
            sc = _score_L(a, Lgen)
            if np.isfinite(sc) and sc < best_score:
                best_a = a.copy()
                best_score = float(sc)
                print(
                    "refined start",
                    idx,
                    "score",
                    f"{best_score:.10f}",
                    "t=",
                    f"{time.monotonic() - t0:.1f}",
                    flush=True,
                )
            _append_unique(candidates_for_official, a)
        used += frac

    # Gradient-informed multiplicative local polish.
    local_end = min(deadline - 8.0, t0 + 0.10 * total if time.monotonic() < t0 + 0.9 * total else deadline - 8.0)
    if local_end > time.monotonic() + 5.0 and time.monotonic() < deadline - 8.0:
        a, sc = _local_search(best_a, local_end, seed + 10007)
        a = _normalize_sum(a)
        sc = _score_L(a, Lgen)
        if np.isfinite(sc) and sc < best_score:
            best_a = a.copy()
            best_score = float(sc)
            print(
                "local polish score",
                f"{best_score:.10f}",
                "t=",
                f"{time.monotonic() - t0:.1f}",
                flush=True,
            )
        _append_unique(candidates_for_official, a)

    # Squeeze remaining time with a second local search pass.
    if time.monotonic() < deadline - 8.0:
        a, sc = _local_search(best_a, deadline - 8.0, seed + 20013)
        a = _normalize_sum(a)
        sc = _score_L(a, Lgen)
        if np.isfinite(sc) and sc < best_score:
            best_a = a.copy()
            best_score = float(sc)
        _append_unique(candidates_for_official, a)

    _append_unique(candidates_for_official, best_a)

    # Final official validation on a few top candidates.
    best_official = None
    official_score = np.inf
    for cand in candidates_for_official:
        out = [float(x) for x in _normalize_sum(cand)]
        try:
            val = evaluate_sequence(out)
        except Exception:
            val = np.inf
        if np.isfinite(val) and val < official_score:
            official_score = float(val)
            best_official = out

    if best_official is None or len(best_official) != n:
        best_official = [float(x) for x in _power_sequence(n, 0.5, 0.5)]

    print("final internal score", f"{best_score:.10f}", flush=True)
    if np.isfinite(official_score):
        print("final official score", f"{official_score:.10f}", flush=True)

    return best_official
