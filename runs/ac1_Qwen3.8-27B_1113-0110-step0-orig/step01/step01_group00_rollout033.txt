import numpy as np
import time
from scipy.optimize import linprog

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]

def _score(a):
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)

def _alternating_lp(a_init, max_iters, deadline, initial_blend=0.8, blend_decay=0.985):
    """Run alternating LP with adaptive blending and restart on stagnation."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = _score(a)
    blend = initial_blend
    
    # Build convolution matrix once (it changes with f, so we need to rebuild each iter)
    stagnation_count = 0
    last_improvement = best_score
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        M = float(np.max(c))
        
        # Build convolution matrix
        C = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            C[j:j + n, j] = a
        
        c_obj = -np.ones(n)
        b_ub = M * np.ones(2 * n - 1)
        bounds = [(0.0, None)] * n
        
        try:
            result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                           method='highs', options={'presolve': True})
            if result.status != 0:
                break
            g = result.x
        except Exception:
            break
        
        sum_g = float(np.sum(g))
        if sum_g < 1e-12:
            break
        
        g = g / sum_g
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        sum_new = float(np.sum(a_new))
        if sum_new < 1e-12:
            break
        a_new = a_new / sum_new
        
        score = _score(a_new)
        if score < best_score - 1e-10:
            best_score = score
            best_a = a_new.copy()
            stagnation_count = 0
            last_improvement = score
        else:
            stagnation_count += 1
        
        a = a_new
        blend = max(0.05, blend * blend_decay)
        
        # If stagnating for too long, try a perturbation-based restart
        if stagnation_count > 50 and blend < 0.1:

            a = np.maximum(a, 0)
            s = float(np.sum(a))
            a = a / s
            blend = 0.5  # Reset blend
            stagnation_count = 0
    
    return best_a, best_score

"""
Optimizes non-negative step function to minimize J(a) = 2n * max(conv(a,a)) / sum(a)^2.
Uses alternating LP with perturbation restarts, multi-scale exploration, and subgradient polish.
"""
import numpy as np
import time
from scipy.optimize import linprog


def _fft_convolve(a):
    """Fast full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]


def _score(a):
    """Compute J(a) = 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)


def _build_conv_matrix(f):
    """Build the convolution matrix C where (C @ g)[k] = (f*g)[k]."""
    n = len(f)
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = f
    return C


def _lp_one_step(f, M):
    """Solve max sum(g) s.t. C_f @ g <= M, g >= 0. Returns g or None."""
    n = len(f)
    C = _build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.status == 0:
            return result.x
    except Exception:
        pass
    return None


def _alternating_lp(a_init, max_iters, deadline, initial_blend=0.8, blend_decay=0.99,
                    restart_interval=80, noise_scale=0.02):
    """
    Alternating LP with adaptive blending and perturbation-based restarts.
    Returns (best_a, best_score).
    """
    a = a_init.copy().astype(np.float64)
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = _score(a)
    blend = initial_blend
    stagnation = 0
    rng = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(np.max(c))

        g = _lp_one_step(a, M)
        if g is None:
            break

        sum_g = float(np.sum(g))
        if sum_g < 1e-12:
            break

        g = g / sum_g
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        sum_new = float(np.sum(a_new))
        if sum_new < 1e-12:
            break
        a_new = a_new / sum_new

        score = _score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            stagnation = 0
        else:
            stagnation += 1

        a = a_new
        blend = max(0.03, blend * blend_decay)

        # Perturbation restart when stagnating
        if stagnation > restart_interval and blend < 0.15:
            noise = rng.normal(0, noise_scale * max(np.mean(a), 1e-6), n)
            a = a + noise
            a = np.maximum(a, 0.0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            blend = 0.4
            stagnation = 0

    return best_a, best_score


def _sensitivity_at_peak(a, k):
    """Compute the sensitivity (gradient) of (a*a)[k] with respect to a."""
    n = len(a)
    j = np.arange(n)
    s = np.zeros(n)
    # (a*a)[k] = sum_j a_j * a_{k-j}
    # d/d(a_i) = a_{k-i} + a_{i-k} (for valid indices)
    idx1 = k - j
    m1 = (idx1 >= 0) & (idx1 < n)
    s[m1] += a[idx1[m1]]
    idx2 = j - k
    m2 = (idx2 >= 0) & (idx2 < n)
    s[m2] += a[idx2[m2]]
    return s


def _subgradient_polish(a_init, max_iters, lr_init, lr_decay, n_peaks, deadline):
    """Subgradient optimization to flatten the top peaks of autocorrelation."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    best_a = a.copy()
    best_score = _score(a)
    lr = lr_init

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        score = 2.0 * n * float(np.max(c)) / (S * S)

        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Get top peak lags
        if n_peaks >= len(c):
            top_k = np.arange(len(c))
        else:
            top_k = np.argpartition(c, -n_peaks)[-n_peaks:]

        # Accumulate sensitivity
        s = np.zeros(n)
        for k in top_k:
            s += _sensitivity_at_peak(a, int(k))
        s /= len(top_k)

        max_s = float(np.max(s))
        if max_s > 1e-15:
            s = s / max_s
            a = a - lr * s
            a = np.maximum(a, 0.0)
            S_new = float(np.sum(a))
            if S_new > 1e-10:
                a = a * S / S_new

        lr *= lr_decay

    return best_a, best_score

def _interpolate_sequence(src, n_target):
    """Interpolate a sequence to a new length."""
    n_src = len(src)
    if n_src == n_target:
        return src.copy()
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    out = np.interp(x_tgt, x_src, src)
    out = np.maximum(out, 0.0)
    s = float(np.sum(out))
    if s > 1e-10:
        out = out / s
    return out
