import numpy as np
import time

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses projected gradient descent with log-sum-exp smoothing (annealed temperature),
Barzilai-Borwein step sizes, and simplex projection. Tries diverse initializations
across multiple sequence lengths. Final polishing uses direct subgradient on the max.
"""


def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]


def _grad_smoothed(a, t):
    """Gradient of log(sum exp(conv(a,a)/t)) w.r.t. a (naive, not accounting for simplex constraint)."""
    n = len(a)
    c = _fft_convolve(a)
    c_shifted = c - c.max()
    exp_c = np.exp(c_shifted / t)
    w = exp_c / exp_c.sum()

    # grad[j] = (2/t) * sum_k w[k] * a[k-j] (if 0 <= k-j < n)
    # = (2/t) * conv(a_rev, w)[j + n - 1]
    a_rev = a[::-1]
    conv_len = n + len(w) - 1
    fft_sz = _next_pow2(conv_len)
    fa = np.fft.rfft(a_rev, fft_sz)

    conv_result = np.fft.irfft(fa * fw, fft_sz)
    grad_unscaled = conv_result[n - 1:n - 1 + n]

def _project_simplex(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1) / rho > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho_hat = np.max(rho[cond])
    theta = (cssv[rho_hat - 1] - 1.0) / rho_hat
    return np.maximum(v - theta, 0.0)


def _score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a)
    return 2.0 * n * float(np.max(c)) / (s * s)


def _optimize_pgd(a0, time_budget, rng, verbose=True):
    """Projected gradient descent with annealed log-sum-exp smoothing."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, np.inf
    a = a / s

    c = _fft_convolve(a)
    best_score = float(np.max(c))
    best_a = a.copy()

    deadline = time.monotonic() + time_budget

    # Temperature schedule
    t_schedule = [10.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25, 0.18, 0.13, 0.1, 0.075, 0.055, 0.04, 0.03, 0.02, 0.015, 0.01]
    n_t = len(t_schedule)

        t_alloc = max(1.0, min(t_alloc, remaining))

def _optimize_subgrad_final(a0, time_budget, rng):
    """Final polishing with direct subgradient on the argmax."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, np.inf

    best_score = float(np.max(c))
    best_a = a.copy()

    deadline = time.monotonic() + time_budget
    eta = 0.02
    momentum = np.zeros(n)
    mu = 0.8
    no_improve = 0

    while time.monotonic() < deadline:
        c = _fft_convolve(a)
        k_max = int(np.argmax(c))
        score = float(c[k_max])

        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Subgradient
        g = np.zeros(n)
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)

            j_range = np.arange(lo, hi + 1)
            g[lo:hi + 1] = 2.0 * a[k_max - j_range]

        momentum = mu * momentum + (1 - mu) * gc

        a = a - eta * momentum
        a = _project_simplex(a)

    inits = []

    # Sinusoidal various params
    for amp in [0.3, 0.4, 0.5, 0.6]:
        for freq in [1, 2, 3]:
            for phase in [0, np.pi / 4, np.pi / 2, np.pi * 3 / 4, np.pi]:
                x = np.arange(n, dtype=np.float64)
                a = 1.0 + amp * np.sin(2 * np.pi * freq * x / n + phase)
                a = np.maximum(a, 0.01)
                inits.append(a)

    # Cosine
    for amp in [0.3, 0.5]:
        x = np.arange(n, dtype=np.float64)
        a = 1.0 + amp * np.cos(2 * np.pi * x / n)
        a = np.maximum(a, 0.01)
        inits.append(a)

    # Exponential decay from left
    for rate in [0.5, 1.0, 2.0, 3.0]:
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-rate * x / n) + 0.05
        inits.append(a)

    # Exponential decay from right
    for rate in [0.5, 1.0, 2.0]:
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-rate * (n - 1 - x) / n) + 0.05
        inits.append(a)

    # Power law
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        x = np.arange(1, n + 1, dtype=np.float64)
        a = x ** p
        inits.append(a)

    # Off-center Gaussian
    for center_frac in [0.2, 0.3, 0.4, 0.6, 0.7, 0.8]:
        for sigma_frac in [0.1, 0.15, 0.2]:
            x = np.arange(n, dtype=np.float64)
            center = n * center_frac
            sigma = n * sigma_frac
            a = np.exp(-0.5 * ((x - center) / sigma) ** 2) + 0.02
            inits.append(a)

    # Sigmoid
    for center_frac in [0.3, 0.5, 0.7]:
        for sharp in [5, 10, 20]:
            x = np.arange(n, dtype=np.float64)
            a = 1.0 / (1.0 + np.exp(-sharp * (x / n - center_frac)))
            a = a + 0.05
            inits.append(a)

    # Double sinusoid
    x = np.arange(n, dtype=np.float64)
    a = 1.0 + 0.3 * np.sin(2 * np.pi * x / n) + 0.15 * np.sin(4 * np.pi * x / n + np.pi / 3)
    a = np.maximum(a, 0.01)
    inits.append(a)

    # Triangular wave
    x = np.arange(n, dtype=np.float64)
    t = 2 * np.abs(2 * (x / n - np.floor(x / n + 0.5))) - 1
    a = 1.0 + 0.4 * t
    a = np.maximum(a, 0.01)
    inits.append(a)

    # Random
    for _ in range(5):
        a = rng.random(n) + 0.05
        inits.append(a)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Minimizes 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
    Uses projected gradient descent with annealed log-sum-exp smoothing,
    BB step sizes, and simplex projection. Tries diverse initializations
    across multiple sequence lengths, then polishes with direct subgradient.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 20.0

    best_overall = None
    best_overall_score = np.inf

    # Include previous best if available
    try:
        prev = np.array(height_sequence_1, dtype=np.float64)
        prev = np.clip(prev, 0, None)
        if prev.size > 50 and prev.sum() > 0.01:
            s = prev.sum()
            prev_norm = prev / s
            sc = _score(prev_norm)
            print(f"Previous best: n={len(prev)}, score={sc:.6f}")
            if sc < best_overall_score:
                best_overall_score = sc
                best_overall = prev_norm.copy()
    except (NameError, Exception):
        pass

    # Try several lengths
    lengths_to_try = [2000, 3000, 4000, 6000, 8000]

    # Time allocation per length: give more to smaller n (faster per iteration)
    # But also try larger n since the landscape might be different
    total_time_available = budget_s - 60.0

    # First, do a quick sweep of initializations for n=2000 to find the best starting point
    n_quick = 2000
    print(f"\n=== Quick sweep for n={n_quick} ===")
    inits_quick = _make_initializations(n_quick, rng)
    # Add the previous best if it has the right length (or resize)
    if best_overall is not None:
        # Interpolate to n_quick if needed
        if len(best_overall) != n_quick:
            from numpy import interp
            old_x = np.linspace(0, 1, len(best_overall))
            new_x = np.linspace(0, 1, n_quick)
            best_resized = interp(new_x, old_x, best_overall)
            best_resized = np.maximum(best_resized, 0)
            best_resized = best_resized / best_resized.sum()
        else:
            best_resized = best_overall.copy()
        inits_quick.insert(0, best_resized)

    scored = []
    for a0 in inits_quick:
        a0 = np.maximum(a0, 0.01)
        a0 = a0 / a0.sum()
        c0 = _fft_convolve(a0)
        sc0 = float(np.max(c0))
        scored.append((sc0, a0))

    scored.sort(key=lambda x: x[0])
    print(f"  Best initial: {2*n_quick*scored[0][0]:.6f}")
    for i in range(min(5, len(scored))):
        print(f"    {i+1}: {2*n_quick*scored[i][0]:.6f}")

    # Allocate time: top 5 initializations get most time
    n_top = min(5, len(scored))
    # Give 40% of time to top candidate, rest split
    times = []
    remaining_for_sweep = total_time_available * 0.5  # 50% for n=2000 sweep
    for i in range(n_top):
        if i == 0:
            times.append(remaining_for_sweep * 0.35)
        else:
            times.append(remaining_for_sweep * 0.14)

    for i in range(n_top):
        if time.monotonic() > deadline:
            break
        _, a0 = scored[i]
        t_alloc = min(times[i], max(10.0, deadline - time.monotonic()))
        print(f"\n  Optimizing init #{i+1} (score={2*n_quick*scored[i][0]:.6f}), time={t_alloc:.0f}s")

        a_opt, sc = _optimize_pgd(a0, t_alloc, rng, verbose=True)
        eval_sc = evaluate_sequence(a_opt.tolist())
        print(f"  Result: internal={sc:.6f}, eval={eval_sc:.6f}")
        if np.isfinite(eval_sc) and eval_sc < best_overall_score:
            best_overall_score = eval_sc
            best_overall = a_opt.copy()
            print(f"  *** NEW BEST: {eval_sc:.6f} ***")

    # Now try other lengths using the best shape as template
    print(f"\n=== Trying other lengths ===")
    remaining_time = deadline - time.monotonic()
    if best_overall is not None and remaining_time > 60:
        for n_try in lengths_to_try:
            if n_try == n_quick:
                continue
            if time.monotonic() > deadline - 30:
                break

            # Reshape best solution to new length
            old_x = np.linspace(0, 1, len(best_overall))
            new_x = np.linspace(0, 1, n_try)
            a_template = np.interp(new_x, old_x, best_overall)
            a_template = np.maximum(a_template, 0.01)
            a_template = a_template / a_template.sum()

            t_alloc = min(120.0, max(10.0, remaining_time * 0.2))
            remaining_time = deadline - time.monotonic()

            print(f"\n  n={n_try}, init_score={2*n_try*float(np.max(_fft_convolve(a_template))):.6f}, time={t_alloc:.0f}s")
            a_opt, sc = _optimize_pgd(a_template, t_alloc, rng, verbose=True)
            eval_sc = evaluate_sequence(a_opt.tolist())
            print(f"  Result: internal={sc:.6f}, eval={eval_sc:.6f}")
            if np.isfinite(eval_sc) and eval_sc < best_overall_score:
                best_overall_score = eval_sc
                best_overall = a_opt.copy()
                print(f"  *** NEW BEST: {eval_sc:.6f} ***")

    # Also try fresh random inits for n=3000 and n=4000
    remaining_time = deadline - time.monotonic()
    if remaining_time > 120:
        print(f"\n=== Fresh random inits for larger n ===")
        for n_try in [3000, 4000]:
            if time.monotonic() > deadline - 30:
                break
            inits = _make_initializations(n_try, rng)
            # Quick score
            scored2 = []
            for a0 in inits[:20]:
                a0 = np.maximum(a0, 0.01)
                a0 = a0 / a0.sum()
                c0 = _fft_convolve(a0)
                scored2.append((float(np.max(c0)), a0))
            scored2.sort(key=lambda x: x[0])

            t_alloc = min(90.0, max(10.0, (deadline - time.monotonic()) / 2))
            _, a0 = scored2[0]
            print(f"\n  n={n_try}, best_init_score={2*n_try*scored2[0][0]:.6f}, time={t_alloc:.0f}s")
            a_opt, sc = _optimize_pgd(a0, t_alloc, rng, verbose=True)
            eval_sc = evaluate_sequence(a_opt.tolist())
            print(f"  Result: internal={sc:.6f}, eval={eval_sc:.6f}")
            if np.isfinite(eval_sc) and eval_sc < best_overall_score:
                best_overall_score = eval_sc
                best_overall = a_opt.copy()
                print(f"  *** NEW BEST: {eval_sc:.6f} ***")

    # Final polishing with direct subgradient
    remaining = deadline - time.monotonic()
    if best_overall is not None and remaining > 30:
        print(f"\n=== Final polishing ({remaining:.0f}s) ===")
        a_pol, sc_pol = _optimize_subgrad_final(best_overall, remaining * 0.7, rng)
        eval_pol = evaluate_sequence(a_pol.tolist())
        print(f"  Polished: {eval_pol:.6f}")
        if np.isfinite(eval_pol) and eval_pol < best_overall_score:
            best_overall_score = eval_pol
            best_overall = a_pol.copy()
            print(f"  *** NEW BEST: {eval_pol:.6f} ***")

        remaining2 = deadline - time.monotonic()
        if remaining2 > 15:
            # One more PGD pass from the polished solution with very low temperature
            a_pol2, sc_pol2 = _optimize_pgd(best_overall, remaining2, rng, verbose=True)
            eval_pol2 = evaluate_sequence(a_pol2.tolist())
            print(f"  Polished2: {eval_pol2:.6f}")
            if np.isfinite(eval_pol2) and eval_pol2 < best_overall_score:
                best_overall_score = eval_pol2
                best_overall = a_pol2.copy()
                print(f"  *** NEW BEST: {eval_pol2:.6f} ***")

    # Final check
    if best_overall is not None:
        final_score = evaluate_sequence(best_overall.tolist())
        elapsed = time.monotonic() - total_start
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(best_overall)}, time={elapsed:.1f}s ===")
        return [float(x) for x in best_overall]
    else:
        a = np.ones(2000, dtype=np.float64)
        print("Fallback to uniform")
        return a.tolist()
