import time

def solve_lp(a, M):
    """Solve the LP: max sum(b) s.t. b >= 0, (a*convolve*b)[k] <= M for all k."""
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    
    # Build the convolution constraint matrix
    # (a * b)[k] = sum_j a[j] * b[k-j] <= M
    # This is a Toeplitz-like matrix times b <= M

    # For each k (0 to 2n-2), the constraint is:
    # sum_{j: 0<=j<n, 0<=k-j<n} a[j] * b[k-j] <= M

    # Then the constraint is A @ b <= M * ones
    
    # This matrix has 2n-1 rows and n columns
    # A[k, i] = a[k-i] if max(0, k-n+1) <= i <= min(k, n-1), else 0
    
    # For efficiency, we can build it using scipy's toeplitz or similar
    from scipy.sparse import csr_matrix
    import scipy.sparse as sp
    
    # Build the constraint matrix
    # Row k, col i: a[k-i] if 0 <= k-i < n
    # 
    # For row k: non-zero entries at columns i where 0 <= k-i < n
    # i.e., k-n < i <= k, intersected with [0, n-1]
    # i from max(0, k-n+1) to min(k, n-1)
    # value = a[k-i]
    
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    objective = cp.Maximize(cp.sum(b))
    constraints = [A @ b <= M]
    opt_val = cp.Problem(objective, constraints).solve()
    
    return b.value

import numpy as np
import time
import cvxpy as cp
from scipy.sparse import csr_matrix

def fft_convolve(a, b):
    """Compute full linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    c = np.fft.irfft(fa * fb, size)
    return c[:n + m - 1]

def fft_self_conv(a):
    """Compute full self-convolution using FFT."""
    return fft_convolve(a, a)

def build_conv_matrix(a):
    """Build the sparse convolution matrix.
    A[k, i] = a[k-i] if 0 <= k-i < len(a), else 0.
    Shape: (2n-1, n)
    """
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_iteration(a, M, solver=cp.GLOP):
    """Solve LP: max sum(b) s.t. b >= 0, A @ b <= M * ones."""
    n = len(a)
    A = build_conv_matrix(a)
    b = cp.Variable(n, nonneg=True)
    objective = cp.Maximize(cp.sum(b))
    constraints = [A @ b <= M]
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=solver)
    if b.value is not None:
        return np.asarray(b.value).flatten()
    return a.copy()

def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def fft_self_conv(a):
    """Full self-convolution via FFT."""
    return fft_convolve(a, a)


def build_conv_matrix(a):
    """Build sparse matrix A where (A @ b)[k] = (a*b)[k] (linear convolution)."""
    n = len(a)
    N = 2 * n - 1
    indptr = np.zeros(N + 1, dtype=np.int64)
    for k in range(N):
        i_s = max(0, k - n + 1)
        i_e = min(k, n - 1)
        indptr[k + 1] = indptr[k] + (i_e - i_s + 1)
    total = int(indptr[-1])
    cols = np.empty(total, dtype=np.int32)
    data = np.empty(total, dtype=np.float64)
    pos = 0
    for k in range(N):
        i_s = max(0, k - n + 1)
        i_e = min(k, n - 1)
        length = i_e - i_s + 1
        cols[pos:pos + length] = np.arange(i_s, i_e + 1, dtype=np.int32)
        data[pos:pos + length] = a[k - np.arange(i_s, i_e + 1)]
        pos += length
    return csr_matrix((data, cols, indptr), shape=(N, n))
