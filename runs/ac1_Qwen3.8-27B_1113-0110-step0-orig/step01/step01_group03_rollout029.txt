import numpy as np
import time
import scipy.optimize
from scipy.optimize import minimize, linprog

"""
Minimizes 2n*max(convolve(a,a))/sum(a)^2 using:
1. Soft-max L-BFGS with decreasing temperature
2. LP-based best response iterations
3. Active-set subgradient with adaptive step size
4. Random restarts and multi-n exploration
"""

def fft_self_conv(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    if n <= 1:
        return a * a
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:2 * n - 1]

def fft_convolve(a, b):
    """Compute full convolution of a and b using FFT."""
    na, nb = len(a), len(b)
    size = 1
    while size < na + nb - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    c = np.fft.irfft(fa * fb, size)
    return c[:na + nb - 1]

def softmax(x):
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()

def compute_grad_b(b, p, n):
    """Compute gradient of 2n*soft_max w.r.t. b, given softmax weights p."""
    # grad_b[j] = 2n * sum_k p[k] * b[k-j] (valid indices)
    # = 2n * (p * b_rev)[j + n - 1]
    b_rev = b[::-1]
    return 2.0 * n * fft_convolve(p, b_rev)[n-1:2*n-1]

def soft_max_objective(theta, tau, n):
    """Objective: 2n * soft_max(conv(b,b), tau) where b = softmax(theta)."""
    b = softmax(theta)
    c = fft_self_conv(b)
    c_t = c / tau
    c_max = c_t.max()
    exp_c = np.exp(c_t - c_max)
    val = tau * (c_max + np.log(exp_c.sum()))
    return 2.0 * n * val

def soft_max_grad(theta, tau, n):
    """Gradient of soft_max objective w.r.t. theta."""
    b = softmax(theta)
    c = fft_self_conv(b)
    c_t = c / tau
    c_max = c_t.max()
    exp_c = np.exp(c_t - c_max)
    p = exp_c / exp_c.sum()
    
    grad_b = compute_grad_b(b, p, n)
    mean_gb = np.dot(b, grad_b)
    grad_theta = b * (grad_b - mean_gb)
    return grad_theta

def soft_max_obj_and_grad(theta, tau, n):
    """Combined objective and gradient for scipy."""
    return soft_max_objective(theta, tau, n), soft_max_grad(theta, tau, n)

def lp_best_response(a, M, n):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0."""
    # A[k,j] = a[k-j] if 0 <= k-j < n, else 0
    # We need to build the constraint matrix A of shape (2n-1, n)
    # A[k,:] = shifted version of a
    
    # Using the structure: A is a "correlation" matrix
    # A[k,j] = a[k-j] if max(0, k-n+1) <= j <= min(n-1, k), else 0
    
    # For efficiency, build A using scipy.sparse
    from scipy.sparse import lil_matrix, csr_matrix
    
    A = lil_matrix((2*n-1, n))
    for j in range(n):
        for k in range(j, j+n):  # k-j is in [0, n-1]
            A[k, j] = a[k-j]
    A = csr_matrix(A)
    
    # LP: max 1^T b s.t. A b <= M, b >= 0
    # In scipy form: min -1^T b s.t. A b <= M, b >= 0
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return None

def active_set_grad_descent(a, n, max_iters, deadline, best_score, best_a, eval_every=50):
    """Active-set subgradient descent with adaptive step size."""
    a = a.copy()
    S = a.sum()
    a = a / S * n  # normalize to sum = n
    momentum = np.zeros(n)
    beta = 0.8
    lr = 0.05  # start with smaller lr for fine-tuning
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        a = np.maximum(a, 0)
        S = a.sum()
        if S < 0.01:
            a = best_a.copy()
            S = a.sum()
        a = a / S * n
        
        c = fft_self_conv(a)
        M = c.max()
        score = 2.0 * n * M / (n * n)  # S = n so S^2 = n^2
        
        # Find active peaks
        thresh = M * 0.999
        peak_indices = np.where(c >= thresh)[0]
        # Group nearby peaks
        k_stars = []
        for idx in peak_indices[np.argsort(-c[peak_indices])]:
            if all(abs(idx - k) > 3 for k in k_stars):
                k_stars.append(int(idx))
            if len(k_stars) >= 3:
                break
        if not k_stars:
            k_stars = [int(np.argmax(c))]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in k_stars:
            j_min = max(0, k - (n - 1))
            j_max = min(n - 1, k)
            if j_max >= j_min:
                s_start = k - j_max
                s_end = k - j_min + 1
                grad[j_min:j_max + 1] += a[s_start:s_end][::-1]
        grad /= len(k_stars)
        
        # Score gradient: d(2nM/S^2)/da = 4n/S^2 * subgrad_M - 4nM/S^3 * 1
        # With S = n: = 4/n * subgrad_M - 4M/n^2
        grad = (4.0 / n) * grad - (4.0 * M / (n * n))
        
        # Momentum update
        momentum = beta * momentum - lr * grad
        a_new = np.maximum(a + momentum, 0)
        a_new = a_new / a_new.sum() * n
        a = a_new
        
        if (it + 1) % eval_every == 0:
            exact_score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(exact_score) and exact_score < best_score:
                best_a = a.copy()
                best_score = exact_score
                no_improve = 0
                if it < 1000:
                    print(f"    AS it={it+1}: score={best_score:.8f}, lr={lr:.6f}")
            else:
                no_improve += 1
            
            if no_improve >= 20:
                lr *= 0.6
                momentum *= 0.3
                no_improve = 0
                if lr < 1e-8:
                    # Random restart
                    perturb = np.exp(np.random.default_rng(it).normal(0, 0.1, n))
                    a = best_a * perturb
                    a = a / a.sum() * n
                    lr = 0.02
                    momentum = np.zeros(n)
    
    return best_a, best_score

import numpy as np
import time
import scipy.optimize
from scipy.optimize import minimize, linprog
from scipy.sparse import lil_matrix, csr_matrix

"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2:
Phase 1: Soft-max L-BFGS with decreasing temperature (smooth landscape navigation)
Phase 2: LP-based best response iterations (principled direction finding)
Phase 3: Active-set subgradient with line search (fine-tuning)
Phase 4: Additional LP + active-set polish
Uses FFT-based convolution, softmax parameterization for simplex constraint,
and scipy HiGHS solver for LP subproblems.
"""

# Global variables for L-BFGS objective wrapper (avoid closures)
_g_tau = 1.0
_g_n = 1


def fft_self_conv(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    if n <= 1:
        return a * a
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:2 * n - 1]


def fft_convolve(a, b):
    """Compute full linear convolution of a and b using FFT."""
    na, nb = len(a), len(b)
    size = 1
    while size < na + nb - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    c = np.fft.irfft(fa * fb, size)
    return c[:na + nb - 1]


def softmax_fn(x):
    """Numerically stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.full_like(x, 1.0 / len(x))
    return e / s


def compute_grad_b(b, p, n):
    """Gradient of 2n*soft_max w.r.t. b: 2n * (p * b_rev)[n-1:2n-1]."""
    b_rev = b[::-1]
    conv = fft_convolve(p, b_rev)
    return 2.0 * n * conv[n - 1:2 * n - 1]


def soft_max_val_and_p(b, tau, n):
    """Compute soft-max value and softmax weights of convolution."""
    c = fft_self_conv(b)
    c_t = c / tau
    c_max = c_t.max()
    exp_c = np.exp(c_t - c_max)
    s = exp_c.sum()
    val = 2.0 * n * (c_max + tau * np.log(s))
    p = exp_c / s
    return val, p, c


def _lbfgs_wrapper(theta):
    """Wrapper for scipy L-BFGS using global tau and n."""
    b = softmax_fn(theta)
    val, p, c = soft_max_val_and_p(b, _g_tau, _g_n)
    n = _g_n
    grad_b = compute_grad_b(b, p, n)
    mean_gb = np.dot(b, grad_b)
    grad_theta = b * (grad_b - mean_gb)
    return val, grad_theta


def lp_solve_best_response(a, M, n):
    """
    Solve LP: max sum(b) s.t. (a*b)[k] <= M for all k in [0,2n-2], b >= 0.
    Returns b_star (unnormalized) or None if infeasible.
    """
    # Build constraint matrix A of shape (2n-1, n)
    # A[k,j] = a[k-j] if 0 <= k-j < n, else 0
    A = lil_matrix((2 * n - 1, n))
    for j in range(n):
        k_lo = j
        k_hi = min(j + n - 1, 2 * n - 2)
        A[k_lo:k_hi + 1, j] = a[0:k_hi - k_lo + 1]
    A = csr_matrix(A)

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

    Active-set subgradient descent with adaptive learning rate and line search.
    a is normalized to sum=n.
    """

    a = a.copy().astype(np.float64)
    S = a.sum()
    if S < 1e-10:
        return best_a, best_score
    a = a / S * n

    momentum = np.zeros(n)
    beta = 0.75
    lr = 0.03
    no_improve = 0
    best_local = a.copy()
    best_local_score = best_score

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 0.01:
            a = best_local.copy()
            S = a.sum()
        a = a / S * n

        c = fft_self_conv(a)
        M = c.max()

        # Find top peaks (active set)
        thresh = M * 0.995
        candidates = np.where(c >= thresh)[0]
        if len(candidates) == 0:
            candidates = np.array([int(np.argmax(c))])
        # Sort by value and pick distinct peaks
        sorted_cands = candidates[np.argsort(-c[candidates])]
        k_stars = []
        for idx in sorted_cands:
            if all(abs(int(idx) - k) > max(3, n // 100) for k in k_stars):
                k_stars.append(int(idx))
            if len(k_stars) >= 4:
                break
        if not k_stars:
            k_stars = [int(np.argmax(c))]

        # Compute subgradient of max
        subgrad = np.zeros(n)
        for k in k_stars:
            j_min = max(0, k - (n - 1))
            j_max = min(n - 1, k)
            if j_max >= j_min:
                s_start = k - j_max
                s_end = k - j_min + 1
                subgrad[j_min:j_max + 1] += a[s_start:s_end][::-1]
        subgrad /= len(k_stars)

        # Full gradient of score = 2n*M/S^2 with S=n:
        # d(score)/da[j] = (4n/S^2)*subgrad[j] - 4nM/S^3
        # = (4/n)*subgrad[j] - 4M/n^2
        grad = (4.0 / n) * subgrad - (4.0 * M / (n * n))

        # Momentum update
        momentum = beta * momentum - lr * grad
        a_candidate = np.maximum(a + momentum, 0.0)
        sc = a_candidate.sum()
        if sc < 1e-10:
            a_candidate = a
        else:
            a_candidate = a_candidate / sc * n
        a = a_candidate

        # Periodic exact evaluation
        if (it + 1) % eval_every == 0:
            exact_score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(exact_score) and exact_score < best_score:
                best_a = a.copy()
                best_score = exact_score
                best_local = a.copy()
                best_local_score = exact_score
                no_improve = 0
                if it < 2000 or (it + 1) % 500 == 0:
                    print(f"    AS it={it+1}: score={best_score:.8f}, lr={lr:.6f}, peaks={len(k_stars)}")
            else:
                no_improve += 1

            # Adaptive learning rate
            if no_improve >= 15:
                lr *= 0.65
                momentum *= 0.3
                no_improve = 0
                if lr < 1e-7:
                    # Perturb from best
                    rng_local = np.random.default_rng(it * 7919 + 13)
                    perturb = np.exp(rng_local.normal(0, 0.05, n))
                    a = best_local * perturb
                    a = np.maximum(a, 0)
                    a = a / a.sum() * n
                    lr = 0.01
                    momentum = np.zeros(n)
                    no_improve = 0

    return best_a, best_score


def soft_max_phase(b, n, deadline, best_score, best_b):
    """
    Phase 1: Soft-max L-BFGS with decreasing temperature.
    b is normalized to sum=1.
    """
    global _g_tau, _g_n
    _g_n = n

    c0 = fft_self_conv(b)
    M0 = c0.max()

    tau_schedule = [M0 * 0.5, M0 * 0.25, M0 * 0.12, M0 * 0.06, M0 * 0.03, M0 * 0.015, M0 * 0.008]

    current_b = b.copy()
    for tau in tau_schedule:
        if time.monotonic() > deadline - 100:
            break
        _g_tau = tau
        theta = np.log(current_b + 1e-30)

        try:
            result = minimize(
                _lbfgs_wrapper,
                x0=theta,
                jac=True,
                method='L-BFGS-B',
                options={'maxiter': 5000, 'ftol': 1e-18, 'gtol': 1e-14, 'maxfun': 100000}
            )
            b_new = softmax_fn(result.x)
            b_new = np.maximum(b_new, 0.0)
            s = b_new.sum()
            if s > 1e-10:
                b_new = b_new / s
            score = float(evaluate_sequence(b_new.tolist()))
            if np.isfinite(score) and score < best_score:
                best_b = b_new.copy()
                best_score = score
                current_b = b_new.copy()
                print(f"    L-BFGS tau={tau:.6f}: score={score:.8f}")
            else:
                # Try a few intermediate mixes
                for mix in [0.3, 0.5, 0.7]:
                    b_mix = (1 - mix) * current_b + mix * b_new
                    b_mix = np.maximum(b_mix, 0)
                    b_mix = b_mix / b_mix.sum()
                    sc = float(evaluate_sequence(b_mix.tolist()))
                    if np.isfinite(sc) and sc < best_score:
                        best_b = b_mix.copy()
                        best_score = sc
                        current_b = b_mix.copy()
                        print(f"    L-BFGS mix={mix}: score={sc:.8f}")
                        break
        except Exception as e:
            print(f"    L-BFGS error at tau={tau:.6f}: {e}")

    return best_b, best_score


def lp_phase(b, n, deadline, best_score, best_b, max_iters=100, line_search=True):
    """
    Phase 2/4: LP-based best response with optional line search.
    b is normalized to sum=1.
    """
    a_work = b.copy()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fft_self_conv(a_work)
        M = c.max()

        b_star = lp_solve_best_response(a_work, M, n)
        if b_star is None:
            break

        b_star = np.maximum(b_star, 0.0)
        S_star = b_star.sum()
        if S_star < 1e-10:
            break
        b_star_norm = b_star / S_star

        if line_search:
            best_t_score = np.inf
            best_t = 0.0
            for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                a_try = (1 - t) * a_work + t * b_star_norm
                a_try = np.maximum(a_try, 0.0)
                st = a_try.sum()
                if st < 1e-10:
                    continue
                a_try = a_try / st
                sc = float(evaluate_sequence(a_try.tolist()))
                if np.isfinite(sc) and sc < best_t_score:
                    best_t_score = sc
                    best_t = t
                    best_a_try = a_try.copy()
            if best_t > 0 and best_t_score < best_score:
                best_b = best_a_try.copy()
                best_score = best_t_score
                a_work = best_a_try.copy()
                no_improve = 0
                if it % 3 == 0:
                    print(f"    LP it={it}: score={best_score:.8f}, t={best_t}, S*={S_star:.3f}")
            else:
                a_work = (1 - 0.2) * a_work + 0.2 * b_star_norm
                a_work = np.maximum(a_work, 0)
                a_work = a_work / a_work.sum()
                no_improve += 1
        else:
            t = 0.3
            a_new = (1 - t) * a_work + t * b_star_norm
            a_new = np.maximum(a_new, 0.0)
            sn = a_new.sum()
            if sn < 1e-10:
                break
            a_new = a_new / sn
            score = float(evaluate_sequence(a_new.tolist()))
            if np.isfinite(score) and score < best_score:
                best_b = a_new.copy()
                best_score = score
                a_work = a_new.copy()
                no_improve = 0
                if it % 5 == 0:
                    print(f"    LP it={it}: score={score:.8f}, S*={S_star:.3f}")
            else:
                a_work = a_new
                no_improve += 1

        if no_improve >= 25:
            a_work = best_b.copy()
            no_improve = 0

    return best_b, best_score


def propose_candidate(seed=42, budget_s=2380, **kwargs):
    """
    Multi-phase optimizer:
    1. Soft-max L-BFGS (smooth optimization with decreasing temperature)
    2. LP best response with line search (principled improvement)
    3. Active-set subgradient (fine-tuning)
    4. More LP + polish
    """
    rng = np.random.default_rng(seed)

    # Get parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        parent = np.ones(512, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    if parent.size == 0 or not np.isfinite(parent).all() or parent.sum() < 0.01:
        parent = np.ones(512, dtype=np.float64)

    n = len(parent)
    b0 = parent / parent.sum()  # normalize to sum=1

    best_b = b0.copy()
    best_score = float(evaluate_sequence(best_b.tolist()))
    if not np.isfinite(best_score):
        best_b = np.ones(n) / n
        best_score = 2.0

    print(f"Initial: score={best_score:.8f}, n={n}, sum={b0.sum():.4f}")
    start_time = time.monotonic()
    deadline = start_time + budget_s - 10

    # === Phase 1: Soft-max L-BFGS ===
    print(f"\nPhase 1: Soft-max L-BFGS (t={time.monotonic()-start_time:.1f}s)")
    best_b, best_score = soft_max_phase(best_b, n, deadline - 600, best_score, best_b)
    print(f"  After Phase 1: score={best_score:.8f}")

    # === Phase 2: LP best response ===
    print(f"\nPhase 2: LP Best Response (t={time.monotonic()-start_time:.1f}s)")
    remaining = deadline - time.monotonic()
    lp_budget = int(min(remaining * 0.35, 900))
    max_lp_iters = max(20, lp_budget // 10)
    best_b, best_score = lp_phase(best_b, n, time.monotonic() + lp_budget, best_score, best_b,
                                   max_iters=max_lp_iters, line_search=True)
    print(f"  After Phase 2: score={best_score:.8f}")

    # === Phase 3: Active-set subgradient ===
    print(f"\nPhase 3: Active-set gradient (t={time.monotonic()-start_time:.1f}s)")
    a_n = best_b * n  # scale to sum=n
    as_budget = int(min(deadline - time.monotonic(), 1200))
    best_a_n, best_score = active_set_refine(
        a_n, n, time.monotonic() + as_budget, best_score, a_n.copy(),
        max_iters=1000000, eval_every=100
    )
    best_b = best_a_n / n
    print(f"  After Phase 3: score={best_score:.8f}")

    # === Phase 4: More LP if time permits ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\nPhase 4: More LP (t={time.monotonic()-start_time:.1f}s, {remaining:.0f}s left)")
        max_lp4 = max(10, int(remaining * 0.5) // 8)
        best_b, best_score = lp_phase(best_b, n, time.monotonic() + int(remaining * 0.5),
                                       best_score, best_b, max_iters=max_lp4, line_search=True)
        print(f"  After Phase 4: score={best_score:.8f}")

    # === Phase 5: Final active-set polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\nPhase 5: Final polish (t={time.monotonic()-start_time:.1f}s, {remaining:.0f}s left)")
        a_n = best_b * n
        best_a_n, best_score = active_set_refine(
            a_n, n, time.monotonic() + int(remaining * 0.8), best_score, a_n.copy(),
            max_iters=1000000, eval_every=50
        )
        best_b = best_a_n / n

    # Final validation
    result = best_b * n  # scale to sum=n for return
    final_score = float(evaluate_sequence(result.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score
        result = (best_b * n).copy()

    print(f"\nFinal score: {best_score:.8f}, n={n}, time={time.monotonic()-start_time:.1f}s")
    return [float(x) for x in result]
