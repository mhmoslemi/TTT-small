def build_conv_matrix(a):

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

"""
LP-based iterative improvement for minimizing 2n*max(a*a)/sum(a)^2.
Uses convex direction-finding: at each step, solve an LP to find a direction g
that keeps cross-correlation bounded, then blend with current sequence.
Combined with gradient refinement and adaptive restarts.
"""


def fft_self_conv(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    if n <= 1:
        return a.copy()
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:2 * n - 1]


def normalize_sum(a, target):
    """Normalize array to have sum=target, keeping non-negativity."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-10:
        return np.full_like(a, target / len(a))
    return a / S * target


def fast_score(a):
    """Fast score computation via FFT."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_self_conv(a)
    M = c.max()
    return 2.0 * n * M / (S * S)


def build_conv_idx(n):
    """Pre-build (row, col) indices for the convolution matrix C where C@g = a*g."""
    row_list = []
    col_list = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        row_list.extend([k] * (j_max - j_min + 1))
        col_list.extend(range(j_min, j_max + 1))
    return np.array(row_list, dtype=np.int32), np.array(col_list, dtype=np.int32)


def solve_lp_direction(a, M, row_idx, col_idx, max_constraints=None):
    """
    Solve LP: max sum(g) s.t. (a*g)[k] <= M for selected k, g >= 0.
    If max_constraints is given, only use the most binding constraints.
    """
    n = len(a)
    k_arr = row_idx
    j_arr = col_idx
    idx = k_arr - j_arr
    vals = a[idx]

    # Determine which constraints to use
    num_full = 2 * n - 1
    if max_constraints is not None and max_constraints < num_full:
        # Only use top constraints: find which k values have the largest (a*g) potential
        # Use all constraints but group by k; select top-k by current a*a value
        c = fft_self_conv(a)
        top_ks = np.argsort(-c)[:max_constraints]
        mask = np.isin(k_arr, top_ks)
        r = row_idx[mask]
        c_idx = col_idx[mask]
        v = vals[mask]
        C = csr_matrix((v, (r, c_idx)), shape=(max_constraints, n))
        b = np.full(max_constraints, M)
        # Remap: we need c[top_ks] as the RHS, but they're all <= M so M is safe
    else:
        C = csr_matrix((vals, (row_idx, col_idx)), shape=(num_full, n))
        b = np.full(num_full, M)

    res = linprog(c=-np.ones(n), A_ub=C, b_ub=b,
                  bounds=[(0, None)] * n, method='highs',
                  options={'disp': False, 'time_limit': 10})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def gradient_step(a, lr, momentum, beta, num_peaks=1):
    """One gradient step with momentum for the score function."""
    n = len(a)
    c = fft_self_conv(a)
    M = c.max()

    # Find top peak positions
    if num_peaks == 1:
        k_stars = [int(np.argmax(c))]
    else:
        # Find distinct peaks
        sorted_idx = np.argsort(-c)
        k_stars = []
        for idx in sorted_idx:
            if all(abs(idx - k) > 3 for k in k_stars):
                k_stars.append(int(idx))
            if len(k_stars) >= num_peaks:
                break

    # Compute gradient contribution from each peak
    grad = np.zeros(n)
    for k in k_stars:
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        s_start = k - j_max
        s_end = k - j_min + 1
        grad[j_min:j_max + 1] += a[s_start:s_end][::-1]
    grad /= len(k_stars)

    # Full gradient of 2n*M/S^2 (with S=n after normalization)
    grad = (4.0 / n) * grad - (4.0 * M / (n * n))

    momentum = beta * momentum - lr * grad
    return a + momentum, momentum, c, M

    # === Phase 1: LP-based iteration ===
    print("=== Phase 1: LP-based iteration ===")
    it = 0
    no_improve = 0
    t_blend = 0.15
    lp_time_avg = 0.0

    while time.monotonic() < deadline:
        it += 1
        a = normalize_sum(best_a, n)
        c = fft_self_conv(a)
        M = c.max()
        S = a.sum()

        # Solve LP (use all constraints initially, reduce if too slow)
        max_constraints = None
        if lp_time_avg > 2.0:
            max_constraints = min(500, 2 * n - 1)
        elif lp_time_avg > 1.0:
            max_constraints = min(2000, 2 * n - 1)

        t0 = time.monotonic()
        g = solve_lp_direction(a, M, row_idx, col_idx, max_constraints)
        lp_dt = time.monotonic() - t0
        lp_time_avg = 0.9 * lp_time_avg + 0.1 * lp_dt

        if g is None or g.sum() < 1e-10:
            no_improve += 5
        else:
            g = normalize_sum(g, S)

            # Line search over blend parameter
            best_local_score = best_score
            best_local_a = best_a.copy()
            for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
                a_try = (1 - t) * a + t * g
                a_try = normalize_sum(a_try, S)
                sc = fast_score(a_try)
                if sc < best_local_score - 1e-12:
                    best_local_score = sc
                    best_local_a = a_try.copy()

            if best_local_score < best_score - 1e-12:
                best_score = best_local_score
                best_a = best_local_a
                no_improve = 0
                if it % 50 == 1:
                    print(f"  LP it={it}: score={best_score:.8f}, t_lp={lp_dt:.3f}s")
            else:
                no_improve += 1

                # Adaptive: try perturbation if stuck
                if no_improve >= 15:
                    # Multiplicative perturbation
                    noise = np.exp(rng.normal(0, 0.03 + 0.01 * (no_improve // 10), n))
                    a_pert = best_a * noise
                    a_pert = normalize_sum(a_pert, n)
                    sc_pert = fast_score(a_pert)
                    if sc_pert < best_score:
                        best_a = a_pert
                        best_score = sc_pert
                        print(f"  Perturb it={it}: score={best_score:.8f}")
                    no_improve = 0

                # Occasional larger perturbation
                if no_improve >= 30:
                    mask = rng.random(n) < 0.3
                    a_pert = best_a.copy()
                    a_pert[mask] *= rng.uniform(0.3, 3.0, mask.sum())
                    a_pert = normalize_sum(a_pert, n)
                    sc_pert = fast_score(a_pert)
                    if sc_pert < best_score:
                        best_a = a_pert
                        best_score = sc_pert
                        print(f"  Big perturb it={it}: score={best_score:.8f}")
                    no_improve = 0

        if time.monotonic() > deadline - 30:
            break

    print(f"Phase 1 done: it={it}, score={best_score:.8f}")

    # === Phase 2: Gradient refinement ===
    print("=== Phase 2: Gradient refinement ===")
    a = normalize_sum(best_a, n)
    momentum = np.zeros(n)
    lr = 0.05
    beta = 0.8
    grad_it = 0
    no_improve_g = 0

    deadline2 = deadline - 30.0

    while time.monotonic() < deadline2:
        a = normalize_sum(a, n)
        num_peaks = 1 if no_improve_g < 20 else 2
        a, momentum, c, M = gradient_step(a, lr, momentum, beta, num_peaks)
        a = np.maximum(a, 0.0)
        a = normalize_sum(a, n)
        grad_it += 1

        if grad_it % 50 == 0:
            sc = fast_score(a)
            if sc < best_score - 1e-10:
                best_a = a.copy()
                best_score = sc
                no_improve_g = 0
                if grad_it % 500 == 0:
                    exact_sc = float(evaluate_sequence(a.tolist()))
                    if np.isfinite(exact_sc) and exact_sc < best_score:
                        best_score = exact_sc
                        best_a = a.copy()
                    print(f"  Grad it={grad_it}: score={best_score:.8f}")
            else:
                no_improve_g += 1
                if no_improve_g >= 30:
                    lr *= 0.6
                    momentum *= 0.3
                    no_improve_g = 0
                    if lr < 1e-5:
                        # Restart
                        a = best_a * np.exp(rng.normal(0, 0.02, n))
                        a = normalize_sum(a, n)
                        lr = 0.02
                        momentum = np.zeros(n)

    # === Phase 3: Fine-tuning with very small steps ===
    print("=== Phase 3: Fine-tuning ===")
    a = normalize_sum(best_a, n)
    momentum = np.zeros(n)
    lr = 0.003
    fine_it = 0

    while time.monotonic() < deadline + 30:
        a = normalize_sum(a, n)
        a, momentum, c, M = gradient_step(a, lr, momentum, 0.85, 1)
        a = np.maximum(a, 0.0)
        a = normalize_sum(a, n)
        fine_it += 1

        if fine_it % 30 == 0:
            sc = fast_score(a)
            if sc < best_score - 1e-10:
                best_a = a.copy()
                best_score = sc
                if fine_it % 300 == 0:
                    print(f"  Fine it={fine_it}: score={best_score:.8f}")

    # === Final validation ===
    best_a = normalize_sum(best_a, n)
    final_score = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"Final: n={n}, score={best_score:.8f}")
    return [float(x) for x in best_a]
