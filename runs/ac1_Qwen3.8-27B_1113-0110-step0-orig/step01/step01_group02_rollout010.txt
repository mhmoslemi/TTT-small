                momentum = np.zeros(n)

        elif no_improve > 200:
            eta *= 0.98
        else:
            eta = min(eta * 1.005, 0.3)
        
        it += 1
    
    return best_a, best_score

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses multi-peak subgradient descent, LP-based refinement, and multi-frequency
parametric search. Builds upon the sinusoidal modulation insight.

import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def _fft_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]

def _score(a):
    n = len(a)
    c = _fft_conv(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (s * s)

def _peak_grad(a, k):
    """Subgradient of (a*a)[k] w.r.t a."""
    n = len(a)
    k = int(k)
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        g[lo:hi+1] = 2.0 * a[k - np.arange(lo, hi + 1)]
    return g

def _build_lp_matrix(f, M):
    """Build constraint matrix for LP: (f*g)_k <= M."""
    n = len(f)
    # Sparse: row k has non-zeros at columns max(0,k-n+1)..min(n-1,k)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[j])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2*n-1, M)
    return A, b

def _lp_solve(f, M):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M."""
    n = len(f)
    A, b = _build_lp_matrix(f, M)

    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return np.ones(n) / n

        g = _lp_solve(f, M)

"""
Minimizes 2n·max(klconv(a,a))/(sum(a))^2 over non-negative sequences a.

Algorithm:
1. Multi-frequency parametric search for good initial shapes.
2. Multi-peak subgradient descent with momentum.
3. LP-dual iterative refinement (state-of-the-art approach).
4. Final subgradient polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def _fft_conv(a):
    n = len(a)
    sz = 2 * n - 1
    fs = _next_pow2(sz)
    fa = np.fft.rfft(a, fs)
    ca = np.fft.irfft(fa * fa, fs)
    return ca[:sz]

def _score(a):
    n = len(a)
    c = _fft_conv(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (s * s)

def _grad_at(a, k):
    n = len(a)
    ki = int(k)
    g = np.zeros(n)
    lo = max(0, ki - n + 1)
    hi = min(n - 1, ki)
    if lo <= hi:
        g[lo:hi + 1] = 2.0 * a[ki - np.arange(lo, hi + 1)]
    return g

def _sg_refine(a0, t_budget, rng, K=5, verbose=True):
    """Multi-peak subgradient descent."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_s = _score(a)
    eta = 0.12
    mom = np.zeros(n)
    mu = 0.7
    dl = time.monotonic() + t_budget
    it = 0
    nni = 0
    lp = time.monotonic()
    
    while time.monotonic() < dl:
        c = _fft_conv(a)
        s = 2.0 * n * float(np.max(c))
        if s < best_s:
            best_s = s
            best_a = a.copy()
            nni = 0
        else:
            nni += 1
        
        if verbose and time.monotonic() - lp > 15:
            print(f"    [SG] it={it} cur={s:.6f} best={best_s:.6f} eta={eta:.4f}")
            lp = time.monotonic()
        
        # Multi-peak gradient
        topk = np.argsort(c)[-K:]
        gt = np.zeros(n)
        for ki in topk:
            gk = _grad_at(a, ki)
            gm = gk.max()
            if gm > 1e-15:
                gt += gk / gm
        gt /= K
        gm = gt.max()
        if gm < 1e-15:
            break
        gn = gt / gm
        mom = mu * mom + (1 - mu) * gn
        a = a * np.exp(-eta * mom)
        a = np.maximum(a, 1e-20)
        a = a / a.sum()
        
        if nni > 800:
            a = best_a * np.exp(rng.normal(0, 0.08, n))
            a = np.maximum(a, 1e-20)
            a = a / a.sum()
            eta = 0.12
            mom = np.zeros(n)
            nni = 0
        elif nni > 400:
            eta *= 0.97
        elif nni < 50:
            eta = min(eta * 1.005, 0.3)
        
        it += 1
    
    return best_a, best_s

def _lp_refine(a0, t_budget, rng, verbose=True):
    """LP-dual iterative refinement."""
    n = len(a0)
    f = np.maximum(a0.astype(np.float64), 0.0)
    f = f / f.sum()
    best_f = f.copy()
    best_s = _score(f)
    dl = time.monotonic() + t_budget
    it = 0
    lp_time = time.monotonic()
    
    # Pre-compute LP indices
    j_flat = np.repeat(np.arange(n), n)
    o_flat = np.tile(np.arange(n), n)
    row_idx = j_flat + o_flat
    
    # Adaptive blend rate
    blend = 0.3
    nni = 0
    
    while time.monotonic() < dl:
        if time.monotonic() + 120 > dl:
            break
        c = _fft_conv(f)
        M = float(np.max(c))
        s = 2.0 * n * M
        if s < best_s:
            best_s = s
            best_f = f.copy()
            nni = 0
        else:
            nni += 1
        
        if verbose and time.monotonic() - lp_time > 10:
            print(f"    [LP] it={it} cur={s:.6f} best={best_s:.6f} blend={blend:.3f}")
            lp_time = time.monotonic()
        
        t0 = time.monotonic()
        # Build LP
        data = f[j_flat]
        A = csr_matrix((data, (row_idx, j_flat)), shape=(2 * n - 1, n))
        b = np.full(2 * n - 1, M)
        c_obj = -np.ones(n)
        bounds = [(0, None)] * n
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        dt = time.monotonic() - t0
        
        if not res.success:
            if verbose:
                print(f"    [LP] LP failed: {res.message}")
            break
        
        g = res.x
        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs
        
        # Adaptive blend
        f = (1 - blend) * f + blend * g
        f = np.maximum(f, 0.0)
        fs = f.sum()
        if fs < 1e-15:
            break
        f = f / fs
        
        # Adjust blend rate
        new_s = 2.0 * n * float(np.max(_fft_conv(f)))
        if new_s < best_s:
            blend = min(blend * 1.1, 0.7)
        elif nni > 5:
            blend *= 0.9
        blend = max(blend, 0.05)
        
        if verbose:
            print(f"    [LP] it={it} dt={dt:.2f}s new={new_s:.6f} best={best_s:.6f}")
        
        it += 1
    
    return best_f, best_s
