```python
"""
SCA (Successive Convex Approximation) + Smoothed Gradient optimizer
for minimizing J(a) = 2n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Key ideas:
1. SCA: Linearize the quadratic constraints (a*a)[k] <= t at current iterate,
   solve the resulting LP for the optimal step.
2. Smoothed gradient: Use softmax-weighted subgradient for fine refinement.
3. Multiple sequence lengths with interpolation between them.
4. Adaptive damping and random restarts to escape local minima.
"""
import numpy as np
import time
import math
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import spsolve


def _fft_convolve(a):
    """Fast full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]


def _score(a):
    """Compute J(a) = 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)


def _build_sca_matrix(a, n):
    """Build the sparse constraint matrix for SCA LP.
    Row k: [2*a[k-j] for valid j, -1] (coefficient of t)
    """
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            idx = k - j
            if 0 <= idx < n:
                rows.append(k)
                cols.append(j)
                data.append(2.0 * a[idx])
        # Coefficient of t (column n) is -1
        rows.append(k)
        cols.append(n)
        data.append(-1.0)
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n + 1))
    return A


def _sca_lp_solve(a, n, deadline):
    """Solve one SCA LP iteration. Returns (a_new, t_new) or None."""
    c = _fft_convolve(a)
    M = float(np.max(c))
    
    # Build sparse A_ub
    A = _build_sca_matrix(a, n)
    b_ub = c.copy()
    
    # Objective: minimize t (last variable)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Equality: sum(a') = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(-np.inf, None)]
    
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': max(1, deadline - time.monotonic())})
        if result.status == 0:
            return result.x[:n], result.x[n]
    except Exception:
        pass
    return None, None


def _sca_optimize(a_init, n, max_iters, deadline, verbose=True):
    """Run SCA optimization for a sequence of length n."""
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_score = _score(a)
    t_damp = 1.0
    no_improve_count = 0
    last_improve_score = best_score
    
    if verbose:
        print(f"    SCA start: n={n}, score={best_score:.6f}")
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        a_lp, t_lp = _sca_lp_solve(a, n, deadline)
        if a_lp is None:
            if verbose:
                print(f"    SCA iter {it}: LP failed, stopping")
            break
        
        # Ensure non-negativity
        a_lp = np.maximum(a_lp, 0.0)
        s_lp = float(np.sum(a_lp))
        if s_lp < 1e-10:
            break
        a_lp = a_lp / s_lp
        
        # Damped update with backtracking
        found_better = False
        for damp in [t_damp, 0.5 * t_damp, 0.25 * t_damp, 0.1 * t_damp]:
            a_trial = (1.0 - damp) * a + damp * a_lp
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = float(np.sum(a_trial))
            if s_trial < 1e-10:
                continue
            a_trial = a_trial / s_trial
            score_trial = _score(a_trial)
            if score_trial < best_score - 1e-12:
                a = a_trial
                best_score = score_trial
                best_a = a.copy()
                found_better = True
                break
        
        if found_better:
            no_improve_count = 0
            last_improve_score = best_score
            # Gradually reduce damping
            t_damp = max(0.3, t_damp * 0.98)
        else:
            no_improve_count += 1
            t_damp *= 0.8
            if no_improve_count > 50:
                # Try a random perturbation
                if verbose:
                    print(f"    SCA iter {it}: stuck at {best_score:.6f}, perturbing")
                a = a + 0.05 * np.random.default_rng(it).standard_normal(n)
                a = np.maximum(a, 0.0)
                s = float(np.sum(a))
                a = a / s
                t_damp = 1.0
                no_improve_count = 0
        
        if verbose and it % 10 == 0:
            print(f"    SCA iter {it}: score={best_score:.6f}, t_lp={t_lp:.6e}, damp={t_damp:.3f}")
    
    if verbose:
        print(f"    SCA final: n={n}, score={best_score:.6f}")
    return best_a, best_score


def _smoothed_gradient(a, n, T, top_k=20):
    """Compute smoothed gradient using softmax weighting over top peaks."""
    c = _fft_convolve(a)
    # Use top-k peaks for efficiency
    if top_k < len(c):
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        c_top = c[top_idx]
        # Softmax with temperature T
        c_shifted = c_top - c_top.max()
        w = np.exp(T * c_shifted)
        w = w / w.sum()
    else:
        top_idx = np.arange(len(c))
        c_shifted = c - c.max()
        w = np.exp(T * c_shifted)
        w = w / w.sum()
    
    # Gradient: weighted sum of subgradients
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        if w[i] < 1e-12:
            continue
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            idx = k - j
            if 0 <= idx < n:
                grad[j] += w[i] * 2.0 * a[idx]
    return grad, float(np.max(c))


def _smoothed_grad_optimize(a_init, n, max_iters, deadline, lr_init=0.05, lr_decay=0.999, T=20.0, top_k=15):
    """Smoothed gradient descent with simplex projection."""
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = _score(a)
    lr = lr_init
    T_current = T
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        grad, max_c = _smoothed_gradient(a, n, T_current, top_k)
        
        # Gradient step
        a_new = a - lr * grad
        # Project onto simplex {a >= 0, sum(a) = 1}
        a_new = _project_simplex(a_new)
        
        score = _score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        lr *= lr_decay
        # Anneal temperature
        T_current = max(5.0, T_current * 0.999)
    
    return best_a, best_score


def _project_simplex(v):
    """Project vector v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def _interpolate_sequence(src, n_target):
    """Interpolate a sequence to a new length."""
    n_src = len(src)
    if n_src == n_target:
        return src.copy()
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    out = np.interp(x_tgt, x_src, src)
    out = np.maximum(out, 0.0)
    s = float(np.sum(out))
    if s > 1e-10:
        out = out / s
    return out


def _make_structured_init(n, rng, style=0):
    """Create diverse structured initializations."""
    t = np.linspace(0, 1, n)
    if style == 0:
        a = np.ones(n)
    elif style == 1:
        a = 1.0 - 0.5 * t  # Linear taper
    elif style == 2:
        a = np.exp(-2.0 * t)  # Exponential decay
    elif style == 3:
        # Asymmetric bump
        center = 0.3
        a = np.exp(-((t - center) ** 2) / 0.05)
    elif style == 4:
        # Two bumps
        a = np.exp(-((t - 0.2) ** 2) / 0.03) + 0.7 * np.exp(-((t - 0.6) ** 2) / 0.05)
    elif style == 5:
        # Spiky
        a = np.abs(rng.normal(0, 1, n))
    elif style == 6:
        # Cosine taper
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 0.01)
    elif style == 7:
        # Power law
        a = (1.0 - t) ** 0.5
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s > 1e-10:
        a = a / s
    return a


def propose_candidate():
    """
    Multi-phase optimizer using SCA (Linearized LP) and smoothed gradient descent
    to minimize J(a) = 2n*max(conv(a,a))/sum(a)^2.
    
    Phase 1: SCA/LP at multiple lengths (200, 400, 500, 800, 1000)
    Phase 2: Smoothed gradient refinement
    Phase 3: Final LP polish
    """
    total_deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(12345)
    
    best_a = None
    best_score = float('inf')
    
    # Try parent sequence
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            parent_score = _score(parent)
            print(f"Parent: n={parent.size}, score={parent_score:.6f}")
            best_a = parent.copy()
            best_score = parent_score
    except Exception:
        parent = None
    
    # Phase 1: SCA at various lengths
    print("\n=== Phase 1: SCA Optimization ===")
    n_list = [200, 300, 400, 500, 700, 1000]
    sca_deadline_frac = 0.65  # Use 65% of time for SCA
    
    for idx_n, n in enumerate(n_list):
        phase_deadline = min(total_deadline, time.monotonic() + 250.0)
        
        if time.monotonic() > total_deadline * (1 - 0.3):
            # Save time for later phases
            if idx_n > 3:
                break
        
        print(f"\n  --- SCA n={n} ---")
        
        # Multiple initializations
        inits = []
        if best_a is not None:
            inits.append(_interpolate_sequence(best_a, n))
        if parent is not None:
            inits.append(_interpolate_sequence(parent, n))
        inits.append(_make_structured_init(n, rng, 1))
        inits.append(_make_structured_init(n, rng, 3))
        inits.append(_make_structured_init(n, rng, 5))
        
        for init_idx, a_init in enumerate(inits):
            if time.monotonic() > phase_deadline:
                break
            label = f"init{init_idx}"
            a_opt, score = _sca_optimize(a_init, n, 500, phase_deadline, verbose=(init_idx == 0))
            if score < best_score:
                best_score = score
                best_a = a_opt.copy()
                print(f"  ** New best: {best_score:.6f} (n={n}, {label})")
        
        print(f"  Best after n={n}: {best_score:.6f}")
    
    print(f"\nAfter Phase 1: best={best_score:.6f}, n={len(best_a) if best_a is not None else 0}")
    
    # Phase 2: Smoothed gradient refinement
    print("\n=== Phase 2: Smoothed Gradient ===")
    if best_a is not None and time.monotonic() < total_deadline - 200:
        phase_deadline = min(total_deadline, time.monotonic() + 400.0)
        n_best = len(best_a)
        # Try at the best n and a couple nearby
        for n in [n_best, min(n_best + 200, 1200), min(n_best, 800)]:
            if time.monotonic() > phase_deadline:
                break
            a_init = _interpolate_sequence(best_a, n) if len(best_a) != n else best_a.copy()
            a_opt, score = _smoothed_grad_optimize(a_init, n, 50000, phase_deadline,
                                                   lr_init=0.03, lr_decay=0.9995, T=30.0, top_k=12)
            if score < best_score:
                best_score = score
                best_a = a_opt.copy()
                print(f"  ** Smoothed grad improved: {best_score:.6f} (n={n})")
    
    # Phase 3: Final SCA polish on the best solution
    print("\n=== Phase 3: Final SCA Polish ===")
    if best_a is not None and time.monotonic() < total_deadline - 50:
        n_best = len(best_a)
        polish_deadline = total_deadline - 10
        
        # Polish at the best n
        a_opt, score = _sca_optimize(best_a.copy(), n_best, 200, polish_deadline, verbose=True)
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  ** Polish improved: {best_score:.6f}")
        
        # Also try at slightly different n
        for n_try in [max(200, n_best - 100), min(1500, n_best + 100)]:
            if time.monotonic() > polish_deadline:
                break
            a_init = _interpolate_sequence(best_a, n_try)
            a_opt, score = _sca_optimize(a_init, n_try, 150, polish_deadline, verbose=False)
            if score < best_score:
                best_score = score
                best_a = a_opt.copy()
                print(f"  ** Polish at n={n_try}: {best_score:.6f}")
    
    # Phase 4: One last smoothed gradient pass
    if best_a is not None and time.monotonic() < total_deadline - 20:
        n_best = len(best_a)
        a_opt, score = _smoothed_grad_optimize(best_a.copy(), n_best, 20000,
                                               total_deadline - 10, lr_init=0.01,
                                               lr_decay=0.9999, T=50.0, top_k=8)
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  ** Final gradient: {best_score:.6f}")
    
    # Final result
    print(f"\n=== FINAL: score={best_score:.6f}, n={len(best_a) if best_a is not None else 0} ===")
    
    if best_a is not None:
        result = [float(x) for x in best_a]
        # Verify with evaluate_sequence
        try:
            verified = float(evaluate_sequence(result))
            print(f"Verified score: {verified:.6f}")
        except Exception as e:
            print(f"Verification error: {e}")
        return result
    
    # Fallback
    return [1.0] * 100
```
