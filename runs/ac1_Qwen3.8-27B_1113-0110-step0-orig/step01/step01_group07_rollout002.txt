"""
Optimized L-BFGS + subgradient search for minimizing max-convolution score.
Uses smooth-max with geometric temperature annealing, log-domain parameterization,
and scipy L-BFGS-B for fast convergence. Multiple restarts and subgradient polishing.

import time
import numpy as np
from scipy.optimize import minimize


def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = np.where(np.isfinite(z), z, -30.0)
    z = z - z.max()
    z = np.clip(z, -30.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones_like(z) / z.size
    return e / s


def compute_conv(a, L):
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:2 * a.size - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    return np.maximum(c, 0.0)


def smooth_max_val(c, tau):
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0.0:
        return cm
    z = (c - cm) / tau
    expz = np.exp(np.clip(z, -50.0, 0.0))
    expsum = float(expz.sum())
    if not np.isfinite(expsum) or expsum <= 0.0:
        expz = np.zeros_like(c)
        expz[int(np.argmax(c))] = 1.0
        expsum = 1.0
    return cm + tau * float(np.log(expsum)), expz / expsum


def objective_and_grad(z, n, L, tau):
    a = softmax(z)
    c = compute_conv(a, L)
    sm_val, weights = smooth_max_val(c, tau)
    val = 2.0 * n * sm_val

    # Gradient w.r.t. a
    W = weights[::-1]
    Wp = np.zeros(L)
    Wp[:W.size] = W
    HW = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(Wp, L), L)
    g_a = 2.0 * HW[n - 1:2 * n - 1][::-1]
    g_a = np.where(np.isfinite(g_a), g_a, 0.0)

    # Gradient w.r.t. z (logit space)
    mean_g = float(np.dot(a, g_a))
    g_z = a * (g_a - mean_g)
    g_z = np.where(np.isfinite(g_z), g_z, 0.0)

    # Clip gradient for stability
    mg = float(np.max(np.abs(g_z)))
    if mg > 1.0:
        g_z *= 1.0 / mg

    return val, g_z


def objective_and_grad_true(z, n, L):
    """Compute true (non-smooth) score for tracking."""
    a = softmax(z)
    c = compute_conv(a, L)
    cm = float(c.max())
    return 2.0 * n * cm


def true_score_from_a(a):
    n = a.size
    L = next_pow2(2 * n - 1)
    c = compute_conv(a, L)
    return 2.0 * n * float(c.max())


def normalize_a(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        return None
    return a / s


def a_to_z(a):
    a = np.maximum(a, 1e-300)
    z = np.log(a)
    z -= z.mean()
    z = np.clip(z, -30.0, 30.0)
    return z


def lgbfgs_phase(z, n, L, tau, max_iter, tol, max_time, label=""):
    best_z = z.copy()
    best_true = objective_and_grad_true(z, n, L)
    start = time.monotonic()
    state = {"calls": 0}

    def fun(zv):
        state["calls"] += 1
        v, g = objective_and_grad(zv, n, L, tau)
        if not np.isfinite(v):
            return 1e10, np.zeros_like(zv)
        return v, g

    res = minimize(
        fun, z.copy(), method="L-BFGS-B",
        options={
            "maxiter": max_iter,
            "ftol": tol,
            "gtol": tol,
            "maxls": 50,
            "maxfun": max_iter * 100,
        }
    )

    elapsed = time.monotonic() - start
    print(f"  {label}: calls={state['calls']} time={elapsed:.1f}s "
          f"fun={res.fun:.6f} success={res.success} msg={res.message[:40]}")

    # Evaluate true score at the result
    final_z = res.x if np.all(np.isfinite(res.x)) else z
    final_true = objective_and_grad_true(final_z, n, L)

    if final_true < best_true:
        best_true = final_true
        best_z = final_z.copy()

    return best_z, best_true


def subgradient_polish(a, n, L, n_steps, eta0, label=""):
    """Polish with subgradient steps on multiple active lags."""
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_score = true_score_from_a(a)
    eta = eta0

    for step in range(n_steps):
        c = compute_conv(a, L)
        cm = c.max()
        if not np.isfinite(cm) or cm <= 0:
            break
        score = 2.0 * n * cm
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Top-k active lags
        topk = min(10, c.size)
        idx = np.argpartition(c, -topk)[-topk:]
        weights = c[idx] / c[idx].sum()

        # Combined gradient
        g = np.zeros(n)
        for w, k in zip(weights, idx):
            for j in range(n):
                kk = k - j
                if 0 <= kk < n:
                    g[j] += 2.0 * w * a[kk]

        # Update
        g = np.where(np.isfinite(g), g, 0.0)
        a_new = a - eta * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s <= 0:
            a_new = a.copy()
            s = 1.0
        a_new /= s

        # Adaptive step
        s_new = true_score_from_a(a_new)
        if s_new < score * (1 - 0.001):
            a = a_new
            eta = min(eta * 1.05, 0.5)
        else:
            eta *= 0.7

    print(f"  subgrad {label}: final={best_score:.6f}")
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0, **kwargs):
    rng = np.random.default_rng(seed)

    raw_budget = max(0.0, float(budget_s))

        hs = np.maximum(hs, 0.0)
        if hs.size > 0 and float(hs.sum()) > 1e-9:

            s0 = true_score_from_a(a0)
            if np.isfinite(s0) and s0 < best_score:
                best_score = s0
                best_a = a0
    except Exception:
        pass

    if best_a is None:
        best_a = np.ones(N) / N
        best_score = 2.0

    print(f"Initial score: {best_score:.6f} (N={N})")

    # Try to avoid the symmetry trap: perturb the initial point
    z = a_to_z(best_a)

    # Temperature annealing schedule
    tau_schedule = [N * 0.15, N * 0.08, N * 0.04, N * 0.02, N * 0.01, N * 0.005, N * 0.002, N * 0.001]
    max_iters_per_phase = [800, 1000, 1200, 1500, 2000, 2000, 2500, 3000]
    tol = 1e-8

    total_time_used = 0
    for phase, (tau, mi) in enumerate(zip(tau_schedule, max_iters_per_phase)):
        elapsed = time.monotonic() - start_time
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Time per phase: proportional to remaining fraction
        time_frac = 0.10 if phase < 2 else (0.14 if phase < 5 else 0.18)
        time_limit = min(remaining * 0.85, raw_budget * time_frac)
        time_limit = max(time_limit, 20.0)

        try:
            z, true_s = lgbfgs_phase(z, N, L, tau, mi, tol, time_limit, f"phase{phase}(tau={tau:.1f})")
        except Exception as e:
            print(f"  phase{phase} error: {e}")
            break

        a_candidate = softmax(z)
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate /= a_candidate.sum()
        sc = true_score_from_a(a_candidate)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_candidate.copy()
            print(f"  New best: {best_score:.6f}")

    # Subgradient polishing
    elapsed = time.monotonic() - start_time
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"Subgradient polishing, {remaining:.0f}s remaining")
        for polish in range(3):
            if time.monotonic() > deadline - 30:
                break
            n_steps = min(2000, int((deadline - time.monotonic()) * 50))
            a_polished, s_polished = subgradient_polish(
                best_a, N, L, n_steps, eta0=0.02, label=f"r{polish}"
            )
            if s_polished < best_score:
                best_score = s_polished
                best_a = a_polished.copy()

    # Try one more L-BFGS with very low tau from the best
    remaining = deadline - time.monotonic()
    if remaining > 40:
        z_best = a_to_z(best_a)
        tau_final = N * 0.0005
        try:
            z_f, s_f = lgbfgs_phase(z_best, N, L, tau_final, 3000, 1e-10, min(remaining - 30, 120), "final")
            a_f = softmax(z_f)
            a_f = np.maximum(a_f, 0.0)
            a_f /= a_f.sum()
            sc_f = true_score_from_a(a_f)
            if sc_f < best_score:
                best_score = sc_f
                best_a = a_f.copy()
        except Exception as e:
            print(f"  final phase error: {e}")

    # Also try a different length (smaller) to see if it gives better score
    for N2 in [4096, 16384]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break
        L2 = next_pow2(2 * N2 - 1)
        try:
            a2 = np.interp(np.linspace(0, 1, N2), np.linspace(0, 1, best_a.size), best_a)
            a2 = np.maximum(a2, 0.0)
            sa2 = a2.sum()
            if sa2 <= 0:
                continue
            a2 /= sa2
            s2_init = true_score_from_a(a2)
            z2 = a_to_z(a2)
            tau2 = N2 * 0.01
            z2, s2 = lgbfgs_phase(z2, N2, L2, tau2, 2000, 1e-9, min(remaining - 40, 90), f"N={N2}")
            a2 = softmax(z2)
            a2 = np.maximum(a2, 0.0)
            a2 /= a2.sum()
            s2 = true_score_from_a(a2)
            if s2 < best_score:
                best_score = s2
                best_a = a2.copy()
                N = N2
                print(f"  N={N2} improved: {best_score:.6f}")
        except Exception as e:
            print(f"  N={N2} error: {e}")

    # Final: build the output list
    best_a = np.maximum(best_a, 0.0)
    best_a = np.minimum(best_a, 1000.0)
    out = [float(x) for x in best_a]

    # Validate with the actual evaluator on the best and a fallback
    final_seq = out
    final_score = np.inf
    candidates = [out]

    candidates.append([1.0 / N] * N)

    for cand in candidates:

            sc = evaluate_sequence(cand)

            sc = np.inf
        if np.isfinite(sc) and sc < final_score:
            final_score = sc
            final_seq = cand

    print(f"Final best: {final_score:.6f} (N={len(final_seq)})")
    return final_seq
