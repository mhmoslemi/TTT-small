import numpy as np
import time
from scipy.optimize import linprog, minimize
import math

def fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]

def score(a):
    n = len(a)
    c = fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_conv_matrix(f):
    n = len(f)
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = f
    return C

def lp_solve(f, M):
    n = len(f)
    C = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.status == 0:
            g = result.x
            s = float(np.sum(g))
            if s > 1e-15:
                return g / s, s
    except Exception:
        pass
    return None, 0

def alternating_lp(a_init, max_iters, deadline, blend_start=0.7, blend_decay=0.98):
    a = normalize(a_init.astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score(a)
    blend = blend_start
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = fft_convolve(a)
        M = float(np.max(c))
        g, s_g = lp_solve(a, M)
        if g is None:
            break
        # Try multiple blend factors
        best_local = a.copy()
        best_local_score = best_score
        for t in [blend, blend*0.5, blend*1.5, 0.3, 0.5]:
            t = min(max(t, 0.05), 0.98)
            a_new = (1 - t) * a + t * g
            a_new = normalize(a_new)
            s_new = score(a_new)
            if s_new < best_local_score:
                best_local_score = s_new
                best_local = a_new
        a = best_local
        if best_local_score < best_score:
            best_score = best_local_score
            best_a = a.copy()
            if it % 50 == 0:
                print(f"    LP iter {it}: score={best_score:.6f}")
        blend = max(0.1, blend * blend_decay)
        # Check convergence
        if it > 50 and abs(score(a) - best_score) < 1e-10:
            break
    return best_a, best_score

def smooth_max_conv(a, t=50):
    """Smooth approximation of max(conv(a,a))."""
    c = fft_convolve(a)
    # Use log-sum-exp for numerical stability
    tc = t * c
    tc_max = np.max(tc)
    log_sum_exp = tc_max + np.log(np.sum(np.exp(tc - tc_max)))
    return log_sum_exp / t

def smooth_max_grad(a, t=50):
    """Gradient of smooth max of conv(a,a) with respect to a."""
    n = len(a)
    c = fft_convolve(a)
    tc = t * c
    tc_max = np.max(tc)
    weights = np.exp(tc - tc_max) / np.exp(tc_max - tc_max)  # softmax
    weights = weights / np.sum(weights)
    # Gradient of sum_k w_k * conv[k] with respect to a[j]
    grad = np.zeros(n)
    for k in range(2*n-1):
        if weights[k] < 1e-15:
            continue
        for j in range(n):
            i1 = k - j
            i2 = j - k
            if 0 <= i1 < n:
                grad[j] += weights[k] * a[i1]
            if 0 <= i2 < n and i2 != i1:
                grad[j] += weights[k] * a[i2]
    return grad

Multi-phase optimizer minimizing J(a) = 2n*max(conv(a,a))/sum(a)^2.
Uses coordinate descent, alternating LP, and smooth-max gradient methods.
"""

import numpy as np
import time
import math
from scipy.optimize import linprog, minimize_scalar

def fft_conv(a):
    n = len(a)
    size = 2 * n - 1
    fs = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fs)
    return np.fft.irfft(A * A, fs)[:size]

def score(a):
    n = len(a)
    c = fft_conv(a)
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return float(2.0 * n * np.max(c) / (s * s))

def norm1(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def interp_seq(src, n_t):
    if len(src) == n_t:
        return src.copy()
    x = np.linspace(0, 1, len(src))
    y = np.linspace(0, 1, n_t)
    out = np.interp(y, x, src)
    return norm1(out)

def build_C(f):
    n = len(f)
    C = np.zeros((2*n-1, n))
    for j in range(n):
        C[j:j+n, j] = f
    return C

def lp_fix(f, M):
    """Solve max sum(g) s.t. C_f g <= M, g >= 0. Return normalized g or None."""
    n = len(f)
    C = build_C(f)
    res = linprog(-np.ones(n), A_ub=C, b_ub=M*np.ones(2*n-1),
                  bounds=[(0,None)]*n, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.status == 0:
        s = float(np.sum(res.x))
        if s > 1e-15:
            return res.x / s
    return None

def alt_lp(a0, deadline, max_it=500):
    a = norm1(a0.copy())
    n = len(a)
    best = a.copy()
    best_s = score(a)
    blend = 0.7
    for it in range(max_it):
        if time.monotonic() > deadline:
            break
        c = fft_conv(a)
        M = float(np.max(c))
        g = lp_fix(a, M)
        if g is None:
            break
        # Try multiple blends
        improved = False
        for t in [blend, 0.3, 0.5, 0.8]:
            a_new = norm1((1-t)*a + t*g)
            s_new = score(a_new)
            if s_new < best_s - 1e-12:
                best_s = s_new
                best = a_new.copy()
                improved = True
            if s_new <= score(a):
                a = a_new
                break
        if not improved:
            a = norm1((1-blend)*a + blend*g)
        blend = max(0.15, blend * 0.97)
    return best, best_s

def coord_desc(a0, deadline, n_passes_max=5000):
    """Coordinate descent: optimize one coordinate at a time."""
    a = norm1(a0.copy())
    n = len(a)
    best = a.copy()
    best_s = score(a)
    rng = np.random.default_rng(123)
    
    for p in range(n_passes_max):
        if time.monotonic() > deadline:
            break
        for j in rng.permutation(n):
            # Compute contribution of a[j] to each conv[k]
            # conv[k] = base_k + f_k * t + g_k * t^2 where t = a[j]
            aj = a[j]
            f = np.zeros(2*n-1)
            g = np.zeros(2*n-1)
            ki = np.arange(2*n-1) - j  # k - j for all k
            valid = (ki >= 0) & (ki < n)
            # For valid k (where k-j is in [0,n-1]):
            if np.any(valid):
                if j < n:  # always true
                    # g[k] = 1 if k = 2j
                    k2j = 2 * j
                    if 0 <= k2j < 2*n-1:
                        g[k2j] = 1.0
                    # f[k] = 2*a[k-j] if k-j != j, else 0 (since g handles that)
                    mask = valid & (ki != j)
                    f[mask] = 2.0 * a[ki[mask]]
            
            # base = conv_full - f*aj - g*aj^2
            c_full = fft_conv(a)
            base = c_full - f * aj - g * aj * aj
            
            # Minimize max(base + f*t + g*t^2) over t >= 0
            def obj(t):
                if t < 0:
                    return 1e10
                return float(np.max(base + f * t + g * t * t))
            
            # Bounded: t in [0, min(aj*5+1, 5)]
            hi = min(aj * 5 + 0.5, 5.0)
            res = minimize_scalar(obj, bounds=(0.0, hi), method='bounded',
                                  options={'xatol': 1e-8})
            if res.success:
                a[j] = max(0.0, res.x)
        
        a = norm1(a)
        s = score(a)
        if s < best_s:
            best_s = s
            best = a.copy()
    
    return best, best_s

Wait, I'm computing c_full = fft_conv(a) inside the inner loop, which is O(n log n) per coordinate. For n=500 and 500 coordinates per pass, that's 500 * O(500 log 500) ≈ 2.25M per pass. With 1000 passes, that's 2.25B. Too slow.

I need to avoid recomputing the full convolution at each coordinate. Let me precompute it:
