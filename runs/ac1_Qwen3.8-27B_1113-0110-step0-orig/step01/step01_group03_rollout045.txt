def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    c = np.fft.irfft(fa * fb, size)
    return c[:n + m - 1]

def compute_soft_max_weighted(c, alpha):
    """Compute soft_max value and softmax weights."""
    c_max = c.max()
    shifted = alpha * (c - c_max)
    # Clip to avoid underflow
    shifted = np.maximum(shifted, -50)
    exp_vals = np.exp(shifted)
    sum_exp = exp_vals.sum()
    w = exp_vals / sum_exp
    soft_max_val = c_max + np.log(sum_exp) / alpha
    return soft_max_val, w

def compute_gradient(a, c, w, soft_max_val):
    """Compute gradient of score = 2*n*soft_max(c)/S^2 wrt a."""
    n = len(a)
    S = a.sum()
    # cross_correlation: grad_c[i] = sum_k w[k] * dc_k/da_i = 2 * sum_m w[m+i] * a[m]
