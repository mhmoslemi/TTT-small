import time
from scipy.optimize import linprog
from scipy import sparse

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]

def _build_lp_matrix(f, n):
    """Build the sparse constraint matrix for (f*g)[k] for all k."""
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            rows.append(k)
            cols.append(k - i)
            vals.append(f[i])
    
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def _lp_solve(f, M, n, A_matrix=None):
    """Solve LP: max sum(g) s.t. g >= 0, A@g <= M."""
    if A_matrix is None:
        A_matrix = _build_lp_matrix(f, n)
    
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A_matrix, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x, -result.fun
    return None, None

    best_a = a.copy()

`= sum_k p[k] * 2*a[k-j]` (where p[k] = softmax(conv/T))

This gives a weighted combination of all the convolution positions, weighted by their softmax probability. When T is small, this concentrates on the maximum; when T is large, it's more uniform.

This might be smoother and avoid getting stuck.

import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using LP-based SOTA iterations combined
with soft-max subgradient descent for exploration. The LP approach guarantees
non-increasing score at each step, while soft-max subgradient provides smoother
gradients to escape local minima.
"""


def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]


def _build_lp_matrix(f, n):
    """Build sparse matrix A where (A@g)[k] = (f*g)[k]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            rows.append(k)
            cols.append(k - i)
            vals.append(f[i])
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _subgradient_vec(a, k, n):
    """Subgradient of (a*a)[k] w.r.t. a[j]."""
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        j_range = np.arange(lo, hi + 1)
        indices = k - j_range
        g[lo:hi + 1] = 2.0 * a[indices]
    return g


def _soft_max_subgradient(a, c, T, n, top_k=20):
    """Compute soft-max weighted subgradient of the convolution peak."""
    # Use top-k positions with softmax weighting
    if len(c) <= top_k:
        weights = np.exp(c / T)
        weights = weights / weights.sum()
    else:
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        top_vals = c[top_idx]
        w_top = np.exp(top_vals / T)
        w_top = w_top / w_top.sum()
        weights = np.zeros(len(c))
        weights[top_idx] = w_top

    # Gradient: sum_k weights[k] * 2*a[k-j] for each j
    grad = np.zeros(n)
    for k in np.nonzero(weights > 1e-15)[0]:
        w = weights[k]
        if w < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            j_range = np.arange(lo, hi + 1)
            indices = k - j_range
            grad[lo:hi + 1] += w * 2.0 * a[indices]
    return grad


def _optimize_lp(a0, time_budget, rng, verbose=True):
    """LP-based optimization (SOTA approach)."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, np.inf
    a = a / s

    c = _fft_convolve(a)
    best_score = 2.0 * n * float(np.max(c))
    best_a = a.copy()

    deadline = time.monotonic() + time_budget
    it = 0
    no_improve = 0
    last_print = time.monotonic()
    t_blend = 0.15

    while time.monotonic() < deadline:
        c = _fft_convolve(a)
        M = float(np.max(c))
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            if verbose and time.monotonic() - last_print > 5.0:
                print(f"    [LP] it={it}, score={score:.8f}, M={M:.8f}")
                last_print = time.monotonic()
        else:
            no_improve += 1

        if no_improve > 300:
            a = best_a * np.exp(rng.normal(0, 0.03, n))
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            no_improve = 0
            t_blend = 0.15
            it += 1
            continue

        # Build LP matrix
        A_mat = _build_lp_matrix(a, n)

        # Solve LP: max sum(g) s.t. A@g <= M, g >= 0
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        result = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})

        if not result.success or result.x is None:
            a = best_a.copy()
            a += rng.normal(0, 0.02, n)
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            no_improve = 0
            it += 1
            continue

        g = result.x
        sum_g = float(g.sum())
        if sum_g < 1e-15:
            a = best_a.copy()
            a += rng.normal(0, 0.02, n)
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            it += 1
            continue

        # Rescale g to unit sum
        g = g / sum_g

        # Adaptive blend parameter
        if no_improve < 30:
            t_blend = min(0.3, 0.1 + 0.01 * it)
        else:
            t_blend = max(0.02, 0.15 * (0.98 ** no_improve))

        # Blend
        a = (1.0 - t_blend) * a + t_blend * g
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        it += 1

    return best_a, best_score


def _optimize_softmax(a0, time_budget, rng, verbose=True):
    """Soft-max subgradient descent with adaptive temperature."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, np.inf
    a = a / s

    c = _fft_convolve(a)
    best_score = 2.0 * n * float(np.max(c))
    best_a = a.copy()

    T = 0.001  # Temperature for soft-max
    eta = 0.1
    no_improve = 0
    momentum = np.zeros(n)
    mu = 0.7
    deadline = time.monotonic() + time_budget
    it = 0
    last_print = time.monotonic()
    top_k = min(50, n // 4)

    while time.monotonic() < deadline:
        c = _fft_convolve(a)
        M = float(np.max(c))
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            if verbose and time.monotonic() - last_print > 8.0:
                print(f"    [SM] it={it}, score={score:.8f}, T={T:.6f}")
                last_print = time.monotonic()
        else:
            no_improve += 1

        if no_improve > 500:
            a = best_a * np.exp(rng.normal(0, 0.05, n))
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            no_improve = 0
            momentum = np.zeros(n)
            eta = 0.1
            T = 0.001
            it += 1
            continue

        # Compute soft-max subgradient
        grad = _soft_max_subgradient(a, c, T, n, top_k)
        g_max = grad.max()
        if g_max < 1e-15:
            break
        grad = grad / g_max
        momentum = mu * momentum + (1.0 - mu) * grad

        # Multiplicative update
        a = a * np.exp(-eta * momentum)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        # Adaptive parameters
        if no_improve > 300:
            eta *= 0.97
            T *= 0.9
        elif no_improve < 50:
            eta = min(eta * 1.005, 0.3)
            T = min(T * 1.02, 0.05)

        it += 1

    return best_a, best_score

            it += 1
            continue

        if soft_it % lp_iters == 0:

            A_mat = _build_lp_matrix(a, n)
            c_obj = -np.ones(n)
            b_ub = np.full(2 * n - 1, M)

                sum_g = float(g.sum())

                    a = (1.0 - t_blend) * a + t_blend * g

            if g_max > 1e-15:

                a = a / a.sum()

            eta *= 0.97
            T *= 0.95
        elif no_improve < 30:
            eta = min(eta * 1.003, 0.25)

        soft_it += 1
        it += 1

    return best_a, best_score
