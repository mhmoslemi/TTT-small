import numpy as np
import time

try:
    from scipy.optimize import minimize as _scipy_minimize
except Exception:
    _scipy_minimize = None


def _pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _clean(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    return np.maximum(v, 0.0)


def _normalize(v):
    v = _clean(v)
    s = float(v.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.full(v.size, 1.0 / max(1, v.size), dtype=np.float64)
    return v / s


def _project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    v = _clean(v)
    n = v.size
    if n == 0:
        return v.copy()
    x = np.sort(v)[::-1]
    css = np.cumsum(x)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = x > (css - 1.0) / idx
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    r = int(np.nonzero(cond)[0][-1])
    tau = (float(css[r]) - 1.0) / float(r + 1)
    out = np.maximum(v - tau, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def _score(a):
    """Compute 2n*max(conv(a,a))/(sum a)^2. Assumes a >= 0."""
    a = _clean(a)
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 1e-12:
        return np.inf
    a = a / s
    L = _pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    mx = float(np.max(c))
    return 2.0 * n * mx if np.isfinite(mx) else np.inf


def _conv_full(a):
    """Full linear convolution using FFT. Returns cleaned non-negative result."""
    a = _clean(a)
    n = a.size
    L = _pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    return np.maximum(c, 0.0)

def _pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _clean(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    return np.maximum(v, 0.0)


def _normalize(v):
    v = _clean(v)
    s = float(v.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.full(v.size, 1.0 / max(1, v.size), dtype=np.float64)
    return v / s


def _project_simplex(v):
    v = _clean(v)
    n = v.size
    if n == 0:
        return v.copy()
    x = np.sort(v)[::-1]
    css = np.cumsum(x)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = x > (css - 1.0) / idx
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    r = int(np.nonzero(cond)[0][-1])
    tau = (float(css[r]) - 1.0) / float(r + 1)
    out = np.maximum(v - tau, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def _score_vec(a):
    a = _clean(a)
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 1e-12:
        return np.inf
    a = a / s
    L = _pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    mx = float(np.max(c))
    return 2.0 * n * mx if np.isfinite(mx) else np.inf


def _conv(a):
    a = _clean(a)
    n = a.size
    L = _pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _gen_power(n, p, offset=1.0):
    j = np.arange(n, dtype=np.float64)
    v = np.power(np.maximum(j + offset, 1e-8), -p)
    return _project_simplex(v)


def _gen_bimodal(n, center1, center2, width):
    j = np.arange(n, dtype=np.float64)
    v = np.exp(-((j - center1) ** 2) / (2 * width ** 2)) + np.exp(-((j - center2) ** 2) / (2 * width ** 2))
    v += 1e-4
    return _project_simplex(v)


def _gen_u_shaped(n, p=0.5):
    j = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    v = np.power(np.abs(j - mid) / mid + 0.1, -p)
    return _project_simplex(_clean(v))


def _gen_sine_mod(n, p=0.5, amp=0.5, freq=1.0):
    j = np.arange(n, dtype=np.float64)
    base = np.power(j + 1.0, -p)
    mod = 1.0 + amp * np.sin(2.0 * np.pi * freq * j / n)
    v = base * mod
    return _project_simplex(_clean(v))


def _gen_gaussian(n, center, width):
    j = np.arange(n, dtype=np.float64)
    v = np.exp(-((j - center) ** 2) / (2 * width ** 2)) + 1e-6
    return _project_simplex(v)


def _gen_random_smooth(n, seed):
    rng = np.random.default_rng(seed)
    # Random walk in log-space, then smooth
    log_v = np.zeros(n)
    increments = rng.normal(0, 0.3, n)
    log_v = np.cumsum(increments)
    # Smooth with a moving average
    k = max(3, n // 200)
    kernel = np.ones(k) / k
    log_v = np.convolve(log_v, kernel, mode='same')
    v = np.exp(log_v - log_v.max())
    return _project_simplex(_clean(v))

def _soft_max_obj_grad(x, a, n, L, eps):
    """Objective: eps * log(sum(exp(c/eps))) - smooth max of c = a*a.
    x is the sequence itself (on simplex). Returns (f, grad)."""
    A = np.fft.rfft(a, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    cmax = float(c.max())
    z = np.clip((c - cmax) / eps, -700.0, 0.0)
    e = np.exp(z)
    sume = float(e.sum())
    if sume <= 0 or not np.isfinite(sume):
        return 1e30, np.zeros(n)
    f = cmax + eps * np.log(sume)
    w = e / sume
    W = np.fft.rfft(w, n=L)
    ga = 2.0 * np.fft.irfft(W * np.conj(A), n=L)[:n]
    # Project gradient onto tangent space of simplex (sum to zero)
    ga -= float(np.mean(ga))
    return f, ga

    bounds = [(-80.0, 80.0)] * (n - 1)

    def wrapped(x_w):

        if s <= 1e-300:
            a_w = np.full(n, 1.0 / n)
            return 1e30, np.zeros_like(x_w)
        a_w[:-1] = ew / s
        a_w[-1] = e_last / s
        f, g = _soft_max_obj_grad(x_w, a_w, n, L, eps)
        # Transform gradient back to log-ratio space
        if np.all(np.isfinite(g)):
            ga = g.copy()
            ga -= float(np.dot(a_w, g))
            ga *= a_w

        else:
            gy = np.zeros_like(x_w)
        return f, gy

    # Annealing schedule
    eps_start = 0.05 * (best_sc / (2.0 * n))
    eps_min = 1e-9
    eps = eps_start
    no_improve = 0

def _subgrad_nesterov(a0, end_time, seed):
    """Nesterov-accelerated projected subgradient on hard max."""
    a = _project_simplex(a0.copy())
    n = a.size
    L = _pow2(2 * n - 1)
    best = a.copy()
    best_sc = _score_vec(best)
    rng = np.random.default_rng(seed)

    t_nest = 1.0
    y = a.copy()  # lookahead point
    step_base = 0.1
    last_improve = time.monotonic()
    iter_count = 0

    while time.monotonic() < end_time:
        # Compute convolution at lookahead point
        A = np.fft.rfft(y, n=L)
        C = A * A
        c = np.fft.irfft(C, n=L)[:2 * n - 1]
        c = np.maximum(c, 0.0)
        mx = float(c.max())
        if not np.isfinite(mx) or mx <= 1e-300:
            break

        # Active set: lags within 1% of max (or top 5)
        cmax = mx
        threshold = cmax * 0.999
        active_idx = np.where(c >= threshold)[0]
        if len(active_idx) > 20:
            # Keep only top 20
            top_vals = np.sort(c)[::-1][:20]
            threshold = top_vals[-1]
            active_idx = np.where(c >= threshold)[0]
        if len(active_idx) == 0:
            active_idx = np.array([int(np.argmax(c))])

        # Subgradient: g_j = (2/|S|) * sum_{k in S} a[k-j] (valid part)
        g = np.zeros(n, dtype=np.float64)
        for k in active_idx:
            k = int(k)
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                g[j_lo:j_hi + 1] += y[k - j_hi:k - j_lo - 1:-1] if j_hi > j_lo else y[k - j_lo:k - j_lo + 1]
        g *= 2.0 / len(active_idx)

        # Normalize
        g_norm = float(np.sqrt(np.dot(g, g))) + 1e-30

        # Step size
        step = step_base / float(np.sqrt(t_nest))

        # Projected update
        a_new = _project_simplex(a - step * g / g_norm)
        if not np.all(np.isfinite(a_new)):
            a_new = a.copy()

        # Nesterov update
        t_old = t_nest
        t_nest = (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old)) / 2.0
        beta = (t_old - 1.0) / t_nest
        y = a_new + beta * (a_new - a)
        a = a_new

        iter_count += 1

        # Check improvement periodically
        if iter_count % 20 == 0:
            sc = _score_vec(a)
            if np.isfinite(sc) and sc < best_sc - 1e-12:
                best_sc = sc
                best = a.copy()
                last_improve = time.monotonic()
                step_base = min(0.5, step_base * 1.05)
            else:
                now = time.monotonic()
                if now - last_improve > 15.0:
                    step_base = max(0.01, step_base * 0.8)
                    if now - last_improve > 45.0:
                        # Restart
                        a = _project_simplex(best * np.exp(rng.normal(0, 0.03, n)))
                        y = a.copy()
                        t_nest = 1.0
                        step_base = max(0.05, step_base * 1.5)
                        last_improve = now

    # Final check
    sc = _score_vec(a)
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best = a.copy()

    return best, best_sc


def _local_search_refined(a0, end_time, seed):

    a = _project_simplex(a0.copy())
    n = a.size
    L = _pow2(2 * n - 1)
    best = a.copy()
    best_sc = _score_vec(best)
    rng = np.random.default_rng(seed)

    sigma = 0.08
    last_improve = time.monotonic()
    start = time.monotonic()
    consecutive_fail = 0

    while time.monotonic() < end_time:
        remaining = end_time - time.monotonic()
        if remaining < 2.0:
            break

        # Choose perturbation type
        r = rng.random()

        if r < 0.4:
            # Block multiplicative perturbation
            cand = best.copy()
            nblocks = 1 + int(rng.integers(0, 2))
            for _ in range(nblocks):
                maxw = max(3, n // 6)
                w = int(np.exp(rng.uniform(np.log(2.0), np.log(float(maxw)))))
                w = max(1, min(w, n))
                s = int(rng.integers(0, n - w + 1))
                factor = np.exp(rng.normal(0, sigma))
                cand[s:s + w] *= factor
            cand = _project_simplex(cand)

        elif r < 0.6:
            # Coordinate-style: perturb a small number of individual entries
            cand = best.copy()
            ncoords = int(rng.integers(1, 4))
            for _ in range(ncoords):
                idx = int(rng.integers(0, n))
                cand[idx] *= np.exp(rng.normal(0, sigma * 1.5))
            cand = _project_simplex(cand)

        elif r < 0.8:
            # Block shift: lift one block, lower another (mass redistribution)
            cand = best.copy()
            w1 = int(np.exp(rng.uniform(np.log(3), np.log(max(5, n // 4)))))
            w1 = min(w1, n)
            s1 = int(rng.integers(0, n - w1 + 1))
            lift = np.exp(rng.normal(0, sigma))
            cand[s1:s1 + w1] *= lift
            # Compensate elsewhere
            s2_options = list(range(0, n - w1))
            s2 = int(rng.integers(0, n - w1))
            while s2 >= s1 - w1 and s2 < s1 + w1:
                s2 = int(rng.integers(0, n - w1 + 1)) if n > w1 else 0
            if n > w1:
                comp = (1.0 / lift) ** (w1 * float(best[s1:s1 + w1].sum()) / max(1e-15, float(best[s2:s2 + w1].sum())))
                comp = np.clip(comp, 0.5, 2.0)
                cand[s2:s2 + w1] *= comp
            cand = _project_simplex(cand)

        else:
            # Global smooth perturbation
            smooth_pert = np.cumsum(np.cumsum(rng.normal(0, sigma / n, n)))
            smooth_pert = np.exp(np.clip(smooth_pert - smooth_pert.mean(), -3, 3))
            cand = _project_simplex(best * smooth_pert)

        sc = _score_vec(cand)
        if np.isfinite(sc) and sc < best_sc - 1e-12:
            best = cand
            best_sc = sc
            last_improve = time.monotonic()
            sigma = min(0.3, sigma * 1.05)
            consecutive_fail = 0
        else:
            consecutive_fail += 1

        # Adaptive sigma
        if time.monotonic() - last_improve > 5.0:
            sigma = max(0.01, sigma * 0.85)
        if time.monotonic() - last_improve > 25.0 and time.monotonic() - start > 30.0:
            # Re-center
            jitter = np.exp(rng.normal(0, 0.02, n))
            cand = _project_simplex(best * jitter)
            sc2 = _score_vec(cand)
            if np.isfinite(sc2) and sc2 < best_sc:
                best = cand
                best_sc = sc2
                last_improve = time.monotonic()
            else:
                break

    return best, best_sc


def _coordinate_descent(a0, end_time, seed):

    cmax = float(c.max())

    # Sort positions by "importance" (larger values first)
    # Try perturbing positions that contribute most to the max autocorrelation
    order = np.argsort(-a)[:min(200, n)]  # top 200 positions

    consecutive_fail = 0
    last_improve = time.monotonic()

    while time.monotonic() < end_time:
        # Pick a random position from the important set
        idx = int(order[int(rng.integers(0, len(order)))])
        # Try a range of transfer amounts to a random other position
        for frac in (0.01, 0.03, 0.05, 0.1, 0.2):
            target_val = a[idx] * frac
            # Find a random position to move mass to/from
            j = int(rng.integers(0, n))
            if j == idx:
                continue

            cand = best.copy()
            if target_val < a[idx] * 0.9:
                # Move mass from idx to j
                amount = min(target_val, a[idx] * 0.9)
                cand[idx] -= amount
                cand[j] += amount
            else:
                # Move mass from j to idx
                amount = min(target_val, a[j] * 0.9)
                cand[j] -= amount
                cand[idx] += amount

            cand = _project_simplex(cand)
            sc = _score_vec(cand)
            if np.isfinite(sc) and sc < best_sc - 1e-12:
                best = cand
                best_sc = sc
                last_improve = time.monotonic()
                consecutive_fail = 0
                break
        else:
            consecutive_fail += 1

        if consecutive_fail > 50:
            break
        if time.monotonic() - last_improve > 20.0:
            break

    return best, best_sc


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the best non-negative sequence minimizing 2n*max(conv(a,a))/(sum a)^2.
    Returns a list of non-negative floats.
    """
    t0 = time.monotonic()
    budget = max(0.0, float(budget_s) - 15.0)  # safety margin
    deadline = t0 + budget

    # Determine sequence length
    n = 3809
    try:
        h = height_sequence_1
        if isinstance(h, (list, np.ndarray)):
            arr = np.asarray(h, dtype=np.float64)
            if arr.ndim == 1 and 100 <= arr.size <= 20000:
                n = int(arr.size)
    except Exception:
        pass
    if n < 32:
        n = 3809

    print(f"Starting search with n={n}, budget={budget:.1f}s")

    # Emergency fallback
    if budget < 10.0:
        a = _gen_power(n, 0.5, 1.0)
        return [float(x) for x in a]

    # Phase 1: Generate diverse candidates (5% of budget)
    p1_end = t0 + 0.05 * budget
    cands = _generate_diverse(n)
    scored = []
    for i, a in enumerate(cands):
        sc = _score_vec(a)
        if np.isfinite(sc):
            scored.append((sc, i, a))
    scored.sort(key=lambda x: x[0])

    if not scored:
        a = _gen_power(n, 0.5, 1.0)
        return [float(x) for x in a]

    global_best = np.asarray(scored[0][2], dtype=np.float64).copy()
    global_best_sc = float(scored[0][0])
    print(f"Phase 1 done: {len(scored)} candidates, best score = {global_best_sc:.6f}")
    print(f"  Top 3: {[f'{s[0]:.6f}' for s in scored[:3]]}")
    if time.monotonic() > p1_end:
        print("  (Phase 1 took longer than allocated, proceeding)")

    # Phase 2: Smooth max annealing on top candidates (30%)
    p2_end = min(deadline, t0 + 0.35 * budget)
    top_cands = [s[2] for s in scored[:4]]

    for idx, a0 in enumerate(top_cands):
        alloc_end = p2_end
        if idx == 0:
            alloc_end = min(deadline, t0 + 0.20 * budget)
        elif idx == 1:
            alloc_end = min(deadline, t0 + 0.28 * budget)
        elif idx == 2:
            alloc_end = min(deadline, t0 + 0.33 * budget)
        else:
            alloc_end = p2_end

        if time.monotonic() >= alloc_end - 3.0:
            continue

        print(f"Phase 2: smoothing candidate {idx} (score { _score_vec(a0):.6f})")
        a, sc = _optimize_smooth_max(a0, alloc_end, int(seed) + 100 * (idx + 1))
        if np.isfinite(sc) and sc < global_best_sc:
            global_best_sc = sc
            global_best = a.copy()
            print(f"  Improved to {sc:.6f}")

    print(f"After Phase 2: best = {global_best_sc:.6f}")

    # Phase 3: Subgradient on hard max (25%)
    p3_end = min(deadline, t0 + 0.60 * budget)
    if time.monotonic() < p3_end:
        print("Phase 3: subgradient descent on hard max")
        a, sc = _subgrad_nesterov(global_best, p3_end, int(seed) + 999)
        if np.isfinite(sc) and sc < global_best_sc:
            global_best_sc = sc
            global_best = a.copy()
            print(f"  Improved to {sc:.6f}")
    else:
        print("Phase 3: skipped (no time)")

    print(f"After Phase 3: best = {global_best_sc:.6f}")

    # Phase 4: Local search (20%)
    p4_end = min(deadline, t0 + 0.80 * budget)
    if time.monotonic() < p4_end:
        print("Phase 4: local search")
        a, sc = _local_search_refined(global_best, p4_end, int(seed) + 555)
        if np.isfinite(sc) and sc < global_best_sc:
            global_best_sc = sc
            global_best = a.copy()
            print(f"  Improved to {sc:.6f}")
    else:
        print("Phase 4: skipped (no time)")

    # Phase 5: Coordinate descent polish (10%)
    if time.monotonic() < deadline:
        print("Phase 5: coordinate descent polish")
        coord_end = min(deadline, t0 + 0.90 * budget)
        if time.monotonic() < coord_end:
            a, sc = _coordinate_descent(global_best, coord_end, int(seed) + 777)
            if np.isfinite(sc) and sc < global_best_sc:
                global_best_sc = sc
                global_best = a.copy()
                print(f"  Improved to {sc:.6f}")

    # Phase 6: Final ultra-fine local search
    if time.monotonic() < deadline:
        print("Phase 6: final polish")
        a, sc = _local_search_refined(global_best, deadline, int(seed) + 12345)
        if np.isfinite(sc) and sc < global_best_sc:
            global_best_sc = sc
            global_best = a.copy()
            print(f"  Improved to {sc:.6f}")

    global_best = _project_simplex(global_best)
    final_sc = _score_vec(global_best)
    print(f"Final score (internal): {final_sc:.6f}")

    # Verify with official evaluator
    out = [float(x) for x in global_best]
    try:
        val = evaluate_sequence(out)
        print(f"Official evaluator: {val:.6f}")
        if not np.isfinite(val):
            print("  Fallback to power sequence")
            out = [float(x) for x in _gen_power(len(out), 0.5, 1.0)]
    except Exception as e:
        print(f"  Evaluator error: {e}")

    print(f"Returning sequence of length {len(out)}")
    return out
