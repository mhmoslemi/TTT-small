"""
Search for non-negative sequences minimizing
    2*n*max(convolutions(a,a))/(sum(a))**2

Method:
- Normalize all sequences to sum 1; objective becomes 2*n*max(conv(a,a)).
- Use FFT-based exact convolution and FFT-based aggregation of
  weighted sub-gradients of selected active convolution coefficients.
- Combine multiplicative log-simplex descent with adaptive active-set
  weighting (top-k, softmax, moment weights), momentum, line search,
  symmetrization/candidate blending, and periodic multiplicative restarts.
- Optionally run a linear-programming best-response step for moderate n.
- Optionally optimize low-dimensional Fourier/quadratic parameterizations
  to escape mild basins.
- Return the best finite sequence found before the time limit.
"""

def _next_pow2(n):
    """Smallest power of two >= n."""
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _clean_positive(a):
    """Return normalized non-negative float array, or None."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        return None
    a = a / s
    a = np.maximum(a, 1e-18)
    s = float(a.sum())
    if s > 0.0:
        a = a / s
    return a


def _append_start(starts, name, n, pattern):
    """Safely add a candidate pattern if it is valid."""
    arr = _clean_positive(pattern)
    if arr is not None:
        starts.append((name, int(n), arr))


def _unique_add(items, item):
    """Add item if its name has not been used before."""
    nm = item[1]
    for old in items:
        if old[1] == nm:
            return
    items.append(item)


def _add_history(history, score, a, cap=5):
    """Maintain a small list of good sequences (no sorting)."""
    if a is None:
        return
    if len(history) >= cap:
        max_i = 0
        for i in range(len(history)):
            if history[i][0] > history[max_i][0]:
                max_i = i
        if score >= history[max_i][0]:
            return
        history.pop(max_i)
    history.append((float(score), np.array(a, dtype=np.float64, copy=True)))

def _fft_convolve(a):
    """Linear convolution a*a, with small negative FFT error clamped."""
    n = int(len(a))
    if n <= 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = _next_pow2(size)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    if c.size > size:
        c = c[:size]
    return np.maximum(c, 0.0)


def _score_normalized(a):
    """Score assuming a has been normalized to sum 1."""
    n = int(len(a))
    if n <= 0:
        return np.inf
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    return float(2.0 * n * float(np.max(c)))


def _rfft_and_c(a):
    """Compute a larger-scale FFT of a and its self-convolution."""
    n = int(len(a))
    if n <= 0:
        fa = np.array([], dtype=np.complex128)
        c = np.array([], dtype=np.float64)
        N = 1
        return fa, c, N
    need = 3 * n - 2
    if need < 1:
        need = 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    size = 2 * n - 1
    if c.size > size:
        c = c[:size]
    c = np.maximum(c, 0.0)
    return fa, c, N


def _get_grad_maps(n):
    """Precompute index maps for gradient aggregation by weights."""
    key = "grad_" + str(int(n))
    if key in _MAP_CACHE:
        return _MAP_CACHE[key]
    if n <= 0:
        need = 1
    else:
        need = max(1, 3 * n - 2)
    N = _next_pow2(need)
    L = 2 * int(n) - 2
    idx = (N - np.arange(L + 1, dtype=np.int64)) % N
    out = (N - np.arange(int(n), dtype=np.int64)) % N
    _MAP_CACHE[key] = (N, idx, out)
    return _MAP_CACHE[key]


def _b_from_fa(fa, w, n, N):
    """
    For weights w over convolution lags, compute b_j =
        sum_k w_k a_{k-j}
    using the already computed FFT fa of a at length N.
    """
    _, idx, out = _get_grad_maps(n)
    fw = np.zeros(N, dtype=np.float64)
    fw[idx] = w
    prod = np.fft.irfft(fa * fw, N)
    return prod[out]


def _weight_vec(c, name, param):
    """Construct normalized non-negative weights over convolution lags."""
    L = max(0, int(len(c)) - 1)
    if len(c) == 0:
        return np.array([], dtype=np.float64)

    M = float(np.max(c))
    mean = 1.0 / float(len(c))
    gap = M - mean

    if gap <= 1e-15:
        return np.full(len(c), 1.0 / len(c), dtype=np.float64)

    p = None
    if name == "one":
        tol = 1e-12 * max(1.0, abs(M))
        idx = np.where(c >= M - tol)[0]
        p = np.zeros(len(c), dtype=np.float64)
        if idx.size == 0:
            idx = np.array([int(np.argmax(c))], dtype=np.int64)
        p[idx] = 1.0
    elif name == "top":
        frac = max(1e-6, float(param))
        thresh = M - frac * gap
        p = (c >= thresh).astype(np.float64)
    elif name == "soft":
        T = max(1e-15, float(param) * gap)
        x = (c - M) / T
        x = np.maximum(x, -50.0)
        p = np.exp(x)
    elif name == "moment":
        q = max(1.0, float(param))
        p = np.power(np.maximum(c, 0.0), q)
    else:
        p = np.ones(len(c), dtype=np.float64)

    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 1e-300:
        idx = np.array([int(np.argmax(c))], dtype=np.int64)
        p = np.zeros(len(c), dtype=np.float64)
        p[idx] = 1.0
        s = 1.0
    p = p / s
    return p

def _perturb_positive(a, rng, sigma):
    """Multiplicative random perturbation, kept positive and normalized."""
    a = np.asarray(a, dtype=np.float64)
    sig = max(1e-4, float(sigma))
    noisy = a * np.exp(rng.normal(0.0, sig, size=a.shape))

def _make_features(n, K):
    """Low-dimensional non-negative-friendly feature basis."""
    x = np.linspace(0.0, 1.0, int(n), endpoint=False)
    feats = []
    feats.append(x)
    feats.append((x - 0.5) ** 2)
    for m in range(1, int(K) + 1):
        feats.append(np.cos(2.0 * np.pi * m * x))
        feats.append(np.sin(2.0 * np.pi * m * x))
    F = np.vstack(feats).astype(np.float64)
    F -= np.mean(F, axis=1, keepdims=True)
    return F


def _model_positive(F, coef):
    """Positive normalized density from exp(F @ coef)."""
    raw = F.dot(coef)
    raw = raw - float(np.mean(raw))
    raw = raw - float(np.max(raw))
    a = np.exp(raw)
    a = np.maximum(a, 1e-18)
    s = float(a.sum())
    if s <= 0.0:
        return None
    a = a / s
    return a


def _fit_features(F, a):
    """Least-squares fit of log(a) by feature coefficients."""
    y = np.log(np.maximum(a, 1e-300))
    A = F.T
    D = int(F.shape[0])
    try:
        ATA = A.T.dot(A)
        ATA += 1e-12 * np.eye(D, dtype=np.float64)
        ATy = A.T.dot(y)
        coef = np.linalg.solve(ATA, ATy)
    except Exception:
        try:
            coef = np.linalg.lstsq(A, y, rcond=None)[0]
        except Exception:
            coef = np.zeros(D, dtype=np.float64)
    return coef


def _param_single(n, F, coef0, allowed_s, rng):
    """Optimize one low-dimensional parameterization for allowed_s seconds."""
    if allowed_s <= 0.0 or n <= 0 or F is None or F.size == 0:
        a = _model_positive(F, coef0)
        if a is None:
            return np.ones(n, dtype=np.float64) / n, np.inf
        return a, _score_normalized(a)

    D = int(F.shape[0])
    coef = np.asarray(coef0, dtype=np.float64).copy()

    if a is None:
        a = np.ones(n, dtype=np.float64) / n

    best_a = a.copy()
    best_score = _score_normalized(a)

    e = 0.1
    m = np.zeros(D, dtype=np.float64)
    v = np.zeros(D, dtype=np.float64)
    t = 0
    last_progress = time.monotonic()
    last_restart = last_progress
    idx = 0
    prev_score = best_score
    deadline = time.monotonic() + float(allowed_s)

        score = float(2.0 * n * float(np.max(c)))

            best_score = score
            best_a = a.copy()

        prev_score = score

        plateau = time.monotonic() - last_progress
        if plateau > 2.0:
            idx = (idx + 1) % len(_SMOOTH_SETTINGS)
            e = 0.08
            m *= 0.25
            v *= 0.25
            t = 0

        if plateau > 10.0 and time.monotonic() - last_restart > 4.0:

            coef = coef + rng.normal(0.0, sig, size=D)
            m = np.zeros(D, dtype=np.float64)
            v = np.zeros(D, dtype=np.float64)

            e = 0.08
            a = _model_positive(F, coef)

        name, param = _SMOOTH_SETTINGS[idx]

        b = _b_from_fa(fa, w, n, N)

        if dmax <= 1e-16:
            idx = (idx + 1) % len(_SMOOTH_SETTINGS)
            continue
        d = d / dmax

        h = a * d
        g = F.dot(h)
        gnorm = float(np.linalg.norm(g, ord=2))
        if not np.isfinite(gnorm) or gnorm <= 1e-16:

            continue
        g = g / gnorm

        t += 1
        b1 = 0.9
        b2 = 0.999
        m = b1 * m + (1.0 - b1) * g
        v = b2 * v + (1.0 - b2) * (g * g)
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)

            a = trial_a
            score = sc

                best_a = np.asarray(a_trial, dtype=np.float64).copy()
                best_score = float(sc_trial)
        except Exception:

    return best_a, best_score

def _optimize_smooth(a0, budget_s, rng):

    l = np.log(np.maximum(a, 1e-18))
    vel = np.zeros(n, dtype=np.float64)
    eta = 0.5
    mu = 0.6

    it = 0
    last_progress = time.monotonic()
    last_restart = last_progress
    prev_score = best_score
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s))
    idx = 0

    while time.monotonic() < deadline - 1.0:
        fa, c, N = _rfft_and_c(a)
        score = float(2.0 * n * float(np.max(c)))

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            last_progress = time.monotonic()
            _add_history(history, best_score, a, cap=5)

        if score < prev_score - 1e-12:
            last_progress = time.monotonic()
        prev_score = score

        plateau = time.monotonic() - last_progress

        # Periodic symmetrization / blending.
        do_sym = (it % 90 == 45) or (plateau > 4.0 and it % 25 == 0)
        if do_sym:
            sym = 0.5 * (a + a[::-1])
            sym = _clean_positive(sym)
            if sym is not None:
                sc_sym = _score_normalized(sym)
                if sc_sym < score - 1e-12:
                    a = sym
                    score = sc_sym
                    l = np.log(np.maximum(a, 1e-18))
                    vel *= 0.2
                    prev_score = score
                    last_progress = time.monotonic()
                    if score < best_score - 1e-12:
                        best_score = score
                        best_a = a.copy()
                        _add_history(history, best_score, a, cap=5)

            if len(history) > 1 and plateau > 3.0:
                tries = 0
                for t in (0.3, 0.5, 0.7):
                    for i in range(min(2, len(history))):
                        hh = history[i][1]
                        cand = (1.0 - t) * a + t * hh
                        cand = _clean_positive(cand)
                        if cand is None:
                            continue
                        sc = _score_normalized(cand)
                        tries += 1
                        if sc < score - 1e-12:
                            a = cand
                            score = sc
                            l = np.log(np.maximum(a, 1e-18))
                            vel *= 0.2
                            prev_score = score
                            last_progress = time.monotonic()
                            if score < best_score - 1e-12:
                                best_score = score
                                best_a = a.copy()
                                _add_history(history, best_score, a, cap=5)
                            break
                        if tries >= 12:
                            break
                    if tries >= 12:
                        break

        if plateau > 1.5:
            idx = (idx + 1) % len(_SMOOTH_SETTINGS)
            vel *= 0.3
            eta = 0.35

        if plateau > 9.0 and time.monotonic() - last_restart > 5.0:
            last_restart = time.monotonic()
            sig = 0.03 + min(0.12, plateau * 0.003)
            a = _perturb_positive(best_a, rng, sig)
            if a is None:
                a = best_a.copy()
            l = np.log(np.maximum(a, 1e-18))
            vel = np.zeros(n, dtype=np.float64)
            eta = 0.35
            idx = 0
            prev_score = _score_normalized(a)

        if plateau > 45.0 and time.monotonic() - last_restart > 8.0:
            last_restart = time.monotonic()
            F = _make_features(n, 3)
            coef = rng.normal(0.0, 1.0, size=F.shape[0])
            a2 = _model_positive(F, coef)
            if a2 is not None:
                a = a2
            else:
                a = _perturb_positive(best_a, rng, 0.18)
            l = np.log(np.maximum(a, 1e-18))
            vel = np.zeros(n, dtype=np.float64)
            eta = 0.35
            idx = 0
            prev_score = _score_normalized(a)

        name, param = _SMOOTH_SETTINGS[idx]
        w = _weight_vec(c, name, param)
        b = _b_from_fa(fa, w, n, N)
        d = b - float(np.mean(b))
        dmax = float(np.max(np.abs(d)))
        if dmax <= 1e-14:
            idx = (idx + 1) % len(_SMOOTH_SETTINGS)
            continue
        d = d / dmax

        vel = mu * vel + (1.0 - mu) * d
        vel = vel - float(np.mean(vel))
        vmax = float(np.max(np.abs(vel)))
        if vmax <= 1e-14:
            idx = (idx + 1) % len(_SMOOTH_SETTINGS)
            continue
        vel = vel / vmax

        accepted = False
        best_trial = None
        best_trial_score = score
        e = eta
        for _ in range(4):
            step = e * vel
            trial = np.exp(l - step)
            tsum = float(trial.sum())
            if not np.isfinite(tsum) or tsum <= 1e-200:
                e *= 0.5
                continue
            trial = trial / tsum
            sc = _score_normalized(trial)

            if np.isfinite(sc) and sc < best_trial_score:
                best_trial = trial
                best_trial_score = sc

            if np.isfinite(sc) and sc < score - 1e-10:
                a = trial
                score = sc
                l = np.log(np.maximum(a, 1e-18))
                prev_score = score
                last_progress = time.monotonic()
                eta = min(e * 1.08, 2.0)
                accepted = True
                break
            e *= 0.5

        if not accepted:
            if best_trial is None:
                eta = max(eta * 0.6, 0.02)
            elif best_trial_score < score and (score - best_trial_score) < 1e-6:
                a = best_trial
                score = best_trial_score
                l = np.log(np.maximum(a, 1e-18))
                prev_score = score
                last_progress = time.monotonic()
                eta = max(e * 0.7, 0.02)
            else:
                eta = max(e * 0.6, 0.015)

        it += 1

    return best_a, best_score

def _build_lp_indices(n):
    """
    Build sparse index structure for constraints f*g <= M.
    For row k and column j, coefficient is f_{k-j} when valid.
    """
    key = "lp_" + str(int(n))
    if key in _LP_CACHE:
        return _LP_CACHE[key]

    rows = 2 * int(n) - 1
    total = int(n) * int(n)
    col = np.empty(total, dtype=np.int32)
    src = np.empty(total, dtype=np.int32)
    indptr = np.empty(rows + 1, dtype=np.int32)

    pos = 0
    for k in range(rows):
        lo = max(0, k - int(n) + 1)
        hi = min(int(n) - 1, k)
        m = hi - lo + 1
        indptr[k] = pos
        cs = np.arange(lo, hi + 1, dtype=np.int32)
        col[pos:pos + m] = cs
        src[pos:pos + m] = (k - cs).astype(np.int32)
        pos += m
    indptr[rows] = pos

    _LP_CACHE[key] = (indptr, col, src)
    return _LP_CACHE[key]


def _lp_best_response(a, time_limit):
    """
    For fixed f (scaled to max 1), solve:
        maximize sum(g)  s.t.  f*g <= M, g >= 0.
    Return normalized g or None.
    """
    if not _HAVE_SCIPY_OPT:
        return None
    n = int(len(a))
    if n <= 0 or n > 2000:
        return None
    a = _clean_positive(a)
    if a is None:
        return None

    # Scale to improve condition; also avoid extremely small coefficients.
    a_lp = np.maximum(a, 1e-8)
    mx = float(a_lp.max())
    if mx <= 0.0:
        return None
    a_lp = a_lp / mx
    c_lp = _fft_convolve(a_lp)
    M = float(np.max(c_lp))
    if not np.isfinite(M) or M <= 0.0:
        return None

    indptr, col, src = _build_lp_indices(n)
    data = a_lp[src].astype(np.float64, copy=False)
    A = csr_matrix((data, col, indptr), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(
            c_obj,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"time_limit": float(max(1.0, time_limit))},
        )
    except Exception:
        return None

    x = getattr(res, "x", None)
    if x is None or len(x) != n:
        return None
    try:
        if not res.success:
            return None
    except Exception:
        pass

    g = np.asarray(x, dtype=np.float64)
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s <= 1e-300:
        return None
    return g / s


def _lp_polish(a0, budget_s, rng):
    """Use LP best-response as occasional global/local jumps."""
    if not _HAVE_SCIPY_OPT:
        a = _clean_positive(a0)
        if a is None:
            a = np.ones(1000, dtype=np.float64) / 1000.0
        return a, _score_normalized(a)

    a = _clean_positive(a0)
    if a is None:
        a = np.ones(1000, dtype=np.float64) / 1000.0
    n = int(len(a))
    if n > 2000:
        return a, _score_normalized(a)

    best_a = a.copy()
    best_score = _score_normalized(a)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s))
    no_imp = 0

    while time.monotonic() < deadline - 10.0:
        rem = deadline - time.monotonic()
        if rem < 20.0:
            break
        tlim = min(35.0, rem * 0.35)
        try:
            h = _lp_best_response(best_a, tlim)
        except Exception:
            break
        finally:
            if gc is not None:
                gc.collect()

        if h is None:
            break

        improved = False
        for t in (0.05, 0.1, 0.2, 0.35, 0.55):
            cand = (1.0 - t) * best_a + t * h
            cand = _clean_positive(cand)
            if cand is None:
                continue
            sc = _score_normalized(cand)
            if sc < best_score - 1e-10:
                best_a = cand
                best_score = sc
                improved = True
                no_imp = 0
                break

        if not improved:
            no_imp += 1
            if no_imp >= 3:
                break

    return best_a, best_score

def _gen_starts_for_n(n, rng, provided=None):
    """Generate diverse smooth starts of length n."""
    starts = []
    n = int(n)
    if n <= 0:
        return starts

    x = np.linspace(0.0, 1.0, n, endpoint=False)

    if provided is not None and len(provided) > 0:
        prov = _clean_positive(provided)
        if prov is not None:
            if len(prov) == n:
                _append_start(starts, "provided_%d" % n, n, prov)
            else:
                xs = np.linspace(0.0, 1.0, len(prov), endpoint=False)
                y = np.log(np.maximum(prov, 1e-300))
                ynew = np.interp(x, xs, y)
                _append_start(starts, "interp_" + str(len(prov)) + "_to_%d" % n, n,
                              np.exp(ynew - float(np.max(ynew))))
            _append_start(starts, "provsym_%d" % n, n, 0.5 * (prov + prov[::-1]))
            _append_start(starts, "provrev30_%d" % n, n, 0.7 * prov + 0.3 * prov[::-1])
            _append_start(starts, "provrev70_%d" % n, n, 0.3 * prov + 0.7 * prov[::-1])

    _append_start(starts, "uniform_%d" % n, n, np.ones(n))
    _append_start(starts, "rampup_%d" % n, n, 0.2 + 0.8 * x)
    _append_start(starts, "rampdown_%d" % n, n, 0.2 + 0.8 * (1.0 - x))
    _append_start(starts, "quadcenter_%d" % n, n, 0.1 + x * (1.0 - x))
    _append_start(starts, "edgepeak_%d" % n, n, np.exp(-5.0 * (x - 0.5) ** 2) + 0.05)
    _append_start(starts, "centerpeak_%d" % n, n, 1.0 / (0.05 + x * (1.0 - x)))
    _append_start(starts, "decay_%d" % n, n, np.exp(-2.0 * x) + 0.05)
    _append_start(starts, "twodecay_%d" % n, n, np.exp(-3.0 * x) + np.exp(-3.0 * (1.0 - x)) + 0.05)

    for amp in (0.3, 0.5, 0.7):
        for ph in (0.0, 0.5, 1.1, 1.9):
            a = 1.0 + amp * np.sin(2.0 * np.pi * x + ph)
            _append_start(starts, "sin2_%.1f_%.1f_%d" % (amp, ph, n), n, np.maximum(a, 0.04))

    for amp2 in (0.35, 0.6):
        a = 1.0 + 0.4 * np.sin(2.0 * np.pi * x + 0.6) + amp2 * np.cos(4.0 * np.pi * x + 0.9)
        _append_start(starts, "polyharm_%.1f_%d" % (amp2, n), n, np.maximum(a, 0.03))

    for ctr in (0.25, 0.5, 0.75):
        for sig in (0.08, 0.15, 0.25):
            a = np.exp(-0.5 * ((x - ctr) / sig) ** 2) + 0.02
            _append_start(starts, "gauss_%.2f_%.2f_%d" % (ctr, sig, n), n, a)

    for c1, c2, h2 in ((0.25, 0.75, 0.5), (0.35, 0.65, 0.75), (0.18, 0.82, 0.35)):
        sig = 0.10
        a = (np.exp(-0.5 * ((x - c1) / sig) ** 2)
             + h2 * np.exp(-0.5 * ((x - c2) / sig) ** 2)
             + 0.02)
        _append_start(starts, "twogauss_%.2f_%.2f_%.2f_%d" % (c1, c2, h2, n), n, a)

    # Random smooth Fourier starts
    rnd_cap = 6 if n > 6000 else 10
    for r in range(rnd_cap):
        s = np.zeros(n, dtype=np.float64)
        for m in range(1, 6):
            amp = rng.normal(0.0, 1.0 / (m + 1.0))
            phase = rng.uniform(0.0, 2.0 * np.pi)
            s += amp * np.cos(2.0 * np.pi * m * x + phase)
            if m <= 3:
                amp2 = rng.normal(0.0, 0.7 / (m + 1.0))
                phase2 = rng.uniform(0.0, 2.0 * np.pi)
                s += amp2 * np.sin(2.0 * np.pi * m * x + phase2)
        s = s - float(np.max(s))
        a = np.exp(s) + 0.01
        _append_start(starts, "randfourier%d_%d" % (r, n), n, a)

    return starts


def _score_all_starts(starts):
    """Score starts and return list [(score, name, n, array)]."""
    scored = []
    for name, n, arr in starts:
        try:
            sc = _score_normalized(arr)
            if np.isfinite(sc):
                scored.append((float(sc), name, int(n), arr))
        except Exception:
            pass
    scored.sort(key=operator.itemgetter(0))
    return scored

def propose_candidate(seed=42, budget_s=2380.0, **kwargs):
    """
    Returns a non-negative height sequence minimizing the given evaluator.
    """
    rng = np.random.default_rng(int(seed) if seed is not None else 42)
    total_start = time.monotonic()
    try:
        total_budget = max(120.0, min(2390.0, float(budget_s) if budget_s is not None else 2380.0))
    except Exception:
        total_budget = 2380.0
    deadline = total_start + max(60.0, total_budget - 30.0)

    # Load any provided previous construction.
    provided = None
    try:
        prov_raw = np.asarray(height_sequence_1, dtype=np.float64)
        prov_raw = np.where(np.isfinite(prov_raw), prov_raw, 0.0)
        prov_raw = np.maximum(prov_raw, 0.0)
        if prov_raw.size > 50:
            provided = prov_raw
    except Exception:
        provided = None

    # Decide lengths. If provided length is sensible, use it and multiples.
    base_n = 2000
    if provided is not None:
        pr = int(len(provided))
        if 500 <= pr <= 5000:
            base_n = pr

    lengths = []
    for cand in (base_n, 2 * base_n, 3 * base_n, 2000, 4000, 6000):
        c = int(cand)
        if c >= 800 and c <= 8000 and c not in lengths:
            lengths.append(c)
        if len(lengths) >= 3:
            break

    # Generate starts.
    all_starts = []
    for n in lengths:
        try:
            st = _gen_starts_for_n(n, rng, provided)
            all_starts.extend(st)
        except Exception:
            pass

    # Fallback if no valid starts.
    if not all_starts:
        one = np.ones(1000, dtype=np.float64) / 1000.0
        all_starts.append(("fallback", 1000, one))

    scored = _score_all_starts(all_starts)
    if not scored:
        one = np.ones(1000, dtype=np.float64) / 1000.0
        return one.tolist()

    # Build selected worklist: best starts, provided, and guarantee variety.
    selected = []
    # Top by score.
    for item in scored[:8]:
        _unique_add(selected, item)

    # Ensure provided is included if present.
    for item in scored:
        if "provided" in item[1] or "interp" in item[1]:
            _unique_add(selected, item)
        if len(selected) >= 12:
            break

    # Ensure at least one longest length if possible.
    if lengths:
        max_len = max(lengths)
        for item in scored:
            if item[2] == max_len:
                _unique_add(selected, item)
                break

    # Limit selected count and reorder by current score only as tie-break.
    if len(selected) > 8:
        # Keep best 8.
        selected = sorted(selected, key=operator.itemgetter(0))[:8]
    else:
        selected = sorted(selected, key=operator.itemgetter(0))

    if not selected:
        one = np.ones(1000, dtype=np.float64) / 1000.0
        return one.tolist()

    best_global_score = float(scored[0][0])
    best_global_a = np.array(scored[0][3], dtype=np.float64, copy=True)

    print("Initial best score: %.8f, n=%d" % (best_global_score, len(best_global_a)))

    # Time weights for candidates.
    weights = []
    nt = len(selected)
    for i, item in enumerate(selected):
        sc, name, n, arr = item
        w = float(nt - i + 1)
        if "provided" in name:
            w *= 3.0
        if "interp" in name:
            w *= 1.8
        if n == base_n:
            w *= 1.5
        if n >= 4000:
            w *= 1.1
        weights.append(max(0.5, w))

    total_w = float(sum(weights)) if weights else 1.0

    for idx, item in enumerate(selected):
        if time.monotonic() > deadline - 60.0:
            break
        sc, name, n, a0 = item
        rem = deadline - time.monotonic()
        if rem <= 30.0:
            break

        alloc = rem * (weights[idx] / total_w)
        alloc = min(alloc, rem - 20.0)
        alloc = max(30.0, alloc)

        print("\n=== Candidate %d: %s, n=%d, initial_score=%.8f, allocating %.1fs ==="
              % (idx + 1, name, n, sc, alloc))

        try:
            # Primary full-space polish.
            sm_time = min(alloc * 0.62, max(20.0, (deadline - time.monotonic()) - 60.0))
            a1, s1 = _optimize_smooth(a0, sm_time, rng)
            if a1 is not None and np.isfinite(s1) and s1 < best_global_score - 1e-12:
                best_global_score = float(s1)
                best_global_a = np.asarray(a1, dtype=np.float64).copy()
                print("  Smooth improved to %.8f" % best_global_score)

            a_work = a1 if a1 is not None else a0

            # Low-dimensional param exploration if time remains.
            rem2 = deadline - time.monotonic()
            if rem2 > 120.0 and n <= 6000:
                p_time = min(alloc * 0.22, rem2 * 0.35, 260.0)
                if p_time > 20.0:
                    a2, s2 = _optimize_param(a_work, p_time, rng)
                    if a2 is not None and np.isfinite(s2) and s2 < best_global_score - 1e-12:
                        best_global_score = float(s2)
                        best_global_a = np.asarray(a2, dtype=np.float64).copy()
                        print("  Param improved to %.8f" % best_global_score)
                    else:
                        a2w = a2 if a2 is not None else a_work
                        # Short full polish on param output.
                        rem3 = deadline - time.monotonic()
                        if rem3 > 60.0:
                            a3, s3 = _optimize_smooth(a2w, min(rem3 * 0.25, 120.0), rng)
                            if a3 is not None and np.isfinite(s3) and s3 < best_global_score - 1e-12:
                                best_global_score = float(s3)
                                best_global_a = np.asarray(a3, dtype=np.float64).copy()
                                print("  Post-param smooth improved to %.8f" % best_global_score)

            # Optional LP best response for moderate lengths.
            rem4 = deadline - time.monotonic()
            if rem4 > 80.0 and n <= 2000:
                lp_time = min(alloc * 0.15, rem4 * 0.35, 180.0)
                if lp_time > 20.0:
                    a4, s4 = _lp_polish(a_work, lp_time, rng)
                    if a4 is not None and np.isfinite(s4) and s4 < best_global_score - 1e-12:
                        best_global_score = float(s4)
                        best_global_a = np.asarray(a4, dtype=np.float64).copy()
                        print("  LP improved to %.8f" % best_global_score)

        except Exception as ex:
            print("  Candidate exception: %s" % repr(ex))

    # Final polishing stage.
    try:
        rem = deadline - time.monotonic()
        if rem > 60.0:
            print("\n=== Final polishing for %.1fs ===" % rem)
            a_pol, s_pol = _optimize_smooth(best_global_a, rem * 0.45, rng)
            if a_pol is not None and np.isfinite(s_pol) and s_pol < best_global_score - 1e-12:
                best_global_score = float(s_pol)
                best_global_a = np.asarray(a_pol, dtype=np.float64).copy()

        rem = deadline - time.monotonic()
        if rem > 50.0 and len(best_global_a) <= 2000:
            a_lp2, s_lp2 = _lp_polish(best_global_a, min(rem * 0.55, 260.0), rng)
            if a_lp2 is not None and np.isfinite(s_lp2) and s_lp2 < best_global_score - 1e-12:
                best_global_score = float(s_lp2)
                best_global_a = np.asarray(a_lp2, dtype=np.float64).copy()

        rem = deadline - time.monotonic()
        if rem > 30.0:
            a_pol2, s_pol2 = _optimize_smooth(best_global_a, rem * 0.85, rng)
            if a_pol2 is not None and np.isfinite(s_pol2) and s_pol2 < best_global_score - 1e-12:
                best_global_score = float(s_pol2)
                best_global_a = np.asarray(a_pol2, dtype=np.float64).copy()
    except Exception as ex:
        print("Final polish exception: %s" % repr(ex))

    # Final sanitize.
    final_a = _clean_positive(best_global_a)
    if final_a is None:
        final_a = np.ones(1000, dtype=np.float64) / 1000.0

    # Optional evaluator verification.
    final_list = [float(x) for x in final_a]
    try:
        ev = evaluate_sequence(final_list)
        if np.isfinite(ev):
            print("\nEvaluator final: %.8f, n=%d, elapsed=%.2fs"
                  % (ev, len(final_list), time.monotonic() - total_start))
        else:
            # If evaluator rejects (should not happen), simplify.
            final_a = np.ones(1000, dtype=np.float64) / 1000.0
            final_list = final_a.tolist()
            print("\nEvaluator invalid; fallback uniform, elapsed=%.2fs"
                  % (time.monotonic() - total_start))
    except Exception as ex:
        print("Evaluator check exception: %s" % repr(ex))

    return final_list
