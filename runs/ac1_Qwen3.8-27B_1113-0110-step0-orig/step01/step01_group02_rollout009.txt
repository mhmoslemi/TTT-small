import numpy as np
import time

def _fft_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    fb = np.fft.rfft(b, fft_size)
    return np.fft.irfft(fa * fb, fft_size)[:size]

def _compute_grad_smooth(a, p):
    """Compute gradient of (sum b[k]^p)^(1/p) where b = conv(a,a)."""
    n = len(a)
    b = _fft_convolve(a, a)  # length 2n-1
    b_pos = np.maximum(b, 1e-30)  # avoid issues
    wp1 = b_pos ** (p - 1)
    
    # c2[j] = sum_m a[m] * w[j-m] = conv(a, w)[j]
    c2 = _fft_convolve(a, wp1)[:n]
    
    # c1[j] = sum_m a[m] * w[m+j] = conv(a_flipped, w)[j+n-1]
    a_flip = a[::-1]
    conv_aw = _fft_convolve(a_flip, wp1)
    c1 = conv_aw[n-1 : 2*n-1]
    
    f = (b_pos ** p).sum() ** (1.0/p)
    if f < 1e-30:
        return np.zeros(n), f
    grad = f ** (1.0 - p) * (c1 + c2)
    return grad, f

def _optimize_smooth(a0, p, time_budget, rng, p_anneal=True):
    """Optimize using smooth max with projected gradient descent."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, np.inf
    a = a / s  # normalize to sum 1
    
    b = _fft_convolve(a, a)
    best_score = 2.0 * n * float(np.max(b))
    best_a = a.copy()
    
    eta = 0.5 / n  # initial step size (for normalized sequence)
    no_improve = 0
    momentum = np.zeros(n)
    mu = 0.8
    deadline = time.monotonic() + time_budget
    it = 0
    last_print = time.monotonic()
    current_p = p
    
    while time.monotonic() < deadline:
        if p_anneal:
            # Gradually increase p to focus more on the peak
            progress = min(1.0, it / 5000.0)
            current_p = p * (1.0 + 3.0 * progress)  # p goes from p to 4p
        
        grad, f = _compute_grad_smooth(a, current_p)
        
        # Score based on actual max
        b = _fft_convolve(a, a)
        score = 2.0 * n * float(np.max(b))
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            if time.monotonic() - last_print > 5.0:
                print(f"    [smooth p={current_p:.1f}] it={it}, score={score:.6f}")
                last_print = time.monotonic()
        else:
            no_improve += 1
        
        # Gradient step
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm
        
        momentum = mu * momentum + (1.0 - mu) * grad
        a = a - eta * momentum
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            eta *= 0.5
            it += 1
            continue
        a = a / s  # renormalize
        
        # Adaptive step size
        if no_improve > 300:
            eta *= 0.95
        elif no_improve < 50:
            eta = min(eta * 1.02, 5.0/n)
        
        # Restart if stuck
        if no_improve > 1000:
            a = best_a * np.exp(rng.normal(0, 0.1, n))
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            no_improve = 0
            eta = 0.5 / n
            momentum = np.zeros(n)
        
        it += 1
    
    # Final evaluation
    b = _fft_convolve(best_a, best_a)
    final_score = 2.0 * n * float(np.max(b))
    return best_a, final_score

        grad = _smooth_grad(a, p)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        momentum = mu * momentum + (1.0 - mu) * grad
        a_new = a - eta * momentum
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a = best_a.copy()
            eta *= 0.5
            it += 1
            continue
        a = a_new / s_new

        # Evaluate every 30 iterations
        if it % 30 == 0 or it < 5:
            b = _fft_convolve(a, a)
            score = 2.0 * n * float(np.max(b))
            if score < best_score:
                best_score = score
                best_a = a.copy()
                no_improve = 0
                if time.monotonic() - last_print > 8.0:
                    elapsed = time.monotonic() - (deadline - time_budget)
                    print(f"    [{seed_print_tag}] it={it} p={p:.0f} score={score:.6f} (t={elapsed:.0f}s)")
                    last_print = time.monotonic()
            else:
                no_improve += 1

        # Adaptive learning rate
        if no_improve > 400:
            eta *= 0.93
        elif no_improve < 40:
            eta = min(eta * 1.015, 20.0)

        # Stuck: restart with noise
        if no_improve > 1200:
            noise_scale = 0.1 + 0.1 * rng.random()
            a = best_a * np.exp(rng.normal(0, noise_scale, n))
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            no_improve = 0
            eta = 3.0
            momentum = np.zeros(n)
            mu = 0.7 + 0.15 * rng.random()

        it += 1

    b_final = _fft_convolve(best_a, best_a)
    final_score = 2.0 * n * float(np.max(b_final))
    return best_a,
