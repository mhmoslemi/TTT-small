def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.

    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def next_pow2(m: int) -> int:
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def normalize_positive(a, floor: float = 0.0) -> np.ndarray:
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, float(floor), None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def softmax_seq(z) -> np.ndarray:
    z = np.asarray(z, dtype=np.float64).ravel()
    z = z - float(np.max(z))
    z = np.clip(z, -60.0, 0.0)
    e = np.exp(z)
    e = np.maximum(e, 1e-300)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(z.size, 1.0 / float(z.size), dtype=np.float64)
    return e / s


def convolve_seq(a, N=None) -> np.ndarray:
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if N is None:
        N = next_pow2(2 * n - 1)
    else:
        need = next_pow2(2 * n - 1)
        if int(N) < int(need):
            N = need
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[: 2 * n - 1]


def convolve_and_fft(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if N is None:
        N = next_pow2(3 * n - 2)
    else:
        need = next_pow2(3 * n - 2)
        if int(N) < int(need):
            N = need
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[: 2 * n - 1]
    return c, fa


def grad_from_weights(w, fa, N, n) -> np.ndarray:
    w = np.asarray(w, dtype=np.float64).ravel()
    w_rev = w[::-1]
    fw = np.fft.rfft(w_rev, N)
    conv = np.fft.irfft(fa * fw, N)
    return 2.0 * conv[n - 1 : 2 * n - 1][::-1]


def compute_weights_floor(c, cmax, tau, floor: float = 0.02) -> np.ndarray:
    c = np.asarray(c, dtype=np.float64).ravel()
    L = c.size
    if L == 0:
        return c
    if cmax <= 0.0 or not np.isfinite(cmax) or tau <= 0.0:
        return np.full(L, 1.0 / float(L), dtype=np.float64)
    z = (c - float(cmax)) / float(tau)
    z = np.clip(z, -50.0, 0.0)
    ez = np.exp(z)
    s = float(ez.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(L, 1.0 / float(L), dtype=np.float64)
    w = ez / s
    if floor > 0.0:
        w = (1.0 - floor) * w + (floor / float(L))
    return w


def smooth_value_from_c(c, cmax, tau) -> float:
    c = np.asarray(c, dtype=np.float64).ravel()
    cmax = float(cmax)
    tau = float(tau)
    if c.size == 0 or cmax <= 0.0 or not np.isfinite(cmax):
        return 0.0
    if tau <= 0.0:
        return cmax
    z = (c - cmax) / tau
    z = np.clip(z, -50.0, 0.0)
    s = float(np.exp(z).sum())
    if not np.isfinite(s) or s <= 0.0:
        return cmax
    return float(cmax + tau * np.log(s))


def internal_score(a, N=None) -> float:
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.inf
    a = a / s
    c = convolve_seq(a, N)
    return float(2.0 * n * float(np.max(c)))


def make_uniform(n: int) -> np.ndarray:
    return np.ones(max(1, int(n)), dtype=np.float64) / float(max(1, int(n)))


def make_sine_sequence(n: int, alphas) -> np.ndarray:
    n = max(2, int(n))
    x = (np.arange(n, dtype=np.float64) - 0.5 * (n - 1)) / max(1.0, 0.5 * n)
    base = np.ones(n, dtype=np.float64)
    for m, alpha in enumerate(alphas, start=1):
        base = base + float(alpha) * np.sin(np.pi * float(m) * x)
    return normalize_positive(base, floor=1e-12)


def make_gaussian_sequence(n: int, mu: float, sigma: float) -> np.ndarray:
    n = max(2, int(n))
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    sigma = max(1e-3, float(sigma))
    return normalize_positive(np.exp(-((x - float(mu)) ** 2) / (2.0 * sigma * sigma)), floor=1e-12)


def make_exponential_sequence(n: int, slope: float) -> np.ndarray:
    n = max(2, int(n))
    t = np.linspace(-1.0, 1.0, n, dtype=np.float64)
    return normalize_positive(np.exp(float(slope) * t), floor=1e-12)

def resample_seq(a, n: int) -> np.ndarray:
    a = normalize_positive(a, floor=0.0)
    old_n = a.size
    n = max(2, int(n))
    if old_n == 1:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    if old_n == n:
        return a.astype(np.float64, copy=True)
    old_x = np.linspace(0.0, 1.0, old_n, dtype=np.float64)
    new_x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    b = np.interp(new_x, old_x, a)
    return normalize_positive(b, floor=1e-12)

def build_logparam_basis(n: int) -> np.ndarray:
    n = max(8, int(n))
    i = np.arange(n, dtype=np.float64)
    x = (i - 0.5 * (n - 1)) / max(1.0, 0.5 * n)
    cols = []

    # Antisymmetric odd modes, useful for reducing the central convolution coefficient.
    for m in range(1, 5):
        cols.append(np.sin(np.pi * float(m) * x))

    # Smooth polynomial modes.
    cols.append(x.copy())
    cols.append(x * x - np.mean(x * x))
    cols.append(x * x * x - np.mean(x * x * x))
    cols.append(x ** 4 - np.mean(x ** 4))

    # Periodic modes.
    for m in range(1, 5):
        t = 2.0 * np.pi * float(m) * i / float(n)
        cols.append(np.sin(t) - np.mean(np.sin(t)))
        cols.append(np.cos(t) - np.mean(np.cos(t)))

    B = np.column_stack(cols)
    B -= B.mean(axis=0)
    norms = np.linalg.norm(B, axis=0)
    norms[norms <= 1e-14] = 1.0
    B = B / norms
    return B


def theta_value(theta, n, B, tau, N) -> float:
    theta = np.asarray(theta, dtype=np.float64).ravel()
    a = softmax_seq(B @ theta)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[: 2 * n - 1]
    cmax = float(np.max(c))
    if cmax <= 0.0 or not np.isfinite(cmax):
        return 0.0
    tau_eff = max(float(tau), 1e-12)
    val = smooth_value_from_c(c, cmax, tau_eff)
    return float(2.0 * n * val)


def theta_grad(theta, n, B, tau, N) -> np.ndarray:
    theta = np.asarray(theta, dtype=np.float64).ravel()
    a = softmax_seq(B @ theta)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[: 2 * n - 1]
    cmax = float(np.max(c))
    D = B.shape[1]
    if cmax <= 0.0 or not np.isfinite(cmax):
        return np.zeros(D, dtype=np.float64)

    tau_eff = max(float(tau), 1e-12)
    z = (c - cmax) / tau_eff
    z = np.clip(z, -50.0, 0.0)
    ez = np.exp(z)
    s = float(ez.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.zeros(D, dtype=np.float64)

    w = ez / s
    ga = grad_from_weights(w, fa, N, n)
    total = float(np.dot(a, ga))
    aB = a @ B
    agaB = (a * ga) @ B
    gtheta = agaB - total * aB
    return (2.0 * n) * gtheta


def optimize_lowdim(n, deadline, seed, initial_a=None, max_starts: int = 4):
    n = max(32, int(n))
    B = build_logparam_basis(n)
    D = B.shape[1]
    N = next_pow2(3 * n - 2)
    rng = np.random.default_rng(int(seed))

    starts = []
    if initial_a is not None:
        try:
            a0 = normalize_positive(np.asarray(initial_a, dtype=np.float64).ravel(), floor=1e-12)
            if a0.size == n:
                z = np.log(np.maximum(a0, 1e-12))
                z -= z.mean()
                theta_proj, _, _, _ = np.linalg.lstsq(B, z, rcond=None)
                starts.append(np.asarray(theta_proj, dtype=np.float64).ravel())
        except Exception:
            pass

    for std in (0.30, 0.60, 1.00, 1.50):
        starts.append(rng.normal(0.0, std, size=D))

    for ci in range(min(D, 5)):
        th = np.zeros(D, dtype=np.float64)
        th[ci] = 1.0
        starts.append(th)

    starts = starts[: max(1, int(max_starts))]

    best_a = None
    best_score = np.inf

    for theta0 in starts:
        if time.monotonic() > deadline:
            break
        theta = np.asarray(theta0, dtype=np.float64).ravel().copy()
        a = softmax_seq(B @ theta)
        sc = internal_score(a, N)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if _scipy_optimize is None:
            break

        max_iter = 20 if n <= 1536 else 12
        factors = (0.40, 0.15, 0.06, 0.025)
        for fac in factors:
            if time.monotonic() > deadline:
                break
            tau = fac / float(n)
            try:
                res = _scipy_optimize.minimize(
                    theta_value,
                    theta,
                    method="L-BFGS-B",
                    jac=theta_grad,
                    args=(n, B, tau, N),
                    options={"maxiter": max_iter, "ftol": 1e-12, "gtol": 1e-6, "maxls": 12},
                )
                if res is not None and res.x is not None and np.all(np.isfinite(res.x)):
                    theta = np.asarray(res.x, dtype=np.float64).ravel()
            except Exception:
                pass

            a = softmax_seq(B @ theta)
            sc = internal_score(a, N)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                print("lowdim n={} score={:.8f}".format(n, best_score), flush=True)

    if best_a is None:
        best_a = make_uniform(n)
        best_score = internal_score(best_a, N)
    return best_a, best_score


def optimize_mirror(a, deadline, verbose: bool = False):
    a = normalize_positive(a, floor=1e-15)
    n = a.size
    N = next_pow2(3 * n - 2)

    best_a = a.copy()
    best_score = internal_score(a, N)

    eta = 0.65
    beta = 0.18
    w_floor = 0.045
    stagnant = 0
    it = 0
    last_best = best_score

    while time.monotonic() < deadline:
        c, fa = convolve_and_fft(a, N)
        cmax = float(np.max(c))
        score = float(2.0 * n * cmax)
        if np.isfinite(score) and score < best_score * (1.0 - 1e-12):
            best_score = score
            best_a = a.copy()

        if cmax <= 0.0 or not np.isfinite(cmax):
            break

        tau = beta * cmax
        w = compute_weights_floor(c, cmax, tau, floor=w_floor)
        g = grad_from_weights(w, fa, N, n)
        g = g - g.mean()
        scale = float(np.abs(g).max())
        if not np.isfinite(scale) or scale <= 1e-15:
            break
        g = g / scale

        loga = np.log(a + 1e-20)
        J_cur = smooth_value_from_c(c, cmax, tau)
        accepted = False

        for e in (eta, 0.5 * eta, 0.25 * eta, 0.125 * eta):
            if e <= 1e-9:
                continue
            loga_try = loga - e * g
            loga_try = loga_try - float(np.max(loga_try))
            a_try = np.exp(np.clip(loga_try, -60.0, 0.0))
            a_try = np.maximum(a_try, 1e-300)
            a_try = a_try / float(a_try.sum())
            a_try = normalize_positive(a_try, floor=1e-15)

            c_try, _ = convolve_and_fft(a_try, N)
            cmax_try = float(np.max(c_try))
            score_try = float(2.0 * n * cmax_try)
            J_try = smooth_value_from_c(c_try, cmax_try, tau)

            if np.isfinite(score_try):
                if score_try < score * (1.0 - 1e-10):
                    accepted = True
                    a = a_try
                    score = score_try
                    break
                if np.isfinite(J_try) and J_try < J_cur * (1.0 - 1e-7) and score_try < score * 1.05:
                    accepted = True
                    a = a_try
                    score = score_try
                    break

        if accepted:
            stagnant = 0
            eta = min(1.25, eta * 1.025)
            beta = max(0.004, beta * 0.999)
            w_floor = max(0.0, w_floor * 0.995)
        else:
            stagnant += 1
            eta = max(1e-4, eta * 0.62)
            beta = max(0.003, beta * 0.82)
            if stagnant % 80 == 0:
                beta = min(0.25, beta * 1.7)
                w_floor = min(0.08, w_floor + 0.005)
                stagnant = 0
            if eta < 1e-5 or stagnant > 700:
                break

        it += 1
        if verbose and it % 200 == 0:
            print("mirror n={} iter={} best={:.8f}".format(n, it, best_score), flush=True)

        if it % 500 == 0 and best_score > last_best - 1e-7:
            # Very occasional aggressive escape.
            tmp = a * np.exp(np.clip(np.random.default_rng(it + n).normal(0.0, 0.05, n), -1.0, 1.0))
            tmp = normalize_positive(tmp, floor=1e-15)
            sc_tmp = internal_score(tmp, N)
            if np.isfinite(sc_tmp) and sc_tmp < best_score:
                a = tmp
                best_score = sc_tmp
                best_a = a.copy()
            last_best = best_score

    return best_a, best_score


def binary_from_values(a0, m: int) -> np.ndarray:
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    m = int(max(0, min(n, m)))
    b = np.zeros(n, dtype=np.float64)
    if m <= 0:
        return b
    if m >= n:
        b.fill(1.0)
        return b

    is_binary = np.all((a0 <= 0.5) | (a0 >= 0.5))
    if is_binary:
        vals = (a0 > 0.5).astype(np.float64)
        cur = int(vals.sum())
        if cur == m:
            return vals
        b = vals.copy()
        if cur > m:
            idx = np.nonzero(b > 0.5)[0]
            if idx.size:
                b[idx[-(cur - m):]] = 0.0
        else:
            idx = np.nonzero(b < 0.5)[0]
            if idx.size:
                b[idx[: m - cur]] = 1.0
        return b

    order = np.argsort(a0, kind="stable")
    b[order[-m:]] = 1.0
    return b


def choose_by_pressure(g, rng, pick_high: bool, temp: float = 0.12) -> int:
    g = np.asarray(g, dtype=np.float64).ravel()
    m = g.size
    if m == 0:
        return -1
    if m == 1:
        return 0
    if not np.all(np.isfinite(g)):
        g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
    vmin = float(np.min(g))
    vmax = float(np.max(g))
    range_g = vmax - vmin
    if range_g <= 1e-14:
        return int(rng.integers(0, m))
    denom = max(float(temp), 1e-12) * range_g
    if pick_high:
        z = (g - vmax) / denom
    else:
        z = (vmin - g) / denom
    z = np.clip(z, -50.0, 0.0)
    p = np.exp(z)
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        return int(rng.integers(0, m))
    p = p / s
    return int(rng.choice(m, p=p))


def optimize_binary(a0, deadline, seed: int = 0):
    a0 = np.asarray(a0, dtype=np.float64).ravel()
