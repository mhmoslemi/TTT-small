"""
Multi-stage search for non-negative coefficient sequences minimizing
2*n*max(convolve(a,a))/sum(a)^2.

Strategy:
- Generate many smooth/asymmetric initial templates, including the provided
  previous best if available.
- Warm up candidates with soft-max multiplicative/inverse fixed-point updates.
- Polish top candidates with annealed soft gradient updates, scipy L-BFGS-B on
  log-softmax parameters with a smooth max approximation, and (when available)
  fixed-f linear-programming directions from the previous state-of-art idea.
- Final exact high-temperature polish and validation.
"""

import numpy as np
import time
import math

_MINIMIZE = None
_LINPROG = None
_CSR_MATRIX = None
try:
    from scipy.optimize import minimize as _sp_minimize
    from scipy.optimize import linprog as _sp_linprog
    from scipy.sparse import csr_matrix as _sp_csr
    _MINIMIZE = _sp_minimize
    _LINPROG = _sp_linprog
    _CSR_MATRIX = _sp_csr
except Exception:
    pass

_LP_CSR_CACHE = {}


def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _fft_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([], dtype=float)
    size = 2 * n - 1
    m = _next_pow2(size)
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:size]


def _normalize(a):
    a = np.asarray(a, dtype=float).ravel()
    if a.size == 0:
        return a
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-15:
        return np.ones_like(a) / max(1, a.size)
    return a / s


def _score(a):
    a = _normalize(a)
    n = len(a)
    if n == 0:
        return np.inf
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    c = np.maximum(c, 0.0)
    m = float(np.max(c))
    if not np.isfinite(m):
        return np.inf
    return 2.0 * float(n) * m


def _append_candidate(lst, a):
    if a is None:
        return
    a = np.asarray(a, dtype=float).ravel()
    if len(a) < 50:
        return
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-15:
        return
    a = a / s
    lst.append(a)


def _resample(a, n):
    a = _normalize(a)
    n0 = len(a)
    if n0 == n:
        return a.copy()
    if n0 == 0:
        return np.ones(n) / float(n)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n)
    y = np.interp(x1, x0, a)
    return _normalize(y)


def _softmax_w(w):
    w = np.asarray(w, dtype=float)
    m = float(np.max(w))
    e = np.exp(w - m)
    s = float(e.sum())
    if s < 1e-300:
        return np.ones_like(w) / float(w.size)
    return e / s


def _softmax_weights(q, beta):
    if beta <= 1e-8:
        return np.full_like(q, 1.0 / max(1, len(q)))
    q = np.asarray(q, dtype=float)
    m = float(np.max(q))
    z = beta * (q - m)
    e = np.exp(np.maximum(z, -745.0))
    s = float(e.sum())
    if s < 1e-300:
        out = np.zeros_like(q)
        out[np.argmax(q)] = 1.0
        return out
    return e / s


def _shift_sum_direct(p, a):
    n = len(a)
    S = np.zeros(n, dtype=float)
    nz = np.nonzero(p > 1e-12)[0]
    for kk in nz:
        k = int(kk)
        pk = float(p[k])
        if pk <= 0.0:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            idx_start = k - hi
            idx_end = k - lo
            S[lo:hi + 1] += pk * a[idx_start:idx_end + 1][::-1]
    return S


def _shift_sum_fft(p, a):
    n = len(a)
    size = 2 * n - 1
    m = _next_pow2(size)
    fa = np.fft.fft(a, m)
    fp = np.fft.fft(p, m)
    c = np.fft.ifft(np.conj(fa) * fp, m).real
    return c[:n]


def _shift_sum(p, a):
    n = len(a)
    nz = np.nonzero(p > 1e-12)[0]
    if len(nz) <= 64:
        return _shift_sum_direct(p, a)
    return _shift_sum_fft(p, a)


def _optimize_soft_fixed(a0, time_budget, rng, beta=5000.0, eta=0.22, mode=0):
    a = _normalize(a0)
    n = len(a)
    if n == 0 or time_budget <= 0:
        return a

    q = _fft_convolve(a)
    q = np.maximum(q, 0.0)
    best_score = 2.0 * float(n) * float(np.max(q))
    best_a = a.copy()

    deadline = time.monotonic() + float(time_budget)
    momentum = np.zeros(n, dtype=float)
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        q = _fft_convolve(a)
        q = np.maximum(q, 0.0)
        score = 2.0 * float(n) * float(np.max(q))

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 900:
            a = best_a * np.exp(rng.normal(0.0, 0.05, n))
            a = np.maximum(a, 1e-20)
            block = max(1, n // 60)
            start = int(rng.integers(0, max(1, n - block)))
            a[start:start + block] *= np.exp(rng.normal(0.0, 0.18))
            a = _normalize(a)
            momentum = np.zeros(n, dtype=float)
            no_improve = 0
            it = 0

        p = _softmax_weights(q, beta)
        S = _shift_sum(p, a)
        smax = float(np.max(S))
        if smax <= 1e-16:
            break

        if mode == 0:
            g = S / smax
            momentum = 0.72 * momentum + 0.28 * g
            a = a * np.exp(-eta * momentum)
        elif mode == 1:
            inv = 1.0 / (S + 1e-10 * smax + 1e-20)
            inv = inv / float(inv.sum())
            a = 0.35 * a + 0.65 * inv
        else:
            target = 1.0 / (S + 1e-10 * smax + 1e-20)
            target = target / float(target.sum())
            aa = np.maximum(a, 1e-300)
            tt = np.maximum(target, 1e-300)
            e = min(max(float(eta), 0.02), 0.6)
            a = np.power(aa, 1.0 - e) * np.power(tt, e)

        a = np.maximum(a, 1e-20)
        s = float(a.sum())
        if s < 1e-15:
            a = best_a.copy()
            eta *= 0.5
        else:
            a = a / s

        if no_improve > 700:
            eta *= 0.96
        elif no_improve < 40:
            eta = min(eta * 1.004, 0.6)

        it += 1

    return best_a


def _soft_value_jac(w, beta):
    n = len(w)
    a = _softmax_w(w)
    q = _fft_convolve(a)
    q = np.maximum(q, 0.0)
    m = float(np.max(q))

    if beta <= 1e-8:
        p = np.full_like(q, 1.0 / max(1, len(q)))
        val = m
    else:
        z = beta * (q - m)
        e = np.exp(np.maximum(z, -745.0))
        denom = float(e.sum())
        if denom < 1e-300:
            p = np.zeros_like(q)
            p[int(np.argmax(q))] = 1.0
            val = m
        else:
            p = e / denom
            val = m + math.log(denom) / beta

    S = _shift_sum(p, a)
    grad_a = 2.0 * S
    mean_grad = float(np.dot(a, grad_a))
    grad_w = a * (grad_a - mean_grad)

    if not np.all(np.isfinite(grad_w)):
        grad_w = np.zeros_like(w)

    return float(val), grad_w


def _lbfgs_refine(a0, time_budget, beta):
    if _MINIMIZE is None or time_budget <= 0:
        return _normalize(a0)
    a = _normalize(a0)
    if len(a) == 0:
        return a
    w0 = np.log(np.maximum(a, 1e-300))
    w0 = w0 - float(np.mean(w0))

    maxiter = int(min(80, max(15, float(time_budget) * 0.8)))
    if len(a) > 3000:
        maxiter = max(10, maxiter // 2)

    bounds = [(-100.0, 100.0)] * len(w0)
    try:
        res = _MINIMIZE(
            _soft_value_jac,
            w0,
            args=(beta,),
            jac=True,
            method="L-BFGS-B",
            bounds=bounds,
            options={
                "maxiter": maxiter,
                "ftol": 1e-12,
                "gtol": 1e-8,
                "maxls": 20,
                "disp": False,
            },
        )
    except Exception:
        return a

    if res is not None and hasattr(res, "x") and res.x is not None:
        a2 = _softmax_w(res.x)
        a2 = _normalize(a2)
        if np.all(np.isfinite(a2)) and a2.sum() > 0.01:
            return a2
    return a


def _get_lp_matrix(n):
    if n in _LP_CSR_CACHE:
        return _LP_CSR_CACHE[n]

    if _CSR_MATRIX is None:
        return None

    m = 2 * n - 1
    total = n * n
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    pos = 0
    for j in range(n):
        lo = j
        hi = min(2 * n - 2, j + n - 1)
        cnt = hi - lo + 1
        rows[pos:pos + cnt] = np.arange(lo, hi + 1, dtype=np.int32)
        cols[pos:pos + cnt] = np.full(cnt, j, dtype=np.int32)
        pos += cnt

    A = _CSR_MATRIX((np.ones(total, dtype=float), (rows, cols)), shape=(m, n))
    row_for_data = np.repeat(np.arange(m, dtype=np.int64), np.diff(A.indptr))
    val_idx = (A.indices.astype(np.int64) - row_for_data).astype(np.int32)
    val_idx = np.clip(val_idx, 0, n - 1)
    _LP_CSR_CACHE[n] = (A, val_idx)
    return _LP_CSR_CACHE[n]


def _lp_direction(f, margin, time_limit):
    if _LINPROG is None:
        return None
    f = _normalize(f)
    n = len(f)
    if n < 500 or n > 2000:
        return None

    cab = _get_lp_matrix(n)
    if cab is None:
        return None
    A, val_idx = cab
    A.data[:] = f[val_idx]

    q = _fft_convolve(f)
    q = np.maximum(q, 0.0)
    M = float(np.max(q)) * max(1.0, float(margin)) * (1.0 + 1e-9)
    b = np.full(2 * n - 1, M, dtype=float)
    c = -np.ones(n, dtype=float)
    bounds = [(0.0, None)] * n

    try:
        res = _LINPROG(
            c,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "disp": False, "time_limit": float(time_limit)},
        )
    except Exception:
        try:
            res = _LINPROG(
                c,
                A_ub=A,
                b_ub=b,
                bounds=bounds,
                method="highs",
                options={"presolve": True, "disp": False},
            )
        except Exception:
            return None

    if res is None:
        return None
    status = getattr(res, "status", 0)
    if status not in (0, 1):
        return None
    g = np.asarray(getattr(res, "x", None), dtype=float)
    if g is None or g.size != n or not np.all(np.isfinite(g)):
        return None
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s < 1e-12:
        return None
    g = g / s
    return g


def _make_pool(rng):
    candidates = []

    try:
        h = np.array(height_sequence_1, dtype=float).ravel()
        h = np.where(np.isfinite(h), h, 0.0)
        h = np.clip(h, 0.0, 1000.0)
        if h.size >= 50 and float(h.sum()) > 0.0:
            hb = _normalize(h)
            _append_candidate(candidates, hb)
            base_n = len(hb)
            for mult in [0.75, 1.0, 1.25, 1.5, 2.0]:
                n_r = int(round(base_n * mult))
                if 800 <= n_r <= 8000:
                    _append_candidate(candidates, _resample(hb, n_r))
        _PROVIDED_BASE = None
    except NameError:
        _PROVIDED_BASE = None
    except Exception:
        _PROVIDED_BASE = None

    n_list = [1500, 2000, 2500, 3000, 4000]
    for n in n_list:
        x = np.linspace(0.0, 1.0, n)
        eps = 1e-9

        for freq in [1, 2]:
            for amp in [0.2, 0.35, 0.5, 0.7, 0.9]:
                for phase in [0.0, math.pi / 8.0, math.pi / 4.0, math.pi / 2.0, 3.0 * math.pi / 4.0]:
                    a = 1.0 + amp * np.sin(2.0 * math.pi * freq * x + phase)
                    _append_candidate(candidates, np.maximum(a, eps))

        for alpha in [0.5, 1.0, 2.0, 4.0, 8.0]:
            _append_candidate(candidates, np.exp(-alpha * x) + eps)
            _append_candidate(candidates, np.exp(alpha * x) + eps)

        for p in [0.2, 0.5, 1.0, 2.0, 3.0]:
            _append_candidate(candidates, x ** p + eps)
            _append_candidate(candidates, (1.0 - x) ** p + eps)

        if n <= 4000:
            for c0 in [-2.0, 0.0, 2.0]:
                for c1 in [-2.0, 0.0, 2.0]:
                    for c2 in [-1.0, 0.0, 1.0]:
                        z = c0 * x + c1 * x * x + c2 * x * x * x
                        z = z - float(np.max(z))
                        _append_candidate(candidates, np.exp(z) + eps)

        for _ in range(4):
            a = np.full(n, 0.05, dtype=float)
            nbumps = int(rng.integers(1, 4))
            for _b in range(nbumps):
                c = float(rng.uniform(0.1, 0.9))
                s = float(rng.uniform(0.05, 0.25))
                amp = float(rng.uniform(0.5, 2.0))
                a += amp * np.exp(-0.5 * ((x - c) / s) ** 2)
            _append_candidate(candidates, a)

    return candidates


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    total_start = time.monotonic()
    deadline = total_start + max(60.0, float(budget_s)) - 45.0
    rng = np.random.default_rng(seed)

    best_a = None
    best_score = np.inf

    def condemn():
        pass

    def consider(a):
        nonlocal best_a, best_score
        if a is None:
            return
        a = _normalize(a)
        if len(a) == 0:
            return
        sc = _score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"    new best internal score = {best_score:.9f}, n={len(best_a)}")

    print(f"Starting search with {budget_s}s budget, deadline margin 45s.")

    pool = _make_pool(rng)
    scored = []
    idx = 0
    for a in pool:
        a = _normalize(a)
        sc = _score(a)
        if np.isfinite(sc):
            scored.append((float(sc), idx, a))
        idx += 1

    scored.sort()
    print(f"Generated {len(pool)} pool candidates, {len(scored)} finite scores.")
    for k in range(min(8, len(scored))):
        print(f"  pool {k + 1}: score={scored[k][0]:.9f}, n={len(scored[k][2])}")

    if len(scored) == 0:
        best_a = np.ones(2000, dtype=float) / 2000.0
        consider(best_a)

    remaining = deadline - time.monotonic()
    if remaining > 120 and len(scored) > 0:
        warm_budget = min(160.0, remaining * 0.25)
        n_warm = min(6, len(scored))
        per = warm_budget / max(1, n_warm)
        warm_results = []

        print("\nWarm-up阶段...")
        for i in range(n_warm):
            if time.monotonic() > deadline - 60:
                break
            a0 = scored[i][2]
            t1 = min(per, max(0.0, deadline - time.monotonic() - 30.0))
            a1 = a0
            if t1 > 3.0:
                a1 = _optimize_soft_fixed(a0, t1, rng, beta=2000.0, mode=0, eta=0.22)
                consider(a1)
            t2 = min(8.0, max(0.0, deadline - time.monotonic() - 30.0))
            a2 = a1
            if t2 > 2.0:
                a2 = _optimize_soft_fixed(a1, t2, rng, beta=30000.0, mode=2, eta=0.28)
                consider(a2)
            scw = _score(a2)
            if np.isfinite(scw):
                warm_results.append((float(scw), i, a2))

        active = []
        for i in range(min(4, len(scored))):
            active.append((scored[i][0], -1000 + i, scored[i][2]))
        for sc, ord_i, a in warm_results:
            active.append((float(sc), ord_i * 100 + 1 if ord_i is not None else 0, a))
        active.sort()
    else:
        active = []
        for i in range(min(4, len(scored))):
            active.append((scored[i][0], -1000 + i, scored[i][2]))

    if best_a is not None and len(active) > 0:
        active.insert(0, (float(best_score), -5000, best_a.copy()))
        active.sort()

    detailed = active[:4]
    L = len(detailed)
    weights = []
    for i in range(L):
        weights.append(max(1, L - i))
    sum_w = sum(weights)

    detail_budget = max(0.0, deadline - time.monotonic() - 90.0)
    detail_budget = min(detail_budget, 1700.0)

    print(f"\nDetailed optimization: {L} candidates, budget {detail_budget:.1f}s")

    for idxd in range(L):
        if time.monotonic() > deadline - 30:
            break
        sc0, _ord, a_start = detailed[idxd]
        t_alloc = detail_budget * float(weights[idxd]) / float(sum_w)
        if t_alloc < 15.0:
            continue

        print(f"\nCandidate {idxd + 1}: initial {sc0:.9f}, n={len(a_start)}, alloc={t_alloc:.1f}s")
        a_cur = a_start

        if t_alloc > 30.0:
            t1 = min(t_alloc * 0.25, max(0.0, deadline - time.monotonic() - 60.0))
            if t1 > 5.0:
                a_cur = _optimize_soft_fixed(a_cur, t1, rng, beta=1500.0, mode=0, eta=0.24)
                consider(a_cur)

            t2 = min(t_alloc * 0.25, max(0.0, deadline - time.monotonic() - 60.0))
            if t2 > 5.0:
                a_cur = _optimize_soft_fixed(a_cur, t2, rng, beta=30000.0, mode=2, eta=0.27)
                consider(a_cur)

            t3 = min(t_alloc * 0.30, max(0.0, deadline - time.monotonic() - 60.0))
            if t3 > 5.0:
                a_cur = _optimize_soft_fixed(a_cur, t3, rng, beta=250000.0, mode=0, eta=0.11)
                consider(a_cur)

            t4 = min(t_alloc * 0.20, max(0.0, deadline - time.monotonic() - 45.0))
            if t4 > 5.0:
                a_cur = _optimize_soft_fixed(a_cur, t4, rng, beta=250000.0, mode=2, eta=0.22)
                consider(a_cur)

        if _MINIMIZE is not None and len(a_cur) <= 3200 and (deadline - time.monotonic()) > 30:
            tb = min(60.0, max(0.0, deadline - time.monotonic() - 45.0))
            if tb > 10.0:
                a_lb = _lbfgs_refine(a_cur, tb, beta=20000.0)
                consider(a_lb)
                a_cur = a_lb

            tb = min(60.0, max(0.0, deadline - time.monotonic() - 30.0))
            if tb > 10.0:
                a_lb2 = _lbfgs_refine(a_cur, tb, beta=250000.0)
                consider(a_lb2)
                a_cur = a_lb2

        if _LINPROG is not None and 800 <= len(a_cur) <= 2000 and (deadline - time.monotonic()) > 90:
            f_base = _normalize(a_cur)
            margins = [1.0 + 1e-7, 1.05, 1.15, 1.35, 1.8]
            for mg in margins:
                if (deadline - time.monotonic()) < 30:
                    break
                tl = min(60.0, max(5.0, (deadline - time.monotonic()) * 0.25))
                d = _lp_direction(f_base, mg, tl)
                if d is None:
                    continue
                for t in [0.02, 0.05, 0.1, 0.2, 0.35, 0.6, 1.0]:
                    aa = (1.0 - t) * f_base + t * d
                    consider(aa)
                if best_a is not None:
                    f_base = _normalize(best_a)

    if best_a is not None:
        rem = deadline - time.monotonic()
        if rem > 60:
            print(f"\nFinal polish, remaining {rem:.1f}s")
            t1 = min(110.0, rem * 0.35)
            a_pol = _optimize_soft_fixed(best_a, t1, rng, beta=250000.0, mode=0, eta=0.09)
            consider(a_pol)

            rem = deadline - time.monotonic()
            if rem > 45:
                t2 = min(90.0, rem * 0.45)
                a_pol2 = _optimize_soft_fixed(a_pol, t2, rng, beta=250000.0, mode=2, eta=0.20)
                consider(a_pol2)
                a_pol = a_pol2

            rem = deadline - time.monotonic()
            if rem > 45:
                pert = best_a * np.exp(rng.normal(0.0, 0.035, len(best_a)))
                pert = _normalize(pert)
                t3 = min(60.0, rem - 15.0)
                a_pol3 = _optimize_soft_fixed(pert, t3, rng, beta=120000.0, mode=0, eta=0.14)
                consider(a_pol3)

    if best_a is None:
        best_a = np.ones(2000, dtype=float) / 2000.0

    best_a = _normalize(best_a)
    final_internal = _score(best_a)
    print(f"Final internal score: {final_internal:.9f}, length={len(best_a)}")

    try:
        final_eval = evaluate_sequence([float(x) for x in best_a])
        print(f"Final evaluate_sequence score: {final_eval:.9f}")
        if not np.isfinite(final_eval):
            best_a = np.ones(2000, dtype=float) / 2000.0
    except Exception:
        pass

    out = []
    for val in best_a:
        v = float(val)
        if not np.isfinite(v) or v < 0.0:
            v = 0.0
        if v > 1000.0:
            v = 1000.0
        out.append(v)
    return out
