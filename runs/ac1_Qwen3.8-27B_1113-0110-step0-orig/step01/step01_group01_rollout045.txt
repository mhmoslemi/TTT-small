Actually, I realize I should also think about what "projected subgradient" means in practice.

We want to minimize f(a) = 2n * max_k c_k(a) / (sum a)^2 over a >= 0, sum a = 1 (WLOG, since f is homogeneous of degree 0).

On the simplex S = {a >= 0, sum a = 1}, f(a) = 2n * max_k c_k(a).

The projected subgradient method:
1. Compute g = subgradient of max c_k at a
2. a_new = project(a - eta * g, S)
3. If f(a_new) < f(a), accept; otherwise, reduce eta.

The projection onto S is:
Given v, find the point p in S closest to v (in Euclidean distance).
Algorithm (Duchi et al. 2008):
- Sort v in decreasing order
- Find the largest rho such that sum_{i: v[i] > v_bar} (v[i] - v_bar) > 1
- p[i] = max(v[i] - v_bar, 0)

def project_simplex(v):
    n = v.size
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) - (cv - 1) > 0)[0]
    if rho.size == 0:
        return np.ones(n, dtype=np.float64) / n
    rho = rho[-1]
    theta = (cv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    for k in indices:

        # a[k-j] is nonzero when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        lo1 = max(0, k - n + 1)
        hi1 = min(n - 1, k)
        if lo1 <= hi1:

        # a[j-k] is nonzero when 0 <= j-k < n, i.e., k <= j <= k+n-1

        hi2 = min(n - 1, k + n - 1)
        if lo2 <= hi2:

    temp = 0.3
    stale = 0
    it = 0

    one_idx = np.nonzero(b > 0.5)[0]
    zero_idx = np.nonzero(b < 0.5)[0]

    while time.monotonic() < deadline:
        if sc < best_sc * (1.0 - 1e-12):
            best_sc = sc
            best_b = b.copy()
            stale = 0

        # Choose swap based on pressure (gradient)
        # Simple: pick random i from ones, j from zeros
        i = one_idx[rng.integers(0, one_idx.size)]
        j = zero_idx[rng.integers(0, zero_idx.size)]

        # Try swap
        c[i:i + n] -= 2.0 * b
        c[2 * i] -= 1.0
        c[j:j + n] += 2.0 * b
        c[2 * j] += 1.0
        b[i] = 0.0
        b[j] = 1.0
        one_idx = np.concatenate([one_idx[one_idx != i], [j]])
        zero_idx = np.concatenate([zero_idx[zero_idx != j], [i]])

        new_cmax = float(c.max())
        new_sc = float(2.0 * n * new_cmax / (m * m))
        delta = new_sc - sc

        accept = False
        if delta < 0:
            accept = True
        elif temperature_ok(delta, temp, rng):
            accept = True

        if accept:
            sc = new_sc
            temp = max(0.005, temp * 0.9995)
        else:
            # Revert
            c[j:j + n] -= 2.0 * b
            c[2 * j] -= 1.0
            c[i:i + n] += 2.0 * b
            c[2 * i] += 1.0
            b[j] = 0.0
            b[i] = 1.0
            one_idx = np.concatenate([one_idx[one_idx != j], [i]])
            zero_idx = np.concatenate([zero_idx[zero_idx != i], [j]])
            stale += 1
            temp = max(0.003, temp * 0.997)
            if stale > 500:
                temp = min(0.5, temp * 3.0)
                stale = 0

        it += 1
        if verbose and it % 5000 == 0:
            print("  binary_sa it={} temp={:.4f} best={:.9f}".format(
                it, temp, best_sc), flush=True)

        if best_sc < 1.4:
            break

    return normalize(best_b, floor=0.0), best_sc


def temperature_ok(delta, temp, rng):
    if delta <= 0:
        return True
    if delta > 2.0:
        return False
    return rng.random() < np.exp(-delta / max(temp, 1e-10))


def resample(a, n_new):
    a = normalize(a, floor=0.0)
    n_old = a.size
    n_new = max(4, int(n_new))
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    b = np.interp(x_new, x_old, a)
    return normalize(b, floor=1e-12)


def generate_varied_inits(n, rng):
    """Generate diverse initial sequences."""
    inits = []
    # Uniform
    inits.append(np.ones(n) / n)
    # Ramped
    x = np.linspace(0, 1, n)
    inits.append(normalize(x + 0.1, floor=1e-6))
    inits.append(normalize(1 - x + 0.1, floor=1e-6))
    # Sinusoidal
    for k in (1, 2, 3, 5):
        for amp in (0.3, 0.6, 0.9):
            v = 1.0 + amp * np.sin(2 * np.pi * k * x + rng.uniform(0, 2 * np.pi))
            inits.append(normalize(v, floor=1e-6))
    # Gaussian windows
    for mu in (0.3, 0.5, 0.7):
        for sig in (0.1, 0.2, 0.3):
            v = np.exp(-((x - mu) ** 2) / (2 * sig * sig))
            inits.append(normalize(v, floor=1e-6))
    # Random Fourier
    for _ in range(5):
        coeffs = rng.normal(0, 1, size=n)
        v = np.fft.irfft(np.abs(coeffs))
        inits.append(normalize(v, floor=1e-6))
    # Bimodal
    for mu1, mu2 in ((0.25, 0.75), (0.2, 0.8), (0.3, 0.7)):
        v = np.exp(-((x - mu1) ** 2) / 0.02) + np.exp(-((x - mu2) ** 2) / 0.02)
        inits.append(normalize(v, floor=1e-6))
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    start = time.monotonic()
    total_budget = max(100.0, float(budget_s) - 30.0)
    deadline = start + total_budget

    print("Total budget: {:.1f}s".format(total_budget), flush=True)

    # Get given sequence
    given_a = None
    try:
        seq = globals().get("height_sequence_1", None)
        if seq is not None:
            given_a = normalize(np.asarray(seq, dtype=np.float64).ravel(), floor=0.0)
            sc = compute_score(given_a)
            print("Given sequence: n={}, score={:.9f}".format(given_a.size, sc), flush=True)
    except Exception:
        pass

    best_a = given_a.copy() if given_a is not None else np.ones(1024) / 1024
    best_score = compute_score(best_a)

    # Phase 1: Try multiple initializations and quick optimizations
    phase1_deadline = start + total_budget * 0.15
    rng = np.random.default_rng(seed)

    for n_try in (1024, 2048):
        if time.monotonic() > phase1_deadline:
            break
        a_start = resample(best_a, n_try) if best_a.size != n_try else best_a.copy()
        inits = generate_varied_inits(n_try, rng)
        for a0 in inits:
            if time.monotonic() > phase1_deadline:
                break
            sc = compute_score(a0)
            if sc < best_score:
                best_score = sc
                best_a = a0.copy()
                print("  Phase1 init n={} score={:.9f}".format(n_try, sc), flush=True)

    # Phase 2: Subgradient optimization on best length
    phase2_deadline = start + total_budget * 0.35
    n_best = best_a.size
    print("Phase 2: Subgradient on n={}".format(n_best), flush=True)
    a1, sc1 = optimize_subgradient(best_a, phase2_deadline, seed=seed + 1, verbose=True)
    if sc1 < best_score:
        best_score = sc1
        best_a = a1.copy()

    # Phase 3: Mirror descent v2
    phase3_deadline = start + total_budget * 0.55
    print("Phase 3: Mirror v2 on n={}".format(n_best), flush=True)
    a2, sc2 = optimize_mirror_v2(best_a, phase3_deadline, seed=seed + 2, verbose=True)
    if sc2 < best_score:
        best_score = sc2
        best_a = a2.copy()

    # Phase 4: Binary SA on multiple lengths
    phase4_deadline = start + total_budget * 0.75
    for n_b in (n_best, n_best * 2):
        if time.monotonic() > phase4_deadline:
            break
        a0_b = resample(best_a, n_b)
        allow = min(120.0, phase4_deadline - time.monotonic())
        if allow < 20:
            break
        print("Phase 4: Binary SA on n={}".format(n_b), flush=True)
        a3, sc3 = optimize_binary_sa(a0_b, time.monotonic() + allow, seed=seed + 3, verbose=True)
        if sc3 < best_score:
            best_score = sc3
            best_a = a3.copy()

    # Phase 5: Multi-length subgradient
    phase5_deadline = start + total_budget * 0.90
    for n_m in (n_best // 2, n_best, n_best * 2):
        if time.monotonic() > phase5_deadline:
            break
        if n_m < 64 or n_m > 16384:
            continue
        a0_m = resample(best_a, n_m)
        allow = min(80.0, phase5_deadline - time.monotonic())
        if allow < 15:
            break
        print("Phase 5: Subgradient n={}".format(n_m), flush=True)
        a4, sc4 = optimize_subgradient(a0_m, time.monotonic() + allow, seed=seed + 4 + n_m, verbose=True)
        if sc4 < best_score:
            best_score = sc4
            best_a = a4.copy()

    # Phase 6: Final mirror descent polish
    phase6_deadline = deadline - 10
    if time.monotonic() < phase6_deadline:
        print("Phase 6: Final mirror polish on n={}".format(best_a.size), flush=True)
        a5, sc5 = optimize_mirror_v2(best_a, phase6_deadline, seed=seed + 5, verbose=True)
        if sc5 < best_score:
            best_score = sc5
            best_a = a5.copy()

    # Final validation
    best_a = normalize(best_a, floor=0.0)
    out_list = [float(x) for x in best_a]
    try:
        final_eval = float(evaluate_sequence(out_list))
    except Exception:
        final_eval = float("inf")

    if not np.isfinite(final_eval):
        best_a = np.ones(1024) / 1024
        out_list = [float(x) for x in best_a]
        final_eval = 2.0

    print("FINAL: internal={:.12f}, evaluated={:.12f}, n={}".format(
        best_score, final_eval, len(out_list)), flush=True)
    return out_list
