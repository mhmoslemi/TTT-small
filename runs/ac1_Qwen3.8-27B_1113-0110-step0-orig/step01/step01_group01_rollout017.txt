"""
Search for non-negative step heights minimizing

    2 * n * max(convolve(a, a)) / (sum(a))**2

via a hybrid:
  - corrected binary / 0-1 local search with prime quadratic-residue seeds,
  - low-dimensional block-softmax optimization,
  - full multiplicative mirror descent on the simplex,
  - full L-BFGS polishing on log-softmax parameters,
  - exact O(n) continuous swap polishing.

The best sequence found before the deadline is returned.
"""

def next_pow2(m: int) -> int:
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def normalize_positive(a, floor: float = 0.0) -> np.ndarray:
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, float(floor), None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def softmax_seq(z) -> np.ndarray:
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return np.ones(1, dtype=np.float64)
    if not np.all(np.isfinite(z)):
        return np.full(z.size, 1.0 / float(z.size), dtype=np.float64)
    z = z - float(np.max(z))
    z = np.clip(z, -60.0, 0.0)
    e = np.exp(z)
    e = np.maximum(e, 1e-300)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(z.size, 1.0 / float(z.size), dtype=np.float64)
    return e / s


def convolve_seq(a, N=None) -> np.ndarray:
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = next_pow2(2 * n - 1)
    if N is None:
        N = need
    else:
        N = max(int(N), need)
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[: 2 * n - 1]


def weighted_grad(a, w, N=None, n=None) -> np.ndarray:
    a = np.asarray(a, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    if n is None:
        n = a.size
    L = 2 * n - 1
    if w.size != L:
        if w.size > L:
            w = w[:L]
        else:
            w = np.pad(w, (0, L - w.size), mode="constant")
    if N is None:
        N = next_pow2(3 * n - 2)
    else:
        N = max(int(N), next_pow2(3 * n - 2))
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w[::-1], N)
    c = np.fft.irfft(fa * fw, N)
    return 2.0 * c[n - 1 : 2 * n - 1][::-1]


def smooth_value(b, m, tau) -> float:
    b = np.asarray(b, dtype=np.float64).ravel()
    if b.size == 0:
        return float("inf")
    m = float(m)
    if not np.isfinite(m) or m <= 0.0:
        return 0.0
    tau = max(float(tau), 1e-15)
    z = (b - m) / tau
    z = np.clip(z, -50.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return m
    return float(m + tau * math.log(s))


def compute_weights(b, m, tau, floor: float = 0.0) -> np.ndarray:
    b = np.asarray(b, dtype=np.float64).ravel()
    L = b.size
    if L == 0:
        return np.array([], dtype=np.float64)
    m = float(m)
    if not np.isfinite(m) or m <= 0.0:
        return np.full(L, 1.0 / float(L), dtype=np.float64)
    if tau <= 0.0 or not np.isfinite(tau):
        w = np.zeros(L, dtype=np.float64)
        w[np.abs(b - m) <= 1e-12 * max(1.0, abs(m))] = 1.0
        return w / float(w.sum())
    z = (b - m) / float(tau)
    z = np.clip(z, -50.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(L, 1.0 / float(L), dtype=np.float64)
    w = e / s
    if floor > 0.0:
        w = (1.0 - float(floor)) * w + float(floor) / float(L)
    return w


def internal_score(a, N=None) -> float:
    a = normalize_positive(np.asarray(a, dtype=np.float64).ravel(), floor=0.0)
    n = a.size
    if n == 0:
        return float("inf")
    if N is None:

    c = convolve_seq(a, N)
    m = float(np.max(c))
    if not np.isfinite(m):

def resample_seq(a, n: int) -> np.ndarray:
    a = normalize_positive(np.asarray(a, dtype=np.float64).ravel(), floor=0.0)
    n = max(2, int(n))
    if a.size == 1:
        return np.full(n, 1.0 / float(n), dtype=np.float64)

def is_prime(n: int) -> bool:
    n = int(n)
    if n < 2:
        return False
    if n in (2, 3):
        return True
    if n % 2 == 0 or n % 3 == 0:
        return False
    f = 5
    while f * f <= n:
        if n % f == 0 or n % (f + 2) == 0:
            return False
        f += 6
    return True


def primes_3mod4(max_n: int):
    max_n = int(max_n)
    out = []
    for x in range(3, max_n + 1, 2):
        if x % 4 == 3 and is_prime(x):
            out.append(x)
    return out


def qr_base(p: int) -> np.ndarray:
    p = int(p)
    a = np.zeros(p, dtype=np.float64)
    if p < 5 or not is_prime(p):
        return a
    e = (p - 1) // 2
    for x in range(1, p):
        if pow(x, e, p) == 1:
            a[x] = 1.0
    return a


def binary_from_values(a0, m: int) -> np.ndarray:
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    m = int(max(0, min(n, m)))
    b = np.zeros(n, dtype=np.float64)
    if m <= 0:
        return b
    if m >= n:
        b.fill(1.0)
        return b
    is_binary = bool(np.all((a0 <= 0.5) | (a0 >= 0.5)))
    if is_binary:
        vals = (a0 > 0.5).astype(np.float64)
        cur = int(vals.sum())
        if cur == m:
            return vals
        b = vals.copy()
        if cur > m:
            idx = np.nonzero(b > 0.5)[0]
            if idx.size:
                b[idx[-(cur - m):]] = 0.0
        else:
            idx = np.nonzero(b < 0.5)[0]
            if idx.size:
                b[idx[: m - cur]] = 1.0
        return b
    order = np.argsort(a0, kind="stable")
    b[order[-m:]] = 1.0
    return b


def generic_swap(c, p, i, j, n) -> None:
    i = int(i)
    j = int(j)
    if i == j:
        return
    d = float(p[j]) - float(p[i])
    if d == 0.0:
        c[2 * i] += 0.0
        c[i + j] += 0.0
        c[2 * j] += 0.0
        return
    two_d = 2.0 * d
    c[i:i + n] += two_d * p
    c[j:j + n] -= two_d * p
    d2 = d * d
    c[2 * i] += d2
    c[i + j] -= 2.0 * d2
    c[2 * j] += d2
    pi = float(p[i])
    pj = float(p[j])
    p[i] = pj
    p[j] = pi


def sample_pressure(values, rng, pick_high: bool = True, temperature=None) -> int:
    values = np.asarray(values, dtype=np.float64).ravel()
    m = values.size
    if m == 0:
        return -1
    if m == 1:
        return 0
    if not np.all(np.isfinite(values)):
        return int(rng.integers(0, m))
    if temperature is None:
        rng_vals = float(values.max() - values.min())
        temperature = 0.25 * rng_vals if rng_vals > 1e-14 else 1.0
    temperature = float(max(1e-12, temperature))
    if pick_high:
        z = (values - float(values.max())) / temperature
    else:
        z = (float(values.min()) - values) / temperature
    z = np.clip(z, -50.0, 0.0)
    w = np.exp(z)
    s = float(w.sum())
    if not np.isfinite(s) or s <= 0.0:
        return int(rng.integers(0, m))
    w = w / s
    return int(rng.choice(m, p=w))


def binary_penalty(c, cmax) -> float:
    cmax = float(cmax)
    t1 = float(np.sum(c >= cmax - 1.0))
    t2 = float(np.sum(c >= cmax - 2.0))
    t3 = float(np.sum(c >= cmax - 3.0))
    return float(t1 + 0.18 * t2 + 0.05 * t3)


def binary_local(a0, n, m, deadline, seed=0):
    n = int(max(4, n))
    m = int(max(2, min(n - 2, m)))
    a = binary_from_values(a0, m).astype(np.float64).copy()
    N = next_pow2(2 * n - 1)
    c = convolve_seq(a, N)
    c[np.abs(c - np.rint(c)) < 1e-7] = np.rint(c[np.abs(c - np.rint(c)) < 1e-7])

    cmax = float(np.max(c))
    score = float(2.0 * n * cmax / (float(m) * float(m)))
    best_a = a.copy()
    best_cmax = cmax
    best_score = score

    rng = np.random.default_rng(int(seed))
    base_step = float(2.0 * n / (float(m) * float(m)))
    T = max(1e-6, 0.9 * base_step)
    stale = 0
    it = 0
    recompute = 0

    while time.monotonic() < deadline:
        cmax = float(np.max(c))
        cur_score = float(2.0 * n * cmax / (float(m) * float(m)))

        if cur_score < best_score * (1.0 - 1e-12):
            best_score = cur_score
            best_cmax = cmax
            best_a = a.copy()

        if cmax <= 0.0 or not np.isfinite(cmax):
            break

        tau_c = max(1.0, 0.11 * cmax)
        w = compute_weights(c, cmax, tau_c, floor=0.0)
        h = weighted_grad(a, w, N, n)

        one_idx = np.nonzero(a > 0.5)[0]
        zero_idx = np.nonzero(a < 0.5)[0]
        if one_idx.size == 0 or zero_idx.size == 0:
            break

        h_one = h[one_idx]
        h_zero = h[zero_idx]
        hr = float(np.max(h) - np.min(h))
        temp_h = max(1e-12, 0.22 * hr)

        cur_pen = binary_penalty(c, cmax)
        improved_local = False

        for _ in range(9):
            si = sample_pressure(h_one, rng, True, temp_h)
            ti = sample_pressure(h_zero, rng, False, temp_h)
            if si < 0 or ti < 0:
                continue
            i = int(one_idx[si])
            j = int(zero_idx[ti])
            if i == j:
                continue

            generic_swap(c, a, i, j, n)

            new_cmax = float(np.max(c))
            new_score = float(2.0 * n * new_cmax / (float(m) * float(m)))
            new_pen = binary_penalty(c, new_cmax)

            accept = False
            if new_score < cur_score * (1.0 - 1e-12):
                accept = True
            elif new_score < cur_score * (1.0 + 1e-10) and new_pen < cur_pen - 1e-9:
                accept = True
            elif new_score >= cur_score:
                delta = new_score - cur_score
                if delta > 0.0 and delta < 8.0 * base_step and rng.random() < math.exp(-delta / max(T, 1e-12)):
                    accept = True

            if accept:
                cur_score = new_score
                cmax = new_cmax
                stale = 0
                recompute += 1
                T = max(base_step * 0.18, T * 0.9985)
                if recompute % 450 == 0:
                    c = convolve_seq(a, N)
                    c[np.abs(c - np.rint(c)) < 1e-6] = np.rint(
                        c[np.abs(c - np.rint(c)) < 1e-6]
                    )
                improved_local = True
            else:
                generic_swap(c, a, i, j, n)
                stale += 1
                T = max(base_step * 0.05, T * 0.975)

        if not improved_local:
            if stale > 700:
                for _ in range(18):
                    if one_idx.size == 0 or zero_idx.size == 0:
                        break
                    i = int(one_idx[int(rng.integers(0, one_idx.size))])
                    j = int(zero_idx[int(rng.integers(0, zero_idx.size))])
                    if i != j:
                        generic_swap(c, a, i, j, n)
                c = convolve_seq(a, N)
                c[np.abs(c - np.rint(c)) < 1e-6] = np.rint(c[np.abs(c - np.rint(c)) < 1e-6])
                cmax = float(np.max(c))
                cur_score = float(2.0 * n * cmax / (float(m) * float(m)))
                stale = 0
                T = max(base_step * 0.8, T)
            elif stale > 220:
                for _ in range(8):
                    if one_idx.size == 0 or zero_idx.size == 0:
                        break
                    i = int(one_idx[int(rng.integers(0, one_idx.size))])
                    j = int(zero_idx[int(rng.integers(0, zero_idx.size))])
                    if i != j:
                        generic_swap(c, a, i, j, n)
                c = convolve_seq(a, N)
                c[np.abs(c - np.rint(c)) < 1e-6] = np.rint(c[np.abs(c - np.rint(c)) < 1e-6])
                stale = 0

        it += 1

    p = a / float(max(1, int(a.sum())))
    return p, best_score


def build_block_data(n: int, B: int):
    n = int(n)
    B = int(max(1, min(n, B)))
    edges = np.linspace(0, n, B + 1).astype(np.int64)
    bid = np.zeros(n, dtype=np.int64)
    blen = np.zeros(B, dtype=np.float64)
    for b in range(B):
        s = int(edges[b])
        e = int(edges[b + 1])
        if e > s:
            bid[s:e] = b
            blen[b] = float(e - s)
        else:
            bid[s] = b
            blen[b] = 1.0
    blen[blen <= 0.0] = 1.0
    return bid, blen


def block_value(theta, n, B, N, tau_fac, block_id, block_len):
    m = softmax_seq(theta)
    p = m[block_id] / block_len[block_id]
    c = convolve_seq(p, N)
    mc = float(np.max(c))
    tau = float(tau_fac) * max(mc, 1e-15)
    return float(2.0 * n * smooth_value(c, mc, tau))


def block_grad(theta, n, B, N, tau_fac, block_id, block_len):
    m = softmax_seq(theta)
    p = m[block_id] / block_len[block_id]
    c = convolve_seq(p, N)
    mc = float(np.max(c))
    tau = float(tau_fac) * max(mc, 1e-15)
    w = compute_weights(c, mc, tau, floor=0.0)
    hp = weighted_grad(p, w, N, n)
    sens = np.bincount(block_id, weights=hp, minlength=B).astype(np.float64)
    sens /= block_len
    dot = float(np.dot(m, sens))
    g = m * (sens - dot)
    return 2.0 * float(n) * g


def optimize_blocks(a0, deadline, seed=0):
    a = normalize_positive(a0, floor=1e-12)
    n = a.size
    if n < 16:
        return a, internal_score(a)
    N = next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = internal_score(a, N)
    rng = np.random.default_rng(int(seed))

    B_list = (8, 16, 32, 64, 128)
    if n <= 128:
        B_list = (4, 8, 16, 32)
    elif n <= 256:
        B_list = (8, 16, 32, 64)

    for B in B_list:
        if time.monotonic() > deadline:
            break
        if B > n:
            B = n
        block_id, block_len = build_block_data(n, B)

        avg = np.bincount(block_id, weights=a, minlength=B).astype(np.float64)
        avg = np.maximum(avg / block_len, 1e-12)
        theta0s = [
            np.zeros(B, dtype=np.float64),
            np.log(avg) - float(np.mean(np.log(avg))),
            rng.normal(0.0, 0.35, size=B),
            rng.normal(0.0, 0.80, size=B),
        ]

        for th0 in theta0s:
            if time.monotonic() > deadline:
                break
            theta = np.asarray(th0, dtype=np.float64).ravel().copy()
            p = softmax_seq(theta)
            sc = internal_score(p, N)
            if sc < best_score * (1.0 - 1e-12):
                best_score = sc
                best_a = p.copy()

            if _so is None:
                continue

            for fac in (0.08, 0.04):
                if time.monotonic() > deadline:
                    break
                try:
                    res = _so.minimize(
                        block_value,
                        theta,
                        jac=block_grad,
                        args=(n, B, N, fac, block_id, block_len),
                        method="L-BFGS-B",
                        options={"maxiter": 35, "ftol": 1e-13, "gtol": 1e-7, "maxls": 20},
                    )
                    if res is not None and res.x is not None and np.all(np.isfinite(res.x)):
                        theta = np.asarray(res.x, dtype=np.float64).ravel()
                        p = softmax_seq(theta)
                        sc = internal_score(p, N)
                        if sc < best_score * (1.0 - 1e-12):
                            best_score = sc
                            best_a = p.copy()
                            print("blocks n={} B={} score={:.9f}".format(n, B, best_score), flush=True)
                except Exception:
                    pass
    return best_a, best_score


def optimize_mirror_v2(a0, deadline, seed=0, verbose=False):
    a = normalize_positive(a0, floor=1e-15)
    n = a.size
    if n < 8:
        return a, internal_score(a)
    N = next_pow2(2 * n - 1)
    c = convolve_seq(a, N)
    m = float(np.max(c))
    best_a = a.copy()
    best_score = float(2.0 * n * m)
    loga = np.log(np.maximum(a, 1e-300))
    loga -= float(np.max(loga))
    rng = np.random.default_rng(int(seed))

    eta = 0.50
    beta = 0.07
    floor_w = 0.0
    stale = 0
    it = 0
    last_best = best_score

    while time.monotonic() < deadline:
        c = convolve_seq(a, N)
        m = float(np.max(c))
        score = float(2.0 * n * m)
        if np.isfinite(score) and score < best_score * (1.0 - 1e-12):
            best_score = score
            best_a = a.copy()

        if m <= 0.0 or not np.isfinite(m):
            break

        tau = beta * m
        w = compute_weights(c, m, tau, floor=floor_w)
        h = weighted_grad(a, w, N, n)
        g = h - float(np.mean(h))
        sc = float(np.max(np.abs(g)))
        if not np.isfinite(sc) or sc <= 1e-16:
            break
        g = g / sc

        smooth_cur = smooth_value(c, m, tau)
        accepted = False
        a_best_try = None
        score_best_try = score
        smooth_best_try = smooth_cur

        for st in (eta, 0.55 * eta, 0.30 * eta, 0.16 * eta, 0.08 * eta):
            if st <= 1e-10:
                continue
            loga_try = loga - st * g
            a_try = softmax_seq(loga_try)
            a_try = np.maximum(a_try, 1e-300)
            a_try = a_try / float(a_try.sum())
            c_try = convolve_seq(a_try, N)
            m_try = float(np.max(c_try))
            score_try = float(2.0 * n * m_try)
            smooth_try = smooth_value(c_try, m_try, tau)

            ok = False
            if np.isfinite(score_try) and score_try < score * (1.0 - 1e-11):
                ok = True
            elif np.isfinite(smooth_try) and smooth_try < smooth_cur - 1e-11 and score_try < score * 1.015:
                ok = True
            if ok and (score_try < score_best_try - 1e-12 or
                       (abs(score_try - score_best_try) <= 1e-12 and smooth_try < smooth_best_try)):
                a_best_try = a_try
                score_best_try = score_try if np.isfinite(score_try) else score
                smooth_best_try = smooth_try if np.isfinite(smooth_try) else smooth_cur
                accepted = True
                break

        if accepted and a_best_try is not None:
            loga = np.log(np.maximum(a_best_try, 1e-300))
            loga -= float(np.max(loga))
            a = a_best_try.copy()
            stale = 0
            eta = min(1.35, eta * 1.02)
            beta = max(0.010, beta * 0.995)
            floor_w = max(0.0, floor_w * 0.994)
        else:
            stale += 1
            eta = max(0.02, eta * 0.68)
            beta = max(0.005, beta * 0.90)
            if stale % 120 == 0:
                beta = min(0.14, beta * 1.6)
                floor_w = min(0.04, floor_w + 0.004)
                stale = 0
            if stale > 650 or eta < 0.021:
                noise = rng.normal(0.0, 0.06, size=n)
                loga = loga + noise
                loga -= float(np.max(loga))
                a = softmax_seq(loga)
                a = np.maximum(a, 1e-300)
                a = a / float(a.sum())
                stale = 0
                eta = 0.35
                beta = 0.09
                floor_w = 0.002

        it += 1
        if verbose and (it % 150 == 0):
            print("mirror n={} iter={} best={:.9f}".format(n, it, best_score), flush=True)

        if it % 700 == 0 and best_score > last_best - 1e-8:
            tmp = a * np.exp(rng.normal(0.0, 0.045, size=n))
            tmp = normalize_positive(tmp, floor=1e-15)
            sc_tmp = internal_score(tmp, N)
            if np.isfinite(sc_tmp) and sc_tmp < best_score:
                a = tmp.copy()
                best_score = sc_tmp
                best_a = a.copy()
            last_best = best_score

    return best_a, best_score


def lbf_value(y, n, tau_fac, N):
    p = softmax_seq(y)
    c = convolve_seq(p, N)
    m = float(np.max(c))
    tau = float(tau_fac) * max(m, 1e-15)
    return float(2.0 * n * smooth_value(c, m, tau))


def lbf_grad(y, n, tau_fac, N):
    p = softmax_seq(y)
    c = convolve_seq(p, N)
    m = float(np.max(c))
    tau = float(tau_fac) * max(m, 1e-15)
    w = compute_weights(c, m, tau, floor=0.0)
    hp = weighted_grad(p, w, N, n)
    dot = float(np.dot(p, hp))
    g = p * (hp - dot)
    return 2.0 * float(n) * g


def optimize_full_lbf(a0, deadline, seed=0):
    a = normalize_positive(a0, floor=1e-15)
    n = a.size
    if n < 32 or _so is None:
        return a, internal_score(a)
    N = next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = internal_score(a, N)
    y = np.log(np.maximum(a, 1e-300))
    y -= float(np.mean(y))

    for fac in (0.08, 0.045, 0.025, 0.012, 0.006):
        if time.monotonic() > deadline:
            break
        try:
            res = _so.minimize(
                lbf_value,
                y,
                jac=lbf_grad,
                args=(n, fac, N),
                method="L-BFGS-B",
                options={"maxiter": 34, "ftol": 1e-13, "gtol": 1e-8, "maxls": 20},
            )
            if res is not None and res.x is not None and np.all(np.isfinite(res.x)):
                y = np.asarray(res.x, dtype=np.float64).ravel()
                p = softmax_seq(y)
                sc = internal_score(p, N)
                if np.isfinite(sc) and sc < best_score * (1.0 - 1e-12):
                    best_score = sc
                    best_a = p.copy()
                    print("lbf n={} fac={} score={:.9f}".format(n, fac, best_score), flush=True)
        except Exception:
            pass

    return best_a, best_score


def swap_polish(a0, deadline, seed=0):
    p = normalize_positive(a0, floor=1e-15)
    n = p.size
    if n < 8:
        return p, internal_score(p)
    N = next_pow2(2 * n - 1)
    c = convolve_seq(p, N)
    m = float(np.max(c))
    best_p = p.copy()
    best_score = float(2.0 * n * m)
    rng = np.random.default_rng(int(seed))
    stale = 0
    it = 0

    while time.monotonic() < deadline:
        m = float(np.max(c))
        score = float(2.0 * n * m)
        if np.isfinite(score) and score < best_score * (1.0 - 1e-12):
            best_score = score
            best_p = p.copy()

        if m <= 0.0 or not np.isfinite(m):
            break

        tau = max(1e-12, 0.055 * m)
        w = compute_weights(c, m, tau, floor=0.0)
        h = weighted_grad(p, w, N, n)

        hr = float(h.max() - h.min())
        if hr <= 1e-14:
            break
        hn = (h - float(h.min())) / hr
        pr = float(p.max() - p.min())
        if pr <= 1e-14:
            pn = np.zeros_like(p)
        else:
            pn = (p - float(p.min())) / pr
        s = 0.72 * hn + 0.28 * pn
        order = np.argsort(s)
        k = max(4, n // 16)
        lows = order[:k]
        highs = order[-k:]

        improved = False
        for _ in range(8):
            if lows.size == 0 or highs.size == 0:
                break
            i = int(highs[int(rng.integers(0, highs.size))])
            j = int(lows[int(rng.integers(0, lows.size))])
            if i == j:
                continue
            generic_swap(c, p, i, j, n)
            nm = float(np.max(c))
            ns = float(2.0 * n * nm)
            if np.isfinite(ns) and ns < score * (1.0 - 1e-11):
                score = ns
                stale = 0
                improved = True
                if score < best_score * (1.0 - 1e-12):
                    best_score = score
                    best_p = p.copy()
                break
            generic_swap(c, p, i, j, n)

        if not improved:
            stale += 1
            if stale > 350:
                for _ in range(10):
                    i = int(rng.integers(0, n))
                    j = int(rng.integers(0, n))
                    if i != j:
                        generic_swap(c, p, i, j, n)
                c = convolve_seq(p, N)
                stale = 0

        it += 1

    return best_p, best_score


def propose_candidate(seed=42, budget_s=2370, **kwargs):
    start = time.monotonic()
    try:
        try_budget = float(budget_s)
    except Exception:
        try_budget = 2370.0
    try_budget = max(180.0, min(2400.0, try_budget))
    total = try_budget - 25.0
    if total < 120.0:
        total = try_budget - 15.0
    deadline = start + float(total)

    rng_main = np.random.default_rng(int(seed))

    best_a = make_uniform(1024)
    best_score = internal_score(best_a)
    print("initial fallback score = {:.9f}".format(best_score), flush=True)

    given = get_given_sequence()
    if given is not None:
        sc = internal_score(given)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = given.copy()
            print("given sequence score = {:.9f}, n={}".format(best_score, best_a.size), flush=True)

    # Phase 1: prime QR binary searches. These are cheap for small p and sometimes
    # produce extremely flat sum-convolution patterns.
    qr_start = time.monotonic()
    qr_end = min(deadline, qr_start + max(20.0, float(total) * 0.22))
    primes = primes_3mod4(2500)
    primes = [p for p in primes if p >= 113]

    for p in primes:
        if time.monotonic() > qr_end - 1.0:
            break
        base = qr_base(p)
        if int(base.sum()) < 2:
            continue
        if p <= 251:
            per_start = 1.8
            offsets = (0, 1, 2, p // 3, p // 5)
        elif p <= 700:
            per_start = 2.6
            offsets = (0, 1, 2, p // 3)
        else:
            per_start = 4.0
            offsets = (0, 1, p // 3)

        starts = []
        for off in offsets:
            starts.append(np.roll(base, off))
        starts.append(1.0 - np.roll(base, 0))
        starts.append(1.0 - np.roll(base, max(1, p // 3)))
        starts.append((rng_main.random(p) < 0.5).astype(np.float64))
        starts.append((rng_main.random(p) < 0.5).astype(np.float64))

        for a0 in starts:
            if time.monotonic() > qr_end - 0.5:
                break
            cur = time.monotonic()
            allow = min(float(per_start), qr_end - cur - 0.5)
            if allow < 0.35:
                break
            m0 = int(round(0.5 * p))
            if a0.size == p and bool(np.all((a0 <= 0.5) | (a0 >= 0.5))):
                cnt = int(np.sum(a0 > 0.5))
                if 2 <= cnt <= p - 2:
                    m0 = cnt
            else:
                m0 = int(round(0.5 * p))
            try:
                b1, sc1 = binary_local(a0, p, m0, min(qr_end, cur + allow), seed=int(seed + p))
            except Exception:
                continue
            if b1 is not None and np.isfinite(sc1) and sc1 < best_score * (1.0 - 1e-12):
                best_score = sc1
                best_a = normalize_positive(b1, floor=0.0)
                print("binary QR n={} m={} score={:.9f}".format(p, m0, best_score), flush=True)

    # Phase 2: block-softmax low-dimensional search at several lengths.
    block_start = time.monotonic()
    block_end = min(deadline, block_start + max(20.0, float(total) * 0.12))
    for n in (512, 1024, 2048):
        if time.monotonic() > block_end - 5.0:
            break
        a0 = resample_seq(best_a, n)
        try:
            b1, sc1 = optimize_blocks(a0, min(block_end, time.monotonic() + 45.0), seed=int(seed + 9000 + n))
        except Exception:
            continue
        if b1 is not None and np.isfinite(sc1) and sc1 < best_score * (1.0 - 1e-12):
            best_score = sc1
            best_a = normalize_positive(b1, floor=0.0)
            print("block selected n={} score={:.9f}".format(n, best_score), flush=True)

    # Phase 3: full mirror descent on good lengths.
    first = 2048
    second = 1024
    third = int(best_a.size) if 128 <= int(best_a.size) <= 4096 else 2048
    fourth = 4096
    mirror_lengths = []
    for nn in (first, second, third, fourth):
        if nn not in mirror_lengths and nn >= 128:
            mirror_lengths.append(int(nn))

    mirror_start = time.monotonic()
    mirror_budget = max(60.0, float(total) * 0.38)
    mirror_end = min(deadline, mirror_start + mirror_budget)
    for n in mirror_lengths:
        if time.monotonic() > mirror_end - 10.0:
            break
        remaining = min(mirror_end - time.monotonic(), 300.0)
        if remaining < 25.0:
            break
        a0 = resample_seq(best_a, n)
        try:
            b1, sc1 = optimize_mirror_v2(a0, min(deadline, time.monotonic() + remaining), seed=int(seed + 12345 + n), verbose=(n == 2048))
        except Exception:
            continue
        if b1 is not None and np.isfinite(sc1) and sc1 < best_score * (1.0 - 1e-12):
            best_score = sc1
            best_a = normalize_positive(b1, floor=0.0)
            print("mirror selected n={} score={:.9f}".format(n, best_score), flush=True)

    # Phase 4: full-dimensional L-BFGS polishing.
    lbf_start = time.monotonic()
    lbf_budget = max(30.0, float(total) * 0.14)
    lbf_end = min(deadline, lbf_start + lbf_budget)
    for n in (2048, 1024):
        if time.monotonic() > lbf_end - 5.0:
            break
        remaining = min(lbf_end - time.monotonic(), 170.0)
        if remaining < 18.0:
            break
        a0 = resample_seq(best_a, n)
        try:
            b1, sc1 = optimize_full_lbf(a0, min(deadline, time.monotonic() + remaining), seed=int(seed + 777 + n))
        except Exception:
            continue
        if b1 is not None and np.isfinite(sc1) and sc1 < best_score * (1.0 - 1e-12):
            best_score = sc1
            best_a = normalize_positive(b1, floor=0.0)
            print("lbf selected n={} score={:.9f}".format(n, best_score), flush=True)

    # Phase 5: binary targeting from the best continuous shape.
    bin_start = time.monotonic()
    bin_budget = max(30.0, float(total) * 0.22)
    bin_end = min(deadline, bin_start + bin_budget)
    for n in (2048, 1024):
        if time.monotonic() > bin_end - 8.0:
            break
        remaining = min(bin_end - time.monotonic(), 160.0)
        if remaining < 25.0:
            break
        a0 = resample_seq(best_a, n)
        m0 = n // 2
        try:
            b1, sc1 = binary_local(a0, n, m0, min(deadline, time.monotonic() + remaining), seed=int(seed + 50000 + n))
        except Exception:
            continue
        if b1 is not None and np.isfinite(sc1) and sc1 < best_score * (1.0 - 1e-12):
            best_score = sc1
            best_a = normalize_positive(b1, floor=0.0)
            print("binary polish selected n={} score={:.9f}".format(n, best_score), flush=True)

    # Phase 6: exact continuous swap polish until near deadline.
    if time.monotonic() < deadline - 12.0:
        try:
            b1, sc1 = swap_polish(best_a, deadline - 5.0, seed=int(seed + 9999999))
        except Exception:
            b1, sc1 = best_a, best_score
        if b1 is not None and np.isfinite(sc1) and sc1 < best_score * (1.0 - 1e-12):
            best_score = sc1
            best_a = normalize_positive(b1, floor=0.0)
            print("swap polish selected score={:.9f}".format(best_score), flush=True)

    # Final sanity: ensure a valid list. Resample absurdly large sequences if needed.
    best_a = normalize_positive(best_a, floor=0.0)
    n_out = int(best_a.size)
    if n_out == 0 or n_out > 20000:
        best_a = resample_seq(best_a, 2048)
        n_out = 2048

    out_list = [float(max(0.0, min(1000.0, x))) for x in best_a]
    out_sum = float(sum(out_list))
    if not np.isfinite(out_sum) or out_sum < 0.01:
        out_list = [1.0 / 1024.0] * 1024

    try:
        final_eval = float(evaluate_sequence(out_list))
    except Exception:
        final_eval = float(best_score)

    if not np.isfinite(final_eval):
        out_list = [float(x) for x in make_uniform(1024)]
        final_eval = 2.0

    print("final internal score = {:.12f}, length = {}".format(float(best_score), n_out), flush=True)
    print("final evaluated score = {:.12f}".format(float(final_eval)), flush=True)
    return out_list
