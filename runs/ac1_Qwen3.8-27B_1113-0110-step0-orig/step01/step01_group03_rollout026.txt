```python
import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
import time

def fft_conv(a, b):
    """Full linear convolution using FFT."""
    n1, n2 = len(a), len(b)
    size = 1
    while size < n1 + n2 - 1:
        size *= 2
    fa = rfft(a, size)
    fb = rfft(b, size)
    c = irfft(fa * fb, size)
    return c[:n1 + n2 - 1]

def fft_self_conv(a):
    """Full self-convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = rfft(a, size)
    c = irfft(fa * fa, size)
    return c[:2 * n - 1]

def score_from_conv(a, c):
    """Compute score given a and its self-convolution c."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (S * S)

def smooth_score_and_grad(a, tau, normalize=True):
    """
    Compute smooth approximation of score and its gradient.
    If normalize=True, first normalize a to sum = n.
    Returns (score, gradient, c).
    """
    n = len(a)
    a = np.maximum(a, 0)
    S = a.sum()
    if S < 0.01:
        return np.inf, np.zeros(n), np.zeros(2*n-1)
    if normalize:
        a = a * n / S
    
    c = fft_self_conv(a)
    
    # Softmax over c (all 2n-1 values)
    max_c = c.max()
    exp_c = np.exp(tau * (c - max_c))
    Z = exp_c.sum()
    p = exp_c / Z
    
    M_smooth = (max_c + np.log(Z)) / tau
    score = 2.0 * n * M_smooth / (n * n)  # = 2*M_smooth/n since S=n
    
    # Gradient: ∂M/∂a_j = (2/Z) * sum_m a[m] * p[j+m]
    # = (2/Z) * cross_correlation(a, p)[j]
    # Using FFT: R = ifft(fft(p, N) * conj(fft(a, N)))
    
    N = 1
    while N < n + (2*n-1) - 1:
        N *= 2
    
    fa = rfft(a, N)
    fp = rfft(p, N)
    R = irfft(fp * np.conj(fa), N)[:n]
    
    grad_M = 2.0 * R / Z
    grad_score = (2.0 * n / (n*n)) * grad_M  # = 2/n * grad_M
    
    return score, grad_score, c
```
