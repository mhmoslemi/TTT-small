def fft_self_conv(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    if n <= 1:
        return a * a
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:2 * n - 1]

def soft_peak_gradient(a, c, k_stars, weights):
    """
    Compute weighted gradient of max-convolution w.r.t. a.
    k_stars: indices of peaks to consider
    weights: weights for each peak (sum to 1)
    """
    n = len(a)
    S = a.sum()
    M = c.max()
    
    # For each peak k, the gradient contribution is 2*a[k-j] for j in [0, n-1] where k-j is valid
    # i.e., j in [max(0, k-n+1), min(n-1, k)]
    grad = np.zeros(n)
    for idx, k in enumerate(k_stars):
        w = weights[idx]
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)

            s_start = k - j_max
            s_end = k - j_min
            grad[j_min:j_max+1] += 2 * w * a[s_start:s_end+1][::-1]
