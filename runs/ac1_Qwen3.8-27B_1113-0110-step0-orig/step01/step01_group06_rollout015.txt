```python
"""
Search for nonnegative step-function heights minimizing
    2*n*max(convolve(a,a)) / (sum(a))**2.

The program uses two complementary parameterizations:
1. An anti-palindromic ansatz a_i + a_{n-1-i} = constant, optimizing m variables.
2. A full softmax parameterization for refinement and symmetry breaking.

The optimized surrogate is a smooth log-sum-exp of the convolution coefficients
plus a variance penalty, with negative/positive weights on low/high coefficients.
Coordinates are optimized with SciPy L-BFGS-B when available, with a simple
fallback mirror/gradient procedure otherwise.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize as _scipy_minimize
except Exception:
    _scipy_minimize = None

# ---------------------------------------------------------------------------
# Globals used to record the best iterate during L-BFGS objective evaluations.
# ---------------------------------------------------------------------------
REC_TRACK = False
REC_KIND = 0
REC_N = 1
REC_M = 1
REC_REF = 0
REC_SCORE = float("inf")
REC_STATE = None
REC_A = None

X_EPS = 1e-8
_BOUND_CACHE = {}


def _pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _fft_self(a, N=None):
    n = int(a.size)
    L = 2 * n - 1
    if N is None:
        N = _pow2(L)
    if N < L:
        N = _pow2(L)
    fa = np.fft.rfft(a, n=N)
    q = np.fft.irfft(fa * fa, n=N).real[:L]
    if q.size and float(q.min()) < 0.0:
        np.maximum(q, 0.0, out=q)
    return q


def _grad_from_dw(dw, a, Ng=None):
    n = int(a.size)
    L = 2 * n - 1
    if Ng is None:
        Ng = _pow2(L + n - 1)
    if Ng < L + n - 1:
        Ng = _pow2(L + n - 1)
    ar = np.ascontiguousarray(a[::-1], dtype=np.float64)
    fdw = np.fft.rfft(dw, n=Ng)
    fa = np.fft.rfft(ar, n=Ng)
    h = np.fft.irfft(fdw * fa, n=Ng).real
    return 2.0 * h[n - 1:2 * n - 1]


def _loss_and_grad_common(a, beta, lam_l2, Nq, Ng):
    q = _fft_self(a, Nq)
    if q.size == 0:
        return 1.0e300, None, q, 0.0
    maxq = float(q.max())
    if not np.isfinite(maxq) or maxq <= 0.0:
        return 1.0e300, None, q, 0.0
    if not np.isfinite(beta) or beta <= 1e-300:
        beta = 1e-300

    z = (q - maxq) / beta
    with np.errstate(over="ignore", under="ignore"):
        ez = np.exp(z)
    s = float(ez.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.zeros_like(q)
        p[int(np.argmax(q))] = 1.0
    else:
        p = ez / s

    dw = p / beta
    loss = maxq / beta + float(np.log(s)) if s > 0 else maxq / beta

    if lam_l2:
        avg = 1.0 / float(q.size)
        delta = q - avg
        loss += float(lam_l2) * float(np.dot(delta, delta))
        dw = dw + 2.0 * float(lam_l2) * delta

    if not np.isfinite(loss):
        return 1.0e300, None, q, maxq

    ga = _grad_from_dw(dw, a, Ng)
    if ga is None or not np.all(np.isfinite(ga)):
        ga = np.zeros_like(a, dtype=np.float64)
    return float(loss), ga, q, maxq


def _anti_from_x(x, m=None):
    x = np.asarray(x, dtype=np.float64).ravel()
    if m is None:
        m = int(x.size)
    c = 1.0 / float(m)
    a = np.empty(2 * m, dtype=np.float64)
    a[:m] = c * x
    a[m:] = c * (1.0 - x[::-1])
    return a


def _normalize_array(a, fallback_n=8192):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        out = np.ones(int(fallback_n), dtype=np.float64)
        out /= out.size
        return out
    out = a / s
    return out


def _score_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return float("inf")
    a = a / s
    q = _fft_self(a)
    if q.size == 0:
        return float("inf")
    return float(2.0 * a.size * q.max())


def _resample_to_n(a, n):
    a = _normalize_array(a, n)
    n = int(n)
    if a.size == n:
        return a.astype(np.float64, copy=True)
    if a.size <= 1:
        return np.full(n, 1.0 / max(1, n), dtype=np.float64)
    old = np.linspace(0.0, a.size - 1, a.size)
    new = np.linspace(0.0, a.size - 1, n)
    out = np.interp(new, old, a)
    s = float(out.sum())
    if s <= 1e-15:
        out = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        out = out / s
    return out


def _project_to_anti(a_old, n):
    n = int(n)
    m = n // 2
    a = _normalize_array(a_old, n)
    if a.size != n:
        a = _resample_to_n(a, n)
    left = a[:m]
    right = a[n - 1:n - m - 1:-1]
    sums = left + right
    x = np.where(sums > 1e-15, left / np.maximum(sums, 1e-15), 0.5)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    return np.clip(x, X_EPS, 1.0 - X_EPS)


def _rec_reset(kind, n, m, score, state, a):
    global REC_TRACK, REC_KIND, REC_N, REC_M, REC_REF, REC_SCORE, REC_STATE, REC_A
    REC_TRACK = True
    REC_KIND = int(kind)
    REC_N = int(n)
    REC_M = int(m)
    REC_REF = 0
    REC_SCORE = float(score)
    REC_STATE = np.asarray(state, dtype=np.float64).copy()
    REC_A = np.asarray(a, dtype=np.float64).copy()


def _anti_loss_jac(x, m, beta, lam_l2, Nq, Ng):
    global REC_TRACK, REC_SCORE, REC_STATE, REC_A

    m = int(m)
    x = np.asarray(x, dtype=np.float64).ravel()
    if x.size != m:
        return 1.0e300, np.zeros(m, dtype=np.float64)
    x = np.clip(x, X_EPS, 1.0 - X_EPS)
    if not np.all(np.isfinite(x)):
        return 1.0e300, np.zeros(m, dtype=np.float64)

    a = _anti_from_x(x, m)
    loss, ga, q, maxq = _loss_and_grad_common(a, beta, lam_l2, Nq, Ng)
    n = 2 * m
    if not np.isfinite(loss):
        return 1.0e300, np.zeros(m, dtype=np.float64)

    if REC_TRACK:
        score = float(2.0 * n * maxq)
        if np.isfinite(score) and score < REC_SCORE:
            REC_SCORE = score
            REC_STATE = x.copy()
            REC_A = a.copy()

    c = 1.0 / float(m)
    gx = c * (ga[:m] - ga[m:][::-1])
    if not np.all(np.isfinite(gx)):
        gx = np.zeros(m, dtype=np.float64)
    return float(loss), gx


def _full_loss_jac_ref(z, n, ref, beta, lam_l2, Nq, Ng):
    global REC_TRACK, REC_SCORE, REC_STATE, REC_A

    n = int(n)
    ref = int(ref)
    if ref < 0:
        ref = 0
    if ref >= n:
        ref = n - 1
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n - 1:
        return 1.0e300, np.zeros(n - 1, dtype=np.float64)
    if not np.all(np.isfinite(z)):
        return 1.0e300, np.zeros(n - 1, dtype=np.float64)

    if ref == n - 1:
        u = np.empty(n, dtype=np.float64)
        u[:-1] = z
        u[-1] = 0.0
    else:
        u = np.concatenate((z[:ref], [0.0], z[ref:]))

    mx = float(np.max(u))
    with np.errstate(over="ignore", under="ignore"):
        b = np.exp(u - mx)
    amt = float(b.sum())
    if not np.isfinite(amt) or amt <= 1e-300:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = b / amt

    loss, ga, q, maxq = _loss_and_grad_common(a, beta, lam_l2, Nq, Ng)
    if not np.isfinite(loss):
        return 1.0e300, np.zeros(n - 1, dtype=np.float64)

    if REC_TRACK:
        score = float(2.0 * n * maxq)
        if np.isfinite(score) and score < REC_SCORE:
            REC_SCORE = score
            REC_STATE = z.copy()
            REC_A = a.copy()

    dot = float(np.dot(a, ga))
    gfull = a * (ga - dot)
    if ref == n - 1:
        gz = gfull[:ref]
    else:
        gz = np.concatenate((gfull[:ref], gfull[ref + 1:]))
    if not np.all(np.isfinite(gz)):
        gz = np.zeros(n - 1, dtype=np.float64)
    return float(loss), gz


def _a_to_z_ref(a, ref):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    ref = int(ref)
    if ref < 0:
        ref = 0
    if ref >= n:
        ref = n - 1
    a = np.maximum(a, 1e-12)
    base = float(a[ref])
    if base <= 1e-15:
        base = 1e-15
    z = np.log(a / base)
    z = np.clip(z, -50.0, 50.0)
    if ref == n - 1:
        return z[:ref].copy()
    return np.concatenate((z[:ref], z[ref + 1:])).astype(np.float64, copy=False)


def _get_anti_bounds(m):
    m = int(m)
    if m not in _BOUND_CACHE:
        _BOUND_CACHE[m] = [(X_EPS, 1.0 - X_EPS)] * m
    return _BOUND_CACHE[m]


def _maxfun_for_n(n, full):

    if full:
        if n <= 4096:
            return 1000
        if n <= 8192:
            return 700
        if n <= 16384:

        return 240
    else:

            return 950

            return 600
        return 380


def _simple_anti_optimize(x0, n, time_budget, rng):
    m = int(n) // 2
    n = int(n)
    if m <= 0:
        return None, float("inf")
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(time_budget))

    x = np.clip(np.asarray(x0, dtype=np.float64).ravel(), X_EPS, 1.0 - X_EPS)
    a = _anti_from_x(x, m)
    Nq = _pow2(2 * n - 1)
    Ng = _pow2(3 * n - 2)
    q = _fft_self(a, Nq)
    score = float(2.0 * n * q.max())

    best_x = x.copy()
    best_a = a.copy()
    best_score = score

    if time.monotonic() >= deadline - 5.0:
        return best_a, best_score

    s = best_score / (2.0 * n)
    beta = max(0.5 * s, 1.0 / (500.0 * n))
    lam = 0.0
    it = 0
    no_improve = 0
    max_no = 80

    while time.monotonic() < deadline - 2.0:
        if it > 0 and it % 80 == 0:
            beta = max(beta * 0.7, 1.0 / (1200.0 * n))
            if it % 160 == 0:
                lam = min(lam + 400.0, 2000.0)

        a = _anti_from_x(x, m)
        loss, ga, q, maxq = _loss_and_grad_common(a, beta, lam, Nq, Ng)
        if not np.isfinite(loss):
            x = np.clip(best_x + 0.05 * rng.normal(size=m), X_EPS, 1.0 - X_EPS)
            continue

        score = float(2.0 * n * maxq)
        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            best_x = x.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no:
            x = 0.8 * best_x + 0.2 * rng.normal(0.5, 0.25, size=m)
            x = np.clip(x, X_EPS, 1.0 - X_EPS)
            no_improve = 0
            max_no = int(max_no * 1.3)
            s = best_score / (2.0 * n)
            beta = max(0.5 * s, 1.0 / (500.0 * n))
            lam = 0.0 if (it % 3 == 0) else min(1000.0, lam + 500.0)
            continue

        c = 1.0 / float(m)
        gx = c * (ga[:m] - ga[m:][::-1])
        gmax = float(np.max(np.abs(gx))) if gx.size else 0.0
        if not np.isfinite(gmax) or gmax < 1e-16:
            break

        step = 0.05 / gmax
        x_new = np.clip(x - step * gx, X_EPS, 1.0 - X_EPS)
        if not np.any(x_new != x):
            x_new = np.clip(x + 0.03 * rng.normal(size=m), X_EPS, 1.0 - X_EPS)

        a_new = _anti_from_x(x_new, m)
        q_new = _fft_self(a_new, Nq)
        sc_new = float(2.0 * n * q_new.max())

        if np.isfinite(sc_new) and (sc_new < best_score - 1e-14 or sc_new <= score + 1e-15):
            x = x_new
            a = a_new
            score = sc_new
        else:
            x = np.clip(x + 0.02 * rng.normal(size=m), X_EPS, 1.0 - X_EPS)

        it += 1

    a_best = _anti_from_x(best_x, m)
    q_best = _fft_self(a_best, Nq)
    sc_best = float(2.0 * n * q_best.max())
    if np.isfinite(sc_best) and sc_best < best_score:
        best_score = sc_best
        best_a = a_best.copy()
    return best_a, best_score


def _optimize_anti_start(x0, n, time_budget, rng):
    n = int(n)

    t0 = time.monotonic()

    if not np.isfinite(score):

        a = _anti_from_x(x, m)
        q = _fft_self(a, Nq)
        score = float(2.0 * n * q.max())

    best_x = x.copy()
    best_a = a.copy()
    best_score = score

    if time.monotonic() >= deadline - 8.0:
        return best_a, best_score

    if _scipy_minimize is None:
        if time.monotonic() < deadline - 5.0:

    bounds = _get_anti_bounds(m)

        beta = max(frac * s, 1.0 / (500.0 * n))

        args = (best_x, m, beta, lam, Nq, Ng)

            "ftol": 1e-12,

            pass

def _optimize_full_start(a0, n, time_budget, rng):
    n = int(n)
    if n <= 2:
        a0 = _normalize_array(a0, n)
        return a0.copy(), _score_a(a0)

    a = _normalize_array(a0, n)
    if a.size != n:
        a = _resample_to_n(a, n)

    Nq = _pow2(2 * n - 1)
    Ng = _pow2(3 * n - 2)
    score = _score_a(a)

    best_a = a.copy()
    best_score = score
    ref = int(np.argmax(best_a))
    best_z = _a_to_z_ref(best_a, ref)

    deadline = time.monotonic() + max(0.0, float(time_budget))
    if not np.isfinite(best_score) or time.monotonic() >= deadline - 8.0:
        return best_a, best_score

    schedule = [
        (0.35, 0.0),
        (0.12, 600.0),
        (0.04, 1200.0),
        (0.012, 2000.0),
        (0.003, 0.0),
        (0.0008, 0.0),
    ]

    stage_index = 0
    while time.monotonic() < deadline - 4.0 and stage_index < len(schedule):
        frac, lam = schedule[stage_index]
        stage_index += 1

        remaining = max(0.0, deadline - time.monotonic())
        stage_time = min(remaining - 2.0, max(12.0, 0.20 * float(time_budget)))
        if stage_time < 10.0:
            break

        s = best_score / (2.0 * n)
        beta = max(frac * s, 1.0 / (500.0 * n))
        if not np.isfinite(beta) or beta <= 0.0:
            beta = 1.0 / (500.0 * n)

        _rec_reset(2, n, n, best_score, best_z, best_a)
        REC_REF = ref

        t_meas = time.monotonic()
        try:
            _full_loss_jac_ref(best_z, n, ref, beta, lam, Nq, Ng)
            dt_eval = max(1e-5, time.monotonic() - t_meas)
        except Exception:
            dt_eval = 0.01

        base_maxfun = _maxfun_for_n(n, True)
        maxfun = int(min(base_maxfun, max(60, int(0.85 * stage_time / dt_eval))))
        if maxfun < 60:
            break

        options = {
            "maxiter": max(30, maxfun // 6),
            "maxfun": maxfun,
            "ftol": 1e-12,
            "gtol": 1e-6,
            "maxls": 20,
            "m": 20,
            "disp": False,
        }

        try:
            _scipy_minimize(
                _full_loss_jac_ref,
                best_z,
                args=(n, ref, beta, lam, Nq, Ng),
                method="L-BFGS-B",
                bounds=None,
                options=options,
            )
        except Exception:
            pass

        if REC_A is not None and REC_STATE is not None and REC_SCORE < best_score:
            best_score = REC_SCORE
            best_a = REC_A.copy()
            new_ref = int(np.argmax(best_a))
            if new_ref != ref:
                best_z = _a_to_z_ref(best_a, new_ref)
                ref = new_ref
            else:
                best_z = REC_STATE.copy()

        a_cur = _normalize_array(best_a, n)
        sc_cur = _score_a(a_cur)
        if np.isfinite(sc_cur) and sc_cur < best_score:
            best_score = sc_cur
            best_a = a_cur.copy()
            new_ref = int(np.argmax(best_a))
            best_z = _a_to_z_ref(best_a, new_ref)
            ref = new_ref

    a_best = _normalize_array(best_a, n)
    sc_best = _score_a(a_best)
    if np.isfinite(sc_best) and sc_best < best_score:
        best_score = sc_best
        best_a = a_best.copy()

    return best_a, best_score


def _make_anti_starts(best_a, n, count, rng):
    m = int(n) // 2
    starts = []
    idx = np.arange(m, dtype=np.float64)

    if best_a is not None:
        starts.append(_project_to_anti(best_a, n))

    x = np.full(m, 0.618, dtype=np.float64)
    x += 0.02 * rng.normal(size=m)
    starts.append(np.clip(x, X_EPS, 1.0 - X_EPS))

    x = np.full(m, 0.382, dtype=np.float64)
    x += 0.02 * rng.normal(size=m)
    starts.append(np.clip(x, X_EPS, 1.0 - X_EPS))

    x = np.linspace(0.18, 0.82, m, dtype=np.float64)
    x += 0.03 * rng.normal(size=m)
    starts.append(np.clip(x, X_EPS, 1.0 - X_EPS))

    for f, amp in [(1, 0.25), (2, 0.35), (4, 0.45), (8, 0.55)]:
        phi = float(rng.uniform(0.0, 2.0 * np.pi))
        x = 0.5 + amp * np.sin(2.0 * np.pi * f * idx / float(m) + phi)
        x += 0.02 * rng.normal(size=m)
        starts.append(np.clip(x, X_EPS, 1.0 - X_EPS))

    for _ in range(2):
        x = np.full(m, 0.5, dtype=np.float64)
        nf = 32
        for f in range(1, nf + 1):
            coeff = 0.5 / float(f + 2)
            phi = float(rng.uniform(0.0, 2.0 * np.pi))
            x += coeff * np.sin(2.0 * np.pi * f * idx / float(m) + phi)
        x += 0.03 * rng.normal(size=m)
        starts.append(np.clip(x, X_EPS, 1.0 - X_EPS))

    for delta in [0.25, 0.40, 0.48]:
        half = m // 2
        bits = np.zeros(m, dtype=np.float64)
        if half > 0:
            sel = rng.choice(m, size=half, replace=False)
            bits[sel] = 1.0
        x = 0.5 + float(delta) * (2.0 * bits - 1.0)
        x += 0.005 * rng.normal(size=m)
        starts.append(np.clip(x, X_EPS, 1.0 - X_EPS))

    x = 0.5 + 0.22 * rng.normal(size=m)
    starts.append(np.clip(x, X_EPS, 1.0 - X_EPS))

    x = 0.5 + 0.35 * np.tanh(0.5 * rng.normal(size=m))
    starts.append(np.clip(x, X_EPS, 1.0 - X_EPS))

    while len(starts) < int(count):
        base = starts[len(starts) % len(starts)].copy()
        x = base + 0.05 * rng.normal(size=m)
        starts.append(np.clip(x, X_EPS, 1.0 - X_EPS))

    return [np.asarray(s, dtype=np.float64).ravel() for s in starts[: int(count)]]


def _random_anti_x(m, rng, mode):
    m = int(m)
    if m <= 0:
        return np.array([], dtype=np.float64)
    mode = int(mode) % 3
    idx = np.arange(m, dtype=np.float64)

    if mode == 0:
        half = m // 2
        bits = np.zeros(m, dtype=np.float64)
        if half > 0:
            sel = rng.choice(m, size=half, replace=False)
            bits[sel] = 1.0
        delta = 0.30 + 0.20 * float(rng.random())
        x = 0.5 + delta * (2.0 * bits - 1.0)
        x += 0.005 * rng.normal(size=m)
    elif mode == 1:
        f = 1 + int(rng.integers(0, 9))
        amp = 0.25 + 0.30 * float(rng.random())
        phi = float(rng.uniform(0.0, 2.0 * np.pi))
        x = 0.5 + amp * np.sin(2.0 * np.pi * f * idx / float(m) + phi)
        x += 0.025 * rng.normal(size=m)
    else:
        amp = 0.12 + 0.18 * float(rng.random())
        x = 0.5 + amp * rng.normal(size=m)

    return np.clip(x, X_EPS, 1.0 - X_EPS)
