Strategy: Full-vector exponentiated-gradient optimization with ADAM-style
adaptive step sizes, adaptive log-sum-exp focus (smooth max approximation),

import numpy as np
import time


def _pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _fft_square(a):
    n = int(a.size)
    L = 2 * n - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, n=N)
    b = np.fft.irfft(fa * fa, n=N).real[:L]
    b = np.maximum(b, 0.0)
    return b


def _fft_grad(weights, a):
    n = int(a.size)
    L = int(weights.size)
    N = _pow2(L + n - 1)
    fw = np.fft.rfft(weights, n=N)
    fa = np.fft.rfft(a, n=N)
    c = np.fft.irfft(fw * np.conj(fa), n=N)[:n]
    return 2.0 * c


def _internal_score(a, b=None):
    if b is None:
        b = _fft_square(a)
    s = float(a.sum())
    if s <= 1e-300:
        return float("inf"), b, 0.0
    maxb = float(b.max())
    score = float(2.0 * a.size * maxb / (s * s))
    return score, b, maxb


def _make_weights(b, focus, eps=0.01):
    L = int(b.size)
    maxb = float(b.max())
    if maxb <= 1e-300:
        return np.full(L, 1.0 / L, dtype=np.float64), maxb
    with np.errstate(over="ignore", under="ignore"):
        z = (maxb - b) / max(b.max(), 1e-300)
        p = np.exp(-z * float(focus))
    s = float(p.sum())
    if s <= 1e-12:
        p = np.zeros(L, dtype=np.float64)
        p[int(np.argmax(b))] = 1.0
    else:
        p = (1.0 - eps) * (p / s) + eps / L
    return p, maxb


def _random_positive(n, rng, amp=3.0):
    n = int(n)
    z = rng.normal(0.0, amp, size=n)
    z -= z.max()
    a = np.exp(z)
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    return a


def _multi_sinusoid_init(n, rng, n_freqs=5, max_amp=0.8):
    n = int(n)
    x = np.arange(n, dtype=np.float64)
    v = np.ones(n, dtype=np.float64)
    for _ in range(n_freqs):
        f = rng.integers(2, max(3, n // 4))
        amp = rng.uniform(0.1, max_amp)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        v += amp * np.cos(2.0 * np.pi * f * x / n + phase)
    v = np.maximum(v, 0.01)
    v /= v.sum()
    return v


def _anti_log_init(n, rng, amp=3.0):
    n = int(n)
    m = n // 2
    if m < 1:
        return np.ones(n, dtype=np.float64) / max(1, n)
    v = rng.normal(0.0, 1.0, m)

    std = float(v.std())
    if std > 1e-12:

    z = np.zeros(n, dtype=np.float64)
    z[:m] = v
    if n % 2 == 0:
        z[m:] = -v[::-1]
    else:
        z[m + 1:] = -v[::-1]
        z[m] = 0.0
    z = z - z.max()
    a = np.exp(z)
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    return a


def _chirp_init(n, rng, chirp_rate=2.0):
    n = int(n)
    x = np.arange(n, dtype=np.float64)
    t = (x / n) * 2.0 - 1.0
    v = 1.0 + 0.5 * np.cos(chirp_rate * np.pi * t * t)
    v = np.maximum(v, 0.01)
    v = v * np.exp(rng.normal(0.0, 0.1, size=n))
    v /= v.sum()
    return v


def _optimize_adam(a0, time_limit, rng, verbose=False, label=""):
    """Full-vector optimizer with ADAM, adaptive focus, and SA acceptance."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = int(a.size)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-300:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a /= s
    a = np.maximum(a, 1e-15)
    a /= a.sum()

    b = None
    score = float("inf")
    best_score = float("inf")
    best_a = a.copy()

    # ADAM state
    beta1 = 0.9
    beta2 = 0.999
    m_v = np.zeros(n, dtype=np.float64)
    v_v = np.zeros(n, dtype=np.float64)
    adam_t = 0

    # Schedule
    focus = 5.0
    focus_inc = 1.015
    focus_max = 2000.0
    lr = 0.5
    lr_min = 0.01

    # SA temperature
    T = 0.05
    T_decay = 0.9995
    T_min = 0.001

    no_improve = 0
    max_no_improve = 800
    resets = 0
    max_resets = 40
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        if b is None:
            b = _fft_square(a)
            score, _, _ = _internal_score(a, b)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and (it % 500 == 0):
            print(
                f"[{label}] it={it} t={time.monotonic()-t0:.1f}s "
                f"score={score:.6f} best={best_score:.6f} "
                f"focus={focus:.1f} lr={lr:.4f} T={T:.4f} resets={resets}"
            )

        if no_improve > max_no_improve:
            resets += 1
            if resets > max_resets:
                break
            no_improve = 0
            max_no_improve = int(max_no_improve * 1.2)
            # Diverse restart
            rtype = rng.integers(0, 4)
            if rtype == 0:
                a = 0.5 * best_a + 0.5 * _random_positive(n, rng, amp=2.0)
            elif rtype == 1:
                a = best_a * np.exp(rng.normal(0.0, 0.5, size=n))
            elif rtype == 2:
                a = _multi_sinusoid_init(n, rng, n_freqs=rng.integers(3, 8))
                a = 0.5 * best_a + 0.5 * a
            else:
                a = _anti_log_init(n, rng, amp=rng.uniform(2.0, 5.0))
                a = 0.5 * best_a + 0.5 * a
            a = np.maximum(a, 1e-15)
            a /= a.sum()
            b = None
            score = float("inf")
            focus = float(rng.choice([3.0, 5.0, 10.0, 20.0, 50.0]))
            lr = 0.5
            T = max(T, 0.03)
            m_v = np.zeros(n, dtype=np.float64)
            v_v = np.zeros(n, dtype=np.float64)
            adam_t = 0
            continue

        # Compute gradient with log-sum-exp
        p, _ = _make_weights(b, focus)
        grad = _fft_grad(p, a)
        # Demean to preserve sum (approximately)
        grad = grad - float(np.dot(a, grad))

        # ADAM update
        adam_t += 1
        m_v = beta1 * m_v + (1 - beta1) * grad
        v_v = beta2 * v_v + (1 - beta2) * (grad * grad)
        m_hat = m_v / (1 - beta1 ** adam_t)
        v_hat = v_v / (1 - beta2 ** adam_t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # Exponentiated update (maintains positivity)
        a_new = a * np.exp(-step)
        a_new = np.maximum(a_new, 1e-300)
        ss = float(a_new.sum())
        if ss <= 1e-300:
            break
        a_new /= ss

        b_new = _fft_square(a_new)
        score_new, _, _ = _internal_score(a_new, b_new)

        # SA acceptance
        delta = score_new - score
        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            a = a_new
            b = b_new
            score = score_new
        # lr adaptation
        if score < best_score - 1e-12:
            lr = min(lr * 1.1, 2.0)
        else:
            lr = max(lr * 0.995, lr_min)

        focus = min(focus * focus_inc, focus_max)
        T = max(T * T_decay, T_min)
        it += 1

    return best_a, float(best_score)


def _optimize_pair_adam(x0, time_limit, rng, verbose=False, label=""):
    """Optimize in anti-palindromic parameterization with ADAM."""
    x = np.asarray(x0, dtype=np.float64).copy()
    m = int(x.size)
    n = 2 * m
    x_eps = 1e-8
    x = np.clip(x, x_eps, 1.0 - x_eps)

    a = _a_from_x(x, n)
    b = None
    score = float("inf")
    best_score = float("inf")
    best_x = x.copy()

    beta1 = 0.9
    beta2 = 0.999
    m_v = np.zeros(m, dtype=np.float64)
    v_v = np.zeros(m, dtype=np.float64)
    adam_t = 0

    focus = 5.0
    focus_inc = 1.02
    focus_max = 1500.0
    lr = 0.3
    lr_min = 0.005

    T = 0.03
    T_decay = 0.9995
    T_min = 0.0005

    no_improve = 0
    max_no = 1000
    resets = 0
    max_resets = 30
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        if b is None:
            b = _fft_square(a)
            score, _, _ = _internal_score(a, b)

        if score < best_score - 1e-12:
            best_score = score
            best_x = x.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and (it % 500 == 0):
            print(
                f"[{label}] pair it={it} t={time.monotonic()-t0:.1f}s "
                f"score={score:.6f} best={best_score:.6f} "
                f"focus={focus:.1f} lr={lr:.4f} resets={resets}"
            )

        if no_improve > max_no:
            resets += 1
            if resets > max_resets:
                break
            no_improve = 0
            max_no = int(max_no * 1.2)
            x = 0.5 * best_x + 0.5 * rng.uniform(0.2, 0.8, size=m)
            x = np.clip(x, x_eps, 1.0 - x_eps)
            a = _a_from_x(x, n)
            b = None
            score = float("inf")
            focus = float(rng.choice([3.0, 5.0, 10.0, 30.0, 100.0]))
            lr = 0.3
            T = max(T, 0.02)
            m_v = np.zeros(m, dtype=np.float64)
            v_v = np.zeros(m, dtype=np.float64)
            adam_t = 0
            continue

        p, _ = _make_weights(b, focus)
        ga = _fft_grad(p, a)
        gx = ga[:m] - ga[m:][::-1]

        adam_t += 1
        m_v = beta1 * m_v + (1 - beta1) * gx
        v_v = beta2 * v_v + (1 - beta2) * (gx * gx)
        m_hat = m_v / (1 - beta1 ** adam_t)
        v_hat = v_v / (1 - beta2 ** adam_t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        x_new = x - step
        x_new = np.clip(x_new, x_eps, 1.0 - x_eps)

        if not np.any(x_new != x):
            x_new = np.clip(x + rng.normal(0.0, 0.02, size=m), x_eps, 1.0 - x_eps)

        a_new = _a_from_x(x_new, n)
        b_new = _fft_square(a_new)
        score_new, _, _ = _internal_score(a_new, b_new)

        delta = score_new - score
        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            x = x_new
            a = a_new
            b = b_new
            score = score_new

        if score < best_score - 1e-12:
            lr = min(lr * 1.08, 1.5)
        else:
            lr = max(lr * 0.995, lr_min)

        focus = min(focus * focus_inc, focus_max)
        T = max(T * T_decay, T_min)
        it += 1

    return _a_from_x(best_x, n), float(best_score)


def _a_from_x(x, n=None):
    x = np.asarray(x, dtype=np.float64).ravel()
    m = x.size
    if n is None:
        n = 2 * m
    t = 1.0 / m
    a = np.empty(n, dtype=np.float64)
    a[:m] = t * x
    a[m:] = t * (1.0 - x[::-1])
    return a


def _greedy_x(n, rng, shuffle=True):
    m = n // 2
    counts = np.zeros(2 * n - 1, dtype=np.int64)
    selected = np.empty(m, dtype=np.int64)
    order = np.arange(m, dtype=np.int64)
    if shuffle:
        rng.shuffle(order)
    t = 0
    for idx_val in order:
        idx = int(idx_val)
        l = idx
        r = n - 1 - idx
        if t == 0:
            y = l
        else:
            s = selected[:t]
            affected_l = int(np.max(counts[s + l])) if t > 0 else 0
            self_l = int(counts[2 * l])
            metric_l = max(affected_l + 2, self_l + 1)
            affected_r = int(np.max(counts[s + r])) if t > 0 else 0
            self_r = int(counts[2 * r])
            metric_r = max(affected_r + 2, self_r + 1)
            y = l if metric_l <= metric_r else r
        if t > 0:
            counts[selected[:t] + y] += 1
        counts[2 * y] += 1
        selected[t] = y
        t += 1
    x = np.full(m, 1e-6, dtype=np.float64)
    left = selected < m
    if np.any(left):
        x[sel[left]] = 1.0 - 1e-6 if False else 1.0 - 1e-6
    # Fix: select where the greedy chose left positions to be "high"
    x = np.full(m, 0.5, dtype=np.float64)
    x[sel[left]] = 0.9
    x[~left] = 0.1
    return x


def _safe_evaluate(seq):
    ev = globals().get("evaluate_sequence")
    if ev is None or not callable(ev):
        return float("inf")
    try:
        return float(ev(seq))
    except Exception:
        return float("inf")


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    seed = int(seed)
    total = max(120.0, float(budget_s))
    now0 = time.monotonic()
    final_deadline = now0 + total - 20.0

    print("starting step-function convolution optimizer (ADAM+SA)")
    best_a = None
    best_score = float("inf")

    # Try to use given sequence as a starting point
    given_a = None
    try:
        given = globals().get("height_sequence_1")
        if given is not None:
            g = np.asarray(given, dtype=np.float64).ravel()
            g = np.nan_to_num(g, nan=0.0, posinf=1000.0, neginf=0.0)
            g = np.clip(g, 0.0, 1000.0)
            s = float(g.sum())
            if s > 0.01 and g.size > 10:
                gn = g / s
                sc, _, _ = _internal_score(gn)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = gn.copy()
                if g.size <= 33000:
                    given_a = gn
                print(f"given internal score: {sc:.6f}, n={gn.size}")
    except Exception:
        given_a = None

    # Phase 1: Quick exploration (first ~40% of budget)
    explore_deadline = now0 + total * 0.35
    full_deadline = now0 + total * 0.75

    tasks = []
    if given_a is not None and given_a.size <= 33000:
        tasks.append((given_a.size, "full", 0, 60.0))

    # Multiple n values with different initializations
    for i, n_val in enumerate([8192, 16384, 16384, 32768]):
        for j, init_type in enumerate(["random", "multi_sin", "anti", "chirp"]):
            if i + j > 7:
                break
            tasks.append((n_val, init_type, i * 10 + j, 45.0))

    task_idx = 0
    for n_val, kind, task_seed, alloc in tasks:
        now = time.monotonic()
        if now >= explore_deadline:
            break
        alloc = min(alloc, explore_deadline - now)
        if alloc < 15.0:
            break

        trng = np.random.default_rng((seed * 1000003 + task_idx * 7919 + task_seed * 131) % (2**32))
        print(f"explore {task_idx}: n={n_val} kind={kind} alloc={alloc:.0f}s best={best_score:.6f}")
        t0 = time.monotonic()

        try:
            if kind == "full" and given_a is not None:
                a0 = given_a.copy()
                if a0.size != n_val:
                    # Resample to target n
                    if a0.size > n_val:
                        idx = np.linspace(0, a0.size - 1, n_val).astype(int)
                        a0 = a0[idx]
                    else:
                        idx = np.linspace(0, 1, n_val)
                        old_idx = np.linspace(0, 1, a0.size)
                        a0 = np.interp(idx, old_idx, a0)
                cand_a, cand_score = _optimize_adam(a0, alloc, trng, label=f"full{n_val}")
            elif kind == "random":
                a0 = _random_positive(n_val, trng, amp=rng.uniform(2.0, 5.0) if False else trng.uniform(2.0, 5.0))
                cand_a, cand_score = _optimize_adam(a0, alloc, trng, label=f"rand{n_val}")
            elif kind == "multi_sin":
                a0 = _multi_sinusoid_init(n_val, trng, n_freqs=trng.integers(3, 10))
                cand_a, cand_score = _optimize_adam(a0, alloc, trng, label=f"msin{n_val}")
            elif kind == "anti":
                a0 = _anti_log_init(n_val, trng, amp=trng.uniform(2.0, 5.0))
                cand_a, cand_score = _optimize_adam(a0, alloc, trng, label=f"anti{n_val}")
            elif kind == "chirp":
                a0 = _chirp_init(n_val, trng, chirp_rate=trng.uniform(1.0, 5.0))
                cand_a, cand_score = _optimize_adam(a0, alloc, trng, label=f"chirp{n_val}")
            else:
                continue

            if cand_a is not None and np.isfinite(cand_score):
                cand_a = np.nan_to_num(cand_a, nan=0.0, posinf=0.0, neginf=0.0)
                cand_a = np.clip(cand_a, 0.0, 1000.0)
                cc = float(cand_a.sum())
                if cc > 0.01:
                    cand_a /= cc
                    internal, _, _ = _internal_score(cand_a)
                    if np.isfinite(internal) and internal < cand_score:
                        cand_score = internal
                    if cand_score < best_score:
                        best_score = cand_score
                        best_a = cand_a.copy()
                        print(f"  NEW BEST: {cand_score:.6f} (n={cand_a.size})")
        except Exception as exc:
            print(f"  task failed: {repr(exc)}")

        task_idx += 1

    print(f"exploration done. best={best_score:.6f}")

    # Phase 2: Extended optimization on best candidates
    if best_a is not None:
        remaining = min(full_deadline, final_deadline) - time.monotonic()
        if remaining > 30.0:
            # Run several extended optimizations from the best and nearby points
            n_run = min(4, int(remaining // 60) + 1)
            for i in range(n_run):
                now = time.monotonic()
                if now >= remaining - 10:
                    break
                alloc = min(remaining - now - 10.0, 180.0)
                if alloc < 20.0:
                    break
                trng = np.random.default_rng((seed + 500000 + i * 31337) % (2**32))
                print(f"extend {i}: alloc={alloc:.0f}s best={best_score:.6f}")

                # Create a slightly perturbed version of the best
                pert = 0.3 / (i + 1)
                a_start = best_a * np.exp(trng.normal(0.0, pert, size=best_a.size))
                a_start = np.maximum(a_start, 1e-15)
                a_start /= a_start.sum()

                try:
                    cand_a, cand_score = _optimize_adam(
                        a_start, alloc, trng,
                        verbose=(i == 0),
                        label=f"ext{i}"
                    )
                    if cand_a is not None and np.isfinite(cand_score):
                        if cand_score < best_score:
                            best_score = cand_score
                            best_a = cand_a.copy()
                            print(f"  NEW BEST: {cand_score:.6f}")
                except Exception as exc:
                    print(f"  extend failed: {repr(exc)}")

    # Phase 3: Final refinement
    if best_a is not None:
        remaining = final_deadline - time.monotonic()
        if remaining > 30.0:
            refine_time = min(remaining - 10.0, 120.0)
            if refine_time > 15.0:
                trng = np.random.default_rng((seed + 999999) % (2**32))
                print(f"final refinement: time={refine_time:.0f}s")
                try:
                    refined_a, refined_score = _optimize_adam(
                        best_a, refine_time, trng,
                        verbose=True, label="final"
                    )
                    if refined_a is not None and np.isfinite(refined_score):
                        if refined_score < best_score:
                            best_score = refined_score
                            best_a = refined_a
                except Exception as exc:
                    print(f"  refine failed: {repr(exc)}")

    # Ensure we have a valid output
    if best_a is None:
        best_a = np.ones(1024, dtype=np.float64) / 1024.0

    best_a = np.nan_to_num(best_a, nan=0.0, posinf=1000.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    ss = float(best_a.sum())
    if ss <= 0.01:
        best_a = np.ones(1024, dtype=np.float64) / 1024.0
        ss = 1.0
    else:
        best_a /= ss

    final_seq = [float(x) for x in best_a]

    # Verify with exact evaluation if possible
    if best_a.size <= 33000 and time.monotonic() < final_deadline - 5.0:
        exact = _safe_evaluate(final_seq)
        if np.isfinite(exact):
            print(f"final: internal={best_score:.6f} exact={exact:.6f} n={best_a.size}")
            if exact < best_score:
                best_score = exact
        else:
            print(f"final: internal={best_score:.6f} exact=N/A n={best_a.size}")
    else:
        print(f"final: internal={best_score:.6f} n={best_a.size}")

    return final_seq
