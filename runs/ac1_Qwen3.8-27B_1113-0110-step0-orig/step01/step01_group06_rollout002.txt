def _optimize_full(a0, time_limit, rng, verbose=False):

    s = float(a.sum())
    if s <= 1e-300:
        a = np.ones(n) / n
        s = 1.0
    a /= s  # normalize to sum=1
    
    b = _fft_square(a)
    score, _ = _score(a, b)
    best_a = a.copy()
    best_score = score
    
    t0 = time.monotonic()
    it = 0
    
    # Exponentiated gradient (mirror descent) with increasing focus
    focus = 5.0
    eta = 0.5
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        b = _fft_square(a)
        score, _ = _score(a, b)
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            if it > 2000:
                break
            # Perturb
            a = a * np.exp(rng.normal(0, 0.3, n))
            a = np.maximum(a, 1e-15)
            a /= a.sum()
            no_improve = 0
            focus = rng.choice([5, 20, 50, 100])
            continue
        
        # Weight peak lags
        maxb = float(b.max())
        if maxb <= 1e-300:
            break
        z = (maxb - b) / maxb  # 0 at peak, 1 at minimum
        with np.errstate(over='ignore', under='ignore'):
            w = np.exp(-z * focus)
        wsum = w.sum()
        if wsum <= 0:
            w = np.zeros_like(b)
            w[np.argmin(z)] = 1.0
        else:
            w /= wsum
        
        # Gradient via FFT cross-correlation
        N = _pow2(n + len(b))
        fw = np.fft.rfft(w, n=N)
        fa = np.fft.rfft(a, n=N)
        grad = 2.0 * np.fft.irfft(fw * np.conj(fa), n=N)[:n]
        
        # Mirror descent: a_new = a * exp(-eta * (grad - mean(a*grad)))
        weighted_grad = a * grad
        eta_new = eta * (1.0 - weighted_grad.mean())  # normalize
        # Actually for mirror descent with KL, the update is:
        # a_new[i] = a[i] * exp(-eta * grad[i]) / Z
        # But we want to preserve sum(a)=1
        a_new = a * np.exp(-eta_new * grad)
        a_new = np.maximum(a_new, 1e-300)
        a_new /= a_new.sum()
        
        b_new = _fft_square(a_new)
        score_new, _ = _score(a_new, b_new)
        
        if score_new < score * 1.001:
            a = a_new
            eta = min(eta * 1.02, 10.0)
        else:
            eta = max(eta * 0.8, 0.01)
        
        focus = min(focus * 1.01, 500.0)
        it += 1
        
        if verbose and it % 1000 == 0:
            print(f"  full n={n} it={it} t={time.monotonic()-t0:.1f}s score={score:.6f} best={best_score:.6f} focus={focus:.1f} eta={eta:.3f}")
    
    return best_a, best_score

"""
Optimize non-negative step function to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses anti-palindromic parameterization with Adam optimizer, peak-directed
subgradient, exponentiated gradient for full-space refinement, and LP-based
template method. Multi-scale search across n=4096..32768 with diverse init.
"""

import numpy as np
import time


def _pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _fft_square(a):
    n = int(a.size)
    L = 2 * n - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, n=N)
    b = np.fft.irfft(fa * fa, n=N).real[:L]
    return np.maximum(b, 0.0)


def _score(a, b=None):
    if b is None:
        b = _fft_square(a)
    s = float(a.sum())
    if s <= 1e-300:
        return float('inf')
    maxb = float(b.max())
    return float(2.0 * a.size * maxb / (s * s))


def _grad_conv(w, a):
    """Gradient of sum_k w[k]*(a*a)[k] w.r.t. a, via FFT cross-correlation."""
    n = int(a.size)
    L = int(w.size)
    N = _pow2(n + L)
    fw = np.fft.rfft(w, n=N)
    fa = np.fft.rfft(a, n=N)
    return 2.0 * np.fft.irfft(fw * np.conj(fa), n=N)[:n]


def _peak_weights(b, focus, top_k=20):
    """Create weights emphasizing the top lags of the convolution."""
    L = b.size
    maxb = b.max()
    if maxb <= 1e-300:
        return np.full(L, 1.0 / L)
    z = (maxb - b) / maxb
    with np.errstate(over='ignore', under='ignore'):
        p = np.exp(-z * float(focus))
    s = p.sum()
    if s <= 1e-12:
        p = np.zeros(L)
        p[int(np.argmax(b))] = 1.0
    else:
        p /= s
        # Extra emphasis on top-k
        idx = np.argsort(b)[::-1][:top_k]
        p[idx] *= 3.0
        p /= p.sum()
    return p


def _a_from_x(x, n):
    m = x.size
    t = 1.0 / m
    a = np.empty(n, dtype=np.float64)
    a[:m] = t * x
    a[m:] = t * (1.0 - x[::-1])
    return a


def _optimize_anti(n, time_limit, rng, label=""):
    """Adam optimizer on anti-palindromic parameters x in [0,1]^m."""
    m = n // 2
    x = np.full(m, 0.5)
    x_eps = 1e-10
    x = np.clip(x, x_eps, 1.0 - x_eps)

    best_x = x.copy()
    best_score = float('inf')

    t0 = time.monotonic()
    it = 0
    lr = 0.5
    beta1 = 0.8
    beta2 = 0.95
    m_t = np.zeros(m)
    v_t = np.zeros(m)
    focus = 3.0
    focus_inc = 1.015
    no_improve = 0
    max_no = 300
    resets = 0

    while time.monotonic() - t0 < time_limit:
        a = _a_from_x(x, n)
        b = _fft_square(a)
        score = _score(a, b)

        if score < best_score - 1e-9:
            best_score = score
            best_x = x.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no:
            resets += 1
            if resets > 30:
                break
            no_improve = 0
            max_no = int(max_no * 1.4)
            # Perturb from best
            x = best_x + rng.normal(0, 0.1, m)
            x = np.clip(x, x_eps, 1.0 - x_eps)
            focus = rng.choice([3.0, 10.0, 30.0, 100.0])
            lr = 0.3
            continue

        w = _peak_weights(b, focus, top_k=min(30, len(b)//20))
        ga = _grad_conv(w, a)
        gx = (ga[:m] - ga[m:][::-1]) / m

        it += 1
        m_t = beta1 * m_t + (1 - beta1) * gx
        v_t = beta2 * v_t + (1 - beta2) * gx * gx
        m_hat = m_t / (1 - beta1 ** it)
        v_hat = v_t / (1 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        x_new = x - step
        x_new = np.clip(x_new, x_eps, 1.0 - x_eps)

        a_new = _a_from_x(x_new, n)
        b_new = _fft_square(a_new)
        score_new = _score(a_new, b_new)

        if score_new <= score * 1.005:
            x = x_new
            lr = min(lr * 1.01, 2.0)
        else:
            lr = max(lr * 0.9, 0.01)
            if lr < 0.05 and rng.random() < 0.1:
                x = x + rng.normal(0, 0.05, m)
                x = np.clip(x, x_eps, 1.0 - x_eps)

        focus = min(focus * focus_inc, 500.0)

        if it % 2000 == 0:
            print(f"  [{label}] it={it} t={time.monotonic()-t0:.1f}s "
                  f"score={score:.6f} best={best_score:.6f} focus={focus:.1f} lr={lr:.4f}")

    a_best = _a_from_x(best_x, n)
    return a_best, best_score


def _optimize_full(a0, time_limit, rng, label=""):
    """Exponentiated gradient (mirror descent) on full vector."""
    a = np.asarray(a0, dtype=np.float64).copy().ravel()
    n = a.size
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s <= 1e-300:
        a = np.ones(n) / n
    else:
        a /= s
    a = np.maximum(a, 1e-15)
    a /= a.sum()

    best_a = a.copy()
    best_score = _score(a)

    t0 = time.monotonic()
    it = 0
    eta = 1.0
    focus = 5.0
    no_improve = 0
    max_no = 400
    resets = 0

    while time.monotonic() - t0 < time_limit:
        b = _fft_square(a)
        score = _score(a, b)

        if score < best_score - 1e-9:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no:
            resets += 1
            if resets > 20:
                break
            no_improve = 0
            max_no = int(max_no * 1.5)
            a = best_a * np.exp(rng.normal(0, 0.5, n))
            a = np.maximum(a, 1e-15)
            a /= a.sum()
            focus = rng.choice([3.0, 10.0, 30.0, 100.0])
            eta = 0.5
            continue

        w = _peak_weights(b, focus, top_k=min(30, len(b)//20))
        grad = _grad_conv(w, a)

        # Mirror descent: a_new = a * exp(-eta * grad) / normalizer
        # Subtract weighted mean to preserve direction
        gd = grad - float(np.dot(a, grad))
        a_new = a * np.exp(-eta * gd)
        a_new = np.maximum(a_new, 1e-300)
        a_new /= a_new.sum()

        b_new = _fft_square(a_new)
        score_new = _score(a_new, b_new)

        wt_old = float(np.dot(w, b))
        wt_new = float(np.dot(w, b_new))

        accept = score_new < score or (score_new < score * 1.002 and wt_new < wt_old)

        if accept:
            a = a_new
            eta = min(eta * 1.03, 15.0)
        else:
            eta = max(eta * 0.85, 0.05)
            if eta < 0.1:
                a = a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 1e-15)
                a /= a.sum()

        focus = min(focus * 1.012, 400.0)
        it += 1

        if it % 2000 == 0:
            print(f"  [{label}] it={it} t={time.monotonic()-t0:.1f}s "
                  f"score={score:.6f} best={best_score:.6f} focus={focus:.1f} eta={eta:.3f}")

    return best_a, best_score


def _sinusoidal_x(m, rng, nfreq=5):
    """Generate anti-palindromic init using sum of sinusoids."""
    x = np.zeros(m)
    for _ in range(nfreq):
        f = rng.integers(1, m // 2 + 1)
        amp = rng.uniform(0.3, 1.0)
        phase = rng.uniform(0, 2 * np.pi)
        t = np.arange(m)
        x += amp * np.sin(2 * np.pi * f * t / m + phase)
    x = 0.5 + 0.3 * x
    x = np.clip(x, 1e-6, 1.0 - 1e-6)
    return x


def _lp_step(a, n, max_time=10.0):
    """LP template: max sum(b) s.t. b>=0, (a*b)[k]<=M for all k."""
    from scipy.optimize import linprog
    M = float(_fft_square(a).max())
    t0 = time.monotonic()

    # For large n, subsample constraints to keep LP tractable
    n_con = 2 * n - 1
    if n > 4096:
        # Use only every other constraint + top 100
        b_full = _fft_square(a)
        top_idx = np.argsort(b_full)[::-1][:min(200, n_con)]
        other_idx = np.arange(0, n_con, max(1, n_con // 4000))
        idx = np.unique(np.concatenate([top_idx, other_idx]))
    else:
        idx = np.arange(n_con)

    # Build constraint matrix A[idx, j] = a[idx - j] if 0 <= idx-j < n
    A_rows = []
    for k in idx:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo > hi:
            A_rows.append(np.zeros(n))
        else:
            row = np.zeros(n)
            # b[j] contributes a[k-j] to constraint k when 0<=k-j<n
            # So row[j] = a[k-j] if 0<=k-j<n
            js = np.arange(lo, hi + 1)
            row[js] = a[k - js]
            A_rows.append(row)
    A = np.array(A_rows)
    b_ub = np.full(A.shape[0], M)

    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs', options={'time_limit': max_time, 'presolve': True})

    if res.status in (0, 3) and res.x is not None:
        b_opt = np.maximum(res.x, 0.0)
        s = b_opt.sum()
        if s > 1e-10:
            return b_opt / s  # normalize to sum=1
    return None


def _lp_refine(a, time_limit, rng, label=""):
    """Iteratively apply LP template method."""
    a = a.copy()
    n = a.size
    best_a = a.copy()
    best_score = _score(a)
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        b_opt = _lp_step(a, n, max_time=min(15.0, time_limit / 4))
        if b_opt is None:
            break

        # Mix: a_new = (1-t)*a + t*b_opt
        t = 0.3
        a_new = (1 - t) * a + t * b_opt
        a_new = np.maximum(a_new, 1e-15)
        a_new /= a_new.sum()

        sc_new = _score(a_new)
        if sc_new < best_score:
            a = a_new
            best_a = a.copy()
            best_score = sc_new
            t = min(t * 1.2, 0.7)
        else:
            t = max(t * 0.7, 0.05)
            if t < 0.06:
                break

        it += 1
        if it % 5 == 0:
            print(f"  [{label} LP] it={it} t={time.monotonic()-t0:.1f}s best={best_score:.6f}")

    return best_a, best_score
