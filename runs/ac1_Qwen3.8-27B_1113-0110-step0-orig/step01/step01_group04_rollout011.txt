```python
"""
Search for non-negative step-function heights minimizing
    2 * n * max(convolve(a, a)) / sum(a)^2.

This implementation uses the exact convolution objective with FFT-based
accelerated convolution and, importantly, uses the correct gradient of the
convolution-squared coefficients:
    d/d a_i (a*a)_k = 2 a_{k-i}.

The main search combines:
  * diverse analytic initial shapes, including resampling of the previous
    best sequence when available,
  * L-BFGS-B on log-ratio simplex parameters for several surrogate
    objectives: L2 and L3 norms of the convolution coefficients and
    active-peak average objectives,
  * a smooth maximum / log-sum-exp objective with decreasing temperature,
  * multiplicative block local search,
  * a final gradient-based peak-balancing polish.

The goal is to escape local minima of plain smooth-max optimization and to
enforce peak equalization/KKT-type conditions over the most populated
convolution coefficients.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize as _minimize
except Exception:
    _minimize = None


_BOUND = 60.0
_DEFAULT_N = 3809


def _pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _normalize(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    v = np.maximum(v, 0.0)
    s = float(v.sum())
    if not np.isfinite(s) or s <= 1e-300:
        n = v.size
        if n == 0:
            return v
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    return v / s


def _resample_sequence(v, n):
    arr = _normalize(v)
    n = int(n)
    if arr.size == n:
        return arr
    if arr.size == 0 or n == 0:
        return np.full(n, 1.0 / max(1, n), dtype=np.float64)
    if arr.size == 1:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, arr.size)
    x_new = np.linspace(0.0, 1.0, n)
    out = np.interp(x_new, x_old, arr)
    return _normalize(out)


def _score_fft_len(n):
    return _pow2(2 * int(n) - 1)


def _grad_fft_len(n):
    return _pow2(3 * int(n) - 2)


def _score_unit(v):
    """Exact score for a sequence, after non-negativity clipping/normalization."""
    v = _normalize(v)
    n = v.size
    if n <= 0:
        return np.inf
    L = _score_fft_len(n)
    A = np.fft.rfft(v, n=L)
    c = np.fft.irfft(A * A, n=L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return np.inf
    return 2.0 * float(n) * mx


def _c_A_from_a(a):
    """Convolution coefficients and FFT vector using a longer FFT length."""
    a = _normalize(a)
    n = a.size
    L = _grad_fft_len(n)
    A = np.fft.rfft(a, n=L)
    c = np.fft.irfft(A * A, n=L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return A, c


def _hankel_matvec(w, a, A=None, L=None):
    """
    For convolution-squared coefficients, the weighted gradient matrix acts as

        (T_w a)_i = sum_{k} w_k a_{k-i},  valid indices.

    This returns T_w a efficiently by an FFT convolution.
    """
    a = _normalize(a)
    n = a.size
    Lc = 2 * n - 1
    if L is None:
        L = _grad_fft_len(n)
    if A is None:
        A = np.fft.rfft(a, n=L)
    if n == 1:
        ww = max(0.0, float(np.asarray(w, dtype=np.float64).sum()))
        return np.array([ww * float(a[0])], dtype=np.float64)

    w = np.asarray(w, dtype=np.float64)
    if w.size != Lc:
        tmp = np.zeros(Lc, dtype=np.float64)
        m = min(w.size, Lc)
        if m > 0:
            tmp[:m] = w[:m]
        w = tmp

    wr = w[::-1]
    H = np.fft.rfft(wr, n=L)
    y = np.fft.irfft(A * H, n=L)
    # y[Lc-1], y[Lc-2], ..., y[n-1]
    return y[2 * n - 2:n - 1:-1]


def _ratios_from_a(a):
    """
    Positive-simplex parameterization: log(a_i / a_ref), all i except ref.
    Ref is fixed at the center to avoid endpoint degeneracy.
    """
    a = _normalize(a)
    n = a.size
    if n <= 1:
        return np.zeros(0, dtype=np.float64)
    ref = n // 2
    ref_val = float(a[ref])
    if ref_val <= 1e-300:
        a = a + 1e-15
        a = _normalize(a)
        ref_val = float(a[ref])
    safe = np.maximum(a, 1e-300)
    x_all = np.log(safe / ref_val)
    x_all = np.where(np.isfinite(x_all), x_all, 0.0)

    x = np.empty(n - 1, dtype=np.float64)
    if ref > 0:
        x[:ref] = x_all[:ref]
    if n - ref - 1 > 0:
        x[ref:] = x_all[ref + 1:]
    return np.clip(x, -_BOUND, _BOUND)


def _a_from_ratios(x):
    """Reconstruct positive simplex a from log-ratios to fixed center ref."""
    x = np.asarray(x, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    n = x.size + 1
    if n <= 1:
        return np.ones(1, dtype=np.float64)

    ref = n // 2
    m = 0.0
    if x.size:
        mx = float(np.max(x))
        if mx > m:
            m = mx

    e = np.exp(np.clip(x - m, -745.0, 709.0))
    e_ref = float(np.exp(-m))
    s = float(e.sum()) + e_ref
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(n, 1.0 / float(n), dtype=np.float64)

    a = np.empty(n, dtype=np.float64)
    if ref > 0:
        a[:ref] = e[:ref]
    if n - ref - 1 > 0:
        a[ref + 1:] = e[ref:]
    a[ref] = e_ref
    return a / s


def _tangent_grad(a, g):
    """
    Convert unconstrained gradient g wrt a into gradients wrt
    log-ratio variables (all indices except fixed ref).
    """
    n = a.size
    if n <= 1:
        return np.zeros(0, dtype=np.float64)
    dot = float(np.dot(a, g))
    gfull = a * (g - dot)
    ref = n // 2
    out = np.empty(n - 1, dtype=np.float64)
    if ref > 0:
        out[:ref] = gfull[:ref]
    if n - ref - 1 > 0:
        out[ref:] = gfull[ref + 1:]
    out = np.where(np.isfinite(out), out, 0.0)
    return out


def _soft_f_jac(x, eps):
    """Smoothed maximum objective and correct gradient."""
    a = _a_from_ratios(x)
    n = a.size
    if n <= 1:
        return 0.0, np.zeros(0, dtype=np.float64)

    A, c = _c_A_from_a(a)
    if c.size == 0:
        return 0.0, np.zeros(n - 1, dtype=np.float64)

    cmax = float(np.max(c))
    if cmax <= 1e-300:
        return 0.0, np.zeros(n - 1, dtype=np.float64)

    eps = float(eps)
    if eps <= 1e-300:
        eps = 1e-12

    z = (c - cmax) / eps
    z = np.clip(z, -745.0, 0.0)
    e = np.exp(z)
    sume = float(e.sum())
    if not np.isfinite(sume) or sume <= 0.0:
        p = np.full(2 * n - 1, 1.0 / float(2 * n - 1), dtype=np.float64)
        F = 2.0 * n * cmax
    else:
        p = e / sume
        F = 2.0 * n * (cmax + eps * float(np.log(sume)))

    L = _grad_fft_len(n)
    h = _hankel_matvec(p, a, A, L)
    g = 2.0 * h * (2.0 * n)
    gy = _tangent_grad(a, g)

    if not np.all(np.isfinite(gy)):
        gy = np.zeros(n - 1, dtype=np.float64)
        F = 1e100
    if not np.isfinite(F):
        F = 1e100
        gy = np.zeros(n - 1, dtype=np.float64)

    return float(F), gy


def _q_f_jac(x, q):
    """Sum of c_k^q objective and correct gradient."""
    a = _a_from_ratios(x)
    n = a.size
    if n <= 1:
        return 0.0, np.zeros(0, dtype=np.float64)

    A, c = _c_A_from_a(a)
    if c.size == 0:
        return 0.0, np.zeros(n - 1, dtype=np.float64)

    qq = float(q)
    if qq <= 1.0:
        F = 2.0 * n * float(np.sum(c))
        w = np.ones_like(c, dtype=np.float64)
    else:
        F = 2.0 * n * float(np.sum(np.power(c, qq)))
        if qq == 2.0:
            w = 2.0 * c
        else:
            w = qq * np.power(c, qq - 1.0)

    L = _grad_fft_len(n)
    h = _hankel_matvec(w, a, A, L)
    g = 2.0 * h * (2.0 * n)
    gy = _tangent_grad(a, g)

    if not np.all(np.isfinite(gy)):
        gy = np.zeros(n - 1, dtype=np.float64)
        F = 1e100
    if not np.isfinite(F):
        F = 1e100
        gy = np.zeros(n - 1, dtype=np.float64)

    return float(F), gy


def _active_f_jac(x, gap_frac):
    """
    Objective: average of the most active convolution coefficients.

    active set = {k : c_k >= (1-gap) max(c)}.
    This encourages peak equalization over the currently dominant lags.
    """
    a = _a_from_ratios(x)
    n = a.size
    if n <= 1:
        return 0.0, np.zeros(0, dtype=np.float64)

    A, c = _c_A_from_a(a)
    if c.size == 0:
        return 0.0, np.zeros(n - 1, dtype=np.float64)

    M = float(np.max(c))
    if M <= 1e-300:
        return 0.0, np.zeros(n - 1, dtype=np.float64)

    gap = float(gap_frac)
    if gap < 0.0:
        gap = 0.0
    if gap > 1.0:
        gap = 1.0

    thr = max(0.0, M * (1.0 - gap))
    active = c >= thr
    if not np.any(active):
        active[int(np.argmax(c))] = True

    cnt = float(np.count_nonzero(active))
    F = 2.0 * n * float(np.mean(c[active]))

    p = np.zeros(2 * n - 1, dtype=np.float64)
    p[active] = 1.0 / cnt

    L = _grad_fft_len(n)
    h = _hankel_matvec(p, a, A, L)
    g = 2.0 * h * (2.0 * n)
    gy = _tangent_grad(a, g)

    if not np.all(np.isfinite(gy)):
        gy = np.zeros(n - 1, dtype=np.float64)
        F = 1e100
    if not np.isfinite(F):
        F = 1e100
        gy = np.zeros(n - 1, dtype=np.float64)

    return float(F), gy


def _run_minimize(fun, x, args, maxiter, bounds=None):
    """One L-BFGS-B block, with a simple fallback if SciPy is unavailable."""
    x = np.asarray(x, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    if x.size == 0:
        return x, 1

    if _minimize is None:
        lr = 0.05
        _, f0 = (0.0, None)
        try:
            f0 = float(fun(x, *args)[0])
        except Exception:
            f0 = 1e100
        best_f = f0
        for _ in range(int(maxiter)):
            try:
                val, grad = fun(x, *args)
                f = float(val)
                gr = np.asarray(grad, dtype=np.float64)
            except Exception:
                break
            if not np.isfinite(f):
                break
            gr = np.where(np.isfinite(gr), gr, 0.0)
            stp = lr * gr
            if np.any(stp):
                x = np.clip(x - stp, -_BOUND, _BOUND)
            if f < best_f - 1e-16:
                best_f = f
                lr = min(1.0, lr * 1.2)
            else:
                lr = max(1e-4, lr * 0.8)
        return x, 0

    if bounds is None:
        bounds = [(-_BOUND, _BOUND)] * int(x.size)

    options = {
        "maxiter": int(maxiter),
        "disp": 0,
        "maxfun": int(max(120, 4 * maxiter)),
        "ftol": 1e-16,
        "gtol": 1e-11,
    }

    try:
        res = _minimize(
            fun,
            x,
            args=args,
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options=options,
        )
    except Exception:
        return x, 99

    x_new = np.asarray(res.x, dtype=np.float64)
    x_new = np.where(np.isfinite(x_new), x_new, 0.0)
    x_new = np.clip(x_new, -_BOUND, _BOUND)

    try:
        status = int(res.status)
    except Exception:
        status = 0

    return x_new, status


def _lbfgs_stage(fun, args, x, bounds, end_time, best_a, best_score, best_x):
    """
    Run repeated short L-BFGS blocks on a fixed surrogate objective.
    Keep the best exact score seen, and revert the parameter vector if
    the surrogate run drifts badly.
    """
    if best_x is None:
        best_x = _ratios_from_a(best_a)

    while time.monotonic() < end_time:
        x, status = _run_minimize(fun, x, args, 35, bounds)
        a = _a_from_ratios(x)
        sc = _score_unit(a)

        if np.isfinite(sc) and sc < best_score - 1e-13:
            best_a = a.copy()
            best_score = float(sc)
            best_x = x.copy()
        elif np.isfinite(sc) and np.isfinite(best_score) and sc > best_score * 1.35:
            x = best_x.copy()

        if status == 1:
            break

    return x, best_a, best_score, best_x


def _shape_power_beta(n, alpha, off, beta=0.0, off1=0.05):
    """One-sided power decay with optional endpoint taper (1-t)^beta."""
    n = int(n)
    if n <= 1:
        return np.ones(n, dtype=np.float64)
    j = np.arange(n, dtype=np.float64)
    v = np.power(j + float(off), -float(alpha))
    if float(beta) != 0.0:
        t = j / float(n - 1)
        y = np.maximum(1.0 - t + float(off1), 1e-12)
        v = v * np.power(y, float(beta))
    return _normalize(v)


def _shape_exponential(n, alpha, off, gamma):
    n = int(n)
    if n <= 1:
        return np.ones(n, dtype=np.float64)
    j = np.arange(n, dtype=np.float64)
    t = j / float(n - 1)
    v = np.power(j + float(off), -float(alpha)) * np.exp(-float(gamma) * t)
    return _normalize(v)


def _shape_random_cos(n, seed, D=8, amp=0.65):
    n = int(n)
    if n <= 1:
        return np.ones(n, dtype=np.float64)
    rng = np.random.default_rng(seed)
    t = np.linspace(0.0, 1.0, n)
    base = np.ones(n, dtype=np.float64)
    for m in range(1, int(D) + 1):
        coef = (float(amp) / float(m * m)) * float(np.exp(rng.normal(0.0, 1.0)))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        base = base + coef * np.cos(np.pi * float(m) * t + phase)
    return _normalize(base)


def _generate_candidates(n):
    n = int(n)
    cands = []

    h = globals().get("height_sequence_1")
    base_h = None
    if h is not None and isinstance(h, (list, np.ndarray)):
        arr = np.asarray(h, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and float(np.sum(np.maximum(arr, 0.0))) > 0.0:
            base_h = _resample_sequence(arr, n)
            cands.append(base_h.copy())

            if base_h.size == n:
                j = np.arange(n, dtype=np.float64)
                for delta in (-0.12, -0.06, 0.0, 0.06, 0.12, 0.20):
                    v = base_h * np.power(j + 1.0, float(delta))
                    cands.append(_normalize(v))

                for sigma in (0.04, 0.08, 0.15, 0.25):
                    sgm = int(round(float(sigma) * 100))
                    rng = np.random.default_rng(9000 + sgm)
                    v = base_h * np.exp(rng.normal(0.0, float(sigma), size=n))
                    cands.append(_normalize(v))

    # Uniform, known baseline.
    cands.append(np.full(n, 1.0 / float(n), dtype=np.float64))

    # One-sided power and endpoint-tapered powers.
    alphas = (0.40, 0.45, 0.48, 0.50, 0.52, 0.55, 0.60, 0.70)
    offs = (0.30, 0.60, 1.00, 1.50, 2.50)
    betas = (0.0, 0.05, 0.12, 0.25, -0.08, -0.16)
    for alpha in alphas:
        for off in offs:
            for beta in betas:
                cands.append(_shape_power_beta(n, alpha, off, beta, 0.05))

    # Exponential/powers hybrids.
    for alpha in (0.45, 0.50, 0.55, 0.60):
        for gamma in (0.0, 0.20, 0.50, 1.00, 2.00):
            cands.append(_shape_exponential(n, alpha, 1.0, gamma))

    # Low-frequency smooth random shapes.
    for seed in range(12):
        cands.append(_shape_random_cos(n, seed, D=8, amp=0.65))

    # A few heavier endpoint-taper variants.
    for alpha in (0.48, 0.52):
        for beta in (-0.05, -0.10, 0.05, 0.10, 0.20):
            cands.append(_shape_power_beta(n, alpha, 0.75, beta, 0.10))

    return cands


def _sort_candidates(cands):
    out = []
    for i, a in enumerate(cands):
        try:
            sc = _score_unit(a)
        except Exception:
            sc = np.inf
        if np.isfinite(sc):
            out.append((float(sc), int(i), np.asarray(a, dtype=np.float64)))
    out.sort()
    return out


def _random_local_search(a0, end_time, seed):
    """Multiplicative block local search, accepting only true score gains."""
    a = _normalize(a0)
    best = a.copy()
    best_score = _score_unit(best)
    n = best.size
    if n <= 1 or not np.isfinite(best_score):
        return best, best_score

    rng = np.random.default_rng(seed)
    sigma = 0.08
    last_best = time.monotonic()
    start = last_best
    maxw = max(3, n // 8)

    while time.monotonic() < end_time:
        cand = best.copy()
        nblocks = 2 if rng.random() < 0.5 else 1

        for _ in range(nblocks):
            if maxw < 2:
                w = 1
            else:
                w = int(np.exp(rng.uniform(np.log(2.0), np.log(float(maxw)))))
            w = max(1, min(w, n))
            s = int(rng.integers(0, n - w + 1))
            std = sigma
            cand[s:s + w] *= np.exp(rng.normal(0.0, std, size=w))

        cand = _normalize(cand)
        sc = _score_unit(cand)

        if np.isfinite(sc) and sc < best_score - 1e-13:
            best = cand
            best_score = float(sc)
            last_best = time.monotonic()
            sigma = min(0.22, sigma * 1.06)
        else:
            elapsed = time.monotonic() - last_best
            if elapsed > 4.0:
                sigma = max(0.015, sigma * 0.78)
            if elapsed > 15.0 and time.monotonic() - start > 25.0:
                cand2 = _normalize(best * np.exp(rng.normal(0.0, 0.06, size=n)))
                sc2 = _score_unit(cand2)
                if np.isfinite(sc2) and sc2 < best_score:
                    best = cand2
                    best_score = float(sc2)
                    last_best = time.monotonic()
                    sigma = min(0.15, sigma * 1.2)

    return best, best_score


def _gradient_local_search(a0, end_time, seed):
    """
    Short first-order polish using active-peak gradient direction
    g_i = 2 sum_k p_k a_{k-i}, projected onto the simplex tangent direction.
    """
    a = _normalize(a0)
    best = a.copy()
    best_score = _score_unit(best)
    n = best.size
    if n <= 1 or not np.isfinite(best_score):
        return best, best_score
    if time.monotonic() + 6.0 >= end_time:
        return best, best_score

    rng = np.random.default_rng(seed)
    lr = 0.05
    gap = 0.20
    last_best = time.monotonic()

    while time.monotonic() < end_time:
        A, c = _c_A_from_a(best)
        M = float(np.max(c)) if c.size else 0.0
        if not np.isfinite(M) or M <= 1e-300:
            break

        temp = max(1e-300, M * gap)
        z = (c - M) / temp
        z = np.clip(z, -745.0, 0.0)
        p = np.exp(z)
        s = float(p.sum())
        if not np.isfinite(s) or s <= 0.0:
            p = np.full(2 * n - 1, 1.0 / float(2 * n - 1), dtype=np.float64)
        else:
            p = p / s

        L = _grad_fft_len(n)
        h = _hankel_matvec(p, best, A, L)
        grad = 2.0 * h
        move = grad - float(np.mean(grad))
        rr = float(np.dot(move, move))
        if not np.isfinite(rr) or rr <= 1e-30:
            break
        move = move / np.sqrt(rr)

        improved = False
        for alpha in (lr, lr * 0.5, lr * 1.8, lr * 0.25):
            cand = _normalize(best - alpha * move)
            sc = _score_unit(cand)
            if np.isfinite(sc) and sc < best_score - 1e-13:
                best = cand
                best_score = float(sc)
                last_best = time.monotonic()
                improved = True
                break

        if improved:
            lr = min(0.18, lr * 1.15)
        else:
            lr = max(0.004, lr * 0.78)
            elapsed = time.monotonic() - last_best
            if elapsed > 4.0:
                gap = max(0.02, gap * 0.75)
            if elapsed > 12.0 and rng.random() < 0.25:
                cand2 = _normalize(best * np.exp(rng.normal(0.0, 0.05, size=n)))
                sc2 = _score_unit(cand2)
                if np.isfinite(sc2) and sc2 < best_score:
                    best = cand2
                    best_score = float(sc2)
                    last_best = time.monotonic()

    return best, best_score


def _refine_start(a0, end_time, seed):
    """
    Refine one initial point by staged surrogates and local search.
    """
    a = _normalize(a0)
    best_a = a.copy()
    best_score = _score_unit(best_a)
    start = time.monotonic()
    total = end_time - start
    n = best_a.size

    if not np.isfinite(best_score) or total < 10.0:
        return best_a, best_score

    best_x = _ratios_from_a(best_a)
    bounds = [(-_BOUND, _BOUND)] * int(best_x.size)

    local_time = max(8.0, 0.12 * total)
    local_end = end_time - local_time
    if local_end < start + 3.0:
        local_end = end_time

    # Staged surrogates:
    # 1. L2 norm of convolution coefficients (global smoothing),
    # 2. L3 norm (more peak-sensitive),
    # 3. active-average objectives with decreasing active gaps.
    stages = [
        (_q_f_jac, (2.0,), 0.09),
        (_q_f_jac, (3.0,), 0.04),
        (_active_f_jac, (0.50,), 0.04),
        (_active_f_jac, (0.25,), 0.04),
        (_active_f_jac, (0.12,), 0.04),
        (_active_f_jac, (0.05,), 0.04),
    ]

    for fun, args, frac in stages:
        if time.monotonic() >= local_end:
            break
        stage_len = max(4.0, total * frac)
        stage_end = min(local_end, time.monotonic() + stage_len)
        x = best_x.copy()
        x, best_a, best_score, best_x = _lbfgs_stage(
            fun, args, x, bounds, stage_end, best_a, best_score, best_x
        )

    # Smooth maximum stage with decreasing temperature and occasional heat.
    cur_peak = max(1e-12, float(best_score) / (2.0 * float(n)))
    eps = max(1e-10, cur_peak * 0.8)
    eps_min = max(1e-13, cur_peak * 1e-6)
    chunk_count = 0
    heat_count = 0
    prev = float(best_score)

    while time.monotonic() < local_end:
        rem = local_end - time.monotonic()
        chunk = min(rem, max(8.0, 0.18 * rem))
        stage_end = time.monotonic() + chunk
        x = best_x.copy()

        x, best_a, best_score, best_x = _lbfgs_stage(
            _soft_f_jac, (eps,), x, bounds, stage_end, best_a, best_score, best_x
        )

        if np.isfinite(best_score) and best_score < prev - 1e-13:
            prev = float(best_score)
            chunk_count = 0
            cur_peak = max(1e-12, float(best_score) / (2.0 * float(n)))
            eps_min = max(1e-13, cur_peak * 1e-6)
            eps = max(eps_min, eps * 0.72)
        else:
            chunk_count += 1
            if chunk_count >= 2:
                sigma = 0.08 + 0.25 * min(1.0, eps / max(cur_peak, 1e-30))
                if eps <= eps_min * 1.2 and heat_count < 4:
                    eps = max(eps_min, cur_peak * 0.35)
                    sigma = 0.30
                    heat_count += 1
                rng = np.random.default_rng(int(seed) + chunk_count * 31 + heat_count * 17)
                if best_x.size:
                    best_x = np.clip(
                        best_x + rng.normal(0.0, sigma, size=best_x.size),
                        -_BOUND,
                        _BOUND,
                    )
                chunk_count = 0

            if eps > eps_min * 1.05:
                eps = max(eps_min, eps * 0.60)
            elif chunk_count >= 4:
                break

    if time.monotonic() < end_time:
        a2, sc2 = _random_local_search(best_a, end_time, int(seed) + 7919)
        if np.isfinite(sc2) and sc2 < best_score:
            best_a = a2
            best_score = float(sc2)

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Return the best non-negative sequence found before the time budget.
    """
    t0 = time.monotonic()
    budget = max(0.0, float(budget_s) - 30.0)

    n = _DEFAULT_N
    try:
        h = globals().get("height_sequence_1")
        if h is not None and isinstance(h, (list, np.ndarray)):
            arr = np.asarray(h, dtype=np.float64)
            if arr.ndim == 1 and 2500 <= arr.size <= 5000:
                n = int(arr.size)
    except Exception:
        pass

    if n < 32:
        n = _DEFAULT_N

    # Very quick fallback.
    if budget < 20.0:
        try:
            h = globals().get("height_sequence_1")
            if h is not None and isinstance(h, (list, np.ndarray)):
                arr = np.asarray(h, dtype=np.float64)
                if arr.ndim == 1 and arr.size > 0:
                    out = _normalize(arr)
                    return [float(x) for x in out]
        except Exception:
            pass
        a = _shape_power_beta(n, 0.5, 1.0, 0.0, 0.05)
        return [float(x) for x in a]

    deadline = t0 + budget

    cands = _generate_candidates(n)
    scored = _sort_candidates(cands)

    if not scored:
        a = _shape_power_beta(n, 0.5, 1.0, 0.0, 0.05)
        return [float(x) for x in a]

    global_best_a = np.asarray(scored[0][2], dtype=np.float64).copy()
    global_best_score = float(scored[0][0])

    try:
        print("Initial candidate best score: %.10f, length=%d, n_cands=%d"
              % (global_best_score, global_best_a.size, len(scored)), flush=True)
    except Exception:
        pass

    top = scored[:4]

    # Allocate most time to the best starting point, but keep some
    # independent runs to avoid a single local basin.
    allocations = [
        (0.58, 0),
        (0.22, 1),
        (0.10, 2),
        (0.05, 3),
    ]

    used = 0.0
    for frac, idx in allocations:
        if idx >= len(top):
            continue
        cur = t0 + used * budget
        alloc_end = min(deadline, t0 + (used + frac) * budget)
        if alloc_end - cur >= 6.0:
            a0 = np.asarray(top[idx][2], dtype=np.float64).copy()
            a1, sc1 = _refine_start(a0, alloc_end, int(seed) + 137 * (idx + 1))
            if np.isfinite(sc1) and sc1 < global_best_score:
                global_best_score = float(sc1)
                global_best_a = np.asarray(a1, dtype=np.float64).copy()
                try:
                    print("After start %d: %.10f at %.1fs"
                          % (idx, global_best_score, time.monotonic() - t0), flush=True)
                except Exception:
                    pass
        used += frac

    # Final independent polish on the global best.
    if time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        local_end = time.monotonic() + 0.65 * remaining
        if local_end > time.monotonic() + 3.0:
            a2, sc2 = _random_local_search(global_best_a, local_end, int(seed) + 10009)
            if np.isfinite(sc2) and sc2 < global_best_score:
                global_best_score = float(sc2)
                global_best_a = np.asarray(a2, dtype=np.float64).copy()

    if time.monotonic() < deadline:
        a3, sc3 = _gradient_local_search(global_best_a, deadline, int(seed) + 20021)
        if np.isfinite(sc3) and sc3 < global_best_score:
            global_best_score = float(sc3)
            global_best_a = np.asarray(a3, dtype=np.float64).copy()

    global_best_a = _normalize(global_best_a)
    out = [float(x) for x in global_best_a]

    # Final validation against the official evaluator.
    try:
        val = evaluate_sequence(out)
        if not np.isfinite(val):
            raise ValueError("invalid final sequence")
    except Exception:
        try:
            h = globals().get("height_sequence_1")
            if h is not None and isinstance(h, (list, np.ndarray)):
                arr = np.asarray(h, dtype=np.float64)
                if arr.ndim == 1 and arr.size > 0:
                    out = [float(x) for x in _normalize(arr)]
        except Exception:
            out = [float(x) for x in _shape_power_beta(len(out) if len(out) else _DEFAULT_N,
                                                       0.5, 1.0, 0.0, 0.05)]

    try:
        print("Final returned score (internal): %.10f, length=%d, t=%.1fs"
              % (global_best_score, len(out), time.monotonic() - t0), flush=True)
    except Exception:
        pass

    return out
```
