def fourier_shape_optimize(n, deadline, seed=0):
    """Optimize the log-magnitude of the FFT directly."""
    N = next_pow2(2 * n - 1)
    rng = np.random.default_rng(int(seed))
    
    # Start from the given sequence or a good initial guess
    a_given = get_sequence()
    if a_given is not None:
        a = normalize_positive(a_given, floor=1e-15)
        if a.size != n:
            x_old = np.linspace(0, 1, a.size)
            x_new = np.linspace(0, 1, n)
            a = normalize_positive(np.interp(x_new, x_old, a), floor=1e-15)
    else:
        a = np.ones(n, dtype=np.float64) / n
    
    # Work in log-magnitude domain of FFT
    fa = np.fft.rfft(a, N)
    mag = np.abs(fa)
    phase = np.angle(fa)
    
    # We want to modify the magnitude to flatten the inverse FFT of |F|^2
    # while keeping the phase (to maintain real-valued, non-negative a)
    
    best_a = a.copy()
    best_score = score_fn(a, N)
    
    eta = 0.05
    stagnant = 0
    it = 0
    
    while time.monotonic() < deadline:
        # Reconstruct a from current mag and phase
        F = mag * np.exp(1j * phase)
        a = np.fft.irfft(F, N)[:n]
        a = np.maximum(a, 0)
        s = float(a.sum())
        if s < 1e-15:
            break
        a = a / s
        
        c, _ = convolve_fft(a, N)
        cmax = float(np.max(c))
        score = float(2.0 * n * cmax)
        
        if np.isfinite(score) and score < best_score * (1 - 1e-10):
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        # Compute "pressure" on the frequency domain
        # The peak of c is at position k*, and we want to reduce it
        # This corresponds to reducing the "coherence" at that frequency
        k_star = int(np.argmax(c))
        # The contribution to c[k_star] comes from all frequency pairs (f1, f2) with f1+f2 ≈ k_star
        # Simplification: reduce the magnitude at frequencies that contribute most to c[k_star]
        
        # Alternative: use the gradient in the magnitude domain
        # dc[k]/dmag[f] is complex and hard to compute directly
        # Instead, use a simpler heuristic: reduce magnitudes that correlate with the peak
        
        # Compute the "sensitivity" of c[k_star] to each frequency bin
        # c = ifft(|F|^2), so c[k] = (1/N) sum_f |F[f]|^2 * exp(2πi*k*f/N)
        # dc[k_star]/d|F[f]| = (2/N) * |F[f]| * exp(2πi*k_star*f/N)
        # The real part of this tells us the direction to change |F[f]|
        
        f_idx = np.arange(len(mag))
        sensitivity = 2.0 * mag * np.cos(2.0 * np.pi * k_star * f_idx / N)
        sensitivity = sensitivity - sensitivity.mean()
        
        # Update magnitudes: reduce where sensitivity is high
        mag_change = eta * sensitivity / (np.abs(sensitivity).max() + 1e-10)
        mag = np.maximum(mag * (1.0 + mag_change), 0.01 * mag.max())
        
        # Renormalize to keep sum(a) = 1
        F = mag * np.exp(1j * phase)
        a_check = np.fft.irfft(F, N)[:n]
        a_check = np.maximum(a_check, 0)
        s_check = float(a_check.sum())
        if s_check > 1e-10:
            mag = mag * (1.0 / (s_check / float(a.sum())) ** 0.5)
        
        stagnant += 1
        if stagnant > 200:
            eta = max(0.005, eta * 0.7)
            if eta < 0.006:
                break
        it += 1
    
    return best_a, best_score

"""
Optimize non-negative step-function heights minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses adaptive-annealed mirror descent with structured FFT-mode perturbations,
multi-scale exploration, and bottleneck-targeting local search.

    import scipy.optimize as _scipy
except Exception:
    _scipy = None


def next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def normalize_positive(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, float(floor), None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def convolve_fft(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([]), np.array([])
    if N is None:
        N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return c, fa


def score_fn(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s <= 1e-15 or not np.isfinite(s):
        return np.inf
    a = a / s
    c, _ = convolve_fft(a, N)
    return float(2.0 * n * np.max(c))

    try:
        seq = globals().get("height_sequence_1", None)
        if seq is not None:
            a = np.asarray(seq, dtype=np.float64).ravel()
            if a.size > 10 and np.all(np.isfinite(a)):
                return normalize_positive(a, floor=0.0)
    except Exception:
        pass
    return None


def compute_gradient(a, c, cmax, tau, N, n):
    """Compute gradient of smoothed max objective w.r.t. a."""
    z = (c - cmax) / max(tau, 1e-12)
    z = np.clip(z, -50.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if s < 1e-300:
        return np.zeros(n, dtype=np.float64)
    w = e / s
    # g_i = sum_k w_k * a_{k-i} (cross-correlation)
    w_rev = w[::-1].copy()
    w_rev_padded = np.zeros(N, dtype=np.float64)
    w_rev_padded[:w_rev.size] = w_rev
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w_rev_padded, N)
    g = np.fft.irfft(fa * fw, N)[:n]
    return g


def mirror_descent_v2(a, deadline, seed=0, verbose=False):
    """Mirror descent with adaptive annealing and structured perturbations."""
    a = normalize_positive(a, floor=1e-15)
    n = a.size
    N = next_pow2(2 * n - 1)
    rng = np.random.default_rng(int(seed))

    best_a = a.copy()
    best_score = score_fn(a, N)

    tau = 0.25
    tau_min = 0.015
    tau_max = 0.6
    eta = 0.45

    stagnant = 0
    it = 0
    last_best_it = 0

    while time.monotonic() < deadline:
        c, _ = convolve_fft(a, N)
        cmax = float(np.max(c))
        score = float(2.0 * n * cmax)

        if np.isfinite(score) and score < best_score * (1.0 - 1e-9):
            best_score = score
            best_a = a.copy()
            stagnant = 0
            last_best_it = it
            eta = min(1.2, eta * 1.005)
        else:
            stagnant += 1

        # Compute gradient
        g = compute_gradient(a, c, cmax, tau, N, n)
        g = g - g.mean()
        g_scale = float(np.abs(g).max())
        if g_scale < 1e-14:
            break
        g = g / g_scale

        # Try multiple step sizes
        accepted = False
        for e in (eta, eta * 0.6, eta * 0.3):
            if e < 1e-6:
                continue
            loga = np.log(a + 1e-30)
            loga_new = loga - e * g
            loga_new -= float(np.max(loga_new))
            a_try = np.exp(np.clip(loga_new, -60.0, 0.0))
            a_try = normalize_positive(a_try, floor=1e-15)

            c_try, _ = convolve_fft(a_try, N)
            score_try = float(2.0 * n * np.max(c_try))

            if np.isfinite(score_try) and score_try < score * (1.0 - 1e-8):
                a = a_try
                accepted = True
                break
            # Smooth criterion
            tau_eff = max(tau, 1e-12)
            z_try = (c_try - float(np.max(c_try))) / tau_eff
            z_try = np.clip(z_try, -50.0, 0.0)
            smooth_try = float(np.max(c_try)) + tau_eff * float(np.log(np.exp(z_try).sum() + 1e-300))
            z_old = (c - cmax) / tau_eff
            z_old = np.clip(z_old, -50.0, 0.0)
            smooth_old = cmax + tau_eff * float(np.log(np.exp(z_old).sum() + 1e-300))
            if np.isfinite(smooth_try) and smooth_try < smooth_old * (1 - 5e-5):
                a = a_try
                accepted = True
                break

        if accepted:
            eta = min(1.5, eta * 1.01)
        else:
            eta = max(0.02, eta * 0.75)

        # Adaptive tau
        if stagnant < 30:
            tau = max(tau_min, tau * 0.998)
        elif stagnant < 100:
            tau = max(tau_min, tau * 0.995)
        else:
            # Reheat
            tau = min(tau_max, tau * 1.8)
            eta = min(1.0, eta * 1.4)
            # Structured perturbation using low-frequency FFT modes
            if stagnant > 300:
                k_perturb = int(rng.integers(1, 8))
                mode = np.sin(2.0 * np.pi * k_perturb * np.arange(n) / n)
                mode = mode - mode.mean()
                pert_strength = 0.08 * (stagnant / 500.0)
                pert_strength = min(pert_strength, 0.25)
                a = normalize_positive(a * (1.0 + pert_strength * mode / (np.abs(mode).max() + 1e-10)), floor=1e-15)
                stagnant = 0
            elif stagnant > 150:
                # Smaller perturbation
                if rng.random() < 0.15:
                    a = normalize_positive(a * np.exp(rng.normal(0, 0.05, n)), floor=1e-15)

        it += 1
        if verbose and it % 1000 == 0:
            print(f"    MD2 n={n} it={it} score={score:.8f} best={best_score:.8f} tau={tau:.4f} eta={eta:.4f} stag={stagnant}", flush=True)

        if best_score < 1.500:
            break

    return best_a, best_score


def bottleneck_targeting(a, deadline, seed=0):
    """Target the specific bottleneck convolution coefficients."""
    a = normalize_positive(a, floor=1e-15)
    n = a.size
    N = next_pow2(2 * n - 1)
    rng = np.random.default_rng(int(seed))

    best_a = a.copy()
    best_score = score_fn(a, N)
    it = 0

    while time.monotonic() < deadline:
        c, _ = convolve_fft(a, N)
        cmax = float(np.max(c))
        score = float(2.0 * n * cmax)

        if np.isfinite(score) and score < best_score * (1 - 1e-9):
            best_score = score
            best_a = a.copy()

        # Find top coefficients
        top_k = max(2, min(10, int(n * 0.005)))
        top_indices = np.argsort(c)[-top_k:]
        top_avg = c[top_indices].mean()

        # Compute "excess pressure" for each coefficient
        excess = np.zeros(c.size)
        excess[top_indices] = c[top_indices] - top_avg

        # Gradient targeting only the excess
        w = np.maximum(excess, 0)
        w_sum = float(w.sum())
        if w_sum < 1e-15:
            # Uniform pressure
            w = np.ones(c.size) / c.size
        else:
            w = w / w_sum

        # Compute gradient
        w_rev = w[::-1].copy()
        w_padded = np.zeros(N)
        w_padded[:w_rev.size] = w_rev
        fa = np.fft.rfft(a, N)
        fw = np.fft.rfft(w_padded, N)
        g = np.fft.irfft(fa * fw, N)[:n]
        g = g - g.mean()
        gs = float(np.abs(g).max())
        if gs < 1e-14:
            break
        g = g / gs

        # Multiplicative update
        loga = np.log(a + 1e-30)
        step = 0.2
        loga_new = loga - step * g
        loga_new -= float(np.max(loga_new))
        a_try = np.exp(np.clip(loga_new, -60.0, 0.0))
        a_try = normalize_positive(a_try, floor=1e-15)

        c_try, _ = convolve_fft(a_try, N)
        score_try = float(2.0 * n * np.max(c_try))

        if np.isfinite(score_try) and score_try < score:
            a = a_try
        else:
            # Try with random perturbation added
            a_try2 = normalize_positive(a_try * np.exp(rng.normal(0, 0.03, n)), floor=1e-15)
            score_try2 = score_fn(a_try2, N)
            if np.isfinite(score_try2) and score_try2 < score:
                a = a_try2
            else:
                # Small random move
                idx = rng.integers(0, n)
                a[idx] *= 1.5
                a = normalize_positive(a, floor=1e-15)

        it += 1
        if it % 2000 == 0:
            print(f"    bottleneck n={n} it={it} best={best_score:.8f}", flush=True)

    return best_a, best_score


def cosine_mode_polish(a, deadline, seed=0, max_modes=12):
    """Polish by adjusting low-frequency cosine modes."""
    a = normalize_positive(a, floor=1e-15)
    n = a.size
    N = next_pow2(2 * n - 1)
    rng = np.random.default_rng(int(seed))

    best_a = a.copy()
    best_score = score_fn(a, N)

    # Build mode matrix
    x = np.arange(n, dtype=np.float64)
    modes = []
    for m in range(1, max_modes + 1):
        modes.append(np.sin(2.0 * np.pi * m * x / n))
        modes.append(np.cos(2.0 * np.pi * m * x / n))
    B = np.column_stack(modes)
    B -= B.mean(axis=0, keepdims=True)
    norms = np.linalg.norm(B, axis=0)
    norms[norms < 1e-10] = 1.0
    B = B / norms

    D = B.shape[1]
    # Project current a onto the mode subspace
    loga = np.log(a + 1e-30)
    loga_centered = loga - loga.mean()
    theta = B.T @ loga_centered

    for m in range(D):
        for direction in (1.0, -1.0):
            for step in (0.1, 0.05, 0.02, 0.01):
                if time.monotonic() > deadline:
                    break
                theta[m] += direction * step
                loga_try = loga_centered + B @ theta
                a_try = np.exp(loga_try - loga_try.max())
                a_try = normalize_positive(a_try, floor=1e-15)
                sc = score_fn(a_try, N)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_try.copy()
                    loga_centered = loga_centered + B @ (np.eye(D)[m] * direction * step)
                    break
                else:
                    theta[m] -= direction * step

    return best_a, best_score


def generate_structured_inits(n, seed):
    """Generate various structured initial sequences."""
    rng = np.random.default_rng(int(seed))
    x = np.linspace(0, 1, n)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine window
    inits.append(normalize_positive(0.5 + 0.5 * np.cos(2 * np.pi * x), floor=1e-12))

    # Sinusoidal
    for freq in (1, 2, 3, 5):
        inits.append(normalize_positive(1.0 + 0.3 * np.sin(2 * np.pi * freq * x), floor=1e-12))

    # Bell-shaped
    for mu in (0.3, 0.5, 0.7):
        for sig in (0.1, 0.2, 0.3):
            inits.append(normalize_positive(np.exp(-((x - mu) ** 2) / (2 * sig ** 2)), floor=1e-12))

    # Two-peaked
    inits.append(normalize_positive(np.exp(-((x - 0.3) ** 2) / 0.02) + np.exp(-((x - 0.7) ** 2) / 0.02), floor=1e-12))
    inits.append(normalize_positive(np.exp(-((x - 0.2) ** 2) / 0.05) + np.exp(-((x - 0.8) ** 2) / 0.05), floor=1e-12))

    # Random smooth
    coeffs = rng.normal(0, 1, size=8) / np.arange(1, 9)
    vals = np.ones(n)
    for m, c in enumerate(coeffs, 1):
        vals += c * np.sin(2 * np.pi * m * x + rng.uniform(0, 2 * np.pi))
    inits.append(normalize_positive(vals, floor=1e-12))

    # Quadratic
    inits.append(normalize_positive((x - 0.5) ** 2 + 0.1, floor=1e-12))
    inits.append(normalize_positive(1.0 - 4.0 * (x - 0.5) ** 2 + 0.1, floor=1e-12))

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main search function. Returns best sequence found within time budget."""
    start = time.monotonic()
    deadline = start + max(100.0, float(budget_s) - 25.0)

    # Get starting point
    a_given = get_given_sequence()
    if a_given is not None:
        n_base = a_given.size
        a = a_given.copy()
        best_score = score_fn(a)
        print(f"Given sequence: n={n_base}, score={best_score:.9f}", flush=True)
    else:
        n_base = 2048
        a = np.ones(n_base) / n_base
        best_score = 2.0
        print(f"No given sequence; uniform n={n_base}, score=2.0", flush=True)

    best_a = a.copy()

    # Phase 0: Evaluate structured inits at base size
    t_phase0_end = start + 30
    if best_score > 1.55:
        inits = generate_structured_inits(n_base, seed)
        for a_init in inits:
            sc = score_fn(a_init)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_init.copy()
                print(f"  structured init: score={sc:.9f}", flush=True)

    # Phase 1: Mirror descent with annealing at base size (main workhorse)
    t1_end = min(deadline - 300, start + (deadline - start) * 0.40)
    print(f"Phase 1: Mirror descent anneal, n={n_base}, budget={t1_end - start:.0f}s", flush=True)
    a1, s1 = mirror_descent_v2(best_a, t1_end, seed=seed, verbose=True)
    if s1 < best_score:
        best_score = s1
        best_a = a1.copy()
        print(f"  Phase 1 result: score={best_score:.9f}", flush=True)

    # Phase 2: Try larger size
    t2_start = time.monotonic()
    t2_end = min(deadline - 200, t2_start + (deadline - t2_start) * 0.30)
    n_large = min(8192, n_base * 2)
    if n_large >= 1024 and t2_end - t2_start > 30:
        print(f"Phase 2: Mirror descent at n={n_large}, budget={t2_end - t2_start:.0f}s", flush=True)
        a_large = resample(best_a, n_large)
        a2, s2 = mirror_descent_v2(a_large, t2_end, seed=seed + 500, verbose=True)
        if s2 < best_score:
            best_score = s2
            best_a = a2.copy()
            print(f"  Phase 2 result: score={best_score:.9f}", flush=True)

    # Phase 3: Bottleneck targeting
    t3_start = time.monotonic()
    t3_end = min(deadline - 100, t3_start + 150)
    if t3_end - t3_start > 30:
        # Use the size that gave best score
        n_for_wf = best_a.size
        print(f"Phase 3: Bottleneck targeting, n={n_for_wf}, budget={t3_end - t3_start:.0f}s", flush=True)
        a3, s3 = bottleneck_targeting(best_a, t3_end, seed=seed + 1000)
        if s3 < best_score:
            best_score = s3
            best_a = a3.copy()
            print(f"  Phase 3 result: score={best_score:.9f}", flush=True)

    # Phase 4: Final mirror descent polish
    t4_end = deadline - 10
    if time.monotonic() < t4_end - 20:
        print(f"Phase 4: Final polish, n={best_a.size}, budget={t4_end - time.monotonic():.0f}s", flush=True)
        a4, s4 = mirror_descent_v2(best_a, t4_end, seed=seed + 2000, verbose=False)
        if s4 < best_score:
            best_score = s4
            best_a = a4.copy()
            print(f"  Phase 4 result: score={best_score:.9f}", flush=True)

    # Final: ensure output is valid
    best_a = normalize_positive(best_a, floor=0.0)
    # Round extremely small values to 0 to avoid issues
    best_a[best_a < 1e-12] = 0.0
    best_a = normalize_positive(best_a, floor=0.0)
    out_list = [float(x) for x in best_a]

    # Validate
    try:
        final_eval = float(evaluate_sequence(out_list))
    except Exception:
        final_eval = float('inf')

    if not np.isfinite(final_eval) or final_eval > 2.0:
        # Fallback to uniform
        out_list = [1.0 / 2048.0] * 2048
        final_eval = 2.0

    print(f"FINAL: internal={best_score:.10f}, evaluated={final_eval:.10f}, n={len(out_list)}", flush=True)
    print(f"Total time: {time.monotonic() - start:.1f}s", flush=True)
    return out_list
