import numpy as np
import time
import math
from scipy.optimize import linprog

def _fft_convolve(a):
    """Fast full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]

def _score(a):
    """Compute J(a) = 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)

def _build_conv_matrix(f):
    """Build the convolution matrix C where (C @ g)[k] = (f*g)[k]."""
    n = len(f)
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = f
    return C

def _lp_solve(f, M, n_constraints=None):
    """Solve max sum(g) s.t. C_f @ g <= M, g >= 0."""
    n = len(f)
    C = _build_conv_matrix(f)
    if n_constraints is not None and n_constraints < C.shape[0]:
        # Only keep the most active constraints
        # We'll handle this differently - just use all constraints
        pass
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 30})

def _fft_convolve(a):
    """Fast full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]


def _score(a):
    """Compute J(a) = 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)


def _build_conv_matrix(f):
    """Build the convolution matrix C where (C @ g)[k] = (f*g)[k]."""
    n = len(f)
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = f
    return C


def _lp_solve(f, M, time_limit=20):
    """Solve max sum(g) s.t. C_f @ g <= M, g >= 0. Returns g or None."""
    n = len(f)
    C = _build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': time_limit})
        if result.status == 0:
            return result.x
    except Exception:
        pass
    return None

    out = np.maximum(out, 0.0)
    s = float(np.sum(out))
    if s > 1e-10:
        out = out / s
    return out

    phase2_deadline = min(total_deadline, time.monotonic() + 400.0)
    inits_2 = [_interpolate_sequence(best_a, n2)] if best_a is not None else []
    for v in [2, 5, 10, 12, 0, 7]:
        inits_2.append(_make_initial(n2, v, rng))

    for idx, a_init in enumerate(inits_2):
        if time.monotonic() > phase2_deadline:
            break
        label = "from_best" if idx == 0 else f"v{idx}"
        a_opt, score = _alternating_lp(a_init, 400, phase2_deadline,
                                       initial_blend=0.75, perturb_every=70,
                                       perturb_mag=0.003)
        print(f"  init={label}: score={score:.6f}")
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  ** New best: {best_score:.6f} (n={n2})")

    print(f"After Phase 2: best={best_score:.6f}")

    # ===================== PHASE 3: LP at n=300, 400 =====================
    print("\n=== Phase 3: LP at intermediate lengths ===")
    phase3_deadline = min(total_deadline, time.monotonic() + 300.0)
    for n3 in [300, 400]:
        if time.monotonic() > phase3_deadline:
            break
        a_init = _interpolate_sequence(best_a, n3)
        a_opt, score = _alternating_lp(a_init, 300, phase3_deadline,
                                       initial_blend=0.7, perturb_every=60,
                                       perturb_mag=0.004)
        print(f"  n={n3}: score={score:.6f}")
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  ** New best: {best_score:.6f} (n={n3})")

    # ===================== PHASE 4: LP at n=1000 =====================
    print("\n=== Phase 4: LP at n=1000 ===")
    n4 = 1000
    phase4_deadline = min(total_deadline, time.monotonic() + 350.0)
    a_init_4 = _interpolate_sequence(best_a, n4)
    a_opt, score = _alternating_lp(a_init_4, 300, phase4_deadline,
                                   initial_blend=0.6, perturb_every=80,
                                   perturb_mag=0.003, use_reduced=True)
    print(f"  n={n4}: score={score:.6f}")
    if score < best_score:
        best_score = score
        best_a = a_opt.copy()
        print(f"  ** New best: {best_score:.6f} (n={n4})")

    # Also try reduced LP at n=500 with more iterations
    if time.monotonic() < total_deadline - 200:
        print("\n  Extra LP at n=500 (reduced constraints, more iters):")
        n5 = 500
        a_init_5 = _interpolate_sequence(best_a, n5)
        extra_deadline = min(total_deadline, time.monotonic() + 200.0)
        a_opt, score = _alternating_lp(a_init_5, 600, extra_deadline,
                                       initial_blend=0.5, perturb_every=50,
                                       perturb_mag=0.005, use_reduced=True)
        print(f"  n={n5} (reduced): score={score:.6f}")
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  ** New best: {best_score:.6f} (n={n5})")

    # ===================== PHASE 5: Subgradient refinement =====================
    print("\n=== Phase 5: Subgradient refinement ===")
    n_best = len(best_a)
    n_sg = min(n_best, 1000)
    a_sg_init = _interpolate_sequence(best_a, n_sg)
    phase5_deadline = min(total_deadline, time.monotonic() + 150.0)
    a_opt, score = _subgradient_vec(a_sg_init, 5000, lr_init=0.3,
                                    lr_decay=0.99995, n_peaks=10,
                                    deadline=phase5_deadline, stagnation_limit=120)
    if n_best != n_sg:
        a_opt_up = _interpolate_sequence(a_opt, n_best)
        score = _score(a_opt_up)
    else:
        a_opt_up = a_opt
    print(f"  Subgradient (n={n_sg}): score={score:.6f}")
    if score < best_score:
        best_score = score
        best_a = a_opt_up.copy()
        print(f"  ** New best: {best_score:.6f}")

    # ===================== PHASE 6: Peak shaving polish =====================
    print("\n=== Phase 6: Peak shaving polish ===")
    n_polish = min(len(best_a), 500)
    a_polish = _interpolate_sequence(best_a, n_polish)
    phase6_deadline = min(total_deadline, time.monotonic() + 100.0)
    a_opt, score = _peak_shave(a_polish, 500, phase6_deadline)
    if len(best_a) != n_polish:
        a_opt = _interpolate_sequence(a_opt, len(best_a))
        score = _score(a_opt)
    print(f"  Peak shave (n={n_polish}): score={score:.6f}")
    if score < best_score:
        best_score = score
        best_a = a_opt.copy()
        print(f"  ** New best: {best_score:.6f}")

    # ===================== PHASE 7: Final LP polish on best =====================
    print("\n=== Phase 7: Final LP polish ===")
    n_final = len(best_a)
    n_lp = min(n_final, 500)
    a_final_init = _interpolate_sequence(best_a, n_lp)
    phase7_deadline = total_deadline - 20.0
    if time.monotonic() < phase7_deadline:
        a_opt, score = _alternating_lp(a_final_init, 300, phase7_deadline,
                                       initial_blend=0.3, perturb_every=40,
                                       perturb_mag=0.002)
        if n_final != n_lp:
            a_opt = _interpolate_sequence(a_opt, n_final)
            score = _score(a_opt)
        print(f"  Final LP (n={n_lp}): score={score:.6f}")
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  ** New best: {best_score:.6f}")

    # Final verification
    print(f"\n=== FINAL: score={best_score:.6f}, n={len(best_a)} ===")
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    verified = float(evaluate_sequence(result))
    print(f"Verified by evaluate_sequence: {verified:.6f}")
    if verified != float('inf') and verified < best_score:
        best_score = verified
    return result
