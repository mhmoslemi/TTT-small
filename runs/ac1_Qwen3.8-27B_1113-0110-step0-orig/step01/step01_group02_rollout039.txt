from scipy.optimize import linprog

def lp_step(a, M):
    """Solve the LP to find g that maximizes sum(g) subject to a*g <= M and g >= 0."""
    n = len(a)
    # Variables: g[0], ..., g[n-1]
    # Objective: maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: for each k = 0, ..., 2n-2
    # sum_j a[j] * g[k-j] <= M
    # where j ranges over valid indices
    
    # Build constraint matrix A_ub (shape: (2n-1, n))
    # Row k: A_ub[k, i] = a[k-i] if 0 <= k-i <= n-1, else 0
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A_ub[k, i] = a[j]
    
    b_ub = np.full(2*n-1, M)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return res.x

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csc_matrix


"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using LP-based acceleration combined with
subgradient polishing. The LP step finds the optimal direction that maximizes sum(g)
subject to cross-correlation bounds, enabling larger and more informed steps than
pure subgradient methods.
"""


def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]


def _cross_convolve(a, g):
    n1, n2 = len(a), len(g)
    size = n1 + n2 - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    fg = np.fft.rfft(g, fft_size)
    cg = np.fft.irfft(fa * fg, fft_size)
    return cg[:size]


def _build_lp_matrix(a):
    n = len(a)
    total_nnz = sum(min(n, k + 1, 2 * n - 1 - k) for k in range(2 * n - 1))
    rows = np.empty(total_nnz, dtype=np.int32)
    cols = np.empty(total_nnz, dtype=np.int32)
    vals = np.empty(total_nnz, dtype=np.float64)
    ptr = 0
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        count = i_max - i_min + 1
        idx = np.arange(i_min, i_max + 1)
        rows[ptr:ptr + count] = k
        cols[ptr:ptr + count] = idx
        vals[ptr:ptr + count] = a[k - idx]
        ptr += count
    return csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _solve_lp(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A_ub = _build_lp_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def _subgradient_vec(a, k_max):
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k_max - n + 1)
    hi = min(n - 1, k_max)
    if lo <= hi:
        j_range = np.arange(lo, hi + 1)
        indices = k_max - j_range
        g[lo:hi + 1] = 2.0 * a[indices]
    return g

    eta = 0.12

            no_improve = 0
