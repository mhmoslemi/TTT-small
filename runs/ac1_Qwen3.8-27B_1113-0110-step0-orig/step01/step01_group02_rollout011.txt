import numpy as np
import time

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses Adam optimizer with log-sum-exp (soft-max) subgradient for smooth
gradient information across all convolution entries. Adaptive temperature,
multiple restarts, and structured perturbations for escaping local minima.
"""

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]

def _score_from_c(c, n):
    return 2.0 * n * float(np.max(c))

def _softgrad(a, c, T):
    n = len(a)
    c_shift = (c - c.max()) / T
    w = np.exp(c_shift)
    ws = w.sum()
    if ws < 1e-300:
        w = np.full(len(c), 1.0 / len(c))
    else:
        w = w / ws
    size = 2 * n
    Fw = np.fft.rfft(w, size)
    Fa = np.fft.rfft(a, size)
    cc = np.fft.irfft(Fw * np.conj(Fa), size)[:n]
    return (2.0 / T) * cc

def _optimize_adam(a0, time_budget, rng, verbose=True):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), np.inf
    a = a / s
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.005
    
    c = _fft_convolve(a)
    T = max(1e-4, (c.max() - c.mean()) * 0.5)
    
    best_score = _score_from_c(c, n)
    best_a = a.copy()
    
    deadline = time.monotonic() + time_budget
    no_improve = 0
    last_print = time.monotonic()
    
    while time.monotonic() < deadline:
        c = _fft_convolve(a)
        score = _score_from_c(c, n)
        
        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            if verbose and time.monotonic() - last_print > 15:
                print(f"    [adam] score={score:.8f}, T={T:.6f}, lr={lr:.6f}")
                last_print = time.monotonic()
        else:
            no_improve += 1
        
        # Adaptive temperature
        T = max(5e-5, T * 0.9995)
        
        g = _softgrad(a, c, T)
        
        t += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            t += 1
            continue
        a = a / s
        
        # Adaptive LR
        if no_improve > 1000:
            lr *= 0.9
        elif no_improve > 500:
            lr *= 0.97
        elif no_improve > 100:
            lr *= 0.99
        elif no_improve == 0 and lr < 0.02:
            lr = min(lr * 1.005, 0.02)
        
        # Structured restart
        if no_improve > 3000:
            a = best_a.copy()
            c_best = _fft_convolve(best_a)
            # Reduce values at positions contributing to top peaks
            k_max = int(np.argmax(c_best))
            # Find j values that contribute most to c[k_max]
            lo = max(0, k_max - n + 1)
            hi = min(n - 1, k_max)
            contributors = np.arange(lo, hi + 1)
            contrib_vals = best_a[k_max - contributors]
            # Reduce top contributors
            top_idx = contributors[np.argsort(contrib_vals)[-min(20, len(contributors)):]]
            a[top_idx] *= 0.9
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve = 0
            lr = 0.005
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            T = max(1e-4, (c_best.max() - c_best.mean()) * 0.5)
        
        # Occasional random perturbation
        if no_improve > 1500 and rng.random() < 0.05:
            a = best_a * np.exp(rng.normal(0, 0.05, n))
            a = np.maximum(a, 1e-20)
            a = a / a.sum()
            no_improve = 0
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr = 0.003
    
    return best_a, best_score

import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Combines Adam optimizer with log-sum-exp soft gradient, LP-based iterative
method (maximize next-step sum subject to convolution bound), and multiplicative
subgradient descent. Uses adaptive temperature, structured restarts, and
multiple initialization strategies to escape local minima.
"""


def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]


def _score_from_c(c, n):
    return 2.0 * n * float(np.max(c))


def _softgrad(a, c, T):
    """Gradient of log-sum-exp(c/T) w.r.t. a via FFT cross-correlation."""
    n = len(a)
    c_shift = (c - c.max()) / T
    w = np.exp(c_shift)
    ws = w.sum()
    if ws < 1e-300:
        w = np.full(len(c), 1.0 / len(c))
    else:
        w = w / ws
    size = 2 * n
    Fw = np.fft.rfft(w, size)
    Fa = np.fft.rfft(a, size)
    cc = np.fft.irfft(Fw * np.conj(Fa), size)[:n]
    return (2.0 / T) * cc


def _lp_solve(a, M):
    """Solve max sum(b) s.t. b>=0, (a*b)[k]<=M for all k. Returns b>=0."""
    n = len(a)
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for m in range(lo, hi + 1):
            coeff = a[k - m]
            if coeff > 1e-15:
                rows.append(k)
                cols.append(m)
                data.append(coeff)
    A_ub = sparse.coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    b_ub = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def _optimize_adam(a0, time_budget, rng, verbose=True):
    """Adam optimizer with log-sum-exp soft gradient."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), np.inf
    a = a / s

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.008

    c = _fft_convolve(a)
    T = max(1e-4, (c.max() - c.mean()) * 0.4)
    best_score = _score_from_c(c, n)
    best_a = a.copy()

    deadline = time.monotonic() + time_budget
    no_improve = 0
    last_print = time.monotonic()
    it = 0

    while time.monotonic() < deadline:
        c = _fft_convolve(a)
        score = _score_from_c(c, n)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            if verbose and time.monotonic() - last_print > 12:
                print(f"    [adam] it={it}, score={score:.8f}, T={T:.2e}, lr={lr:.5f}")
                last_print = time.monotonic()
        else:
            no_improve += 1

        T = max(5e-5, T * 0.9997)

        g = _softgrad(a, c, T)
        t += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            it += 1
            continue
        a = a / s

        if no_improve > 2000:
            lr *= 0.9
        elif no_improve > 1000:
            lr *= 0.97
        elif no_improve > 500:
            lr *= 0.99
        elif no_improve == 0 and lr < 0.015:
            lr = min(lr * 1.003, 0.015)

        if no_improve > 5000:
            a = best_a.copy()
            c_best = _fft_convolve(best_a)
            top_ks = np.argsort(c_best)[-min(30, n)//2:]
            for k in top_ks:
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                idx = np.arange(lo, hi + 1)
                vals = best_a[k - idx]
                top_j = idx[np.argsort(vals)[-5:]]
                a[top_j] *= 0.92
            a = np.maximum(a, 1e-20)
            a = a / a.sum()
            no_improve = 0
            lr = 0.005
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            T = max(1e-4, (c_best.max() - c_best.mean()) * 0.4)

        if no_improve > 3000 and rng.random() < 0.03:
            a = best_a * np.exp(rng.normal(0, 0.03, n))
            a = np.maximum(a, 1e-20)
            a = a / a.sum()
            no_improve = 0
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr = 0.004

        it += 1

    return best_a, best_score


def _optimize_lp(a0, time_budget, rng, verbose=True):
    """LP-based iterative method: maximize next-step subject to convolution bound."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), np.inf
    a = a / s

    c = _fft_convolve(a)
    M = float(np.max(c))
    best_score = 2.0 * n * M
    best_a = a.copy()

    deadline = time.monotonic() + time_budget
    t_param = 0.15
    it = 0
    last_print = time.monotonic()
    no_improve = 0

    while time.monotonic() < deadline:
        b = _lp_solve(a, M)
        if b is None:
            break
        sum_b = float(b.sum())
        if sum_b < 1e-10:
            # LP can't improve; perturb and retry
            a = best_a * np.exp(rng.normal(0, 0.02, n))
            a = np.maximum(a, 1e-20)
            a = a / a.sum()
            c = _fft_convolve(a)
            M = float(np.max(c))
            t_param = 0.15
            no_improve = 0
            it += 1
            continue

        a_new = (a + t_param * b) / (1.0 + t_param * sum_b)
        a_new = np.maximum(a_new, 0.0)
        c_new = _fft_convolve(a_new)
        M_new = float(np.max(c_new))
        score_new = 2.0 * n * M_new

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            M = M_new
            no_improve = 0
            if verbose and time.monotonic() - last_print > 20:
                print(f"    [lp] it={it}, score={score_new:.8f}, sum_b={sum_b:.4f}")
                last_print = time.monotonic()
        else:
            no_improve += 1
            t_param *= 0.85
            if t_param < 1e-3:
                a = best_a.copy()
                c = _fft_convolve(a)
                M = float(np.max(c))
                t_param = 0.15
                no_improve = 0

        it += 1

    return best_a, best_score


def _optimize_mult(a0, time_budget, rng, verbose=True):
    """Multiplicative subgradient with soft-max."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), np.inf
    a = a / s

    c = _fft_convolve(a)
    T = max(1e-4, (c.max() - c.mean()) * 0.4)
    best_score = _score_from_c(c, n)
    best_a = a.copy()

    eta = 0.04
    deadline = time.monotonic() + time_budget
    no_improve = 0
    it = 0
    last_print = time.monotonic()

    while time.monotonic() < deadline:
        c = _fft_convolve(a)
        score = _score_from_c(c, n)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            if verbose and time.monotonic() - last_print > 15:
                print(f"    [mult] it={it}, score={score:.8f}")
                last_print = time.monotonic()
        else:
            no_improve += 1

        T = max(5e-5, T * 0.999)
        g = _softgrad(a, c, T)
        g_max = g.max()
        if g_max < 1e-15:
            break
        g_n = g / g_max

        a = a * np.exp(-eta * g_n)
        a = np.maximum(a, 1e-20)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            it += 1
            continue
        a = a / s

        if no_improve > 800:
            eta *= 0.94
        elif no_improve == 0:
            eta = min(eta * 1.008, 0.12)

        if no_improve > 3000:
            a = best_a * np.exp(rng.normal(0, 0.04, n))
            a = np.maximum(a, 1e-20)
            a = a / a.sum()
            no_improve = 0
            eta = 0.04
            T = max(1e-4, (c.max() - c.mean()) * 0.4)

        it += 1

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Minimizes 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
    Uses Adam+softmax, LP-based iterative, and multiplicative subgradient
    optimizers with multiple initializations and structured restarts.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 25.0

    best_overall = None
    best_overall_score = np.inf

    # Build initializations
    candidates = []

    # Provided sequence
    try:
        hs = np.array(height_sequence_1, dtype=np.float64)
        hs = np.clip(hs, 0, None)
        if hs.size > 100 and hs.sum() > 0.01:
            candidates.append(("provided", hs.copy()))
    except Exception:
        pass

    # Sinusoidal variants (the best family from previous run)
    n_main = 2000
    x = np.arange(n_main, dtype=np.float64)
    for amp in [0.35, 0.4, 0.45, 0.5]:
        for phase in [np.pi / 4, np.pi / 3, np.pi / 2, 2 * np.pi / 3, 3 * np.pi / 4]:
            a = 1.0 + amp * np.sin(2 * np.pi * x / n_main + phase)
            a = np.maximum(a, 0.01)
            candidates.append((f"sin_{amp}_{phase:.3f}", a))

    # Cosine
    for amp in [0.4, 0.5]:
        a = 1.0 + amp * np.cos(2 * np.pi * x / n_main)
        a = np.maximum(a, 0.01)
        candidates.append((f"cos_{amp}", a))

    # Two-frequency
    a = 1.0 + 0.3 * np.sin(2 * np.pi * x / n_main) + 0.15 * np.sin(4 * np.pi * x / n_main + 0.5)
    a = np.maximum(a, 0.01)
    candidates.append(("two_freq", a))

    # Decay
    a = np.exp(-0.7 * x / n_main) + 0.1
    candidates.append(("decay", a))

    # Asymmetric ramp
    a = np.linspace(0.5, 1.5, n_main)
    candidates.append(("ramp", a))

    # Try n=2500
    x2500 = np.arange(2500, dtype=np.float64)
    a = 1.0 + 0.4 * np.sin(2 * np.pi * x2500 / 2500 + np.pi / 4)
    a = np.maximum(a, 0.01)
    candidates.append(("sin_2500", a))

    # Rank by initial score
    scored = []
    for name, a0 in candidates:
        a0 = np.maximum(a0, 0)
        s = a0.sum()
        if s < 1e-15:
            continue
        a0 = a0 / s
        c0 = _fft_convolve(a0)
        sc = _score_from_c(c0, len(a0))
        scored.append((sc, name, a0))

    scored.sort(key=lambda x: x[0])
    print(f"Total candidates: {len(scored)}")
    print("Top 8 initial scores:")
    for i in range(min(8, len(scored))):
        print(f"  {i+1}. {scored[i][1]}: {scored[i][0]:.6f} (n={len(scored[i][2])})")

    # Allocate time: top candidate gets most
    n_top = min(5, len(scored))
    remaining = deadline - time.monotonic()

    # Time allocation: 50% top, 20% second, 10% third, 10% fourth, 10% fifth
    if n_top >= 5:
        fracs = [0.48, 0.18, 0.12, 0.10, 0.10]
    elif n_top == 4:
        fracs = [0.52, 0.20, 0.14, 0.14]
    elif n_top == 3:
        fracs = [0.55, 0.22, 0.23]
    elif n_top == 2:
        fracs = [0.60, 0.40]
    else:
        fracs = [0.95]

    for i in range(n_top):
        if time.monotonic() > deadline:
            break
        sc, name, a0 = scored[i]
        t_alloc = remaining * fracs[i]
        t_alloc = min(t_alloc, max(10.0, deadline - time.monotonic()))

        print(f"\n[Phase {i+1}] Optimizing '{name}' (n={len(a0)}, init={sc:.6f}, t={t_alloc:.0f}s)")

        # Adam (55% of phase time)
        t_adam = t_alloc * 0.55
        a1, s1 = _optimize_adam(a0, t_adam, rng)
        ev1 = evaluate_sequence(a1.tolist())
        print(f"  Adam: {ev1:.6f}")
        if np.isfinite(ev1) and ev1 < best_overall_score:
            best_overall_score = ev1
            best_overall = a1.copy()
            print(f"  *** NEW BEST: {ev1:.6f} ***")

        # LP-based (30%)
        t_lp = deadline - time.monotonic()
        t_lp = min(t_alloc * 0.30, max(8.0, deadline - time.monotonic()))
        start_lp = a1 if np.isfinite(s1) and s1 < sc else a0
        a2, s2 = _optimize_lp(start_lp, t_lp, rng)
        ev2 = evaluate_sequence(a2.tolist())
        print(f"  LP:   {ev2:.6f}")
        if np.isfinite(ev2) and ev2 < best_overall_score:
            best_overall_score = ev2
            best_overall = a2.copy()
            print(f"  *** NEW BEST: {ev2:.6f} ***")

        # Multiplicative (15%)
        if time.monotonic() < deadline - 10:
            t_mult = min(t_alloc * 0.15, max(5.0, deadline - time.monotonic()))
            best_phase = a2 if np.isfinite(s2) and (not np.isfinite(s1) or s2 < s1) else a1
            a3, s3 = _optimize_mult(best_phase, t_mult, rng)
            ev3 = evaluate_sequence(a3.tolist())
            print(f"  Mult: {ev3:.6f}")
            if np.isfinite(ev3) and ev3 < best_overall_score:
                best_overall_score = ev3
                best_overall = a3.copy()
                print(f"  *** NEW BEST: {ev3:.6f} ***")

    # Final polishing phase
    remaining = deadline - time.monotonic()
    if best_overall is not None and remaining > 40:
        print(f"\n=== Final polishing ({remaining:.0f}s) ===")

        # Adam polish
        t1 = remaining * 0.45
        a_p, sp = _optimize_adam(best_overall, t1, rng)
        ev_p = evaluate_sequence(a_p.tolist())
        print(f"  Adam polish: {ev_p:.6f}")
        if np.isfinite(ev_p) and ev_p < best_overall_score:
            best_overall_score = ev_p
            best_overall = a_p.copy()

        # LP polish
        t2 = deadline - time.monotonic()
        t2 = min(remaining * 0.35, max(10.0, deadline - time.monotonic()))
        a_q, sq = _optimize_lp(best_overall, t2, rng)
        ev_q = evaluate_sequence(a_q.tolist())
        print(f"  LP polish:   {ev_q:.6f}")
        if np.isfinite(ev_q) and ev_q < best_overall_score:
            best_overall_score = ev_q
            best_overall = a_q.copy()

        # Adam polish again from best
        t3 = deadline - time.monotonic()
        if t3 > 15:
            a_r, sr = _optimize_adam(best_overall, t3 * 0.9, rng)
            ev_r = evaluate_sequence(a_r.tolist())
            print(f"  Adam polish2: {ev_r:.6f}")
            if np.isfinite(ev_r) and ev_r < best_overall_score:
                best_overall_score = ev_r
                best_overall = a_r.copy()

    # Final verification
    if best_overall is not None:
        final_score = evaluate_sequence(best_overall.tolist())
        elapsed = time.monotonic() - total_start
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(best_overall)}, time={elapsed:.1f}s ===")
        return [float(x) for x in best_overall]
    else:
        print("Falling back to uniform sequence")
        return [1.0] * 1000
