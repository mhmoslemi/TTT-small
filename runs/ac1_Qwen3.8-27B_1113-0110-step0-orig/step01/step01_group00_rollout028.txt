import numpy as np
import time
from scipy.optimize import linprog

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]

def _score(a):
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)

def _build_conv_matrix(f):
    n = len(f)
    size = 2 * n - 1
    C = np.zeros((size, n), dtype=np.float64)
    for j in range(n):
        C[j:j+n, j] = f
    return C

def _lp_solve(f, M, prev_g=None):
    n = len(f)
    C = _build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.status == 0:
            return result.x
    except Exception:
        pass
    return None

def _alternating_lp_v2(a_init, n_iters=200, deadline=None, blend_start=0.9, blend_end=0.3, blend_decay=0.98):
    """Improved alternating LP with adaptive M and blend."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = _score(a)
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        t = it / max(1, n_iters - 1)
        blend = blend_start * (blend_end / blend_start) ** t
        
        c = _fft_convolve(a)
        M = float(np.max(c))
        
        g = _lp_solve(a, M)
        if g is None:
            break
        sum_g = float(np.sum(g))
        if sum_g < 1e-12:
            break
        g = g / sum_g
        
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        sum_new = float(np.sum(a_new))
        if sum_new < 1e-12:
            break
        a_new = a_new / sum_new
        
        score = _score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_score

    """Given f, find g with sum(g)=1, g>=0 that minimizes max_k (f*g)[k]."""
    n = len(f)
    size = 2 * n - 1
    C = _build_conv_matrix(f)  # shape (size, n)
    
    # Variables: [g_0, ..., g_{n-1}, M]
    # Objective: minimize M
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize M
    
    # Inequality: C @ g - M <= 0 → [C | -1] @ [g; M] <= 0
    A_ub = np.zeros((size, n + 1))
    A_ub[:, :n] = C
    A_ub[:, n] = -1.0
    b_ub = np.zeros(size)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: g_i >= 0, M >= 0
    bounds = [(0.0, None)] * (n + 1)
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

"""
Hybrid alternating LP + subgradient optimizer for minimizing
J(a) = 2n * max(conv(a,a)) / sum(a)^2 over non-negative sequences.

Uses a cleaner LP formulation (directly minimize peak M) combined with
diverse initializations, adaptive blending, and subgradient polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _fft_convolve(a):
    """Fast full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]


def _score(a):
    """Compute J(a) = 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)


def _build_conv_matrix(f):
    """Build the convolution matrix C where (C @ g)[k] = (f*g)[k]."""
    n = len(f)
    size = 2 * n - 1
    C = np.zeros((size, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = f
    return C


def _lp_min_peak(f):
    """
    Given fixed f, solve: minimize M s.t. (f*g)[k] <= M for all k,
    sum(g) = 1, g >= 0.
    Returns (g, M) or (None, None).
    """
    n = len(f)
    size = 2 * n - 1
    C = _build_conv_matrix(f)

    # Variables: x = [g_0, ..., g_{n-1}, M]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    # Inequality: C @ g - M <= 0
    A_ub = np.zeros((size, n + 1))
    A_ub[:, :n] = C
    A_ub[:, n] = -1.0
    b_ub = np.zeros(size)

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * (n + 1)

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 30})
        if result.status == 0:
            g = result.x[:n]
            M_val = float(result.x[n])
            return g, M_val
    except Exception:
        pass
    return None, None


def _alternating_lp(a_init, n_iters, deadline):
    """
    Alternating LP: fix a, minimize peak of (a*g), set g; fix g, minimize peak of (g*a), set a.
    With adaptive blending and tracking best.
    """
    a = a_init.copy().astype(np.float64)
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_score = _score(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        # Adaptive blend: start high, decay
        t = it / max(1, n_iters - 1)
        blend = 0.85 * (0.4 / 0.85) ** t

        # Fix a, find g
        g, M_g = _lp_min_peak(a)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        sg = float(np.sum(g))
        if sg < 1e-12:
            break
        g = g / sg

        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        sn = float(np.sum(a_new))
        if sn < 1e-12:
            break
        a_new = a_new / sn

        score = _score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        # Fix g, find a
        a_sol, M_a = _lp_min_peak(g)
        if a_sol is not None:
            a_sol = np.maximum(a_sol, 0.0)
            sa = float(np.sum(a_sol))
            if sa > 1e-12:
                a_sol = a_sol / sa
                a_new2 = (1.0 - blend) * a_new + blend * a_sol
                a_new2 = np.maximum(a_new2, 0.0)
                sn2 = float(np.sum(a_new2))
                if sn2 > 1e-12:
                    a_new2 = a_new2 / sn2
                    score2 = _score(a_new2)
                    if score2 < best_score:
                        best_score = score2
                        best_a = a_new2.copy()
                    a = a_new2
                else:
                    a = a_new
            else:
                a = a_new
        else:
            a = a_new

    return best_a, best_score


def _subgradient_refine(a_init, n_iters, deadline, top_k=5, lr_init=0.5, lr_end=0.01):
    """
    Subgradient method to flatten the convolution peaks.
    """
    a = a_init.copy().astype(np.float64)
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_score = _score(a)

    lr_ratio = lr_end / lr_init
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        lr = lr_init * (lr_ratio ** (it / max(1, n_iters - 1)))
        c = _fft_convolve(a)

        # Get top-k peak indices
        if top_k >= len(c):
            peak_idx = np.arange(len(c))
        else:
            peak_idx = np.argpartition(c, -top_k)[-top_k:]

        # Compute subgradient (sum of sensitivity vectors at peaks)
        grad = np.zeros(n)
        for k in peak_idx:
            k = int(k)
            j = np.arange(n)
            # Sensitivity: d(a*a)[k]/da[j] = a[k-j] + a[j+k] (with bounds)
            idx1 = k - j
            idx2 = j - k
            m1 = (idx1 >= 0) & (idx1 < n)
            m2 = (idx2 >= 0) & (idx2 < n)
            sg = np.zeros(n)
            sg[m1] += a[idx1[m1]]
            sg[m2] += a[idx2[m2]]
            grad += sg

        grad /= max(1, len(peak_idx))

        # Gradient step
        a = a - lr * grad
        a = np.maximum(a, 0.0)
        sn = float(np.sum(a))
        if sn < 1e-10:
            a = best_a.copy()
            continue
        a = a / sn

        score = _score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score


def _make_init(n, variant, rng):
    """Create diverse initializations."""
    t = np.linspace(0, 1, n)
    if variant == 0:
        a = np.ones(n) / n
    elif variant == 1:
        a = np.exp(-4.0 * t)
    elif variant == 2:
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))  # Hann window
    elif variant == 3:
        a = 0.42 - 0.5 * np.cos(2.0 * np.pi * t) + 0.08 * np.cos(4.0 * np.pi * t)  # Blackman
    elif variant == 4:
        # Two-bump Gaussian
        a = np.exp(-((t - 0.3) ** 2) / 0.02) + 0.5 * np.exp(-((t - 0.7) ** 2) / 0.05)
    elif variant == 5:
        # Three-bump
        a = (np.exp(-((t - 0.2) ** 2) / 0.03) + np.exp(-((t - 0.5) ** 2) / 0.02) +
             np.exp(-((t - 0.8) ** 2) / 0.03))
    elif variant == 6:
        # Raised cosine with modulation
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t)) * (1.0 + 0.3 * np.cos(2.0 * np.pi * 3.0 * t))
    elif variant == 7:
        # Random smooth
        a = np.abs(rng.normal(0, 1, n)).cumsum()
        a = a / a[-1]
    elif variant == 8:
        # Parabolic
        a = 1.0 - 4.0 * (t - 0.5) ** 2
    elif variant == 9:
        # Flat top with tapered edges
        w = 0.1
        a = np.ones(n)
        fade = int(w * n)
        for i in range(fade):
            a[i] = 0.5 * (1 - np.cos(np.pi * i / fade))
            a[n - 1 - i] = a[i]
    elif variant == 10:
        # Sine-squared
        a = np.sin(np.pi * t) ** 2
    elif variant == 11:
        # Inverse Gaussian (peaked in middle)
        a = np.exp(-((t - 0.5) ** 2) / 0.05)
    else:
        a = np.ones(n) / n

    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def _interpolate_to_length(src, n_target):
    """Interpolate a sequence to a new length, preserving non-negativity."""
    n_src = len(src)
    if n_src == n_target:
        return src.copy()
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    out = np.interp(x_tgt, x_src, src)
    out = np.maximum(out, 0.0)
    s = float(np.sum(out))
    if s > 1e-10:
        out = out / s
    else:
        out = np.ones(n_target) / n_target
    return out
