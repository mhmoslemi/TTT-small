"""
Minimizes J(a) = 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Uses a multi-phase approach:
1. LP-based alternating optimization at multiple sequence lengths
2. Log-sum-exp smoothed subgradient with Nesterov momentum
3. Coordinate descent with exact 1D line search
4. Final LP polish
Key innovation: adaptive log-sum-exp beta, Polyak step-size, and

def _fft_convolve(a):
    """Fast full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]


def _score(a):
    """Compute J(a) = 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)


def _score_and_conv(a):
    """Compute score and full convolution simultaneously."""
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf'), c
    return 2.0 * n * max_c / (s * s), c


def _build_conv_matrix(f):
    """Build the convolution matrix C where (C @ g)[k] = (f*g)[k]."""
    n = len(f)
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = f
    return C


def _lp_solve(f, M):
    """Solve max sum(g) s.t. C_f @ g <= M, g >= 0. Returns g or None."""
    n = len(f)
    C = _build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.status == 0:
            return result.x
    except Exception:
        pass
    return None


def _alternating_lp(a_init, max_iters, deadline, initial_blend=0.8):
    """Run alternating LP: fix f, solve LP for g, blend, repeat."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = _score(a)
    blend = initial_blend

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        M = float(np.max(c))
        g = _lp_solve(a, M)
        if g is None:
            break
        sum_g = float(np.sum(g))
        if sum_g < 1e-12:
            break
        g = g / sum_g
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        sum_new = float(np.sum(a_new))
        if sum_new < 1e-12:

        a_new = a_new / sum_new
        score = _score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        blend = max(0.1, blend * 0.97)

    return best_a, best_score


def _logsumexp_gradient(a, beta, top_k=10):
    """Compute gradient of (1/beta)*log(sum exp(beta*c[k])) for top-k peaks.
    
    Returns (grad, max_c) where max_c is the true max of c.
    """
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    
    if top_k >= len(c):
        top_idx = np.arange(len(c))
    else:
        top_idx = np.argpartition(c, -top_k)[-top_k:]
    
    c_top = c[top_idx]
    # Shift for numerical stability
    c_shifted = c_top - c_top.max()
    weights = np.exp(beta * c_shifted)
    w_sum = weights.sum()
    if w_sum < 1e-300:
        weights = np.ones(len(top_idx)) / len(top_idx)
    else:
        weights = weights / w_sum
    
    # grad[i] = 2 * sum_k weights[k] * a[k-i] (for valid k-i in [0, n))
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        k = int(k)
        w = weights[i]
        # For coordinate j, dc[k]/da[j] = 2*a[k-j] if 0 <= k-j < n
        # So grad[j] += 2*w*a[k-j] for j in [max(0,k-n+1), min(n-1,k)]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            js = np.arange(lo, hi + 1)
            grad[lo:hi + 1] += 2.0 * w * a[k - js]
    
    return grad, max_c


def _subgradient_lse(a_init, max_iters, deadline, beta_start=500, beta_end=20000,

    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s  # normalize to sum=1
    
    if target_M is None:
        target_M = 1.0 / (2 * n - 1)  # theoretical lower bound
    
    best_a = a.copy()
    best_score = _score(a)
    best_M = best_score / (2.0 * n)
    
    velocity = np.zeros(n)

    n_effective = max(1, max_iters)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Interpolate beta
        frac = it / max(n_effective - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 50000.0)  # cap to avoid overflow
        
        # Keep a on the fly with momentum
        a_moment = a + momentum * velocity

def _coordinate_descent(a_init, max_iters, deadline):
    """Coordinate descent: for each coordinate, optimize it exactly given others.
    
    For fixed a with a[i] = t, the rest scaled to maintain sum=1:
    a'[j] = a[j]*(1-t)/(1-a[i]) for j != i
    c[k] = 2*t*a'[k-i] + b[k] where b[k] doesn't involve a[i]
    We minimize max_k c[k] over t in [0, 1-a[i]_max_constraint].
    """
    a = a_init.copy().astype(np.float64)
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, _score(a)
    a = a / s
    
    best_a = a.copy()
    best_score = _score(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            
            saved = a[i]
            rest_sum = s - saved  # sum of all other elements
            
            if rest_sum < 1e-15:
                continue
            
            # If we set a[i] = t, then all other elements scale by (1-t)/rest_sum
            # c[k] = 2*t * (a_other_scaled[k-i]) + (a_other_scaled @ a_other_scaled shifted)
            # = 2*t * a[k-i]*(1-t)/rest_sum + ((1-t)/rest_sum)^2 * c_other[k]
            
            # Precompute c_other[k] = sum_{j!=i} a[j]*a[k-j] (only terms not involving a[i])
            # c_other[k] = c_full[k] - 2*a[i]*a[k-i] (for valid k-i)
            a_others = a.copy()
            a_others[i] = 0.0
            c_others = _fft_convolve(a_others)
            
            # For t in [0, 1]:
            # c[k](t) = 2*t * a_others[k-i] * (1-t)/rest_sum + c_others[k] * ((1-t)/rest_sum)^2
            # Let u = (1-t)/rest_sum, then t = rest_sum*(1-u), u in [0, 1/rest_sum]
            # Actually, let's just do a 1D search over t.
            
            if saved < 1e-15:
                # Try small values of t
                t_candidates = np.array([0.0, 0.01/rest_sum, 0.05/rest_sum, 0.1/rest_sum])
            else:
                # Trial around current value
                t_center = saved
                t_spread = max(0.1 * t_center, 0.01 / rest_sum)
                t_candidates = np.linspace(max(0, t_center - 2*t_spread), 
                                           min(rest_sum, t_center + 2*t_spread), 20)
            
            best_t = saved
            best_local_M = float('inf')
            
            for t in t_candidates:
                t = max(0.0, min(t, rest_sum))
                scale = (s - t) / rest_sum if rest_sum > 1e-15 else 1.0
                a_trial = a_others * scale
                a_trial[i] = t
                # Compute max of conv
                c_trial = _fft_convolve(a_trial)
                M_trial = float(np.max(c_trial))
                if M_trial < best_local_M:
                    best_local_M = M_trial
                    best_t = t
            
            if abs(best_t - saved) > 1e-10:
                a[i] = best_t
                # Renormalize (should already be sum=1 by construction)
                s_new = float(np.sum(a))
                if abs(s_new - s) > 1e-10:
                    a = a / s_new
                improved = True
        
        score = _score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if not improved:
            # Try random restarts within coordinate descent
            for _ in range(5):
                i = np.random.randint(n)
                a[i] = np.random.random() * 0.1 / n
                a = np.maximum(a, 0)
                a = a / a.sum()
    
    return best_a, best_score


def _make_initial(n, variant, rng):
    """Create diverse initial non-negative sequences."""
    t = np.linspace(0, 1, n)
    if variant == 0:
        a = 1.0 - t
    elif variant == 1:
        a = np.exp(-3.0 * t)
    elif variant == 2:
        a = (1.0 - t) * (1.0 + 0.5 * np.cos(2.0 * np.pi * 3.0 * t))
        a = np.maximum(a, 0.0)
    elif variant == 3:
        a = (1.0 - t) * (1.0 + 0.7 * np.sin(2.0 * np.pi * 5.0 * t))
        a = np.maximum(a, 0.0)
    elif variant == 4:
        a = np.abs(rng.normal(0, 1, n))
    elif variant == 5:
        a = np.exp(-((t - 0.3) ** 2) / 0.02) + 0.5 * np.exp(-((t - 0.7) ** 2) / 0.05)
    elif variant == 6:
        a = t ** 2 * (3.0 - 2.0 * t)
    elif variant == 7:
        a = (1.0 - t) ** 1.5 * (1.0 + 0.4 * np.cos(2.0 * np.pi * 2.0 * t + 1.0))
        a = np.maximum(a, 0.0)
    elif variant == 8:
        a = (1.0 - t) * (1.0 + 0.3 * np.cos(2.0 * np.pi * 7.0 * t))
        a = np.maximum(a, 0.0)
    elif variant == 9:
        a = np.exp(-2.0 * t) * (1.0 + 0.5 * np.sin(2.0 * np.pi * 4.0 * t))
        a = np.maximum(a, 0.0)
    elif variant == 10:
        # Two-bump with tuning
        a = np.exp(-((t - 0.25) ** 2) / 0.015) + 0.8 * np.exp(-((t - 0.75) ** 2) / 0.03)
    elif variant == 11:
        # Three bump
        a = np.exp(-((t - 0.2) ** 2) / 0.02) + 0.7 * np.exp(-((t - 0.5) ** 2) / 0.02) + 0.9 * np.exp(-((t - 0.8) ** 2) / 0.02)
    elif variant == 12:
        # Oscillatory with slow decay
        a = (1.0 - t) * (1.0 + 0.6 * np.cos(2.0 * np.pi * 11.0 * t))
        a = np.maximum(a, 0.0)
    elif variant == 13:
        # Ring-like
        a = np.exp(-((t - 0.5) ** 2) / 0.01) * (1.0 + 0.5 * np.cos(2.0 * np.pi * 8.0 * t))
        a = np.maximum(a, 0.0)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def _interpolate_sequence(src, n_target):
    """Interpolate a sequence to a new length."""
    n_src = len(src)
    if n_src == n_target:
        return src.copy()
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    out = np.interp(x_tgt, x_src, src)
    out = np.maximum(out, 0.0)
    s = float(np.sum(out))
    if s > 1e-10:
        out = out / s
    return out


def _spectral_flatten(a_init, max_iters, deadline):
    """Spectral domain flattening: modify phase in frequency domain to flatten
    the autocorrelation, then project back to non-negative time domain.
    
    Uses the fact that we want |A(omega)|^2 to be as flat as possible
    (subject to DC constraint).
    """

    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_score = _score(a)
    
    # Use a larger FFT size to handle linear convolution
    fft_size = 1 << (2 * n - 1).bit_length()
    
    momentum = 0.5
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Go to frequency domain
        A = np.fft.rfft(a, fft_size)
        
        # Compute current autocorrelation (linear)
        c = _fft_convolve(a)
        max_c = float(np.max(c))
        if 2.0 * n * max_c < best_score:
            best_score = 2.0 * n * max_c
            best_a = a.copy()
        
        # Target: make |A|^2 flat (minus DC)
        # |A[0]|^2 = (sum a)^2 = 1, so target magnitude for non-DC = sqrt(1/(n-1)) approx
        # But we can't change |A[0]|, so let's try to minimize the variance of |A[m]|^2 for m>0
        # 
        # Alternative: modify the phase of A to make IFFT more uniform
        # This is like a "phase retrieval" approach
        
        # Compute target: we want IFFT(|A|^2) to be flat
        # |A|^2 is fixed in magnitude (since |A| is determined by a), but we can 
        # exploit the fact that changing the sign/phase of individual frequency
        # components changes the time-domain signal while keeping |A| fixed.
        
        # Actually for real non-negative signals, we can't freely change phase.
        # Instead, let's use a different approach:
        # 
        # "Water-filling" in frequency domain:
        # We want to minimize max of inverse-DFT of |A|^2 (which is the autocorrelation)
        # Since |A|^2 is determined by a (not freely choosable), this doesn't directly help.
        
        # Better approach: use the frequency domain to compute a "pressure" signal
        # that tells us where to add/subtract in time domain.
        
        # The autocorrelation in frequency domain is |A|^2.
        # The "excess" at each frequency is |A[m]|^2 - mean(|A|^2).
        # The time-domain signal corresponding to this excess is the "autocorrelation excess".
        # We want to minimize the peak of this excess.
        
        # Gradient approach in frequency domain:
        # d(max autocorr)/da = gradient as before, but we can compute it via FFT
        # 
        # Actually, let's just use this as a different initialization strategy:
        # Create a signal whose |FFT| is as flat as possible, then take its absolute value.
        
        break  # Skip this for now, it's complex and may not help
    
    return best_a, best_score


def _polynomial_fit_optimize(n, deadline):
    """Try to find optimal sequences of the form a[j] = P(j/n) where P is a polynomial.
    
    We parameterize by polynomial coefficients and use scipy.optimize.
    """
    def _make_poly_a(coeffs, n):
        """Create sequence from polynomial coefficients."""
        t = np.linspace(0, 1, n)
        a = np.zeros(n)
        for i, c in enumerate(coeffs):
            a += c * t ** i
        a = np.maximum(a, 0.0)
        s = float(np.sum(a))
        if s < 1e-10:
            return np.ones(n) / n
        return a / s
    
    def _objective(coeffs):
        a = _make_poly_a(coeffs, n)
        return _score(a)
    
    best_a = None
    best_score = float('inf')
    
    # Try different polynomial orders and initial points
    for order in [2, 3, 4, 5, 6]:
        if time.monotonic() > deadline:
            break
        rng_local = np.random.default_rng(order * 7)
        for trial in range(3):
            if time.monotonic() > deadline:
                break
            # Initialize with random coefficients
            coeffs = np.concatenate([np.ones(1), rng_local.normal(0, 0.5, order)])
            # Ensure non-negative at endpoints
            t_test = np.linspace(0, 1, 100)
            val = np.zeros(100)
            for i, c in enumerate(coeffs):
                val += c * t_test ** i
            if np.min(val) < -1:
                continue
            try:
                result = minimize(_objective, coeffs, method='Nelder-Mead',
                                  options={'maxiter': 2000, 'xatol': 1e-8, 'fatol': 1e-10})
                if result.fun < best_score:
                    best_score = result.fun
                    best_a = _make_poly_a(result.x, n)
                    print(f"      Poly order={order}, trial={trial}: score={result.fun:.6f}")
            except Exception:
                pass
    
    return best_a, best_score


def _multi_start_subgradient(a_init, n_restarts, deadline, max_iters_each=5000):
    """Run subgradient from multiple perturbed starts."""
    best_a = a_init.copy()
    best_score = _score(a_init)
    rng = np.random.default_rng(123)
    n = len(a_init)
    
    per_restart = max(1000, max_iters_each // max(1, n_restarts))
    
    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break
        
        if r == 0:
            a_start = a_init.copy()
        else:
            # Perturb
            a_start = a_init + rng.normal(0, 0.01 / n, n)
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()
        
        a_opt, score = _subgradient_lse(a_start, per_restart, 
                                         min(deadline, time.monotonic() + 200.0),
                                         beta_start=200, beta_end=10000,
                                         top_k=8, momentum=0.8)
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
    
    return best_a, best_score
