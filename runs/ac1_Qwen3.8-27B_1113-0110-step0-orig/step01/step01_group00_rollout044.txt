import numpy as np
import time
from scipy.optimize import linprog, minimize

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]

def _score(a):
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)

def _build_conv_matrix(f):
    n = len(f)
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        C[j:j+n, j] = f
    return C

def _lp_min_M(f):
    """Solve min M s.t. C_f @ g <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, inf)."""
    n = len(f)
    C = _build_conv_matrix(f)
    # Variables: [g(0), ..., g(n-1), M]
    # Objective: minimize M -> c_obj = [0,...,0,1]
    # Constraints: C_f @ g - M <= 0 for each row
    #              sum(g) = 1
    #              g >= 0, M >= 0 (well, M can be anything but it'll be positive)
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize M
    
    # Inequality constraints: C_f @ g - M <= 0
    A_ub = np.hstack([C, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    
    # Equality constraint: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        if result.status == 0:
            g = result.x[:n]
            M = result.x[n]
            return g, M
    except Exception:
        pass
    return None, float('inf')

def _alternating_lp_v2(a_init, max_iters, deadline, initial_blend=0.7):
    """Improved alternating LP with direct min-M formulation."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = _score(a)
    blend = initial_blend
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        g, M = _lp_min_M(a)
        if g is None:
            break
        sum_g = float(np.sum(g))
        if sum_g < 1e-12:
            break
        g = g / sum_g
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        sum_new = float(np.sum(a_new))
        if sum_new < 1e-12:
            break
        a_new = a_new / sum_new
        score = _score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            if it % 50 == 0:
                print(f"      iter {it}: score={best_score:.6f} (M={M:.6e})")
        a = a_new
        blend = max(0.05, blend * 0.98)
    
    return best_a, best_score

            break

        if sum_g < 1e-12:

        a_new = (1.0 - blend) * a + blend * g

            best_a = a_new.copy()
            if it % 100 == 0:

Actually, I realize I should also think about what makes this problem hard. The function J(a) = 2n * max_k c_k / s^2 is non-smooth (due to the max). The LP approach handles this by linearizing at the current point. The gradient approach handles it by either:
1. Subgradient (use the subgradient of the max function)
2. Smooth approximation (log-sum-exp)

The log-sum-exp approach is probably the best for gradient methods:
L_beta(a) = (1/beta) * log(sum_k exp(beta * c_k))

This is smooth and converges to max(c_k) as beta -> infinity.

The gradient of L_beta is:
dL_beta/da[j] = (1/beta) * sum_k exp(beta*c_k) * dc_k/da[j] / sum_k exp(beta*c_k)
= sum_k w_k * dc_k/da[j]
where w_k = exp(beta*c_k) / sum_k exp(beta*c_k) (softmax weights)

And dc_k/da[j] = 2*a[k-j] * 1_{0<=k-j<n} (as derived above).

So the gradient is:
dL_beta/da[j] = 2 * sum_k w_k * a[k-j] * 1_{0<=k-j<n}
= 2 * (w ⊛ a)[j]  (cross-correlation)

from scipy.optimize import linprog

def _fft_convolve(a):
    """Fast full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)

    """Compute J(a) = 2n * max(conv(a,a)) / sum(a)^2."""

def _gradient_polish(a_init, deadline, beta_start=20, beta_end=200, max_iters=50000):
    """Polish using projected gradient on log-sum-exp smoothed objective."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = _score(a)
    
    # Learning rate schedule
    lr = 0.1
    lr_decay = 0.9999
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Increase beta over time (from smooth to sharp)
        progress = min(it / max_iters, 1.0)
        beta = beta_start + (beta_end - beta_start) * progress
        
        c = _fft_convolve(a)
        max_c = float(np.max(c))
        
        # Softmax weights
        w = np.exp(beta * (c - max_c))
        w_sum = float(np.sum(w))
        if w_sum < 1e-30:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        else:
            w /= w_sum
        
        # Compute gradient: grad[j] = 2 * sum_k w[k] * a[k-j] (valid k-j)
        # Use FFT cross-correlation
        fft_size = 1 << (3*n - 2).bit_length()
        W = np.zeros(fft_size)
        W[:2*n-1] = w
        B = np.zeros(fft_size)
        B[:n] = a
        FW = np.fft.rfft(W)
        FB = np.fft.rfft(B)
        corr = np.fft.irfft(FW * np.conj(FB), fft_size)
        grad = 2.0 * corr[:n]
        
        # Gradient of the full score J = 2n * L / s^2
        # dJ/da[j] = (2n/s^2) * dL/da[j] - (4n*L/s^3) * 1
        # For the projected gradient, we only need the direction, so:
        # direction[j] = dL/da[j] - (L/s) * (1/s)  ... actually let's just use dL/da[j]
        # and project onto sum(a)=1, a>=0.
        
        # Projected gradient step
        max_grad = float(np.max(np.abs(grad)))
        if max_grad < 1e-15:
            lr *= lr_decay
            continue
        
        a_new = a - lr * grad / max_grad
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s_new
        
        # Check if we improved
        score = _score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            if it % 5000 == 0:
                print(f"      grad iter {it}: score={best_score:.8f} beta={beta:.1f} lr={lr:.6f}")
            a = a_new
        else:
            # If not improved, reduce learning rate
            lr *= 0.95
        lr = max(lr, 0.001)
        lr *= lr_decay
    
    print(f"    Gradient polish final: score={best_score:.8f}")
    return best_a, best_score

def _interpolate_sequence(src, n_target):

    out = np.maximum(out, 0.0)
    s = float(np.sum(out))
    if s > 1e-10:
        out = out / s
    return out

2. Taking a step towards it

This is very simple! And it's known to converge (albeit slowly, as O(1/t)).

1. Find k* = argmax_k c_k

3. Find j* = argmin_j s[j] (the FW oracle)

def _fft_convolve(a):
    """Fast full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]


def _score(a):
    """Compute J(a) = 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    c = _fft_convolve(a)

    s = float(np.sum(a))
    if s < 0.01:

def _build_conv_matrix(f):
    """Build convolution matrix C where (C @ g)[k] = (f*g)[k]."""
    n = len(f)
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = f
    return C

    print(f"    Gradient polish final: score={best_score:.8f}")
    return best_a, best_score


def _make_initial(n, variant, rng):
    """Create diverse initial non-negative sequences."""
    t = np.linspace(0, 1, n)
    if variant == 0:
        a = 1.0 - t
    elif variant == 1:
        a = np.exp(-3.0 * t)
    elif variant == 2:
        a = (1.0 - t) * (1.0 + 0.5 * np.cos(2.0 * np.pi * 3.0 * t))
        a = np.maximum(a, 0.0)
    elif variant == 3:
        # Double Gaussian (best performer)
        a = np.exp(-((t - 0.3) ** 2) / 0.02) + 0.5 * np.exp(-((t - 0.7) ** 2) / 0.05)
    elif variant == 4:
        # Asymmetric double bump
        a = np.exp(-((t - 0.2) ** 2) / 0.03) + 0.7 * np.exp(-((t - 0.8) ** 2) / 0.025)
    elif variant == 5:
        # Tighter double bump
        a = np.exp(-((t - 0.25) ** 2) / 0.015) + 0.6 * np.exp(-((t - 0.75) ** 2) / 0.02)
    elif variant == 6:
        a = (1.0 - t) * (1.0 + 0.4 * np.sin(2.0 * np.pi * 4.0 * t))
        a = np.maximum(a, 0.0)
    elif variant == 7:
        # Three bumps
        a = np.exp(-((t - 0.15) ** 2) / 0.01) + 0.5 * np.exp(-((t - 0.5) ** 2) / 0.02) + 0.8 * np.exp(-((t - 0.85) ** 2) / 0.015)
    elif variant == 8:
        a = np.exp(-2.0 * t) * (1.0 + 0.6 * np.cos(2.0 * np.pi * 6.0 * t))
        a = np.maximum(a, 0.0)
    elif variant == 9:
        # Epanechnikov
        a = np.sqrt(np.maximum(1.0 - (2.0 * t - 1.0) ** 2, 0.0))
    elif variant == 10:
        # Quartic (very flat top)
        a = 1.0 - (2.0 * t - 1.0) ** 4
    elif variant == 11:
        # Wide double bump
        a = np.exp(-((t - 0.35) ** 2) / 0.04) + 0.8 * np.exp(-((t - 0.65) ** 2) / 0.04)
    elif variant == 12:
        # Smooth random
        a = np.abs(rng.normal(0, 1, n))
        kernel = np.exp(-0.5 * (np.arange(-10, 11) / 3.0) ** 2)
        kernel /= kernel.sum()
        a = np.convolve(a, kernel, mode='same')
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def _interpolate_sequence(src, n_target):
    """Interpolate a sequence to a new length."""
    n_src = len(src)
    if n_src == n_target:
        return src.copy()
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    out = np.interp(x_tgt, x_src, src)
    out = np.maximum(out, 0.0)
    s = float(np.sum(out))
    if s > 1e-10:
        out = out / s
    return out


def _annealing_restart(a_current, best_score, deadline, rng, n_restarts=5):
    """Try to escape local minima via perturb + LP."""
    a = a_current.copy()
    n = len(a)
    improved = False

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break
        # Perturb: add noise and renormalize
        noise_scale = 0.1 * (1.0 + 0.5 * r)
        noise = rng.uniform(-noise_scale, noise_scale, n)
        a_pert = a + noise * np.abs(a) * 2.0
        a_pert = np.maximum(a_pert, 0.0)
        s = float(np.sum(a_pert))
        if s < 1e-10:
            continue
        a_pert = a_pert / s

        # Run LP from perturbed point
        restart_deadline = min(deadline, time.monotonic() + 120.0)
        a_opt, score = _alternating_lp(a_pert, 200, restart_deadline,
                                       blend_init=0.6, blend_decay=0.98, blend_min=0.03)
        if score < best_score:
            best_score = score
            a = a_opt.copy()
            improved = True
            print(f"      restart {r}: new best {score:.8f}")

    return a, best_score, improved


def propose_candidate():
    """
    Multi-phase optimizer minimizing J(a)=2n*max(conv(a,a))/sum(a)^2.
    Phase 1: Alternating LP at multiple n values with diverse initializations.
    Phase 2: Log-sum-exp gradient polishing.
    Phase 3: LP re-polish and annealing restarts.
    Phase 4: Try larger n via interpolation + LP.
    """
    total_deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(7)

    best_a = None
    best_score = float('inf')

    # Load parent sequence
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            parent_score = _score(parent)
            print(f"Parent sequence: n={parent.size}, score={parent_score:.6f}")
            best_a = parent.copy()
            best_score = parent_score
    except Exception:
        pass

    # Phase 1: Alternating LP - the workhorse
    print("\n=== Phase 1: Alternating LP ===")
    n_list = [100, 150, 200, 250, 300, 400, 500]

    for n in n_list:
        if time.monotonic() > total_deadline - 600:
            break
        # Allocate time: more time for intermediate n
        if n <= 200:
            phase_time = 250.0
        elif n <= 400:
            phase_time = 300.0
        else:
            phase_time = 200.0
        phase_deadline = min(total_deadline - 600, time.monotonic() + phase_time)
        print(f"\n  --- n={n} ---")

        inits = []
        labels = []
        if best_a is not None:
            inits.append(_interpolate_sequence(best_a, n))
            labels.append("from_best")

        # Best-performing initializations first
        for v in [3, 4, 5, 3, 11, 0, 6, 7, 8, 10, 1]:
            inits.append(_make_initial(n, v, rng))
            labels.append(f"v{v}")

        for label, a_init in zip(labels, inits):
            if time.monotonic() > phase_deadline:
                break
            print(f"    init={label:12s} ", end="", flush=True)
            a_opt, score = _alternating_lp(a_init, 500, phase_deadline,
                                           blend_init=0.7, blend_decay=0.985, blend_min=0.03)
            print(f"score={score:.8f}")
            if score < best_score:
                best_score = score
                best_a = a_opt.copy()
                print(f"    ** New best: {best_score:.8f} (n={n}, init={label})")

    print(f"\nAfter Phase 1: best_score={best_score:.8f}, n={len(best_a) if best_a is not None else 0}")

    # Phase 2: Gradient polishing with log-sum-exp
    print("\n=== Phase 2: Gradient Polish ===")
    if best_a is not None and time.monotonic() < total_deadline - 300:
        n_best = len(best_a)
        polish_deadline = min(total_deadline - 200, time.monotonic() + 400.0)
        print(f"  Polishing at n={n_best}")
        a_opt, score = _gradient_polish(best_a, polish_deadline,
                                        beta_start=30, beta_end=500, max_iters=200000)
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  ** New best after gradient: {best_score:.8f}")

    # Phase 3: LP re-polish + annealing
    print("\n=== Phase 3: LP Polish + Annealing ===")
    if best_a is not None and time.monotonic() < total_deadline - 150:
        n_best = len(best_a)
        n_lp = min(n_best, 400)
        polish_deadline = min(total_deadline - 80, time.monotonic() + 250.0)
        if n_best <= 400:
            a_ds = best_a.copy()
        else:
            a_ds = _interpolate_sequence(best_a, n_lp)
        print(f"  LP polish at n={n_lp}")
        a_lp, score_lp = _alternating_lp(a_ds, 400, polish_deadline,
                                         blend_init=0.5, blend_decay=0.99, blend_min=0.02)
        if n_best != n_lp:
            a_up = _interpolate_sequence(a_lp, n_best)
            score_up = _score(a_up)
        else:
            a_up = a_lp
            score_up = score_lp
        if score_up < best_score:
            best_score = score_up
            best_a = a_up.copy()
            print(f"  ** New best after LP: {best_score:.8f}")
        else:
            # Use the LP result for annealing even if not better (different local min)
            best_a = a_up.copy()  # Keep n_best version for consistency

        # Annealing restarts
        if time.monotonic() < total_deadline - 80:
            print(f"  Annealing restarts at n={len(best_a)}")
            a_ann, score_ann, _ = _annealing_restart(best_a, best_score,
                                                      total_deadline - 50, rng, n_restarts=4)
            if score_ann < best_score:
                best_score = score_ann
                best_a = a_ann.copy()
                print(f"  ** New best after annealing: {best_score:.8f}")

    # Phase 4: Try larger n
    print("\n=== Phase 4: Larger n ===")
    if best_a is not None and time.monotonic() < total_deadline - 100:
        for n_large in [600, 800, 1000]:
            if time.monotonic() > total_deadline - 50:
                break
            a_init = _interpolate_sequence(best_a, n_large)
            phase_deadline = min(total_deadline - 30, time.monotonic() + 120.0)
            print(f"  n={n_large}: ", end="", flush=True)
            a_opt, score = _alternating_lp(a_init, 200, phase_deadline,
                                           blend_init=0.5, blend_decay=0.99, blend_min=0.02)
            print(f"score={score:.8f}")
            if score < best_score:
                best_score = score
                best_a = a_opt.copy()
                print(f"  ** New best: {best_score:.8f} (n={n_large})")

    # Final: one more gradient polish + LP at the best n
    if best_a is not None and time.monotonic() < total_deadline - 20:
        final_deadline = total_deadline - 5
        a_opt, score = _gradient_polish(best_a, final_deadline,
                                        beta_start=100, beta_end=1000, max_iters=50000)
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()

    # Final verification
    print(f"\n=== Final: score={best_score:.8f}, n={len(best_a) if best_a is not None else 0} ===")
    if best_a is not None:
        result = [float(max(0.0, x)) for x in best_a]
        verified = float(evaluate_sequence(result))
        print(f"Verified: {verified:.8f}")
        return result

    return [1.0] * 100
