"""
Search for a non-negative coefficient sequence minimizing
    2 * n * max(coeff(P(z)^2)) / (sum(coeff))^2
where P(z) = sum_k a_k z^k.

This implementation combines:
  * strong finite-length initial candidates, including one-sided regularized
    power-law shapes, central-binomial prefixes, tapers, modulations, and
    resamplings of the latest known construction;
  * a mirror-descent / active-set style optimizer for the maximum coefficient
    of the polynomial square, with adaptive step sizes, adaptive weighting of
    tight convolution coefficients, and periodic stochastic reseeding;
  * a final random multiplicative block local search.

The returned sequence is normalized so that its sum is 1.
"""

import numpy as np
import time

_MAX_N = 8192
_DEFAULT_N = 3809
_FLOOR = 1e-18


def _next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _normalize(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    v = np.maximum(v, _FLOOR)
    s = float(v.sum())
    if not np.isfinite(s) or s <= 1e-300:
        n = max(1, int(v.size))
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    v /= s
    return v


def _score_and_fft(a, L=None):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.inf, None, 0.0, None

    if L is None:
        L = _next_pow2(2 * n - 1)

    A = np.fft.rfft(a, n=L)
    C = A * A
    b = np.fft.irfft(C, n=L)[:2 * n - 1]
    np.maximum(b, 0.0, out=b)

    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.inf, b, 0.0, A

    M = float(np.max(b)) if b.size else 0.0
    if not np.isfinite(M) or M <= 0.0:
        return np.inf, b, 0.0, A

    sc = 2.0 * float(n) * M / (s * s)
    return float(sc), b, M, A


def _score_fast(a, L=None):
    sc, _, _, _ = _score_and_fft(a, L)
    return sc


def _power_unnorm(n, p, off):
    i = np.arange(n, dtype=np.float64)
    x = np.maximum(i + float(off), 1e-12)
    return np.power(x, -float(p))


def _power_taper_unnorm(n, p, off, right_q, right_off):
    i = np.arange(n, dtype=np.float64)
    left = np.power(np.maximum(i + float(off), 1e-12), -float(p))
    right_base = np.maximum((float(n) - 1.0 - i) + float(right_off), 1e-12)
    right = np.power(right_base, -float(right_q))
    return left * right


def _central_binomial_unnorm(n):
    v = np.empty(n, dtype=np.float64)
    if n <= 0:
        return v
    v[0] = 1.0
    for k in range(1, n):
        v[k] = v[k - 1] * float(2 * k - 1) / float(2 * k)
    return v


def _upscale(a, N):
    a = _normalize(a)
    m = int(a.size)
    N = int(N)
    if N <= 0:
        return np.empty(0, dtype=np.float64)
    if m == N:
        return a.copy()
    if m == 1:
        val = float(a[0]) if np.isfinite(a[0]) else 0.0
        return np.full(N, val, dtype=np.float64)

    x0 = np.linspace(0.0, 1.0, m, dtype=np.float64)
    x1 = np.linspace(0.0, 1.0, N, dtype=np.float64)
    out = np.interp(x1, x0, a)
    return _normalize(out)


def _generate_candidates(n, base_list):
    cands = []

    n = int(n)
    if n <= 0:
        return cands

    for base in base_list:
        if base is None:
            continue
        arr = np.asarray(base, dtype=np.float64)
        if arr.size == 0:
            continue
        if arr.size == n:
            cands.append(_normalize(arr))
        else:
            cands.append(_upscale(arr, n))

    cands.append(np.full(n, 1.0 / float(n), dtype=np.float64))

    # One-sided power laws.
    for p in (0.40, 0.45, 0.48, 0.50, 0.52, 0.55, 0.60):
        for off in (0.5, 1.0, 1.5):
            cands.append(_normalize(_power_unnorm(n, p, off)))

    # Mild right tapers.
    for p in (0.45, 0.50, 0.55):
        for qr in (0.15, 0.35, 0.70):
            cands.append(_normalize(_power_taper_unnorm(n, p, 1.0, qr, 0.5)))

    # Weak two-sided variants.
    for p in (0.45, 0.50, 0.55):
        for tw in (0.08, 0.25, 0.70):
            i = np.arange(n, dtype=np.float64)
            left = np.power(np.maximum(i + 1.0, 1e-12), -float(p))
            right = np.power(np.maximum(float(n) - i, 1e-12), -float(p))
            v = left + float(tw) * right
            cands.append(_normalize(v))

    # Central-binomial prefix: its square has a long exact constant plateau.
    cb = _normalize(_central_binomial_unnorm(n))
    cands.append(cb)

    # Mixes of central-binomial and one-sided power.
    p5 = _normalize(_power_unnorm(n, 0.5, 1.0))
    for mix in (0.25, 0.50, 0.75):
        v = (1.0 - mix) * cb + mix * p5
        cands.append(_normalize(v))

    # Low-frequency modulations on a one-sided power law.
    i = np.arange(n, dtype=np.float64)
    denom = max(1, n - 1)
    for amp in (0.15, 0.30):
        for freq in (2.0, 4.0, 8.0):
            mod = 1.0 + amp * np.abs(np.sin(freq * i / float(denom)))
            v = np.power(np.maximum(i + 1.0, 1e-12), -0.5) * mod
            cands.append(_normalize(np.maximum(v, 0.0)))

    # A few deterministic randomized structured starts.
    for seed in (101, 202, 303):
        rng = np.random.default_rng(seed)
        v = np.exp(rng.normal(0.0, 0.45, size=n))
        v *= np.power(np.maximum(i + 1.0, 1e-12), -0.5)
        cands.append(_normalize(v))

    return cands


def _score_candidates(cands, L):
    scored = []
    for i, a in enumerate(cands):
        arr = np.asarray(a, dtype=np.float64)
        sc = _score_fast(arr, L)
        if np.isfinite(sc):
            scored.append((float(sc), i, arr))
    scored.sort()
    return scored


def _mirror_search(a0, end_time, seed, label="mirror"):
    a = _normalize(a0)
    n = int(a.size)
    L = _next_pow2(2 * n - 1)

    best = a.copy()
    best_sc = _score_fast(a, L)
    start = time.monotonic()

    rng = np.random.default_rng(seed)
    alpha = 0.12
    q_decay = 1.0
    no_accept = 0
    last_best_time = time.monotonic()
    last_print = time.monotonic()

    stage_q = (2.0, 5.0, 12.0, 30.0, 80.0)
    stage_limits = (0.20, 0.45, 0.70, 0.90)
    stage = 0

    while time.monotonic() < end_time:
        rem = end_time - time.monotonic()
        if rem < 0.75:
            break

        sc, b, M, A = _score_and_fft(a, L)
        if not np.isfinite(sc) or M <= 0.0 or b is None:
            a = best.copy()
            sc = _score_fast(a, L)
            continue

        if sc < best_sc - 1e-14:
            best = a.copy()
            best_sc = float(sc)
            last_best_time = time.monotonic()
            no_accept = 0
        else:
            no_accept += 1

        if time.monotonic() - last_print > 25.0:
            print(
                f"    {label}: t={time.monotonic()-start:7.1f}s, "
                f"current={sc:.10f}, best={best_sc:.10f}, "
                f"alpha={alpha:.4f}, q_decay={q_decay:.2f}"
            )
            last_print = time.monotonic()

        frac = min(1.0, (time.monotonic() - start) / max(1e-6, end_time - start))
        while stage < len(stage_limits) and frac > stage_limits[stage]:
            stage += 1
        q = float(stage_q[stage]) * float(q_decay)
        if no_accept > 50:
            q *= 1.0 + 0.05 * min(10.0, float(no_accept) / 10.0)
        q = float(min(150.0, max(1.5, q)))

        z = b / max(M, 1e-300)
        z = np.maximum(z, 1e-12)

        # Occasionally use a hard active set; otherwise use soft active set.
        if rng.random() < 0.20:
            thresh = 0.999
            if no_accept > 100:
                thresh = 0.995
            w = (b >= M * thresh).astype(np.float64)
            cnt = float(w.sum())
            if cnt <= 0.0:
                w[int(np.argmax(b))] = 1.0
                cnt = 1.0
        else:
            w = np.power(z, q)
            cnt = float(w.sum())
            if not np.isfinite(cnt) or cnt <= 0.0:
                w = np.full(b.size, 1.0 / float(b.size), dtype=np.float64)
                cnt = 1.0

        E0 = float(np.dot(w, b))
        if not np.isfinite(E0) or E0 <= 1e-300:
            w = np.full(b.size, 1.0 / float(b.size), dtype=np.float64)
            E0 = 1.0

        # Scale so that sum_k w_k b_k = M.
        w = w * (M / E0)

        W = np.fft.rfft(w, n=L)
        h = np.fft.irfft(W * A.conj(), n=L)[:n]
        np.maximum(h, 0.0, out=h)

        r = h / max(M, 1e-300)
        if not np.all(np.isfinite(r)):
            r = a.copy()

        # Remove a weighted-average offset to make the multiplicative step stable.
        r_shift = float(np.dot(a, r))
        if not np.isfinite(r_shift) or r_shift <= 1e-15:
            r_shift = float(np.median(r))
        rr = r - r_shift
        rr = np.clip(rr, -9.0, 9.0)

        # Try a few line-search step sizes.
        best_cand = a
        best_cand_sc = float(sc)
        for al in (alpha, 0.5 * alpha, 0.25 * alpha):
            if al <= 1e-15:
                continue
            e = np.exp(-al * rr)
            cand = np.maximum(a * e, _FLOOR)
            cand = _normalize(cand)
            sc_try = _score_fast(cand, L)
            if sc_try < best_cand_sc - 1e-14:
                best_cand_sc = float(sc_try)
                best_cand = cand

        if best_cand_sc < sc - 1e-14:
            a = best_cand
            sc = float(best_cand_sc)
            if sc < best_sc - 1e-14:
                best = a.copy()
                best_sc = float(sc)
                last_best_time = time.monotonic()
                no_accept = 0
                alpha = min(0.60, alpha * 1.05)
                q_decay = min(1.0, q_decay * 1.08)
            else:
                no_accept += 1
        else:
            no_accept += 1
            alpha = max(0.02, alpha * 0.85)

            # If stalled, perturb the working point.  The stored best is kept.
            stall = time.monotonic() - last_best_time
            if no_accept > 80 or stall > 45.0:
                sigma = 0.015 + 0.08 * min(1.0, float(no_accept) / 1000.0)
                arr = best * np.exp(rng.normal(0.0, sigma, size=n))

                # Sometimes mix back a robust one-sided shape to escape very bad basins.
                if rng.random() < 0.30:
                    p = 0.45 + 0.20 * rng.random()
                    off = 0.5 + 1.5 * rng.random()
                    res = _normalize(_power_unnorm(n, p, off))
                    arr = 0.60 * arr + 0.40 * res

                a = _normalize(np.maximum(arr, _FLOOR))
                sc = _score_fast(a, L)
                no_accept = 30
                alpha = max(0.08, alpha * 0.75)
                q_decay = max(0.25, q_decay * 0.55)

            if no_accept > 1500:
                break

    return best, float(best_sc)


def _block_search(a0, end_time, seed, label="block"):
    a = _normalize(a0)
    n = int(a.size)
    L = _next_pow2(2 * n - 1)

    best = a.copy()
    best_sc = _score_fast(a, L)
    current_sc = best_sc

    rng = np.random.default_rng(seed)
    sigma = 0.09
    no = 0
    start = time.monotonic()
    last_best = start
    last_print = start

    log_lo = np.log(2.0)
    max_size = max(4, n // 25)
    log_hi = np.log(float(max_size))

    while time.monotonic() < end_time:
        rem = end_time - time.monotonic()
        if rem < 0.75:
            break

        if current_sc < best_sc - 1e-14:
            best = a.copy()
            best_sc = float(current_sc)
            last_best = time.monotonic()
            no = 0

        if time.monotonic() - last_print > 25.0:
            print(
                f"    {label}: t={time.monotonic()-start:7.1f}s, "
                f"current={current_sc:.10f}, best={best_sc:.10f}, sigma={sigma:.4f}"
            )
            last_print = time.monotonic()

        nblocks = int(rng.integers(1, 3))
        cand = a.copy()
        for _ in range(nblocks):
            w = int(np.exp(rng.uniform(log_lo, log_hi)))
            w = max(1, min(w, n))
            sidx = int(rng.integers(0, n - w + 1))
            factor = float(np.exp(rng.normal(0.0, sigma)))
            cand[sidx:sidx + w] *= factor

        cand = _normalize(np.maximum(cand, _FLOOR))
        sc_try = _score_fast(cand, L)

        if sc_try < current_sc - 1e-14:
            a = cand
            current_sc = float(sc_try)
            if sc_try < best_sc - 1e-14:
                best = cand.copy()
                best_sc = float(sc_try)
                last_best = time.monotonic()
                no = 0
                sigma = min(0.35, sigma * 1.06)
            else:
                no = max(0, no - 2)
        else:
            no += 1
            if no > 30:
                sigma = max(0.015, sigma * 0.78)

            if time.monotonic() - last_best > 70.0 and time.monotonic() - start > 40.0:
                sigma2 = 0.02 + 0.08 * min(1.0, float(no) / 1000.0)
                a = _normalize(best * np.exp(rng.normal(0.0, sigma2, size=n)))
                current_sc = _score_fast(a, L)
                no = 15
                sigma = 0.10

    return best, float(best_sc)


def _optimize_start(a0, end_time, seed, label):
    now = time.monotonic()
    if end_time <= now:
        return _normalize(a0), _score_fast(a0)

    # Use most of the allocation for mirror descent, a little for local polish.
    block_time = max(6.0, 0.10 * (end_time - now))
    mirror_end = end_time - block_time

    a, sc = _mirror_search(a0, mirror_end, seed, label + "/mirror")
    if time.monotonic() < end_time:
        a, sc = _block_search(a, end_time, seed + 1009, label + "/block")
    return a, sc


def _run_sorted_starts(scored, end_time, seed_base, label, max_starts=3, fracs=(0.55, 0.30, 0.25)):
    best_a, best_sc = scored[0][2], scored[0][0]
    starts = scored[:max_starts]

    for i in range(len(starts)):
        if time.monotonic() >= end_time:
            break

        rem = max(0.0, end_time - time.monotonic())
        if i == len(starts) - 1:
            sub_end = end_time
        else:
            frac = fracs[i] if i < len(fracs) else 0.0
            sub_end = time.monotonic() + max(1.0, frac * rem)

        sc0, idx0, a0 = starts[i]
        a1, sc1 = _optimize_start(a0, sub_end, seed_base + 1000 * (i + 1) + idx0, label + f"._{i}")
        if np.isfinite(sc1) and sc1 < best_sc:
            best_a = a1
            best_sc = float(sc1)
            print(f"  {label}: start {i} improved to {best_sc:.10f}")

    return best_a, best_sc

                global_a = a
                global_sc = float(sc)

    # Final global local search on the best sequence found so far.
    if time.monotonic() < deadline:
        a, sc = _block_search(global_a, deadline, int(seed) + 54321, "final_block")
        if np.isfinite(sc) and sc < global_sc:
            global_a = a
            global_sc = float(sc)

    global_a = _normalize(global_a)

    # Build a clean Python list.
    out = []
    for x in global_a:
        y = float(x)
        if not np.isfinite(y) or y < 0.0:
            y = 0.0
        if y > 1000.0:
            y = 1000.0
        out.append(y)

    # Optional official sanity check.  This is called at most once.
    try:
        val = evaluate_sequence(out)
        print(f"official evaluate_sequence: {val if np.isfinite(val) else 'inf'}")
        if not np.isfinite(val):
            fallback = _normalize(_power_unnorm(len(out), 0.5, 1.0))
            out = [float(max(0.0, min(1000.0, x))) for x in fallback]
    except Exception:
        pass

    return out
