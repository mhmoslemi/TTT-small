def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def normalize_positive(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, float(floor), None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def make_uniform(n):

def resample_seq(a, n):
    a = normalize_positive(a, floor=0.0)
    old_n = a.size

        return np.full(n, 1.0 / float(n), dtype=np.float64)

        return a.astype(np.float64, copy=True)
    old_x = np.linspace(0.0, 1.0, old_n, dtype=np.float64)
    new_x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    b = np.interp(new_x, old_x, a)
    return normalize_positive(b, floor=1e-12)


def compute_conv_and_grad(a, tau, N):
    """Compute convolution, score, softmax weights, and gradient wrt a.
    Returns (c, cmax, score, w, grad_a) where grad_a = d(score)/da."""
    n = len(a)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    cmax = float(np.max(c))
    score = 2.0 * n * cmax
    # Softmax weights
    z = (c - cmax) / max(tau, 1e-15)
    z = np.clip(z, -100.0, 0.0)
    ez = np.exp(z)
    s = float(ez.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)
    else:
        w = ez / s
    # Gradient: 2n * sum_k w[k] * dc[k]/da
    w_rev = w[::-1]
    fw = np.fft.rfft(w_rev, N)
    conv = np.fft.irfft(fa * fw, N)
    grad_a = 2.0 * n * (2.0 * conv[n - 1:2 * n - 1][::-1])
    return c, cmax, score, w, grad_a


def adam_step(z, m_z, v_z, step, lr, beta1, beta2, eps, grad_z):
    """Perform an Adam step on z, returning updated z, m_z, v_z."""
    m_z = beta1 * m_z + (1 - beta1) * grad_z
    v_z = beta2 * v_z + (1 - beta2) * (grad_z * grad_z)
    m_hat = m_z / (1 - beta1 ** step)
    v_hat = v_z / (1 - beta2 ** step)
    z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
    return z, m_z, v_z


def optimize_adam(a_init, deadline, seed, tau_start=0.5, tau_end=0.03,
                  lr_start=0.08, lr_end=0.005, max_iters=200000):
    """Adam optimization on smooth max objective.
    a_init: initial normalized sequence (sum=1).
    Returns (best_a, best_score)."""
    n = len(a_init)
    N = next_pow2(3 * n - 2)
    z = np.log(a_init + 1e-20)
    z -= z.mean()
    m_z = np.zeros(n, dtype=np.float64)
    v_z = np.zeros(n, dtype=np.float64)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    best_score = 2.0 * n * float(np.max(np.convolve(a_init, a_init, mode='full')))
    best_a = a_init.copy()
    step = 0
    stagnant = 0
    no_improve = 0

    while time.monotonic() < deadline and step < max_iters:
        # Anneal parameters
        frac = min(1.0, step / float(max_iters))
        tau = tau_start * (tau_end / tau_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac

        # Compute a from z
        z_max = float(np.max(z))
        a = np.exp(z - z_max)
        a = np.maximum(a, 1e-300)
        a = a / float(a.sum())

        # Compute gradient
        c, cmax, score, w, grad_a = compute_conv_and_grad(a, tau, N)

        if score < best_score * (1.0 - 1e-10):
            best_score = score
            best_a = a.copy()
            no_improve = 0
            stagnant = 0
        else:
            no_improve += 1

        # Convert to z-gradient: dF/dz_j = a_j * (sum_i grad_a[i]*a_i - grad_a[j])
        dot_ha = float(np.dot(grad_a, a))
        grad_z = a * (dot_ha - grad_a)

        # Normalize gradient for stability
        gnorm = float(np.linalg.norm(grad_z))
        if gnorm > 1e-12:
            grad_z = grad_z / gnorm * 0.1

        # Adam step
        step += 1
        z, m_z, v_z = adam_step(z, m_z, v_z, step, lr, beta1, beta2, eps, grad_z)

        # Periodic: check if stuck and restart
        if no_improve > 5000:
            # Try a random perturbation
            rng = np.random.default_rng(step + seed)
            perturb = rng.normal(0, 0.3, n)
            z = z + perturb
            m_z = np.zeros(n, dtype=np.float64)
            v_z = np.zeros(n, dtype=np.float64)
            step = 0
            no_improve = 0
            beta1, beta2 = 0.9, 0.999

        if no_improve > 50000:
            break

    # Final: return best
    best_a = normalize_positive(best_a, floor=0.0)
    return best_a, best_score

    print("fallback score = {:.9f}".format(best_score), flush=True)

    given = get_given_sequence()
    if given is not None:
        sc = 2.0 * len(given) * float(np.max(np.convolve(given, given, mode='full')))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = given.copy()
            print("given sequence: n={} score={:.9f}".format(len(given), best_score), flush=True)

    # Phase 1: Quick Adam optimization on given sequence (n=2048)
    if time.monotonic() < deadline - 900:
        n1 = len(best_a)
        if n1 >= 256:
            t1 = deadline - 900
            a1, sc1 = optimize_adam_with_sa(best_a, t1, seed=seed + 1,
                                            tau_start=0.5, tau_end=0.03,
                                            lr_start=0.07, lr_end=0.004,
                                            max_total_iters=200000)
            if sc1 < best_score:
                best_score = sc1
                best_a = a1.copy()
                print("phase1 adam: n={} score={:.9f}".format(n1, best_score), flush=True)

    # Phase 2: Try different n values with Adam
    for target_n in [1024, 2048, 4096, 4095, 2047]:
        if time.monotonic() > deadline - 600:
            break
        allow = min(90.0, deadline - time.monotonic() - 500)
        if allow < 25:
            break
        a0 = resample_seq(best_a, target_n)
        a_opt, sc = optimize_adam(a0, time.monotonic() + allow, seed=seed + 100 + target_n,
                                   tau_start=0.35, tau_end=0.025,
                                   lr_start=0.05, lr_end=0.003,
                                   max_iters=60000)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print("phase2 n={} score={:.9f}".format(target_n, best_score), flush=True)

    # Phase 3: Binary optimization
    for target_n in [2048, 4096, 8192]:
        if time.monotonic() > deadline - 300:
            break
        allow = min(150.0, deadline - time.monotonic() - 200)
        if allow < 30:
            break
        a0 = resample_seq(best_a, target_n)
        # Also try a structured binary start (quadratic residues)
        sq = np.zeros(target_n, dtype=np.float64)
        for x in range(target_n):
            if (x * x) % target_n < target_n // 2:
                sq[(x * x) % target_n] = 1.0
        b_opt, b_sc = optimize_binary改进(a0, time.monotonic() + allow // 2, seed=seed + 5000 + target_n)
        if b_sc < best_score:
            best_score = b_sc
            best_a = b_opt.copy()
            print("phase3 binary n={} score={:.9f}".format(target_n, best_score), flush=True)

        if time.monotonic() < deadline - 200:
            b_opt2, b_sc2 = optimize_binary改进(sq, time.monotonic() + allow // 2, seed=seed + 6000 + target_n)
            if b_sc2 < best_score:
                best_score = b_sc2
                best_a = b_opt2.copy()
                print("phase3 binary-sq n={} score={:.9f}".format(target_n, best_score), flush=True)

    # Phase 4: Final intensive Adam on best candidate
    if time.monotonic() < deadline - 20:
        allow = deadline - time.monotonic() - 20
        if allow > 30:
            a_final, sc_final = optimize_adam_with_sa(
                best_a, time.monotonic() + allow, seed=seed + 99999,
                tau_start=max(0.05, 0.3 * (best_score / (2.0 * len(best_a)))),
                tau_end=0.01,
                lr_start=0.03, lr_end=0.002,
                max_total_iters=100000)
            if sc_final < best_score:
                best_score = sc_final
                best_a = a_final.copy()
                print("phase4 final: score={:.9f}".format(best_score), flush=True)

    # Resample to a good size if needed
    n_final = len(best_a)
    if n_final < 512:
        best_a = resample_seq(best_a, 1024)
        best_score = 2.0 * 1024 * float(np.max(np.convolve(best_a, best_a, mode='full')))

    best_a = normalize_positive(best_a, floor=0.0)
    out_list = [float(x) for x in best_a]

    # Final validation
    try:
        final_eval = float(evaluate_sequence(out_list))
    except Exception:
        final_eval = float('inf')

    if not np.isfinite(final_eval):
        best_a = make_uniform(1024)
        out_list = [float(x) for x in best_a]
        final_eval = 2.0

    print("FINAL: score={:.12f}, n={}".format(final_eval, len(out_list)), flush=True)
    return out_list
