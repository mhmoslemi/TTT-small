    """Compute 2*n*max(conv(a,a)) / sum(a)^2"""

"""
Optimized search for non-negative sequences minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-based gradient computation with natural gradient correction and adaptive smoothing.
"""

import time
import numpy as np

def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()

def normalize_positive(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s

def fast_score(a):
    """Compute scale-invariant score: 2*n*max(conv(a,a))/sum(a)^2"""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n < 1:
        return np.inf
    s = float(a.sum())
    if s <= 0:
        return np.inf
    a = a / s  # normalize
    L = next_pow2(2 * n - 1)
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0:
        return np.inf
    return float(2.0 * n * cm)

def compute_gradient(a, tau, L):
    """
    Compute the gradient of the smooth max objective w.r.t. a (on the simplex).
    Returns (true_score, smooth_score, grad) where grad is the raw gradient
    (tangent to the simplex, i.e., sum(grad) = 0).
    """
    n = a.size
    # a is assumed normalized (sum = 1)
    
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    
    cm = float(c.max())
    if cm <= 0 or not np.isfinite(cm):
        return np.inf, np.inf, np.zeros(n)
    
    # Smooth max weights
    z = (c - cm) / tau
    # Clip for numerical stability
    z = np.clip(z, -50.0, 0.0)
    expz = np.exp(z)
    expsum = float(expz.sum())
    if expsum <= 0 or not np.isfinite(expsum):
        # Fallback: uniform weights on top entries
        k = max(1, int(0.05 * len(c)))
        idx = np.argsort(c)[-k:]
        expz = np.zeros_like(c)
        expz[idx] = 1.0
        expsum = float(k)
    
    w = expz / expsum  # probability distribution over convolution entries
    
    # Compute gradient: grad_p[i] = sum_j a[j] * w[i+j] (cross-correlation)
    # Using FFT: grad_p = ifft(fft(a) * conj(fft(w_rev)))... 
    # Actually: grad_p[i] = sum_{j=0}^{n-1} a[j] * w[i+j]
    # This is the cross-correlation of a and w at positive lags.
    # 
    # Method: rev_w = w[::-1] (length 2n-1)
    # conv(a, rev_w) has length n + 2n-2 = 3n-2
    # grad_p[i] = conv_result[2n-2-i] for i=0..n-1
    # So grad_p = conv_result[n-1:2n-1][::-1]
    
    L_conv = next_pow2(3 * n - 2)
    rev_w = w[::-1]
    A_f = np.fft.rfft(a, L_conv)
    W_f = np.fft.rfft(rev_w, L_conv)
    conv_result = np.fft.irfft(A_f * W_f, L_conv)[:3 * n - 2]
    
    grad_p = conv_result[n-1:2*n-1][::-1]
    grad_p = np.where(np.isfinite(grad_p), grad_p, 0.0)
    
    # Project to tangent space of simplex: subtract mean weighted by a
    # Actually, for the constraint sum(a)=1, the tangent space is {v: sum(v)=0}
    # The projection is: grad - (sum(a * grad) / sum(a^2)) * a ... no
    # For the simplex constraint, the tangent direction should satisfy sum(v) = 0
    # So: grad_tangent = grad_p - mean(grad_p)
    # But wait, we're on the probability simplex, so the constraint is sum(a) = 1.
    # The projection of a gradient onto the tangent space is:
    # v = grad_p - (sum_i grad_p_i) / n * 1 (if we use the Euclidean metric)
    # But for the "natural" metric on the simplex (Fisher information), it's different.
    # For simplicity, let's just subtract the mean:
    mean_g = float(np.mean(grad_p))
    grad = grad_p - mean_g
    
    # Alternative: subtract a-weighted mean (for the log-domain parameterization)
    # a_mean_g = float(np.dot(a, grad_p) / float(np.dot(a, a)))  # not standard
    # Or simply: a_mean_g = float(np.dot(a, grad_p))
    # grad = grad_p - a * a_mean_g  (this is the gradient w.r.t. log-domain z)
    
    true_score = 2.0 * n * cm
    smooth_score = 2.0 * n * (cm + tau * float(np.log(expsum)))
    
    return float(true_score), float(smooth_score), grad

def compute_z_gradient(a, tau, L):
    """
    Compute the gradient of the smooth objective w.r.t. the log-domain parameter z.
    If a = softmax(z), then d_obj/dz_i = a_i * (d_obj/da_i - sum_j a_j * d_obj/da_j)
    (times the appropriate scaling for the normalized case)
    
    This is the "natural gradient" which accounts for the exponential parameterization.
    """
    n = a.size
    _, _, grad_p = compute_gradient(a, tau, L)
    
    # d_obj/dz_i = a_i * (grad_p_i - sum_j a_j * grad_p_j)
    # (since the objective depends on a only through the normalized a, and
    #  d(softmax(z))/dz_j = a_j * (delta_ij - a_i))
    mean_a_g = float(np.dot(a, grad_p))
    z_grad = a * (grad_p - mean_a_g)
    
    return z_grad

def cosine_shape(n, exponent=2.0):
    """Create a cosine-like shape: (1 - cos(2*pi*x/2))^exponent for x in [0,1]"""
    x = np.linspace(0, 1, n)
    h = (1 - np.cos(np.pi * x)) ** exponent  # = sin^2(pi*x/2)^... 
    # Actually: 1 - cos(pi*x) = 2*sin^2(pi*x/2), so this is 2^exponent * sin^(2*exponent)(pi*x/2)
    return normalize_positive(h)

def ramp_shape(n, power=1.0):
    """Create a power-law ramp: x^power for x in [0,1]"""
    x = np.linspace(0, 1, n)
    h = x ** power
    return normalize_positive(h)

def two_bump_shape(n, alpha=0.3, beta=0.7, width=0.1):
    """Two bumps at positions alpha and beta"""
    x = np.linspace(0, 1, n)
    h = np.exp(-(x-alpha)**2/(2*width**2)) + 0.8*np.exp(-(x-beta)**2/(2*width**2))
    return normalize_positive(h)

def cheb_shape(n, coeffs, mode=0, eps=0.1):
    """Create a shape from Chebyshev polynomial coefficients"""
    D = len(coeffs)
    x = np.linspace(0, 1, n)
    t = 2*x - 1
    t = np.clip(t, -1, 1)
    ang = np.arccos(t)
    val = np.zeros(n)
    for m_idx, c in enumerate(coeffs):
        val += c * np.cos((m_idx+1) * ang)
    if mode == 0:
        h = 1.0 + eps + val
    else:
        h = np.exp(np.clip(val, -8, 8))
    return normalize_positive(h)

def adam_optimize(p0, duration_s, tau_init, tau_min, lr_init, lr_decay,

    Adam optimizer in the log-domain with adaptive tau.
    """
    rng = np.random.default_rng(seed)
    n = p0.size
    L = next_pow2(2 * n - 1)
    L_conv = next_pow2(3 * n - 2)

    z = np.log(np.maximum(p0, 1e-300))
    z = np.where(np.isfinite(z), z, -50.0)
    z -= z.mean()
    z = np.clip(z, -50.0, 50.0)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-15
    step_count = 0
    
    tau = float(tau_init)
    lr = float(lr_init)
    
    best_p = p0.copy()
    best_score = fast_score(p0)
    no_improve = 0
    
    deadline = time.monotonic() + duration_s
    start = time.monotonic()
    
    while time.monotonic() < deadline:

        tau = max(tau_min, tau * 0.9995)  # slow decay

        # Compute gradient
        true_s, smooth_s, grad = compute_gradient(p0, tau, L)
        if not np.isfinite(true_s):
            break
        
        # Update best
        if true_s < best_score - 1e-10:
            best_score = true_s
            best_p = p0.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute z-gradient (natural gradient)
        mean_a_g = float(np.dot(p0, grad))
        z_grad = p0 * (grad - mean_a_g)
        z_grad = np.where(np.isfinite(z_grad), z_grad, 0.0)
        
        # Clip gradient for stability
        max_g = np.max(np.abs(z_grad))
        if max_g > 1.0:
            z_grad *= 1.0 / max_g
        
        # Adam update
        step_count += 1
        m = beta1 * m + (1 - beta1) * z_grad
        v = beta2 * v + (1 - beta2) * z_grad ** 2
        m_hat = m / (1 - beta1 ** step_count)
        v_hat = v / (1 - beta2 ** step_count)
        
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        z = np.where(np.isfinite(z), z, 0.0)
        z -= z.mean()
        z = np.clip(z, -50.0, 50.0)
        
        p0 = _softmax(z)
        
        # Restart if stuck

            if verbose:

            # Perturb from best

            z = np.where(np.isfinite(z), z, -50.0)
            z += rng.normal(0, 0.3, n)
            z -= z.mean()
            z = np.clip(z, -50, 50)
            m *= 0.1
            v *= 0.1

def _softmax(z):
    z = np.clip(z, -50, 0)
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s <= 0:
        return np.ones(len(z)) / len(z)
    return e / s
