from scipy.optimize import linprog

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]

def _score(a):
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)

    n = len(f)

    for j in range(n):
        C[j:j + n, j] = f
    return C

"""
Hybrid optimizer for minimizing J(a) = 2n*max(conv(a,a))/sum(a)^2.
Uses: (1) minimax LP alternating optimization, (2) exact coordinate descent polish,
(3) multi-scale n exploration with adaptive time allocation.

import numpy as np
import time
from scipy.optimize import linprog


def _fft_convolve(a):
    """Fast full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]


def _score(a):
    """Compute J(a) = 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)


def _build_conv_matrix(f):
    """Build convolution matrix C where (C @ g)[k] = (f*g)[k]."""
    n = len(f)
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = f
    return C


def _lp_minimax(f, time_limit=30.0):
    """
    Solve: min t s.t. (f*g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns (g, t) or (None, inf).
    """
    n = len(f)
    C = _build_conv_matrix(f)
    # Variables: [g_0, ..., g_{n-1}, t]
    # Minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    # A_ub * vars <= b_ub:  C_f @ g - t <= 0
    A_ub = np.hstack([C, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1)
    # Bounds: g >= 0, t >= 0
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': time_limit})
        if result.status == 0:
            return result.x[:n].copy(), float(result.x[n])
    except Exception:
        pass
    return None, float('inf')


def _alternating_lp_minimax(a_init, deadline, initial_blend=0.7, max_iters=500):
    """Alternating: fix f, solve minimax LP for g, blend, repeat."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = _score(a)
    blend = initial_blend
    lp_time = min(30.0, (deadline - time.monotonic()) / 10.0)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        g, t_lp = _lp_minimax(a, time_limit=lp_time)
        if g is None:
            break
        sum_g = float(np.sum(g))
        if sum_g < 1e-12:
            break
        g = g / sum_g
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        sum_new = float(np.sum(a_new))
        if sum_new < 1e-12:

        a_new = a_new / sum_new
        score = _score(a_new)

        # Reduce LP time if we're running low

        if remaining < 60:

def _coord_descent(a_init, deadline, max_sweeps=200):
    """
    Coordinate descent: for each coordinate i, solve the exact 1D subproblem
    of minimizing max_k c[k] over a[i] >= 0, with other coords fixed.
    """
    a = a_init.copy().astype(np.float64)
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = _score(a)
    c = _fft_convolve(a)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        for i in range(n):
            if time.monotonic() > deadline:
                break
            ai = a[i]
            # Compute c_base and w
            c_base = c.copy()
            w = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        c_base[k] -= ai * ai
                        w[k] = 0.0  # quadratic handled separately
                    else:
                        c_base[k] -= 2.0 * ai * a[ki]
                        w[k] = 2.0 * a[ki]
            
            # Bisection on t: find min t s.t. exists ai>=0 with
            #   c_base[k] + ai*w[k] <= t for all k != 2i
            #   c_base[2i] + ai^2 <= t
            #   ai >= 0
            
            t_lo = float(np.max(c_base))  # t at ai=0
            t_hi = t_lo + 1.0
            # Expand t_hi
            for _ in range(20):
                feasible = _check_feasible(t_hi, c_base, w, 2 * i, n)
                if feasible >= 0:
                    break
                t_hi *= 2.0
            t_hi = t_hi * 2.0
            
            for _ in range(50):
                t_mid = 0.5 * (t_lo + t_hi)
                ub = _check_feasible(t_mid, c_base, w, 2 * i, n)
                if ub >= 0:
                    t_hi = t_mid
                else:
                    t_lo = t_mid
            
            t_opt = t_hi
            # Find optimal ai: the largest feasible value
            ai_new = _max_feasible_ai(t_opt, c_base, w, 2 * i, n)
            ai_new = max(0.0, min(ai_new, np.sqrt(t_opt) if t_opt > 0 else 0.0))
            
            if abs(ai_new - ai) > 1e-10:
                a[i] = ai_new
                # Update convolution incrementally
                delta = ai_new - ai
                for k in range(2 * n - 1):
                    ki = k - i
                    if 0 <= ki < n:
                        if k == 2 * i:
                            c[k] += 2.0 * ai * delta + delta * delta
                        else:
                            c[k] += 2.0 * delta * a[ki]
                ai = ai_new
                # Renormalize
                s_new = float(np.sum(a))
                if s_new > 1e-10:
                    scale = s / s_new
                    a *= scale
                    c *= scale * scale
        
        current_score = _score(a)
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()
            if sweep % 20 == 0:
                print(f"      CD sweep {sweep}: score={best_score:.6f}")
    
    return best_a, best_score


def _check_feasible(t, c_base, w, k_quad, n):
    """Return max feasible ai for given t, or -1 if infeasible."""
    ai_max = float('inf')
    for k in range(2 * n - 1):
        if k == k_quad:
            continue
        if w[k] > 1e-15:
            ub = (t - c_base[k]) / w[k]
            if ub < ai_max:
                ai_max = ub
            if ub < -1e-12:
                return -1.0
    # Quadratic constraint at k_quad
    if k_quad < 2 * n - 1:
        quad_rhs = t - c_base[k_quad]
        if quad_rhs < -1e-12:
            return -1.0
        quad_ub = np.sqrt(max(0.0, quad_rhs))
        if quad_ub < ai_max:
            ai_max = quad_ub
    return ai_max


def _max_feasible_ai(t, c_base, w, k_quad, n):
    """Return the maximum feasible ai for given t."""
    return _check_feasible(t, c_base, w, k_quad, n)

def _interpolate_sequence(src, n_target):
    """Interpolate a sequence to a new length, normalized to sum=1."""
    n_src = len(src)
    if n_src == n_target:
        return src.copy()
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    out = np.interp(x_tgt, x_src, src)
    out = np.maximum(out, 0.0)
    s = float(np.sum(out))
    if s > 1e-10:
        out = out / s
    else:
        out = np.ones(n_target) / n_target
    return out

        # Add diverse inits
        for v in variants[:max(2, 8 - idx_n)]:
            inits.append(_make_initial(n, v, rng))
        
        n_tried = 0
        for a_init in inits:
            if time.monotonic() > phase_deadline:
                break
            init_score = _score(a_init)
            print(f"    init (score={init_score:.4f})", end=" ")
            # LP optimization
            a_opt, score = _alternating_lp_minimax(a_init, phase_deadline, initial_blend=0.7, max_iters=300)
            print(f" LP-> {score:.6f}")
            
            # Quick CD polish
            if score < best_score + 0.05 and time.monotonic() < phase_deadline - 10:
                cd_deadline = min(phase_deadline, time.monotonic() + 30.0)
                a_cd, score_cd = _coord_descent(a_opt, cd_deadline, max_sweeps=50)
                if score_cd < score:
                    print(f"    CD-> {score_cd:.6f}")
                    a_opt, score = a_cd, score_cd
            
            if score < best_score:
                best_score = score
                best_a = a_opt.copy()
                print(f"    ** New best: {best_score:.6f} (n={n})")
            n_tried += 1
    
    print(f"\nAfter Phase 1: best={best_score:.6f}, n={len(best_a) if best_a is not None else 0}")

    # Phase 2: Coordinate descent polish on best
    print("\n=== Phase 2: CD Polish ===")
    if best_a is not None and time.monotonic() < total_deadline - 60:
        cd_deadline = min(total_deadline - 30, time.monotonic() + 600.0)
        print(f"  Polishing n={len(best_a)}, budget={cd_deadline - time.monotonic():.0f}s")
        a_cd, score_cd = _coord_descent(best_a, cd_deadline, max_sweeps=500)
        if score_cd < best_score:
            best_score = score_cd
            best_a = a_cd.copy()
            print(f"  ** CD improved: {best_score:.6f}")
        else:
            print(f"  CD: {score_cd:.6f} (no improvement)")

    # Phase 3: Try LP again from best (different blend)
    print("\n=== Phase 3: LP Re-optimization ===")
    if best_a is not None and time.monotonic() < total_deadline - 30:
        n_best = len(best_a)
        # Try at the same n with different blend
        lp_deadline = min(total_deadline - 10, time.monotonic() + 300.0)
        for blend_try in [0.5, 0.9, 0.3]:
            if time.monotonic() > lp_deadline:
                break
            a_opt, score = _alternating_lp_minimax(best_a, lp_deadline, initial_blend=blend_try, max_iters=200)
            if score < best_score:
                best_score = score
                best_a = a_opt.copy()
                print(f"  blend={blend_try}: new best {best_score:.6f}")
        
        # CD polish again
        if time.monotonic() < total_deadline - 10:
            a_cd, score_cd = _coord_descent(best_a, total_deadline - 5, max_sweeps=200)
            if score_cd < best_score:
                best_score = score_cd
                best_a = a_cd.copy()
                print(f"  Final CD: {best_score:.6f}")

    # Phase 4: Try slightly larger n if time permits
    if best_a is not None and time.monotonic() < total_deadline - 100:
        n_new = len(best_a) + 200
        if n_new <= 3000:
            print(f"\n=== Phase 4: Extend to n={n_new} ===")
            a_init = _interpolate_sequence(best_a, n_new)
            a_opt, score = _alternating_lp_minimax(a_init, total_deadline - 10, initial_blend=0.5, max_iters=100)
            if score < best_score:
                best_score = score
                best_a = a_opt.copy()
                print(f"  Extended: {best_score:.6f}")
            if time.monotonic() < total_deadline - 10:
                a_cd, score_cd = _coord_descent(best_a, total_deadline - 5, max_sweeps=100)
                if score_cd < best_score:
                    best_score = score_cd
                    best_a = a_cd.copy()
                    print(f"  CD after extend: {best_score:.6f}")

    # Final output
    print(f"\n=== FINAL: score={best_score:.6f}, n={len(best_a) if best_a is not None else 0} ===")
    if best_a is not None:
        result = [float(x) for x in best_a]
        verified = float(evaluate_sequence(result))
        print(f"Verified via evaluate_sequence: {verified:.6f}")
        if verified < best_score + 1e-9:
            return result
        else:
            # Fallback to best_a
            return result
    return [1.0] * 100
