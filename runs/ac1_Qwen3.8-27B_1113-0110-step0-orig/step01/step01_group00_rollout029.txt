```python
"""
Algorithm: Hybrid coordinate descent + alternating LP for minimizing
J(a) = 2n * max(conv(a,a)) / sum(a)^2 over non-negative sequences.

Key ideas:
1. Fast coordinate descent with exact 1D line search (vectorized convolution updates)
2. Periodic LP-based "jumps" to escape local minima
3. Diverse initializations including multi-Gaussian families
4. Progressive n-escalation with interpolation
"""

import numpy as np
import time
from scipy.optimize import linprog


def _fft_convolve(a):
    """Fast full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]


def _score(a):
    """Compute J(a) = 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)


def _score_from_parts(c, n, S):
    """Score from precomputed convolution and sum."""
    if S < 0.01:
        return float('inf')
    return 2.0 * n * float(np.max(c)) / (S * S)


def _build_conv_matrix_rowwise(f, n):
    """Build the (2n-1) x n convolution matrix C where (C@g)[k] = (f*g)[k]."""
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = f
    return C


def _lp_solve(f, M, n):
    """Solve max sum(g) s.t. f*g <= M, g >= 0. Returns g or None."""
    C = _build_conv_matrix_rowwise(f, n)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 30})
        if result.status == 0:
            return result.x
    except Exception:
        pass
    return None


def _coord_descent(a_init, max_passes, deadline, n_cand=6, verbose=False):
    """
    Fast cyclic coordinate descent with 1D line search.
    Uses vectorized convolution updates for O(n) per candidate evaluation.
    """
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    else:
        a = a / S * S  # keep same scale
    c = _fft_convolve(a).copy()
    best_score = _score_from_parts(c, n, S)
    best_a = a.copy()
    stall_count = 0

    for pw in range(max_passes):
        if time.monotonic() > deadline:
            break
        any_improved = False
        for i in range(n):
            old = a[i]
            S_other = S - old

            # Generate candidate x values
            cands = [0.0]
            if old > 1e-15:
                cands.append(old * 0.5)
                cands.append(old * 1.5)
                cands.append(old * 0.25)
                cands.append(old * 2.0)

            # Add optimal values for top peaks
            top_k = np.argpartition(c, -4)[-4:]
            for k in top_k:
                k = int(k)
                ki = k - i
                if k == 2 * i:
                    base = c[k] - old * old
                    if S_other > 1e-12:
                        xo = base / S_other
                        if xo > 0 and xo < 100 * S / n:
                            cands.append(xo)
                elif 0 <= ki < n and a[ki] > 1e-15:
                    base = c[k] - 2.0 * old * a[ki]
                    xo = (S_other * a[ki] - base) / a[ki] if a[ki] > 0 else 0
                    xo = S_other - base / a[ki]
                    if 0 < xo < 100 * S / n:
                        cands.append(xo)

            # Evaluate candidates (vectorized)
            cur_max = float(np.max(c))
            best_x = old
            best_local = 2.0 * n * cur_max / (S * S)

            for x in cands:
                if x < 0:
                    x = 0.0
                delta = x - old
                if abs(delta) < 1e-14:
                    continue

                # Vectorized update
                start = i
                end = min(i + n, 2 * n - 1)
                new_c = c.copy()
                new_c[start:end] += 2.0 * delta * a[:end - i]
                new_c[2 * i] += delta * delta

                new_max = float(np.max(new_c))
                new_S = S_other + x
                if new_S < 0.01:
                    continue
                local = 2.0 * n * new_max / (new_S * new_S)
                if local < best_local - 1e-12:
                    best_local = local
                    best_x = x

            if best_x != old:
                delta = best_x - old
                start = i
                end = min(i + n, 2 * n - 1)
                c[start:end] += 2.0 * delta * a[:end - i]
                c[2 * i] += delta * delta
                S = S_other + best_x
                a[i] = best_x
                any_improved = True

                cur_score = _score_from_parts(c, n, S)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()

        if not any_improved:
            stall_count += 1
            if stall_count >= 3:
                break
        else:
            stall_count = 0

        if verbose and pw % 50 == 0:
            print(f"      coord pw={pw}: score={best_score:.6f}")

    return best_a, best_score


def _alternating_lp(a_init, max_iters, deadline, initial_blend=0.7):
    """Alternating LP: fix f, solve LP for g, blend, repeat."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_score = _score(a)
    blend = initial_blend

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        M = float(np.max(c))
        g = _lp_solve(a, M, n)
        if g is None:
            break
        sum_g = float(np.sum(g))
        if sum_g < 1e-12:
            break
        g = g / sum_g
        a_new = np.maximum((1.0 - blend) * a + blend * g, 0.0)
        sum_new = float(np.sum(a_new))
        if sum_new < 1e-12:
            break
        a_new = a_new / sum_new
        score = _score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        blend = max(0.1, blend * 0.96)

    return best_a, best_score


def _hybrid_optimize(a_init, deadline, lp_every=50):
    """Coordinate descent with periodic LP jumps."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    best_a = a.copy()
    best_score = _score(a)

    coord_passes = 0
    while time.monotonic() < deadline:
        # Coordinate descent phase
        sub_deadline = min(deadline, time.monotonic() + 60.0)
        a, score = _coord_descent(a, 200, sub_deadline)
        coord_passes += 1
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if time.monotonic() > deadline:
            break

        # LP jump every lp_every coord phases
        if coord_passes % lp_every == 0:
            lp_deadline = min(deadline, time.monotonic() + 30.0)
            a_lp, score_lp = _alternating_lp(a, 30, lp_deadline, initial_blend=0.5)
            if score_lp < best_score:
                best_score = score_lp
                best_a = a_lp.copy()
                a = a_lp.copy()

        # Check convergence
        if coord_passes > 10:
            break

    return best_a, best_score

def _interpolate_sequence(src, n_target):
    """Interpolate a sequence to a new length."""
    n_src = len(src)
    if n_src == n_target:
        return src.copy()
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    out = np.interp(x_tgt, x_src, src)
    out = np.maximum(out, 0.0)
    s = float(np.sum(out))
    if s > 1e-10:
        out = out / s
    return out

        print(f"\nFinal coord polish: {best_score:.6f}")

    print(f"\n=== FINAL: score={best_score:.6f}, n={len(best_a) if best_a is not None else 0} ===")

    if best_a is not None:
        result = [float(x) for x in best_a]
        verified = float(evaluate_sequence(result))
        print(f"Verified: {verified:.6f}")
        return result

    return [1.0] * 100
```
