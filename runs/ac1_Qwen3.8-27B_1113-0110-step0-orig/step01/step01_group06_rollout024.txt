- For anti-palindromic a: A(omega) = (1/n) * sum_i exp(-2πi*omega*i/n) + sum_i d_i * (exp(-2πi*omega*i/n) - exp(-2πi*omega*(n-1-i)/n))

- The first term is a delta at omega=0 (DC component)
- So |A(omega)|^2 ≈ |sum_i d_i * exp(-2πi*omega*i/n) * (1 - exp(-2πi*omega*(n-1)/n))|^2 + (DC term)

"""
Optimize non-negative step function to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses anti-palindromic parameterization with top-K gradient focusing,
adaptive step size, multiple restarts, and coordinate descent refinement.
"""

import numpy as np
import time

def _pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()

def _fft_conv(a, b):
    n = len(a)
    m = len(b)
    L = n + m - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    result = np.fft.irfft(fa * fb, N)[:L]
    return np.maximum(result, 0.0)

def _fft_conv_square(a):
    n = len(a)
    L = 2 * n - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, N)
    b = np.fft.irfft(fa * fa, N)[:L]
    return np.maximum(b, 0.0)

def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    b = _fft_conv_square(a)
    return 2.0 * n * b.max() / (s * s)

def _score_with_b(a, b):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * b.max() / (s * s)

def _anti_palindromic_a(d, n):
    """Construct anti-palindromic sequence from d vector."""
    m = n // 2
    a = np.empty(n, dtype=np.float64)
    c = 1.0 / n
    a[:m] = c + d
    a[m:] = c - d[::-1]
    return a

def _a_to_d(a, n):
    """Extract d from anti-palindromic sequence."""
    m = n // 2
    c = 1.0 / n
    d = (a[:m] - c).copy()
    return d

def _grad_b_wrt_d(b, a, n):
    """Compute gradient of b_k with respect to d for all k.
    Returns array of shape (n, 2n-1) where grad[i, k] = db_k/dd_i.
    But this is too large. Instead, compute gradient of a weighted sum.
    """
    m = n // 2
    # b_k = sum_{i+j=k} a_i * a_j
    # da_i/dd_i = 1 for i < m, da_{n-1-i}/dd_i = -1 for i < m
    # db_k/dd_i = a_{k-i} - a_{n-1-k+(n-1-i)} ... let me think more carefully
    
    # Actually, let's compute the gradient of sum_k w_k * b_k wrt d.
    # sum_k w_k * b_k = sum_k w_k * sum_{i+j=k} a_i * a_j
    # = sum_{i,j} w_{i+j} * a_i * a_j
    # d/dd_i: for i < m, da_i/dd_i = 1, da_{n-1-i}/dd_i = -1
    # d/dd_i [sum_{i',j'} w_{i'+j'} a_{i'} a_{j'}]
    # = sum_{j'} w_{i+j'} a_{j'} - sum_{j'} w_{(n-1-i)+j'} a_{j'}
    # = (w * a)[i] - (w * a)[n-1-i]  (where w*a is the convolution of w with a)
    # Wait, (w * a)[k] = sum_j w_k * a_j? No, that's not standard conv.
    # Let me define: g[k] = sum_j w[k] * a[j] where the sum is over valid j.
    # Actually: sum_{j'} w_{i+j'} * a_{j'} where 0 <= i+j' <= 2n-2 and 0 <= j' <= n-1
    # This is the cross-correlation-like quantity.
    # = (a * w)[i] where we're doing a specific convolution.
    # Let me just compute it directly.
    
    # g[k] = sum_{j valid} w[k-j] * a[j] for k = 0, ..., n-1
    # where "valid" means 0 <= k-j <= 2n-2 and 0 <= j <= n-1
    # This is the convolution of a (length n) with w (length 2n-1)
    # conv(a, w)[k] = sum_j a[j] * w[k-j]
    # We need g[k] = sum_j a[j] * w[k-j] for k = 0, ..., n-1
    # = conv(a, w)[0:n]
    
    # So grad_d[i] = g[i] - g[n-1-i]
    # where g = conv(a, w)[0:n]
    
    return None  # We'll compute differently

def _weighted_grad(w, a, n):
    """Compute gradient of sum_k w[k]*b[k] with respect to d.
    Returns array of length m = n//2.
    """
    m = n // 2
    # g[k] = sum_j a[j] * w[k-j] for k = 0, ..., 2n-2
    # This is conv(a, w) but we need to be careful about dimensions
    # Actually, let's use FFT
    L = n + (2*n - 1) - 1  # = 3n - 2
    # We only need g[0:n]
    # g[k] = sum_{j=0}^{n-1} a[j] * w[k-j] for 0 <= k-j <= 2n-2
    # = sum_{j=max(0,k-(2n-2))}^{min(n-1,k)} a[j] * w[k-j]
    
    # Use FFT: g = ifft(fft(a) * fft(w))
    N = _pow2(3 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    g = np.fft.irfft(fa * fw, N)[:n]
    
    # grad_d[i] = g[i] - g[n-1-i] for i = 0, ..., m-1
    grad = g[:m] - g[n-1:m-1:-1]
    return grad

def _topk_weights(b, K, eps=0.01):
    """Create weights that focus on the top K coefficients."""
    L = len(b)
    idx = np.argsort(b)[::-1][:K]
    w = np.zeros(L, dtype=np.float64)
    top_val = b[idx[0]]
    for i in idx:
        w[i] = 1.0
    # Add small weight everywhere
    w += eps
    w /= w.sum()
    return w

def _softmax_weights(b, focus, eps=0.01):
    """Create softmax weights focused on large coefficients."""
    L = len(b)
    maxb = b.max()
    if maxb < 1e-300:
        return np.full(L, 1.0/L)
    z = (maxb - b) / maxb
    p = np.exp(-z * focus)
    s = p.sum()
    if s < 1e-30:
        w = np.zeros(L)
        w[np.argmax(b)] = 1.0
        return w
    p = (1-eps) * p/s + eps/L
    return p

def _optimize_anti_palindromic(n, d0, time_limit, rng, verbose=False):
    """Optimize the anti-palindromic parameterization."""
    m = n // 2
    d = d0.copy()
    bound = 1.0 / n
    d = np.clip(d, -bound, bound)
    
    a = _anti_palindromic_a(d, n)
    b = _fft_conv_square(a)
    score = _score_with_b(a, b)
    
    best_score = score
    best_d = d.copy()
    
    eta = 0.1
    focus = 5.0
    topK = 10
    no_improve = 0
    max_no = 500
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        if score < best_score - 1e-10:
            best_score = score
            best_d = d.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no:
            # Restart
            d = rng.uniform(-bound, bound, size=m) * 0.5
            a = _anti_palindromic_a(d, n)
            b = _fft_conv_square(a)
            score = _score_with_b(a, b)
            eta = 0.1
            focus = rng.choice([5.0, 10.0, 20.0, 50.0, 100.0])
            topK = rng.choice([5, 10, 20, 50, 100])
            no_improve = 0
            it += 1
            continue
        
        # Compute weights
        if it % 3 == 0:
            w = _topk_weights(b, topK)
        else:
            w = _softmax_weights(b, focus)
        
        # Compute gradient
        grad = _weighted_grad(w, a, n)
        maxg = np.max(np.abs(grad))
        if maxg < 1e-18:
            eta *= 0.5
            if eta < 1e-6:
                break
            it += 1
            continue
        
        # Normalize gradient
        grad = grad / maxg
        
        # Step

        a_new = _anti_palindromic_a(d_new, n)

            a = a_new

            score = score_new

def _coordinate_descent(a, time_limit, rng, n_top=5):
    """Coordinate descent refinement on the full sequence."""
    n = len(a)
    a = a.copy()
    s = a.sum()
    a = a / s  # Normalize to sum=1
    score = _score(a)
    best_score = score
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    step_size = 0.01
    
    while time.monotonic() - t0 < time_limit:
        b = _fft_conv_square(a)
        # Find top n_top coefficients
        top_idx = np.argsort(b)[::-1][:n_top]
        
        improved = False
        for k in top_idx:
            # For b_k = sum_{i+j=k} a_i * a_j, try to reduce it
            # by adjusting individual a_i
            # Simple approach: for each i that contributes to b_k,
            # try reducing a_i slightly
            for i in range(max(0, k - n + 1), min(n, k + 1)):
                j = k - i
                if j < 0 or j >= n:
                    continue
                # Current contribution: a[i] * a[j]
                # If we reduce a[i] by delta, b_k decreases by delta * a[j]
                # But other b's change too
                delta = step_size * a[i] * 0.1
                if delta < 1e-15:
                    continue
                a[i] -= delta
                if a[i] < 0:
                    a[i] = 0
                # Compensate by increasing another element
                a[(i + 1) % n] += delta
                
                b_new = _fft_conv_square(a)
                score_new = _score_with_b(a, b_new)
                if score_new < score:
                    score = score_new
                    improved = True
                    if score < best_score:
                        best_score = score
                        best_a = a.copy()
                else:
                    a[i] += delta
                    a[(i + 1) % n] -= delta
        
        if not improved:
            step_size *= 0.8
            if step_size < 1e-10:
                break
        else:
            step_size = min(step_size * 1.05, 0.1)
        
        it += 1
        if it % 100 == 0:
            print(f"  coord descent it={it} score={score:.6f} best={best_score:.6f} t={time.monotonic()-t0:.1f}s")
    
    return best_a, best_score

Let me think of a faster approach.

**Key insight**: Instead of doing a full FFT for each coordinate update, I can compute the change in b analytically.

If I change a[i] by δ, the change in b[k] is:
- For k = i+j where j is any valid index: δ * a[j]
- For k = 2i: additionally δ^2 (if we're changing a[i] itself)
- More precisely:
  - Δb[k] = δ * a[k-i] for 0 ≤ k-i < n, k ≠ 2i
  - Δb[2i] = δ * a[i] + δ^2 (the cross term plus the square)

- For k = i+q (q ≠ i): b[k] increases by δ * a[q]
- For k = 2i: b[k] increases by δ * a[i] + δ^2 (since both p and q change)
- Actually for k = 2i: the term a[i]*a[i] becomes (a[i]+δ)*a[i] + a[i]*(a[i]+δ) - a[i]*a[i] = 2δ*a[i] + δ^2

Minimize 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Uses anti-palindromic parameterization with efficient coordinate descent,
FFT-based gradient optimization, and diverse multi-start strategy.

import numpy as np

def _pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _fft_conv_square(a):
    n = len(a)
    L = 2 * n - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, N)
    b = np.fft.irfft(fa * fa, N)[:L]
    return np.maximum(b, 0.0)


def _compute_score(a, b=None):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return float('inf')
    if b is None:
        b = _fft_conv_square(a)
    return 2.0 * n * float(b.max()) / (s * s)


def _anti_pal_a(d, n):
    m = n // 2
    c = 1.0 / n
    a = np.empty(n, dtype=np.float64)
