```python
def _lp_step(a, M, n):
    """Solve the LP: max sum b_j s.t. (a*b)[k] <= M, b >= 0."""
    from scipy.optimize import linprog
    from scipy import sparse
    
    # (a*b)[k] = sum_i a[i] * b[k-i] for valid i
    # This is linear in b.
    # Build the constraint matrix A_ub where A_ub[k, j] = a[j] if k-j is valid index in a, else 0.
    # i.e., A_ub[k, j] = a[k-j] if 0 <= k-j < n, else 0.
    
    # For efficiency, use sparse matrix.
    # A_ub has shape (2n-1, n)
    # A_ub[k, j] = a[k-j] if 0 <= k-j <= n-1, else 0
    
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[idx])
    
    A_ub = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    c_obj = -np.ones(n)  # minimize -sum = max sum
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    return res.x
```

def _lp_direction(a, M, n):
    """
    Solve LP: max sum(b) s.t. conv(a,b)[k] <= M for all k, b >= 0.
    Returns b or None.
    """
    try:
        from scipy.optimize import linprog
        from scipy import sparse
    except:
        return None
    
    # Build cross-convolution matrix: A[k,j] = a[k-j] if 0 <= k-j <= n-1
    # A is (2n-1) x n, each column j has nonzeros at rows j..j+n-1
    total_nz = n * n  # each of n columns has n nonzeros
    rows = np.empty(total_nz, dtype=np.int32)
    cols = np.empty(total_nz, dtype=np.int32)
    vals = np.empty(total_nz, dtype=np.float64)
    
    idx = 0
    for j in range(n):
        r0 = j
        r1 = j + n  # rows j to j+n-1
        rows[idx:idx+n] = np.arange(r0, r1)
        cols[idx:idx+n] = j
        vals[idx:idx+n] = a[::-1]  # a[n-1-j], a[n-2-j], ..., a[0]
        # Wait: A[k,j] = a[k-j]. For k = j, j+1, ..., j+n-1: a[0], a[1], ..., a[n-1]
        vals[idx:idx+n] = a
        idx += n
    
    A_ub = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    c_obj = -np.ones(n)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0,None)]*n, method='highs',
                  options={'presolve': True})
    if res and res.success:
        b = np.asarray(res.x, dtype=np.float64)
        b = np.clip(b, 0, None)
        if b.sum() > 0:
            return b
    return None
