import time
from scipy.optimize import minimize, minimize_scalar
from scipy.special import expit

def _pow2(m):
    m = int(max(1, m))
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()

def _clean(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    return np.maximum(v, 0.0)

def _normalize(v):
    v = _clean(v)
    s = v.sum()
    if s <= 1e-15:
        return np.full(v.size, 1.0/v.size if v.size else 0.0)
    return v / s

def _score(a, L=None):
    """Compute 2*n*max(conv)/(sum^2) using FFT."""
    v = _clean(a)
    n = v.size
    if n == 0:
        return np.inf
    s = v.sum()
    if s < 1e-12:
        return np.inf
    v = v / s
    if L is None:
        L = _pow2(2*n - 1)
    A = np.fft.rfft(v, n=L)
    c = np.fft.irfft(A * A, n=L)[:2*n-1]
