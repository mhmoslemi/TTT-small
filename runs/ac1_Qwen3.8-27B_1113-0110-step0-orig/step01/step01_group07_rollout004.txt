        step += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        b1 = 1 - beta1 ** step
        b2 = 1 - beta2 ** step
        z = z - lr * (m / b1) / (np.sqrt(v / b2) + eps)
        z = np.where(np.isfinite(z), z, 0.0)
        z -= z.mean()
        z = np.clip(z, -50, 50)
    
    return best_z, best_s

def cheb_basis(n, D):
    x = np.linspace(0, 1, n)
    t = 2 * x - 1
    t = np.clip(t, -1, 1)
    ang = np.arccos(t)
    out = np.empty((n, D))
    for m in range(1, D + 1):
        out[:, m - 1] = np.cos(m * ang)
    return out
