The minimum of sum a_i * a_{n-1-i} subject to sum a_i = 1 and a_i >= 0 is 0 (achieved when a is supported on one side, e.g., a_0 = 1, rest 0, then c_{n-1} = a_0 * a_{n-1} = 0). But then c_0 = 1, which is huge.

So there's a trade-off: minimizing the center lag increases the edge lags, and vice versa.

The optimal solution balances all lags to be as equal as possible. This is reminiscent of a "flat-top" design.

"""
Hybrid optimization for non-negative coefficient sequences minimizing
2*n*max(autocorrelation) / (sum)^2.

Uses:
- Semi-smooth subgradient (top-K lags) with Adam
- Frank-Wolfe with exact subgradient
- Multiplicative update heuristic
- Local coordinate search
- Multiple sequence lengths with zoom-in strategy
"""

import time
import numpy as np

def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()

def normalize_positive(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s

def fast_score(a):
    p = normalize_positive(a)
    if p is None:
        return np.inf
    n = p.size
    if n <= 0:
        return np.inf
    L = next_pow2(2 * n - 1)
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0.0:
        return np.inf
    return float(2.0 * n * cm)

def compute_convolution(a, L=None):
    n = a.size
    if L is None:
        L = next_pow2(2 * n - 1)
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    return c

def softmax_func(z):
    z = np.asarray(z, dtype=np.float64)
    z = np.where(np.isfinite(z), z, -50.0)
    z = z - z.max()
    z = np.clip(z, -50.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        e = np.ones_like(z)
        s = float(e.size)
    return e / s

def exact_max_subgrad(a, c, K=1):
    """
    Compute subgradient of 2n * max_k c_k w.r.t. a.
    Uses top-K lags for semi-smooth approximation.
    Returns gradient (not yet adjusted for simplex constraint).
    """
    n = a.size
    # Get top-K indices
    if K >= len(c):
        idx = np.argsort(c)[::-1]
    else:
        # Use argpartition for efficiency
        part = np.argpartition(c, -K)[-K:]
        idx = part[np.argsort(c[part])[::-1]]
    
    # Gradient of c_k w.r.t. a: (grad c_k)_m = 2 * a_{k-m} (a_j=0 for invalid j)
    g = np.zeros(n, dtype=np.float64)
    for k in idx:
        lo = max(0, int(k) - n + 1)
        hi = min(n - 1, int(k))
        if lo > hi:
            continue
        m_range = np.arange(lo, hi + 1)
        a_idx = int(k) - m_range
        g[lo:hi+1] += 2.0 * a[a_idx]
    
    g /= len(idx)
    return 2.0 * n * g

def simplex_projection(v, S=1.0):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - S))[0]
    if len(rho) == 0:
        return np.full(n, S / n)
    rho = rho[-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def adam_exact(a_init, duration_s, rng, K=10, lr0=0.01, tau_schedule='decay'):
    """
    Adam optimizer using exact (or semi-smooth) max subgradient.
    Works in log-domain (softmax parametrization).
    """
    p = normalize_positive(a_init)
    if p is None:
        return None, np.inf
    
    n = p.size
    L = next_pow2(2 * n - 1)
    deadline = time.monotonic() + max(0.0, float(duration_s))
    
    z = np.log(np.maximum(p, 1e-300))
    z = np.where(np.isfinite(z), z, -50.0)
    z -= z.mean()
    z = np.clip(z, -50.0, 50.0)
    
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    bias_step = 0
    
    lr = float(lr0)
    best_p = p.copy()
    best_score = fast_score(p)
    no_improve = 0
    step = 0
    
    while time.monotonic() < deadline:
        c = compute_convolution(p, L)
        cm = float(c.max())
        true_s = 2.0 * n * cm
        
        if true_s < best_score - 1e-12:
            best_score = true_s
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive K: increase K when stuck (more information)
        K_eff = K + no_improve // 100
        K_eff = min(K_eff, 200)
        
        # Subgradient
        g_a = exact_max_subgrad(p, c, K=K_eff)
        
        # Transform to log-domain gradient
        mean_g = float(np.dot(p, g_a))
        grad_z = g_a - p * mean_g
        
        # Clip gradient
        max_g = float(np.max(np.abs(grad_z)))
        if max_g > 1e-1:
            grad_z *= (1e-1 / max_g)
        
        # Adam update
        bias_step += 1
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        b1 = 1.0 - beta1 ** bias_step
        b2 = 1.0 - beta2 ** bias_step
        z = z - lr * (m / b1) / (np.sqrt(v / b2) + eps)
        z = np.where(np.isfinite(z), z, 0.0)
        z -= z.mean()
        z = np.clip(z, -50.0, 50.0)
        p = softmax_func(z)
        step += 1
        
        # Learning rate schedule
        if step % 5000 == 0:
            lr = max(0.0005, lr * 0.92)
        
        # Restart if stuck
        if no_improve > 5000:
            zz = np.log(np.maximum(best_p, 1e-300))
            zz = np.where(np.isfinite(zz), zz, -50.0)
            zz = zz + rng.normal(0.0, 0.2, n)
            zz -= zz.mean()
            z = np.clip(zz, -50.0, 50.0)
            m *= 0.1
            v *= 0.1
            no_improve = 0
            lr = max(0.002, lr * 1.5)
            print(f"  adam restart at step {step}, score {best_score:.6f}")
        
        if step % 10000 == 0:
            print(f"  adam_exact step {step} score {true_s:.6f} best {best_score:.6f} lr {lr:.5f} K {K_eff}")
    
    return best_p, best_score

        return None, np.inf

    it = 0
    
    while time.monotonic() < deadline:

        cm = float(c.max())
        current_s = 2.0 * n * cm

            best_s = current_s
            best_a = a.copy()

"""
Optimization for non-negative coefficient sequences.
Minimizes 2*n*max(autocorr) / (sum)^2 using:
- Semi-smooth subgradient Adam (top-K lags)
- Frank-Wolfe refinement
- Multi-length zoom-in strategy
- Local coordinate search
"""

import time
import numpy as np

# --- Helper functions ---

def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()

def normalize_positive(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s

def compute_convolution(a, L=None):
    n = a.size
    if L is None:
        L = next_pow2(2 * n - 1)
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    return c

def score_from_conv(c, n):
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0:
        return np.inf
    return 2.0 * n * cm

def fast_score(a):
    p = normalize_positive(a)
    if p is None:
        return np.inf
    n = p.size
    c = compute_convolution(p)
    return score_from_conv(c, n)

def softmax_func(z):
    z = np.asarray(z, dtype=np.float64)
    z = np.where(np.isfinite(z), z, -50.0)
    z = z - z.max()
    z = np.clip(z, -50.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones_like(z) / z.size
    return e / s

def topk_subgrad(a, c, K=10):
    """
    Semi-smooth subgradient of 2n*max(c) using top-K lags.
    (grad c_k)_m = 2 * a_{k-m} (a_j=0 for j notin [0,n-1])
    """
    n = a.size
    total_c = c.size
    K = min(K, total_c)
    
    if K == total_c:
        idx = np.arange(total_c)
    else:
        # Get top K indices
        part = np.argpartition(c, -K)[-K:]
        idx = part[np.argsort(c[part])[::-1]]
    
    g = np.zeros(n, dtype=np.float64)
    for k in idx:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        # a_{k-m} for m in [lo, hi]
        # m goes from lo to hi, k-m goes from k-lo to k-hi
        a_idx = k - np.arange(lo, hi + 1)
        g[lo:hi+1] += 2.0 * a[a_idx]
    
    g /= float(K)
    return 2.0 * n * g

def log_domain_adam(a_init, duration_s, rng, K=20, lr0=0.015, verbose=True):
    """Adam in log-domain with semi-smooth subgradient."""
    p = normalize_positive(a_init)
    if p is None:
        return None, np.inf
    
    n = p.size
    L = next_pow2(2 * n - 1)
    deadline = time.monotonic() + max(0.0, float(duration_s))
    
    z = np.log(np.maximum(p, 1e-300))
    z = np.where(np.isfinite(z), z, -50.0)
    z -= z.mean()
    z = np.clip(z, -50.0, 50.0)
    
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    bias_step = 0
    
    lr = float(lr0)
    best_p = p.copy()
    best_score = score_from_conv(compute_convolution(p, L), n)
    no_improve = 0
    step = 0
    
    while time.monotonic() < deadline:
        c = compute_convolution(p, L)
        true_s = score_from_conv(c, n)
        
        if true_s < best_score - 1e-12:
            best_score = true_s
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive K
        K_eff = min(K + no_improve // 50, 200)
        
        g_a = topk_subgrad(p, c, K=K_eff)
        mean_g = float(np.dot(p, g_a))
        grad_z = g_a - p * mean_g
        
        # Gradient clipping
        max_g = float(np.max(np.abs(grad_z)))
        if max_g > 0.1:
            grad_z *= (0.1 / max_g)
        
        bias_step += 1
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        b1 = 1.0 - beta1 ** bias_step
        b2 = 1.0 - beta2 ** bias_step
        z = z - lr * (m / b1) / (np.sqrt(v / b2) + eps)
        z = np.where(np.isfinite(z), z, 0.0)
        z -= z.mean()
        z = np.clip(z, -50.0, 50.0)
        p = softmax_func(z)
        step += 1
        
        if step % 3000 == 0:
            lr = max(0.0003, lr * 0.93)
        
        if no_improve > 4000:
            # Restart with perturbation
            zz = np.log(np.maximum(best_p, 1e-300))
            zz = np.where(np.isfinite(zz), zz, -50.0)
            zz += rng.normal(0.0, 0.25, n)
            zz -= zz.mean()
            z = np.clip(zz, -50.0, 50.0)
            m *= 0.05
            v *= 0.05
            no_improve = 0
            lr = max(0.003, lr * 2.0)
            if verbose:
                print(f"  restart@{step} score={best_score:.6f}")
        
        if verbose and step % 20000 == 0:
            print(f"  adam step={step} cur={true_s:.6f} best={best_score:.6f} lr={lr:.5f} K={K_eff}")
    
    return best_p, best_score

def multiplicative_update(a_init, duration_s, K=20, verbose=True):
    """
    Multiplicative update heuristic:
    a_i <- a_i / (1 + lambda * sum_{k in topK} a_{k-i})
    This reduces a_i when it contributes to peak lags.
    """
    a = normalize_positive(a_init)
    if a is None:
        return None, np.inf
    
    n = a.size
    L = next_pow2(2 * n - 1)
    deadline = time.monotonic() + max(0.0, float(duration_s))
    best_a = a.copy()
    best_s = score_from_conv(compute_convolution(a, L), n)
    step = 0
    
    while time.monotonic() < deadline:
        c = compute_convolution(a, L)
        cur_s = score_from_conv(c, n)
        
        if cur_s < best_s - 1e-12:
            best_s = cur_s
            best_a = a.copy()
        
        # Get top-K peak lags
        K_eff = min(K, c.size)
        part = np.argpartition(c, -K_eff)[-K_eff:]
        
        # Compute "pressure" on each a_i
        pressure = np.zeros(n, dtype=np.float64)
        for k in part:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo > hi:
                continue
            a_idx = k - np.arange(lo, hi + 1)
            pressure[lo:hi+1] += a[a_idx]
        
        # Multiplicative update
        lam = 0.1
        a = a / (1.0 + lam * pressure / (a + 1e-30))
        a = np.maximum(a, 0.0)
        s = float(a.sum())
        if s > 0:
            a /= s
        
        step += 1
        if step % 500 == 0 and verbose:
            print(f"  mult step={step} score={score_from_conv(compute_convolution(a, L), n):.6f} best={best_s:.6f}")
    
    return best_a, best_s

def resize_sequence(seq, m):
    p = normalize_positive(seq)
    m = int(m)
    if p is None:
        return np.ones(m, dtype=np.float64) / float(m)
    n = p.size
    if n == m:
        return p.copy()
    if n == 1:
        return np.full(m, 1.0 / float(m), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, m)
    out = np.interp(x_new, x_old, p)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(m, dtype=np.float64) / float(m)
    return out / s

def local_search(a_init, duration_s, rng, n_trials_per_step=50, verbose=True):
    """
    Local search: randomly perturb a few coordinates and accept if improved.
    """
    a = normalize_positive(a_init)
    if a is None:

    best_a = a.copy()
    best_s = score_from_conv(compute_convolution(a, L), n)
    it = 0
    
    while time.monotonic() < deadline:
        c = compute_convolution(a, L)
        cur_s = score_from_conv(c, n)
        
        improved = False
        for _ in range(n_trials_per_step):
            if time.monotonic() > deadline:
                break
            
            # Random perturbation: increase one coordinate, decrease another
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            delta = rng.uniform(0.001, 0.02)
            if a[i] < delta:
                continue
            
            a_try = a.copy()
            a_try[i] -= delta
            a_try[j] += delta
            
            c_try = compute_convolution(a_try, L)
            s_try = score_from_conv(c_try, n)
            
            if s_try < best_s - 1e-10:
                best_s = s_try
                best_a = a_try.copy()
                a = a_try
                improved = True
        
        if not improved:
            it += 1
            if it > 50:
                break
        
        if verbose and it % 20 == 0:
            print(f"  local it={it} best={best_s:.6f}")
    
    return best_a, best_s
