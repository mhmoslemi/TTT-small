"""
Improvement over the previous prototype-expansion + coordinate-descent approach.

Main ideas:
 1) Search in a block-mass subsystem: the final sequence is m blocks of length L,
    each block constant.  The exact score of such a block sequence can be computed
    from the short convolution q = p * p in O(m^2 + m), which makes strong
    refinement feasible.
 2) Use a stronger LP polishing step.  Given current block masses p, solve LPs for
    directions g so that the cross-convolution p*g is small in exactly the two
    families of constraints that determine the block-expanded sequence:
         L * (p*g)_k,   (p*g)_k + (L-1)(p*g)_{k-1}.
    Both a min-max LP and fixed-bound packing LPs are used.
 3) Polish the block masses with an exact one-coordinate line search formulated
    directly for the block-expanded score.
 4) Expand the best block prototypes and polish full-length sequences with the
    previous-style coordinate descent, using remaining time.
"""

import numpy as np
import time

INF = float("inf")


def _candidate_key(cand):
    return cand[0]


def _clean_array(arr):
    a = np.asarray(arr, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    np.clip(a, 0.0, 1000.0, out=a)
    return a


def _normalize(a):
    a = _clean_array(a)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(a.size if a.size > 0 else 1, dtype=np.float64)
    return a / s


def _score_normalized(p, c):
    n = int(p.size)
    if n <= 0:
        return INF
    if c.size == 0:
        return INF
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return INF
    return float(2.0 * n * max(0.0, mx))


def _score_array(a):
    a = _normalize(a)
    if a.size <= 0:
        return INF
    c = np.convolve(a, a)
    return _score_normalized(a, c)

def _two_level(n):
    n = max(1, int(n))
    if n == 1:
        return np.ones(1, dtype=np.float64)
    m = max(1, n // 2)
    p = np.zeros(n, dtype=np.float64)
    p[:m] = (2.0 / 3.0) / float(m)
    rest = n - m
    if rest > 0:
        p[m:] = (1.0 / 3.0) / float(rest)
    return _normalize(p)


def _paired_initial(n, rng, alpha_w=1.0, split=0.3):
    n = max(1, int(n))
    if n == 1:
        return np.ones(1, dtype=np.float64)
    m = (n + 1) // 2
    weights = rng.dirichlet(np.full(m, max(0.05, float(alpha_w)), dtype=np.float64))
    p = np.zeros(n, dtype=np.float64)
    split = max(0.05, float(split))
    for r in range(m):
        a = r
        b = n - 1 - r
        if a == b:
            p[a] = weights[r]
        else:
            if rng.random() < 0.72:
                if rng.random() < 0.5:
                    q0, q1 = 1.0, 0.0
                else:
                    q0, q1 = 0.0, 1.0
            else:
                q = rng.dirichlet(np.full(2, split, dtype=np.float64))
                q0, q1 = float(q[0]), float(q[1])
            p[a] = weights[r] * q0
            p[b] = weights[r] * q1
    return _normalize(p)


def _make_initial(n, strategy, rng):
    n = max(1, int(n))
    if n == 1:
        return np.ones(1, dtype=np.float64)
    s = int(strategy) % 16
    if s == 0:
        p = np.ones(n, dtype=np.float64)
    elif s == 1:
        p = rng.dirichlet(np.full(n, 1.0, dtype=np.float64))
    elif s == 2:
        p = rng.dirichlet(np.full(n, 0.5, dtype=np.float64))
    elif s == 3:
        p = rng.dirichlet(np.full(n, 0.2, dtype=np.float64))
    elif s == 4:
        p = np.exp(rng.normal(0.0, 1.0, n))
    elif s == 5:
        p = np.exp(rng.normal(0.0, 0.4, n))
    elif s == 6:
        p = _paired_initial(n, rng, 0.7, 0.25)
    elif s == 7:
        p = _paired_initial(n, rng, 1.2, 0.8)
    elif s == 8:
        p = _two_level(n)
    elif s == 9:
        x = np.linspace(0.0, 1.0, n, dtype=np.float64)
        p = (0.2 + 0.8 * (np.abs(2.0 * x - 1.0) ** 2)) * (1.0 + 0.5 * rng.random(n))
    elif s == 10:
        x = np.linspace(0.0, 1.0, n, dtype=np.float64)
        p = 0.05 + 0.95 * (np.abs(2.0 * x - 1.0) ** 8)
    elif s == 11:
        x = np.linspace(0.0, 1.0, n, dtype=np.float64)
        p = np.sin(np.pi * x) ** 2
        p = 0.1 * (1.0 - p) + p
    elif s == 12:
        x = np.linspace(0.0, 1.0, n, dtype=np.float64)
        p = np.exp(-((x - 0.5) ** 2) / (2.0 * 0.12 * 0.12))
    elif s == 13:
        p = np.sort(rng.random(n)) ** 3
    elif s == 14:
        p = _two_level(n) * (1.0 + 0.25 * rng.random(n))
    else:
        x = np.linspace(0.0, 1.0, n, dtype=np.float64)
        p = np.exp(rng.normal(0.0, 0.8, n)) * (1.0 + 0.5 * np.abs(np.sin(np.pi * x)))
    return _normalize(p)


def _add_candidate(candidates, seq, score, limit=6):
    if seq is None or not np.isfinite(score):
        return
    arr = _normalize(seq).copy()
    if arr.size == 0:
        return
    candidates.append([float(score), arr])
    candidates.sort(key=_candidate_key)
    while len(candidates) > int(limit):
        candidates.pop()


def _downsample_block_masses(a, m):
    a = _normalize(a)
    m = max(1, int(m))
    n = a.size
    if m <= 1:
        return np.ones(1, dtype=np.float64)
    if n % m == 0:
        p = a.reshape(m, n // m).sum(axis=1)
    else:
        p = np.zeros(m, dtype=np.float64)
        for j in range(m):
            lo = (j * n) // m
            hi = max(lo + 1, ((j + 1) * n) // m)
            p[j] = float(np.sum(a[lo:hi]))
    p = np.clip(p, 0.0, None)
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(m, dtype=np.float64) / float(m)
    return p / s


def _expand_blocks(p, L):
    p = _normalize(p)
    m = int(p.size)
    L = max(1, int(L))
    n = m * L
    out = np.empty(n, dtype=np.float64)
    inv = 1.0 / float(L)
    for i in range(m):
        out[i * L:(i + 1) * L] = p[i] * inv
    return _normalize(out)


def _score_from_q(q, L, n):
    q = np.asarray(q, dtype=np.float64)
    L = max(1, int(L))
    n = max(1, int(n))
    if q.size == 0:
        return INF
    mx = float(np.max(q))
    if not np.isfinite(mx):
        return INF
    if L == 1:
        return float(2.0 * n * mx)
    alpha = float(L - 1)
    v1 = float(L) * mx
    v2 = 0.0
    if q.size > 1:
        v2 = float(np.max(q[1:] + alpha * q[:-1]))
    best = max(v1, v2)
    if not np.isfinite(best):
        return INF
    return float(2.0 * n * best / float(L * L))


def _score_block(p, L):
    p = _normalize(p)
    m = int(p.size)
    L = max(1, int(L))
    n = m * L
    if m <= 0:
        return INF
    q = np.convolve(p, p)
    return _score_from_q(q, L, n)


def _max_numerator(q, L):
    q = np.asarray(q, dtype=np.float64)
    L = max(1, int(L))
    if q.size == 0:
        return 0.0
    mx = float(np.max(q))
    if not np.isfinite(mx):
        return 0.0
    if L == 1:
        return mx
    alpha = float(L - 1)
    v1 = float(L) * mx
    v2 = 0.0
    if q.size > 1:
        v2 = float(np.max(q[1:] + alpha * q[:-1]))
    return max(v1, v2)


def _mix_score(qf, qg, qx, t, L, n):
    t = float(t)
    at = 1.0 - t
    bt = t
    qt = at * at * qf + 2.0 * at * bt * qx + bt * bt * qg
    return _score_from_q(qt, L, n)


def _line_search_mix(qf, qg, qx, current_score, L, n):
    best_t = 0.0
    best_val = float(current_score)
    ts = np.geomspace(1e-6, 1.0, 90)
    ts = np.unique(np.concatenate((np.array([0.0, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0], dtype=np.float64), ts)))
    for t in ts:
        val = _mix_score(qf, qg, qx, float(t), L, n)
        if np.isfinite(val) and val < best_val:
            best_val = float(val)
            best_t = float(t)
    if best_val < float(current_score) - 1e-15:
        lo = max(0.0, best_t - 0.08)
        hi = min(1.0, best_t + 0.08)
        if hi > lo:
            ts2 = np.linspace(lo, hi, 41)
            for t in ts2:
                val = _mix_score(qf, qg, qx, float(t), L, n)
                if np.isfinite(val) and val < best_val:
                    best_val = float(val)
                    best_t = float(t)
    return float(best_t), float(best_val)


def _build_A(p):
    m = int(p.size)
    A = np.zeros((2 * m - 1, m), dtype=np.float64)
    for i in range(m):
        A[i:i + m, i] = p
    return A


def _linprog_fixed(A_cons, M):
    if A_cons is None or A_cons.size == 0 or not np.isfinite(M) or M <= 0.0:
        return None
    try:
        from scipy.optimize import linprog
    except Exception:
        return None
    m = int(A_cons.shape[1])
    c = -np.ones(m, dtype=np.float64)
    try:
        res = linprog(
            c,
            A_ub=A_cons,
            b_ub=np.full(A_cons.shape[0], float(M), dtype=np.float64),
            bounds=[(0.0, None)] * m,
            method="highs",
            options={"presolve": True}
        )
    except Exception:
        return None
    if res is None or not getattr(res, "success", False):
        return None
    g = np.asarray(res.x, dtype=np.float64)
    np.clip(g, 0.0, None, out=g)
    s = float(np.sum(g))
    if not np.isfinite(s) or s < 1e-12:
        return None
    return g / s


def _linprog_minmax(A_cons, m):
    if A_cons is None or A_cons.size == 0:
        return None
    try:
        from scipy.optimize import linprog
    except Exception:
        return None
    ncons = int(A_cons.shape[0])
    m = int(m)
    c_obj = np.concatenate((np.zeros(m, dtype=np.float64), np.array([1.0], dtype=np.float64)))
    A_ub = np.hstack((A_cons, -np.ones((ncons, 1), dtype=np.float64)))
    b_ub = np.zeros(ncons, dtype=np.float64)
    A_eq = np.zeros((1, m + 1), dtype=np.float64)
    A_eq[0, :m] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)
    bounds = [(0.0, None)] * m + [(0.0, None)]
    try:
        res = linprog(
            c_obj,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"presolve": True}
        )
    except Exception:
        return None
    if res is None or not getattr(res, "success", False):
        return None
    g = np.asarray(res.x[:m], dtype=np.float64)
    np.clip(g, 0.0, None, out=g)
    s = float(np.sum(g))
    if not np.isfinite(s) or s < 1e-12:
        return None
    return g / s

def _block_grid(p, i, m, grid_count):
    old = float(p[i]) if i < p.size and np.isfinite(p[i]) else 0.0
    if old < 0.0:
        old = 0.0
    scale = max(1.0 / float(m), 1e-12)
    rel = np.logspace(-6.0, 2.0, max(4, int(grid_count)))
    ys1 = old * rel if old > 0.0 else np.zeros(1, dtype=np.float64)
    lo = max(1e-15, scale * 1e-6)
    hi = min(1.0, max(scale * 100.0, old * 10.0 if old > 0.0 else 0.0, 20.0 / float(m)))
    if hi <= lo:
        hi = min(1.0, lo * 10.0)
    ys2 = np.logspace(np.log10(lo), np.log10(hi), max(4, int(grid_count)))
    extras = np.array([0.0, old, scale, 0.5 * scale, 2.0 * scale, 5.0 * scale, 10.0 * scale], dtype=np.float64)
    ys = np.concatenate((ys1, ys2, extras))
    ys = ys[np.isfinite(ys) & (ys >= 0.0) & (ys <= 1.0)]
    return np.unique(ys)


def _block_eval_ys(q_aff, lin_q, is_diag, const_e, lin_e, diag_e,

    else:
        maxE = np.full(ys.size, E_unaff, dtype=np.float64)
    R1_vals = float(L) * (q_aff[None, :] + dy[:, None] * lin_q[None, :] + quad[:, None] * is_diag[None, :])
    maxR1 = np.maximum(R1_unaff, np.max(R1_vals, axis=1))
    mx = np.maximum(maxR1, maxE)
    vals = np.full(ys.size, INF, dtype=np.float64)
    valid = S > 1e-15
    coeff = 2.0 * float(m) / float(L)
    vals[valid] = coeff * mx[valid] / (S[valid] * S[valid])
    idx = int(np.argmin(vals))
    return float(ys[idx]), float(vals[idx])


def _block_coord_update(p, q, score, i, L, grid_count, refine_count):
    m = int(p.size)
    if m <= 1 or i < 0 or i >= m:
        return p, q, float(score), False
    L = max(1, int(L))
    old = float(p[i])
    if not np.isfinite(old) or old < 0.0:
        old = 0.0
        p = p.copy()
        p[i] = 0.0

    ys = _block_grid(p, i, m, grid_count)
    if ys.size == 0:
        return p, q, float(score), False

    q_aff = q[i:i + m].copy()
    lin_q = 2.0 * p.copy()
    lin_q[i] = 0.0
    is_diag = np.zeros(m, dtype=np.float64)
    is_diag[i] = 1.0

    R1_unaff = 0.0
    if i > 0:
        R1_unaff = max(R1_unaff, float(np.max(q[:i])))
    if i + m < q.size:
        R1_unaff = max(R1_unaff, float(np.max(q[i + m:])))
    R1_unaff = float(L) * R1_unaff

    alpha = float(L - 1)
    if m > 1:
        E_full = q[1:] + alpha * q[:-1]
        ka = max(1, i)
        kb = min(2 * m - 2, i + m)
        e_start = ka - 1
        e_end = kb - 1
        E_unaff = 0.0
        if e_start > 0:
            E_unaff = max(E_unaff, float(np.max(E_full[:e_start])))
        if e_end + 1 < E_full.size:
            E_unaff = max(E_unaff, float(np.max(E_full[e_end + 1:])))
        const_e = E_full[e_start:e_end + 1].copy()
        e_len = const_e.size
        lin_e = np.zeros(e_len, dtype=np.float64)
        diag_e = np.zeros(e_len, dtype=np.float64)
        for idx in range(e_len):
            k = ka + idx
            if i <= k < i + m:
                r = k - i
                lin_e[idx] += lin_q[r]
                if k == 2 * i:
                    diag_e[idx] += 1.0
            if i <= k - 1 < i + m:
                r = (k - 1) - i
                lin_e[idx] += alpha * lin_q[r]
                if (k - 1) == 2 * i:
                    diag_e[idx] += alpha
    else:
        const_e = np.zeros(0, dtype=np.float64)
        lin_e = np.zeros(0, dtype=np.float64)
        diag_e = np.zeros(0, dtype=np.float64)
        E_unaff = 0.0

    S_base = 1.0 - old
    best_y, best_val = _block_eval_ys(
        q_aff, lin_q, is_diag, const_e, lin_e, diag_e,
        R1_unaff, E_unaff, S_base, old, L, m, ys
    )

    if refine_count > 0 and ys.size > 1:
        idx = int(np.searchsorted(ys, best_y))
        if idx >= ys.size:
            idx = ys.size - 1
        left = 0.0 if idx == 0 else float(ys[idx - 1])
        right = float(ys[idx + 1]) if idx + 1 < ys.size else max(best_y * 10.0, 1.0 / float(m))
        if best_y > 0.0:
            local = best_y * np.logspace(-1.0, 1.0, max(4, int(refine_count)))
        else:
            lo = max(0.0, left)
            hi = max(1e-12, right)
            if hi > lo:
                local = np.linspace(lo, hi, max(4, int(refine_count)))
            else:
                local = np.logspace(-8.0, -2.0, max(4, int(refine_count)))
        local = np.unique(np.concatenate((local, np.array([left, best_y, right], dtype=np.float64))))
        local = local[np.isfinite(local) & (local >= 0.0) & (local <= 1.0)]
        if local.size > 0:
            ly, lv = _block_eval_ys(
                q_aff, lin_q, is_diag, const_e, lin_e, diag_e,
                R1_unaff, E_unaff, S_base, old, L, m, local
            )
            if np.isfinite(lv) and lv < best_val:
                best_y, best_val = ly, lv

    tol = 1e-12 * max(1.0, float(score))
    if not np.isfinite(best_val) or best_val >= float(score) - tol:
        return p, q, float(score), False

    S = S_base + best_y
    if S <= 1e-15:
        return p, q, float(score), False
    S2 = S * S
    dy = best_y - old
    quad = dy * (best_y + old)

    p_new = p / S
    p_new[i] = best_y / S
    p_new = _normalize(p_new)

    q_new = q.copy()
    q_new[:i] = q[:i] / S2
    q_new[i:i + m] = (q_aff + dy * lin_q + quad * is_diag) / S2
    q_new[i + m:] = q[i + m:] / S2
    np.clip(q_new, 0.0, None, out=q_new)

    return p_new, q_new, float(best_val), True


def _block_coordinate_refine(p, L, deadline, rng, max_cycles=10):
    p = _normalize(p.copy())
    m = int(p.size)
    L = max(1, int(L))
    if m <= 0:
        return p, INF
    if m == 1:
        return p, _score_block(p, L)
    q = np.convolve(p, p)
    score = _score_block(p, L)
    best_p = p.copy()
    best_score = float(score)
    grid_count, refine_count = _block_grid_params(m)
    stagnant = 0
    perts = 0
    noise_scales = (0.35, 0.22, 0.12, 0.06)
    max_cycles = max(0, int(max_cycles))
    for _ in range(max_cycles):
        if time.monotonic() > float(deadline) - 1.0:
            break
        order = rng.permutation(m)
        improved = False
        for i in order:
            if time.monotonic() > float(deadline) - 0.5:
                break
            p, q, score, imp = _block_coord_update(p, q, score, int(i), L, grid_count, refine_count)
            if imp:
                improved = True
        p = _normalize(p)
        q = np.convolve(p, p)
        score = _score_block(p, L)
        if not np.isfinite(score):
            p = best_p.copy()
            q = np.convolve(p, p)
            score = best_score
        tol = 1e-12 * max(1.0, best_score)
        if score < best_score - tol:
            best_score = float(score)
            best_p = p.copy()
            stagnant = 0
        else:
            if score > best_score * 1.2 + 1e-9:
                p = best_p.copy()
                q = np.convolve(p, p)
                score = best_score
            if improved:
                stagnant = 0
            else:
                stagnant += 1

        if stagnant >= 2 and perts < 4:
            scale = float(noise_scales[min(perts, len(noise_scales) - 1)])
            p = best_p * np.exp(rng.normal(0.0, scale, m))
            p = _normalize(p)
            q = np.convolve(p, p)
            score = _score_block(p, L)
            if np.isfinite(score) and score < best_score - 1e-12:
                best_p = p.copy()
                best_score = float(score)
            perts += 1
            stagnant = 0
        elif stagnant >= 6:
            break
    return best_p, float(best_score)


def _full_grid_params(n):
    n = max(1, int(n))
    if n <= 128:
        return 50, 12
    if n <= 512:
        return 34, 8
    if n <= 1200:
        return 26, 6
    return 18, 4


def _full_coord_update(p, c, score, i, grid_count, refine_count):
    n = int(p.size)
    if n <= 1 or i < 0 or i >= n:
        return p, c, float(score), False
    old = float(p[i])
    if not np.isfinite(old) or old < 0.0:
        old = 0.0
        p = p.copy()
        p[i] = 0.0
    D = 1.0 - old
    if D < 0.0:
        D = 0.0
    ks = slice(i, i + n)
    c_k = c[ks]
    coeff = 2.0 * p.copy()
    const = -2.0 * old * p.copy()
    mid = int(i)
    coeff[mid] = 0.0
    const[mid] = -old * old
    max_unaff = 0.0
    if i > 0:
        max_unaff = max(max_unaff, float(np.max(c[:i])))
    end = i + n
    if end < c.size:
        max_unaff = max(max_unaff, float(np.max(c[end:])))

    xs = np.concatenate((np.array([0.0], dtype=np.float64), np.logspace(-8.0, 2.0, grid_count)))
    xs = np.append(xs, old)
    xs = xs[np.isfinite(xs) & (xs >= 0.0) & (xs <= 1e4)]
    if xs.size == 0:
        return p, c, float(score), False

    aff = c_k[None, :] + coeff[None, :] * xs[:, None] + const[None, :]
    aff[:, mid] += xs * xs
    max_aff = np.max(aff, axis=1)
    maxc = np.maximum(max_unaff, max_aff)
    S = D + xs
    vals = np.full(xs.size, INF, dtype=np.float64)
    valid = S > 1e-15
    vals[valid] = (2.0 * n) * maxc[valid] / (S[valid] * S[valid])
    idx = int(np.argmin(vals))
    best_x = float(xs[idx])
    best_val = float(vals[idx])

    tol = 1e-12 * max(1.0, float(score))
    if not np.isfinite(best_val) or best_val >= float(score) - tol:
        return p, c, float(score), False

    if refine_count > 0 and xs.size > 1:
        left = 0.0 if idx == 0 else float(xs[idx - 1])
        right = float(xs[idx]) * 10.0 if idx == xs.size - 1 else float(xs[idx + 1])
        if best_x > 0.0:
            local = best_x * np.logspace(-1.0, 1.0, int(refine_count))
        else:
            base_left = max(0.0, left)
            base_right = max(1e-12, right)
            local = np.linspace(base_left, base_right, int(refine_count))
        local = np.unique(np.concatenate((local, np.array([left, best_x, right], dtype=np.float64))))
        local = local[np.isfinite(local) & (local >= 0.0) & (local <= 1e4)]
        if local.size > 0:
            aff2 = c_k[None, :] + coeff[None, :] * local[:, None] + const[None, :]
            aff2[:, mid] += local * local
            max_aff2 = np.max(aff2, axis=1)
            maxc2 = np.maximum(max_unaff, max_aff2)
            S2 = D + local
            vals2 = np.full(local.size, INF, dtype=np.float64)
            valid2 = S2 > 1e-15
            vals2[valid2] = (2.0 * n) * maxc2[valid2] / (S2[valid2] * S2[valid2])
            idx2 = int(np.argmin(vals2))
            if float(vals2[idx2]) < best_val:
                best_x = float(local[idx2])
                best_val = float(vals2[idx2])

    if not np.isfinite(best_val) or best_val >= float(score) - tol:
        return p, c, float(score), False

    S = D + best_x
    if S <= 1e-15:
        return p, c, float(score), False
    S2 = S * S
    aff_best = c_k + coeff * best_x + const
    aff_best[mid] += best_x * best_x

    c_new = c / S2
    c_new[ks] = aff_best / S2
    p_new = p / S
    p_new[mid] = best_x / S
    p_new = _normalize(p_new)
    np.clip(c_new, 0.0, None, out=c_new)

    snew = float(np.sum(p_new))
    if not np.isfinite(snew) or snew <= 0.0:
        return p, c, float(score), False
    p_new = p_new / snew
    c_new = c_new / (snew * snew)
    score_new = _score_normalized(p_new, c_new)
    if not np.isfinite(score_new) or score_new >= float(score) - tol:
        return p, c, float(score), False
    return p_new, c_new, float(score_new), True


def _full_coordinate_refine(p, deadline, rng, max_perturb=2):
    p = _normalize(p.copy())
    n = int(p.size)
    if n <= 0:
        return p, INF
    c = np.convolve(p, p)
    score = _score_normalized(p, c)
    if not np.isfinite(score):
        score = _score_array(p)
    best_p = p.copy()
    best_score = float(score)
    grid_count, refine_count = _full_grid_params(n)
    xs_grid = np.concatenate((np.array([0.0], dtype=np.float64), np.logspace(-8.0, 2.0, grid_count)))
    stagnant = 0
    perts = 0
    noise_scales = (0.55, 0.30, 0.14, 0.06)
    max_perturb = max(0, int(max_perturb))
    while time.monotonic() < float(deadline) - 0.3:
        order = rng.permutation(n)
        improved = False
        for i in order:
            if time.monotonic() > float(deadline) - 0.2:
                break
            p, c, score, imp = _full_coord_update(p, c, score, int(i), xs_grid, refine_count)
            if imp:
                improved = True
        p = _normalize(p)
        s = float(np.sum(p))
        if not np.isfinite(s) or s <= 0.0:
            break
        c = np.convolve(p, p)
        np.clip(c, 0.0, None, out=c)
        score = _score_normalized(p, c)
        if not np.isfinite(score):
            p = best_p.copy()
            c = np.convolve(p, p)
            score = best_score
        tol = 1e-12 * max(1.0, best_score)
        if score < best_score - tol:
            best_score = float(score)
            best_p = p.copy()
            stagnant = 0
        else:
            if score > best_score * 1.25 + 1e-8:
                p = best_p.copy()
                c = np.convolve(p, p)
                score = best_score
            if not improved:
                stagnant += 1

        if stagnant >= 2 and perts < max_perturb:
            scale = float(noise_scales[min(perts, len(noise_scales) - 1)])
            p = best_p * np.exp(rng.normal(0.0, scale, n))
            p = np.clip(p, 0.0, None)
            p = _normalize(p)
            c = np.convolve(p, p)
            np.clip(c, 0.0, None, out=c)
            score = _score_normalized(p, c)
            if np.isfinite(score) and score < best_score - 1e-10:
                best_p = p.copy()
                best_score = float(score)
            perts += 1
            stagnant = 0
        elif stagnant >= 6:
            break
    return best_p, float(best_score)


def propose_candidate(seed=42, budget_s=2350):
    """Search for a low-score nonnegative coefficient sequence."""
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2350.0
    if not np.isfinite(budget) or budget <= 1.0:
        budget = 2350.0
    budget = min(2350.0, max(10.0, budget))
    deadline = t0 + max(1.0, budget - 25.0)
    rng = np.random.default_rng(int(seed))

    target_n = 4096
    fallback = []
    best_seq = None
    best_score = INF

    def update_best(seq, sc):
        nonlocal best_seq, best_score
        if seq is None or not np.isfinite(sc):
            return
        add = _normalize(seq).copy()
        if add.size == 0:
            return
        _add_candidate(fallback, add, sc, limit=8)
        if sc < best_score:
            best_score = float(sc)
            best_seq = add.copy()
            print(f"[{time.monotonic()-t0:.1f}s] new best internal score = {best_score:.9f}, len={add.size}", flush=True)

    gseq = _get_global_sequence()
    if gseq is not None:
        seq = _normalize(gseq)
        sc = _score_array(seq)
        update_best(seq, sc)
        print(f"initial global sequence score = {sc:.9f}, len={seq.size}", flush=True)

    base1 = _two_level(target_n)
    update_best(base1, _score_array(base1))
    base2 = np.ones(target_n, dtype=np.float64) / float(target_n)
    update_best(base2, _score_array(base2))

    seed_full = None
    if best_seq is not None:
        seed_full = _normalize(best_seq)

    # Short full polish on the initial global seed, if available and useful.
    if seed_full is not None and time.monotonic() < deadline - 180.0:
        quick_dl = min(t0 + 60.0, deadline - 120.0)
        if quick_dl > time.monotonic() + 20.0:
            try:
                seq2, sc2 = _full_coordinate_refine(seed_full, quick_dl, rng, max_perturb=2)
                update_best(seq2, sc2)
                if seq2 is not None and np.isfinite(sc2):
                    seed_full = _normalize(seq2)
            except Exception:
                pass

    reserve_final = max(260.0, 0.38 * budget)
    proto_deadline = deadline - reserve_final
    if proto_deadline < t0 + 120.0:
        proto_deadline = deadline - 60.0
    if proto_deadline < t0 + 60.0:
        proto_deadline = deadline - 30.0

    plan = [
        (16, 6, 30),
        (24, 5, 45),
        (32, 6, 60),
        (48, 5, 75),
        (64, 6, 110),
        (96, 5, 140),
        (128, 5, 190),
        (192, 4, 240),
        (256, 5, 300),
        (384, 4, 330),
        (512, 5, 400),
        (768, 3, 260),
    ]

    for m, trials, t_slice in plan:
        if time.monotonic() >= proto_deadline:
            break
        L = max(1, int(round(target_n / float(m))))
        n = m * L
        m_deadline = min(proto_deadline, time.monotonic() + max(20.0, float(t_slice)))

        if m <= 32:
            lp_iters = 34
        elif m <= 64:
            lp_iters = 26
        elif m <= 128:
            lp_iters = 18
        elif m <= 256:
            lp_iters = 12
        elif m <= 512:
            lp_iters = 8
        else:
            lp_iters = 4
        if m <= 32:
            coord_cycles = 24
        elif m <= 64:
            coord_cycles = 18
        elif m <= 128:
            coord_cycles = 12
        elif m <= 256:
            coord_cycles = 8
        else:
            coord_cycles = 4

        use_lp = m <= 512
        use_minmax = m <= 512

        for trial in range(int(trials)):
            if time.monotonic() >= m_deadline:
                break
            if trial == 0 and seed_full is not None:
                p = _downsample_block_masses(seed_full, m)
            else:
                st = int(rng.integers(0, 16))
                p = _make_initial(m, st, rng)
            sc = _score_block(p, L)
            if np.isfinite(sc):
                update_best(_expand_blocks(p, L), sc)

            if use_lp and time.monotonic() < m_deadline - 8.0:
                lp_dl = min(m_deadline - 4.0, time.monotonic() + max(10.0, 0.52 * float(t_slice)))
                try:
                    p, sc = _lp_refine_block(p, L, lp_dl, lp_iters, rng, use_minmax=use_minmax)
                    if np.isfinite(sc):
                        update_best(_expand_blocks(p, L), sc)
                except Exception:
                    pass

            if time.monotonic() < m_deadline - 4.0:
                coord_dl = min(m_deadline - 2.0, time.monotonic() + max(5.0, 0.35 * float(t_slice)))
                try:
                    p, sc = _block_coordinate_refine(p, L, coord_dl, rng, max_cycles=coord_cycles)
                    if np.isfinite(sc):
                        update_best(_expand_blocks(p, L), sc)
                except Exception:
                    pass

    if best_seq is None:
        base = _two_level(target_n)
        best_seq = base.copy()
        best_score = _score_array(best_seq)
        _add_candidate(fallback, best_seq, best_score, limit=8)

    # Final full-length polish.  Use remaining time, keep the best sequence.
    fallback.sort(key=_candidate_key)
    first_polished = False
    for idx in range(min(3, len(fallback))):
        if time.monotonic() > deadline - 35.0:
            break
        cand_score, cand_seq = fallback[idx]
        if not np.isfinite(cand_score):
            continue
        remaining = max(0.0, deadline - 30.0 - time.monotonic())
        if idx == 0:
            polish_dl = deadline - 25.0
        else:
            polish_dl = time.monotonic() + min(80.0, max(0.0, 0.25 * remaining))
        if polish_dl <= time.monotonic() + 15.0:
            continue
        try:
            pol, pol_sc = _full_coordinate_refine(_normalize(cand_seq), polish_dl, rng, max_perturb=2)
            if np.isfinite(pol_sc):
                update_best(pol, pol_sc)
        except Exception:
            pass
        first_polished = True

    if not first_polished and best_seq is not None:
        try:
            pol, pol_sc = _full_coordinate_refine(_normalize(best_seq), deadline - 20.0, rng, max_perturb=2)
            if np.isfinite(pol_sc):
                update_best(pol, pol_sc)
        except Exception:
            pass

    # Choose final sequence using the authoritative evaluator when available.
    candidates = []
    if best_seq is not None:
        candidates.append(best_seq)
    for item in fallback:
        candidates.append(item[1])
    if not candidates:
        candidates.append(_two_level(target_n))

    chosen = None
    chosen_score = INF
    for seq in candidates:
        if seq is None or seq.size == 0:
            continue
        seqn = _normalize(seq)
        try:
            sc_ext = float(evaluate_sequence([float(x) for x in seqn]))
        except Exception:
            sc_ext = _score_array(seqn)
        if np.isfinite(sc_ext) and sc_ext < chosen_score:
            chosen = seqn
            chosen_score = sc_ext

    if chosen is None:
        chosen = _normalize(best_seq if best_seq is not None else _two_level(target_n))

    out = _normalize(chosen)
    out = np.clip(out, 0.0, 1000.0)
    return [float(x) for x in out]
