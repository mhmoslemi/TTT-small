from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=500, **kwargs):
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)

        initial_h = np.random.rand(n_points)

        scaled = h_clipped * factor

        return scaled_clipped

        corr = np.correlate(h_repaired, 1 - h_repaired, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time.time()
    result_de = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
