import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=500, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values)
    
    def objective(h):
        one_minus_h = 1 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    res = minimize(objective, initial_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-7, options={'maxiter': 1000})
    
    if res.success:
        best_h = res.x
    else:
        best_h = initial_h
    
    c5_val = objective(best_h)
    return best_h.tolist(), c5_val, n_points
