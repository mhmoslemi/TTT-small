        h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
    n_points = len(h_initial)
    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h, n_points)
    time_start = time.time()
    
    def repair(h_vec):
        h_clipped = np.clip(h_vec, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return np.zeros_like(h_vec)
        scale = (n_points / 2) / total
        return h_clipped * scale
    
    def compute_c5(h, n_points):
        dx = 2.0 / n_points
        max_overlap = 0.0
        for k in range(n_points):
            current_overlap = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                current_overlap += h[i] * (1.0 - h[j])
            max_overlap = max(max_overlap, current_overlap)
        return max_overlap * dx
    
    def objective(x):
        h_repaired = repair(x)
        return compute_c5(h_repaired, n_points)
    
    result = opt.minimize(
        fun=objective,
        x0=h_initial,
        method='nelder-mead',
        bounds=[(0, 1)] * n_points,
        options={'maxiter': 1000, 'xatol': 1e-8, 'ftol': 1e-8}
    )
    
    if result.success:
        h_candidate = repair(result.x)
        c5_candidate = compute_c5(h_candidate, n_points)
        if c5_candidate < best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate
    
    final_h = repair(best_h)
    final_c5 = compute_c5(final_h, n_points)
    elapsed_time = time.time() - time_start
    
    if elapsed_time > budget_s:
        return (final_h, final_c5, n_points)
    
    return (final_h, final_c5, n_points)
