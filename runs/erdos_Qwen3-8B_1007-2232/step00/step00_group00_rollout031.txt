        i, j = np.random.choice(n_points, 2, replace=False)
        h_i, h_j = h[i], h[j]
        delta_min = np.max([-h_j, -(1.0 - h_i)])
        delta_max = np.min([1.0 - h_i, h_j])
        if delta_min >= delta_max:
            continue
        delta = np.random.uniform(delta_min, delta_max)
        new_h = h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        if not (0 <= new_h[i] <= 1 and 0 <= new_h[j] <= 1):
            continue
        corr_new = np.correlate(new_h, 1.0 - new_h, mode='full')
        new_c5 = np.max(corr_new) * dx
        if new_c5 < current_c5:
            h = new_h
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
        else:
            if np.random.rand() < np.exp((current_c5 - new_c5) / temp):
                h = new_h
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_h = new_h.copy()
                    best_c5 = new_c5
        temp *= cooling_rate
    h_sum = np.sum(best_h)
    if not np.isclose(h_sum, required_sum):
        delta_adj = (required_sum - h_sum) / n_points
        best_h += delta_adj
    corr_final = np.correlate(best_h, 1.0 - best_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    return (best_h.tolist(), final_c5, n_points)
