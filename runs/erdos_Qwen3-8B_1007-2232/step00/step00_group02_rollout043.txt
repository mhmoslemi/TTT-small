def adjust_sum(h, target_sum):
    current_sum = np.sum(h)
    delta = target_sum - current_sum
    if delta == 0:
        return h
    h += delta / len(h)
    h = np.clip(h, 0, 1)
    return h

    h_clip = np.clip(h, 0, 1)
    corr = np.correlate(h_clip, 1 - h_clip, mode='full')
    val = np.max(corr) * dx
    target_sum = len(h) / 2
    penalty = 1e6 * (np.sum(h_clip) - target_sum)**2
    return val + penalty
