import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=500, **kwargs):
    global initial_h_values
    if 'initial_h_values' in globals():
        h_initial = initial_h_values
    else:
        n_points = 45
        h_initial = np.ones(n_points) * 0.5
    n_points = len(h_initial)
    dx = 2.0 / n_points
    
    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    start_time = time.time()
    result = minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)

        return h_initial, objective(h_initial), n_points
    
    h_values = result.x
    c5_bound = objective(h_values)
    
    # Check integral constraint
    assert np.isclose(np.sum(np.clip(h_values, 0, 1)) * dx, 1.0, atol=1e-4)
    return h_values, c5_bound, n_points
