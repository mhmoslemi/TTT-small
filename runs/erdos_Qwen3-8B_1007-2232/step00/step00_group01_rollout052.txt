            trial_h = h.copy()
            trial_h[i] += 0.01
            trial_h = repair(trial_h, n_points)
            current_c5 = evaluate_erdos_solution(trial_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = trial_h.copy()
                improved = True
            # Try decreasing
            trial_h = h.copy()
            trial_h[i] -= 0.01
            trial_h = repair(trial_h, n_points)
            current_c5 = evaluate_erdos_solution(trial_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = trial_h.copy()
                improved = True
        if not improved:
            old_n = n_points
            n_points *= 2
            h = resample(best_h, old_n, n_points)
            best_h = h.copy()
            best_c5 = evaluate_erdos_solution(best_h)
        elapsed_time = time.time() - start_time
        iterations += 1
    return (best_h, best_c5, n_points)
