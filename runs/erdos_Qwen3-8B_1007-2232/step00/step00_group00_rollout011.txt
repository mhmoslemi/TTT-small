import numpy as np
import time

def run(seed=42, budget_s=500, **kwargs):
    # Assume initial_h_values is provided
    if 'initial_h_values' not in globals():
        raise ValueError("initial_h_values is required")
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # PSO parameters
    n_particles = 20
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_iter = 1000
    epsilon = 1e-5

    # Initialize particles
    particles = np.random.rand(n_particles, n_points)
    velocities = np.random.rand(n_particles, n_points) * 0.1
    # Set first particle to initial_h_values
    particles[0, :] = initial_h.copy()
    # Adjust other particles to have correct sum
    for i in range(1, n_particles):
        h = np.random.rand(n_points)
        current_sum = h.sum()
        target_sum = n_points / 2
        delta = target_sum - current_sum
        h += delta / n_points
        h = np.clip(h, 0, 1)
        particles[i, :] = h

    pbest = particles.copy()
    pbest_fitness = np.full(n_particles, np.inf)
    gbest = particles[0].copy()
    gbest_fitness = np.inf

    start_time = time.time()
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        # Evaluate particles
        for i in range(n_particles):
            h = particles[i]
            if np.any(h < 0) or np.any(h > 1):
                fitness = np.inf
            else:

                if abs(current_sum - target_sum) > epsilon:
                    fitness = np.inf
                else:
