import numpy as np
import random
import time

def run(seed=42, budget_s=500, **kwargs):
    np.random.seed(seed)
    initial_h = kwargs.get('initial_h_values', None)
    n_points = kwargs.get('n_points', 45)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    if initial_h is not None:
        if len(initial_h) != n_points:
            raise ValueError("Initial h length mismatch")
        h_pop = np.array(initial_h, dtype=float)
    else:
        h_pop = np.random.uniform(0, 1, n_points)
        h_pop = (h_pop - np.min(h_pop)) / (np.max(h_pop) - np.min(h_pop))
        h_pop = h_pop * (target_sum - np.sum(h_pop)) / (n_points - np.sum(h_pop)) + h_pop
        h_pop = np.clip(h_pop, 0, 1)

    def feasibility_check(h):
        if np.any(h < 0) or np.any(h > 1):
            return False
        return abs(np.sum(h) - target_sum) < 1e-5

    def repair(h):
        if feasibility_check(h):
            return h
        s = np.sum(h)
        delta = (target_sum - s) / n_points
        h += delta
        h = np.clip(h, 0, 1)
        return h

    def objective(h):
        h = repair(h)
        if not feasibility_check(h):
            return -1000  # penalty
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return -max_corr * dx  # minimize C5

    n_particles = 20
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
    personal_best = h_pop.copy()
    global_best = h_pop.copy()
    best_fitness = objective(h_pop)
    start_time = time.time()

    for _ in range(100000):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

            r1, r2 = np.random.rand(2)
            velocities[i] = inertia * velocities[i] + cognitive * r1 * (personal_best[i] - h_pop[i]) + social * r2 * (global_best - h_pop[i])
