import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=500, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

        c5_bound = np.max(corr) * dx
