import numpy as np
import time

def run(seed=42, budget_s=500, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=np.float64)
    assert np.isclose(np.sum(h) * dx, 1.0), "Initial sum invalid"
    
    def evaluate(h_vec):
        corr = np.correlate(h_vec, 1.0 - h_vec, mode='full')
        return np.max(corr) * dx
    
    current_h = h.copy()
    current_c5 = evaluate(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    temp = 1.0
    cooling = 0.99995
    
    start_time = time.time()
    iteration = 0
    max_steps = int(1e6)
    
    while iteration < max_steps and time.time() - start_time < budget_s:
        i, j = np.random.randint(0, n_points, size=2)
        if i == j:
            j = (j + 1) % n_points
        
        delta_i = np.random.normal(0, 0.0005)
        delta_j = np.random.normal(0, 0.0005)
        
        current_h[i] += delta_i
        current_h[j] += delta_j
        
        current_h[i] = np.clip(current_h[i], 0.0, 1.0)
        current_h[j] = np.clip(current_h[j], 0.0, 1.0)
        
        current_sum = np.sum(current_h)
        delta_sum = 2.0 - current_sum
        
        current_h[i] += delta_sum / 2
        current_h[j] += delta_sum / 2
        
        current_h[i] = np.clip(current_h[i], 0.0, 1.0)
        current_h[j] = np.clip(current_h[j], 0.0, 1.0)
        
        new_c5 = evaluate(current_h)
        
        if new_c5 < current_c5:
            current_c5 = new_c5
        else:
            prob = np.exp(-(new_c5 - current_c5) / temp)
            if np.random.rand() < prob:
                current_c5 = new_c5
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
        
        temp *= cooling
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)
