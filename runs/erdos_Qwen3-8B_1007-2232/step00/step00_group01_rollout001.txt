import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=500, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def clip_to_bounds(h):
        return np.clip(h, 0, 1)

    def scale_to_integral(h, target_sum):
        current_sum = np.sum(h)
        scaling_factor = target_sum / current_sum if current_sum != 0 else 1.0
        return h * scaling_factor

    def objective(h):
        h_repaired = clip_to_bounds(h)
        h_repaired = scale_to_integral(h_repaired, n_points / 2)
        val = np.correlate(h_repaired, 1.0 - h_repaired, mode='full')
        return np.max(val) * dx

    x0 = initial_h_values.copy()
    start_time = time.time()

    result = minimize(
        fun=objective,
        x0=x0,
        method='nelder-mead',

        options={'maxiter': 1000, 'xatol': 1e-5}

    best_h = result.x
    best_h = clip_to_bounds(best_h)
    best_h = scale_to_integral(best_h, n_points / 2)
    c5_bound = objective(best_h)
