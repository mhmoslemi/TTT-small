    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Set up differential evolution options
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=50,  # Reduced iterations for speed
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        disp=False
    )

    best_h = result.x
    c5_bound = result.fun
    return (best_h, c5_bound, n_points)
