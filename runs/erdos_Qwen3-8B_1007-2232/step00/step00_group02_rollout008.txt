import numpy as np
import scipy.optimize as opt
from scipy.signal import correlate

def run(seed=42, budget_s=500, **kwargs):
    if 'initial_h_values' in kwargs:
        h = np.array(kwargs['initial_h_values'])
    else:
        n_points = 50
        h = np.full(n_points, 0.5)
    n_points = len(h)
    dx = 2.0 / n_points
    sum_h = n_points / 2
    h = np.clip(h, 0, 1)
    if not np.isclose(np.sum(h), sum_h):
        h = h * (sum_h / np.sum(h))

    def obj_func(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode='full') * dx
        return np.max(corr)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - sum_h
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    result = opt.minimize(
        obj_func,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        tol=1e-6,
        options={'maxiter': 200, 'ftol': 1e-9, 'gtol': 1e-9}
    )
    
    best_h = result.x
    best_c5 = obj_func(best_h)
    
    return (best_h, best_c5, n_points)
