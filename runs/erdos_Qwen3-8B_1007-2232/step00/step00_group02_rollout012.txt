import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=500, **kwargs):
    try:
        initial_h = initial_h_values
    except NameError:
        n_points = 45
        dx = 2.0 / n_points
        initial_h = np.full(n_points, 0.5)
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h_flat):
        h = np.clip(h_flat, 0, 1)
        sum_h = np.sum(h) * dx
        if abs(sum_h - 1.0) > 1e-6:
            scale = 1.0 / sum_h
            h = h * scale
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)

    # Differential Evolution
    print("Starting differential evolution")
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.9,
        tol=1e-6,
        maxiter=100,
        seed=seed
    )
    if time.time() - start_time >= budget_s:
        return (best_h, best_c5, n_points)
    best_h = result.x
    best_c5 = objective(best_h)

    # Nelder-Mead refinement
    print("Refining with Nelder-Mead")
    result_local = minimize(
        objective,
        x0=best_h,
        method='Nelder-Mead',
        bounds=bounds,
        options={'xatol': 1e-6, 'fatol': 1e-6, 'maxiter': 100}
    )
    if time.time() - start_time >= budget_s:
        return (result_local.x, objective(result_local.x), n_points)
    best_h = result_local.x
    best_c5 = objective(best_h)

    # BFGS refinement
    print("Refining with BFGS")
    result_bfgs = minimize(
        objective,
        x0=best_h,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 50, 'ftol': 1e-6}
    )
    if time.time() - start_time >= budget_s:
        return (result_bfgs.x, objective(result_bfgs.x), n_points)
    best_h = result_bfgs.x
    best_c5 = objective(best_h)

    return (best_h, best_c5, n_points)
