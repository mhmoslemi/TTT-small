        method='Nelder-Mead',
        bounds=bounds,
        options={'xatol': 1e-7, 'fatol': 1e-7}
    )

    final_h = repair_h(result_local.x)
    final_c5 = objective(final_h)
    return (final_h, final_c5, n_points)
