import numpy as np
import time

def run(seed=42, budget_s=500, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=float)
        n_points = len(initial_h)
    else:
        n_points = 45
        initial_h = np.full(n_points, 0.5)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_h += (target_sum - np.sum(initial_h)) / n_points
    initial_h = np.clip(initial_h, 0, 1)
    
    n_particles = 20
    inertia = 0.7
    cognitive = 2.0
    social = 2.0
    max_vel = 0.5
    
    particles = np.full((n_particles, n_points), 0.5)
    particles[0] = initial_h
    for i in range(1, n_particles):
        particles[i] = np.random.uniform(0, 1, n_points)
        particles[i] += (target_sum - np.sum(particles[i])) / n_points
        particles[i] = np.clip(particles[i], 0, 1)
    
    velocities = np.zeros((n_particles, n_points))
    fitness = np.full(n_particles, -1000)
    personal_best = np.copy(particles)
    personal_best_fitness = np.full(n_particles, -1000)
    global_best = np.copy(particles[0])
    best_fitness = -1000
    
    def eval_fitness(h):
        if not (np.all(h >= 0) and np.all(h <= 1)):
            return -1000
        s = np.sum(h)
        if abs(s - target_sum) > 1e-5:
            return -1000
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = dx * np.max(corr)
        return -c5
    
    for i in range(n_particles):
        fitness[i] = eval_fitness(particles[i])
    personal_best_fitness[:] = fitness[:]
    best_fitness = np.max(fitness)
    global_best = particles[np.argmax(fitness)]
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = inertia * velocities[i] + cognitive * r1 * (personal_best[i] - particles[i]) + social * r2 * (global_best - particles[i])
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            s = np.sum(particles[i])
            if abs(s - target_sum) > 1e-5:
                delta = (target_sum - s) / n_points
                particles[i] += delta
                particles[i] = np.clip(particles[i], 0, 1)
            current_fit = eval_fitness(particles[i])
            if current_fit > personal_best_fitness[i]:
                personal_best[i] = particles[i].copy()
                personal_best_fitness[i] = current_fit
            if current_fit > best_fitness:
                best_fitness = current_fit
                global_best = particles[i].copy()
    
    h_values = global_best
    c5_bound = -best_fitness
    return h_values, c5_bound, n_points
