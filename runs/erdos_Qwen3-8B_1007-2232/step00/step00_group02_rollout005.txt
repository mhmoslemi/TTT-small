import numpy as np
import random
import time

def run(seed=42, budget_s=500, **kwargs):
    import numpy as np
    import random
    import time
    
    # Start with the provided initial solution
    try:
        initial_h = initial_h_values
    except NameError:
        # Fallback if initial_h_values is not available
        n_points = 45
        initial_h = np.random.uniform(0, 1, n_points)
        initial_h = initial_h / np.sum(initial_h) * (n_points / 2)
        initial_h = np.clip(initial_h, 0, 1)
    
    n_points = len(initial_h)
    dx = 2.0 / n_points
    current_h = np.copy(initial_h).astype(float)
    best_h = np.copy(current_h)
    best_c5 = compute_c5(current_h)
    
    # Parameters for SA
    initial_temp = 1.0
    cooling_rate = 0.99995
    max_steps = 10**6
    step = 0
    start_time = time.time()
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        corr_max = np.max(corr)
        return corr_max * dx
    
    while time.time() - start_time < budget_s:
        # Perturb two indices
        i, j = np.random.choice(n_points, 2, replace=False)
        delta_i = np.random.normal(0, 0.0005)
        delta_j = np.random.normal(0, 0.0005)
        
        h_new = np.copy(current_h)
        h_new[i] += delta_i
        h_new[j] += delta_j
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust sum to meet constraint
        sum_new = np.sum(h_new)
        target_sum = n_points / 2
        error = sum_new - target_sum
        if abs(error) > 1e-8:
            idx = np.random.randint(n_points)
            adjustment = target_sum - sum_new
            h_new[idx] += adjustment
            h_new[idx] = np.clip(h_new[idx], 0, 1)
        
        new_c5 = compute_c5(h_new)
        delta = best_c5 - new_c5
        temp = initial_temp * (cooling_rate ** step)
        
        if new_c5 < best_c5 or random.random() < np.exp(-delta / temp):
            current_h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
        
        step += 1
        
        # Early stopping if no improvement for 10^5 steps
        if step % 100000 == 0 and np.abs(best_c5 - compute_c5(current_h)) < 1e-8:
            break
    
    return (current_h, best_c5, n_points)
