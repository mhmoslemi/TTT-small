    sum_h = np.sum(h_initial)
    if np.abs(sum_h - n_points / 2) > 1e-6:
        h_initial = h_initial / sum_h * (n_points / 2)
    
    def objective(x):
        return compute_c5(x, n_points)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
    
    best_h = h_initial
    best_c5 = compute_c5(best_h, n_points)
    
    for _ in range(5):
        perturb = np.random.uniform(-0.1, 0.1, n_points)
        new_h = best_h + perturb
        new_h = np.clip(new_h, 0.0, 1.0)
        sum_new = np.sum(new_h)
        if sum_new > 1e-6:
            scale = (n_points / 2) / sum_new
            new_h *= scale
        
        res = minimize(objective, new_h, bounds=bounds, constraints=cons, method='L-BFGS-B')
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
    
    return (best_h.tolist(), best_c5, n_points)
