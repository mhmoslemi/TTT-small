import numpy as np
import scipy.optimize as optimize

def run(seed=42, budget_s=500, **kwargs):
    global initial_h_values
    h = np.array(initial_h_values)
    n = len(h)
    dx = 2.0 / n

    def repair(h_vec):
        h_clipped = np.clip(h_vec, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return np.zeros_like(h_clipped)
        scale = (n / 2) / total
        return h_clipped * scale

    def compute_c5(h_vec):
        h_shifted_all = np.array([np.roll(h_vec, k) for k in range(n)])
        overlaps = np.sum(h_vec * (1 - h_shifted_all), axis=1)
        return np.max(overlaps) * dx

    def objective(params):
        h_repaired = repair(params)
        return compute_c5(h_repaired)

    h = repair(h)
    result = optimize.minimize(objective, h, method='Nelder-Mead', options={'maxiter': 1000})
    best_h = repair(result.x)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
