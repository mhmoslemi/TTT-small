    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - sum_target}]

    # Initial minimize
    result = opt.minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints)
    best_h = result.x
    best_c5 = result.fun

    # Perturb and re-optimize
    for _ in range(20):
        perturb = np.random.normal(0, 0.1, size=n_points)
        new_h = x0 + perturb
        new_h = clip_and_rescale(new_h)
        result = opt.minimize(objective, new_h, method='SLSQP', bounds=bounds, constraints=constraints)
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
            x0 = result.x

    return best_h, best_c5, n_points
