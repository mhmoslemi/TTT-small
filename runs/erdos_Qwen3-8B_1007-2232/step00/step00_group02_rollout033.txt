import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def repair(h, dx, n_points):
    h_clipped = np.clip(h, 0.0, 1.0)
    current_sum = np.sum(h_clipped)
    target_sum = n_points / 2.0
    if np.isclose(current_sum, target_sum, atol=1e-6):
        return h_clipped
    scale = target_sum / current_sum
    h_scaled = np.clip(h_clipped * scale, 0.0, 1.0)
    return h_scaled

def objective(h_candidate, dx, n_points):
    h_repaired = repair(h_candidate, dx, n_points)
    corr = np.correlate(h_repaired, 1 - h_repaired, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=500, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values)
    initial_h_repaired = repair(initial_h, dx, n_points)
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    result_de = None
    try:
        result_de = differential_evolution(
            lambda x: objective(x, dx, n_points),
            bounds, 
            strategy='best1bin',
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.9,
            maxiter=500,
            seed=seed,
            disp=False
        )
    except:
        pass
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (initial_h_repaired, objective(initial_h_repaired, dx, n_points), n_points)
    
    best_h_de = result_de.x if result_de else initial_h_repaired
    best_h_repaired = repair(best_h_de, dx, n_points)
    best_c5 = objective(best_h_repaired, dx, n_points)
    
    try:
        result_nm = minimize(
            lambda x: objective(x, dx, n_points),
            best_h_repaired,
            method='Nelder-Mead',
            bounds=bounds,
            options={'xatol': 1e-6, 'fatol': 1e-6, 'maxiter': 200}
        )
    except:
        result_nm = None
    
    if result_nm is not None:
        best_h_nm = result_nm.x
        best_h_nm_repaired = repair(best_h_nm, dx, n_points)
        best_c5_nm = objective(best_h_nm_repaired, dx, n_points)
        if best_c5_nm < best_c5:
            best_h = best_h_nm_repaired
            best_c5 = best_c5_nm
        else:
            best_h = best_h_repaired
    else:
        best_h = best_h_repaired
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (best_h, best_c5, n_points)
    
    return (best_h, best_c5, n_points)
