        candidate[j] = np.clip(candidate[j], 0.0, 1.0)

        sum_candidate = np.sum(candidate)
        delta_sum = sum_candidate - target_sum
        if delta_sum != 0:
            candidate[0] += delta_sum
            candidate[0] = np.clip(candidate[0], 0.0, 1.0)
            sum_candidate = np.sum(candidate)
            delta_sum = sum_candidate - target_sum

        old_c5 = compute_c5(h)
        new_c5 = compute_c5(candidate)

        if np.random.rand() < np.exp((old_c5 - new_c5) / temp):
            h = candidate

        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = candidate.copy()

        temp *= cooling_rate
        steps += 1

    return (best_h, best_c5, n_points)
