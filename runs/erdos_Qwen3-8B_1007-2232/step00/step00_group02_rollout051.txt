import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=500, **kwargs):
    # Handle initial_h_values
    if 'initial_h_values' in globals():
        h_initial = initial_h_values
    else:
        n = 45
        h_initial = np.random.rand(n)
        h_initial = h_initial / np.sum(h_initial) * (n / 2)
        h_initial = np.clip(h_initial, 0, 1)
    n = len(h_initial)
    target_sum = n / 2
    h_initial = h_initial[:-1]

    def obj(h):
        h_full = np.concatenate([h, [target_sum - np.sum(h)]])
        h_full = np.clip(h_full, 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * (2/n)

    bounds = [(0, 1) for _ in h_initial]
    result = opt.differential_evolution(obj, bounds, strategy='best1bin',
                                        maxiter=100, popsize=15, tol=1e-6,
                                        mutation=(0.5, 1), recombination=0.8, seed=seed)
    h_opt = result.x
    h_full = np.concatenate([h_opt, [target_sum - np.sum(h_opt)]])
    h_full = np.clip(h_full, 0, 1)
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    c5 = np.max(corr) * (2 / n)
    return (h_full, c5, n)
