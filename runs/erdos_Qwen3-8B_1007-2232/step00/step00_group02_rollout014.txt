import numpy as np
import time
import random

def run(seed=42, budget_s=500, **kwargs):
    np.random.seed(seed)
    initial_h_values = globals().get('initial_h_values', None)
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:

    def obj(h):

        sum_h = np.sum(h_clip) * dx

        c5 = np.max(corr) * dx
        penalty = 1e5 * max(0, abs(sum_h - 1.0))**2
        return c5 + penalty
    num_particles = 20
    particles = [h.copy() for _ in range(num_particles)]
    velocities = [np.zeros_like(p) for p in particles]
    pbest = [p.copy() for p in particles]
    gbest = particles[0].copy()
    best_obj = obj(gbest)
    start_time = time.time()
    for _ in range(500):
        if time.time() - start_time > budget_s:

            r1, r2 = np.random.rand(2)
            velocities[i] = 0.9 * velocities[i] + 1.5 * r1 * (pbest[i] - particles[i]) + 1.5 * r2 * (gbest - particles[i])

            current_obj = obj(particles[i])
