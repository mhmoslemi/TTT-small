def run(seed=42, budget_s=500, **kwargs):
    import numpy as np
    import time

    # Use the initial_h_values
    h_initial = initial_h_values
    n = len(h_initial)
    dx = 2.0 / n
    sum_target = n / 2

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    # Function to repair the solution
    def repair(h):
        h = np.clip(h, 0.0, 1.0)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h) + 1.0 / n * sum_target
        scaling = sum_target / total
        h = h * scaling
        return h

    h_current = np.array(h_initial, dtype=np.float64)
    best_h = np.array(h_current, dtype=np.float64)
    best_c5 = compute_c5(h_current)

    start_time = time.time()
    iterations = 0

    while time.time() - start_time < budget_s:
        # Perturb the current solution
        noise = np.random.normal(0, 0.05, size=n)
        h_perturbed = h_current + noise
        h_perturbed = repair(h_perturbed)

        # Evaluate the new solution
        current_c5 = compute_c5(h_perturbed)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_perturbed
            h_current = h_perturbed
        iterations += 1

    return best_h, best_c5, n
