    def compute_c5_fft(h):
        h2 = 1.0 - h
        n = len(h)
        fft_size = 2 * n - 1
        fft_h = np.fft.fft(h, fft_size)
        fft_h2 = np.fft.fft(h2, fft_size)
        corr = np.fft.ifft(fft_h * np.conj(fft_h2)).real
        return corr.max() * dx
    
    for _ in range(steps):
        if time.time() - start_time > budget_s:
            break
        i = np.random.randint(n_points)
        j = np.random.randint(n_points)
        while i == j:
            j = np.random.randint(n_points)
        delta = 0.0005 * np.random.randn()
        h_i = h[i] + delta
        h_j = h[j] - delta
        h_i = np.clip(h_i, 0.0, 1.0)
        h_j = np.clip(h_j, 0.0, 1.0)
        h_new = h.copy()
        h_new[i] = h_i
        h_new[j] = h_j
        current_sum = np.sum(h_new)
        delta_sum = target_sum - current_sum
        if abs(delta_sum) > 1e-6:
            h_new[0] += delta_sum
            h_new[0] = np.clip(h_new[0], 0.0, 1.0)
        new_c5 = compute_c5_fft(h_new)
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
            best_h = h_new.copy()
        else:
            temp = initial_temp * (cooling_rate) ** _
            prob = np.exp((new_c5 - best_c5) / temp)
            if random.random() < prob:
                h = h_new.copy()
    return (best_h.tolist(), best_c5, n_points)
