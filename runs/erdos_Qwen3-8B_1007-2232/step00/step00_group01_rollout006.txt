import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=500, **kwargs):
    np.random.seed(seed)
    n_points = 47
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()
    best_h = initial_h.copy()
    best_c5 = np.inf
    start_time = time.time()
    
    def repair(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h) + 0.5
        scale = (n_points / 2) / total
        return h * scale
    
    def objective(h_vec):
        h = repair(h_vec)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def optimize():
        nonlocal best_c5, best_h
        for _ in range(10):
            if time.time() - start_time > budget_s - 10:
                return
            perturbed = initial_h + 0.1 * np.random.randn(n_points)
            perturbed = np.clip(perturbed, 0, 1)
            total = np.sum(perturbed)
            if total == 0:
                perturbed = np.zeros_like(perturbed) + 0.5
            else:
                scale = (n_points / 2) / total

                    curr_c5 = objective(curr_h)

            except:
                pass
    
    optimize()
    return (best_h, best_c5, n_points)
