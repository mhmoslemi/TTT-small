import numpy as np
from scipy.optimize import minimize
import time
import copy

def run(seed=42, budget_s=500, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)
    dx = 2.0 / n_points
    sum_target = n_points / 2

    def compute_c5(h_flat):
        h = h_flat.reshape(n_points)
        h_1_minus = 1 - h
        corr = np.correlate(h, h_1_minus, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    def objective(h_flat):
        return compute_c5(h_flat)

    def repair_h(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        scale = sum_target / current_sum if current_sum != 0 else 1.0
        h *= scale
        return h.flatten()

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h_flat: np.sum(h_flat) - sum_target})

    def optimize(start_h):
        result = minimize(
            objective, 
            start_h, 
            method='L-BFGS-B', 
            bounds=bounds, 
            constraints=cons,
            tol=1e-8,
            options={'maxls': 20, 'ftol': 1e-9}
        )
        if result.success:
            return repair_h(result.x)
        return start_h

    best_h = copy.copy(h)
    best_c5 = compute_c5(best_h)
    best_n = n_points
    elapsed = 0.0

    if budget_s > 10:
        for _ in range(3):
            if elapsed >= budget_s:
                break
            start_time = time.time()
            best_h = optimize(best_h)
            best_c5 = compute_c5(best_h)
            elapsed = time.time() - start_time

    return (best_h, best_c5, best_n)
