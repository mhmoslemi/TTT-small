import numpy as np

def run(seed=42, budget_s=500, **kwargs):

    import numpy as np

    # Assume initial_h_values is available

    n_points = len(initial_h_values)

    h = np.array(initial_h_values, dtype=np.float64)

    dx = 2.0 / n_points

    # Compute initial C5

    def compute_c5(h):

        corr = np.correlate(h, 1 - h, mode='full')

        return np.max(corr) * dx

    current_c5 = compute_c5(h)

    best_h = h.copy()

    best_c5 = current_c5

    # Set up parameters

    T = 1.0

    cooling_rate = 0.999

    max_iterations = int(budget_s / 0.01)  # 500 seconds / 0.01 sec per iteration → 50000 iterations

    import time

    start_time = time.time()

    for step in range(max_iterations):

        if time.time() - start_time > budget_s:

            break

        # Generate indices i and j

        i = np.random.randint(0, n_points)

        j = np.random.randint(0, n_points)

        if i == j:

            j = (i + 1) % n_points

        # Perturb

        delta_range = 0.05

        new_h = h.copy()

        hi = new_h[i]

        hj = new_h[j]

        delta = np.random.uniform(-delta_range, delta_range)

        new_hi = hi + delta

        new_hi = np.clip(new_hi, 0., 1.)

        change_i = new_hi - hi

        new_hj = hj - change_i

        new_hj = np.clip(new_hj, 0., 1.)

        new_h[i] = new_hi

        new_h[j] = new_hj

        # Compute new C5

        new_c5 = compute_c5(new_h)

        # Simulated annealing acceptance

        if new_c5 < current_c5:

            current_c5 = new_c5

            h = new_h.copy()

        else:

            # Compute probability

            prob = np.exp((current_c5 - new_c5) / T)

            if np.random.rand() < prob:

                current_c5 = new_c5

                h = new_h.copy()

        # Update best found

        if current_c5 < best_c5:

            best_c5 = current_c5

            best_h = h.copy()

        # Cool temperature

        T *= cooling_rate

    # After iterations, return best_h and best_c5

    return (best_h.tolist(), best_c5, n_points)
