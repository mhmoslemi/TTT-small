import numpy as np

import time

def run(seed=42, budget_s=500, **kwargs):
    import numpy as np
    from scipy.optimize import Bounds, differential_evolution, minimize

    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        h_clipped = np.clip(h, 0, 1)
        sum_clipped = np.sum(h_clipped) * dx
        if abs(sum_clipped - 1.0) > 1e-6:
            scale = 1.0 / sum_clipped
            h_clipped *= scale
            h_clipped = np.clip(h_clipped, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    start_time = time.time()

    # Differential evolution with early exit
    result = differential_evolution(objective, bounds, popsize=15, strategy='best1bin', tol=1e-6, seed=seed)
    best_h = result.x

    # Local optimization with Nelder-Mead
    res = minimize(objective, best_h, method='Nelder-Mead', bounds=bounds, tol=1e-6, options={'xatol': 1e-6, 'fatol': 1e-6})

    final_h = res.x
    h_valid = np.clip(final_h, 0, 1)
    sum_valid = np.sum(h_valid) * dx
    if abs(sum_valid - 1.0) > 1e-6:
        scale = 1.0 / sum_valid
        h_valid *= scale

    corr = np.correlate(h_valid, 1 - h_valid, mode='full')
    c5 = np.max(corr) * dx
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (final_h, c5, n_points)
    return (h_valid, c5, n_points)
