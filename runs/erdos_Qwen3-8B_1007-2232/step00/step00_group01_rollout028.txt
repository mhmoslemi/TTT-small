def run(seed=42, budget_s=500, **kwargs):
    # Initialize with provided initial values
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Repair function to maintain integral constraint
    def repair(h, n):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return np.zeros_like(h_clipped)
        factor = (n / 2) / total
        return h_clipped * factor
    
    # Initial repair
    h = repair(h, n_points)
    
    # Best solution tracking
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    
    # Iterative improvement loop
    while time.time() - start_time < budget_s:
        # Perturb a random index

        # Generate small random delta

        h_new = h.copy()
        h_new[i] += delta
        
        # Clip and repair
        h_new = np.clip(h_new, 0, 1)
