import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=500, **kwargs):
    # Use initial_h_values if provided, else generate a random one and scale
    initial_h = kwargs.get('initial_h_values', None)
    if initial_h is None:
        n_points = 100  # Use 100 samples for faster optimization
        initial_h = np.random.rand(n_points)
        initial_h = np.clip(initial_h, 0, 1)
        target_sum = n_points / 2
        current_sum = np.sum(initial_h)
        initial_h = initial_h * target_sum / current_sum
    else:
        n_points = len(initial_h)
        # Ensure initial_h is clipped and scaled for the integral constraint
        initial_h = np.clip(initial_h, 0, 1)
        target_sum = n_points / 2
        current_sum = np.sum(initial_h)
        initial_h = initial_h * target_sum / current_sum
    
    dx = 2.0 / n_points
    best_h = initial_h.copy()
    best_c5 = float('inf')

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr)

    def constraint_func(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Warm start with initial_h
    result = minimize(objective, initial_h, method='L-BFGS-B', bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_func}], tol=1e-6)

    if result.success:
        best_h = result.x
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * dx

    # Perform a few more iterations with Nelder-Mead
    start_time = time.time()
    while time.time() - start_time < budget_s:
        try:
            res = minimize(objective, best_h, method='Nelder-Mead', bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_func}], tol=1e-5)
            if res.success:
                curr_corr = np.correlate(res.x, 1 - res.x, mode='full')
                curr_c5 = np.max(curr_corr) * dx
                if curr_c5 < best_c5:
                    best_h = res.x
                    best_c5 = curr_c5
        except:
            break

    # Final evaluation
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx

    return (best_h, final_c5, n_points)
