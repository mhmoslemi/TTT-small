import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=500, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def objective(h):
        h_array = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_array) * dx
        if np.abs(sum_h - 1.0) > 1e-8:
            scale = 1.0 / sum_h
            h_array = h_array * scale
        corr = np.correlate(h_array, 1 - h_array, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=20,
        maxiter=200,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        disp=False
    )

    h_opt = result.x
    h_clipped = np.clip(h_opt, 0, 1)
    sum_h = np.sum(h_clipped) * dx
    scale = 1.0 / sum_h
    h_scaled = h_clipped * scale
    corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
    c5_bound = np.max(corr) * dx

    return (h_scaled, c5_bound, n_points)
