        h_initial = np.clip(h_initial * scale, 0, 1)
    n_points = len(h_initial)
    bounds = [(0.0, 1.0)] * n_points
    
    def objective(h):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        penalty = 1e5 * (np.sum(h) - n_points / 2) ** 2
        return c5 + penalty
    
    result = differential_evolution(objective, bounds, 
                                   strategy='best1bin', 
                                   popsize=15, 
                                   maxiter=100, 
                                   tol=1e-6, 
                                   mutation=(0.5, 1), 
                                   recombination=0.8, 
                                   seed=seed, 
                                   disp=False)
    h_opt = result.x
    c5_opt = np.max(np.correlate(h_opt, 1 - h_opt, mode='full')) * (2.0 / len(h_opt))
    return (h_opt, c5_opt, len(h_opt))
