        h = np.clip(h, 0, 1)
    
    def objective(h_vec):

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - required_sum})

    start_time = time.time()
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 200, 'ftol': 1e-8})
