import numpy as np
import time

def run(seed=42, budget_s=500, **kwargs):
    np.random.seed(seed)
    try:
        h = initial_h_values.copy()
    except NameError:
        h = np.random.rand(47)
    n_points = len(h)
    
    def repair(h, n):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        scale = (n / 2.0) / total if total != 0 else 1.0
        return h * scale
    
    h = repair(h, n_points)
    
    def compute_overlap(h, n):
        dx = 2.0 / n
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = compute_overlap(h, n_points)
    time_start = time.time()
    
    while time.time() - time_start < budget_s:
        improved = False
        for i in range(n_points):
            new_h = h.copy()
            for delta in [0.05, -0.05]:
                new_h = np.clip(new_h, 0, 1)
                new_h[i] = np.clip(new_h[i] + delta, 0, 1)
                new_h = repair(new_h, n_points)
                c5 = compute_overlap(new_h, n_points)
                if c5 < best_c5:
                    best_h = new_h
                    best_c5 = c5
                    improved = True
        if not improved:
            n_points *= 2
            x_old = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h_new = np.interp(x_new, x_old, h)
            h_new = repair(h_new, n_points)
            best_h = h_new
            best_c5 = compute_overlap(h_new, n_points)
    
    return best_h, best_c5, n_points
