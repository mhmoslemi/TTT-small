    """Compute the maximum overlap C5 for a step function h of length n.
    The overlap is defined by C5 = max_k ∫ h(x)(1-h(x+k)) dx, approximated by the discretization.
    Args:
        obj: Discrete function values of h (array-like)
        dx: Grid spacing (dx = 2/n)
        n: Number of grid points
        use_mode_full: Whether to use 'full' convolution (default is True)
    Returns:
        Maximum overlap (C5)
    """
    # Option to use 'full' or 'valid' or 'same' mode
    mode = 'full' if use_mode_full else 'valid'
