import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=500, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Initialize with provided initial_h_values if available
    try:
        initial_h = initial_h_values
        n_points = len(initial_h)
        if n_points == 0:
            n_points = 47
            initial_h = np.full(n_points, 0.5)
    except NameError:
        n_points = 47
        initial_h = np.full(n_points, 0.5)

    # Repair function: clip and scale
    def repair_h(h):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return h_clipped
        scale = (n_points / 2) / total
        return h_clipped * scale

    # Compute C5
    def compute_c5(h):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr

    # Objective function
    def objective(x):
        h_repaired = repair_h(x)
        return compute_c5(h_repaired)

    # Initial optimization
    result = minimize(
        objective, 
        x0=initial_h, 
        method='nelder-mead',
        options={'xatol': 1e-5, 'fatol': 1e-5, 'maxiter': 1000, 'maxfev': 5000}
    )

    # Early exit if time is up
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        best_h = repair_h(result.x)
        best_c5 = compute_c5(best_h)
        return best_h, best_c5, n_points

    # Perturb and restart if time remains
    for _ in range(2):
        if elapsed >= budget_s:
            break
        # Perturb and scale
        x0_perturb = result.x + np.random.normal(0, 0.05, n_points)
        x0_perturb = np.clip(x0_perturb, 0, 1)
        x0_perturb = repair_h(x0_perturb)

        # Run optimization
        result_perturb = minimize(
            objective,
            x0=x0_perturb,
            method='nelder-mead',
            options={'xatol': 1e-5, 'fatol': 1e-5, 'maxiter': 300, 'maxfev': 1000}
        )
        if compute_c5(result_perturb.x) < compute_c5(result.x):
            result = result_perturb
        elapsed = time.time() - start_time

    best_h = repair_h(result.x)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
