import numpy as np
import time

def run(seed=42, budget_s=500, **kwargs):
    np.random.seed(seed)
    initial_h_values = globals().get('initial_h_values')
    n_points = len(initial_h_values) if initial_h_values is not None else 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize swarm
    n_particles = 20
    particles = np.random.rand(n_particles, n_points)
    if initial_h_values is not None:
        particles[0] = initial_h_values
    velocities = np.random.rand(n_particles, n_points) * 0.1
    p_best = particles.copy()
    p_fitness = np.full(n_particles, np.inf)
    g_best = particles[0].copy()
    g_fitness = np.inf

    def repair(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        delta = target_sum - current_sum
        if abs(delta) < 1e-5:
            return h
        h += delta / n_points
        h = np.clip(h, 0, 1)
        return h

    def eval_fitness(h):
        h = repair(h)
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        return -max_corr * dx  # Minimize C, so negated for maximization

    start_time = time.time()
    for _ in range(10000):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        for i in range(n_particles):
            h = particles[i]
            h_fitness = eval_fitness(h)
            if h_fitness < p_fitness[i]:
                p_fitness[i] = h_fitness
                p_best[i] = h.copy()
            if h_fitness < g_fitness:
                g_fitness = h_fitness
                g_best = h.copy()
        
        for i in range(n_particles):
            h = particles[i]
            vel = velocities[i]
            r1, r2 = np.random.rand(), np.random.rand()
            cognitive = 2.0
            social = 2.0
            vel = 0.7 * vel + cognitive * r1 * (p_best[i] - h) + social * r2 * (g_best - h)
            velocities[i] = vel
            particles[i] = h + vel
            particles[i] = np.clip(particles[i], 0, 1)

    final_h = g_best
    final_c5 = -g_fitness
    return (final_h, final_c5, n_points)
