        # Generate delta_i and delta_j

        delta_i = np.random.normal(0, 0.0005)
        delta_j = np.random.normal(0, 0.0005)
        new_h = h.copy()
        new_h[i] = np.clip(new_h[i] + delta_i, 0, 1)
        new_h[j] = np.clip(new_h[j] - delta_j, 0, 1)

        # Check sum again
        current_sum = np.sum(new_h)
        delta_sum = current_sum - target_sum
        # If delta_sum is not zero, it's due to clipping. But proceed anyway
        # Compute new C5
