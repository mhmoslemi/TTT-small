import numpy as np
import time

def run(seed=42, budget_s=500, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def feasibility_check(h):
        return np.all(h >= 0) and np.all(h <= 1) and abs(np.sum(h) - target_sum) < 1e-5

    def repair(h):
        h = h.copy()
        s = np.sum(h)
        delta = (target_sum - s) / n_points
        h += delta
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        delta = (target_sum - s) / n_points
        h += delta
        return h

    def compute_C(h):
        h_arr = np.array(h, dtype=float)
        corr = np.correlate(h_arr, 1 - h_arr, mode="full")
        return dx * np.max(corr)

    def evaluate_fitness(h):
        h = np.array(h)
        if not feasibility_check(h):
            return -1000
        return -compute_C(h)

    def update_particle(pos, vel, pbest, gbest, inertia, c1, c2, bounds):
        r1, r2 = np.random.rand(), np.random.rand()
        vel = inertia * vel + c1 * r1 * (pbest - pos) + c2 * r2 * (gbest - pos)
        vel = np.clip(vel, -0.5, 0.5)
        pos = pos + vel
        pos = np.clip(pos, bounds[0], bounds[1])
        pos = repair(pos)
        return pos, vel

    n_particles = 20
    particles = np.array([np.array(initial_h) + np.random.uniform(-0.1, 0.1, n_points) for _ in range(n_particles)])
    velocities = np.random.uniform(-0.5, 0.5, (n_particles, n_points))
    pbest = particles.copy()
    pbest_fitness = np.array([evaluate_fitness(p) for p in particles])
    gbest = particles[np.argmax(pbest_fitness)]
    gbest_fitness = np.max(pbest_fitness)
    best_h = gbest
    best_c5 = np.inf
    start_time = time.time()

    iterations = 0
    while time.time() - start_time < budget_s:
        iterations += 1
        for i in range(n_particles):
            pos = particles[i]
            vel = velocities[i]
            new_pos, new_vel = update_particle(pos, vel, pbest[i], gbest, 0.7, 2.0, 2.0, (0.0, 1.0))
            new_fitness = evaluate_fitness(new_pos)
            if new_fitness > pbest_fitness[i]:
                pbest[i] = new_pos
                pbest_fitness[i] = new_fitness
                if new_fitness > gbest_fitness:
                    gbest = new_pos
                    gbest_fitness = new_fitness
            particles[i] = new_pos
            velocities[i] = new_vel
        if np.abs(gbest_fitness - best_c5) < 1e-6:
            break
        best_c5 = gbest_fitness
        best_h = gbest

    final_h = repair(best_h)
    final_c5 = compute_C(final_h)
    return final_h, final_c5, n_points
