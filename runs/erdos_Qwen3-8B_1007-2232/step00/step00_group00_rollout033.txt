import numpy as np
import time

def run(seed=42, budget_s=500, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    assert np.isclose(np.sum(initial_h) * dx, 1.0), "Initial h violates constraint"

    def compute_c5(h):
        conv = np.correlate(h, 1.0 - h, mode='full')
        return np.max(conv) * dx

    current_h = initial_h.copy()
    best_h = current_h.copy()
    best_c5 = compute_c5(current_h)
    temp = 1.0
    cooling_rate = 0.995
    start_time = time.time()

    while time.time() - start_time < budget_s:
        i, j = np.random.choice(n_points, size=2, replace=False)
        h_i, h_j = current_h[i], current_h[j]
        max_delta = min(1.0 - h_i, h_j)
        delta = np.random.uniform(-max_delta, max_delta)
        new_h = current_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5:
            current_h, best_h, best_c5 = new_h, new_h, new_c5
        else:
            delta_c = new_c5 - best_c5
            if np.random.random() < np.exp(-delta_c / temp):
                current_h = new_h
        temp *= cooling_rate

    return (best_h, best_c5, n_points)
