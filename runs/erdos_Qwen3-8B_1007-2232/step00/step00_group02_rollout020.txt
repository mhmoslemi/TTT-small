    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(45, dtype=np.float64)
        h[0:15] = 1.0
        h = h / np.sum(h)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5

    temp = 1.0
    cooling_rate = 0.99995
    steps = 1000000
    start_time = time.time()

    for _ in range(steps):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

        i, j = np.random.randint(0, n_points, size=2)
        if i == j:
            j = (j + 1) % n_points

        delta_i = np.random.normal(0, 0.0005)
        delta_j = np.random.normal(0, 0.0005)

        h[i] = np.clip(h[i] + delta_i, 0.0, 1.0)
        h[j] = np.clip(h[j] - delta_j, 0.0, 1.0)

        current_sum = np.sum(h)
        delta_sum = target_sum - current_sum

        if abs(delta_sum) > 1e-6:
            old_val = h[0]
            new_val = old_val + delta_sum
            h[0] = np.clip(new_val, 0.0, 1.0)

        new_c5 = compute_c5(h)

        if new_c5 < current_c5:
            current_c5 = new_c5
            h = h.copy()
        else:
            prob = np.exp(-(new_c5 - current_c5) / temp)
            if np.random.rand() < prob:
                current_c5 = new_c5
                h = h.copy()

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        temp *= cooling_rate

    return (best_h.tolist(), best_c5, n_points)
