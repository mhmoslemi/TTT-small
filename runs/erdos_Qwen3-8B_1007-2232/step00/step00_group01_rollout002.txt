import numpy as np
import scipy.optimize

def run(seed=42, budget_s=500, **kwargs):
    h_initial = initial_h_values
    n_points = len(h_initial)
    dx = 2.0 / n_points
    sum_target = n_points / 2.0

    def compute_c5(h):
        cross_corr = np.correlate(h, 1 - h, mode='full')
        return np.max(cross_corr) * dx

    def objective(h_vec):
        return compute_c5(h_vec)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - sum_target}]

    result = scipy.optimize.minimize(
        fun=objective,
        x0=h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={
            'maxiter': 200,
            'ftol': 1e-8,
            'disp': False
        }
    )

    best_h = result.x
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_points)
