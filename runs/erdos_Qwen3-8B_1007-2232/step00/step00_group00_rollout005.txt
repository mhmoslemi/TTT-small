            h_initial += delta / n_points
            h_initial = np.clip(h_initial, 0, 1)

    else:
        # Generate random with adjusted sum
        h_base = np.random.uniform(0, 1, n_points)
        sum_h = np.sum(h_base)
        delta = (n_points / 2) - sum_h
        h_base += delta / n_points

        h = np.clip(h_vec, 0, 1)
        sum_h = np.sum(h)
        if abs(sum_h - n_points / 2) > 1e-5:
