def run(seed=42, budget_s=500, **kwargs):

    import numpy as np
    import random

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    assert np.isclose(np.sum(initial_h) * dx, 1.0), "Initial h not satisfying integral constraint"

    # Convert to numpy array for easier manipulation
    h = np.array(initial_h, dtype=np.float64)

    # Function to compute C5
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5

    # Parameters for optimization
    max_iterations = int(budget_s / 0.01)  # Assuming 0.01 seconds per iteration
    # Adjust max_iterations to fit within budget_s
    max_iterations = int(budget_s / 0.01)  # Simplified, but adjust as needed

    # Perturbation parameters
    for iteration in range(max_iterations):
        # Select two distinct indices
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        while j == i:
            j = np.random.randint(0, n_points)

        # Compute delta_lo and delta_hi
        delta_lo = max(-h[i], - (1 - h[j]))
        delta_hi = min(1 - h[i], h[j])

        if delta_lo > delta_hi:
            continue  # No valid delta for this pair

        delta = np.random.uniform(delta_lo, delta_hi)
        # Apply delta to indices i and j
        new_h_i = h[i] + delta
        new_h_j = h[j] - delta

        # Create new h
        new_h = h.copy()
        new_h[i] = new_h_i
        new_h[j] = new_h_j

        # Compute C5 for the new h
        new_c5 = compute_c5(new_h)

        # Accept the new solution if it's better
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
        elif np.random.rand() < np.exp((best_c5 - new_c5) / (0.1 * iteration + 1)):
            # Simulated annealing: accept with some probability
            best_c5 = new_c5
            best_h = new_h.copy()

        # Also, check if the current best is better than the initial
        if best_c5 < current_c5:
            current_c5 = best_c5

        # Optional: print progress
        # print(f"Iteration {iteration}, Best C5: {best_c5}")

    return (best_h.tolist(), best_c5, n_points)
