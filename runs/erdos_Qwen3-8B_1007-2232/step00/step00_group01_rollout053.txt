import scipy.optimize as opt
import time

def run(seed=42, budget_s=500, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    best_c5 = compute_c5(initial_h)
    best_h = initial_h.copy()

    for _ in range(20):
        np.random.seed(seed + _)
        perturbed = initial_h + np.random.uniform(-0.2, 0.2, n_points)
        perturbed = np.clip(perturbed, 0.0, 1.0)
        sum_perturbed = np.sum(perturbed)
        delta = (n_points / 2) - sum_perturbed
        perturbed += delta / n_points
        perturbed = np.clip(perturbed, 0.0, 1.0)

        result = opt.minimize(
            compute_c5,
            perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            tol=1e-6,
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        if time.time() - start_time > budget_s:
