import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=500, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=float)
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h
    h *= scale
    best_h = h.copy()
    best_c5 = compute_c5(h)
    time_start = time.time()
    time_remaining = budget_s
    n_current = n_points
    max_iter = 10000
    iterations = 0
    
    while time_remaining > 1e-3 and iterations < max_iter:
        improved = False
        for _ in range(100):  # Perturbation attempts per iteration
            i = np.random.randint(n_current)
            delta = np.random.uniform(-0.05, 0.05)
            h_i = h[i] + delta
            h_i = np.clip(h_i, 0, 1)
            h[i] = h_i
            sum_h = np.sum(h)
            scale = (n_current / 2) / sum_h
            h *= scale
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                improved = True
        if not improved:
            n_new = n_current * 2
            h_new = np.repeat(h, 2)
            sum_h_new = np.sum(h_new)
            scale = (n_new / 2) / sum_h_new
            h_new *= scale
            h = h_new.copy()
            best_h = h.copy()
            best_c5 = compute_c5(h)
            n_current = n_new
        time_used = time.time() - time_start
        time_remaining = budget_s - time_used
        iterations += 1
    
    return best_h, best_c5, n_current

def compute_c5(h):
    dx = 2.0 / len(h)
    kernel = 1 - h
    corr = correlate(h, kernel, mode='full')
    return np.max(corr) * dx
