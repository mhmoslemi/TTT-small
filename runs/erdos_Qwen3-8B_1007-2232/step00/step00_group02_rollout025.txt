import numpy as np
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=500, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    delta = n_points / 2 - sum_h
    h[0] += delta
    h = np.clip(h, 0, 1)
    
    def obj_func(h_vec):
        h_clip = np.clip(h_vec, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        max_corr = np.max(corr) * dx
        sum_penalty = 1e2 * (np.sum(h_clip) - n_points / 2) ** 2
        return max_corr + sum_penalty
    
    bounds = [(0, 1)] * n_points
    result = differential_evolution(obj_func, bounds, strategy='best1bin', maxiter=100, popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.8, disp=False)
    
    best_h = result.x
    final_h = np.clip(best_h, 0, 1)
    corr = np.correlate(final_h, 1 - final_h, mode='full')
    c5_bound = np.max(corr) * dx
    return (final_h, c5_bound, n_points)
