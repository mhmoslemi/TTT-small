import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=500, **kwargs):
    # Use existing initial_h_values if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=np.float64)
        n = len(initial_h)
    else:
        # Fallback to a basic 50%-filled array with uniform spacing
        n = 47
        initial_h = np.full(n, 0.5, dtype=np.float64)
    
    dx = 2.0 / n
    
    # Repair function to clip and normalize
    def repair(h_vec):
        h = np.clip(h_vec, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 1.0 / len(h)) * (len(h) / 2)
        scale = (n / 2) / total
        return h * scale
    
    # Objective function
    def objective(h_params):
        h = repair(h_params)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Optimization settings
    options = {
        'maxiter': 500,
        'xatol': 1e-5,
        'fatol': 1e-5,
        'adaptive': True
    }
    
    start_time = time.time()
    result = minimize(objective, x0=initial_h, method='Nelder-Mead', options=options)
    elapsed = time.time() - start_time
    
    # Extract best solution
    best_h = repair(result.x)
    c5_bound = objective(best_h)
    
    # Optional: Add early termination if time is near budget
    if elapsed > budget_s - 10:
        pass  # Return what we have
    
    return (best_h, c5_bound, n)
