import numpy as np
from scipy.optimize import differential_evolution

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def objective(h, n_points):
    c5 = compute_c5(h, n_points)
    target_sum = n_points / 2
    penalty = 1e4 * abs(np.sum(h) - target_sum)
    return c5 + penalty

def run(seed=42, budget_s=600, **kwargs):
    # Choose n_points
    n_points = 100  # or other value
    target_sum = n_points / 2
    # Generate initial guess
    np.random.seed(seed)
    h_initial = np.random.uniform(0, 1, n_points)
    h_initial = h_initial / np.sum(h_initial) * target_sum
    h_initial = np.clip(h_initial, 0, 1)
    # Bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    # Run differential evolution
    result = differential_evolution(
        fun=lambda x: objective(x, n_points),
        bounds=bounds,
        strategy='best1bin',
        popsize=15,
        maxiter=100,
        tol=1e-6,
