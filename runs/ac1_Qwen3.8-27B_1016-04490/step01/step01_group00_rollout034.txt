        if time.monotonic() > deadline - 1.5:
            break
        sc_choice = rng.choice(len(scales))
        dx = scales[sc_choice] * S * rng.standard_normal(n)
        x_try = np.clip(best + dx, 0.0, 1000.0)
        if x_try.sum() > 0.01:
            x_try *= S / x_try.sum()
            sc = compute_score(x_try)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best = x_try.copy()

    fc = compute_score(best)
    if not np.isfinite(fc):
        fc = compute_score(parent)
        best = parent
    if not np.isfinite(fc):
        return [1.0]
    return [float(v) for v in best]
