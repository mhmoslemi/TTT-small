```python
from scipy.sparse import coo_matrix
from scipy.optimize import linprog

def solve_lp(f, n, M):
    """Solve max sum(b) s.t. A@b <= M, b >= 0"""
    # Build sparse convolution matrix A (2n-1 x n)
    # A[k,j] = f[k-j] if 0 <= k-j < n
    
    # Vectorized construction
    k = np.arange(2*n-1)
    # For each k, valid j range: max(0, k-n+1) to min(n-1, k)
    j_min = np.maximum(0, k - n + 1)
    j_max = np.minimum(n-1, k)
    counts = j_max - j_min + 1
    
    # Build sparse matrix using coo
    # For row k, columns are j_min[k]..j_max[k], values are f[k-j]
    # = f[k-j_max[k]]..f[k-j_min[k]]
    
    # Total nnz
    total_nnz = counts.sum()
    
    rows = np.zeros(total_nnz, dtype=np.int32)
    cols = np.zeros(total_nnz, dtype=np.int32)
    vals = np.zeros(total_nnz, dtype=np.float64)
    
    offset = 0
    for k in range(2*n-1):
        c = counts[k]
        rows[offset:offset+c] = k
        cols[offset:offset+c] = np.arange(j_min[k], j_max[k]+1)
        vals[offset:offset+c] = f[k - j_max[k] : k - j_min[k] + 1]
        offset += c
    
    A = coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()
    
    # max sum(b) s.t. A@b <= M, b >= 0
    c = -np.ones(n)  # minimize -sum(b)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return None
```

```python
        b_ub = np.full(2 * n - 1, M)
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n,
                          method='highs',
                          options={'time_limit': 20, 'maxiter': 80000})
        except Exception:
            break
        if not res.success or res.x is None:
            break
        g = np.maximum(res.x, 0.0)
        gs = g.sum()
        if gs < 0.01:
            break
        g *= target_sum / gs
        g = np.clip(g, 0.0, 1000.0)
        x_new = (1.0 - mix_t) * x + mix_t * g
        x_new = np.clip(x_new, 0.0, 1000.0)
        s = x_new.sum()
        if s < 0.01:
            break
        x_new *= target_sum / s
        x_new = np.clip(x_new, 0.0, 1000.0)
        x = x_new
        iters1 += 1
        if iters1 % 2 == 0:
            sc = float(evaluate_sequence(x.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best = x.copy()
                no_improve = 0
            else:
                no_improve += 2
        mix_t = max(min_mix, mix_t * 0.96)
        if no_improve >= 12:
            x += 0.08 * rng.standard_normal(n)
            x = np.maximum(x, 0.0)
            x *= target_sum / x.sum()
            mix_t = 0.25
            no_improve = 0

    # Phase 2: Gradient polishing with Nesterov momentum (45% of budget)
    phase2_start = time.monotonic()
    remaining = max(30.0, deadline - phase2_start)
    T = 0.2
    min_T = 0.005
    lr = 0.15
    vel = np.zeros(n)
    beta = 0.92
    fft_size = next_pow2(2 * n - 1)
    corr_size = next_pow2(3 * n)
    x = best.copy()
    iters2 = 0
    no_imp2 = 0

    while time.monotonic() < deadline - 5.0:
        X = sfft.rfft(x, fft_size)
        conv = np.real(sfft.irfft(X * X, fft_size))[:2 * n - 1]
        conv = np.maximum(conv, 0.0)
        cmax = conv.max()
        p = np.exp((conv - cmax) / T)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros(2 * n - 1)
            p[(2 * n - 1) // 2] = 1.0
        else:
            p /= p_sum
        P = sfft.rfft(p, corr_size)
        X2 = sfft.rfft(x, corr_size)
        corr = np.real(sfft.irfft(P * np.conj(X2), corr_size))[:n]
        g = 2.0 * corr
        g -= g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm > 1e-12:
            g = g / gnorm * min(gnorm, 5.0 / max(n, 1))
            g = np.clip(g, -5.0 / n, 5.0 / n)
        vel = beta * vel - lr * g
        x_new = x + vel
        x_new = np.clip(x_new, 0.0, 1000.0)
        s = x_new.sum()
        if s < 0.01:
            x_new = np.maximum(x_new + 0.05, 0.0)
            s = x_new.sum()
        x_new *= target_sum / s
        x_new = np.clip(x_new, 0.0, 1000.0)
        x = x_new
        iters2 += 1
        T = max(min_T, T * 0.9997)
        if iters2 > 5000:
            lr = max(0.02, lr * 0.9998)
            beta = min(0.97, beta * 1.0001)
        if iters2 % 800 == 0:
            sc = float(evaluate_sequence(x.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best = x.copy()
                no_imp2 = 0
            elif iters2 > 20000 and no_imp2 >= 3:
                x = best + 0.03 * rng.standard_normal(n)
                x = np.maximum(x, 0.0)
                x *= target_sum / x.sum()
                vel = np.zeros(n)
                T = max(0.05, T * 3.0)
                lr = 0.1
                no_imp2 = 0
            else:
                no_imp2 += 1

    # Final validation
    if not np.isfinite(best_score):
        return [float(v) for v in parent]
    fc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(fc):
        return [float(v) for v in parent]
    return [float(v) for v in best]
```
