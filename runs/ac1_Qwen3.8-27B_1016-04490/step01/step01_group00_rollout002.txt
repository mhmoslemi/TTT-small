        X2 = sfft.rfft(x, corr_size)
        grad_raw = np.real(sfft.irfft(P * np.conj(X2), corr_size))[:n]
        g = 2.0 * grad_raw
        g -= g.mean()
        
        # Adam update
        g_norm = np.sqrt(np.mean(g**2))
        if g_norm < 1e-15:
            t_adam += 1
            continue
        g = g / g_norm  # normalize gradient
        
        t_adam += 1
        m_adam = beta1 * m_adam + (1 - beta1) * g
        v_adam = beta2 * v_adam + (1 - beta2) * (g * g)
        m_hat = m_adam / (1 - beta1**t_adam)
        v_hat = v_adam / (1 - beta2**t_adam)
        
        lr_t = lr0 * np.sqrt(1 - beta2**t_adam) / (1 - beta1**t_adam)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        x = np.maximum(x, 0.0)
        x = normalize(x, target_sum)
        x = np.minimum(x, 1000.0)
        
        # Anneal temperature
        T = max(T_min, T * 0.99985)
        
        # Small noise periodically
        if t_adam % 2000 == 0 and t_adam > 0:
            x += 0.003 * rng.standard_normal(n)
            x = np.maximum(x, 0.0)
            x = normalize(x, target_sum)
        
        # Time check inside loop
        if t_adam % 500 == 0 and time.monotonic() > deadline - 25.0:
            break
    
    # Final eval
    sc = compute_score(x)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best = x.copy()
    
    if not np.isfinite(best_score):
        return [float(vv) for vv in parent]
    return [float(vv) for vv in best]

def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

    if s < 1e-10:
