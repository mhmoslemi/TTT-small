            if np.isfinite(sc) and sc < bs - 1e-9:
                bs = sc
                best = x.copy()
                stall = 0
            else:
                stall += eval_int

            if stall > 4500:
                amp = 0.015 + 0.005 * (stall // 4500)
                x = best.copy() + amp * rng.standard_normal(n)
                x = norm_seq(x, n, ts)
                T = 0.1
                lr = 0.35
                stall = 0

        if it % 4000 == 0 and it > 0:
            x = norm_seq(x + 4e-4 * rng.standard_normal(n), n, ts)

    # Polishing phase with lower temperature
    if time.monotonic() < deadline - 5.0:
        x = best.copy()
        T = 0.005
        lr = 0.08
        it2 = 0
        stall2 = 0
        while time.monotonic() < deadline - 3.0 and it2 < 80000:
            X = sfft.rfft(x, fs)
            cv = np.real(sfft.irfft(X * X, fs))[:2 * n - 1]
            cv = np.maximum(cv, 0.0)
            cm = cv.max()
            if cm < 1e-15:
                break
            p = np.exp((cv - cm) / T)
            ps = p.sum()
            if ps < 1e-300:
                p = np.zeros(2 * n - 1)
                p[cv.argmax()] = 1.0
            else:
                p /= ps
            P = sfft.rfft(p, cs)
            X2 = sfft.rfft(x, cs)
            gr = np.real(sfft.irfft(P * np.conj(X2), cs))[:n]
            g = 2.0 * gr
            g -= g.mean()
            gn = np.sqrt((g * g).sum())
            if gn < 1e-14:
                break
            g /= gn
            xn = norm_seq(x - lr * g, n, ts)
            x = xn
            T = max(0.001, T * 0.9999)
            it2 += 1
            if it2 % 300 == 0:
                sc = float(evaluate_sequence(x.tolist()))
                if np.isfinite(sc) and sc < bs - 1e-9:
                    bs = sc
                    best = x.copy()
                    stall2 = 0
                else:
                    stall2 += 300
                if stall2 > 6000:
                    break

    fc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(fc):
        return [float(v) for v in parent]
    return [float(v) for v in best]
