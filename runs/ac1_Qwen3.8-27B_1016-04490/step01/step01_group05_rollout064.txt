def _fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a) + len(b) - 1
    L = 1
    while L < n:
        L <<= 1
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    return np.fft.irfft(fa * fb, L)[:n]
