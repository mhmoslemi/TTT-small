        a = np.maximum(a, 1e-30)
        a /= a.sum()

        it += 1
        if it % 3 == 0:
            tau = max(tau * 0.997, min_tau)
            lr = max(lr * 0.99995, 0.3)

        if it % eval_every == 0:
            sc = float(evaluate_sequence(a.tolist()))
            if sc < best_score - 1e-12:
                best_score = sc
                best = a.copy()
                stalled = 0
            else:
                stalled += 1
            if stalled >= 5:
                scale = 0.004 * rng.uniform(1.0, 3.0)
                a += scale * rng.standard_normal(n)
                a = np.maximum(a, 0.0)
                a /= a.sum()
                stalled = 0
                if rng.random() < 0.4:
                    sym = 0.5 * (a + a[::-1])
                    sym = np.maximum(sym, 0.0)
                    sym /= sym.sum()
                    if float(evaluate_sequence(sym.tolist())) < best_score:
                        a = sym.copy()
                        best = a.copy()
                        best_score = float(evaluate_sequence(a.tolist()))

    sc = float(evaluate_sequence(a.tolist()))
    if sc < best_score:
        best = a.copy()
        best_score = sc
    return best, best_score


def _local_refine(a, deadline, rng):
    n = a.size
    a = np.maximum(a, 0.0).astype(np.float64).copy()
    a /= a.sum()
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    while time.monotonic() < deadline - 2:
        b = _fft_conv(a, a)
        peak_k = int(b.argmax())
        lo = max(0, peak_k - n + 1)
        hi = min(n - 1, peak_k)
        contribs = np.zeros(n)
        contribs[lo:hi + 1] = a[lo:hi + 1] * a[peak_k - hi:peak_k - lo + 1][::-1]

        top_m = min(12, max(3, n // 200))
        top_idx = np.argsort(-contribs)[:top_m]

        improved = False
        for idx in top_idx:
            for factor in (0.8, 0.65, 1.2, 1.4):
                a2 = a.copy()
                a2[idx] *= factor
                s2 = a2.sum()
                if s2 < 1e-10:
                    continue
                a2 /= s2
                sc = float(evaluate_sequence(a2.tolist()))
                if sc < best_score - 1e-12:
                    best_score = sc
                    best = a2.copy()
                    a = a2
                    improved = True
                    break
            if improved:
                break

        if not improved:
            a += 0.003 * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            a /= a.sum()

    return best, best_score


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 12.0)

    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.ndim != 1 or parent.size < 2 or not np.isfinite(parent).all():
        parent = np.ones(2535)
    n = parent.size
    parent = np.maximum(parent, 0.0)
    ps = parent.sum()
    if ps < 1e-10:
        parent = np.ones(n) / n
    else:
        parent /= parent.sum()

    j = np.arange(n, dtype=np.float64)
    x = j / max(n - 1, 1)

    inits = [
        parent.copy(),
        1.0 + 0.6 * (2.0 * x - 1.0) ** 2,
        1.0 + 1.2 * (2.0 * x - 1.0) ** 3 * np.abs(2.0 * x - 1.0),
        np.exp(-40.0 * (x - 0.2) ** 2) + np.exp(-40.0 * (x - 0.8) ** 2) + 0.003,
        (1.0 + j) ** (-0.25) + (1.0 + (n - 1 - j)) ** (-0.25) + 0.01,
    ]

    for i in range(len(inits)):
        inits[i] = np.maximum(inits[i], 0.0).astype(np.float64)
        ss = inits[i].sum()
        if ss < 1e-10:
            inits[i] = np.ones(n) / n
        else:
            inits[i] /= inits[i].sum()

    best_all = parent.copy()
    best_sc = float(evaluate_sequence(parent.tolist()))

    remaining = max(deadline - time.monotonic(), 5.0)
    reserve = min(180.0, remaining * 0.18)
    main_budget = deadline - reserve

    fracs = [0.42, 0.14, 0.14, 0.15, 0.15]
    for i, init in enumerate(inits):
        if time.monotonic() >= main_budget:
            break
        alloc = fracs[min(i, len(fracs) - 1)] * remaining
        dl = min(time.monotonic() + alloc, main_budget - 1.0)
        b, sc = _optimize(init, dl, rng)
        if sc < best_sc:
            best_sc = sc
            best_all = b.copy()

    # Symmetrization attempt on best
    sym = 0.5 * (best_all + best_all[::-1])
    sym = np.maximum(sym, 0.0)
    sym /= sym.sum()
    sym_sc = float(evaluate_sequence(sym.tolist()))
    if sym_sc < best_sc:
        best_sc = sym_sc
        best_all = sym

    # Local refinement
    if time.monotonic() < deadline - 10:
        refine_dl = time.monotonic() + min(reserve * 0.7, deadline - time.monotonic() - 10)
        b, sc = _local_refine(best_all, refine_dl, rng)
        if sc < best_sc:
            best_sc = sc
            best_all = b.copy()

        # Second round of refinement
        if time.monotonic() < deadline - 5:
            b, sc = _local_refine(best_all, deadline - 3, rng)
            if sc < best_sc:
                best_sc = sc
                best_all = b.copy()

    final = np.clip(best_all, 0.0, 1000.0)
    fs = float(evaluate_sequence(final.tolist()))
    if not np.isfinite(fs) or fs >= 2.0:
        final = np.clip(parent, 0.0, 1000.0)
        if float(evaluate_sequence(final.tolist())) >= 2.0:
            final = np.ones(n, dtype=np.float64) / n

    return [float(v) for v in final]
