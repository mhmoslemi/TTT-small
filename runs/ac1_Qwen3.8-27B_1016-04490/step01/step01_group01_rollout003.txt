def _fft_conv(a, N):
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)

def _subgrad(a, n, S, M, pk, pw):
    wa = np.zeros(n)
    for w, k in zip(pw, pk):
        k = int(k)
        lo = max(0, k - n + 1)
