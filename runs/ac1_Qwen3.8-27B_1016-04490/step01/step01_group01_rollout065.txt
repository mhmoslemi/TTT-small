def _fft_conv(a, N):

def _top_peaks_grad(a, b, n, K):
    """Compute weighted subgradient from top K peaks of b."""
    m = 2 * n - 1
    K = min(K, m)
    idx = np.argpartition(b, -K)[-K:] if K < m else np.arange(m)
    pv = b[idx]
    ws = pv ** 2
    wsum = ws.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    ws = ws / wsum
    grad = np.zeros(n)
    for wi, ki in zip(ws, idx):
        k = int(ki)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            j = np.arange(lo, hi + 1)
            grad[lo:hi + 1] += wi * a[k - j]
    return 2.0 * grad
