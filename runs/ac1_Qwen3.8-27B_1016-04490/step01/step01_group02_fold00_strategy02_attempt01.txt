    ca = _convolve_fft(c[:len(a)], a)
    return ca[len(a)-1:len(a)+1] * 2.0  # Gradient slice

def _convolve_fft(x, y):
    # Sparse FFT convolution for performance
    n1, n2 = len(x), len(y)
    m = n1 + n2 - 1
    sz = 1
    while sz < m:
        sz *= 2
    x_pad = np.pad(x if isinstance(x, np.ndarray) else np.array(x), (0, sz - n1), 'constant')
    y_pad = np.pad(y if isinstance(y, np.ndarray) else np.array(y), (0, sz - n2), 'constant')
    x_fft = np.fft.rfft(x_pad)
    y_fft = np.fft.rfft(y_pad)
    conv_fft = np.fft.irfft(x_fft * y_fft)[:m]
    return conv_fft.astype(np.float64)

def _scale_to_sum(seq, target_sum):
    current_sum = sum(seq)
    factor = target_sum / max(current_sum, 1e-6)
    return [min(1000.0, s * factor) for s in seq]
