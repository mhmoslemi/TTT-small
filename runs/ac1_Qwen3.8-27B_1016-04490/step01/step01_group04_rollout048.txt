import numpy as np
import time


def _p2(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def _score_conv(a, fl):
    n = a.size
    f = np.fft.rfft(a, n=fl)
    c = np.fft.irfft(f * f, n=fl)
    c = c[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    M = float(c.max())
    s = float(a.sum())
    sc = 2.0 * n * M / (s * s) if s > 1e-300 else np.inf
    return sc, c, M


def _grad_step(w, a, fl):
    n = a.size
    ra = a[::-1]
    wf = np.fft.rfft(w, n=fl)
    rf = np.fft.rfft(ra, n=fl)
    cc = np.fft.irfft(wf * rf, n=fl)
    return 2.0 * cc[n - 1:2 * n - 1]
