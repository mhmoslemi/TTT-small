import numpy as np
import time


def _fft_conv(a, N):
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)


def _score_fast(a, n, N):
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = _fft_conv(a, N)
    return 2.0 * n * b.max() / (S * S)


def _softmax_top(b, beta, n_active):
    n_active = min(n_active, len(b))
    idx = np.argpartition(b, -n_active)[-n_active:]
    bl = b[idx]
    x = beta * (bl - bl.max())
    w = np.exp(x)
    tot = w.sum()
    if tot < 1e-30:
        w = np.ones_like(w) / len(w)
    else:
        w = w / tot
    return idx, w


def _peak_grad(a, n, idx, w, dm):
    dm[:] = 0.0
    n1 = n - 1
    for wi, jj in zip(w, idx):
        if wi < 1e-10:
            continue
        j = int(jj)
        lo = j - n1
        if lo < 0:
            lo = 0
        hi = n1
        if hi > j:
            hi = j
        # left part k in [lo, j]: index = j-k
        if 0 <= lo <= j:
            seg = a[0:(j - lo + 1)][::-1]
            dm[lo:(j + 1)] += 2.0 * wi * seg
        # right part k in [j+1, hi]: index = k-j
        if j + 1 <= hi:
            seg = a[1:(hi - j + 1)]
            dm[(j + 1):(hi + 1)] += 2.0 * wi * seg
        # diagonal correction when 2k == j
        if (j & 1) == 0:
            k = j >> 1
            if lo <= k <= hi:
                dm[k] -= wi * a[k]
    return dm


def _gen_init(n, rng, kind):
    i = np.arange(n)
    if kind == 1:
        f1 = rng.integers(1, 8)
        f2 = rng.integers(1, 8)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * i / n) + 0.3 * np.cos(2 * np.pi * f2 * i / n)
    elif kind == 2:
        c = n // 2
        a = np.exp(-0.5 * ((i - c) / (n * 0.2)) ** 2)
    elif kind == 3:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * rng.integers(1, 6) * i / n)
    else:
        a = 1.0 + 0.3 * rng.standard_normal(n)
    a = np.clip(a, 0.05, 1000.0)
    return a * (1.0 / a.max())


def _normalize(a, target):
    s = a.sum()
    if s <= 1e-30:
        return a * 0
    return a * (target / s)


def _adam_run(a, n, N, deadline, seed):
    rng = np.random.default_rng(seed)
    target = np.sqrt(2.0 * n)
    a = _normalize(a.copy(), target)
    best = a.copy()
    best_score = _score_fast(a, n, N)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    beta = 8.0
    n_active = 64
    lr = 0.004
    dm = np.zeros(n)
    patience = 0
    max_pat = 250

    while time.monotonic() < deadline:
        b = _fft_conv(a, N)
        M = b.max()
        S = a.sum()
        sc = 2.0 * n * M / (S * S)
        if sc < best_score:
            best = a.copy()
            best_score = sc
            patience = 0
        else:
            patience += 1

        if patience >= max_pat:
            if best_score < 1.51:
                break
            a = best * (1.0 + 0.03 * rng.standard_normal(n))
            a = _normalize(np.clip(a, 0.0, 1000.0), target)
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            patience = 0
            max_pat = max(80, max_pat // 2)
            if max_pat < 50:
                break
            beta = min(beta * 1.2, 40.0)

        idx, w = _softmax_top(b, beta, n_active)
        _peak_grad(a, n, idx, w, dm)

        pref = 2.0 * n / (S * S * S)
        grad = pref * (S * dm - 2.0 * M)

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        mh = m / (1.0 - beta1 ** t)
        vh = v / (1.0 - beta2 ** t)
        step = lr * mh / (np.sqrt(vh) + eps)
