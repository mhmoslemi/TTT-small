from scipy import fft as sfft


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def subgrad_step(x, fft_size, corr_size, T, lr, target_sum):
    n = len(x)
    X = sfft.rfft(x, fft_size)
    conv = np.real(sfft.irfft(X * X, fft_size))[:2 * n - 1]
    conv = np.maximum(conv, 0.0)
    cmax = conv.max()
    if T > 1e-10:
        p = np.exp((conv - cmax) / T)
        ps = p.sum()
        if ps > 1e-300:
            p /= ps
        else:
            p = np.zeros(2 * n - 1)
            p[n - 1] = 1.0
    else:
        p = np.zeros(2 * n - 1)
        idx = np.where(conv >= cmax - 1e-8)[0]
        p[idx] = 1.0 / len(idx)
    P = sfft.rfft(p, corr_size)
    X2 = sfft.rfft(x, corr_size)
    corr = np.real(sfft.irfft(P * np.conj(X2), corr_size))[:n]
    g = 2.0 * corr
    g -= g.mean()
    x = np.maximum(x - lr * g, 0.0)

        x = np.ones(n) * 0.01
        s = x.sum()

    return x, conv.max()

        n_iters = 4000 if att < 10 else 2000

        x = refine(x, fft_size, corr_size, target_sum, T0, lr0, n_iters,
                   hard_deadline - 20)

        sc = float(evaluate_sequence(x.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = x.copy()

    # Final: one more polish from the best
    if time.monotonic() < hard_deadline - 10:
        x = best.copy()
        x = refine(x, fft_size, corr_size, target_sum, 0.005, 0.1, 5000,
                   hard_deadline - 5)
        sc = float(evaluate_sequence(x.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = x.copy()

    result = [float(v) for v in best]
    final_sc = float(evaluate_sequence(result))
    if not np.isfinite(final_sc):
        result = [float(v) for v in parent]
    return result
