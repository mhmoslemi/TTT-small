import numpy as np
import time
import scipy.optimize as sopt

def _fft_autocorr(a):
    n = a.size
    if n == 0:
        return np.zeros(0)
    L = 2 * n - 1
    N = 1 << max(0, (L - 1).bit_length())
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    return np.real(np.clip(c, 0.0, None))

def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.full(z.shape, 1.0 / z.size)
    return e / s

def _obj(z, beta, n, N):
    p = _softmax(z)
    c = _fft_autocorr(p)
    u = n * c
    um = u.max()
    se = np.exp(beta * (u - um))
    val = (beta * um + np.log(se.sum())) / beta
    return val

def _grad(z, beta, n, N):
    p = _softmax(z)
    c = _fft_autocorr(p)
    u = n * c
    um = u.max()
    se = np.exp(beta * (u - um))
    w = se / se.sum()
    L = 2 * n - 1
    N2 = 1 << max(0, (L + n - 2).bit_length())
    fp = np.fft.rfft(p, N2)
    fw = np.fft.rfft(w, N2)
    conv_part = np.fft.irfft(fw * fp, N2)[:n]
    ww = np.zeros(N2)
    ww[:L] = w
    pp = np.zeros(N2)
    pp[:n] = p
    corr_part = np.fft.irfft(np.fft.rfft(ww) * np.conj(np.fft.rfft(pp)), N2)[:n]
    gp = n * (corr_part + conv_part)
    dot = np.dot(p, gp)
    gz = p * (gp - dot)
    if not np.isfinite(gz).all():
        return np.zeros(n)
    return gz

def _peak_shave(p, deadline, rng):

    p = p.copy()

    if not np.isfinite(ps):

    N = 1 << max(0, (2 * n - 2).bit_length())
    step = 0.008
    fails = 0
    while time.monotonic() < deadline - 20 and fails < 200:
        c = _fft_autocorr(p)
        kp = int(np.argmax(c))
        # gradient of c[kp] w.r.t. p[m] = p[kp-m] if 0<=kp-m<n
        g = np.zeros(n)
        lo = max(0, kp - n + 1)
        hi = min(n - 1, kp)
        idx = np.arange(lo, hi + 1)
        g[idx] = c[0] * 0 + p[kp - idx]
        # Also the symmetric: conv(p,p)[kp] = sum_i p[i]*p[kp-i]
        # d/dp[m] = p[kp-m] (if valid)
        # Direction: reduce p where g is large
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        # Move p in direction -g, projected to keep sum=1 and p>=0
        dp = -step * g
        dp = dp - dp.mean()  # keep sum unchanged

            step = min(0.03, step * 1.03)

            fails += 1
            step *= 0.92

                break
    return p, ps
