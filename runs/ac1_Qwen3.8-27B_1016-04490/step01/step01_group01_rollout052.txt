import numpy as np

def _fft_conv(a, N):
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)


def _score_fft(a, n, N):
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = _fft_conv(a, N)
    return 2.0 * n * b.max() / (S * S)


def _make_grad(a, b, n, n_peaks):
    S = a.sum()
    if S < 1e-30:
        return np.zeros(n)
    mean_a = S / n
    idx = np.argpartition(b, -n_peaks)[-n_peaks:]
    vals = b[idx]
    w_sum = vals.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    w = vals / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, idx):
        ki = int(k)
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if lo <= hi:
            seg = a[ki - hi: ki - lo + 1][::-1]
            grad[lo:hi + 1] += wi * (seg - mean_a)
    return grad
