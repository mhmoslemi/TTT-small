import numpy as np
import time


def _fft_len(n):
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    return L


def _conv(a, n, L):
    fa = np.fft.rfft(a, L)
    return np.fft.irfft(fa * fa, L)[:2 * n - 1]


def _grad(a, w, n, L):
    fa = np.fft.rfft(a, L)
    fw = np.fft.rfft(w, L)
    t1 = np.fft.irfft(fw * fa.conj(), L)[:n]
    t2 = np.fft.irfft(fa * fw, L)[:n]
    return t1 + t2
