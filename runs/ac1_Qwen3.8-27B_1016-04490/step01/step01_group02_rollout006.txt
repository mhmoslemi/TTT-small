def _fft_conv(x, y):
    """FFT-based convolution, O(n log n)."""
    m = len(x) + len(y) - 1
    sz = 1
    while sz < m:
        sz <<= 1
    fx = np.fft.rfft(x, sz)
    fy = np.fft.rfft(y, sz)
    return np.fft.irfft(fx * fy, sz)[:m]

def _softmax_topk(x, tau, k=None):
