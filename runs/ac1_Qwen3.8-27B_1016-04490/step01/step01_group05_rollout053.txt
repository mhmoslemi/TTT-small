                    a /= ss
                b = np.convolve(a, a)
                cur_max = b.max()
                y = a.copy()
                r = 1.0
                last_improve = step

        step += 1
        tau_scale *= (1.0 - 0.003)
        if tau_scale < 0.02:
            tau_scale = 0.02

        if step % 40 == 0:
            s = eval_a(a)
            if s < best_score:
                best_score = s
                best_a = a.copy()
            if time.monotonic() > deadline:
                break

    s_final = eval_a(best_a)
    if np.isfinite(s_final) and s_final < best_score:
        best_score = s_final

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if not result or sum(result) < 0.01:
        result = [float(max(0.0, min(1000.0, x))) for x in parent]
    return result
