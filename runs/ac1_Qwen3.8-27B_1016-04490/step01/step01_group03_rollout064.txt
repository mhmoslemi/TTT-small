    sd = 0.1 * n
    L.append((np.exp(-((j - t1) / sd) ** 2) + np.exp(-((j - t2) / sd) ** 2) + 0.005).copy())
    d = np.minimum(j, n - 1 - j) + 1.0
    for pv in (0.5, 1.0, 2.0, 3.0):
        L.append(d ** pv)
    for pv in (0.5, 1.0, 2.0):
        L.append((np.abs(j - m) / (m + 1.0) ** pv + 0.05).copy())
    return L

def lse_run(z0):
    global GB
    if sopt is None:
        return z0
    z = z0 - z0.mean()
    z = np.clip(z, -25, 25)
    betas = (8, 16, 32, 64, 128, 256, 512, 1024, 2048, 4096)
    mits = (40, 50, 70, 90, 100, 130, 160, 200, 250, 300)
    vb = lse_val(z)
    for b, mi in zip(betas, mits):
        if time.monotonic() > DEADLINE - 20:
            break
        GB = float(b)
        try:
            r = sopt.minimize(lse_val, z, jac=lse_grd, method='L-BFGS-B',
                              callback=_tcb,
                              options={'maxiter': mi, 'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 50})
        except Exception:
            r = None
        if r is not None and hasattr(r, 'x') and r.x is not None and np.isfinite(r.fun):
            if r.fun <= vb + 1e-8:
                z = np.asarray(r.x, dtype=np.float64)
                vb = float(r.fun)
        p = np.exp(z - z.max())
        p /= p.sum()
        p = normclip(p)
        if p is not None:
            upb(p)
    return z

def peak_shave(p, iters):
    global DEADLINE, GNF, GN
    p = p.copy()
    fc = sc_f(p)
    for _ in range(iters):
        if time.monotonic() > DEADLINE - 15:
            break
        c = conve(p, GNF)
        pk = float(c.max())
        if pk < 1e-15:
            break
        idx = np.argsort(c)[::-1]
        top_k = idx[:30]
        thr = pk * 0.95
        active = []
        for k in top_k:
            if c[k] >= thr:
                active.append(int(k))
            else:
                break
        if not active:
            break
        na = len(active)
        wts = np.array([c[k] for k in active])
        wts /= wts.sum()
        g = np.zeros(GN, dtype=np.float64)
        for ii in range(na):
            k = active[ii]
            wi = wts[ii]
            lo = max(0, k - (GN - 1))
            hi = min(GN - 1, k)
            g[lo:hi + 1] += wi * p[k - hi:k - lo + 1][::-1]
        g *= 2.0 * float(GN)
        d = float(p @ g)
        g0 = g - d
        found = False
        for t in (0.08, 0.04, 0.02, 0.01, 0.005, 0.002):
            p2 = p - t * g0
            if p2.min() < 0:
                p2 = np.maximum(p2, 0)
                s2 = p2.sum()
                if s2 < 1e-15:
                    break
                p2 /= s2
            else:
                p2 /= p2.sum()
            sc2 = sc_f(p2)
            if np.isfinite(sc2) and sc2 < fc - 1e-13:
                p = p2
                fc = sc2
                found = True
                break
        if not found:
            break
    upb(p)
    return p

def sym_try(p):
    global GNF, GN
    psym = (p + p[::-1]) / 2.0
    psym = normclip(psym)
    if psym is not None:
        upb(psym)
    return psym

def propose_candidate(seed=42, budget_s=1100, **kwargs):

    return [float(x) for x in BEST_P]
