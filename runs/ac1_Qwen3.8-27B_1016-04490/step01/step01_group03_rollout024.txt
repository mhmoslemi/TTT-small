import numpy as np
import time
import scipy.optimize as sopt

def fast_conv(a):
    """Full linear convolution via FFT."""
    n = len(a)
    N = 1 << (2*n-2).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa*fa, N)[:2*n-1]
    return np.maximum(c, 0)  # nonnegative for nonnegative input

def score(a):
    s = np.sum(a)
    if s < 0.01:
        return inf
    c = fast_conv(a)
    return 2*len(a)*np.max(c)/s**2

def lse_objective(z, beta):
    p = softmax(z)
    c = fast_conv(p)
    u = len(p) * c
    umax = np.max(u)
    ex = np.exp(beta*(u - umax))
    s = np.sum(ex)
    return (beta*umax + np.log(s))/beta

def lse_gradient(z, beta):
    p = softmax(z)
    n = len(p)
    c = fast_conv(p)
    u = n * c
    umax = np.max(u)
    ex = np.exp(beta*(u - umax))
    w = ex / np.sum(ex)
    # Gradient via FFT cross-correlation/convolution
    N = 1 << (2*n-2).bit_length()
    fp = np.fft.rfft(p, N)
    fw = np.fft.rfft(w, N)
    # conv(p, w) gives d_conv[k]/d_p scaled
    cp = np.fft.irfft(fp * fw, N)  # conv(p, w)
    cw = np.fft.irfft(fp * np.conj(fw), N)  # corr(p, w)
    # gradient = (n/beta) * (cp[0:n] + reversed(cw[n-1:2n-1]))
    # ... (similar to previous program)
    g = (n/beta) * (cp[:n] + np.flip(cw[1:2*n-1])[:n])  # approximate
    dot = np.dot(p, g)
    grad = p * (g - dot)

import time
import scipy.optimize as sopt

def _norm(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1 or a.size == 0:
        return None
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01 or not np.isfinite(s):
        return None
    return a / s

def _fxcorr(a):
    n = len(a)
    N = 1 << (2*n - 2).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1]
    return np.maximum(c, 0.0)

def _score(a):
    p = _norm(a)
    if p is None:
        return float('inf')
    c = _fxcorr(p)
    return float(2.0 * len(p) * c.max())

def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    return e / s if s > 0 else np.full_like(z, 1.0/len(z))

def _lse_obj(z, beta, n, N):
    p = _softmax(z)
    c = _fxcorr(p)
    u = n * c
    um = u.max()
    ex = np.exp(beta * (u - um))
    s = ex.sum()
    if s <= 0 or not np.isfinite(s):
        return 1e100
    return float((beta * um + np.log(s)) / beta)

def _lse_grad(z, beta, n, N):
    p = _softmax(z)
    c = _fxcorr(p)
    u = n * c
    um = u.max()
    ex = np.exp(beta * (u - um))
    w = ex / ex.sum()

    fw = np.fft.rfft(w, N)
    conv_pw = np.fft.irfft(fp * fw, N)
    corr_pw = np.fft.irfft(fp * np.conj(fw), N)
    dmin = conv_pw[:n]
    dplus = np.empty(n)
    dplus[0] = corr_pw[0]
    if n > 1:
