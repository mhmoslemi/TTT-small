def _fft_conv(a, b, n):
    L = 1
    while L < 2*n - 1:
        L <<= 1
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    return np.fft.irfft(fa*fb, L)[:2*n-1]

def _grad_lag(a, k, n):
