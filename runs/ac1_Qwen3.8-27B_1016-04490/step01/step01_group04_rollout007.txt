```python
def compute_gradient(a, conv, n, target, top_k=10):
    g = np.zeros(n)
    # Get top k indices
    if n < top_k:
        peaks = np.argsort(conv)[::-1]
    else:
        top_idx = np.argpartition(conv, -top_k)[-top_k:]
        peaks = top_idx[np.argsort(conv[top_idx])[::-1]]
    
    for k in peaks:
        weight = conv[k] - target
        if weight <= 0:
            continue
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        j_lo = k - hi
        j_hi = k - lo
        # a_ij = a[lo:hi+1], a_j = a[j_lo:j_hi+1][::-1]
        a_i = a[lo:hi+1]
        a_j = a[j_lo:j_hi+1][::-1]
        # gradient for each i in [lo,hi]: 2*a_j (or a_j if i==j)
        contrib = 2.0 * a_j
        # Fix diagonal: where i == j, i.e., lo + t = k - (hi - t) => 2*(lo+t) = k => t = (k-2*lo)/2
        diag_t = (k - 2*lo) / 2.0
        if diag_t == int(diag_t) and 0 <= int(diag_t) <= (hi - lo):
            contrib[int(diag_t)] *= 0.5
        g[lo:hi+1] += weight * contrib
    return g
```

For a periodic extension: if a has a Fourier component at frequency f, the circular convolution has a component at frequency 0 (DC) with power |A_f|^2. To minimize the peak, we want the power to be spread across frequencies.
