import numpy as np
import time
from scipy.optimize import minimize

_BETA = 1.0

def _fft_conv(a, b):
    n = len(a) + len(b) - 1
    N = 1 << (n - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N), N)[:n]

def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.full(z.shape, 1.0 / len(z))
    return e / s

def _cmax(p):
    c = _fft_conv(p, p)
    return np.maximum(c, 0.0).max()

def _fun(z):
    p = _softmax(z)
    c = np.maximum(_fft_conv(p, p), 0.0)
    bc = _BETA * c
    bc -= bc.max()
    e = np.exp(bc)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return 1e100
    return float(_BETA * c.max() + np.log(s))

def _grad(z):
    n = len(z)
    p = _softmax(z)
    c = np.maximum(_fft_conv(p, p), 0.0)
    bc = _BETA * c
    bc -= bc.max()
    e = np.exp(bc)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.zeros(n)
    w = e / s
    r = _fft_conv(p, w[::-1])
    ga = r[n - 1:2 * n - 1][::-1]
    if not np.isfinite(ga).all():
        return np.zeros(n)
    dot = float(p @ ga)
    g = p * (ga - dot)
    if not np.isfinite(g).all():
        return np.zeros(n)
    return g

def _subgrad_step(p, eta):
    n = len(p)
    c = np.maximum(_fft_conv(p, p), 0.0)

    p2 = np.maximum(p2, 0.0)
