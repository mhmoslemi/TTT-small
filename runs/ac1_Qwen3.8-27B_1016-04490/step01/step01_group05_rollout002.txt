def _fft_convolve(a, b):
    """Circular-free convolution via FFT."""
    n = len(a)
    m = len(b)
    L = 1
    while L < n + m - 1:
        L <<= 1
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    return np.fft.irfft(fa * fb, L)[:n + m - 1]
