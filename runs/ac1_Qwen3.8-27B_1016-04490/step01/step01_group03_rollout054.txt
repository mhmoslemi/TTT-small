import numpy as np
import time

def fft_conv(a):
    n = len(a)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    return c

def score_of(a):
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fft_conv(a)
    return 2.0 * len(a) * c.max() / (s * s)

def grad_at_lag(a, k, n):
    # k is in [0, n-1]
    # grad[m] = 2*a[k-m] for m=0..k, 0 for m>k
    g = np.zeros(n)
    if k >= 0 and k < n:
        g[:k+1] = 2.0 * a[k::-1]
    return g
```

def subgradient_descent(a, time_budget, deadline):
    n = len(a)
    a = a.astype(np.float64).copy()
    # Normalize to sum 1
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best = a.copy()
    best_score = score_of(a)
    
    momentum = np.zeros(n)
    alpha = 0.01
    beta = 0.9  # momentum coefficient
    
    stall = 0
    eval_count = 0
    
    while time.monotonic() < deadline:
        c = fft_conv(a)
        k_max = c.argmax()
        # Use symmetry: map to [0, n-1]
        if k_max > n - 1:
            k_max = 2 * n - 2 - k_max
        
        # Find lags near max (within 0.5%)
        threshold = c[k_max] * 0.995
        near_max = np.where(c >= threshold)[0]
        # Map to [0, n-1]
        near_max = [k if k < n else 2*n-2-k for k in near_max]
        
        # Average gradient
        g = np.zeros(n)
        for k in near_max:
            g += grad_at_lag(a, k, n)
        g /= len(near_max)
        
        # Momentum update
        momentum = beta * momentum - alpha * g
        a_new = a + momentum
        
        # Project: clip to 0, normalize
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a * 0.5 + np.ones(n) * 0.5 / n
            s = a_new.sum()
        a_new = a_new / s
        
        # Evaluate
        new_score = score_of(a_new)
        if new_score < best_score:
            best = a_new.copy()
            best_score = new_score
            stall = 0
            # Slightly increase step
            alpha = min(alpha * 1.02, 0.1)
        else:
            stall += 1
            alpha *= 0.95
            a = a  # Keep old (or partially revert)
            # Revert momentum
            momentum *= 0.5
        
        if stall > 100:
            alpha *= 0.5
            stall = 0
            if alpha < 1e-8:
                break
        
        eval_count += 1
        if eval_count % 100 == 0:
            # Check time more frequently
            if time.monotonic() > deadline - 5:
                break
    
    return best
```

def fft_conv(a):

def compute_L_and_grad(a, beta, N):

    c = fft_conv(a)
    cmax = c.max()
    ex = np.exp(beta * (c - cmax))
    s = ex.sum()
    L = cmax + np.log(s) / beta

    if n > 1:
