        parent = np.ones(1024, dtype=np.float64)

    if parent.size == 0 or not np.all(np.isfinite(parent)) or np.sum(parent) < 1e-12:
        parent = np.ones(1024, dtype=np.float64)
    n = parent.size
    L = _nextpow2(3 * n + 4)

    best_a = parent.copy()
    best_score = _quick_score(best_a, n, L)
    es = evaluate_sequence(best_a.tolist())
    if np.isfinite(es):
        best_score = es

    for ci in _seed_inits(n, rng):
        ci = np.clip(ci, 0.0, 1000.0).astype(np.float64)
        if np.sum(ci) < 1e-12:
            continue
        si = _quick_score(ci, n, L)
        if si < best_score * 0.9995:
            ta = _try_normalize(ci)
            if ta is not None:
                best_score = si
                best_a = ta

    a = best_a.copy()
    momentum = np.zeros(n, dtype=np.float64)
    mu = 0.85
    step = 0
    last_improve = 0
    tau_frac0 = 0.045

    while time.monotonic() < deadline:
        c = _conv_square(a, L)
        cur_max = float(np.max(c))
        ssum = float(np.sum(a))
        if cur_max <= 0.0 or ssum < 1e-14:
            break

        tau_frac = min(0.06, max(0.008, tau_frac0 * (0.9997 ** step)))
        tau = tau_frac * cur_max
        if tau < 1e-15:
            tau = cur_max * 0.01
        sh = np.minimum(c / tau - cur_max / tau, 45.0)
        w = np.exp(sh)
        ws = float(np.sum(w))
        if ws < 1e-30:
            break
        w = w / ws

        g = _fft_correlate(w, a, L)[:n]
        aa = float(np.dot(a, a))
        if aa > 0.0:
            g = g - (float(np.dot(g, a)) / aa) * a
        gn = float(np.linalg.norm(g))
        if gn < 1e-18:
            break
        g = g / gn

        momentum = mu * momentum + g
        dn = float(np.linalg.norm(momentum))
        direction = momentum / dn if dn > 1e-18 else g

        taken = False
        for lr in (0.4, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005):
            a_t = _try_normalize(a - lr * direction)
            if a_t is None:
                continue
            if float(np.max(_conv_square(a_t, L))) < cur_max:
                a = a_t
                taken = True
                break

        if not taken:
            a_t = _try_normalize(0.5 * a + 0.5 * best_a)
            if a_t is None:
                break
            a = a_t

        step += 1

        if step % 30 == 0:
            sc = _quick_score(a, n, L)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = step

        if step - last_improve > 4000:
            a_t = _try_normalize(0.65 * a + 0.35 * best_a + 0.2 * rng.normal(0.0, 0.15, n))
            if a_t is None:
                a = best_a.copy()
            else:
                a = a_t
            momentum = np.zeros(n, dtype=np.float64)
            last_improve = step

    for cand in (a, best_a):
        cc = np.clip(cand, 0.0, 1000.0)
        if cc.size:
            sc = _quick_score(cc, n, L)
            if sc < best_score:
                best_score = sc
                best_a = cc.copy()

    es = evaluate_sequence(best_a.tolist())
    if np.isfinite(es):
        best_score = es
    best_a = np.clip(best_a, 0.0, 1000.0)

    if best_a.size == 0 or not np.all(np.isfinite(best_a)):
        best_a = np.clip(parent, 0.0, 1000.0)

    out = [float(x) for x in best_a]
    if not out or not all(np.isfinite(x) for x in out):
        out = [float(x) for x in np.clip(parent, 0.0, 1000.0)]
    if not out:
        out = [1.0]
    return out
