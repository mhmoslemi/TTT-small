import numpy as np

def _fft_conv(a, N):
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)


def _xcorr(w, a, N):
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    return np.fft.irfft(W * np.conj(A), N)


def _normalize(a, target):
    s = a.sum()
    if s > 1e-30:
