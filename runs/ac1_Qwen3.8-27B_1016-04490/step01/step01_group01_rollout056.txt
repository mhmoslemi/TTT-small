def _fft_conv_full(a, N):

def _score_of(a, n, N):

def _compute_grad(a, b, n, n_peaks, temp):
    top_k = np.argpartition(b, -n_peaks)[-n_peaks:]
    pk_vals = b[top_k]

    logits -= logits.max()
    w = np.exp(logits)
