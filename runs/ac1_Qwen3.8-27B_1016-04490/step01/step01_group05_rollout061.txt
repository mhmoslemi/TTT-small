Actually, one more thought about the mirror descent update. The update is:
a_new[m] = a[m] * exp(-lr * g[m] / max(g))

import numpy as np

def _fft_conv(a, b):
    n = len(a) + len(b) - 1
    L = 1 << (n - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)[:n]


def _cross_corr(w, a):
    n = len(a)
    L = 1 << (3 * n - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(w, L) * np.conj(np.fft.rfft(a, L)), L)[:n]

    s = float(np.sum(a))
    if s < 1e-10:
        return 1e18
    return 2.0 * len(a) * float(np.max(_fft_conv(a, a))) / (s * s)
