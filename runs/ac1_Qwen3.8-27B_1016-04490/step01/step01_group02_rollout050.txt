import time

def _fft_conv(x, y):
    m = len(x) + len(y) - 1
    sz = 1
    while sz < m:
        sz <<= 1
    fx = np.fft.rfft(x, sz)
    fy = np.fft.rfft(y, sz)
    return np.fft.irfft(fx * fy, sz)[:m]

def _softmax_w(s, tau):
    z = (s - s.max()) / tau
    np.clip(z, -60, 0, out=z)
    w = np.exp(z)
    ss = w.sum()
    if ss < 1e-300:
        return np.full_like(s, 1.0 / len(s))
    return w / ss

def _grad(a, w):
    n = a.size
    ca = _fft_conv(w[::-1], a)
    return 2.0 * ca[n - 1: 2 * n - 1][::-1]

def _eval_score(a):
    n = a.size
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _fft_conv(a, a)
    return 2.0 * n * c.max() / (s * s)

import numpy as np
import time


def _fft_conv(x, y):
    m = len(x) + len(y) - 1
    sz = 1
    while sz < m:
        sz <<= 1
    fx = np.fft.rfft(x, sz)
    fy = np.fft.rfft(y, sz)
    return np.fft.irfft(fx * fy, sz)[:m]


def _softmax_w(s, tau):
    z = (s - s.max()) / tau
    np.clip(z, -70.0, 0.0, out=z)
    w = np.exp(z)
    ss = w.sum()
    if ss < 1e-300:
        return np.full_like(s, 1.0 / len(s))
    return w / ss


def _grad(a, w):
    n = a.size
    ca = _fft_conv(w[::-1], a)
    return 2.0 * ca[n - 1:2 * n - 1][::-1]


def _eval_score(a):
    n = a.size
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _fft_conv(a, a)
    return float(2.0 * n * c.max() / (s * s))
