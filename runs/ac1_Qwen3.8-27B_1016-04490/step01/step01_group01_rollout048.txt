def _fft_conv(a, N):
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)

def _eval(a, n, N):
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = _fft_conv(a, N)
    return 2.0 * n * b.max() / (S * S)

def _grad(a, b, n, N, lam):
    M = b.max()
    if M <= 0:
        return np.zeros(n)
    bs = b - M  # shift for numerical stability
    exp_bs = np.exp(lam * bs)
