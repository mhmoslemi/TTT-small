    p = 1
    while p < m:
        p <<= 1
    return p

    x0 = parent.copy()

    if s0 < 0.01:

        s0 = n
    x0 = x0 / s0
    
    def eval_score(arr):
        return float(evaluate_sequence(arr.tolist()))
    
    best = x0.copy()
    best_score = eval_score(best)
    if not np.isfinite(best_score):
        best_score = float('inf')

    candidates = [x0.copy()]

        xm = np.maximum(xm, 0.01)
        xm = xm / xm.sum()
        candidates.append(xm)
    
    # Block structure
    for bp in [4, 8, 16]:
        xm = np.ones(n) * (1.0 + 0.5 * (np.arange(n) % bp < bp//2).astype(float))
        xm = xm / xm.sum()
        candidates.append(xm)
    
    # Select best initial
    for c in candidates:
        sc = eval_score(c)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = c.copy()

    momentum = np.zeros(n)
    beta = 0.85
    T = 0.3
    lr = 0.05
    T_min = 0.005
    T_decay = 0.9997
    iters_since_improve = 0

    restart_count = 0
    max_restarts = 30

            break

        conv = np.real(sfft.irfft(X * X, fs))[:2*n-1]

        p = np.exp((conv - cmax) / T)
        ps = p.sum()
        if ps < 1e-300:

            p[(2*n-1)//2] = 1.0
        else:

        momentum = beta * momentum + (1.0 - beta) * g

        if s < 1e-10:
            x = best.copy()

        if iteration > 100000:
            lr = max(0.01, lr * 0.99995)
        
        # Periodic evaluation
        if iteration % 5000 == 0:
            sc = eval_score(x)
            if np.isfinite(sc) and sc < best_score - 1e-8:
                best_score = sc
                best = x.copy()
                iters_since_improve = 0

                iters_since_improve += 5000

                T = 0.3

    # Final validation
    fc = eval_score(best)
    if not np.isfinite(fc):
        return [float(v) for v in parent]
    return [float(v) for v in best]
