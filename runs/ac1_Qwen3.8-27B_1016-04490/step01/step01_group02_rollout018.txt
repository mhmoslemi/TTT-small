import numpy as np
import time
from scipy import sparse
from scipy.optimize import linprog


def _fft_conv(x, y):
    m = len(x) + len(y) - 1
    sz = 1
    while sz < m:
        sz <<= 1
    fx = np.fft.rfft(x, sz)
    fy = np.fft.rfft(y, sz)
    return np.fft.irfft(fx * fy, sz)[:m]


def _build_conv_mat(a):
    n = len(a)
    diags, offsets = [], []
    for d in range(-(n - 1), n):
        diags.append(np.full(n - abs(d), float(a[abs(d)])))
        offsets.append(d)
    return sparse.diags(diags, offsets, shape=(2 * n - 1, n), format='csc')


def _lp_solve(a, M, tlimit):
    n = len(a)
    A = _build_conv_mat(a)
    b = np.full(2 * n - 1, M)
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': tlimit})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None

    tau = 1.5 / n
    lr = 1.0
    it = 0
    stalled = 0
    last_lp = -9999

    while time.monot

onic() < deadline - 5:
        M = conv.max()
        cz = conv / tau
        cz -= cz.max()
        p = np.exp(cz)
        p /= p.sum()
        pa = _fft_conv(p[::-1], a)
        g = 2.0 * pa[n - 1:2 * n - 1][::-1]
        step = np.minimum(lr * g, 50.0)
        a_new = a * np.exp(-step)
        a_new = np.maximum(a_new, 1e-300)
        a_new /= a_new.sum()
        conv_new = _fft_conv(a_new, a_new)
        sv = 2.0 * n * conv_new.max()
        if sv <= best_score + 1e-9:
            a, conv = a_new, conv_new
            if sv < best_score - 1e-12:

            lr = min(lr * 1.008, 30.0)
