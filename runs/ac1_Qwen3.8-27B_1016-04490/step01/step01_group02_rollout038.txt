            if stalled % 10 == 0:
                tau *= 0.95
                tau = max(tau, 1e-5/n)
            if stalled >= 30:
                a += 0.02 * rng.standard_normal(n)
                a = np.maximum(a, 0.0)
                a /= a.sum()
                stalled = 0
                lr = 0.5
                tau = max(tau * 5, 2.0/n)
        
        if time.monotonic() >= deadline:
            break
    
    sc = _score(a)
    if sc < best_sc:
        best_sc = sc
        best = a.copy()
    return best, best_sc
