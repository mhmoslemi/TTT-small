1. Compute gradient g

3. Update: a ← (1-t)a + t * e_{argmin g}

import numpy as np
import time

def _fft_grad(a, w, n, L):
    fa = np.fft.rfft(a, L)
    fw = np.fft.rfft(w, L)
    g = np.fft.irfft(fa * fw.conj(), L)
    return 2.0 * g[2*n-2::-1][:n]

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 20.0)
    
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    ok = parent.size > 0 and np.all(np.isfinite(parent))
    n = parent.size if ok else 256
    if not ok:
        parent = np.ones(n)
    
    i = np.arange(n, dtype=np.float64)
    
    # Initializations
    inits = []
    p = parent.copy()
    if p.sum() > 1e-10:
        inits.append(p / p.sum())
    for alpha in (0.5, 0.4, 0.6, 0.3, 0.7):
        c = (i + 1.0) ** (-alpha)
        inits.append(c / c.sum())
    c = np.exp(-3.0 * (2.0*i/(n-1) - 1.0)**2)
    inits.append(c / c.sum())
    
    best_a = inits[0].copy()
    best_score = np.inf
    for c in inits:
        b = np.convolve(c, c)
        sc = 2.0 * n * b.max()
        if sc < best_score:
            best_score = sc
            best_a = c.copy()
    
    a = best_a.copy()
    L = 1
    while L < 3 * n:
        L <<= 1
    
    step = 0
    last_improve = 0
    
    while time.monotonic() < deadline:
        # Early exit safety
        if time.monotonic() > deadline - 30:
            break
        
        b = np.convolve(a, a)
        cur_max = b.max()
        cur_sum = a.sum()
        if cur_sum < 1e-10 or cur_max < 1e-15:
            break
        
        # Phase-based tau
        if step < 500:
            tau_frac = 0.08
        elif step < 2000:
            tau_frac = 0.04
        elif step < 5000:
            tau_frac = 0.02
        else:
            tau_frac = max(0.008, 0.02 * 0.9995 ** (step - 5000))
        
        tau = cur_max * tau_frac
        w = np.exp((b - cur_max) / tau)
        w /= w.sum()
        
        g = _fft_grad(a, w, n, L)
        gn = np.linalg.norm(g)
        if gn < 1e-18:
            break
        
        # Gradient descent with line search
        lr = min(0.4, 20.0 / (step + 1.0))
        for _ in range(6):
            a_try = a - lr * g / gn
            a_try = np.maximum(a_try, 0.0)
            ss = a_try.sum()
            if ss < 1e-14:
                lr *= 0.5
                continue
            a_try /= ss
            b_try = np.convolve(a_try, a_try)
            if b_try.max() <= cur_max * 1.002:
                a = a_try
                if b_try.max() < cur_max * 0.9995:
                    last_improve = step
                break
            lr *= 0.5
        
        step += 1
        
        # Periodic best check
        if step % 80 == 0:
            b_ck = np.convolve(a, a)
            sc = 2.0 * n * b_ck.max() / (a.sum()**2)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        # Restart strategies
        if step > 2000 and step - last_improve > 2000:
            # Multiplicative perturbation
            noise = rng.normal(0, 0.15, n)
            a = np.maximum(a * np.exp(noise), 1e-15)
            a /= a.sum()
            last_improve = step
            step = 0
        
        if step > 6000 and step - last_improve > 5000:
            # Blend with best
            a = 0.5 * a + 0.5 * best_a
            a = np.maximum(a, 1e-15)
            a /= a.sum()
            last_improve = step
            step = 0
    
    # Final evaluation
    fs = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    return [float(x) for x in best_a]
