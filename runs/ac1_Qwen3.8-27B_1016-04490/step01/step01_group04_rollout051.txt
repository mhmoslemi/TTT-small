def next_pow2(m):
    L = 1
    while L < m:
        L <<= 1
    return L

def norm1(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.ones_like(a) / a.size
    return a / s

def autocorr(a, flen):
    af = np.fft.rfft(a, n=flen)
    c = np.fft.irfft(af * af, n=flen)
    n = a.size
    return np.maximum(c[:2*n - 1], 0.0)

def grad_smooth(a, w, flen):
    n = a.size
    arf = np.fft.rfft(a[::-1], n=flen)
    wf = np.fft.rfft(w, n=flen)
    conv = np.fft.irfft(arf * wf, n=flen)
    return 2.0 * np.maximum(conv[n - 1:n - 1 + n], 0.0)

def softmax_w(c, tau):
    cm = c.max()
    if cm < 1e-300:
        w = np.zeros_like(c)
        w[0] = 1.0
        return w
    z = np.clip((c - cm) / max(tau, 1e-30), -60.0, 0.0)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-30:
        w = np.zeros_like(c)
        w[c.argmax()] = 1.0
        return w
    return e / s

def score_fft(a, flen):
    c = autocorr(a, flen)
    return 2.0 * a.size * c.max()
