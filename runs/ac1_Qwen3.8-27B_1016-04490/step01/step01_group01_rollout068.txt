def _fft_conv(a, N):
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)


def _score(a, n, N):
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = _fft_conv(a, N)
    return float(2.0 * n * b.max() / (S * S))


def _subgrad_dir(a, b, n, k):
    idx = np.argpartition(b, -k)[-k:]
    w = b[idx].copy()
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w = w / ws
    g = np.zeros(n)
    for wi, m in zip(w, idx):

        lo = max(0, mi - n + 1)
        hi = min(n - 1, mi)
        if lo <= hi:
            seg = a[mi - hi:mi - lo + 1][::-1]
            g[lo:hi + 1] += wi * seg
    g = g - g.sum() / n
    return g


def _clip_norm(a, target):

def _sawtooth(n, p, rng):
    if p <= 1 or p >= n:
        return np.ones(n)
    i = np.arange(n)
    return ((i + int(rng.integers(0, p))) % p).astype(np.float64) + 1.0


def _period_proj(a, n, p):
    if p <= 1 or p > n:
        return a

    if m < p:
        return a
    blk = a[:m].reshape(-1, p).mean(axis=0)
    return np.tile(blk, (n + p - 1) // p)[:n].copy()
