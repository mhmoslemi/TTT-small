import time
import scipy.optimize as sopt

def _fft_conv(a):
    n = a.size
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:L]

def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones(z.size) / z.size
    return e / s

def _score(p):
    # p is normalized (sum=1), nonnegative
    conv = _fft_conv(p)
    return float(2.0 * p.size * conv.max())

def _sg(z, beta):
    """Objective and gradient for softmax parameterization."""
    p = _softmax(z)
    n = p.size
    conv = _fft_conv(p)
    bc = beta * conv
    bcs = bc - bc.max()
    se = np.exp(bcs)
    w = se / se.sum()
    val = (bc.max() + np.log(se.sum())) / beta
    
    # grad_p[m] = 2 * sum_j p[j] * w[m+j]
    M = 1 << (3 * n - 3).bit_length()
    P = np.fft.fft(p, M)
    W = np.fft.fft(w, M)
    cc = np.fft.ifft(P * np.conj(W))[:n].real
    grad_p = 2.0 * cc
    
    # grad_z = p * (grad_p - p·grad_p)
    dot = np.dot(p, grad_p)
    grad_z = p * (grad_p - dot)
    return val, grad_z
