    while L < 2 * n:
        L <<= 1
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa, L)[:2*n-1]
    return c

def _grad_soft(a, c, tau, n):
    """Compute gradient of soft-max of auto-convolution."""
    L = 1
    while L < 3 * n:
        L <<= 1
    # soft-max weights
    c_shifted = c - c.max()
    w = np.exp(c_shifted / tau)
    w /= w.sum()
    # gradient via cross-correlation
    fa = np.fft.rfft(a, L)
    fw = np.fft.rfft(w, L)
    g = np.fft.irfft(fa * fw.conj(), L)[:3*n-2]
    grad = 2.0 * g[2*n-2::-1][:n]
    return grad, w
