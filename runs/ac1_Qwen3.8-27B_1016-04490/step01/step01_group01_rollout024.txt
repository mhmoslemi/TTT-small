                if mpat < 35:
                    break

            P = 64
            g = _pgrad(a, b, n, P)
            g -= g.mean()

            a *= np.exp(-eta * g)
            a = _norm(np.clip(a, 0.0, 1000.0), T)

            eta *= 0.9975
            if eta < 0.012:
                eta = 0.012

            if rng.random() < 0.05:
                w = int(rng.integers(max(3, n // 120), max(5, n // 25)))
                s0 = int(rng.integers(0, n - w))
                a[s0:s0 + w] *= np.exp(rng.normal(0, 0.15))
                a = _norm(np.clip(a, 0.0, 1000.0), T)

    # Fine polish phase
    if best is not None and time.monotonic() < t_end - 5:
        a = best.copy()
        eta_f = 0.04
        for _ in range(500):
            if time.monotonic() > t_end - 5:
                break
            b = _fft_conv(a, N)
            g = _pgrad(a, b, n, 48)
            g -= g.mean()
            a *= np.exp(-eta_f * g)
            a = _norm(np.clip(a, 0.0, 1000.0), T)
            sc = _score_fast(a, n, N)
            if sc < best_sc:
                best_sc = sc
                best = a.copy()
            eta_f *= 0.9985
            if eta_f < 0.004:
                eta_f = 0.004

    # Ultra-fine final touches
    if best is not None and time.monotonic() < t_end - 2:
        a = best.copy()
        eta_u = 0.01
        for _ in range(200):
            if time.monotonic() > t_end - 2:
                break
            b = _fft_conv(a, N)
            g = _pgrad(a, b, n, 32)
            g -= g.mean()
            a *= np.exp(-eta_u * g)
            a = _norm(np.clip(a, 0.0, 1000.0), T)
            sc = _score_fast(a, n, N)
            if sc < best_sc:
                best_sc = sc
                best = a.copy()
            eta_u *= 0.999
            if eta_u < 0.002:
                eta_u = 0.002

    if best is None or not np.all(np.isfinite(best)) or best.sum() < 0.01:
        best = np.ones(n)

    out = [float(x) for x in best]
    ev = evaluate_sequence(out)
    if not np.isfinite(ev):
        out = [1.0] * n
    return out
