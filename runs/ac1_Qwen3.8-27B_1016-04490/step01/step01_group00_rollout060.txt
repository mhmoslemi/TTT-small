def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def norm_seq(a, target):
    a = np.asarray(a, dtype=np.float64).copy()
    if a.size == 0:
        return a
    bad = ~np.isfinite(a)
    if bad.any():
        a[bad] = 0.0
    a[a < 0.0] = 0.0
    s = float(a.sum())
    if s <= 1e-12:
        a = np.ones_like(a)
        s = float(a.size)
    a *= float(target) / s
    a = np.clip(a, 0.0, 1000.0)
    s2 = float(a.sum())
    if s2 <= 1e-12:
        a = np.ones_like(a)
        a *= float(target) / float(a.size)
    return a

                ex = exact_score(x)
                last_exact = it
                if np.isfinite(ex) and ex < best_exact - 1e-10:
                    best_exact = ex
                    best_exact_x = x.copy()

        if cmax <= 1e-14:
            break

        diff = (conv - cmax) / max(T, 1e-12)
        np.clip(diff, -40.0, 0.0, out=diff)
        p = np.exp(diff)
        psum = float(p.sum())
        if psum <= 1e-300:
            p = np.zeros_like(p)
            p[(2 * n - 1) // 2] = 1.0
            psum = 1.0
        else:
            p /= psum

        P = sfft.rfft(p, L)
        prod = np.conj(X) * P
        if L > 1:
            prod[-1] = prod[-1].real
        corr = np.real(sfft.irfft(prod, L))[:n]
        g = 2.0 * corr
        g -= g.mean()

        vel = beta * vel + (1.0 - beta) * g
        drms = float(np.sqrt(np.mean(vel * vel))) + 1e-14
        xrms = float(np.sqrt(np.mean(x * x))) + 1e-14
        alpha = lr * (xrms / drms)
        if alpha > 1.0:
            alpha = 1.0

        x_trial = x - alpha * vel
        np.maximum(x_trial, 0.0, out=x_trial)
        s = float(x_trial.sum())
        if s <= 1e-10:
            x = best_surr_x.copy()
            vel = np.zeros_like(vel)
            it += 1
            continue
        x_trial *= target / s
        x_trial = np.clip(x_trial, 0.0, 1000.0)
        x = x_trial

        if np.isfinite(sc):
            if sc <= old_surr + 1e-10:
                lr = min(lr_max, lr * 1.04)
            else:
                lr = max(lr_min, lr * 0.80)
            old_surr = sc

        T = max(Tmin, T * 0.9998)
        Tcap = 0.35 * cmax
        if T > Tcap:
            T = Tcap

        if it > 0 and it % 3000 == 0:
            noise = smooth_noise(n, rng, 0.02 * xrms)
            x = norm_seq(x + noise, target)
            vel *= 0.5

        if stall > patience:
            restarts += 1
            xrms = float(np.sqrt(np.mean(x * x))) + 1e-14
            if restarts % 3 == 2 and init_ptr < len(cand_arrays):
                x = cand_arrays[init_ptr].copy()
                init_ptr += 1
            else:
                x = best_surr_x.copy()
                noise = smooth_noise(n, rng, 0.10 * xrms)
                x = norm_seq(x + noise, target)
            vel = np.zeros_like(vel)
            T = max(0.15, 0.25 * best_surr)
            lr = 0.15
            stall = 0
            last_exact = it

        it += 1

    fallback = parent_norm
    best_final = fallback
    best_final_score = float("inf")

    if best_exact_x is not None and np.isfinite(best_exact):
        best_final = best_exact_x
        best_final_score = best_exact
    else:
        fwd = exact_score(fallback)
        if np.isfinite(fwd):
            best_final = fallback
            best_final_score = fwd

    if time.monotonic() < deadline and best_surr_x is not None:
        if best_surr < best_final_score - 1e-8:
            fwd = exact_score(best_surr_x)
            if np.isfinite(fwd) and fwd < best_final_score:
                best_final = best_surr_x
                best_final_score = fwd

    if time.monotonic() < deadline and x is not None:
        fwd = exact_score(x)
        if np.isfinite(fwd) and fwd < best_final_score:
            best_final = x
            best_final_score = fwd

    if best_final_score >= 1e100 or not np.all(np.isfinite(best_final)):
        best_final = parent_norm

    return np.asarray(best_final, dtype=np.float64).tolist()
