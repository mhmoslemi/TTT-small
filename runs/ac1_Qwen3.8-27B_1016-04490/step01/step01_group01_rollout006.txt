```python
def _centered_grad(a, b, n, N, n_peaks):
    S = a.sum()
    mean_a = S / n
    peak_idx = np.argpartition(b, -n_peaks)[-n_peaks:]
    pv = b[peak_idx]
    w_sum = pv.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    grad = np.zeros(n)
    for w, k in zip(pv / w_sum, peak_idx):
        ki = int(k)
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if lo <= hi:
            seg = a[ki - hi: ki - lo + 1][::-1]
            grad[lo:hi + 1] += w * (seg - mean_a)
    return grad
```
