def _score_unnorm(a):
    """Score assuming sum(a)=1: 2n*max(conv(a,a))"""
    b = _fft_conv(a, a)
    return 2.0 * len(a) * b.max()

def _project_simple(a, target):
    """Project a onto {x >= 0, sum(x) = target} using water-filling."""
    a_pos = np.maximum(a, 0.0)
    s = a_pos.sum()
    if s < 1e-15:
        return np.full(len(a), target / len(a))
    if abs(s - target) < 1e-10 * target:
        return a_pos / s * target
    lo = a.min() - 1.0
    hi = a.max() + 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.maximum(a - mid, 0.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    nu = 0.5 * (lo + hi)
    result = np.maximum(a - nu, 0.0)
    s = result.sum()
    if s > 1e-15:
        result *= target / s
    return result
