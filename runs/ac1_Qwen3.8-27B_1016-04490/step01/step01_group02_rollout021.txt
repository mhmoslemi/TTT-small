import numpy as np
import time
from scipy.sparse import lil_matrix, csr_matrix
from scipy.optimize import linprog

def _fft_conv(x, y):
    m = len(x) + len(y) - 1
    sz = 1
    while sz < m:
        sz <<= 1
    fx = np.fft.rfft(x, sz)
    fy = np.fft.rfft(y, sz)
    return np.fft.irfft(fx * fy, sz)[:m]

def _build_conv_matrix(f, n):
    """Build sparse (2n-1) x n matrix T where (T@b)[k] = (f*b)[k]."""
    rows, cols, vals = [], [], []
    for k in range(2*n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(k, n - 1)
        for j in range(jmin, jmax + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def _solve_lp(f, M, n):
    """Solve max 1^T b s.t. T@b <= M, b >= 0."""
    T = _build_conv_matrix(f, n)
    # Convert to minimization: min -1^T b
    c = -np.ones(n)
    # Inequality: T@b <= M*1
    A_ub = T
    b_ub = M * np.ones(2*n - 1)
    # Bounds: b >= 0
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

    n = len(a)
    S = a.sum()

    C = _fft_conv(a, a)
    return 2.0 * n * C.max() / (S * S)
