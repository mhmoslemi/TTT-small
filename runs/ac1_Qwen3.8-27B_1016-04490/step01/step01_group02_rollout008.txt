        np.exp(-8.0 * (t - 0.25) ** 2) + np.exp(-8.0 * (t - 0.75) ** 2) + 0.01,
        0.5 + 0.3 * np.cos(2.0 * np.pi * 3.7 * t) + 0.2 * np.cos(2.0 * np.pi * 8.1 * t),
        1.0 - 0.4 * t + 0.2 * np.sin(2.0 * np.pi * 2.3 * t),
    ]

    for i in range(len(inits)):
        inits[i] = np.maximum(inits[i], 0.0).astype(np.float64)
        s = inits[i].sum()
        if s < 1e-10:
            inits[i][:] = 1.0
        else:
            inits[i] /= s

    rem = max(deadline - time.monotonic(), 5.0)
    t_per = rem / len(inits)

    best_all = parent.copy()
    best_sc = _score(parent)

    for idx in range(len(inits)):
        now = time.monotonic()
        if now >= deadline:
            break
        t_end = min(now + t_per, deadline)
        if t_end - now < 3.0:
            break
        b, sc = _optimize(inits[idx].copy(), t_end, rng)
        if sc < best_sc:
            best_sc = sc
            best_all = b.copy()

    # Polish with remaining time
    rem2 = deadline - time.monotonic()
    if rem2 > 10.0:
        b, sc = _optimize(best_all.copy(), deadline, rng)
        if sc < best_sc:
            best_sc = sc
            best_all = b.copy()

    best_all = np.maximum(best_all, 0.0)
    bs = best_all.sum()
    if bs < 0.01:
        best_all = np.ones(n, dtype=np.float64) / n

    fl = [float(v) for v in best_all]
    fs = evaluate_sequence(fl)
    if not np.isfinite(fs):
        fl = [float(v) for v in parent]
        fs = evaluate_sequence(fl)
        if not np.isfinite(fs):
            fl = [1.0 / n] * n

    return fl
