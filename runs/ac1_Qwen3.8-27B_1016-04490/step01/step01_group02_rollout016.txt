def _fft_conv(x, y):
    n = len(x) + len(y) - 1
    sz = 1
    while sz < n:

    return np.fft.irfft(np.fft.rfft(x, sz) * np.fft.rfft(y, sz), sz)[:n]

    s = _fft_conv(a, a)
    return float(2.0 * n * s.max() / (a.sum()**2))
