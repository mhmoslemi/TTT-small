    s = float(h.sum())
    if s < 1e-9:
        h = np.ones_like(h)
        s = float(h.size)
    return int(h.size), h / s


def _build_inits(n, parent, rng):
    j = np.arange(n, dtype=np.float64)
    x = j / float(n)
    raw = [
        parent.copy(),
        np.sin(np.pi * (j + 0.5) / float(n)) ** 1.5 + 1e-6,
        np.exp(-6.0 * (x - 0.25) ** 2) + np.exp(-6.0 * (x - 0.75) ** 2) + 0.03,
        (1.0 + np.cos(2.0 * np.pi * x)) ** 2 + 1e-4,
        (1.0 - np.cos(2.0 * np.pi * x)) ** 2 + 1e-4,
        np.exp(-4.0 * (x - 0.5) ** 2) + 0.03,
    ]
    out = []
    for a in raw:
        a = np.maximum(a, 0.0)
        s = float(a.sum())
        out.append(a / s if s > 1e-9 else np.ones(n, dtype=np.float64))
    return out


def _run(a, deadline, rng):
    n = int(np.size(a))
    a = np.maximum(np.asarray(a, dtype=np.float64).copy(), 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-9:
        a = np.ones(n, dtype=np.float64)
        s = float(n)
    a = a / s
    best = a.copy()
    best_score = _score(best)
    if not np.isfinite(best_score):
        best_score = float("inf")
    it = 0
    last_eval = time.monotonic()
    r0 = 0.7
    rmin = 0.005
    tau_cool = max(400.0, 0.5 * float(n))
    while True:
        now = time.monotonic()
        if now >= deadline:
            break
        c = _fft_conv(a, a)
        cm = float(c.max())
        if not np.isfinite(cm) or cm <= 0.0:
            break
        r = r0 * math.exp(-it / tau_cool)
        if r < rmin:
            r = rmin
        tau = r * cm + 1e-300
        z = c / tau
        z = z - float(z.max())
        e = np.exp(z)
        es = float(e.sum())
        if es < 1e-300:
            p = np.full_like(c, 1.0 / max(1, c.size))
        else:
            p = e / es
        ca = _fft_conv(p[::-1], a)
        g = 2.0 * ca[n - 1:2 * n - 1][::-1]
        g = np.maximum(g, 0.0)
        gm = float(g.max())
        if gm > 0.0 and np.isfinite(gm):
            eta = 1.0 / gm
            a = a * np.exp(-eta * g)
            a = np.maximum(a, 1e-300)
            sa = float(a.sum())
            a = a / sa if sa > 0.0 else np.ones(n, dtype=np.float64)
        it += 1
        if (it % 150 == 0) or (now - last_eval > 0.3):
            sc = _score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best = a.copy()
            last_eval = time.monotonic()
    sc = _score(best)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
    return best, best_score


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    budget = float(budget_s)
    deadline = t0 + max(1.0, budget - 10.0)
    n, parent = _get_parent()
    inits = _build_inits(n, parent, rng)
    best = parent.copy()
    best_score = _score(best)
    if not np.isfinite(best_score):
        best = np.ones(n, dtype=np.float64)
        best_score = _score(best)
    i = 0
    while True:
        now = time.monotonic()
        if now >= deadline:
            break
        remaining = deadline - now
        if i == 0:
            rt = min(120.0, 0.35 * budget, remaining)
        else:
            rt = min(30.0, remaining - 2.0)
        if rt < 5.0:
            break
        if i < len(inits):
            a0 = inits[i].copy()
        else:
            amp = 0.08 * (0.2 + 0.8 * (remaining / max(1.0, budget)))
            a0 = best * (1.0 + amp * rng.standard_normal(n))
            a0 = np.maximum(a0, 0.0)
            sa = float(a0.sum())
            a0 = a0 / sa if sa > 0.0 else np.ones(n, dtype=np.float64)
        b, sc = _run(a0, now + rt, rng)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = b.copy()
        i += 1
    vf = np.nan_to_num(np.asarray(best, dtype=np.float64), nan=0.0, posinf=1.0, neginf=0.0)
    vf = np.clip(vf, 0.0, 1000.0)
    if vf.size == 0 or not np.isfinite(vf).all() or float(vf.sum()) < 0.01:
        vf = np.ones(n, dtype=np.float64)
    return [float(v) for v in vf]
