import numpy as np, time

def _renorm(a, target):
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s <= 1e-300:
        return np.zeros_like(a)
    return a * (target / s)

def _conv_score(a, n, N):
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    cv = c[:2*n-1]
    S = float(np.sum(a))
    if S < 0.01:
        return cv, float('inf')
    return cv, float(2.0 * n * float(cv.max()) / (S * S))

def _pair_b(a, cv, n, K=8, tau_frac=0.005):
    max_c = float(cv.max())
    if not np.isfinite(max_c) or max_c <= 0:
        return np.zeros(n)
    m = cv.size
    if K >= m:
       idx = np.arange(m)
    else:
       idx = np.argpartition(cv, m-K)[-K:]
    vals = cv[idx]
    tau = max(max_c * tau_frac, 1e-12)
    w = np.exp((vals - max_c) / tau)
    wsum = float(w.sum())
    if wsum <= 1e-300:
       w = np.zeros_like(w); w[0]=1.0
    else: w /= wsum
    b = np.zeros(n)
    # loop
    for t in range(K):
       wt = float(w[t])
       if wt <= 1e-12: continue
       pos = int(idx[t])
       lo = pos - n + 1
       if lo < 0: lo = 0
       hi = pos
       if hi >= n: hi = n-1
       if lo <= hi:
          b[lo:hi+1] += wt * a[pos-hi:pos-lo+1][::-1]
    b -= b.mean()
    return b

import numpy as np
import time

np.seterr(over="ignore", under="ignore", invalid="ignore")


def _renorm(a, target):
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s <= 1e-300:
        return np.zeros_like(a)
    return a * (target / s)


def _conv_score(a, n, N):
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    m = 2 * n - 1
    cv = c[:m]
    S = float(np.sum(a))
    if S < 0.01:
        return cv, float("inf")
    return cv, float(2.0 * n * float(np.max(cv)) / (S * S))


def _partner_weights(a, cv, n, K):
    max_c = float(np.max(cv))
    if not np.isfinite(max_c) or max_c <= 0.0:
        return np.zeros(n, dtype=np.float64)

    m = cv.size
    if K >= m:
        idx = np.arange(m)
    else:
        idx = np.argpartition(cv, m - K)[-K:]

    L = int(idx.size)
    vals = cv[idx].astype(np.float64, copy=False)
    w = np.clip(vals, 0.0, None)
    wsum = float(w.sum())
    if wsum <= 1e-300:
        w = np.full(L, 1.0 / max(L, 1), dtype=np.float64)
    else:
        w = w / wsum

    b = np.zeros(n, dtype=np.float64)
    for t in range(L):
        wt = float(w[t])
        if wt <= 1e-14:
            continue
        pos = int(idx[t])
        lo = pos - n + 1
        if lo < 0:
            lo = 0
        hi = pos
        if hi >= n:
            hi = n - 1
        if lo <= hi:
            b[lo:hi + 1] += wt * a[pos - hi:pos - lo + 1][::-1]
    return b


def _init_ramp(n, rev):
    if n <= 1:
        return np.ones(n, dtype=np.float64)
    x = np.arange(n, dtype=np.float64) / float(n - 1)
    a = 0.7 + 1.3 * x
    if rev:
        a = a[::-1]
    a = np.clip(a, 1e-6, 1000.0)
    s = float(a.sum())
    if s <= 1e-300:
        return a
    return a / s


def _init_cos(n, freq, amp):
    if n <= 1:
        return np.ones(n, dtype=np.float64)
    x = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.cos(2.0 * np.pi * float(freq) * x / float(n))
    a = np.clip(a, 1e-6, 1000.0)
    s = float(a.sum())
    if s <= 1e-300:
        return a
    return a / s


def _init_chirp(n, amp):
    if n <= 1:
        return np.ones(n, dtype=np.float64)
    x = np.arange(n, dtype=np.float64)
    phase = (2.0 * np.pi / float(n + 1)) * x * x
    a = 1.0 + amp * np.cos(phase)
    a = np.clip(a, 1e-6, 1000.0)
    s = float(a.sum())
    if s <= 1e-300:
        return a
    return a / s


def _init_thue(n, amp):
    t = np.zeros(n, dtype=np.int8)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    sgn = (1 - 2 * t).astype(np.float64)
    a = 1.0 + amp * sgn
    a = np.clip(a, 1e-6, 1000.0)
    sv = float(a.sum())
    if sv <= 1e-300:
        return a
    return a / sv


def _init_random(n, rng, amp):
    a = 1.0 + amp * rng.standard_normal(n)
    a = np.clip(a, 1e-6, 1000.0)
    s = float(a.sum())
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s


def _gather_inits(n, rng, N, parent):
    starts = []
    if parent is not None:
        starts.append(parent)
        p = _renorm(parent, 1.0)
        if float(np.sum(p)) > 1e-300:
            starts.append(np.sort(p)[::-1])
            starts.append(np.sort(p))
            pmax = float(np.max(p))
            if pmax > 1e-300:
                starts.append(1.0 / (p + 0.1 * pmax))

    starts.append(_init_ramp(n, False))
    starts.append(_init_ramp(n, True))
    starts.append(_init_cos(n, 1, 0.70))
    starts.append(_init_cos(n, 2, 0.70))
    starts.append(_init_chirp(n, 0.70))
    starts.append(_init_thue(n, 0.50))
    starts.append(_init_random(n, rng, 0.50))
    starts.append(_init_random(n, rng, 0.80))

    cands = []
    idx = 0
    for arr in starts:
        a0 = np.asarray(arr, dtype=np.float64).copy()

        if np.isfinite(sc0):

            idx += 1

    return cands
