    n = len(f)
    m = 2 * n - 1
    
    # Build sparse convolution matrix A (m x n)
    # A[k, j] = f[k-j] if 0 <= k-j <= n-1
    col_indices = []
    row_indices = []
    data = []
    
    # Use vectorized approach
    # For k = 0 to n-1: entries at (k, 0), (k, 1), ..., (k, k) with values f[k], f[k-1], ..., f[0]
    # For k = n to 2n-2: entries at (k, k-n+1), (k, k-n+2), ..., (k, n-1) with values f[k-n], f[k-n+1], ..., f[2n-2-k]

            row_indices.append(k)
            col_indices.append(j)
            data.append(f[k - j])
    
    # Lower triangle

        for j in range(n - n_entries, n):
            row_indices.append(k)
            col_indices.append(j)
            data.append(f[k - j])

    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.HIGHS, max_time=time_limit)
    
    if b.value is None:
        return None
    return b.value

def _fft_conv(x, y):
    m = len(x) + len(y) - 1
    sz = 1
    while sz < m:
        sz <<= 1
    fx = np.fft.rfft(x, sz)
    fy = np.fft.rfft(y, sz)
    return np.fft.irfft(fx * fy, sz)[:m]

    z -= z.max()

    return sp.csc_matrix((vals, (rows, cols)), shape=(m, n))

import numpy as np
import time
import scipy.sparse as sp
import cvxpy as cp

def _fft_conv(x, y):
    m = len(x) + len(y) - 1
    sz = 1
    while sz < m:
        sz <<= 1
    fx = np.fft.rfft(x, sz)
    fy = np.fft.rfft(y, sz)
    return np.fft.irfft(fx * fy, sz)[:m]

def _softmax(x, tau):
    z = x.astype(np.float64) / tau
    z -= z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.full(len(x), 1.0 / len(x), dtype=np.float64)
    return e / s

def _fast_score(a):
    n = a.size
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = _fft_conv(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _compute_grad(a, tau):
    n = a.size
    sz1 = 1
    while sz1 < 2 * n - 1:
        sz1 <<= 1
    fa = np.fft.rfft(a, sz1)
    s = np.fft.irfft(fa * fa, sz1)[:2 * n - 1]
    p = _softmax(s, tau)
    sz2 = 1
    while sz2 < 3 * n - 2:
        sz2 <<= 1
    fc = np.fft.rfft(p[::-1], sz2)
    fa2 = np.fft.rfft(a, sz2)
    ca = np.fft.irfft(fc * fa2, sz2)[:3 * n - 2]
    return 2.0 * ca[n - 1:2 * n - 1][::-1]

def _build_conv_mat(f):
    n = len(f)
    m = 2 * n - 1
    total = n * n
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    vals = np.empty(total, dtype=np.float64)
    idx = 0
    for k in range(m):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        cnt = j_hi - j_lo + 1
        jj = np.arange(j_lo, j_hi + 1)
        rows[idx:idx + cnt] = k
        cols[idx:idx + cnt] = jj
        vals[idx:idx + cnt] = f[k - jj]
        idx += cnt
    return sp.csc_matrix((vals, (rows, cols)), shape=(m, n))
