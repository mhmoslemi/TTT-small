def _fft_conv(x, y):
    m = len(x) + len(y) - 1
    sz = 1
    while sz < m:
        sz <<= 1
    fx = np.fft.rfft(x, sz)
    fy = np.fft.rfft(y, sz)
    return np.fft.irfft(fx * fy, sz)[:m]

    b = _fft_conv(a, a)
    return float(2 * len(a) * b.max() / a.sum()**2)

    # b has length 2n-1
    z = b / tau
