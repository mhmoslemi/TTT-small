import numpy as np

def _fft_conv(x, y):
    m = len(x) + len(y) - 1
    sz = 1
    while sz < m:
        sz <<= 1
    fx = np.fft.rfft(x, sz)
    fy = np.fft.rfft(y, sz)
    return np.fft.irfft(fx * fy, sz)[:m]


def _soft_weights(s, tau, topk):
    n = s.size
    if topk and 0 < topk < n:
        idx = np.argpartition(s, -topk)[-topk:]
        vals = s[idx]
        vals = vals - vals.max()
        e = np.exp(vals / tau)
        p = np.zeros(n)
        p[idx] = e / e.sum()
    else:
        z = s / tau
        z = z - z.max()
        e = np.exp(z)
        p = e / e.sum()
    return p
