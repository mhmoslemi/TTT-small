from scipy import fft as sfft

def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def do_conv(x, fs):
    X = sfft.rfft(x, fs)
    c = np.real(sfft.irfft(X * X, fs))[:2 * len(x) - 1]
    return np.maximum(c, 0.0)

def do_xcorr(p, x, fs):
    P = sfft.rfft(p, fs)
    X = sfft.rfft(x, fs)
    return np.real(sfft.irfft(P * np.conj(X), fs))[:len(x)]
