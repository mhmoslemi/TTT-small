import numpy as np
import time

def _convolve_fft(a, b):
    """Linear convolution via FFT."""
    n = len(a) + len(b) - 1
    L = 1
    while L < n:
        L <<= 1
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    return np.fft.irfft(fa * fb, L)[:n]
