    """Linear autocorrelation via FFT, O(n log n)."""
    n = len(a)
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    fa = np.fft.rfft(a, L)
    return np.fft.irfft(fa * fa.conj(), L)[:2 * n - 1]

    ac = _fft_autocorr(a / s)
