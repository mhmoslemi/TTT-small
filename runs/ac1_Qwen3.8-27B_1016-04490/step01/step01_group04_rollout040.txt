def targeted_reduce(a, c, k_hot, flen):
    """For each hot lag k, find the dominant contributing pairs and reduce them."""
    b = a.copy()
    n = a.size
    for k in k_hot:
        # c[k] = sum_j a[j] * a[k-j] for valid j
        # Find j that maximizes a[j] * a[k-j]
        # For efficient computation:
        if k < n:
            # c[k] = sum_{j=0}^{k} a[n-1-j] * a[n-1-j+k] ... 
            # Actually for np.convolve: c[k] = sum_j a[j]*a[k-j] where 0<=j<n, 0<=k-j<n
            # So j ranges from max(0, k-n+1) to min(k, n-1)
            jmin = max(0, k - n + 1)
            jmax = min(k, n - 1)
            if jmin > jmax:
                continue
            j = np.arange(jmin, jmax + 1)
            products = a[j] * a[k - j]
            top_idx = j[np.argmax(products)]
            # Reduce the larger element
            if a[top_idx] > a[k - top_idx]:
                b[top_idx] *= 0.95
            else:
                b[k - top_idx] *= 0.95
    return b
