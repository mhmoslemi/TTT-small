Key math:
- Scale invariant, normalize sum=1
- Score = 2*n*max(conv(a,a))
- Gradient of max(conv(a,k)) wrt a[i] = 2 * sum_k p[k] * a[k-i] where p is softmax of conv or point mass at max
- conv(p[::-1], a)[n-1:2n-1][::-1] gives the gradient (divided by 2)

For the LP:
- max sum(g) s.t. g >= 0, (a*g)[k] <= M
- (a*g)[k] = sum_j a[j] * g[k-j] where g[x]=0 outside [0,n-1]
- Matrix A: A[k,i] = a[k-i] if 0<=k-i<=n-1, 0<=i<=n-1
- For column i: nonzeros at rows i to i+n-1, values a[0], a[1], ..., a[n-1]

    n = a.size

    # Use scipy.sparse with coo format
    # rows = [i, i+1, ..., i+n-1] for each i

Actually, for the exact max gradient:
- Find k* = argmax(c)
- grad[i] = 2 * a[k* - i] (for valid i)
- This is just 2 * a reversed and shifted

This is MUCH cheaper than the softmax version (no need for second FFT).

Let me use a hybrid:
- Most iterations: exact max gradient (cheap: 1 FFT for c + O(n) for grad)
- Every K iterations: softmax gradient (more expensive but smoother)

With exact max gradient:
```

k = c.argmax()
# grad[i] = 2 * a[k-i] for valid i
# a[k-i] is nonzero for 0 <= k-i <= n-1, i.e., k-n+1 <= i <= k
# For i in [0, n-1]:
#   if k-i in [0, n-1]: grad[i] = 2*a[k-i]
#   else: grad[i] = 0
```

This is O(n) after the FFT. Much faster!

But the exact subgradient only points in one direction (toward reducing the single max). The softmax gives a weighted combination of all the "hot" spots, which helps flatten the whole convolution.

Strategy:
- Use softmax with moderate tau most of the time
- When the top few c values are close, reduce tau to focus on the max
- When stuck, increase tau to explore

Let me also add a "pairing" trick: if c[k] is high, we want to reduce a's contribution at lag k. The gradient tells us which a[i] to reduce. The multiplicative update naturally does this.

import numpy as np
import time
from scipy import sparse
from scipy.optimize import linprog

def _fft_conv(x, y):
    m = len(x) + len(y) - 1
    sz = 1
    while sz < m:
        sz <<= 1
    fx = np.fft.rfft(x, sz)
    fy = np.fft.rfft(y, sz)
    return np.fft.irfft(fx * fy, sz)[:m]

def _score(a):
    n = a.size
    if n == 0:
        return float('inf')
    c = _fft_conv(a, a)
    return 2.0 * n * c.max()

def _grad_softmax(a, tau):
    n = a.size
    c = _fft_conv(a, a)
    z = c / tau
    z -= z.max()
    p = np.exp(z)
    s = p.sum()
    if s < 1e-300:
        idx = c.argmax()
        p = np.zeros(2*n-1)
        p[idx] = 1.0
    else:
        p /= s
    ca = _fft_conv(p[::-1], a)
    return 2.0 * ca[n-1:2*n-1][::-1]

def _grad_exact(a):
    n = a.size
    c = _fft_conv(a, a)
    k = int(c.argmax())
    g = np.zeros(n)
    lo = max(0, k - (n-1))
    hi = min(n-1, k)
    for i in range(lo, hi+1):
        g[i] = 2.0 * a[k - i]
    return g

def _optimize(a, deadline, rng):
    n = a.size
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n

    best_sc = _score(a)
    tau = 1.0 / n
    lr = 1.0
    stalled = 0
    it = 0
    eval_every = 10
    
    while time.monotonic() < deadline:
        # Alternate between softmax and exact gradient
        if it % 3 == 0:
            g = _grad_softmax(a, tau)
        else:
            g = _grad_exact(a)
        
        # Multiplicative update
        a_new = a * np.exp(-lr * g)
        a_new = np.maximum(a_new, 1e-30)
        sm = a_new.sum()
        if sm < 1e-10:
            break
        a_new /= sm
        a = a_new
        
        it += 1
        if it % eval_every == 0:
            sc = _score(a)
            if sc < best_sc - 1e-10:
                best_sc = sc
                best = a.copy()
                stalled = 0
            else:
                stalled += 1
            
            # Adapt parameters
            if stalled == 3:
                tau *= 0.9
                lr *= 0.85
            elif stalled >= 5 and stalled % 5 == 0:
                # Gentle perturbation
                a += 0.005 * rng.standard_normal(n)
                a = np.maximum(a, 0.0)
                a /= a.sum()
                best = a.copy()
                best_sc = _score(a)
                stalled = 0
                lr = 1.0
                tau = max(tau * 3, 1.0/n)
            
            if stalled >= 15:
                # Stronger restart
                a += 0.02 * rng.standard_normal(n)
                a = np.maximum(a, 0.0)
                a /= a.sum()
                stalled = 0
                tau = 2.0 / n

        best = a.copy()
    return best, best_sc

def _lp_refine(a, deadline):
    n = a.size
    M = _fft_conv(a, a).max()
    
    # Build sparse constraint matrix
    # A[k, i] = a[k-i] for 0<=k-i<=n-1
    # Column i: rows i to i+n-1, values a[0]...a[n-1]

    A = sparse.csr_matrix((vals_arr, (rows_arr, cols_arr)), shape=(2*n-1, n))

    c_lp = np.full(n, -1.0)

                                            if bs2 > 1e-10:

                                                    a = blend2
                                                    M = _fft_conv(a, a).max()
                                except:
                                    pass
    except:
        pass
    
    return a

    x = (j + 0.5) / n
    
    inits = []
    
    # 1. Parent
    p = np.maximum(parent, 0.0).copy()
    ps = p.sum()
    if ps > 1e-10:
        p /= ps
    else:
        p[:] = 1.0/n
    inits.append(p)
    
    # 2. Cosine window (raised cosine)
    c1 = np.cos(np.pi * x)**2
    c1 = np.maximum(c1, 0.01)
    inits.append(c1 / c1.sum())
    
    # 3. Uniform
    inits.append(np.ones(n) / n)
    
    # 4. Gaussian
    mu = 0.5
    sig = 0.25
    g1 = np.exp(-0.5 * ((x - mu)/sig)**2)
    inits.append(g1 / g1.sum())
    
    # 5. Two Gaussian peaks
    g2 = np.exp(-0.5 * ((x-0.25)/0.15)**2) + np.exp(-0.5 * ((x-0.75)/0.15)**2) + 0.01
    inits.append(g2 / g2.sum())
    
    # 6. Triangle (peak at center)
    t1 = 1.0 - np.abs(2*x - 1)
    t1 = np.maximum(t1, 0.01)
    inits.append(t1 / t1.sum())
    
    return inits

import time

def _fft_conv(x, y):
    m = len(x) + len(y) - 1
    sz = 1
    while sz < m:
        sz <<= 1
    return np.fft.irfft(np.fft.rfft(x, sz) * np.fft.rfft(y, sz), sz)[:m]

def _score(a):
    c = _fft_conv(a, a)
    return 2.0 * a.size * c.max()

def _grad_softmax(a, tau):

    c = _fft_conv(a, a)
    z = c / tau
