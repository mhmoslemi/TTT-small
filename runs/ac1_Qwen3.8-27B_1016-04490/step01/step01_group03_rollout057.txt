import numpy as np
import time
import math

try:
    import scipy.optimize as _sopt
except Exception:
    _sopt = None

def _pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()

def _conv(a):
    n = a.size
    L = 2 * n - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    return np.clip(c.real, 0.0, None)

def _norm(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return None
    p = a / s
    return p if np.isfinite(p).all() else None

def _sf(p):
    if p is None:
        return float('inf')
    c = _conv(p)
    return float(2.0 * p.size * c.max())

def _update(a, best, bs):
    p = _norm(a)
    if p is None:
        return best, bs
    s = float(evaluate_sequence(p.tolist()))
    if np.isfinite(s) and s < bs:
        return p.copy(), s
    return best, bs

def _obj(z, beta, n, N1):
    zm = z.max()
    e = np.exp(z - zm)
    p = e / e.sum()
    c = _conv(p)
    u = float(n) * c
    um = float(u.max())
    ex = np.exp(beta * (u - um))
    s = float(ex.sum())
    if s < 1e-300:
        return 1e10
    return float((beta * um + math.log(s)) / beta)

def _grad(z, beta, n, N1, N2):
    zm = z.max()
    e = np.exp(z - zm)
    p = e / e.sum()
    c = _conv(p)
    u = float(n) * c
    um = float(u.max())
    ex = np.exp(beta * (u - um))
    s = float(ex.sum())
    if s < 1e-300:
        return np.zeros(n)
    w = ex / s
    fp = np.fft.rfft(p, N2)
    fw = np.fft.rfft(w, N2)
    A = np.fft.irfft(np.conj(fp) * fw, N2)[:n].real
    B = np.fft.irfft(fp * fw, N2)[:n].real
    gp = beta * float(n) * (A + B)
    dot = float(p @ gp)
    gz = p * (gp - dot)
    if not np.isfinite(gz).all():
        return np.zeros(n)
    return gz

def _lbfgs_opt(z0, n, N1, N2, deadline, max_per_beta):
    z = z0.copy()
    z -= z.mean()
    if np.abs(z).max() > 50.0:
        z *= 50.0 / np.abs(z).max()
    betas = [16, 32, 64, 128, 256, 512, 1024, 2048, 4096]
    for bi, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        mit = min(max_per_beta, 50 + 30 * (bi + 1))
        if _sopt is not None:
            def _f(z_):
                return _obj(z_, beta, n, N1)
            def _g(z_):
                return _grad(z_, beta, n, N1, N2)
            try:
                res = _sopt.minimize(_f, z, jac=_g, method='L-BFGS-B',
                    options={'maxiter': mit, 'ftol': 1e-14, 'gtol': 1e-9, 'maxls': 40})
                if res.x is not None and res.x.size == n and np.isfinite(res.fun):
                    z = res.x.astype(np.float64)
            except Exception:
                pass
        else:
            lr = 1.0 / (beta * n + 1.0)
            for it in range(mit):
                if time.monotonic() > deadline:
                    break
                g = _grad(z, beta, n, N1, N2)
                z = z - lr * g
                z -= z.mean()
                if np.abs(z).max() > 50.0:
                    z *= 50.0 / np.abs(z).max()
        z -= z.mean()
    return z

def _candidates(n, parent):
    j = np.arange(n, dtype=np.float64)
    out = []
    if parent is not None:
        out.append(parent.copy())
    out.append(np.ones(n) / n)
    for p in (-0.4, -0.2, -0.1, 0.1, 0.2, 0.5, 1.0):
        a = (j + 1.0) ** p
        pn = _norm(a)
        if pn is not None:
            out.append(pn)
    for c in (0.2, 0.5, 1.0, 2.0):
        a = np.exp(c * (j / n - 0.5))
        pn = _norm(a)
        if pn is not None:

    for mu, sig in [(0.3, 0.25), (0.5, 0.25), (0.7, 0.25), (0.4, 0.35), (0.6, 0.35)]:
        a = np.exp(-((x - mu) ** 2) / (2.0 * sig * sig))
        pn = _norm(a)
        if pn is not None:

def _polish(p, best, bs, deadline, rng):
    p = p.copy()
    cur = _sf(p)
    eta = 0.02
    kern = np.ones(7) / 7.0
    while time.monotonic() < deadline and eta > 1e-5:
        noise = rng.normal(0.0, eta, p.size)
        noise = np.convolve(noise, kern, mode='same')
        cand = p * np.exp(noise)
        pn = _norm(cand)
        if pn is None:
            eta *= 0.8
            continue
        sc = _sf(pn)
        if np.isfinite(sc) and sc < cur - 1e-12:
            best, bs = _update(pn, best, bs)
            p = pn
            cur = float(bs)
            eta = min(0.15, eta * 1.08)
        else:
            eta *= 0.85
    best, bs = _update(p, best, bs)
    return best, bs

    t0 = time.monotonic()
    b = float(budget_s) if budget_s > 0 else 1100.0

    dl = t0 + b - 45.0

    except Exception:
        pass
    if parent is None:
        parent = np.ones(n) / n

    best = parent.copy()
    bs = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(bs):
        best = np.ones(n) / n
        bs = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(bs):
        bs = 2.0

    N1 = _pow2(2 * n - 1)
    N2 = _pow2(3 * n - 2)

    cands = _candidates(n, parent)
    scored = []
    for c in cands:
        sc = _sf(c)
        scored.append((sc if np.isfinite(sc) else 1e10, c))
    scored.sort(key=lambda x: x[0])

    starts = []
    for sc, c in scored[:6]:
        dup = False
        for s in starts:
            if np.allclose(s, c, rtol=0.01, atol=1e-4):
                dup = True
                break
        if not dup:
            starts.append(c)
        if len(starts) >= 4:
            break

    max_pb = 300 if n <= 8000 else 200
    for si, s in enumerate(starts):
        if time.monotonic() > dl:
            break
        z0 = np.log(s + 1e-300)
        z0 -= z0.mean()
        opt_dl = min(dl, wall_end - 60.0)
        if time.monotonic() > opt_dl:
            break
        z_opt = _lbfgs_opt(z0, n, N1, N2, opt_dl, max_pb)
        e = np.exp(z_opt - z_opt.max())
        p_new = e / e.sum()
        pn = _norm(p_new)
        if pn is not None:
            best, bs = _update(pn, best, bs)

    if time.monotonic() < wall_end - 20.0:
        best, bs = _polish(best, best, bs, wall_end, rng)

    best, bs = _update(best, best, bs)
    if best is None or best.size == 0 or not np.isfinite(best).all():
        best = np.ones(n) / n
    return [float(x) for x in best]
