import numpy as np
import time


def _npo2(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def _norm(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s < 1e-300:
        a = np.ones_like(a)
        s = float(a.size)
    return a / s


def _fft_conv(a):
    n = a.size
    flen = _npo2(2 * n - 1)
    af = np.fft.rfft(a, flen)
    c = np.fft.irfft(af * af, flen)
    return c[:2 * n - 1]


def _wgrad(a, w, gflen):
    n = a.size
    r = a[::-1]
    out = np.fft.irfft(np.fft.rfft(w, gflen) * np.fft.rfft(r, gflen), gflen)
    return out[n - 1:2 * n - 1]


def _score(a):
    n = a.size
    c = _fft_conv(a)
    M = float(np.max(c))
    s = float(np.sum(a))
    if not np.isfinite(M) or s < 0.01:
        return np.inf, c, M
    return 2.0 * n * M / (s * s), c, M


def _softmax(conv, tau, M):
    if tau <= 1e-14:
        tol = 1e-12 * max(1.0, M)
        w = (conv >= M - tol).astype(np.float64)
    else:

        w = np.zeros(conv.size, dtype=np.float64)
        w[int(np.argmax(conv))] = 1.0

    best_sc = scored[0][0]

    # best start for main run
    a = scored[0][1].copy()
    pool = [s[1].copy() for s in scored[:6]]
    try:
        best_a, best_sc = _run(a, n, flen, gflen, 0.1, target_M, rng,
                               deadline * 0.62 + start, best_a, best_sc, pool)
    except Exception:
        pass

    # polish the global best with a fresh focus run on remaining time
    deadline2 = start + max(1.0, float(budget_s) - 25.0)
    if best_sc > target_b:
        try:
            rng2 = np.random.default_rng(seed + 7)
            pool2 = [best_a.copy()] + pool[-3:]
            best_a, best_sc = _run(best_a, n, flen, gflen, 0.05, target_M, rng2,
                                   deadline2, best_a, best_sc, pool2)
        except Exception:
            pass

    # final exact check over a few candidates
    cands = [best_a]
    if parent is not None and parent.size == n and float(np.sum(parent)) > 0.0:
        cands.append(parent)
    for s in scored[:3]:
        if s[1] is not best_a:
            cands.append(s[1])
    final_a, final_sc = best_a, best_sc
    use_exact = 0 < n <= 20000
    for arr in cands:
        lst = _finish(arr, use_exact)
        s = np.inf
        if use_exact:
            try:
                s = float(evaluate_sequence(lst))
            except Exception:
                s = np.inf
        if not np.isfinite(s):
            s = _score(_norm(arr))[0]
        if np.isfinite(s) and s < final_sc:
            final_sc = s
            final_a = _norm(arr)
    return _finish(final_a, use_exact)
