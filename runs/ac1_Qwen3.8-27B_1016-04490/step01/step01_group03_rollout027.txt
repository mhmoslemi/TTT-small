def lp_step(f, M, n):
    # Maximize sum(g) subject to g >= 0 and conv(f, g)(k) <= M for all k
    # Build constraint matrix A (sparse)
    # A has shape (2n-1, n)

    # Row k has nonzeros at columns j where 0 <= k-j < n, i.e., max(0, k-n+1) <= j <= min(n-1, k)
    
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        j_start = max(0, k-n+1)
        j_end = min(n-1, k)
        for j in range(j_start, j_end+1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k-j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # linprog minimizes, so negate objective
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=M*np.ones(2*n-1), bounds=[(0, None)]*n, method='highs')
    
    if res.success:

    return None

    #   rows j to min(j+n-1, 2n-2) are filled with f[0] to f[j+n-1-j] = f[n-1]
    #   Actually rows j to j+n-1 (clipped at 2n-2)

    # For j=0: min(n, 2n-1) = n
    # For j=n-1: min(n, n) = n
    # For j=n: min(n, n-1) = n-1
    # ...

    # For j <= n-1: 2n-1-j >= n, so nonzeros = n
    # Wait, 2n-1-j for j=n-1 is n, for j=0 is 2n-1.
    # So all columns have exactly n nonzeros (since 2n-1-j >= n for j <= n-1).

def _lp_step(f, M):
    """Solve LP to find g maximizing sum(g) s.t. g>=0, conv(f,g)<="
    n = f.size
    # Build sparse matrix A (2n-1) x n where A[k,j] = f[k-j] if 0<=k-j<n
    cols = np.repeat(np.arange(n), n)
    rows = np.arange(n) + cols
    vals = np.tile(f, n)
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    c = -np.ones(n)
    b = M + 1e-12
    res = linprog(c, A_ub=A, b_ub=b*np.ones(2*n-1), bounds=[(0,None)]*n, method='highs',
                  options={'maxiter': 5000})
    if res.success and res.x is not None and np.isfinite(res.x).all():
        return np.clip(res.x, 0.0, None)
    return None
