```python
def _fft_conv(a, b):
    n = len(a)
    m = len(b)
    L = 1
    while L < n + m - 1:
        L <<= 1
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    return np.fft.irfft(fa * fb, L)[:n + m - 1]

def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

```python
def _smooth_gradient(p, a, n):
    """Compute smooth gradient of softmax max of auto-correlation."""
    a_flip = a[::-1]
    L = 1
    while L < 2*n - 1:
        L <<= 1
    fp = np.fft.rfft(p, L)
    fa = np.fft.rfft(a, L)
    fa_flip = np.fft.rfft(a_flip, L)
    conv_pa = np.fft.irfft(fp * fa, L)[:2*n-1]
    conv_paf = np.fft.irfft(fp * fa_flip, L)[:2*n-1]
    v = conv_paf[n-1:n-1+n] + conv_pa[:n]
    return v
```

```python
def _project_simplex(v):
    n = len(v)
    s = np.sum(v)
    if s <= 0 or not np.isfinite(s):
```

```python
import numpy as np
import time

def _fft_conv2(a, b):
    """Linear convolution of a and b via FFT. O(n log n)."""
    na, nb = len(a), len(b)
    L = 1
    while L < na + nb - 1:
        L <<= 1
    return np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)[:na + nb - 1]

def _proj_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    if not np.all(np.isfinite(v)) or np.sum(v) <= 0:
        return np.ones(n) / n
    us = np.sort(v)[::-1]
    cu = np.cumsum(us)
    rho = np.nonzero(us * np.arange(1, n + 1) > (cu - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    theta = (cu[rho[-1]] - 1.0) / (rho[-1] + 1.0)
    return np.maximum(v - theta, 0.0)

def _smooth_grad(p, a, n):
    """
    Smooth gradient of softmax_max_k c_k where c = a*a.
    p: softmax weights over c, length 2n-1
    a: input, length n
    Returns gradient vector of length n.
    v_j = sum_k p_k (a_{k-j} + a_{j-k})
       = (p * a_rev)[n-1+j] + (p * a)[j]
    """

    L = 1
    while L < 3 * n - 2:
        L <<= 1
    fp = np.fft.rfft(p, L)
    out1 = np.fft.irfft(fp * np.fft.rfft(a_rev, L), L)[:3 * n - 2]
    out2 = np.fft.irfft(fp * np.fft.rfft(a, L), L)[:3 * n - 2]
    return out1[n - 1:2 * n - 1] + out2[:n]
```
