def _fft_conv(a, N):

    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:2 * a.size - 1]

    """Compute the AC1 score."""

    s = a.sum()
    if s > 1e-30:
        a = a * (target / s)
    return np.clip(a, 0.0, 1000.0)

    g_i ∝ (S * contrib_i / M - 2) where contrib_i = sum_k (b_k/sum(b_top)) * a_{k-i}

def _fft_conv(a, N):

    return np.fft.irfft(A * A, N)[:2 * a.size - 1]

def _grad_mult(a, b, n, topk):

    if M < 1e-30 or S < 1e-30:
        return np.zeros(n)
    idx = np.argpartition(b, -topk)[-topk:]
