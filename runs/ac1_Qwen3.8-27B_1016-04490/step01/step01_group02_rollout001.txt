        ss = a.sum()
        if ss < 1e-10:
            a[:] = 1.0
            a /= a.sum()
        else:
            a /= ss

        it += 1
        if it % 10 == 0:
            tau *= 0.997
            if tau < 1e-5 * tau_init:
                tau = 1e-5 * tau_init
            lr *= 0.9995

        if it % 40 == 0:
            sc = float(evaluate_sequence(a.tolist()))
            if sc < best_score - 1e-12:
                best_score = sc
                best = a.copy()
                stalled = 0
                best_tau = tau
                best_lr = lr
            else:
                stalled += 1
            if stalled >= 6:
                sc = 0.003 * rng.standard_normal(n)
                a = np.maximum(a + sc, 0.0)
                a /= a.sum()
                tau = best_tau * 2.0
                lr = best_lr * 0.5
                stalled = 0

    sc = float(evaluate_sequence(a.tolist()))
    if sc < best_score:
        best = a.copy()
        best_score = sc
    return best, best_score


def _local_refine(a, deadline, rng):
    n = a.size
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    evals = 0
    max_evals = 300

    for it in range(60):
        if time.monotonic() > deadline or evals >= max_evals:
            break
        for _ in range(10):
            if time.monotonic() > deadline or evals >= max_evals:
                break
            i, j = rng.choice(n, size=2, replace=False)
            total = best[i] + best[j]
            if total < 1e-15:
                continue
            improved = False
            for frac in (0.3, 0.35, 0.4, 0.45, 0.55, 0.6, 0.65, 0.7):
                cand = best.copy()
                cand[i] = total * frac
                cand[j] = total * (1.0 - frac)
                evals += 1
                sc = float(evaluate_sequence(cand.tolist()))
                if sc < best_score - 1e-12:
                    best_score = sc
                    best = cand.copy()
                    improved = True
                    break
            if not improved:
                noise = rng.standard_normal(n) * 0.0005 / np.sqrt(n)
                cand = np.maximum(best + noise, 0.0)
                cand /= cand.sum()
                evals += 1
                sc = float(evaluate_sequence(cand.tolist()))
                if sc < best_score - 1e-12:
                    best_score = sc
                    best = cand.copy()
    return best, best_score


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 10.0)

    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.ndim != 1 or parent.size < 2 or not np.isfinite(parent).all():
        parent = np.ones(2535, dtype=np.float64)
    n = parent.size
    parent = np.maximum(parent, 0.0)
    ps = parent.sum()
    if ps < 1e-10:
        parent[:] = 1.0
    else:
        parent /= ps

    parent_score = float(evaluate_sequence(parent.tolist()))
    best_all = parent.copy()
    best_sc = parent_score

    j = np.arange(n, dtype=np.float64)
    x = j / n
    inits = [
        parent.copy(),
        np.sin(np.pi * (j + 0.5) / n) ** 2 + 1e-4,
        np.exp(-15.0 * (x - 0.3) ** 2) + np.exp(-15.0 * (x - 0.7) ** 2) + 1e-3,
        np.abs(rng.standard_normal(n)) ** 0.6 + 0.01,
        (1.0 + 5.0 * x) ** (-0.35) + 0.01,
        parent.copy() * np.exp(-3.0 * (x - 0.5) ** 2) + 0.005,
    ]

    n_inits = len(inits)
    time_per = max(5.0, (deadline - time.monotonic()) * 0.75 / n_inits)

    tau_init = 0.4 / n
    lr_init = 2.5

    for init in inits:
        if time.monotonic() >= deadline:
            break
        ia = np.maximum(init, 0.0).astype(np.float64)
        ss = ia.sum()
        if ss < 1e-10:
            ia[:] = 1.0
        else:
            ia /= ss
        dl = min(time.monotonic() + time_per, deadline)
        topk = 25 if ia is not inits[0] else 0
        b, sc = _run_opt(ia, dl, rng, tau_init, lr_init, topk)
        if sc < best_sc:
            best_sc = sc
            best_all = b.copy()

    # Second pass: refine best with smaller tau
    if time.monotonic() < deadline - 30.0:
        dl2 = min(time.monotonic() + 80.0, deadline - 30.0)
        b2, sc2 = _run_opt(best_all.copy(), dl2, rng, 0.1 / n, 1.0, 15)
        if sc2 < best_sc:
            best_sc = sc2
            best_all = b2

    # Local refinement
    if time.monotonic() < deadline - 15.0:
        dl3 = min(time.monotonic() + 60.0, deadline - 15.0)
        best_all, best_sc = _local_refine(best_all, dl3, rng)

    # Final validation
    final_sc = float(evaluate_sequence(best_all.tolist()))
    if not np.isfinite(final_sc) or final_sc > parent_score:
        best_all = parent
        best_sc = parent_score

    return [float(v) for v in best_all]
