- Optimize only those K constraints (ignoring the others)

- Check if the solution is valid (other entries are below the max)
- If not, add the violated constraint and repeat

import numpy as np
import time
import scipy.optimize as sopt


def _softmax(z):
    z = np.asarray(z, dtype=np.float64).copy()
    z -= z.max()
    e = np.exp(z)
    s = e.sum()
    if s <= 0.0 or not np.isfinite(s):
        return np.ones(z.shape, dtype=np.float64) / z.size
    return e / s


def _fast_conv(a):
    n = a.size
    L = 2 * n - 1
    N = 1 << max(0, (L - 1).bit_length())
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:L]


def _exact_score(f):
    f = np.clip(np.asarray(f, dtype=np.float64), 0.0, 1000.0)
    n = f.size
    s = f.sum()
    if s < 0.01:
        return float('inf')
    c = _fast_conv(f)
    return float(2.0 * n * c.max() / (s * s))


def _obj(z, beta):
    f = _softmax(z)
    c = _fast_conv(f)
    u = float(f.size) * c
    um = u.max()
    v = np.exp(beta * (u - um))
    s = v.sum()
    if s <= 0.0 or not np.isfinite(s):
        return 1e100
    return float((beta * um + np.log(s)) / beta)


def _grad(z, beta):
    f = _softmax(z)
    n = f.size
    c = _fast_conv(f)
    u = n * c
    um = u.max()
    w = np.exp(beta * (u - um))
    ws = w.sum()
    if ws <= 0.0 or not np.isfinite(ws):
        return np.zeros(n, dtype=np.float64)
    w /= ws
    N = 1 << max(0, (3 * n - 2).bit_length())
    Ff = np.fft.rfft(f, N)
    Fw = np.fft.rfft(w, N)
    corr = np.fft.irfft(np.conj(Ff) * Fw, N)
    gf = 2.0 * corr[:n]
    dot = float(f @ gf)
    gz = f * (gf - dot)
    if not np.isfinite(gz).all():
        return np.zeros(n, dtype=np.float64)
    return gz


_DEADLINE = 0.0


class _TO(Exception):
    pass


def _cb(_):
    if time.monotonic() > _DEADLINE:
        raise _TO


def _run_lbfgs(z0, t_end, n):
    global _DEADLINE
    _DEADLINE = t_end
    z = z0.copy()
    z -= z.mean()
    best_z = z.copy()
    best_v = _obj(z, 8.0)
    phases = [(8, 80), (16, 100), (32, 120), (64, 150), (128, 150), (256, 200), (512, 200), (1024, 200), (2048, 200), (4096, 200)]
    for beta, mx in phases:
        if time.monotonic() > t_end - 12.0:
            break
        zv = z - z.mean()
        am = zv.max()
        if am > 40.0 and am > 0.0:
            zv *= 40.0 / am
        z = zv
        try:
            res = sopt.minimize(
                lambda zz: _obj(zz, beta),
                z,
                jac=lambda zz: _grad(zz, beta),
                method='L-BFGS-B',
                options={'maxiter': mx, 'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 25},
                callback=_cb
            )
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64)
                if x.size == n and np.isfinite(x).all() and np.isfinite(res.fun):
                    vn = _obj(x, beta)
                    if vn <= best_v + 1e-9:
                        z = x
                        best_v = vn
                        best_z = x.copy()
        except _TO:
            break
        except Exception:
            pass
    return best_z


def _make_cands(n, parent):
    out = [parent.copy()]
    j = np.arange(n, dtype=np.float64)
    for p in (-0.7, -0.5, -0.35, -0.2, -0.1, 0.0, 0.1, 0.3, 0.6):
        a = (j + 1.0) ** p
        a = np.clip(a, 1e-8, None)
        out.append(a / a.sum())
    for fr in (0.25, 0.35, 0.5, 0.65, 0.75):
        m = int(fr * n)
        for r in (2.0, 4.0, 8.0, 20.0):
            a = np.ones(n)
            a[m:] = r
            out.append(a / a.sum())
            b = np.ones(n)
            b[:m] = r
            out.append(b / b.sum())
    for m1, m2 in ((int(0.3*n), int(0.6*n)), (int(0.4*n), int(0.7*n)), (int(0.5*n), int(0.8*n))):
        a = np.ones(n)
        a[m1:m2] = 3.0
        out.append(a / a.sum())
        b = np.linspace(0.3, 1.0, n)
        b[:m1] *= 1.5
        b[m2:] *= 0.5
        out.append(b / b.sum())
    base = (j + 1.0) ** (-0.3)
    for amp in (0.15, 0.3, 0.5):
        for k in (1, 2, 3):
            t = base * (1.0 + amp * np.sin(2.0 * np.pi * k * j / n))
            t = np.clip(t, 1e-6, None)
            out.append(t / t.sum())

def _polish(f, t_end, rng, n):
    best = f.copy()
    bs = _exact_score(best)
    eta = 0.004
    ker = np.array([1.0, 2.0, 3.0, 2.0, 1.0]) / 9.0
    while time.monotonic() < t_end - 18.0 and eta > 1e-8:
        noise = rng.normal(0.0, eta, n)
        noise = np.convolve(noise, ker, mode='same')
        cand = best * (1.0 + noise)
        cand = np.clip(cand, 0.0, 1000.0)
        cs = cand.sum()
        if cs > 0.01:
            cand /= cs
            s = _exact_score(cand)
            if np.isfinite(s) and s < bs:
                bs = s
                best = cand.copy()
                eta = min(0.015, eta * 1.06)
            else:
                eta *= 0.82
        else:
            eta *= 0.7
    return best, bs
