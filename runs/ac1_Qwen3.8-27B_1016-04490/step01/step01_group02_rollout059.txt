        if sc2_eval < best_sc - 1e-10:
            best_sc = sc2_eval
            best = b2.copy()

    bv = np.maximum(best, 0.0).astype(np.float64)
    bv = np.clip(bv, 0.0, 1000.0)
    if bv.sum() < 0.01:
        bv = np.full(n, 0.01 / n)
    return [float(v) for v in bv]
