import numpy as np
import time

def _fft_conv(a, b, N):
    """Linear convolution via FFT."""
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    return np.fft.irfft(A * B, N)

def _score_fast(a, n, N):
    b = _fft_conv(a, a, N)
    bmax = b[:2*n-1].max()
    S = a.sum()
    if S < 0.01:
        return float('inf')
    return 2.0 * n * bmax / (S * S)

def _full_gradient(a, b, n, N, tau, p):
    """Compute gradient using weighted peaks."""
    bmax = b[:2*n-1].max()
    if bmax <= 0:
        return np.zeros(n)

    ratio = b[:2*n-1] / bmax
    lam = ratio ** p

    g1_full = _fft_conv(lam, a_flip, N)
    g1 = g1_full[n-1 : 2*n-1]
    g = g1[:n] + g2[:n]
    return g
