def _fft_conv(x, y):
    m = len(x) + len(y) - 1
    sz = 1
    while sz < m:
        sz <<= 1
    fx = np.fft.rfft(x, sz)
    fy = np.fft.rfft(y, sz)
    return np.fft.irfft(fx * fy, sz)[:m]


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _score(a):

        z -= z.max()
