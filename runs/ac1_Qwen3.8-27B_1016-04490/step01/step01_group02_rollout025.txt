    s = a.sum()
    if s < 1e-10:
        return 1e30
    return 2.0 * n * float(b.max()) / (s * s)

def _softmax_grad(a, tau):
    n = a.size
    b = _fft_conv(a, a)
    z = b / tau
    z -= z.max()
    e = np.exp(z)
    p = e / e.sum()
    c = p[::-1]
    ca = _fft_conv(c, a)
    return 2.0 * ca[n-1:2*n-1][::-1]

def _optimize(a0, deadline, rng):
    n = a0.size
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        a[:] = 1.0
    else:
        a /= s
    
    best = a.copy()
    best_sc = _fast_score(a)
    
    tau = 2.0 / n
    tau_min = 1.0 / (n * 10)
    lr = 1.0
    it = 0
    stall = 0
    eval_every = 200
    
    while time.monotonic() < deadline:
        g = _softmax_grad(a, tau)
        # Clip to avoid overflow/underflow
        step = np.clip(lr * g, -5.0, 5.0)
        a = a * np.exp(-step)
        a = np.maximum(a, 1e-30)
        a /= a.sum()
        it += 1
        
        # Decay schedule
        if it % 50 == 0:
            tau *= 0.97
            if tau < tau_min:
                tau = tau_min
            lr *= 0.9995
        
        # Periodic eval
        if it % eval_every == 0:
            sc = _fast_score(a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best = a.copy()
                stall = 0
            else:
                stall += 1
            
            if stall >= 8:
                # Perturb and reheat
                a = a + 0.005 * rng.standard_normal(n)
                a = np.maximum(a, 0.0)
                a /= a.sum()
                tau = 2.0 / n
                lr = 1.0
                stall = 0
    
    sc = _fast_score(a)
    if sc < best_sc:
        best = a.copy()
        best_sc = sc
    return best, best_sc
