def _next_pow2(x):

    return np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)[:n]

    """cc[m] = sum_i a[i]*w[i+m], for m=0 to len(w)-1"""
    L = _next_pow2(len(a) + len(w) - 1)
    return np.fft.irfft(np.conj(np.fft.rfft(a, L)) * np.fft.rfft(w, L), L)[:len(w)]


def _score(a, n):
    b = _fft_conv(a, a)
    return 2.0 * n * b.max()


def _make_inits(n):

    mid = (n - 1) / 2.0
    inits = []
    inits.append(np.ones(n))

    inits.append(bm)
    # Cosine
    inits.append(np.cos(np.pi * i / (2 * n)) ** 2)
    # Inverse
    inits.append(1.0 / (1.0 + d / (n * 0.1)))
    return inits
