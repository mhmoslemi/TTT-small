def _next_pow2(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def _autocorr(a):
    """Compute linear autocorrelation of a via FFT."""
    n = len(a)
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, L)
    return np.fft.irfft(fa * fa)[:2 * n - 1]


def _grad_autocorr(a, w):
    """Gradient of sum_m w[m]*c[m] where c=autocorr(a). Returns vector of length n."""

import numpy as np
import time


def _next_pow2(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def _autocorr(a):
    n = len(a)
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, L)
    return np.fft.irfft(fa * fa)[:2 * n - 1]


def _grad_softmax(a, w):
    """2 * cross_correlation(w, a) computed via FFT."""
    n = len(a)
    L = _next_pow2(3 * n - 2)
    fa = np.fft.rfft(a, L)
    fw = np.fft.rfft(w, L)
    return 2.0 * np.fft.irfft(fw * np.conj(fa), L)[:n]


def _simplex_proj(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    v = np.asarray(v, dtype=np.float64).copy()
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-14:
        return np.full(v.shape, target_sum / v.size)
    return v * (target_sum / s)
