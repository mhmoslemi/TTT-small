    """Full linear convolution of p with itself via FFT."""

    c = np.fft.irfft(fp * fp, N, N)[:L]
    return np.maximum(c, 0.0)

    """Score = 2*n*max(conv)/sum^2, for normalized p (sum=1): 2*n*max(conv)."""
    c = _fft_conv(p)
    return 2.0 * p.size * np.max(c)

def _lso_objective(z, beta, Nfft):
    """Log-sum-exp of beta * (n * conv(softmax(z), softmax(z)))."""

    e = np.exp(z)
    p = e / e.sum()
    c = _fft_conv(p)
    u = n * beta * c
    umax = np.max(u)
    s = np.sum(np.exp(u - umax))
    if s <= 0: return 1e100

    for frac in (1.0, 0.85, 0.7, 0.5, 0.35):
        w = max(10, int(frac * n))
        off = (n - w) // 2
        x = np.arange(w, dtype=np.float64)
        prof = np.sin(np.pi * (x + 0.5) / w)
        full = np.zeros(n, dtype=np.float64)
        full[off:off + w] = prof
        if full.sum() > 0:
            cands.append(full / full.sum())
    for sf in (0.12, 0.18, 0.25, 0.35, 0.5):
        sig = sf * n
        prof = np.exp(-0.5 * ((j - 0.5 * n) / sig) ** 2)
        if prof.sum() > 0:
            cands.append(prof / prof.sum())
    for alpha in (0.5, 1.0, 1.5, 2.0, 3.0):
        x = 2.0 * j / n - 1.0
        prof = np.where(np.abs(x) < 1.0, (1.0 - x * x) ** alpha, 0.0)
        if prof.sum() > 0.01:
            cands.append(prof / prof.sum())
    for asym in (0.35, 0.65):
        ctr = asym * n
        sig = 0.12 * n
        prof = np.exp(-0.5 * ((j - ctr) / sig) ** 2)
        if prof.sum() > 0.01:
            cands.append(prof / prof.sum())
    return cands


def _polish(p, deadline, rng):
    n = p.size
    best_p = p.copy()
    best_sc = _score(p)
    eta = 0.005
    for _ in range(2000):
        if time.monotonic() > deadline - 10:
            break
        idx = rng.integers(0, n, size=15)
        p_try = best_p.copy()
        p_try[idx] *= np.exp(rng.normal(0, eta, 15))
        p_try = np.maximum(p_try, 0)
        s = p_try.sum()
        if s < 1e-6:
            eta *= 0.7
            continue
        p_try /= s
        sc = _score(p_try)
        if np.isfinite(sc) and sc < best_sc - 1e-12:
            best_sc = sc
            best_p = p_try.copy()
            eta = min(0.15, eta * 1.1)
        else:
            eta *= 0.85
        if eta < 1e-6:
            break
    return best_p, best_sc
