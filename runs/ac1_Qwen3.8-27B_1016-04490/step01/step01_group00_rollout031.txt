These are equal iff x[i] = x[n-1-i] for all i (i.e., x is palindromic).

So for a general non-palindromic x, the convolution is NOT symmetric. This means we can't exploit symmetry.

import numpy as np
import time
from scipy import fft as sfft


def _npow2(m):
    p = 1
    while p < m:
        p *= 2
    return p


def _norm(x, ts, n):
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s < 0.01:
        x = np.full(n, 0.01)
        s = x.sum()
    return x * (ts / s)


def _sc(x):
    return float(evaluate_sequence(x.tolist()))


def _fft_conv(x, fs, L):
    X = sfft.rfft(x, fs)
    c = np.real(sfft.irfft(X * X, fs))[:L]
    return np.maximum(c, 0.0)


def _grad_step(x, T, lr, fs, L, n):
    conv = _fft_conv(x, fs, L)
    cm = conv.max()
    if cm < 1e-30:
        return x
    p = np.exp((conv - cm) / T)
    ps = p.sum()
    if ps < 1e-300:
        p = np.zeros(L)
        p[np.argmax(conv)] = 1.0
    else:
        p /= ps
    P = sfft.rfft(p, fs)
    X = sfft.rfft(x, fs)
    c = np.real(sfft.irfft(P * np.conj(X), fs))[:n]
    g = 2.0 * c
    g -= g.mean()
    return x - lr * g


def _optimize(x, T0, lr0, miters, efreq, fs, L, n, ts, dl):
    T = T0
    lr = lr0
    xb = x.copy()
    sb = _sc(x)
    stall = 0
    for it in range(miters):
        if time.monotonic() > dl - 5.0:
            break
        x = _grad_step(x, T, lr, fs, L, n)
        x = _norm(x, ts, n)
        T = max(0.002, T * 0.9996)
        lr = max(0.02, lr * 0.99995)
        if (it + 1) % efreq == 0:
            s = _sc(x)
            if np.isfinite(s) and s < sb - 1e-12:
                sb = s
                xb = x.copy()
                stall = 0
            else:
                stall += efreq
            if stall > efreq * 50:
                break
    return xb, sb
