import numpy as np
import time


def npt(x):
    L = 1
    while L < x:
        L <<= 1
    return L


def fft_conv(a, b, sz):
    fa = np.fft.rfft(a, sz)
    fb = np.fft.rfft(b, sz)
    return np.fft.irfft(fa * fb, sz)


def autocorr(a, sz):
    f = np.fft.rfft(a, sz)
    return np.fft.irfft(f * f, sz)


def compute_score(a):
    n = a.size
    sz = npt(2 * n - 1)
    c = autocorr(a, sz)[:2 * n - 1]
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    return 2.0 * n * float(c.max()) / (s * s)


def compute_grad(a, sz1, sz2, tau):
    n = a.size
    c = autocorr(a, sz1)[:2 * n - 1]
    M = float(c.max())
    if M <= 1e-18:
        return np.zeros(n), 0.0
    z = np.clip((c - M) / tau, -50.0, 0.0)
    e = np.exp(z)
    w = e / e.sum()
    ar = a[n - 1::-1]
    corr = fft_conv(ar, w, sz2)
    g = 2.0 * corr[n - 1:2 * n - 1]
    g = np.where(np.isfinite(g), g, 0.0)
    return g, M
