    return np.maximum(conv, 0.0)

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 12.0)

    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(512, dtype=np.float64)
    n = parent.size
    target_sum = np.sqrt(2.0 * n)
    fft_size = next_pow2(2 * n)
    corr_size = next_pow2(3 * n)

    x = normalize_seq(parent.copy(), target_sum, n)
    best = x.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(best_score):
        x = np.full(n, target_sum / n)
        best = x.copy()
        best_score = float(evaluate_sequence(best.tolist()))

    v = np.zeros(n)
    beta = 0.82
    T = max(0.5, best_score * 0.3)
    lr = 1.2
    iters_since_improve = 0
    eval_int = 150
    K_top = max(4, n // 500)
    use_topk = (iteration_phase := 0) == 0  # always True initially, swap later

    iteration = 0
    hard_limit = 800000

    while iteration < hard_limit:
        if time.monotonic() > deadline - 6.0:
            break

        conv = compute_conv(x, fft_size, n)

        # Hybrid gradient: soft + top-K
        g_soft = soft_gradient(x, conv, T, n, corr_size)
        g_top = topk_gradient(x, conv, K_top, n, corr_size)
        g = 0.6 * g_soft + 0.4 * g_top

        # Nesterov momentum
        v = beta * v + (1.0 - beta) * g
        x = x - lr * v
        x = normalize_seq(x, target_sum, n)

        # Anneal temperature and lr
        if iteration < 30000:
            T = max(0.02, T0 * (0.02 / T0) ** (iteration / 30000))
        else:
            T = max(0.005, T * 0.9997)
        lr = max(0.03, lr * (0.99992 if iteration > 15000 else 0.99999))

        # Periodic exact evaluation
        if iteration % eval_int == 0:
            sc = float(evaluate_sequence(x.tolist()))
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best = x.copy()
                iters_since_improve = 0
            else:
                iters_since_improve += eval_int

            # Restart on stagnation
            if iters_since_improve > 10000:
                nscale = target_sum * 0.02 * (1.0 + 0.5 * min(iters_since_improve / 10000.0, 3.0))
                x = best + nscale * rng.standard_normal(n)
                x = normalize_seq(x, target_sum, n)
                v = np.zeros(n)
                T = max(0.1, best_score * 0.2)
                lr = 1.0
                beta = 0.82
                iters_since_improve = 0

                # Occasionally try a fresh structure
                if iters_since_improve > 30000:
                    j = np.arange(n)
                    fres = np.ones(n)
                    for k in range(1, 6):
                        fres += (0.3 / k) * np.sin(2.0 * np.pi * k * 7.0 * j / n + rng.uniform(0, 6.28))
                    fres = np.maximum(fres, 0.05)
                    fres = normalize_seq(fres, target_sum, n)
                    s2 = float(evaluate_sequence(fres.tolist()))
                    if np.isfinite(s2) and s2 < best_score:
                        best_score = s2
                        best = fres.copy()
                        x = fres.copy()
                    iters_since_improve = 0

        iteration += 1

    # Final polish: very low temp, tiny steps
    if time.monotonic() < deadline - 8.0:
        x = best.copy()
        v = np.zeros(n)
        T = 0.01
        lr = 0.05
        for _ in range(3000):
            if time.monotonic() > deadline - 6.0:
                break
            conv = compute_conv(x, fft_size, n)
            g = soft_gradient(x, conv, T, n, corr_size)
            v = 0.9 * v + 0.1 * g
            x = x - lr * v
            x = normalize_seq(x, target_sum, n)
            sc = float(evaluate_sequence(x.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best = x.copy()

    fc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(fc) or len(best) == 0:
        return [float(v) for v in parent]
    return [float(v) for v in best]
