def _fft_conv(x, y):
    m = len(x) + len(y) - 1
    sz = 1
    while sz < m:
        sz <<= 1
    return np.fft.irfft(np.fft.rfft(x, sz) * np.fft.rfft(y, sz), sz)[:m]


def _grad(a, tau):
    n = a.size
    s = _fft_conv(a, a)
    z = s / tau - s.max() / tau
    e = np.exp(z)
    p = e / e.sum()
    c = p[::-1]
    ca = _fft_conv(c, a)
    return 2.0 * ca[n - 1:2 * n - 1][::-1]
