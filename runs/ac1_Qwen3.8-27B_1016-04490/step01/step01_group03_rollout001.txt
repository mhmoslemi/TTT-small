import time
import math
try:
    import scipy.optimize as sopt
except Exception:
    sopt = None

DL = 0.0
_CB = 1.0

class ST(Exception):
    pass

def tc(_):
    if time.monotonic() > DL:
        raise ST

def norm(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1 or a.size == 0 or not np.isfinite(a).all():
        return None
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01 or not np.isfinite(s):
        return None
    return a / s

def acol(a):
    n = a.size
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    f = np.fft.rfft(a, N)
    return np.clip(np.real(np.fft.irfft(f * f, N)[:L]), 0, None)

def fsc(a):
    p = norm(a)
    if p is None:
        return 1e9
    return float(2 * p.size * np.max(acol(p)))

def smx(z):
    z = z - z.max()
    e = np.exp(z)
    s = float(e.sum())
    return e / s if s > 0 and np.isfinite(s) else np.full_like(z, 1.0 / z.size)

def lob(z):
    p = smx(z)
    u = float(p.size) * acol(p)
    m = float(u.max())
    e = np.exp(_CB * (u - m))
    s = float(e.sum())
    return (_CB * m + math.log(max(s, 1e-300))) / _CB if s > 0 else 1e100

def lgr(z):
    n = z.size
    p = smx(z)
    u = float(n) * acol(p)
    m = float(u.max())
    e = np.exp(_CB * (u - m))
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        return np.zeros(n)
    w = e / s
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fp, fw = np.fft.rfft(p, N), np.fft.rfft(w, N)
    cm = np.real(np.fft.irfft(fp * np.conj(fw), N))
    cv = np.real(np.fft.irfft(fp * fw, N))
    dp = np.empty(n)
    dp[0] = cm[0]
    if n > 1:
        dp[1:] = cm[N - 1:N - n:-1]
    g = (float(n) / _CB) * (dp + cv[:n])
    if not np.isfinite(g).all():
        return np.zeros(n)
    return p * (g - float(np.dot(p, g)))

def run_lp(p0, best, bs, dl):
    n = p0.size
    z = np.log(p0 + 1e-300)
    z -= z.mean()
    for bt, mi in [(16, 40), (32, 40), (64, 35), (128, 30), (256, 30), (512, 25), (1024, 20)]:
        if time.monotonic() > dl - 40:
            break
        _CB = bt
        a = z - z.mean()
        if a.max() > 25:
            a *= 25.0 / a.max()
        try:
            r = sopt.minimize(lob, a, jac=lgr, method='L-BFGS-B',
                              callback=tc, options={'maxiter': mi, 'ftol': 1e-13, 'gtol': 1e-9, 'maxls': 12})
            if r is not None and r.x is not None and np.isfinite(r.fun):
                z = r.x - r.x.mean()
        except Exception:

        if p is not None:

            if np.isfinite(sc) and sc < bs:

    return best, bs

def step_cands(n):
    c = []
    c.append(np.ones(n, dtype=np.float64))
    for i in range(1, 35):
        k = max(1, min(n - 1, int(round(i * n / 35.0))))
        for r in (0.2, 0.4, 0.6, 1.5, 2.5, 5.0, 10.0):
            a = np.ones(n, dtype=np.float64)
            a[:k] = r
            c.append(a)
    for i in range(1, 20):
        k1 = max(1, int(round(i * n / 20.0)))
        k2 = max(k1 + 1, min(n - 1, int(round((i + 3) * n / 20.0))))
        for r1, r2 in [(2.0, 0.5), (1.5, 0.3), (3.0, 0.4), (0.5, 2.0)]:
            a = np.ones(n, dtype=np.float64)
            a[:k1] = r1
            a[k1:k2] = 0.1
            a[k2:] = r2
            c.append(a)
    j = np.arange(n, dtype=np.float64)
    for p in (-1.0, -0.7, -0.5, -0.3, 0.5, 1.0):
        try:
            a = (1.0 + j / n) ** p
            if np.isfinite(a).all() and a.min() > 0:
                c.append(a)
        except Exception:
            pass
    return c

def quant_opt(p, k, dl):
    n = p.size
    srt = np.sort(p)
    edges = np.quantile(srt, np.linspace(0, 1, k + 1))
    q = np.searchsorted(edges[1:-1], p)
    template = np.zeros(n, dtype=np.float64)
    masks = [q == i for i in range(k)]
    vals0 = np.array([p[m].mean() for m in masks if m.any()])
    if vals0.size < 2:
        return None
    def ob(v):
        a = template.copy()
        for i, m in enumerate(masks):
            a[m] = max(0.0, v[i])
        s = float(a.sum())
        if s < 0.01:
            return 1e10
        a /= s
        return float(2.0 * n * np.max(acol(a)))
    try:
        r = sopt.minimize(ob, vals0, method='Nelder-Mead',
                          options={'maxiter': 3000, 'xatol': 1e-12, 'fatol': 1e-12})
        if r is not None and np.isfinite(r.fun):
            a = template.copy()
            for i, m in enumerate(masks):
                a[m] = max(0.0, r.x[i])
            s = float(a.sum())
            if s >= 0.01:
                return a / s
    except Exception:
        pass
    return None

def polish(p, best, bs, dl, rng):
    p = p.copy()
    fb = fsc(p)
    eta = 0.015
    kern = np.ones(7, dtype=np.float64) / 7.0
    while time.monotonic() < dl - 20 and eta > 1e-8:
        nz = rng.normal(0, eta, p.size)
        nz = np.convolve(nz, kern, 'same')
        c = norm(p * np.exp(nz))
        if c is None:
            eta *= 0.8
            continue
        fc = fsc(c)
        if fc < fb - 1e-11:
            p, fb = c, fc
            sc = float(evaluate_sequence(p.tolist()))
            if np.isfinite(sc) and sc < bs:
                best, bs = p.copy(), sc
            eta = min(0.25, eta * 1.06)
        else:
            eta *= 0.84
    return best, bs
