import numpy as np
import time
from scipy.optimize import minimize

def fast_conv(a):
    """Compute full linear convolution of a with itself via FFT."""
    n = len(a)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    return np.maximum(c, 0)  # clip small negatives

def cross_corr(a, b):
    """Cross-correlation (a ⊛ b)[j] = sum_k a[k] * b[k-j] = sum_k a[k+j] * b[k]."""
    n = len(a)
    lb = len(b)
    L = n + lb - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    result = np.fft.irfft(fa * np.conj(fb), N)
    return result[:n]  # indices 0 to n-1

def softmax_obj(z, beta, n):
    """Log-sum-exp objective in z-space."""
    p = np.exp(z - np.max(z))
    s = p.sum()
    p = p / s
    c = fast_conv(p)
    u = n * c
    u_max = u.max()
    e = np.exp(beta * (u - u_max))
    val = (u_max + np.log(e.sum()) / beta)  # /beta is the 1/beta factor
    # Actually: (1/beta) * log(sum exp(beta*u)) = (1/beta)*(beta*u_max + log(sum exp(beta*(u-u_max))))
    # = u_max + log(sum exp(beta*(u-u_max)))/beta
    return val

def softmax_grad(z, beta, n):
    p = np.exp(z - np.max(z))
    s = p.sum()
    p = p / s
    c = fast_conv(p)
    u = n * c
    u_max = u.max()
    e = np.exp(beta * (u - u_max))
    w = e / e.sum()
    
    # grad_p[j] = 2*n * (w ⊛ f)[j] where f = p
    # cross_corr(w, p)[j] = sum_k w[k] * p[k-j]
    # But we need sum_m w[m] * p[m-j] for m-j in [0,n-1]
    # This is cross_corr with w as "left" and p as "right"
    grad_p = 2.0 * n * cross_corr(w, p)
    
    # Convert to z-space
    dot = np.dot(p, grad_p)
    grad_z = p * (grad_p - dot)
    return grad_z

    from scipy.optimize import linprog

    # A[k, j] = p[k-j] if 0 <= k-j <= n-1
    diags_list = []

        s = g.sum()
        if s > 0.01:
            return g / s

import numpy as np

from scipy.optimize import minimize

def _np2(m):
    return 1 << max(0, (m - 1).bit_length())

def _autocorr(p):
    n = p.size
    N = _np2(2 * n - 1)
    fp = np.fft.rfft(p, N)
    c = np.fft.irfft(fp * fp, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

def _xcorr(w, p):
    n = p.size
    N = _np2(2 * n - 1)
    return np.fft.irfft(np.fft.rfft(w, N) * np.conj(np.fft.rfft(p, N)), N)[:n]
