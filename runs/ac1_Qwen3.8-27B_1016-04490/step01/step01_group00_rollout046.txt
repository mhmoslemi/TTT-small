        p = np.exp((conv - cm) / T)
        ps = p.sum()
        if ps < 1e-300:
            np.fill(p, 0.0)
            p[n-1] = 1.0
        else:
            p /= ps
        Pf = sfft.rfft(p, fsz)
        g = 2.0 * np.real(sfft.irfft(Pf * np.conj(X), fsz))[:n]
        g -= g.mean()
        
        v = mom * v + (1.0 - mom) * g
        xn = np.maximum(x - lr * v, 0.0)
        x = _norm(xn)
        
        T = max(T_min, T * 0.99985)
        if it > 15000:
            lr = max(0.003, lr * 0.99997)
        
        if it % 300 == 0:
            sc = _sc(x)
            if np.isfinite(sc) and sc < best_sc - 1e-9:
                best_sc = sc
                best = x.copy()
                stuck = 0
            else:
                stuck += 300
            if stuck > stuck_lim:
                rsi += 1
                if rsi % 3 == 0:
                    # Big restart
                    jj = np.arange(n, dtype=np.float64)
                    x = 1.0 + 0.3 * np.sin(2*np.pi*(3+rsi)*jj/n) + 0.15*np.sin(2*np.pi*(7+2*rsi)*jj/n)
                    x = _norm(np.maximum(x, 0.05))
                else:
                    # Small restart from best
                    x = best * (1.0 + 0.08 * rng.standard_normal(n))
                    x = _norm(np.maximum(x, 0.0))
                T = max(0.15, best_sc * 0.05)
                lr = 0.15
                v[:] = 0.0
                stuck = 0
        
        if it % 2000 == 0 and it > 0:
            x += 0.001 * rng.standard_normal(n)
            x = _norm(np.maximum(x, 0.0))
        
        it += 1
    
    if not np.isfinite(best_sc):
        return [float(tv) for tv in par]
    return [float(tv) for tv in best]
