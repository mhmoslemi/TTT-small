        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = init.copy()
    
    # Run gradient descent from best init
    x = best.copy()

    T = 0.4

    stall = 0

        conv = np.maximum(conv, 0.0)

        cmax = conv.max()
        logits = (conv - cmax) / T
        np.clip(logits, -50.0, 0.0, out=logits)

        ps = p.sum()
        if ps > 1e-300:
            p /= ps
        else:
            p[:] = 0.0
            p[n-1] = 1.0

        P = rfft(p, fft_sz)
        g = 2.0 * np.real(irfft(P * np.conj(X), fft_sz))[:n]
        g -= g.mean()
        
        # Update
        v = beta * v + (1.0 - beta) * g
        x = x - lr * v
        x = norm1(x)
        
        # Annealing (two phases)
        if it < 100000:
            T = max(0.05, T * 0.9999)
            lr = max(0.05, lr * 0.99999)
        else:

            lr = max(0.02, lr * 0.99998)

                stall = 0
            else:
                stall += 500

                # Restart
                x = norm1(init_sine(n, rng))
                v[:] = 0.0
                T = 0.3
                lr = 0.25
                stall = 0

            x = norm1(x)
        
        it += 1

    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best = x.copy()
    
    if best is None:

import numpy as np
import time
from scipy.fft import rfft, irfft

def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def norm1(x):
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s < 1e-15:
        return np.full(len(x), 1.0 / len(x), dtype=np.float64)
    return x / s

def init_sine(n, rng):
    j = np.arange(n, dtype=np.float64)

        x += 0.35 * np.sin(2.0 * np.pi * freq * j / n + rng.uniform(0.0, 6.28318))
    return np.maximum(x, 0.005)
