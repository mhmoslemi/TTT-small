    return 2.0 * a.size * M / (S * S), c

def _topk_gradient(a, c, K, tau):

        top = np.arange(2 * n - 1)

    tmax = float(tv.max())
    denom = max(tau * M, 1e-30)
    w = np.exp((tv - tmax) / denom)
    ws = float(w.sum())
