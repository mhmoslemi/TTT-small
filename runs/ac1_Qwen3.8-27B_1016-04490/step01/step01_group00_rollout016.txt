```python
import numpy as np
import time
from scipy import fft as sfft
from scipy import sparse, optimize

def build_conv_matrix(n, f):
    """Build sparse Toeplitz matrix for convolution with f."""
    # A[k, j] = f[k-j] if 0 <= k-j <= n-1
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_min <= j_max:
            js = range(j_min, j_max + 1)
            rows.extend([k] * (j_max - j_min + 1))
            cols.extend(js)
            vals.extend(f[k - j] for j in js)
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A
```
