    fp = np.fft.rfft(p, N)
    c = np.fft.irfft(fp * fp, N)[:L]

def _acf(p):
    n = p.size
    L = 2 * n - 1
    N = 1 << max(0, (L - 1).bit_length())
    fp = np.fft.rfft(p, N)
    c = np.fft.irfft(fp * fp, N)[:L]
    return np.clip(c.real, 0.0, None)


def _sc(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    s = p.sum()
    if s < 0.01:
        return float('inf')
    c = _acf(p)
    return float(2.0 * p.size * c.max() / (s * s))


def _nm(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.clip(p, 0.0, None)
    s = p.sum()
    if s <= 0 or not np.isfinite(s):
        return None
    q = p / s
    return q if np.isfinite(q).all() else None


def _sm(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s <= 0:
        return np.full(z.shape, 1.0 / z.size)
    return e / s

def _obj(z, B):
    p = _sm(z)
    c = _acf(p)
    n = p.size
    u = n * c
    um = u.max()
    e = np.exp(B * (u - um))
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return 1e100
    return float((B * um + np.log(s)) / B)


def _grad(z, B):
    p = _sm(z)
    n = p.size
    c = _acf(p)
    u = n * c
    um = u.max()
    e = np.exp(B * (u - um))
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.zeros(n)
    w = e / s
    N = 1 << max(0, (2 * n - 1).bit_length())
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p, N)
    corr = np.fft.irfft(fw * np.conj(fp), N)[:n]
    g = 2.0 * n * corr
    dot = float(p @ g)
    grad = p * (g - dot)
    return grad if np.isfinite(grad).all() else np.zeros(n)
