    umax = u.max()

def lse_gradient(z, beta, lam):

            if s2 < 0.01:
                break
            np2 /= s2
            np2 /= n
            sc = _score(np2)
            if sc < res[1]:
                res[0] = p * np2 / np2.mean()
                res[1] = sc
    return res[0], res[1]

def _lp_fp(f, BD, steps=30, shrink=0.9):
    n = f.size
    M0 = float(np.max(_autocorr(f)))
    N = 1 << (2 * n - 2).bit_length()
    ff = np.fft.rfft(f, N)
    g = f.copy()
    for _ in range(steps):
        if time.monotonic() > BD - 10:
            break
        gw = np.fft.rfft(g, N)
        cg = np.clip(np.fft.irfft(ff * gw, N)[:2 * n - 1], 0.0, None)
        viol = (cg - M0) / M0
        dp = np.zeros(n)
        vp = np.fft.rfft(np.clip(viol, 0, None), N)
        dp[:] = np.real(np.fft.irfft(np.conj(ff) * vp, N)[:n])
        dp2 = np.fft.irfft(ff * vp, N)
        dp[:] += np.real(dp2[:n])
        dp *= 0.5 * n / M0
        g *= np.exp(-shrink * np.maximum(dp, 0))
        g = np.clip(g, 1e-18, None)
        g = g / g.sum()
    return g, float(np.max(_autocorr(f * g)))

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    try:
        b = float(budget_s)
    except Exception:
        b = 1100.0
    if b < 5:
        b = 5.0
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    BD = t0 + max(3.0, b - 30)
    p_best = None
    s_best = float('inf')
    n = 1
    par = None
    try:
        rv = np.asarray(height_sequence_1, dtype=np.float64)
        if rv.ndim == 1 and rv.size > 0:
            n = int(rv.size)
            par = _norm(rv)
    except Exception:
        pass
    if par is None:
        par = np.ones(n, dtype=np.float64)
    s0 = _score(par)
    p_best = par.copy()
    s_best = s0
    cands = [par.copy()].extend(_gen_cands(n, rng))
    for c in cands:
        if time.monotonic() > BD - 60:
            break
        pc = _norm(c)
        if pc is None:
            continue
        sc = _score(pc)
        if sc < s_best:
            s_best = sc
            p_best = pc.copy()
    starts = [p_best.copy()]
    sc_all = sorted([(_score(_norm(c)), _norm(c)) for c in cands if _norm(c) is not None])
    for sc, pc in sc_all:
        if len(starts) >= min(4, max(2, n // 500)):
            break
        if sc > s_best * 1.15:
            break
        if not any(np.mean(np.abs(pc - s).flatten()) < 0.02 for s in starts):
            starts.append(pc)
    if n <= 2000:
        bts = (16, 32, 64, 128, 256, 512, 1024)
        its_t = (40, 50, 60, 70, 80, 90, 100)
    elif n <= 5000:
        bts = (16, 32, 64, 128, 256)
        its_t = (30, 40, 50, 60, 70)
    else:
        bts = (8, 16, 32, 64)
        its_t = (25, 30, 40, 50)
    max_starts = 2 if n <= 3000 else 1
    lam = 1e-5
    for i, p0 in enumerate(starts[:max_starts]):
        if time.monotonic() > BD - 20:
            break
        scale = max(0.3, 1.0 - 0.15 * i)
        res = _run_lse(p0, BD, list(bts), [int(x * scale) for x in its_t], lam)
        if res is not None:
            s = _score(res)
            if s < s_best:
                s_best = s
                p_best = res.copy()
    if time.monotonic() < BD - 30:
        gc, cg = _lp_fp(p_best, BD, steps=25, shrink=0.8)
        s = _score(gc)
        if s < s_best:
            s_best = s
            p_best = gc
            p_best = _run_lse(gc, BD, (64, 256), (40, 50), 1e-5)
            if p_best is not None:
                s = _score(p_best)
                if s < s_best:
                    s_best = s
    if time.monotonic() < BD - 20:
        pq, sq = _quant_topK(p_best, 5, 1e-4, 0.5)
        if sq < s_best:
            s_best = sq
            p_best = pq
            p_best = _run_lse(p_best, BD, (64, 128), (30, 40), 1e-5)
            if p_best is not None:
                s2 = _score(p_best)
                if s2 < s_best:
                    s_best = s2
    if time.monotonic() < BD - 12:
        eta = 0.008
        kd = cKDTree(p_best.reshape(-1, 1))
        while time.monotonic() < BD - 10 and eta > 1e-5:
            nd2, idx = kd.query(p_best.reshape(-1, 1), k=3)
            nrm = np.maximum(nd2[:, 0] ** 0.5, 1e-8)
            nw = rng.normal(0, eta / nrm, p_best.size)
            cc = _norm(p_best * np.exp(nw))
            if cc is not None:
                sc = _score(cc)
                if sc < s_best - 1e-12:
                    s_best = sc
                    p_best = cc
                    kd = cKDTree(p_best.reshape(-1, 1))
                    eta = min(0.05, eta * 1.08)
                else:
                    eta *= 0.85
    try:
        ef = evaluate_sequence([float(max(0.0, x)) for x in p_best])
        if np.isfinite(ef):
            p_best = np.clip(np.asarray(p_best, dtype=np.float64), 0.0, 1000.0)
    except Exception:
        p_best = par.copy()
    if p_best is None or p_best.size == 0 or not np.all(np.isfinite(p_best)):
        p_best = np.ones(n, dtype=np.float64)
    if float(np.sum(p_best)) < 0.01:
        p_best = np.ones(n, dtype=np.float64)
    return [float(max(0.0, x)) for x in p_best]
