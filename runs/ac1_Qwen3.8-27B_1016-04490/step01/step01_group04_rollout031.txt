            donors = np.argpartition(contrib, n - D)[-D:]
            receivers = np.argpartition(grad, D - 1)[:D]
            for mf in (0.003, 0.01, 0.03):
                b = a.copy()
                ds = float(b[donors].sum())
                mv = min(mf, 0.4 * ds)
                if mv < 1e-15:
                    continue
                wd = b[donors]
                ws = float(wd.sum())
                if ws > 0:
                    b[donors] -= mv * wd / ws
                    b[receivers] += mv / float(len(receivers))
                b = _norm(b)
                scb, _, _ = _score(b, L)
                if np.isfinite(scb) and scb < sc - 1e-13:
                    a = b
                    sc = scb
                    break

    out = _norm(best)
    out = np.clip(out, 0.0, 1000.0)
    result = [float(x) for x in out]

    try:
        ex = float(evaluate_sequence(result))
        if not np.isfinite(ex):
            result = [1.0 / float(n)] * n
    except Exception:
        result = [1.0 / float(n)] * n

    return result
