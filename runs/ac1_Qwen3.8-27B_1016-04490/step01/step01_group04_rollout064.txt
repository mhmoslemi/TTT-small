def _next_p2(m):
    L = 1
    while L < m:
        L <<= 1
    return L

def _normalize(a):

    s = a.sum()
    if s < 1e-300:
        return np.ones_like(a) / a.size
    return a / s

def _autocorr(a, L):
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa, L)
