def _self_conv(a):
    n = len(a)
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    fa = np.fft.rfft(a, L)
    return np.fft.irfft(fa * fa, L)[:2 * n - 1]


def _cross_corr(a, b):
    """Linear cross-correlation: c[m] = sum_i a[i+m]*b[i], m=0..len(a)+len(b)-2"""
    total = len(a) + len(b) - 1
    L = 1
    while L < total:
        L <<= 1
    return np.fft.irfft(np.fft.rfft(a, L) * np.conj(np.fft.rfft(b, L)), L)[:total]


def _obj(a):
    b = _self_conv(a)
    return b.max()


def _score_fast(a):
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * len(a) * _obj(a) / (s * s)


def _grad_smoothed(a, tau):
    b = _self_conv(a)
    bmax = b.max()
    w = np.exp(b / tau - bmax / tau)
    ws = w.sum()
    if ws < 1e-30:

    w /= ws
    cc = _cross_corr(w, a)
    return 2.0 * cc[:len(a)], bmax
