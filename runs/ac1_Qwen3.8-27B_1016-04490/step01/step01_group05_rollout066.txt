import time
from scipy import sparse as sp
from scipy import optimize


def _fft_conv(a, b):
    n = len(a) + len(b) - 1
    L = 1
    while L < n:
        L <<= 1
    return np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)[:n]


def _score(a, n):
    b = _fft_conv(a, a)
    s = a.sum()
    if s < 1e-12:
        return 1e30
    return 2.0 * n * b.max() / (s * s)


def _normalize(a):
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _subgrad(f, k_star, n):
    sg = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    if lo <= hi:
        sg[lo:hi + 1] = 2.0 * f[k_star - np.arange(lo, hi + 1)]
    return sg


def _build_lp(n):
    rp, cp = [], []
    for k in range(2 * n - 1):
        jlo = max(0, k - n + 1)
        jhi = min(n - 1, k)
        js = np.arange(jlo, jhi + 1)
        rp.append(np.full(len(js), k, dtype=np.int32))
        cp.append(js.astype(np.int32))
    rows = np.concatenate(rp)
    cols = np.concatenate(cp)
    offs = (rows.astype(np.int64) - cols.astype(np.int64)).astype(np.int32)
    return rows, cols, offs


def _solve_lp(f, M, n, rows, cols, offs, cvec, bounds, tl):
    try:
        vals = f[offs]
        A = sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        bub = M * np.ones(2 * n - 1)
        res = optimize.linprog(cvec, A_ub=A, b_ub=bub, bounds=bounds,
                               method='highs', options={'time_limit': tl})
        if res.status == 0 and res.x is not None:
            g = np.maximum(np.asarray(res.x, dtype=np.float64), 0.0)
            if g.sum() > 1e-15 and np.all(np.isfinite(g)):
                return g / g.sum()
    except Exception:
        pass
    return None


def _grad_step(a, n, b, lr):
    k_star = int(np.argmax(b))
    sg = _subgrad(a, k_star, n)
    gn = np.linalg.norm(sg)
    if gn < 1e-15:
        return a, b
    for t in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.04]:
        a2 = a - t * sg / gn
        a2 = np.maximum(a2, 0.0)
        s2 = a2.sum()
        if s2 < 1e-12:
            continue
        a2 /= s2
        b2 = _fft_conv(a2, a2)
        if b2.max() <= b.max() * 1.002:
            return a2, b2
    return a, b
