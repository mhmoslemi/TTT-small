import numpy as np

def fft_conv(a, b=None):
    """Full linear convolution via FFT."""
    if b is None:
        b = a
    na, nb = len(a), len(b)
    L = na + nb - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:L]

def simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    v = v.copy()
    # Find threshold via sorting
    sv = np.sort(v)
    # We need: sum_{i: v_i > theta} (v_i - theta) = 1
    # Let sorted descending: s_0 >= s_1 >= ... >= s_{n-1}
    # Try k: theta_k = (sum_{i=0}^{k-1} s_i - 1) / k
    # Valid if s_k <= theta_k < s_{k-1} (for the k-th active set)
    desc = np.sort(v)[::-1]
    cumsum = np.cumsum(desc)
    # For active set of size k (1-indexed): theta = (cumsum[k-1] - 1) / k
    # Valid if desc[k] <= theta < desc[k-1] (i.e., desc[k-1] > theta >= desc[k])
    k = np.arange(1, n+1, dtype=np.float64)
    theta = (cumsum - 1.0) / k
    # Valid: theta < desc[k-1] and (k==n or theta >= desc[k])
    valid = np.zeros(n, dtype=bool)
    valid[0] = True  # k=1 always try
    for i in range(1, n):
        valid[i] = (theta[i] <= desc[i-1]) and (i == n-1 or theta[i] >= desc[i])
    # Find largest valid k
    idx = np.where(valid)[0]
    if len(idx) == 0:
        # Fallback: uniform
        return np.full(n, 1.0/n)
    k_best = idx[-1] + 1  # 1-indexed
    theta_best = theta[k_best - 1]
    p = np.maximum(v - theta_best, 0.0)
    p /= p.sum()
    return p
