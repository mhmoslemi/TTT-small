    """Compute linear convolution a*b via FFT, return real result of length len(a)+len(b)-1."""
    A = sfft.rfft(a, fs)
    B = sfft.rfft(b, fs)
    out = sfft.irfft(A * B, fs).real[:len(a)+len(b)-1]
    return out

        TW = sfft.irfft(XR * W, fs).real[:3*n-2]
        grad = TW[n-1:2*n-1]
        
        # Adam
        t_ad += 1
        m_a = 0.9 * m_a + 0.1 * grad
        v_a = 0.999 * v_a + 0.001 * grad * grad
        m_hat = m_a / (1.0 - 0.9 ** t_ad)
        v_hat = v_a / (1.0 - 0.999 ** t_ad)
        
        x = x - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        x = np.maximum(x, 0.0)
        xs = x.sum()
        if xs < 1e-14:
            x = best.copy()
        else:
            x = x / xs
        
        if cur < best_score - 1e-13:
            best_score = cur
            best = x.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        if no_imp > 5000:
            x = best + 0.01 * rng.standard_normal(n) * best.mean()
