        # We want grad[i] = 2 * sum_{k=0}^{n-1} w[k] * a[k-i] (for 0<=k-i<n)
        # Let j = k-i: grad[i] = 2 * sum_{j=-i}^{n-1-i} w[j+i] * a[j]
        # = 2 * sum_{j=0}^{n-1} w[j+i] * a[j] (with w[k]=0 for k outside [0,n-1])
        # Now (a * w_rev)[n-1-i] = sum_j a[j] * w_rev[n-1-i-j]
        # w_rev[n-1-i-j] = w[n-1-(n-1-i-j)] = w[i+j] (when 0<=i+j<n)
        # So (a * w_rev)[n-1-i] = sum_j a[j] * w[i+j] (for 0<=i+j<n)
        # This matches! So grad[i] = 2 * (a*w_rev)[n-1-i]
        
        grad = 2.0 * grad_full[n-1:-1:-1] if n > 1 else 2.0 * grad_full[:1]
        # Actually: grad_full has length fs. We need indices n-1, n-2, ..., 0.
        # grad_full[n-1-i] for i=0,...,n-1 -> indices n-1, n-2, ..., 0
        grad = 2.0 * grad_full[n-1::-1][:n]
        
        # Subtract mean to maintain normalization (projected gradient)
        grad = grad - grad.mean()
        
        # Update
        x_new = x - lr * grad
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-10:
            x_new = np.full(n, ts/n)
            sx = ts
        x_new = x_new * (ts / sx)
        x_new = np.clip(x_new, 0, 1000)
        x = x_new
        
        # Anneal
        T = max(0.01, T * 0.9997)
        if no_improve > 5000:
            # Restart
            x = best + rng.normal(0, max(best.std()*0.1, 0.01), n)
            x = np.maximum(x, 0)
            sx = x.sum()
            if sx > 1e-10:
                x = x * (ts/sx)
            T = T0
            lr = 0.3
            no_improve = 0
        
        it += 1
        
        if it % 20000 == 0:
            lr *= 0.5
            T = max(0.01, T * 0.5)
