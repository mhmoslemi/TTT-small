def cff(a, N):
    n = a.size
    fa = np.fft.rfft(a, N)
    return np.real(np.fft.irfft(fa * fa, N)[:2 * n - 1])

def ns(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.clip(a, 0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return None
    p = a / s
    return p if np.isfinite(p).all() else None

def _upd(p, N, n):

def ev(p, N, n):
    return float(2.0 * n * cff(p, N).max())

def soft_o(z, be, N, n):
    z = z - z.max()
    p = np.exp(z)
    S = p.sum()
    if S < 1e-300:
        return 1e100
    p /= S
    u = n * cff(p, N)
    um = u.max()
    ex = np.exp(be * (u - um))
    S2 = ex.sum()
    if S2 < 1e-300:
        return 1e100
    return (be * um + math.log(S2)) / be

def soft_g(z, be, N, n):
    z = z - z.max()
    p = np.exp(z)
    S = p.sum()
    if S < 1e-300:
        return np.zeros(n)
    p /= S
    u = n * cff(p, N)
    um = u.max()
    ex = np.exp(be * (u - um))
    S2 = ex.sum()
    if S2 < 1e-300:
        return np.zeros(n)
    w = ex / S2
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p, N)
    cv = np.real(np.fft.irfft(fw * fp, N)[:n])
    gp = 2.0 * n * cv
    return p * (gp - p @ gp)

def pshave(p, N, n, mx, dl):

    s = p.sum()
    if s < 1e-15:
        return p

        g = 2.0 * n * cv
