            cur = 2.0 * n * conv.max()
            if cur < best_score - 1e-13:
                best_score = cur
                best = x.copy()
                no_imp = 0
                if best_score <= 1.5028:
                    break
            else:
                no_imp += 1

            if no_imp >= ilim:
                x = best.copy()
                s = rng.integers(1, 5)
                ph = rng.uniform(0, 2 * np.pi)
                x += 0.02 * np.cos(2 * np.pi * s * t / n + ph)
                x += 0.01 * rng.standard_normal(n)
                x = np.maximum(x, 1e-12)
                x = x / x.sum()
                vel[:] = 0.0
                no_imp = 0

            grad = peak_grad(conv, x, m, work)
            grad -= grad.mean()
            vel = bt * vel + grad
            gn = np.linalg.norm(vel)
            if gn > 1e-18:
                step = sf * (1.0 / n) * vel / gn
                x = np.maximum(x - step, 0.0)
                sm = x.sum()
                x = x / sm if sm > 1e-14 else np.ones(n) / n
            it += 1
            if it % 5000 == 0 and no_imp > ilim // 2:
                no_imp += ilim // 4
        if best_score <= 1.5028:
            break
        ep += 1
        vel[:] = 0.0
        no_imp = 0

    # Phase 2: Fine polish with small m
    if time.monotonic() < deadline - 12.0:
        x = best.copy()
        vel[:] = 0.0
        for _ in range(200000):
            if time.monotonic() > deadline - 12.0:
                break
            conv = fft_conv(x, fs)
            c2 = 2.0 * n * conv.max()
            if c2 < best_score - 1e-14:
                best_score = c2
                best = x.copy()
            grad = peak_grad(conv, x, 3, work)
            grad -= grad.mean()
            vel = 0.4 * vel + grad
            gn = np.linalg.norm(vel)
            if gn < 1e-19:
                break
            x = np.maximum(x - 0.005 / n * vel / gn, 0.0)
            sm = x.sum()
            x = x / sm if sm > 1e-14 else np.ones(n) / n

    # Phase 3: Ultra-fine single-peak
    if time.monotonic() < deadline - 5.0:
        x = best.copy()
        for _ in range(100000):
            if time.monotonic() > deadline - 5.0:
                break
            conv = fft_conv(x, fs)
            c2 = 2.0 * n * conv.max()
            if c2 < best_score - 1e-14:
                best_score = c2
                best = x.copy()
            grad = peak_grad(conv, x, 1, work)
            grad -= grad.mean()
            gn = np.linalg.norm(grad)
            if gn < 1e-19:
                break
            x = np.maximum(x - 0.002 / n * grad / gn, 0.0)
            sm = x.sum()
            x = x / sm if sm > 1e-14 else np.ones(n) / n

    # Final validation with exact evaluator
    sc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(sc):
        pn = np.maximum(parent, 0.0)
        pn = pn / pn.sum()
        sc2 = float(evaluate_sequence(pn.tolist()))
        if np.isfinite(sc2):
            return [float(v) for v in pn]
        return [1.0]
    return [float(v) for v in best]
