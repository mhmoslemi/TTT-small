                x = _normalize(np.maximum(x, 0.0), target_sum)
                # Also try perturbed best
                if rng.random() < 0.5:
                    x = best + 0.03 * rng.standard_normal(n)
                    x = _normalize(np.maximum(x, 0.0), target_sum)
                mom.fill(0.0)
                T = max(0.02, best_sc * 0.04)
                lr = 0.5
                beta = 0.82
                no_imp = 0

        it += 1

    # --- Phase 2: Local polish with very low temperature ---
    if time.monotonic() < deadline - 10.0:
        x = best.copy()
        T = 1e-5
        lr = 0.05
        mom2 = np.zeros(n)
        for _ in range(4000):
            if time.monotonic() > deadline - 10.0:
                break
            conv = _fft_conv(x, Nc)[:2 * n - 1].clip(0.0)
            cmax = conv.max()
            if cmax < 1e-300:
                break
            logits = np.clip((conv - cmax) / max(T, 1e-8), -90.0, 0.0)
            p = np.exp(logits)
            ps = p.sum()
            if ps < 1e-300:
                p = np.zeros(2 * n - 1)
                p[(2 * n - 1) // 2] = 1.0
            else:
                p /= ps
            g = _grad_step(x, p, Ngrad)
            g -= g.mean()
            gm = np.abs(g).max()
            if gm > 1e5:
                g /= gm
            mom2[:] = 0.9 * mom2 + 0.1 * g
            x = _normalize(np.maximum(x - lr * mom2, 0.0), target_sum)
            if _ in range(0, 4000, 40):
                sc = _do_score(x)
                if sc < best_sc:
                    best_sc = sc
                    best = x.copy()
                T = max(1e-7, T * 0.9)

    # --- Phase 3: Try a few step-function perturbations ---
    if time.monotonic() < deadline - 5.0:
        for frac in [0.3, 0.4, 0.5, 0.6, 0.7]:
            if time.monotonic() > deadline - 5.0:
                break
            m = max(1, int(n * frac))
            for ratio in [0.15, 0.3, 0.5, 0.7]:
                x2 = np.empty(n)
                x2[:m] = 1.0
                x2[m:] = ratio
                x2 = _normalize(x2, target_sum)
                sc = _do_score(x2)
                if sc < best_sc:
                    best_sc = sc
                    best = x2.copy()

    # Final validation
    fc = _do_score(best)
    if not np.isfinite(fc):
        best = parent.copy()
        fc = _do_score(best)
    if not np.isfinite(fc):
        best = np.ones(n)
    return [float(v) for v in best]
