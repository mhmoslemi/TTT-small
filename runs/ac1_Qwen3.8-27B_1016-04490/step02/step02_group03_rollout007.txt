def subgrad_descent(p, iters=500):

        w[active] = 1.0 / len(active)
        # Gradient: g[m] = 2 * sum_k w[k] p[k-m]

        fw = np.fft.rfft(w, GNF)
        r = np.fft.irfft(np.conj(fp) * fw, GNF)
        g = 2.0 * np.real(r[:n])
        # Projected gradient step on simplex
        d = float(p @ g)
        g = g - d  # project gradient to tangent space
        # Adaptive step size
        base_lr = 1.0 / (n * n * 0.01)
        for lr in (base_lr * 10, base_lr * 5, base_lr * 2, base_lr, base_lr * 0.5, base_lr * 0.1):

def peak_shave(p, iters=300):

    n = GN
    p = p.copy()

    for _ in range(iters):

            break
        c = conv(p, GNF)
        pk_c = float(c.max())
        if pk_c < 1e-15:
            break
        idx = np.argsort(c)[::-1]
        active = []
        for k in idx[:50]:
            if c[k] >= pk_c * 0.97:
                active.append(int(k))
            else:
                break
        if not active:
            break
        wts = np.array([c[k] for k in active])
        wts /= wts.sum()
        g = np.zeros(n)
        for ii in range(len(active)):
            k = active[ii]
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            seg = p[lo:hi + 1].copy()
            g[k - hi:k - lo + 1] += wts[ii] * seg[::-1]
            # Also the symmetric part
        # Actually for autocorrelation, gradient at max is 2*p[k-m]
        # Let me just use the FFT approach
        w = np.zeros(2 * n - 1)
        for ii in range(len(active)):
            w[active[ii]] = wts[ii]
        fp = np.fft.rfft(p, GNF)
        fw = np.fft.rfft(w, GNF)
        r = np.fft.irfft(np.conj(fp) * fw, GNF)
        g = 2.0 * np.real(r[:n])
        d = float(p @ g)
        g -= d
        found = False
        for t in (0.05, 0.03, 0.01, 0.005, 0.002, 0.001):
            p2 = p - t * g
            if p2.min() < 0:
                p2 = np.maximum(p2, 0)
                s2 = p2.sum()
                if s2 < 1e-15:
                    continue
                p2 /= s2
            else:
                p2 /= p2.sum()

            if np.isfinite(sc2) and sc2 < pk - 1e-10:
                p = p2
                pk = sc2
                found = True
                break
