    fw = np.fft.rfft(w, GNF)
    r = np.fft.irfft(np.conj(fp) * fw, GNF)
    gp = float(n) * 2.0 * np.real(r[:n])
    d = float(p @ gp)
    g = p * (gp - d)
    return g

def _obj(z):
    return lse_val(z, CUR_GAMMA)

def _grad(z):
    return lse_grd(z, CUR_GAMMA)

def lse_run(z0):
    global CUR_GAMMA
    if sopt is None:
        return
    z = z0 - z0.mean()
    z = np.clip(z, -25, 25)
    best_v = lse_val(z, 8)
    betas = (8, 16, 32, 64, 128, 256, 512, 1024, 2048)
    mits = (50, 60, 80, 100, 120, 150, 200, 250, 300)
    for b, mi in zip(betas, mits):
        if time.monotonic() > DEADLINE - 25:
            break
        CUR_GAMMA = float(b)
        try:
            r = sopt.minimize(_obj, z, jac=_grad, method='L-BFGS-B',
                           callback=_tcb,
                           options={'maxiter': mi, 'ftol': 1e-15, 'gtol': 1e-11, 'maxls': 40})
        except Exception:
            break
        if r is not None and r.x is not None and np.isfinite(r.fun):
            if r.fun <= best_v + 1e-8:
                z = np.asarray(r.x, dtype=np.float64)
                best_v = float(r.fun)
        p = np.exp(z - z.max())
        p /= max(p.sum(), 1e-300)
        p = normclip(p)
        if p is not None:
            upb(p)

def peak_shave(p, iters):
    p = p.copy()
    fc = sc_f(p)
    c_prev = conve(p, GNF)
    steps = (0.06, 0.04, 0.025, 0.015, 0.01, 0.006, 0.003, 0.001)
    for _ in range(iters):
        if time.monotonic() > DEADLINE - 8:
            break
        pk = float(c_prev.max())
        if pk < 1e-15:
            break
        idx = np.argsort(c_prev)[::-1]
        top_k = idx[:80]
        thr = pk * 0.99
        active = []
        for k in top_k:
            if c_prev[k] >= thr:
                active.append(int(k))
            else:
                break
        if not active:
            break
        na = len(active)
        cvals = np.array([c_prev[k] for k in active])
        wts = np.maximum(cvals - thr * 0.95, 1e-10)
        wts /= wts.sum()
        g = np.zeros(GN, dtype=np.float64)
        for ii in range(na):
            k = active[ii]
            wi = wts[ii]
            lo = max(0, k - (GN - 1))
            hi = min(GN - 1, k)
            g[lo:hi + 1] += wi * p[k - hi:k - lo + 1][::-1]
        g *= 2.0 * float(GN)
        d = float(p @ g)
        g0 = g - d
        gn = np.linalg.norm(g0)
        if gn < 1e-18:
            break
        g0 /= gn
        found = False
        for t in steps:
            p2 = p - t * gn * g0
            if p2.min() < 0:
                p2 = np.maximum(p2, 0)
                s2 = p2.sum()
                if s2 < 1e-15:
                    break
                p2 /= s2
            else:
                p2 /= p2.sum()
            c2 = conve(p2, GNF)
            sc2 = float(GN) * c2.max()
            if np.isfinite(sc2) and sc2 < fc - 1e-14:
                p = p2
                fc = sc2
                c_prev = c2
                found = True
                break
        if not found:
            break
    upb(p)
    return p

def sym_tv(p):
    psym = (p + p[::-1]) / 2.0
    psym = normclip(psym)
    if psym is not None:
        upb(psym)
    return psym

def inits(n):
    L = []
    j = np.arange(n, dtype=np.float64)
    m = 0.5 * (n - 1)
    L.append(np.ones(n, dtype=np.float64))
    L.append(0.5 * (1.0 + np.cos(np.pi * j / n)))
    L.append(0.5 * (1.0 - np.cos(np.pi * j / n)) + 0.01)
    for p in (-0.3, -0.7, -1.2, -2.0):
        L.append((j + 1.0) ** p)
    for a in (0.5, 1.0, 2.0, 4.0):
        L.append(np.maximum(1.0 - np.abs(j - m) / (m + 1.0), 0.0) ** a + 0.01)
    for f in (1, 2, 3, 5):
        mod = 1.0 + 0.3 * np.cos(2.0 * np.pi * f * j / n)
        L.append(0.5 * (1.0 + np.cos(np.pi * j / n)) * mod + 0.01)
    t1 = 0.35 * n
    t2 = 0.65 * n
    sd = 0.1 * n
    L.append(np.exp(-((j - t1) / sd) ** 2) + np.exp(-((j - t2) / sd) ** 2) + 0.005)
    d = np.minimum(j, n - 1 - j) + 1.0
    for pv in (0.5, 1.5, 3.0):
        L.append(d ** pv)
    return L

def _sort_key(x):
    return x[0]

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    global DEADLINE, BEST_P, BEST_SC, GN, GNF
    t0 = time.monotonic()
    wall = t0 + budget_s - 50.0
    DEADLINE = wall
    rng = np.random.default_rng(seed)
    n = 2535
    parent = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64)
        if raw.ndim == 1 and raw.size > 0:
            n = int(raw.size)
            parent = normclip(raw)
    except Exception:
        pass
    if parent is None:
        parent = np.ones(n, dtype=np.float64)
    GN = n
    GNF = 1 << (2 * n - 2 - 1).bit_length()
    BEST_P = parent.copy()
    BEST_SC = float(evaluate_sequence(parent.tolist()))
    if not np.isfinite(BEST_SC):
        BEST_SC = 2.0
        BEST_P = np.ones(n).copy()
    # Generate and score initializations
    cands = inits(n)
    scored = []
    for a in cands:
        if time.monotonic() > wall - 90:
            break
        p = normclip(a)
        if p is None:
            continue
        sc = float(evaluate_sequence(p.tolist()))
        scored.append([sc, p])
    scored.append([BEST_SC, BEST_P.copy()])
    scored.sort(key=_sort_key)
    # Pick diverse starts
    starts = []
    for sc, p0 in scored:
        if len(starts) >= 5:
            break
        dup = False
        for q in starts:
            if np.linalg.norm(p0 - q) < 0.2:
                dup = True
                break
        if not dup:
            starts.append(p0.copy())
    # Run log-sum-exp optimization from each start
    for p0 in starts:
        if time.monotonic() > wall - 70:
            break
        z0 = np.log(np.maximum(p0, 1e-15))
        z0 -= z0.mean()
        z0 = np.clip(z0, -25, 25)
        lse_run(z0)
    # Peak shaving on global best
    upb(BEST_P)
    p_out = peak_shave(BEST_P.copy(), 800)
    # Try palindromic
    psym = sym_tv(p_out)
    if psym is not None:
        p_out2 = peak_shave(psym, 400)
    # Peak shave other starts
    for p0 in starts[:4]:
        if time.monotonic() > wall - 20:
            break
        peak_shave(p0.copy(), 300)
    # Fine polish on best
    upb(BEST_P)
    p_final = fine_polish(BEST_P.copy(), 500)
    psym2 = sym_tv(p_final)
    if psym2 is not None:
        fine_polish(psym2, 200)
    upb(BEST_P)
    if BEST_P is None or BEST_P.size == 0 or not np.isfinite(BEST_P).all():
        BEST_P = np.ones(n, dtype=np.float64)
    return [float(x) for x in BEST_P]
