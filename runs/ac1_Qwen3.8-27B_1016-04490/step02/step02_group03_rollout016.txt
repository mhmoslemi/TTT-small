    if p.sum() < 0.01:
        return
    s = float(evaluate_sequence(p.tolist()))
    if np.isfinite(s) and s < BEST_SC:
        BEST = p.copy()
        BEST_SC = s

def _lse_val(z):
    a = np.exp(z - z.max())
    s = a.sum()
    if s < 1e-300:
        return 1e100
    a /= s
    c = _conv(a)
    cm = c.max()
    e = np.exp(CB * (c - cm))
    ss = e.sum()
    if ss < 1e-300:
        return 1e100
    return float(cm + np.log(ss) / CB)

def _lse_grd(z):
    n = N
    a = np.exp(z - z.max())
    s = a.sum()
    if s < 1e-300:
        return np.zeros(n)
    a /= s
    c = _conv(a)
    cm = c.max()
    e = np.exp(CB * (c - cm))
    ss = e.sum()
    if ss < 1e-300:
        return np.zeros(n)
    w = e / ss
    A = np.fft.rfft(a, NF)
    W = np.fft.rfft(w, NF)
    ga = 2.0 * np.real(np.fft.irfft(np.conj(A) * W, NF))[:n]
    d = float(np.dot(a, ga))
    return a * (ga - d)

def _run_lse(z0, betas, its):
    global CB
    z = z0 - z0.mean()
    z = np.clip(z, -30.0, 30.0)
    for b, mi in zip(betas, its):
        if time.monotonic() > DL - 25:
            break
        if sopt is None:
            break
        CB = float(b)
        try:
            r = sopt.minimize(_lse_val, z, jac=_lse_grd, method='L-BFGS-B',
                             callback=_tc,
                             options={'maxiter': mi, 'ftol': 1e-15, 'gtol': 1e-11, 'maxls': 40})
            if r.x is not None and np.isfinite(r.fun):
                z = np.asarray(r.x, dtype=np.float64)
        except Exception:
            pass
        a = np.exp(z - z.max())
        a /= a.sum()
        _upb(a)
    return z

def _shave(p, niter):
    p = p.copy()
    c = _conv(p)
    fc = float(c.max())
    for _ in range(niter):
        if time.monotonic() > DL - 8:
            break
        c = _conv(p)
        cm = float(c.max())
        if cm < 1e-18:
            break
        idx = np.argsort(c)[::-1]
        topk = idx[:60]
        thr = cm * 0.90
        act = [int(i) for i in topk if c[i] >= thr]
        if not act:
            break
        wts = np.array([c[kk] for kk in act])
        wts /= wts.sum()
        pw = np.zeros(2 * N - 1)
        for ii, kk in enumerate(act):
            pw[kk] = wts[ii]
        A = np.fft.rfft(p, NF)
        W = np.fft.rfft(pw, NF)
        g = 2.0 * float(N) * np.real(np.fft.irfft(np.conj(A) * W, NF))[:N]
        d = float(np.dot(p, g))
        gd = g - d
        done = False
        for t in (0.06, 0.04, 0.025, 0.015, 0.008, 0.004, 0.002):
            p2 = p - t * gd
            p2 = np.maximum(p2, 0.0)
            s2 = p2.sum()
            if s2 < 1e-15:
                break
            p2 /= s2
            c2 = _conv(p2)
            if c2.max() < fc - 1e-14:
                p = p2
                fc = float(c2.max())
                done = True
                break
        if not done:
            break
    _upb(p)
    return p

def _inits(n):
    L = []
    j = np.arange(n, dtype=np.float64)
    m = 0.5 * (n - 1)
    L.append(np.ones(n, dtype=np.float64))
    L.append(0.5 * (1.0 + np.cos(np.pi * j / max(n - 1, 1))))
    L.append(0.5 * (1.0 - np.cos(np.pi * j / max(n - 1, 1))) + 0.01)
    for p in (-0.4, -0.7, -1.0, -1.5, -2.0, -3.0):
        L.append((j + 1.0) ** p)
    for a in (1.0, 2.0, 4.0, 7.0):
        L.append((1.0 - np.abs(j - m) / (m + 1.0)) ** a + 0.01)
    for f in (1, 2, 3, 5):
        mod = 1.0 + 0.4 * np.cos(2.0 * np.pi * f * j / n)
        L.append((0.5 * (1.0 + np.cos(np.pi * j / max(n - 1, 1))) * mod + 0.01).copy())
    for f in (4, 7, 13):
        L.append(0.5 + 0.4 * np.cos(2.0 * np.pi * f * j / n) + 0.01)
    return L
