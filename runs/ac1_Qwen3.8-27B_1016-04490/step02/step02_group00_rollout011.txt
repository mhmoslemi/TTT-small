    if sc < best_score:
        best = a.copy()
        best_score = sc
    return best, best_score


def _peak_shave(a, deadline, rng):
    n = a.size
    fft_sz = _make_fft_size(n)
    a = a.copy()
    best = a.copy()
    best_score = _score(a)
    it = 0
    lr = 0.3

    while time.monotonic() < deadline:
        fa = np.fft.rfft(a, fft_sz)
        s_conv = np.fft.irfft(fa * fa, fft_sz)[:2 * n - 1]
        k_star = int(np.argmax(s_conv))
        idx = np.arange(n)
        valid = (k_star - idx >= 0) & (k_star - idx < n)
        dconv = np.zeros(n)
        j = idx[valid]
        dconv[j] = a[j] + a[k_star - j]
        dconv = dconv / max(dconv.max(), 1e-15)
        a *= np.exp(-lr * dconv)
        a = np.maximum(a, 1e-15)
        a = a / a.sum() * n
        lr *= 0.995
        it += 1
        if it % 50 == 0:
            sc = _score(a)
            if sc < best_score - 1e-10:
                best_score = sc
                best = a.copy()

    sc = _score(a)
    if sc < best_score:
        best = a.copy()
        best_score = sc
    return best, best_score


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 15.0)

    parent = np.asarray(height_sequence_1, dtype=np.float64)
    if parent.ndim != 1 or parent.size < 2 or not np.isfinite(parent).all():
        parent = np.ones(2000, dtype=np.float64)
    n = parent.size
    parent = np.clip(parent, 0.0, 1000.0)
    if parent.sum() < 1e-10:
        parent = np.ones(n)
    parent = parent / parent.sum()

    j = np.arange(n, dtype=np.float64)
    x = j / max(n - 1, 1)
    c = 2.0 * x - 1.0

    inits = [
        parent.copy(),
        np.exp(-3.0 * c * c),
        0.5 + 0.5 * np.cos(np.pi * c),
        np.maximum(1.0 - np.abs(c), 0.05) ** 1.5,
        np.exp(-1.0 * c * c) + 0.3 * np.exp(-8.0 * c * c),
        (1.0 / (1.0 + 2.0 * c * c)) ,
    ]

    scored = []
    for ia in inits:
        ia = np.maximum(ia, 0.0).astype(np.float64)
        ia = _normalize(ia)
        sc = _score(ia)
        if np.isfinite(sc):
            scored.append((sc, ia.copy()))
    scored.sort(key=lambda t: t[0])

    if not scored:
        return [float(v) for v in parent]

    best_all = scored[0][1].copy()
    best_sc = scored[0][0]

    rem = max(deadline - time.monotonic(), 20.0)
    top_k = min(5, len(scored))
    time_per = rem * 0.75 / top_k

    for k in range(top_k):
        if time.monotonic() >= deadline - 60:
            break
        dl = min(time.monotonic() + time_per, deadline - 60.0)
        b, sc = _optimize(scored[k][1].copy(), dl, rng, use_adam=(k % 2 == 1))
        if sc < best_sc:
            best_sc = sc
            best_all = b.copy()

    rem2 = deadline - time.monotonic()
    if rem2 > 80.0:
        b2, sc2 = _optimize(best_all.copy(), deadline - 40.0, rng, use_adam=True)
        if sc2 < best_sc:
            best_sc = sc2
            best_all = b2.copy()

    rem3 = deadline - time.monotonic()
    if rem3 > 30.0:
        b3, sc3 = _peak_shave(best_all.copy(), deadline - 5.0, rng)
        if sc3 < best_sc:
            best_sc = sc3
            best_all = b3.copy()

    best_all = np.clip(best_all, 0.0, 1000.0)
    if not np.isfinite(best_all).all() or best_all.sum() < 0.01:
        best_all = parent.copy()

    sc_final = _score(best_all)
    if not np.isfinite(sc_final):
        best_all = parent.copy()

    return [float(v) for v in best_all]
