    s = v.sum()
    if s < 1e-10:
        return np.full(n, ts / n)
    return np.clip(v * (ts / s), 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    dl = time.monotonic() + max(0.0, float(budget_s) - 15.0)

    try:
        par = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except (NameError, TypeError):
        par = np.ones(4498)

    n = par.size
    fs = _np2(2 * n + 1)
    ts = np.sqrt(2.0 * n)
    jj = np.arange(n, dtype=np.float64)

    x = _normalize(par.copy(), n, ts)
    best = x.copy()
    bsc = _score_fft(best, n, fs)

    # Initialization exploration
    for trial in range(40):
        if time.monotonic() > dl - 60.0:
            break
        t = par.astype(np.float64).copy()
        style = trial % 5
        if style == 0:
            t += rng.uniform(0.05, 0.4) * np.sin(
                2 * np.pi * rng.uniform(1, 25) * jj / n + rng.uniform(0, 6.283))
        elif style == 1:
            t *= 1.0 + rng.uniform(0.05, 0.3) * np.cos(
                2 * np.pi * rng.uniform(1, 18) * jj / n + rng.uniform(0, 6.283))
        elif style == 2:
            t += rng.standard_normal(n) * par.std() * rng.uniform(0.05, 0.25)
        elif style == 3:
            nb = int(rng.integers(3, 12))
            for _ in range(nb):
                b = int(rng.integers(0, n - 50))
                w = int(rng.integers(10, 300))
                t[b:b + w] *= 1.0 + rng.uniform(-0.25, 0.25)
        else:
            t += 0.08 * par * np.sin(2 * np.pi * rng.uniform(5, 50) * jj / n)
        t = np.maximum(t, 0.0)
        t = _normalize(t, n, ts)
        sc = _score_fft(t, n, fs)
        if np.isfinite(sc) and sc < bsc:
            bsc = sc
            best = t.copy()
            x = t.copy()

    # Main optimization loop
    T = 0.08
    lr = 0.4
    no_imp = 0
    it = 0
    max_patience = 20000

    while time.monotonic() < dl - 4.0:
        it += 1

        X = sfft.rfft(x, fs)
        conv = np.maximum(np.real(sfft.irfft(X * X, fs))[:2 * n - 1], 0.0)
        cm = conv.max()
        sx = x.sum()
        cur = 2.0 * n * cm / (sx * sx)

        if cur < bsc - 1e-10:
            bsc = cur
            best = x.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Softmax weights
        scaled = (conv - cm) / T
        mx = scaled.max()
        ex = np.exp(scaled - mx)
        es = ex.sum()
        if es > 1e-300:
            p = ex / es
        else:
            p = np.ones(2 * n - 1) / (2 * n - 1)

        # Full gradient: g = (p_rev * x) shifted + (p * x) shifted
        pr = p[::-1]
        P1 = sfft.rfft(pr, fs)
        conv1 = np.real(sfft.irfft(P1 * X, fs))
        P2 = sfft.rfft(p, fs)
        conv2 = np.real(sfft.irfft(P2 * X, fs))
        g = conv1[n - 1:2 * n - 1][::-1] + conv2[:n]
        g -= g.mean()

        # Gradient step
        x_new = x - lr * g
        x_new = np.maximum(x_new, 0.0)
        x_new = _normalize(x_new, n, ts)

        # Quick acceptance: compare scores
        sc_new = _score_fft(x_new, n, ts)
        if sc_new < cur * 1.001:
            x = x_new
        else:
            # Try smaller step
            x_s = x - lr * 0.15 * g
            x_s = np.maximum(x_s, 0.0)
            x_s = _normalize(x_s, n, ts)
            if _score_fft(x_s, n, fs) < cur:
                x = x_s

        # Schedule
        T = max(0.001, T * 0.99985)
        if it > 8000:
            lr = max(0.03, lr * 0.9999)

        # Periodic noise
        if it % 4000 == 0 and it > 0:
            scale = max(1e-5, best.std() * 0.015)
            x += rng.standard_normal(n) * scale
            x = _normalize(np.maximum(x, 0.0), n, ts)
            T = max(T, 0.02)

        # Large jumps (step-function-style perturbation)
        if it > 0 and it % 12000 == 0:
            nseg = int(rng.integers(4, 15))
            seg_len = n // nseg
            shifts = rng.uniform(-0.3, 0.3, nseg)
            for s_i in range(nseg):
                lo = s_i * seg_len
                hi = lo + seg_len if s_i < nseg - 1 else n
                x[lo:hi] *= (1.0 + shifts[s_i])
            x = _normalize(np.maximum(x, 0.0), n, ts)

        # Restart if stuck
        if no_imp > max_patience:
            mag = max(best.std() * 0.12, 0.005)
            x = best + rng.normal(0, mag, n)
            x = _normalize(np.maximum(x, 0.0), n, ts)
            T = max(0.04, T * 0.5)
            lr = 0.35
            max_patience = max(5000, int(max_patience * 0.6))
            no_imp = 0

    # Final verification
    fb = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(fb):
        fb2 = float(evaluate_sequence(par.tolist()))
        if np.isfinite(fb2):
            return par.tolist()
        return [1.0] * n
    return best.tolist()
