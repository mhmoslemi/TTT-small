def subgradient_shave(p, max_iter=2000):
    n = len(p)
    p = p.copy()
    c = conv_fft(p)
    pk = float(c.max())
    st = time.monotonic()
    for it in range(max_iter):
        if dead():
            break
        if time.monotonic() - st > 5:  # timeout per call
            break
        # Find peaks
        thr = pk * (1 - 1e-4)
        K = np.where(c >= thr)[0]
        if len(K) == 0:
            K = np.array([int(np.argmax(c))])
        # Limit to top peaks
        if len(K) > 50:
            K = K[np.argsort(c[K])[::-1][:50]]
        # Compute subgradient
        grad = np.zeros(n)
        for k in K:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi+1] += 2.0 * p[k-hi:k-lo+1][::-1]
        # Project to sum-zero (tangent to simplex)
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        grad /= gn
        # Find max step
        pos = grad > 1e-15
        if pos.any():
            t_max = float(np.min(p[pos] / grad[pos])) * 0.99
        else:
            t_max = 0.01
        # Backoff line search
        improved = False
        for frac in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625):
            t = t_max * frac
            if t < 1e-10:
                break
            p_new = p - t * grad
            if p_new.min() < -1e-12:
                continue
            p_new = np.maximum(p_new, 0)
            p_new /= p_new.sum()
            c_new = conv_fft(p_new)
            pk_new = float(c_new.max())
            if pk_new < pk - 1e-12:
                p = p_new
                c = c_new
                pk = pk_new
                improved = True
                break
        if not improved:
            # Try even smaller step
            t = t_max * 0.001
            if t > 1e-12:
                p_new = np.maximum(p - t*grad, 0)
                p_new /= p_new.sum()
                c_new = conv_fft(p_new)
                if c_new.max() < pk - 1e-14:
                    p = p_new
                    c = c_new
                    pk = float(c_new.max())
                    improved = True
        if not improved:
            break
    update_best(p)
    return p

import numpy as np
import time

DEADLINE = 0.0
BEST_P = None
BEST_SC = float('inf')
N = 0
NF = 0
_LB = 64.0

def dead():
    return time.monotonic() > DEADLINE

def conv_fft(a):
    n = len(a)
    Nf = 1 << (2 * n - 1 - 1).bit_length()
    fa = np.fft.rfft(a, Nf)
    return np.real(np.fft.irfft(fa * fa, Nf)[:2 * n - 1])

def score(p):
    n = len(p)
    s = p.sum()
    if s < 0.01:
        return float('inf')
    return float(2.0 * n * conv_fft(p).max() / (s * s))

    global BEST_P, BEST_SC

    a = np.clip(np.asarray(a, dtype=np.float64), 0, 1e6)
    s = a.sum()
    if s < 1e-10 or not np.isfinite(s):
        return None
    return a / s

def lse_obj(z):
    zm = z - z.max()
    p = np.exp(zm)
    s = p.sum()
    if s < 1e-300:
        return 1e100
    p /= s
    c = conv_fft(p)
    u = float(N) * c
    um = u.max()
    B = _LB
    ex = np.exp(B * (u - um))
    s2 = ex.sum()
    if s2 < 1e-300:
        return 1e100
    return float((B * um + np.log(s2)) / B)

def lse_grad(z):
    n = N
    zm = z - z.max()
    p = np.exp(zm)
    s = p.sum()
    if s < 1e-300:
        return np.zeros(n)
    p /= s
    c = conv_fft(p)

    um = u.max()
    B = _LB
    ex = np.exp(B * (u - um))

    fw = np.fft.rfft(w, NF)
    cross = np.fft.irfft(fw * np.conj(fp), NF)
    grad_p = 2.0 * cross[:n]
    d = float(p @ grad_p)
    return p * (grad_p - d)

            if r is not None and np.isfinite(r.fun) and r.fun <= cv + 1e-6:

        except Exception:

        p = np.exp(z - z.max())

def subgrad_shave(p, max_it=3000, time_limit=8.0):
    n = len(p)
    p = p.copy()
    c = conv_fft(p)
    pk = float(c.max())
