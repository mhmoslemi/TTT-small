def _peak_and_grad(p, N, top_k=80, thresh=0.92):
    """Return (max_val, convolution, gradient of max w.r.t. p)."""
    n = p.size
    L = 2 * n - 1
    fa = np.fft.rfft(p, N)
    c = np.real(np.fft.irfft(fa * fa, N)[:L])
    cm = c.max()
    if cm < 1e-15:
        return cm, c, np.zeros(n)
    thr = thresh * cm
    idx = np.where(c >= thr)[0]
    if len(idx) > top_k:
        order = np.argsort(c[idx])[::-1][:top_k]
        idx = idx[order]
    # weights proportional to c values
    wv = c[idx]
    wsum = wv.sum()
    wv = wv / wsum
    # gradient: g[j] = 2 * sum_{k in active} w[k] * p[k-j] (if 0<=k-j<n)
    g = np.zeros(n)
    for ii in range(len(idx)):
        k = idx[ii]
        wi = wv[ii]
        if wi < 1e-15:
            continue
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        seg = p[k - hi : k - lo + 1][::-1]
        g[lo : hi + 1] += 2.0 * wi * seg
    return cm, c, g

    return p / s

def peak_shave(p0, N, iters, deadline, top_k=80, thresh=0.92):
    """Iterative peak shaving with line search."""
    n = p0.size
    p = normalize(p0.copy())
    if p is None:
        return p0, float('inf')
    best_p = p.copy()
    best_sc = _score(p, N)
    stall = 0
    for it in range(iters):
        if time.monotonic() > deadline:
            break
        cm, c, g = _peak_and_grad(p, N, top_k, thresh)
        # Project to maintain sum
        g = g - g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-13:
            break
        g = g / gn  # normalize gradient
        
        # Line search: try decreasing step sizes
        improved = False
        for t in (0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005):
            p2 = p - t * g
            p2 = np.maximum(p2, 0.0)

            sc2 = _score(p2, N)

                    best_p = p.copy()
                improved = True
                break
        if not improved:

            if stall > 30:
                break
        else:
            stall = 0
    return best_p, best_sc
