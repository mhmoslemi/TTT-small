        gn = np.linalg.norm(g)
        if gn > 1e-14:
            g /= gn
        mom = beta2 * mom + (1.0 - beta2) * g
        x = x - lr2 * mom
        x = np.maximum(x, 0.0)
        x = _norm(x, ts)
        if k % 150 == 0:
            sc = float(evaluate_sequence(x.tolist()))
            if np.isfinite(sc) and sc < bs:
                bs = sc
                best = x.copy()
            lr2 *= 0.997

    # Final verification
    fc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(fc):
        return [float(v) for v in par]
    return [float(v) for v in best]
