import time
from scipy import fft as sfft


def _np2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _sc(v):
    return float(evaluate_sequence(v.tolist()))


def _norm(x, ts):
    s = x.sum()
    if s < 1e-12:
        x[:] = ts / x.size
    else:
        x *= ts / s
    np.clip(x, 0.0, 1000.0, out=x)
