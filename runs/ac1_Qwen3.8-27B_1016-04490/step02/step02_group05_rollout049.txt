import numpy as np
import time
from scipy import fft as sfft


def _pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _norm(x, s):
    x = np.maximum(x, 0.0)
    tot = x.sum()
    if tot < 1e-12:
        return np.full(len(x), s / len(x))
    return x * (s / tot)


def _auto(x, fs):
    X = sfft.rfft(x, fs)
    c = np.real(sfft.irfft(X * X, fs))
    return c[:2 * len(x) - 1]


def _xcorr(x, p, cs):
    P = sfft.rfft(p, cs)
    Xc = sfft.rfft(x, cs)
    return np.real(sfft.irfft(P * np.conj(Xc), cs))


def _sstep(x, T, lr, be, mo, fs, cs):
    n = len(x)
    conv = _auto(x, fs)
    cmax = conv.max()
    if cmax <= 0:
        return x
    w = (conv - cmax) / T
    w = np.clip(w, -80.0, 0.0)
    p = np.exp(w)
    ps = p.sum()
    if ps < 1e-300:
        p = np.zeros(2 * n - 1)
        p[n - 1] = 1.0
    else:
        p /= ps
    g = 2.0 * _xcorr(x, p, cs)[:n]
    g -= g.mean()
    gm = np.abs(g).max()
    if gm > 1e5:
        g /= gm
    mo[:] = be * mo + (1.0 - be) * g
    xn = x - lr * mo
    return _norm(xn, x.sum())
