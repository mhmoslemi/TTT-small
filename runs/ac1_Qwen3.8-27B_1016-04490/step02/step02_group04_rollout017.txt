    # Build sparse constraint matrix A: (2n-1) x n
    # A[k, i] = x[k-i] if 0 <= k-i < n, else 0

    cols = []
    vals = []
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        for j in range(jmin, jmax + 1):
            rows.append(k)
            cols.append(k - j)
            vals.append(x[j])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

    return None
