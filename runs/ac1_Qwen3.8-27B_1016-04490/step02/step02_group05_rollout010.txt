import numpy as np
import time
from scipy import fft as sfft


def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _normalize(x, s):
    total = x.sum()
    if total < 0.01:
        x = np.maximum(x, 0.01)
        total = x.sum()
    return np.clip(x * (s / total), 0.0, 1000.0)


def _autocorr(x, fs, n):
    X = sfft.rfft(x, fs)
    return np.real(sfft.irfft(X * X, fs))[: 2 * n - 1]


def _grad_step(x, T, lr, mom, beta, fs, cs, n):
    c = _autocorr(x, fs, n)
    c = np.maximum(c, 0.0)
    cm = c.max()
    lg = np.clip((c - cm) / T, -50.0, 0.0)
    p = np.exp(lg)
    ps = p.sum()
    if ps < 1e-300:
        p = np.zeros(2 * n - 1)
        p[n - 1] = 1.0
    else:
        p /= ps
    P = sfft.rfft(p, cs)
    Xc = sfft.rfft(x, cs)
    cr = np.real(sfft.irfft(P * np.conj(Xc), cs))
    g = 2.0 * cr[:n]
    g -= g.mean()
    gm = np.abs(g).max()
    if gm > 1e6:
        g *= 1e6 / gm
    mom[:] = beta * mom + (1.0 - beta) * g
    xn = np.clip(x - lr * mom, 0.0, 1000.0)
    return xn, c
