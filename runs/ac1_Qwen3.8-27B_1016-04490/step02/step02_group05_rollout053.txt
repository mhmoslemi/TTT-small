def _step(x, T, lr, mom, beta, fsz, csz, n):
    X = sfft.rfft(x, fsz)
    cv = np.real(sfft.irfft(X*X, fsz))
    cv[n-1:] = cv[n-1:]  # keep full
    cv = cv[:2*n-1]
    cv = np.maximum(cv, 0.0)
    cm = cv.max()
    lg = np.clip((cv - cm) / max(T, 1e-12), -50.0, 0.0)
    pm = np.exp(lg)
    ps = pm.sum()
    if ps < 1e-300:
        pm = np.zeros(2*n-1)
        pm[(2*n-1)//2] = 1.0
    else:
        pm /= ps
    P = sfft.rfft(pm, csz)
    Xc = sfft.rfft(x, csz)
    cr = np.real(sfft.irfft(P * np.conj(Xc), csz))
    g = 2.0 * cr[:n].copy()
    g -= g.mean()
    gm = np.abs(g).max()
    if gm > 1e8:
        g *= 1e-8 / (gm * 1e-16)
    mom[:] = beta * mom + (1.0-beta) * g
    xn = x - lr * mom
    np.maximum(xn, 0.0, out=xn)
    return xn
