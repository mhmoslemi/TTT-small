```python
def _lp_direction(f, M):
    """Solve LP: max sum(b) s.t. b>=0, (f*b)_k <= M for all k."""
    n = len(f)
    # Build sparse constraint matrix A of shape (2n-1, n)
    # A[k, j] = f[k-j] if 0 <= k-j < n else 0
    rows, cols, vals = [], [], []
    for k in range(2*n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    # Linprog: minimize -sum(b) s.t. A@b <= M, b >= 0
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=M*np.ones(2*n-1), bounds=(0, None), method='highs')
    if res.success:
        return res.x
    return None
```

```python
def _build_A(a):
    n = len(a)
    from scipy.sparse import coo_matrix
    # Use the structure: A is a band matrix
    # Row k, column j: A[k,j] = a[k-j] if 0<=k-j<n
    # Let d = k-j (the "diagonal offset")
    # For each diagonal d (range -(n-1) to n-1):
    #   A[k, k-d] = a[d] for valid k, k-d
    #   k-d in [0, n-1] => k in [d, d+n-1]
    #   k in [0, 2n-2]
    #   So k in [max(0,d), min(2n-2, d+n-1)]
    
    rows_all = []
    cols_all = []
    vals_all = []
    for d in range(-(n-1), n):
        val = a[d] if 0 <= d <= n-1 else 0  # d is always in [-(n-1), n-1]
        if val == 0:
            continue
        k_lo = max(0, d)
        k_hi = min(2*n-2, d+n-1)
        ks = np.arange(k_lo, k_hi+1)
        js = ks - d
        rows_all.append(ks)
        cols_all.append(js)
        vals_all.append(np.full(len(ks), val))
    
    rows = np.concatenate(rows_all)
    cols = np.concatenate(cols_all)
    vals = np.concatenate(vals_all)
    A = coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()
    return A
```

```python
import numpy as np
import time
from scipy.sparse import coo_matrix
from scipy.optimize import linprog


def _score(a):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    fs = 1
    while fs < 2 * n - 1:
        fs <<= 1
    fa = np.fft.rfft(a, fs)
    conv = np.fft.irfft(fa * fa, fs)[:2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _conv_aa(a, fs):
    fa = np.fft.rfft(a, fs)
    return np.fft.irfft(fa * fa, fs)[:2 * len(a) - 1]


def _grad_tophard(conv, a, top_m):
    n = len(a)
    top_idx = np.argpartition(conv, -top_m)[-top_m:]
    grad = np.zeros(n)
    for k in top_idx:
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        lo = k - j_hi
        hi = k - j_lo
        grad[j_lo:j_hi + 1] += 2.0 * a[lo:hi + 1][::-1]
    return grad


def _optimize(a, deadline, rng, top_m=30):
    n = len(a)
    a = _normalize(a.astype(np.float64).copy())
    fs = 1
    while fs < 2 * n - 1:
        fs <<= 1

    best = a.copy()
    best_sc = _score(a)
    lr = 0.8
    it = 0
    stalled = 0
    eval_iv = 40

    while time.monotonic() < deadline:
        conv = _conv_aa(a, fs)
        grad = _grad_tophard(conv, a, top_m)
        gm = grad.max()
        if gm > 1e-15:
            a *= np.exp(-lr * grad / gm)
        a = np.maximum(a, 1e-15)
        a /= a.sum()

        it += 1
        if it % 30 == 0:
            lr *= 0.997
        if it % eval_iv == 0:
            sc = _score(a)
            if sc < best_sc - 1e-11:
                best_sc = sc
                best = a.copy()
                stalled = 0
            else:
                stalled += 1
            if stalled >= 15:
                mask = rng.random(n) < 0.25
                a[mask] *= np.exp(0.04 * rng.standard_normal(mask.sum()))
                a = _normalize(a)
                lr = min(lr * 1.6, 2.0)
                stalled = 0

    sc = _score(a)
    if sc < best_sc:
        best = a.copy()
        best_sc = sc
    return best, best_sc


def _build_sparse_A(a):
    n = len(a)
    rows_p = []
    cols_p = []
    vals_p = []
    for d in range(-(n - 1), n):
        val = a[d]
        if val < 1e-18:
            continue
        k_lo = max(0, d)
        k_hi = min(2 * n - 2, d + n - 1)
        ks = np.arange(k_lo, k_hi + 1)
        js = ks - d
        rows_p.append(ks)
        cols_p.append(js)
        vals_p.append(np.full(len(ks), val))
    rows = np.concatenate(rows_p)
    cols = np.concatenate(cols_p)
    vals = np.concatenate(vals_p)
    return coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()


def _lp_refine(a, deadline, n_lp=3):
    n = len(a)
    a = _normalize(a.copy())
    best = a.copy()
    best_sc = _score(a)
    for _ in range(n_lp):
        if time.monotonic() >= deadline:
            break
        conv = _conv_aa(a, 1 << (len(a).bit_length()))
        M = conv.max()
        A = _build_sparse_A(a)
        c = -np.ones(n)
        b_ub = M * np.ones(2 * n - 1)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'time_limit': 60, 'presolve': True})
        if not res.success:
            break
        b = np.maximum(res.x, 0.0)
        bs = b.sum()
        if bs < 1e-10:
            break
        b /= bs
        a = _normalize(0.5 * a + 0.5 * b)
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best = a.copy()
    return best, best_sc

    deadline = time.monotonic() + max(0.0, float(budget_s) - 8.0)

    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)

    if parent is None or parent.ndim != 1 or parent.size < 2 or not np.isfinite(parent).all():
        parent = None

    if parent is not None and parent.size > 10 and parent.sum() > 1e-10:
        n = parent.size
        n = n  # keep parent length
    else:
        n = 2535

    return [float(v) for v in best_all]
```
