    for pw in (-0.4, -0.8, -1.2, -1.8):
        L.append((j + 1.0) ** pw)
    for a in (0.5, 1.0, 1.5, 2.5, 4.0):
        L.append((1.0 - np.abs(j - m) / (m + 1.0)) ** a + 0.01)
    for f in (2, 5, 11):
        L.append(0.5 * (1.0 + np.cos(np.pi * j / max(n-1,1))) * (1+0.3*np.cos(2*np.pi*f*j/n)) + 0.01)
    t1, t2, sd = 0.35*n, 0.65*n, 0.12*n
    L.append(np.exp(-((j-t1)/sd)**2) + np.exp(-((j-t2)/sd)**2) + 0.005)
    d = np.minimum(j, n-1-j) + 1.0
    for pv in (0.5, 1.0, 2.0):
        L.append(d ** pv)
    return L

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    global DEADLINE, BEST_P, BEST_SC, GN, GNF
    t0 = time.monotonic()
    wall = t0 + budget_s - 55.0
    DEADLINE = wall
    rng = np.random.default_rng(seed)
    n = 2535
    parent = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64)
        if raw.ndim == 1 and raw.size > 0:
            n = int(raw.size)
            parent = normclip(raw)
    except Exception:
        pass
    if parent is None:
        parent = np.ones(n, dtype=np.float64)
    GN = n
    GNF = 1 << (2 * n - 1).bit_length()
    BEST_P = parent.copy()
    BEST_SC = float(evaluate_sequence(parent.tolist()))
    if not np.isfinite(BEST_SC):
        BEST_SC = 2.0
        BEST_P = np.ones(n, dtype=np.float64)
    upb(BEST_P)

    cands = gen_inits(n)
    cands.append(BEST_P.copy())
    cands.append(BEST_P[::-1].copy())
    cands.append((BEST_P + BEST_P[::-1]) / 2.0)
    cands.append(BEST_P * (1 + 0.05 * rng.standard_normal(n)))
    scored = []
    for a in cands:
        if time.monotonic() > wall - 100:
            break
        p = normclip(a)
        if p is None:
            continue
        sc = float(evaluate_sequence(p.tolist()))
        scored.append((sc, p))
    scored.sort(key=lambda x: x[0])
    starts = []
    for sc, p0 in scored:
        if len(starts) >= 5:
            break
        dup = False
        for q in starts:
            if np.linalg.norm(p0 - q) < 0.35:
                dup = True
                break
        if not dup:
            starts.append(p0.copy())

    for i, p0 in enumerate(starts):
        if time.monotonic() > wall - 25:
            break
        z0 = np.log(p0 + 1e-15)
        z0 -= z0.mean()
        z0 = np.clip(z0, -30, 30)
        t_rem = wall - 25 - i * 8
        lse_run(z0, max(t0 + 50, t_rem))

    upb(BEST_P)
    p_out = peak_shave(BEST_P.copy(), 1200)
    p_out = refine_active(p_out, 300)

    psym = (p_out + p_out[::-1]) / 2.0
    psym = normclip(psym)
    if psym is not None:
        p_s = peak_shave(psym, 600)
        p_s = refine_active(p_s, 200)

    for p0 in starts[:3]:
        if time.monotonic() > wall - 20:
            break
        p_r = peak_shave(p0.copy(), 500)
        p_r = refine_active(p_r, 150)

    upb(BEST_P)

    if BEST_P is None or BEST_P.size == 0 or not np.isfinite(BEST_P).all():
        BEST_P = np.ones(n, dtype=np.float64)

    final_sc = float(evaluate_sequence(BEST_P.tolist()))
    if not np.isfinite(final_sc) or final_sc > 2.0:
        BEST_P = np.ones(n, dtype=np.float64)

    return [float(x) for x in BEST_P]
