import time
from scipy import fft as sfft

def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def _fft_conv(a, fft_size):
    A = sfft.rfft(a, fft_size)
    return np.real(sfft.irfft(A * A, fft_size))[:2 * len(a) - 1]

def _cross_corr(p, a, fft_size):
    P = sfft.rfft(p, fft_size)
    A = sfft.rfft(a, fft_size)
    return np.real(sfft.irfft(P * np.conj(A), fft_size))[:len(a)]
