        hi = min(n - 1, k)
        gp[lo:hi + 1] += wi * p[k - hi:k - lo + 1][::-1]
    gp *= 2.0 * n
    dsum = float(p @ gp)
    return p * (gp - dsum)

def lse_run(z0, t_lim):
    global GB
    if sopt is None:
        return z0
    z = z0 - z0.mean()
    z = np.clip(z, -20, 20)
    bv = lse_obj(z)
    for beta in (32, 256, 2048):
        if time.monotonic() > t_lim:
            break
        GB = float(beta)
        try:
            r = sopt.minimize(lse_obj, z, jac=lse_grad, method='L-BFGS-B',
                             options={'maxiter': 150, 'ftol': 1e-16, 'gtol': 1e-12, 'maxls': 30})
            if r is not None and hasattr(r, 'x') and r.x is not None and np.isfinite(r.fun):
                if r.fun < bv:
                    bv = float(r.fun)
                z = np.asarray(r.x, dtype=np.float64)
        except Exception:
            pass
        p = np.exp(z - z.max())
        p /= p.sum()
        upb(p)
    return z

def make_inits(n):
    L = []
    j = np.arange(n, dtype=np.float64)
    m = 0.5 * (n - 1)
    L.append(np.ones(n, dtype=np.float64))
    L.append(0.5 * (1.0 + np.cos(np.pi * j / (n - 1))))
    for p in (-0.5, -1.2, -2.0):
        L.append((j + 1.0) ** p)
    for a in (0.5, 1.0, 2.5):
        L.append((1.0 - np.abs(j - m) / (m + 1.0)) ** a + 0.01)
    t1, t2, sd = 0.3 * n, 0.7 * n, 0.08 * n
    L.append(np.exp(-((j - t1) / sd) ** 2) + np.exp(-((j - t2) / sd) ** 2) + 0.005)
    d = np.minimum(j, n - 1 - j) + 1.0
    L.append(d ** 0.5)
    L.append(d ** 1.5)
    L.append(0.5 * (1.0 + np.cos(np.pi * j / (n - 1))) * (1.0 + 0.3 * np.cos(2.0 * np.pi * 2.0 * j / n)) + 0.01)
    return L
