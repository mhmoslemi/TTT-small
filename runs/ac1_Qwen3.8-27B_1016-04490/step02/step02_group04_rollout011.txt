import numpy as np
import time


def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _proj_simplex(v, c):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    rho = np.nonzero(u * idx > (cssv - c))[0]
    if len(rho) == 0:
        return np.full(n, c / n)
    rho = rho[-1]
    theta = (cssv[rho] - c) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
