    # Polishing: exact subgradient, decreasing step
    if time.monotonic() < deadline - 12.0:
        x = best.copy()
        for m_p, step_p in [(1, 0.02), (1, 0.01), (1, 0.005), (2, 0.004), (1, 0.003), (1, 0.001)]:
            if time.monotonic() > deadline - 12.0:
                break
            for _ in range(300):
                if time.monotonic() > deadline - 12.0:
                    break
                conv = fft_conv(x, fs)
                cur = 2.0 * n * conv.max()
                if cur < best_score - 1e-13:
                    best_score = cur
                    best = x.copy()
                subgrad(conv, x, m_p, work)
                gn = np.linalg.norm(work)
                if gn < 1e-18:
                    break
                x_new = np.maximum(x - step_p * x.mean() * work / gn, 0.0)
                sl = x_new.sum()
                if sl < 1e-14:
                    break
                x_new = x_new / sl
                conv_n = fft_conv(x_new, fs)
                if 2.0 * n * conv_n.max() <= cur:
                    x = x_new
                else:
                    x = x_new * 0.5 + x * 0.5
                    s2 = x.sum()
                    if s2 > 1e-14:
                        x = x / s2

    sc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(sc):
        pn = np.maximum(parent, 0.0)
        pn = pn / pn.sum()
        sc2 = float(evaluate_sequence(pn.tolist()))
        if np.isfinite(sc2):
            return [float(v) for v in pn]
        return [1.0]
    return [float(v) for v in best]
