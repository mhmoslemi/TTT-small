import numpy as np

from scipy import fft as sfft

def npow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def norm_seq(v, ts, n):
    v = np.clip(v, 0.0, 1000.0)
    s = v.sum()
    if s < 0.01:
        return np.full(n, ts / n)
    v *= ts / s
    return v

def do_autocorr(x, fs, n):
    X = sfft.rfft(x, fs)
    c = np.real(sfft.irfft(X * X, fs))[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def do_grad(p, x, cs, n):
