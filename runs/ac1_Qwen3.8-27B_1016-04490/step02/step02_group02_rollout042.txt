    p = 1
    while p < m:

    return p

    A = sfft.rfft(a, fs)
    out = sfft.irfft(A * A, fs).real[:2 * len(a) - 1]
    return np.maximum(out, 0.0)


def topm_grad(conv, a, m, work):
    n = len(a)
    L = len(conv)
    k = min(m, L)
    idx = np.argpartition(conv, -k)[-k:]
    vals = conv[idx]
    vs = vals.sum()

        work[:n] = 0.0
        return
    ws = vals / vs
    work[:n] = 0.0

        kk = int(idx[j])
        w = float(ws[j])
        lo = max(0, kk - (n - 1))
        hi = min(kk, n - 1)
        if lo <= hi:
