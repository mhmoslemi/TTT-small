    X = sfft.rfft(x, fs)

def _grad(x, p, fs, n):
    X = sfft.rfft(x, fs)
    pr = p[::-1]
    P = sfft.rfft(pr, fs)
    gr = np.real(sfft.irfft(P * X, fs))
    g = 2.0 * gr[n - 1:2 * n - 1][::-1]
    g -= g.mean()
    return g, X


def _softmax_weights(conv, T, topk):
    cm = conv.max()
    if topk is not None and topk < len(conv):
        idx = np.argpartition(conv, -topk)[-topk:]
        mask = np.zeros(len(conv), dtype=bool)
        mask[idx] = True
        c2 = np.where(mask, conv, -np.inf)
    else:
        c2 = conv
    sc = (c2 - cm) / T
    ex = np.exp(sc - sc.max())
    es = ex.sum()
    if es < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(conv)] = 1.0
        return p
    return ex / es


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    dl = time.monotonic() + max(0.0, float(budget_s) - 20.0)

    par = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if par.ndim != 1 or par.size == 0 or not np.all(np.isfinite(par)):
        par = np.ones(1024)
    n = par.size
    tgt = np.sqrt(2.0 * n)
    fs = _np2(2 * n + 1)
    jj = np.arange(n, dtype=np.float64)
    clen = 2 * n - 1

    # Generate diverse initializations
    cands = [_proj(par.copy(), tgt)]
    # Smooth parametric perturbations
    for f in [1.0, 2.0, 3.0, 5.0]:
        for ph in [0.0, 1.0, 2.5]:
            w = 0.15 * np.sin(2 * np.pi * f * jj / n + ph)
            cands.append(_proj(par + tgt * w, tgt))
    # Edge/center weighting
    cands.append(_proj(par + 0.2 * tgt * np.cos(np.pi * jj / (n - 1)), tgt))
    cands.append(_proj(par - 0.15 * tgt * np.sin(np.pi * jj / (n - 1)), tgt))
    # Random smooth
    for _ in range(6):
        freqs = rng.uniform(0.5, 8, 3)
        amps = rng.uniform(0.03, 0.12, 3)
        phs = rng.uniform(0, 6.28, 3)
        w = sum(a * np.sin(2 * np.pi * f * jj / n + p)
                for a, f, p in zip(amps, freqs, phs))
        cands.append(_proj(par + tgt * w, tgt))

    best = cands[0].copy()
    best_sc = _sc(best)
    x = best.copy()
    for c in cands[1:]:
        s = _sc(c)
        if s < best_sc:
            best_sc = s
            best = c.copy()
            x = c.copy()

    # Main optimization with adaptive parameters
    T = 0.06
    lr = 0.45
    mom = 0.82
    vel = np.zeros(n)
    it = 0
    stale = 0
    topk = 8

    while time.monotonic() < dl - 10.0:
        conv = _conv(x, fs, n)
        cm = conv.max()

        if cm < best_sc - 1e-13:
            best_sc = cm
            best = x.copy()
            stale = 0
        else:
            stale += 1

        # Adaptive topk: focus more as we converge
        if stale < 2000:
            topk_eff = 5
        elif stale < 8000:
            topk_eff = 10
        else:
            topk_eff = None  # full softmax

        p = _softmax_weights(conv, T, topk_eff)
        g, X = _grad(x, p, fs, n)

        # Shrink learning rate in later stages
        lr_eff = lr * (1.0 / (1.0 + 0.01 * stale))
        vel = mom * vel - lr_eff * g
        x = np.clip(x + vel, 0.0, 1000.0)
        sx = x.sum()
        if sx < 1e-10:
            x = np.full(n, tgt / n)
        else:
            x = x * (tgt / sx)

        it += 1

        # Anneal temperature
        T = max(0.0003, T * 0.9998)
        if it > 10000:
            lr = max(0.03, lr * 0.9997)
        # Decay momentum to avoid oscillation
        mom = max(0.5, mom * 0.99995)

        # Periodic small perturbation
        if it % 4000 == 0:
            x += rng.standard_normal(n) * 0.003 * max(x.std(), 0.01)
            x = _proj(x, tgt)

        # Restart if stuck
        if stale > 12000:
            scale = 0.02 * max(best.std(), 0.01)
            x = best + rng.normal(0, scale, n)
            x = _proj(x, tgt)
            T = max(0.02, best_sc * 0.05)
            lr = 0.4
            mom = 0.8
            vel = np.zeros(n)
            stale = 0

    # Fine-tune phase: very low temperature, small steps
    T2 = 0.0005
    lr2 = 0.08
    mom2 = 0.6
    vel2 = np.zeros(n)
    x = best.copy()
    for _ in range(5000):
        if time.monotonic() > dl:
            break
        conv = _conv(x, fs, n)
        cm = conv.max()
        if cm < best_sc - 1e-13:
            best_sc = cm
            best = x.copy()

        p = _softmax_weights(conv, T2, 3)
        g, X = _grad(x, p, fs, n)
        vel2 = mom2 * vel2 - lr2 * g
        x = np.clip(x + vel2, 0.0, 1000.0)
        sx = x.sum()
        if sx > 1e-10:
            x = x * (tgt / sx)
        lr2 = max(0.005, lr2 * 0.9995)

    # Final verification
    fs_final = _sc(best)
    if not np.isfinite(fs_final):
        par_sc = _sc(par)
        if np.isfinite(par_sc):
            return par.tolist()
        return [float(tgt / n)] * n
    return best.tolist()
