        w = w / w.sum()
    else:
        w = np.full(k, 1.0 / k)
    work[:] = 0.0
    for j in range(k):
        kk = int(idx[j])
        wj = float(w[j])
        lo = kk - (n - 1)
        if lo < 0:
            lo = 0
        hi = kk
        if hi > n - 1:
            hi = n - 1
        if lo <= hi:
            seg = x[kk - hi:kk - lo + 1][::-1]
            work[lo:hi + 1] += 2.0 * wj * seg
    return work


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    deadline = time.monotonic() + max(60.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(seed)
    n = len(height_sequence_1)
    fs = next_pow2(2 * n + 1)

    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    ps = parent.sum()
    if ps < 0.01 or not np.isfinite(parent).all():
        x = np.ones(n) / n
    else:
        x = parent / ps

    conv = fft_autoconv(x, fs)
    best_score = 2.0 * n * conv.max()
    best = x.copy()

    work = np.zeros(n)
    vel = np.zeros(n)
    m = 6
    step_frac = 0.05
    beta = 0.3
    no_imp = 0
    it = 0

    while time.monotonic() < deadline - 10.0:
        conv = fft_autoconv(x, fs)
        cur = 2.0 * n * conv.max()

        if cur < best_score - 1e-14:
            best_score = cur
            best = x.copy()
            no_imp = 0
        else:
            no_imp += 1

        grad = compute_grad(conv, x, m, work)
        grad -= grad.mean()
        vel = beta * vel + grad
        vn = np.linalg.norm(vel)
        if vn > 1e-18:
            x = np.maximum(x - step_frac * x.mean() * vel / vn, 0.0)
            s = x.sum()
            if s < 1e-14:
                x = np.ones(n) / n
            else:
                x = x / s
        else:
            no_imp += 500

        it += 1

        if no_imp >= 2500:
            pert = 0.02 + 0.015 * min(it // 25000, 4)
            x = best + pert * rng.standard_normal(n)
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            vel[:] = 0.0
            no_imp = 0
            step_frac = max(0.01, step_frac * 0.85)
            m = min(25, m + 2)

        if it % 4000 == 0:
            sc = float(evaluate_sequence(x.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best = x.copy()
            if best_score <= 1.50275:
                break

    # Polish phase: focus on the single dominant peak
    if time.monotonic() < deadline - 5.0:
        x = best.copy()
        m2 = 2
        for _ in range(20000):
            if time.monotonic() > deadline - 5.0:
                break
            conv = fft_autoconv(x, fs)
            c2 = 2.0 * n * conv.max()
            if c2 < best_score - 1e-14:
                best_score = c2
                best = x.copy()
            grad = compute_grad(conv, x, m2, work)
            grad -= grad.mean()
            gn = np.linalg.norm(grad)
            if gn < 1e-18:
                break
            x = np.maximum(x - 0.006 * x.mean() * grad / gn, 0.0)
            s = x.sum()
            if s < 1e-14:
                break
            x = x / s

    # Second polish: top-3 with very small steps
    if time.monotonic() < deadline - 3.0:
        x = best.copy()
        for _ in range(10000):
            if time.monotonic() > deadline - 3.0:
                break
            conv = fft_autoconv(x, fs)
            c2 = 2.0 * n * conv.max()
            if c2 < best_score - 1e-14:
                best_score = c2
                best = x.copy()
            grad = compute_grad(conv, x, 3, work)
            grad -= grad.mean()
            gn = np.linalg.norm(grad)
            if gn < 1e-18:
                break
            x = np.maximum(x - 0.004 * x.mean() * grad / gn, 0.0)
            s = x.sum()
            if s < 1e-14:
                break
            x = x / s

    sc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(sc):
        return [1.0] * n
    return [float(v) for v in best]
