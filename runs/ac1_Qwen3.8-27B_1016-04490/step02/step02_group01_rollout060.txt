import time
from scipy import fft as sfft


def npow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def autocorr(x, fs, L):
    X = sfft.rfft(x, fs)
    c = np.real(sfft.irfft(X * X, fs))[:L]
    np.maximum(c, 0.0, out=c)
    return c


def xgrad(p, x, cs, n):
    P = sfft.rfft(p, cs)
    Xc = sfft.rfft(x, cs)
    g = 2.0 * np.real(sfft.irfft(P * np.conj(Xc), cs))[:n]
    g -= g.mean()
    return g


def renorm(x, ts):
    np.clip(x, 0.0, 1000.0, out=x)
    s = x.sum()
    if s < 0.01:
        return False
    x *= ts / s
    np.clip(x, 0.0, 1000.0, out=x)
    return True


def compute_p(c, T, L):
    cm = c.max()
    e = np.exp((c - cm) / T)
    se = e.sum()
    if se > 1e-300:
        return e / se
    p = np.zeros(L)
    p[int(np.argmax(c))] = 1.0
    return p
