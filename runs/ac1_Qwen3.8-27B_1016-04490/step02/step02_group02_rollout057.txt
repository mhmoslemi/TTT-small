import time
from scipy import fft as sfft


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def fft_conv(a, fs):
    A = sfft.rfft(a, fs)
    return np.maximum(sfft.irfft(A * A, fs).real[:2 * len(a) - 1], 0.0)


def topm_grad(conv, x, m, work):
    n = len(x)
    k = min(m, len(conv))
    idx = np.argpartition(conv, -k)[-k:]
    vs = conv[idx].sum()
    if vs < 1e-30:
        work[:] = 0.0
        return
    ws = conv[idx] / vs
    work[:] = 0.0
    for j in range(k):
        kk = int(idx[j])
        w = 2.0 * float(ws[j])
        lo = max(0, kk - n + 1)
        hi = min(kk, n - 1)
        if lo <= hi:
            work[lo:hi + 1] += w * (x[lo:hi + 1] + x[kk - hi:kk - lo + 1][::-1])


def soft_grad(conv, x, T, fs):

    p = np.exp((conv - conv.max()) / T)
    p /= p.sum()
    p_r = p[::-1]
    S1f = sfft.irfft(sfft.rfft(p, fs) * sfft.rfft(np.ones(n), fs), fs).real

    S2 = S2f[n - 1:2 * n - 1][::-1]
    return S1 * x + S2
