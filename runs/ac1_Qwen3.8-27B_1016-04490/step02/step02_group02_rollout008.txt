        x = line_search_min(x, vel, cur, fs, n)
        it1 += 1
        if cur < best_score - 1e-12:
            best_score = cur
            best = x.copy()
            no_imp = 0
        else:
            no_imp += 1
        if no_imp >= IMP1:
            x = best + 0.04 * rng.standard_normal(n) / n
            x = np.maximum(x, 1e-12)
            x = x / x.sum()
            vel[:] = 0.0
            no_imp = 0
            m1 = min(128, m1 + 16)
        if it1 % 3000 == 0:
            m1 = min(256, m1 + 4)

    # Phase 2: medium with top-m=4, smaller steps
    x = best.copy()
    vel[:] = 0.0
    m2 = 4
    step2 = 0.02
    beta2 = 0.3
    it2 = 0
    no_imp = 0
    IMP2 = 800
    phase2_end = time.monotonic() + (deadline - time.monotonic()) * 0.35

    while time.monotonic() < phase2_end:
        conv = fft_conv(x, fs)
        cur = 2.0 * n * conv.max()
        grad = topm_grad(conv, x, m2, work)
        grad -= grad.mean()
        vel = beta2 * vel + grad
        gn = np.linalg.norm(vel)
        if gn > 1e-18:
            x_try = np.maximum(x - step2 * x.mean() * vel / gn, 0.0)
            s = x_try.sum()
            if s > 1e-14:
                x = x_try / s
        it2 += 1
        if cur < best_score - 1e-13:
            best_score = cur
            best = x.copy()
            no_imp = 0
        else:
            no_imp += 1
        if no_imp >= IMP2:
            x = best + 0.02 * rng.standard_normal(n) / n
            x = np.maximum(x, 1e-12)
            x = x / x.sum()
            vel[:] = 0.0
            no_imp = 0
            m2 = min(16, m2 + 2)
        if it2 % 2000 == 0:
            step2 = max(0.004, step2 * 0.97)
        if it2 % 10000 == 0:
            x = 0.9995 * x + 0.0005 / n
            x = x / x.sum()

    # Phase 3: fine with m=1 exact subgradient, very small steps
    x = best.copy()
    step3 = 0.006
    it3 = 0
    while time.monotonic() < deadline - 8.0:
        conv = fft_conv(x, fs)
        c2 = 2.0 * n * conv.max()
        if c2 < best_score - 1e-14:
            best_score = c2
            best = x.copy()
        grad = topm_grad(conv, x, 1, work)
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-18:
            break
        x = np.maximum(x - step3 * x.mean() * grad / gn, 0.0)
        s = x.sum()
        if s < 1e-14:
            break
        x = x / s
        step3 *= 0.99995
        it3 += 1

    # Phase 4: ultra-fine with m=2
    x = best.copy()
    step4 = 0.003
    while time.monotonic() < deadline - 3.0:
        conv = fft_conv(x, fs)
        c2 = 2.0 * n * conv.max()
        if c2 < best_score - 1e-14:
            best_score = c2
            best = x.copy()
        grad = topm_grad(conv, x, 2, work)
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-18:
            break
        x = np.maximum(x - step4 * x.mean() * grad / gn, 0.0)
        s = x.sum()
        x = x / s if s > 1e-14 else np.ones(n) / n
        step4 *= 0.99998

    sc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(sc):
        pn = np.maximum(parent, 0.0)
        pn = pn / pn.sum()
        sc2 = float(evaluate_sequence(pn.tolist()))
        if np.isfinite(sc2):
            return [float(v) for v in pn]
        return [1.0]
    return [float(v) for v in best]
