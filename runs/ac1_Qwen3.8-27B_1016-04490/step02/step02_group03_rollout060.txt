import numpy as np
import time


def _proj(v):
    """Project vector v onto the probability simplex {x>=0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    ar = np.arange(1, n + 1, dtype=np.float64)
    rho = int(np.sum(u * ar >= cv - 1.0))
    if rho == 0:
        return np.ones(n) / n
    tau = (cv[rho - 1] - 1.0) / rho
    out = np.maximum(v - tau, 0.0)
    return out


def _conv(a, N):
    """Full linear self-convolution of a via zero-padded FFT."""
    fa = np.fft.rfft(a, N)
    return np.real(np.fft.irfft(fa * fa, N))[:2 * a.size - 1]


def _grad_from_peaks(a, c, beta, K):
    """Smoothed log-sum-exp gradient restricted to the K largest weights."""
    n = a.size
    cm = c.max()
    w = np.exp(beta * (c - cm))
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= s
    K = min(K, c.size)
    top = np.argpartition(-w, K - 1)[:K] if K < c.size else np.arange(c.size)
    g = np.zeros(n, dtype=np.float64)
    for k in top:
        wk = 2.0 * w[k]
        if wk <= 0.0:
            continue
        lo = k - (n - 1)
        if lo < 0:
            lo = 0
        hi = k
        if hi > n - 1:
            hi = n - 1
        if lo <= hi:
            g[lo:hi + 1] += wk * a[k - hi:k - lo + 1][::-1]
    return g
