    it = 0
    stalled = 0
    
    while time.monotonic() < deadline:
        g, c = _grad(a, tau, fft_sz)
        gm = g.max()
        if gm > 1e-15:
            g_norm = g / gm
            a *= np.exp(-lr * g_norm)
        a = np.maximum(a, 1e-15)
        a /= a.sum()
        
        it += 1
        if it % 50 == 0:
            tau *= 0.995
            lr *= 0.998
            if tau < 5e-5 / n:
                tau = 5e-5 / n
        
        if it % eval_every == 0:
            sc = _score(a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best = a.copy()
                stalled = 0
            else:
                stalled += 1
            if stalled >= 15:
                mask = rng.random(n) < 0.25
                a[mask] *= np.exp(0.03 * rng.standard_normal(int(mask.sum())))
                a = _normalize(a)
                stalled = 0
    
    sc = _score(a)
    if sc < best_sc:
        best = a.copy()
        best_sc = sc
    return best, best_sc
