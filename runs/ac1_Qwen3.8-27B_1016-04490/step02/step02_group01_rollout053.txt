from scipy import fft as sfft

def npow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def autocorr(x, fs, n):
    X = sfft.rfft(x, fs)
    c = np.real(sfft.irfft(X * X, fs))[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def make_grad(p, x, cs, n):
    P = sfft.rfft(p, cs)
    Xc = sfft.rfft(x, cs)
    g = 2.0 * np.real(sfft.irfft(P * np.conj(Xc), cs))[:n]
    g -= g.mean()
    return g
