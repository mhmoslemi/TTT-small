        p = softmax_wts(c, T, K)
        g = xgrad(p, x, cs, n)

        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break

        # Adaptive step with momentum
        step = LR * (ts / n)
        v = beta * v - step * (g / gn)
        v -= v.mean()

        x_new = normalize(x + v, ts)
        T = max(Tm, T * 0.9997)
        LR = max(LR_min, LR * 0.99998)
        x = x_new
        it += 1

        # Periodic exact evaluation
        if it % 800 == 0:
            sv = sc(x)
            if np.isfinite(sv) and sv < bsc - 1e-12:
                bsc = sv
                best = x.copy()
                stuck = 0
                v *= 0.5  # damp momentum on improvement
            else:
                stuck += 800
            restarts = 0
            if stuck > SL:
                # Asymmetric perturbation restart
                d = rng.standard_normal(n)
                d += 0.4 * np.linspace(-1, 1, n)
                d /= np.linalg.norm(d)
                x = normalize(best + 0.05 * (ts / n) * d, ts)
                v = np.zeros(n)
                T = max(0.05, bsc * 0.08)
                LR = 0.45
                stuck = 0
                restarts += 1
                # Limit total restarts
                if restarts > 0 and stuck == 0 and it > 50000:
                    # Reduce further restarts by increasing patience
                    SL = 12000

    # Final polish: targeted peak reduction
    if time.monotonic() < dl - 3.0:
        best, bsc = polish_peaks(best, fs, cs, n, ts, bsc, dl)

    if not np.isfinite(bsc):
        x = np.clip(x, 0.0, 1000.0)
        if x.sum() < 0.01:
            x = np.ones(n)
        return [float(vv) for vv in x]
    return [float(vv) for vv in best]
