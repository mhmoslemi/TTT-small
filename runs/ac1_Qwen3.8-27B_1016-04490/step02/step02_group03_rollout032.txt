import time
import scipy.optimize as sopt
import scipy.sparse as sp

DEADLINE = 0.0
BEST_P = None
BEST_SC = float('inf')
GN = 0
GNF = 0
BETA = 4.0

def fft_conv(a, N):
    n = a.size
    fa = np.fft.rfft(a, N)
    return np.real(np.fft.irfft(fa * fa, N)[:2 * n - 1])

def get_score(p):
    c = fft_conv(p, GNF)
    return float(2.0 * GN * c.max())

def upd(p):
    global BEST_P, BEST_SC
    if p is None or p.size == 0:
        return
    p = np.clip(p.astype(np.float64), 0.0, 1000.0)
    s = p.sum()
    if s < 0.01:
        return
    p = p / s
    sc = get_score(p)
    if np.isfinite(sc) and sc < BEST_SC:
        BEST_P = p.copy()
        BEST_SC = sc

def lse_f(z):
    n = GN
    beta = BETA
    zm = z - z.max()
    p = np.exp(zm)
    s = p.sum()
    if s < 1e-300:
        return 1e100
    p = p / s
    c = fft_conv(p, GNF)
    u = n * c
    um = u.max()
    ex = np.exp(beta * (u - um))
    s2 = ex.sum()
    if s2 < 1e-300:
        return 1e100
    return (beta * um + np.log(s2)) / beta

def lse_g(z):
    n = GN
    beta = BETA
    zm = z - z.max()
    p = np.exp(zm)
    s = p.sum()
    if s < 1e-300:
        return np.zeros(n)
    p = p / s
    c = fft_conv(p, GNF)
    u = n * c
    um = u.max()
    ex = np.exp(beta * (u - um))
    s2 = ex.sum()
    if s2 < 1e-300:
        return np.zeros(n)
    w = ex / s2
    fp = np.fft.rfft(p, GNF)
    fw = np.fft.rfft(w, GNF)
    t1 = np.fft.irfft(np.conj(fp) * fw, GNF)[:n]
    t2 = np.fft.irfft(fp * fw, GNF)[:n]
    gp = float(n) * (t1 + t2)
    d = float(p @ gp)
    return p * (gp - d)

def lse_optimize(z0):
    global BETA
    z = z0 - z0.mean()
    z = np.clip(z, -30, 30)
    vb = lse_f(z)
    stages = [(4, 30), (8, 40), (16, 55), (32, 65), (64, 80),
              (128, 100), (256, 130), (512, 160), (1024, 200), (2048, 250)]
    for b, mi in stages:
        if time.monotonic() > DEADLINE - 40:
            break
        BETA = float(b)
        try:
            r = sopt.minimize(lse_f, z, jac=lse_g, method='L-BFGS-B',
                              callback=timeout_cb,
                              options={'maxiter': mi, 'ftol': 1e-15,
                                       'gtol': 1e-11, 'maxls': 40})
            if hasattr(r, 'x') and r.x is not None and np.isfinite(r.fun):
                if r.fun <= vb + 1e-9:
                    z = np.clip(r.x, -40, 40)
                    vb = float(r.fun)
        except Exception:
            z = np.clip(z, -40, 40)
            break
    p = np.exp(z - z.max())
    p /= p.sum()
    upd(p)
    return z

def peak_shave(p, max_iters=500):
    p = p.copy()

        g *= 2.0 * GN
        d = float(p @ g)
        g0 = g - d

def lp_step(p, M, tl):
    n = GN
    L = 2 * n - 1
    r_list, c_list, v_list = [], [], []
    for k in range(L):
        jlo = max(0, k - n + 1)
        jhi = min(n - 1, k)
        cnt = jhi - jlo + 1
        r_list.append(np.full(cnt, k, dtype=np.int32))
        c_list.append(np.arange(jlo, jhi + 1, dtype=np.int32))
        v_list.append(p[k - jhi:k - jlo + 1])
    A = sp.csr_matrix((np.concatenate(v_list),
                       (np.concatenate(r_list), np.concatenate(c_list))),
                      shape=(L, n))
    res = sopt.linprog(-np.ones(n), A_ub=A, b_ub=np.full(L, M),
                       bounds=[(0, None)] * n, method='highs',
                       options={'time_limit': max(1, int(tl))})
    if res.status == 0:
        b = np.maximum(res.x, 0)
        s = b.sum()
        if s > 1e-15:
            return b / s
    return None

def make_inits(n, rng, parent):
    cands = []
    j = np.arange(n, dtype=np.float64)
    m = 0.5 * (n - 1)
    cands.append(np.ones(n))
    cands.append(0.5 * (1 + np.cos(np.pi * j / (n - 1))))
    cands.append(0.5 * (1 - np.cos(np.pi * j / (n - 1))) + 0.01)
    for pw in [-0.3, -0.6, -1.0, -1.5]:
        cands.append((j + 1) ** pw)
    for a in [0.5, 1.0, 1.5, 2.0, 3.0]:
        cands.append((1.0 - np.abs(j - m) / (m + 1)) ** a + 0.01)
    for f in [1, 2, 3, 5]:
        mod = 1 + 0.3 * np.cos(2 * np.pi * f * j / n)
        cands.append(0.5 * (1 + np.cos(np.pi * j / (n - 1))) * mod + 0.01)
    d = np.minimum(j, n - 1 - j) + 1
    for pv in [0.5, 1.0, 2.0]:
        cands.append(d ** pv)
    if parent is not None:
        cands.append(parent.copy())
        for _ in range(3):
            nz = rng.standard_normal(n) * 0.08
            pp = np.maximum(parent + nz, 0)
            s = pp.sum()
            if s > 0.01:
                cands.append(pp / s)
    return cands
