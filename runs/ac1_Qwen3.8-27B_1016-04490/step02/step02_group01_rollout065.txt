from scipy import fft as sfft

def npow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def autocorr(x, fs, n):
    X = sfft.rfft(x, fs)
    c = np.real(sfft.irfft(X * X, fs))[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def fast_score(c, n, sa):
    if sa < 0.01:
        return np.inf
    return float(2.0 * n * c.max() / (sa * sa))

def proj_simplex(v, S):
    n = v.size
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    ar = np.arange(1, n + 1, dtype=np.float64)
    idx = np.nonzero(u * ar > (cv[-1] - S))[0]
    if len(idx) == 0:
        t = np.maximum(v, 0.0)
        s = t.sum()
        if s < 1e-300:
            return np.full(n, S / n)
        return t * (S / s)
    rho = idx[0]
    theta = (cv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
