                do_normalize(cand, ts)
                s_ = fast_score(cand, n)
                if np.isfinite(s_) and s_ < bsc - 1e-14:
                    bsc = s_
                    best = cand.copy()
                    improved = True
                    break
            if improved:
                peak_list.append(peak_list.pop(0))
                break
        if not improved:
            # Try one more cycle with softened peak
            c = autocorr(best, fs, n)
            p2 = softmax_wts(c, 0.05, 64)
            g = xgrad(p2, best, cs, n)
            gn = np.linalg.norm(g)
            if gn < 1e-15:
                break
            for st in (0.005, 0.002, 0.001):
                cand = best - st * (ts / n) * g / gn
                do_normalize(cand, ts)
                s_ = fast_score(cand, n)
                if np.isfinite(s_) and s_ < bsc - 1e-14:
                    bsc = s_
                    best = cand.copy()
                    improved = True
                    break
        if not improved:
            break

    # Final exact evaluation with the authoritative evaluator
    try:
        s_final = evaluate_sequence([float(v) for v in best])
        if np.isfinite(s_final):
            bsc = s_final
    except Exception:
        pass

    do_normalize(best, ts)
    if best.sum() < 0.01:
        best = np.ones(n) * 1.0
    return [float(v) for v in best]
