            if sc < best_score - 1e-14:
                best_score = sc
                best = a.copy()
                stalled = 0
            else:
                stalled += 1
            lr *= 0.997
            if stalled >= 10:
                a = best.copy()
                mask = rng.random(n) < 0.2
                cnt = int(mask.sum())
                if cnt > 0:
                    a[mask] *= np.exp(0.08 * rng.standard_normal(cnt))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                stalled = 0
                lr = 1.0
            if abs(best_score - 2.0) < 1e-7:
                break

    sc = _true_score(a)
    if sc < best_score:
        best = a.copy()
        best_score = sc
    return best, best_score


def _get_parent():
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim != 1 or p.size < 2 or not np.isfinite(p).all():
            p = np.ones(2535, dtype=np.float64)
    except NameError:
        p = np.ones(2535, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.maximum(p, 0.0)
    if p.sum() < 1e-10:
        p = np.ones(p.size, dtype=np.float64)
    p = p / p.sum()
    return p


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 10.0)

    parent = _get_parent()
    n = parent.size

    j = np.arange(n, dtype=np.float64)
    x = j / max(n - 1, 1)

    inits = [
        parent.copy(),
        np.ones(n, dtype=np.float64),
        j + 1.0,
        n - j,
        np.exp(2.0 * x),
        1.0 + 0.5 * np.sin(np.pi * x),
        (1.0 + 0.3 * np.cos(2.0 * np.pi * x)) ** 2,
    ]

    scored = []
    for ia in inits:
        ia = np.maximum(ia.astype(np.float64, copy=True), 0.0)
        si = ia.sum()
        if si < 1e-10:
            continue
        ia = ia / si
        sc = _true_score(ia)
        if np.isfinite(sc):
            scored.append((sc, ia))
    scored.sort()

    if not scored:
        return [1.0] * n

    best_all = scored[0][1].copy()
    best_sc = scored[0][0]

    rem = max(deadline - time.monotonic(), 1.0)
    top_k = min(4, len(scored))
    time_per = rem / (top_k + 1.0)

    for k in range(top_k):
        if time.monotonic() >= deadline - 40.0:
            break
        dl = min(time.monotonic() + time_per, deadline - 40.0)
        b, sc = _grad_desc(scored[k][1].copy(), dl, rng)
        if sc < best_sc:
            best_sc = sc
            best_all = b.copy()

    rem2 = deadline - time.monotonic()
    if rem2 > 40.0:
        b2, sc2 = _grad_desc(best_all.copy(), deadline - 5.0, rng)
        if sc2 < best_sc:
            best_sc = sc2
            best_all = b2.copy()

    if not np.isfinite(best_all).all() or best_all.sum() < 0.01:
        best_all = np.ones(n, dtype=np.float64)

    fs = evaluate_sequence(best_all.tolist())
    if not np.isfinite(fs):
        best_all = np.ones(n, dtype=np.float64)

    return [float(v) for v in best_all]
