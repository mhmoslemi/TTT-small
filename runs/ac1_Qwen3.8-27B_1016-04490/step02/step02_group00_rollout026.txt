import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix


def _fft_conv(x, y):
    m = len(x) + len(y) - 1
    sz = 1
    while sz < m:
        sz <<= 1
    return np.fft.irfft(np.fft.rfft(x, sz) * np.fft.rfft(y, sz), sz)[:m]


def _score(a):
    return float(evaluate_sequence(a.tolist()))


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 1e-15 else np.ones_like(a) / len(a)


def _grad_dir(a, tau, fft_sz):
    n = a.size
    fx = np.fft.rfft(a, fft_sz)
    sc = np.fft.irfft(fx * fx, fft_sz)[:2 * n - 1]
    z = sc / tau - sc.max()
    e = np.exp(z)
    ss = e.sum()
    p = e / ss if ss > 1e-300 else np.full(2 * n - 1, 1.0 / (2 * n - 1))
    fy = np.fft.rfft(p[::-1], fft_sz)
    ca = np.fft.irfft(fx * fy, fft_sz)[:3 * n - 2]
    return 2.0 * ca[n - 1:2 * n - 1][::-1]


def _lp_dir(f, M, topk, deadline):
    n = len(f)
    cv = _fft_conv(f, f)
    idx = np.argsort(cv)[::-1][:topk]
    A = lil_matrix((topk, n))
    for r, k in enumerate(idx):
        lo, hi = max(0, k - n + 1), min(n, k + 1)
        if hi > lo:
            cols = np.arange(lo, hi)
            A[r, cols] = f[k - cols]
    A = A.tocsr()
    tl = deadline - time.monotonic()
    if tl < 4.0:
        return None
    res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(topk, M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'time_limit': max(1.0, min(tl * 0.35, 20.0))})
    if res.success and res.x is not None and np.isfinite(res.x).all():
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def _line_search(f, g, cur_sc, deadline):
    best_t, best_sc = 0.0, cur_sc
    for t in (0.1, 0.25, 0.5, 0.75, 1.0):
        if time.monotonic() > deadline - 5.0:
            break
        c = _normalize((1 - t) * f + t * g)
        sc = _score(c)
        if sc < best_sc:
            best_t, best_sc = t, sc
    return best_t, best_sc


def _grad_refine(a, deadline, rng, eval_every=150):
    n = a.size
    fft_sz = 1
    while fft_sz < 2 * n - 1:
        fft_sz <<= 1
    tau, lr = 2.0 / n, 1.0
    best, best_sc = a.copy(), _score(a)
    stalled, cnt = 0, 0
    while time.monotonic() < deadline:
        g = _grad_dir(a, tau, fft_sz)
        gm = g.max()
        if gm > 1e-15:
            a *= np.exp(-lr * g / gm)
        a = _normalize(np.maximum(a, 1e-15))
        tau = max(tau * 0.996, 0.3 / n)
        lr = min(lr * 1.0005, 4.0)
        cnt += 1
        if cnt % eval_every == 0:
            sc = _score(a)
            if sc < best_sc - 1e-9:
                best, best_sc, stalled = a.copy(), sc, 0
            else:
                stalled += 1
            if stalled >= 12:
                m = rng.random(n) < 0.25
                a[m] *= np.exp(0.02 * rng.standard_normal(m.sum()))
                a = _normalize(a)
                tau = 1.5 / n
                stalled = 0
    sc = _score(a)
    if sc < best_sc:
        best, best_sc = a.copy(), sc
    return best, best_sc


def _lp_phase(a, deadline, topk, max_iters=300):
    n = len(a)
    best, best_sc = a.copy(), _score(a)
    tk = topk
    for _ in range(max_iters):
        if time.monotonic() > deadline:
            break
        cv = _fft_conv(a, a)
        M = cv.max()
        g = _lp_dir(a, M, tk, deadline)
        if g is None:
            break
        g = _normalize(g)
        cur = _score(a)
        t, sc = _line_search(a, g, cur, deadline)
        if t > 0:
            a = _normalize((1 - t) * a + t * g)
        if sc < best_sc:
            best, best_sc = a.copy(), sc
        tk = max(25, tk - 1)
    return best, best_sc
