def nrm(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None

    s = a.sum()
    if s < 0.01:
        return None

    p = np.exp(zm)
    s = p.sum()
    if s < 1e-300:
        return 1e100
    p /= s
    u = float(GN) * cv(p)
    um = u.max()
    ex = np.exp(VB * (u - um))
    return float((VB * um + np.log(max(ex.sum(), 1e-300))) / VB)

def lgr(z):
    zm = z - z.max()
    p = np.exp(zm)
    s = p.sum()

    fw = np.fft.rfft(w, NF)
