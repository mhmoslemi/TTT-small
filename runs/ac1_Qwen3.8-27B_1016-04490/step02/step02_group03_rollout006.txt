def peak_shave(p, max_iters=500):
    global DEADLINE, N1, GN
    p = p.copy()
    n = GN
    N = N1
    fc = 2.0 * n * convoreal(p, N).max()
    for it in range(max_iters):
        if time.monotonic() > DEADLINE - 10:
            break
        c = convoreal(p, N)
        pk = c.max()
        if pk < 1e-15:
            break
        # Find top peaks
        top_k = np.argsort(c)[::-1][:20]
        thr = pk * 0.9
        active = [int(k) for k in top_k if c[k] >= thr]
        if not active:
            break
        # Weighted gradient toward reducing peaks
        wts = np.array([c[k] for k in active])
        wts /= wts.sum()
        # g[m] = sum_j w_j * 2*a[k_j - m]
        g = np.zeros(n)
        for jj, k in enumerate(active):
            wi = wts[jj]
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            # a[k-m] for m in range: m = k-j for j in [0, n-1]
            # So m ranges from k-(n-1) to k, clipped to [0, n-1]
            # g[m] += wi * a[k-m] for valid m
            # m = k - j, j = 0..n-1, so m = k, k-1, ..., k-(n-1)
            # g[m] += wi * a[j] where j = k-m
            # For m in [lo, hi]: j = k-m, range [k-hi, k-lo]
            g[lo:hi+1] += wi * p[k-hi:k-lo+1][::-1]
        g *= 2.0
        # Project gradient
        d = float(p @ g)
        g0 = g - d
        # Find step
        found = False
        for t in (0.05, 0.02, 0.01, 0.005, 0.002, 0.001):
            p2 = p - t * g0
            if p2.min() < 0:
                p2 = np.maximum(p2, 0)
            s2 = p2.sum()
            if s2 < 1e-15:
                continue
            p2 /= s2
            sc2 = 2.0 * n * convoreal(p2, N).max()
            if sc2 < fc - 1e-12:
                p = p2
                fc = sc2
                found = True
                break
        if not found:
            break
    update_best(p)
    return p
