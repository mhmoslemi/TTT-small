import numpy as np
import time
from scipy import fft as sfft

def npow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def proj_simplex(v, S):
    """Project v onto {a: a_i >= 0, sum(a) = S}"""
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)

def proj_simplex(v, S):
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    # Find rho: largest index where u[rho] > (cv[rho] - S) / (rho+1)
    # Equivalently: (rho+1)*u[rho] > cv[rho] - S
    idx = np.arange(1, n+1, dtype=np.float64)
    cond = u * idx > (cv - S)
    if not np.any(cond):
        return np.zeros(n)
    rho = int(np.nonzero(cond)[0][-1])
    tau = (cv[rho] - S) / (rho + 1)

import numpy as np
import time
from scipy import fft as sfft

def npow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def proj_simplex(v, S):
    n = v.size
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cv - S)
    if not np.any(cond):
        return np.maximum(v, 0.0)
    rho = int(np.nonzero(cond)[0][-1])
    tau = (cv[rho] - S) / (rho + 1)
    return np.maximum(v - tau, 0.0)

def fft_conv(x, fs, n):
    X = sfft.rfft(x, fs)
    c = np.real(sfft.irfft(X * X, fs))[:2 * n - 1]
    return c

def fft_subgrad(p_vec, x, cs, n):
    P = sfft.rfft(p_vec, cs)
    A = sfft.rfft(x, cs)
    g = 2.0 * np.real(sfft.irfft(np.conj(P) * A, cs))[:n]
    return g
