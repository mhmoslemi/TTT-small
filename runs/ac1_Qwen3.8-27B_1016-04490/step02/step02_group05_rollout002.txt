            pm1[:] = 0.9 * pm1 + 0.1 * g_norm
            pm2[:] = 0.97 * pm2 + 0.03 * g_norm
            v = pm1 - pm2
            x_new = x - lr * v
            x_new = np.clip(x_new, 0.0, 1000.0)
            x = _normalize(x_new, target_sum)
            sc = float(evaluate_sequence(x.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best = x.copy()

    # Phase 3: coordinate perturbation polish
    if time.monotonic() < deadline - 5.0:
        step = max(1, n // 500)
        for trial in range(200):
            if time.monotonic() > deadline - 5.0:
                break
            idx = rng.integers(0, n, size=step)
            for i in range(step):
                j = idx[i]
                orig = x[j]
                for delta in (-0.5, 0.5, -0.1, 0.1, -0.01, 0.01):
                    x[j] = max(0.0, orig + delta * np.mean(x))
                    sc = float(evaluate_sequence(x.tolist()))
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best = x.copy()
                    x[j] = orig
                break

    fc = float(evaluate_sequence(best.tolist()))
    if np.isfinite(fc) and fc <= best_score:
        return [float(v) for v in best]
    return [float(v) for v in parent]
