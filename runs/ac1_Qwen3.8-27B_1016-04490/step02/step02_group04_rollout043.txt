            x = _norm(x + 8e-4 * rng.standard_normal(n), ts)

        # Full restart from best with noise if stuck
        if noi > 5000:
            x = _norm(best + 0.025 * rng.standard_normal(n), ts)
            T = max(0.25 * bsc, 0.05)
            lr = 0.5
            noi = 0

        it += 1

    # Final exact-score verification and fallback
    fs_final = _score(best)
    if not np.isfinite(fs_final):
        psc = _score(par)
        if np.isfinite(psc):
            return par.tolist()
        return [float(ts / n)] * n
    return best.tolist()
