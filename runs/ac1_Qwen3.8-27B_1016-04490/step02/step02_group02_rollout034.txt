import numpy as np
import time


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def fft_conv(a, fs):
    A = np.fft.rfft(a, fs)
    out = np.fft.irfft(A * A, fs).real[:2 * len(a) - 1]
    return np.maximum(out, 0.0)


def soft_grad(conv, a, m, beta, work):
    n = len(a)
    mm = min(m, len(conv))
    idx = np.argpartition(conv, -mm)[-mm:]
    tv = conv[idx]
    mx = tv.max()
    wts = np.exp(beta * (tv - mx))
    wts /= wts.sum()
    order = np.argsort(tv)[::-1]
    idx = idx[order]
    wts = wts[order]
    work[:] = 0.0
    for j in range(mm):
        kk = int(idx[j])
        w = float(wts[j])
        lo = max(0, kk - (n - 1))
        hi = min(kk, n - 1)
        if lo <= hi:
            seg = a[kk - hi:kk - lo + 1][::-1]
            work[lo:hi + 1] += 2.0 * w * seg


def reproject(x):

        s = x.sum()
    return x / s
