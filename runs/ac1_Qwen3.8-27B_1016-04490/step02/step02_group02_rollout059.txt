import numpy as np
import time
from scipy import fft as sfft


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def fft_conv_grad(a, P):
    n = len(a)
    A = sfft.rfft(a, P)
    conv = np.maximum(sfft.irfft(A * A, P).real[:2 * n - 1], 0.0)
    total = conv.sum()
    if total < 1e-30:
        return conv, np.zeros(n)
    C = sfft.rfft(conv, P)
    cc = sfft.irfft(C * np.conj(A), P).real[:n]
    return conv, 2.0 * cc / total


def fft_conv_only(a, P):
    n = len(a)
    A = sfft.rfft(a, P)
    return np.maximum(sfft.irfft(A * A, P).real[:2 * n - 1], 0.0)


def topm_grad(conv, a, m, work):
    n = len(a)
    k = min(m, len(conv))
    idx = np.argpartition(conv, -k)[-k:]
    vals = conv[idx]
    vs = vals.sum()
    if vs < 1e-30:
        work[:] = 0.0
        return work
    ws = vals / vs
    work[:] = 0.0
    for j in range(k):
        kk = int(idx[j])
        w = float(ws[j])
        lo = max(0, kk - n + 1)
        hi = min(kk, n - 1)
        if lo <= hi:
            work[lo:hi + 1] += 2.0 * w * a[kk - hi:kk - lo + 1][::-1]
    return work


def _nesterov_step(x, vel, grad, eta, beta, n):
    vel[:] = beta * vel + grad
    gn = np.linalg.norm(vel)
    if gn < 1e-22:
        return x
    x = np.maximum(x - eta * x.mean() * vel / gn, 0.0)
    s = x.sum()
    if s < 1e-14:
        return x  # caller will handle reset
    return x / s


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    total = max(1.0, float(budget_s))
    hard_end = t0 + total - 40.0

    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = parent.size
    if n < 2:
        return [1.0]

    P = next_pow2(3 * n - 2)
    x = np.maximum(parent, 0.0)
    xs = x.sum()
    if xs < 0.01:
        x = np.ones(n)
        xs = float(n)
    x = x / xs

    conv, _ = fft_conv_grad(x, P)
    best = x.copy()
    best_score = 2.0 * n * conv.max()

    vel = np.zeros(n)
    work = np.zeros(n)
    noise_scale = max(0.05 * best.max(), 1e-10 / n)

    # Phase 1: smooth gradient, broad exploration (35% of time)
    eta, beta = 0.06, 0.85
    no_imp, ilim = 0, 400
    end1 = t0 + 0.35 * (total - 40.0)
    while time.monotonic() < end1:
        conv, grad = fft_conv_grad(x, P)
        cur = 2.0 * n * conv.max()
        if cur < best_score - 1e-13:
            best_score, best, no_imp = cur, x.copy(), 0
        else:
            no_imp += 1
        grad -= grad.mean()
        xn = _nesterov_step(x, vel, grad, eta, beta, n)
        if xn.sum() < 1e-14:
            x = best + noise_scale * rng.standard_normal(n)
            x = np.maximum(x, 1e-10)
            x /= x.sum()
            vel[:] = 0.0
        else:
            x = xn
        if no_imp >= ilim:
            x = best + noise_scale * (1.0 + no_imp / ilim) * rng.standard_normal(n)
            x = np.maximum(x, 1e-10)
            x /= x.sum()
            vel[:] = 0.0
            no_imp, ilim = 0, min(ilim + 150, 3000)

    # Phase 2: top-m gradient, medium refinement (40% of time)
    m, eta, beta = 5, 0.04, 0.92
    no_imp, ilim = 0, 600
    end2 = t0 + 0.75 * (total - 40.0)
    noise_scale = max(0.03 * best.max(), 1e-10 / n)
    while time.monotonic() < end2:
        conv = fft_conv_only(x, P)
        cur = 2.0 * n * conv.max()
        if cur < best_score - 1e-13:
            best_score, best, no_imp = cur, x.copy(), 0
        else:
            no_imp += 1
        grad = topm_grad(conv, x, m, work)
        grad -= grad.mean()
        xn = _nesterov_step(x, vel, grad, eta, beta, n)
        if xn.sum() < 1e-14:
            x = best + noise_scale * rng.standard_normal(n)
            x = np.maximum(x, 1e-10)
            x /= x.sum()
            vel[:] = 0.0
        else:
            x = xn
        if no_imp >= ilim:
            x = best + noise_scale * rng.standard_normal(n)
            x = np.maximum(x, 1e-10)
            x /= x.sum()
            vel[:] = 0.0
            no_imp, ilim = 0, min(ilim + 200, 4000)

    # Phase 3: fine polish with top-1 (remaining time)
    eta, beta = 0.012, 0.95
    while time.monotonic() < hard_end:
        conv = fft_conv_only(x, P)
        cur = 2.0 * n * conv.max()
        if cur < best_score - 1e-13:
            best_score, best = cur, x.copy()
        grad = topm_grad(conv, x, 1, work)
        grad -= grad.mean()
        xn = _nesterov_step(x, vel, grad, eta, beta, n)
        if xn.sum() >= 1e-14:
            x = xn

    # Final verification
    sc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(sc):
        pn = np.maximum(parent, 0.0)
        pn /= pn.sum()
        return [float(v) for v in pn]
    return [float(v) for v in best]
