        L.append(np.exp(-2.5 * ((j - t1) / (0.12 * n)) ** 2) + np.exp(-2.5 * ((j - t2) / (0.12 * n)) ** 2) + 0.005)
    L.append(0.5 * (1.0 + np.cos(2.0 * np.pi * j / (n - 1))))
    L.append(0.5 * (1.0 - np.cos(np.pi * j / (n - 1))) + 0.02)
    L.append(np.ones(n) * 0.5 + 0.25 * np.cos(2.0 * np.pi * 2.0 * j / n))
    return L

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    global DEADLINE, BEST_P, BEST_SC, GN, GNF
    t0 = time.monotonic()
    wall = t0 + budget_s - 60.0
    DEADLINE = wall

    n = 2535
    parent = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64)
        if raw.ndim == 1 and raw.size > 0:
            n = int(raw.size)
            parent = norm_clip(raw)
    except Exception:
        pass
    if parent is None:
        j = np.arange(n, dtype=np.float64)
        m = 0.5 * (n - 1)
        parent = (1.0 - np.abs(j - m) / (m + 1.0)) ** 2.0 + 0.01

    GN = n
    GNF = 1 << (2 * n - 1).bit_length()
    BEST_P = parent.copy()
    BEST_SC = float(evaluate_sequence(parent.tolist()))
    if not np.isfinite(BEST_SC):
        BEST_SC = 2.0
        BEST_P = np.ones(n).copy()
    update_best(BEST_P)

    cands = gen_inits(n)
    scored = []
    for a in cands:
        if time.monotonic() > wall - 100:
            break
        p = norm_clip(a)
        if p is None:
            continue
        sc = float(evaluate_sequence(p.tolist()))
        scored.append((sc, p))
    scored.append((BEST_SC, BEST_P.copy()))
    scored.sort(key=lambda x: x[0])

    starts = []
    for sc, p0 in scored:
        if len(starts) >= 5:
            break
        dup = False
        for q in starts:
            if np.linalg.norm((p0 - q) / p0.sum(), ord=1) < 0.3:
                dup = True
                break
        if not dup:
            starts.append(p0.copy())

    for p0 in starts:
        if time.monotonic() > wall - 50:
            break
        z0 = np.log(p0 + 1e-15)
        z0 = z0 - z0.mean()
        z0 = np.clip(z0, -25, 25)
        lse_run(z0)

    p_out = peak_shave(BEST_P.copy(), 700)

    psym = (p_out + p_out[::-1]) / 2.0
    psym = norm_clip(psym)
    if psym is not None:
        update_best(psym)
        peak_shave(psym.copy(), 400)

    for p0 in starts[:3]:
        if time.monotonic() > wall - 20:
            break
        peak_shave(p0.copy(), 250)

    update_best(BEST_P)

    if BEST_P is None or BEST_P.size == 0 or not np.isfinite(BEST_P).all():
        BEST_P = np.ones(n, dtype=np.float64)
    return [float(x) for x in BEST_P]
