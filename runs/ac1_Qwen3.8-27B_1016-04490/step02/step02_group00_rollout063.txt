            scored.append((sc, ia))
    scored.sort(key=_by_sc)
    if not scored:
        ln = j + 1.0
        return [float(v) for v in ln / ln.sum()]
    best_all = scored[0][1].copy()
    best_sc = scored[0][0]
    rem = max(deadline - time.monotonic(), 5.0)
    tk = min(4, len(scored))
    tp = rem / tk
    for k in range(tk):
        if time.monotonic() >= deadline - 15.0:
            break
        dl = min(time.monotonic() + tp, deadline - 15.0)
        b, sc = _optimize(scored[k][1].copy(), dl, rng, topk=5)
        if sc < best_sc:
            best_sc, best_all = sc, b.copy()
    if deadline - time.monotonic() > 30.0:
        b2, sc2 = _optimize(best_all.copy(), deadline - 5.0, rng, topk=3)
        if sc2 < best_sc:
            best_sc, best_all = sc2, b2.copy()
    for _ in range(200):
        if time.monotonic() > deadline - 5.0:
            break
        at = best_all.copy()
        mk = rng.random(n) < 0.2
        if mk.any():
            at[mk] *= np.exp(0.02 * rng.standard_normal(int(mk.sum())))
            at = _norm(at)
            st = _score(at)
            if st < best_sc:
                best_sc, best_all = st, at.copy()
    if not np.isfinite(best_all).all() or best_all.sum() < 0.01:
        ln = j + 1.0
        best_all = ln / ln.sum()
    return [float(v) for v in best_all]
