For the softmax version (which handles multiple near-max positions), the gradient is a weighted sum of these shifted/reversed versions.

Let me implement both:
- Phase 1: Use the exact subgradient at the argmax (cheap, just a shifted/reversed array)
- Phase 2: Use the softmax gradient (more global, handles multiple peaks)

Actually, for the exact subgradient, the multiplicative update would be:
a *= exp(-lr * 2 * a_rev_shifted)

Where a_rev_shifted[j] = a[k*-j] (valid range).

This means: positions j where a[k*-j] is large get decreased more. This makes sense: if a[k*-j] is large, then the product a[j]*a[k*-j] contributes to conv[k*], so reducing a[j] helps.

But wait, this is a very specific direction. It might oscillate or get stuck. The softmax version is more stable.

Let me combine: use the softmax gradient but with a lower temperature (sharper focus on peaks).

import numpy as np
import time


def _conv_self(a, sz):
    n = a.size
    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa * fa, sz)[:2 * n - 1]


def _score_fast(a):
    n = a.size
    m = 2 * n - 1
    sz = 1
    while sz < m:
        sz <<= 1
    conv = _conv_self(a, sz)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * conv.max() / (s * s))


def _run_opt(a, deadline, rng):
    n = a.size
    a = np.maximum(a, 0.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a[:] = 1.0 / n
    a = a / s
    
    m = 2 * n - 1
    sz = 1
    while sz < m:
        sz <<= 1
    
    best = a.copy()
    best_score = _score_fast(a)
    
    tau = 3.0 / n
    lr = 0.8
    it = 0
    stall = 0
    eval_gap = 30
    
    while time.monotonic() < deadline:
        fa = np.fft.rfft(a, sz)
        conv = np.fft.irfft(fa * fa, sz)[:m]
        
        z = conv / tau
        z -= z.max()
        e = np.exp(z)
        tot = e.sum()
        if tot < 1e-300:
            p = np.full(m, 1.0 / m)
        else:
            p = e / tot
        
        p_rev = p[::-1]
        fp = np.fft.rfft(p_rev, sz)
        cross = np.fft.irfft(fa * fp, sz)[:n + m - 1]
        grad = 2.0 * cross[n - 1:2 * n - 1][::-1]
        
        gm = grad.max()
        if gm > 1e-18:
            a *= np.exp(-lr * grad / gm)
        
        a = np.maximum(a, 1e-15)
        a /= a.sum()
        it += 1
        
        if it % 40 == 0:
            tau *= 0.994
            if tau < 0.5 / n:
                tau = 0.5 / n
            lr *= 0.999
        
        if it % eval_gap == 0:
            sc = _score_fast(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best = a.copy()
                stall = 0
            else:
                stall += 1
            
            if stall >= 10:
                strength = 0.03 + 0.02 * (stall % 3)
                mask = rng.random(n) < 0.15
                cnt = mask.sum()
                if cnt > 0:
                    a[mask] *= np.exp(strength * rng.standard_normal(cnt))
                    a = np.maximum(a, 1e-15)
                    a /= a.sum()
                stall = 0
    
    sc = _score_fast(a)
    if sc < best_score:
        best = a.copy()
        best_score = sc
    return best, best_score


def _local_refine(a, deadline, rng):
    n = a.size
    a = np.maximum(a, 0.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    m = 2 * n - 1
    sz = 1
    while sz < m:
        sz <<= 1
    
    best = a.copy()
    best_score = _score_fast(a)
    
    step = 0.01
    it = 0
    while time.monotonic() < deadline:
        fa = np.fft.rfft(a, sz)
        conv = np.fft.irfft(fa * fa, sz)[:m]
        kstar = int(np.argmax(conv))
        
        # Subgradient at kstar: grad[j] = 2*a[kstar-j] if valid
        lo = max(0, kstar - (n - 1))
        hi = min(n - 1, kstar)
        lo2 = max(0, kstar)
        hi2 = min(n - 1, kstar + n - 1)
        grad = np.zeros(n)
        if hi >= lo:
            idx = np.arange(lo, hi + 1)
            src = kstar - idx
            grad[idx] = 2.0 * a[src]
        
        gm = grad.max()
        if gm < 1e-18:
            break
        
        # Try multiplicative step
        a_try = a * np.exp(-step * grad / gm)
        a_try = np.maximum(a_try, 1e-15)
        a_try /= a_try.sum()
        sc_try = _score_fast(a_try)
        
        if sc_try < best_score - 1e-14:
            best_score = sc_try
            best = a_try.copy()
            a = a_try.copy()
            step = min(step * 1.05, 0.15)
            it = 0
        else:

            it += 1
            if it > 80:
                # Apply small perturbation
                idx = rng.integers(0, n, size=max(1, n // 50))
                a[idx] *= np.exp(0.01 * rng.standard_normal(len(idx)))
                a = np.maximum(a, 1e-15)
                a /= a.sum()
                it = 0
                step = 0.01
    
    return best
