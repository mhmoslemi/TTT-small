    """Find g maximizing sum(g) s.t. (f*g)[k_j] <= M, g >= 0."""

    L = len(conv)
    if M is None:
        M = conv.max()

    k = min(m, L)
    idx = np.argpartition(conv, -k)[-k:]
    order = np.argsort(conv[idx])[::-1]
    idx = idx[order]

    p = 1

        p <<= 1
    return p

def fft_conv(a, fs):
    A = sfft.rfft(a, fs)
    return np.maximum(sfft.irfft(A * A, fs).real[:2 * len(a) - 1], 0.0)

def topm_grad(conv, a, m, work):
    n = len(a)
    k = min(m, len(conv))
    idx = np.argpartition(conv, -k)[-k:]
    order = np.argsort(conv[idx])[::-1]
    idx = idx[order]
    vals = conv[idx]

    if vs < 1e-30:
        work[:n] = 0.0
        return work
    ws = vals / vs
    work[:n] = 0.0

        lo = max(0, kk - n + 1)
