import numpy as np
import time
from scipy import fft as sfft

def npow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def sc(x):
    return float(evaluate_sequence(x.tolist()))

def autocorr(x, fs, n):
    X = sfft.rfft(x, fs)
    c = np.real(sfft.irfft(X * X, fs))[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def softmax_wts(c, T, K):
    cm = c.max()
    if cm <= 0:
        return np.ones(c.size) / c.size
    above = np.where(c >= 0.6 * cm)[0]
    if above.size <= K * 2:
        idx = above
    else:
        idx = np.argpartition(c, -K)[-K:]
        top_above = above[np.argsort(c[above])[-K:]]
        idx = np.unique(np.concatenate([idx, top_above]))
    pv = np.exp((c[idx] - cm) / T)
    s = pv.sum()
    p = np.zeros(c.size)
    if s > 1e-300:
        p[idx] = pv / s
    else:
        p[int(np.argmax(c))] = 1.0
    return p

def xgrad(p, x, cs, n):
    P = sfft.rfft(p, cs)
    Xc = sfft.rfft(x, cs)
    g = np.real(sfft.irfft(P * (Xc + np.conj(Xc)), cs))[:n]
    g -= g.mean()
    return g
