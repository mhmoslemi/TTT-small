def topm_grad_w(conv, a, m, work):

    k = min(m, len(conv))
    idx = np.argpartition(conv, -k)[-k:]
    order = np.argsort(conv[idx])[::-1]
    idx = idx[order]
    vals = conv[idx]
    # Exponential weighting: top peak gets weight 1, next gets weight exp(-gap), etc.
    if vals[0] < 1e-30:
        work[:n] = 0.0
        return work[:n]
    # Normalize: w_j = vals[j] / vals[0]
    ws = vals / vals[0]
    work[:n] = 0.0
    for j in range(k):
        if ws[j] < 0.01:  # skip negligible peaks
            break
        kk = int(idx[j])
        lo = max(0, kk - (n - 1))
        hi = min(kk, n - 1)
        if lo <= hi:
            seg = a[kk - hi:kk - lo + 1][::-1]
            work[lo:hi + 1] += 2.0 * ws[j] * seg
            mid = kk // 2
            if kk % 2 == 0 and lo <= mid <= hi:
                work[mid] -= ws[j] * a[mid]
    return work[:n]
