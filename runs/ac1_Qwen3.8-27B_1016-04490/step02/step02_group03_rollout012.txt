def lse_val(z):
    global GB
    zm = z - z.max()
    p = np.exp(zm)
    s = p.sum()
    if s < 1e-300:
        return 1e100
    p /= s
    c = conve(p, GNF)
    u = float(GN) * c
    um = u.max()
    ex = np.exp(GB * (u - um))
    s2 = ex.sum()
    if s2 < 1e-300:
        return 1e100
    return float((GB * um + np.log(s2)) / GB)

    global GB
    n = GN
    zm = z - z.max()
    p = np.exp(zm)
    s = p.sum()

    s2 = ex.sum()

    w = ex / s2
    fp = np.fft.rfft(p, GNF)
    fw = np.fft.rfft(w, GNF)
    r = np.fft.irfft(np.conj(fp) * fw, GNF)
    gp = float(n) * 2.0 * np.real(r[:n])
    d = float(p @ gp)
    g = p * (gp - d)
    return g

        g *= 2.0 * float(GN)
        d = float(p @ g)
        g0 = g - d
        # Try decreasing step sizes
        found = False
        for t in (0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001):
