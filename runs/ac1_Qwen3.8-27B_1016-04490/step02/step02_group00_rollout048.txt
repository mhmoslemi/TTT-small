from scipy.optimize import linprog

def _score(a):
    return float(evaluate_sequence(a.tolist()))


def _fft_conv(a):
    n = len(a)
    m = 2 * n - 1
    sz = 1
    while sz < m:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa * fa, sz)[:m]


def _build_T(a):
    """Build sparse constraint matrix T where T[k,i] = a[k-i]."""
    n = len(a)
    m = 2 * n - 1
    ks = np.arange(m)
    i0 = np.maximum(0, ks - n + 1)
    i1 = np.minimum(n - 1, ks)
    counts = i1 - i0 + 1
    total = int(counts.sum())
    
    rows = np.repeat(ks, counts)
    # Build cols vectorized
    indices = np.arange(total)
    row_start = np.concatenate(([0], np.cumsum(counts[:-1])))
    local_idx = indices - np.repeat(row_start, counts)
    cols = np.repeat(i0, counts) + local_idx
    vals = a[rows - cols]
    
    return csr_matrix((vals, (rows, cols)), shape=(m, n))


def _lp_solve(a, M, time_limit):

                      method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None
