Let me also think about the "active set" idea from the LP description. The idea is:
- The max is achieved at some set of peaks {k_1, ..., k_r}
- At the optimum, the gradients of the active constraints are linearly dependent (KKT)
- We can iterate: identify active set, solve the subproblem, update

A practical version:

2. Find the max value M and the set of peaks where b[k] >= M - epsilon
3. For these active peaks, compute the "anti-correlation" direction: the direction that reduces all active peaks
4. Take a step in that direction
5. Repeat

    return _softmax(z)

def _peak_shave(p, iters):
    p = p.copy().astype(np.float64)
    n = GN
    cur = _score(p)
    for _ in range(iters):
        if time.monotonic() > DEADLINE - 35:
            break
        c = _conv(p)
        pk = c.max()
        if pk < 1e-18:
            break
        thr = pk * 0.965
        top = np.argsort(c)[::-1]
        active = []
        for k in top[:80]:
            ki = int(k)
            if c[ki] >= thr:
                active.append(ki)
            else:
                break
        if not active:
            break
        wts = np.array([c[k] for k in active])
        wts /= wts.sum()
        g = np.zeros(n)
        for ii in range(len(active)):
            k = active[ii]
            wi = wts[ii]
            lo1 = max(0, k - n + 1)
            hi1 = min(n - 1, k)
            g[lo1:hi1 + 1] += wi * p[k - hi1:k - lo1 + 1][::-1]
            lo2 = max(0, k)
            hi2 = min(n - 1, k + n - 1)
            g[lo2:hi2 + 1] += wi * p[lo2 - k:hi2 - k + 1]
        g *= float(n)
        g0 = g - g.mean()
        found = False
        for t in (0.1, 0.05, 0.03, 0.01, 0.005, 0.002, 0.001):
            p2 = p - t * g0
            if (p2 < 0).any():
                p2 = np.maximum(p2, 0.0)
            s2 = p2.sum()
            if s2 < 1e-10:
                break
            p2 /= s2
            sc2 = _score(p2)
            if sc2 < cur - 1e-15:
                p = p2
                cur = sc2
                found = True
                break
        if not found:
            break
    _update(p)
    return p

def _gen_starts(n):
    L = []
    j = np.arange(n, dtype=np.float64)
    m = 0.5 * (n - 1)
    L.append(np.ones(n))
    L.append(0.5 * (1.0 + np.cos(np.pi * j / (n - 1))))
    L.append(0.5 * (1.0 - np.cos(np.pi * j / (n - 1))) + 0.01)
    for pv in (-0.3, -0.6, -1.0, -1.5, -2.5):
        L.append((j + 1.0) ** pv)
    for a in (0.5, 1.0, 2.0, 4.0, 8.0):
        L.append((1.0 - np.abs(j - m) / (m + 1.0)) ** a + 0.01)
    for f in (1, 2, 3, 5):
        L.append(0.5 * (1.0 + np.cos(np.pi * j / (n - 1))) * (1.0 + 0.4 * np.cos(2.0 * np.pi * f * j / n)) + 0.01)
    t1 = 0.3 * n
    t2 = 0.7 * n
    sd = 0.08 * n
    L.append(np.exp(-((j - t1) / sd) ** 2) + np.exp(-((j - t2) / sd) ** 2) + 0.005)
    d = np.minimum(j, n - 1 - j) + 1.0
    for pv in (0.5, 1.0, 2.0):
        L.append(d ** pv)
    return L

def _sort_key(x):
    return x[0]
