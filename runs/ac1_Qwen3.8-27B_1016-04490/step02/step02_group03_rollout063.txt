import numpy as np
import time
import math
try:
    from scipy.optimize import minimize as _min
except Exception:
    _min = None

DEADLINE=0.0
BEST_A=None
BEST_SC=float('inf')
N=0
NFFT=0
GBETA=1.0
BIG=1.0e150

def _stop(x):
    if time.monotonic() > DEADLINE:
        raise TimeoutError

def tuple_key(x):
    return x[0]

def clean(a):
    a=np.asarray(a,dtype=np.float64).ravel()
    if a.size==0: return None
    a=np.clip(a,0.0,1000.0)
    s=float(a.sum())
    if s<0.01 or not math.isfinite(s): return None
    p=a/s
    if not np.isfinite(p).all(): return None
    return p

def update_best(a):
    global BEST_A,BEST_SC
    p=clean(a)
    if p is None: return
    try:
        sc=float(evaluate_sequence(p.tolist()))
    except Exception:
        sc=float('inf')
    if math.isfinite(sc) and sc<BEST_SC:
        BEST_A=p.copy(); BEST_SC=sc

def conv(a):
    fa=np.fft.rfft(a,NFFT)
    c=np.fft.irfft(fa*fa,NFFT)
    return np.real(c[:2*int(a.size)-1])

def fast_score(a):
    p=clean(a)
    if p is None: return float('inf')
    b=conv(p)
    sc=2.0*float(N)*float(b.max())
    return sc if math.isfinite(sc) else float('inf')

def softmax_z(z):
    z=np.asarray(z,dtype=np.float64)
    if not np.isfinite(z).all(): return None,None
    zm=z-z.max()
    a=np.exp(zm)
    s=float(a.sum())
    if s<1e-300 or not math.isfinite(s):
        return np.full(N,1.0/float(N)), zm
    return a/s, zm

def corr_weights(a,w):
    L=2*N-1
    wr=w[::-1]
    fa=np.fft.rfft(a,NFFT)
    fw=np.fft.rfft(wr,NFFT)
    c=np.fft.irfft(fa*fw,NFFT)
    return np.real(c[N:L][::-1])

def make_weights(s,beta,floor,normalize):
    sm=float(s.max())
    w=np.exp(beta*(s-sm))
    if floor>0.0:
        w=w+float(floor)
    if normalize:
        ss=float(w.sum())
        if ss>1e-300:
            w=w/ss
    return w

def lse_value(z):
    a,_=softmax_z(z)
    if a is None: return BIG
    b=conv(a)
    u=float(N)*b
    um=float(u.max())
    if not math.isfinite(um): return BIG
    ex=np.exp(GBETA*(u-um))
    s=float(ex.sum())
    if s<1e-300 or not math.isfinite(s): return BIG
    return float((GBETA*um+math.log(s))/GBETA)

def lse_grad(z):
    n=N
    a,_=softmax_z(z)
    if a is None: return np.zeros(n)
    b=conv(a)
    u=float(n)*b
    um=float(u.max())
    if not math.isfinite(um): return np.zeros(n)
    ex=np.exp(GBETA*(u-um))
    s=float(ex.sum())
    if s<1e-300 or not math.isfinite(s): return np.zeros(n)
    w=ex/s
    corr=corr_weights(a,w)
    ga=2.0*float(n)*corr
    gz=a*(ga-float(np.dot(a,ga)))
    if not np.isfinite(gz).all(): return np.zeros(n)
    return gz

def run_lse(a,base=120):
    global GBETA
    a=clean(a)
    if a is None or _min is None: return a
    z=np.log(np.maximum(a,1e-300))
    z=z-z.mean()
    z=np.clip(z,-30.0,30.0)
    f0_all=lse_value(z)
    if not math.isfinite(f0_all):
        return a
    if base >= 100:
        sched=((24,70),(96,90),(384,110),(1536,130),(6144,150),(24576,180))
    else:
        sched=((96,50),(384,60),(1536,70),(6144,80))
    scale=max(0.4,min(1.5,base/120.0))
    for beta0,it0 in sched:
        if time.monotonic() > DEADLINE - 160:
            break
        GBETA=float(max(8,int(beta0*scale)))
        iters=int(max(20,it0*scale))
        f0=lse_value(z)
        if not math.isfinite(f0): break
        try:
            res=_min(lse_value,z,jac=lse_grad,method='L-BFGS-B',callback=_stop,
                     options={'maxiter':iters,'maxls':25,'ftol':1e-15,'gtol':1e-9})
        except Exception:
            continue
        if res is not None and getattr(res,'x',None) is not None and np.isfinite(getattr(res,'fun',BIG)):

def apply_L(fa,v):
    fv=np.fft.rfft(v,NFFT)
    y=np.fft.irfft(fa*fv,NFFT)
    return np.real(y[:2*N-1])

def apply_LT(fa,r):
    L=2*N-1
    rr=r[::-1]
    fr=np.fft.rfft(rr,NFFT)
    q=np.fft.irfft(fa*fr,NFFT)
    return np.real(q[N:L][::-1])

def newton_direction(a,b,w):
    fa=np.fft.rfft(a,NFFT)
    rhs=-apply_LT(fa,w*b)
    rhs-=rhs.mean()
    rn=float(np.abs(rhs).sum())
    if rn<1e-20: return None
    rhs=rhs/rn
    p=rhs.copy()
    r=rhs.copy()
    x=np.zeros(N,dtype=np.float64)
    rs=float(np.dot(r,r))
    if rs<1e-300: return None
    a_safe=np.maximum(a,1e-8)
    reg=1e-8/a_safe
    for _ in range(32):
        if time.monotonic() > DEADLINE - 50:
            break
        Lp=apply_L(fa,p)
        Ap=apply_LT(fa,w*Lp)
        Ap=Ap+reg*p
        Ap-=Ap.mean()
        pa=float(np.dot(p,Ap))
        if not math.isfinite(pa) or abs(pa)<1e-300:
            break
        alpha=rs/pa
        if not math.isfinite(alpha) or alpha<0.0:
            break
        x=x+alpha*p
        r=rhs-alpha*Ap
        r-=r.mean()
        rs2=float(np.dot(r,r))
        if not math.isfinite(rs2) or rs2<1e-22*rs:
            break
        p=r+(rs2/rs)*p
        p-=p.mean()
        rs=rs2
    x-=x.mean()
    if not np.isfinite(x).all(): return None
    nd=float(np.abs(x).sum())
    if nd<1e-20: return None
    return x/nd

    for beta_f, it_f, mbeta in ((30.0, 240, 4096.0), (50.0, 140, 8192.0)):
        if time.monotonic() > wall - 100 or BEST_A is None:
            break
        a = npol(BEST_A.copy(), int(it_f), float(beta_f))
        if a is None:
            break
        a = mshave(a, int(0.8 * it_f), float(mbeta))
        upd(a)
        sym = clean(0.5 * (a + a[::-1]))
        if sym is not None:
            upd(sym)
            s2 = npol(sym, int(0.4 * it_f), float(beta_f) + 5.0)
            if s2 is not None:
                s2 = mshave(s2, int(0.3 * it_f), float(mbeta))
                upd(s2)

    if BEST_A is None or BEST_A.size == 0 or not np.isfinite(BEST_A).all() or BEST_A.sum() < 0.01:
        BEST_A = np.full(N, 1.0 / float(N) if N else 1.0)
    return [float(x) for x in np.clip(BEST_A, 0.0, 1000.0)]
