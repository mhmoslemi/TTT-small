def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def fft_conv(a, fs):
    A = sfft.rfft(a, fs)
    out = sfft.irfft(A * A, fs).real[:2 * len(a) - 1]
    return np.maximum(out, 0.0)


def topm_grad(conv, a, m, work, power=1.0):

    idx = np.argpartition(conv, -k)[-k:]
    order = np.argsort(conv[idx])[::-1]

    if power != 1.0:
        wts = vals ** power
    else:
        wts = vals
    ws = wts.sum()
    if ws < 1e-30:
        work[:] = 0.0
        return work
