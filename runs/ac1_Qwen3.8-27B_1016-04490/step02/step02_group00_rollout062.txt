def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 10.0)
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.ndim != 1 or parent.size < 2 or not np.isfinite(parent).all():
        parent = np.ones(2535, dtype=np.float64)
    n = parent.size
    parent = np.maximum(parent, 0.0)
    if parent.sum() < 1e-10:
        parent = np.ones(n)
    parent = parent / parent.sum()
    j = np.arange(n, dtype=np.float64)
    x = j / (n - 1.0)
    inits = [
        parent.copy(),
        j + 1.0,
        n - j,
        (j + 1.0) ** 1.3,
        (j + 1.0) ** 0.7,
        np.exp(2.0 * x),
        1.0 + 0.5 * np.cos(np.pi * x),
        np.exp(-4.0 * (x - 0.3)) + 0.5 * np.exp(-4.0 * (x - 0.7)) + 0.1,
        1.0 / (1.0 + 8.0 * x) + 0.05,
        0.5 + 0.3 * np.sin(2.0 * np.pi * x) + 0.3,
    ]
    scored = []
    for ia in inits:
        ia = np.maximum(ia, 0.0).astype(np.float64, copy=True)
        ia = _normalize(ia)
        sc = _score(ia)
        if np.isfinite(sc):
            scored.append((sc, ia))
    scored.sort(key=_sort_key)
    if not scored:
        lin = j + 1.0
        lin /= lin.sum()
        return [float(v) for v in lin]
    best_all = scored[0][1].copy()
    best_sc = scored[0][0]
    rem = max(deadline - time.monotonic(), 1.0)
    top_k = min(5, len(scored))
    wts = np.array([2.0, 1.5, 1.1, 0.8, 0.6][:top_k])
    wts /= wts.sum()
    time_alloc = wts * (rem * 0.88)
    for k in range(top_k):
        if time.monotonic() >= deadline - 30.0:
            break
        dl = min(time.monotonic() + time_alloc[k], deadline - 30.0)
        b, sc = _run_opt(scored[k][1].copy(), dl, rng)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_all = b.copy()
    rem2 = deadline - time.monotonic()
    if rem2 > 25.0:
        b2, sc2 = _run_opt(best_all.copy(), deadline - 3.0, rng)
        if np.isfinite(sc2) and sc2 < best_sc:
            best_sc = sc2
            best_all = b2.copy()
    if not np.isfinite(best_all).all() or best_all.sum() < 1e-10:
        lin = j + 1.0
        best_all = lin / lin.sum()
    return [float(v) for v in best_all]
