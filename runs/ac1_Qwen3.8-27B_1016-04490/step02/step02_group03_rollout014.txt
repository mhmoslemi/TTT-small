    return np.real(np.fft.irfft(fa * fa, N))[:2*n-1]

def _score(a):
    """Compute the AC1 upper bound."""
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = _conv(a, _NFFT)
    return float(2.0 * _N * c.max() / (s * s))

def _update_best(p):
    """Update global best if p scores better."""
    global _BEST, _BEST_S
    if p is None:
        return
    p = np.clip(p, 0, 1000)
    if p.sum() < 0.01:
        return
    s = float(evaluate_sequence(p.tolist()))
    if np.isfinite(s) and s < _BEST_S:

        _BEST_S = s

def _norm(p):
    """Normalize to sum 1, clip to [0, 1000]."""
    p = np.clip(p, 0.0, 1000.0)
    s = p.sum()
    if s < 1e-15:
        return None
    return p / s

def _lse_fp(z, beta):
    """Log-sum-exp of the max autocorrelation, given log-params z."""
    a = np.exp(z - z.max())
    a /= a.sum()
    c = _conv(a, _NFFT)
    u = float(_N) * c
    um = u.max()
    s = np.exp(beta * (u - um)).sum()
    return float((beta * um + np.log(s)) / beta)

def _lse_grad(z, beta):
    """Gradient of the log-sum-exp objective."""
    a = np.exp(z - z.max())
    a /= a.sum()
    c = _conv(a, _NFFT)
    u = float(_N) * c
    um = u.max()
    w = np.exp(beta * (u - um))
    w /= w.sum()
    # Gradient: g[j] = 2 * sum_k w[k] * a[k-j] = 2 * cross_corr(w, a)[j]
    # FFT: IFFT(FFT(w) * conj(FFT(a)))
    N2 = 1 << (len(w) + _N - 2 - 1).bit_length()
    fw = np.fft.rfft(w, N2)
    fa = np.fft.rfft(a, N2)
    r = np.fft.irfft(fw * np.conj(fa), N2)
    gp = 2.0 * float(_N) * r[:_N]
    d = a @ gp
    g = a * (gp - d)
    return g

def _lp_step(f, time_limit=10.0):
    """One LP-based step: find g minimizing max(f*g) s.t. g>=0, sum(g)=1."""
    n = f.size
    # Build sparse convolution matrix A: A[k,j] = f[j] if 0<=k-j<n
    rows_parts = []
    cols_parts = []
    vals_parts = []
    for j in range(n):
        r_lo = j
        r_hi = min(j + n - 1, 2 * n - 2)
        r = np.arange(r_lo, r_hi + 1)
        rows_parts.append(r)
        cols_parts.append(np.full(len(r), j, dtype=np.int32))
        vals_parts.append(np.full(len(r), f[j]))
    rows = np.concatenate(rows_parts)
    cols = np.concatenate(cols_parts)
    vals = np.concatenate(vals_parts)
    A = csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    # LP: min M' s.t. A@g - M' <= 0, sum(g) = 1, g >= 0, M' >= 0
    # Variables: [g (n), M' (1)]
    # c for objective: [0,...,0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality: A@g - M' <= 0
    # A_ineq = [A, -e] where e is the last basis vector
    A_ineq_parts = [A]
    # Append -e column: row k, col n has value -1
    neg_e_rows = np.arange(2 * n - 1)
    neg_e_cols = np.full(2 * n - 1, n, dtype=np.int32)
    neg_e_vals = np.ones(2 * n - 1)
    from scipy.sparse import hstack, csc_matrix as csc2, identity as spy
    A_ineq = hstack([A, csc2((neg_e_vals, (neg_e_rows, neg_e_cols)), shape=(2 * n - 1, 1))])
    
    b_ineq = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    t_start = time.monotonic()
    try:
        res = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq,
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                      method='highs', options={'time_limit': time_limit, 'presolve': True})
    except Exception:
        return None
    
    if res is None or not res.success:
        return None
    
    g = res.x[:n]
    g = np.clip(g, 0, 1000)
    gs = g.sum()
    if gs < 1e-15:
        return None
    return g / gs

    n = f.size
    # For each column j, rows are j to min(j+n-1, 2n-2)
    # The pattern is: column 0 has n entries, column 1 has n entries, ..., column n-1 has n entries
    # Wait, column j has entries in rows j to j+n-1 (capped at 2n-2)
    # So column j has min(n, 2n-1-j) entries
    # Total: sum_{j=0}^{n-1} min(n, 2n-1-j)
    # = sum_{j=0}^{n-1} n (since 2n-1-j >= n for j <= n-1)
    # = n * n = n^2

    # And k must be in [0, 2n-2]
    # So k in [j, min(j+n-1, 2n-2)]
    # Length: min(j+n-1, 2n-2) - j + 1 = min(n, 2n-1-j)

    # For j = n-1: min(n, n) = n

    g = np.clip(r.x[:n], 0, 1000)

def _shave(f, iters):

        c = _conv(f)
        M = c.max()

        if len(peaks) == 0: break
        wts = np.exp(60.0 * (c[peaks] - M))
        wts /= wts.sum()
        g = np.zeros(_N)
        for k, w in zip(peaks, wts):
            k = int(k)
            lo = max(0, k - _N + 1)
            hi = min(_N - 1, k)
            g[lo:hi+1] += w * f[k-hi:k-lo+1][::-1]
        g *= 2.0 * _N
        g -= f @ g
        found = False
        for t in (0.05, 0.02, 0.01, 0.005, 0.002, 0.001):

                s2 = _score(f2)
                if s2 < _score(f) - 1e-13:
                    f = f2
                    found = True
                    break
        if not found: break
    _upd(f)
    return f
