def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _normalize(x, target):
    s = x.sum()
    if s < 1e-10:
        x[:] = 1.0
        s = len(x)
    x *= target / s
    np.clip(x, 0.0, 1000.0, out=x)
    return x


def _softmax_p(conv, T, n):
    cmax = conv.max()
    if cmax < 1e-15 or T < 1e-8:
        p = np.zeros(2 * n - 1)
        p[np.argmax(conv)] = 1.0
        return p

    return g
