import numpy as np

from scipy import fft as sfft


def _np2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _proj_simplex(v, S):
    n = v.size
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_all = np.arange(1, n + 1, dtype=np.float64)
    cond = u - (cv - S) / rho_all > 0
    if not np.any(cond):
        return np.full(n, S / n)
    rho = int(np.max(rho_all[cond]))
    theta = (cv[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def _scandl(x, n, fs):
    X = sfft.rfft(x, fs)
    return np.maximum(np.real(sfft.irfft(X * X, fs))[:2 * n - 1], 0.0)
