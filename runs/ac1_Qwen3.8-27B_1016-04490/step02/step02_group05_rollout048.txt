def _grad_fn(y, T, fft_size, corr_size, n):
    X = sfft.rfft(y, fft_size)
    conv = np.real(sfft.irfft(X * X, fft_size))[:2 * n - 1]
    conv = np.maximum(conv, 0.0)
    cmax = conv.max()
    logits = np.clip((conv - cmax) / T, -50.0, 0.0)
    p = np.exp(logits)
    ps = p.sum()
    if ps < 1e-300:
        p = np.zeros(2 * n - 1)
        p[(2 * n - 1) // 2] = 1.0
    else:
        p /= ps
    P = sfft.rfft(p, corr_size)
    Yc = sfft.rfft(y, corr_size)
    corr = np.real(sfft.irfft(P * np.conj(Yc), corr_size))
    g = 2.0 * corr[:n]
    g -= g.mean()
    gm = np.abs(g).max()
    if gm > 1e5:
        g *= 1e5 / gm
    return g


def _fista_block(x_init, T0, lr0, fft_size, corr_size, n, target_sum,
                 max_iters, deadline, eval_every, parent):
    x = x_init.copy()
    x_prev = x.copy()
    t_curr = 1.0
    t_prev = 1.0
    T = T0
    lr = lr0
    local_best = x.copy()
    local_best_sc = 1e18
    no_imp = 0
    it = 0
    while it < max_iters and time.monotonic() < deadline:
        if it == 0:
            y = x
        else:
            factor = min(1.0, (t_prev - 1.0) / t_curr)
            y = x + factor * (x - x_prev)
        g = _grad_fn(y, T, fft_size, corr_size, n)
        x_new = y - lr * g
        np.maximum(x_new, 0.0, out=x_new)
        _norm_sum(x_new, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_curr * t_curr)) / 2.0
        x_prev[:] = x
        x[:] = x_new
        t_prev = t_curr
        t_curr = t_new
        T = max(0.001, T * 0.9996)
        lr = max(0.01, lr * (0.99998 if it > 10000 else 1.0))
        it += 1
        if it % eval_every == 0:
            sc = evaluate_sequence(x.tolist())
            if np.isfinite(sc) and sc < local_best_sc:
                local_best_sc = sc
                local_best = x.copy()
                no_imp = 0
            else:
                no_imp += eval_every
            if it % 20000 == 0:
                x += 2e-5 * np.random.default_rng(it).standard_normal(n)
                x = np.maximum(x, 0.0)
                _norm_sum(x, target_sum)
                x_prev[:] = x
                t_curr = 1.0
                t_prev = 1.0
    return local_best, local_best_sc, no_imp


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 15.0)

    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256)
    n = parent.size

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(best_score):
        best = np.ones(n)
        best_score = float(evaluate_sequence(best.tolist()))

    target_sum = np.sqrt(2.0 * n)
    fft_size = _next_pow2(2 * n)
    corr_size = _next_pow2(3 * n - 1)

    # Phase 1: Main FISTA search from parent
    x0 = _norm_sum(parent.copy(), target_sum)
    x0 += 0.02 * rng.standard_normal(n)
    x0 = np.maximum(x0, 0.01)
    _norm_sum(x0, target_sum)

    x_res, sc_res, _ = _fista_block(
        x0, best_score * 0.10, 0.45, fft_size, corr_size, n, target_sum,
        max_iters=2000000, deadline=deadline - 60.0,
        eval_every=400, parent=best)
    if np.isfinite(sc_res) and sc_res < best_score:
        best_score = sc_res
        best = x_res.copy()

    # Phase 2: Restart with structured init
    if time.monotonic() < deadline - 40.0:
        x1 = _init_seq(n, rng, best, 1)
        _norm_sum(x1, target_sum)
        x_res2, sc_res2, _ = _fista_block(
            x1, best_score * 0.08, 0.40, fft_size, corr_size, n, target_sum,
            max_iters=1500000, deadline=deadline - 40.0,
            eval_every=400, parent=best)
        if np.isfinite(sc_res2) and sc_res2 < best_score:
            best_score = sc_res2
            best = x_res2.copy()

    # Phase 3: Polishing with low T on the best
    if time.monotonic() < deadline - 15.0:
        xp = _norm_sum(best.copy(), target_sum)
        xp_res, xp_sc, _ = _fista_block(
            xp, 0.003, 0.03, fft_size, corr_size, n, target_sum,
            max_iters=800000, deadline=deadline - 10.0,
            eval_every=200, parent=best)
        if np.isfinite(xp_sc) and xp_sc < best_score:
            best_score = xp_sc
            best = xp_res.copy()

    # Final verification
    fc = float(evaluate_sequence(best.tolist()))
    if np.isfinite(fc) and fc <= best_score + 1e-10:
        return [float(v) for v in best]
    fc2 = float(evaluate_sequence(parent.tolist()))
    if np.isfinite(fc2):
        return [float(v) for v in parent]
    return [float(v) for v in np.ones(n)]
