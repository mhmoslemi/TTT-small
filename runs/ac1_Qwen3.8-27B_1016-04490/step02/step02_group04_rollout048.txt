import numpy as np
import time
from scipy import fft as sfft

def _np2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def _score(v):

def _norm(x, ts):
    sx = x.sum()
    if sx < 1e-10:
        return np.full(x.shape, ts / x.size)
    return np.clip(x * (ts / sx), 0.0, 1000.0)

def _autocorr(x, fs):
    X = sfft.rfft(x, fs)
    return np.maximum(np.real(sfft.irfft(X * X, fs, n=fs))[:2 * len(x) - 1], 1e-12)

def _grad(x, p, fs):
    pr = p[::-1]
    X = sfft.rfft(x, fs)
    P = sfft.rfft(pr, fs)
    gr = np.real(sfft.irfft(P * X, fs, n=fs))
    return 2.0 * gr[len(x) - 1:2 * len(x) - 1][::-1]
