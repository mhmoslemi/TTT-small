def _project(b, target_sum):
    """Project b onto {a >= 0, sum(a) = target_sum}"""
    # Water filling
    b_sorted = np.sort(b)[::-1]
    cumsum = np.cumsum(b_sorted)
    # Find largest k such that b_sorted[k] > (target_sum - cumsum[k])/k
    # Actually, find t such that sum(max(0, b-t)) = target_sum
    # Try: t = (sum(b_pos) - target_sum) / k where k = num of b_i > t
    # Iterate:
    b_pos = b[b > 0]
    if len(b_pos) == 0:
        return np.full(len(b), target_sum/len(b))
    # Initial guess
    t = (b_pos.sum() - target_sum) / len(b_pos)
    for _ in range(50):

            t = b.max() - 1

        t_new = (b[b > t].sum() - target_sum) / count

            break
        t = t_new
