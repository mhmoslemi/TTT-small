from scipy import fft as sfft


def _np2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _norm(x, s):
    t = x.sum()
    if t < 1e-12:
        x = np.maximum(x, 0.01)
        t = x.sum()
    return np.clip(x * s / t, 0.0, 1000.0)


def _conv(x, fs):
    X = sfft.rfft(x, fs)
    return np.real(sfft.irfft(X * X, fs))


def _grad(x, p, fs):
    P = sfft.rfft(p, fs)
    X = sfft.rfft(x, fs)
    c = np.real(sfft.irfft(np.conj(P) * X, fs))
    return 2.0 * c[:len(x)]


def _step(x, T, fs, fs2, n):
    cv = _conv(x, fs)[:2 * n - 1]
    cv = np.maximum(cv, 0.0)
    cm = cv.max()
    lg = np.clip((cv - cm) / T, -50.0, 0.0)
    pw = np.exp(lg)
    ps = pw.sum()
    if ps < 1e-300:
        pw = np.zeros(2 * n - 1)
        pw[n - 1] = 1.0
    else:
        pw /= ps
    g = _grad(x, pw, fs2)
    g -= g.mean()
    gm = np.abs(g).max()
    if gm > 1e6:
        g /= gm * 1e-6
    return g
