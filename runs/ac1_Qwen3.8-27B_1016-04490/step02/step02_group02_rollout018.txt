```python
def solve_lp_direction(f, M, n, max_constraints=200):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for active k, g >= 0"""
    from scipy.optimize import linprog
    from scipy import sparse as sp
    
    conv_f = fft_conv(f, next_pow2(2*n))  # f*f
    # Select top constraints
    k_top = np.argpartition(conv_f, -max_constraints)[-max_constraints:]
    k_top = k_top[k_top >= 0]  # ensure valid
    m_c = len(k_top)
    
    # Build constraint matrix (sparse)
    rows = []
    cols = []
    vals = []
    for idx_k in range(m_c):
        k = int(k_top[idx_k])
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            rows.append(idx_k)
            cols.append(j)
            vals.append(f[j])
    
    A = sp.csc_array((vals, (rows, cols)), shape=(m_c, n))
    b = M * np.ones(m_c)
    c = -np.ones(n)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return np.maximum(res.x, 0)
    return None
```

```python
def lp_direction(f, M, n, max_act=100):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for active k, g >= 0"""
    from scipy.sparse import csr_matrix
    from scipy.optimize import linprog
    
    conv_f = fft_conv(f, next_pow2(2*n))
    # Select active constraints (top max_act)
    idx = np.argpartition(conv_f, -max_act)[-max_act:]
    idx = idx[idx >= 0]
    idx = np.sort(idx)[::-1]  # descending by conv value
    mc = len(idx)
    
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for ii in range(mc):
        k = int(idx[ii])
        jlo = max(0, k - n + 1)
        jhi = min(k, n - 1)
        for j in range(jlo, jhi + 1):
            rows.append(ii)
            cols.append(j)
            vals.append(f[j])
```

```python
        x = best.copy()
        step = 0.006
        m = 32
        for _ in range(30000):
            if time.monotonic() > deadline - 15.0:
                break
            conv = fft_conv(x, fs)
            c2 = 2.0 * n * conv.max()
            if c2 < best_score - 1e-12:
                best_score = c2
                best = x.copy()
            compute_subgrad(conv, x, m, work)
            work -= work.mean()
            vn = np.linalg.norm(work)
            if vn < 1e-18:
                break
            x = np.maximum(x - step * x.mean() * work / vn, 0.0)
            ss = x.sum()
            if ss < 1e-14:
                break
            x = x / ss

    # Phase 4: Ultra-fine
    if time.monotonic() < deadline - 5.0:
        x = best.copy()
        step = 0.002
        m = 60
        for _ in range(20000):
            if time.monotonic() > deadline - 5.0:
                break
            conv = fft_conv(x, fs)
            c2 = 2.0 * n * conv.max()
            if c2 < best_score - 1e-12:
                best_score = c2
                best = x.copy()
            compute_subgrad(conv, x, m, work)
            work -= work.mean()
            vn = np.linalg.norm(work)
            if vn < 1e-18:
                break
            x = np.maximum(x - step * x.mean() * work / vn, 0.0)
            ss = x.sum()
            if ss < 1e-14:
                break
            x = x / ss

    sc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(sc):
        pn = np.maximum(parent, 1e-10)
        pn = pn / pn.sum()
        sc2 = float(evaluate_sequence(pn.tolist()))
        if np.isfinite(sc2):
            return [float(v) for v in pn]
        return [1.0]
    return [float(v) for v in best]
```
