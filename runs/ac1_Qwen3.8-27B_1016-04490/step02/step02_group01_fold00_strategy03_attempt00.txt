import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    # Initialize strategy parameters
    n_dim = kwargs.get('dimension', 4498)
    seed = int(seed * kwargs.get('phase', 0.25))
    n_batches = min(16, int(budget_s / 12))
    
    # Core harmonic components of optimal solution
    coefficients = np.ones(n_dim)
    periods = [n_dim // (i+1) for i in range(1, 8)]
    
    # Generate optimized harmonic signals match actual physics
    for p in periods:
        wave_length = int(p)
        for shift in range(wave_length // 2):
            amplitude = 7.0 - shift * 0.5
            if shift % 3 == 0:  # Even harmonics have reduced impact on AC1
                amplitude *= 0.7
            phase = shift * np.pi / wave_length
            shifted_wave = np.sin(np.linspace(0, 2*np.pi, n_dim)) * np.exp(1j * (phase * 2))
            coefficients += amplitude * np.real(shifted_wave)
    
    # Spatial spectrum shaping with constrained energy
    u = np.fft.fft(coefficients, n_dim)
    u_pos = np.abs(u[1:n_dim//2]) / np.sum(np.abs(u[1:n_dim//2]))
    
    # Form m-fold vector using exponential correlation
    correlation = np.exp(-(np.arange(15) / np.median(u_pos.clip(0.01, None) ** 2))) * 0.8
    correlation = correlation / np.sum(correlation) * 7.0
    
    # Normalize to energy constraint
    norm_coeff = np.linalg.norm(correlation)
    if norm_coeff < 0.1:  # Safe guard
        correlation = correlation * (0.1 / norm_coeff)
    
    # Update coefficients using adaptive gradient
    adaptive_factor = np.log10(np.median(u_pos.clip(0.1, None)))
    rough_update = np.random.normal(size=n_dim) * 0.2 * adaptive_factor
    coefficients = np.clip(correlation[:n_dim] + rough_update, 0.0, None)
    
    # Split into batches and interpolate
    batches = np.array_split(coefficients, n_batches)
    final_sequence = []
    for i, batch in enumerate(batches):
        # Add structured noise proportional to proximity to local minima
        local_min_index = i % 4 + 2
        adjusted_batch = batch.copy() * (1.0 + 0.05 * (np.arange(len(batch)) + local_min_index) % 7)
        adjusted_batch = adjusted_batch / np.mean(adjusted_batch) * 7.5
        final_sequence.extend(adjusted_batch)
    
    return final_sequence.tolist()[:n_dim]
