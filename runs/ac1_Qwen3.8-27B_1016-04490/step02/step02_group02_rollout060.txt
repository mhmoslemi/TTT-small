from scipy import fft as sfft


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    r = np.arange(1, 1 + n).astype(np.float64)
    cond = u - (cv - 1.0) / r > 0
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    k = int(idx[-1])
    theta = (cv[k] - 1.0) / (k + 1)
    return np.maximum(v - theta, 0.0)
