    c = np.fft.irfft(X * X, fs)[:2*n-1].real
    np.maximum(c, 0.0, out=c)
    return c

def grad_fft(p, x, cs, n):
    P = np.fft.rfft(p, cs)
    Xc = np.fft.rfft(x, cs)
    g = 2.0 * np.fft.irfft(P * np.conj(Xc), cs)[:n].real
    g -= g.mean()
    return g

def score(x):
    return float(evaluate_sequence(x.tolist()))
