import numpy as np
import time
from scipy import fft as sfft


def _np2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _norm(x, n, ts):
    s = x.sum()
    if s < 1e-12:
        return np.full(n, ts / n)
    return x * (ts / s)


def _fscore(x, L):
    X = sfft.rfft(x, L)
    b = np.maximum(np.real(sfft.irfft(X * X, L))[:2 * len(x) - 1], 0.0)
    s = x.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * len(x) * b.max() / (s * s)
