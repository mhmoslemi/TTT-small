import numpy as np
import time
from scipy import fft as sfft

def npow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def sc(x):
    return float(evaluate_sequence(x.tolist()))

def autocorr(x, fs):
    X = sfft.rfft(x, fs)
    c = np.real(sfft.irfft(X * X, fs))[:2 * len(x) - 1]
    np.maximum(c, 0.0, out=c)
    return c

def grad_fft(p, x, cs):
    P = sfft.rfft(p, cs)
    Xc = sfft.rfft(x, cs)
    g = 2.0 * np.real(sfft.irfft(P * np.conj(Xc), cs))[:len(x)]
    g -= g.mean()
    return g

def normalize(x, ts):
    np.clip(x, 0.0, 1000.0, out=x)
    s = x.sum()
    if s < 0.01:
        x[:] = ts / len(x)
        return
    x *= ts / s
    np.clip(x, 0.0, 1000.0, out=x)

def run_search(x0, ts, fs, cs, deadline, rng):
    n = len(x0)
    x = x0.copy()
    normalize(x, ts)
    best = x.copy()
    bsc = sc(best)
    it = 0
    while time.monotonic() < deadline:
        T = max(0.002, bsc * (0.08 * 0.9997 ** it))
        c = autocorr(x, fs)
        K = min(48, c.size)
        idx = np.argpartition(c, -K)[-K:]
        vals = c[idx]
        pv = np.exp((vals - vals.max()) / T)
        sw = pv.sum()
        p = np.zeros(c.size)
        if sw > 1e-300:
            p[idx] = pv / sw
        else:
            p[int(np.argmax(c))] = 1.0
        g = grad_fft(p, x, cs)
        gn = np.linalg.norm(g)
        if gn > 1e-14:
            lr = 0.35 * min(1.0, T / 0.04)
            x -= lr * (ts / n) * g / gn
            normalize(x, ts)
        it += 1
        if it % 300 == 0:
            s_ = sc(x)
            if np.isfinite(s_) and s_ < bsc - 1e-9:
                bsc = s_
                best = x.copy()
            elif it % 3000 == 0:
                x[:] = best + 0.015 * bsc * rng.standard_normal(n)
                normalize(x, ts)
    return best, bsc

def polish(best, bsc, ts, fs, cs, deadline):
    n = len(best)
    while time.monotonic() < deadline:
        c = autocorr(best, fs)
        improved = False
        top2 = np.argpartition(c, -2)[-2:]
        for w1, w2 in [(1.0, 0.0), (0.7, 0.3), (0.5, 0.5)]:
            pd = np.zeros(2 * n - 1)
            pd[top2[0]] = w1
            if w2 > 0:
                pd[top2[1]] = w2
            g = grad_fft(pd, best, cs)
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                continue
            for st in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001]:
                cand = best - st * ts * g / (n * gn)
                normalize(cand, ts)
                sv = sc(cand)
                if np.isfinite(sv) and sv < bsc - 1e-12:
                    bsc = sv
                    best = cand.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            top3 = np.argpartition(c, -3)[-3:]
            if len(set(top3.tolist())) < 2:
                break
            pv3 = np.exp(-c[top3] / max(c[top3].max() * 0.1, 0.01))
            pv3 /= pv3.sum()
            pd3 = np.zeros(2 * n - 1)
            for j in range(3):
                pd3[top3[j]] = pv3[j]
            g3 = grad_fft(pd3, best, cs)
            gn3 = np.linalg.norm(g3)
            if gn3 > 1e-14:
                for st in [0.03, 0.015, 0.007, 0.003]:
                    cand = best - st * ts * g3 / (n * gn3)
                    normalize(cand, ts)
                    sv = sc(cand)
                    if np.isfinite(sv) and sv < bsc - 1e-12:
                        bsc = sv
                        best = cand.copy()
                        improved = True
                        break
        if not improved:
            break
    return best, bsc

    if not np.isfinite(bsc) or not np.all(np.isfinite(best)):
