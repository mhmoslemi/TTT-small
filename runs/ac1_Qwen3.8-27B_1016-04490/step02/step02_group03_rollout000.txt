def inits(n):
    L = []
    j = np.arange(n, dtype=np.float64)
    m = 0.5 * (n - 1)
    L.append(np.ones(n, dtype=np.float64).copy())
    L.append(0.5 * (1.0 + np.cos(np.pi * j / (n - 1))).copy())
    L.append(0.5 * (1.0 - np.cos(np.pi * j / (n - 1))) + 0.01)
    for p in (-0.3, -0.5, -0.8, -1.2, -1.8, -2.5):
        L.append((j + 1.0) ** p)
    for a in (0.5, 1.0, 1.5, 2.5, 4.0):
        L.append((1.0 - np.abs(j - m) / (m + 1.0)) ** a + 0.01)
    for f in (1, 2, 3, 5, 8):
        mod = 1.0 + 0.4 * np.cos(2.0 * np.pi * f * j / n)
        L.append((0.5 * (1.0 + np.cos(np.pi * j / (n - 1))) * mod + 0.01).copy())
    t1 = 0.35 * n
    t2 = 0.65 * n
    sd = 0.1 * n
    L.append((np.exp(-((j - t1) / sd) ** 2) + np.exp(-((j - t2) / sd) ** 2) + 0.005).copy())
    d = np.minimum(j, n - 1 - j) + 1.0
    for pv in (0.5, 1.0, 2.0, 3.0):
        L.append(d ** pv)
    for pv in (0.5, 1.0, 2.0):
        L.append((np.abs(j - m) / (m + 1.0)) ** pv + 0.05)
    return L
