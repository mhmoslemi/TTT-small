    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    try:
        par = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except Exception:
        par = np.ones(512)
    if par.ndim != 1 or par.size < 2 or not np.all(np.isfinite(par)):
        par = np.ones(512)

    n = par.size
    ts = np.sqrt(2.0 * n)
    fs = npow2(2 * n - 1)
    cs = npow2(3 * n - 2)

    x = par.copy()
    x = renorm(x, ts)
    best = x.copy()
    bsc = sc(best)

    # Phase 1: Heavy-ball momentum gradient descent
    v = np.zeros(n)
    mu = 0.80
    T = max(0.04, bsc * 0.08)
    T_init = T
    T_min = 0.002
    base_lr = 0.055
    it = 0
    stuck = 0
    eval_iv = 300

    while time.monotonic() < deadline - 30.0:
        c = autocorr(x, fs, n)
        K = max(4, min(80, int(50 * (T / T_init + 0.2))))
        p = topk_weights(c, T, K)
        g = grad_fft(p, x, cs, n)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break

        step_mag = base_lr * (ts / n) * min(1.0, 0.5 + T / T_init)
        v = mu * v - step_mag * (g / gn)
        x = x + v
        renorm(x, ts)

        T = max(T_min, T * 0.9985)
        it += 1

        if it % eval_iv == 0:
            sv = sc(x)
            if np.isfinite(sv) and sv < bsc - 1e-10:
                bsc = sv
                best = x.copy()
                stuck = 0
            else:
                stuck += eval_iv
            if stuck > 2500:
                x = best + 0.015 * (ts / n) * rng.standard_normal(n)
                renorm(x, ts)
                v = np.zeros(n)
                T = T_init * 0.6
                stuck = 0

    # Phase 2: Fine-tune without momentum, fewer peaks
    x = best.copy()
    T2 = 0.003
    lr2 = 0.20
    for it2 in range(8000):
        if time.monotonic() > deadline - 20.0:
            break
        c = autocorr(x, fs, n)
        K2 = max(2, min(20, int(8 + 4 * T2 / 0.003)))
        p = topk_weights(c, T2, K2)
        g = grad_fft(p, x, cs, n)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        x -= lr2 * (ts / n) * (g / gn)
        renorm(x, ts)
        T2 = max(0.0005, T2 * 0.999)
        if it2 % 400 == 0:
            sv = sc(x)
            if np.isfinite(sv) and sv < bsc - 1e-10:
                bsc = sv
                best = x.copy()

    # Phase 3: Polish with exact subgradient at single max peak
    xp = best.copy()
    for _ in range(300):
        if time.monotonic() > deadline - 5.0:
            break
        c = autocorr(xp, fs, n)
        ks = int(np.argmax(c))
        pd = np.zeros(2 * n - 1)
        pd[ks] = 1.0
        g = grad_fft(pd, xp, cs, n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        imp = False
        for st in (0.03, 0.015, 0.007, 0.003, 0.001):
            cand = xp - st * (ts / n) * g / gn
            renorm(cand, ts)
            sv = sc(cand)
            if np.isfinite(sv) and sv < bsc:
                bsc = sv
                best = cand.copy()
                xp = best.copy()
                imp = True
                break
        if not imp:
            break

    # Phase 4: Random micro-perturbations around best
    for _ in range(200):
        if time.monotonic() > deadline - 1.5:
            break
        for amp in (0.004, 0.002, 0.001):
            cand = best + amp * (ts / n) * rng.standard_normal(n)
            renorm(cand, ts)
            sv = sc(cand)
            if np.isfinite(sv) and sv < bsc:
                bsc = sv
                best = cand.copy()
                break

    if not np.isfinite(bsc):
        x = np.clip(x, 0.0, 1000.0)
        if x.sum() < 0.01:
            x = np.ones(n)
        return [float(vv) for vv in x]

    return [float(vv) for vv in best]
