    sz = 1
    while sz < m:
        sz <<= 1
    fx = np.fft.rfft(x, sz)
    fy = np.fft.rfft(y, sz)
    return np.fft.irfft(fx * fy, sz)[:m]

    cols = []

    n = len(a)

        b, sc = _optimize(scored[k][1].copy(), dl, rng, rows, cols)
        if sc < best_sc:
            best_sc = sc
            best_all = b.copy()
    rem2 = deadline - time.monotonic()
    if rem2 > 25:
        b2, sc2 = _optimize(best_all.copy(), deadline - 6.0, rng, rows, cols, mult_per_lp=40)
        if sc2 < best_sc:
            best_sc = sc2
            best_all = b2.copy()
    best_all = np.maximum(best_all, 0.0)
    if best_all.sum() < 0.01:
        best_all = np.ones(n) / n
    return [float(v) for v in best_all]
