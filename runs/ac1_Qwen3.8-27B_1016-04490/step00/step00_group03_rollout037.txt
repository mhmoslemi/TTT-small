def build_conv_matrix(f):
    """Build sparse matrix A where A[k,j] = f[j] if 0<=k-j<n."""
    n = f.size
    rows = (np.arange(n, dtype=np.int32)[:, None] + np.arange(n, dtype=np.int32)[None, :]).ravel()
    cols = np.repeat(np.arange(n, dtype=np.int32), n)
    vals = np.repeat(f, n)
    return scipy.sparse.coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()

def lp_step(f, M, time_limit=60):
    """Solve max 1'b s.t. (f*b)_k <= M, b>=0. Return b or None."""
    n = f.size

    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs', options={'time_limit': time_limit})
        if res.success and res.x is not None:
