import numpy as np
import time


def conv_fft(a):
    n = len(a)
    size = 2 * n
    fa = np.fft.rfft(a, size)
    b = np.fft.irfft(fa * fa, size)
    return b[:2 * n - 1]


def evaluate_fast(a):
    n = len(a)
    b = conv_fft(a)
    max_b = float(np.max(b))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    n = len(height_sequence_1)
    a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0).copy()

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    idx = np.arange(n, dtype=np.float64)
    ipos = np.arange(n)

    inits = [
        rng.uniform(0.0, 1000.0, n),
        np.where(rng.random(n) < 0.5, 1000.0, 0.0),
        np.where(ipos % 3 < 2, 1000.0, 0.0),
        np.where(ipos % 7 < 5, 1000.0, 0.0),
        np.where(ipos % 13 < 8, 1000.0, 50.0),
        500.0 + 450.0 * np.sin(2.0 * np.pi * idx * 3.0 / n),
        500.0 + 450.0 * np.sin(2.0 * np.pi * idx * 5.0 / n + 0.7),
        500.0 + 450.0 * np.cos(2.0 * np.pi * idx * 11.0 / n),
        1000.0 * (0.3 + 0.7 * np.abs(np.sin(2.0 * np.pi * idx * 2.0 / n))),
        np.where(ipos < n // 2, 1000.0, 200.0),
        1000.0 * np.where((ipos % 5) < 3, 1.0, 0.2),
    ]

    for init in inits:
        init = np.clip(init, 0.0, 1000.0)
        s = float(evaluate_sequence(init.tolist()))
        if np.isfinite(s) and s < best_score:
            best = init.copy()
            best_score = s

    a = best.copy()
    eval_count = 0

    while time.monotonic() < deadline:
        b = conv_fft(a)
        k_star = int(np.argmax(b))
        max_b = float(b[k_star])
        sum_a = float(np.sum(a))

        if sum_a < 0.01:
            a = np.full(n, 500.0)
            continue

        a_try = a.copy()
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)

        if hi >= lo:
            m = np.arange(lo, hi + 1)
            other = k_star - m
            weight = 2.0 * a[other]
            diag = m == other
            if np.any(diag):
                weight[diag] = a[m[diag]]
        else:
            weight = np.zeros(n)

        batch = min(500, max(10, n // 10))
        ii = rng.integers(0, n, batch)
        jj = rng.integers(0, n, batch)
        mask = weight[ii] > weight[jj]
        ii = ii[mask]
        jj = jj[mask]
        if len(ii) > 0:
            d = 40.0 * rng.random(len(ii))
            a_try[ii] -= d
            a_try[jj] += d

        if rng.random() < 0.25:
            wn = int(rng.integers(max(1, n // 80), max(2, n // 15)))
            ws = int(rng.integers(0, n - wn))
            a_try[ws:ws + wn] += rng.normal(0.0, 60.0, wn)

        a_try = np.clip(a_try, 0.0, 1000.0)
        s_try = float(evaluate_sequence(a_try.tolist()))
        eval_count += 1

        if np.isfinite(s_try) and s_try < best_score:
            best = a_try.copy()
            best_score = s_try
            a = a_try.copy()
        else:
            r = rng.random()
            if r < 0.05:
                a = a_try.copy()
            else:
                a = best.copy()

    final_score = float(evaluate_sequence([float(x) for x in best]))
    if not np.isfinite(final_score):
        best = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)

    return [float(x) for x in best]
