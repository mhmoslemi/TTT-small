import time

def conv_self(a):
    """Full self-convolution via FFT. O(n log n)."""
    n = a.size
    size = 2 * n - 1
    m = 1
    while m < size:
        m <<= 1
    fa = np.fft.rfft(a, m)
    return np.fft.irfft(fa * fa, m)[:size]

def ac1_obj(a):
    """AC1 objective (same formula as evaluator, but no list conversion)."""
    n = a.size
    s = a.sum()
    if s < 0.01:
        return 1e18
    c = conv_self(a)
    return 2.0 * n * c.max() / (s * s)

def ac1_grad(a):
    """Subgradient of AC1 objective. Returns (gradient, tau, max_b)."""
    n = a.size
    s = a.sum()
    c = conv_self(a)
    tau = int(np.argmax(c))
    mb = c[tau]
    v = np.zeros(n)

    lo1 = max(0, tau - n + 1)
    hi1 = min(n - 1, tau)
