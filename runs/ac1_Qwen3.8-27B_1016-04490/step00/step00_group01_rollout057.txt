import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse


def fft_conv(a):
    """Linear autocorrelation (convolution of a with itself) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)


def fast_score(a):
    """Matches evaluate_sequence for clamped nonnegative arrays."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = fft_conv(a)
    return float(2.0 * n * c.max() / (s * s))


def make_lp_matrix(a, lags, n):
    """Sparse K x n matrix: row k has a[k-j] in column j for valid j."""
    K = len(lags)
    rows, cols, vals = [], [], []
    for idx, k in enumerate(lags):
        j0 = max(0, k - n + 1)
        j1 = min(k, n - 1)
        if j1 < j0:
            continue
        js = np.arange(j0, j1 + 1)
        rows.extend([idx] * len(js))
        cols.extend(js.tolist())
        vals.extend(np.asarray(a[k - js], dtype=np.float64).tolist())
    if len(vals) == 0:
        return sparse.csc_matrix((K, n))
    return sparse.csc_matrix((vals, (rows, cols)), shape=(K, n))
