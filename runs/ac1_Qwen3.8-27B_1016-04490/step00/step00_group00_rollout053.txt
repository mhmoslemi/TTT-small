        a_new = a + alpha * direction / dn
    else:
        a_new = a + rng.normal(0.0, 0.01, n)
    return np.clip(a_new, 0.0, 1000.0), M


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(60.0, float(budget_s) - 25.0)

    try:
        n = len(height_sequence_1)
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except Exception:
        n = 128
        parent = np.ones(n)
    if n < 2 or not np.all(np.isfinite(parent)):
        n = 128
        parent = np.ones(n)

    # Initialize from structured (non-uniform) starting point
    a = _cos_init(rng, n)
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Check parent as well
    p_sc = float(evaluate_sequence(parent.tolist()))
    if np.isfinite(p_sc) and p_sc < best_score:
        best = parent.copy()
        best_score = p_sc
        a = parent.copy()

    alpha = 0.05
    stall = 0

    while time.monotonic() < deadline:
        # Gradient step
        a_new, _ = _grad_step(a, alpha, rng)
        score = float(evaluate_sequence(a_new.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a_new.copy()
            best_score = score
            a = a_new
            stall = 0
            alpha = min(alpha * 1.03, 0.8)
        else:
            stall += 1
            if stall >= 12:
                # Block perturbation (medium scale)
                a_new = _block_perturb(a, rng, 0.10, max(3, n // 60))
                score2 = float(evaluate_sequence(a_new.tolist()))
                if np.isfinite(score2) and score2 < best_score:
                    best = a_new.copy()
                    best_score = score2
                    a = a_new
                    stall = 0
                    alpha = 0.05
                else:
                    a = best.copy()
                    alpha = max(0.008, alpha * 0.82)
                    stall = 0
            elif stall >= 4:
                # Small targeted nudge on a random segment
                w = int(rng.integers(max(2, n // 200), max(4, n // 40)))
                s = int(rng.integers(0, max(1, n - w)))
                a[s:s + w] *= float(np.exp(rng.normal(0.0, 0.04)))
                a = np.clip(a, 0.0, 1000.0)
                score3 = float(evaluate_sequence(a.tolist()))
                if np.isfinite(score3) and score3 < best_score:
                    best = a.copy()
                    best_score = score3
                    stall = 0

    return [float(x) for x in np.clip(best, 0.0, 1000.0)]
