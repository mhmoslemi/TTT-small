import time


def _symm(b):
    return 0.5 * (b + b[::-1])


def _norm1(b):
    s = b.sum()
    if s < 1e-15:
        return np.ones_like(b) / len(b)
    return b / s


def _topk_grads(b, k_star_list):
    n = len(b)
    grad = np.zeros(n)
    for ks in k_star_list:
        m_lo = max(0, ks - n + 1)
        m_hi = min(n - 1, ks)
        if m_lo > m_hi:
            continue
        m_arr = np.arange(m_lo, m_hi + 1)
        comp = ks - m_arr
        grad[m_arr] += 2.0 * b[comp]
        if ks % 2 == 0:
            mid = ks // 2
            if m_lo <= mid <= m_hi:
                grad[mid] -= b[mid]
    return grad
