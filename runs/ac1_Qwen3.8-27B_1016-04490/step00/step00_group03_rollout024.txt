from scipy import sparse
from scipy.optimize import linprog

def build_conv_matrix(a):

    n = len(a)
    rows = []
    cols = []
    vals = []

import numpy as np
import time
from scipy import sparse
from scipy.optimize import linprog


def build_conv_pattern(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def lp_refine(a, n, max_time=60.0):
    """One or a few LP-based refinement steps."""
    if max_time <= 0:
        return a
    start = time.monotonic()
    pat_r, pat_c = build_conv_pattern(n)
    kc = pat_r - pat_c
    t_mix = 0.1
    c = np.convolve(a, a)
    M = c.max()
    for _ in range(20):
        if time.monotonic() - start > max_time - 10:
            break
        vals = a[kc]
        C = sparse.csr_matrix((vals, (pat_r, pat_c)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n)
        b_ub = M * np.ones(2 * n - 1)
        rem = max_time - (time.monotonic() - start) - 5
        if rem < 5:
            break
        res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=(0, None),
                      method='highs', options=dict(time_limit=rem))
        if res.status != 0:
            break
        g = np.clip(res.x, 0, None)
        gs = g.sum()
        if gs < 1e-12:
            break
        g /= gs
        a = (1 - t_mix) * a + t_mix * g
        a = np.clip(a, 1e-12, None)
        a /= a.sum()
        c = np.convolve(a, a)
        M = c.max()
    return a
