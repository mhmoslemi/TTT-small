import time
from scipy import sparse
from scipy.optimize import linprog


def _conv(a):
    return np.convolve(a, a)


def _mult_step(a, eta, top_p):
    n = a.size
    c = _conv(a)
    M = c.max()
    if M < 1e-30:
        return a
    p = min(top_p, c.size)
    ki = np.argpartition(c, -p)[-p:]
    g = np.zeros(n)
    for k in ki:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]
    g *= 2.0
    a2 = a / (1.0 + eta * g / M)
    a2 = np.maximum(a2, 0.0)
    s = a2.sum()
    if s > 1e-15:
        a2 /= s
    else:
        a2 = a.copy()
    return a2


def _lp_step(a, max_time):
    n = a.size
    c = _conv(a)
    M = c.max()
    t0 = time.monotonic()
    rem = max_time - (time.monotonic() - t0)
    if rem < 2.0:
        return None

        dd = np.repeat(np.arange(n, dtype=np.int64), n)

def _chip_step(v, n_peaks=5):
    n = v.size
    c = _conv(v)
    p = min(n_peaks, c.size)
    ki = np.argpartition(c, -p)[-p:]
    a2 = v.copy()
    for k in ki:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # pairs: (j, k-j) for j in [lo, hi]
        js = np.arange(lo, hi + 1)
        ps = k - js
        # reduce the max of each pair
        aj = a2[js]
        ap = a2[ps]
        m = np.maximum(aj, ap)
        mn = np.minimum(aj, ap)
        # reduce m by small fraction of mn
        delta = 0.05 * mn

import numpy as np
import time
from scipy import sparse
from scipy.optimize import linprog


def _conv(a):
    return np.convolve(a, a)


def _mult_step(a, eta, top_p):
    n = a.size
    c = _conv(a)
    M = c.max()
    if M < 1e-30:
        return a
    p = min(top_p, c.size)
    ki = np.argpartition(c, -p)[-p:]
    g = np.zeros(n)
    for k in ki:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]
    g *= 2.0
    a2 = a / (1.0 + eta * g / M)
    a2 = np.maximum(a2, 0.0)
    s = a2.sum()
    if s > 1e-15:
        a2 /= s
    else:
        a2 = a.copy()
    return a2


def _lp_step(a, max_time):
    n = a.size
    c = _conv(a)
    M = c.max()
    if M < 1e-30:
        return None
    t0 = time.monotonic()
    rem = max_time - (time.monotonic() - t0)
    if rem < 2.0:
        return None
    try:
        jj = np.tile(np.arange(n, dtype=np.int64), n)
        dd = np.repeat(np.arange(n, dtype=np.int64), n)
        kk = jj + dd
        vv = np.repeat(a, n)
        A = sparse.coo_matrix((vv, (kk, jj)), shape=(2 * n - 1, n)).tocsr()
        del jj, dd, kk, vv
        rem2 = max_time - (time.monotonic() - t0)
        if rem2 < 2.0:
            return None
        res = linprog(
            -np.ones(n), A, np.full(2 * n - 1, M),
            bounds=[(0.0, None)] * n,
            method='highs',
            options={'time_limit': max(1.0, rem2 - 1.0)}
        )
        del A
        if res.status == 0 and res.x is not None:
            b = np.asarray(res.x, dtype=np.float64)
            b = np.maximum(b, 0.0)
            bs = b.sum()
            if bs > 1e-15:
                b /= bs
                return b
    except Exception:
        pass
    return None
