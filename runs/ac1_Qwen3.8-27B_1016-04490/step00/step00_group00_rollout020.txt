import time
import scipy.sparse as sp
from scipy.optimize import linprog

def build_conv_structure(n):
    """Build the sparse structure (rows, cols pattern) for the conv matrix."""
    i_vals = np.tile(np.arange(n), n)
    j_vals = np.repeat(np.arange(n), n)
    rows = i_vals + j_vals
    return rows, j_vals  # rows and cols pattern, data will be set separately

def make_conv_matrix(rows, cols, a):
    """Build the actual sparse conv matrix given the structure and values a."""
    n = len(a)
    data = np.tile(a, n)
    C = sp.coo_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return C.tocsr()

def solve_lp_step(C, n):
    """Solve the LP: min t s.t. C@b <= t, sum(b)=1, b>=0, t>=0."""
    # Variables: [b_0, ..., b_{n-1}, t]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # A_ub: [C, -1] (2n-1) x (n+1)
    one_col = sp.csr_matrix(np.full((2*n-1, 1), -1.0))
    A_ub = sp.hstack([C, one_col], format='csr')
    b_ub = np.zeros(2*n-1)

    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)

    if res.success:

        return b, res.x[-1]

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog


def _build_coo_structure(n):
    off = np.tile(np.arange(n), n)
    jv = np.repeat(np.arange(n), n)
    rv = (off + jv).astype(np.int32)
    cv = jv.astype(np.int32)
    C = sp.coo_matrix((np.zeros(n * n), (rv, cv)), shape=(2 * n - 1, n))
    return C


def _solve_lp(C, n, time_budget):
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    nc = 2 * n - 1
    neg_one = sp.csr_matrix(np.full((nc, 1), -1.0))
    A_ub = sp.hstack([C, neg_one], format='csr')
    b_ub = np.zeros(nc)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'time_limit': max(3.0, time_budget)})
    return res


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 5.0

    parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
    parent = np.clip(parent, 0.0, 1000.0)
    n = parent.size
    s = parent.sum()
    a = parent / s if s > 0.01 else np.ones(n) / n

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Try a couple of alternative initializations
    for init in _alt_inits(n, rng):
        sc = float(evaluate_sequence(init.tolist()))
        if sc < best_score:
            best = init.copy()
            best_score = sc

    # Build COO structure once

    stagnant = 0
    while time.monotonic() < deadline - 40.0:
        rem = deadline - 40.0 - time.monotonic()
        if rem < 8.0:
            break
        C0.data[:] = np.tile(a, n)
        C = C0.tocsr()
        res = _solve_lp(C, n, rem * 0.5)
        del C
        if res.success:
            b = np.maximum(res.x[:n], 0.0)
            sb = b.sum()
            if sb > 0.01:
                b /= sb
                sc = float(evaluate_sequence(b.tolist()))
                if sc < best_score:

                    best_score = sc
                    stagnant = 0
                else:
                    stagnant += 1
                a = b
            if stagnant >= 8:
                break
        else:
            break

    # Local search polish
    while time.monotonic() < deadline - 2.0:
        cand = best.copy()
        w = max(1, n // 80)

        sc = float(evaluate_sequence(cand.tolist()))
        if sc < best_score:
            best = cand.copy()
            best_score = sc

    return [float(x) for x in best]


def _alt_inits(n, rng):
    """Yield a few structured initializations normalized to sum=1."""
    # Cosine bump
    x = np.arange(n, dtype=np.float64)
    t = x / (n - 1)
    for p in (1.5, 2.0, 3.0, 4.0):
        v = (0.5 + 0.5 * np.cos(2 * np.pi * t)) ** p
        v -= v.min()
        if v.sum() > 0.01:
            v /= v.sum()
            yield v.copy()
    # Two-plateau
    v = np.zeros(n)
    v[:n // 3] = 1.0
    v[2 * n // 3:] = 1.0
    v /= v.sum()
    yield v
    # Random

    v += 0.1
    v /= v.sum()
    yield v

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog


def _build_coo(n):
    off = np.tile(np.arange(n), n)
    jv = np.repeat(np.arange(n), n)
    rv = (off + jv).astype(np.int32)
    cv = jv.astype(np.int32)
    C = sp.coo_matrix((np.zeros(n * n), (rv, cv)), shape=(2 * n - 1, n))
    del off, jv, rv, cv
    return C


def _lp_step(C, n, tlim):
    nc = 2 * n - 1
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    neg = sp.csr_matrix(np.full((nc, 1), -1.0))
    A_ub = sp.hstack([C, neg], format='csr')
    b_ub = np.zeros(nc)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)
    return linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                   bounds=bounds, method='highs',
                   options={'time_limit': max(3.0, tlim), 'presolve': True})


def _alt_inits(n, rng):
    x = np.arange(n, dtype=np.float64)
    t = x / max(1, n - 1)
    for p in (1.0, 2.0, 3.0, 5.0):
        v = (0.5 + 0.5 * np.cos(2.0 * np.pi * t)) ** p
        v -= v.min()
        if v.sum() > 1e-6:
            yield v / v.sum()
    v = np.zeros(n)
    v[:n // 3] = 1.0
    v[2 * n // 3:] = 1.0
    yield v / v.sum()
    v = np.ones(n)

    v[idx] = 0.3
    yield v / v.sum()
    v = rng.random(n) + 0.05
    yield v / v.sum()
