def lp_update(p, n, alpha, mix, inner, inner_deadline, rng):
    S = float(p.sum())
    M = float(np.convolve(p,p).max())
    if M <= 0 or not math.isfinite(M): return p.copy(), M
    g = p.copy()
    for _ in range(inner):
        if time.monotonic() > inner_deadline: break
        y = np.convolve(p, g)
        maxy = float(y.max())
        if maxy <= 0 or not math.isfinite(maxy): break
        if maxy > M:
            scale = M / maxy
            g *= scale
            y = y * scale
            maxy = M
        u = y / M
        slack = 1.0 - u
        np.clip(slack, 0.0, None, out=slack)
        h = np.correlate(slack, p, mode='valid')
        hmax = float(h.max())
        if hmax <= 1e-300 or not math.isfinite(hmax): break
        # damp alpha sometimes
        g *= np.exp(alpha * (h / hmax))
        s = float(g.sum())
        if s <= 0 or not math.isfinite(s): break
        g *= S / s
        g = np.clip(g, 0.0, 1000.0)
        s = float(g.sum())
        if s > 0: g *= S / s
    g = np.clip(g,0,1000)
    s=float(g.sum())
    if s<=0: return p.copy(), M
    g*=S/s
    q = (1.0 - mix)*p + mix*g
    q=np.clip(q,0,1000)
    s=float(q.sum())
    if s>0: q*=S/s
    return q, M

def normalize_vec(v, total=1.0):
    v = np.asarray(v, dtype=np.float64)
    v = np.clip(v, 0.0, 1000.0)
    s = float(v.sum())
    if s <= 0.0 or not math.isfinite(s):
        v = np.zeros_like(v)
        return v
    v *= total / s
    return v


def fast_score(v, n):
    v = np.asarray(v, dtype=np.float64)
    s = float(v.sum())
    if s <= 1e-12 or not math.isfinite(s):
        return float("inf")
    c = np.convolve(v, v)
    m = float(c.max())
    if not math.isfinite(m) or m <= 0.0:
        return float("inf")
    return float(2.0 * n * m / (s * s))

def make_deterministic_seqs(n):
    out = []
    x = np.linspace(0.0, 1.0, n)

    out.append(np.ones(n, dtype=np.float64))
    for a in (0.5, 1.0, 2.0):
        out.append(np.power(np.maximum(x, 1e-9), a))
        out.append(np.power(np.maximum(1.0 - x, 1e-9), a))
        out.append(np.exp(2.0 * a * (x - 0.5)))
        out.append(np.exp(-2.0 * a * (x - 0.5)))
    for m in (1, 2, 3, 4, 6, 8):
        for ph in (0.0, 0.7, 1.4, math.pi):
            out.append(1.0 + 0.45 * np.cos(m * math.pi * x + ph))
            out.append(1.0 + 0.45 * np.sin(m * math.pi * x + ph))
    for f in (0.3, 0.5, 0.7):
        k = int(round(f * n))
        for r in (0.3, 0.5, 0.7):
            s = np.empty(n, dtype=np.float64)
            s[:k] = r
            s[k:] = 1.0
            out.append(s)

def make_cosine_seq(n, rng, terms):

def lp_update(p, n, alpha, mix, inner, inner_deadline):

    if s <= 0.0 or not math.isfinite(s):
        return p.copy()

        if time.monotonic() > inner_deadline:

        u = y / M

        if wsum <= 1e-300 or not math.isfinite(wsum):

        slack = slack / wsum
        d = np.zeros(n, dtype=np.float64)
        for kk, ww in zip(idx, slack):
            add_lag_derivative(p, n, kk, float(ww), d)
        dmax = float(d.max())
        if dmax <= 1e-300 or not math.isfinite(dmax):
            break
        factor = np.exp(float(alpha) * (d / dmax))
        g *= factor
        sm = float(g.sum())
        if sm <= 1e-300 or not math.isfinite(sm):

        g *= s / sm
        g = np.clip(g, 0.0, 1000.0)

    sm = float(g.sum())
    if sm <= 0.0 or not math.isfinite(sm):

    q = (1.0 - float(mix)) * p + float(mix) * g

def peak_shave(p, n, rng):
    c = np.convolve(p, p)
    m = float(c.max())
    if m <= 0.0 or not math.isfinite(m):
        return p.copy()

    tau = m * float(rng.uniform(0.84, 0.995))
    active = np.nonzero(c >= tau)[0]
    if active.size == 0:
        active = np.array([int(np.argmax(c))], dtype=np.int64)
    if active.size > 64:
        active = np.argpartition(c, 64)[-64:]
    weights = c[active] - tau
    wsum = float(weights.sum())
    if wsum <= 1e-300 or not math.isfinite(wsum):
        weights = np.ones(active.size, dtype=np.float64)
    else:
        weights = weights / wsum

    d = np.zeros(n, dtype=np.float64)
    for kk, ww in zip(active, weights):
        add_lag_derivative(p, n, int(kk), float(ww), d)

    dmax = float(d.max())
    q = p.copy()
    if dmax > 1e-300 and math.isfinite(dmax):
        eta = float(rng.uniform(0.05, 0.60))
        q *= np.exp(-eta * (d / dmax))

    q *= np.exp(rng.normal(0.0, 0.03, size=n))
    q = np.clip(q, 0.0, 1000.0)
    sm = float(q.sum())
    if sm > 0.0 and math.isfinite(sm):
        q *= 1.0 / sm
    else:
        q = p.copy()
    return q


def block_perturb(p, rng):
    q = p.copy()
    n = q.size
    for _ in range(3):
        lo = max(1, n // 200)
        hi = max(2, n // 12)
        if hi <= lo:
            hi = lo + 1
        width = int(rng.integers(lo, hi))
        width = min(max(1, width), n)
        stop = n - width + 1
        start = int(rng.integers(0, stop))
        q[start:start + width] *= float(rng.uniform(0.5, 1.7))
    if rng.random() < 0.25:
        q *= np.exp(rng.normal(0.0, 0.05, size=n))
    q = np.clip(q, 0.0, 1000.0)
    sm = float(q.sum())
    if sm > 0.0 and math.isfinite(sm):
        q *= 1.0 / sm
    else:
        q = p.copy()
    return q


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    budget = max(30.0, float(budget_s))
    hard_deadline = time.monotonic() + max(1.0, budget - 7.0)

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    if parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)
    n = int(parent.size)
    parent = np.clip(parent, 0.0, 1000.0)
    ps = float(parent.sum())
    if ps > 0.0 and math.isfinite(ps):
        parent *= 1.0 / ps
    else:
        parent = np.ones(n, dtype=np.float64) / float(n)

    pool = [[parent.copy(), fast_score(parent, n)]]

    for seq in make_deterministic_seqs(n):
        pool.append([seq, fast_score(seq, n)])

    init_deadline = time.monotonic() + min(75.0, 0.18 * budget)
    generated = 0
    max_generated = 260 if n <= 20000 else 80
    while time.monotonic() < init_deadline and generated < max_generated:
        generated += 1
        seq = make_cosine_seq(n, rng, int(rng.integers(3, 8)))
        pool.append([seq, fast_score(seq, n)])
        seq = make_block_seq(n, rng, int(rng.integers(4, 11)))
        pool.append([seq, fast_score(seq, n)])

    valid_pool = []
    for item in pool:
        if np.isfinite(item[1]) and item[0].size == n:
            valid_pool.append([item[0].copy(), float(item[1])])
    if not valid_pool:
        valid_pool = [[parent.copy(), 2.0]]
    valid_pool.sort(key=pool_key)

    best_seq = valid_pool[0][0].copy()
    best_score = float(valid_pool[0][1])
    n_starts = min(5, len(valid_pool))
    starts = [[valid_pool[i][0].copy(), float(valid_pool[i][1])] for i in range(n_starts)]

    temperature = 0.06
    search_deadline = time.monotonic() + max(0.0, budget - 8.0)

    while time.monotonic() < search_deadline:
        for si in range(n_starts):
            if time.monotonic() >= search_deadline:
                break
            p = starts[si][0]
            sc = starts[si][1]
            r = float(rng.random())

            if r < 0.38:
                sub_deadline = time.monotonic() + min(0.65, max(0.08, (search_deadline - time.monotonic()) * 0.008))
                alpha = float(rng.uniform(0.05, 0.26))
                mix = float(rng.uniform(0.10, 0.35))
                q = lp_update(p, n, alpha, mix, int(rng.integers(4, 10)), sub_deadline)
            elif r < 0.90:
                q = peak_shave(p, n, rng)
            else:
                q = block_perturb(p, rng)

            qs = fast_score(q, n)
            if math.isfinite(qs):
                if qs < best_score:
                    best_score = float(qs)
                    best_seq = q.copy()
                accept = qs <= sc * (1.0 + temperature)
                if not accept and qs < best_score + 0.08:
                    accept = float(rng.random()) < 0.25
                if accept:
                    starts[si][0] = q
                    starts[si][1] = float(qs)
                    temperature = max(0.005, temperature * 0.998)

    if time.monotonic() < search_deadline - 3.0:
        p = best_seq.copy()
        sc = best_score
        polish_deadline = time.monotonic() + min(60.0, max(1.0, search_deadline - time.monotonic()))
        while time.monotonic() < polish_deadline:
            q = peak_shave(p, n, rng)
            qs = fast_score(q, n)
            if math.isfinite(qs) and qs < sc:
                sc = float(qs)
                p = q.copy()
                best_score = sc
                best_seq = p.copy()

    ans = []
    for x in best_seq:
        x = float(x)
        if not math.isfinite(x):
            x = 0.0
        if x < 0.0:
            x = 0.0
        if x > 1000.0:
            x = 1000.0
        ans.append(x)

    if len(ans) == 0:
        ans = [1.0 / float(max(1, n))]

    final_score = evaluate_sequence(ans)
    if not math.isfinite(final_score):
        fb = np.ones(max(1, n), dtype=np.float64) / float(max(1, n))
        ans = [float(x) for x in fb]
        evaluate_sequence(ans)

    return ans
