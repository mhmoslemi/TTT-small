import numpy as np
import time


def compute_load(a, hot):
    """Compute load[i] = sum_j a[j]*hot[i+j] via one convolution."""
    n = a.size
    conv = np.convolve(a[::-1], hot)
    return conv[n - 1: n - 1 + n]


def fast_score(a, n):
    """Compute 2*n*max(convolve(a,a)) assuming sum(a)=1."""
    c = np.convolve(a, a)
    return float(2.0 * n * c.max())

    s = a.sum()

        a[:] = 1.0
        s = float(n)
    a /= s

    best_score = fast_score(a, n)

    t_alpha = 0.20
    hot_frac = 0.90

    while time.monotonic() < deadline:

        cur_score = fast_score(a, n)
        if cur_score < best_score - 1e-14:

            best_score = cur_score

        else:
