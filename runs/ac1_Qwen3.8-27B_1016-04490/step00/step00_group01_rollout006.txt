def mirror_blend(vec, rng):
    """Convex blend with reversed sequence."""
    t = float(rng.uniform(0.1, 0.4))
    return np.clip((1.0 - t) * vec + t * vec[::-1], 0.0, 1000.0)


def smooth_perturb(vec, rng):
    """Add a low-frequency sinusoidal perturbation."""
    n = vec.size
    k = int(min(15, max(2, n // 40)))
    idx = np.arange(n, dtype=np.float64)
    delta = np.zeros(n, dtype=np.float64)
    for _ in range(k):
        f = int(rng.integers(1, max(2, n // 4)))
        p = float(rng.uniform(0.0, 2.0 * np.pi))
        a = float(rng.uniform(0.005, 0.03)) * float(np.max(vec))
        delta += a * np.sin(2.0 * np.pi * f * idx / n + p)
    return np.clip(vec + delta, 0.0, 1000.0)


def nudge_toward_uniform(vec, rng):
    """Step slightly toward the arithmetic mean (reduces variance)."""
    t = float(rng.uniform(0.02, 0.12))
    m = float(np.mean(vec))
    return np.clip((1.0 - t) * vec + t * m, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --- Initialize from parent ---
    raw = np.asarray(height_sequence_1, dtype=np.float64)
    if raw.ndim != 1 or raw.size == 0 or not np.all(np.isfinite(raw)):
        raw = np.ones(512, dtype=np.float64)
    parent = np.clip(raw, 0.0, 1000.0)
    n = parent.size

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(best_score):
        best = np.ones(n, dtype=np.float64) if n > 0 else np.ones(64, dtype=np.float64)
        n = best.size
        best_score = float(evaluate_sequence(best.tolist()))
        if not np.isfinite(best_score):
            return [1.0] * 64

    lo = max(2, n // 400)
    hi = max(lo + 1, n // 6)
    consec = 0
    max_consec = 300
    save_hs = time.monotonic()
    fs_hs = int(np.sum(np.isfinite(best)))

    while time.monotonic() < deadline:
        r = rng.random()
        if r < 0.30:
            cand = block_perturb(best, rng, lo, hi)
        elif r < 0.45:
            cand = two_block_perturb(best, rng, lo, hi)
        elif r < 0.60:
            cand = mirror_blend(best, rng)
        elif r < 0.80:
            cand = smooth_perturb(best, rng)
        elif r < 0.90:
            cand = nudge_toward_uniform(best, rng)
        else:
            # Wide soft block
            w = int(rng.integers(max(3, n // 50), max(4, n // 3)))
            st = int(rng.integers(0, max(1, n - w + 1)))
            sig = float(rng.uniform(0.3, 1.5))
            center = (st + st + w) / 2.0
            span = max(1.0, w / 2.0)
            gauss = np.exp(-0.5 * ((np.arange(st, st + w) - center) / span) ** 2)
            cand = best.copy()
            scale = float(np.exp(rng.normal(0.0, 0.2)))
            cand[st:st + w] *= (1.0 - 0.3 * gauss) + 0.3 * gauss * scale
            cand = np.clip(cand, 0.0, 1000.0)

        # Fast filter (allow tiny numerical slack)
        fs = fast_score(cand)
        if np.isfinite(fs) and fs < best_score * 1.0005:
            exact = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(exact) and exact < best_score:
                best = cand
                best_score = exact
                consec = 0
            else:
                consec += 1
        else:
            consec += 1

        # Periodic re-seed if stuck
        if consec >= max_consec:
            consec = 0
            mix = float(rng.uniform(0.3, 0.7))
            uni = np.ones(n, dtype=np.float64) * float(np.mean(best))
            best = np.clip(mix * best + (1.0 - mix) * uni, 0.0, 1000.0)
            best_score = float(evaluate_sequence(best.tolist()))
            # Also try pure uniform occasionally
            if rng.random() < 0.2:
                uni_score = float(evaluate_sequence(np.ones(n, dtype=np.float64).tolist()))
                if uni_score < best_score:
                    best = np.ones(n, dtype=np.float64)
                    best_score = uni_score

    # Final guard
    fs = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(fs):
        return [1.0] * n
    return [float(x) for x in best]
