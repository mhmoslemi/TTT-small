import numpy as np
import time
from scipy.fft import fft, ifft, next_fast_len

def _fft_conv(a):
    n = a.size
    size = next_fast_len(2 * n - 1)
    fa = fft(a, size)
    return np.real(ifft(fa * fa, size))[:2 * n - 1]

def _grad_at_peak(a, tau, M, S, n):
    g = np.zeros(n)
    lo = max(0, tau - n + 1)
    hi = min(n - 1, tau)
    if lo <= hi:
        ks = np.arange(lo, hi + 1)
        g[ks] = 2.0 * a[tau - ks]
    # dF/da[k] = 2n*(dM/da[k])/S - 4nM/S^2
    g *= 2.0 * n / S
    g -= 4.0 * n * M / (S * S)
    return g

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    n = len(height_sequence_1)
    a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Phase 1 – break symmetry with block perturbations
    t1 = time.monotonic() + min(150.0, 0.2 * float(budget_s))
    while time.monotonic() < min(t1, deadline):
        c = a.copy()
        nb = int(rng.integers(2, 6))
        for _ in range(nb):
            w = max(2, int(n * float(rng.uniform(0.02, 0.2))))
            s = int(rng.integers(0, max(1, n - w)))
            c[s:s + w] *= float(np.exp(rng.normal(0.0, 0.4)))
        c = np.clip(c, 0.0, 1000.0)
        if c.sum() < 0.01:
            continue
        sc = float(evaluate_sequence(c.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = c.copy()
            best_score = sc
            a = c
        elif np.isfinite(sc) and sc < best_score + 0.5 and rng.random() < 0.3:
            a = c  # anneal

    # Phase 2 – gradient descent at peak lag (FFT-based, O(n log n) per step)
    a = best.copy()
    S = float(a.sum())
    lr = 2.0 / n
    steps_since_eval = 0
    while time.monotonic() < deadline - 80.0:
        b = _fft_conv(a)
        tau = int(np.argmax(b))
        M = float(b[tau])
        S = float(a.sum())
        if S < 0.01:

        g = _grad_at_peak(a, tau, M, S, n)
        a = a - lr * g
        a = np.clip(a, 0.0, 1000.0)
        S2 = float(a.sum())
        if S2 > 0.01:
            a *= S / S2
        lr *= 0.999
        steps_since_eval += 1
        if steps_since_eval >= 300:
            steps_since_eval = 0
            sc = float(evaluate_sequence(a.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best = a.copy()
                best_score = sc
            # Occasionally re-inject perturbation
            if rng.random() < 0.15:
                a = a + float(rng.normal(0, 0.02 / S)) * a
                a = np.clip(a, 0.0, 1000.0)

    # Phase 3 – fine-tune with small block perturbations
    while time.monotonic() < deadline - 2.0:
        c = best.copy()
        w = max(1, n // 400)

        sc = float(evaluate_sequence(c.tolist()))

            best_score = sc
