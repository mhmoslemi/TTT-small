def compute_score(a):
    """Fast score computation (same as evaluate_sequence but avoids list conversion)."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = np.convolve(a, a)
    max_c = np.max(c)
    return float(2 * n * max_c / (S * S))
