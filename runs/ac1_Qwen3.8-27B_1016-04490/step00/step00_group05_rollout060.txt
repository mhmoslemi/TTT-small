def _fft_conv(a, b):
    n_out = len(a) + len(b) - 1
    size = 1 << (n_out - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n_out]


def _fast_score(x):

    conv = _fft_conv(x, x)
    M = conv.max()
    S = x.sum()
    if S < 1e-15:
        return np.inf
    return 2.0 * n * M / (S * S)


def _subgrad(x, n_act=None):
    n = len(x)
    conv = _fft_conv(x, x)
    if n_act is None:

    n_act = min(n_act, len(conv))
    if n_act < len(conv):

        top = np.arange(len(conv))
    grad = np.zeros(n)
    for k in top:

        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            lo = k - i_hi
            hi = k - i_lo
            grad[i_lo:i_hi + 1] += x[lo:hi + 1][::-1]
    return grad, conv


def _normalize(x):
    s = x.sum()
    if s < 1e-14:
        return np.ones(len(x)) / len(x)
    return x / s


def _log_update(x, grad, lr):
