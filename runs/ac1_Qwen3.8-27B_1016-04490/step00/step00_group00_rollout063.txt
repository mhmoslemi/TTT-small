import numpy as np
import time

def _fft_conv(a):
    """FFT-based full convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def _grad_from_lags(a, lags):
    """Compute sum of gradients from given lags."""
    n = len(a)
    grad = np.zeros(n)
    for k in lags:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi >= lo:
            grad[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]
    return grad

    best = parent.copy()

    # Generate initial candidates
