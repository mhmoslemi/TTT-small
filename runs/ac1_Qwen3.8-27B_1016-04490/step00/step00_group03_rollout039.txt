def propose_candidate(seed=42, budget_s=1100, **kwargs):

    # Local search with block perturbations
    while time.monotonic() < deadline:
        candidate = best.copy()
        # Random block
        width = int(rng.integers(max(2, n//50), max(3, n//5)))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        # Multiplicative perturbation
        candidate[start:stop] *= np.exp(rng.normal(0, 0.05, stop-start))
        # Also try gradient-like step toward flattening the convolution
        # Clip
        np.clip(candidate, 0.001, 1000.0, out=candidate)
        
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
    
    return [float(x) for x in best]
