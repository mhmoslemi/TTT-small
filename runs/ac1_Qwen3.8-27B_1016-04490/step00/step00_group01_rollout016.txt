    n_out = len(a) + len(b) - 1

    A = np.fft.rfft(a, size)

    return np.fft.irfft(A * B, size)[:n_out]

    """Subgradient of max(b) w.r.t. a.
    d b[t] / d a[j] = 2*a[t-j] if 0 <= t-j < n, else 0.
