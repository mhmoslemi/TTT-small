        lo = max(0, p - n + 1)
        hi = min(p, n - 1)

        # Gradient of the peak: d(peak)/dx_i = 2*x_{p-i} for valid i
        dpeak = np.zeros(n)
        dpeak[lo:hi + 1] = 2.0 * x[p - np.arange(lo, hi + 1)]

        # Descent direction for the score (sum-preserving)
        direction = 2.0 * peak - dpeak * S
        direction -= np.mean(direction)

        # Occasional "dual" (LP-inspired) direction: mass where x is small
        if iters % 80 == 0 and no_improve > 60:
            eps = 0.02 * S / n
            g = 1.0 / (x + eps)
            g = g * (S / np.sum(g))
            direction = 0.5 * direction + 0.5 * (g - x)
            direction -= np.mean(direction)

        nnorm = np.linalg.norm(direction)
        if nnorm < 1e-14:
            direction = rng.standard_normal(n)
            direction -= np.mean(direction)
            nnorm = np.linalg.norm(direction)

        scale = step * S / n
        x_try = x + scale * direction / nnorm
        x_try = np.clip(x_try, 0.0, 1000.0)
        if np.sum(x_try) < 0.01:
            x_try = x + 0.01 * scale * rng.standard_normal(n)
            x_try = np.clip(x_try, 0.0, 1000.0)
            if np.sum(x_try) < 0.01:
                x_try = best.copy()
                continue

        s = float(evaluate_sequence(x_try.tolist()))

        if np.isfinite(s) and s < best_score:
            best_score = s
            best = x_try.copy()
            x = x_try.copy()
            no_improve = 0
            step = min(step * 1.02, 5.0)
        else:
            no_improve += 1
            if no_improve > 500:
                # Iterated local search: kick from best
                amp = 0.06 * best.mean()
                x = best + amp * rng.standard_normal(n)
                x = np.clip(x, 0.0, 1000.0)
                no_improve = 0
                step = max(0.2, step * 0.5)
            else:
                x = best.copy()
                if no_improve % 150 == 0:
                    step = max(step * 0.75, 0.05)

    return [float(v) for v in best]
