import numpy as np
import time

np.seterr(over="ignore", under="ignore", invalid="ignore")


def next_power_two_m(m):
    m = int(m)
    if m <= 1:
        return 1
    L = 1
    while L < m:
        L <<= 1
    return L


def negative_index_vector(n, L):
    if n <= 1:
        return np.array([0], dtype=np.int64)
    out = np.empty(n, dtype=np.int64)
    out[0] = 0
    out[1:] = L - np.arange(1, n, dtype=np.int64)
    return out


def add_sequence_cand(pool, arr):
    a = np.asarray(arr, dtype=np.float64)
    if a.ndim != 1:
        a = a.reshape(-1)
    if a.size == 0:
        return
    if not np.all(np.isfinite(a)):
        a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s > 1e-12:
        pool.append(a / s)


def log_normalize(x, floor_log):
    arr = np.asarray(x, dtype=np.float64)
    if not arr.flags.writeable:
        arr = np.array(arr, dtype=np.float64, copy=True)
    np.maximum(arr, floor_log, out=arr)
    m = float(np.max(arr))
    if not np.isfinite(m):
        arr.fill(floor_log)
        return arr
    arr -= m
    s = float(np.sum(np.exp(arr)))
    if not np.isfinite(s) or s <= 0.0:
        arr.fill(floor_log)
        return arr
    arr -= np.log(s)
    return arr


def fft_score_normalized(a, L, W):
    fa = np.fft.fft(a, L)
    c = np.fft.ifft(fa * fa)[:W].real
    if c.size == 0:
        return float("inf")
    cmax = float(np.max(c))
    if not np.isfinite(cmax) or cmax < 0.0:
        return float("inf")
    return 2.0 * float(len(a)) * cmax

    start_time = time.monotonic()
    budget = max(1.0, float(budget_s))
    hard_end = start_time + budget
    headroom = max(20.0, 0.02 * budget)
    deadline = hard_end - headroom
    if deadline < start_time + 1.0:
        deadline = hard_end - max(1.0, 0.25 * headroom)

    rng = np.random.default_rng(seed)

    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1:
            parent_arr = parent_arr.reshape(-1)
        if parent_arr.size > 0 and bool(np.all(np.isfinite(parent_arr))):
            parent = np.clip(parent_arr, 0.0, 1000.0)
    except Exception:
        parent = None

    if parent is None or parent.size == 0:
        n = 4806
        parent = np.ones(n, dtype=np.float64)
    else:
        n = int(parent.size)
        if n <= 0:
            n = 4806
            parent = np.ones(n, dtype=np.float64)

    psum = float(np.sum(parent))
    parent_norm = parent / max(psum, 1e-300) if psum > 1e-12 else None

    W = 2 * n - 1
    L = next_power_two_m(max(1, 3 * n - 2))
    idx = negative_index_vector(n, L)
    floor_log = np.log(max(1e-300, 1e-12 / float(n)))

    pool = []
    add_sequence_cand(pool, parent)
    add_sequence_cand(pool, np.ones(n, dtype=np.float64))

    t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    for amp in (0.25, 0.5, 0.75, 0.95):
        for freq in (1, 2, 3, 5):
            add_sequence_cand(pool, 1.0 + amp * np.cos(2.0 * np.pi * float(freq) * t))
            add_sequence_cand(pool, 1.0 + amp * np.cos(np.pi * t))
            add_sequence_cand(pool, 1.0 + amp * np.sin(np.pi * t))

    for sigma in (0.3, 0.7, 1.3, 2.0):
        add_sequence_cand(pool, np.exp(rng.normal(0.0, sigma, n)))

    for q in (0.5, 0.6, 0.7, 0.8, 0.9):
        add_sequence_cand(pool, (rng.random(n) < q).astype(np.float64))

    if parent_norm is not None:
        for sigma in (0.1, 0.3, 0.7):
            add_sequence_cand(pool, parent_norm * np.exp(rng.normal(0.0, sigma, n)))

    if not pool:
        pool = [np.ones(n, dtype=np.float64) / float(n)]

    best_score = float("inf")
    best_a = None
    for cand in pool:
        sc = fft_score_normalized(cand, L, W)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = cand.copy()

    if best_a is None:
        best_a = np.ones(n, dtype=np.float64) / float(n)
        best_score = 2.0

    x = np.log(np.maximum(best_a, float(np.exp(floor_log))))
    x = log_normalize(x, floor_log)

    now = time.monotonic()
    remaining = max(0.0, deadline - now)
    normal_end = now + max(5.0, 0.6 * remaining)
    if normal_end > deadline - 20.0:
        normal_end = deadline - 20.0
    if normal_end > now + 3.0:
        best_a, best_score, x = run_search(
            x, rng, L, W, idx, floor_log, n, normal_end,
            best_a, best_score, pool, False
        )

    polish_end = deadline - 2.0
    if time.monotonic() + 5.0 < polish_end:
        x = np.log(np.maximum(best_a, float(np.exp(floor_log))))
        x = log_normalize(x, floor_log)
        polish_pool = [best_a]
        if parent_norm is not None:
            polish_pool.append(parent_norm.copy())
        best_a, best_score, x = run_search(
            x, rng, L, W, idx, floor_log, n, polish_end,
            best_a, best_score, polish_pool, True
        )

    final = np.clip(np.asarray(best_a, dtype=np.float64), 0.0, 1000.0)
    fs = float(np.sum(final))
    if not np.isfinite(fs) or fs < 0.01:
        final = np.ones(n, dtype=np.float64) / float(n)
    else:
        final = final / fs

    candidates = []
    add_sequence_cand(candidates, final)
    add_sequence_cand(candidates, np.ones(n, dtype=np.float64) / float(n))
    if parent_norm is not None:
        add_sequence_cand(candidates, parent_norm)
    if not candidates:
        candidates = [np.ones(max(1, n), dtype=np.float64) / float(max(1, n))]

    best_out = None
    best_out_score = float("inf")

    if time.monotonic() < hard_end - max(5.0, 0.5 * headroom):
        for cand in candidates:
            try:
                sc = evaluate_sequence(cand.tolist())
            except Exception:
                sc = float("inf")
            if np.isfinite(sc) and sc < best_out_score:
                best_out_score = sc
                best_out = cand

    if best_out is None:
        for cand in candidates:
            sc = fft_score_normalized(cand, L, W)
            if np.isfinite(sc) and sc < best_out_score:
                best_out_score = sc
                best_out = cand

    if best_out is None:
        best_out = np.ones(n, dtype=np.float64) / float(n)

    return [float(v) for v in best_out]
