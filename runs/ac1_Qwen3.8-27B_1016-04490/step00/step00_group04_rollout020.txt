def fft_conv(a):
    """Full linear autocorrelation via FFT, O(n log n) memory/time."""
    n = a.shape[0]
    size = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]
