import time

def compute_conv(a):
    """Compute np.convolve(a, a) efficiently."""
    return np.convolve(a, a)

def compute_score(a):
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * len(a) * c.max() / (s * s)
