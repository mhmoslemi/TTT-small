import numpy as np

def periodic_template(rng, n, period):
    """Create a periodic template with random zeros/scale."""
    template = rng.uniform(0.5, 2.0, size=period)
    nz = rng.integers(1, max(2, period // 2 + 1))
    zpos = rng.choice(period, size=nz, replace=False)
    template[zpos] = 0.0
    reps = n // period + 1
    out = np.tile(template, reps)[:n]
    return np.clip(out, 0.0, 1000.0)

def block_perturb(x, rng, n):
    """Scale a random block."""
    out = x.copy()
    w = rng.integers(1, max(2, n // 10))
    s = rng.integers(0, max(1, n - w + 1))
    f = rng.uniform(0.3, 2.5)
    out[s:s+w] *= f
    return np.clip(out, 0.0, 1000.0)

def block_reduce(x, rng, n):
    """Reduce (toward zero) a random block."""
    out = x.copy()

def center_reduce(x, rng, n):

def smooth_perturb(x, rng, n):
