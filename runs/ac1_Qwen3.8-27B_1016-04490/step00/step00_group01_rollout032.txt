def _subgrad_at_lag(a, k, n):
    """Subgradient of b[k] = sum_j a[j]*a[k-j] w.r.t. a."""
    s = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    seg = a[k - hi : k - lo + 1][::-1]
    s[lo:hi+1] = 2.0 * seg
    return s

def _softmax_grad(a, b, w, n):
    """Smoothed gradient: sum_k w[k] * subgrad(b[k])."""
    # g[i] = 2 * sum_k w[k] * a[k-i] (where valid)
    # = 2 * sum_j a[j] * w[i+j] (where valid)
    # Compute via convolution: conv(a, w_rev)[2n-2-i]
    w_rev = w[::-1]
    # Use np.convolve (O(n * 2n) = O(n^2), fine for n=6121)
    conv = np.convolve(a, w_rev, mode='full')  # length 3n-2
    g = 2.0 * conv[2*n-2 : n-2 : -1]  # g[i] = 2*conv[2n-2-i] for i=0..n-1
    return g[:n]

import time


def _init_series(n):
    i = np.arange(n, dtype=np.float64)
    a0 = np.ones(n)
    a1 = 1.0 / np.sqrt(i + 0.5)
    a2 = np.ones(n, dtype=np.float64)
    a3 = a1 ** 1.2
    a4 = a1 ** 0.8
    a5 = np.cos(0.5 * np.pi * i / max(n - 1, 1)) ** 0.5
    return [a0, a1, a2, a3, a4, a5]


def _smoothed_grad(a, b, n):
    L = 1
    while L < 3 * n:
        L <<= 1
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    g = np.fft.irfft(fa * fb.conj(), L)[:3 * n - 2]
    return 2.0 * g[2 * n - 2::-1][:n]
