# FFT-based convolution
def fft_convolve(a, b):
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    return np.fft.irfft(A * B, size)[:n]

def fft_conv(a):
    n = a.size
    size = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, size)
    res = np.fft.irfft(A * A, size)
    return res[:2 * n - 1]
