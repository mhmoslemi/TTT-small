import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def _precompute_rows_cols(n):
    m = 2 * n - 1
    rows_parts = [np.arange(j, j + n) for j in range(n)]
    rows = np.concatenate(rows_parts)
    cols = np.repeat(np.arange(n), n)
    return rows, cols, m


def _solve_harmonize_lp(a, rows, cols, m, time_limit=20.0):
    n = a.size
    vals = np.repeat(a, n)
    A = csr_matrix((vals, (rows, cols)), shape=(m, n))
    conv_aa = np.convolve(a, a)
    M = float(np.max(conv_aa))
    c_obj = -np.ones(n)
    b_ub = np.full(m, M)
    bnd = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bnd,
                  method='highs', options={'time_limit': time_limit, 'presolve': True})
    if res is not None and res.success and res.x is not None:
        return np.maximum(res.x, 0.0), M
    return None, M


def _block_perturb(base, rng, n):
    cand = base.copy()
    width = int(rng.integers(max(1, n // 200), max(2, n // 10)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.15)))
    for _ in range(3):
        j = int(rng.integers(0, n))
        cand[j] *= float(np.exp(rng.normal(0.0, 0.10)))
    if rng.random() < 0.3:
        half = n // 2
        cand[:half] = 0.5 * (cand[:half] + cand[half-1:half-1+half:-1])
    return np.clip(cand, 0.0, 1000.0)
