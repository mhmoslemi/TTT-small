import time

def _fft_conv(x, y):
    n = len(x) + len(y) - 1
    sz = 1
    while sz < n:
        sz <<= 1
    return np.fft.irfft(np.fft.rfft(x, sz) * np.fft.rfft(y, sz), sz)[:n]

def _softmax(x, tau):
    z = x / tau
    z -= z.max()
    e = np.exp(z)
    s = e.sum()
    return e / s if s > 0 else np.full(len(x), 1.0/len(x))

def _proj_simplex(v):
    """Project v onto {x >= 0, sum x = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where((u - (cssv - 1) / np.arange(1, n+1)) > 0)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

    return 2.0 * ca[n - 1:2 * n - 1][::-1]


def _run(a, deadline, rng):
    n = a.size
    a = np.maximum(a, 0.0).astype(np.float64)
    s0 = a.sum()
    if s0 < 1e-10:
        a[:] = 1.0
    a /= a.sum()

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    tau = 1.0 / n
    lr = 1.0
    eval_every = 3000
    stalled = 0

    for it in range(20000000):
        if time.monotonic() >= deadline:
            break

        g = _smooth_grad(a, tau)
        a *= np.exp(-lr * g)
        a = np.maximum(a, 1e-30)
        a /= a.sum()

        if it % 20 == 0:
            tau = max(tau * 0.997, 1e-5 / n)
            lr *= 0.999

        if (it + 1) % eval_every == 0:
            sc = float(evaluate_sequence(a.tolist()))
            if sc < best_score - 1e-12:
                best_score = sc
                best = a.copy()
                stalled = 0
            else:
                stalled += 1
            if stalled > 5:
                a += 0.005 * rng.standard_normal(n)
                a = np.maximum(a, 0.0)
                a /= a.sum()
                stalled = 0

    sc = float(evaluate_sequence(a.tolist()))
    if sc < best_score:
        best = a.copy()
        best_score = sc
    return best, best_score


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    parent = np.clip(
        np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.ndim != 1 or parent.size < 2 or not np.isfinite(parent).all():
        parent = np.ones(2535, dtype=np.float64)
    n = parent.size
    parent = np.maximum(parent, 0.0)
    ps = parent.sum()
    if ps < 1e-10:
        parent[:] = 1.0
    else:
        parent /= ps

    j = np.arange(n, dtype=np.float64)
    x = j / n
    inits = [
        parent.copy(),
        np.sin(np.pi * (j + 0.5) / n) ** 2,
        np.exp(-10.0 * (x - 0.25) ** 2) + np.exp(-10.0 * (x - 0.75) ** 2) + 0.01,
        np.abs(rng.standard_normal(n)) + 0.05,
        (j + 1.0) ** (-0.4),
    ]

    rem = deadline - time.monotonic()
    time_per = max(rem / len(inits), 2.0)

    best_all = parent.copy()
    best_sc = float(evaluate_sequence(parent.tolist()))

    for ia in inits:
        if time.monotonic() >= deadline:
            break
        ia = np.maximum(ia, 0.0).astype(np.float64)
        ss = ia.sum()
        if ss < 1e-10:
            ia[:] = 1.0
        else:
            ia /= ss
        dl = min(time.monotonic() + time_per, deadline)
        b, sc = _run(ia, dl, rng)
        if sc < best_sc:
            best_sc = sc
            best_all = b.copy()

    return [float(v) for v in best_all]
