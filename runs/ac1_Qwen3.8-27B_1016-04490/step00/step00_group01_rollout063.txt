import numpy as np
import time


def _fft_conv(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    total = n + m - 1
    N = 1
    while N < total:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:total]


def _score(a, n):
    b = _fft_conv(a, a)
    s = a.sum()
    if s < 1e-30:
        return 1e18
    return 2.0 * n * b.max() / (s * s)


def _sgd(a, n, lr, K=16):
    b = _fft_conv(a, a)
    K = min(K, len(b))
    top = np.argpartition(b, -K)[-K:]
    w = b[top].astype(np.float64)
    ws = w.sum()
    if ws < 1e-30:
        return a, b.max()
    w /= ws
    sg = np.zeros(n)
    for j in range(K):
        k = int(top[j])
        wt = 2.0 * w[j]
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            sg[lo:hi + 1] += wt * np.flip(a[k - hi:k - lo + 1])
    a_new = a - lr * sg
    np.maximum(a_new, 0.0, out=a_new)
    s = a_new.sum()
    if s < 1e-30:
        return a, b.max()
    a_new /= s
    return a_new, b.max()


def _lp_refine(f, n, deadline, max_it=3000):
    M = _fft_conv(f, f).max() * 1.001
    g = f.copy()
    cg = _fft_conv(f, g)
    g *= M / max(cg.max(), 1e-30)
    np.maximum(g, 0.0, out=g)
    eta = 1.0 / max(f.sum() * M + 1e-10, 1e-10)
    N2 = 1
    while N2 < 3 * n:
        N2 <<= 1
    F = np.fft.rfft(f, N2)
    for _ in range(max_it):
        if time.monotonic() > deadline:
            break
        b = _fft_conv(f, g)
        viol = np.maximum(0.0, b - M)
        if viol.max() < 1e-14 * M:
            break
        V = np.fft.rfft(viol, N2)
        corr = np.fft.irfft(V * np.conj(F), N2)
        grad = corr[:n]
        g -= eta * grad
        np.maximum(g, 0.0, out=g)
        if g.sum() < 1e-30:
            break
        eta *= 0.999
    return g
