def fft_convolve(a, n):

    size = 1 << (2*n - 1).bit_length()  # next power of 2 for efficiency

def fft_convolve(a):

    size = 1 << (2*n-1).bit_length()
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa*fa, size)[:2*n-1]

def fft_convolve(a):
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def subgrad_peak(a, n, m):
    """Subgradient of conv(a,a)[m] w.r.t. a."""
    L = max(0, m - n + 1)
    U = min(n - 1, m)
    sg = np.zeros(n)
    if L <= U:
        j = np.arange(L, U + 1)
        sg[L:U + 1] += a[m - j]
    jlo = max(0, m - U)
    jhi = min(n - 1, m - L)
    if jlo <= jhi:
        j = np.arange(jlo, jhi + 1)
        sg[jlo:jhi + 1] += a[m - j]
    return sg

import time
from scipy import sparse
from scipy.optimize import linprog


def _fft_conv(a):
    n = a.size
    sz = 1 << (2 * n - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) ** 2, sz)[:2 * n - 1]


def _sg(a, n, m):
    L = max(0, m - n + 1)
    U = min(n - 1, m)
    sg = np.zeros(n)
    if L <= U:
        j = np.arange(L, U + 1)
        sg[L:U + 1] += a[m - j]
    lo = max(0, m - U)
    hi = min(n - 1, m - L)
    if lo <= hi:
        j = np.arange(lo, hi + 1)
        sg[lo:hi + 1] += a[m - j]
    return sg


def _lp_dir(a, n, M, tgt):

    va = np.tile(a, n)
    T = sparse.coo_matrix((va, (ri, ci)), shape=(2 * n - 1, n)).tocsr()
    res = linprog(-np.ones(n), A_ub=T, b_ub=M * np.ones(2 * n - 1),
                  bounds=[(0, None)] * n, method='highs',
                  options={'time_limit': tgt})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        s = g.sum()
        if s > 1e-12:
            return g * n / s
    return None
