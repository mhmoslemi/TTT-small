def _rng_block_perturb(base, rng, n):
    cand = np.array(base, dtype=np.float64, copy=True)
    width = int(rng.integers(max(2, n // 400), max(3, n // 16)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    r = rng.random()
    if r < 0.5:
        cand[start:stop] *= np.exp(rng.normal(0.0, 0.25, stop - start))
    elif r < 0.75:
        scale = 0.5 * max(1e-9, float(np.mean(base)))
        cand[start:stop] += rng.normal(0.0, scale, stop - start)
    else:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _peak_shave_step(base, rng, eta):
    """Reduce the dominant convolution lag by a multiplicative descent step."""
    a = np.array(base, dtype=np.float64, copy=True)
    c = np.convolve(a, a)
    k = int(np.argmax(c))
    lo = max(0, k - a.size + 1)
    hi = min(a.size, k + 1)
    if hi <= lo:
        return a
    j = np.arange(lo, hi)
    partner = k - j
    weight = a[partner] * a[j]
    if weight.max() <= 0.0:
        return a
    a[j] *= np.exp(-eta * weight / a[partner].max())
    if rng.random() < 0.15:
        a *= rng.uniform(0.97, 1.03)
    return np.clip(a, 0.0, 1000.0)
