import numpy as np
import time


def _grad_soft_conv(x, beta):
    """Return gradient of soft-max(conv(x,x)) w.r.t. x, plus conv and max."""
    n = x.size
    conv = np.convolve(x, x)
    max_c = conv.max()
    h = np.exp(np.clip(beta * (conv - max_c), -60.0, 0.0))
    xr = x[::-1]
    conv_xr_h = np.convolve(xr, h)
    cross = conv_xr_h[n - 1:2 * n - 1]
    h_cum = np.empty(2 * n, dtype=np.float64)
    h_cum[0] = 0.0
    np.cumsum(h, out=h_cum[1:])
    sums_h = h_cum[n:] - h_cum[:n]
    grad = cross + x * sums_h
    return grad, conv, max_c
