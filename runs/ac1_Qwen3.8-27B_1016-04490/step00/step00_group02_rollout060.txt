def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Next power of 2
    N = 1
    while N < size:
        N <<= 1
    fa = np.fft.rfft(a, N)
    ac = np.fft.irfft(fa * fa, N)[:size]
    return ac

def parametric_init(n, kind, param, rng):
