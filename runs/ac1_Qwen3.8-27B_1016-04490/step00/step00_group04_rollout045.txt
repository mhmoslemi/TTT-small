def gen_power_ramp(n, p, scale=1000.0):

    b = np.convolve(a, a)
    max_b = b.max()
    sum_a = a.sum()
    return float(2.0 * n * max_b / (sum_a ** 2))


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    n = len(height_sequence_1)

    # Try several power-ramp exponents to find the best initialization
    best = None
    best_score = np.inf
    for p in [1.0, 1.5, 2.0, 2.5, 3.0, 3.5, 4.0, 5.0, 6.0]:
        cand = gen_power_ramp(n, p)
        s = evaluate_fast(cand)
        if np.isfinite(s) and s < best_score:
            best_score = s
            best = cand.copy()

    # Verify with the authoritative evaluator
    best_score = float(evaluate_sequence(best.tolist()))

    # Local search: perturb random blocks multiplicatively
    while time.monotonic() < deadline:
        candidate = best.copy()
        width = int(rng.integers(max(1, n // 200), max(2, n // 8)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        multiplier = np.exp(rng.normal(0.0, 0.08, size=stop - start))
        candidate[start:stop] *= multiplier
        np.clip(candidate, 0.0, 1000.0, out=candidate)

        score = evaluate_fast(candidate)
        if np.isfinite(score) and score < best_score - 1e-10:
            best = candidate
            best_score = score

    # Final verification
    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):
        fallback = gen_power_ramp(n, 3.0)
        return [float(x) for x in fallback]

    return [float(x) for x in best]
