def propose_candidate(seed=42, budget_s=1100, **kwargs):

    val = float(np.clip(val, 0.01, 1000.0))

    best = [val] * n
    best_score = float(evaluate_sequence(best))

    # Check parent in case of any edge case
    parent = [float(x) for x in np.clip(parent_arr, 0.0, 1000.0)]
    if len(parent) == n and all(np.isfinite(x) for x in parent):
        ps = float(evaluate_sequence(parent))
        if np.isfinite(ps) and ps < best_score:
            best = parent
            best_score = ps

    # Brief confirmation search: try to reduce max convolution
    # (mathematically expected to find nothing better than 2)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    arr = np.asarray(best, dtype=np.float64)
    stagnate = 0

    while time.monotonic() < deadline and stagnate < 30:
        try:
            b = np.convolve(arr, arr)
        except Exception:
            break
        max_idx = int(np.argmax(b))
        max_b = float(b[max_idx])
        sum_a = float(arr.sum())
        if sum_a < 0.01:
            break

        # Gradient of the peak convolution value wrt a: a reversed/shifted
        # Steepest descent on the peak term
        c = arr.copy()
        step = rng.uniform(0.002, 0.02)
        # The partial d(b[k])/d(a[j]) = a[k-j] (with bounds)
        lo = max(0, max_idx - n + 1)
        hi = min(n - 1, max_idx)
        for j in range(lo, hi + 1):
            c[j] -= step * arr[max_idx - j]
        c = np.clip(c, 0.0, 1000.0)

            stagnate = 0
