import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time


def build_constr_matrix(a, k_list):
    """Build sparse Toeplitz sub-matrix for given lags."""
    n = a.size
    K = len(k_list)
    rows = []
    cols = []
    vals = []
    for idx in range(K):
        k = int(k_list[idx])
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_hi < i_lo:
            continue
        js = np.arange(i_lo, i_hi + 1)
        seg = a[k - js]
        rows.extend([idx] * len(seg))
        cols.extend(js.tolist())
        vals.extend(seg.tolist())
    if not vals:
        return sp.csr_matrix((K, n))
    return sp.csr_matrix((vals, (rows, cols)), shape=(K, n))
