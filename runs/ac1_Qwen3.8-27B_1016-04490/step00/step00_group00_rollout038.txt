import numpy as np
import time


def _uniform(n):
    """Return a uniform nonnegative sequence of length n."""
    return np.ones(n, dtype=np.float64)


def _block_perturb(base, rng):
    """O(n) perturbation: scale a random block and blend toward uniform."""
    c = base.copy()
    n = c.size
    w = max(1, int(rng.integers(n // 50, n // 3)))
    s = int(rng.integers(0, max(1, n - w)))
    c[s:s + w] *= float(np.exp(rng.normal(0.0, 0.1)))
    c = 0.5 * c + 0.5 * np.mean(c)
    return np.clip(c, 0.0, 1000.0)
