def _conv_fft(a):
    n = a.size
    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, L)
    return np.fft.irfft(fa * fa, L)[:size]

def _grad_from_weights(w, a):
    n = a.size
    b = a[::-1]
    conv_len = (2*n - 1) + n - 1  # 3n-2
    L = 1 << (conv_len - 1).bit_length()
    fw = np.fft.rfft(w, L)
    fb = np.fft.rfft(b, L)
    conv_wb = np.fft.irfft(fw * fb, L)[:conv_len]
    return 2.0 * conv_wb[n-1 : n-1+n]

def _softmax_weights(conv, beta):
    M = conv.max()
    e = np.exp(beta * (conv - M))

    s = w.sum()
    if s < 1e-30:
        w = np.zeros_like(conv)
        w[conv == M] = 1.0 / (conv == M).sum()
    else:
        w /= s
    return w

def _remove_scale_component(g, a, S):
    """Project gradient onto tangent of the unit-sum simplex."""
    Sr = float(np.dot(a, g))
    return g - (Sr / (S * S)) * a


def _conv_fft(a):
    n = a.size
    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, L)
    return np.fft.irfft(fa * fa, L)[:size]


def _grad_from_weights(w, a):
    n = a.size
    b = a[::-1]

    fw = np.fft.rfft(w, L)
    fb = np.fft.rfft(b, L)

    skips = [0, 1, 2, 3, 5, 7, 10, 13]
    for sk in skips:
        t = (np.arange(n) + -0.5 * sk) / float(n)
        c = np.cos(np.pi * t)
        cand = np.clip(c * np.cos(np.pi * t), 0.0, 1.0)
        if cand.sum() < 0.01:
            continue
        s = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(s) and s < best_score:
            best_score = s
            best = cand.copy()

    S = best.sum()
    if S < 0.01:
        best = parent.copy()
        S = best.sum()
        best_score = float(evaluate_sequence(best.tolist()))
    a = best / S

    lr = 0.5
    beta = 50.0
    floor = 1e-10
    stall = 0

    while time.monotonic() < deadline:
        t0 = time.monotonic()
        conv = _conv_fft(a)
        w = _lase_weights(conv, beta)
        g = _grad_from_weights(w, a)
        g = _remove_scale_component(g, a, 1.0)
        step = lr * g
        a2 = a - step
        a2 = np.clip(a2, floor, None)
        a2 /= a2.sum()
        cand = np.clip(a2 * S, 0.0, 1000.0)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            a = a2.copy()
            best = cand.copy()
            stall = 0
            lr = min(lr * 1.05, 2.0)
        else:
            stall += 1
            if stall == 1:
                lr *= 0.5
            elif stall % 8 == 0:
                beta = rng.uniform(15.0, 120.0)
                lr = 0.5
            if stall >= 60:
                lam = 0.15 + 0.25 * min(stall / 200.0, 1.0)
                a = (1.0 - lam) * a + lam * rng.dirichlet(np.ones(n))
                cand = np.clip(a * S, 0.0, 1000.0)
                sc = float(evaluate_sequence(cand.tolist()))
                if np.isfinite(sc):
                    if sc < best_score:
                        best_score = sc
                        best = cand.copy()
                stall = 0
                lr = 0.5
                beta = rng.uniform(20.0, 80.0)
        if time.monotonic() - t0 > 30.0:
            break

    return [float(x) for x in best]
