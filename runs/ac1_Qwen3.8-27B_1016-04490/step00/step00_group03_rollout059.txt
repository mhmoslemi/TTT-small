```python
import numpy as np
import time
from scipy import sparse
from scipy.optimize import linprog


def build_conv_matrix(f):
    """Build sparse (2n-1) x n matrix A so that (A@g)[k] = (f*g)[k]."""
    n = len(f)
    m = 2 * n - 1
    i_vals = np.tile(np.arange(n), n)
    j_vals = np.repeat(np.arange(n), n)
    rows = j_vals + i_vals
    vals = np.tile(f, n)
    return sparse.coo_matrix((vals, (rows, j_vals)), shape=(m, n)).tocsr()


def quick_score(a, n):
    """Compute AC1 score inline (faster than full evaluator for line search)."""
    conv = np.convolve(a, a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * float(np.max(conv)) / (s * s)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # Initialization from parent
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)
    f = parent.astype(np.float64).copy()
    n = f.size
    s = f.sum()
    if s < 1e-10:
        f[:] = 1.0 / n
    else:
        f /= s

    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(best_score):
        f[:] = 1.0 / n
        best = f.copy()
        best_score = float(evaluate_sequence(best.tolist()))

    no_improve = 0
    max_no_improve = 150

    for it in range(5000):
        wall = time.monotonic()
        if wall > deadline - 12.0:
            break

        # Compute current peak
        conv_ff = np.convolve(f, f)
        M = float(np.max(conv_ff))
        if M < 1e-18:
            break

        # Build sparse convolution matrix
        A = build_conv_matrix(f)

        # LP: max sum(g) s.t. A@g <= M*e, g >= 0
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        tl = max(5.0, min(40.0, (deadline - wall) * 0.35))

        try:
            res = linprog(
                c, A_ub=A, b_ub=b_ub,
                bounds=[(0.0, None)] * n,
                method="highs",
                options={"time_limit": tl, "presolve": True},
            )
        except Exception:
            break

        if time.monotonic() > deadline - 5.0:
            break

        if not res.success or res.x is None or not np.isfinite(res.x).all():
            # Fallback: random block perturbation
            if time.monotonic() > deadline - 5.0:
                break
            width = max(2, n // 80)
            start = int(rng.integers(0, max(1, n - width)))
            f_p = f.copy()
            f_p[start:start + width] *= float(np.exp(rng.normal(0.0, 0.15)))
            f_p = np.clip(f_p, 0.0, None)
            sp = f_p.sum()
            if sp > 1e-10:
                f_p /= sp
                sc = quick_score(f_p, n)
                if np.isfinite(sc) and sc < best_score:
                    best = f_p.copy()
                    best_score = sc
                    f = f_p
                    no_improve = 0
                else:
                    no_improve += 1
                    if no_improve >= max_no_improve:
                        break
            continue

        g = np.clip(res.x, 0.0, None)
        g_sum = float(g.sum())
        if g_sum < 1e-12:
            break
        g /= g_sum

        # If g too close to f, we're at a fixed point
        if np.allclose(g, f, atol=1e-7, rtol=1e-7):
            no_improve += 1
            if no_improve >= 40:
                break
            # Small perturbation to escape
            f = f + rng.normal(0.0, 0.005, n)
            f = np.clip(f, 0.0, None)
            fs = f.sum()
            if fs > 1e-10:
                f /= fs
            continue

        # Line search with a few candidate step sizes
        improved = False
        for step in (0.45, 0.3, 0.2, 0.1, 0.05, 0.02):
            f_new = (1.0 - step) * f + step * g
            f_new = np.clip(f_new, 0.0, None)
            s2 = f_new.sum()
            if s2 < 1e-10:
                continue
            f_new /= s2
            sc = quick_score(f_new, n)
            if np.isfinite(sc) and sc < best_score - 1e-14:
                best = f_new.copy()
                best_score = sc
                f = f_new
                improved = True
                no_improve = 0
                break

        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break

    # Final validation with authoritative evaluator
    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):
        # Absolute fallback: uniform
        best = np.ones(n) / n
        final_score = float(evaluate_sequence(best.tolist()))

    return [float(x) for x in best]
```
