import numpy as np
import time


def _project_simplex(a):
    """Project onto {x >= 0, sum(x) = 1} in O(n)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-18:
        n = a.size
        return np.full(n, 1.0 / n)
    return a / s


def _grad_bk(a, k, n):
    """Gradient of (a*a)full[k] w.r.t. a, O(n) memory."""
    g = np.zeros(n)
    j_lo = max(0, k - n + 1)
    j_hi = min(n - 1, k)
    if j_lo > j_hi:
        return g
    idx = np.arange(j_lo, j_hi + 1)
    g[idx] = 2.0 * a[k - idx]
    if k % 2 == 0:
        jm = k // 2
        if j_lo <= jm <= j_hi:
            g[jm] = a[jm]
    return g


def _subgradient_loop(a, n, deadline, rng):
    """Projected subgradient to minimize max(conv(a,a)) on simplex."""
    best_a = a.copy()
    best_mb = float(np.max(np.convolve(a, a)))
    t = 0
    stale = 0
    while time.monotonic() < deadline:
        t += 1
        b = np.convolve(a, a)
        cmb = float(np.max(b))
        if cmb < best_mb - 1e-14:
            best_mb = cmb
            best_a = a.copy()
            stale = 0
        else:
            stale += 1
        if stale > 3000:
            a = best_a + rng.normal(0.0, 1.0 / (n * 5), n)
            a = _project_simplex(a)
            stale = 0
            continue
        ks = int(np.argmax(b))
        g = _grad_bk(a, ks, n)
        lr = 4.0 / (n * np.sqrt(float(t)))
        a2 = np.maximum(a - lr * g, 0.0)
        s2 = a2.sum()
        if s2 < 1e-18:
            a2 = np.full(n, 1.0 / n)
        else:
            a2 /= s2
        a = a2
    return best_a, best_mb


def _polish(a, n, deadline, rng):
    """Small random-walk polish from current best."""
    best = a.copy()
    best_mb = float(np.max(np.convolve(a, a)))
    step = 1.0 / n
    while time.monotonic() < deadline:
        idx = rng.integers(0, n)
        delta = rng.normal(0.0, step * 0.3)
        a[idx] = max(0.0, a[idx] + delta)
        s = a.sum()
        if s < 1e-18:
            a = best.copy()
            continue
        a /= s
        mb = float(np.max(np.convolve(a, a)))
        if mb < best_mb:
            best_mb = mb
            best = a.copy()
    return best, best_mb

    hard_deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)
    n = parent.size

    # Fallback: the parent itself
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Normalize to simplex for optimization (score is scale-invariant)
    s0 = parent.sum()
    if s0 < 1e-10:
        a0 = np.full(n, 1.0 / n)
    else:
        a0 = parent / s0

    # Phase 1: Projected subgradient (main solver)
    sub_deadline = min(hard_deadline - 120.0, time.monotonic() + 800.0)
    a_sub, _ = _subgradient_loop(a0.copy(), n, sub_deadline, rng)

    # Phase 2: Quick polish with local random walk
    pol_deadline = min(hard_deadline - 30.0, time.monotonic() + 200.0)
    a_pol, _ = _polish(a_sub.copy(), n, pol_deadline, rng)

    # Evaluate candidates with the authoritative evaluator
    for cand in (a_sub, a_pol, a0):

        if np.isfinite(sc) and sc < best_score:

            best = cand.copy()

    # Build return list
    result = [float(max(0.0, min(1000.0, x))) for x in best]
    if sum(result) < 0.01:
        result = [float(1.0 / n)] * n
    return result
