    """Linear auto-convolution via FFT, O(n log n)."""
    n = a.size
    M = 1
    while M < 2 * n - 1:
        M *= 2
    fa = np.fft.fft(a, M)
    return np.fft.ifft(fa * fa, M)[:2 * n - 1].real

    """Approximate score for fast filtering."""
    S = a.sum()
    if S < 0.01:
        return np.inf
    n = a.size
    ac = fast_autoconv(a)
    return 2.0 * n * ac.max() / (S * S)

    n = a.size
