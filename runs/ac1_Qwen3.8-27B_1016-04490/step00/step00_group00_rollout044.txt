def _build_conv_matrix_rows_cols(n):
    """Precompute row/col indices for the convolution matrix."""
    m = 2 * n - 1
    # Column j has entries in rows j to j+n-1
    rows_parts = [np.arange(j, j + n) for j in range(n)]
    rows = np.concatenate(rows_parts)
    cols = np.repeat(np.arange(n), n)
    return rows, cols, m

def _solve_lp(a, rows, cols, m):
    """Solve the harmonization LP."""
