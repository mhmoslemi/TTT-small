import time

def _fft_conv(a):
    """Linear self-convolution via FFT."""
    n = a.size
    N = 1 << (2*n - 1).bit_length()
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:2*n-1]

def _smooth_objective(b, alpha):
    """Smooth approximation of max(b) via log-sum-exp."""
    bmax = b.max()
    shifted = alpha * (b - bmax)
    shifted = np.clip(shifted, -100, 100)
    return bmax + np.log(np.sum(np.exp(shifted))) / alpha

def _smooth_gradient(b, a, alpha):
    """Gradient of smooth max w.r.t. a. O(n log n) via FFT."""
    n = a.size
    bmax = b.max()
    shifted = alpha * (b - bmax)
    shifted = np.clip(shifted, -100, 100)
    w = np.exp(shifted) / np.sum(np.exp(shifted))
    # w has length 2n-1
    # grad[j] = sum_k w[k] * db[k]/da[j] = sum_k w[k] * 2*a[k-j] (if 0<=k-j<n)
    # = 2 * sum_k w[k] * a[k-j]
    # Let R[j] = sum_k w[k] * a[k-j] for k=0..2n-2, j=0..n-1
    # = sum_{m=0}^{n-1} a[m] * w[m+j]  (substituting k=m+j)
    # This is cross-correlation of a and w.
    # Compute via: conv(flip(a), w) and extract
    a_flip = a[::-1].copy()
    N = 1 << (n + 2*n - 2).bit_length()  # need N >= n + 2n-2
    Af = np.fft.rfft(a_flip, N)
    Wf = np.fft.rfft(w, N)
    corr = np.fft.irfft(Af * Wf, N)
    # corr[k] = sum_i a_flip[i] * w[k-i] = conv(a_flip, w)[k]
    # R[j] = corr[n-1+j]
    grad = 2.0 * corr[n-1:n-1+n]
    return grad

def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

import numpy as np
import time


def _fft_conv(a):
    """Linear self-convolution of a via real FFT. O(n log n)."""
    n = a.size
    N = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:2 * n - 1]


def _smooth_grad(b, a, alpha):
    """Gradient of log-sum-exp smooth max of b w.r.t. a, via FFT cross-correlation."""
    n = a.size
    bmax = b.max()
    shifted = np.clip(alpha * (b - bmax), -80, 80)
    w = np.exp(shifted)
    w /= w.sum()
    # R[j] = sum_m a[m]*w[m+j] = conv(flip(a), w)[n-1+j]
    a_flip = a[::-1].copy()
    Nc = 1 << (3 * n - 2).bit_length()
    Af = np.fft.rfft(a_flip, Nc)
    Wf = np.fft.rfft(w, Nc)
    corr = np.fft.irfft(Af * Wf, Nc)
    return 2.0 * corr[n - 1:n - 1 + n]


def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if rho.size == 0:
        return np.full(n, 1.0 / n)
    theta = (cssv[rho[-1]] - 1.0) / (rho[-1] + 1.0)
    return np.maximum(v - theta, 0.0)

    peak = b.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * peak / (s * s))
