import numpy as np
import time

def gen_smooth_perturbation(n, rng, nfreq=5):
    """Generate a smooth zero-mean perturbation of length n."""
    t = np.arange(n, dtype=np.float64) / n
    g = np.zeros(n, dtype=np.float64)
    for _ in range(nfreq):
        freq = rng.uniform(1.0, 4.0) * np.pi
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.uniform(0.3, 1.0)
        g += amp * np.sin(freq * t + phase)
    g -= g.mean()
    return g

def smooth_perturb(base, rng, sigma=0.1):
    """Multiplicative smooth perturbation preserving nonnegativity."""
    n = base.size
    g = gen_smooth_perturbation(n, rng)
    g = g / (np.std(g) + 1e-12)
    candidate = base * np.exp(sigma * g)
    return np.clip(candidate, 0.0, 1000.0)

def block_perturb(base, rng, sigma=0.15):
    """Perturb a random block of the sequence."""
    n = base.size
    wmax = max(2, n // 10)

    start = int(rng.integers(0, n - width + 1))
    candidate = base.copy()
    candidate[start:start+width] *= np.exp(rng.normal(0.0, sigma))
    return np.clip(candidate, 0.0, 1000.0)
