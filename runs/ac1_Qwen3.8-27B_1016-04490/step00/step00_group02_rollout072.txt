def _conv(a,b):
   need=len(a)+len(b)-1
   if need <= 0: return np.array([], dtype=np.float64)
   if need < 4096:
       return np.convolve(a,b)
   size=1
   while size < need:
       size <<= 1
   fa=np.fft.rfft(a,size)
   fb=np.fft.rfft(b,size)
   out=np.fft.irfft(fa*fb,size)
   return out[:need]

import numpy as np
import time


def _finite_nonneg(arr):
    try:
        a = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    return np.clip(a, 0.0, 1000.0)


def _normalize(arr):
    a = _finite_nonneg(arr)
    if a.size == 0:
        a = np.ones(1, dtype=np.float64)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(a.size, 1.0 / float(a.size if a.size else 1))
    return a / s


def _update_best(seq, best, best_score):
    try:
        lst = [float(x) for x in seq]
        sc = evaluate_sequence(lst)
    except Exception:
        sc = np.inf
    if np.isfinite(sc) and sc < best_score:
        arr = np.asarray(seq, dtype=np.float64).ravel().copy()
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        if arr.size and float(arr.sum()) > 0.0:
            return arr, float(sc)
    return best, float(best_score)


def _conv(a, b):
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    need = a.size + b.size - 1
    if need <= 0:
        return np.array([], dtype=np.float64)
    if need < 4096:
        return np.convolve(a, b)
    size = 1
    while size < need:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    out = np.fft.irfft(fa * fb, size)
    return out[:need]


def _fft_score(a):
    p = _normalize(a)
    s = float(p.sum())
    if p.size == 0 or s <= 1e-300:
        return np.inf
    c = _conv(p, p)
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    if c.size == 0:
        return np.inf
    return 2.0 * p.size * float(np.max(c)) / (s * s)


def _make_power(n, alpha, lin):
    if n <= 1:
        return np.ones(max(n, 1), dtype=np.float64)
    i = np.arange(1, n + 1, dtype=np.float64)
    logv = -float(alpha) * np.log(i)
    if lin != 0.0:
        x = i / float(n)
        f = 1.0 + float(lin) * x
        f = np.where(f > 1e-8, f, 1e-8)
        logv += np.log(f)
    logv -= float(np.max(logv))
    v = np.exp(logv)
    v = np.where(np.isfinite(v), v, 0.0)
    s = float(v.sum())
    if not np.isfinite(s) or s <= 0.0:
        v = np.ones(n, dtype=np.float64)
        s = float(n)
    return v / s


def _make_exp(n, lam):
    if n <= 1:
        return np.ones(max(n, 1), dtype=np.float64)
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    v = np.exp(float(lam) * x)
    v = np.where(np.isfinite(v), v, 0.0)
    s = float(v.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(n, 1.0 / float(n))
    return v / s


def _add_to_top(scores, arrs, score, arr, cap):
    if not np.isfinite(score):
        return
    score = float(score)
    if len(arrs) < cap:
        scores.append(score)
        arrs.append(np.asarray(arr, dtype=np.float64).copy())
        return
    mx = scores[0]
    mi = 0
    for k in range(1, len(scores)):
        if scores[k] > mx:
            mx = scores[k]
            mi = k
    if score < mx:
        scores[mi] = score
        arrs[mi] = np.asarray(arr, dtype=np.float64).copy()


def _weights_from_conv(c, cmax):
    n = c.size
    if n == 0 or cmax <= 1e-300:
        return np.ones(max(n, 1), dtype=np.float64)
    tau = 0.5 * cmax
    for _ in range(6):
        if tau <= 0.0:
            break
        sh = (c - cmax) / tau
        sh = np.where(sh > -12.0, sh, -12.0)
        w = np.exp(sh)
        w = np.where(np.isfinite(w), w, 0.0)
        sw = float(w.sum())
        if np.isfinite(sw) and sw > 0.0:
            w /= sw
            if float(np.max(w)) < 0.45:
                return w
        tau *= 2.0
    return np.full(n, 1.0 / float(n), dtype=np.float64)


def _gradient_for_weights(p, w):
    n = p.size
    if n == 0:
        return np.array([], dtype=np.float64)
    m = w.size - 1
    wr = w[::-1]
    cp = _conv(p, wr)
    if cp.size < m + n:
        return np.zeros(n, dtype=np.float64)
    a = cp[m - n + 1:m + 1][::-1]
    pref = np.empty(2 * n, dtype=np.float64)
    pref[0] = 0.0
    np.cumsum(w, out=pref[1:])
    s = pref[n:n + n] - pref[:n]
    g = a + p * s
    g = np.where(np.isfinite(g), g, 0.0)
    return np.maximum(g, 0.0)


def _local_refine(start, seed, deadline, best, best_score):
    n = start.size
    if n < 2:
        return best, _update_best(start, best, best_score)
    p = _normalize(start)
    best, best_score = _update_best(p, best, best_score)

    rng = np.random.default_rng(seed)
    record_score = _fft_score(p)
    record_arr = p.copy()
    exact_limit = 0
    stagnant = 0
    it = 0

    while it < 8000:
        it += 1
        if time.monotonic() >= deadline:
            break

        s = float(p.sum())
        if s <= 1e-300:
            break
        c = _conv(p, p)
        c = np.where(np.isfinite(c), c, 0.0)
        c = np.maximum(c, 0.0)
        c /= (s * s)
        if c.size == 0:
            break
        cmax = float(np.max(c))
        if cmax <= 1e-300:
            break
        old = 2.0 * n * cmax

        w = _weights_from_conv(c, cmax)
        g = _gradient_for_weights(p, w)
        gmax = float(np.max(g))
        if gmax <= 1e-300 or not np.isfinite(gmax):
            p = _normalize(p * np.exp(0.03 * rng.normal(size=n)))
            continue
        g = g / gmax

        accepted = False
        for eta in (0.04, 0.12, 0.35, 0.8, 1.6):
            pc = p * np.exp(-eta * g)
            pc = np.where(np.isfinite(pc), pc, 0.0)
            pcs = float(pc.sum())
            if pcs <= 1e-300:
                continue
            pc = pc / pcs
            cc = _conv(pc, pc)
            cc = np.where(np.isfinite(cc), cc, 0.0)
            cc = np.maximum(cc, 0.0)
            sc2 = 2.0 * n * float(np.max(cc)) / (float(pc.sum()) ** 2)
            if np.isfinite(sc2) and sc2 < old - 1e-13:
                p = pc
                old = sc2
                accepted = True
                break

        if accepted:
            stagnant = 0
            if old < record_score - 1e-12:
                record_score = old
                record_arr = p.copy()
                exact_limit += 1
                if (
                    exact_limit <= 4
                    or exact_limit % 6 == 0
                    or it % 32 == 0
                ) and time.monotonic() < deadline - 3.0:
                    best, best_score = _update_best(p, best, best_score)
        else:
            stagnant += 1
            if stagnant == 1:
                p = _normalize(p * np.exp(0.04 * rng.normal(size=n)))
            elif stagnant == 4:
                mix = _make_power(n, 0.5, float(rng.uniform(-0.3, 1.5)))
                p = _normalize(0.6 * p + 0.4 * mix)
            elif stagnant >= 18:
                break

        if it % 160 == 0 and time.monotonic() < deadline - 3.0:
            best, best_score = _update_best(p, best, best_score)

    if time.monotonic() < deadline - 2.0:
        best, best_score = _update_best(p, best, best_score)
    if time.monotonic() < deadline - 0.5:
        best, best_score = _update_best(record_arr, best, best_score)
    return best, best_score
