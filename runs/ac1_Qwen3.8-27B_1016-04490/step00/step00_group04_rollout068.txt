import time

    def try_cand(cand):
        nonlocal best, best_score
        cand = np.clip(np.asarray(cand, dtype=np.float64), 0.0, 1000.0)
        if cand.size != n or np.sum(cand) < 0.01:
            return
        s = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(s) and s < best_score:
            best = cand.copy()
            best_score = s

    # Grid search: step functions [A]*k + [B]*(n-k)
    for ratio in np.linspace(1.8, 2.5, 8):
        for frac in np.linspace(0.42, 0.58, 9):
            k = int(n * frac)
            if k < 1 or k >= n:
                continue

            cand[:k] = ratio
            cand[k:] = 1.0
            try_cand(cand)

    for ratio in [1.8, 2.0, 2.2, 2.5]:
