import numpy as np
import time


def _fft_conv(a, N):
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)


def _score(a, n, N):
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = _fft_conv(a, N)
    return 2.0 * n * b.max() / (S * S)


def _centered_grad(a, b, n, N, n_peaks):
    S = a.sum()
    mean_a = S / n
    peak_idx = np.argpartition(b, -n_peaks)[-n_peaks:]
    pv = b[peak_idx]
    w_sum = pv.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    grad = np.zeros(n)
    for w, k in zip(pv / w_sum, peak_idx):
        ki = int(k)
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if lo <= hi:
            seg = a[ki - hi: ki - lo + 1][::-1]
            grad[lo:hi + 1] += w * (seg - mean_a)
    return grad


def _normalize(a, target_sum):
    s = a.sum()
    if s > 1e-30:
        a = a * (target_sum / s)
    return np.clip(a, 0.0, 1000.0)


def _param_init(n, rng, kind):
    i = np.arange(n)
    if kind == 0:
        return np.ones(n)
    elif kind == 1:
        return 1.0 + 0.3 * np.cos(2 * np.pi * i / n)
    elif kind == 2:
        return 1.0 + 0.3 * np.sin(2 * np.pi * i / n)
    elif kind == 3:
        c = n // 2
        return np.exp(-0.5 * ((i - c) / (n * 0.15)) ** 2)
    elif kind == 4:
        c = n // 4
        c2 = 3 * n // 4
        return np.exp(-0.5 * ((i - c) / (n * 0.1)) ** 2) + np.exp(-0.5 * ((i - c2) / (n * 0.1)) ** 2)
    elif kind == 5:
        return np.ones(n) * (1.0 + 0.2 * rng.standard_normal(n))
    elif kind == 6:
        f1 = rng.integers(1, 8)
        f2 = rng.integers(1, 8)
        return 1.0 + 0.25 * np.cos(2 * np.pi * f1 * i / n) + 0.25 * np.cos(2 * np.pi * f2 * i / n)
    else:
        return np.ones(n) * (1.0 + 0.5 * rng.standard_normal(n))
