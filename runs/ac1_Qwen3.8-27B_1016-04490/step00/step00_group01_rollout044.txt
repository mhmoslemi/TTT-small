        best = np.ones(n, dtype=np.float64)
        best_score = 2.0

    result = [float(x) for x in best]
    # Verify returned list is valid
    fs = float(evaluate_sequence(result))
    if not np.isfinite(fs):
        result = [1.0] * n
    return result
