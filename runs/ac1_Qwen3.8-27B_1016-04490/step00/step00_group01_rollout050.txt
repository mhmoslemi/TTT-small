def _conv(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    N = 1
    while N < n + m - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]
