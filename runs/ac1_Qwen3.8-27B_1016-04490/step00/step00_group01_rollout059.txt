import numpy as np
import time

def compute_grad(a, c, n, K):
    grad = np.zeros(n)
    peaks = np.argpartition(c, -K)[-K:]
    for k in peaks:
        k = int(k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        js = np.arange(j_lo, j_hi + 1)
        grad[js] += a[k - js]
    return grad
