def _project(v, S):
    """Project v onto {x >= 0, sum(x) = S} via 1D bisection on tau."""
    n = v.size
    lo = 0.0
    hi = v.max() - S / n + 1.0
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        if np.maximum(v - mid, 0.0).sum() > S:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    return np.maximum(v - tau, 0.0)
