# lo_i = max(0, n-1-k_hi+i)

hi = np.minimum(n - 1, n - 1 - k_lo + np.arange(n))

def grad_from_peaks(a, c, K, n):

    grad = np.zeros(n)

    for k in top_k:

        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            grad[i_lo:i_hi+1] += a[k - i_hi : k - i_lo - 1 : -1]

    peak_idx = np.argpartition(c, -K)[-K:]
