import numpy as np

def quick_score(a):
    """Fast score using NumPy directly (avoids list conversion overhead)."""
    n = a.size
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    conv = np.convolve(a, a)
    mb = float(np.max(conv))
    return float(2.0 * n * mb / (s * s))


def smooth_step(a, window):
    """Moving-average smoothing, O(n)."""
    n = a.size
    if window <= 1:
        return a.copy()
    kernel = np.ones(window, dtype=np.float64) / window
    smoothed = np.convolve(a, kernel, mode='same')
    return np.clip(smoothed, 0.0, 1000.0)
