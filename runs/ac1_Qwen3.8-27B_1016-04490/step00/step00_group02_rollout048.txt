import time

def compute_score_fast(a):
    """Compute the score without going through the evaluator (for speed during search)."""
    n = len(a)
    b = np.convolve(a, a)
    max_b = np.max(b)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    return 2.0 * n * max_b / (S * S)

    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
