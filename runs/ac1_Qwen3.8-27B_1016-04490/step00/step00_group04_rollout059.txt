```python
import numpy as np
import time


def fft_conv(a, b=None):
    """Linear convolution via FFT, O(n log n)."""
    if b is None:
        b = a
    out_len = a.size + b.size - 1
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:out_len]


def subgrad_peak(a, k_star):
    """Subgradient of (a*a)[k_star] w.r.t. a, O(n)."""
    n = a.size
    grad = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n, k_star + 1)
    j_idx = range(lo, hi)
    for j in j_idx:
        p = k_star - j
        if 0 <= p < n:
            grad[j] = 2.0 * a[p]
    return grad
```

```python
import numpy as np
import time


def fft_conv(a, b=None):
    if b is None:
        b = a
    out_len = a.size + b.size - 1
    size = 1 << max(1, (out_len - 1).bit_length())
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:out_len]


def subgrad_peak(a, k_star):
    n = a.size
    grad = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n, k_star + 1)
    j_idx = np.arange(lo, hi)
    p_idx = k_star - j_idx
    valid = (p_idx >= 0) & (p_idx < n)
    grad[j_idx[valid]] = 2.0 * a[p_idx[valid]]
    return grad


def multi_subgrad(a, b, frac=0.95):
    n = a.size
    b_max = b.max()
    threshold = frac * b_max
    peak_idx = np.where(b >= threshold)[0]
    if peak_idx.size == 0:
        peak_idx = np.array([int(np.argmax(b))])
    grad = np.zeros(n)
    count = 0
    for k in peak_idx:
        lo = max(0, int(k) - n + 1)
        hi = min(n, int(k) + 1)
        j_idx = np.arange(lo, hi)
        p_idx = int(k) - j_idx
        valid = (p_idx >= 0) & (p_idx < n)
        grad[j_idx[valid]] += a[p_idx[valid]]
        count += 1
    if count > 0:
        grad *= 2.0 / (count * b_max)
    return grad


def block_perturb(a, rng, strength=0.15):
    n = a.size
    out = a.copy()
    nblocks = rng.integers(2, 8)
    for _ in range(nblocks):
        w = int(rng.integers(max(3, n // 200), max(4, n // 8)))
        start = int(rng.integers(0, n - w + 1))
        scaling = np.exp(rng.normal(0, strength, size=w))
        out[start:start + w] *= scaling
    out = np.clip(out, 0, 1000)
    s = out.sum()
    if s > 0:
        out *= a.sum() / s
    return out


def sinusoid_perturb(a, rng, n_harm=3):
    n = a.size
    x = np.arange(n, dtype=np.float64)
    delta = np.zeros(n)
    for k in range(1, n_harm + 1):
        amp = rng.uniform(0.02, 0.12)
        phase = rng.uniform(0, 2.0 * np.pi)
        delta += amp * np.sin(2.0 * np.pi * k * x / n + phase)
    out = np.clip(a * (1.0 + delta), 0, 1000)
    s = out.sum()
    if s > 0:
        out *= a.sum() / s
    return out
```
