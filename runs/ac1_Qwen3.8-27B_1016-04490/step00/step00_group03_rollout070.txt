def set_opt_for_n(n):
    global OPT_N, OPT_L, OPT_NA, OPT_NG
    if n == OPT_N:
        return
    OPT_N = int(n)
    OPT_L = 2 * OPT_N - 1

    nlin = OPT_N + OPT_L - 1

def softmax(z):
    z = np.asarray(z, dtype=np.float64)
    if z.size == 0:
        return np.ones(0, dtype=np.float64)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(z.shape, 1.0 / float(z.size), dtype=np.float64)
    return e / s


def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.ones(0, dtype=np.float64)
    L = 2 * n - 1
    N = 1 << max(0, (L - 1).bit_length())
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    c = np.real(c)
    return np.clip(c, 0.0, None)


def norm_p(a):
    if a is None:
        return None
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1 or a.size == 0:
        return None
    if not np.isfinite(a).all():
        return None
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s < 0.01:
        return None
    p = a / s
    if not np.isfinite(p).all():
        return None
    return p

def logsumexp_scalar(p):
    set_opt_for_n(p.size)
    beta = float(OPT_BETA)
    c = fast_autocorr(p)
    u = float(p.size) * c
    if u.size == 0:
        return 0.0
    umax = float(np.max(u))
    ex = np.exp(beta * (u - umax))
    s = float(np.sum(ex))
    if s <= 0.0 or not np.isfinite(s):
        return 1e100
    return float((beta * umax + np.log(s)) / beta)

    val = logsumexp_scalar(p)
    if not np.isfinite(val):
        return 1e100
    m = float(np.mean(z))
    val += float(OPT_LAMBDA) * m * m
    return float(val)


def logsumexp_gradient(z):
    set_opt_for_n(z.size)
    n = int(OPT_N)
    beta = float(OPT_BETA)
    p = softmax(z)

    c = fast_autocorr(p)
    u = float(n) * c
    umax = float(np.max(u))
    ex = np.exp(beta * (u - umax))
    s = float(np.sum(ex))
    if s <= 0.0 or not np.isfinite(s):
        return np.zeros(n, dtype=np.float64)
    w = ex / s

    N = int(OPT_NG)
    fp = np.fft.rfft(p, N)
    fw = np.fft.rfft(w, N)

    corr = np.fft.irfft(fp * np.conj(fw), N)
    conv = np.fft.irfft(fp * fw, N)

    dminus = conv[:n]
    dplus = np.empty(n, dtype=np.float64)
    dplus[0] = corr[0]
    if n > 1:
        dplus[1:] = corr[N - 1:N - n:-1]

    g = (float(n) / beta) * (dplus + dminus)
    if not np.isfinite(g).all():
        return np.zeros(n, dtype=np.float64)

    dot = float(np.dot(p, g))
    grad = p * (g - dot)
    grad += 2.0 * float(OPT_LAMBDA) * float(np.mean(z))
    if not np.isfinite(grad).all():
        return np.zeros(n, dtype=np.float64)
    return grad

            break

        OPT_BETA = float(beta)
        z = current_z - float(np.mean(current_z))
        amax = float(np.max(np.abs(z)))
        if amax > 30.0 and amax > 0.0:
            z *= 30.0 / amax

        val_before = logsumexp_objective(z)
        if not np.isfinite(val_before):
            break

        mit = max(5, int(mit0 * scale))
        try:
            res = _sopt.minimize(
                logsumexp_objective,
                z,
                jac=logsumexp_gradient,
                method="L-BFGS-B",
                callback=_timeout_callback,
                options={
                    "maxiter": mit,
                    "ftol": 1e-10,
                    "gtol": 1e-7,
                    "maxls": 20
                }
            )
        except Exception:
            res = None

        if res is not None and res.x is not None:
            x = np.asarray(res.x, dtype=np.float64)
            if x.size == n and np.isfinite(x).all() and np.isfinite(res.fun):
                z_new = x - float(np.mean(x))
                p_new = norm_p(softmax(z_new))
                if p_new is not None and float(res.fun) <= val_before + 1e-7:
                    current_z = z_new

        p_cur = norm_p(softmax(current_z - float(np.mean(current_z))))
        if p_cur is None:
            break
        best, best_score = core_update(p_cur, best, best_score)

    return best, best_score


def final_polish(p_in, best, best_score, wall_deadline, rng):
    p = norm_p(p_in)
    if p is None:
        return best, best_score

    p = p.copy()
    fast_best = fast_score(p)
    if not np.isfinite(fast_best):
        fast_best = best_score

    eta = 0.012
    kernel = np.ones(5, dtype=np.float64) / 5.0

    while time.monotonic() < wall_deadline - 25.0 and eta > 1e-6:
        noise = rng.normal(0.0, eta, p.size)
        noise = np.convolve(noise, kernel, mode="same")
        cand = p * np.exp(noise)
        cand = norm_p(cand)
        if cand is None:
            eta *= 0.8
            continue

        fc = fast_score(cand)
        if np.isfinite(fc) and fc < fast_best - 1e-10:
            best, best_score = core_update(cand, best, best_score)
            p = cand.copy()
            fast_best = fc
            eta = min(0.18, eta * 1.05)
        else:
            eta *= 0.86

    best, best_score = core_update(p, best, best_score)
    return best, best_score


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    global TIME_DEADLINE

    try:
        b = float(budget_s)
    except Exception:
        b = 1100.0
    if b <= 0.0:
        b = 1.0

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    wall_deadline = start + max(5.0, b - 25.0)
    TIME_DEADLINE = wall_deadline - 8.0

    n = 2535
    parent = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64)
        if raw.ndim == 1 and raw.size > 0:
            n = int(raw.size)
            parent = norm_p(raw)
    except Exception:
        parent = None

    if parent is None:
        parent = np.ones(n, dtype=np.float64)

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(best_score):
        best = np.ones(n, dtype=np.float64)
        best_score = float(evaluate_sequence(best.tolist()))
        if not np.isfinite(best_score):
            best_score = 2.0

    top = []
    add_candidate(best, best, best_score, top, 5)

    for cand in make_candidates(n):
        if time.monotonic() > wall_deadline - 60.0:
            break
        best, best_score = add_candidate(cand, best, best_score, top, 5)

    set_opt_for_n(n)

    starts = []
    for _score, p in top[:5]:
        if len(starts) >= 3 or time.monotonic() > wall_deadline - 140.0:
            break
        skip = False
        for q in starts:
            if np.allclose(p, q, rtol=1e-2, atol=1e-5):
                skip = True
                break
        if not skip:
            starts.append(p.copy())

    for p0 in starts:
        if time.monotonic() > wall_deadline - 100.0:
            break
        best, best_score = run_full_optimize(p0, best, best_score, wall_deadline)

    if time.monotonic() < wall_deadline - 35.0:
        best, best_score = final_polish(best, best, best_score, wall_deadline, rng)

    best, best_score = core_update(best, best, best_score)
    if best is None or best.size == 0 or not np.isfinite(best).all():
        best = np.ones(n, dtype=np.float64)

    return [float(x) for x in best]
