def build_sparse_rows(f, top_lags):
    """Build sparse matrix rows for selected lags of convolution with f."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in top_lags:
        js = np.arange(max(0, k - n + 1), min(n - 1, k) + 1)
        if len(js) == 0:
            continue
        rows.extend([int(k)] * len(js))
        cols.extend(js.tolist())
        vals.extend(f[k - js].tolist())
    return (np.array(rows, dtype=np.int32),

def _build_row(f, k):
    n = len(f)
    js = np.arange(max(0, k - n + 1), min(n - 1, k) + 1)
    if js.size == 0:
        return js, np.empty(0)
    return js, f[k - js]


def _lp_direction(f, deadline, K=40):
    n = len(f)
    ff = np.convolve(f, f)
    M = float(np.max(ff))
    if M <= 0:
        return None
    top = np.argsort(ff)[::-1][:K]
    rr, cc, vv = [], [], []
    for k in top:
        js, vs = _build_row(f, int(k))
        if js.size:
            rr.extend([int(k)] * js.size)
            cc.extend(js.tolist())
            vv.extend(vs.tolist())
    if not rr:
        return None
    T = sp.csr_matrix((np.array(vv), (np.array(rr, dtype=np.int32),
                                       np.array(cc, dtype=np.int32))),
                      shape=(K, n))
    tl = max(1.0, float(deadline) - time.monotonic() - 20.0)
    res = linprog(-np.ones(n), A_ub=T, b_ub=np.full(K, M),
                  bounds=[(0.0, None)] * n, method='highs',
                  options={'time_limit': tl})
    if not getattr(res, 'success', False):
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0.0, None)
    return g if g.sum() > 1e-12 else None
