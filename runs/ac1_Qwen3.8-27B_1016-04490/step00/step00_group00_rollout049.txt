import numpy as np
import time
from scipy.fft import fft, ifft, next_fast_len

def fft_convolve(a):
    """Full convolution via FFT, O(n log n)."""
    n = len(a)
    size = next_fast_len(2*n - 1)
    fa = fft(a, size)
    b = np.real(ifft(fa * fa, size))[:2*n-1]
    return b

def compute_gradient(a, tau_star, M):
    """Compute gradient of 2n*M/S^2 at the peak lag."""
    n = len(a)
    S = np.sum(a)
    # grad[k] = 2n * (dM/da[k] / S - 2M/S^2)
    # dM/da[k] = 2*a[tau_star - k] if valid
    grad = np.zeros(n)
    # a[tau_star - k] is valid when 0 <= tau_star - k < n
    # i.e., tau_star - n < k <= tau_star, clamped to [0, n-1]
    k_min = max(0, tau_star - n + 1)
    k_max = min(n - 1, tau_star)
    valid = np.arange(k_min, k_max + 1)
    idx = tau_star - valid  # these are the valid indices for a[tau_star - k]
    grad[valid] = 2.0 * a[idx]
    # Full gradient of F
    grad = 2.0 * n * (grad / S - 2.0 * M / (S * S))
    return grad

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 5.0)

    # Evaluate initial

    best_score = float(evaluate_sequence(best.tolist()))
    
    # Phase 1: Gradient descent
    lr = 1.0 / n  # learning rate
    for iteration in range(1000000):
        if time.monotonic() > deadline - 10.0:
            break
        
        # Compute convolution
        b = fft_convolve(a)
        tau_star = int(np.argmax(b))
        M = b[tau_star]
        S = np.sum(a)
        
        # Compute gradient
        grad = compute_gradient(a, tau_star, M)
        
        # Gradient step
        a = a - lr * grad
        a = np.maximum(a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        
        # Renormalize to keep sum stable
        S_new = np.sum(a)
        if S_new > 0:
            a = a * (S / S_new)  # or a = a / S_new * S
        
        # Periodic evaluation
        if iteration % 50 == 0:
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < best_score:
                best = a.copy()
                best_score = score
        
        # Adaptive learning rate
        lr *= 0.9999

        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            a = candidate
            best_score = score
    
    return [float(x) for x in best]
