import time


def _block_perturb(arr, rng):
    n = arr.size
    out = arr.copy()
    # Pick 1-3 blocks to perturb
    nblocks = rng.integers(1, 4)
    for _ in range(nblocks):
        width = rng.integers(max(2, n // 200), max(3, n // 8))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + int(width))
        out[start:stop] *= float(np.exp(rng.normal(0.0, 0.08)))
    return np.clip(out, 0.0, 1000.0)


def _smoothing_perturb(arr, rng):
    n = arr.size
    out = arr.copy()
    # Blend with a smoothed version
    window = int(rng.integers(3, max(4, n // 50)))

    kernel = np.ones(window) / window
    smoothed = np.convolve(out, kernel, mode='same')

    out = (1 - t) * out + t * smoothed
    return np.clip(out, 0.0, 1000.0)

import numpy as np
import time


def _block_perturb(arr, rng):
    n = arr.size
    out = arr.copy()
    nblocks = int(rng.integers(1, 4))
    for _ in range(nblocks):
        width = int(rng.integers(max(2, n // 200), max(3, n // 8)))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        out[start:stop] *= float(np.exp(rng.normal(0.0, 0.08)))
    return np.clip(out, 0.0, 1000.0)


def _smoothing_perturb(arr, rng):
    n = arr.size
    out = arr.copy()
    window = int(rng.integers(3, max(4, n // 50)))
    if window < 3:
        window = 3
    kernel = np.ones(window, dtype=np.float64) / window
    smoothed = np.convolve(out, kernel, mode='same')
    t = float(rng.uniform(0.05, 0.3))
    out = (1.0 - t) * out + t * smoothed
    return np.clip(out, 0.0, 1000.0)
