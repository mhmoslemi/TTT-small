def _compute_gradient(a, k_star, n):
    """Compute subgradient of c[k_star] w.r.t. a."""
    g = np.zeros(n)

    hi = min(n - 1, k_star)
    g[lo:hi+1] = 2.0 * a[k_star - hi : k_star - lo + 1][::-1]
    return g

    """Remove the uniform component (tangent to the simplex)."""
    return g - np.mean(g)
