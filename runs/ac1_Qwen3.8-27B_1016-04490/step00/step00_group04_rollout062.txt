import numpy as np
import time


def _make_geo(n, v):
    r = np.exp(-v / n)
    return np.power(r, np.arange(n, dtype=np.float64))


def _make_cosine(n, alpha):
    j = np.arange(n, dtype=np.float64)
    return np.clip(1.0 + alpha * np.cos(2.0 * np.pi * j / n), 0.01, None)


def _make_power(n, p):
    return np.power(0.3 + 0.7 * np.linspace(0.0, 1.0, n, dtype=np.float64), p)


def _score(arr):
    return float(evaluate_sequence(arr.tolist()))


def _subgradient_step(a, eta, k_star_override=None):
    conv = np.convolve(a, a)
    if k_star_override is None:
        k_star = int(np.argmax(conv))
    else:
        k_star = k_star_override
    n = a.size
    grad = np.zeros(n, dtype=np.float64)
    lo = max(0, k_star - (n - 1))
    hi = min(n - 1, k_star)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        grad[idx] = a[k_star - idx]
    a_new = a - eta * grad
    a_new = np.clip(a_new, 0.0, 1000.0)
    s = a_new.sum()
    if s < 1e-12:
        return a.copy()
    a_new *= a.sum() / s
    return a_new


def _block_perturb(a, rng, width, noise_scale):
    trial = a.copy()
    w = max(2, int(width))
    start = int(rng.integers(0, max(1, a.size - w)))
    stop = min(a.size, start + w)
    trial[start:stop] *= np.exp(rng.normal(0.0, noise_scale))
    trial = np.clip(trial, 0.0, 1000.0)
    return trial

    best = parent.copy()

        pw = _make_power(n, p)
        if pw.sum() > 0.01:

    # Symmetric geometric (product of two ramps)
