import numpy as np
import time


def make_linear_tilt(n, t):
    x = np.arange(n, dtype=np.float64)
    if n > 1:
        x = 2.0 * x / (n - 1.0) - 1.0
    return np.maximum(0.0, 1.0 + t * x)


def make_exp_tilt(n, t):
    x = np.arange(n, dtype=np.float64)
    if n > 1:
        x = 2.0 * x / (n - 1.0) - 1.0
    return np.exp(t * x)


def make_quad_tilt(n, t):
    x = np.arange(n, dtype=np.float64)
    if n > 1:
        x = 2.0 * x / (n - 1.0) - 1.0
    return np.maximum(0.0, 1.0 + t * x * x - 0.5 * t)


def make_v_shape(n, t):
    x = np.arange(n, dtype=np.float64)
    if n > 1:
        c = (n - 1.0) / 2.0
        x = (x - c) / c
    return np.maximum(0.01, 1.0 + t * x * x)


def block_perturb(base, rng):
    n = base.size
    cand = base.copy()
    mode = rng.random()
    if mode < 0.5:
        w = int(rng.integers(max(1, n // 500), max(2, n // 10)))
        s = int(rng.integers(0, max(1, n - w)))
        e = min(n, s + w)
        cand[s:e] *= float(np.exp(rng.normal(0, 0.15)))
    elif mode < 0.8:
        # Anti-palindromic tweak: adjust pairs (i, n-1-i)
        for _ in range(int(rng.integers(5, 50))):
            i = int(rng.integers(0, n))
            j = n - 1 - i
            if i == j:
                continue
            d = rng.normal(0, 0.05) * min(cand[i], cand[j])
            cand[i] = max(0.0, cand[i] + d)
            cand[j] = max(0.0, cand[j] - d)
    else:
        # Two-block transfer
        w = int(rng.integers(max(1, n // 300), max(2, n // 20)))
        s1 = int(rng.integers(0, max(1, n - w)))
        s2 = int(rng.integers(0, max(1, n - w)))
        if abs(s1 - s2) < w:
            s2 = (s1 + n // 2) % max(1, n - w)
        amt = rng.uniform(0.05, 0.3) * cand[s1:s1 + w].mean()
        cand[s1:s1 + w] = np.maximum(0.0, cand[s1:s1 + w] - amt)
        cand[s2:s2 + w] = np.minimum(1000.0, cand[s2:s2 + w] + amt)
    return np.clip(cand, 0.0, 1000.0)
