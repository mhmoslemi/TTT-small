def _fft_convolve(a, N):
    """Compute linear convolution of a with itself via FFT."""
    n = len(a)
    out = np.zeros(N)
    out[:n] = a
    A = np.fft.rfft(out)
    B = A * A
    conv = np.fft.irfft(B, N)[:2*n-1]
    return conv

def _score_fft(a, N):
    """Compute score using FFT for speed."""
    n = len(a)
    conv = _fft_convolve(a, N)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)
