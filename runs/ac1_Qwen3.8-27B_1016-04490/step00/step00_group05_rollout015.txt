def score_arr(a):
    s = a.tolist()
    return float(evaluate_sequence(s))

    return np.exp(lam * j / (n - 1.0))


def gen_power(n, p, base):
    j = np.arange(n, dtype=np.float64)
    t = j / (n - 1.0)
    return np.clip((base + t) ** p, 0.0, 1000.0)


def gen_two_rate(n, l1, l2):
    j = np.arange(n, dtype=np.float64)
    x = np.exp(l1 * j / (n - 1.0))
    w = 0.5 * (1.0 + np.tanh(2.0 * (j - 0.5 * (n - 1.0)) / (0.1 * n)))
    x2 = np.exp(l2 * j / (n - 1.0))
    return np.clip(x * (1.0 - w) + x2 * w, 0.0, 1000.0)


def block_perturb(x, rng, scale, wmin, wmax):
    n = x.size
    c = x.copy()
    width = int(rng.integers(wmin, wmax + 1))
    width = min(width, n)
    start = int(rng.integers(0, n - width + 1))
    stop = min(n, start + width)
    c[start:stop] *= np.exp(rng.normal(0.0, scale, size=width))
    c = np.clip(c, 0.0, 1000.0)
    if rng.random() < 0.25:
        c[:] *= np.exp(rng.normal(0.0, scale * 0.3))
    return np.clip(c, 0.0, 1000.0)


def smooth_ratio_perturb(x, rng, scale):
    n = x.size
    c = x.copy()
    half = max(2, n // 8)
    left = 0.5 * (c[:half] + c[:half][::-1])
    right = 0.5 * (c[n - half:] + c[n - half:][::-1])
    c[:half] = left * np.exp(rng.normal(0.0, scale, size=half))
    c[n - half:] = right * np.exp(rng.normal(0.0, scale, size=half))

            best_score = s

    for p in [1.5, 2.0, 2.5, 3.0, 4.0, 5.0, 7.0, 10.0]:
        for base_v in [1.0, 2.0, 5.0]:
            if time.monotonic() > deadline:
                break
            c = gen_power(n, float(p), float(base_v))
            s = score_arr(c)
            if np.isfinite(s) and s < best_score:
                best = c.copy()
                best_score = s

    for l1 in [2.0, 4.0, 6.0]:
        for l2 in [3.0, 5.0, 7.0]:
            if time.monotonic() > deadline:
                break
            c = gen_two_rate(n, float(l1), float(l2))
            s = score_arr(c)
            if np.isfinite(s) and s < best_score:
                best = c.copy()
                best_score = s

    # Local search with adaptive exploration
    steps_since_improve = 0
    max_patience = 80
    base_scale = 0.08
    while time.monotonic() < deadline:
        phase = 1 + (steps_since_improve > 30) + (steps_since_improve > 60)
        scale = base_scale * (0.6 ** (phase - 1))
        wmin = max(1, n // 2000)
        wmax = max(wmin + 1, n // (60 * phase + 20))

        if rng.random() < 0.6:
            cand = block_perturb(best, rng, scale, wmin, wmax)
        else:
            cand = smooth_ratio_perturb(best, rng, scale)

        s = score_arr(cand)
        if not np.isfinite(s):
            steps_since_improve += 1
            continue

        acc_prob = 1.0
        if s > best_score:
            t = 0.02 / (1.0 + 0.005 * steps_since_improve)
            acc_prob = np.exp(-min(50.0, (s - best_score) / max(t, 1e-8)))

        if s < best_score:
            best = cand.copy()
            best_score = s
            steps_since_improve = 0
        elif rng.random() < acc_prob:
            best = cand.copy()

        steps_since_improve += 1

    return [float(v) for v in best]
