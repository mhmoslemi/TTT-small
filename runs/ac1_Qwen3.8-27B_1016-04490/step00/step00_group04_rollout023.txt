```python
    # These differ: k-n+1+j vs j+k, a shift of n-1
    # So g[k] = (u_rev * v)[n-1+k]
    
    u_rev = a[::-1]
    conv = np.convolve(u_rev, w)  # length n + (2n-1) - 1 = 3n-2
    grad = conv[n-1:n-1+n]  # g[k] = conv[n-1+k] for k=0,...,n-1
    
    return grad
```
