import numpy as np
import time

def project_simplex(v, target=1.0):
    """O(n log n) projection onto {x>=0, sum(x)=target}."""
    n = v.size
    if n == 0:
        return v.copy()
    sv = np.sort(v)[::-1]
    cssv = np.cumsum(sv)
    r = np.arange(1, n + 1, dtype=np.float64)
    cond = sv > (cssv - target) / r
    if not cond.any():
        return np.full(n, target / n, dtype=np.float64)
    rho = int(np.where(cond)[0][0])
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0).astype(np.float64)

def init_cosine(n):
    i = np.arange(n, dtype=np.float64)

    a = np.cos(np.clip(x, -1.0, 1.0) * np.pi / 2.0) ** 2
    a = np.maximum(a, 1e-10)
    return a / a.sum()

import numpy as np
import time


def project_simplex(v, target=1.0):
    """Project v onto {x>=0, sum(x)=target} in O(n log n)."""
    n = v.size
    if n == 0:
        return v.copy()
    sv = np.sort(v)[::-1]
    cssv = np.cumsum(sv)
    r = np.arange(1, n + 1, dtype=np.float64)
    cond = sv > (cssv - target) / r
    if not cond.any():
        return np.full(n, target / n, dtype=np.float64)
    rho = int(np.where(cond)[0][0])
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0).astype(np.float64)


def init_cosine(n):
    i = np.arange(n, dtype=np.float64)
    x = (i - (n - 1) / 2.0) / (n / 2.0)
    a = np.cos(np.clip(x, -1.0, 1.0) * np.pi / 2.0) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def init_gaussian(n, sf):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - (n - 1) / 2.0) / (sf * n)) ** 2)
    return a / a.sum()


def init_poly(n, p):
    """Polynomial bump: (1-x^2)^p where x in [-1,1]."""
    i = np.arange(n, dtype=np.float64)
    x = (i - (n - 1) / 2.0) / (n / 2.0)
    a = np.maximum(1.0 - x * x, 0.0) ** p
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def run_subgrad(a, lr_init, n_iter, top_k, seed, deadline):
    """Subgradient descent on max(conv(a,a)), simplex constraint."""
    n = a.size
    rng = np.random.default_rng(seed)
    a = project_simplex(a.copy(), 1.0)
    lr = lr_init
    best_local = a.copy()
    best_local_maxc = np.max(np.convolve(a, a))
    x = np.arange(n, dtype=np.int64)
    stall = 0

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        c = np.convolve(a, a)
        if top_k <= 1:
            ks = int(np.argmax(c))
            idx = ks - x
            valid = (idx >= 0) & (idx < n)
            grad = np.zeros(n, dtype=np.float64)
            grad[valid] = 2.0 * a[idx[valid]]
        else:
            ti = np.argsort(c)[::-1][:top_k]
            grad = np.zeros(n, dtype=np.float64)
            for ks in ti:
                idx = int(ks) - x
                valid = (idx >= 0) & (idx < n)
                grad[valid] += 2.0 * a[idx[valid]]
            grad /= float(top_k)

        a = a - lr * grad
        a = project_simplex(a, 1.0)

        if it % 50 == 49:
            cm = np.max(np.convolve(a, a))
            if cm < best_local_maxc - 1e-12:
                best_local_maxc = cm
                best_local = a.copy()
                lr = min(lr * 1.03, 1.0)
                stall = 0
            else:
                stall += 1
                lr = max(lr * 0.97, 0.0005)
            if stall > 300:
                a = best_local + 0.01 * rng.standard_normal(n)
                a = project_simplex(a, 1.0)
                lr = lr_init
                stall = 0

    return best_local, best_local_maxc


def run_deflation(a, n_iter, deadline, seed):
    """Greedy deflation: reduce heaviest pair at argmax lag."""
    n = a.size
    rng = np.random.default_rng(seed)
    a = project_simplex(a.copy(), 1.0)
    best = a.copy()
    best_mc = np.max(np.convolve(a, a))
    eta = 0.02

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        c = np.convolve(a, a)
        ks = int(np.argmax(c))
        # Find pair (i, ks-i) with max product
        i = np.arange(n, dtype=np.int64)
        j = ks - i
        valid = (j >= 0) & (j < n)
        ii = i[valid]
        jj = j[valid]
        prods = a[ii] * a[jj]
        pidx = int(np.argmax(prods))
        ip = int(ii[pidx])
        jp = int(jj[pidx])
        # Reduce both
        red = eta * (a[ip] + a[jp]) + 1e-12
        a[ip] = max(0.0, a[ip] - red / 2.0)
        a[jp] = max(0.0, a[jp] - red / 2.0)
        a = project_simplex(a, 1.0)

        if it % 30 == 29:
            mc = np.max(np.convolve(a, a))
            if mc < best_mc:
                best_mc = mc
                best = a.copy()
                eta = min(eta * 1.02, 0.1)
            else:
                eta = max(eta * 0.98, 0.005)
                if it > 100 and best_mc - mc > 1e-8:
                    a = best + 0.005 * rng.standard_normal(n)
                    a = project_simplex(a, 1.0)

    return best, best_mc
