        np.clip(cand, 0.0, 1000.0, out=cand)
        sc = _safe_eval(cand)
        if sc < best_score - 1e-12:
            best = cand
            best_score = sc
            stall = 0
        else:
            stall += 1
        if time.monotonic() - t0 > float(budget_s) * 0.6:
            break

    # ---- final safety: guarantee a valid, finite, non-empty list ----
    best = np.clip(np.asarray(best, dtype=np.float64), 0.0, 1000.0)
    if best.size == 0 or not np.isfinite(best).all() or best.sum() < 0.01:
        best = np.ones(n, dtype=np.float64)
    if _safe_eval(best) > 2.0 + 1e-9:
        fallback = np.ones(n, dtype=np.float64)
        best = fallback

    return [float(x) for x in best]
