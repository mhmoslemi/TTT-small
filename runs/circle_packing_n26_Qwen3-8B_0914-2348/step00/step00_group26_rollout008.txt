from scipy.optimize import minimize
from scipy.optimize import Bounds

    n = 26
    rows = 6  # 6 rows in the hexagonal grid
    cols = 5  # 5 columns in each row (adjust based on circle size)
    spacing = 0.2  # initial spacing

                x = col * spacing
                y = row * spacing * np.sqrt(3)
            else:
                x = (col + 0.5) * spacing
                y = row * spacing * np.sqrt(3)

    radii = np.full(n, 0.05)

    def repulsion(centers, radii, sigma=0.5, theta=1.5):
        repulsion_energy = 0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist > 0:
                    repulsion_energy += (radii[i] + radii[j]) / (dist**theta) * sigma
        return repulsion_energy

    def objective(x, centers, radii):

                return np.inf

                return np.inf

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
