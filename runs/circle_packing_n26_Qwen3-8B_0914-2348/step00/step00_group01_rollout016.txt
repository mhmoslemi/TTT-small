import scipy.optimize as opt

def run_packing():
    # Initial grid configuration for 26 circles
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Create a grid of 5x5 with 1 circle extra
    for i in range(5):
        for j in range(5):
            idx = i * 5 + j
            if idx >= num_circles:
                break
            centers[idx] = [i / 4, j / 4]

    # Initial radii: All circles are of the same size
    initial_radius = 0.1
    radii[:] = initial_radius

    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize radii

    def constraints(x):

        centers_flat = x[:2 * num_circles].reshape((num_circles, 2))
        radii_flat = x[2 * num_circles:]
        radii = radii_flat

        # Constraint 1: All circles lie within the unit square
        for i in range(num_circles):
            x, y = centers_flat[i]
            r = radii[i]
            if not (0 <= x - r and x + r <= 1):
                return False
            if not (0 <= y - r and y + r <= 1):
                return False
        # Constraint 2: All circles are non-overlapping
        for i in range(num_circles):
            for j in range(i + 1, num_circles):

                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:

        return True

    # Initial guess
    initial_guess = np.concatenate([centers.flatten(), radii])

    # Optimization
    result = opt.minimize(
