import scipy.optimize as opt

    # Hexagonal layout: rows alternate in x-position

    num_circles = 26
    n_rows = 5
    n_cols = 6

    initial_radius = 0.1

    centers = []

        return -np.sum(x)

    def constraint_center(x, i):
        return x[i * 2] - radii[i] >= -1e-12
    def constraint_center2(x, i):
        return x[i * 2] + radii[i] <= 1 + 1e-12
    def constraint_center3(x, i):
        return x[i * 2 + 1] - radii[i] >= -1e-12
    def constraint_center4(x, i):
        return x[i * 2 + 1] + radii[i] <= 1 + 1e-12

    bounds = []

        bounds.extend([(0, 1 + 1e-12)])

    def constraint_overlap(x, i, j):
