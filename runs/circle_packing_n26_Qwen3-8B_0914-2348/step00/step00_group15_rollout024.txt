import scipy.optimize as optimize
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    # Initial base grid configuration for 26 circles (5x5 grid with one extra)
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Step 1: Initial grid configuration
    # Arrange in a grid with one extra circle
    x_coords = np.linspace(0.1, 0.9, 5)
    y_coords = np.linspace(0.1, 0.9, 5)
    idx = 0
    for i in range(5):
        for j in range(5):
            centers[idx] = [x_coords[i], y_coords[j]]
            idx += 1
    # Add an extra circle near the top-right
    centers[25] = [0.65, 0.8]
    
    # Initial guess for radii
    initial_radius = 0.08
    radii[:] = initial_radius

    # Define constraints for optimization
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraint_overlap(radii):
        dist_matrix = cdist(centers, centers)
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-6:
                    return -1
        return 0

    def constraint_boundary(radii):
        for i in range(num_circles):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return -1
        return 0

    # Run optimization
    result = minimize(
        fun=objective,
        x0=radii,
        bounds=[(0, 0.15) for _ in range(num_circles)],
        constraints=[
            {'type': 'ineq', 'fun': constraint_overlap},
            {'type': 'ineq', 'fun': constraint_boundary}
        ],
        method='SLSQP'
    )

    centers = centers
    radii = result.x
    sum_radii = np.sum(radii)
    
    # Validate the packing
    status, msg = validate_packing(centers, radii)
    if not status:
        raise ValueError(f"Validation failed: {msg}")

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
