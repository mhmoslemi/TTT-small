import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds

def run_packing():
    # Define initial guess for centers and radii
    # We'll use a hybrid approach of placing circles in hexagonal grids and filling empty spaces
    
    # Initial center positions (we'll use a hexagonal grid pattern with some randomness)
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    
    # Hexagonal grid of circles in the square
    # Generate positions in a hexagonal grid
    max_row = 0
    max_col = 0
    for i in range(26):
        row = i // 7
        col = i % 7
        x = col / 6
        y = row / 6
        centers[i] = [x, y]

    # We will optimize the radii while preserving the center positions to maximize the sum of radii
    # We'll use a constrained optimization problem with the boundary and non-overlapping constraints
    
    # Bounds for radii (non-negative)
    radii_bounds = [(0, 1) for _ in range(26)]
    
    # Objective function: maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Negative for minimization
    
    # Constraint function: ensure circles are within the square
    def square_bounds(radii):
        return np.array([radii[i] for i in range(26)])
    
    # Constraint function: ensure circles do not overlap
    def non_overlap(radii):
        overlaps = np.zeros(26)
        for i in range(26):
            for j in range(i + 1, 26):
                dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if dist < radii[i] + radii[j] - 1e-8:
                    overlaps[i] += 1
                    overlaps[j] += 1
        return overlaps
    
    # Combine all constraints
    def constraints(radii):
        square = square_bounds(radii)
        non_overlap = non_overlap(radii)
        return np.concatenate((square, non_overlap))
    
    # Optimization problem
    result = optimize.minimize(
        objective,
        x0=radii,
        bounds=radii_bounds,
        constraints=optimize.constraints.Constraints(
            'ineq',
            fun=constraints,
            jac=None
        )
    )
    
    # Extract result
    radii_opt = result.x
    centers_opt = centers
    
    # Check if the result is feasible
    if not result.success:
        raise ValueError("Optimization failed")
    
    sum_radii = np.sum(radii_opt)
    return centers_opt, radii_opt, sum_radii
