import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii using a
    constrained optimization approach with a hexagonal packing initialization.

    Returns:
        tuple of (centers, radii, sum_radii)
    """

    # Initial hexagonal grid layout - 26 circles in a hexagonal pattern
    # We use 6 rows: row 0 has 1 circle, row 1 has 2, ..., row 5 has 6 circles
    num_rows = 6
    num_cols = [1, 2, 3, 4, 5, 6]  # Number of circles in each row
    hex_radius = 1.0 / np.sqrt(3)  # Geometric radius for hexagonal packing

    # Initial centers based on hexagonal grid
    centers = []
    for row_idx in range(num_rows):
        row = []
        for col_idx in range(num_cols[row_idx]):
            x = 0.5 + (col_idx - (num_cols[row_idx] - 1) / 2) * hex_radius
            y = row_idx * (np.sqrt(3) * hex_radius) + 0.5
            row.append([x, y])
        centers.extend(row)

    # Initial guess for radii - same as hexagonal grid
    radii = [hex_radius] * len(centers)

    # Convert to NumPy arrays
    centers = np.array(centers)
    radii = np.array(radii)

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        # Constraint: all circles must be within the square
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return False
        # Constraint: no two circles can overlap
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    def constraint_func(radii):
        return constraints(radii)

    # Optimization parameters
    bounds = Bounds([0.0] * len(radii), [1.0] * len(radii))

    result = minimize(
        objective,
        radii,
        method='SLSQP',

    if not result.success:
        raise RuntimeError(f"Optimization failed: {result.message}")

    centers = np.array(centers)

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
