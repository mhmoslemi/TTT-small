from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint

def run_packing():

    # Generating hexagonal grid with 25 circles
    num_circles = 25
    centers = []
    for i in range(num_circles):
        row = i // 5
        col = i % 5
        x = 0.5 + (col - 2.5) * 0.2

    def constraint_distance(i, j, radii, centers):
        return np.linalg.norm(centers[i] - centers[j]) - (radii[i] + radii[j])
    
    # Constraints: distance between every pair of circles >= sum of their radii
    cons = []

        radii,
        method='SLSQP',

    )
    
    # Extract results

    radii = res.x

    valid, msg = validate_packing(centers, radii)

def validate_packing(centers, radii):
    n = centers.shape[0]
    
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"
    
    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

            if dist < radii[i] + radii[j] - 1e-12:
