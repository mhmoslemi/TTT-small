from scipy.optimize import Bounds

    radii = []

    grid_y = np.linspace(0.1, 0.9, 5)

        for y in grid_y:
            centers.append([x, y])
    
    # Add an extra circle in the center
    centers.append([0.5, 0.5])
    centers = np.array(centers)
    
    # Initial guess for radii (all the same)

    # Optimization: maximize sum of radii
    def objective(r):
        return -np.sum(r)  # Minimize negative sum to maximize sum

    # Equality constraints: circle radii must be positive
    def equality_constraints(r):
        return r

    # Inequality constraints: circles must not overlap, must be within square
    def inequality_constraints(x, r):
        # x is the vector of centers, r is the vector of radii
        # Check overlapping
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dist = np.linalg.norm(x[i] - x[j])
                if dist < r[i] + r[j] - 1e-8:
                    return False
        # Check boundary constraints
        for i in range(num_circles):
            if x[i, 0] - r[i] < 0 or x[i, 0] + r[i] > 1:

            if x[i, 1] - r[i] < 0 or x[i, 1] + r[i] > 1:
