import scipy.optimize as optimize

def run_packing():

        for i in range(n_rows):
            for j in range(n_cols):
                x = j * side_length + offset_x
                y = i * (np.sqrt(3)/2) * side_length + offset_y
                centers.append((x, y))
        return np.array(centers)

    def check_constraints(centers, radii):
        n = len(centers)
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return False, f"Circle {i} out of bounds"
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return False, f"Overlap between circle {i} and circle {j}"
        return True, "Valid packing"

    # Initial guess: layer of 8 circles in hexagonal pattern
    initial_centers = hexagonal_grid(2, 4, 0.2, 0.2, 0.2)
