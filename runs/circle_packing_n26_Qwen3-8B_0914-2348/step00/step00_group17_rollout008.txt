import numpy as np
from scipy.optimize import minimize
from scipy.spatial import Voronoi, voronoi_plot_2d
from scipy.spatial.distance import pdist, squareform
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    This implementation uses a hybrid approach combining Voronoi diagram and optimization.
    """

    # Initial layout: 5 rows of 5 circles with one extra at the bottom
    # The initial centers are placed in a grid
    initial_centers = []
    for i in range(5):
        for j in range(5):
            x = j / 4.0
            y = i / 4.0
            initial_centers.append([x, y])
    
    # Add one more circle at bottom right
    initial_centers.append([0.75, 0.75])

    initial_centers = np.array(initial_centers)

    def objective(radii):
        return -np.sum(radii)  # Negative because we're minimizing

    def constraint_min_distance(radii, centers):
        n = len(centers)
        distances = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if distances[i, j] < radii[i] + radii[j] - 1e-12:
                    return -np.inf
        return 0

    # Initial guess for radii: all equal to 0.1
    initial_radii = np.full(26, 0.1)

    # Constraints: radii must be non-negative
    constraints = [
        {'type': 'ineq', 'fun': lambda radii: radii}
    ]

    # Optimization problem
    result = minimize(
        objective,
        initial_radii,
        method='SLSQP',
        constraints=constraints,
        bounds=[(0, 1)] * 26,
        options={'maxiter': 200, 'disp': False}
    )

    if not result.success:
        raise ValueError("Optimization failed")

    radii = result.x
    centers = initial_centers.copy()

    # Now perform Voronoi refinement to improve packing
    # Adjust centers based on Voronoi diagram to minimize distances
    # We will refine the positions of the circles to maximize the minimum distance
    # This is a local refinement step

    def voronoi_refinement(centers, radii, n_iterations=5):
        for _ in range(n_iterations):
            vor = Voronoi(centers)
            # Find the Voronoi vertices
            vertices = vor.vertices
            # For each site, find the nearest Voronoi vertex
            new_centers = []
            for i in range(len(centers)):
                # Find the nearest vertex to the site
                dists = np.sum((vertices - centers[i]) ** 2, axis=1)
                if np.any(dists > 1e-12):
                    nearest_vertex = vertices[np.argmin(dists)]
                    new_centers.append(nearest_vertex)
                else:
                    new_centers.append(centers[i])
            centers = np.array(new_centers)
        return centers

    centers = voronoi_refinement(centers, radii)

    # Final check for boundary constraints
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            raise ValueError(f"Circle {i} at ({x},{y}) r={r} is outside the unit square")

    return centers, radii, np.sum(radii)
