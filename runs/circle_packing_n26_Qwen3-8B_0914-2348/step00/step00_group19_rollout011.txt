import numpy as np
import cvxpy as cp

def run_packing():
    # Define the number of circles and grid dimensions
    n = 26
    rows = 5
    cols = 5

    # Initialize centers in a hexagonal grid
    centers = np.zeros((n, 2))
    for i in range(n):
        col = i % cols
        row = i // cols
        x = col + 0.5 * (row % 2)
        y = row * np.sqrt(3) / 2
        centers[i] = [x, y]

    # Convert to unit square by scaling
    scale = 1.0 / (np.max(centers[:, 0]) - np.min(centers[:, 0])) if np.max(centers[:, 0]) > np.min(centers[:, 0]) else 1.0
    centers /= scale
    centers -= np.min(centers, axis=0)

    # Define variables for optimization
    r = cp.Variable(n)
    pos = cp.Variable((n, 2))

    # Objective: maximize total radii
    obj = cp.Maximize(cp.sum(r))

    # Constraints
    constraints = [
        # Each circle's center must lie within [0, 1]^2
        cp.constraints.Inequality(0 <= pos[:, 0]),
        cp.constraints.Inequality(pos[:, 0] <= 1),
        cp.constraints.Inequality(0 <= pos[:, 1]),
        cp.constraints.Inequality(pos[:, 1] <= 1),
        # Each circle's radius must be non-negative
        r >= 0,
        # Ensure circles are not overlapping
        cp.constraints.Distinct(pos[i], pos[j]) >= r[i] + r[j] - 1e-8 for i in range(n) for j in range(i + 1, n)
    ]

    # Solve the optimization problem
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.SCS, eps=1e-6)
    except:
        raise RuntimeError("Optimization failed.")

    # Extract results
    radii = r.value
    centers = pos.value
    total_radii = np.sum(radii)

    return centers, radii, total_radii
