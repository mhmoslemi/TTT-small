import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Step 1: Initial hexagonal packing configuration
    num_circles = 26
    # Approximate number of rows in hexagon packing
    rows = 5
    cols = 5
    # Grid spacing; radius to be determined
    grid_spacing = 0.25
    # Initial centers in hexagonal grid
    centers = []
    for row in range(rows):
        for col in range(cols):
            x = col * grid_spacing
            y = row * (grid_spacing * np.sqrt(3)) / 2
            centers.append([x, y])
    # Add corner circles
    # Top-left corner
    centers.append([0.0, 1.0])
    # Top-right corner
    centers.append([1.0, 1.0])
    # Bottom-left corner
    centers.append([0.0, 0.0])
    # Bottom-right corner
    centers.append([1.0, 0.0])
    centers = np.array(centers)
    
    # Step 2: Optimization function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum (maximize sum)

    def constraints(radii):
        # Constraints: non-overlap and within unit square
        n = len(radii)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        for i in range(n):
            x, y = centers[i]
            if x - radii[i] < -1e-8 or x + radii[i] > 1 + 1e-8 or y - radii[i] < -1e-8 or y + radii[i] > 1 + 1e-8:
                return False
        return True

    # Initial guess: all circles have the same initial radius
    initial_radius = 0.05
    initial_radii = initial_radius * np.ones((num_circles,))

    # Optimization to maximize total area
    result = minimize(
        objective,
        initial_radii,
        constraints=lambda r: constraints(r),
        method='SLSQP',
        bounds=[(0, 1) for _ in range(num_circles)]
    )

    radii = result.x
    centers = np.array(centers)
    total_sum = np.sum(radii)
    
    return (centers, radii, total_sum)
