from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def initial_config(n=26):
    """Generate a hexagonal grid packing within the unit square."""
    rows = int(np.sqrt(n))
    cols = n // rows + (1 if n % rows else 0)
    
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = j + 0.5 * (i % 2)
            y = i + 0.5
            if x < 1 and y < 1:
                centers.append([x, y])
    return np.array(centers)

def objective(radii, centers):
    """Maximize the sum of radii."""
    return -np.sum(radii)

def constraints(radii, centers):

    initial_radii = np.full(n, 0.1)
