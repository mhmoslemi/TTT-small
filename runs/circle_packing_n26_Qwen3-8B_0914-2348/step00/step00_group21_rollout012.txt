import cvxpy as cp

def run_packing():

    grid_size = 0.2
    num_subgrids = int(1 / grid_size) + 1  # 5x5 grid
    subgrid_radius = grid_size / 2 - 0.01  # Slightly smaller than half of subgrid size
    max_radius = max(0.1, subgrid_radius)

    # Generate grid of centers
    centers = []
    for i in range(num_subgrids):
        for j in range(num_subgrids):
            x = i * grid_size + grid_size / 2
            y = j * grid_size + grid_size / 2
            centers.append([x, y])
    centers = np.array(centers)

    # Initial radii for all subgrid circles
    radii = np.full(len(centers), subgrid_radius)

    # Add a corner circle with max_radius
    corner_circle = np.array([0.1, 0.1])
    centers = np.vstack((centers, corner_circle))
    radii = np.append(radii, max_radius)

    # Function to compute pairwise distances
    def pairwise_distances(matrix):
        n = matrix.shape[0]
        dists = np.zeros((n, n))
        for i in range(n):
            for j in range(i + 1, n):
                dists[i, j] = np.sqrt(np.sum((matrix[i] - matrix[j]) ** 2))
                dists[j, i] = dists[i, j]
        return dists

    # Optimization variables
    radii_var = cp.Variable(len(centers))
    constraints = []
    constraints.append(radii_var >= 0)  # Non-negative radii

    # Boundary constraints
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii_var[i]
        constraints.append(x - r >= -1e-12)
        constraints.append(x + r <= 1 + 1e-12)
        constraints.append(y - r >= -1e-12)
        constraints.append(y + r <= 1 + 1e-12)

    # Pairwise distance constraints
    dists = pairwise_distances(centers)
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            constraints.append(dists[i, j] >= radii_var[i] + radii_var[j] - 1e-12)

    # Maximize sum of radii
    objective = cp.Maximize(cp.sum(radii_var))

    # Solve the QP
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.OSQP)
    except:

    # Return final values
    return centers, radii_var.value, np.sum(radii_var.value)
