import numpy as np
import cvxpy as cp

def run_packing():
    # Define a hexagonal grid in the unit square with 26 circles
    def grid_points(num_points):
        # Generate a hexagonal grid in the unit square
        indices = np.arange(num_points)
        row_idx = (indices // 7).astype(int)
        col_idx = (indices % 7).astype(int)
        x = 0.1 + (col_idx + 0.5) * (0.8 / 6)
        y = 0.1 + row_idx * (0.8 / 6) + 0.5 * np.sin(2 * np.pi * row_idx / 7)
        return np.column_stack((x, y))

    centers = grid_points(26)

    # Define variables
    # We fix the centers but allow radius optimization
    # We also allow for some repositioning of centers via a small perturbation
    num_circles = 26
    radii = cp.Variable(num_circles)

    # Define weights to prefer smaller radii near boundaries
    boundary_weights = np.zeros(num_circles)
    for i in range(num_circles):
        x, y = centers[i]
        if x < 0.1 or x > 0.9 or y < 0.1 or y > 0.9:
            boundary_weights[i] = 1000  # Penalize circles near boundaries more

    # Objective: Maximize the total radius, weighted by boundary proximity
    obj = cp.Maximize(cp.sum(radii) - cp.sum(boundary_weights * radii))

    # Constraints
    constraints = []

    # Ensure no overlaps between circles
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            dist = cp.norm(centers[i] - centers[j])
            constraints.append(dist >= radii[i] + radii[j] - 1e-12)

    # Ensure all circles are within the unit square
    for i in range(num_circles):
        x, y = centers[i]
        constraints.append(x - radii[i] >= -1e-12)
        constraints.append(x + radii[i] <= 1 + 1e-12)
        constraints.append(y - radii[i] >= -1e-12)
        constraints.append(y + radii[i] <= 1 + 1e-12)

    # Solve the problem
    problem = cp.Problem(obj, constraints)
    problem.solve(solver=cp.OSQP, eps=1e-6, verbose=False)

    # Extract the optimized results
    centers_opt = centers
    radii_opt = radii.value

    if radii_opt is None:
        raise ValueError("Optimization failed")

    total_sum = np.sum(radii_opt)
    return (centers_opt, radii_opt, total_sum)
