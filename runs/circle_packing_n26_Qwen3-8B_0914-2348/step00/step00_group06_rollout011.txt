import cvxpy as cp

def run_packing():

    #             [0,0]   [0.5,0]  [1,0] ...

    # Define initial positions

    cols = 5

    n = len(centers)
    radii = cp.Variable(n)

    # Objective: Maximize total sum of radii
    obj = cp.Maximize(cp.sum(radii))
