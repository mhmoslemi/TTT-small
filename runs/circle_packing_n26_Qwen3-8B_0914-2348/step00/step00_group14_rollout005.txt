def run_packing():

    num_circles = 26

    rows = 5
    cols = 5

            centers[i * cols + j] = [x, y]

        for i in range(num_circles):
            x, y = centers[i]

            if (x - r < -1e-12 or x + r > 1 + 1e-12 or

        # Constraint 2: non-overlapping
