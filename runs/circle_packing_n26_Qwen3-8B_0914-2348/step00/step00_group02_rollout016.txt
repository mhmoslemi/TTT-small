import scipy.optimize as opt

def run_packing():

    num_circles = 26
    centers = []

        for col in range(5):
            x = col + 0.5 * (row % 2)
            y = row + 0.5
            centers.append([x, y])
    centers = np.array(centers)
    
    # Initial radius estimate (approximate based on grid spacing)
    initial_radii = np.full(num_circles, 0.1)

    # Define the objective: maximize the sum of radii
    def objective(x):
        r = x[num_circles:]
        return -np.sum(r)  # Minimize negative sum to maximize radii

    # Bounds: (x, y) ∈ [0, 1], radii ≥ 0
    bounds = []
    for i in range(num_circles):
        bounds.extend([(0.0, 1.0), (0.0, 1.0)])  # For x, y
        bounds.extend([(0.0, 1.0)])            # For radius
    bounds = np.array(bounds).reshape(-1, 2)

    # Constraints: circle must be fully inside square
    def inside_box(x):
        r = x[num_circles:]
        for i in range(num_circles):
            x_i, y_i = x[i*2 : i*2+2]
            r_i = r[i]
            if (x_i - r_i < 0 or x_i + r_i > 1 or
                y_i - r_i < 0 or y_i + r_i > 1):
                return -1
        return 0

    # Constraints: circles must not overlap
    def non_overlapping(x):
        r = x[num_circles:]
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = x[i*2] - x[j*2]
                dy = x[i*2+1] - x[j*2+1]
                dist = np.sqrt(dx*dx + dy*dy)

                    return -1
        return 0

    # Setup optimization
    x0 = np.concatenate([centers.flatten(), initial_radii])

        x0,

            {'type': 'ineq', 'fun': non_overlapping}
        ]
