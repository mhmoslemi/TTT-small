from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():

    N = 26

    # Initialize centers on a hexagonal grid inside the unit square
    # We'll use a hexagonal packing with rows
    rows = 5
    cols = 5
    spacing = 1.0 / (rows + 1)
    centers = np.zeros((N, 2))
    offset = 0.5 * spacing
    for i in range(rows):

            x = j * spacing + offset
            y = i * spacing + offset
            if i % 2 == 1:
                x += 0.5 * spacing
            centers[i * cols + j] = [x, y]

    # Define the radii as variables to optimize
    radii = np.ones(N) * 0.05  # Initial guess

    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraint_overlap(radii):
        distances = squareform(pdist(centers))
        return distances - (radii[:, np.newaxis] + radii[np.newaxis, :]) + 1e-12

    # Objective function with bounds
    bounds = Bounds(lb=0, ub=np.inf)

    # Run the optimization
    result = minimize(
        objective,
        radii,
        method="SLSQP",
        bounds=bounds,
        constraints={"type": "ineq", "fun": constraint_overlap},

    sum_radii = np.sum(radii_opt)
