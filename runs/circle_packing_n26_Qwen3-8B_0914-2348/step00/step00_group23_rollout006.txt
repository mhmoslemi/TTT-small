import numpy as np
import cvxpy as cp

def run_packing():
    # Define the number of circles
    n = 26

    # Set up the problem: maximize the sum of radii
    # Variables
    centers = cp.Variable((n, 2))
    radii = cp.Variable(n, non_negative=True)

    # Objective function: maximize sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []

    # Center of each circle must lie within the square
    for i in range(n):
        x, y = centers[i]
        constraints.append(x >= 0)
        constraints.append(x <= 1)
        constraints.append(y >= 0)
        constraints.append(y <= 1)

    # Ensure each circle lies entirely inside the unit square
    for i in range(n):
        r = radii[i]
        x, y = centers[i]
        constraints.append(x - r >= -1e-12)
        constraints.append(x + r <= 1 + 1e-12)
        constraints.append(y - r >= -1e-12)
        constraints.append(y + r <= 1 + 1e-12)

    # Ensure circles do not overlap
    for i in range(n):
        for j in range(i + 1, n):
            x_i, y_i = centers[i]
            x_j, y_j = centers[j]
            dist = cp.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
            constraints.append(dist >= radii[i] + radii[j] - 1e-12)

    # Solve the optimization problem
    problem = cp.Problem(objective, constraints)
    problem.solve()

    # Extract solution
    centers = centers.value
    radii = radii.value
    sum_radii = np.sum(radii)

    return (centers, radii, sum_radii)
