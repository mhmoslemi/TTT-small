import numpy as np
import math
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():

    num_circles = 26
    grid_size = math.ceil(math.sqrt(num_circles))
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Grid-based initial placement
    for i in range(num_circles):
        row = i // grid_size
        col = i % grid_size
        centers[i, 0] = col / (grid_size - 1)
        centers[i, 1] = row / (grid_size - 1)

    def objective(x):
        # x has shape (num_circles * 2 + num_circles) = 26 * 3 = 78
        centers_flat = x[:num_circles * 2].reshape(num_circles, 2)
        radii_flat = x[num_circles * 2:].reshape(num_circles)

            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if (x_i - r_i < 0 or x_i + r_i > 1 or
                y_i - r_i < 0 or y_i + r_i > 1):
                # Penalize boundary violation
                return -float('inf')

                    # Penalize overlap
                    return -float('inf')

        bounds.append((0.0, 1.0))

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):

        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
