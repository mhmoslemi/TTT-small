import numpy as np
from scipy.optimize import minimize
import time

def run_packing():
    # Step 1: Initial grid placement
    n = 26
    rows = 5
    cols = 6
    
    # Grid coordinates
    x_coords = np.linspace(0, 1, cols, endpoint=False)
    y_coords = np.linspace(0, 1, rows, endpoint=False)
    
    # Initial grid centers
    centers = np.empty((n, 2))
    idx = 0
    for y in y_coords:
        for x in x_coords:
            centers[idx] = [x, y]
            idx += 1
    
    # Step 2: Initial radius calculation based on distance to boundaries
    radii = np.zeros(n)
    for i in range(n):
        x, y = centers[i]
        distances = [x, 1 - x, y, 1 - y]
        radii[i] = min(distances)
    
    # Step 3: Optimize radii to maximize sum while maintaining non-overlap and boundary constraints
    def objective(radii_vec):
        return -np.sum(radii_vec)  # Minimize negative sum to maximize

        x1, y1 = centers[i]
        x2, y2 = centers[j]
        dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
        return dist - (radii_vec[i] + radii_vec[j])  # >= 0
    
    def constraint_boundary(i, radii_vec):
        x, y = centers[i]
        return min(x - radii_vec[i], 1 - x - radii_vec[i], y - radii_vec[i], 1 - y - radii_vec[i])  # >= 0

        objective,
        radii,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12

            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
