import cvxpy as cp

def run_packing():

    radii = np.zeros(26)

    rows = 5
    cols = 5
    h = 1.0 / (rows - 1)
    w = 1.0 / (cols - 1)

        for j in range(cols):
            x = i * w
            y = j * h
            centers[i * cols + j] = [x, y]
            radii[i * cols + j] = 0.05

    radii[-1] = 0.05

    # We will define variables for centers and radii
    # We will use a sparse relaxation approach: maximize sum of radii, with constraints
    # that circles are not overlapping and are within [0,1]^2

    x_centers = cp.Variable((26, 2))
    x_radii = cp.Variable(26)

    # Objective: Maximize sum of radii
    obj = cp.Maximize(cp.sum(x_radii))

    # Constraints
    constraints = []

    # Each circle must be within the unit square
    for i in range(26):
        x, y = x_centers[i]
        r = x_radii[i]

            x >= r,
            x <= 1 - r,
            y >= r,
            y <= 1 - r
        ]

            constraints += [d >= x_radii[i] + x_radii[j]]

    # Solve the problem
    problem = cp.Problem(obj, constraints)
    problem.solve(solver=cp.SCS, eps=1e-4)

    # Convert to numpy arrays
    centers = np.array(x_centers.value)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
