import scipy.optimize as opt

def run_packing():

            grid_centers.append([x, y])

    corner_circles = [

    n = len(centers)

    initial_radii = np.full(n, 0.1)

    def objective(radii):

    def constraint_func(radii):

        for i in range(n):

                dx = centers[i, 0] - centers[j, 0]
