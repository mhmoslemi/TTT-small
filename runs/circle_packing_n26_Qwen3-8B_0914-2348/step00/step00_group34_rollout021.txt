```python
import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.spatial.distance import pdist
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint

def run_packing():
    # Initial arrangement: 5 rows of 5 circles (excluding one)
    # with some circles placed at corners for better packing
    # Define initial centers
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    
    # First row: 5 circles near top
    for i in range(5):
        centers[i] = [i / 4.0, 1 - 0.1]
        radii[i] = 0.1
    
    # Second row: 5 circles near middle
    for i in range(5, 10):
        centers[i] = [i - 5 - 0.25, 0.5]
        radii[i] = 0.1
    
    # Third row: 5 circles near middle
    for i in range(10, 15):
        centers[i] = [i - 10 - 0.25, 0.5]
        radii[i] = 0.1
    
    # Fourth row: 5 circles near bottom
    for i in range(15, 20):
        centers[i] = [i - 15 - 0.25, 0.1]
        radii[i] = 0.1
    
    # Fifth row: 5 circles near bottom
    for i in range(20, 25):
        centers[i] = [i - 20 - 0.25, 0.1]
        radii[i] = 0.1
    
    # Add one circle in the corner
    centers[25] = [0.05, 0.05]
    radii[25] = 0.05

    def objective(r):
        return -sum(r)  # Minimize negative sum to maximize sum

    def constraint1(r):
        # All radii non-negative
        return r >= 0

    def constraint2(r):
        # Boundary constraints
        for i in range(26):
            x, y = centers[i]
            r_i = r[i]
            if x - r_i < -1e-12 or x + r_i > 1 + 1e-12 or y - r_i < -1e-12 or y + r_i > 1 + 1e-12:
                return -np.inf
        return 0

    def constraint3(r):
        # Circle-circle non-overlap
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < r[i] + r[j] - 1e-12:
                    return -np.inf
        return 0

    # Set up optimization problem
    bounds = Bounds(0, np.inf)
    constraints = [
        LinearConstraint(np.eye(26), lb=0, ub=np.inf),
        {'type': 'ineq', 'fun': constraint2},
        {'type': 'ineq', 'fun': constraint3}
    ]

    # Initial guess
    initial_guess = radii.copy()

    # Run optimization
    result = minimize(
        objective,
        initial_guess,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP'
    )

    radii = result.x
    sum_radii = -result.fun

    return centers, radii, sum_radii
```
