import scipy.optimize as optimize

def run_packing():

    def hexagonal_grid(n):

        cols = int(np.ceil(n / rows))
        centers = []
        for i in range(rows):
            for j in range(cols):
                x = j / (cols - 1) if cols > 1 else 0.5
                y = i / (rows - 1) if rows > 1 else 0.5
                centers.append([x, y])
        return np.array(centers)

    def distance(i, j):
        return np.linalg.norm(centers[i] - centers[j])

    # Objective: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum

    # Constraints
    constraints = []

    # Ensure circles are within the unit square
    for i in range(n):
