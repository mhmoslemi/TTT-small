import scipy.optimize as opt

def run_packing():

    num_circles = 26
    rows = 5
    cols = 5

    radii = np.zeros(num_circles)

    for i in range(rows):

            idx = i * cols + j

    def objective(radii, centers):
        return -np.sum(radii)
