    horizontal_spacing = 0.3  # Horizontal spacing between circles

    # Create initial center positions
    centers = []
    for layer in range(num_layers):
        for i in range(circles_per_layer[layer]):
            x = i * horizontal_spacing
            y = layer * layer_height
            centers.append([x, y])

    centers = np.array(centers)

    # Initial radius for all circles
    initial_radius = 0.1
    radii = initial_radius * np.ones(len(centers))

    # Optimization parameters
    bounds = Bounds(
        lb=np.array([[0.0, 0.0] for _ in range(len(centers))]),
        ub=np.array([[1.0, 1.0] for _ in range(len(centers))])
    )

    # Define objective function: maximize the sum of radii
    def objective(x):
        return -np.sum(x)  # Minimize negative sum to maximize

    # Define constraints
    def constraint_overlap(x, r):
        # Ensure circle i and j are not overlapping
        for i in range(len(x)):
            for j in range(i+1, len(x)):
                dx = x[i, 0] - x[j, 0]
                dy = x[i, 1] - x[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < r[i] + r[j] - 1e-8:
                    return 1  # Constraint violated
        return 0

    def constraint_boundary(x, r):

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
