import numpy as np
import scipy.optimize as optimize

def run_packing():

    # 5 rows with 5, 5, 5, 5, 6 circles
    rows = 5
    cols = [5, 5, 5, 5, 6]
    x = []
    y = []
    for i in range(rows):
        for j in range(cols[i]):
            x.append(j + i / 2)
            y.append(i + 0.5)
    
    centers = np.array([x, y]).T
    initial_radii = np.full(26, 0.05)

    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum, i.e., maximize sum

    def constraint1(radii):

        for i in range(26):

            if x - radii[i] < -1e-8 or x + radii[i] > 1 + 1e-8 or y - radii[i] < -1e-8 or y + radii[i] > 1 + 1e-8:
                return -1
        return 0

    def constraint2(radii):
        # Circle-circle constraints (non-overlapping)
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0

    # Optimization bounds: radii must be non-negative
    bounds = [(0, 1) for _ in range(26)]

    # Run optimization
    result = optimize.minimize(
        objective,
        initial_radii,
        bounds=bounds,
        constraints=[

            {'type': 'ineq', 'fun': constraint2}
        ],
        method='SLSQP'
    )
