import scipy.optimize as optimize

def run_packing():

    base_radius = 0.08  # Approximate initial radius used to seed the pack

    centers = []

            # Adjust to fit in the unit square
            x = x / (5 + 0.5)
            y = y / (4 * np.sqrt(3) / 2)
            centers.append([x, y])

    centers = np.array(centers)

    radii = np.full(centers.shape[0], base_radius)
