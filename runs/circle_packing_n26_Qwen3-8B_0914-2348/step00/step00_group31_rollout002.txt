def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial grid: 5 rows x 5 columns
    num_circles = 26
    rows, cols = 5, 5
    # Base grid configuration
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)
    for i in range(rows):
        for j in range(cols):
            index = i * cols + j
            if index >= num_circles:
                break
            x = j / (cols - 1)

            a = 0.1  # Initial guess for radius
            centers[index, 0] = x

            radii[index] = a

    def objective(radii_flat):

        # Radius cannot exceed the distance to the boundary
        return radii_flat - (1 - np.abs(centers[:, 0] - 0.5) - np.abs(centers[:, 1] - 0.5))

        # Distances between circles must be >= r_i + r_j
        r = radii_flat
        distances = np.sqrt(np.sum((centers[:, None] - centers[None, :]) ** 2, axis=2))
        return distances - (r[:, None] + r[None, :])

    initial_guess = radii_flat.copy()

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
