import numpy as np
import cvxpy as cp

def run_packing():
    # Define initial grid pattern with 3 rows and 9 columns (3*9=27, one circle is removed)
    num_circles = 26
    num_rows = 3
    num_cols = 9
    
    # Initial grid positions
    centers = []
    for i in range(num_rows):
        for j in range(num_cols):
            x = j / (num_cols - 1) if num_cols > 1 else 0.5
            y = i / (num_rows - 1) if num_rows > 1 else 0.5
            centers.append([x, y])
    centers = np.array(centers)
    
    # Define optimization variables
    r = cp.Variable(num_circles, non_negative=True)
    
    # Objective: Maximize sum of radii
    objective = cp.Maximize(cp.sum(r))
    
    # Constraints: Each circle lies within the square
    constraints = []
    for i in range(num_circles):
        x_i, y_i = centers[i]
        constraints.append(x_i - r[i] >= -1e-8)
        constraints.append(x_i + r[i] <= 1 + 1e-8)
        constraints.append(y_i - r[i] >= -1e-8)
        constraints.append(y_i + r[i] <= 1 + 1e-8)
    
    # Constraints: Ensure non-overlapping between all circle pairs
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            constraints.append(dist >= r[i] + r[j] - 1e-8)
    
    # Create and solve the optimization problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.OSQP, eps=1e-8, verbose=False)
    
    # Validate the solution
    if not prob.status == 'optimal':
        raise ValueError("Optimization failed to find an optimal solution.")
    
    # Extract the results
    centers_final = centers
    radii_final = r.value
    sum_radii = np.sum(radii_final)
    
    return (centers_final, radii_final, sum_radii)
