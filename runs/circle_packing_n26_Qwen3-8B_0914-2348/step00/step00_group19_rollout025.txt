import cvxpy as cp

def run_packing():

        return -np.sum(x)  # Minimize negative of sum to maximize it

    # 1. Each circle must be within [0, 1] x [0, 1]
    # 2. Each circle must not overlap with others
    # 3. Radii >= 0

    # Define variables
    radii = cp.Variable(num_circles, nonneg=True)
    centers = cp.Variable((num_circles, 2))

    # Objective: Maximize sum of radii
    obj = cp.Maximize(cp.sum(radii))

    # Constraints:
    constraints = []

    # Each circle must be within the unit square
    for i in range(num_circles):
        x, y = centers[i]
        r = radii[i]
        constraints += [
            x - r >= -1e-8,
            x + r <= 1 + 1e-8,
            y - r >= -1e-8,
            y + r <= 1 + 1e8,
        ]

            dist = cp.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)

    # Solve the optimization
    problem = cp.Problem(obj, constraints)
    problem.solve(solver='SLSQP')  # Sequential Least Squares Programming
