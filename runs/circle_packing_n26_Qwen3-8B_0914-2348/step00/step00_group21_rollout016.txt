import math

def place_hexagonal(n_circles):

    Place n_circles in a hexagonal pattern starting with the bottom row centered.
    Returns (centers, radii)

    centers = []
    radii = []

    row_height = 2 * r * math.sqrt(3) / 2
    max_rows = int((1 - 2 * r) / row_height) + 1

    circles_per_row = [int(math.ceil(n_circles / max_rows))] * max_rows
    while sum(circles_per_row) > n_circles:
        circles_per_row[-1] -= 1
    
    # Calculate the x positions for each row
    x_positions = [np.linspace(0.1, 0.9, cols) for cols in circles_per_row]

    row_y = 0.1

        y = row_y + i * row_height
        for j in range(cols):
            x = x_positions[i][j]
            centers.append(np.array([x, y]))
            radii.append(r)

        dist_to_edge = min(x, 1 - x, y, 1 - y)
        adjusted_radii.append(min(r, dist_to_edge))
    
    return np.array(centers), np.array(adjusted_radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
