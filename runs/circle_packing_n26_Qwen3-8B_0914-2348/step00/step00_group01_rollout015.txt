import cvxpy as cp

def run_packing():

    n = 26

    # 5x5 grid with some extra space

    # Introduce variables for radii
    radii = cp.Variable(n)

    # Objective: Maximize sum of radii
    objective = cp.Maximize(cp.sum(radii))

        constraints.append(radii[i] <= x - 1e-6)

        constraints.append(radii[i] <= 1 - y - 1e-6)

    # Non-overlapping constraint: distance between centers >= sum of radii

    problem = cp.Problem(objective, constraints)
