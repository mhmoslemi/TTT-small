from scipy.optimize import minimize

def run_packing():

    ])

    # Add a central circle

    n = len(centers)

    # Initial guess for radii: same radius for all circles
    initial_radii = np.full(n, 0.1)

    # Optimization objective: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Optimization constraints
    def constraint_distances(radii):

        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < radii[i] + radii[j] - 1e-8:

    cons = (

        {'type': 'ineq', 'fun': constraint_boundary}
