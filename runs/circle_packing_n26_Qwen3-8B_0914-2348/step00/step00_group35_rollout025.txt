import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
from itertools import combinations

def run_packing():

    # Radius will be calculated based on the spacing and boundary constraints

        for i in range(rows):
            for j in range(cols):
                x = j * (1 + r) + r / 2
                y = i * (sqrt(3) * r) + r / 2
                # Ensure x and y stay within the square
                x = np.clip(x, r, 1 - r)
                y = np.clip(y, r, 1 - r)
                centers.append([x, y])
        return np.array(centers)

    def get_distances(c1, c2):
        return np.sqrt((c1[0] - c2[0]) ** 2 + (c1[1] - c2[1]) ** 2)

    # Step 2: Heuristically estimate initial radius for top layer
    # Use radius formula for hexagonal packing in square: r = 0.2

    # Create base lower layer
    lower_centers = create_hex_grid(r1)
    lower_radii = np.full(len(lower_centers), r1)

    # Step 3: Use optimization to maximize total area with constraints
    def objective(radii, centers):
        return -np.sum(radii)  # minimize negative sum, i.e., maximize sum

    def constraint_overlap(centers, radii):
        constraints = []
        for i, j in combinations(range(len(centers)), 2):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            constraints.append(dist - (radii[i] + radii[j]) >= -1e-8)
        return constraints

    def constraint_boundary(centers, radii):
        constraints = []
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            constraints.append(x - r >= -1e-8)
            constraints.append(x + r <= 1 + 1e-8)
            constraints.append(y - r >= -1e-8)
            constraints.append(y + r <= 1 + 1e-8)
        return constraints

    def optimize_packing(initial_centers, initial_radii):
        x0 = initial_radii.copy()
        bounds = [(0, 1) for _ in range(len(x0))]

        constraints = constraint_overlap(initial_centers, x0)
        constraints += constraint_boundary(initial_centers, x0)
        constraints = [lambda x, i=i: constraints[i](x) for i in range(len(constraints))]

        result = minimize(
            objective, x0, bounds=bounds, constraints=constraints,
            method='SLSQP', tol=1e-6, options={'maxiter': 1000}
        )

        return result.x

    initial_radii = lower_radii
