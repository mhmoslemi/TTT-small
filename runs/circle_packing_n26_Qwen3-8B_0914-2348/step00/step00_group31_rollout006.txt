import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds

def run_packing():
    # Define the number of circles
    n = 26
    # Initial grid-based placement (5x5 grid with 1 extra circle)
    # Start with a 5x5 grid (25 circles)
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Square grid with some extra circles
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col / 4.0
        y = row / 4.0
        centers[i, 0] = x
        centers[i, 1] = y
    
    # First guess for radii: same for all
    radii = np.full(n, 1.0 / 4.0)  # Initial guess
    
    # Define objective function to maximize sum of radii
    def objective(x):
        # x contains all radii (n variables)
        return -np.sum(x)  # Minimize negative sum (i.e., maximize sum)
    
    # Define bounds for each radius: non-negative
    bounds = Bounds(0, np.inf)
    
    # Define constraints for non-overlapping and boundary conditions
    def constraint(x):
        r = x
        for i in range(n):
            xi, yi = centers[i]
            ri = r[i]
            # Check boundary constraints: circle must be within [0, 1] × [0, 1]
            if (xi - ri < -1e-12 or xi + ri > 1 + 1e-12 or
                yi - ri < -1e-12 or yi + ri > 1 + 1e-12):
                return False
        # Check non-overlapping between every pair of circles
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < r[i] + r[j] - 1e-12:
                    return False
        return True
    
    # Define the optimization call
    result = optimize.minimize(
        fun=objective,
        x0=radii,
        bounds=bounds,
        constraints={
            'type': 'constraint',
            'fun': constraint,
            'method': 'SLSQP'
        }
    )
    
    # Extract the optimized radii
    radii = result.x
    # Update centers to use the same configuration as before
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col / 4.0
        y = row / 4.0
        centers[i, 0] = x
        centers[i, 1] = y
    
    # Return centers, radii, and sum of radii
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
