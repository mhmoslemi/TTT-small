import numpy as np
import scipy.optimize as opt
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)  # Initial random positions
    radii = np.ones(n) * 0.05  # Initial guess for radii

    def objective(radii):
        return -np.sum(radii)  # Maximize sum of radii

    # Boundary constraints: each circle's center must be within [0,1]^2
    def boundary_constraint(radii):
        return np.array([(centers[i, 0] - radii[i] >= 0) &
                        (centers[i, 1] - radii[i] >= 0) &
                        (centers[i, 0] + radii[i] <= 1) &
                        (centers[i, 1] + radii[i] <= 1)
                        for i in range(n)])
    
    # Non-overlapping constraint: distance between centers >= sum of radii
    def non_overlap_constraint(radii):
        dist = squareform(pdist(centers))
        return np.array([dist[i, j] - radii[i] - radii[j] for i in range(n) for j in range(i + 1, n)])
    
    # Optimization bounds
    bounds = [(0, 1.0) for _ in range(n)]

    # Set up constraints
    constraints = []
    constraints.append({'type': 'ineq', 'fun': boundary_constraint})
    constraints.append({'type': 'ineq', 'fun': lambda r: non_overlap_constraint(r)})

    # Optimization method
    result = opt.minimize(

        radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'disp': False, 'ftol': 1e-8, 'maxiter': 500}
