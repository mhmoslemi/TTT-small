import numpy as np
from scipy.optimize import minimize
import matplotlib.pyplot as plt

def run_packing():
    n = 26

    # Define the hexagonal packing pattern, starting with a 5x5 grid with some shifts
    def hexagonal_grid(n):
        grid = []
        for i in range(n):
            for j in range(n):
                x = j + (i % 2) * 0.5
                y = i * np.sqrt(3) / 2
                grid.append([x, y])
        # Add extra circles to reach size 26
        if n * n > 26:
            grid = grid[:26]
        return np.array(grid)

    centers = hexagonal_grid(26)
    radii = np.full(n, 0.05)  # Initial guess

    def objective(x):
        return -np.sum(x)

    # Define non-overlap and boundary constraints
    def constraint_overlap(x, i, j):
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        dist = np.sqrt(dx**2 + dy**2)
        return x[i] + x[j] - dist

    def constraint_boundary(x, i):
        x_center, y_center = centers[i]
        r = x[i]
        if x_center - r < 0:
            return x_center - r
        if x_center + r > 1:
            return x_center + r - 1
        if y_center - r < 0:
            return y_center - r
        if y_center + r > 1:
            return y_center + r - 1
        return 0

    # Define bounds for radii: non-negative
    bounds = [(0, 1) for _ in range(n)]

    # Use different constraints for all circle pairs (i < j)
    constraints = []

            constraints.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: constraint_overlap(x, i, j)})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: constraint_boundary(x, i)})

    result = minimize(
        fun=objective,
        x0=radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
