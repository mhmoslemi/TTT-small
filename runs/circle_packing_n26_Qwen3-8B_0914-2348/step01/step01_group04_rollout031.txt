import numpy as np
import scipy.optimize as opt
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)  # Random initial positions
    radii = np.random.rand(n) * 0.1  # Initial random radii

    def objective(radii, centers):
        return -np.sum(radii)  # Maximize sum of radii

    def non_overlap_constraint(radii, centers):
        dist = squareform(pdist(centers))
        return np.array([dist[i, j] - radii[i] - radii[j] for i in range(n) for j in range(i + 1, n)])

    def boundary_constraint(radii, centers):
        return np.array([(centers[i, 0] - radii[i] >= 0) &
                        (centers[i, 1] - radii[i] >= 0) &
                        (centers[i, 0] + radii[i] <= 1) &
                        (centers[i, 1] + radii[i] <= 1)
                        for i in range(n)])

    # Define bounds and constraints
    bounds = [(0, 1.0) for _ in range(n)]

    constraints = []
    constraints.append({'type': 'ineq', 'fun': lambda r: non_overlap_constraint(r, centers)})
    constraints.append({'type': 'ineq', 'fun': lambda r: boundary_constraint(r, centers)})

    # Run optimization
    result = opt.minimize(
        lambda r: objective(r, centers),
        radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'ftol': 1e-8, 'maxiter': 500}
    )

    # Extract results
    centers_final = centers.copy()
    radii_final = result.x
    sum_radii = np.sum(radii_final)

    return (centers_final, radii_final, sum_radii)
