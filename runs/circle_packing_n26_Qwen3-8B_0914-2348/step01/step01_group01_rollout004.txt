import cvxpy as cp

def run_packing():

    # Create a hexagonal grid with two rows of large circles and three rows of smaller ones
    # This is more compact than a regular hexagonal grid

    centers = []
    radii = []

    for i in range(13):
        x = i / 12
        y = 0.15
        centers.append([x, y])

        x = i / 12
        y = 0.5
        centers.append([x, y])

    centers = np.array(centers)

    # Objective: maximize the sum of radii
    radii = cp.Variable(n)
    objective = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []

    for i in range(n):
        x, y = centers[i]
        constraints.append(x - radii[i] >= 0)
        constraints.append(x + radii[i] <= 1)
        constraints.append(y - radii[i] >= 0)

    # Non-overlapping constraints: distance between centers >= sum of radii

    prob = cp.Problem(objective, constraints)
