import numpy as np
import cvxpy as cp

def run_packing():
    n = 26
    # Initial placement: remove symmetry and allow for more dynamic placement
    centers = np.random.rand(n, 2)
    radii = cp.Variable(n)

    # Constraints
    constraints = []

    # Boundary constraints: each circle's center must be within [0,1] and radii must be nonnegative
    for i in range(n):
        # Center x must be within [0, 1] and radius must be nonnegative
        constraints.append(centers[i, 0] <= 1 - radii[i])
        constraints.append(centers[i, 0] >= radii[i])
        constraints.append(centers[i, 1] <= 1 - radii[i])
        constraints.append(centers[i, 1] >= radii[i])

    for i in range(n):

            dist = cp.sqrt((centers[i, 0] - centers[j, 0]) ** 2 + (centers[i, 1] - centers[j, 1]) ** 2)
            constraints.append(dist >= radii[i] + radii[j])

    # Objective: maximize sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Solve the optimization problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS, eps=1e-8, max_iter=1000)
