import scipy.optimize as opt
from scipy.spatial.distance import pdist, squareform

def run_packing():

    radii = np.ones(n) * 0.05  # Initial guess for radii

    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (row % 2)
        y = row * np.sqrt(3) / 2
        centers[i] = [x, y]

    # Scale to unit square
    scaling = 1.0 / np.max(centers, axis=0)
    centers = centers * scaling
    radii = radii * scaling

        return -np.sum(radii)  # Minimize negative of sum to maximize sum

    constraints = []

                        (centers[i, 0] + radii[i] <= 1) &
                        (centers[i, 1] + radii[i] <= 1)
                        for i in range(n)])
    # Add boundary constraints
    for i in range(n):
        def constraint_i(r):
            return centers[i, 0] - r[i]
        constraints.append({'type': 'ineq', 'fun': constraint_i})
