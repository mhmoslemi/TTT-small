    # We'll use a spiral-like layout with circles distributed along a spiral path
    # This ensures they are placed closer to the square boundaries and corners
    theta = np.linspace(0, 2 * np.pi, n, endpoint=False)

    centers[:, 1] = r * np.sin(theta) + 0.5
