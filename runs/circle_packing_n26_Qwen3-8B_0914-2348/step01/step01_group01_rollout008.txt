import numpy as np
import cvxpy as cp

def run_packing():
    n = 26

    # Generate initial positions using a randomized grid
    centers = np.random.rand(n, 2)
    radii = cp.Variable(n)

    # Define the objective function: maximize sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []

    # Each circle must be entirely within the unit square
    for i in range(n):
        # x - r >= 0
        constraints.append(centers[i, 0] - radii[i] >= 0)
        # y - r >= 0
        constraints.append(centers[i, 1] - radii[i] >= 0)
        # x + r <= 1
        constraints.append(centers[i, 0] + radii[i] <= 1)
        # y + r <= 1
        constraints.append(centers[i, 1] + radii[i] <= 1)

    # Non-overlap: distance between centers >= sum of radii
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append(cp.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2) >= radii[i] + radii[j])

    # Solve the optimization problem
    problem = cp.Problem(objective, constraints)
    problem.solve(solver='SCS')

    # Extract results
    centers_final = centers.value
    radii_final = radii.value
    sum_radii = np.sum(radii_final)

    return (centers_final, radii_final, sum_radii)
