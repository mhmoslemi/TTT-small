import numpy as np
import scipy.optimize as opt
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.05  # Initial guess for radii

    # Generate a staggered square packing
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (row % 2)
        y = row
        centers[i] = [x, y]

    # Scale to fit in the unit square
    max_x = np.max(centers[:, 0])
    max_y = np.max(centers[:, 1])
    centers /= np.array([max_x, max_y])

    def objective(radii):
        return -np.sum(radii)

    # Boundary constraint functions
    def boundary_left(radii):
        return centers[:, 0] - radii

    def boundary_bottom(radii):
        return centers[:, 1] - radii

    def boundary_right(radii):
        return 1 - (centers[:, 0] + radii)

    def boundary_top(radii):
        return 1 - (centers[:, 1] + radii)

    # Non-overlapping constraint function
    def non_overlap(radii):
        dist = squareform(pdist(centers))
        return np.array([dist[i, j] - radii[i] - radii[j] for i in range(n) for j in range(i + 1, n)])

    # Define optimization bounds
    bounds = [(0, 1.0) for _ in range(n)]

        {'type': 'ineq', 'fun': non_overlap}
    ]

    # Run optimization
    result = opt.minimize(

        bounds=bounds,
        constraints=constraints,
        method='SLSQP'
    )

    # Extract final results
    centers_final = centers.copy()
    radii_final = result.x
    sum_radii = np.sum(radii_final)
