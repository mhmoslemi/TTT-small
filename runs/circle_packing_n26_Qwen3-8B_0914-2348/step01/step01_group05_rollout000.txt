import numpy as np
import scipy.optimize as opt
from scipy.spatial.distance import pdist, squareform

def run_packing():

    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.05  # Initial guess

    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (row % 2)
        y = row * np.sqrt(3) / 2
        centers[i] = [x, y]

    # Normalize center positions to [0, 1]^2
    centers /= np.max(centers, axis=0)

    def objective(radii):
        return -np.sum(radii)  # Maximize sum of radii

    # Constraint: each circle must be inside the unit square
    def boundary_constraint(radii):
        return np.array([
            centers[i, 0] - radii[i] >= 0,
            centers[i, 1] - radii[i] >= 0,
            1 - (centers[i, 0] + radii[i]) >= 0,
            1 - (centers[i, 1] + radii[i]) >= 0

        ])

    # Constraint: circles must not overlap
    def non_overlap_constraint(radii):
        dist = squareform(pdist(centers))
        return np.array([
            dist[i, j] - radii[i] - radii[j]
            for i in range(n)
            for j in range(i + 1, n)

    bounds = [(0, 1.0) for _ in range(n)]

    # Optimization with SLSQP
