import numpy as np
import scipy.optimize as opt
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Step 1: Use an improved initial layout (not a grid-based hexagonal)
    # Generate improved layout: staggered rows with circles placed toward edges
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        if row % 2 == 0:
            x = col + 0.5
        else:
            x = col + 0.5 - 0.25
        y = row * np.sqrt(3) / 2
        centers[i] = [x, y]

    # Step 2: Scale the layout to the unit square
    # We will allow some circles to touch the boundary later
    # But first, scale to fit inside
    max_x = np.max(centers[:, 0])
    min_x = np.min(centers[:, 0])
    max_y = np.max(centers[:, 1])
    min_y = np.min(centers[:, 1])
    scaling = np.min([1.0 / (max_x - min_x), 1.0 / (max_y - min_y)])
    centers *= scaling

    # Step 3: Set initial radii
    radii = np.ones(n) * 0.05

    # Step 4: Define the objective (maximize the sum of radii)
    def objective(radii):
        return -np.sum(radii)  # Minimize negative of sum to maximize sum

    # Step 5: Define constraints
    # Constraint 1: Each circle's center must be in [0,1]^2
    # Constraint 2: Distance between each pair of centers is at least the sum of their radii

    # Step 6: Bounds for radii (non-negative)
    bounds = [(0, 1.0) for _ in range(n)]

    # Custom constraints
    def constraint_1(radii):
        return np.array([centers[i, 0] - radii[i] for i in range(n)])

    def constraint_2(radii):
        return np.array([centers[i, 1] - radii[i] for i in range(n)])

    def constraint_3(radii):
        return np.array([1.0 - (centers[i, 0] + radii[i]) for i in range(n)])

    def constraint_4(radii):
        return np.array([1.0 - (centers[i, 1] + radii[i]) for i in range(n)])

    # Constraint for non-overlapping
    def non_overlap_constraint(radii):
        dist = squareform(pdist(centers))
        return np.array([dist[i, j] - radii[i] - radii[j] for i in range(n) for j in range(i + 1, n)])

    # Solve using SLSQP
    constraints = [

    ]

    result = opt.minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'iprint': 0, 'ftol': 1e-8, 'maxiter': 1000}
    )
