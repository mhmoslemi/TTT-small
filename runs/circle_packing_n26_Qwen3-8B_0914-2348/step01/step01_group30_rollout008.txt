import numpy as np
from scipy.optimize import minimize
import warnings

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)  # Random initial positions for better exploration
    radii = np.random.uniform(0, 0.2, n)  # Initial radii are small to allow optimization

    # Objective: maximize the sum of the radii
    def objective(radii):
        return -np.sum(radii)  # Negative because scipy minimizes

    # Constraints:
    # 1. All circles must be inside the unit square
    # 2. No two circles can overlap
    # 3. All radii must be non-negative

    # Constraint function for boundaries
    def boundary_constraint(radii):
        violations = []
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                violations.append(1)
            else:
                violations.append(0)
        return np.sum(violations)

    # Constraint function for non-overlapping circles
    def non_overlap_constraints(radii):
        violations = []
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    violations.append(1)
                else:
                    violations.append(0)
        return np.sum(violations)

    # Combine objective and constraints
    def combined_objective(radii):
        return objective(radii) + boundary_constraint(radii) + non_overlap_constraints(radii)

    # Bounds for radii: each radius must be non-negative
    bounds = [(0, 1) for _ in range(n)]

    # Use L-BFGS-B to find the optimal radii
    result = minimize(
        fun=combined_objective,
        x0=radii,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-8},
    )

    # Extract final centers and radii
    radii = result.x
    centers = centers

    # Validate the packing
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise ValueError(msg)

    # Return centers, radii, and sum of radii
    return centers, radii, np.sum(radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
