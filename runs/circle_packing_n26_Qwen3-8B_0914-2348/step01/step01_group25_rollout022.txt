from scipy.optimize import minimize

    centers = []

            if i % 2 == 0:
                x = j * 0.3
                y = i * 0.3
            else:
                x = j * 0.3 + 0.15
                y = i * 0.3
            centers.append([x, y])

    # Add one more circle at the bottom right
    centers.append([0.8, 0.8])

    def objective(x):

    def constraint_boundary(i, x):

        x_center, y_center = centers[i]
        if x_center - r < -1e-8 or x_center + r > 1 + 1e-8:
            return x_center + r - 1 - 1e-8
        if y_center - r < -1e-8 or y_center + r > 1 + 1e-8:
            return y_center + r - 1 - 1e-8
        return 0

    # Constraint: Circles do not overlap
    def constraint_overlap(i, j, x):

        r_j = x[j]
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        dist = np.sqrt(dx**2 + dy**2)
        if dist < r_i + r_j - 1e-8:
            return r_i + r_j - dist - 1e-8
        return 0

    def constraint_non_negative(i, x):
        r = x[i]
        return -r

    # Build constraints
    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: constraint_boundary(i, x)})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: constraint_non_negative(i, x)})

    result = minimize(

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
