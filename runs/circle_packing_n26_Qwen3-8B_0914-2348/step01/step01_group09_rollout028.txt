import scipy.optimize as opt
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Generate a dense square grid of 26 points in [0,1]^2
    centers = np.zeros((n, 2))

        row = i // 5
        col = i % 5
        centers[i] = [col / 4, row / 4]

    radii = np.ones(n) * 0.05

        return -np.sum(radii)  # Minimize negative of sum to maximize sum

    constraints.append({'type': 'ineq', 'fun': lambda r: np.array([centers[i, 1] - r[i] for i in range(n)])})
