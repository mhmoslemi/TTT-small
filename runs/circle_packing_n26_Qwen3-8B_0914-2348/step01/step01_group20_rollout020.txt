        col = (i - 8) % 5
        x = 0.2 + col * 0.2

        centers[i] = [x, y]

    # Initial radii based on proximity to boundaries

            radii[i] = 0.1

    def constraint_top(i):
        return lambda r: 1.0 - centers[i, 1] - r

    result = minimize(objective, x0, bounds=bounds, constraints=constraints)
