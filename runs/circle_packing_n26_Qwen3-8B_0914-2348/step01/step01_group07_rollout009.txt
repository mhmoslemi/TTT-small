        0.03, 0.03, 0.03, 0.03, 0.03, 0.03
    ])

    min_x, max_x = np.min(centers[:, 0]), np.max(centers[:, 0])
    min_y, max_y = np.min(centers[:, 1]), np.max(centers[:, 1])
    scale = np.array([1.0 / (max_x - min_x), 1.0 / (max_y - min_y)])
    centers = (centers - np.array([min_x, min_y])) * scale

    def objective(radii):
        return -np.sum(radii)  # Maximize sum by minimizing negative sum

    # Constraints
    constraints = []

    # Boundary constraint: Each circle must be within [0, 1]x[0, 1]
    for i in range(26):
        constraints.append({

            'fun': lambda r, i=i: centers[i, 0] - r[i]
        })

    result = opt.minimize(

        bounds=bounds,
