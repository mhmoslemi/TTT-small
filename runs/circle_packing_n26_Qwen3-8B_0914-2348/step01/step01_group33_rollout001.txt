from scipy.optimize import minimize
import matplotlib.pyplot as plt

def run_packing():
    n = 26
    num_columns = 6
    num_rows = 5
    grid_spacing = 0.2
    total_width = 1.0

    # Initialize centers using a hexagonal grid
    centers = []
    for i in range(num_rows):
        for j in range(num_columns):
            x = j * grid_spacing
            y = i * grid_spacing + (i % 2) * (grid_spacing / 2)
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers.append([x, y])

    # Add a few additional circles near the boundary to fill gaps
    additional_centers = []
    # Add a circle near the top-right corner
    additional_centers.append([0.9, 0.9])
    # Add a circle near the bottom-left corner
    additional_centers.append([0.1, 0.1])
    # Add another circle in the second row
    additional_centers.append([0.4, 0.25])
    # Add another circle in the third row
    additional_centers.append([0.6, 0.45])

    # Combine to get final centers
    centers = np.array(centers + additional_centers)

    initial_radius = grid_spacing / 2
    radii = np.full(n, initial_radius)

    # Define a penalty function to penalize overlaps, out-of-bound placements, and negative radii
    def penalty(x):
        # Penalty for negative radius
        neg_radius = np.sum(x < 0)

        # Penalty for out-of-bound placement
        boundary_penalty = 0
        for i in range(n):
            r = x[i]
            x_center, y_center = centers[i]
            if x_center - r < -1e-8 or x_center + r > 1 + 1e-8:
                boundary_penalty += max(0, (x_center + r - 1 - 1e-8) + (1 - (x_center - r + 1e-8)))
            if y_center - r < -1e-8 or y_center + r > 1 + 1e-8:
                boundary_penalty += max(0, (y_center + r - 1 - 1e-8) + (1 - (y_center - r + 1e-8)))

        # Penalty for overlapping circles
        overlap_penalty = 0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < x[i] + x[j] - 1e-8:
                    overlap_penalty += max(0, (x[i] + x[j] - dist - 1e-8))

        # Total penalty
        return neg_radius + boundary_penalty + overlap_penalty

    bounds = [(0, 0.5) for _ in range(n)]

    # Use SLSQP to minimize the total penalty while maximizing the sum of radii
    result = minimize(
        fun=penalty,
        x0=radii,
        bounds=bounds,
        method='SLSQP',

    return centers, radii, np.sum(radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
