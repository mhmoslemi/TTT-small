import scipy.optimize as opt
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Define a hybrid arrangement: 5 rows with 5 circles each, with one additional circle to be placed near the corner
    # Primary grid
    grid = []
    for row in range(5):
        for col in range(5):
            x = col / 4.0 + 0.05
            y = row / 4.0 + 0.05
            grid.append([x, y])
    # Place the 26th circle in the bottom-right corner
    grid.append([0.9, 0.9])
    centers = np.array(grid)
    
    # Initial guess for radii (all same)
    radii = np.ones(n) * 0.05

    def objective(radii):
        return -np.sum(radii)  # Objective to maximize sum of radii by minimizing negative sum

    def boundary_constraint(radii):
        return np.array([(centers[i, 0] - radii[i] >= 0) &
                        (centers[i, 1] - radii[i] >= 0) &
                        (centers[i, 0] + radii[i] <= 1) &
                        (centers[i, 1] + radii[i] <= 1)
                        for i in range(n)])

    def non_overlap_constraint(radii):
        dist = squareform(pdist(centers))
        return np.array([dist[i, j] - radii[i] - radii[j] for i in range(n) for j in range(i + 1, n)])

    # Set up optimization problem

    constraints.append({'type': 'ineq', 'fun': non_overlap_constraint})
