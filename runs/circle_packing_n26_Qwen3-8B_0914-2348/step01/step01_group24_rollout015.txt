import cvxpy as cp

def run_packing():

    n = 26

    # We'll use a more flexible model with free placement
    centers = cp.Variable((n, 2))
    radii = cp.Variable(n, nonneg=True)

    # Objective: maximize the sum of radii
    objective = cp.Maximize(cp.sum(radii))

    constraints = []
