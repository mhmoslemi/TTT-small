import scipy.optimize as opt
from scipy.spatial.distance import pdist, squareform

def run_packing():

    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    corners = np.array([
        [0, 0],
        [1, 0],
        [1, 1],
        [0, 1]
    ])

        radii[i] = 0.2  # Initial radius for corner circles

    remaining = n - 4

    max_col = 5

        # Normalize to the unit square
        x_norm = x / (max_col + 0.5)
        y_norm = y / (max_row * np.sqrt(3)/2)
        centers[4 + i] = [x_norm, y_norm]
        radii[4 + i] = 0.1  # Initial guess for radii

    def objective(radii):
        return -np.sum(radii)  # Minimize negative of sum to maximize sum
