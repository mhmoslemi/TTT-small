import numpy as np
import random
from scipy.spatial.distance import pdist, squareform
import math

def run_packing():

    population_size = 100
    generations = 100
    mutation_rate = 0.1
    mutation_strength = 0.05
    elite_ratio = 0.2
    max_radius = 0.5
    min_radius = 0.01

    # Generate initial population (random centers and radii)
    population = []
    for _ in range(population_size):
        centers = np.random.rand(n, 2)
        radii = np.random.rand(n) * max_radius
        population.append((centers, radii))

    def fitness(centers, radii):
        # Check if circles stay within the unit square
        for i in range(n):
            if (centers[i, 0] - radii[i] < 0 or
                centers[i, 0] + radii[i] > 1 or
                centers[i, 1] - radii[i] < 0 or
                centers[i, 1] + radii[i] > 1):
                return 0.0

        # Check for circle overlaps
        dist_matrix = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-12:
                    return 0.0

        # Calculate total radii
        return np.sum(radii)

    # Selection function (tournament selection)
    def select_parent(population):
        tournament = random.sample(population, 5)
        return max(tournament, key=lambda x: fitness(x[0], x[1]))

    # Crossover function
    def crossover(parent1, parent2):
        centers1, radii1 = parent1
        centers2, radii2 = parent2
        centers = np.zeros((n, 2))
        radii = np.zeros(n)
        for i in range(n):
            if random.random() < 0.5:
                centers[i] = centers1[i]
                radii[i] = radii1[i]
            else:
                centers[i] = centers2[i]
                radii[i] = radii2[i]
        return (centers, radii)

    def mutate(centers, radii):
        for i in range(n):
            # Move center
            if random.random() < mutation_rate:
                centers[i, 0] += random.uniform(-mutation_strength, mutation_strength)
                centers[i, 1] += random.uniform(-mutation_strength, mutation_strength)
                # Clamp to [0, 1]
                centers[i, 0] = np.clip(centers[i, 0], 0, 1)
                centers[i, 1] = np.clip(centers[i, 1], 0, 1)
            # Resize radius

        return (centers, radii)

            child = crossover(parent1, parent2)
            child = mutate(child[0], child[1])
            new_population.append(child)

        population = new_population

    # Find best individual
    best = max(population, key=lambda x: fitness(x[0], x[1]))
    centers_final, radii_final = best
    sum_radii = np.sum(radii_final)

    return (centers_final, radii_final, sum_radii)
