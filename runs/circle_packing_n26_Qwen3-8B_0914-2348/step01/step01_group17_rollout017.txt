from scipy.optimize import minimize
from itertools import combinations

    radii = np.zeros(num_circles)

    hex_positions = []
    # Generate 5 rows with 5 circles each (total 25)
    for row in range(5):

            x = x_start + col * 0.2
            y = 0.1 + (row % 2) * 0.15
            if row % 2 == 1:
                y += 0.15
            hex_positions.append([x, y])

    extra_circle = [0.5, 0.5]
    hex_positions.append(extra_circle)

    # Assign the positions
    positions[:] = hex_positions[:num_circles]

        dy = np.min([positions[i, 1], 1 - positions[i, 1]])
        radii[i] = np.min([dx, dy]) * 0.5

    def objective(radii):

    constraints = []

    for i in range(num_circles):
