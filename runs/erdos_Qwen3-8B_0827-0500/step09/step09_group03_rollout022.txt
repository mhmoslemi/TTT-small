    initial_guesses.append(generate_structured_initial(n_points, 12))
    initial_guesses.append(generate_structured_initial(n_points, 15))
    initial_guesses.append(generate_structured_initial(n_points, 20))
    initial_guesses.append(generate_random_initial(n_points))
    initial_guesses.append(generate_random_initial(n_points))
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        # Define objective function with normalization
        def objective(h):
            sum_h = np.sum(h)
            if sum_h == 0:
                return np.inf
            scaling_factor = (n_points / 2) / sum_h
            h_normalized = np.clip(h * scaling_factor, 0.0, 1.0)
            corr = np.correlate(h_normalized, 1 - h_normalized, mode='full')
            max_corr = np.max(corr)
            dx = 2.0 / n_points
            return max_corr * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution
        result = differential_evolution(
            objective, bounds, 
            strategy='best1bin', 
            popsize=15, 
            mutation=0.5, 
            recombination=0.7, 
            tol=1e-8, 
            maxiter=500, 
            seed=seed
        )
        
        # Check if optimization succeeded
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
