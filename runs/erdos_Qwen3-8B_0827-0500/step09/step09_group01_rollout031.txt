    pulse_indices = np.linspace(0, n_points - 1, num_pulses, endpoint=False).astype(int)
    structured[pulse_indices] = 1.0
    target_sum = n_points / 2
    current_sum = np.sum(structured)
    scaling_factor = target_sum / current_sum
    structured *= scaling_factor
    return structured

def generate_random_initial(n_points):
    h = np.random.rand(n_points)
    target_sum = n_points / 2
    current_sum = np.sum(h)
    scaling_factor = target_sum / current_sum
    h *= scaling_factor
    return h

def generate_symmetric_initial(n_points):
    half = n_points // 2
    h = np.random.rand(half)
    target_sum = n_points / 2
    current_sum = np.sum(h)
    scaling_factor = target_sum / current_sum
    h = h * scaling_factor
    h = np.concatenate([h, h[::-1]])
    if n_points % 2 == 1:
        h = np.concatenate([h, [0.0]])
    return h

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    initial_guesses = [
        initial_h_values,
        generate_perturbed_initial(initial_h_values, 0.01),
        generate_perturbed_initial(initial_h_values, 0.03),
        generate_perturbed_initial(initial_h_values, 0.05),
        generate_perturbed_initial(initial_h_values, 0.1),
        generate_structured_initial(n_points, 5),
        generate_structured_initial(n_points, 10),
        generate_structured_initial(n_points, 15),
        generate_structured_initial(n_points, 20),
        generate_random_initial(n_points),
        generate_symmetric_initial(n_points),
    ]
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        A_eq = np.ones(n_points)
        b_eq = n_points / 2
        cons = [LinearConstraint(A_eq, b_eq, b_eq)]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=cons,
                          tol=1e-10, options={'maxiter': 300, 'ftol': 1e-12, 'disp': False})
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
