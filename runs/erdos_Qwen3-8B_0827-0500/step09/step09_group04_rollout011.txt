            h[-1] = target_sum - np.sum(variables)
            h[-1] = np.clip(h[-1], 0.0, 1.0)
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            dx = 2.0 / n_points
            return max_corr * dx
        bounds = [(0.0, 1.0) for _ in range(n_points - 1)]
        try:
            result = differential_evolution(objective, bounds, strategy='best1bin', maxiter=100, popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.8, disp=False)
            best_variables = result.x
            h = np.zeros(n_points)
            h[:-1] = best_variables
            h[-1] = target_sum - np.sum(best_variables)
            h[-1] = np.clip(h[-1], 0.0, 1.0)
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h
        except:
            pass
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
