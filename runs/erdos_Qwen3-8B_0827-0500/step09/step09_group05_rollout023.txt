import numpy as np
import time
from scipy.optimize import minimize, LinearConstraint
from scipy.interpolate import interp1d
from multiprocessing import Pool, cpu_count

def generate_perturbed_initial(h, noise_level=0.05):
    perturbed = h + noise_level * np.random.randn(len(h))
    perturbed = np.clip(perturbed, 0.0, 1.0)
    return perturbed

def generate_structured_initial(n_points, num_pulses):
    structured = np.zeros(n_points)
    pulse_indices = np.linspace(0, n_points - 1, num_pulses, endpoint=False).astype(int)
    structured[pulse_indices] = 1.0
    target_sum = n_points / 2
    current_sum = np.sum(structured)
    scaling_factor = target_sum / current_sum
    structured *= scaling_factor
    return structured

def generate_random_initial(n_points):
    random_initial = np.random.rand(n_points)
    target_sum = n_points / 2
    current_sum = np.sum(random_initial)
    scaling_factor = target_sum / current_sum
    random_initial *= scaling_factor
    random_initial = np.clip(random_initial, 0.0, 1.0)
    return random_initial

def optimize_h(initial_guess, n_points, dx, evaluate_erdos_solution):
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    A_eq = np.ones(n_points)
    b_eq = n_points / 2
    cons = [LinearConstraint(A_eq, b_eq, b_eq)]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = minimize(objective, initial_guess, method='SLSQP', bounds=bounds, constraints=cons,
                      tol=1e-10, options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
    
    if result.success:
        return (result.x, result.fun)
    else:
        return (initial_guess, np.inf)

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    initial_guesses = [
        initial_h_values,
        generate_perturbed_initial(initial_h_values, noise_level=0.05),
        generate_perturbed_initial(initial_h_values, noise_level=0.1),
        generate_structured_initial(n_points, 5),
        generate_structured_initial(n_points, 10),
        generate_structured_initial(n_points, 15),
        generate_structured_initial(n_points, 20),
        generate_random_initial(n_points)
    ]
    
    best_h = initial_h_values
    best_c5 = evaluate_erdos_solution(best_h)
    start_time = time.time()
    
    max_workers = min(10, len(initial_guesses))
    with Pool(processes=max_workers) as pool:
        results = []
        for guess in initial_guesses:
            if time.time() - start_time >= budget_s:
                break
            result = pool.apply_async(optimize_h, (guess, n_points, dx, evaluate_erdos_solution))
            results.append(result)
        
        for result in results:
            if time.time() - start_time >= budget_s:
                break
            try:
                h, c5 = result.get()
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h
            except:
                pass
    
    return (best_h, best_c5, n_points)
