    random_initial = np.clip(random_initial, 0.0, 1.0)
    return random_initial

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    
    # Generate a focused set of initial guesses
    initial_guesses = [initial_h_values]
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.05))
    initial_guesses.append(generate_structured_initial(n_points, 5))
    initial_guesses.append(generate_structured_initial(n_points, 10))
    initial_guesses.append(generate_structured_initial(n_points, 15))
    initial_guesses.append(generate_structured_initial(n_points, 20))
    initial_guesses.append(generate_random_initial(n_points))
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        # Define objective function
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            dx = 2.0 / n_points
            return max_corr * dx
        
        # Constraints
        A_eq = np.ones(n_points)
        b_eq = n_points / 2
        cons = [LinearConstraint(A_eq, b_eq, b_eq)]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization settings
        result = minimize(objective, guess, method='L-BFGS-B', bounds=bounds, constraints=cons,
                          tol=1e-10, options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
        
        # Check if optimization succeeded
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
