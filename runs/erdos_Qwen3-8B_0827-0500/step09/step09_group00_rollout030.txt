def evaluate_C5(h, dx):
    n = len(h)
    max_integral = 0.0
    for m in range(-n + 1, n):
        integral = 0.0
        for i in range(n):
            j = i + m
            if 0 <= j < n:
                integral += h[i] * (1 - h[j]) * dx
        max_integral = max(max_integral, integral)
    return max_integral
