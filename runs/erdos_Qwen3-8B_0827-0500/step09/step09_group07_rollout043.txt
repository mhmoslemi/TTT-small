    structured = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            structured[i] = 1.0
    target_sum = n_points / 2
    current_sum = np.sum(structured)
    scaling_factor = target_sum / current_sum
    structured *= scaling_factor
    return structured

def generate_structured_initial(n_points, num_pulses=2, spacing_factor=1.0):
    structured = np.zeros(n_points)
    pulse_length = 1.0 / num_pulses
    spacing = (2.0 - num_pulses * pulse_length) / (num_pulses - 1)
    for i in range(num_pulses):
        start = i * (pulse_length + spacing) if i > 0 else 0.0
        end = start + pulse_length
        idx_start = int(start / (2.0 / n_points))
        idx_end = int(end / (2.0 / n_points))
        structured[idx_start:idx_end] = 1.0
    target_sum = n_points / 2
    current_sum = np.sum(structured)
    scaling_factor = target_sum / current_sum
    structured *= scaling_factor
    return structured

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    
    # Generate diverse initial guesses
    initial_guesses = [initial_h_values]
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.01))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.05))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.1))
    
    # Generate structured initial guesses with varied parameters
    for num_pulses in [3, 4, 5]:
        for spacing_factor in [0.8, 1.0, 1.2]:
            initial_guesses.append(generate_structured_initial(n_points, num_pulses=num_pulses, spacing_factor=spacing_factor))
    
    initial_guesses.append(generate_alternating_initial(n_points))
    initial_guesses.append(np.random.rand(n_points))
    
    # Normalize random initial guess
    random_initial = initial_guesses[-1]
    target_sum = n_points / 2
    current_sum = np.sum(random_initial)
    scaling_factor = target_sum / current_sum
    random_initial *= scaling_factor
    random_initial = np.clip(random_initial, 0.0, 1.0)
    initial_guesses[-1] = random_initial
    
    best_h = None
    best_c5 = float('inf')
    elapsed_time = 0.0
    start_time = np.datetime64('now')
    
    for guess in initial_guesses:
        # Define objective function with penalty for constraint violation
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            dx = 2.0 / n_points
            obj = max_corr * dx
            # Penalty for constraint violation
            constraint_violation = np.abs(np.sum(h) - n_points / 2)
            penalty = 1e4 * constraint_violation
            return obj + penalty
        
        # Define bounds
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution
        result = differential_evolution(objective, bounds, strategy='best1bin', 
                                       popsize=15, mutation=(0.5, 1), recombination=0.8, 
                                       tol=1e-12, maxiter=100, 
                                       disp=False)
        
        # Check if optimization succeeded
        if result.success:
            # Compute actual C5 without penalty
            corr = np.correlate(result.x, 1 - result.x, mode='full')
            max_corr = np.max(corr)
            dx = 2.0 / n_points
            current_c5 = max_corr * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Track time usage
        elapsed_time = np.datetime64('now') - start_time
        if np.less(elapsed_time, np.timedelta64(budget_s, 's')):
            continue
        else:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / n_points
        best_c5 = max_corr * dx
    
    return (best_h, best_c5, n_points)
