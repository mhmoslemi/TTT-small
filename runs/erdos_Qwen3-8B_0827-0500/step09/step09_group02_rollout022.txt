    np.random.seed(seed)
    n_points = len(initial_h_values)
    
    # Generate diverse initial guesses
    initial_guesses = [initial_h_values]
    # Add perturbations with varying noise levels
    for noise_level in [0.01, 0.03, 0.05, 0.1]:
        initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=noise_level))
    # Add structured initial guesses with varying pulse counts
    for num_pulses in range(2, 21):  # 2 to 20 pulses
        initial_guesses.append(generate_structured_initial(n_points, num_pulses))
    # Add more random initial guesses
    for _ in range(5):
        initial_guesses.append(generate_random_initial(n_points))
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        # Define objective function
        def objective(h):
            n = len(h)
            one_minus_h = 1.0 - h
            dx = 2.0 / n
            max_integral = 0.0
            for m in range(-n + 1, n):
                if m >= 0:
                    if n - m - 1 >= 0:
                        sum_val = np.dot(h[:n - m], one_minus_h[m:])
                    else:
                        sum_val = 0.0
                else:
                    k = -m
                    if n - k - 1 >= 0:
                        sum_val = np.dot(h[k:], one_minus_h[:n - k])
                    else:
                        sum_val = 0.0
                integral = sum_val * dx
                if integral > max_integral:
                    max_integral = integral
            return max_integral
        
        # Constraints
        A_eq = np.ones(n_points)
        b_eq = n_points / 2
        cons = [LinearConstraint(A_eq, b_eq, b_eq)]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization settings
        result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=cons,
                          tol=1e-10, options={'maxiter': 500, 'ftol': 1e-12, 'disp': False})
        
        # Check if optimization succeeded
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
