    best_c5 = float('inf')
    elapsed_time = 0.0
    start_time = np.datetime64('now')
    
    for guess in initial_guesses:
        # Define objective function
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            dx = 2.0 / n_points
            return max_corr * dx
        
        # Constraints
        A_eq = np.ones(n_points)
        b_eq = n_points / 2
        cons = [LinearConstraint(A_eq, b_eq, b_eq)]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization settings
        result = minimize(objective, guess, method='L-BFGS-B', bounds=bounds, constraints=cons,
                          tol=1e-10, options={'maxiter': 2000, 'ftol': 1e-10, 'gtol': 1e-10})
        
        # Check if optimization succeeded
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Track time usage
        elapsed_time = np.datetime64('now') - start_time
        if np.less(elapsed_time, np.timedelta64(budget_s, 's')):
            continue
        else:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
