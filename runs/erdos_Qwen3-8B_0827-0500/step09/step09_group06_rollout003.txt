import numpy as np
import time
from scipy.optimize import minimize, LinearConstraint
from multiprocessing import Pool, cpu_count

def generate_perturbed_initial(h, noise_level=0.05):
    perturbed = h + noise_level * np.random.randn(len(h))
    perturbed = np.clip(perturbed, 0.0, 1.0)
    return perturbed

def generate_structured_initial(n_points, num_pulses):
    structured = np.zeros(n_points)
    pulse_indices = np.linspace(0, n_points - 1, num_pulses, endpoint=False).astype(int)
    structured[pulse_indices] = 1.0
    target_sum = n_points / 2
    current_sum = np.sum(structured)
    scaling_factor = target_sum / current_sum
    structured *= scaling_factor
    return structured

def generate_random_initial(n_points):
    random_initial = np.random.rand(n_points)
    target_sum = n_points / 2
    current_sum = np.sum(random_initial)
    scaling_factor = target_sum / current_sum
    random_initial *= scaling_factor
    random_initial = np.clip(random_initial, 0.0, 1.0)
    return random_initial

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    
    # Generate diverse initial guesses
    initial_guesses = [initial_h_values]
    initial_guesses.append(generate_perturbed_initial(initial_h_values, 0.05))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, 0.1))
    for num_pulses in [5, 8, 12, 15, 20, 25, 30]:
        initial_guesses.append(generate_structured_initial(n_points, num_pulses))
    initial_guesses.append(generate_random_initial(n_points))
    
    # Objective function
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / n_points
        return max_corr * dx
    
    # Constraints
    A_eq = np.ones(n_points)
    b_eq = n_points / 2
    cons = [LinearConstraint(A_eq, b_eq, b_eq)]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization function for parallel processing
    def optimize(initial_guess):
        result = minimize(objective, initial_guess, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False})
        return (result.fun, result.x) if result.success else (float('inf'), initial_guess)
    
    # Run optimizations in parallel
    with Pool(processes=cpu_count()) as pool:
        results = pool.map(optimize, initial_guesses)
    
    # Select the best result
    best_c5 = min(results, key=lambda x: x[0])[0]
    best_h = min(results, key=lambda x: x[0])[1]
    
    return (best_h, best_c5, n_points)
