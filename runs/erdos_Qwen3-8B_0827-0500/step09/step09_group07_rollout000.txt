            recombination=0.8, 
            seed=seed, 
            maxiter=100
        )
        
        # Check if optimization succeeded
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Track time usage
        elapsed_time = np.datetime64('now') - start_time
        if np.less(elapsed_time, np.timedelta64(budget_s, 's')):
            continue
        else:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
