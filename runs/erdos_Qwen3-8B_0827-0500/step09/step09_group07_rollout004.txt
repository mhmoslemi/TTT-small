            if elapsed >= budget_s:
                break
            done, not_done = pool.poll(timeout=0.1, iteration=1)
            for job in done:
                results.append(job.get())
            if not not_done:
                break
    
    # Find the best result
    best_h = initial_h_values
    best_c5 = np.correlate(initial_h_values, 1 - initial_h_values, mode='full').max() * (2.0 / n_points)
    for result in results:
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return (best_h, best_c5, n_points)
