            if 0 <= j < n_points:
                current_sum += h[i] * (1.0 - h[j])
        max_corr = max(max_corr, current_sum)
    return max_corr * dx

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    
    # Generate diverse initial guesses
    initial_guesses = [initial_h_values]
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.05))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.1))
    initial_guesses.append(generate_structured_initial(n_points, 5))
    initial_guesses.append(generate_structured_initial(n_points, 8))
    initial_guesses.append(generate_structured_initial(n_points, 12))
    initial_guesses.append(generate_structured_initial(n_points, 15))
    initial_guesses.append(generate_structured_initial(n_points, 20))
    initial_guesses.append(generate_random_initial(n_points))
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    # Define constraints
    A_eq = np.ones(n_points)
    b_eq = n_points / 2
    cons = [LinearConstraint(A_eq, b_eq, b_eq)]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Use ThreadPoolExecutor for parallel optimization
    with ThreadPoolExecutor(max_workers=10) as executor:
        futures = []
        for guess in initial_guesses:
            # Submit optimization task
            future = executor.submit(minimize, objective, guess, method='SLSQP', bounds=bounds, constraints=cons,
                                     tol=1e-10, options={'maxiter': 500, 'ftol': 1e-12, 'disp': False})
            futures.append(future)
        
        # Process results
        for future in futures:
            result = future.result()
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            return (best_h, best_c5, n_points)
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
