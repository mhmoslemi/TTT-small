    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Generate diverse initial guesses
    initial_guesses = [initial_h_values]
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.01))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.03))
    initial_guesses.append(generate_structured_initial(n_points, 5))
    initial_guesses.append(generate_structured_initial(n_points, 7))
    initial_guesses.append(generate_structured_initial(n_points, 10))
    initial_guesses.append(generate_structured_initial(n_points, 12))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.05))
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        # Define objective function with penalty for sum constraint
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            penalty = 1e4 * np.abs(np.sum(h) - n_points / 2)
            return max_corr * dx + penalty
        
        # Optimization settings
        result = minimize(objective, guess, method='Nelder-Mead', 
                          bounds=[(0.0, 1.0) for _ in range(n_points)],
                          options={'xatol': 1e-6, 'fatol': 1e-6, 'maxiter': 200})
        
        # Check if optimization succeeded
        if result.success:
            # Compute the actual max_corr without penalty
            corr = np.correlate(result.x, 1 - result.x, mode='full')
            max_corr = np.max(corr)
            current_c5 = max_corr * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
