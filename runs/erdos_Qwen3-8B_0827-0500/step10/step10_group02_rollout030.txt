import numpy as np
import time
from scipy.optimize import minimize, LinearConstraint
from multiprocessing import Pool, cpu_count

def generate_perturbed_initial(h, noise_level=0.05):
    perturbed = h + noise_level * np.random.randn(len(h))
    perturbed = np.clip(perturbed, 0.0, 1.0)
    return perturbed

def generate_structured_initial(n_points, num_pulses, spread=1.0):
    structured = np.zeros(n_points)
    pulse_indices = np.round(np.linspace(0, n_points - 1, num_pulses, endpoint=False) + np.random.uniform(-spread, spread)).astype(int)
    pulse_indices = np.unique(np.clip(pulse_indices, 0, n_points - 1))
    structured[pulse_indices] = 1.0
    target_sum = n_points / 2
    current_sum = np.sum(structured)
    scaling_factor = target_sum / current_sum
    structured *= scaling_factor
    return structured

def evaluate_erdos_solution(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def objective(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def optimize_single(guess, n_points, dx, initial_h_values, seed):
    np.random.seed(seed)
    A_eq = np.ones(n_points)
    b_eq = n_points / 2
    cons = [LinearConstraint(A_eq, b_eq, b_eq)]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = minimize(objective, guess, args=(dx,), method='SLSQP', bounds=bounds, constraints=cons,
                      tol=1e-10, options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False})
    return result.x, result.fun

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    initial_guesses = []
    initial_guesses.append(initial_h_values)
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.01))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.03))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.05))
    
    for num_pulses in [5, 10, 15, 20, 25, 30]:
        initial_guesses.append(generate_structured_initial(n_points, num_pulses, spread=1.5))
    
    best_h = initial_h_values
    best_c5 = evaluate_erdos_solution(best_h)
    start_time = time.time()
    
    # Use multiprocessing to parallelize optimization
    pool = Pool(processes=min(cpu_count(), len(initial_guesses)))
    results = []
    
    for i, guess in enumerate(initial_guesses):
        if time.time() - start_time >= budget_s:
            break
        seed_i = seed + i
        results.append(pool.apply_async(optimize_single, (guess, n_points, dx, initial_h_values, seed_i)))
    
    for res in results:
        h_opt, c5_opt = res.get()
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    pool.close()
    pool.join()
    
    return (best_h, best_c5, n_points)
