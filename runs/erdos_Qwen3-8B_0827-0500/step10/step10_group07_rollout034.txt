            dx = 2.0 / n_points
            penalty = 1e6 * (np.sum(h) - n_points / 2)**2
            return max_corr * dx + penalty
        
        # Bounds for differential evolution
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution
        result = differential_evolution(objective, bounds, strategy='best1bin', maxiter=100, popsize=15, tol=1e-12, mutation=(0.5, 1), recombination=0.8, disp=False)
        
        # Check if optimization succeeded
        if result.success:
            current_c5 = result.fun - penalty  # Subtract penalty to get the actual objective
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
