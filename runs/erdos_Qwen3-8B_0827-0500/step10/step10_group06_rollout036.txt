    initial_guesses.append(generate_structured_initial(n_points, 20))
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    # Use multiprocessing to run optimizations in parallel
    def optimize_guess(guess):
        # Define objective function
        def objective_func(h):
            return objective(h, n_points)
        
        # Constraints
        A_eq = np.ones(n_points)
        b_eq = n_points / 2
        cons = [LinearConstraint(A_eq, b_eq, b_eq)]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization settings
        result = minimize(objective_func, guess, method='trust-constr', bounds=bounds, constraints=cons,
                          tol=1e-8, options={'maxiter': 1000, 'disp': False})
        
        # Check if optimization succeeded
        if result.success:
            current_c5 = result.fun
            return (current_c5, result.x)
        else:
            return (float('inf'), guess)  # Fallback to initial guess
    
    # Create a pool of workers
    with Pool(processes=cpu_count()) as pool:
        results = pool.map(optimize_guess, initial_guesses)
    
    # Process results
    for c5, h in results:
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
