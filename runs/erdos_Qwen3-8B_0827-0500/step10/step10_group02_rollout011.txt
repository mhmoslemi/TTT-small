    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        try:
            result = differential_evolution(
                objective, bounds, x0=guess, strategy='best1bin', 
                maxiter=100, popsize=15, tol=1e-6, mutation=(0.5, 1), 
                recombination=0.8, disp=False, polish=True
            )
            if result.success:
                current_c5 = result.fun - 1e6 * (np.sum(result.x) - n_points / 2) ** 2
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
