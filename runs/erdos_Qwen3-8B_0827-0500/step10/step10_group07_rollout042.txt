            for task in done:
                tasks.remove(task)
                try:
                    current_c5, current_h = task.get()
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = current_h
                except Exception as e:
                    pass
        
        for task in tasks:
            try:
                current_c5, current_h = task.get()
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h
            except Exception as e:
                pass
    
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
