    random_initial = np.clip(random_initial, 0.0, 1.0)
    return random_initial

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Higher resolution for better approximation
    target_sum = n_points / 2
    
    # Generate initial guesses
    initial_guesses = []
    # Interpolate initial_h_values to n_points=100
    if 'initial_h_values' in globals():
        initial_h_values_interp = interp1d(np.linspace(0, 2, len(initial_h_values)), initial_h_values, kind='linear', fill_value='extrapolate')
        h_interp = initial_h_values_interp(np.linspace(0, 2, n_points))
        initial_guesses.append(h_interp)
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.05))
    initial_guesses.append(generate_structured_initial(n_points, 5))
    initial_guesses.append(generate_structured_initial(n_points, 10))
    initial_guesses.append(generate_structured_initial(n_points, 15))
    initial_guesses.append(generate_random_initial(n_points))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.1))
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        # Define objective function with penalty
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            dx = 2.0 / n_points
            penalty = 1e6 * (np.sum(h) - target_sum)**2
            return max_corr * dx + penalty
        
        # Bounds for differential evolution
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution
        result = differential_evolution(objective, bounds, strategy='best1bin', popsize=15, mutation=(0.5, 1.0), recombination=0.8, tol=1e-10, maxiter=1000, disp=False)
        
        # Check if optimization succeeded
        if result.success:
            current_c5 = result.fun - 1e6 * (np.sum(result.x) - target_sum)**2  # Subtract penalty to get actual objective
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
