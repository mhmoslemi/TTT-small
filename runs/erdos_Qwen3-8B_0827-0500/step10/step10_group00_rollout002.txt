    perturbed = np.clip(perturbed, 0.0, 1.0)
    return perturbed

def generate_evenly_spaced_initial(n_points, num_pulses):
    positions = np.linspace(0, 2, num_pulses)
    structured = np.zeros(n_points)
    for pos in positions:
        idx = int(np.round(pos * n_points / 2))
        if 0 <= idx < n_points:
            structured[idx] = 1.0
    target_sum = n_points / 2
    current_sum = np.sum(structured)
    scaling_factor = target_sum / current_sum
    structured *= scaling_factor
    return structured

def generate_symmetric_initial(n_points, num_pulses):
    structured = np.zeros(n_points)
    for i in range(num_pulses // 2):
        pos = 1.0 - (i + 0.5) * (2.0 / num_pulses)
        idx = int(np.round(pos * n_points / 2))
        if 0 <= idx < n_points:
            structured[idx] = 1.0
        pos = 1.0 + (i + 0.5) * (2.0 / num_pulses)
        idx = int(np.round(pos * n_points / 2))
        if 0 <= idx < n_points:
            structured[idx] = 1.0
    target_sum = n_points / 2
    current_sum = np.sum(structured)
    scaling_factor = target_sum / current_sum
    structured *= scaling_factor
    return structured

def generate_random_initial(n_points):
    random_initial = np.random.rand(n_points)
    target_sum = n_points / 2
    current_sum = np.sum(random_initial)
    scaling_factor = target_sum / current_sum
    random_initial *= scaling_factor
    random_initial = np.clip(random_initial, 0.0, 1.0)
    return random_initial

def run(seed=42, budget_s=100, **kwargs):

    # Add more structured initial guesses

        initial_guesses.append(generate_evenly_spaced_initial(n_points, num_pulses))
        initial_guesses.append(generate_symmetric_initial(n_points, num_pulses))
    
    # Add more random initial guesses
    for _ in range(5):
        initial_guesses.append(generate_random_initial(n_points))
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        # Define objective function using cross-correlation
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Constraints
        A_eq = np.ones(n_points)
        b_eq = n_points / 2
        cons = [LinearConstraint(A_eq, b_eq, b_eq)]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization settings
        result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=cons,
                          tol=1e-10, options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
        
        # Check if optimization succeeded
        if result.success:
            current_c5 = result.fun

        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

    if best_h is None:
