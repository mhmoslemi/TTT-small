import numpy as np
from scipy.optimize import minimize, LinearConstraint

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    
    # Define objective function
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / n_points
        return max_corr * dx
    
    # Constraints
    A_eq = np.ones(n_points)
    b_eq = n_points / 2
    cons = [LinearConstraint(A_eq, b_eq, b_eq)]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization settings
    start_time = time.time()
    result = minimize(
        objective,
        initial_h_values,
        method='differential_evolution',
        bounds=bounds,
        constraints=cons,
        options={
            'maxiter': 100,
            'popsize': 20,
            'mutation': 0.5,
            'recombination': 0.8,
            'tol': 1e-10
        }
    )
    
    # Check if optimization succeeded
    if result.success:
        current_c5 = result.fun
        best_h = result.x
    else:
        # Fallback to initial guess
        best_h = initial_h_values
        current_c5 = evaluate_erdos_solution(best_h)
    
    # Ensure we respect the budget
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (best_h, current_c5, n_points)
    
    return (best_h, current_c5, n_points)
