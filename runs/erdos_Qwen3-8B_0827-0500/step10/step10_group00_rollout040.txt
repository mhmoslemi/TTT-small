        # For simplicity, use SLSQP for all, but with different parameters
        # Or use different methods for different guesses
        # Let's use COBYLA for some guesses and SLSQP for others
        # Let's alternate methods based on the index of the guess
        if len(initial_guesses) % 2 == 0:
            method = 'SLSQP'
        else:
            method = 'COBYLA'
        
        # Optimization settings
        result = minimize(objective, guess, method=method, bounds=bounds, constraints=cons,
                          tol=1e-10, options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False})
        
        # Check if optimization succeeded
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
