            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for the first n-1 variables
        bounds = [(0.0, 1.0) for _ in range(n_points - 1)]
        
        # Run differential evolution
        result = differential_evolution(objective, bounds, strategy='best1bin', 
                                       popsize=15, tol=1e-10, mutation=(0.5, 1), recombination=0.8, 
                                       maxiter=100, disp=False)
        
        # Check if optimization succeeded
        if result.success:
            current_h = np.zeros(n_points)
            current_h[:-1] = result.x
            current_h[-1] = (n_points / 2) - np.sum(current_h[:-1])
            current_h = np.clip(current_h, 0.0, 1.0)
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
