import numpy as np
import time
from scipy.optimize import minimize, LinearConstraint
from scipy.interpolate import interp1d
from joblib import Parallel, delayed

def generate_perturbed_initial(h, noise_level=0.05):
    perturbed = h + noise_level * np.random.randn(len(h))
    perturbed = np.clip(perturbed, 0.0, 1.0)
    return perturbed

def generate_structured_initial(n_points, num_pulses):
    structured = np.zeros(n_points)
    pulse_indices = np.linspace(0, n_points - 1, num_pulses, endpoint=False).astype(int)
    structured[pulse_indices] = 1.0
    target_sum = n_points / 2
    current_sum = np.sum(structured)
    scaling_factor = target_sum / current_sum
    structured *= scaling_factor
    return structured

def generate_random_initial(n_points):
    random_initial = np.random.rand(n_points)
    target_sum = n_points / 2
    current_sum = np.sum(random_initial)
    scaling_factor = target_sum / current_sum
    random_initial *= scaling_factor
    random_initial = np.clip(random_initial, 0.0, 1.0)
    return random_initial

def generate_two_pulse_initial(n_points):
    structured = np.zeros(n_points)
    pulse_width = 0.5 * n_points  # 50% of the interval
    start1 = int(n_points / 4)
    end1 = start1 + int(pulse_width)
    start2 = int(3 * n_points / 4) - int(pulse_width)
    end2 = start2 + int(pulse_width)
    structured[start1:end1] = 1.0
    structured[start2:end2] = 1.0
    target_sum = n_points / 2
    current_sum = np.sum(structured)
    scaling_factor = target_sum / current_sum
    structured *= scaling_factor
    return structured

def evaluate_erdos_solution(h):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    dx = 2.0 / len(h)
    return max_corr * dx

def optimize_single_guess(guess, n_points, initial_h_values, seed):

    # Define objective function
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / n_points
        return max_corr * dx
    
    # Constraints
    A_eq = np.ones(n_points)
    b_eq = n_points / 2
    cons = [LinearConstraint(A_eq, b_eq, b_eq)]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization settings
    result = minimize(objective, guess, method='L-BFGS-B', bounds=bounds, constraints=cons,

    # Check if optimization succeeded
    if result.success:

    else:
        return guess, evaluate_erdos_solution(guess)
