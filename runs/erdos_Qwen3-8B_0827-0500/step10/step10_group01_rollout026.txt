    current_sum = np.sum(structured)
    scaling_factor = target_sum / current_sum
    structured *= scaling_factor
    return structured

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Generate diverse initial guesses
    initial_guesses = [initial_h_values]
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.01))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.03))
    initial_guesses.append(generate_structured_initial(n_points, 5))
    initial_guesses.append(generate_structured_initial(n_points, 7))
    initial_guesses.append(generate_structured_initial(n_points, 10))
    initial_guesses.append(generate_structured_initial(n_points, 12))
    initial_guesses.append(generate_structured_initial(n_points, 15))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.05))
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        # Define objective function with penalty for sum constraint
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            penalty = 1e4 * np.abs(np.sum(h) - n_points / 2)
            return max_corr * dx + penalty
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution
        result = differential_evolution(
            objective, bounds, 
            strategy='best1bin', 
            maxiter=100, 
            popsize=15, 
            tol=1e-10, 
            mutation=(0.5, 1), 
            recombination=0.8,
            disp=False
        )
        
        # Check if the time is within budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        # Extract the best solution
        current_h = result.x
        current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
