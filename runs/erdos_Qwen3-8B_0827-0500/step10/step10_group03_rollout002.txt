        perturbed = np.clip(perturbed, 0.0, 1.0)
        initial_guesses.append(perturbed)
    # Add structured guesses
    for num_pulses in [2, 3, 4, 5, 6, 7, 8, 9, 10, 12, 14, 16, 18, 20]:
        structured = np.zeros(n_points)
        pulse_indices = np.linspace(0, n_points - 1, num_pulses, endpoint=False).astype(int)
        structured[pulse_indices] = 1.0
        structured = structured * (target_sum / np.sum(structured))
        initial_guesses.append(structured)
    # Add random guesses
    for _ in range(8):
        random_initial = np.random.rand(n_points)
        random_initial = random_initial * (target_sum / np.sum(random_initial))
        random_initial = np.clip(random_initial, 0.0, 1.0)
        initial_guesses.append(random_initial)

    best_h = initial_h_values
    best_c5 = evaluate_erdos_solution(initial_h_values)

    start_time = time.time()

    for guess in initial_guesses:
        # Define bounds
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        # Define objective with penalty
        def objective_with_penalty(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            penalty = 1e4 * abs(np.sum(h) - target_sum)
            return max_corr * dx + penalty

        # Run differential evolution
        result = differential_evolution(objective_with_penalty, bounds, strategy='best1bin', 
                                       popsize=15, mutation=(0.5, 1.0), recombination=0.8, 
                                       tol=1e-9, maxiter=100, disp=False)
        # Check if result is successful
        if result.success:
            # Compute the actual C5 value
            corr = np.correlate(result.x, 1 - result.x, mode='full')
            max_corr = np.max(corr)
            current_c5 = max_corr * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        # Check time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

    return (best_h, best_c5, n_points)
