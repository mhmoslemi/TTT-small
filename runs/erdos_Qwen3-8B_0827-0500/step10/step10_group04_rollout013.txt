    random_initial *= scaling_factor
    random_initial = np.clip(random_initial, 0.0, 1.0)
    return random_initial

def objective_with_penalty(h, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    dx = 2.0 / n_points
    obj = max_corr * dx
    constraint_violation = np.abs(np.sum(h) - n_points / 2)
    penalty = 1e4 * constraint_violation
    return obj + penalty

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    
    # Generate initial guesses
    initial_guesses = [initial_h_values]
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.05))
    initial_guesses.append(generate_structured_initial(n_points, 5))
    initial_guesses.append(generate_structured_initial(n_points, 10))
    initial_guesses.append(generate_structured_initial(n_points, 15))
    initial_guesses.append(generate_random_initial(n_points))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.1))
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    # Define the bounds for differential evolution
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Use multiprocessing to run optimizations in parallel
    with ProcessPoolExecutor(max_workers=10) as executor:
        futures = []
        for guess in initial_guesses:
            # Submit the optimization task
            future = executor.submit(differential_evolution, 
                                    objective_with_penalty, 
                                    bounds, 
                                    args=(n_points,), 
                                    strategy='best1bin', 
                                    popsize=15, 
                                    maxiter=100, 
                                    tol=1e-10)
            futures.append(future)
        
        # Process results as they complete
        for future in as_completed(futures):
            try:
                result = future.result()
                # Extract the best solution from the result
                h_opt = result.x
                # Compute the objective value
                corr = np.correlate(h_opt, 1 - h_opt, mode='full')
                max_corr = np.max(corr)
                dx = 2.0 / n_points
                current_c5 = max_corr * dx
                # Check if this is better
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_opt
            except Exception as e:
                print(f"Optimization failed: {e}")
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            # If time is up, return the best found so far
            return (best_h, best_c5, n_points)
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
