    np.random.seed(seed)
    n_points = len(initial_h_values)
    
    # Generate diverse initial guesses with varied structured patterns
    initial_guesses = [initial_h_values]
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.05))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.10))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.15))
    initial_guesses.append(generate_structured_initial(n_points, 5))
    initial_guesses.append(generate_structured_initial(n_points, 8))
    initial_guesses.append(generate_structured_initial(n_points, 12))
    initial_guesses.append(generate_structured_initial(n_points, 15))
    initial_guesses.append(generate_structured_initial(n_points, 20))
    initial_guesses.append(generate_random_initial(n_points))
    initial_guesses.append(generate_random_initial(n_points))
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        # Define objective function with penalty for sum constraint
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            dx = 2.0 / n_points
            # Penalty for sum constraint
            penalty = 1e6 * (np.sum(h) - n_points / 2)**2
            return max_corr * dx + penalty
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution with adjusted parameters
        result = differential_evolution(objective, bounds, strategy='best1bin', 
                                       popsize=15, mutation=(0.5, 1.0), recombination=0.8, 
                                       tol=1e-10, maxiter=100, disp=False)
        
        # Check if optimization succeeded
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
