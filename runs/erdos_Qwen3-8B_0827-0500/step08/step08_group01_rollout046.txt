    start_time = time.time()
    
    with Pool() as pool:
        results = pool.map(lambda guess: optimize_guess(guess, n_points), initial_guesses)
    
    for h, c5 in results:
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
