    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.05))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.1))
    initial_guesses.append(generate_structured_initial(n_points, num_pulses=2, spacing_factor=1.2, pulse_width=0.4))
    initial_guesses.append(generate_structured_initial(n_points, num_pulses=3, spacing_factor=1.0, pulse_width=0.5))

    initial_guesses.append(generate_structured_initial(n_points, num_pulses=5, spacing_factor=0.7, pulse_width=0.5))
    initial_guesses.append(generate_structured_initial(n_points, num_pulses=6, spacing_factor=0.6, pulse_width=0.5))
    initial_guesses.append(generate_structured_initial(n_points, num_pulses=7, spacing_factor=0.5, pulse_width=0.5))

    initial_guesses.append(np.random.rand(n_points))
    
    # Normalize random initial guess
    random_initial = initial_guesses[-1]
    current_sum = np.sum(random_initial)
    scaling_factor = target_sum / current_sum
    random_initial *= scaling_factor
    random_initial = np.clip(random_initial, 0.0, 1.0)
    initial_guesses[-1] = random_initial
    
    best_h = None
    best_c5 = float('inf')
    elapsed_time = 0.0
    
    for guess in initial_guesses:
        # Define objective function for differential evolution
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            dx = 2.0 / n_points
            return max_corr * dx + 1e6 * (np.sum(h) - target_sum)**2  # penalty term
        
        # Bounds for differential evolution
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Perform differential evolution
        result = differential_evolution(objective, bounds, strategy='best1bin', popsize=15, maxiter=100, tol=1e-10, mutation=(0.5, 1), recombination=0.8, disp=False)
        
        # Refine with SLSQP
        def objective_slsqp(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            dx = 2.0 / n_points
            return max_corr * dx
        
        # Constraints for SLSQP
        A_eq = np.ones(n_points)
        b_eq = target_sum
        cons = [({'type': 'eq', 'fun': lambda h: np.sum(h) - b_eq})]
        bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
        
        # Initial guess for SLSQP is the result from differential evolution
        result_slsqp = minimize(objective_slsqp, result.x, method='SLSQP', bounds=bounds_slsqp, constraints=cons, tol=1e-10, options={'maxiter': 500, 'ftol': 1e-10, 'gtol': 1e-10})
        
        # Check if optimization succeeded
        if result_slsqp.success:
            current_c5 = result_slsqp.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x
        
        # Track time usage
        # Estimating time based on iterations, but this is approximate
        elapsed_time += 10  # Assuming each DE run takes about 10 seconds
        if elapsed_time >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
