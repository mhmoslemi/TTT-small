        idx_end = int(end / (2.0 / n_points))
        structured[idx_start:idx_end] = 1.0
    target_sum = n_points / 2
    current_sum = np.sum(structured)
    scaling_factor = target_sum / current_sum
    structured *= scaling_factor
    return structured

def optimize_guess(guess, n_points, evaluate_erdos_solution):
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / n_points
        return max_corr * dx
    
    A_eq = np.ones(n_points)
    b_eq = n_points / 2
    cons = [LinearConstraint(A_eq, b_eq, b_eq)]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=cons,
                      tol=1e-12, options={'maxiter': 5000, 'ftol': 1e-12, 'gtol': 1e-12})
    
    if result.success:
        return {'h': result.x, 'c5': result.fun}
    else:
        # Fallback to normalized guess
        guess_sum = np.sum(guess)
        normalized_guess = guess * (n_points / 2) / guess_sum
        normalized_guess = np.clip(normalized_guess, 0.0, 1.0)
        c5 = evaluate_erdos_solution(normalized_guess)
        return {'h': normalized_guess, 'c5': c5}

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    
    # Generate diverse initial guesses
    initial_guesses = [initial_h_values]
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.01))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.05))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.1))
    for num_pulses in range(2, 11):
        initial_guesses.append(generate_structured_initial(n_points, num_pulses=num_pulses))
    initial_guesses.append(generate_alternating_initial(n_points))
    initial_guesses.append(np.random.rand(n_points))
    
    # Normalize random initial guess
    random_initial = initial_guesses[-1]
    target_sum = n_points / 2
    current_sum = np.sum(random_initial)
    scaling_factor = target_sum / current_sum
    random_initial *= scaling_factor
    random_initial = np.clip(random_initial, 0.0, 1.0)
    initial_guesses[-1] = random_initial
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    # Use multiprocessing to run optimizations in parallel
    with ProcessPoolExecutor(max_workers=10) as executor:
        futures = []
        for guess in initial_guesses:
            # Normalize the guess to ensure sum is n_points/2
            guess_sum = np.sum(guess)
            guess = guess * (n_points / 2) / guess_sum
            guess = np.clip(guess, 0.0, 1.0)
            # Submit the optimization task
            future = executor.submit(optimize_guess, guess, n_points, evaluate_erdos_solution)
            futures.append(future)
        
        # Process results as they come in
        for future in as_completed(futures):
            try:
                result = future.result()
                current_c5 = result['c5']
                current_h = result['h']
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h
            except Exception as e:
                print(f"Error in optimization: {e}")
            
            # Check if time budget is exceeded
            elapsed_time = time.time() - start_time
            if elapsed_time >= budget_s:
                break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
