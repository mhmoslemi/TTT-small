def optimize_guess(guess, n_points):
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / n_points
        return max_corr * dx
    
    A_eq = np.ones(n_points)
    b_eq = n_points / 2
    cons = [LinearConstraint(A_eq, b_eq, b_eq)]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=cons,
                      tol=1e-10, options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False})
    
    if result.success:
        return (result.fun, result.x)
    else:
        return (float('inf'), guess)

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    
    initial_guesses = [initial_h_values]
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.01))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.03))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.05))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.1))
    
    for num_pulses in [2, 3, 4, 5, 6, 7, 8, 9, 10, 12, 15, 20, 25, 30]:
        initial_guesses.append(generate_structured_initial(n_points, num_pulses))
    
    for _ in range(5):
        initial_guesses.append(generate_random_initial(n_points))
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    with Pool(processes=cpu_count()) as pool:
        results = [pool.apply_async(optimize_guess, (guess, n_points)) for guess in initial_guesses]
        
        while results:
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break
            result = results.pop(0)
            try:
                current_c5, current_h = result.get()
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h
            except Exception as e:
                pass
        
        for result in results:
            try:
                current_c5, current_h = result.get()
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h
            except Exception as e:
                pass
    
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
