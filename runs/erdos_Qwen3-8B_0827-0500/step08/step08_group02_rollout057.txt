    n_points = len(initial_h_values)
    
    initial_guesses = [initial_h_values]
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.01))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.03))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.05))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.1))
    
    for num_pulses in [2, 3, 4, 5, 6, 7, 8, 9, 10, 12, 15, 20, 25, 30]:
        initial_guesses.append(generate_structured_initial(n_points, num_pulses))
    
    for _ in range(5):
        initial_guesses.append(generate_random_initial(n_points))
    
    initial_guesses = [np.array(g) for g in initial_guesses]
    initial_guesses = list(set(tuple(g) for g in initial_guesses))
    initial_guesses = [np.array(g) for g in initial_guesses]
    
    best_h = initial_h_values
    best_c5 = evaluate_erdos_solution(best_h)
    start_time = time.time()
    
    with Pool(processes=10) as pool:
        results = []
        for guess in initial_guesses:
            results.append(pool.apply_async(optimize_initial_guess, (guess, n_points)))
        
        while results:
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break
            
            result = results.pop(0)
            h_opt, c5_opt = result.get()
            
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
    
    if best_c5 == float('inf'):
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
