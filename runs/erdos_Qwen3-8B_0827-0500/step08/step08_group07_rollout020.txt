    spacing = (2.0 - total_width) / (num_pulses - 1)
    for i in range(num_pulses):
        start = i * (pulse_width + spacing) if i > 0 else 0.0
        end = start + pulse_width
        idx_start = int(start / (2.0 / n_points))
        idx_end = int(end / (2.0 / n_points))
        structured[idx_start:idx_end] = 1.0
    target_sum = n_points / 2
    current_sum = np.sum(structured)
    scaling_factor = target_sum / current_sum
    structured *= scaling_factor
    return structured

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    start_time = time.time()
    
    # Generate initial guesses
    initial_guesses = []
    initial_guesses.append(initial_h_values.copy())
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.05))
    initial_guesses.append(generate_structured_initial(n_points, num_pulses=5, spacing_factor=0.8, pulse_width=0.4))
    initial_guesses.append(generate_structured_initial(n_points, num_pulses=6, spacing_factor=0.7, pulse_width=0.35))
    initial_guesses.append(generate_structured_initial(n_points, num_pulses=7, spacing_factor=0.6, pulse_width=0.3))
    
    best_h = None
    best_c5 = float('inf')
    
    for guess in initial_guesses:
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            dx = 2.0 / n_points
            return max_corr * dx
        
        A_eq = np.ones(n_points)
        b_eq = n_points / 2
        cons = [LinearConstraint(A_eq, b_eq, b_eq)]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        result = minimize(objective, guess, method='L-BFGS-B', bounds=bounds, constraints=cons,
                          tol=1e-8, options={'maxiter': 500, 'ftol': 1e-8, 'gtol': 1e-8})
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
    
    if best_h is None:
        best_h = initial_h_values
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
