from scipy.interpolate import interp1d

def generate_perturbed_initial(h, noise_level=0.05):
    perturbed = h + noise_level * np.random.randn(len(h))
    perturbed = np.clip(perturbed, 0.0, 1.0)
    return perturbed

def generate_structured_initial(n_points, num_pulses=2, spacing_factor=1.0, pulse_width=0.5):
    structured = np.zeros(n_points)
    total_width = num_pulses * pulse_width
    spacing = (2.0 - total_width) / (num_pulses - 1)
    for i in range(num_pulses):
        start = i * (pulse_width + spacing) if i > 0 else 0.0
        end = start + pulse_width
        idx_start = int(start / (2.0 / n_points))
        idx_end = int(end / (2.0 / n_points))
        structured[idx_start:idx_end] = 1.0
    target_sum = n_points / 2
    current_sum = np.sum(structured)
    scaling_factor = target_sum / current_sum
    structured *= scaling_factor
    return structured

    best_c5 = float('inf')

    for guess in initial_guesses:
        # Define objective function with zero-padding
        def objective(h):
            padded_h = np.pad(h, (n_points, n_points), 'constant')
            one_minus_h = 1.0 - h
            corr = np.correlate(padded_h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            dx = 2.0 / n_points
            return max_corr * dx

        # Track time usage

        if elapsed_time >= budget_s:

    # Fallback to initial guess if no improvement found

        best_h = initial_h_values
