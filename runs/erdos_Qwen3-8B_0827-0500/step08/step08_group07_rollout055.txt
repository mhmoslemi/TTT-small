    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization settings
    result = minimize(objective, guess, method='L-BFGS-B', bounds=bounds, constraints=cons,
                      tol=1e-10, options={'maxiter': 3000, 'ftol': 1e-10, 'gtol': 1e-10})
    
    # Check if optimization succeeded
    if result.success:
        return (result.x, result.fun)
    else:
        return (guess, float('inf'))  # Return the guess as a fallback

def objective(h, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    dx = 2.0 / n_points
    return max_corr * dx
