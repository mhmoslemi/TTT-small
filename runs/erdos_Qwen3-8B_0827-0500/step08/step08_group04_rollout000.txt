            h_candidate, c5_candidate = result
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate
            elapsed = time.time() - start_time
            if elapsed > budget_s:
                break
    
    if best_h is None:
        best_h = initial_h_values
        best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * (2.0 / n_points)
    
    return (best_h, best_c5, n_points)
