    scaling_factor = target_sum / current_sum
    random_initial *= scaling_factor
    random_initial = np.clip(random_initial, 0.0, 1.0)
    initial_guesses[-1] = random_initial
    
    best_h = None
    best_c5 = float('inf')
    
    for guess in initial_guesses:
        # Define objective function with penalty for sum constraint
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            dx = 2.0 / n_points
            # Penalty for sum constraint
            sum_h = np.sum(h)
            penalty = 1e3 * np.abs(sum_h - n_points / 2)
            return max_corr * dx + penalty
        
        # Bounds for each variable
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution
        result = differential_evolution(objective, bounds, strategy='best1bin', popsize=15, mutation=(0.5, 1.0), recombination=0.8, tol=1e-10, maxiter=100)
        
        # Check if optimization succeeded
        if result.success:
            current_c5 = result.fun - 1e3 * np.abs(np.sum(result.x) - n_points / 2)  # Subtract penalty to get actual objective
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Check elapsed time
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
