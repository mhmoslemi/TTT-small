                          tol=1e-8, options={'maxiter': 1000, 'disp': False})
        return result.fun, result.x
    
    with Pool(processes=min(cpu_count(), len(initial_guesses))) as pool:
        results = pool.map(optimize_guess, initial_guesses)
    
    for c5, h in results:
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (best_h, best_c5, n_points)
    
    return (best_h, best_c5, n_points)
