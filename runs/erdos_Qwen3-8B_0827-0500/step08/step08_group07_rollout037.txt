    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Generate diverse initial guesses
    initial_guesses = []
    # Perturb the initial guess
    initial_guesses.append(initial_h_values + np.random.normal(0, 0.05, n_points))
    initial_guesses.append(initial_h_values + np.random.normal(0, 0.1, n_points))
    # Structured initial guesses
    for num_pulses in [2, 3, 4, 5, 6, 7, 8, 9, 10]:
        spacing_factor = 1.0 if num_pulses <= 5 else 0.8
        pulse_width = 0.5 if num_pulses <= 5 else 0.6
        structured = np.zeros(n_points)
        total_width = num_pulses * pulse_width
        spacing = (2.0 - total_width) / (num_pulses - 1)
        for i in range(num_pulses):
            start = i * (pulse_width + spacing) if i > 0 else 0.0
            end = start + pulse_width
            idx_start = int(start / (2.0 / n_points))
            idx_end = int(end / (2.0 / n_points))
            structured[idx_start:idx_end] = 1.0
        target_sum = n_points / 2
        current_sum = np.sum(structured)
        scaling_factor = target_sum / current_sum
        structured *= scaling_factor
        initial_guesses.append(structured)
    # Random initial guesses
    for _ in range(3):
        random_initial = np.random.rand(n_points)
        target_sum = n_points / 2
        current_sum = np.sum(random_initial)
        scaling_factor = target_sum / current_sum
        random_initial *= scaling_factor
        random_initial = np.clip(random_initial, 0.0, 1.0)
        initial_guesses.append(random_initial)
    
    best_h = None
    best_c5 = float('inf')
    elapsed_time = 0.0
    
    # Run differential evolution for each initial guess
    for guess in initial_guesses:
        guess = np.clip(guess, 0.0, 1.0)
        # Run differential evolution
        result = differential_evolution(objective, bounds, popsize=20, mutation=0.5, recombination=0.8, tol=1e-10, maxiter=100)
        # Check if optimization succeeded
        if result.success:
            # Subtract penalty to get actual overlap value
            current_c5 = result.fun - 1000 * (np.sum(result.x) - n_points / 2)**2
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        # Track time usage
        elapsed_time += result.message.split()[-1] if 'time' in result.message else 0.1
        if elapsed_time >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = objective(best_h) - 1000 * (np.sum(best_h) - n_points / 2)**2
    
    return (best_h, best_c5, n_points)
