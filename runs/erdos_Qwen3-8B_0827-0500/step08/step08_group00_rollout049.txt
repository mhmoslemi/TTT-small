    for guess in initial_guesses:
        # Define objective function
        def objective(h):
            cross_corr = np.correlate(h, 1 - h, mode='full')
            max_val = np.max(cross_corr) * dx
            return max_val
        
        # Constraints
        A_eq = np.ones(n_points)
        b_eq = n_points / 2
        cons = [LinearConstraint(A_eq, b_eq, b_eq)]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization settings
        result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=cons,
                          tol=1e-10, options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False})
        
        # Check if optimization succeeded
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
