from scipy.optimize import minimize, LinearConstraint
from scipy.interpolate import interp1d

def generate_perturbed_initial(h, noise_level=0.05):
    perturbed = h + noise_level * np.random.randn(len(h))
    perturbed = np.clip(perturbed, 0.0, 1.0)
    return perturbed

def generate_alternating_initial(n_points):
    structured = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            structured[i] = 1.0
    target_sum = n_points / 2
    current_sum = np.sum(structured)
    scaling_factor = target_sum / current_sum
    structured *= scaling_factor
    return structured

def generate_structured_initial(n_points, num_pulses=2, spacing_factor=1.0):
    structured = np.zeros(n_points)
    pulse_length = 1.0 / num_pulses
    spacing = (2.0 - num_pulses * pulse_length) / (num_pulses - 1)
    for i in range(num_pulses):
        start = i * (pulse_length + spacing) if i > 0 else 0.0
        end = start + pulse_length
        idx_start = int(start / (2.0 / n_points))
        idx_end = int(end / (2.0 / n_points))
        structured[idx_start:idx_end] = 1.0
    target_sum = n_points / 2
    current_sum = np.sum(structured)
    scaling_factor = target_sum / current_sum
    structured *= scaling_factor
    return structured
