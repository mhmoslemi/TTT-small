    target_sum = n_points / 2
    current_sum = np.sum(random_initial)
    scaling_factor = target_sum / current_sum
    random_initial *= scaling_factor
    random_initial = np.clip(random_initial, 0.0, 1.0)
    initial_guesses[-1] = random_initial
    
    best_h = None
    best_c5 = float('inf')
    elapsed_time = 0.0
    start_time = np.datetime64('now')
    
    for guess in initial_guesses:
        # Define objective function
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            dx = 2.0 / n_points
            return max_corr * dx
        
        # Constraints
        A_eq = np.ones(n_points)
        b_eq = n_points / 2
        cons = [LinearConstraint(A_eq, b_eq, b_eq)]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization settings
        # Try differential evolution for some guesses
        if np.random.rand() < 0.5:  # 50% chance to use differential evolution
            result = differential_evolution(
                objective,
                bounds,
                strategy='best1bin',
                popsize=15,
                mutation=(0.5, 1),
                recombination=0.8,
                tol=1e-12,
                maxiter=100,
                disp=False
            )
            current_h = result.x
            current_c5 = result.fun
        else:
            result = minimize(
                objective,
                guess,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                tol=1e-12,
                options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            current_h = result.x
            current_c5 = result.fun
        
        # Check if optimization succeeded
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
        
        # Track time usage
        elapsed_time = np.datetime64('now') - start_time
        if np.less(elapsed_time, np.timedelta64(budget_s, 's')):
            continue
        else:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
