            maxiter=50, 
            popsize=15, 
            tol=1e-6, 
            mutation=(0.5, 1), 
            recombination=0.8, 
            disp=False,
            x0=guess
        )
        current_h = result.x
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return (best_h, best_c5, n_points)
