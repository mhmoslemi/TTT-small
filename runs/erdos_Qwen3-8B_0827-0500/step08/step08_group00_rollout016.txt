import time
from scipy.optimize import minimize, LinearConstraint

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Generate diverse initial guesses
    initial_guesses = [initial_h_values]
    # Add perturbations of the initial guess
    for noise_level in [0.01, 0.03, 0.05]:
        perturbed = initial_h_values + noise_level * np.random.randn(n_points)
        perturbed = np.clip(perturbed, 0.0, 1.0)
        initial_guesses.append(perturbed)
    # Add structured initial guesses with varying pulse counts
    for num_pulses in [5, 10, 15]:
        structured = np.zeros(n_points)
        pulse_indices = np.linspace(0, n_points - 1, num_pulses, endpoint=False).astype(int)
        structured[pulse_indices] = 1.0

        current_sum = np.sum(structured)

        structured *= scaling_factor
        initial_guesses.append(structured)
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        # Define objective function using numpy correlate for efficiency
        def objective(h):

            return np.max(corr) * dx
