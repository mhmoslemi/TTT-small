from scipy.optimize import minimize, LinearConstraint

def generate_perturbed_initial(h, noise_level=0.05):

def generate_structured_initial(n_points, num_pulses):

    for i in range(num_pulses):
        start = i * (pulse_length + spacing) if i > 0 else 0.0
        end = start + pulse_length
        dx = 2.0 / n_points
        idx_start = int(start / dx)
        idx_end = int(end / dx)
        idx_end = min(idx_end, n_points)
        structured[idx_start:idx_end] = 1.0
    target_sum = n_points / 2
    current_sum = np.sum(structured)
    scaling_factor = target_sum / current_sum
    structured *= scaling_factor
    return structured

def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    corr = np.correlate(h_values, 1 - h_values, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def objective_function(h, n_points):
    corr = np.correlate(h, 1 - h, mode='full')

    dx = 2.0 / n_points
    return max_corr * dx
