    random_initial = np.clip(random_initial, 0.0, 1.0)
    initial_guesses[-1] = random_initial
    
    best_h = None
    best_c5 = float('inf')
    elapsed_time = 0.0
    start_time = np.datetime64('now')
    
    with Pool(processes=cpu_count()) as pool:
        result_futures = [pool.apply_async(optimize_h, (guess, n_points)) for guess in initial_guesses]
        
        while result_futures:
            elapsed_time = np.datetime64('now') - start_time
            if np.greater(elapsed_time, np.timedelta64(budget_s, 's')):
                break
            
            try:
                result_future = result_futures.pop(0)
                current_c5, current_h = result_future.get(timeout=0.1)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h
            except Exception as e:
                continue
        
        if best_h is None:
            best_h = initial_h_values
            best_c5 = objective(best_h, n_points)
    
    return (best_h, best_c5, n_points)
