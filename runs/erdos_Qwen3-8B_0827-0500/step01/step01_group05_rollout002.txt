import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    n_points = 100  # Use 100 points for better accuracy and balance
    dx = 2.0 / n_points
    # Initialize h with the provided initial values, rescaled if needed
    # Assuming initial_h_values is for n_points=80, we'll scale it to 100
    # For simplicity, we'll use the initial values and pad with zeros
    # However, to ensure sum(h) = n_points / 2, we'll adjust
    h = np.zeros(n_points)
    if len(initial_h_values) > 0:
        # Interpolate initial_h_values to n_points
        h[:len(initial_h_values)] = initial_h_values
        # Normalize to meet the integral constraint
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Default to uniform distribution if no initial values are provided
        h = np.full(n_points, 0.5)
    
    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def find_k_max(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.argmax(corr)
    
    def compute_gradient(h_vec, k):
        n = len(h_vec)
        grad = np.zeros(n)
        for j in range(n):
            j_plus_k = j + k
            if j_plus_k < n:
                term1 = 1 - h_vec[j_plus_k]
            else:
                term1 = 0
            j_minus_k = j - k
            if j_minus_k >= 0:
                term2 = h_vec[j_minus_k]
            else:
                term2 = 0
            grad[j] = (term1 - term2) * dx
        return grad
    
    def gradient(h_vec):
        k = find_k_max(h_vec)
        return compute_gradient(h_vec, k)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    
    # Run optimization with SLSQP
    result = minimize(objective, h, method='SLSQP', jac=gradient, bounds=bounds, constraints=cons, options={'maxiter': 100})
    
    h_opt = result.x
    c5_bound = objective(h_opt)
    return h_opt, c5_bound, n_points
