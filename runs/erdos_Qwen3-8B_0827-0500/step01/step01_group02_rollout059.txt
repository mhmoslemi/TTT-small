import numpy as np
from scipy.optimize import minimize
from concurrent.futures import ThreadPoolExecutor

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is a global variable
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Define the objective function
    def objective(h):
        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    # Define constraints
    def constraint(h):
        return np.sum(h) - n_points / 2

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Function to generate random initial guess
    def generate_random_initial_guess(n_points):
        h = np.random.rand(n_points)
        h_sum = np.sum(h)
        scaling_factor = (n_points / 2) / h_sum
        h *= scaling_factor
        return h

    # Generate multiple initial guesses
    initial_guesses = [initial_h]
    for _ in range(5):  # generate 5 random initial guesses
        initial_guesses.append(generate_random_initial_guess(n_points))

    # Function to run optimization for a single guess
    def optimize_guess(h_initial):
        result = minimize(
            objective,
            h_initial,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 100, 'ftol': 1e-6, 'eps': 1e-6}
        )
        if result.success:
            return result.fun, result.x
        else:
            return objective(h_initial), h_initial

    # Run optimizations in parallel
    best_c5 = float('inf')
    best_h = initial_h.copy()

    with ThreadPoolExecutor(max_workers=10) as executor:
        results = list(executor.map(optimize_guess, initial_guesses))

    for c5, h in results:
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    return best_h, best_c5, n_points
