import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from provided values and normalize
    initial_h = initial_h_values.copy()
    sum_h = np.sum(initial_h)
    scale = (n_points / 2.0) / sum_h
    initial_h = initial_h * scale
    initial_h = np.clip(initial_h, 0.0, 1.0)
    
    # Objective function: minimize the maximum overlap integral
    def objective(h):
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Constraints: sum of h must equal n_points / 2
    constraints = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0})
    
    # Bounds: h[i] ∈ [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run optimization with SLSQP
    result = minimize(
        objective,
        initial_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        tol=1e-6,
        options={'maxiter': 100}
    )
    
    # Extract the best solution found
    h_opt = result.x
    c5_bound = objective(h_opt)
    
    return (h_opt, c5_bound, n_points)
