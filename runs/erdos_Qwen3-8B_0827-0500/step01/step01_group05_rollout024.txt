import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values

    dx = 2.0 / n_points

    initial_learning_rate = 0.1
    decay_rate = 0.995
    start_time = time.time()
    
    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def find_k_max(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.argmax(corr)
    
    def compute_gradient_for_k(h, k):
        grad = np.zeros_like(h)
        epsilon = 1e-5
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += epsilon
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            val_plus = corr_plus[k]
            h_minus = h.copy()
            h_minus[i] -= epsilon
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            val_minus = corr_minus[k]
            grad[i] = (val_plus - val_minus) / (2 * epsilon)
        return grad
    
    # Normalize initial h
    h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = compute_C5(h)
    best_h = h.copy()
    
    for iteration in range(max_iterations):
