def run(seed=42, budget_s=100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    def constraint(h):
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def perturb(h, seed=None):

        perturbation = np.random.uniform(-0.2, 0.2, size=len(h))

    best_c5 = float('inf')
    best_h = initial_h.copy()

    num_restarts = 20  # Increased from 5

    for i in range(num_restarts):
        h_initial = perturb(initial_h, seed=seed + i)
        result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, options={'maxiter': 200})
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
