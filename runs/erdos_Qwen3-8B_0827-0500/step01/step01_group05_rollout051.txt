import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Adam optimizer parameters
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    learning_rate = 0.1  # Initial learning rate
    t = 0  # Time step
    
    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def find_k_max(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.argmax(corr)
    
    # Initialize Adam variables
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    while time.time() - start_time < budget_s:
        t += 1
        k_max = find_k_max(h)
        # Compute analytical gradient for k_max
        grad = np.zeros(n_points)
        for j in range(n_points):
            # Compute term1: 1 - h[j + k_max] if j + k_max < n_points else 1
            if j + k_max < n_points:
                term1 = 1.0 - h[j + k_max]
            else:
                term1 = 1.0
            # Compute term2: h[j - k_max] if j - k_max >= 0 else 0
            if j - k_max >= 0:
                term2 = h[j - k_max]
            else:
                term2 = 0.0
            grad[j] = dx * (term1 - term2)
        
        # Update Adam parameters
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h = h - learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Clip h to [0, 1]

        # Normalize h to maintain the integral constraint
        current_sum = np.sum(h)
        if current_sum != 0:
            h = h * (n_points / 2.0) / current_sum
    
    final_C5 = compute_C5(h)
    return h, final_C5, n_points
