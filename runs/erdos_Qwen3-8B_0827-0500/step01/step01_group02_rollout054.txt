import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is a global variable
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Define the objective function with penalty
    def objective(h):
        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        penalty = 1e6 * np.abs(np.sum(h) - n_points / 2)
        return max_corr * dx + penalty

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run differential evolution with multiple restarts
    best_c5 = float('inf')
    best_h = initial_h.copy()
    num_restarts = 5

    for i in range(num_restarts):
        # Set seed for reproducibility
        np.random.seed(seed + i)
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed + i
        )
        current_c5 = objective(result.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x

    return best_h, best_c5, n_points
