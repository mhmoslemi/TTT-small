import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is a global variable with n_points=42
    n_points = len(initial_h_values)
    # Generate multiple initial guesses
    initial_guesses = [initial_h_values]
    # Add a few random initial guesses
    for _ in range(3):
        # Generate a random h with sum n_points / 2
        h = np.random.rand(n_points)
        # Scale to have sum n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        initial_guesses.append(h)
    # Define the objective function
    def objective(h):
        dx = 2.0 / n_points
        one_minus_h = 1.0 - h
        conv = np.correlate(h, one_minus_h, mode='full')
        c5 = np.max(conv) * dx
        # Add a small penalty for sum deviation
        penalty = 1e-6 * (np.sum(h) - n_points / 2)**2
        return c5 + penalty
    # Define the constraint
    def constraint(h):
        return np.sum(h) - n_points / 2
    cons = ({'type': 'eq', 'fun': constraint})
    # Bounds for each variable
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    # Best solution so far
    best_c5 = float('inf')
    best_h = None
    # Start timer
    start_time = time.time()
    # Run optimization for each initial guess
    for guess in initial_guesses:
        # Run optimization
        result = minimize(objective, x0=guess, method='L-BFGS-B', constraints=cons, bounds=bounds, tol=1e-6, options={'maxiter': 200})
        # Check if we have time left
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        # Update best solution
        h_opt = result.x
        dx = 2.0 / n_points
        one_minus_h = 1.0 - h_opt
        conv = np.correlate(h_opt, one_minus_h, mode='full')
        c5_bound = np.max(conv) * dx
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt
    # Return the best solution found
