    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )

    # Extract the best h_half
    best_h_half = result.x
    # Generate the full h vector
    if n_points % 2 == 0:
        best_h_full = np.concatenate([best_h_half, best_h_half[::-1]])
    else:
        best_h_full = np.concatenate([best_h_half, best_h_half[:-1][::-1]])
    # Normalize to meet the sum constraint
    sum_best = np.sum(best_h_full)
    best_h_normalized = best_h_full * (n_points / 2) / sum_best

    return best_h_normalized, result.fun, n_points
