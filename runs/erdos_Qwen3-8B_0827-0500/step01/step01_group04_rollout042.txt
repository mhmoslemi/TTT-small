import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points

        max_corr = np.max(corr)
        penalty = (np.sum(h) - target_sum) ** 2 * 1e4
        return max_corr * dx + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run differential evolution with multiple trials
    best_h = initial_h_values.copy()
    best_c5 = objective(best_h)
    for _ in range(3):
        result = differential_evolution(
            objective, bounds, strategy='best1bin',

            tol=1e-6, maxiter=100, seed=np.random.randint(0, 1000)
        )

        current_c5 = objective(current_h)
        if current_c5 < best_c5:

            best_c5 = current_c5
