    def constraint_high(h_vec):
        return 1 - h_vec

    cons = (
        {'type': 'eq', 'fun': constraint_sum},
        {'type': 'ineq', 'fun': constraint_low},
        {'type': 'ineq', 'fun': constraint_high}
    )

    result = minimize(
        fun=objective,
        x0=h,
        method='SLSQP',
        jac=gradient,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
    )

    best_h = result.x
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, n_points
