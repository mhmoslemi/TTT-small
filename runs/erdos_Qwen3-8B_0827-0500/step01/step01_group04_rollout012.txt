        return np.sum(h) - n_points / 2
    
    # First, use differential evolution for global search
    result_de = differential_evolution(
        objective,
        bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        method='bounded',
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        tol=1e-6,
        maxiter=100
    )
    
    # Refine with SLSQP
    def objective_slsqp(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / n_points
        return max_corr * dx
    
    A_eq = np.ones(n_points)
    b_eq = n_points / 2
    cons_slsqp = [LinearConstraint(A_eq, b_eq, b_eq)]
    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
    
    result_slsqp = minimize(
        objective_slsqp,
        result_de.x,
        method='SLSQP',
        bounds=bounds_slsqp,
        constraints=cons_slsqp
    )
    
    best_h = result_slsqp.x
    best_c5 = objective_slsqp(best_h)
    
    return (best_h, best_c5, n_points)
