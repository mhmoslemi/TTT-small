def run(seed=42, budget_s=100, **kwargs):
    # Initial setup
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points

        initial_h = initial_h * (n_points / 2) / sum_h
    h = initial_h.copy()
    # Parameters
    learning_rate = 0.05
    max_iterations = 5000
    time_start = time.time()
    for iteration in range(max_iterations):
        # Compute current C5
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx

        m = np.argmax(corr)
        k_max = m - (n_points - 1)
        # Compute gradient
        grad = np.zeros(n_points)

            if 0 <= i + k_max < n_points:

            if 0 <= i - k_max < n_points:
                term2 = h[i - k_max]
            grad[i] = dx * (term1 - term2)
        # Update h
        h -= learning_rate * grad

        if sum_h != 0:
