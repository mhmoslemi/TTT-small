    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points

    # Define the objective function
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    # Define the gradient function
    def gradient(h_flat):
        h = h_flat.reshape(n_points)
        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(cross_corr)
        # Compute the gradient numerically
        epsilon = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += epsilon
            cross_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            cross_plus_val = cross_plus[max_index]
            h_minus = h.copy()
            h_minus[i] -= epsilon
            cross_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            cross_minus_val = cross_minus[max_index]
            grad[i] = (cross_plus_val - cross_minus_val) / (2 * epsilon)
        return grad * dx

    # Define constraints
    constraints = [{'type': 'eq', 'fun': lambda h_flat: np.sum(h_flat) - n_points / 2}]

    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]

    # Initial guess
    h_initial = h.copy()

    # Run optimization
    start_time = time.time()
    result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=constraints, jac=gradient, options={'maxiter': 100})
    elapsed_time = time.time() - start_time

    # Return the best solution
    best_h = result.x
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    return best_h, best_c5, n_points
