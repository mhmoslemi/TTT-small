import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    # Initial setup
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    # Ensure sum is correct
    sum_h = np.sum(initial_h)
    if abs(sum_h - n_points / 2) > 1e-6:
        initial_h = initial_h * (n_points / 2) / sum_h
    h_initial = initial_h.copy()
    
    # Define the objective function
    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    # Define the constraints
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization with SLSQP
    result = minimize(
        objective,
        h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    # Fallback if optimization fails
    if result.success:
        h_opt = result.x
    else:
        h_opt = h_initial
    
    # Compute final C5
    corr_final = np.correlate(h_opt, 1 - h_opt, mode='full')
    c5_bound = np.max(corr_final) * dx
    
    return (h_opt, c5_bound, n_points)
