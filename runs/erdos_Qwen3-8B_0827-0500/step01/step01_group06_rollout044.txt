import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    # Initial setup
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    # Ensure sum is correct
    sum_h = np.sum(initial_h)
    if abs(sum_h - n_points / 2) > 1e-6:
        initial_h = initial_h * (n_points / 2) / sum_h
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = np.inf
    
    # Parameters
    learning_rate_start = 0.05
    learning_rate_end = 0.001
    max_iterations = 1000
    epsilon = 1e-4  # Increased for better gradient estimation
    time_start = time.time()
    
    for iteration in range(max_iterations):
        # Compute current C5
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        # Find the maximum shift index
        k_max = np.argmax(corr)
        # Compute gradient numerically
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_perturbed = h.copy()
            h_perturbed[i] += epsilon
            corr_perturbed = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
            grad[i] = (corr_perturbed[k_max] - corr[k_max]) / epsilon
        # Update h with decreasing learning rate
        learning_rate = learning_rate_start * (learning_rate_end / learning_rate_start) ** (iteration / max_iterations)
        h -= learning_rate * grad
        # Clip to [0, 1]
        h = np.clip(h, 0.0, 1.0)
        # Normalize sum
        sum_h = np.sum(h)
        if sum_h != 0:
            h = h * (n_points / 2) / sum_h
        # Check time budget
        if time.time() - time_start > budget_s:
            break
    
    # After optimization, compute the final C5
    corr_final = np.correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(corr_final) * dx
