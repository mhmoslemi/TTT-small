            grad[i] = (val_plus - val_minus) / (2 * epsilon)
        return grad
    
    def objective(h):
        return compute_C5(h)
    
    def gradient(h):
        k_max = find_k_max(h)
        return compute_gradient_for_k(h, k_max)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    result = minimize(objective, h, method='SLSQP', jac=gradient, bounds=bounds, constraints=cons, options={'maxiter': 100})
    
    # Check if time is up
    if time.time() - start_time > budget_s:
        # If time is up, return the best solution found so far
        pass
    
    return result.x, result.fun, n_points
