from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    # Set number of points and compute dx
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h with initial_h_values and normalize

    initial_h = initial_h * (n_points / 2) / sum_h

    def objective(h):

        corr = np.correlate(h, ones_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    
    # Bounds for each h[i]

    # Run the optimization with SLSQP
    result = minimize(
        objective, 
        initial_h, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=cons, 
        tol=1e-6, 
        options={'maxiter': 100}
