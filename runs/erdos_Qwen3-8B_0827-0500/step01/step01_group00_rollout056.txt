import scipy.optimize as opt
from scipy.optimize import Bounds

def run(seed=42, budget_s=100, **kwargs):

    # Choose n_points
    n_points = 64
    dx = 2.0 / n_points

    initial_guesses = []

    initial_guesses.append(initial_h_values)
    # Generate perturbed versions of initial_h_values
    for _ in range(5):
        perturb = np.random.normal(0, 0.05, n_points)
        h_perturb = initial_h_values + perturb
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb / np.sum(h_perturb) * (n_points / 2)
        initial_guesses.append(h_perturb)
    # Add a few random guesses
    for _ in range(3):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand / np.sum(h_rand) * (n_points / 2)
        initial_guesses.append(h_rand)
    
    best_c5 = float('inf')
    best_h = initial_h_values
    
    for guess in initial_guesses:
        # Convert to numpy array
        h_initial = np.array(guess)
        # Define the objective function
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Define constraints
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Set up optimization problem
        result = opt.minimize(
            fun=objective,
            x0=h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 1500, 'ftol': 1e-9, 'eps': 1e-8}
        )
        
        if result.success:
            current_c5 = objective(result.x)

                best_c5 = current_c5
                best_h = result.x
