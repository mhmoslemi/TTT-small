            term1 = 0.0
            if 0 <= i_plus < n_points:
                term1 = 1.0 - h_vec[i_plus]
            term2 = 0.0
            if 0 <= i_minus < n_points:
                term2 = h_vec[i_minus]
            grad[i] = dx * (term1 - term2)
        return grad
    
    constraint = {'type': 'eq', 'fun': lambda h_vec: np.sum(h_vec) - n_points / 2}
    
    start_time = time.time()
    result = minimize(
        objective,
        h,
        method='SLSQP',
        jac=gradient,
        constraints=constraint,
        options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return h, compute_c5(h, dx), n_points
    
    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    cross_corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(cross_corr)
    return max_corr * dx
