import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    # Initial setup
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    # Ensure sum is correct
    sum_h = np.sum(initial_h)
    if abs(sum_h - n_points / 2) > 1e-6:
        initial_h = initial_h * (n_points / 2) / sum_h
    h = initial_h.copy()
    
    # Define objective function
    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    # Define constraint function
    def constraint(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Define gradient function
    def gradient(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(h_vec)
        epsilon = 1e-6
        for i in range(len(h_vec)):
            h_perturbed = h_vec.copy()
            h_perturbed[i] += epsilon
            corr_perturbed = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
            grad[i] = (corr_perturbed[k_max] - corr[k_max]) / epsilon
        return grad
    
    # Optimization settings
    options = {
        'maxiter': 50,
        'ftol': 1e-8,
        'eps': 1e-8,
        'disp': False
    }
    
    # Perform optimization
    result = scipy.optimize.minimize(
        fun=objective,
        x0=h,
        method='SLSQP',
        constraints={'type': 'eq', 'fun': constraint},
        jac=gradient,
        options=options
    )
    
    # Check success and return result
    if result.success:
        h_opt = result.x
        corr_final = np.correlate(h_opt, 1 - h_opt, mode='full')
        c5_bound = np.max(corr_final) * dx
        return (h_opt, c5_bound, n_points)
    else:
        corr_initial = np.correlate(h, 1 - h, mode='full')
        c5_bound = np.max(corr_initial) * dx
        return (h, c5_bound, n_points)
