    # Define the objective function
    def objective(h):
        # Compute the overlap integral for all valid shifts k
        # This is an approximation using full cross-correlation
        cross_corr = np.correlate(h, 1 - h, mode='full')
        # The valid shifts are those where the overlap is non-zero
        # For simplicity, we take the max of the full cross-correlation
        max_corr = np.max(cross_corr)
        return max_corr * dx

    # Define constraints
    def constraint(h):
        return np.sum(h) - n_points / 2

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Function to generate a random initial guess
    def generate_random_initial(n_points):
        h = np.random.uniform(0, 1, n_points)
        h_sum = np.sum(h)
        scaling = (n_points / 2) / h_sum
        h = h * scaling
        return h

    # Function to perturb the initial guess
    def perturb(h, seed=None):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        return np.clip(h + perturbation, 0.0, 1.0)

    # Run multiple optimizations
    best_c5 = float('inf')
    best_h = initial_h.copy()

    # Number of restarts
    num_restarts = 10

    for i in range(num_restarts):
        # Generate a random initial guess
        h_initial = generate_random_initial(n_points)
        # Perform optimization
        result = minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, options={'maxiter': 100})
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # If optimization fails, try the perturbed guess
            current_c5 = objective(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_initial
