        for i in range(len(x)):
            x_perturbed = x.copy()
            x_perturbed[i] += epsilon
            # Normalize sum
            sum_perturbed = np.sum(x_perturbed)
            if sum_perturbed != 0:
                x_perturbed = x_perturbed * (n_points / 2) / sum_perturbed
            corr_perturbed = np.correlate(x_perturbed, 1 - x_perturbed, mode='full')
            # Get the value at k_max
            max_corr_perturbed = corr_perturbed[k_max] * dx
            # Original value
            max_corr_original = corr[k_max] * dx
            grad[i] = (max_corr_perturbed - max_corr_original) / epsilon
        return grad
    
    # Constraints
    constraints = [
        {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2},
        {'type': 'ineq', 'fun': lambda x: x},  # lower bound 0
        {'type': 'ineq', 'fun': lambda x: 1 - x}  # upper bound 1
    ]
    
    # Optimization with time budget
    start_time = time.time()
    result = optimize.minimize(
        fun=objective,
        x0=h,
        method='SLSQP',
        jac=gradient,
        constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
    )
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        # Early return if time is exceeded
        corr_final = np.correlate(h, 1 - h, mode='full')
        c5_bound = np.max(corr_final) * dx
        return (h, c5_bound, n_points)
    
    best_h = result.x
    # Compute the final C5
    corr_final = np.correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(corr_final) * dx
    return (best_h, c5_bound, n_points)
