import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h_initial = initial_h_values.copy()
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraints: sum(h) = n_points / 2, 0 <= h[i] <= 1
    constraints = [
        {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    ]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization settings
    result = minimize(
        objective,
        h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        tol=1e-6,
        options={'maxiter': 100}
    )
    
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    else:
        best_h = h_initial
        best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
