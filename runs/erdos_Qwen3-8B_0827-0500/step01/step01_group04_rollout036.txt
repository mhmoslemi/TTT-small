            dx = 2.0 / n_points
            return max_corr * dx
        
        A_eq = np.ones(n_points)
        b_eq = n_points / 2
        cons = [LinearConstraint(A_eq, b_eq, b_eq)]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Use L-BFGS-B method for better convergence
        result = minimize(objective, guess, method='L-BFGS-B', bounds=bounds, constraints=cons)
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        # Check if time budget is exceeded
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
    
    if best_h is None:
        best_h = initial_h_values
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
