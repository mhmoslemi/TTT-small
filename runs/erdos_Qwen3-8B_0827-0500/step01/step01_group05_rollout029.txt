import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    max_iterations = 10000
    start_time = time.time()
    
    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def find_k_max(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.argmax(corr)
    
    # Adam optimizer parameters
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    learning_rate = 0.01
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    t = 0
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        t += 1
        k_max = find_k_max(h)
        grad = np.zeros_like(h)
        epsilon = 1e-4
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += epsilon
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            val_plus = corr_plus[k_max]
            h_minus = h.copy()
            h_minus[i] -= epsilon
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            val_minus = corr_minus[k_max]
            grad[i] = (val_plus - val_minus) / (2 * epsilon)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)

        h = np.clip(h, 0, 1)

        if current_sum != 0:
            h = h * (target_sum / current_sum)
    
    final_C5 = compute_C5(h)
