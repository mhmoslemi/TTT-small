    best_h = initial_h_values
    
    for guess in initial_guesses:
        # Convert to numpy array
        h_initial = np.array(guess)
        # Define the objective function
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Define constraints
        def constraint_sum(h):
