    def objective(x):
        h_current = x.copy()
        cross_corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    # Bounds: each h[i] ∈ [0, 1]
    bounds = Bounds(0, 1)
    
    # Initial guess
    x0 = h.copy()
    
    # Use SLSQP method with analytical gradients if possible
    # For simplicity, use numerical gradient
    result = minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Early stopping if time is up
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
