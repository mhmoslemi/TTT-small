    for i in range(num_restarts):
        h_initial = perturb(initial_h, seed=seed + i)
        result = minimize(objective, h_initial, method='SLSQP', bounds=bounds,
                          constraints={'type': 'eq', 'fun': constraint},
                          options={'maxiter': 100})
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            current_c5 = objective(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_initial

    # Run differential evolution for global exploration
    def de_objective(h):
        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        penalty = 1e6 * abs(np.sum(h) - n_points / 2)
        return max_corr + penalty

    # Run DE with initial population around initial_h
    initial_population = [np.clip(initial_h + np.random.uniform(-0.1, 0.1, size=n_points), 0, 1) for _ in range(5)]
    result_de = differential_evolution(de_objective, bounds, strategy='best1bin',
                                      popsize=10, maxiter=50, tol=1e-6,
                                      mutation=(0.5, 1), recombination=0.8, seed=seed)
    current_c5 = result_de.fun
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = result_de.x

    return best_h, best_c5, n_points
