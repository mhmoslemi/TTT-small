```python
            grad[i] = (val_plus - val_minus) / (2 * epsilon)
        return grad
    
    constraints = [
        {'type': 'eq', 'fun': lambda h_vec: np.sum(h_vec) - n_points / 2}
    ]
    bounds = [(0, 1) for _ in range(n_points)]
    
    result = minimize(
        objective,
        h,
        method='L-BFGS-B',
        jac=gradient,
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return h, compute_C5(h), n_points
    
    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
```
