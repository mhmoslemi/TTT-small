import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    # Initial setup
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    # Normalize initial_h to meet the integral constraint
    sum_h = np.sum(initial_h)
    initial_h = initial_h * (n_points / 2) / sum_h
    h = initial_h.copy()
    
    # Objective function to minimize the maximum overlap integral
    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint: sum of h must be n_points / 2
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Bounds: each h_i is between 0 and 1
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization settings
    options = {
        'maxiter': 200,  # More iterations for better convergence
        'ftol': 1e-8,    # Tolerance for function value
        'eps': 1e-8,     # Step size for finite differences
        'disp': False    # Suppress optimization output
    }
    
    # Perform constrained optimization
    result = minimize(
        fun=objective,
        x0=h,
        method='SLSQP',
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        bounds=bounds,
        options=options
    )
    
    # Extract the best solution found
    best_h = result.x
    c5_bound = objective(best_h)
    
    return (best_h, c5_bound, n_points)
