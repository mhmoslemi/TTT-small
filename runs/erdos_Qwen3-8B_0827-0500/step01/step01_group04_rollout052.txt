import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / n_points
        penalty = 1e3 * abs(np.sum(h) - target_sum)
        return max_corr * dx + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Generate initial guesses
    initial_guesses = [initial_h_values]
    # Random initial guess
    random_guess = np.random.rand(n_points)
    random_guess = random_guess * target_sum / np.sum(random_guess)
    random_guess = np.clip(random_guess, 0.0, 1.0)
    initial_guesses.append(random_guess)
    
    # Binary pattern initial guess
    binary_guess = np.zeros(n_points)
    binary_guess[::2] = 1.0
    current_sum = np.sum(binary_guess)
    diff = target_sum - current_sum
    if diff != 0:
        zero_indices = np.where(binary_guess == 0)[0]
        if zero_indices.size > 0:
            diff_per_zero = diff / zero_indices.size
            binary_guess[zero_indices[:int(diff_per_zero * zero_indices.size)]] += 0.5
    initial_guesses.append(binary_guess)
    
    best_h = initial_h_values
    best_c5 = objective(best_h)
    
    # Run differential evolution for each initial guess
    for guess in initial_guesses:
        result = differential_evolution(
            objective,
            bounds,
            popsize=15,
            strategy='best1bin',
            tol=1e-6,
            maxiter=100,
            seed=seed
        )
        if result.success:
            current_h = result.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
    
    return (best_h, best_c5, n_points)
