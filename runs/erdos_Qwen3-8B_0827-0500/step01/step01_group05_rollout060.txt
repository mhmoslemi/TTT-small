            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            val_minus = corr_minus[k]
            grad[i] = (val_plus - val_minus) / (2 * epsilon)
        return grad
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        # Compute current maximum k
        k_max = find_k_max(h)
        # Compute gradient for this k
        grad = compute_gradient_for_k(h, k_max)
        # Adam update
        t = iteration + 1
        alpha = initial_lr * (1 - beta1**t) / (1 - beta2**t)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h -= alpha * m_hat / (np.sqrt(v_hat) + epsilon)
        # Clip h to [0, 1]
        h = np.clip(h, 0, 1)
        # Adjust to maintain integral constraint
        current_sum = np.sum(h)
        target_sum = n_points / 2
        if current_sum != 0:
            h = h * (target_sum / current_sum)
    
    final_C5 = compute_C5(h)
    return h, final_C5, n_points
