    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    start_time = time.time()
    max_iterations = 100  # Adjust based on time budget
    
    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        
        def objective(h_vec):
            corr = np.correlate(h_vec, 1 - h_vec, mode='full')
            return corr[k_max] * dx
        
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
        result = minimize(objective, h, method='SLSQP', constraints=cons, options={'ftol': 1e-8})
        h = result.x
        h = np.clip(h, 0, 1)
    
    final_C5 = compute_C5(h)
    return h, final_C5, n_points
