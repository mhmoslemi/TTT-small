def run(seed=42, budget_s=100, **kwargs):
    # Choose n_points
    n_points = 60
    dx = 2.0 / n_points
    # Generate initial guess: binary with 30 ones and 30 zeros
    np.random.seed(seed)
    indices = np.random.choice(n_points, size=n_points//2, replace=False)
    h_initial = np.zeros(n_points)
    h_initial[indices] = 1.0
    # Define the objective function
    def objective(h):
        one_minus_h = 1.0 - h
        conv = np.correlate(h, one_minus_h, mode='full')
        c5 = np.max(conv) * dx
        return c5
    # Define the constraint
    def constraint(h):
        return np.sum(h) - n_points / 2
    cons = ({'type': 'eq', 'fun': constraint})
    # Bounds for each variable
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    # Initial guess
    x0 = h_initial
    # Start timer
    start_time = time.time()
    # Run optimization
    result = minimize(objective, x0=x0, method='SLSQP', constraints=cons, bounds=bounds, tol=1e-6, options={'maxiter': 100})
    # Check elapsed time
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        # If time is exceeded, return the best solution found so far (result.x)
        h_opt = result.x
        dx = 2.0 / n_points
        one_minus_h = 1.0 - h_opt
        conv = np.correlate(h_opt, one_minus_h, mode='full')
        c5_bound = np.max(conv) * dx
        return h_opt, c5_bound, n_points
    else:
        # Compute the final C5 value
        h_opt = result.x
        dx = 2.0 / n_points
        one_minus_h = 1.0 - h_opt
        conv = np.correlate(h_opt, one_minus_h, mode='full')
        c5_bound = np.max(conv) * dx
        return h_opt, c5_bound, n_points
