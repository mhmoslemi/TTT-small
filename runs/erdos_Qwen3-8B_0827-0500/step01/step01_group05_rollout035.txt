import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    start_time = time.time()
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def gradient(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(h_flat)
        for i in range(n_points):
            term1 = 0.0
            if i + k_max < n_points:
                term1 = 1.0 - h[i + k_max]
            term2 = 0.0
            if i - k_max >= 0:
                term2 = h[i - k_max]
            grad[i] = term1 - term2
        return grad
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [
        {'type': 'eq', 'fun': lambda h_flat: np.sum(h_flat) - n_points / 2}
    ]
    
    result = opt.minimize(
        fun=objective,
        x0=h,
        method='SLSQP',
        jac=gradient,
        bounds=bounds,
        constraints=constraints,
        options={
            'maxiter': 100,
            'ftol': 1e-8,
            'eps': 1e-8,
            'disp': False
        }
    )
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return h, compute_C5(h), n_points
    
    return result.x, result.fun, n_points

def compute_C5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))
