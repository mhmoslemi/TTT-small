        k = max_k_index - (n_points - 1)
        # Compute gradient for this shift k
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Compute term1: 1 - h[i + k] if within bounds, else 1
            if 0 <= i + k < n_points:
                term1 = 1.0 - h[i + k]
            else:
                term1 = 1.0
            # Compute term2: h[i - k] if within bounds, else 0
            if 0 <= i - k < n_points:
                term2 = h[i - k]
            else:
                term2 = 0.0
            grad[i] = dx * (term1 - term2)
        # Update h using gradient descent
        step_size = 0.01
        h_new = h - step_size * grad
        # Clamp to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        # Adjust sum to be n_points / 2
        sum_h_new = np.sum(h_new)
        delta = (n_points / 2) - sum_h_new
        h_new += delta / n_points
        # Clamp again to ensure no overflow
        h_new = np.clip(h_new, 0, 1)
        h = h_new
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    cross_corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(cross_corr)
    return max_corr * dx
