import numpy as np
from scipy.optimize import minimize, LinearConstraint
from scipy.optimize import Bounds

def generate_random_initial(n_points):
    h = np.random.rand(n_points)
    target_sum = n_points / 2
    current_sum = np.sum(h)
    scaling_factor = target_sum / current_sum
    h *= scaling_factor
    h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    
    # Generate initial guesses
    initial_guesses = [initial_h_values]
    for _ in range(4):  # generate 4 more random initial guesses
        random_initial = generate_random_initial(n_points)
        initial_guesses.append(random_initial)
    
    best_h = None
    best_c5 = float('inf')
    
    for guess in initial_guesses:
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            dx = 2.0 / n_points
            return max_corr * dx
        
        A_eq = np.ones(n_points)
        b_eq = n_points / 2
        cons = [LinearConstraint(A_eq, b_eq, b_eq)]
        bounds = Bounds(0.0, 1.0)
        
        result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 1000, 'ftol': 1e-8})
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    if best_h is None:
        best_h = initial_h_values
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
