        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(cross_corr)
        k = max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            term1 = 0.0
            if 0 <= i + k < n_points:
                term1 = 1.0 - h[i + k]
            term2 = 0.0
            if 0 <= i - k < n_points:
                term2 = h[i - k]
            grad[i] = dx * (term1 - term2)
        return grad.flatten()
    
    # Constraints
    cons = [{'type': 'eq', 'fun': lambda h_flat: np.sum(h_flat) - n_points / 2}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization
    result = opt.minimize(
        fun=objective,
        x0=h,
        method='SLSQP',
        jac=gradient,
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    # Check if the optimization was successful
    if result.success:
        h_opt = result.x
        c5_opt = compute_c5(h_opt, dx)
        return h_opt, c5_opt, n_points
    else:
        # If optimization fails, return the initial guess
        return h, best_c5, n_points

def compute_c5(h, dx):
    cross_corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(cross_corr)
    return max_corr * dx
