import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    # Scale to meet the integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def gradient(h):
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(h)
        epsilon = 1e-4
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += epsilon
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            val_plus = corr_plus[k_max]
            h_minus = h.copy()
            h_minus[i] -= epsilon
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            val_minus = corr_minus[k_max]
            grad[i] = (val_plus - val_minus) / (2 * epsilon)
        return grad
    
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    
    start_time = time.time()
    result = minimize(objective, h, method='SLSQP', jac=gradient, constraints=cons, tol=1e-6, options={'maxiter': 100})
    
    # Ensure we don't exceed the budget
    if time.time() - start_time > budget_s:
        return h, compute_C5(h), n_points
    
    return result.x, result.fun, n_points

def compute_C5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))
