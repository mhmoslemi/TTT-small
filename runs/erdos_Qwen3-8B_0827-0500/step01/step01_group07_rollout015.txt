import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is a global variable with n_points=42
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Define the objective function with normalization
    def objective(h):
        target_sum = n_points / 2
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.inf
        normalized_h = h * (target_sum / current_sum)
        one_minus_h = 1.0 - normalized_h
        conv = np.correlate(normalized_h, one_minus_h, mode='full')
        c5 = np.max(conv) * dx
        return c5
    
    # Bounds for each variable
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Use differential evolution with strategy and parameters tuned for efficiency
    result = differential_evolution(
        objective, bounds, 
        strategy='best1bin', 
        popsize=20, 
        maxiter=100, 
        tol=1e-6, 
        mutation=(0.5, 1), 
        recombination=0.8, 
        seed=seed
    )
    
    # Extract the optimal h_values
    h_opt = result.x
    
    # Compute the final C5 value
    target_sum = n_points / 2
    current_sum = np.sum(h_opt)
    normalized_h = h_opt * (target_sum / current_sum)
    one_minus_h = 1.0 - normalized_h
    conv = np.correlate(normalized_h, one_minus_h, mode='full')
    c5_bound = np.max(conv) * dx
    
    return h_opt, c5_bound, n_points
