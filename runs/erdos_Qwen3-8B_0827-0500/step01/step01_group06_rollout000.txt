    learning_rate = 0.01
    beta1 = 0.9

    epsilon = 1e-8
    # Initialize first and second moments
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    # Time tracking
    time_start = time.time()
    max_iterations = 1000
    for iteration in range(max_iterations):
        # Compute correlation and find k_max
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        # Compute gradient for k_max
        epsilon = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_perturbed = h.copy()
            h_perturbed[i] += epsilon
            corr_perturbed = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
            grad[i] = (corr_perturbed[k_max] - corr[k_max]) / epsilon
        # Update Adam moments
        t = iteration + 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        # Update h
        h = h - learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)

    corr_final = np.correlate(h, 1 - h, mode='full')

import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    # Choose n_points
    n_points = 100
    # Generate initial guess
    np.random.seed(seed)
    h = np.zeros(n_points)
    indices = np.random.choice(n_points, size=n_points // 2, replace=False)
    h[indices] = 1.0
    h = h * (n_points / 2) / np.sum(h)
    # Adam parameters
    learning_rate = 0.01
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    # Initialize first and second moments
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    # Time tracking
    time_start = time.time()
    max_iterations = 1000
    for iteration in range(max_iterations):
        # Compute correlation and find k_max
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        # Compute gradient for k_max
        epsilon = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_perturbed = h.copy()
            h_perturbed[i] += epsilon
            corr_perturbed = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
            grad[i] = (corr_perturbed[k_max] - corr[k_max]) / epsilon
        # Update Adam moments
        t = iteration + 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        # Update h
        h = h - learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
        # Clip to [0, 1]
        h = np.clip(h, 0.0, 1.0)
        # Normalize sum
        sum_h = np.sum(h)
        if sum_h != 0:
            h = h * (n_points / 2) / sum_h
        # Check time budget
        if time.time() - time_start > budget_s:
            break
    # After optimization, compute the final C5
    corr_final = np.correlate(h, 1 - h, mode='full')
    c5_bound = np.max(corr_final) * (2.0 / n_points)
    return (h, c5_bound, n_points)
