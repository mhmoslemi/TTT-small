import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    # Initial setup
    n_points = 80
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()
    sum_h = np.sum(initial_h)
    if abs(sum_h - n_points / 2) > 1e-6:
        initial_h = initial_h * (n_points / 2) / sum_h
    initial_h = np.clip(initial_h, 0.0, 1.0)
    
    # Define objective function
    def objective(h):
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        return np.max(corr) * dx

    cons = (
        {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2},
    )
    
    # Bounds for each h_i
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization settings
    options = {
        'ftol': 1e-6,
        'maxiter': 100,
        'disp': False
    }
    
    start_time = time.time()
    result = minimize(objective, initial_h, method='SLSQP', bounds=bounds, constraints=cons, options=options)
    
    # Extract the best solution
    h_opt = result.x
    c5_bound = objective(h_opt)
    
    # Ensure we respect the time budget
    if time.time() - start_time > budget_s - 1:
        pass  # Early exit if time is up
    
    return (h_opt, c5_bound, n_points)
