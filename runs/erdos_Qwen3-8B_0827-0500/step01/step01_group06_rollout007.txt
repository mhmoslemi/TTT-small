    h = initial_h.copy()
    # Parameters for Adam
    alpha = 0.01  # learning rate
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    # Initialize first and second moments
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    time_start = time.time()
    for iteration in range(1000):  # max iterations
        # Compute current C5
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        # Find the maximum shift index
        k_max = np.argmax(corr)
        # Compute gradient analytically for k_max
        grad = np.zeros(n_points)
        # Map k_max to actual shift
        actual_k = k_max - (n_points - 1)
        for j in range(n_points):
            i_plus = j + int(np.round(actual_k))
            if 0 <= i_plus < n_points:
                term1 = 1 - h[i_plus]
            else:
                term1 = 0
            i_minus = j - int(np.round(actual_k))
            if 0 <= i_minus < n_points:
                term2 = h[i_minus]
            else:
                term2 = 0
            grad[j] = dx * (term1 - term2)
        # Update Adam parameters
        t = iteration + 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        # Update h
        h -= alpha * m_hat / (np.sqrt(v_hat) + epsilon)
        # Clip to [0, 1]
        h = np.clip(h, 0.0, 1.0)
        # Normalize sum

        # Check time budget
        if time.time() - time_start > budget_s:

    # After optimization, compute the final C5
    corr_final = np.correlate(h, 1 - h, mode='full')
    c5_bound = np.max(corr_final) * dx
    return (h, c5_bound, n_points)
