    def constraint(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]

        scaling = (n_points / 2) / h_sum

        return np.clip(h, 0.0, 1.0)

    # Function to perturb the initial guess
    def perturb(h, seed=None):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.2, 0.2, size=len(h))
        return np.clip(h + perturbation, 0.0, 1.0)

    # Function to perform optimization
    def optimize(initial_guess, method='SLSQP'):
        result = minimize(
            objective,
            initial_guess,
            method=method,
            bounds=Bounds(0.0, 1.0),
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 200, 'ftol': 1e-8}
        )
        if result.success:
            return result.x, result.fun
        else:
            return initial_guess, objective(initial_guess)

    best_c5 = float('inf')
    best_h = initial_h.copy()

    # Generate multiple initial guesses
    initial_guesses = [initial_h]
