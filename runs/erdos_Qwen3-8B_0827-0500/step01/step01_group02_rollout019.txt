        perturbation = np.random.uniform(-0.2, 0.2, size=len(h))
        return np.clip(h + perturbation, 0.0, 1.0)

    # Function to generate a random initialization
    def generate_random_initialization():
        h = np.random.rand(n_points)
        current_sum = np.sum(h)
        scale = (n_points / 2) / current_sum
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
        return h

    # Best solution found
    best_c5 = float('inf')
    best_h = initial_h.copy()

    # Number of restarts
    num_restarts = 15

    for i in range(num_restarts):
        # Randomly choose between perturbing the initial guess or generating a new one
        if np.random.rand() < 0.5:
            h_initial = perturb(initial_h, seed=seed + i)
        else:
            h_initial = generate_random_initialization()
        
        # Perform optimization
        result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, options={'maxiter': 200})
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # If optimization fails, try the perturbed guess
            current_c5 = objective(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_initial

    return best_h, best_c5, n_points
