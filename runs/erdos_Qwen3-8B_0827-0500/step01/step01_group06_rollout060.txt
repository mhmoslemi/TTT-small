        c5_bound = np.max(corr_final) * dx
        return (h, c5_bound, n_points)
    
    # Final evaluation
    corr_final = np.correlate(result.x, 1 - result.x, mode='full')
    c5_bound = np.max(corr_final) * dx
    return (result.x, c5_bound, n_points)
