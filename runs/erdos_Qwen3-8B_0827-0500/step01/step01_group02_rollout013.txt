from scipy.optimize import minimize
from multiprocessing import Pool, cpu_count

def run(seed=42, budget_s=100, **kwargs):

    initial_h = initial_h_values

    dx = 2.0 / n_points

    # Define the objective function
    def objective(h):
        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    # Define constraints
    def constraint(h):
        return np.sum(h) - n_points / 2

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Function to perturb the initial guess
    def perturb(h, seed=None):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.05, 0.05, size=len(h))
        return np.clip(h + perturbation, 0.0, 1.0)

    # Run multiple optimizations in parallel
    def optimize_perturbed(h_initial):
        result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8})
        return (result.fun, result.x) if result.success else (objective(h_initial), h_initial)

    # Number of restarts
    num_restarts = 10
    seeds = [seed + i for i in range(num_restarts)]

    # Parallel processing
    with Pool(processes=min(cpu_count(), num_restarts)) as pool:
        results = pool.map(lambda s: optimize_perturbed(perturb(initial_h, seed=s)), seeds)

    # Find the best solution
    best_c5 = float('inf')
    best_h = initial_h.copy()
    for c5, h in results:
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    return best_h, best_c5, n_points
