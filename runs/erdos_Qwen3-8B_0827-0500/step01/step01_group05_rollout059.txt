        top_k_indices = find_top_k(h, num_top=3)
        gradients = []
        for k in top_k_indices:
            grad = compute_gradient_for_k(h, k)
            gradients.append(grad)
        combined_grad = np.mean(gradients, axis=0)
        learning_rate = initial_learning_rate * (1 - iteration / max_iterations)
        h -= learning_rate * combined_grad
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        target_sum = n_points / 2
        if current_sum != 0:
            h = h * (target_sum / current_sum)
    
    final_C5 = compute_C5(h)
    return h, final_C5, n_points
