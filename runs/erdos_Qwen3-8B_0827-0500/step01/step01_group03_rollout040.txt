import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    # Initial setup
    n_points = 80
    initial_h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    if abs(sum_h - n_points / 2) > 1e-6:
        initial_h = initial_h * (n_points / 2) / sum_h
    initial_h = np.clip(initial_h, 0.0, 1.0)
    
    # Parameters
    max_iterations = 5000
    start_time = time.time()
    learning_rate = 0.1
    decay_rate = 0.999
    epsilon = 1e-6
    
    for iteration in range(max_iterations):
        # Compute cross-correlation to find max overlap
        ones_minus_h = 1.0 - initial_h
        corr = np.correlate(initial_h, ones_minus_h, mode='full')
        c5 = np.max(corr) * dx
        max_index = np.argmax(corr)
        
        # Compute gradient numerically
        gradient = np.zeros_like(initial_h)
        for i in range(n_points):
            h_perturbed = initial_h.copy()
            h_perturbed[i] += epsilon
            corr_perturbed = np.correlate(h_perturbed, 1.0 - h_perturbed, mode='full')
            delta = (corr_perturbed[max_index] - corr[max_index]) / epsilon
            gradient[i] = delta
        
        # Update h with adaptive learning rate
        learning_rate *= decay_rate
        initial_h -= learning_rate * gradient
        
        # Normalize sum
        sum_h = np.sum(initial_h)
        initial_h = initial_h * (n_points / 2) / sum_h
        # Clamp values to [0, 1]
        initial_h = np.clip(initial_h, 0.0, 1.0)
        
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final evaluation
    ones_minus_h = 1.0 - initial_h
    corr = np.correlate(initial_h, ones_minus_h, mode='full')
    c5_bound = np.max(corr) * dx
    
    return (initial_h, c5_bound, n_points)
