from scipy.optimize import minimize
from scipy.optimize import Bounds

    # Define the objective function
    def objective(h):
        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    # Define constraints
    def constraint(h):
        return np.sum(h) - n_points / 2

    # Bounds for each h[i]
    bounds = Bounds([0.0] * n_points, [1.0] * n_points)

    # Function to perturb the initial guess
    def perturb(h, seed=None):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.2, 0.2, size=len(h))
        return np.clip(h + perturbation, 0.0, 1.0)

    # Run multiple optimizations
    best_c5 = float('inf')
    best_h = initial_h.copy()

    # Number of restarts
    num_restarts = 20

    for i in range(num_restarts):
        # Perturb the initial guess
        h_initial = perturb(initial_h, seed=seed + i)
        # Perform optimization
        result = minimize(objective, h_initial, method='trust-constr', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, options={'maxiter': 200})
        if result.success:
