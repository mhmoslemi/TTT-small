import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    max_iterations = 1000
    learning_rate = 0.05
    start_time = time.time()
    
    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def find_k_max(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.argmax(corr)
    
    def compute_gradient_for_k(h, k):
        grad = np.zeros_like(h)
        epsilon = 1e-4
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += epsilon
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            val_plus = corr_plus[k]
            h_minus = h.copy()
            h_minus[i] -= epsilon
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            val_minus = corr_minus[k]
            grad[i] = (val_plus - val_minus) / (2 * epsilon)
        return grad
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        k_max = find_k_max(h)
        grad = compute_gradient_for_k(h, k_max)
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        target_sum = n_points / 2
        if current_sum != 0:
            h = h * (target_sum / current_sum)
    
    final_C5 = compute_C5(h)
    return h, final_C5, n_points
