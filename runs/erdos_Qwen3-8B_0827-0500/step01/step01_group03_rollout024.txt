import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    # Start with a moderate number of points for balance between speed and resolution
    n_points = 100
    initial_h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    # Normalize initial guess to meet the integral constraint
    if abs(sum_h - n_points / 2) > 1e-6:
        initial_h = initial_h * (n_points / 2) / sum_h
    initial_h = np.clip(initial_h, 0.0, 1.0)
    
    # Define the objective function: max overlap integral
    def objective(h):
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Define numerical gradient for the objective function
    def gradient(h):
        grad = np.zeros_like(h)
        epsilon = 1e-5
        for i in range(len(h)):
            h_perturbed = h.copy()
            h_perturbed[i] += epsilon
            corr_perturbed = np.correlate(h_perturbed, 1.0 - h_perturbed, mode='full')
            max_corr_perturbed = np.max(corr_perturbed) * dx
            grad[i] = (max_corr_perturbed - objective(h)) / epsilon
        return grad
    
    # Set bounds for each variable (0 <= h[i] <= 1)
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Equality constraint: sum(h) = n_points / 2
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    
    # Optimization using SLSQP with bounds and constraints
    result = minimize(
        objective, 
        initial_h, 
        method='SLSQP', 
        jac=gradient, 
        bounds=bounds, 
        constraints=cons, 
        tol=1e-6, 
        options={'maxiter': 100}
    )
    
    # Extract the best solution found within the time budget
    h_values = result.x
    c5_bound = objective(h_values)
    return (h_values, c5_bound, n_points)
