    # Generate initial guesses
    initial_guesses = generate_initial_guesses(n_points)
    
    # Define bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    A_eq = np.ones(n_points)
    b_eq = n_points / 2
    cons = [LinearConstraint(A_eq, b_eq, b_eq)]
    
    # Try different optimization methods
    for guess in initial_guesses:
        # Check remaining time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        # Try SLSQP
        result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=cons)
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Try differential evolution with penalty
        def objective_penalty(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            dx = 2.0 / n_points
            penalty = 1e4 * np.abs(np.sum(h) - n_points / 2)
            return max_corr * dx + penalty
        # Run differential evolution with limited iterations
        result_de = differential_evolution(objective_penalty, bounds, strategy='best1bin', maxiter=50, popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.8)
        if result_de.success:
            current_c5_de = result_de.fun - 1e4 * np.abs(np.sum(result_de.x) - n_points / 2)
            if current_c5_de < best_c5:
                best_c5 = current_c5_de
                best_h = result_de.x
        
        # Check remaining time again
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return (best_h, best_c5, n_points)
