import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    # Set parameters
    n_points = 100  # Reduced for speed while maintaining resolution
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()
    sum_h = np.sum(initial_h)
    if abs(sum_h - n_points / 2) > 1e-6:
        initial_h = initial_h * (n_points / 2) / sum_h
    initial_h = np.clip(initial_h, 0.0, 1.0)
    
    # Define objective function
    def objective(h):
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Define constraints
    constraints = [
        {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    ]
    
    # Define bounds for each variable
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization setup
    start_time = time.time()
    result = minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    elapsed_time = time.time() - start_time
    
    # Early termination if budget is exceeded
    if elapsed_time > budget_s - 1:
        # Return best solution found so far
        final_h = result.x
    else:
        final_h = result.x
    
    # Final evaluation
    ones_minus_h = 1.0 - final_h
    corr = np.correlate(final_h, ones_minus_h, mode='full')
    c5_bound = np.max(corr) * dx
    
    return (final_h, c5_bound, n_points)
