import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    h = initial_h_values.copy()

    dx = 2.0 / n_points

            else:
                term_plus = 0

            if 0 <= idx_minus < n_points:
                term_minus = h[idx_minus] * dx
            else:
                term_minus = 0
            grad[i] = term_plus - term_minus
        return grad

    cons = [{'type': 'eq', 'fun': constraint}]

    if result.success:

    else:
        h_opt = h_initial
    
    h_opt = np.clip(h_opt, 0, 1)
    sum_h = np.sum(h_opt)
    if sum_h != 0:
