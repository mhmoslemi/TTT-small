from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):

    n_points = 80

    sum_h = np.sum(h_initial)
    scale = (n_points / 2) / sum_h

        corr = np.correlate(h, ones_minus_h, mode='full')

    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Optimization constraints and bounds

    # Start optimization with SLSQP
    start_time = time.time()
    result = minimize(
        objective,

        method='SLSQP',
        bounds=bounds,
        constraints=cons,
