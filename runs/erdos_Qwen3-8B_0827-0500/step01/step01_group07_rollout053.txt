import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):

    # Define the objective function
    def objective(h):
        one_minus_h = 1.0 - h
        conv = np.correlate(h, one_minus_h, mode='full')
        c5 = np.max(conv) * dx
        return c5
    
    # Define the constraint
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    cons = ({'type': 'eq', 'fun': constraint})

    # Initial guess

    # Start timer
    start_time = time.time()
    
    # Run optimization with Nelder-Mead method
    result = minimize(objective, x0=x0, method='Nelder-Mead', constraints=cons, bounds=bounds, tol=1e-5, options={'maxiter': 100})
