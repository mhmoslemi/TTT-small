import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    # Initial setup
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    # Ensure sum is correct
    sum_h = np.sum(initial_h)
    if abs(sum_h - n_points / 2) > 1e-6:
        initial_h = initial_h * (n_points / 2) / sum_h
    # Objective function
    def objective(h):
        h_array = np.array(h)
        corr = np.correlate(h_array, 1 - h_array, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    # Constraints: sum(h) = n_points / 2
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    # Time tracking
    time_start = time.time()
    # Optimization
    result = minimize(objective, initial_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
    # Check time budget and return best solution
    elapsed_time = time.time() - time_start
    if elapsed_time >= budget_s:
        # Early exit if time is up
        h_values = initial_h
        corr_final = np.correlate(h_values, 1 - h_values, mode='full')
        c5_bound = np.max(corr_final) * dx
        return (h_values, c5_bound, n_points)
    # Final evaluation
    h_values = result.x
    corr_final = np.correlate(h_values, 1 - h_values, mode='full')
    c5_bound = np.max(corr_final) * dx
    return (h_values, c5_bound, n_points)
