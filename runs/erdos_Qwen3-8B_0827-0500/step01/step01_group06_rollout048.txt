import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Initial setup
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    # Ensure sum is correct
    sum_h = np.sum(initial_h)
    if abs(sum_h - n_points / 2) > 1e-6:
        initial_h = initial_h * (n_points / 2) / sum_h
    h = initial_h.copy()
    # Define objective function
    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    # Define gradient function
    def gradient(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_val = np.max(corr)
        k_max = np.argmax(corr)
        grad = np.zeros_like(h_vec)
        epsilon = 1e-5
        for i in range(len(h_vec)):
            h_perturbed = h_vec.copy()
            h_perturbed[i] += epsilon
            corr_perturbed = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
            grad[i] = (corr_perturbed[k_max] - corr[k_max]) / epsilon
        return grad
    # Constraints: sum(h) = n_points / 2
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    # Bounds: 0 <= h[i] <= 1
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    # Optimization
    time_start = time.time()
    result = minimize(objective, h, method='SLSQP', jac=gradient, constraints=cons, bounds=bounds, options={'maxiter': 100})
    # Check time budget
    elapsed = time.time() - time_start
    if elapsed > budget_s:
        return (h, np.max(np.correlate(h, 1 - h, mode='full')) * dx, n_points)
    # Return the best solution
    return (result.x, result.fun, n_points)
