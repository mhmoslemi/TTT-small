            # Check if the sum is close to the required value
            if abs(np.sum(result.x) - n_points / 2) < 1e-5:
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
    
    return (best_h, best_c5, n_points)
