from scipy.optimize import minimize, LinearConstraint

def generate_random_initial(n_points):
    h = np.random.rand(n_points)
    target_sum = n_points / 2
    current_sum = np.sum(h)
    scaling_factor = target_sum / current_sum
    h *= scaling_factor
    h = np.clip(h, 0.0, 1.0)
    return h

def generate_perturbed_initial(base_h, perturbation_scale=0.1):
    perturbed = base_h + np.random.uniform(-perturbation_scale, perturbation_scale, size=len(base_h))
    perturbed = np.clip(perturbed, 0.0, 1.0)
    target_sum = len(base_h) / 2
    current_sum = np.sum(perturbed)
    scaling_factor = target_sum / current_sum
