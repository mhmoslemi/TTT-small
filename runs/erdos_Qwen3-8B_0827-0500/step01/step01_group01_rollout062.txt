        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += epsilon
            h_minus = h.copy()
            h_minus[j] -= epsilon
            cross_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            cross_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            cross_plus_val = cross_plus[max_k_index]
            cross_minus_val = cross_minus[max_k_index]
            grad[j] = (cross_plus_val - cross_minus_val) / (2 * epsilon)
        # Perform line search to find optimal step size
        step_sizes = [0.01, 0.001, 0.0001]
        best_step = 0.01
        best_delta_c5 = -np.inf
        for step in step_sizes:
            h_new = h - step * grad
            sum_h_new = np.sum(h_new)
            delta = (n_points / 2) - sum_h_new
            h_new += delta / n_points
            h_new = np.clip(h_new, 0, 1)
            cross_new = np.correlate(h_new, 1 - h_new, mode='full')
            new_c5 = cross_new[max_k_index] * dx
            if new_c5 < current_c5:
                best_step = step
                best_delta_c5 = new_c5
        # Update h with the best step size
        h = h - best_step * grad
        sum_h = np.sum(h)
        delta = (n_points / 2) - sum_h
        h += delta / n_points
        h = np.clip(h, 0, 1)
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    cross_corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(cross_corr)
    return max_corr * dx
