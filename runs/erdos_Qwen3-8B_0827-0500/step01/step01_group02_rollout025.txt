                best_c5 = current_c5
                best_h = result.x
        else:
            # If optimization fails, try the initial guess
            current_c5 = objective(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_initial

    return best_h, best_c5, n_points
