        h_rand = h_rand / np.sum(h_rand) * (n_points / 2)
        initial_guesses.append(h_rand)
    
    best_c5 = float('inf')
    best_h = initial_h_values
    
    # Define the objective function with penalty
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        penalty = 1e4 * abs(np.sum(h) - n_points / 2)
        return max_corr * dx + penalty
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution for each initial guess
    for guess in initial_guesses:
        h_initial = np.array(guess)
        
        # Use differential evolution with the initial guess as a starting point
        result = opt.differential_evolution(
            fun=objective,
            bounds=bounds,
            popsize=15,
            maxiter=100,
            tol=1e-6,
            seed=seed
        )
        
        if result.success:
            current_h = result.x
            corr = np.correlate(current_h, 1 - current_h, mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
    
    return (best_h, best_c5, n_points)
