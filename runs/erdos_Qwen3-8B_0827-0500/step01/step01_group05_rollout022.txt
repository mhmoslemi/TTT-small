import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    max_iterations = 10000
    learning_rate = 0.001
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    start_time = time.time()
    
    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def find_k_max(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.argmax(corr)
    
    def compute_gradient_for_k(h, k):
        grad = np.zeros_like(h)
        epsilon = 1e-4
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += epsilon
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            val_plus = corr_plus[k]
            h_minus = h.copy()
            h_minus[i] -= epsilon
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            val_minus = corr_minus[k]
            grad[i] = (val_plus - val_minus) / (2 * epsilon)
        return grad
    
    # Initialize Adam parameters
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    t = 1
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        k_max = find_k_max(h)
        grad = compute_gradient_for_k(h, k_max)
        
        # Update Adam parameters
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        t += 1
        
        # Update h
        h -= learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        target_sum = n_points / 2
        if current_sum != 0:
            h = h * (target_sum / current_sum)
    
    final_C5 = compute_C5(h)
    return h, final_C5, n_points
