            best_c5 = current_c5
            best_h = h.copy()
        # Find the k with maximum cross-correlation
        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_k_index = np.argmax(cross_corr)
        # Compute corresponding shift k
        k = max_k_index - (n_points - 1)
        # Compute gradient numerically
        grad = np.zeros(n_points)
        epsilon = 1e-5
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += epsilon
            h_minus = h.copy()
            h_minus[j] -= epsilon
            cross_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            cross_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            # Get the cross-correlation value for the current k
            cross_plus_val = cross_plus[max_k_index]
            cross_minus_val = cross_minus[max_k_index]
            grad[j] = (cross_plus_val - cross_minus_val) / (2 * epsilon)
        # Line search for optimal step size
        alpha = 0.1  # Initial step size
        beta = 0.5   # Reduction factor
        t = 1.0
        while True:
            h_new = h - t * grad
            # Enforce constraint sum(h) = n_points / 2
            sum_h_new = np.sum(h_new)
            delta = (n_points / 2) - sum_h_new
            h_new += delta / n_points
            h_new = np.clip(h_new, 0, 1)
            # Compute new_c5
            new_c5 = compute_c5(h_new, dx)
            if new_c5 < current_c5:
                break
            t *= beta
        h = h_new
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    cross_corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(cross_corr)
    return max_corr * dx
