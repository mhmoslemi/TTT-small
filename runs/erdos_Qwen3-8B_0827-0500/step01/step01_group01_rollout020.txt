import scipy.optimize as opt
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    # Assume initial_h_values is a global variable as per the problem statement
    h = initial_h_values.copy()

    start_time = time.time()
    # Run optimization with SLSQP
    result = opt.minimize(
