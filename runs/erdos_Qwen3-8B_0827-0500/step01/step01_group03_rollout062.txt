import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    # Start with a higher resolution for better precision
    n_points = 100
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()

    # Normalize initial_h to meet the integral constraint
    initial_h = initial_h * (n_points / 2) / sum_h
    initial_h = np.clip(initial_h, 0.0, 1.0)

    def objective(h):
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Constraint: sum(h) must equal n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Optimization bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint})
    
    # Use COBYLA for robustness with bounds and constraints
    result = minimize(
        objective, 
        initial_h, 
        method='COBYLA', 
        bounds=bounds, 
        constraints=cons, 
        tol=1e-6, 
        options={'maxiter': 100}
    )
    
    # Extract the best solution found
    best_h = result.x
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
