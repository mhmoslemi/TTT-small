import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is a global variable with n_points=42
    n_points = len(initial_h_values)
    # Define the objective function with penalty
    def objective(h):
        dx = 2.0 / n_points
        one_minus_h = 1.0 - h
        conv = np.correlate(h, one_minus_h, mode='full')
        c5 = np.max(conv) * dx
        # Penalty for constraint violation
        penalty = 1e6 * abs(np.sum(h) - n_points / 2)
        return c5 + penalty
    # Bounds for each variable
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    # Set seed for reproducibility
    np.random.seed(seed)
    # Run differential evolution
    start_time = time.time()
    result = differential_evolution(
        objective, 
        bounds, 
        strategy='best1bin', 
        maxiter=100, 
        popsize=15, 
        tol=1e-6, 
        mutation=(0.5, 1), 
        recombination=0.8, 
        seed=seed
    )
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        # If time is exceeded, return the best solution found so far (result.x)
        h_opt = result.x
        dx = 2.0 / n_points
        one_minus_h = 1.0 - h_opt
        conv = np.correlate(h_opt, one_minus_h, mode='full')
        c5_bound = np.max(conv) * dx
        return h_opt, c5_bound, n_points
    else:
        # Compute the final C5 value
        h_opt = result.x
        dx = 2.0 / n_points
        one_minus_h = 1.0 - h_opt
        conv = np.correlate(h_opt, one_minus_h, mode='full')
