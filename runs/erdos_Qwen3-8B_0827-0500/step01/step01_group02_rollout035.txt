from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    from scipy.optimize import minimize

    # Predefined initial guess (global variable)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def constraint(h):

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Generate diverse initial guesses

        guesses = [base_guess.copy()]
        for _ in range(num_extra):
            # Random initial guess with sum constraint

            h_rand_sum = np.sum(h_rand)

            guesses.append(h_rand)

    # Generate multiple initial guesses
    initial_guesses = generate_initial_guesses(initial_h, num_extra=4)

    max_iterations = 100
    num_restarts = len(initial_guesses)

    # Run optimization for each initial guess
    for i, h_initial in enumerate(initial_guesses):
        # Use SLSQP for constrained optimization
        result = minimize(
