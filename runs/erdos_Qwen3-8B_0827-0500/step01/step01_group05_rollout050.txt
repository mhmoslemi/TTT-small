import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        return compute_C5(h)
    
    def constraint_sum(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum
    
    # Initial guess
    h_flat = h.copy()
    
    # Optimization settings
    max_iter = 1000
    time_start = time.time()
    
    # Use L-BFGS-B optimizer for better convergence
    result = minimize(
        objective,
        h_flat,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={
            'maxiter': max_iter,
            'ftol': 1e-8,
            'gtol': 1e-8,
            'disp': False
        }
    )
    
    # Check time budget
    elapsed = time.time() - time_start
    if elapsed > budget_s:
        return h, compute_C5(h), n_points
    
    # Return the best solution found
    h_opt = result.x
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * (target_sum / np.sum(h_opt)) if np.sum(h_opt) != 0 else h_opt
    return h_opt, compute_C5(h_opt), n_points
