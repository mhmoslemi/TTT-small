import numpy as np
from scipy.optimize import minimize
from scipy.stats import dirichlet
import time

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is a global variable with n_points=42
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate multiple initial guesses
    np.random.seed(seed)
    num_initial_guesses = 5
    initial_guesses = []
    for _ in range(num_initial_guesses):
        # Dirichlet distribution to generate valid initial guesses
        alpha = [1.0] * n_points
        sample = dirichlet.rvs(alpha, size=1)[0]
        sample = sample * target_sum
        initial_guesses.append(sample)
    # Include the initial guess provided as well
    initial_guesses.append(initial_h_values)
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            conv = np.correlate(h, one_minus_h, mode='full')
            return np.max(conv) * dx
        
        # Constraint function
        def constraint(h):
            return np.sum(h) - target_sum
        
        # Optimization setup
        cons = ({'type': 'eq', 'fun': constraint})
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run optimization
        result = minimize(objective, x0=guess, method='L-BFGS-B', constraints=cons, bounds=bounds,
                          tol=1e-6, options={'maxiter': 200})
        
        # Check success and update best solution
        if result.success:
            h_opt = result.x
            one_minus_h = 1.0 - h_opt
            conv = np.correlate(h_opt, one_minus_h, mode='full')
            c5 = np.max(conv) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
    
    return best_h, best_c5, n_points
