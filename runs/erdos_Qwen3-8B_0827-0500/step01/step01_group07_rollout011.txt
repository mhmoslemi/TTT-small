import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 42  # Use the same number of points as the initial guess
    target_sum = n_points / 2.0
    
    # Generate a random initial guess with correct sum
    h_rand = np.random.rand(n_points)
    h_rand = h_rand * target_sum / np.sum(h_rand)

    def objective(h):
        dx = 2.0 / n_points
        one_minus_h = 1.0 - h
        conv = np.correlate(h, one_minus_h, mode='full')
        c5 = np.max(conv) * dx
        return c5
    
    # Constraint: sum of h must equal target_sum
    def constraint(h):
        return np.sum(h) - target_sum
    
    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run optimization with tighter tolerances and more iterations
    result = minimize(
        objective, 
        x0=h_rand, 
        method='SLSQP',
