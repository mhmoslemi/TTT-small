        max_corr = np.max(corr)
        dx = 2.0 / n_points
        return max_corr * dx
    
    # Define the constraint
    A_eq = np.ones(n_points)
    b_eq = n_points / 2
    cons = [LinearConstraint(A_eq, b_eq, b_eq)]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Try different optimization methods
    methods = ['SLSQP', 'L-BFGS-B', 'COBYLA']
    
    for guess in initial_guesses:
        # Try different methods
        for method in methods:
            # Check if time is up
            if time.time() - start_time > budget_s - 1:  # leave some margin
                break
            # Run the optimization
            result = minimize(objective, guess, method=method, bounds=bounds, constraints=cons)
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
    
    if best_h is None:
        best_h = initial_h_values
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
