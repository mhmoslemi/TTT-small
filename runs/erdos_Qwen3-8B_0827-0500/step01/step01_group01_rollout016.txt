def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    start_time = time.time()
    
    # Optimization parameters
    max_iterations = 100
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        # Compute current C5
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Define objective function
        def objective(x):
            h_new = x
            cross_corr = np.correlate(h_new, 1 - h_new, mode='full')
            return np.max(cross_corr) * dx
        
        # Constraints
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Optimization
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = res.x
        else:
            break
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    cross_corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(cross_corr)
    return max_corr * dx
