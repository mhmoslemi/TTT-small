import numpy as np
import time
import scipy.optimize as opt
from scipy.optimize import Bounds

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    # Define objective function
    def objective(x):
        h_current = x
        cross_corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx
    # Define constraints
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    # Initial guess
    x0 = h.copy()
    # Track time
    start_time = time.time()
    # Run optimization
    result = opt.minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-6,
        options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    # Check time and return best result
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        return h, compute_c5(h, dx), n_points
    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    cross_corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(cross_corr)
    return max_corr * dx
