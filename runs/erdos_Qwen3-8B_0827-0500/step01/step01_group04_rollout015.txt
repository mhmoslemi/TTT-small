            return max_corr * dx
        
        A_eq = np.ones(n_points)
        b_eq = n_points / 2
        cons = [LinearConstraint(A_eq, b_eq, b_eq)]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Use SLSQP with tighter tolerances
        result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                elapsed_time += result.message.split()[-2]  # Approximate time based on message
                if elapsed_time >= budget_s:
                    break
    
    # Fallback to initial guess if no improvement
    if best_h is None:
        best_h = np.array(initial_h_values)
        best_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), best_h)
        best_h = np.clip(best_h, 0.0, 1.0)
        target_sum = n_points / 2
        current_sum = np.sum(best_h)
        best_h *= target_sum / current_sum
        best_h = np.clip(best_h, 0.0, 1.0)
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
