import numpy as np
from scipy.optimize import minimize, LinearConstraint
from scipy.optimize import Bounds
import time

def generate_random_initial(n_points):
    h = np.random.rand(n_points)
    target_sum = n_points / 2
    current_sum = np.sum(h)
    scaling_factor = target_sum / current_sum
    h *= scaling_factor
    h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)

    start_time = time.time()

    initial_guesses = [initial_h_values]
    for _ in range(5):  # Generate 5 random initial guesses
        random_initial = generate_random_initial(n_points)
        initial_guesses.append(random_initial)
    
    best_h = None
    best_c5 = float('inf')
    
    for guess in initial_guesses:
        # Define objective function
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            dx = 2.0 / n_points
            return max_corr * dx
        
        # Define constraints
        A_eq = np.ones(n_points)
        b_eq = n_points / 2
        cons = [LinearConstraint(A_eq, b_eq, b_eq)]
        bounds = Bounds(0.0, 1.0)
        
        # Use SLSQP with tighter tolerances
        result = minimize(
            objective,
            guess,
            method='SLSQP',

            constraints=cons,
            tol=1e-8,
            options={'maxiter': 1000}
        )
