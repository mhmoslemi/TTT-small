    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    with ProcessPoolExecutor(max_workers=10) as executor:
        futures = []
        for guess in initial_guesses:
            future = executor.submit(optimization_task, guess, n_points)
            futures.append(future)
        
        for future in as_completed(futures):
            try:
                result_h, result_c5 = future.result()
                if result_c5 < best_c5:
                    best_c5 = result_c5
                    best_h = result_h
            except Exception as e:
                print(f"Error in optimization task: {e}")
            
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break
    
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
