import numpy as np
import time
from scipy.optimize import minimize, LinearConstraint

def generate_perturbed_initial(h, noise_level=0.1):
    perturbed = h + noise_level * np.random.randn(len(h))
    perturbed = np.clip(perturbed, 0.0, 1.0)
    return perturbed

def generate_structured_initial(n_points, num_pulses, spacing=1.0):
    structured = np.zeros(n_points)
    pulse_indices = np.round(np.linspace(0, n_points - 1, num_pulses)).astype(int)
    structured[pulse_indices] = 1.0
    target_sum = n_points / 2
    current_sum = np.sum(structured)
    scaling_factor = target_sum / current_sum
    structured *= scaling_factor
    return structured

def generate_random_initial(n_points):
    h = np.random.rand(n_points)
    h = h / np.sum(h) * (n_points / 2)
    return h
