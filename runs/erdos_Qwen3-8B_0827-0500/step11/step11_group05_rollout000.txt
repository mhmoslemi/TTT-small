    perturbed = np.clip(perturbed, 0.0, 1.0)
    return perturbed

def generate_structured_initial(n_points, num_pulses):
    structured = np.zeros(n_points)
    pulse_indices = np.linspace(0, n_points - 1, num_pulses, endpoint=False).astype(int)
    structured[pulse_indices] = 1.0
    target_sum = n_points / 2
    current_sum = np.sum(structured)
    scaling_factor = target_sum / current_sum
    structured *= scaling_factor
    return structured

def generate_random_initial(n_points):
    random_initial = np.random.rand(n_points)
    target_sum = n_points / 2
    current_sum = np.sum(random_initial)
    scaling_factor = target_sum / current_sum
    random_initial *= scaling_factor
    random_initial = np.clip(random_initial, 0.0, 1.0)
    return random_initial

def compute_c5(h, dx):
    n_points = len(h)
    max_c5 = 0.0
    for k in range(n_points):
        # Positive shift k
        valid_x = range(0, n_points - k)
        integral_pos = np.sum(h[valid_x] * (1 - h[valid_x + k])) * dx
        # Negative shift k
        valid_x = range(k, n_points)
        integral_neg = np.sum(h[valid_x] * (1 - h[valid_x - k])) * dx
        current_max = max(integral_pos, integral_neg)
        if current_max > max_c5:
            max_c5 = current_max
    return max_c5

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Generate diverse initial guesses
    initial_guesses = [initial_h_values]
    # Add perturbations
    for noise_level in [0.01, 0.03, 0.05, 0.1]:
        initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level))
    # Add structured initial guesses
    for num_pulses in [2, 3, 4, 5, 6, 7, 8, 9, 10, 12, 14, 16, 18, 20]:
        initial_guesses.append(generate_structured_initial(n_points, num_pulses))
    # Add random initial guesses
    for _ in range(8):
        initial_guesses.append(generate_random_initial(n_points))
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        # Define objective function
        def objective(h):
            return compute_c5(h, dx)
        
        # Constraints
        A_eq = np.ones(n_points)
        b_eq = n_points / 2
        cons = [LinearConstraint(A_eq, b_eq, b_eq)]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization settings
        result = minimize(objective, guess, method='Trust-Constr', bounds=bounds, constraints=cons,
                          tol=1e-9, options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False})
        
        # Check if optimization succeeded
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = compute_c5(best_h, dx)
    
    return (best_h, best_c5, n_points)
