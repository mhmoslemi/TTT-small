import numpy as np
import time
from scipy.optimize import minimize, LinearConstraint
from joblib import Parallel, delayed

def generate_perturbed_initial(h, noise_level=0.05):
    perturbed = h + noise_level * np.random.randn(len(h))
    perturbed = np.clip(perturbed, 0.0, 1.0)
    return perturbed

def generate_structured_initial(n_points, num_pulses):
    structured = np.zeros(n_points)
    pulse_indices = np.linspace(0, n_points - 1, num_pulses, endpoint=False).astype(int)
    structured[pulse_indices] = 1.0
    target_sum = n_points / 2
    current_sum = np.sum(structured)
    scaling_factor = target_sum / current_sum
    structured *= scaling_factor
    return structured

def generate_random_initial(n_points):
    random_initial = np.random.rand(n_points)
    target_sum = n_points / 2
    current_sum = np.sum(random_initial)
    scaling_factor = target_sum / current_sum
    random_initial *= scaling_factor
    random_initial = np.clip(random_initial, 0.0, 1.0)
    return random_initial

def evaluate_erdos_solution(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def optimize_h(h_initial, n_points, dx, seed):
    np.random.seed(seed)
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    A_eq = np.ones(n_points)
    b_eq = n_points / 2
    cons = [LinearConstraint(A_eq, b_eq, b_eq)]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=cons,
                      tol=1e-10, options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
    return result.x, result.fun

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    initial_guesses = [initial_h_values]
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.005))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.02))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.04))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.06))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.08))
    
    for num_pulses in range(2, 21):  # Increase pulse count range
        initial_guesses.append(generate_structured_initial(n_points, num_pulses))

        initial_guesses.append(generate_random_initial(n_points))
    
    best_h = initial_h_values
    best_c5 = evaluate_erdos_solution(initial_h_values)
    start_time = time.time()
    
    # Use parallel processing to run optimizations in parallel
    results = Parallel(n_jobs=10, backend='threading')(
