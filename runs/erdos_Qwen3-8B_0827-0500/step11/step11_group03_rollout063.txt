    initial_guesses.append(generate_structured_initial(n_points, 10))
    initial_guesses.append(generate_structured_initial(n_points, 15))
    initial_guesses.append(generate_structured_initial(n_points, 20))
    initial_guesses.append(generate_structured_initial(n_points, 25))
    initial_guesses.append(generate_structured_initial(n_points, 30))
    initial_guesses.append(generate_structured_initial(n_points, 35))
    initial_guesses.append(generate_structured_initial(n_points, 40))
    initial_guesses.append(generate_random_initial(n_points))
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        def objective(h):
            corr = correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            penalty = 1e4 * np.abs(np.sum(h) - n_points / 2)
            return max_corr * dx + penalty
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        try:
            result = differential_evolution(
                objective, bounds, strategy='best1bin',
                maxiter=100, popsize=15, tol=1e-10,
                mutation=(0.5, 1), recombination=0.8, disp=False
            )
            current_c5 = result.fun - 1e4 * np.abs(np.sum(result.x) - n_points / 2)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        except:
            pass
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
