    best_h = initial_guess
    best_c5 = evaluate_erdos_solution(initial_guess)

    # Define a helper function to run DE with a given population size
    def run_de(popsize=15, maxiter=100):
        result = differential_evolution(
            objective, bounds, strategy='best1bin',
            maxiter=maxiter, popsize=popsize, tol=1e-10,
            mutation=(0.5, 1), recombination=0.8, seed=seed
        )
        h = result.x
        c5 = result.fun - 1e4 * np.abs(np.sum(h) - target_sum)
        return h, c5

    # Try multiple DE runs with different parameters
    for popsize in [15, 20]:
        for maxiter in [100, 150]:
            h, c5 = run_de(popsize, maxiter)
            if c5 < best_c5:
                best_h = h
                best_c5 = c5
            # Early exit if time is exhausted
            if time.time() - start_time >= budget_s:
                break

    # Fallback to initial guess if no improvement found
    if best_c5 > evaluate_erdos_solution(initial_guess):
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)

    return (best_h, best_c5, n_points)
