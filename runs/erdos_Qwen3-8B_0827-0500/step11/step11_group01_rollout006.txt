import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        penalty = 1e6 * abs(np.sum(h) - target_sum)
        return max_corr * dx + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Generate diverse initial population
    initial_population = [initial_h_values]
    # Add perturbations
    for _ in range(5):
        perturbed = initial_h_values + 0.05 * np.random.randn(n_points)
        perturbed = np.clip(perturbed, 0.0, 1.0)
        initial_population.append(perturbed)
    # Add structured guesses
    for num_pulses in [3, 5, 7, 10]:

        pulse_indices = np.random.choice(n_points, size=num_pulses, replace=False)
        structured[pulse_indices] = 1.0
        current_sum = np.sum(structured)
        scaling_factor = target_sum / current_sum
        structured *= scaling_factor
        initial_population.append(structured)
    
    # Run differential evolution
    result = differential_evolution(

        bounds, 
        popsize=20, 
        strategy='best1bin', 
        maxiter=100, 
        tol=1e-10, 
        mutation=(0.5, 1), 
        recombination=0.8, 
        seed=seed, 
        disp=False
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
