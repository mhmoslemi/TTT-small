import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Define objective function with penalty for constraint
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        penalty = 1e5 * np.abs(np.sum(h) - n_points / 2)
        return max_corr * dx + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution with multiple strategies
    result = differential_evolution(
        objective, bounds, 
        strategy='best1bin', 
        popsize=20, 
        maxiter=100, 
        tol=1e-10,
        mutation=(0.5, 1.0),
        recombination=0.8
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
