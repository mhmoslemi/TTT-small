                                       popsize=15, mutation=(0.5, 1.0), recombination=0.8, 
                                       tol=1e-10, maxiter=100, seed=seed)
        
        # Check if optimization succeeded
        if result.success:
            current_c5 = result.fun - 1e6 * np.abs(np.sum(result.x) - n_points / 2)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
