    target_sum = n_points / 2
    
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        penalty = 1e3 * (np.sum(h) - target_sum) ** 2
        return max_corr * dx + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    result = differential_evolution(
        objective, bounds, strategy='best1bin', popsize=15, tol=1e-10,
        mutation=(0.5, 1), recombination=0.8, maxiter=100, polish=True
    )
    elapsed = time.time() - start_time
