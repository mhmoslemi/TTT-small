import numpy as np
import time
from scipy.optimize import minimize, LinearConstraint
from scipy.optimize import differential_evolution

def generate_perturbed_initial(h, noise_level=0.05):
    perturbed = h + noise_level * np.random.randn(len(h))
    perturbed = np.clip(perturbed, 0.0, 1.0)
    return perturbed

def generate_structured_initial(n_points, num_pulses, spacing=1.0):
    structured = np.zeros(n_points)
    pulse_indices = np.round(np.linspace(0, n_points - 1, num_pulses)).astype(int)
    structured[pulse_indices] = 1.0
    target_sum = n_points / 2
    current_sum = np.sum(structured)
    scaling_factor = target_sum / current_sum
    structured *= scaling_factor
    return structured

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate diverse initial guesses
    initial_guesses = [initial_h_values]
    for num_pulses in [5, 7, 10, 12, 15]:
        initial_guesses.append(generate_structured_initial(n_points, num_pulses, spacing=1.2))
    for noise_level in [0.01, 0.03, 0.05]:
        initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=noise_level))
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        # Define objective function
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Constraints
        A_eq = np.ones(n_points)
        b_eq = target_sum
        cons = [LinearConstraint(A_eq, b_eq, b_eq)]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization settings
        result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=cons,
                          tol=1e-12, options={'maxiter': 800, 'ftol': 1e-12, 'disp': False})
        
        # Check if optimization succeeded
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
