import time
from scipy.optimize import minimize, LinearConstraint

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Generate diverse initial guesses
    initial_guesses = [initial_h_values]
    # Add perturbations
    for noise_level in [0.01, 0.03, 0.05]:
        perturbed = initial_h_values + noise_level * np.random.randn(n_points)
        perturbed = np.clip(perturbed, 0.0, 1.0)
        initial_guesses.append(perturbed)
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        # Define objective function
        def objective(h):

        b_eq = n_points / 2
        cons = [LinearConstraint(A_eq, b_eq, b_eq)]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization settings
        result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=cons,
                          tol=1e-8, options={'maxiter': 300, 'ftol': 1e-12, 'disp': False})

        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:

        if elapsed >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = compute_c5(best_h, dx)
