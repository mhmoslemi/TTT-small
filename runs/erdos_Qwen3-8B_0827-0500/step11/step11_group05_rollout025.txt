import numpy as np
import time
from scipy.optimize import minimize, LinearConstraint

def compute_c5(h, dx):
    n_points = len(h)
    max_c5 = 0.0
    for k in range(n_points):
        # Positive shift k
        if k < n_points:
            valid_x = h[:n_points - k]
            valid_1_minus_h = 1.0 - h[k:]
            integral_pos = np.dot(valid_x, valid_1_minus_h) * dx
        else:
            integral_pos = 0.0
        # Negative shift k
        if k < n_points:
            valid_x = h[k:]
            valid_1_minus_h = 1.0 - h[:n_points - k]
            integral_neg = np.dot(valid_x, valid_1_minus_h) * dx
        else:
            integral_neg = 0.0
        current_max = max(integral_pos, integral_neg)
        if current_max > max_c5:
            max_c5 = current_max
    return max_c5
