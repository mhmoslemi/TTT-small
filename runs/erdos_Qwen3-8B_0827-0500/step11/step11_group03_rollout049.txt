    h = dirichlet_samples[0] * (n_points / 2)
    return h

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Generate diverse initial guesses
    initial_guesses = [initial_h_values]
    # Add perturbed versions
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.01))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.03))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.05))
    # Add structured initial guesses
    for num_pulses in [5, 10, 15, 20, 25, 30, 35, 40]:
        initial_guesses.append(generate_structured_initial(n_points, num_pulses))
    # Add random initial guesses
    for _ in range(5):
        initial_guesses.append(generate_random_initial(n_points))
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        # Define objective function with penalty
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            penalty = 1e6 * abs(np.sum(h) - n_points / 2)
            return max_corr * dx + penalty
        
        # Bounds for differential evolution
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution
        result = differential_evolution(objective, bounds, strategy='best1bin', 
                                       maxiter=100, popsize=15, tol=1e-10, mutation=(0.5, 1), 
                                       recombination=0.8, polish=True, disp=False)
        
        # Check if optimization succeeded
        if result.success:
            # Calculate the actual C5 value by subtracting the penalty
            # Assuming the penalty is negligible
            current_c5 = result.fun - 1e6 * abs(np.sum(result.x) - n_points / 2)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
