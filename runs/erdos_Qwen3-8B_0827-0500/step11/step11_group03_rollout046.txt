        def objective_slsqp(h):
            corr = correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        cons = [({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})]
        bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
        
        result_slsqp = minimize(objective_slsqp, result_de.x, method='SLSQP', 
                               bounds=bounds_slsqp, constraints=cons, 
                               tol=1e-10, options={'maxiter': 500, 'ftol': 1e-12, 'disp': False})
        
        # Check if optimization succeeded
        if result_slsqp.success:
            current_c5 = result_slsqp.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
