import numpy as np
import time
from scipy.optimize import minimize, LinearConstraint
from multiprocessing import Pool, cpu_count

def generate_perturbed_initial(h, noise_level=0.05):
    perturbed = h + noise_level * np.random.randn(len(h))
    perturbed = np.clip(perturbed, 0.0, 1.0)
    return perturbed

def generate_structured_initial(n_points, num_pulses):
    structured = np.zeros(n_points)
    pulse_indices = np.linspace(0, n_points - 1, num_pulses, endpoint=False).astype(int)
    structured[pulse_indices] = 1.0
    target_sum = n_points / 2
    current_sum = np.sum(structured)
    scaling_factor = target_sum / current_sum
    structured *= scaling_factor
    return structured

def generate_random_initial(n_points):
    random_initial = np.random.rand(n_points)
    target_sum = n_points / 2
    current_sum = np.sum(random_initial)
    scaling_factor = target_sum / current_sum
    random_initial *= scaling_factor
    random_initial = np.clip(random_initial, 0.0, 1.0)
    return random_initial

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def objective(h, dx):
    return compute_c5(h, dx)

def optimize_h(guess, dx, n_points):
    A_eq = np.ones(n_points)
    b_eq = n_points / 2
    cons = [LinearConstraint(A_eq, b_eq, b_eq)]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = minimize(lambda h: objective(h, dx), guess, method='SLSQP', bounds=bounds, constraints=cons,
                      tol=1e-9, options={'maxiter': 200, 'ftol': 1e-12, 'disp': False})
    return result.x, result.fun if result.success else None

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    initial_guesses = [initial_h_values]
    for noise_level in [0.01, 0.03, 0.05, 0.1]:
        initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level))
    for num_pulses in [2, 3, 4, 5, 6, 7, 8, 9, 10, 12, 14, 16, 18, 20]:
        initial_guesses.append(generate_structured_initial(n_points, num_pulses))
    for _ in range(8):
        initial_guesses.append(generate_random_initial(n_points))
    
    best_h = initial_h_values
    best_c5 = compute_c5(best_h, dx)
    start_time = time.time()
    
    with Pool(processes=cpu_count()) as pool:
        results = []
        for guess in initial_guesses:
            if time.time() - start_time >= budget_s:
                break
            results.append(pool.apply_async(optimize_h, (guess, dx, n_points)))
        
        for res in results:
            h_opt, c5_val = res.get()
            if c5_val is not None and c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt
