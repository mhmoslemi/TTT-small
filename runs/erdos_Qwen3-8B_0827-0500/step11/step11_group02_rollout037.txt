            max_corr = np.max(corr)
            dx = 2.0 / n_points
            return max_corr * dx
        
        # Bounds for each variable
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution
        result = differential_evolution(objective, bounds, strategy='best1bin', 
                                       popsize=15, maxiter=100, tol=1e-10, 
                                       mutation=(0.5, 1), recombination=0.8, 
                                       disp=False)
        
        # Check if optimization succeeded
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
