        bounds = [(0.0, 1.0) for _ in range(n_points)]
        # Run differential evolution with initial guess
        result = differential_evolution(
            objective, bounds, strategy='best1bin',
            maxiter=100, popsize=15, tol=1e-10,
            mutation=(0.5, 1), recombination=0.8,
            x0=guess, seed=seed
        )
        
        # Check if optimization succeeded
        if result.success:
            current_sum = np.sum(result.x)
            penalty = 1e4 * (current_sum - target_sum)**2
            current_c5 = result.fun - penalty
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return (best_h, best_c5, n_points)
