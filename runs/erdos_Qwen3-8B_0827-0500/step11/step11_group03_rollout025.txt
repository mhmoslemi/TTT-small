        return max_corr * dx
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    initial_guess = initial_h_values.copy()
    
    # Run differential evolution with adjusted parameters
    result = differential_evolution(
        objective, bounds, 
        strategy='best1bin', 
        popsize=15, 
        mutation=(0.5, 1.0), 
        recombination=0.8, 
        tol=1e-10, 
        maxiter=100, 
        seed=seed
    )
    
    best_h = result.x
    best_c5 = result.fun
    
    return (best_h, best_c5, n_points)
