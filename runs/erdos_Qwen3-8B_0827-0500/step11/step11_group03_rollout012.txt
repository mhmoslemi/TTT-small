    initial_guesses.append(generate_structured_initial(n_points, 30))
    initial_guesses.append(generate_structured_initial(n_points, 35))
    initial_guesses.append(generate_structured_initial(n_points, 40))
    initial_guesses.append(generate_random_initial(n_points))  # Add random initial guess
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        # Define objective function using numpy correlate
        def objective(h):
            corr = correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Constraints
        A_eq = np.ones(n_points)
        b_eq = n_points / 2
        cons = [({'type': 'eq', 'fun': lambda h: np.sum(h) - b_eq})]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Try differential evolution for some guesses
        if np.random.rand() < 0.5:  # 50% chance to use DE
            # Use differential evolution with penalty for constraint
            def objective_de(h):
                corr = correlate(h, 1 - h, mode='full')
                max_corr = np.max(corr)
                penalty = 1e4 * (np.sum(h) - n_points / 2)**2
                return max_corr * dx + penalty
            
            result = differential_evolution(objective_de, bounds, strategy='best1bin', 
                                           popsize=15, maxiter=100, tol=1e-6, 
                                           disp=False)
            # Refine with SLSQP
            result_slsqp = minimize(objective, result.x, method='SLSQP', bounds=bounds, 
                                   constraints=cons, tol=1e-10, options={'maxiter': 500, 'ftol': 1e-12, 'disp': False})
            current_c5 = result_slsqp.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x
        else:
            # Use SLSQP
            result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=cons,
                              tol=1e-10, options={'maxiter': 500, 'ftol': 1e-12, 'disp': False})
            # Check if optimization succeeded
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
