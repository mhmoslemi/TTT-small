            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Constraints
        A_eq = np.ones(n_points)
        b_eq = n_points / 2
        cons = [LinearConstraint(A_eq, b_eq, b_eq)]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization settings
        # Randomly choose between SLSQP and differential evolution
        if np.random.rand() < 0.5:
            # Use differential evolution with penalty for sum constraint
            def objective_with_penalty(h):
                corr = np.correlate(h, 1 - h, mode='full')
                max_corr = np.max(corr)
                penalty = 1e6 * np.abs(np.sum(h) - n_points / 2)
                return max_corr * dx + penalty
            
            result = minimize(objective_with_penalty, guess, method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False})
        else:
            result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=cons,
                              tol=1e-10, options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False})
        
        # Check if optimization succeeded
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
