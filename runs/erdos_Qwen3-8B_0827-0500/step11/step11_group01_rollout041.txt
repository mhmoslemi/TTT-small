import numpy as np
import time
from scipy.optimize import minimize, LinearConstraint
from concurrent.futures import ThreadPoolExecutor, as_completed

def generate_perturbed_initial(h, noise_level=0.05):
    perturbed = h + noise_level * np.random.randn(len(h))
    perturbed = np.clip(perturbed, 0.0, 1.0)
    target_sum = len(h) / 2
    current_sum = np.sum(perturbed)
    scaling_factor = target_sum / current_sum
    perturbed *= scaling_factor
    return perturbed

def generate_structured_initial(n_points, num_pulses):
    structured = np.zeros(n_points)
    pulse_indices = np.random.choice(n_points, size=num_pulses, replace=False)
    structured[pulse_indices] = 1.0
    target_sum = n_points / 2
    current_sum = np.sum(structured)
    scaling_factor = target_sum / current_sum
    structured *= scaling_factor
    return structured

def optimization_task(guess, n_points, dx):
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    A_eq = np.ones(n_points)
    b_eq = n_points / 2
    cons = [LinearConstraint(A_eq, b_eq, b_eq)]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = minimize(objective, guess, method='trust-constr', bounds=bounds, constraints=cons,
                      tol=1e-10, options={'maxiter': 500, 'ftol': 1e-12, 'disp': False})
    
    if result.success:
        return (result.fun, result.x)
