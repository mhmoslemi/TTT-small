        current_sum = np.sum(initial_guesses[-1])
        scaling_factor = target_sum / current_sum
        initial_guesses[-1] *= scaling_factor
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        # Define objective function
        def objective(h):
            return compute_c5(h, dx)
        
        # Constraints
        A_eq = np.ones(n_points)
        b_eq = n_points / 2
        cons = [({'type': 'eq', 'fun': lambda h: np.sum(h) - b_eq})]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization settings
        result = minimize(objective, guess, method='COBYLA', bounds=bounds, constraints=cons,
                          tol=1e-6, options={'maxiter': 200, 'disp': False})
        
        # Check if optimization succeeded
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = compute_c5(best_h, dx)
    
    return (best_h, best_c5, n_points)
