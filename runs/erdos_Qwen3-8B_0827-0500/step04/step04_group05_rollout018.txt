import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    # Load initial guess
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Define the objective function
    def objective(h):
        cross_corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    # Define constraint
    def constraint(h):
        return np.sum(h) - n_points / 2

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Function to generate a perturbed guess
    def perturb(h, seed=None):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        return np.clip(h + perturbation, 0.0, 1.0)

    # Function to normalize h to meet the sum constraint
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)

    # Initialize best solution
    best_h = initial_h.copy()
    best_c5 = float('inf')

    # Number of restarts and iterations
    num_restarts = 50  # Reduced for efficiency
    max_iter = 100     # Increased for deeper optimization

    # Optimization settings
    method = 'L-BFGS-B'  # More efficient for bounded problems

    for i in range(num_restarts):
        # Perturb the current best solution
        h_initial = perturb(best_h, seed=seed + i)
        h_initial = normalize(h_initial)  # Ensure sum constraint

        # Perform optimization
        result = minimize(
            objective,
            h_initial,
            method=method,
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': max_iter, 'ftol': 1e-8, 'eps': 1e-8}
        )

        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # Fallback to perturbed guess if optimization fails
            current_c5 = objective(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_initial

    return best_h, best_c5, n_points
