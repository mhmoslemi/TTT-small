from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        cross_corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        penalty = 1e-6 * (np.sum(h) - n_points / 2)**2
        return max_corr * dx + penalty

    def constraint(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def random_initial_guess():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        return h

    def perturb(h, seed=None):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.05, 0.05, size=len(h))
        h_perturbed = np.clip(h + perturbation, 0.0, 1.0)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        return h_perturbed

    best_c5 = float('inf')
    best_h = initial_h.copy()

    num_restarts = 150
    maxiter = 250

    for i in range(num_restarts):
        if i % 3 == 0:
            h_initial = random_initial_guess()

            h_initial = perturb(best_h, seed=seed + i)
        
        result = minimize(
            objective, 
            h_initial, 
            method='SLSQP', 
            bounds=bounds, 
            constraints={'type': 'eq', 'fun': constraint},
