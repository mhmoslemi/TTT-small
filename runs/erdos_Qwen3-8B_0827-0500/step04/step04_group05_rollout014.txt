import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is a global variable
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Define the objective function
    def objective(h_candidate):
        # Normalize to meet the sum constraint
        h = h_candidate * (n_points / 2) / np.sum(h_candidate)
        # Compute cross-correlation
        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run differential evolution with multiple restarts
    best_c5 = float('inf')
    best_h = initial_h.copy()

    # Parameters for differential evolution
    strategy = 'best1bin'
    popsize = 15
    mutation = (0.5, 1)
    recombination = 0.8
    tol = 1e-8
    maxiter = 100
    num_restarts = 5  # Number of independent runs

    for i in range(num_restarts):
        # Generate a random initial guess based on initial_h
        np.random.seed(seed + i)
        initial_population = np.random.rand(popsize, n_points)
        initial_population = initial_population * (n_points / 2) / np.sum(initial_population, axis=1, keepdims=True)
        initial_population = np.vstack([initial_population, initial_h])
        initial_population = np.vstack([initial_population, np.random.rand(n_points) * (n_points / 2) / np.sum(np.random.rand(n_points))])

        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy=strategy,
            popsize=popsize,
            mutation=mutation,
            recombination=recombination,
            tol=tol,
            maxiter=maxiter,
            seed=seed + i,
            x0=initial_population
        )

        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # Fallback to evaluating the initial population
            for candidate in initial_population:
                h = candidate * (n_points / 2) / np.sum(candidate)
                cross_corr = np.correlate(h, 1 - h, mode='full')
                current_c5 = np.max(cross_corr) * dx
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = candidate

    return best_h, best_c5, n_points
