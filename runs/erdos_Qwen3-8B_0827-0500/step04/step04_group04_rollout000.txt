import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        cross_corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def random_initial_guess():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        return h

    def perturb(h, seed=None):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.2, 0.2, size=len(h))
        return np.clip(h + perturbation, 0.0, 1.0)

    best_c5 = float('inf')
    best_h = initial_h.copy()

    num_restarts = 200  # Increased number of restarts
    max_iter = 200      # Increased number of iterations per optimization
    ftol = 1e-8        # Tighter tolerance
    eps = 1e-8         # Tighter step size

    for i in range(num_restarts):
        # Generate a random initial guess
        h_initial = random_initial_guess()
        # Run optimization
        result = minimize(
            objective, 
            h_initial, 
            method='SLSQP',

            options={'maxiter': max_iter, 'ftol': ftol, 'eps': eps}
