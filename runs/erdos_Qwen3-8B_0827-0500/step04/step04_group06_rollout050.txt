                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint},
                options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-9}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
            else:
                # Fallback to evaluating the perturbed guess directly
                current_c5 = objective(perturbed)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = perturbed

    return best_h, best_c5, n_points
