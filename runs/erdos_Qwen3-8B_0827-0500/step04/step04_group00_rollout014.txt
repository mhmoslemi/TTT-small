import numpy as np
from scipy.optimize import minimize, LinearConstraint
from scipy.interpolate import interp1d
from multiprocessing import Pool, cpu_count

def generate_perturbed_initial(h, noise_level=0.05):
    perturbed = h + noise_level * np.random.randn(len(h))
    perturbed = np.clip(perturbed, 0.0, 1.0)
    return perturbed

def generate_structured_initial(n_points, num_pulses):
    structured = np.zeros(n_points)
    pulse_indices = np.linspace(0, n_points - 1, num_pulses, endpoint=False).astype(int)
    structured[pulse_indices] = 1.0
    target_sum = n_points / 2
    current_sum = np.sum(structured)
    scaling_factor = target_sum / current_sum
    structured *= scaling_factor
    return structured

def generate_random_initial(n_points):
    random_initial = np.random.rand(n_points)
    target_sum = n_points / 2
    current_sum = np.sum(random_initial)
    scaling_factor = target_sum / current_sum
    random_initial *= scaling_factor
    random_initial = np.clip(random_initial, 0.0, 1.0)
    return random_initial

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    
    # Generate diverse initial guesses
    initial_guesses = [initial_h_values]
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.03))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.05))
    for num_pulses in [3, 4, 5, 6, 7]:
        initial_guesses.append(generate_structured_initial(n_points, num_pulses))
    initial_guesses.append(generate_random_initial(n_points))
    
    # Function to run optimization for a single guess
    def optimize_guess(guess):
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            dx = 2.0 / n_points
            return max_corr * dx
        
        A_eq = np.ones(n_points)
        b_eq = n_points / 2
        cons = [LinearConstraint(A_eq, b_eq, b_eq)]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=cons,
                          tol=1e-12, options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False})
        return result
    
    pool = Pool(processes=min(cpu_count(), len(initial_guesses)))
    results = []
    best_h = None
    best_c5 = float('inf')
    start_time = np.datetime64('now')
    
    # Submit all initial guesses to the pool
    jobs = [pool.apply_async(optimize_guess, (guess,)) for guess in initial_guesses]
    
    # Collect results as they finish
    while jobs:
        for job in jobs:
            if job.ready():
                result = job.get()
                if result.success:
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result.x
                jobs.remove(job)
        elapsed = np.datetime64('now') - start_time
        if elapsed.total_seconds() >= budget_s:
            break
    
    pool.close()
    pool.join()
    
    if best_h is None:
        best_h = initial_h_values
        best_c5 = np.correlate(best_h, 1 - best_h, mode='full').max() * (2.0 / n_points)
    
    return (best_h, best_c5, n_points)
