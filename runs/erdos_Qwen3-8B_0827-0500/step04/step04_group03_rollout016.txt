    best_c5 = float('inf')
    elapsed_time = 0.0
    
    with Pool(processes=min(cpu_count(), len(initial_guesses))) as pool:
        results = [pool.apply_async(optimization_task, (guess, n_points)) for guess in initial_guesses]
        
        while results:
            result = results.pop(0).get()
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
            elapsed_time += 0.1  # Approximate time tracking
            if elapsed_time >= budget_s:
                break
    
    if best_h is None:
        best_h = initial_h_values
        best_c5 = objective_function(best_h, n_points)
    
    return (best_h, best_c5, n_points)
