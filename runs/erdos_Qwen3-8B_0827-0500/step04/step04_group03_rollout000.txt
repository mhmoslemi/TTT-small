    initial_guesses.append(generate_structured_initial(n_points, num_pulses=4))
    initial_guesses.append(generate_structured_initial(n_points, num_pulses=5))
    initial_guesses.append(np.random.rand(n_points))
    
    # Normalize random initial guess
    random_initial = initial_guesses[-1]
    target_sum = n_points / 2
    current_sum = np.sum(random_initial)
    scaling_factor = target_sum / current_sum
    random_initial *= scaling_factor
    random_initial = np.clip(random_initial, 0.0, 1.0)
    initial_guesses[-1] = random_initial
    
    best_h = None
    best_c5 = float('inf')
    start_time = np.datetime64('now')
    
    for guess in initial_guesses:
        # Define objective function
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            dx = 2.0 / n_points
            return max_corr * dx
        
        # Constraints
        A_eq = np.ones(n_points)
        b_eq = n_points / 2
        cons = [LinearConstraint(A_eq, b_eq, b_eq)]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization settings
        result = minimize(objective, guess, method='L-BFGS-B', bounds=bounds, constraints=cons,
                          tol=1e-10, options={'maxiter': 1000, 'ftol': 1e-10, 'gtol': 1e-10})
        
        # Check if optimization succeeded
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Track time usage
        elapsed_time = np.datetime64('now') - start_time
        if np.isscalar(elapsed_time) and elapsed_time.total_seconds() >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
