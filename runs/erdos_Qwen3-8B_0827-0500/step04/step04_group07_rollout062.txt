import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is a global variable
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

        return max_corr * dx

    # Define constraints

        return np.sum(h) - n_points / 2

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run differential evolution with multiple restarts and constraints
    best_c5 = float('inf')
    best_h = initial_h.copy()

    # Parameters for differential evolution
    popsize = 15
    maxiter = 100
    tol = 1e-8
    mutation = (0.5, 1)
    recombination = 0.8

    num_restarts = 10

        # Generate a random initial guess or perturb the initial_h
        if i % 3 == 0:
            # Use the initial guess with perturbation
            h_initial = initial_h.copy()
            np.random.seed(seed + i)
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            h_initial += perturbation
            h_initial = np.clip(h_initial, 0.0, 1.0)
        elif i % 3 == 1:

            h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        else:

            h_initial = np.zeros(n_points)
            h_initial[::2] = 1.0
            # Adjust the last element to meet the sum constraint
            h_initial[-1] = 0.5 if n_points % 2 else 1.0

        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            popsize=popsize,
            maxiter=maxiter,

            mutation=mutation,
            recombination=recombination,

        )

        if result.success:

                best_c5 = current_c5
                best_h = result.x

            current_c5 = objective(h_initial)
            if current_c5 < best_c5:
