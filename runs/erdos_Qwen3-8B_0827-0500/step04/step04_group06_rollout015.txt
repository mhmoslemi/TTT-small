    num_restarts = 10  # Reduced to save time

    for i in range(num_restarts):
        # Run differential evolution with a different seed
        result = differential_evolution(objective, bounds, strategy='best1bin', 
                                       popsize=15, mutation=(0.5, 1.0), recombination=0.8, 
                                       tol=1e-6, maxiter=100, seed=seed + i)
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # If optimization fails, try the initial guess
            current_c5 = objective(initial_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = initial_h

    return best_h, best_c5, n_points
