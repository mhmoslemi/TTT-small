    bounds_de = [(0.0, 1.0) for _ in range(n_points)]

    # Run differential evolution to find a good initial guess
    result_de = differential_evolution(objective_de, bounds_de, strategy='best1bin', maxiter=100, popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.8, polish=True)
    best_h_de = result_de.x

    # Now, use this as an initial guess for SLSQP
    # Define the objective function for SLSQP
    def objective_slsqp(h):
        cross_corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    # Define the constraint for SLSQP
    def constraint_slsqp(h):
        return np.sum(h) - target_sum

    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]

    # Perform SLSQP optimization
    result_slsqp = minimize(objective_slsqp, best_h_de, method='SLSQP', bounds=bounds_slsqp, constraints={'type': 'eq', 'fun': constraint_slsqp}, 
                           options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8})
    best_h = result_slsqp.x
    best_c5 = result_slsqp.fun

    return best_h, best_c5, n_points
