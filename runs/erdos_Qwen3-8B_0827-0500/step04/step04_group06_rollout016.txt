import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h_values):
        sum_first = np.sum(h_values)
        last = (n_points / 2) - sum_first
        full_h = np.concatenate([h_values, [last]])
        full_h[-1] = np.clip(last, 0.0, 1.0)
        cross_corr = correlate(full_h, 1 - full_h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    bounds = [(0.0, 1.0) for _ in range(n_points - 1)]
    initial_guess = initial_h[:-1].copy()

    # Run differential evolution with multiple restarts
    best_c5 = float('inf')
    best_h = initial_h.copy()

    # Use a smaller population and more iterations for better exploration
    result = differential_evolution(
        objective, bounds, strategy='best1bin', popsize=15, mutation=(0.5, 1),
        recombination=0.8, tol=1e-9, maxiter=100, seed=seed
    )
    best_h = np.concatenate([result.x, [(n_points / 2) - np.sum(result.x)]])
    best_h[-1] = np.clip((n_points / 2) - np.sum(result.x), 0.0, 1.0)
    best_c5 = result.fun

    return best_h, best_c5, n_points
