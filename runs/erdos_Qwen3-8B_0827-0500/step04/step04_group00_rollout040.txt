    A_eq = np.ones(n_points)
    b_eq = n_points / 2
    cons = [LinearConstraint(A_eq, b_eq, b_eq)]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = minimize(objective, initial_guess, method='L-BFGS-B', bounds=bounds, constraints=cons,
                      tol=1e-9, options={'maxiter': 800, 'ftol': 1e-10, 'disp': False})
    
    if result.success:
        return (result.x, result.fun)
    else:
        return (initial_guess, evaluate_erdos_solution(initial_guess))

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    
    initial_guesses = [initial_h_values]
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.03))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.05))
    for num_pulses in [3, 4, 5, 6]:
        initial_guesses.append(generate_structured_initial(n_points, num_pulses))
    initial_guesses.append(generate_random_initial(n_points))
    
    best_h = initial_h_values
    best_c5 = evaluate_erdos_solution(initial_h_values)
    elapsed_time = 0
    
    with Pool(processes=cpu_count()) as pool:
        results = []
        for guess in initial_guesses:
            results.append(pool.apply_async(optimize_h, (guess, n_points)))
        
        for result in results:
            h, c5 = result.get()
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
            elapsed_time += 10  # Approximate time per optimization
            if elapsed_time >= budget_s:
                break
    
    return (best_h, best_c5, n_points)
