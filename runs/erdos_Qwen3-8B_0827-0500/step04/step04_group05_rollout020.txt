def run(seed=42, budget_s=100, **kwargs):

    def normalize_h(h_candidate):

        scaled = h_candidate * (n_points / 2) / s

        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx
