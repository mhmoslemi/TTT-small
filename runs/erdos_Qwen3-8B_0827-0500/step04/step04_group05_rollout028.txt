import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    # Predefined initial guess with 95 samples
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Define the objective function
    def objective(h):
        cross_corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    # Define constraints
    def constraint(h):
        return np.sum(h) - n_points / 2

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Function to generate a random initial guess
    def random_initial_guess():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        return h

    # Function to perturb the initial guess
    def perturb(h, seed=None):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        return np.clip(h + perturbation, 0.0, 1.0)

    # Function to generate a structured initial guess (alternating pattern)
    def structured_initial_guess():
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 3 == 0:
                h[i] = 1.0
            elif i % 3 == 1:
                h[i] = 0.5
            else:
                h[i] = 0.0
        h = h * (n_points / 2) / np.sum(h)
        return h

    # Main optimization loop
    best_c5 = float('inf')
    best_h = initial_h.copy()
    num_restarts = 200  # Increased for better exploration
    time_used = 0.0

    for i in range(num_restarts):
        # Generate a structured or random initial guess
        if i % 5 == 0:
            h_initial = structured_initial_guess()
        else:
            h_initial = random_initial_guess()
        
        # Early stopping if time is exhausted
        if time_used >= budget_s - 0.1:
            break
        
        # Perform optimization with L-BFGS-B method
        result = minimize(
            objective, 
            h_initial, 
            method='L-BFGS-B', 
            bounds=bounds, 
            constraints={'type': 'eq', 'fun': constraint}, 
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
        )
        
        # Track time usage (approximate)

            h_initial_perturbed = perturb(best_h, seed=seed + i)
            current_c5 = objective(h_initial_perturbed)
            if current_c5 < best_c5:
                best_c5 = current_c5
