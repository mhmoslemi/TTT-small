import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        cross_corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        # Penalty for constraint violation
        penalty = 1e6 * (np.sum(h) - n_points / 2)**2
        return max_corr * dx + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Initial guess
    x0 = initial_h.copy()

    # Run differential evolution with multiple restarts and parameter tuning
    best_c5 = float('inf')
    best_h = initial_h.copy()
    num_restarts = 5  # Reduced for speed, but allows multiple diverse initializations

    for i in range(num_restarts):
        # Perturb initial guess for diversity
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        x0_perturbed = np.clip(x0 + perturbation, 0.0, 1.0)
        
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            maxiter=100,
            tol=1e-8,
            mutation=(0.5, 1),
            recombination=0.8,
            x0=x0_perturbed
        )
        
        current_c5 = objective(result.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x

    return best_h, best_c5, n_points
