    def random_initial_guess():
        variables = np.random.rand(n_points - 1)
        return variables

    # Function to generate initial guess from initial_h_values
    def initial_h_guess():
        variables = initial_h[:n_points-1].copy()
        return variables

    # Run multiple optimizations
    best_c5 = float('inf')
    best_h = initial_h.copy()

    # Number of runs
    num_runs = 15

    for i in range(num_runs):
        # Generate initial guess
        if i % 3 == 0:
            # Use initial_h_values as initial guess
            initial_variables = initial_h_guess()
        elif i % 2 == 0:
            # Use symmetric initial guess
            initial_variables = symmetric_initial_guess()
        else:
            # Use random initial guess
            initial_variables = random_initial_guess()
        
        # Run differential evolution
        result = differential_evolution(
            objective, 
            bounds, 
            strategy='best1bin', 
            maxiter=50, 
            popsize=15, 
            tol=1e-6, 
            mutation=(0.5, 1), 
            recombination=0.8, 
            disp=False
        )
        if result.success:
            current_variables = result.x
            # Compute h
            h = np.zeros(n_points)
            h[:n_points-1] = current_variables
            h[-1] = (n_points / 2) - np.sum(current_variables)
            # Compute cross-correlation
            cross_corr = correlate(h, 1 - h, mode='full')
            current_c5 = np.max(cross_corr) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h
        else:
            # If optimization fails, try the initial guess
            h = np.zeros(n_points)
            h[:n_points-1] = initial_variables
            h[-1] = (n_points / 2) - np.sum(initial_variables)
            cross_corr = correlate(h, 1 - h, mode='full')
            current_c5 = np.max(cross_corr) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h

    return best_h, best_c5, n_points
