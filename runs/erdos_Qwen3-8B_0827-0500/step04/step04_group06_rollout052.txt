import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        # Normalize to meet the sum constraint
        h_normalized = h * (n_points / 2) / np.sum(h)
        # Compute cross-correlation with 1 - h_normalized
        cross_corr = correlate(h_normalized, 1 - h_normalized, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run multiple DE optimizations with different seeds and parameters
    best_c5 = float('inf')
    best_h = initial_h.copy()

    # Parameters for DE
    strategy = 'best1bin'
    popsize = 15
    mutation = (0.5, 1)
    recombination = 0.9
    tol = 1e-9
    maxiter = 100
    num_restarts = 5

    for i in range(num_restarts):
        # Use seed to vary initial population
        np.random.seed(seed + i)
        result = differential_evolution(
            objective, bounds, strategy=strategy, popsize=popsize,
            mutation=mutation, recombination=recombination, tol=tol,
            maxiter=maxiter, seed=seed + i
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # Fallback to initial guess if optimization fails
            current_c5 = objective(initial_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = initial_h.copy()

    return best_h, best_c5, n_points
