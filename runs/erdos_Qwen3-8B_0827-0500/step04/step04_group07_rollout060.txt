import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        # Normalize h to meet the integral constraint
        sum_h = np.sum(h)
        if sum_h == 0:
            h_normalized = np.zeros_like(h)
        else:
            h_normalized = h * (n_points / 2) / sum_h
        # Compute cross-correlation with 1 - h_normalized
        cross_corr = correlate(h_normalized, 1 - h_normalized, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run differential evolution with tuned parameters
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=30,
        maxiter=150,
        tol=1e-8,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )

    best_h = result.x
    best_c5 = result.fun

    return best_h, best_c5, n_points
