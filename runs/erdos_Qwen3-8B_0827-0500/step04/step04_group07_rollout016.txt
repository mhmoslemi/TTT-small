    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def random_initial_guess():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        return h

    def structured_initial_guess():
        h = np.zeros(n_points)
        h[::2] = 1.0
        h[-1] = 0.5 if n_points % 2 else 1.0
        return h

    best_c5 = float('inf')
    best_h = initial_h.copy()

    # Run DE with initial_h
    result = differential_evolution(objective, bounds, strategy='best1bin', 
                                   popsize=15, mutation=(0.5, 1), recombination=0.8, 
                                   tol=1e-8, seed=seed, maxiter=100)
    if result.success:
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x

    # Run DE with random initial guess
    result = differential_evolution(objective, bounds, strategy='best1bin', 
                                   popsize=15, mutation=(0.5, 1), recombination=0.8, 
                                   tol=1e-8, seed=seed + 1, maxiter=100)
    if result.success:
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x

    # Run DE with structured initial guess
    result = differential_evolution(objective, bounds, strategy='best1bin', 
                                   popsize=15, mutation=(0.5, 1), recombination=0.8, 
                                   tol=1e-8, seed=seed + 2, maxiter=100)
    if result.success:
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x

    return best_h, best_c5, n_points
