import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 64  # Smaller number for faster optimization
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Generate structured initial guess with evenly spaced pulses
    num_pulses = 5  # Adjust based on problem symmetry
    pulse_positions = np.linspace(0, 2, num_pulses, endpoint=False)
    h_initial = np.zeros(n_points)
    for pos in pulse_positions:
        idx = int(pos / dx)
        h_initial[idx] = 1.0
    current_sum = np.sum(h_initial)
    scaling_factor = target_sum / current_sum
    h_initial *= scaling_factor
    h_initial = np.clip(h_initial, 0.0, 1.0)

    # Define objective function with penalty
    def objective(h):
        sum_h = np.sum(h)
        penalty = 1e6 * abs(sum_h - target_sum)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx + penalty

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run differential evolution with tuned parameters
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )

    # Extract the best h and C5
    best_h = result.x
    best_c5 = result.fun - 1e6 * abs(np.sum(best_h) - target_sum)  # Subtract penalty to get actual C5
