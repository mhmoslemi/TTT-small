    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.03))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.05))
    for num_pulses in [3, 4, 5, 6, 7]:
        initial_guesses.append(generate_structured_initial(n_points, num_pulses))
    initial_guesses.append(generate_random_initial(n_points))
    
    best_h = initial_h_values
    best_c5 = evaluate_c5(initial_h_values, dx)
    elapsed_time = 0.0
    
    # Use parallel processing to run optimizations
    with ProcessPoolExecutor(max_workers=10) as executor:
        futures = []
        for guess in initial_guesses:
            # Submit both DE and SLSQP optimizations
            futures.append(executor.submit(optimize_with_de, guess, dx, n_points, bounds, constraints))
            futures.append(executor.submit(optimize_with_slsqp, guess, dx, n_points, bounds, constraints))
        
        for future in as_completed(futures):
            try:
                h_opt, c5_val = future.result()
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_opt
            except Exception as e:
                pass
            
            # Track time usage (approximate)
            elapsed_time += 0.1  # Placeholder for actual time tracking
            if elapsed_time >= budget_s:
                break
    
    return (best_h, best_c5, n_points)
