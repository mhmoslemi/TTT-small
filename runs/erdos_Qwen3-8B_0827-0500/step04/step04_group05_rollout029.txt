import numpy as np
from scipy.optimize import minimize, differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is a global variable
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Define the objective function
    def objective(h):
        cross_corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - target_sum
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Function to generate a random initial guess
    def random_initial_guess():
        h = np.random.rand(n_points)
        # Normalize to meet the sum constraint
        h = h * (target_sum) / np.sum(h)
        return h
    
    # Function to perturb the initial guess
    def perturb(h, seed=None):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        return np.clip(h + perturbation, 0.0, 1.0)
    
    # Function to evaluate the objective with constraint handling
    def evaluate_with_constraint(h):
        # Enforce constraint by normalizing
        h_normalized = h.copy()
        h_normalized = h_normalized * target_sum / np.sum(h_normalized)
        return objective(h_normalized)
    
    # Run multiple optimizations
    best_c5 = float('inf')
    best_h = initial_h.copy()
    best_h_normalized = initial_h.copy()
    
    # Number of restarts
    num_restarts = 150  # Increased for better exploration
    
    for i in range(num_restarts):
        # Generate a random initial guess or perturb the current best
        if i % 3 == 0:
            # Use random initial guess
            h_initial = random_initial_guess()
        else:
            # Perturb the current best solution
            h_initial = perturb(best_h, seed=seed + i)
        
        # Perform optimization with adjusted parameters
        result = minimize(
            evaluate_with_constraint, 
            h_initial, 
            method='SLSQP', 
            bounds=bounds, 
            constraints={'type': 'eq', 'fun': constraint}, 
            options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-9}
        )

                best_h = h_initial
