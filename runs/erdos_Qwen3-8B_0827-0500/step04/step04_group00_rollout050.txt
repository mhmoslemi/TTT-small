                              tol=1e-9, options={'maxiter': 1000, 'ftol': 1e-10, 'disp': False})
            return result
        
        results.append(pool.apply_async(optimize, (guess,)))
    
    # Wait for all results
    for result in results:
        res = result.get()
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    pool.close()
    pool.join()
    
    return (best_h, best_c5, n_points)
