import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is a global variable
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Define the objective function
    def objective(h):
        # Normalize h to meet the sum constraint
        h_normalized = h * (n_points / 2) / np.sum(h)
        # Compute cross-correlation with 1 - h
        cross_corr = correlate(h_normalized, 1 - h_normalized, mode='full')
        # Find the maximum value in the cross-correlation
        max_corr = np.max(cross_corr)
        return max_corr * dx

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run differential evolution optimization with multiple restarts
    best_c5 = float('inf')
    best_h = initial_h.copy()

    # Generate a few different initial guesses
    initial_guesses = []
    # Use initial_h as a starting point
    initial_guesses.append(initial_h)
    # Generate a random initial guess
    initial_guesses.append(np.random.rand(n_points) * (n_points / 2) / np.sum(np.random.rand(n_points)))
    # Generate a structured initial guess (alternating 1 and 0)
    structured_h = np.zeros(n_points)
    structured_h[::2] = 1.0
    # Adjust the last element to meet the sum constraint
    structured_h[-1] = 0.5 if n_points % 2 else 1.0
    initial_guesses.append(structured_h)

    # Run differential evolution from each initial guess
    for guess in initial_guesses:
        # Ensure the guess is within bounds and normalized
        guess = np.clip(guess, 0.0, 1.0)
        guess = guess * (n_points / 2) / np.sum(guess)
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # If optimization fails, try the perturbed guess
            current_c5 = objective(guess)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = guess

    return best_h, best_c5, n_points
