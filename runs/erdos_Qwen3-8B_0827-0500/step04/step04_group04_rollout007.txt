from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

        cross_corr = correlate(h, 1 - h, mode='full')

        perturbation = np.random.normal(0, 0.1, size=len(h))
        return np.clip(h + perturbation, 0.0, 1.0)

    best_c5 = float('inf')
    best_h = initial_h.copy()

    num_restarts = 200  # Increased number of restarts

            h_initial = initial_h.copy()

            h_initial = perturb(best_h, seed=seed + i)

            objective, 
            h_initial, 
            method='SLSQP',
