import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is a global variable
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Define the objective function
    def objective(h):
        cross_corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    # Define constraints
    def constraint(h):
        return np.sum(h) - n_points / 2

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Function to generate a random initial guess
    def random_initial_guess():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        return h

    # Function to perturb the initial guess
    def perturb(h, seed=None):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.2, 0.2, size=len(h))
        return np.clip(h + perturbation, 0.0, 1.0)

    # Function to generate a structured initial guess (alternating pattern)
    def structured_initial_guess():
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 2 == 0:
                h[i] = 1.0
            else:
                h[i] = 0.0
        # Normalize to meet the sum constraint
        h = h * (n_points / 2) / np.sum(h)
        return h

    # Initialize best solution
    best_c5 = float('inf')
    best_h = initial_h.copy()

    # Number of restarts
    num_restarts = 200  # Increased for more exploration

    # Optimization parameters
    opt_options = {
        'maxiter': 150,
        'ftol': 1e-6,
        'eps': 1e-6,
        'disp': False
    }

    # Try different initial guess strategies
    initial_guess_strategies = [
        lambda: initial_h.copy(),
        random_initial_guess,
        structured_initial_guess
    ]

    for i in range(num_restarts):
        # Select a random initial guess strategy
        strategy_idx = np.random.randint(len(initial_guess_strategies))
        h_initial = initial_guess_strategies[strategy_idx]()

        # Run optimization with COBYLA method
        try:
            result = minimize(
                objective, 
                h_initial, 
                method='COBYLA', 
                bounds=bounds, 
                constraints={'type': 'eq', 'fun': constraint}, 
                options=opt_options
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            # Fallback to evaluating the initial guess directly
            current_c5 = objective(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_initial

    return best_h, best_c5, n_points
