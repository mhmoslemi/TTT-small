    # Run multiple differential evolution with perturbed initial guesses
    for i in range(5):
        np.random.seed(seed + i)
        initial_guess = initial_h[:-1].copy()
        perturbation = np.random.uniform(-0.1, 0.1, size=len(initial_guess))
        initial_guess += perturbation
        initial_guess = np.clip(initial_guess, 0.0, 1.0)

        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            maxiter=100,
            tol=1e-8,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed + i
        )

        best_h_part = result.x
        sum_part = np.sum(best_h_part)
        h_last = (n_points / 2) - sum_part
        h_last = np.clip(h_last, 0.0, 1.0)
        current_h = np.concatenate([best_h_part, [h_last]])
        current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h

    return best_h, best_c5, n_points
