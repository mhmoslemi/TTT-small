    best_c5_de = objective_slsqp(best_h_de)

    # Now use SLSQP to refine the result
    result_slsqp = minimize(objective_slsqp, best_h_de, method='SLSQP', bounds=bounds_slsqp, constraints={'type': 'eq', 'fun': constraint}, 
                          options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-9})
    if result_slsqp.success:
        best_c5 = result_slsqp.fun
        best_h = result_slsqp.x
    else:
        # Fallback to the differential evolution result
        best_c5 = best_c5_de
        best_h = best_h_de

    # Perform multiple restarts with different initial guesses
    num_restarts = 50  # Adjust based on time
    best_c5 = best_c5
    best_h = best_h

    for i in range(num_restarts):
        # Generate a random initial guess
        h_initial = random_initial_guess()
        # Perform SLSQP optimization
        result = minimize(objective_slsqp, h_initial, method='SLSQP', bounds=bounds_slsqp, constraints={'type': 'eq', 'fun': constraint}, 
                          options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-9})
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # Fallback to the initial guess
            current_c5 = objective_slsqp(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_initial

    return best_h, best_c5, n_points
