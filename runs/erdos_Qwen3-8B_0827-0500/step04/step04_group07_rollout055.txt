import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        cross_corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    def constraint(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Initial guess: perturb the given initial_h
    h_initial = initial_h.copy()
    np.random.seed(seed)
    perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
    h_initial = np.clip(h_initial + perturbation, 0.0, 1.0)

    # Optimization with tighter tolerances and more iterations
    result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, 
                      constraints={'type': 'eq', 'fun': constraint}, 
                      options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-9})

    if result.success:
        best_h = result.x
        best_c5 = result.fun
    else:
        # Fallback to the perturbed initial guess if optimization fails
        best_h = h_initial
        best_c5 = objective(h_initial)

    return best_h, best_c5, n_points
