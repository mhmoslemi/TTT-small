import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is a global variable
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Run differential evolution
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        maxiter=100,
        disp=False,
        polish=True
    )

    # Extract the best solution
    best_h = result.x

    return best_h, best_c5, n_points
