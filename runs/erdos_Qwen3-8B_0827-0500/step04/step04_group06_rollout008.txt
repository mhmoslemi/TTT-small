import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        cross_corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    def constraint(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def random_initial_guess():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        return h

    def perturb(h, seed=None):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        return np.clip(h + perturbation, 0.0, 1.0)

    best_c5 = float('inf')
    best_h = initial_h.copy()
    num_restarts = 50
    time_used = 0.0

    for i in range(num_restarts):
        if i % 3 == 0:
            h_initial = perturb(initial_h, seed=seed + i)
        elif i % 3 == 1:
            h_initial = random_initial_guess()
        else:
            h_initial = random_initial_guess()

        try:
            result = minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint},
                              options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-9})
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:

            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_initial

        time_used += 0.5  # Approximate time per restart
        if time_used >= budget_s:
            break

    return best_h, best_c5, n_points
