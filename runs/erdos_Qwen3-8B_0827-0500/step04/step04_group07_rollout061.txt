import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is a global variable
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Define the objective function with penalty for constraint violation
    def objective(h):
        cross_corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        # Penalty for constraint violation
        penalty = 1e6 * (np.sum(h) - n_points / 2)**2
        return max_corr * dx + penalty

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run differential evolution with multiple restarts
    best_h = initial_h.copy()
    best_c5 = float('inf')
    num_restarts = 10  # Reduced for speed

    for i in range(num_restarts):
        # Perturb the initial guess to avoid local minima
        np.random.seed(seed + i)
        h_initial = initial_h.copy()
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        h_initial += perturbation
        h_initial = np.clip(h_initial, 0.0, 1.0)
        # Normalize to meet the sum constraint
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)

        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            maxiter=100,
            tol=1e-8,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed + i
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # Fallback to perturbed guess if optimization fails
            current_c5 = objective(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_initial

    return best_h, best_c5, n_points
