import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        # Normalize to ensure the sum constraint is met
        h_normalized = h * (n_points / 2) / np.sum(h)

        objective, bounds,

        popsize=30, 
        maxiter=150, 
        tol=1e-6, 
        mutation=(0.5, 1.0), 
        recombination=0.8, 
        seed=seed
