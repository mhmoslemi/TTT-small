import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
from multiprocessing import Pool, cpu_count

def run(seed=42, budget_s=100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        cross_corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def objective_with_penalty(h):
        obj = objective(h)
        penalty = 1e4 * abs(np.sum(h) - n_points / 2)
        return obj + penalty

    def evaluate_solution(h):
        return objective(h)

    # Use multiprocessing to run multiple differential evolution optimizations in parallel
    def parallel_optimize():
        # Run multiple differential evolution optimizations in parallel
        with Pool(processes=cpu_count()) as pool:
            results = pool.map(
                lambda _: differential_evolution(
                    objective_with_penalty,
                    bounds,
                    strategy='best1bin',
                    popsize=15,
                    maxiter=50,
                    tol=1e-6,
                    mutation=(0.5, 1),
                    recombination=0.8,
                    seed=np.random.randint(0, 100000)
                ),
                range(5)  # Run 5 parallel optimizations
            )
        # Collect the best solution from all runs
        best_h = initial_h
        best_c5 = float('inf')
        for result in results:
            if result.success:
                current_h = result.x
                current_c5 = evaluate_solution(current_h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h
        return best_h, best_c5, n_points

    # Start the parallel optimization
    best_h, best_c5, n_points = parallel_optimize()
    return best_h, best_c5, n_points
