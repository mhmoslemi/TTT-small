import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    
    # Define objective function with penalty for sum constraint
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / n_points
        # Penalty term to enforce sum constraint
        penalty = 1e4 * (np.sum(h) - n_points / 2)**2
        return max_corr * dx + penalty
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess based on provided initial_h_values
    initial_guess = initial_h_values.copy()
    
    # Run differential evolution with adjusted parameters for better exploration
    result = differential_evolution(
        objective, bounds, 
        strategy='best1bin', 
        maxiter=100, 
        popsize=20, 
        tol=1e-6, 
        mutation=(0.5, 1), 
        recombination=0.8, 
        seed=seed
    )
    
    # Extract the best solution
    best_h = result.x
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
