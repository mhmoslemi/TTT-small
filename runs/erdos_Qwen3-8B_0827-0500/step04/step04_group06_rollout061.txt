import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is a global variable
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Define the objective function
    def objective(h):
        # Normalize h to meet the sum constraint
        h_normalized = h * (n_points / 2) / np.sum(h)
        # Compute the cross-correlation
        cross_corr = correlate(h_normalized, 1 - h_normalized, mode='full')
        # Find the maximum value in the cross-correlation
        max_corr = np.max(cross_corr)
        return max_corr * dx

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run differential evolution with multiple restarts
    best_c5 = float('inf')
    best_h = initial_h.copy()
    num_restarts = 10  # Reduced for speed while maintaining diversity

    for i in range(num_restarts):
        # Set seed for reproducibility
        np.random.seed(seed + i)
        # Generate a random initial guess based on initial_h
        h_initial = np.random.rand(n_points)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        # Perturb initial guess to avoid local minima
        h_initial += np.random.uniform(-0.05, 0.05, size=n_points)
        h_initial = np.clip(h_initial, 0.0, 1.0)

        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            tol=1e-9,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed + i,
            maxiter=100,
            polish=True
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # Fallback to the perturbed initial guess
            current_c5 = objective(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_initial

    return best_h, best_c5, n_points
