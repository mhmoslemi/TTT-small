import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is a global variable
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Define the objective function
    def objective(h):
        # Compute cross-correlation with 1 - h
        cross_corr = correlate(h, 1 - h, mode='full')
        # Find the maximum value in the cross-correlation
        max_corr = np.max(cross_corr)
        return max_corr * dx

    # Define constraints
    def constraint(h):
        return np.sum(h) - n_points / 2

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Function to generate a random initial guess
    def random_initial_guess():
        h = np.random.rand(n_points)
        # Normalize to meet the sum constraint
        h = h * (n_points / 2) / np.sum(h)
        return h

    # Run multiple optimizations
    best_c5 = float('inf')
    best_h = initial_h.copy()

    # Number of restarts
    num_restarts = 200  # Increased for more exploration

    for i in range(num_restarts):
        # Generate a random initial guess
        h_initial = random_initial_guess()
        
        # Perform optimization with adjusted parameters
        result = minimize(
            objective, 
            h_initial, 
            method='SLSQP', 
            bounds=bounds, 
            constraints={'type': 'eq', 'fun': constraint}, 
            options={'maxiter': 100, 'ftol': 1e-6, 'eps': 1e-6}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # If optimization fails, try the initial guess
            current_c5 = objective(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_initial

    return best_h, best_c5, n_points
