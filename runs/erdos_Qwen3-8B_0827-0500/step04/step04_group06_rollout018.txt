    dx = 2.0 / n_points

    def objective(x):
        h = np.zeros(n_points)
        h[:n_points-1] = x
        last = (n_points / 2) - np.sum(x)
        last = np.clip(last, 0.0, 1.0)
        h[-1] = last
        cross_corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    bounds = [(0.0, 1.0) for _ in range(n_points - 1)]

    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    num_restarts = 20
    for i in range(num_restarts):
        if i % 3 == 0:
            initial_guess = np.clip(initial_h[:n_points-1], 0.0, 1.0)
        elif i % 3 == 1:
            np.random.seed(seed + i)
            initial_guess = np.random.rand(n_points - 1)
        else:
            np.random.seed(seed + i * 2)
            initial_guess = np.random.rand(n_points - 1)

        result = differential_evolution(
            objective, bounds, strategy='best1bin',
            popsize=15, mutation=(0.5, 1), recombination=0.9,
            tol=1e-9, maxiter=50, seed=seed + i
        )
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_x = result.x
            best_h = np.zeros(n_points)
            best_h[:n_points-1] = best_x
            last = (n_points / 2) - np.sum(best_x)
            last = np.clip(last, 0.0, 1.0)
            best_h[-1] = last

    return best_h, best_c5, n_points
