import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is a global variable
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Define the objective function
    def objective(h):
        # Normalize h to meet the sum constraint
        sum_h = np.sum(h)
        if sum_h == 0:
            return float('inf')
        h_normalized = h * (n_points / 2) / sum_h
        # Compute cross-correlation with 1 - h
        cross_corr = correlate(h_normalized, 1 - h_normalized, mode='full')
        # Find the maximum value in the cross-correlation
        max_corr = np.max(cross_corr)
        return max_corr * dx

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run differential evolution with multiple restarts
    best_c5 = float('inf')
    best_h = initial_h.copy()

    # Parameters for differential evolution
    popsize = 15
    maxiter = 100
    strategy = 'best1bin'
    mutation = (0.5, 1)
    recombination = 0.9
    tol = 1e-9

    # Number of restarts
    num_restarts = 5

    for i in range(num_restarts):
        # Set seed for reproducibility
        np.random.seed(seed + i)
        # Generate a random initial guess or perturb the initial_h
        if i % 2 == 0:
            # Use the initial guess with perturbation
            h_initial = initial_h.copy()
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            h_initial += perturbation
            h_initial = np.clip(h_initial, 0.0, 1.0)
        else:
            # Generate a random initial guess
            h_initial = np.random.rand(n_points)
            h_initial = h_initial * (n_points / 2) / np.sum(h_initial)

        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy=strategy,
            popsize=popsize,
            maxiter=maxiter,
            tol=tol,
            mutation=mutation,
            recombination=recombination,
            seed=seed + i
        )

        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # Fallback to the initial guess if optimization fails
            current_c5 = objective(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_initial
