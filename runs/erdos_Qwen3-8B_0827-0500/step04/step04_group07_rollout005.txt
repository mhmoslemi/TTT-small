import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    # Predefined initial guess
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Define the objective function
    def objective(h):
        cross_corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    # Define constraint
    def constraint(h):
        return np.sum(h) - n_points / 2

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Generate diverse initial guesses
    def random_initial_guess():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        return h

    def symmetric_initial_guess():
        h = np.zeros(n_points)
        for i in range(n_points // 2):
            h[i] = 1.0
            h[n_points - 1 - i] = 1.0
        # Adjust for even/odd n_points
        if n_points % 2 == 1:
            h[n_points // 2] = 0.5
        return h

    def structured_initial_guess():
        h = np.zeros(n_points)
        h[::3] = 1.0
        h[1::3] = 0.5
        h[2::3] = 0.0
        # Adjust to meet sum constraint
        h = h * (n_points / 2) / np.sum(h)
        return h

    # Perturbation function
    def perturb(h, seed=None):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.15, 0.15, size=len(h))
        return np.clip(h + perturbation, 0.0, 1.0)

    # Track best solution
    best_c5 = float('inf')
    best_h = initial_h.copy()
    start_time = np.datetime64('now')

    # Optimization parameters
    max_restarts = 150
    max_iter_per_restart = 100
    for i in range(max_restarts):
        # Generate initial guess
        if i % 3 == 0:
            h_initial = perturb(initial_h, seed=seed + i)
        elif i % 3 == 1:
            h_initial = random_initial_guess()
        else:
            h_initial = symmetric_initial_guess()

        # Optimization settings
        method = 'L-BFGS-B'
        options = {'maxiter': max_iter_per_restart, 'ftol': 1e-8, 'eps': 1e-8}

        # Perform optimization
        try:
            result = minimize(objective, h_initial, method=method, bounds=bounds, constraints={'type': 'eq', 'fun': constraint},

            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:

                    best_h = result.x

            if current_c5 < best_c5:

        # Check time budget
        elapsed = np.datetime64('now') - start_time
