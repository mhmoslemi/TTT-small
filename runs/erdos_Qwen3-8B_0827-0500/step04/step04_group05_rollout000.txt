import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        # Normalize to meet the sum constraint
        sum_h = np.sum(h)
        h_normalized = h * (n_points / 2) / sum_h if sum_h != 0 else h
        # Compute cross-correlation
        cross_corr = correlate(h_normalized, 1 - h_normalized, mode='full')
        # Find the maximum value in the cross-correlation
        max_corr = np.max(cross_corr)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run differential evolution with a penalty for constraint violation
    result = differential_evolution(
        objective,

        mutation=(0.5, 1),

        tol=1e-8,
