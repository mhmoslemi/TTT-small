import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(vars):
        h = np.zeros(n_points)

    last_initial = np.clip(last_initial, 0.0, 1.0)

    # Run differential evolution with multiple restarts
    best_c5 = float('inf')
    best_h = initial_h.copy()

    # Parameters for differential evolution
    strategy = 'best1bin'
    popsize = 15
    mutation = (0.5, 1)
    recombination = 0.9

    maxiter = 100

            maxiter=maxiter,
            disp=False,

        )
        if result.success:

            best_h[:-1] = best_vars
