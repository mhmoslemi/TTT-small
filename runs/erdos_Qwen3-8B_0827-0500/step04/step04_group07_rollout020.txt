import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is a global variable
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Define the objective function
    def objective(h):
        # Compute cross-correlation with 1 - h
        cross_corr = correlate(h, 1 - h, mode='full')
        # Find the maximum value in the cross-correlation
        max_corr = np.max(cross_corr)
        return max_corr * dx

    # Define constraints
    def constraint(h):
        return np.sum(h) - n_points / 2

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Function to generate a random initial guess
    def random_initial_guess():
        h = np.random.rand(n_points)
        # Normalize to meet the sum constraint
        h = h * (n_points / 2) / np.sum(h)
        return h

    # Function to generate a structured initial guess (evenly spaced 1s)
    def structured_initial_guess():
        h = np.zeros(n_points)
        # Place 1s at evenly spaced positions
        num_ones = int(np.ceil(n_points / 5))
        for i in range(num_ones):
            pos = int(2 * i / num_ones * n_points)
            h[pos] = 1.0
        # Adjust the last element to meet the sum constraint
        h[-1] = (n_points / 2 - np.sum(h)) / (n_points - num_ones)
        return h

    # Function to perturb the initial guess
    def perturb(h, seed=None):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.2, 0.2, size=len(h))
        return np.clip(h + perturbation, 0.0, 1.0)

    # Run multiple optimizations
    best_c5 = float('inf')
    best_h = initial_h.copy()

    # Number of restarts
    num_restarts = 200  # Increased for more exploration

    for i in range(num_restarts):
        # Generate a random initial guess or perturb the initial_h
        if i % 3 == 0:

            h_initial = perturb(initial_h, seed=seed + i)
        elif i % 3 == 1:
            # Generate a random initial guess
            h_initial = random_initial_guess()
        else:
            # Generate a structured initial guess
            h_initial = structured_initial_guess()

        # Perform optimization with adjusted parameters
        result = minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint},
