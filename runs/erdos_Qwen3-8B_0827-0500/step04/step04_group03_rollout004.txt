                best_c5 = current_c5
                best_h = result.x
        
        # Track time usage
        # Approximate time per iteration
        elapsed_time += 0.1
        if elapsed_time >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
