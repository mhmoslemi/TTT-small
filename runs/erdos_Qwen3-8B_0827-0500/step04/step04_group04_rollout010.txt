        # Perform optimization with adjusted parameters
        result = minimize(
            objective, 
            h_initial, 
            method='Nelder-Mead', 
            bounds=bounds, 
            constraints={'type': 'eq', 'fun': constraint}, 
            options={'xatol': 1e-6, 'fatol': 1e-6, 'maxiter': 100}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # If optimization fails, try the perturbed guess
            current_c5 = objective(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_initial

    return best_h, best_c5, n_points
