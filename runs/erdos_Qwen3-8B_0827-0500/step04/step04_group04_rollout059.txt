```python
import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is a global variable
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Define the objective function with penalty for sum constraint
    def objective(h):
        cross_corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        penalty = 1000 * np.abs(np.sum(h) - n_points / 2)
        return max_corr * dx + penalty

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = initial_h.copy()

    # Number of runs
    num_runs = 5

    for i in range(num_runs):
        # Set seed for reproducibility
        np.random.seed(seed + i)
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed + i
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

    return best_h, best_c5, n_points
```
