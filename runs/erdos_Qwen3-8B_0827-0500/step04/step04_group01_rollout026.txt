    initial_guesses.append(generate_random_initial(n_points, seed=seed+2))
    
    best_h = None
    best_c5 = float('inf')
    elapsed_time = 0.0
    
    for guess in initial_guesses:
        # Define objective function with smooth max
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr_smooth = smooth_max(corr)
            dx = 2.0 / n_points
            penalty = 1e4 * (np.sum(h) - n_points / 2)**2
            return max_corr_smooth * dx + penalty
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization settings
        result = minimize(objective, guess, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-8})
        
        # Check if optimization succeeded
        if result.success:
            # Compute actual C5 after optimization
            corr = np.correlate(result.x, 1 - result.x, mode='full')
            max_corr = np.max(corr)
            dx = 2.0 / n_points
            current_c5 = max_corr * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Track time usage
        elapsed_time += result.message.split()[-1] if 'time' in result.message else 0.1
        if elapsed_time >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / n_points
        best_c5 = max_corr * dx
    
    return (best_h, best_c5, n_points)
