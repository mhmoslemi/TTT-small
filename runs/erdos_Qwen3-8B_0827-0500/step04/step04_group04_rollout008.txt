import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is a global variable
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Define the objective function
    def objective(h):
        cross_corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        penalty = 1e4 * (np.sum(h) - target_sum) ** 2  # Tuned penalty coefficient
        return max_corr * dx + penalty

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Initialize population based on initial_h to guide the search
    initial_population = [initial_h + np.random.normal(0, 0.05, n_points) for _ in range(20)]
    initial_population = [np.clip(p, 0, 1) for p in initial_population]

    # Run differential evolution with custom parameters
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=20,
        maxiter=50,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        init=initial_population
    )

    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
