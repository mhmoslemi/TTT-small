        result = minimize(objective, guess, method='L-BFGS-B', bounds=bounds, constraints=cons,
                          tol=1e-9, options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-8})
        
        # Check if optimization succeeded
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Track time usage
        elapsed_time += result.message.split()[-1] if 'time' in result.message else 0.1
        if elapsed_time >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
