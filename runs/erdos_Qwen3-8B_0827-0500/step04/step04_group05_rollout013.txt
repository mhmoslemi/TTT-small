        strategy='best1bin',
        maxiter=100,
        popsize=20,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )

    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
