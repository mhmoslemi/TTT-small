from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Define the objective function
    def objective(h):
        cross_corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    # Define constraints
    def constraint(h):
        return np.sum(h) - n_points / 2

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Function to generate a random initial guess
    def random_initial_guess():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        return h

    # Function to generate a uniform initial guess
    def uniform_initial_guess():
        return np.full(n_points, 0.5)

    # Function to perturb the initial guess
    def perturb(h, seed=None):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.2, 0.2, size=len(h))
        return np.clip(h + perturbation, 0.0, 1.0)

    # Run multiple optimizations
    best_c5 = float('inf')
    best_h = initial_h.copy()

    # Number of restarts
    num_restarts = 200  # Increased for more exploration

    for i in range(num_restarts):
        # Generate a random initial guess or perturb the current best
        if i % 3 == 0:
            # Use initial_h as a starting point
            h_initial = initial_h.copy()
        elif i % 3 == 1:
            # Use uniform initial guess
            h_initial = uniform_initial_guess()
        else:
            # Perturb the current best solution
            h_initial = perturb(best_h, seed=seed + i)
        
        # Perform optimization with adjusted parameters
        result = minimize(
            objective, 
            h_initial, 
            method='L-BFGS-B', 
            bounds=bounds, 
            constraints={'type': 'eq', 'fun': constraint},

        if result.success:
