import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
from multiprocessing import Pool, cpu_count

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is a global variable
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Define the objective function
    def objective(h):
        # Compute cross-correlation with 1 - h
        cross_corr = correlate(h, 1 - h, mode='full')
        # Find the maximum value in the cross-correlation
        max_corr = np.max(cross_corr)
        return max_corr * dx

    # Define constraints
    def constraint(h):
        return np.sum(h) - n_points / 2

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Function to generate a random initial guess
    def random_initial_guess():
        h = np.random.rand(n_points)
        # Normalize to meet the sum constraint
        h = h * (n_points / 2) / np.sum(h)
        return h

    # Function to generate a structured initial guess (alternating 1 and 0)
    def structured_initial_guess():
        h = np.zeros(n_points)
        h[::2] = 1.0
        # Adjust the last element to meet the sum constraint
        h[-1] = 0.5 if n_points % 2 else 1.0
        return h

    # Function to perturb the initial guess
    def perturb(h, seed=None):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        return np.clip(h + perturbation, 0.0, 1.0)

    # Function to run a single optimization
    def optimize(initial_h):
        # Perform optimization with adjusted parameters
        result = minimize(objective, initial_h, method='Nelder-Mead', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, 
                          options={'maxiter': 200, 'xatol': 1e-8, 'fatol': 1e-8})
        if result.success:
            return (result.fun, result.x)
        else:
            # If optimization fails, try the perturbed guess
            current_c5 = objective(initial_h)
            return (current_c5, initial_h)

    # Number of restarts
    num_restarts = 200  # Increased for more exploration

    # Generate initial guesses
    initial_guesses = []
    for i in range(num_restarts):
        if i % 3 == 0:
            # Use the initial guess with perturbation
            h_initial = perturb(initial_h, seed=seed + i)
        elif i % 3 == 1:
            # Generate a random initial guess
            h_initial = random_initial_guess()
        else:
            # Generate a structured initial guess
            h_initial = structured_initial_guess()
        initial_guesses.append(h_initial)

    # Use multiprocessing to run optimizations in parallel
    with Pool(processes=cpu_count()) as pool:
        results = pool.map(optimize, initial_guesses)

    # Find the best result
    best_c5 = float('inf')
    best_h = initial_h.copy()
    for c5, h in results:
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    return best_h, best_c5, n_points
