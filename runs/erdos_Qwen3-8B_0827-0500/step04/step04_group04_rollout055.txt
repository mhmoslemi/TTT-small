```python
import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        cross_corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        penalty = 1e6 * abs(np.sum(h) - n_points / 2)
        return max_corr * dx + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = initial_h.copy()

    num_restarts = 20
    for i in range(num_restarts):
        np.random.seed(seed + i)
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=30,
            mutation=(0.5, 1),
            recombination=0.8,
            maxiter=100,
            tol=1e-6,
            disp=False
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            current_c5 = objective(initial_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = initial_h

    return best_h, best_c5, n_points
```
