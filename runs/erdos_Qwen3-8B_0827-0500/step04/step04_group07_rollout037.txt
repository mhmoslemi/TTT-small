from scipy.optimize import minimize, differential_evolution

def run(seed=42, budget_s=100, **kwargs):

    def objective(h):
        cross_corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    def constraint(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def random_initial_guess():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        return h

    def structured_initial_guess():
        h = np.zeros(n_points)
        h[::2] = 1.0
        h[-1] = 0.5 if n_points % 2 else 1.0
        return h

    def perturb(h, seed=None):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        return np.clip(h + perturbation, 0.0, 1.0)

    best_c5 = float('inf')
    best_h = initial_h.copy()
    start_time = time.time()

    # Generate initial guesses
    initial_guesses = []
    initial_guesses.append(initial_h)
    initial_guesses.append(random_initial_guess())
    initial_guesses.append(structured_initial_guess())
    initial_guesses.append(perturb(initial_h, seed=seed + 100))
    initial_guesses.append(perturb(initial_h, seed=seed + 200))
    initial_guesses.append(perturb(initial_h, seed=seed + 300))

    for guess in initial_guesses:
        if time.time() - start_time > budget_s - 10:
            break
        try:
            result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint},
