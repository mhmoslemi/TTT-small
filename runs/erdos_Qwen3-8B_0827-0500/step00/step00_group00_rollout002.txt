    dx = 2.0 / n_points
    
    def objective(h):
        # Compute cross-correlation with (1 - h)
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraints: sum(h) = n_points / 2
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    h_initial = initial_h.copy()
    
    # Optimization
    result = minimize(
        objective,
        h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 100}
    )
    
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    else:
        best_h = h_initial
        best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
