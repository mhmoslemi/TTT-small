from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 70  # Start with the initial sample size
    h_initial = initial_h_values  # Assume this is a global variable as per problem statement
    
    # Ensure initial_h_values has the correct length
    if len(h_initial) != n_points:
        raise ValueError("Initial h_values length does not match n_points")
    
    dx = 2.0 / n_points
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr * dx)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    
    # Run optimization with Nelder-Mead
    result = minimize(
        objective,
        x0=h_initial,
        bounds=bounds,
        constraints=constraint,

    elapsed = time.time() - start_time
