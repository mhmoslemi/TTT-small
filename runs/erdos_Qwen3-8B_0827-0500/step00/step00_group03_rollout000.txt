import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is a global variable
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h_initial = initial_h_values.copy()
    
    # Objective function to minimize the maximum overlap
    def objective(h):
        # Compute the correlation between h and 1 - h
        corr = correlate(h, 1 - h, mode='full')
        # The maximum value of the correlation scaled by dx
        return np.max(corr) * dx
    
    # Constraint: sum of h must be n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i] in [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    # Equality constraint
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Optimization settings
    options = {
        'maxiter': 100,
        'ftol': 1e-8,
        'disp': False,
        'eps': 1e-8
    }
    
    # Run optimization using L-BFGS-B for efficiency
    result = minimize(
        objective,
        h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options=options
    )
    
    # Extract the best solution found
    best_h = result.x
    c5_bound = objective(best_h)
    
    return best_h, c5_bound, n_points
