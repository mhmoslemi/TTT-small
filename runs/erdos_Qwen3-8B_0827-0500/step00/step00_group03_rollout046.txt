    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    h_initial = initial_h.copy()
    
    start_time = time.time()
    
    # Optimization settings
    result = minimize(
        objective, 
        h_initial, 
        method='L-BFGS-B', 
        bounds=bounds, 
        constraints=cons, 
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    
    elapsed_time = time.time() - start_time
    
    # If optimization took too long, return the initial solution
    if elapsed_time > budget_s - 1:
        return initial_h, objective(initial_h), n_points
    
    best_h = result.x
    best_c5 = result.fun
    
    return best_h, best_c5, n_points
