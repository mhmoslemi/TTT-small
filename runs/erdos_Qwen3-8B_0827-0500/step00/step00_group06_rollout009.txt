import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 95
    dx = 2.0 / n_points
    initial_h = initial_h_values
    
    # Ensure initial_h has length n_points
    if len(initial_h) != n_points:
        # Resample if needed (for example, by averaging)
        initial_h = np.repeat(initial_h, n_points // len(initial_h) + 1)[:n_points]
    
    # Define the objective function
    def objective(h):
        corr = np.correlate(h, h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define constraints
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    
    # Define bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    h0 = initial_h.copy()
    
    # Start timer
    start_time = time.time()

        constraints=constraints,
        options={'ftol': 1e-9, 'maxiter': 100}
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        # Return the best solution found so far
        return (result.x, result.fun, n_points)
    
    # Extract the best solution
    best_h = result.x
    c5_bound = result.fun
    return (best_h, c5_bound, n_points)
