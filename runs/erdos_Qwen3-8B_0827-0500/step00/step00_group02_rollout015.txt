    start_time = time.time()
    
    # Initial parameters
    n_points = 45
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    # Ensure sum(h) is correct
    if np.abs(np.sum(h) - n_points / 2) > 1e-6:
        h = h / np.sum(h) * (n_points / 2)
    
    def objective(h_values):
        ones_minus_h = 1.0 - h_values
        corr = np.correlate(h_values, ones_minus_h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr
    
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    # Optimization setup
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Perform optimization
    result = opt.minimize(
        objective, 
        h, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    # Check time budget
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (h, evaluate_erdos_solution(h), n_points)
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Check if the optimization improved the result
    if best_c5 < evaluate_erdos_solution(h):
        h = best_h
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(h)
    return (h, final_c5, n_points)
