        h_values = best_h
    
    return best_h, best_c5, n_points

def compute_c5(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def perturb(h_values, n_points, noise_scale=0.05):
    h = np.array(h_values)
    noise = np.random.normal(0, noise_scale, size=h.shape)
    perturbed = h + noise
    # Clip to [0, 1]
    perturbed = np.clip(perturbed, 0.0, 1.0)
    # Scale to maintain sum
    sum_perturbed = np.sum(perturbed)
    scale = (n_points / 2.0) / sum_perturbed
    perturbed *= scale
    return perturbed.tolist()
