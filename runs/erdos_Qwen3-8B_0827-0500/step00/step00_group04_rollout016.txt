import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
import time

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is a global variable provided
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h_initial = initial_h_values.copy()
    
    # Define the objective function
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Define constraints
    bounds = Bounds(0, 1)
    constraint = LinearConstraint(np.ones(n_points), lb=n_points/2, ub=n_points/2)
    
    # Optimization settings
    method = 'L-BFGS-B'
    max_iter = 100  # Limit iterations to stay within time budget
    
    start_time = time.time()
    result = minimize(objective, h_initial, method=method, bounds=bounds, constraints=constraint, options={'maxiter': max_iter})
    
    # Check elapsed time and return best solution
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        # If time is up, return the best solution found so far
        best_h = result.x
        c5_bound = objective(best_h)
        return best_h, c5_bound, n_points
    
    best_h = result.x
    c5_bound = objective(best_h)
    return best_h, c5_bound, n_points
