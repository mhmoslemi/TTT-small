        )
        
        # Check if the optimization was successful
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
            # Check elapsed time
            elapsed_time = time.time() - start_time
            if elapsed_time >= budget_s:
                break
    
    return (best_h, best_c5, n_points)
