    n_points = 100
    dx = 2.0 / n_points
    
    # Generate initial guesses
    initial_guesses = []

        initial_guesses.append(perturbed)

    best_c5 = float('inf')

    elapsed_time = 0.0
    
    for guess in initial_guesses:
        # Convert to numpy array

        # Define the objective function

            max_corr = np.max(corr)

        # Define constraints
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Set up optimization problem

        result = opt.minimize(
