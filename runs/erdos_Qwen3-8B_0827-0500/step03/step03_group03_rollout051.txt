import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points

    # Define the objective function
    def objective(h):
        # Compute cross-correlation with 1 - h
        cross_corr = correlate(h, 1 - h, mode='full')
        # Find the maximum value in the cross-correlation
        max_corr = np.max(cross_corr)
        return max_corr * dx

    # Define constraints
    def constraint(h):
        return np.sum(h) - n_points / 2

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Function to generate a random initial guess
    def random_initial_guess():
        h = np.random.rand(n_points)
        # Normalize to meet the sum constraint
        h = h * (n_points / 2) / np.sum(h)
        return h

    # Function to perturb the initial guess
    def perturb(h, seed=None):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.3, 0.3, size=len(h))
        return np.clip(h + perturbation, 0.0, 1.0)

    # Run multiple optimizations
    best_c5 = float('inf')
    best_h = initial_h.copy()

    # Number of restarts
    num_restarts = 100  # Increased from 50 to 100 for more exploration

        if i % 3 == 0:
            # Use the initial guess with perturbation
            h_initial = perturb(initial_h, seed=seed + i)
        elif i % 3 == 1:
            # Generate a random initial guess
            h_initial = random_initial_guess()
        else:
            # Generate a random initial guess with a different seed
            h_initial = random_initial_guess()

        # Perform optimization with adjusted parameters
        result = minimize(objective, h_initial, method='trust-constr', bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint}],

        if result.success:

            # If optimization fails, try the perturbed guess
