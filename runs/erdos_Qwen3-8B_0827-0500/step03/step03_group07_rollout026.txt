import numpy as np
from scipy.optimize import minimize
from time import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Increased resolution for better optimization
    dx = 2.0 / n_points
    # Generate a random binary initial guess with exactly half 1s
    h_initial = np.zeros(n_points, dtype=np.float64)
    indices = np.random.permutation(n_points)
    h_initial[indices[:n_points//2]] = 1.0
    
    # Define the objective function
    def objective(h):
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr
    
    # Define the constraint for the sum
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h_i
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization settings with tighter tolerances and more iterations
    options = {
        'maxiter': 500,  # Increased from 300 for better convergence
        'ftol': 1e-8,
        'eps': 1e-8,
        'disp': False
    }
    
    # Track start time
    start_time = time()
    
    # Perform optimization
    result = minimize(
        fun=objective,
        x0=h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options=options
    )
    
    # Extract the best solution
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Check if we need to return early due to time budget
    elapsed = time() - start_time
    if elapsed >= budget_s - 1:  # Leave 1 second buffer
        return (best_h, best_c5, n_points)
    
    return (best_h, best_c5, n_points)
