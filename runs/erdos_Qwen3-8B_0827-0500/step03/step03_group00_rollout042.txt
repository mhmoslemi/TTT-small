from scipy.optimize import minimize
from scipy.signal import correlate
from multiprocessing import Pool, cpu_count

    dx = 2.0 / n_points

    def objective(h):
        cross_corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    # Define constraints
    def constraint(h):
        return np.sum(h) - n_points / 2

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def random_initial_guess():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        return h

    def perturb(h, seed=None):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.15, 0.15, size=len(h))
        return np.clip(h + perturbation, 0.0, 1.0)

    # Function to generate a structured initial guess
    def structured_initial_guess():
        # Create a pattern with alternating 1s and 0s
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 3 == 0:
                h[i] = 1.0
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
        return h

    # Function to evaluate a single optimization run
    def evaluate_run(h_initial):
        result = minimize(
            objective, 
            h_initial, 
            method='L-BFGS-B', 
            bounds=bounds, 
            constraints={'type': 'eq', 'fun': constraint}, 
            options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-9}

        if result.success:
            return result.fun, result.x
        else:
            return objective(h_initial), h_initial

    # Number of restarts
    num_restarts = 100  # Increased from 50 for better exploration

    # Use multiprocessing to parallelize evaluations
    pool = Pool(processes=min(cpu_count(), num_restarts))
    results = []
