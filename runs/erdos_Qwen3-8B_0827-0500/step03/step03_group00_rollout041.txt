    for i in range(num_restarts):
        # Choose initial guess
        if i % 3 == 0:
            h_initial = structured_initial.copy()
        elif i % 3 == 1:
            h_initial = random_initial_guess()
        else:
            h_initial = perturb(best_h, seed=seed + i)
        
        # Perform optimization
        result = minimize(
            objective, 
            h_initial, 
            method='L-BFGS-B', 
            bounds=bounds, 
            constraints={'type': 'eq', 'fun': constraint}, 
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            current_c5 = objective(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_initial

    return best_h, best_c5, n_points
