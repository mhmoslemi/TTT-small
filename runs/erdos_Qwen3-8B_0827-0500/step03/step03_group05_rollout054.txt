    pool.join()
    
    # Ensure the best solution is within bounds and satisfies the sum constraint
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_best = np.sum(best_h)
    delta = (n_points / 2) - sum_best
    best_h += delta / n_points
    best_h = np.clip(best_h, 0.0, 1.0)
    
    return best_h, best_c5, n_points

def optimize(guess, dx, n_points, bounds, cons):
    try:
        result = opt.minimize(
            fun=lambda h: np.max(np.correlate(h, 1 - h, mode='full')) * dx,
            x0=guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
        )
        if result.success:
            return result.x, result.fun
        else:
            return guess, np.max(np.correlate(guess, 1 - guess, mode='full')) * dx
    except:
        return guess, np.max(np.correlate(guess, 1 - guess, mode='full')) * dx
