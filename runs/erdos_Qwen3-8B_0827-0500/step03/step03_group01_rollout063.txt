import numpy as np
from scipy.optimize import minimize, LinearConstraint
from scipy.interpolate import interp1d
from multiprocessing import Pool, cpu_count

def generate_perturbed_initial(h, noise_level=0.05):
    perturbed = h + noise_level * np.random.randn(len(h))
    perturbed = np.clip(perturbed, 0.0, 1.0)
    return perturbed

def generate_structured_initial(n_points, num_pulses=3):
    structured = np.zeros(n_points)
    pulse_indices = np.linspace(0, n_points - 1, num_pulses, endpoint=False).astype(int)
    structured[pulse_indices] = 1.0
    target_sum = n_points / 2
    current_sum = np.sum(structured)
    scaling_factor = target_sum / current_sum
    structured *= scaling_factor
    return structured

def generate_random_initial(n_points):
    random_initial = np.random.rand(n_points)
    target_sum = n_points / 2
    current_sum = np.sum(random_initial)
    scaling_factor = target_sum / current_sum
    random_initial *= scaling_factor
    random_initial = np.clip(random_initial, 0.0, 1.0)
    return random_initial

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def optimize_h(initial_guess, n_points, seed):
    np.random.seed(seed)
    def objective(h):
        return evaluate_c5(h, n_points)
    
    A_eq = np.ones(n_points)
    b_eq = n_points / 2
    cons = [LinearConstraint(A_eq, b_eq, b_eq)]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = minimize(objective, initial_guess, method='L-BFGS-B', bounds=bounds, constraints=cons,
                      tol=1e-8, options={'maxiter': 200, 'ftol': 1e-9, 'gtol': 1e-9})
    return result.x, result.fun

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 64  # Increased resolution for better approximation
    initial_guesses = []
    
    # Generate diverse initial guesses
    initial_guesses.append(initial_h_values)
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.03))
    initial_guesses.append(generate_structured_initial(n_points, num_pulses=4))
    initial_guesses.append(generate_structured_initial(n_points, num_pulses=5))
    initial_guesses.append(generate_random_initial(n_points))
    
    best_h = initial_h_values
    best_c5 = evaluate_c5(initial_h_values, n_points)
    elapsed_time = 0.0
    
    # Use parallel processing to evaluate multiple initial guesses
    with Pool(min(cpu_count(), len(initial_guesses))) as pool:
        results = pool.starmap(optimize_h, [(guess, n_points, seed + i) for i, guess in enumerate(initial_guesses)])

            best_h = h
