def run(seed=42, budget_s=100, **kwargs):
    # Set the number of points
    n_points = 100
    dx = 2.0 / n_points
    # Generate a random initial guess with exactly 50 ones
    np.random.seed(seed)
    indices = np.random.choice(n_points, size=50, replace=False)
    h_initial = np.zeros(n_points)
    h_initial[indices] = 1.0
    
    # Define the objective function
    def objective(h):
        ones_minus_h = 1.0 - h
        # Compute the cross-correlation for all shifts
        corr = np.correlate(h, ones_minus_h, mode='full')
        # The maximum overlap integral is the maximum of the cross-correlation values multiplied by dx
        max_corr = np.max(corr) * dx
        return max_corr
    
    # Define the constraint for the sum
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h_i
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization settings with tighter tolerances and more iterations
    options = {
        'maxiter': 500,  # Increase the number of iterations
        'ftol': 1e-9,
        'eps': 1e-9,
        'disp': False
    }
    
    # Perform optimization
    result = minimize(
        fun=objective,
        x0=h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options=options
    )
    
    # Extract the best solution
    best_h = result.x
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
