    initial_guesses.append(generate_structured_initial(n_points, num_pulses=2))
    initial_guesses.append(generate_structured_initial(n_points, num_pulses=3))
    initial_guesses.append(generate_structured_initial(n_points, num_pulses=4))
    initial_guesses.append(np.random.rand(n_points))
    
    # Normalize random initial guess
    random_initial = initial_guesses[-1]
    current_sum = np.sum(random_initial)
    scaling_factor = target_sum / current_sum
    random_initial *= scaling_factor
    random_initial = np.clip(random_initial, 0.0, 1.0)
    initial_guesses[-1] = random_initial
    
    best_h = None
    best_c5 = float('inf')
    elapsed_time = 0.0
    total_time = 0.0
    
    for guess in initial_guesses:
        # Define objective function for DE
        def objective_de(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx + 1e6 * (np.sum(h) - target_sum)**2
        
        # Bounds for DE
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run DE optimization
        result_de = differential_evolution(objective_de, bounds, args=(target_sum, dx), 
                                          strategy='best1bin', popsize=15, mutation=(0.5, 1.0), 
                                          recombination=0.9, tol=1e-6, maxiter=100, disp=False)
        
        # Extract the DE result
        de_h = result_de.x
        de_c5 = objective_de(de_h)
        
        # Refine with SLSQP
        def objective_slsqp(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Constraints for SLSQP
        A_eq = np.ones(n_points)
        b_eq = target_sum
        cons = [LinearConstraint(A_eq, b_eq, b_eq)]
        bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
        
        result_slsqp = minimize(objective_slsqp, de_h, method='SLSQP', bounds=bounds_slsqp, constraints=cons,
                               tol=1e-10, options={'maxiter': 500, 'ftol': 1e-10, 'gtol': 1e-10})
        
        # Check if optimization succeeded
        if result_slsqp.success:
            current_c5 = result_slsqp.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x
        
        # Track time usage
        elapsed_time += result_de.message.split()[-1] if 'time' in result_de.message else 0.1
        elapsed_time += result_slsqp.message.split()[-1] if 'time' in result_slsqp.message else 0.1
        if elapsed_time >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
