        perturbation = np.random.uniform(-0.15, 0.15, size=len(h))
        return np.clip(h + perturbation, 0.0, 1.0)

    # Run multiple optimizations
    best_c5 = float('inf')
    best_h = initial_h.copy()

    # Number of restarts
    num_restarts = 50  # Increased for better exploration

    # Use differential evolution for global search
    def differential_evolution_run():
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            maxiter=100,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        return result.x, result.fun

    # Run differential evolution once
    de_h, de_c5 = differential_evolution_run()
    if de_c5 < best_c5:
        best_c5 = de_c5
        best_h = de_h

    # Run multiple restarts with structured and random initial guesses
    for i in range(num_restarts):
        if i % 3 == 0:
            # Use structured initial guess
            h_initial = structured_initial_guess()
        else:
            # Use random initial guess
            h_initial = random_initial_guess()
        
        # Perturb the initial guess
        h_initial = perturb(h_initial, seed=seed + i)
        
        # Perform optimization with SLSQP
        result = minimize(
            objective, 
            h_initial, 
            method='SLSQP', 
            bounds=bounds, 
            constraints={'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}, 
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # If optimization fails, try the perturbed guess
            current_c5 = objective(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_initial

    return best_h, best_c5, n_points
