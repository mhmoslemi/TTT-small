import numpy as np
import time
import random

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    def evaluate_c5(h):
        cross_corr = np.correlate(h, 1 - h, mode='full')
        return np.max(cross_corr) * dx
    
    def generate_individual():
        h = np.random.rand(n_points)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        delta = (n_points / 2) - sum_h
        if delta != 0:
            h += delta / n_points
            h = np.clip(h, 0.0, 1.0)
        return h
    
    population_size = 20
    population = [initial_h_values.copy()]
    for _ in range(population_size - 1):
        population.append(generate_individual())
    
    best_h = initial_h_values.copy()
    best_c5 = evaluate_c5(best_h)
    
    start_time = time.time()
    
    for gen in range(100):
        fitness = [evaluate_c5(h) for h in population]
        sorted_indices = np.argsort(fitness)
        selected = [population[i] for i in sorted_indices[:int(len(population)*0.5)]]
        
        new_population = []
        while len(new_population) < population_size:
            p1, p2 = random.sample(selected, 2)
            crossover = np.random.randint(1, n_points-1)
            child = np.concatenate([p1[:crossover], p2[crossover:]])
            
            mutation_rate = 0.1
            for i in range(n_points):
                if random.random() < mutation_rate:
                    child[i] += np.random.uniform(-0.05, 0.05)
                    child[i] = np.clip(child[i], 0.0, 1.0)
            
            sum_child = np.sum(child)
            delta = (n_points / 2) - sum_child
            if delta != 0:
                child += delta / n_points
                child = np.clip(child, 0.0, 1.0)

        population = new_population
