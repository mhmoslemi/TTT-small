    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Define the objective function
    def objective(h):
        # Normalize h to meet the sum constraint
        h_normalized = h * (n_points / 2) / np.sum(h)
        # Compute cross-correlation
        cross_corr = correlate(h_normalized, 1 - h_normalized, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = initial_h.copy()

    # Run multiple DE optimizations
    num_restarts = 10  # Number of restarts with different seeds
    for i in range(num_restarts):
        # Set seed for reproducibility
        np.random.seed(seed + i)
        # Run differential evolution
        result = differential_evolution(
            objective, 
            bounds, 
            strategy='best1bin', 
            popsize=20, 
            maxiter=100, 
            tol=1e-9, 
            mutation=(0.5, 1), 
            recombination=0.8, 
            seed=seed + i
        )
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x

    return best_h, best_c5, n_points
