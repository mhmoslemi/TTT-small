import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
from multiprocessing import Pool, cpu_count

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    # Generate initial guesses
    initial_guesses = []
    initial_guesses.append(initial_h_values.copy())
    
    # Generate structured initial guess (alternating 1s and 0s)
    def generate_structured_guess():
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 2 == 0:
                h[i] = 1.0
            else:
                h[i] = 0.0
        current_sum = np.sum(h)
        target_sum = n_points / 2
        if current_sum < target_sum:
            h[0] += (target_sum - current_sum)
        else:
            h[0] -= (current_sum - target_sum)
        return h
    initial_guesses.append(generate_structured_guess())
    
    # Add a few random guesses
    for _ in range(2):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        initial_guesses.append(h)
    
    # Define the objective function with penalty for sum constraint
    def objective(h):
        cross_corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        penalty = 1e4 * abs(np.sum(h) - n_points / 2)
        return max_corr * dx + penalty
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Function to run differential evolution with a given initial guess
    def optimize_with_guess(initial_guess):
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        return result.x, result.fun
    
    # Use multiprocessing to run optimizations in parallel
    with Pool(processes=cpu_count()) as pool:
        results = pool.map(optimize_with_guess, initial_guesses)
    
    # Find the best result
    best_h, best_c5 = min(results, key=lambda x: x[1])
    
    return best_h, best_c5, n_points
