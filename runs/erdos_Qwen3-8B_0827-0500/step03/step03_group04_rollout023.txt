import numpy as np
import time
import scipy.optimize as opt
from scipy.optimize import Bounds
from concurrent.futures import ThreadPoolExecutor

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    
    def objective(h_vec):
        cross_corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(cross_corr) * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    best_h = h_initial.copy()
    best_c5 = objective(best_h)
    
    # Generate diverse initial guesses
    num_initial_guesses = 15
    initial_guesses = []
    for _ in range(num_initial_guesses):
        perturbation = np.random.normal(0, 0.05, n_points)
        perturbed = h_initial + perturbation
        perturbed = np.clip(perturbed, 0.0, 1.0)
        initial_guesses.append(perturbed)
    
    start_time = time.time()
    with ThreadPoolExecutor(max_workers=10) as executor:
        futures = []
        for guess in initial_guesses:
            future = executor.submit(optimize_guess, guess, dx, n_points, bounds, cons, objective)
            futures.append(future)
        
        for future in futures:
            result = future.result()
            if result is not None:
                current_h, current_c5 = result
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break
    
    # Ensure the best solution satisfies the sum constraint
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_best = np.sum(best_h)
    delta = (n_points / 2) - sum_best
    best_h += delta / n_points
    best_h = np.clip(best_h, 0.0, 1.0)
    
    return best_h, best_c5, n_points

def optimize_guess(guess, dx, n_points, bounds, cons, objective):
    try:
        result = opt.minimize(
            fun=objective,
            x0=guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-10, 'disp': False}
        )
        if result.success:
            return result.x, result.fun
        else:
            return None, None
    except:
        return None, None
