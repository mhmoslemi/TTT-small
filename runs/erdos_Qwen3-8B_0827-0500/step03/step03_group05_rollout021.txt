import numpy as np
import time
import scipy.optimize as opt
from scipy.optimize import Bounds

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    # Assume initial_h_values is a global variable as per the problem statement
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Define objective and constraint once
    def objective(h_vec):
        cross_corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(cross_corr) * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Generate multiple initial guesses
    initial_guesses = []
    initial_guesses.append(initial_h_values.copy())
    
    # Generate 3 random guesses
    for _ in range(3):
        h_rand = np.random.rand(n_points)
        sum_h = np.sum(h_rand)
        scale = (n_points / 2) / sum_h
        h_rand = h_rand * scale
        h_rand = np.clip(h_rand, 0.0, 1.0)
        # Adjust sum to be exactly n_points / 2
        sum_rand = np.sum(h_rand)
        delta = (n_points / 2) - sum_rand
        if delta != 0:
            h_rand += delta / n_points
            h_rand = np.clip(h_rand, 0.0, 1.0)
        initial_guesses.append(h_rand)
    
    # Generate a perturbed version of initial_h_values
    perturbed = initial_h_values.copy()
    noise = np.random.normal(0, 0.05, n_points)
    perturbed += noise
    perturbed = np.clip(perturbed, 0.0, 1.0)
    sum_perturbed = np.sum(perturbed)
    delta_perturbed = (n_points / 2) - sum_perturbed
    perturbed += delta_perturbed / n_points
    perturbed = np.clip(perturbed, 0.0, 1.0)
    initial_guesses.append(perturbed)
    
    # Generate structured guesses
    # Structured guess 1: first half is 1, second half is 0
    structured1 = np.zeros(n_points)
    structured1[:n_points//2] = 1.0
    initial_guesses.append(structured1)
    
    # Structured guess 2: single peak in the middle
    structured2 = np.zeros(n_points)
    structured2[n_points//2] = 1.0
    initial_guesses.append(structured2)
    
    # Structured guess 3: alternating 1 and 0 (for even n_points)
    if n_points % 2 == 0:
        structured3 = np.zeros(n_points)
        for i in range(n_points):
            if i % 2 == 0:
                structured3[i] = 1.0
            else:
                structured3[i] = 0.0
        initial_guesses.append(structured3)
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        # Run optimization for this guess
        result = opt.minimize(
            fun=objective,
            x0=guess,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'disp': False}
        )
        
        if result.success:
            current_h = result.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
        else:
            # If optimization fails, use the guess as is
            current_c5 = objective(guess)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = guess
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Ensure the best solution is within bounds and satisfies the sum constraint
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_best = np.sum(best_h)
    delta = (n_points / 2) - sum_best
    best_h += delta / n_points
    best_h = np.clip(best_h, 0.0, 1.0)
    
    return best_h, best_c5, n_points
