from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

        h_normalized = h * (n_points / 2) / np.sum(h)
        cross_corr = correlate(h_normalized, 1 - h_normalized, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Use initial guess to seed the population
    initial_population = [initial_h]
    # Add perturbations to the initial guess to diversify the population

        initial_population.append(perturbed)
    initial_population = np.array(initial_population)

    # Run differential evolution with custom population
    result = differential_evolution(objective, bounds, strategy='best1bin', 
                                   maxiter=100, popsize=15, tol=1e-6, mutation=(0.5, 1), 
                                   recombination=0.8, seed=seed, init=initial_population)
