import scipy.optimize as opt
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Choose n_points
    n_points = 100
    dx = 2.0 / n_points

    initial_guesses = []

    for _ in range(10):  # More perturbations

        perturbed = np.clip(initial_h_values + perturbation, 0, 1)
