                          tol=1e-8, options={'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-8})
        
        # Check if optimization succeeded
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Track time usage
        # Note: Estimating time based on iterations might not be accurate, but for the sake of example
        # we'll assume each optimization takes roughly 0.1 seconds
        elapsed_time += 0.1
        if elapsed_time >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
