    for _ in range(3):  # 3 random guesses
        h_rand = np.random.rand(n_points)
        h_rand = h_rand / np.sum(h_rand) * (n_points / 2)
        initial_guesses.append(h_rand)
    
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    elapsed_time = 0
    
    # Define the objective function for differential evolution
    def objective_de(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        penalty = 1e4 * abs(np.sum(h) - n_points / 2)
        return max_corr * dx + penalty
    
    # Define the objective function for SLSQP
    def objective_slsqp(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define constraints for SLSQP
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Track elapsed time
    start_time = time.time()
    
    for guess in initial_guesses:
        # Convert to numpy array
        h_initial = np.array(guess)
        
        # Run differential evolution
        result_de = opt.differential_evolution(
            fun=objective_de,
            bounds=bounds,
            args=(),
            popsize=15,
            maxiter=100,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        
        # Check if DE was successful
        if result_de.success:
            h_de = result_de.x
            # Run SLSQP on the DE result
            result_slsqp = opt.minimize(
                fun=objective_slsqp,
                x0=h_de,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 200, 'ftol': 1e-9, 'disp': False}
            )
            if result_slsqp.success:
                current_c5 = objective_slsqp(result_slsqp.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
        else:
            # If DE failed, try SLSQP directly
            result_slsqp = opt.minimize(
                fun=objective_slsqp,
                x0=h_initial,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 200, 'ftol': 1e-9, 'disp': False}
            )
            if result_slsqp.success:
                current_c5 = objective_slsqp(result_slsqp.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
        
        # Check elapsed time
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
    
    return (best_h, best_c5, n_points)
