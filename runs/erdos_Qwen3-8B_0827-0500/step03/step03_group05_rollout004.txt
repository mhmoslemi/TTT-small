import numpy as np
import time
import scipy.optimize as opt
from scipy.optimize import Bounds
from concurrent.futures import ThreadPoolExecutor

def optimize_guess(guess, dx, n_points, objective, cons, bounds):
    result = opt.minimize(
        fun=objective,
        x0=guess,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 500, 'ftol': 1e-9, 'disp': False}
    )
    if result.success:
        current_h = result.x
        current_c5 = objective(current_h)
        return current_h, current_c5
    else:
        current_c5 = objective(guess)
        return guess, current_c5

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def objective(h_vec):
        cross_corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(cross_corr) * dx

    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2

    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    initial_guesses = [initial_h_values.copy()]
    for _ in range(5):
        h_rand = np.random.rand(n_points)
        sum_h = np.sum(h_rand)
        scale = (n_points / 2) / sum_h
        h_rand = h_rand * scale
        h_rand = np.clip(h_rand, 0.0, 1.0)
        sum_rand = np.sum(h_rand)
        delta = (n_points / 2) - sum_rand
        if delta != 0:
            h_rand += delta / n_points
            h_rand = np.clip(h_rand, 0.0, 1.0)
        initial_guesses.append(h_rand)

    perturbed = initial_h_values.copy()
    noise = np.random.normal(0, 0.05, n_points)
    perturbed += noise
    perturbed = np.clip(perturbed, 0.0, 1.0)
    sum_perturbed = np.sum(perturbed)
    delta_perturbed = (n_points / 2) - sum_perturbed
    perturbed += delta_perturbed / n_points
    perturbed = np.clip(perturbed, 0.0, 1.0)
    initial_guesses.append(perturbed)

    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    with ThreadPoolExecutor(max_workers=10) as executor:
        futures = []
        for guess in initial_guesses:
            future = executor.submit(optimize_guess, guess, dx, n_points, objective, cons, bounds)
            futures.append(future)

        for future in futures:
            result = future.result()
            if result is not None:
                current_h, current_c5 = result
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h
            elapsed = time.time() - start_time

                break

    best_h = np.clip(best_h, 0.0, 1.0)

    delta = (n_points / 2) - sum_best
    best_h += delta / n_points
    best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, n_points
