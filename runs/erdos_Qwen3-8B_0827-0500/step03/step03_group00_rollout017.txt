from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        cross_corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    def constraint(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def random_initial_guess():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        return h

    def perturb(h, seed=None):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.2, 0.2, size=len(h))
        return np.clip(h + perturbation, 0.0, 1.0)

    best_c5 = float('inf')
    best_h = initial_h.copy()
    num_restarts = 100  # Increased for thorough exploration

    for i in range(num_restarts):
        if i % 3 == 0:
            h_initial = random_initial_guess()
        else:
            h_initial = perturb(best_h, seed=seed + i)
        
        result = minimize(
            objective, 
            h_initial,

        )
