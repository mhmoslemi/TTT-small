from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    n_points = 100
    dx = 2.0 / n_points
    np.random.seed(seed)
    h_initial = np.zeros(n_points)
    indices = np.random.choice(n_points, size=50, replace=False)
    h_initial[indices] = 1.0

    def objective(h):
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    options = {
        'maxiter': 500,
        'ftol': 1e-9,
        'eps': 1e-9,
        'disp': False
    }

    result = minimize(
        fun=objective,
        x0=h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options=options
    )

    best_h = result.x
    best_c5 = objective(best_h)
    return (best_h, best_c5, n_points)
