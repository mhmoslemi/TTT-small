import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Choose n_points
    n_points = 100
    dx = 2.0 / n_points
    
    # Generate initial guesses
    initial_guesses = []
    
    # Generate the interpolated initial guess based on initial_h_values
    dx_old = 2.0 / 42
    h_interpolated = np.zeros(n_points)
    for k in range(n_points):
        x_new = k * dx
        i = int(x_new / dx_old)
        i = np.clip(i, 0, 41)
        h_interpolated[k] = initial_h_values[i]
    initial_guesses.append(h_interpolated)
    
    # Generate perturbed initial guesses based on initial_h_values
    for _ in range(10):  # 10 perturbations
        perturbation = np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(initial_h_values + perturbation, 0, 1)
        perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
        initial_guesses.append(perturbed)
    
    # Generate some random initial guesses
    for _ in range(5):  # 5 random guesses
        h_rand = np.random.rand(n_points)
        h_rand = h_rand / np.sum(h_rand) * (n_points / 2)
        initial_guesses.append(h_rand)
    
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    elapsed_time = 0.0
    
    for guess in initial_guesses:
        # Convert to numpy array
        h_initial = np.array(guess)
        # Define the objective function
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Define constraints
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Set up optimization problem
        start_time = time.time()
        result = opt.minimize(
            fun=objective,
            x0=h_initial,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 500, 'ftol': 1e-9, 'disp': False}
        )
        elapsed_time += time.time() - start_time
        
        if elapsed_time >= budget_s:
            break
        
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return (best_h, best_c5, n_points)
