        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Set up optimization problem
        result = opt.minimize(
            fun=objective,
            x0=h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False}
        )
        
        # Track elapsed time
        start_time = time.time()
        result = opt.minimize(
            fun=objective,
            x0=h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False}
        )
        end_time = time.time()
        elapsed_time += end_time - start_time
        
        if elapsed_time >= budget_s:
            break
        
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return (best_h, best_c5, n_points)
