    # Generate multiple initial guesses by perturbing the initial_h_values
    num_initial_guesses = 5
    initial_guesses = []
    for _ in range(num_initial_guesses):
        perturbation = np.random.normal(0, 0.05, n_points)
        perturbed = h_initial + perturbation
        perturbed = np.clip(perturbed, 0.0, 1.0)
        initial_guesses.append(perturbed)
    
    # Run optimization for each initial guess
    start_time = time.time()
    for i, guess in enumerate(initial_guesses):
        # Run optimization
        result = opt.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 5000, 'ftol': 1e-10, 'disp': False}
        )
        if result.success:
            current_h = result.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
        # Check time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Ensure the best solution is within bounds and satisfies the sum constraint
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_best = np.sum(best_h)
    delta = (n_points / 2) - sum_best
    best_h += delta / n_points
    best_h = np.clip(best_h, 0.0, 1.0)
    
    return best_h, best_c5, n_points
