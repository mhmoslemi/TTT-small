import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is a global variable
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Define the objective function
    def objective(h):
        cross_corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    # Define constraints
    def constraint(h):
        return np.sum(h) - n_points / 2

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Function to generate a random initial guess
    def random_initial_guess():
        h = np.random.rand(n_points)
        # Normalize to meet the sum constraint
        h = h * (n_points / 2) / np.sum(h)
        return h

    # Function to perturb the initial guess
    def perturb(h, seed=None):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.05, 0.05, size=len(h))
        h_perturbed = np.clip(h + perturbation, 0.0, 1.0)
        sum_perturbed = np.sum(h_perturbed)
        scale_factor = (n_points / 2) / sum_perturbed
        h_normalized = h_perturbed * scale_factor
        return h_normalized

    # Track best solution
    best_c5 = float('inf')
    best_h = initial_h.copy()

    # Number of restarts
    num_restarts = 100  # Increased for more exploration

    # Start timing
    import time
    start_time = time.time()

    for i in range(num_restarts):
        # Generate a random initial guess or perturb the initial_h
        if i % 3 == 0:
            # Use the initial guess with perturbation
            h_initial = perturb(initial_h, seed=seed + i)
        elif i % 3 == 1:
            # Generate a random initial guess
            h_initial = random_initial_guess()
        else:
            # Generate a random initial guess with a different seed
            h_initial = random_initial_guess()

        # Perform optimization with adjusted parameters
        result = minimize(objective, h_initial, method='COBYLA', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, 
                          options={'maxiter': 200, 'disp': False})
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # If optimization fails, try the perturbed guess
            current_c5 = objective(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_initial

        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 1:
            break

    return best_h, best_c5, n_points
