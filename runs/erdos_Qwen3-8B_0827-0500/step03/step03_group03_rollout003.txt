from scipy.optimize import minimize
from scipy.signal import correlate
from multiprocessing import Pool, cpu_count

    def objective(h):
        cross_corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    def constraint(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def random_initial_guess():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        return h

    def perturb(h, seed=None):
        np.random.seed(seed)
        perturbation = np.random.normal(0, 0.1, size=len(h))
        return np.clip(h + perturbation, 0.0, 1.0)

    def optimize_single(h_initial):
        result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, 
                          options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-9})
        if result.success:
            return result.fun, result.x
        else:
            return objective(h_initial), h_initial

    num_restarts = 100
    best_c5 = float('inf')
    best_h = initial_h.copy()

    with Pool(processes=cpu_count()) as pool:
        initial_guesses = []
        for i in range(num_restarts):
            if i % 3 == 0:
                h_initial = perturb(initial_h, seed=seed + i)

                h_initial = random_initial_guess()

                h_initial = random_initial_guess()
            initial_guesses.append(h_initial)
