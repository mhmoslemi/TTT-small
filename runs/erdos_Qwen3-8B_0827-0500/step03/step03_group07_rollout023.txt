    def objective(h):
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr

    def constraint_sum(h):

    }
    
    # Perform optimization
    result = minimize(

        method='SLSQP',
        bounds=bounds,
