    for i in range(n_points):
        h_alternating[i] = 1 if i % 2 == 0 else 0
    initial_guesses.append(h_alternating)
    
    # Block binary pattern (first half 1s)
    h_block = np.zeros(n_points)
    h_block[:n_points//2] = 1
    initial_guesses.append(h_block)
    
    # Random binary pattern (exactly 100 ones)
    h_random_binary = np.zeros(n_points)
    indices = np.random.choice(n_points, size=n_points//2, replace=False)
    h_random_binary[indices] = 1
    initial_guesses.append(h_random_binary)
    
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    start_time = time.time()
    
    for guess in initial_guesses:
        # Convert to numpy array
        h_initial = np.array(guess)
        # Define the objective function
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Define constraints
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Set up optimization problem
        result = opt.minimize(
            fun=objective,
            x0=h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
        )
        
        # Check if we've exceeded the budget
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s - 1:  # Leave a buffer
            break
        
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return (best_h, best_c5, n_points)
