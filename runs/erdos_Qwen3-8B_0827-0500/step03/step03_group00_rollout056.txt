from scipy.optimize import minimize
from scipy.signal import correlate
from multiprocessing import Pool, cpu_count
import time

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is a global variable
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Define the objective function
    def objective(h):
        cross_corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    # Define constraint
    def constraint(h):
        return np.sum(h) - n_points / 2

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Function to generate a random initial guess
    def random_initial_guess():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        return h

    # Function to perturb the initial guess
    def perturb(h, seed=None):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        return np.clip(h + perturbation, 0.0, 1.0)

    # Function to run a single optimization
    def optimize(initial_h):
        result = minimize(
            objective, 
            initial_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints={'type': 'eq', 'fun': constraint}, 
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-9}
        )
        if result.success:
            return (result.fun, result.x)
        else:
            return (objective(initial_h), initial_h)

    # Generate initial guesses
    initial_guesses = []
    for i in range(20):
        initial_guesses.append(random_initial_guess())
    for i in range(20):
        initial_guesses.append(perturb(initial_h, seed=seed + i))

    # Run optimizations in parallel
    best_c5 = float('inf')
    best_h = initial_h.copy()
    start_time = time.time()

    with Pool(processes=min(cpu_count(), len(initial_guesses))) as pool:
        results = pool.map(optimize, initial_guesses)
        for c5, h in results:

                best_h = h
            if time.time() - start_time > budget_s - 1:
