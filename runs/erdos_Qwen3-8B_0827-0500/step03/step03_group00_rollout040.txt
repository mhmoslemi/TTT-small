import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Define the objective function
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        cross_corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    # Define the sum constraint
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Function to generate a random initial guess
    def random_initial_guess():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        return h

    # Function to perturb the initial guess
    def perturb(h, seed=None):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.15, 0.15, size=len(h))
        return np.clip(h + perturbation, 0.0, 1.0)

    # Run multiple optimizations
    best_c5 = float('inf')
    best_h = initial_h.copy()

    # Number of restarts
    num_restarts = 50

    for i in range(num_restarts):
        # Generate a random initial guess or perturb the current best
        if i % 3 == 0:
            # Use random initial guess
            h_initial = random_initial_guess()
        else:
            # Perturb the current best solution
            h_initial = perturb(best_h, seed=seed + i)
        
        # Perform optimization with adjusted parameters
        result = minimize(
            objective, 
            h_initial, 
            method='SLSQP', 
            bounds=bounds, 
            constraints={'type': 'eq', 'fun': constraint_sum}, 
            options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # If optimization fails, try the perturbed guess
            current_c5 = objective(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_initial

    return best_h, best_c5, n_points
