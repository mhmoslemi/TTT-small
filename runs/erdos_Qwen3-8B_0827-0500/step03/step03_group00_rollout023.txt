from scipy.optimize import minimize
from scipy.signal import correlate

    def objective(h):
        # Compute cross-correlation with 1 - h
        cross_corr = correlate(h, 1 - h, mode='full')
        # Find the maximum value in the cross-correlation
        max_corr = np.max(cross_corr)
        return max_corr * dx

    # Define constraints
    def constraint(h):
        return np.sum(h) - n_points / 2

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def random_initial_guess():
        h = np.random.rand(n_points)
        # Normalize to meet the sum constraint
        h = h * (n_points / 2) / np.sum(h)
        return h

    # Function to perturb the initial guess
    def perturb(h, seed=None):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.2, 0.2, size=len(h))
        return np.clip(h + perturbation, 0.0, 1.0)

    # Run multiple optimizations
    best_c5 = float('inf')
    best_h = initial_h.copy()

    # Number of restarts
    num_restarts = 100  # Increased for better exploration

    for i in range(num_restarts):

        if i % 3 == 0:

            h_initial = random_initial_guess()
        else:
            # Perturb the current best solution
            h_initial = perturb(best_h, seed=seed + i)
        
        # Perform optimization with adjusted parameters
        result = minimize(
