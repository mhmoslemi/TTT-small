import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    # Use initial_h_values as the starting point
    n_points = 100
    dx = 2.0 / n_points
    h_initial = initial_h_values.copy()
    # Scale to meet the sum constraint
    sum_h = np.sum(h_initial)
    scale = (n_points / 2) / sum_h
    h_initial = h_initial * scale
    h_initial = np.clip(h_initial, 0.0, 1.0)
    # Add small perturbation to avoid local minima
    np.random.seed(seed)
    perturbation = np.random.normal(0, 0.05, size=n_points)
    h_initial += perturbation
    h_initial = np.clip(h_initial, 0.0, 1.0)
    
    # Define the objective function
    def objective(h):
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr
    
    # Define the constraint for the sum
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h_i
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization settings with tighter tolerances and more iterations
    options = {
        'maxiter': 500,
        'ftol': 1e-9,
        'eps': 1e-9,
        'disp': False
    }
    
    # Perform optimization with trust-constr method
    result = minimize(
        fun=objective,
        x0=h_initial,
        method='trust-constr',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options=options
    )
    
    # Extract the best solution
    best_h = result.x
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
