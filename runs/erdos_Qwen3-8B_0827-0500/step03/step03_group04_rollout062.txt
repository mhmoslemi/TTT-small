import numpy as np
import time
import scipy.optimize as opt
from scipy.optimize import Bounds

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    # Assume initial_h_values is a global variable as per the problem statement
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    
    def objective(h_vec):
        cross_corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(cross_corr) * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    best_h = h_initial.copy()
    best_c5 = objective(h_initial)
    
    # Generate multiple initial guesses
    num_initial_guesses = 10
    initial_guesses = []
    
    # Add initial guess based on initial_h_values with perturbation
    perturbation = np.random.normal(0, 0.1, n_points)

    initial_guesses.append(perturbed)
    
    # Generate random initial guesses

        h_scaled = np.clip(h_scaled, 0.0, 1.0)
        initial_guesses.append(h_scaled)
    
    start_time = time.time()
    for guess in initial_guesses:
        # Run optimization
        result = opt.minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-10, 'disp': False}
        )
        if result.success:

            current_c5 = objective(current_h)
            if current_c5 < best_c5:

        # Check time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Ensure the best solution is within bounds and satisfies the sum constraint
    best_h = np.clip(best_h, 0.0, 1.0)
