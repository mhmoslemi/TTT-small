import numpy as np
import time
import scipy.optimize as opt

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    # Assume initial_h_values is a global variable as per the problem statement
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

        sum_penalty = 1000 * np.abs(np.sum(h_vec) - n_points / 2)
        return max_corr + sum_penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Use differential evolution with multiple restarts
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    # Run differential evolution with multiple restarts
    for _ in range(3):
        # Generate a perturbed version of initial_h_values
        guess = initial_h_values.copy()
        noise = np.random.normal(0, 0.05, n_points)
        guess += noise
        guess = np.clip(guess, 0.0, 1.0)
        sum_guess = np.sum(guess)
        delta = (n_points / 2) - sum_guess
        guess += delta / n_points
        guess = np.clip(guess, 0.0, 1.0)
        
        # Run differential evolution
