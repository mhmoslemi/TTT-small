        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=30,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed + run_idx
        )
        # Update best solution
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # Fallback to initial guess if optimization fails
            current_c5 = objective(initial_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = initial_h.copy()

    return best_h, best_c5, n_points
