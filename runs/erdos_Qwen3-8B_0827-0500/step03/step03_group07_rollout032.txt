    h_binary = h_binary * scale
    h_binary = np.clip(h_binary, 0.0, 1.0)
    initial_guesses.append(h_binary)
    
    best_h = None
    best_c5 = float('inf')
    options = {
        'maxiter': 500,
        'ftol': 1e-9,
        'eps': 1e-9,
        'disp': False
    }
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def objective(h):
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    for h_initial in initial_guesses:
        result = minimize(
            fun=objective,
            x0=h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options=options
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return (best_h, best_c5, n_points)
