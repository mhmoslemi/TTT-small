from scipy.optimize import minimize

    global initial_h_values
    h_initial = initial_h_values.copy()
    
    # Normalize to meet the sum constraint
    n_points = len(h_initial)
    sum_h = np.sum(h_initial)
    scale = (n_points / 2) / sum_h
    h_initial = h_initial * scale
    h_initial = np.clip(h_initial, 0.0, 1.0)
    
    dx = 2.0 / n_points
    
    # Define the objective function
    def objective(h):
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr
    
    # Define the constraint for the sum
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h_i
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization settings with tighter tolerances and more iterations
    options = {

        'ftol': 1e-9,
        'eps': 1e-9,
        'disp': False
    }
    
    # Perform optimization using COBYLA method for better handling of bounds
    result = minimize(
        fun=objective,
        x0=h_initial,
        method='COBYLA',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options=options
    )
    
    # Extract the best solution
    best_h = result.x
