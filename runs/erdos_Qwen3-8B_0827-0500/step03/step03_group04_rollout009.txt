    # Assume initial_h_values is a global variable as per the problem statement
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    
    def objective(h_vec):
        cross_corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(cross_corr) * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    best_h = h_initial.copy()
    best_c5 = objective(h_initial)

    num_initial_guesses = 10
    initial_guesses = []

        initial_guesses.append(perturbed)

        initial_guesses.append(h)
    
    start_time = time.time()
    for i, guess in enumerate(initial_guesses):
        # Run optimization
        result = opt.minimize(

        # Check time

    # Ensure the best solution is within bounds and satisfies the sum constraint
    best_h = np.clip(best_h, 0.0, 1.0)
