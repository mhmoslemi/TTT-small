import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    # Use a higher resolution for better accuracy
    n_points = 200
    dx = 2.0 / n_points
    np.random.seed(seed)
    
    # Generate a random initial guess with uniform distribution
    h_initial = np.random.rand(n_points)
    sum_h = np.sum(h_initial)
    scale = (n_points / 2) / sum_h
    h_initial = h_initial * scale
    h_initial = np.clip(h_initial, 0.0, 1.0)
    
    # Define the objective function
    def objective(h):
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr
    
    # Define the constraint for the sum
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h_i
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization settings
    options = {
        'maxiter': 500,
        'ftol': 1e-9,
        'eps': 1e-9,
        'disp': False
    }
    
    # Perform optimization using L-BFGS-B for better convergence
    result = minimize(

        x0=h_initial,

        bounds=bounds,

    )
