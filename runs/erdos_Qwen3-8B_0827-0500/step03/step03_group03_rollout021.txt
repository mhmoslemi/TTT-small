import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is a global variable
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Define the objective function
    def objective(h):
        cross_corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        penalty = 1e4 * np.abs(np.sum(h) - n_points / 2)
        return max_corr * dx + penalty

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Function to generate a structured initial guess
    def structured_initial_guess():
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 3 == 0:
                h[i] = 1.0
            elif i % 3 == 1:
                h[i] = 0.5
            else:
                h[i] = 0.0
        # Normalize to meet the sum constraint
        h = h * (n_points / 2) / np.sum(h)
        return h

    # Run differential evolution with structured initial guess
    result_de = differential_evolution(
        objective, bounds, strategy='best1bin', 
        popsize=15, tol=1e-9, mutation=(0.5, 1), 
        recombination=0.8, maxiter=100, 
        disp=False, seed=seed
    )
    best_h = result_de.x
    best_c5 = objective(best_h)

    # Run SLSQP with multiple restarts for refinement
    def objective_slsqp(h):
        cross_corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    def constraint_slsqp(h):
        return np.sum(h) - n_points / 2

    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]

    # Generate initial guesses
    initial_guesses = []
    for i in range(50):
        if i % 3 == 0:
            h_initial = structured_initial_guess()
        elif i % 3 == 1:
            h_initial = np.random.rand(n_points)
            h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        else:
            h_initial = np.random.rand(n_points)
            h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        initial_guesses.append(h_initial)

    best_c5_slsqp = float('inf')
    best_h_slsqp = best_h.copy()

    for guess in initial_guesses:
        result = minimize(
