import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is a global variable
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Define the objective function with normalization
    def objective(h):
        # Normalize h to meet the sum constraint
        h_normalized = h * (n_points / 2) / np.sum(h)
        cross_corr = correlate(h_normalized, 1 - h_normalized, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run differential evolution with multiple restarts
    best_c5 = float('inf')
    best_h = initial_h.copy()

    # Number of restarts
    num_restarts = 20

    for i in range(num_restarts):
        # Set seed for reproducibility
        np.random.seed(seed + i)
        # Generate a random initial population based on initial_h
        initial_population = np.array([np.random.rand(n_points) * initial_h for _ in range(15)])
        initial_population = np.vstack([initial_population, initial_h])
        initial_population = np.vstack([initial_population, np.random.rand(15, n_points)])
        initial_population = initial_population / np.sum(initial_population, axis=1)[:, np.newaxis] * (n_points / 2)

        result = differential_evolution(
            objective,

            strategy='best1bin',
            maxiter=50,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed + i
        )

        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
