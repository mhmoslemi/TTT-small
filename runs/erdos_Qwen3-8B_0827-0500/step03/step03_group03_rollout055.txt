        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.8,
            tol=1e-6,
            maxiter=100,
            seed=seed + i
        )

        h_reduced = result.x
        sum_h_reduced = np.sum(h_reduced)
        last = (n_points / 2) - sum_h_reduced
        last = np.clip(last, 0.0, 1.0)
        full_h = np.concatenate([h_reduced, [last]])

        cross_corr = correlate(full_h, 1 - full_h, mode='full')
        current_c5 = np.max(cross_corr) * dx

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = full_h

    return best_h, best_c5, n_points
