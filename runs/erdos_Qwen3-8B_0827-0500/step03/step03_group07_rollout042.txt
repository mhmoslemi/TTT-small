from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    # Use a higher resolution for better precision
    n_points = 200

    def objective(h):

        corr = np.correlate(h, ones_minus_h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr

    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h_i
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization settings with tighter tolerances and more iterations
    options = {
        'maxiter': 500,
        'ftol': 1e-9,
        'eps': 1e-9,
        'disp': False
    }
    
    # Perform optimization
    result = minimize(
        fun=objective,
        x0=h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options=options
    )
    
    # Extract the best solution
