import time
import scipy.optimize as opt

    # Assume initial_h_values is a global variable as per the problem statement

        return np.max(cross_corr) * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    best_h = h_initial.copy()
    best_c5 = objective(h_initial)
    
    # Generate multiple initial guesses with diverse perturbations

        perturbed = h_initial + perturbation
        perturbed = np.clip(perturbed, 0.0, 1.0)
        initial_guesses.append(perturbed)
        # Add a random flip of some values
        flip_indices = np.random.choice(n_points, size=5, replace=False)
        perturbed[flip_indices] = 1.0 - perturbed[flip_indices]
        initial_guesses.append(perturbed.copy())
    
    # Deduplicate initial guesses
    initial_guesses = [np.array(g) for g in set(tuple(g) for g in initial_guesses)]
    
    start_time = time.time()
    for i, guess in enumerate(initial_guesses):

        result = opt.minimize(
