    np.random.seed(seed)
    # Assume initial_h_values is a global variable as per the problem statement
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Define objective with penalty
    def objective(h_vec):
        cross_corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_corr = np.max(cross_corr) * dx
        sum_h = np.sum(h_vec)
        penalty = 1e4 * abs(sum_h - target_sum)
        return max_corr + penalty
    
    # Define bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Generate initial guesses
    initial_guesses = []
    initial_guesses.append(initial_h_values.copy())
    
    # Generate 5 random guesses
    for _ in range(5):
        h_rand = np.random.rand(n_points)
        sum_h = np.sum(h_rand)
        scale = (target_sum) / sum_h
        h_rand = h_rand * scale
        h_rand = np.clip(h_rand, 0.0, 1.0)
        # Adjust sum to be exactly target_sum
        sum_rand = np.sum(h_rand)
        delta = target_sum - sum_rand
        if delta != 0:
            h_rand += delta / n_points
            h_rand = np.clip(h_rand, 0.0, 1.0)
        initial_guesses.append(h_rand)
    
    # Generate a perturbed version of initial_h_values
    perturbed = initial_h_values.copy()
    noise = np.random.normal(0, 0.05, n_points)
    perturbed += noise
    perturbed = np.clip(perturbed, 0.0, 1.0)
    sum_perturbed = np.sum(perturbed)
    delta_perturbed = target_sum - sum_perturbed
    perturbed += delta_perturbed / n_points
    perturbed = np.clip(perturbed, 0.0, 1.0)
    initial_guesses.append(perturbed)
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        # Run differential evolution
        result = differential_evolution(
            fun=objective,
            bounds=bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.8,
            tol=1e-6,
            disp=False
        )
        
        current_h = result.x
        current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Ensure the best solution is within bounds and satisfies the sum constraint
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_best = np.sum(best_h)
    delta = target_sum - sum_best
    best_h += delta / n_points
    best_h = np.clip(best_h, 0.0, 1.0)
    
    return best_h, best_c5, n_points
