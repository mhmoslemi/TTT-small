```python
import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is a global variable
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Define the objective function
    def objective(h):
        # Normalize h to meet the sum constraint
        target_sum = n_points / 2
        current_sum = np.sum(h)
        if current_sum == 0:
            return float('inf')
        scale = target_sum / current_sum
        normalized_h = np.clip(h * scale, 0.0, 1.0)
        # Compute cross-correlation
        cross_corr = correlate(normalized_h, 1 - normalized_h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run differential evolution with adjusted parameters
    result = differential_evolution(
        objective,
        bounds,
        popsize=20,
        mutation=(0.5, 1.0),
        recombination=0.8,
        tol=1e-9,
        maxiter=100,
        seed=seed
    )

    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun

    return best_h, best_c5, n_points
```
