        generate_perturbed_initial(initial_h_values, noise_level=0.02),
        generate_perturbed_initial(initial_h_values, noise_level=0.05),
        generate_structured_initial(n_points, num_pulses=3),
        generate_structured_initial(n_points, num_pulses=4),
        generate_structured_initial(n_points, num_pulses=5),
        generate_random_initial(n_points)
    ]
    
    best_h = None
    best_c5 = float('inf')
    
    for guess in initial_guesses:
        # Define objective function
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            dx = 2.0 / n_points
            return max_corr * dx
        
        # Constraints
        A_eq = np.ones(n_points)
        b_eq = n_points / 2
        cons = [LinearConstraint(A_eq, b_eq, b_eq)]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization settings
        result = minimize(objective, guess, method='L-BFGS-B', bounds=bounds, constraints=cons,
                          tol=1e-9, options={'maxiter': 1000, 'disp': False})
        
        # Check if optimization succeeded
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Check elapsed time
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
