    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.05))
    for num_pulses in [3, 4, 5, 6]:
        initial_guesses.append(generate_structured_initial(n_points, num_pulses))
    initial_guesses.append(generate_random_initial(n_points))
    initial_guesses = list(set(tuple(g) for g in initial_guesses))
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    with ProcessPoolExecutor(max_workers=10) as executor:
        future_to_guess = {executor.submit(optimization_task, guess, n_points): guess for guess in initial_guesses}
        for future in as_completed(future_to_guess):
            guess = future_to_guess[future]
            try:
                result = future.result(timeout=10)
                if result is not None:
                    current_h, current_c5 = result
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = current_h
            except Exception as e:
                print(f"Optimization failed: {e}")
    
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        return (best_h, best_c5, n_points)
    return (best_h, best_c5, n_points)

def optimization_task(guess, n_points):
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / n_points
        return max_corr * dx
    
    A_eq = np.ones(n_points)
    b_eq = n_points / 2
    cons = [LinearConstraint(A_eq, b_eq, b_eq)]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = minimize(objective, guess, method='L-BFGS-B', bounds=bounds, constraints=cons,
                      tol=1e-8, options={'maxiter': 200})
    
    if result.success:
        return (result.x, result.fun)
    return None
