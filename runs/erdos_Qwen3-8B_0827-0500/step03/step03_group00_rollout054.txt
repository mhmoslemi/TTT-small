import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        cross_corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)

    # Generate a diverse initial population

        # Normalize each row to meet the sum constraint

        # Clip to [0, 1] to ensure bounds are respected
        population = np.clip(population, 0.0, 1.0)
        return population

    best_c5 = float('inf')
    best_h = initial_h.copy()

    # Run multiple differential evolution optimizations
    num_runs = 20  # Adjusted for better exploration within time budget
    for run_idx in range(num_runs):
        # Generate initial population
        initial_population = generate_initial_population(10)
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            constraints=constraints,
            popsize=10,
            mutation=(0.5, 1),
            recombination=0.8,
            tol=1e-6,
            maxiter=100,
            seed=seed + run_idx
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # Fallback to initial population if optimization fails
            for h in initial_population:
                current_c5 = objective(h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h

    return best_h, best_c5, n_points
