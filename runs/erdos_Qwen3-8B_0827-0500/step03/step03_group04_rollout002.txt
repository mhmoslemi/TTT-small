    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    best_h = h_initial.copy()
    best_c5 = objective(h_initial)
    
    # Generate diverse initial guesses
    initial_guesses = []
    # Perturb initial guess
    for _ in range(3):
        perturbation = np.random.normal(0, 0.05, n_points)
        perturbed = h_initial + perturbation
        perturbed = np.clip(perturbed, 0.0, 1.0)
        initial_guesses.append(perturbed)
    # Random binary vectors
    for _ in range(3):
        indices = np.random.choice(n_points, size=int(n_points / 2), replace=False)
        binary_vec = np.zeros(n_points)
        binary_vec[indices] = 1.0
        initial_guesses.append(binary_vec)
    # Random uniform samples
    for _ in range(3):
        random_vec = np.random.uniform(0, 1, n_points)
        random_vec = np.clip(random_vec, 0.0, 1.0)
        initial_guesses.append(random_vec)
    
    start_time = time.time()
    
    # Run optimization for each initial guess
    for guess in initial_guesses:
        result = opt.minimize(
            fun=objective,
            x0=guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-10, 'eps': 1e-8}
        )
        if result.success:
            current_h = result.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
        # Early exit if time is up
        if time.time() - start_time >= budget_s:
            break
    
    # Final adjustment to ensure constraint satisfaction
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_best = np.sum(best_h)
    delta = (n_points / 2) - sum_best
    best_h += delta / n_points
    best_h = np.clip(best_h, 0.0, 1.0)
    
    return best_h, best_c5, n_points
