        # Normalize h to meet the sum constraint and clip to [0,1]
        sum_h = np.sum(h)
        if sum_h == 0:
            return float('inf')
        scale = (n_points / 2) / sum_h
        h_normalized = h * scale
        h_normalized = np.clip(h_normalized, 0, 1)
        new_sum = np.sum(h_normalized)
        if new_sum == 0:
            return float('inf')
        scale_adj = (n_points / 2) / new_sum
        h_normalized = h_normalized * scale_adj
        # Compute cross-correlation
        cross_corr = correlate(h_normalized, 1 - h_normalized, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    # Define bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run differential evolution with adjusted parameters
    result = differential_evolution(
        objective, bounds, 
        strategy='best1bin', 
        maxiter=100, popsize=20, tol=1e-9, 
        mutation=(0.5, 1), recombination=0.8, 
        seed=seed, 
        disp=False
    )

    best_h = result.x
    best_c5 = result.fun
    return best_h, best_c5, n_points
