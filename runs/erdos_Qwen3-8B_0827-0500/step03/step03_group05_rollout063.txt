from multiprocessing import Pool, cpu_count

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)

        cross_corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(cross_corr) * dx

        return np.sum(h_vec) - n_points / 2

    sum_structured = np.sum(structured_guess)
    scale = (n_points / 2) / sum_structured
    structured_guess *= scale
    structured_guess = np.clip(structured_guess, 0.0, 1.0)

def optimize_guess(guess, dx, n_points, objective, constraint_sum, bounds, cons):
    result = opt.minimize(
        fun=objective,
        x0=guess,
        method='COBYLA',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 1000, 'disp': False}
    )
    
    if result.success:
        current_h = result.x
        current_c5 = objective(current_h)
        return current_h, current_c5
    else:
        # If optimization fails, use the guess as is
        current_c5 = objective(guess)
        return guess, current_c5
