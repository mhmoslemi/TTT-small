import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    n_points = 100
    dx = 2.0 / n_points
    np.random.seed(seed)
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    # Try multiple initial guesses
    for _ in range(5):
        # Generate a random binary initial guess with exactly 50 ones
        h_initial = np.zeros(n_points)
        indices = np.random.choice(n_points, size=n_points // 2, replace=False)
        h_initial[indices] = 1.0
        
        # Define the objective function
        def objective(h):
            ones_minus_h = 1.0 - h
            corr = np.correlate(h, ones_minus_h, mode='full')
            max_corr = np.max(corr) * dx
            return max_corr
        
        # Define the constraint for the sum
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        # Bounds for each h_i
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization settings
        options = {
            'maxiter': 500,
            'ftol': 1e-9,
            'eps': 1e-9,
            'disp': False
        }
        
        # Perform optimization
        result = minimize(
            fun=objective,
            x0=h_initial,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options=options
        )
        
        # Update best solution
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    return (best_h, best_c5, n_points)
