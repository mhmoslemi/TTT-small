    perturbed = initial_h_values.copy()
    noise = np.random.normal(0, 0.05, n_points)
    perturbed += noise
    perturbed = np.clip(perturbed, 0.0, 1.0)
    sum_perturbed = np.sum(perturbed)
    delta_perturbed = (n_points / 2) - sum_perturbed
    perturbed += delta_perturbed / n_points
    perturbed = np.clip(perturbed, 0.0, 1.0)
    initial_guesses.append(perturbed)
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        # Run optimization for this guess
        result = opt.minimize(
            fun=objective,
            x0=guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-9, 'disp': False}
        )
        
        if result.success:
            current_h = result.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
        else:
            # If optimization fails, use the guess as is
            current_c5 = objective(guess)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = guess
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Ensure the best solution is within bounds and satisfies the sum constraint
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_best = np.sum(best_h)
    delta = (n_points / 2) - sum_best
    best_h += delta / n_points
    best_h = np.clip(best_h, 0.0, 1.0)
    
    return best_h, best_c5, n_points
