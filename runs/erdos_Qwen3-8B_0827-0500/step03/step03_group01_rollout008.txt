            dx = 2.0 / n_points
            penalty = 1e3 * np.abs(np.sum(h) - n_points / 2)
            return max_corr * dx + penalty
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = differential_evolution(
            objective, bounds, strategy='best1bin', popsize=15, mutation=(0.5, 1),
            recombination=0.9, tol=1e-6, maxiter=100, seed=seed
        )
        
        if result.success:
            current_c5 = result.fun - 1e3 * np.abs(np.sum(result.x) - n_points / 2)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        elapsed_time += result.message.split()[-1] if 'time' in result.message else 0.1
        if elapsed_time >= budget_s:
            break
    
    if best_h is None:
        best_h = initial_h_values
        best_c5 = np.correlate(best_h, 1 - best_h, mode='full').max() * (2.0 / n_points)
    
    return (best_h, best_c5, n_points)
