    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.05))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.1))
    
    # Add structured initial guesses with varying pulse counts
    for num_pulses in [2, 3, 4, 5, 6, 7, 8]:
        initial_guesses.append(generate_structured_initial(n_points, num_pulses))
    
    # Normalize random initial guesses
    for _ in range(3):
        random_initial = np.random.rand(n_points)
        target_sum = n_points / 2
        current_sum = np.sum(random_initial)
        scaling_factor = target_sum / current_sum
        random_initial *= scaling_factor
        random_initial = np.clip(random_initial, 0.0, 1.0)
        initial_guesses.append(random_initial)
    
    best_h = None
    best_c5 = float('inf')
    start_time = np.datetime64('now')
    
    for guess in initial_guesses:
        # Define objective function with normalization
        def objective(h):
            h = np.clip(h, 0.0, 1.0)
            h = h / np.sum(h) * (n_points / 2)
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            dx = 2.0 / n_points
            return max_corr * dx
        
        # Bounds for differential evolution
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution
        result = differential_evolution(objective, bounds, strategy='best1bin', 
                                       popsize=15, maxiter=100, tol=1e-10, 
                                       mutation=(0.5, 1.0), recombination=0.8, 
                                       polish=True, seed=seed)
        
        # Check if optimization succeeded
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Track time usage
        elapsed_time = np.datetime64('now') - start_time
        if np.greater(elapsed_time, np.timedelta64(budget_s, 's')):
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
