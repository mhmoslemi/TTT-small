    initial_guesses = []
    initial_guesses.append(initial_h_values)
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.02, seed=seed+1))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.05, seed=seed+2))
    initial_guesses.append(generate_structured_initial(n_points, num_pulses=2, seed=seed+3))
    initial_guesses.append(generate_structured_initial(n_points, num_pulses=3, seed=seed+4))
    initial_guesses.append(generate_structured_initial(n_points, num_pulses=4, seed=seed+5))
    initial_guesses.append(generate_structured_initial(n_points, num_pulses=5, seed=seed+6))
    initial_guesses.append(generate_random_initial(n_points, seed=seed+7))
    initial_guesses.append(generate_random_initial(n_points, seed=seed+8))
    initial_guesses.append(generate_random_initial(n_points, seed=seed+9))
    
    # Function to evaluate the objective
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / n_points
        return max_corr * dx
    
    # Constraints
    A_eq = np.ones(n_points)
    b_eq = n_points / 2
    cons = [LinearConstraint(A_eq, b_eq, b_eq)]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Function to run optimization for a single guess
    def optimize(guess):
        result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=cons,
                          tol=1e-10, options={'maxiter': 1000, 'ftol': 1e-10, 'gtol': 1e-10})
        return result
    
    # Use multiprocessing to run optimizations in parallel
    with Pool(processes=cpu_count()) as pool:
        results = pool.map(optimize, initial_guesses)
    
    # Find the best result
    best_h = None
    best_c5 = float('inf')
    for result in results:
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
