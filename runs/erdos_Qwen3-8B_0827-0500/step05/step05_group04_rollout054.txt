        h = np.zeros(n_points)
        h[:-1] = h_vars
        sum_h = np.sum(h_vars)
        h[-1] = S - sum_h
        h[-1] = np.clip(h[-1], 0.0, 1.0)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / n_points
        return max_corr * dx
    
    # Define bounds for the variables
    bounds = [(0.0, 1.0) for _ in range(n_points - 1)]
    
    best_h = None
    best_c5 = float('inf')
    elapsed_time = 0.0
    
    for guess in initial_guesses:
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            popsize=15,
            mutation=(0.5, 1.0),
            recombination=0.9,
            tol=1e-10,
            maxiter=100,
            strategy='best1bin',
            polish=True,
            disp=False
        )
        # Check if optimization succeeded
        if result.success:
            # Convert the result to the full h array
            h_vars_result = result.x
            h_result = np.zeros(n_points)
            h_result[:-1] = h_vars_result
            sum_h = np.sum(h_vars_result)
            h_result[-1] = S - sum_h
            h_result[-1] = np.clip(h_result[-1], 0.0, 1.0)
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_result
        # Track time usage (approximate)
        elapsed_time += 10.0  # Placeholder; actual time tracking would require more complex handling
        if elapsed_time >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        # Convert initial_h_values to the new variable structure
        h_vars_initial = initial_h_values[:-1]
        h_result_initial = np.zeros(n_points)
        h_result_initial[:-1] = h_vars_initial
        sum_h = np.sum(h_vars_initial)
        h_result_initial[-1] = S - sum_h
        h_result_initial[-1] = np.clip(h_result_initial[-1], 0.0, 1.0)
        best_h = h_result_initial
        best_c5 = objective(h_vars_initial)
    
    return (best_h, best_c5, n_points)
