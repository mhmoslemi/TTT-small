    # Use parallel processing to evaluate multiple initial guesses
    results = Parallel(n_jobs=10, backend='threading', timeout=None)(
        delayed(optimize_h)(guess, n_points) for guess in initial_guesses
    )
    
    for h, c5 in results:
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    return (best_h, best_c5, n_points)
