    current_sum = np.sum(random_initial)
    scaling_factor = target_sum / current_sum
    random_initial *= scaling_factor
    random_initial = np.clip(random_initial, 0.0, 1.0)
    return random_initial

def generate_sine_initial(n_points):
    x = np.linspace(0, 2, n_points)
    h = 0.5 + 0.5 * np.sin(np.pi * x)
    target_sum = n_points / 2
    current_sum = np.sum(h)
    scaling_factor = target_sum / current_sum
    h *= scaling_factor
    return h

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    
    # Generate diverse initial guesses with varied structured patterns
    initial_guesses = [initial_h_values]
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.03))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.05))
    for num_pulses in [3, 4, 5, 6, 7]:
        initial_guesses.append(generate_structured_initial(n_points, num_pulses))
    initial_guesses.append(generate_random_initial(n_points))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.1))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.01))
    initial_guesses.append(generate_sine_initial(n_points))  # New initial guess
    
    best_h = None
    best_c5 = float('inf')
    elapsed_time = 0.0
    
    for guess in initial_guesses:
        # Define objective function
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            dx = 2.0 / n_points
            return max_corr * dx
        
        # Constraints
        A_eq = np.ones(n_points)
        b_eq = n_points / 2
        cons = [LinearConstraint(A_eq, b_eq, b_eq)]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization settings
        start_opt_time = time.time()
        result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=cons,
                          tol=1e-8, options={'maxiter': 2000, 'ftol': 1e-10, 'disp': False})
        elapsed_opt_time = time.time() - start_opt_time
        elapsed_time += elapsed_opt_time
        
        # Check if optimization succeeded
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Check budget
        if elapsed_time >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
