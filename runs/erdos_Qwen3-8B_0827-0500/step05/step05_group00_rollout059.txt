    current_sum = np.sum(random_initial)
    scaling_factor = target_sum / current_sum
    random_initial *= scaling_factor
    random_initial = np.clip(random_initial, 0.0, 1.0)
    return random_initial

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    
    # Generate diverse initial guesses
    initial_guesses = [initial_h_values]
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.03))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.05))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.1))
    
    for num_pulses in [2, 3, 4, 5, 6]:
        initial_guesses.append(generate_structured_initial(n_points, num_pulses))
    
    initial_guesses.append(generate_random_initial(n_points))
    initial_guesses.append(generate_random_initial(n_points))
    initial_guesses.append(generate_random_initial(n_points))
    
    # Scale initial guesses to meet the integral constraint
    scaled_initial_guesses = []
    for guess in initial_guesses:
        sum_h = np.sum(guess)
        scaling_factor = (n_points / 2) / sum_h
        scaled_guess = guess * scaling_factor
        scaled_initial_guesses.append(scaled_guess)
    
    # Use the first few scaled guesses as initial population
    initial_population = scaled_initial_guesses[:5]
    
    # Define the objective function
    def objective(h_candidate):
        sum_h = np.sum(h_candidate)
        if sum_h == 0:
            return np.inf
        scaling_factor = (n_points / 2) / sum_h
        h_scaled = h_candidate * scaling_factor
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / n_points
        return max_corr * dx
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution
    result = differential_evolution(
        objective, 
        bounds, 
        popsize=10, 
        strategy='best1bin', 
        tol=1e-10, 
        maxiter=100, 
        mutation=(0.5, 1), 
        recombination=0.8, 
        initial_population=initial_population
    )
    
    # Return the best solution found
    best_h = result.x
    best_c5 = result.fun
    return (best_h, best_c5, n_points)
