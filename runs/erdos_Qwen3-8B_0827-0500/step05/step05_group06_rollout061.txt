    if seed is not None:

    perturbed = np.clip(perturbed, 0.0, 1.0)
    return perturbed

def generate_structured_initial(n_points, num_pulses=2, seed=None):

        np.random.seed(seed)

    target_sum = n_points / 2
    current_sum = np.sum(structured)
