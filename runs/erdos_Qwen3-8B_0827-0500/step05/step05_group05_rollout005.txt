            objective, bounds, popsize=15, maxiter=200, tol=1e-8,
            mutation=(0.5, 1.0), recombination=0.8, strategy='best1bin'
        )
        
        if result.success:
            current_h = result.x
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * (2.0 / n_points)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
        
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
    
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
