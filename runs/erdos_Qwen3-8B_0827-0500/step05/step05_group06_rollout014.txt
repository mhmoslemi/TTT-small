    spacing = (2.0 - num_pulses * pulse_length) / (num_pulses - 1)
    for i in range(num_pulses):
        start = i * (pulse_length + spacing) if i > 0 else 0.0
        end = start + pulse_length
        idx_start = int(start / (2.0 / n_points))
        idx_end = int(end / (2.0 / n_points))
        structured[idx_start:idx_end] = 1.0
    target_sum = n_points / 2
    current_sum = np.sum(structured)
    scaling_factor = target_sum / current_sum
    structured *= scaling_factor
    return structured

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    start_time = time.time()
    
    # Generate diverse initial guesses
    initial_guesses = [initial_h_values]
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.02, seed=seed+1))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.05, seed=seed+2))
    for num_pulses in range(2, 8):  # Add more pulse numbers
        initial_guesses.append(generate_structured_initial(n_points, num_pulses=num_pulses, seed=seed+3 + num_pulses))
    # Add symmetric initial guesses
    for num_pulses in [4, 6]:  # Symmetric pulse numbers
        initial_guesses.append(generate_structured_initial(n_points, num_pulses=num_pulses, seed=seed+3 + num_pulses*2))
    # Add random initial guess
    random_initial = np.random.rand(n_points)
    target_sum = n_points / 2
    current_sum = np.sum(random_initial)
    scaling_factor = target_sum / current_sum
    random_initial *= scaling_factor
    random_initial = np.clip(random_initial, 0.0, 1.0)
    initial_guesses.append(random_initial)
    
    best_h = None
    best_c5 = float('inf')
    
    for guess in initial_guesses:
        # Define objective function
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            dx = 2.0 / n_points
            return max_corr * dx
        
        # Constraints
        A_eq = np.ones(n_points)
        b_eq = n_points / 2
        cons = [LinearConstraint(A_eq, b_eq, b_eq)]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization settings
        result = minimize(objective, guess, method='L-BFGS-B', bounds=bounds, constraints=cons,
                          tol=1e-10, options={'maxiter': 1000, 'ftol': 1e-10, 'gtol': 1e-10})
        
        # Check if optimization succeeded
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Check elapsed time
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
