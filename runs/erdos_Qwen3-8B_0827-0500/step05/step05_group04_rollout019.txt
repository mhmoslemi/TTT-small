        # Differential Evolution optimization
        result = differential_evolution(
            lambda h: objective(h, n_points, target_sum),
            bounds,
            constraints=cons,
            strategy='best1bin',
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.8,
            tol=1e-10,
            maxiter=200,
            disp=False,
            polish=True,
            vectorized=True
        )
        
        # Check if optimization succeeded
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Track time usage
        elapsed_time += result.message.split()[-1] if 'time' in result.message else 0.1
        if elapsed_time >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = objective(best_h, n_points, target_sum)
    
    return (best_h, best_c5, n_points)
