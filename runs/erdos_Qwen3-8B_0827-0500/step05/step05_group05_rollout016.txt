    target_sum = n_points / 2
    current_sum = np.sum(structured)
    scaling_factor = target_sum / current_sum
    structured *= scaling_factor
    return structured

def generate_random_initial(n_points):
    random_initial = np.random.rand(n_points)
    target_sum = n_points / 2
    current_sum = np.sum(random_initial)
    scaling_factor = target_sum / current_sum
    random_initial *= scaling_factor
    random_initial = np.clip(random_initial, 0.0, 1.0)
    return random_initial

def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    max_integral = 0.0
    for m in range(-n_points + 1, n_points):
        integral = 0.0
        for i in range(n_points):
            j = i + m
            if 0 <= j < n_points:
                integral += h_values[i] * (1 - h_values[j]) * dx
        max_integral = max(max_integral, integral)
    return max_integral

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    
    # Generate diverse initial guesses with varied structured patterns
    initial_guesses = [initial_h_values]
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.03))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.05))
    for num_pulses in [3, 4, 5, 6, 7]:
        initial_guesses.append(generate_structured_initial(n_points, num_pulses))
    initial_guesses.append(generate_random_initial(n_points))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.1))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.01))
    
    best_h = None
    best_c5 = float('inf')
    elapsed_time = 0
    total_time = 0
    
    def objective(h):
        return evaluate_erdos_solution(h)
    
    # Constraints
    A_eq = np.ones(n_points)
    b_eq = n_points / 2
    cons = [LinearConstraint(A_eq, b_eq, b_eq)]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization settings
    results = []
    
    def optimize_guess(guess):
        nonlocal elapsed_time
        result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=cons,
                          tol=1e-8, options={'maxiter': 1000, 'ftol': 1e-10, 'disp': False})
        if result.success:
            current_c5 = result.fun
            return (current_c5, result.x)
        return (float('inf'), guess)
    
    # Use multiprocessing to parallelize the optimization
    with mp.Pool(processes=10) as pool:
        results = pool.map(optimize_guess, initial_guesses)
    
    for c5, h in results:
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
