        scaling_factor = target_sum / current_sum
        guess_scaled = guess * scaling_factor
        guess_scaled = np.clip(guess_scaled, 0.0, 1.0)
        result = differential_evolution(
            objective, bounds, popsize=15, tol=1e-8, maxiter=100,
            mutation=(0.5, 1), recombination=0.8, seed=seed
        )
        if result.success:
            current_c5 = result.fun - 1e6 * np.abs(np.sum(result.x) - target_sum)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
    
    if best_h is None:
        best_h = initial_h_values
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
