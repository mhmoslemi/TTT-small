    for guess in initial_guesses:
        # Define objective function with penalty for constraint violation
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            dx = 2.0 / n_points
            c5 = max_corr * dx
            # Penalty for sum(h) not matching target
            penalty = 1e6 * abs(np.sum(h) - target_sum)
            return c5 + penalty
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        # Run differential evolution with early stopping
        result = differential_evolution(
            objective, bounds, popsize=15, mutation=(0.5, 1.0), recombination=0.8,
            tol=1e-6, maxiter=100, disp=False
        )
        
        # Check if optimization succeeded
        if result.success:
            current_c5 = result.fun - 1e6 * abs(np.sum(result.x) - target_sum)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Track time usage (approximate)
        elapsed_time += 0.1  # Simplified time tracking for demonstration
        if elapsed_time >= budget_s:
            break
    
    # Fallback to best guess if no improvement found
    if best_h is None:
        best_h = initial_guesses[0]
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
