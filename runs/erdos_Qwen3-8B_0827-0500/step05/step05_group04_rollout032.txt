import numpy as np
from scipy.optimize import minimize, LinearConstraint
from scipy.interpolate import interp1d

def generate_structured_initial(n_points, num_pulses):
    structured = np.zeros(n_points)
    pulse_length = 1.0 / num_pulses
    spacing = (2.0 - num_pulses * pulse_length) / (num_pulses - 1)
    for i in range(num_pulses):
        start = i * (pulse_length + spacing) if i > 0 else 0.0
        end = start + pulse_length
        idx_start = int(start / (2.0 / n_points))
        idx_end = int(end / (2.0 / n_points))
        structured[idx_start:idx_end] = 1.0
    target_sum = n_points / 2
    current_sum = np.sum(structured)
    scaling_factor = target_sum / current_sum
    structured *= scaling_factor
    return structured

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 64  # Smaller sequence for faster optimization
    
    # Generate diverse initial guesses
    initial_guesses = []
    for num_pulses in [2, 3, 4, 5, 6]:
        initial_guesses.append(generate_structured_initial(n_points, num_pulses))
    initial_guesses.append(np.random.rand(n_points))
    initial_guesses.append(np.zeros(n_points))
    initial_guesses.append(np.ones(n_points))
    
    # Normalize random initial guess
    random_initial = initial_guesses[-2]
    target_sum = n_points / 2
    current_sum = np.sum(random_initial)
    scaling_factor = target_sum / current_sum
    random_initial *= scaling_factor
    random_initial = np.clip(random_initial, 0.0, 1.0)
    initial_guesses[-2] = random_initial
    
    best_h = None
    best_c5 = float('inf')
    elapsed_time = 0.0
    
    for guess in initial_guesses:
        # Define objective function
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            dx = 2.0 / n_points
            return max_corr * dx
        
        # Constraints
        A_eq = np.ones(n_points)
        b_eq = n_points / 2
        cons = [LinearConstraint(A_eq, b_eq, b_eq)]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization settings
        result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=cons,
                          tol=1e-10, options={'maxiter': 2000, 'ftol': 1e-10, 'gtol': 1e-10})
        
        # Check if optimization succeeded
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Track time usage
        elapsed_time += result.message.split()[-1] if 'time' in result.message else 0.1
        if elapsed_time >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_guesses[0]
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
