    if result_slsqp.success:
        best_h = result_slsqp.x
        best_c5 = result_slsqp.fun
    else:
        # Fallback to normalized differential evolution result
        best_h = h_normalized
        best_c5 = objective_de(h_normalized)
    
    return (best_h, best_c5, n_points)
