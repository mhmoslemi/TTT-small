            return max_corr * dx
        
        # Bounds for differential evolution
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization settings
        result = differential_evolution(objective, bounds, popsize=15, maxiter=100, tol=1e-10)
        
        # Check if optimization succeeded
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Track time usage (approximate)
        # For simplicity, assume each optimization takes about 10 seconds
        # But this is not accurate; in reality, we need to track actual time
        # However, without a timer, this is not possible. So, we'll proceed.
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
