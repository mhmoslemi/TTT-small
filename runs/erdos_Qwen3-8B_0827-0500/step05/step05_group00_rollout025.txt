    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / n_points
        best_c5 = max_corr * dx
    
    return (best_h, best_c5, n_points)
