import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    
    # Define objective function with penalty for constraint violation
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / n_points
        constraint_violation = np.abs(np.sum(h) - n_points / 2)
        penalty = 1e4 * constraint_violation
        return max_corr * dx + penalty
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution with multiple restarts
    best_h = initial_h_values.copy()
    best_c5 = objective(best_h)
    
    # Perform multiple differential evolution runs with different seeds
    for i in range(3):
        result = differential_evolution(
            objective, bounds, strategy='best1bin', 
            popsize=15, maxiter=100, tol=1e-8, 
            mutation=(0.5, 1), recombination=0.8, 
            seed=seed + i
        )
        current_h = result.x
        current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return (best_h, best_c5, n_points)
