    initial_guesses.append(generate_random_initial(n_points, seed=seed+3))
    
    best_h = None
    best_c5 = float('inf')
    elapsed_time = 0.0
    
    for guess in initial_guesses:
        # Define objective function with penalty
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            dx = 2.0 / n_points
            penalty = λ * (np.sum(h) - target_sum)**2
            return max_corr * dx + penalty
        
        # Bounds for each variable
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution
        result = differential_evolution(objective, bounds, strategy='best1bin', 
                                       popsize=15, mutation=(0.5, 1.0), recombination=0.8, 
                                       tol=1e-9, maxiter=100, disp=False)
        
        # Check if optimization succeeded
        if result.success:
            # Calculate actual C5 by subtracting penalty
            sum_h = np.sum(result.x)
            penalty_val = λ * (sum_h - target_sum)**2
            current_c5 = result.fun - penalty_val
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Track time usage (approximate)
        elapsed_time += 0.1  # Assuming each iteration takes about 0.1 seconds
        if elapsed_time >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
