    return perturbed

def generate_structured_initial(n_points, num_pulses, spacing_factor, pulse_width):
    structured = np.zeros(n_points)
    total_width = num_pulses * pulse_width
    spacing = (2.0 - total_width) / (num_pulses - 1)
    for i in range(num_pulses):
        start = i * (pulse_width + spacing) if i > 0 else 0.0
        end = start + pulse_width
        idx_start = int(start / (2.0 / n_points))
        idx_end = int(end / (2.0 / n_points))
        structured[idx_start:idx_end] = 1.0
    target_sum = n_points / 2
    current_sum = np.sum(structured)
    scaling_factor = target_sum / current_sum
    structured *= scaling_factor
    return structured

def objective(h, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    dx = 2.0 / n_points
