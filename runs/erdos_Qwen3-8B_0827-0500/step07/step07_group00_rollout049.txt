    start_time = time.time()
    
    with Pool(processes=min(cpu_count(), 10)) as pool:
        results = []
        for guess in initial_guesses:
            results.append(pool.apply_async(optimize_initial_guess, (guess, n_points)))
        
        while results:
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break
            result = results.pop(0)
            optimized_h, current_c5 = result.get()
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
    
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
