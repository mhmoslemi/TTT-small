        scaling_factor = target_sum / current_sum
        structured *= scaling_factor
        initial_guesses.append(structured)
    
    # Generate random initial guesses
    for _ in range(3):
        random_h = np.random.rand(n_points)
        random_h = random_h / np.sum(random_h) * (n_points / 2)
        initial_guesses.append(random_h)
    
    # Define objective function
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / n_points
        return max_corr * dx
    
    # Constraints
    A_eq = np.ones(n_points)
    b_eq = n_points / 2
    cons = [LinearConstraint(A_eq, b_eq, b_eq)]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Function to run optimization for a single guess
    def optimize_guess(guess):
        result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=cons,
                          tol=1e-8, options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False})
        if result.success:
            return (result.fun, result.x)
        else:
            return (np.inf, guess)  # Return worst case if optimization fails
    
    # Use multiprocessing
    num_cpus = min(cpu_count(), len(initial_guesses))
    start_time = time.time()
    best_c5 = np.inf
    best_h = interpolated_h  # Fallback
    
    with Pool(processes=num_cpus) as pool:
        # Submit all tasks
        results = pool.map(optimize_guess, initial_guesses)
        # Check elapsed time
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            # If time is up, return the best found so far
            for c5, h in results:
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h
            return (best_h, best_c5, n_points)
        else:
            # Process results
            for c5, h in results:
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h
            return (best_h, best_c5, n_points)
