            polish_options={'maxiter': 50, 'ftol': 1e-12}
        )
        
        # Check if optimization succeeded
        if result.success:
            current_c5 = result.fun - 1e4 * np.abs(np.sum(result.x) * (2.0 / n_points) - 1)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Check time budget
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
    
    # Fallback to interpolated initial guess if no improvement found
    if best_h is None:
        best_h = interpolated_h
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
