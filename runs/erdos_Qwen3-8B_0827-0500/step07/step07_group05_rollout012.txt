    initial_guesses = []
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.01))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.05))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.1))
    initial_guesses.append(generate_structured_initial(n_points, num_pulses=2, spacing_factor=1.2, pulse_width=0.4))
    initial_guesses.append(generate_structured_initial(n_points, num_pulses=3, spacing_factor=1.0, pulse_width=0.5))
    initial_guesses.append(generate_structured_initial(n_points, num_pulses=4, spacing_factor=0.8, pulse_width=0.6))
    initial_guesses.append(generate_structured_initial(n_points, num_pulses=5, spacing_factor=0.7, pulse_width=0.5))
    initial_guesses.append(generate_structured_initial(n_points, num_pulses=6, spacing_factor=0.6, pulse_width=0.5))
    initial_guesses.append(generate_structured_initial(n_points, num_pulses=7, spacing_factor=0.5, pulse_width=0.5))
    initial_guesses.append(generate_structured_initial(n_points, num_pulses=8, spacing_factor=0.4, pulse_width=0.5))
    initial_guesses.append(generate_structured_initial(n_points, num_pulses=9, spacing_factor=0.3, pulse_width=0.5))
    initial_guesses.append(generate_structured_initial(n_points, num_pulses=10, spacing_factor=0.2, pulse_width=0.5))
    initial_guesses.append(np.random.rand(n_points))
    
    # Normalize random initial guess
    random_initial = initial_guesses[-1]
    target_sum = n_points / 2
    current_sum = np.sum(random_initial)
    scaling_factor = target_sum / current_sum
    random_initial *= scaling_factor
    random_initial = np.clip(random_initial, 0.0, 1.0)
    initial_guesses[-1] = random_initial
    
    best_h = None
    best_c5 = float('inf')
    elapsed_time = 0.0
    
    for guess in initial_guesses:
        # Define objective function
        def objective(h):
            dx = 2.0 / n_points
            max_overlap = 0.0
            for m in range(-n_points + 1, n_points):
                valid_i = []
                for i in range(n_points):
                    j = i + m
                    if 0 <= j < n_points:
                        valid_i.append(i)
                integral = 0.0
                for i in valid_i:
                    j = i + m
                    integral += h[i] * (1 - h[j]) * dx
                max_overlap = max(max_overlap, integral)
            return max_overlap
        
        # Constraints
        A_eq = np.ones(n_points)
        b_eq = n_points / 2
        cons = [LinearConstraint(A_eq, b_eq, b_eq)]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization settings
        result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=cons,
                          tol=1e-10, options={'maxiter': 1000, 'ftol': 1e-10, 'gtol': 1e-10})
        
        # Check if optimization succeeded
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Track time usage
        elapsed_time += result.message.split()[-1] if 'time' in result.message else 0.1
        if elapsed_time >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
