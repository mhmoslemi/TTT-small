import numpy as np
import time
from scipy.optimize import minimize, LinearConstraint
from scipy.interpolate import interp1d
from multiprocessing import Pool, cpu_count

def generate_perturbed_initial(h, noise_level=0.05):
    perturbed = h + noise_level * np.random.randn(len(h))
    perturbed = np.clip(perturbed, 0.0, 1.0)
    return perturbed

def generate_structured_initial(n_points, num_pulses):
    structured = np.zeros(n_points)
    pulse_indices = np.linspace(0, n_points - 1, num_pulses, endpoint=False).astype(int)
    structured[pulse_indices] = 1.0
    target_sum = n_points / 2
    current_sum = np.sum(structured)
    scaling_factor = target_sum / current_sum
    structured *= scaling_factor
    return structured

def evaluate_erdos_solution(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def objective_function(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def optimize_h(initial_guess, n_points, dx):
    # Constraints
    A_eq = np.ones(n_points)
    b_eq = n_points / 2
    cons = [LinearConstraint(A_eq, b_eq, b_eq)]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization settings
    result = minimize(
        lambda h: objective_function(h, dx),
        initial_guess,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        tol=1e-8,
        options={'maxiter': 500, 'ftol': 1e-12}
    )
    
    if result.success:
        return result.x, result.fun
    return initial_guess, float('inf')

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 400  # Increased for better resolution
    
    # Interpolate initial_h_values to n_points
    original_h = initial_h_values
    original_len = len(original_h)
    original_x = np.linspace(0, 2, original_len)

    interpolated_h = np.interp(new_x, original_x, original_h)
    interpolated_h = np.clip(interpolated_h, 0.0, 1.0)
    
    # Generate diverse initial guesses
    initial_guesses = [interpolated_h]
    initial_guesses.append(generate_perturbed_initial(interpolated_h, noise_level=0.01))
    initial_guesses.append(generate_perturbed_initial(interpolated_h, noise_level=0.05))
    initial_guesses.append(generate_perturbed_initial(interpolated_h, noise_level=0.1))
    
    # Add structured initial guesses with varying pulse counts
    for num_pulses in [10, 20, 30, 40, 50]:
        initial_guesses.append(generate_structured_initial(n_points, num_pulses))
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    # Use multiprocessing to parallelize optimization
    pool = Pool(processes=min(cpu_count(), len(initial_guesses)))
    results = []
    
    for guess in initial_guesses:
        dx = 2.0 / n_points
        results.append(pool.apply_async(optimize_h, (guess, n_points, dx)))
