    result = minimize(objective, guess, method='trust-constr', bounds=bounds, constraints=cons,
                      tol=1e-8, options={'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-8})
    if result.success:
        return (result.fun, result.x)
    else:
        return (np.inf, guess)
