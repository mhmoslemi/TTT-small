    target_sum = n_points / 2
    current_sum = np.sum(structured)
    scaling_factor = target_sum / current_sum
    structured *= scaling_factor
    return structured

def generate_random_initial(n_points):
    random_initial = np.random.rand(n_points)
    target_sum = n_points / 2
    current_sum = np.sum(random_initial)
    scaling_factor = target_sum / current_sum
    random_initial *= scaling_factor
    random_initial = np.clip(random_initial, 0.0, 1.0)
    return random_initial

def objective(h, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    dx = 2.0 / n_points
    return max_corr * dx

def optimize_guess(guess, n_points, evaluate_erdos_solution):
    # Define constraints
    A_eq = np.ones(n_points)
    b_eq = n_points / 2
    cons = [LinearConstraint(A_eq, b_eq, b_eq)]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization settings
    result = minimize(objective, guess, method='L-BFGS-B', bounds=bounds, constraints=cons,
                      tol=1e-10, options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
    
    # Check if optimization succeeded
    if result.success:
        return (result.x, result.fun)
    else:
        return (guess, float('inf'))

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    
    # Generate diverse initial guesses
    initial_guesses = [initial_h_values]
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.01))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.03))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.05))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.1))
    
    # Add more structured initial guesses
    for num_pulses in [2, 3, 4, 5, 6, 7, 8, 9, 10, 12, 15, 20]:
        initial_guesses.append(generate_structured_initial(n_points, num_pulses))
    
    # Add more random initial guesses
    for _ in range(5):
        initial_guesses.append(generate_random_initial(n_points))
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    # Use multiprocessing to run optimizations in parallel
    with Pool() as pool:
        results = pool.map(lambda guess: optimize_guess(guess, n_points, evaluate_erdos_solution), initial_guesses)
    
    # Collect results and find the best
    for h, c5 in results:
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
