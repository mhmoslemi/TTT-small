        structured[mirrored_indices] = 1.0
    else:
        pulse_indices = np.linspace(0, n_points - 1, num_pulses, endpoint=False).astype(int)
        pulse_indices = np.sort(pulse_indices)
        center = n_points // 2
        pulse_indices = np.concatenate([pulse_indices, [center]])
        pulse_indices = np.unique(pulse_indices)
        structured[pulse_indices] = 1.0
    target_sum = n_points / 2
    current_sum = np.sum(structured)
    scaling_factor = target_sum / current_sum
    structured *= scaling_factor
    return structured

def generate_random_initial(n_points):
    random_initial = np.random.rand(n_points)
    target_sum = n_points / 2
    current_sum = np.sum(random_initial)
    scaling_factor = target_sum / current_sum
    random_initial *= scaling_factor
    random_initial = np.clip(random_initial, 0.0, 1.0)
    return random_initial

def optimize_h(guess, n_points):
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / n_points
        penalty = 1000 * (np.sum(h) - n_points / 2)**2
        return max_corr * dx + penalty
    
    result = minimize(objective, guess, method='Nelder-Mead',
                      bounds=[(0.0, 1.0) for _ in range(n_points)],
                      options={'maxiter': 1000, 'disp': False})
    
    if result.success:
        return result.x, result.fun
    else:
        return guess, np.inf

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    
    initial_guesses = [initial_h_values]
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.01))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.03))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.05))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.1))
    
    for num_pulses in [2, 3, 4, 5, 6, 7, 8, 9, 10, 12, 15, 20]:
        initial_guesses.append(generate_structured_initial(n_points, num_pulses))
    
    for _ in range(3):
        initial_guesses.append(generate_random_initial(n_points))
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    with Pool(processes=10) as pool:
        results = []
        for guess in initial_guesses:
            results.append(pool.apply_async(optimize_h, (guess, n_points)))
        
        while results:
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break
            
            for i in range(len(results)):
                if results[i].ready():
                    h_opt, c5_opt = results[i].get()
                    if c5_opt < best_c5:
                        best_c5 = c5_opt
                        best_h = h_opt
                    results.pop(i)
                    break
                else:
                    time.sleep(0.1)
    
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
