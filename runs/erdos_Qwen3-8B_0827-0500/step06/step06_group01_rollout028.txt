    return perturbed

def generate_structured_initial(n_points, num_pulses=2, spacing_factor=1.0, pulse_length=0.5):
    structured = np.zeros(n_points)
    spacing = (2.0 - num_pulses * pulse_length) / (num_pulses - 1)
    for i in range(num_pulses):
