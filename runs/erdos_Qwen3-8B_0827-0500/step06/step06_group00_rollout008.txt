    num_workers = min(cpu_count(), len(initial_guesses))
    with Pool(num_workers) as pool:
        results = pool.map(optimize_guess, initial_guesses)
    
    # Find the best result from all optimizations
    for c5, h in results:
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
