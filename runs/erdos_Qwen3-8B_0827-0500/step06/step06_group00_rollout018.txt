    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = 0.0
    for k in k_values:
        shift_index = n_points - 1 + int(k / dx)
        if 0 <= shift_index < len(corr):
            max_corr = max(max_corr, corr[shift_index] * dx)
    return max_corr

def optimize_single(guess, k_values, dx, n_points, constraints, bounds):
    def obj(h):
        return objective(h, k_values, dx, n_points)
    result = minimize(obj, guess, method='L-BFGS-B', bounds=bounds, constraints=constraints,
                      tol=1e-10, options={'maxiter': 1000, 'disp': False})
    return result.x, result.fun

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_guesses = [initial_h_values]
    for num_pulses in [5, 7, 9, 11, 13]:
        initial_guesses.append(generate_structured_initial(n_points, num_pulses))
    for _ in range(3):
        initial_guesses.append(generate_random_initial(n_points))
    k_values = np.linspace(0, 2, 100)
    A_eq = np.ones(n_points)
    b_eq = n_points / 2
    cons = [LinearConstraint(A_eq, b_eq, b_eq)]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    pool = Pool(processes=cpu_count())
    best_h = initial_h_values
    best_c5 = evaluate_erdos_solution(initial_h_values)
    start_time = time.time()
    jobs = []
    for guess in initial_guesses:
        job = pool.apply_async(optimize_single, (guess, k_values, dx, n_points, cons, bounds))
        jobs.append(job)
    for job in jobs:
        if time.time() - start_time >= budget_s:
            break
        result = job.get()
        h_opt, c5 = result
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
    pool.close()
    pool.join()
    return (best_h, best_c5, n_points)
