    scaling_factor = target_sum / current_sum
    random_initial *= scaling_factor
    initial_guesses.append(random_initial)
    
    best_h = None
    best_c5 = float('inf')
    elapsed_time = 0.0
    
    for guess in initial_guesses:
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = differential_evolution(
            fun=lambda h: objective(h, n_points, target_sum),
            bounds=bounds,
            strategy='best1bin',
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.9,
            tol=1e-10,
            maxiter=1000,
            disp=False
        )
        if result.success:
            current_c5 = result.fun - 1e4 * (np.sum(result.x) - target_sum)**2
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        elapsed_time += 10  # Approximate time per optimization
        if elapsed_time >= budget_s:
            break
    
    if best_h is None:
        best_h = initial_h_values
        best_c5 = objective(best_h, n_points, target_sum) - 1e4 * (np.sum(best_h) - target_sum)**2
    
    return (best_h, best_c5, n_points)
