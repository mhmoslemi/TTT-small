from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):

    elapsed_time = 0.0
    start_time = time.time()
    
    # Perform multiple differential evolution runs with different seeds
    for i in range(3):
        result = differential_evolution(
