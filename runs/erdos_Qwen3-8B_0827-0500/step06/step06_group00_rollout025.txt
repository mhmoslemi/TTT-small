    structured[pulse_indices] = 1.0
    target_sum = n_points / 2
    current_sum = np.sum(structured)
    scaling_factor = target_sum / current_sum
    structured *= scaling_factor
    return structured

def generate_random_initial(n_points):
    random_initial = np.random.rand(n_points)
    target_sum = n_points / 2
    current_sum = np.sum(random_initial)
    scaling_factor = target_sum / current_sum
    random_initial *= scaling_factor
    random_initial = np.clip(random_initial, 0.0, 1.0)
    return random_initial

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    
    # Generate diverse initial guesses
    initial_guesses = [initial_h_values]
    for noise_level in [0.01, 0.03, 0.05, 0.1]:
        initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level))
    for num_pulses in [2, 3, 4, 5, 6, 7, 8, 9, 10]:
        initial_guesses.append(generate_structured_initial(n_points, num_pulses))
    for _ in range(3):
        initial_guesses.append(generate_random_initial(n_points))
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        # Define objective function with penalty
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            dx = 2.0 / n_points
            obj = max_corr * dx
            penalty = 1e4 * abs(np.sum(h) - n_points / 2)
            return obj + penalty
        
        # Bounds for differential evolution
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution
        result = differential_evolution(objective, bounds, popsize=15, tol=1e-8, mutation=(0.5, 1.0), recombination=0.8, seed=seed, polish=True, disp=False)
        
        # Check if optimization succeeded
        if result.success:
            current_h = result.x
            # Ensure the sum is correct (due to rounding errors)
            current_sum = np.sum(current_h)
            if abs(current_sum - n_points / 2) > 1e-5:
                # If the constraint is violated, adjust the last value
                current_h[-1] = (n_points / 2) - current_sum + current_h[-1]
                current_h = np.clip(current_h, 0.0, 1.0)
                current_sum = np.sum(current_h)
                if abs(current_sum - n_points / 2) > 1e-5:
                    # If adjustment failed, skip this result
                    continue
            current_c5 = result.fun - 1e4 * abs(np.sum(current_h) - n_points / 2)  # Subtract penalty to get the original objective
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
