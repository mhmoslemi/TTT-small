def generate_perturbed_initial(h, noise_level=0.05):
    perturbed = h + noise_level * np.random.randn(len(h))
    perturbed = np.clip(perturbed, 0.0, 1.0)
    return perturbed

def generate_structured_initial(n_points, num_pulses):
    structured = np.zeros(n_points)
    interval_length = 1.0 / num_pulses
    start_positions = np.array([i * (2.0 / (num_pulses + 1)) for i in range(num_pulses)])
    end_positions = start_positions + interval_length

    dx = 2.0 / n_points
    for i in range(num_pulses):
        start = start_positions[i]
        end = end_positions[i]
        j_start = int(start / dx)
        j_end = int(end / dx)
        j_start = max(0, min(j_start, n_points - 1))
        j_end = max(0, min(j_end, n_points - 1))
        structured[j_start:j_end+1] = 1.0

    current_sum = np.sum(structured)
    target_sum = n_points / 2
    scaling_factor = target_sum / current_sum
    structured *= scaling_factor
    return structured
