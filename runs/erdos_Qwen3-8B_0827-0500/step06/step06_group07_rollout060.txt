    random_initial = np.clip(random_initial, 0.0, 1.0)
    return random_initial

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    
    initial_guesses = [initial_h_values]
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.01))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.05))
    initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=0.1))
    initial_guesses.append(generate_structured_initial(n_points, num_pulses=2))
    initial_guesses.append(generate_structured_initial(n_points, num_pulses=3))
    initial_guesses.append(generate_structured_initial(n_points, num_pulses=4))
    initial_guesses.append(generate_structured_initial(n_points, num_pulses=5))
    initial_guesses.append(generate_structured_initial(n_points, num_pulses=6))
    initial_guesses.append(generate_structured_initial(n_points, num_pulses=8))
    initial_guesses.append(generate_random_initial(n_points, seed=seed+1))
    initial_guesses.append(generate_random_initial(n_points, seed=seed+2))
    initial_guesses.append(generate_random_initial(n_points, seed=seed+3))
    initial_guesses.append(generate_random_initial(n_points, seed=seed+4))
    
    best_h = None
    best_c5 = float('inf')
    elapsed_time = 0.0
    
    for guess in initial_guesses:
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            dx = 2.0 / n_points
            return max_corr * dx
        
        A_eq = np.ones(n_points)
        b_eq = n_points / 2
        cons = [LinearConstraint(A_eq, b_eq, b_eq)]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        result_de = differential_evolution(objective, bounds, constraints=cons, popsize=15, tol=1e-6, maxiter=50, mutation=(0.5, 1.0), recombination=0.8, seed=seed)
        if result_de.success:
            refined_guess = result_de.x
            result_trust = minimize(objective, refined_guess, method='trust-constr', bounds=bounds, constraints=cons,
                                   tol=1e-9, options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-8})
            if result_trust.success:
                current_c5 = result_trust.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_trust.x
        else:
            result_trust = minimize(objective, guess, method='trust-constr', bounds=bounds, constraints=cons,
                                   tol=1e-9, options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-8})
            if result_trust.success:
                current_c5 = result_trust.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_trust.x
        
        elapsed_time += result_de.message.split()[-1] if 'time' in result_de.message else 0.1
        if elapsed_time >= budget_s:
            break
    
    if best_h is None:
        best_h = initial_h_values
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
