    if last_initial < 0 or last_initial > 1:
        # Adjust initial guess to ensure validity
        initial_guess = np.random.rand(n_points - 1)
        sum_initial = np.sum(initial_guess)
        last_initial = (n_points / 2) - sum_initial
        while last_initial < 0 or last_initial > 1:
            initial_guess = np.random.rand(n_points - 1)
            sum_initial = np.sum(initial_guess)
            last_initial = (n_points / 2) - sum_initial
    
    # Run differential evolution
    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-10,
        mutation=(0.5, 1),
        recombination=0.8,
        disp=False,
        polish=True,
        vectorized=True
    )
    
    # Extract the best solution
    best_x = result.x
    sum_best_x = np.sum(best_x)
    best_last = (n_points / 2) - sum_best_x
    best_h = np.concatenate((best_x, [best_last]))
    
    # Check if the best solution is valid
    if best_last < 0 or best_last > 1:
        best_h = initial_h_values
    
    # Compute the best_c5
    best_c5 = np.correlate(best_h, 1 - best_h, mode='full') * dx
    best_c5 = np.max(best_c5)
    
    return (best_h, best_c5, n_points)
