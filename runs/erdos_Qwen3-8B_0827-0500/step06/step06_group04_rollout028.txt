    initial_guesses.append(generate_perturbed_initial(interpolated_h, noise_level=0.05))
    initial_guesses.append(generate_perturbed_initial(interpolated_h, noise_level=0.1))
    
    # Add more structured initial guesses with varying pulse counts
    for num_pulses in [5, 10, 15, 20, 25, 30]:
        initial_guesses.append(generate_structured_initial(n_points, num_pulses))
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        # Define objective function
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            dx = 2.0 / n_points
            return max_corr * dx
        
        # Constraints
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization settings
        result = minimize(objective, guess, method='COBYLA', bounds=bounds, constraints=cons,
                          tol=1e-8, options={'maxiter': 500, 'disp': False})
        
        # Check if optimization succeeded
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Check elapsed time
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
    
    # Fallback to interpolated initial guess if no improvement found
    if best_h is None:
        best_h = interpolated_h
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
