def generate_symmetric_structured_initial(n_points, num_pulses):
    # Generate a symmetric structured initial guess
    structured = np.zeros(n_points)
    # Place pulses symmetrically around the center
    pulse_indices = np.linspace(0, n_points - 1, num_pulses, endpoint=False).astype(int)
    # Mirror the pulses around the center
    pulse_indices = np.sort(np.concatenate((pulse_indices, n_points - 1 - pulse_indices)))
    # Remove duplicates if any
    pulse_indices = np.unique(pulse_indices)
    # Ensure the number of pulses is even or adjust as needed
    if len(pulse_indices) > num_pulses:
        pulse_indices = pulse_indices[:num_pulses]
    structured[pulse_indices] = 1.0
    target_sum = n_points / 2
    current_sum = np.sum(structured)
    scaling_factor = target_sum / current_sum
    structured *= scaling_factor
    return structured
