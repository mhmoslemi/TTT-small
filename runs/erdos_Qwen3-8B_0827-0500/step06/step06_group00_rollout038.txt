import numpy as np
import time
from scipy.optimize import minimize, LinearConstraint
from scipy.interpolate import interp1d
from concurrent.futures import ProcessPoolExecutor, as_completed

def generate_perturbed_initial(h, noise_level=0.05):
    perturbed = h + noise_level * np.random.randn(len(h))
    perturbed = np.clip(perturbed, 0.0, 1.0)
    return perturbed

def generate_structured_initial(n_points, num_pulses):
    structured = np.zeros(n_points)
    pulse_indices = np.linspace(0, n_points - 1, num_pulses, endpoint=False).astype(int)
    structured[pulse_indices] = 1.0
    target_sum = n_points / 2
    current_sum = np.sum(structured)
    scaling_factor = target_sum / current_sum
    structured *= scaling_factor
    return structured

def generate_random_initial(n_points):
    random_initial = np.random.rand(n_points)
    target_sum = n_points / 2
    current_sum = np.sum(random_initial)
    scaling_factor = target_sum / current_sum
    random_initial *= scaling_factor
    random_initial = np.clip(random_initial, 0.0, 1.0)
    return random_initial

def evaluate_erdos_solution(h):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    dx = 2.0 / len(h)
    return max_corr * dx

def optimize_single_guess(guess, n_points, initial_h_values, seed, budget_s):
    np.random.seed(seed)
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / n_points
        return max_corr * dx
    
    A_eq = np.ones(n_points)
    b_eq = n_points / 2
    cons = [LinearConstraint(A_eq, b_eq, b_eq)]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = minimize(objective, guess, method='L-BFGS-B', bounds=bounds, constraints=cons,
                      tol=1e-10, options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False})
    
    if result.success:
        return (result.fun, result.x)
    else:
        return (float('inf'), guess)

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    
    initial_guesses = [initial_h_values]
    for noise_level in [0.01, 0.03, 0.05, 0.1]:
        initial_guesses.append(generate_perturbed_initial(initial_h_values, noise_level=noise_level))
    
    for num_pulses in [2, 3, 4, 5, 6, 7, 8, 9, 10]:
        initial_guesses.append(generate_structured_initial(n_points, num_pulses))
    
    initial_guesses.append(generate_random_initial(n_points))
    initial_guesses.append(generate_random_initial(n_points))
    initial_guesses.append(generate_random_initial(n_points))
    
    best_h = initial_h_values
    best_c5 = evaluate_erdos_solution(initial_h_values)
    start_time = time.time()
    
    with ProcessPoolExecutor(max_workers=10) as executor:
        futures = []
        for i, guess in enumerate(initial_guesses):
            if time.time() - start_time >= budget_s:
                break
            futures.append(executor.submit(optimize_single_guess, guess, n_points, initial_h_values, seed + i, budget_s))
        
        for future in as_completed(futures):
            if time.time() - start_time >= budget_s:
                break
            current_c5, current_h = future.result()
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
