    A_eq = np.ones(n_points)
    b_eq = target_sum
    cons = [LinearConstraint(A_eq, b_eq, b_eq)]
    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]

    result_slsqp = minimize(
        objective_slsqp, best_h_de, method='SLSQP', bounds=bounds_slsqp,
        constraints=cons, tol=1e-9, options={'maxiter': 1500, 'ftol': 1e-10, 'eps': 1e-8}
    )

    if result_slsqp.success:
        best_h = result_slsqp.x
        best_c5 = result_slsqp.fun
    else:
        best_h = best_h_de
        best_c5 = objective(best_h)

    return (best_h, best_c5, n_points)
