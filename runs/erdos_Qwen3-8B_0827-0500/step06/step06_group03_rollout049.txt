        current_h[-1] = np.clip(current_h[-1], 0.0, 1.0)
        current_c5 = result.fun
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
        
        # Track time usage
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_guess
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
