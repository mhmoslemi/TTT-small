from scipy.optimize import minimize, LinearConstraint

def generate_symmetric_initial(n_points, num_pulses=2, pulse_width=0.5):
    structured = np.zeros(n_points)
    total_width = num_pulses * pulse_width
    spacing = (2.0 - total_width) / (num_pulses - 1)
    for i in range(num_pulses // 2):
        start_left = i * (pulse_width + spacing)
        end_left = start_left + pulse_width
        idx_start_left = int(start_left / (2.0 / n_points))
        idx_end_left = int(end_left / (2.0 / n_points))
        structured[idx_start_left:idx_end_left] = 1.0

        end_right = start_right + pulse_width

        structured[idx_start_right:idx_end_right] = 1.0
    # Handle odd number of pulses
    if num_pulses % 2 == 1:
        mid = 1.0
        start = mid - pulse_width / 2
