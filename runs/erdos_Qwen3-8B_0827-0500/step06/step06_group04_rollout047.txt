        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization settings
        result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=cons,
                          tol=1e-8, options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False})
        
        # Check if optimization succeeded
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Track time usage
        elapsed_time += 0.1  # Approximate time per iteration
        if elapsed_time >= budget_s:
            break
    
    # Fallback to interpolated initial guess if no improvement found
    if best_h is None:
        best_h = interpolated_h
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
