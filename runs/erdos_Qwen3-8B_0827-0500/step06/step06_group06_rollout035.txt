    target_sum = n_points / 2
    
    # Define the objective with penalty
    def objective_with_penalty(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / n_points
        original_obj = max_corr * dx
        sum_h = np.sum(h)
        penalty = 1e6 * abs(sum_h - target_sum)
        return original_obj + penalty
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution
    result = differential_evolution(objective_with_penalty, bounds, args=(), popsize=15, maxiter=100, tol=1e-12, mutation=(0.5, 1.0), recombination=0.8, seed=seed)
    
    # Extract the best solution
    best_h = result.x
    # Compute the original objective (without penalty)
    sum_h = np.sum(best_h)
    penalty = 1e6 * abs(sum_h - target_sum)
    best_c5 = result.fun - penalty
    
    return (best_h, best_c5, n_points)
