        initial_guesses.append(generate_structured_initial(n_points, num_pulses))
    initial_guesses.append(generate_two_intervals_initial(n_points))
    
    # Add a random initial guess
    initial_guesses.append(np.random.uniform(0, 1, n_points))
    
    best_h = None
    best_c5 = float('inf')
    elapsed_time = 0.0
    
    for guess in initial_guesses:
        # Define objective function
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            dx = 2.0 / n_points
            return max_corr * dx
        
        # Constraints
        A_eq = np.ones(n_points)
        b_eq = n_points / 2
        cons = [LinearConstraint(A_eq, b_eq, b_eq)]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization settings
        result = minimize(objective, guess, method='L-BFGS-B', bounds=bounds, constraints=cons,
                          tol=1e-8, options={'maxiter': 500, 'disp': False})
        
        # Check if optimization succeeded
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Track time usage
        elapsed_time += 0.1  # Approximate time per iteration
        if elapsed_time >= budget_s:
            break
    
    # Fallback to interpolated initial guess if no improvement found
    if best_h is None:
        best_h = interpolated_h
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
