    initial_guesses.append(generate_random_initial(n_points, seed=seed+1))
    initial_guesses.append(generate_random_initial(n_points, seed=seed+2))
    initial_guesses.append(generate_random_initial(n_points, seed=seed+3))
    initial_guesses.append(generate_random_initial(n_points, seed=seed+4))
    
    best_h = None
    best_c5 = float('inf')
    elapsed_time = 0.0
    
    for guess in initial_guesses:
        # Define objective function
        def objective(h):
            dx = 2.0 / n_points
            return compute_c5(h, dx)
        
        # Constraints
        A_eq = np.ones(n_points)
        b_eq = n_points / 2
        cons = [LinearConstraint(A_eq, b_eq, b_eq)]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization settings
        result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=cons,
                          tol=1e-12, options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-8})
        
        # Check if optimization succeeded
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Track time usage
        elapsed_time += result.message.split()[-1] if 'time' in result.message else 0.1
        if elapsed_time >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = compute_c5(best_h, 2.0 / n_points)
    
    return (best_h, best_c5, n_points)
