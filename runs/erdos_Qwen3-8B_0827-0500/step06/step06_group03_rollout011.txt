import numpy as np
from scipy.optimize import minimize, LinearConstraint
from scipy.interpolate import interp1d
import time
from multiprocessing import Pool, cpu_count

def generate_perturbed_initial(h, noise_level=0.05):
    perturbed = h + noise_level * np.random.randn(len(h))
    perturbed = np.clip(perturbed, 0.0, 1.0)
    return perturbed

def generate_structured_initial(n_points, num_pulses):
    structured = np.zeros(n_points)
    pulse_indices = np.linspace(0, n_points - 1, num_pulses, endpoint=False).astype(int)
    structured[pulse_indices] = 1.0
    target_sum = n_points / 2
    current_sum = np.sum(structured)
    scaling_factor = target_sum / current_sum
    structured *= scaling_factor
    return structured

def generate_random_initial(n_points, seed=None):
    if seed is not None:
        np.random.seed(seed)
    random_initial = np.random.rand(n_points)
    target_sum = n_points / 2
    current_sum = np.sum(random_initial)
    scaling_factor = target_sum / current_sum
    random_initial *= scaling_factor
    random_initial = np.clip(random_initial, 0.0, 1.0)
    return random_initial

def evaluate_erdos_solution(h):
    n_points = len(h)
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def objective_function(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def optimize_single_guess(guess, n_points, dx, seed):
    np.random.seed(seed)
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    A_eq = np.ones(n_points)
    b_eq = n_points / 2
    cons = [LinearConstraint(A_eq, b_eq, b_eq)]
    
    def objective(h):
        return objective_function(h, dx, n_points)
    
    result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=cons,
                      tol=1e-9, options={'maxiter': 1500, 'ftol': 1e-10, 'eps': 1e-8})
    
    if result.success:
        return result.fun, result.x
    else:
        return float('inf'), guess

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 50  # Adjusted for better resolution
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate diverse initial guesses
    initial_guesses = []
    initial_guesses.append(generate_structured_initial(n_points, num_pulses=5))
    initial_guesses.append(generate_structured_initial(n_points, num_pulses=6))
    initial_guesses.append(generate_structured_initial(n_points, num_pulses=7))
    initial_guesses.append(generate_structured_initial(n_points, num_pulses=8))
    initial_guesses.append(generate_random_initial(n_points, seed=seed+1))
    initial_guesses.append(generate_random_initial(n_points, seed=seed+2))
    initial_guesses.append(generate_random_initial(n_points, seed=seed+3))
    initial_guesses.append(generate_perturbed_initial(generate_structured_initial(n_points, num_pulses=5), noise_level=0.05))
    initial_guesses.append(generate_perturbed_initial(generate_structured_initial(n_points, num_pulses=6), noise_level=0.05))
    initial_guesses.append(generate_perturbed_initial(generate_structured_initial(n_points, num_pulses=7), noise_level=0.05))
    initial_guesses.append(generate_perturbed_initial(generate_structured_initial(n_points, num_pulses=8), noise_level=0.05))
    
    best_h = None
    best_c5 = float('inf')
    elapsed_time = 0.0
    start_time = time.time()
    
    # Use multiprocessing to parallelize the optimization
    with Pool(processes=cpu_count()) as pool:
        results = []
        for i, guess in enumerate(initial_guesses):
            seed_i = seed + i
            results.append(pool.apply_async(optimize_single_guess, (guess, n_points, dx, seed_i)))
        
        while results:
            # Check for completion and process results
            completed = [res for res in results if res.ready()]
            for res in completed:
                current_c5, current_h = res.get()
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h
                results.remove(res)
            elapsed_time = time.time() - start_time
            if elapsed_time >= budget_s:
                break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = generate_structured_initial(n_points, num_pulses=5)
