                                       tol=1e-12, maxiter=100, disp=False)
        end_opt = time.time()
        total_elapsed += end_opt - start_opt
        
        # Check if optimization succeeded
        if result.success:
            current_c5 = result.fun - 1e4 * abs(np.sum(result.x) - n_points / 2)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Check budget
        if total_elapsed >= budget_s:
            break
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
