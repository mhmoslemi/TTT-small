import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    # Set n_points to 100 for better resolution
    n_points = 100
    dx = 2.0 / n_points
    # Generate initial guesses
    initial_guesses = []
    # Binary initial guess (first half of points set to 1)
    binary_initial = np.zeros(n_points)
    binary_initial[:n_points//2] = 1.0
    initial_guesses.append(binary_initial)
    # Random initial guess with sum constraint
    random_initial = np.random.rand(n_points)
    random_initial = random_initial * (n_points / 2) / np.sum(random_initial)
    initial_guesses.append(random_initial)
    # Evaluate all initial guesses and track the best result
    best_h = None
    best_c5 = float('inf')
    for h_initial in initial_guesses:
        # Clip values to [0, 1] and ensure sum constraint
        h_initial = np.clip(h_initial, 0.0, 1.0)
        sum_h = np.sum(h_initial)
        if abs(sum_h - n_points / 2) > 1e-6:
            h_initial = h_initial * (n_points / 2) / sum_h
        # Define objective function
        def objective(h):
            ones_minus_h = 1.0 - h
            corr = np.correlate(h, ones_minus_h, mode='full')
            max_corr = np.max(corr) * dx
            return max_corr
        # Define constraint for sum
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        # Bounds for each h_i
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        # Optimization settings
        options = {
            'maxiter': 300,
            'ftol': 1e-8,
            'eps': 1e-8,
            'disp': False
        }
        # Perform optimization
        result = minimize(
            fun=objective,
            x0=h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options=options
        )
        # Check if optimization was successful
        if result.success:
            current_h = result.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
    return (best_h, best_c5, n_points)
