import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Set n_points to 100 for better resolution
    n_points = 100
    dx = 2.0 / n_points
    # Generate a random binary initial guess with 50 ones and 50 zeros
    np.random.seed(seed)
    initial_h = np.zeros(n_points)
    indices = np.random.choice(n_points, size=n_points // 2, replace=False)
    initial_h[indices] = 1.0
    
    # Define the objective function
    def objective(h):
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Define the constraint
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h_i
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    h_initial = initial_h.copy()
    
    # Optimization with SLSQP, increased iterations and tighter tolerances
    result = minimize(objective, h_initial, method='SLSQP',
                      constraints=[{'type': 'eq', 'fun': constraint_sum}],
                      bounds=bounds,
                      options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-9})
    
    # Get the best solution
    best_h = result.x
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
