import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        penalty = 1e4 * np.abs(np.sum(h) - target_sum)
        return max_corr * dx + penalty
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use differential evolution with a seed for reproducibility
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.8,
        tol=1e-8,
        maxiter=100,
        seed=seed
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
