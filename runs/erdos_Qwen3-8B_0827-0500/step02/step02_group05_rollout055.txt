import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Define the objective function with normalization
    def objective(h):
        # Normalize to meet the sum constraint
        h_norm = h * (n_points / 2) / np.sum(h)
        # Compute cross-correlation with 1 - h
        cross_corr = correlate(h_norm, 1 - h_norm, mode='full')
        # Find the maximum value in the cross-correlation
        max_corr = np.max(cross_corr)
        return max_corr * dx

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Use differential evolution with population-based exploration
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 1.0),
        recombination=0.8,
        tol=1e-8,
        maxiter=100,
        seed=seed
    )

    best_h = result.x
    best_c5 = result.fun
    return best_h, best_c5, n_points
