    h_initial = np.random.rand(n_points)
    h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
    initial_guesses.append(h_initial)
    # Add a few more random guesses
    for _ in range(4):
        h_initial = np.random.rand(n_points)
        h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
        initial_guesses.append(h_initial)
    
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    for guess in initial_guesses:
        # Define the objective function
        def objective(h):
            sum_h = np.sum(h)
            if sum_h == 0:
                return float('inf')
            h_normalized = h / sum_h * (n_points / 2)
            corr = np.correlate(h_normalized, 1 - h_normalized, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Use differential evolution
        result = opt.differential_evolution(
            fun=objective,
            bounds=bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            disp=False
        )
        
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return (best_h, best_c5, n_points)
