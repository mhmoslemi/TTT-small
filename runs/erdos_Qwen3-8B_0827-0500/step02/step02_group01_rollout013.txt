            cross_corr = np.correlate(h_vec, 1 - h_vec, mode='full')
            return np.max(cross_corr) * dx
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2
        
        bounds = Bounds([0.0]*n_points, [1.0]*n_points)
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        
        start_opt_time = time.time()
        result = opt.minimize(
            fun=objective,
            x0=h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
        )
        elapsed_opt_time = time.time() - start_opt_time
        elapsed_time += elapsed_opt_time
        
        if elapsed_time >= budget_s:
            break
        
        if result.success:
            current_h = result.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
        else:
            current_c5 = objective(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h
    
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_best = np.sum(best_h)
    delta = (n_points / 2) - sum_best
    best_h += delta / n_points
    best_h = np.clip(best_h, 0.0, 1.0)
    
    return best_h, best_c5, n_points
