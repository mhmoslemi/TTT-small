def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    # Assume initial_h_values is a global variable as per the problem statement

    dx = 2.0 / n_points

    initial_guesses = [h_initial_binary, h_initial_random]

    best_c5 = float('inf')
    start_time = time.time()

        def objective(h_vec):

        def constraint_sum(h_vec):

            method='L-BFGS-B',

            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-9, 'disp': False}
        )
        
        if result.success:

            if time.time() - start_time > budget_s - 1:
