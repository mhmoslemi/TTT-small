from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):

    n_points = 100
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    num_initial_guesses = 5

        initial_h += noise
        initial_h = np.clip(initial_h, 0.0, 1.0)

        sum_h = np.sum(initial_h)
        if abs(sum_h - n_points / 2) > 1e-6:
            initial_h = initial_h * (n_points / 2) / sum_h
        initial_h = np.clip(initial_h, 0.0, 1.0)

            ones_minus_h = 1.0 - h
            corr = np.correlate(h, ones_minus_h, mode='full')
            max_corr = np.max(corr) * dx

        def constraint_sum(h):

        }

        result = minimize(
