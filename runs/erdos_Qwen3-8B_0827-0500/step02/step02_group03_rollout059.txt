import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    # Set number of points and parameters
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_h = initial_h_values.copy()
    sum_initial = np.sum(initial_h)
    initial_h = initial_h * (target_sum / sum_initial)
    initial_h = np.clip(initial_h, 0.0, 1.0)
    
    # Objective function with penalty for sum constraint
    def objective(h):
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        max_corr = np.max(corr) * dx
        penalty = 1e3 * abs(np.sum(h) - target_sum)
        return max_corr + penalty
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution with adjusted parameters
    result = differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )
    
    # Extract the best solution and compute actual C5
    best_h = result.x
    ones_minus_h = 1.0 - best_h
    corr = np.correlate(best_h, ones_minus_h, mode='full')
    best_c5 = np.max(corr) * dx
    
    return (best_h, best_c5, n_points)
