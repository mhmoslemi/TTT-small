def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Choose n_points
    n_points = 100
    dx = 2.0 / n_points
    
    # Generate initial guesses
    initial_guesses = []
    # Add initial_h_values
    initial_guesses.append(initial_h_values)
    # Add uniform distribution
    initial_guesses.append(np.full(n_points, 0.5))
    # Add step function
    step = np.concatenate([np.ones(n_points//2), np.zeros(n_points//2)])
    initial_guesses.append(step)
    # Perturb initial_h_values
    for _ in range(3):
        h_perturb = initial_h_values + np.random.normal(0, 0.05, n_points)
        h_perturb = np.clip(h_perturb, 0.0, 1.0)
        # Ensure the sum is correct
        h_perturb = h_perturb / np.sum(h_perturb) * (n_points / 2)
        initial_guesses.append(h_perturb)
    
    best_c5 = float('inf')
    best_h = initial_h_values
    
    for guess in initial_guesses:
        # Convert to numpy array
        h_initial = np.array(guess)
        # Define the objective function
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Define constraints
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Set up optimization problem
        result = opt.minimize(
            fun=objective,
            x0=h_initial,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return (best_h, best_c5, n_points)
