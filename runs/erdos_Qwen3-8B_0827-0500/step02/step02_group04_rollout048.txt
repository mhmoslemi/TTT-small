import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Set the number of points
    n_points = 100
    dx = 2.0 / n_points
    
    # Generate initial guess with perturbation
    np.random.seed(seed)
    initial_h = initial_h_values.copy()
    # Rescale initial guess to match n_points
    initial_h = np.interp(np.linspace(0, 1, n_points), np.linspace(0, 1, len(initial_h_values)), initial_h_values)
    # Add random noise to perturb the initial guess
    noise = np.random.normal(0, 0.05, size=n_points)
    initial_h += noise
    # Clip to [0,1] and adjust sum
    initial_h = np.clip(initial_h, 0.0, 1.0)
    sum_h = np.sum(initial_h)
    if abs(sum_h - n_points / 2) > 1e-6:
        initial_h = initial_h * (n_points / 2) / sum_h
    
    # Define the objective function
    def objective(h):
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Define the constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h_i
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    h_initial = initial_h.copy()

    result = minimize(objective, h_initial, method='COBYLA',
                      constraints=[{'type': 'eq', 'fun': constraint_sum}],
                      bounds=bounds,
                      options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8})
    
    # Get the best solution
    best_h = result.x
    best_c5 = objective(best_h)
