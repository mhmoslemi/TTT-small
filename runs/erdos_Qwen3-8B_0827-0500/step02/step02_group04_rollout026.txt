import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Generate initial guesses
    n_points = 160
    initial_guesses = []
    # Add initial_h_values
    initial_guesses.append(initial_h_values.copy())
    # Add uniform distribution
    uniform_guess = np.full(n_points, 0.5)
    initial_guesses.append(uniform_guess)
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        # Adjust the guess to meet the sum constraint
        sum_h = np.sum(guess)
        if abs(sum_h - n_points / 2) > 1e-6:
            guess = guess * (n_points / 2) / sum_h
        guess = np.clip(guess, 0.0, 1.0)
        
        # Define the objective function
        def objective(h):
            ones_minus_h = 1.0 - h
            corr = np.correlate(h, ones_minus_h, mode='full')
            return np.max(corr) * (2.0 / n_points)
        
        # Define the constraints
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        # Bounds for each h_i
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Initial guess
        h_initial = guess.copy()
        
        # Optimization with SLSQP
        result = minimize(objective, h_initial, method='SLSQP',
                          constraints=[{'type': 'eq', 'fun': constraint_sum}],
                          bounds=bounds,
                          options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8})
        
        # Check if we have time left
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        # Update best solution
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    return (best_h, best_c5, n_points)
