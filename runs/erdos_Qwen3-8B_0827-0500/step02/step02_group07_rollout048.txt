import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    
    # Generate multiple initial guesses by perturbing the initial_h_values
    num_starts = 5
    initial_guesses = []
    for _ in range(num_starts):
        perturb = np.random.uniform(-0.05, 0.05, n_points)
        perturbed = np.clip(h_initial + perturb, 0, 1)
        initial_guesses.append(perturbed)
    
    best_h = None
    best_c5 = float('inf')
    elapsed = 0
    
    for guess in initial_guesses:
        # Define the objective function
        def objective(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        def constraint_sum(h_flat):
            h = h_flat.reshape(n_points)
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Track elapsed time and exit early if budget is exceeded
        elapsed += 10  # Approximate time per iteration
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
