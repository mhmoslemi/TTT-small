import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is a global variable
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Define the objective function
    def objective(h):
        # Normalize h to meet the sum constraint
        sum_h = np.sum(h)
        if sum_h == 0:
            return float('inf')
        h_normalized = h * (n_points / 2) / sum_h
        # Compute cross-correlation with 1 - h_normalized
        cross_corr = correlate(h_normalized, 1 - h_normalized, mode='full')
        # Find the maximum value in the cross-correlation
        max_corr = np.max(cross_corr)
        return max_corr * dx

    # Define bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Function to generate a random initial guess
    def random_initial_guess():
        h = np.random.rand(n_points)
        # Normalize to meet the sum constraint
        h = h * (n_points / 2) / np.sum(h)
        return h

    # Run multiple differential evolution optimizations
    best_c5 = float('inf')
    best_h = initial_h.copy()

    # Parameters for differential evolution
    strategy = 'best1bin'
    maxiter = 50
    popsize = 20
    tol = 1e-6
    mutation = (0.5, 1)
    recombination = 0.8

    # Number of restarts
    num_restarts = 5

    for i in range(num_restarts):
        # Generate a random initial guess or use initial_h
        if i % 2 == 0:
            h_initial = initial_h.copy()
        else:
            h_initial = random_initial_guess()
        
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy=strategy,
            maxiter=maxiter,
            popsize=popsize,
            tol=tol,
            mutation=mutation,
            recombination=recombination,
            seed=seed + i
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # If optimization fails, try the initial guess
            current_c5 = objective(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_initial

    return best_h, best_c5, n_points
