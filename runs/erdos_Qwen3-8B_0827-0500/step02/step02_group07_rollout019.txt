import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 80
    dx = 2.0 / n_points
    
    # Generate initial guesses
    initial_guesses = []
    # Add the initial_h_values if available
    if 'initial_h_values' in globals():
        initial_guesses.append(initial_h_values.copy())
    # Generate random initial guesses
    for _ in range(5):
        h_random = np.random.rand(n_points)
        h_scaled = h_random / np.sum(h_random) * (n_points / 2)
        initial_guesses.append(h_scaled)
    
    best_c5 = float('inf')
    best_h = None
    
    # For each initial guess, run optimization
    for h_initial in initial_guesses:
        # Define the objective function
        def objective(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Define the constraint
        def constraint_sum(h_flat):
            h = h_flat.reshape(n_points)
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        
        # Run the optimization
        result = minimize(
            objective,
            h_initial,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
        )
        
        # Check if this result is better
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    return best_h, best_c5, n_points
