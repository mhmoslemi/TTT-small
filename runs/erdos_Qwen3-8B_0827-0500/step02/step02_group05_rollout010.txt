import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
from multiprocessing import Pool, cpu_count

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        cross_corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    def constraint(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def random_initial_guess():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        return h

    def perturb(h, seed=None):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.15, 0.15, size=len(h))
        return np.clip(h + perturbation, 0.0, 1.0)

    def optimize_single(h_initial):
        result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-8})
        return (result.fun, result.x) if result.success else (objective(h_initial), h_initial)

    num_restarts = 50
    best_c5 = float('inf')
    best_h = initial_h.copy()

    seeds = [seed + i for i in range(num_restarts)]
    initial_guesses = []
    for i in range(num_restarts):
        if i % 3 == 0:
            initial_guesses.append(perturb(initial_h, seeds[i]))
        elif i % 3 == 1:
            initial_guesses.append(random_initial_guess())
        else:
            initial_guesses.append(np.copy(initial_h))

    with Pool(min(cpu_count(), 10)) as pool:
        results = pool.map(optimize_single, initial_guesses)

    for c5, h in results:
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    return best_h, best_c5, n_points
