from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 80
    dx = 2.0 / n_points

    initial_h = np.random.choice([0, 1], size=n_points)
    initial_h = initial_h.astype(float)

    sum_h = np.sum(initial_h)
    if abs(sum_h - n_points / 2) > 1e-6:
        # Adjust to make sum correct
        diff = (n_points / 2) - sum_h
        # Find indices of 1s to flip to 0 or vice versa
        ones_indices = np.where(initial_h == 1)[0]
        zeros_indices = np.where(initial_h == 0)[0]
        if diff > 0:
            # Need to add more 1s
            flip_indices = np.random.choice(zeros_indices, int(diff), replace=False)
            initial_h[flip_indices] = 1.0
        else:
            # Need to add more 0s
            flip_indices = np.random.choice(ones_indices, -int(diff), replace=False)
            initial_h[flip_indices] = 0.0
    initial_h = np.clip(initial_h, 0.0, 1.0)
    
    # Define the objective function
    def objective(h):
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr
    
    # Define the constraint for the sum
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h_i
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    h_initial = initial_h.copy()

    result = minimize(
