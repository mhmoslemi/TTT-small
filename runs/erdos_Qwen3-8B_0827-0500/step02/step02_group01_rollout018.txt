import numpy as np
import time
import scipy.optimize as opt
from scipy.optimize import Bounds

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 50  # Use smaller n_points for faster optimization
    dx = 2.0 / n_points
    
    # Generate a structured initial guess: 1s on even indices, 0s on odd
    h_initial = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            h_initial[i] = 1.0
    
    def objective(h_vec):
        cross_corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(cross_corr) * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    start_time = time.time()
    result = opt.minimize(
        fun=objective,
        x0=h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
    )
    
    best_h = h_initial.copy()
    best_c5 = objective(h_initial)
    
    if result.success:
        best_h = result.x
        best_c5 = objective(best_h)
    
    # Ensure the best solution is within bounds and satisfies the sum constraint
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_best = np.sum(best_h)
    delta = (n_points / 2) - sum_best
    best_h += delta / n_points
    best_h = np.clip(best_h, 0.0, 1.0)
    
    return best_h, best_c5, n_points
