import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Choose n_points
    n_points = 200
    dx = 2.0 / n_points
    
    # Generate initial guesses
    initial_guesses = []
    
    # Interpolate initial_h_values into new n_points
    dx_old = 2.0 / 100
    h_interpolated = np.zeros(n_points)
    for i in range(n_points):
        x_new = i * dx
        i_old = x_new / dx_old
        i_floor = int(np.floor(i_old))
        i_ceil = i_floor + 1
        i_ceil = np.clip(i_ceil, 0, 99)
        frac = i_old - i_floor
        h_interpolated[i] = (1 - frac) * initial_h_values[i_floor] + frac * initial_h_values[i_ceil]
    initial_guesses.append(h_interpolated)
    
    # Generate random initial guesses
    for _ in range(5):  # 5 random guesses
        h_rand = np.random.rand(n_points)
        h_rand = h_rand / np.sum(h_rand) * (n_points / 2)
        initial_guesses.append(h_rand)
    
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    elapsed_time = 0.0
    
    for guess in initial_guesses:
        # Convert to numpy array
        h_initial = np.array(guess)
        # Define the objective function
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Define constraints
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Set up optimization problem
        result = opt.minimize(
            fun=objective,
            x0=h_initial,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 2000, 'ftol': 1e-9, 'gtol': 1e-9}
        )
        
        # Check if the optimization completed within budget
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Track elapsed time
        elapsed_time += result.message.split()[-2] if 'message' in result else 0.0
        if elapsed_time >= budget_s:
            break
    
    return (best_h, best_c5, n_points)
