import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    # Increase n_points for better resolution
    n_points = 100
    dx = 2.0 / n_points
    # Use initial_h_values as a structured starting point
    initial_h = initial_h_values.copy()
    sum_h = np.sum(initial_h)
    if abs(sum_h - n_points / 2) > 1e-6:
        initial_h = initial_h * (n_points / 2) / sum_h
    initial_h = np.clip(initial_h, 0.0, 1.0)
    
    # Objective function to minimize the maximum overlap
    def objective(h):
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr
    
    # Constraint to maintain integral of h over [0,2] equal to 1
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h_i
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization settings with increased iterations and tolerance
    options = {
        'maxiter': 300,
        'ftol': 1e-8,
        'eps': 1e-8,
        'disp': False
    }
    
    # Perform optimization with L-BFGS-B method
    result = minimize(
        fun=objective,
        x0=initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options=options
    )
    
    # Extract the best solution found
    best_h = result.x
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
