        h = guess.copy()
        # Define objective and constraints
        def objective(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        def constraint_sum(h_flat):
            h = h_flat.reshape(n_points)
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        
        result = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
