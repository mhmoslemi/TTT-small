import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = 100
    dx = 2.0 / n_points
    # Generate initial guess: two intervals of 1's
    initial_h = np.zeros(n_points)
    # First interval: [0, 0.5]
    first_interval_end = int(0.5 / dx)
    initial_h[:first_interval_end + 1] = 1.0
    # Second interval: [1.5, 2]
    second_interval_start = int(1.5 / dx)
    initial_h[second_interval_start:] = 1.0
    # Scale to meet the sum constraint
    sum_h = np.sum(initial_h) * dx
    scale_factor = 1.0 / sum_h
    initial_h *= scale_factor
    # Define the objective function
    def objective(h):
        dx = 2.0 / n_points
        max_c5 = 0.0
        for m in range(n_points + 1):
            if m >= n_points:
                integral = 0.0
            else:
                integral = np.dot(h[:n_points - m], 1 - h[m:]) * dx
            max_c5 = max(max_c5, integral)
        return max_c5
    # Define constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    # Bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    # Initial guess
    h_initial = initial_h.copy()
    # Optimization
    result = minimize(objective, h_initial, method='SLSQP',
                      constraints=[{'type': 'eq', 'fun': constraint_sum}],
                      bounds=bounds,
                      options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8})
    best_h = result.x
    best_c5 = objective(best_h)
    return (best_h, best_c5, n_points)
