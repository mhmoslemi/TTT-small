import numpy as np
from scipy.optimize import minimize
from multiprocessing import Pool, cpu_count

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def generate_initial_guess():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        h = np.clip(h, 0.0, 1.0)
        return h
    
    def objective(h):
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    initial_guesses = [generate_initial_guess() for _ in range(5)]
    initial_guesses.append(initial_h_values.copy() * (target_sum / np.sum(initial_h_values)))
    
    def optimize(guess):
        result = minimize(
            fun=objective,
            x0=guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8, 'disp': False}
