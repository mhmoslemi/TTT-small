import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def constraint_sum(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Use initial guess with perturbation to avoid local minima
    h = initial_h_values.copy()
    perturb = np.random.normal(0, 0.05, size=n_points)
    h += perturb
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    result = minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    best_h = result.x
    best_c5 = result.fun
    return best_h, best_c5, n_points
