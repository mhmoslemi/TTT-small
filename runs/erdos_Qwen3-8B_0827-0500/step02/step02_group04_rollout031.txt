import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Increased for better resolution
    # Generate a random binary initial guess
    h_initial = np.zeros(n_points)
    indices = np.random.choice(n_points, size=n_points//2, replace=False)
    h_initial[indices] = 1.0
    dx = 2.0 / n_points
    
    # Define objective function
    def objective(h):
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Constraint: sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h_i
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization with SLSQP
    result = minimize(objective, h_initial, method='SLSQP',
                      constraints=[{'type': 'eq', 'fun': constraint_sum}],
                      bounds=bounds,
                      options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8})
    
    # Return best solution
    best_h = result.x
    best_c5 = objective(best_h)
    return (best_h, best_c5, n_points)
