import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Choose n_points
    n_points = 64
    dx = 2.0 / n_points
    
    # Generate initial guesses
    initial_guesses = []
    # Add initial_h_values
    initial_guesses.append(initial_h_values)
    # Add structured guess
    structured_guess = np.zeros(n_points)
    structured_guess[0:16] = 1.0
    structured_guess[32:48] = 1.0
    initial_guesses.append(structured_guess)
    # Add random guesses
    for _ in range(3):
        h_rand = np.random.uniform(0, 1, n_points)
        h_rand = h_rand / np.sum(h_rand) * (n_points / 2)
        initial_guesses.append(h_rand)
    
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    elapsed_time = 0.0
    
    for guess in initial_guesses:
        # Convert to numpy array
        h_initial = np.array(guess)
        # Define the objective function
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Define constraints
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Set up optimization problem
        result = opt.minimize(
            fun=objective,
            x0=h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 1000, 'ftol': 1e-9}
        )
        
        # Track time
        elapsed_time += result['nit'] * 0.01  # Approximate time per iteration
        
        if elapsed_time >= budget_s:
            break
        
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return (best_h, best_c5, n_points)
