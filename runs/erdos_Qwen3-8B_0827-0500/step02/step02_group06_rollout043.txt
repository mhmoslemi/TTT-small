        return max_corr * dx
    
    # Define constraints for SLSQP
    def constraint_sum_slsqp(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
    
    # Set up optimization problem
    result_slsqp = opt.minimize(
        fun=objective_slsqp,
        x0=initial_guess,
        method='SLSQP',
        bounds=bounds_slsqp,
        constraints={'type': 'eq', 'fun': constraint_sum_slsqp},
        options={'maxiter': 500, 'ftol': 1e-8}
    )
    
    # Check if the result is successful
    if result_slsqp.success:
        current_c5 = objective_slsqp(result_slsqp.x)
        best_h = result_slsqp.x
        best_c5 = current_c5
    else:
        # Fallback to the differential evolution result
        best_h = initial_guess
        best_c5 = objective_de(initial_guess)
    
    return (best_h, best_c5, n_points)
