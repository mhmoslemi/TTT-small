        )
        
        # Check if the result is within the time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break  # Stop if time is up
        
        # Extract the best solution
        current_h = result.x
        current_c5 = objective(current_h)
        
        # Update the best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
            best_n = n_points
    
    # Return the best solution found
    return best_h, best_c5, best_n
