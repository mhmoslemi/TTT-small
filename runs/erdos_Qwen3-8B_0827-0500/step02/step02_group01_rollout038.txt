import numpy as np
import time
import scipy.optimize as opt
from scipy.optimize import Bounds

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Choose a balance between accuracy and speed
    dx = 2.0 / n_points

    def constraint_sum(h_vec):

    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    start_time = time.time()
    result = opt.minimize(
        fun=objective,
        x0=h,

        bounds=bounds,
        constraints=cons,
        options={'maxiter': 500, 'ftol': 1e-9, 'disp': False}
    )
    
    best_h = h.copy()

        best_h = result.x
        best_c5 = objective(best_h)
