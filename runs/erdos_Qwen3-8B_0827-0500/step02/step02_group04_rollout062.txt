import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Define number of points and initial guesses
    n_points = 80
    dx = 2.0 / n_points
    initial_guesses = []

    # Initial guess 1: provided initial_h_values
    initial_guesses.append(initial_h_values.copy())
    
    # Initial guess 2: uniform distribution
    initial_guesses.append(np.full(n_points, 0.5))
    
    # Initial guess 3: step function (first half 1, second half 0)
    initial_guesses.append(np.concatenate([np.ones(n_points//2), np.zeros(n_points//2)]))
    
    # Initial guess 4: step function (first third 1, rest 0)
    initial_guesses.append(np.concatenate([np.ones(n_points//3), np.zeros(2*n_points//3)]))
    
    # Initial guess 5: alternating pattern (0.5, 0.5, 0.5, 0.5, ...)
    initial_guesses.append(np.full(n_points, 0.5))
    
    # Define objective function
    def objective(h):
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Define constraint function
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds for each h_i
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = None
    best_c5 = float('inf')
    elapsed = 0.0
    
    # Run optimization for each initial guess
    for guess in initial_guesses:
        # Scale guess to meet the sum constraint
        sum_guess = np.sum(guess)
        if abs(sum_guess - n_points / 2) > 1e-6:
            guess = guess * (n_points / 2) / sum_guess
        guess = np.clip(guess, 0.0, 1.0)
        
        # Start timer for this guess
        start_time = time.time()
        
        # Run optimization
        result = minimize(objective, guess, method='SLSQP',
                          constraints=[{'type': 'eq', 'fun': constraint_sum}],
                          bounds=bounds,
                          options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8})
        
        # Calculate elapsed time for this guess
        elapsed += time.time() - start_time
        
        # Check if we've exceeded the budget
        if elapsed >= budget_s:
            break
        
        # Update best result
        current_h = result.x
        current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return (best_h, best_c5, n_points)
