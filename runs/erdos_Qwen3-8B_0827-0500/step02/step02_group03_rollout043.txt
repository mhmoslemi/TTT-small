import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 80
    dx = 2.0 / n_points
    
    # Generate initial guesses
    initial_guesses = []
    # Add the provided initial guess
    initial_guesses.append(initial_h_values.copy())
    # Generate 4 random initial guesses
    for _ in range(4):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        initial_guesses.append(np.clip(h, 0.0, 1.0))
    
    best_c5 = float('inf')
    best_h = None
    
    # Optimization settings
    options = {
        'maxiter': 500,
        'ftol': 1e-8,
        'eps': 1e-8,
        'disp': False
    }
    
    for guess in initial_guesses:
        guess = np.clip(guess, 0.0, 1.0)
        # Define objective function
        def objective(h):
            ones_minus_h = 1.0 - h
            corr = np.correlate(h, ones_minus_h, mode='full')
            max_corr = np.max(corr) * dx
            return max_corr
        
        # Define constraint for the sum
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Perform optimization
        result = minimize(
            fun=objective,
            x0=guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options=options
        )
        
        # Check if this result is better
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return (best_h, best_c5, n_points)
