from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Set number of points
    n_points = 100
    dx = 2.0 / n_points
    # Initial guesses
    initial_guesses = []
    # Add initial_h_values as first guess
    initial_guesses.append(initial_h_values.copy())
    # Generate a random initial guess
    np.random.seed(seed)

    initial_guesses.append(h_rand)
    # Define objective function
    def objective(h):
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        return np.max(corr) * dx
    # Define constraint
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    # Bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    # Best result
    best_c5 = float('inf')

    for h_initial in initial_guesses:

        h_initial = np.clip(h_initial, 0.0, 1.0)
        # Run optimization
        result = minimize(objective, h_initial, method='SLSQP',
