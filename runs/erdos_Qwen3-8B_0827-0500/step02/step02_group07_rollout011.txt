import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points

    def objective(h_flat):
        h = h_flat.copy()
        S = np.sum(h)
        T = n_points / 2
        if S == 0:
            h = np.zeros_like(h)
            h[0] = T
        else:
            scale = T / S
            h = h * scale
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    bounds = [(0, 1) for _ in range(n_points)]

    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )

    best_h = result.x
    best_c5 = result.fun
    return best_h, best_c5, n_points
