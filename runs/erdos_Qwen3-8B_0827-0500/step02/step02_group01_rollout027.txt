    h = np.zeros(n_points, dtype=float)
    h[:n_points // 2] = 1.0
    np.random.shuffle(h)
    
    dx = 2.0 / n_points
    
    def objective(h_vec):
        cross_corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(cross_corr) * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    start_time = time.time()
    result = opt.minimize(
        fun=objective,
        x0=h,
        method='COBYLA',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 1500, 'disp': False}
    )
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    if result.success:
        best_h = result.x
        best_c5 = objective(best_h)
    
    # Ensure the best solution is within bounds and satisfies the sum constraint
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_best = np.sum(best_h)
    delta = (n_points / 2) - sum_best
    best_h += delta / n_points
    best_h = np.clip(best_h, 0.0, 1.0)
    
    return best_h, best_c5, n_points
