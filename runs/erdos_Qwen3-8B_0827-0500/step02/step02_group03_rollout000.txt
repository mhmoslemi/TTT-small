import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Increased to allow finer control
    dx = 2.0 / n_points
    
    # Generate a more informed initial guess with 0s and 1s
    num_ones = n_points // 2
    indices = np.random.choice(n_points, size=num_ones, replace=False)
    initial_h = np.zeros(n_points)
    initial_h[indices] = 1.0
    
    # Objective function to minimize maximum overlap
    def objective(h):
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr
    
    # Constraint to maintain integral of h over [0,2]
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h_i
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization settings with increased iterations and tighter tolerances
    options = {
        'maxiter': 300,
        'ftol': 1e-8,
        'eps': 1e-8,
        'disp': False
    }
    
    # Perform optimization
    result = minimize(
        fun=objective,
        x0=initial_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options=options
    )
    
    # Extract the best solution
    best_h = result.x
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
