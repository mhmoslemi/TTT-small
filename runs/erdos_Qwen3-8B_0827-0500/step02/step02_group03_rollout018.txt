import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Increased points for better resolution
    dx = 2.0 / n_points
    # Generate a random initial guess and scale it to meet the sum constraint
    initial_h = np.random.rand(n_points)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    initial_h = np.clip(initial_h, 0.0, 1.0)
    
    # Define the objective function
    def objective(h):
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr
    
    # Define the constraint for the sum
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h_i
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    h_initial = initial_h.copy()
    
    # Optimization settings
    options = {
        'maxiter': 300,
        'ftol': 1e-8,
        'eps': 1e-8,
        'disp': False
    }
    
    # Perform optimization
    result = minimize(
        fun=objective,
        x0=h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options=options
    )
    
    # Extract the best solution
    best_h = result.x
    best_c5 = objective(best_h)
