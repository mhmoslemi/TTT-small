import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    n_points = 80
    dx = 2.0 / n_points
    # Define bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Define the objective function
    def objective(h):
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr
    
    # Define the constraint for the sum
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Set up the differential evolution
    result = differential_evolution(
        fun=objective,
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.8,
        maxiter=100,
        tol=1e-8,
        disp=False
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
