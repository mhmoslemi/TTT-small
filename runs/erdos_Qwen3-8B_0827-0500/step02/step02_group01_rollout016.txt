    def objective(h_vec):
        n_points = len(h_vec)
        max_integral = 0.0
        dx = 2.0 / n_points
        for k in range(n_points):
            overlap_sum = 0.0
            for x in range(n_points - k):
                overlap_sum += h_vec[x] * (1 - h_vec[x + k])
            integral = overlap_sum * dx
            if integral > max_integral:
                max_integral = integral
        return max_integral
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    start_time = time.time()
    result = opt.minimize(
        fun=objective,
        x0=h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-9, 'disp': False}
    )
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    if result.success:
        best_h = result.x
        best_c5 = objective(best_h)
    
    # Ensure the best solution is within bounds and satisfies the sum constraint
    best_h = np.clip(best_h, 0.0, 1.0)
