import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Objective function that normalizes h to meet the sum constraint
    def objective(h):
        h_sum = np.sum(h)
        if h_sum != 0:
            h = h * (n_points / 2) / h_sum
        cross_corr = np.correlate(h, 1 - h, mode='full')

        return max_corr * dx

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run differential evolution with adaptive parameters
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.8,
        tol=1e-8,
        maxiter=100,
        seed=seed
    )

    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
