import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    # Set the number of points
    n_points = 100
    dx = 2.0 / n_points
    
    # Generate a structured initial guess with periodicity
    np.random.seed(seed)
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        initial_h[i] = 0.5 + 0.25 * np.sin(2 * np.pi * i / n_points)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    initial_h = np.clip(initial_h, 0.0, 1.0)
    
    # Define the objective function
    def objective(h):
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr
    
    # Define the constraint for the sum
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h_i
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    h_initial = initial_h.copy()
    
    # Optimization settings
    options = {
        'maxiter': 300,
        'ftol': 1e-8,
        'eps': 1e-8,
        'disp': False
    }
    
    # Perform optimization
    result = minimize(
        fun=objective,
        x0=h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options=options
    )
    
    # Extract the best solution
    best_h = result.x
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
