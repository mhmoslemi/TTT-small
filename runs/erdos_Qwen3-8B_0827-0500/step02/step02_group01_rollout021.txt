        cross_corr_fft = h_fft * np.conj(one_minus_h_fft)
        cross_corr = np.fft.ifft(cross_corr_fft).real
        return np.max(cross_corr) * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    start_time = time.time()
    result = opt.minimize(
        fun=objective,
        x0=h,
        method='trust-constr',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 1000, 'disp': False}
    )
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    if result.success:
        best_h = result.x
        best_c5 = objective(best_h)
    
    # Ensure the best solution is within bounds and satisfies the sum constraint
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_best = np.sum(best_h)
    delta = (n_points / 2) - sum_best
    best_h += delta / n_points
    best_h = np.clip(best_h, 0.0, 1.0)
    
    return best_h, best_c5, n_points
