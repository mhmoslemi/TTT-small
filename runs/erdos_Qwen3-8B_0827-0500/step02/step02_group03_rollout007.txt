import numpy as np
from scipy.optimize import minimize
from multiprocessing import Pool, cpu_count

def run(seed=42, budget_s=100, **kwargs):
    # Set the number of points
    n_points = 80
    dx = 2.0 / n_points
    initial_guesses = []
    
    # Add the provided initial_h_values
    initial_guesses.append(initial_h_values.copy())
    
    # Generate random initial guesses
    for _ in range(3):
        h = np.random.rand(n_points)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
        initial_guesses.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    # Define objective and constraint once
    def objective(h):
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    options = {
        'maxiter': 200,
        'ftol': 1e-8,
        'eps': 1e-8,
        'disp': False
    }
    
    # Use multiprocessing to parallelize optimization runs
    def optimize(guess):
        # Scale to meet the sum constraint
        sum_h = np.sum(guess)
        if abs(sum_h - n_points / 2) > 1e-6:
            scale = (n_points / 2) / sum_h
            guess = guess * scale
        guess = np.clip(guess, 0.0, 1.0)
        
        # Run optimization
        result = minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options=options
        )
        
        if result.success:
            return (result.fun, result.x)
        return (float('inf'), None)
    
    # Run optimizations in parallel
    with Pool(processes=min(cpu_count(), len(initial_guesses))) as pool:
        results = pool.map(optimize, initial_guesses)
    
    # Find the best result
    for c5, h in results:
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    return (best_h, best_c5, n_points)
