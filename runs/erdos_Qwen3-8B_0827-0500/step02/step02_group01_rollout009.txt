        sum_new = np.sum(new_h)
        delta = (n_points / 2) - sum_new
        new_h += delta / n_points
        new_h = np.clip(new_h, 0.0, 1.0)
        
        # Run optimization
        start_time = time.time()
        result = opt.minimize(
            fun=objective,
            x0=new_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-9, 'disp': False}
        )
        elapsed += time.time() - start_time
        
        if result.success:
            current_h = result.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        
        # Early exit if time is up
        if elapsed >= budget_s - 1:
            break
    
    # Ensure the best solution is within bounds and satisfies the sum constraint
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_best = np.sum(best_h)
    delta = (n_points / 2) - sum_best
    best_h += delta / n_points
    best_h = np.clip(best_h, 0.0, 1.0)
    
    return best_h, best_c5, n_points
