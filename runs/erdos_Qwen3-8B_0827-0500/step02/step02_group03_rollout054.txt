    initial_h = np.random.rand(n_points)

    def objective(h):

        corr = np.correlate(h, ones_minus_h, mode='full')
        max_corr = np.max(corr) * dx

    def constraint_sum(h):

    bounds = [(0.0, 1.0) for _ in range(n_points)]
