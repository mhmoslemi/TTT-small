import numpy as np
import time
import scipy.optimize as opt
from scipy.optimize import Bounds

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    
    def objective(h_vec):
        cross_corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(cross_corr) * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Generate initial guesses
    initial_guesses = [h.copy()]
    for _ in range(5):
        perturb = np.random.normal(0, 0.05, n_points)
        perturbed = h + perturb
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        target_sum = n_points / 2
        delta = (target_sum - current_sum) / n_points
        perturbed += delta
        perturbed = np.clip(perturbed, 0, 1)
        initial_guesses.append(perturbed)
    
    best_h = h.copy()
    best_c5 = objective(h)
    start_time = time.time()
    
    for guess in initial_guesses:
        result = opt.minimize(
            fun=objective,
            x0=guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-9, 'disp': False}
        )
        if result.success:
            current_h = result.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 10:
            break
    
    # Final adjustment for sum constraint
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_best = np.sum(best_h)
    delta = (n_points / 2) - sum_best
    best_h += delta / n_points
    best_h = np.clip(best_h, 0.0, 1.0)
    
    return best_h, best_c5, n_points
