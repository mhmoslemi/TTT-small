import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is a global variable
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Define the objective function with normalization
    def objective(h):
        # Normalize h to meet the sum constraint
        h_normalized = h * (n_points / 2) / np.sum(h)
        # Compute cross-correlation with 1 - h_normalized
        cross_corr = correlate(h_normalized, 1 - h_normalized, mode='full')
        # Find the maximum value in the cross-correlation
        max_corr = np.max(cross_corr)
        return max_corr * dx

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run differential evolution with multiple restarts and parameter tuning
    best_c5 = float('inf')
    best_h = initial_h.copy()

    # Number of restarts
    num_restarts = 10

    for i in range(num_restarts):
        # Set seed for reproducibility
        np.random.seed(seed + i)
        # Generate a random initial guess
        h_initial = np.random.rand(n_points)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        # Run differential evolution

            popsize=15,
            tol=1e-8,

            recombination=0.8,
            seed=seed + i
        )
        
        if result.success:
            current_c5 = result.fun

                best_c5 = current_c5
