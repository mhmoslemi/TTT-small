    initial_guesses.append(h_interpolated)
    
    # Generate perturbed versions of the initial guess
    for _ in range(5):  # 5 perturbed guesses
        perturbation = np.random.normal(0, 0.05, n_points)
        perturbed = h_interpolated + perturbation
        perturbed = np.clip(perturbed, 0.0, 1.0)
        initial_guesses.append(perturbed)
    
    # Generate random initial guesses
    for _ in range(5):  # 5 random guesses
        h_rand = np.random.rand(n_points)
        h_rand = h_rand / np.sum(h_rand) * (n_points / 2)
        initial_guesses.append(h_rand)
    
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    for guess in initial_guesses:
        # Convert to numpy array
        h_initial = np.array(guess)
        # Define the objective function
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Define constraints
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Set up optimization problem
        result = opt.minimize(
            fun=objective,
            x0=h_initial,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 500, 'ftol': 1e-8}
        )
        
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return (best_h, best_c5, n_points)
