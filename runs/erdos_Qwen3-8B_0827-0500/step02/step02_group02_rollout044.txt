from scipy.interpolate import interp1d

def generate_perturbed_initial(h, noise_level=0.05):
    perturbed = h + noise_level * np.random.randn(len(h))
    perturbed = np.clip(perturbed, 0.0, 1.0)
    return perturbed

def generate_structured_initial(n_points):
    structured = np.zeros(n_points)
    pulse_width = 1.0 / 3.0
    pulse_start = [0.5 - pulse_width/2, 1.0 - pulse_width/2, 1.5 - pulse_width/2]
    pulse_end = [0.5 + pulse_width/2, 1.0 + pulse_width/2, 1.5 + pulse_width/2]
    dx = 2.0 / n_points
    for i in range(n_points):
        x = i * dx
        if (x >= pulse_start[0] and x <= pulse_end[0]) or \
           (x >= pulse_start[1] and x <= pulse_end[1]) or \
           (x >= pulse_start[2] and x <= pulse_end[2]):
            structured[i] = 1.0
    return structured

def generate_single_pulse_initial(n_points):
    structured = np.zeros(n_points)
    dx = 2.0 / n_points

    target_sum = n_points / 2

            max_corr = np.max(corr)
            dx = 2.0 / n_points
            return max_corr * dx
