        h = population[i]

            if np.random.rand() < 0.1:
                h[j] += np.random.uniform(-0.1, 0.1)
                h[j] = np.clip(h[j], 0.0, 1.0)
        current_sum = np.sum(h)
        delta = target_sum - current_sum
        if delta != 0:
            h += delta / n_points
            h = np.clip(h, 0.0, 1.0)
        population[i] = h

    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h, dx)

    start_time = time.time()
    while time.time() - start_time < budget_s:
        fitness = []
        for h in population:
            c5 = compute_c5(h, dx)
            fitness.append(c5)
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:

            best_c5 = fitness[best_idx]

            best_candidate = candidates[np.argmin([fitness[i] for i in candidates])]
            parents.append(population[best_candidate])

def compute_c5(h, dx):
    cross_corr = np.correlate(h, 1 - h, mode='full')
    return np.max(cross_corr) * dx
