    initial_vars = symmetric_initial[:half_points]
    
    # Function to generate perturbed initial guesses
    def generate_perturbations(num_perturbations=5):
        perturbations = []
        for _ in range(num_perturbations):
            perturbed = initial_vars + np.random.normal(0, 0.05, initial_vars.shape)
            perturbed = np.clip(perturbed, 0.0, 1.0)
            sum_perturbed = np.sum(perturbed)
            scale = (n_points / 4) / sum_perturbed if sum_perturbed != 0 else 1.0
            perturbed *= scale
            perturbations.append(perturbed)
        return perturbations
    
    perturbations = generate_perturbations(num_perturbations=5)
    best_c5 = float('inf')
    best_h = None
    
    for perturbation in perturbations:
        def objective(vars):
            h = np.zeros(n_points)
            for i in range(half_points):
                h[i] = vars[i]
                h[n_points - 1 - i] = vars[i]
            ones_minus_h = 1.0 - h
            corr = np.correlate(h, ones_minus_h, mode='full')
            max_corr = np.max(corr) * dx
            return max_corr
        
        def constraint_sum(vars):
            return np.sum(vars) - n_points / 4
        
        bounds = [(0.0, 1.0) for _ in range(half_points)]
        h_initial = perturbation.copy()
        
        options = {
            'maxiter': 200,
            'ftol': 1e-8,
            'eps': 1e-8,
            'disp': False
        }
        
        result = minimize(
            fun=objective,
            x0=h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options=options
        )
        
        if result.success:
            current_h = result.x
            full_h = np.zeros(n_points)
            for i in range(half_points):
                full_h[i] = current_h[i]
                full_h[n_points - 1 - i] = current_h[i]
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = full_h
    
    return (best_h, best_c5, n_points)
