import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def constraint_sum(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    result = minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-9}
    )
