        cross_corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        penalty = 1e4 * abs(np.sum(h) - n_points / 2)
        return max_corr * dx + penalty

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run differential evolution
    result_de = differential_evolution(
        objective, 
        bounds, 
        strategy='best1bin', 
        popsize=15, 
        mutation=(0.5, 1), 
        recombination=0.9, 
        tol=1e-6, 
        maxiter=50,
        polish=True
    )

    # Now perform local optimization with SLSQP
    best_h = result_de.x
    # Normalize the sum to meet the constraint
    best_h = best_h * (n_points / 2) / np.sum(best_h)

    # Define the objective function without penalty
    def objective_local(h):
        cross_corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    # Constraints
    def constraint(h):
        return np.sum(h) - n_points / 2

    # Bounds
    bounds_local = [(0.0, 1.0) for _ in range(n_points)]

    # Perform SLSQP optimization
    result_local = minimize(
        objective_local, 
        best_h, 
        method='SLSQP', 
        bounds=bounds_local, 
        constraints={'type': 'eq', 'fun': constraint}, 
        options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
    )

    # Check if the local optimization was successful
    if result_local.success:
        best_c5 = result_local.fun
        best_h = result_local.x
    else:
        best_c5 = result_de.fun
        best_h = result_de.x

    return best_h, best_c5, n_points
