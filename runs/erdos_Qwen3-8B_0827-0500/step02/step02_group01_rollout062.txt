import time

from scipy.optimize import Bounds

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    # Assume initial_h_values is a global variable as per the problem statement
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Generate a few initial guesses
    initial_guesses = [initial_h_values.copy()]

    h_rand = np.random.uniform(0, 1, n_points)

    current_sum = np.sum(h_rand)

    h_rand += delta / n_points
    h_rand = np.clip(h_rand, 0, 1)
    initial_guesses.append(h_rand)
    
    best_h = initial_h_values.copy()
    best_c5 = float('inf')
    elapsed_time = 0
    
    for guess in initial_guesses:
        # Define the objective function
        def objective(h_vec):
            cross_corr = np.correlate(h_vec, 1 - h_vec, mode='full')
            return np.max(cross_corr) * dx
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2
        
        bounds = Bounds([0.0]*n_points, [1.0]*n_points)
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        
        start_time = time.time()
        result = opt.minimize(
            fun=objective,
            x0=guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
        )
        elapsed_time += time.time() - start_time
        
        # Check if the result is successful
        if result.success:

        # Also check the initial guess
        current_c5 = objective(guess)
        if current_c5 < best_c5:
            best_h = guess.copy()

        # Early exit if time is up
