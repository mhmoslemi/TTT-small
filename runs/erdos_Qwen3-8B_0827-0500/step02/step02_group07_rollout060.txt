from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        penalty = 1e6 * np.abs(np.sum(h) - n_points / 2)

    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use differential evolution for global search
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=50,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        disp=False
    )
    
    best_h = result.x
    best_c5 = result.fun
    return best_h, best_c5, n_points
