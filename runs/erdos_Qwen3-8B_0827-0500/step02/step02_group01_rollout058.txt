    best_c5 = objective_slsqp(best_h)
    
    # Ensure the solution is within bounds and satisfies the sum constraint
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_best = np.sum(best_h)
    delta = (n_points / 2) - sum_best
    best_h += delta / n_points
    best_h = np.clip(best_h, 0.0, 1.0)
    
    return best_h, best_c5, n_points
