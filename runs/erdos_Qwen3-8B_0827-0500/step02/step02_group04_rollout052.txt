import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    initial_guesses = []
    
    # Generate initial guesses: one based on initial_h_values scaled to n_points, and a few random
    # Assuming initial_h_values is for n=80, we scale it to n_points=100
    if 'initial_h_values' in globals():
        scaled_initial = np.array(initial_h_values) * (n_points / 80.0)
        scaled_initial = scaled_initial * (n_points / 2.0) / np.sum(scaled_initial)
        scaled_initial = np.clip(scaled_initial, 0.0, 1.0)
        initial_guesses.append(scaled_initial)
    
    # Generate random initial guesses
    for _ in range(2):
        random_h = np.random.rand(n_points)
        random_h = random_h * (n_points / 2.0) / np.sum(random_h)
        random_h = np.clip(random_h, 0.0, 1.0)
        initial_guesses.append(random_h)
    
    # Define objective function
    def objective(h):
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Define constraint for integral
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Bounds for each h_i
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    for h_initial in initial_guesses:
        # Run optimization with more iterations and tighter tolerances
        result = minimize(objective, h_initial, method='SLSQP',
                          constraints=[{'type': 'eq', 'fun': constraint_sum}],
                          bounds=bounds,
                          options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8})
        
        # Check if we have a valid solution
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # If optimization fails, use the last valid solution
            if best_h is None:
                best_h = h_initial
                best_c5 = objective(best_h)
    
        # Early stopping if time is up
        if time.time() - start_time > budget_s - 1:
            break
    
    return (best_h, best_c5, n_points)
