import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        cross_corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        penalty = 1e4 * abs(np.sum(h) - n_points / 2)
        return max_corr * dx + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Generate initial guesses
    initial_guesses = [initial_h.copy()]
    np.random.seed(seed)
    for _ in range(5):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        initial_guesses.append(h)

    best_c5 = float('inf')
    best_h = initial_h.copy()

    # Run differential evolution from each initial guess
    for guess in initial_guesses:
        result = differential_evolution(
            objective, bounds, strategy='best1bin',
            popsize=15, maxiter=100, tol=1e-8,
            mutation=(0.5, 1), recombination=0.8,
            seed=seed + initial_guesses.index(guess)
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            current_c5 = objective(guess)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = guess

    return best_h, best_c5, n_points
