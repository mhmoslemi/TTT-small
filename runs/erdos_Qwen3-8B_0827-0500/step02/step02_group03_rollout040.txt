        corr = np.correlate(h, ones_minus_h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    for guess in initial_guesses:
        guess_sum = np.sum(guess)
        if abs(guess_sum - n_points / 2) > 1e-6:
            guess = guess * (n_points / 2) / guess_sum
        guess = np.clip(guess, 0.0, 1.0)
        
        result = minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options=options
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    if best_h is None:
        best_h = initial_guesses[0]
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
