from scipy.optimize import minimize
from scipy.interpolate import interp1d

def run(seed=42, budget_s=100, **kwargs):
    # Set a higher resolution for better optimization
    n_points = 100
    dx = 2.0 / n_points
    # Generate initial guess based on provided initial_h_values, scaled to n_points
    # Assume initial_h_values is for 80 points; interpolate to 100 points
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        # Interpolate initial_h_values to n_points
        f = interp1d(np.linspace(0, 2, len(initial_h)), initial_h, kind='linear')
        initial_h_new = f(np.linspace(0, 2, n_points))
        initial_h_new = np.clip(initial_h_new, 0.0, 1.0)
        # Ensure sum is correct
        sum_h = np.sum(initial_h_new)
        if abs(sum_h - n_points / 2) > 1e-6:
            initial_h_new = initial_h_new * (n_points / 2) / sum_h
        h_initial = initial_h_new
    else:
        # Fallback to random initial guess
        np.random.seed(seed)
        h_initial = np.random.rand(n_points)
        sum_h = np.sum(h_initial)
        h_initial = h_initial * (n_points / 2) / sum_h
        h_initial = np.clip(h_initial, 0.0, 1.0)
    
    # Define the objective function
    def objective(h):
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr
    
    # Define the constraint for the sum
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h_i
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    options = {
