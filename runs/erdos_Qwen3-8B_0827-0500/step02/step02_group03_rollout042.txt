import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 80
    initial_guesses = []
    # Add the initial_h_values as one guess
    initial_guesses.append(initial_h_values.copy())
    # Generate 4 more random initial guesses
    for _ in range(4):
        h = np.random.rand(n_points)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
        initial_guesses.append(h)
    
    best_h = None
    best_c5 = float('inf')
    time_used = 0.0
    
    for guess in initial_guesses:
        # Ensure the guess satisfies the sum constraint
        sum_h = np.sum(guess)
        if abs(sum_h - n_points / 2) > 1e-6:
            guess = guess * (n_points / 2) / sum_h
        guess = np.clip(guess, 0.0, 1.0)
        
        # Define the objective function
        def objective(h):
            ones_minus_h = 1.0 - h
            corr = np.correlate(h, ones_minus_h, mode='full')
            max_corr = np.max(corr) * (2.0 / n_points)
            return max_corr
        
        # Define the constraint for the sum
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        # Bounds for each h_i
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization settings
        options = {
            'maxiter': 500,
            'ftol': 1e-8,
            'eps': 1e-8,
            'disp': False
        }
        
        # Initial guess for this iteration
        h_initial = guess.copy()
        
        # Perform optimization
        result = minimize(
            fun=objective,
            x0=h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options=options
        )
        
        # Check if the optimization completed successfully
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Track time usage (simulated for the purpose of this function)
        time_used += 10  # Approximate time per iteration
        
        # Early exit if time is exceeded
        if time_used >= budget_s:
            break
    
    return (best_h, best_c5, n_points)
