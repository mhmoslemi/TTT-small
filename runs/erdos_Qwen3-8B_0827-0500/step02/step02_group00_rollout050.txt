import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds
from multiprocessing import Pool, cpu_count

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Choose n_points
    n_points = 100
    dx = 2.0 / n_points
    
    # Generate initial guesses
    initial_guesses = []
    
    # Generate the interpolated initial guess based on initial_h_values
    dx_old = 2.0 / 42
    h_interpolated = np.zeros(n_points)
    for k in range(n_points):
        x_new = k * dx
        i = int(x_new / dx_old)
        i = np.clip(i, 0, 41)
        h_interpolated[k] = initial_h_values[i]
    initial_guesses.append(h_interpolated)
    
    # Generate random initial guesses
    for _ in range(9):  # 10 total initial guesses (1 interpolated + 9 random)
        h_rand = np.random.rand(n_points)
        h_rand = h_rand / np.sum(h_rand) * (n_points / 2)
        initial_guesses.append(h_rand)
    
    # Define a helper function to run the optimization for a given guess
    def optimize_guess(guess):
        h_initial = np.array(guess)
        # Define the objective function
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Define constraints
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Set up optimization problem
        result = opt.minimize(
            fun=objective,
            x0=h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 500, 'ftol': 1e-8}
        )
        
        if result.success:
            current_c5 = objective(result.x)
            return (current_c5, result.x)
        else:
            return (float('inf'), np.zeros(n_points))
    
    # Use multiprocessing to run optimizations in parallel
    num_cpus = min(cpu_count(), len(initial_guesses))
    with Pool(processes=num_cpus) as pool:
        results = pool.map(optimize_guess, initial_guesses)
    
    # Find the best result
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
