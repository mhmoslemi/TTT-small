    initial_guesses = []

    dx_old = 2.0 / 42

        i = int(x_new / dx_old)
        i = np.clip(i, 0, 41)
        h_interpolated[k] = initial_h_values[i]
    initial_guesses.append(h_interpolated)
    
    # Generate perturbed versions of the interpolated guess

        h_perturbed = h_interpolated + 0.05 * np.random.randn(n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)

        initial_guesses.append(h_perturbed)
    
    # Generate random initial guesses
    for _ in range(5):  # 5 random guesses
        h_rand = np.random.rand(n_points)
        h_rand = h_rand / np.sum(h_rand) * (n_points / 2)
        initial_guesses.append(h_rand)
    
    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    for guess in initial_guesses:
        # Convert to numpy array
        h_initial = np.array(guess)
        # Define the objective function
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Define constraints
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Set up optimization problem
        result = opt.minimize(
