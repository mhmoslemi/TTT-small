import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points

    h_initial = np.zeros(n_points)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

                      options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8})
    
    best_h = result.x
    best_c5 = objective(best_h)
