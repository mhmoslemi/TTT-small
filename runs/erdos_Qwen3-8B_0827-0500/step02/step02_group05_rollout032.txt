import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is a global variable
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Define the objective function
    def objective(h):
        # Normalize h to meet the sum constraint
        h_sum = np.sum(h)
        if h_sum == 0:
            return float('inf')
        h_normalized = h * (n_points / 2) / h_sum
        h_normalized = np.clip(h_normalized, 0, 1)
        # Compute cross-correlation
        cross_corr = correlate(h_normalized, 1 - h_normalized, mode='full')
        # Find the maximum value
        max_corr = np.max(cross_corr)
        return max_corr * dx

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run differential evolution with multiple restarts
    best_c5 = float('inf')
    best_h = initial_h.copy()

    # Parameters for differential evolution
    strategy = 'best1bin'
    popsize = 15
    mutation = (0.5, 1)
    recombination = 0.8
    tol = 1e-8
    maxiter = 100
    seed = seed

    # Run multiple differential evolution runs
    num_runs = 5
    for run_idx in range(num_runs):
        # Set seed for reproducibility
        np.random.seed(seed + run_idx)
        # Generate initial population with some diversity
        initial_population = np.random.rand(popsize, n_points)
        initial_population = initial_population * (n_points / 2) / np.sum(initial_population, axis=1, keepdims=True)
        initial_population = np.clip(initial_population, 0, 1)
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy=strategy,
            popsize=popsize,
            mutation=mutation,
            recombination=recombination,
            tol=tol,
            maxiter=maxiter,
            seed=seed + run_idx,
            x0=initial_population
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # If optimization fails, try the initial population
            current_c5 = objective(initial_population[0])
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = initial_population[0]

    return best_h, best_c5, n_points
