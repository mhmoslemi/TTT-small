    # Generate initial guesses
    initial_guesses = []

    def generate_symmetric_guess(n_points):

        for i in range(n_points // 2):
            h[i] = 0.5
        total = np.sum(h)
        h = h / total * (n_points / 2)
        return h
    
    initial_guesses.append(generate_symmetric_guess(n_points))
    
    # Generate random guesses
    for _ in range(2):
        h_random = np.random.rand(n_points)
        h_random = h_random / np.sum(h_random) * (n_points / 2)
        initial_guesses.append(h_random)
    
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    for guess in initial_guesses:
        # Convert to numpy array
        h_initial = np.array(guess)
        # Define the objective function
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Define constraints
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Set up optimization problem
        result = opt.minimize(
