import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    # Generate a structured initial guess: spread 1s evenly
    h_initial = np.zeros(n_points)
    # Distribute 1s with spacing to minimize overlap
    num_ones = int(np.round(n_points / 2))
    indices = np.linspace(0, n_points - 1, num_ones, endpoint=False).astype(int)
    h_initial[indices] = 1.0
    # Ensure sum is correct
    sum_h = np.sum(h_initial)
    if abs(sum_h - n_points / 2) > 1e-6:
        h_initial = h_initial * (n_points / 2) / sum_h
    h_initial = np.clip(h_initial, 0.0, 1.0)
    
    # Objective function
    def objective(h):
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Constraint
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization with COBYLA

    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
