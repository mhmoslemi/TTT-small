    initial_h = np.clip(initial_h, 0.0, 1.0)
    
    # Define the objective function
    def objective(h):
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Define the constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h_i
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    h_initial = initial_h.copy()
    
    # Optimization with L-BFGS-B and enhanced parameters
    result = minimize(objective, h_initial, method='L-BFGS-B',
                      constraints=[{'type': 'eq', 'fun': constraint_sum}],
                      bounds=bounds,
                      options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8})
    
    # Get the best solution
    best_h = result.x
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
