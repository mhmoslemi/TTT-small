import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100  # Increased for better resolution
    dx = 2.0 / n_points
    required_sum = n_points / 2
    
    # Generate a structured initial guess with two intervals
    initial_h = np.zeros(n_points)
    # First interval: 0 to 0.5 (indices 0 to 24)
    initial_h[0:25] = 1.0
    # Second interval: 1.0 to 1.5 (indices 50 to 74)
    initial_h[50:75] = 1.0
    # Ensure sum is correct
    initial_h = initial_h * (required_sum / np.sum(initial_h))
    initial_h = np.clip(initial_h, 0.0, 1.0)
    
    # Define the objective function
    def objective(h):
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Define the constraints
    def constraint_sum(h):
        return np.sum(h) - required_sum
    
    # Bounds for each h_i
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    h_initial = initial_h.copy()
    
    # Optimization with L-BFGS-B
    result = minimize(objective, h_initial, method='L-BFGS-B',
                      constraints=[{'type': 'eq', 'fun': constraint_sum}],
                      bounds=bounds,
                      options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8})
    
    # Get the best solution
    best_h = result.x
    best_c5 = objective(best_h)
