def run(seed=42, budget_s=100, **kwargs):

    initial_guesses = []

    dx_old = 2.0 / 42
    h_interpolated = np.zeros(n_points)
    for k in range(n_points):
        x_new = k * dx
        i = int(x_new / dx_old)
        i = np.clip(i, 0, 41)
        h_interpolated[k] = initial_h_values[i]
    initial_guesses.append(h_interpolated)
    
    # Generate perturbed versions of the interpolated guess
    for _ in range(3):  # 3 perturbations
        perturb = np.random.normal(0, 0.05, n_points)
        h_perturb = h_interpolated + perturb
        h_perturb = np.clip(h_perturb, 0, 1)
        initial_guesses.append(h_perturb)
    
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    for guess in initial_guesses:
        # Convert to numpy array
        h_initial = np.array(guess)
        # Define the objective function
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Define constraints
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Set up optimization problem
        result = opt.minimize(
