import numpy as np
from scipy.optimize import minimize, basinhopping

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def constraint_sum(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Generate a few random initial guesses
    np.random.seed(seed)
    initial_guesses = [h]
    for _ in range(3):
        rand_h = np.random.rand(n_points)
        rand_h = (rand_h * n_points / 2) / np.sum(rand_h)
        initial_guesses.append(rand_h)
    
    best_c5 = float('inf')
    best_h = h
    
    for guess in initial_guesses:
        result = basinhopping(
            objective,
            guess,
            niter=50,
            T=1.0,
            stepsize=0.5,
            minimizer_kwargs={
                'method': 'L-BFGS-B',
                'bounds': bounds,
                'constraints': cons,

            }
        )
        if result.fun < best_c5:

            best_h = result.x
