import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is a global variable
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Define the objective function
    def objective(h):

        h_normalized = h * (n_points / 2) / np.sum(h)

        return max_corr * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run differential evolution
    result_de = differential_evolution(

        tol=1e-8, mutation=(0.5, 1), recombination=0.8, seed=seed
    )
    best_h_de = result_de.x
    best_c5_de = result_de.fun

    # Refine with SLSQP
    def objective_slsqp(h):
        h_normalized = h * (n_points / 2) / np.sum(h)
        cross_corr = correlate(h_normalized, 1 - h_normalized, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]

    result_slsqp = minimize(
        objective_slsqp, best_h_de, method='SLSQP',
        bounds=bounds_slsqp, constraints={'type': 'eq', 'fun': constraint_slsqp},
        options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
    )
    best_h_slsqp = result_slsqp.x

    # Compare and return the best solution
    best_h = best_h_slsqp if best_c5_slsqp < best_c5_de else best_h_de
