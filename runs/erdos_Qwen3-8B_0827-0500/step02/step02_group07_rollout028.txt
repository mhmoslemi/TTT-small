import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    def generate_initial_guess(n_points):
        h = np.random.rand(n_points)
        sum_rest = np.sum(h[:-1])
        h[-1] = (n_points / 2) - sum_rest
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        delta = (n_points / 2) - sum_h
        h += delta / n_points
        return h
    
    h_initial = generate_initial_guess(n_points)
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def constraint_sum(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    result = minimize(
        objective,
        h_initial,
        method='L-BFGS-B',
        bounds=bounds,

    )

    best_c5 = result.fun
    return best_h, best_c5, n_points
