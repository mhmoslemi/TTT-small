import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Define the objective function with penalty for sum constraint
    def objective(h):
        cross_corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        # Penalty for sum constraint
        penalty = 1e6 * abs(np.sum(h) - n_points / 2)
        return max_corr * dx + penalty

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run differential evolution with multiple restarts
    best_c5 = float('inf')
    best_h = initial_h.copy()
    num_restarts = 5

    for i in range(num_restarts):
        # Use initial guess with perturbation
        h_initial = initial_h.copy()
        np.random.seed(seed + i)
        h_initial += np.random.uniform(-0.1, 0.1, size=n_points)
        h_initial = np.clip(h_initial, 0.0, 1.0)
        # Normalize to meet the sum constraint
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)

        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=50,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed + i
        )
        current_c5 = objective(result.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x

    return best_h, best_c5, n_points
