        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2
        
        bounds = Bounds([0.0]*n_points, [1.0]*n_points)
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        
        start_time = time.time()
        result = opt.minimize(
            fun=objective,
            x0=h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
        )
        elapsed_time += time.time() - start_time
        
        if result.success:
            current_h = result.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        else:
            # Use the initial guess if optimization fails
            current_h = h
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
    
    # Ensure the best solution is within bounds and satisfies the sum constraint
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_best = np.sum(best_h)
    delta = (n_points / 2) - sum_best
    best_h += delta / n_points
    best_h = np.clip(best_h, 0.0, 1.0)
    
    return best_h, best_c5, n_points
