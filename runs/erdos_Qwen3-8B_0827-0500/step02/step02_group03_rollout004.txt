import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    # Increase resolution for better precision
    n_points = 128
    dx = 2.0 / n_points
    
    # Generate multiple initial guesses
    initial_guesses = []
    # Use provided initial guess
    initial_guesses.append(initial_h_values.copy())
    # Generate a random initial guess with sum constraint
    np.random.seed(seed)
    random_h = np.random.rand(n_points)
    random_h = random_h * (n_points / 2) / np.sum(random_h)
    initial_guesses.append(random_h)
    
    best_c5 = float('inf')
    best_h = None
    
    for guess in initial_guesses:
        # Clip to [0,1] and normalize
        guess = np.clip(guess, 0.0, 1.0)
        guess = guess * (n_points / 2) / np.sum(guess)
        
        # Objective function
        def objective(h):
            ones_minus_h = 1.0 - h
            corr = np.correlate(h, ones_minus_h, mode='full')
            max_corr = np.max(corr) * dx
            return max_corr
        
        # Constraint for integral
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        # Bounds
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization settings
        options = {
            'maxiter': 300,
            'ftol': 1e-8,
            'eps': 1e-8,
            'disp': False
        }
        
        # Run optimization
        result = minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options=options
        )
        
        # Update best solution
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    return (best_h, best_c5, n_points)
