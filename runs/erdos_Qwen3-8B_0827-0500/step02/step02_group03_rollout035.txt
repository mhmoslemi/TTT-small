import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    # Choose n_points
    n_points = 100
    dx = 2.0 / n_points
    # Generate initial guesses
    initial_guesses = []
    # Generate a random initial guess
    np.random.seed(seed)
    h_random = np.random.rand(n_points)
    sum_h = np.sum(h_random)
    scale = (n_points / 2) / sum_h
    h_random = h_random * scale

    initial_guesses.append(h_random)
    # Generate a pattern-based initial guess
    h_pattern = np.zeros(n_points)
    for i in range(n_points):
        h_pattern[i] = 0.5 + 0.5 * np.sin(np.pi * i / n_points)
    sum_h = np.sum(h_pattern)
    scale = (n_points / 2) / sum_h
    h_pattern = h_pattern * scale
    h_pattern = np.clip(h_pattern, 0.0, 1.0)
    initial_guesses.append(h_pattern)
    # Now, run optimization for each initial guess
    best_c5 = float('inf')
    best_h = None
    for guess in initial_guesses:
        # Define the objective function
        def objective(h):
            ones_minus_h = 1.0 - h
            corr = np.correlate(h, ones_minus_h, mode='full')
            max_corr = np.max(corr) * dx
            return max_corr
        # Define the constraint for the sum
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        # Bounds for each h_i
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        # Optimization settings
        options = {
            'maxiter': 200,
            'ftol': 1e-8,
            'eps': 1e-8,
            'disp': False
        }
        # Perform optimization
        result = minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options=options
        )
        # Check if optimization was successful
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            pass  # Skip failed optimizations
    return (best_h, best_c5, n_points)
