            initial_h = initial_h * (n_points / 2) / sum_h
        initial_h = np.clip(initial_h, 0.0, 1.0)
        
        result = minimize(
            fun=objective,
            x0=initial_h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options=options
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return (best_h, best_c5, n_points)
