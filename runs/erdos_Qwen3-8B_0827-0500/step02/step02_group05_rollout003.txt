import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is a global variable
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Define the objective function with penalty for constraint violation
    def objective(h):
        cross_corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        # Penalize constraint violation
        penalty = 1e6 * abs(np.sum(h) - n_points / 2)
        return max_corr * dx + penalty

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run differential evolution with multiple restarts
    best_c5 = float('inf')
    best_h = initial_h.copy()

    # Number of restarts
    num_restarts = 10

    for i in range(num_restarts):
        # Set seed for reproducibility
        np.random.seed(seed + i)
        # Generate a random initial guess
        h_initial = np.random.rand(n_points)
        # Normalize to meet the sum constraint
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=50,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed + i
        )
        
        # Check if the result is successful
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # If optimization fails, try the perturbed guess
            current_c5 = objective(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_initial

    return best_h, best_c5, n_points
