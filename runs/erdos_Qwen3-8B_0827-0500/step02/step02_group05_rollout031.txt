import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Objective function with normalization
    def objective(h):
        # Normalize h to meet the sum constraint
        h_normalized = h * (n_points / 2) / np.sum(h)
        # Compute cross-correlation with 1 - h
        cross_corr = correlate(h_normalized, 1 - h_normalized, mode='full')
        # Find the maximum value in the cross-correlation
        max_corr = np.max(cross_corr)
        return max_corr * dx

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Use differential evolution with multiple restarts
    best_c5 = float('inf')
    best_h = initial_h.copy()
    num_restarts = 10

    for i in range(num_restarts):
        # Set seed for reproducibility
        np.random.seed(seed + i)
        # Generate a random initial population
        initial_population = np.random.rand(n_points, 15)
        initial_population = initial_population * (n_points / 2) / np.sum(initial_population, axis=0)
        initial_population = initial_population.reshape(-1, n_points)

        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.8,
            tol=1e-8,
            maxiter=100,
            seed=seed + i,
            polish=True,
            x0=initial_population[0]
        )

        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # Fallback to the initial population's first guess
            current_c5 = objective(initial_population[0])
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = initial_population[0]

    return best_h, best_c5, n_points
