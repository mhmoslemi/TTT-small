        indices = np.where(first_half == 0)[0][:extra]
        first_half[indices] = 1
    elif ones_count > 20:
        extra = ones_count - 20
        indices = np.where(first_half == 1)[0][:extra]
        first_half[indices] = 0
    initial_h = np.zeros(n_points)
    initial_h[:half] = first_half
    initial_h[half:] = first_half[::-1]
    h = initial_h.copy()
    dx = 2.0 / n_points
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def constraint_sum(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Use Nelder-Mead with increased iterations and tighter tolerances
    result = minimize(
        objective,
        h,
        method='Nelder-Mead',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 500, 'xatol': 1e-8, 'fatol': 1e-8}
    )
    
    best_h = result.x
    best_c5 = result.fun
    return best_h, best_c5, n_points
