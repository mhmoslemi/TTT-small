        if result.success:
            h_opt = result.x
            cross_corr = correlate(h_opt, 1 - h_opt, mode='full')
            max_corr = np.max(cross_corr)
            current_c5 = max_corr * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_opt
        else:
            # If optimization fails, try the perturbed guess
            h_initial = np.clip(h_initial, 0.0, 1.0)
            cross_corr = correlate(h_initial, 1 - h_initial, mode='full')
            current_c5 = np.max(cross_corr) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_initial

    return best_h, best_c5, n_points
