def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    # Use the initial_h_values as provided
    # Assuming initial_h_values is a global variable with n_points=80
    n_points = 80
    dx = 2.0 / n_points
    initial_guesses = []
    # Add the initial_h_values as the first guess
    initial_guesses.append(initial_h_values.copy())
    # Generate 4 random initial guesses
    for _ in range(4):
        h = np.random.uniform(0, 1, n_points)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = h * scale
        h = np.clip(h, 0, 1)
        initial_guesses.append(h)
    
    best_c5 = float('inf')
    best_h = None

    # For each initial guess, run the optimization
    for guess in initial_guesses:
        # Define the objective function
        def objective(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx

        def constraint_sum(h_flat):
            h = h_flat.reshape(n_points)
            return np.sum(h) - n_points / 2

        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': constraint_sum}]

        # Run the optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
        )

        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    # Return the best result
    return best_h, best_c5, n_points
