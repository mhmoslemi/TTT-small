import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 80
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate random initial guess
    h_initial = np.random.rand(n_points)
    h_initial = h_initial / np.sum(h_initial) * target_sum
    h_initial = np.clip(h_initial, 0.0, 1.0)
    
    # Define the objective function
    def objective(h):
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Define the constraints
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    # Bounds for each h_i
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization with SLSQP
    result = minimize(objective, h_initial, method='SLSQP',
                      constraints=[{'type': 'eq', 'fun': constraint_sum}],
                      bounds=bounds,
                      options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8})
    
    # Get the best solution
    best_h = result.x
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
