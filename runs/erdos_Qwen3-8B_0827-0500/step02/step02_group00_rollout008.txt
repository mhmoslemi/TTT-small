def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Choose n_points
    n_points = 200
    dx = 2.0 / n_points
    
    # Generate initial guesses
    initial_guesses = []
    
    # Interpolated initial guess based on initial_h_values
    h_interpolated = interpolate_initial_h(initial_h_values, n_points)
    initial_guesses.append(h_interpolated)
    
    # Generate random initial guesses
    for _ in range(5):  # More random guesses
        h_rand = np.random.rand(n_points)
        h_rand = np.clip(h_rand, 0, 1)
        scale = (n_points / 2) / np.sum(h_rand)
        h_rand = h_rand * scale
        initial_guesses.append(h_rand)
    
    # Generate structured initial guesses
    structured_guesses = []
    # Example: alternating high-low pattern
    for shift in range(3):
        h_structured = np.zeros(n_points)
        for i in range(n_points):
            if (i + shift) % 4 == 0 or (i + shift) % 4 == 3:
                h_structured[i] = 1.0
            else:
                h_structured[i] = 0.0
        # Normalize to meet the integral constraint
        h_structured = h_structured * (n_points / 2) / np.sum(h_structured)
        structured_guesses.append(h_structured)
    initial_guesses.extend(structured_guesses)
    
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    for guess in initial_guesses:
        # Convert to numpy array
        h_initial = np.array(guess)
        # Define the objective function
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Define constraints
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Set up optimization problem
        result = opt.minimize(
            fun=objective,
            x0=h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 500, 'ftol': 1e-8}
        )
        
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return (best_h, best_c5, n_points)
