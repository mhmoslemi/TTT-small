import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Increased points for better resolution

        corr = np.correlate(h, 1 - h, mode='full')

        return max_corr * dx

    cons = [{'type': 'eq', 'fun': constraint_sum}]

        initial_h_values,
        method='SLSQP',

        constraints=cons,

    )
