import time
import scipy.optimize as opt
from scipy.optimize import Bounds

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 50  # Smaller n_points for faster evaluation
    dx = 2.0 / n_points
    
    # Generate multiple initial guesses
    initial_guesses = []
    # Start with a random initial guess

    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:

        def objective(h_vec):
            cross_corr = np.correlate(h_vec, 1 - h_vec, mode='full')
            return np.max(cross_corr) * dx
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2
        
        bounds = Bounds([0.0]*n_points, [1.0]*n_points)
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        
        # Use L-BFGS-B for better exploration
        result = opt.minimize(
            fun=objective,
            x0=guess,

            bounds=bounds,

        )
