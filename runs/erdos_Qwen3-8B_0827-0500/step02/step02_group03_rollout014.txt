import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 80
    dx = 2.0 / n_points
    
    # Generate multiple initial guesses
    initial_guesses = []

    initial_h = initial_h_values.copy()

    initial_guesses.append(initial_h)

    indices = np.random.choice(n_points, size=int(n_points / 2), replace=False)
    binary_initial = np.zeros(n_points)
    binary_initial[indices] = 1.0
    binary_initial = np.clip(binary_initial, 0.0, 1.0)
    initial_guesses.append(binary_initial)
    
    # Third guess: perturbed version of initial_h_values
    perturbed_initial = initial_h.copy()
    perturbed_initial += np.random.normal(0, 0.05, n_points)
    perturbed_initial = np.clip(perturbed_initial, 0.0, 1.0)
    initial_guesses.append(perturbed_initial)
    
    # Objective function
    def objective(h):
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr
    
    # Constraint for integral
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
