import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    # Set n_points and parameters
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate initial guesses
    initial_guesses = []
    # Add initial_h_values
    initial_guesses.append(initial_h_values.copy())
    # Generate a random initial guess
    np.random.seed(seed)
    h_random = np.random.uniform(0, 1, n_points)
    h_random = h_random * (target_sum / np.sum(h_random))
    h_random = np.clip(h_random, 0.0, 1.0)
    initial_guesses.append(h_random)
    
    # Define the objective function
    def objective(h):
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Define the constraint
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    # Bounds for each h_i
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    # Try each initial guess
    for h_initial in initial_guesses:
        # Optimize with SLSQP
        result = minimize(objective, h_initial, method='SLSQP',
                          constraints=[{'type': 'eq', 'fun': constraint_sum}],
                          bounds=bounds,
                          options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8})
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
            # Check time budget
            if time.time() - start_time > budget_s - 1:
                break
    
    return (best_h, best_c5, n_points)
