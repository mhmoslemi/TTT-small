from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    # Define n_points and initial guesses
    n_points = 100
    dx = 2.0 / n_points
    initial_guesses = []
    
    # Add initial_h_values as the first guess
    initial_guesses.append(initial_h_values.copy())
    
    # Add a random guess
    random_guess = np.random.rand(n_points)
    random_guess = random_guess * (n_points / 2) / np.sum(random_guess)
    initial_guesses.append(random_guess)
    
    # Add a uniform guess
    uniform_guess = np.full(n_points, 0.5)
    initial_guesses.append(uniform_guess)
    
    # Clip all guesses to [0, 1]
    initial_guesses = [np.clip(g, 0.0, 1.0) for g in initial_guesses]
    
    # Track the best solution
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        # Ensure the guess meets the sum constraint
        guess = guess * (n_points / 2) / np.sum(guess)
        
        # Define the objective function
        def objective(h):
            ones_minus_h = 1.0 - h
            corr = np.correlate(h, ones_minus_h, mode='full')
            max_corr = np.max(corr) * dx
            return max_corr
        
        # Define the constraint for the sum
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        # Bounds for each h_i
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization settings
        options = {
            'maxiter': 500,
            'ftol': 1e-8,
            'eps': 1e-8,
            'disp': False
        }
        
        # Initial guess for this iteration
        h_initial = guess.copy()
        
        # Perform optimization
        result = minimize(
            fun=objective,
            x0=h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options=options
        )
        
        # Check if the optimization was successful
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Check time budget
