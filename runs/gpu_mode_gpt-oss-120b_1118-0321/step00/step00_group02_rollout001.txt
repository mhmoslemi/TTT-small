@triton.jit
def tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                   N, H,
                   stride_left_b, stride_left_i, stride_left_k, stride_left_h,
                   stride_right_b, stride_right_j, stride_right_k, stride_right_h,
                   stride_out_b, stride_out_i, stride_out_j, stride_out_h,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_x = tl.program_id(0) # i block index
    pid_y = tl.program_id(1) # j block index
    pid_z = tl.program_id(2) # batch*hidden

    batch = pid_z // H
    h = pid_z % H

    # Offsets
    offs_i = pid_x * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_y * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for i,j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Compute base pointers
    left_base = left_ptr + batch * stride_left_b + h * stride_left_h
    right_base = right_ptr + batch * stride_right_b + h * stride_right_h
    out_base = out_ptr + batch * stride_out_b + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)

    for k_iter in range(num_k):
        k_start = k_iter * BLOCK_K
        k_offset = k_start + offs_k
        mask_k = k_offset < N

        # Load left block [M, K]
        left_ptrs = left_base + offs_i[:, None] * stride_left_i + k_offset[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right block transposed = [K, N]
        right_ptrs = right_base + k_offset[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        # shape (K, N) we need (K, BLOCK_N), use mask_k for rows and mask_j for cols
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Multiply-accumulate
        acc += tl.dot(left, right)

    # Write back result
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def tri_mul_triton(left, right):
    # left, right shape: [B,N,N,H]; assume same dtype and device.
    B, N, _, H = left.shape

    out = torch.empty_like(left) # shape same as left

    # Convert to contiguous to ensure stride mapping.
    left_contig = left.contiguous()
    right_contig = right.contiguous()
    out_contig = out

    # Get strides as int64
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left_contig.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right_contig.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out_contig.stride()

    # Set block sizes (choose e.g., 64,64,32)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = ( (N + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N,

"""
TriMul (outgoing) forward implementation using Triton for the heavy
pairwise contraction.

The operation consists of:
1. LayerNorm over the last channel dimension.
2. Linear projections `left` and `right` (dim → hidden_dim) and corresponding
   sigmoid gates.
3. Masking (if a mask is provided) and gating of the projections.
4. A batched matrix multiplication per hidden channel:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented as a Triton kernel (`tri_mul_kernel`).
5. LayerNorm on the hidden dimension, gating with `out_gate`,
   and a final linear projection back to `dim`.

The Triton kernel processes blocks of the `(i, j)` output matrix while looping
over `k`.  The batch‐hidden index is mapped to `program_id(2)`.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton kernel for the O(N^3) contraction (per hidden channel)
# -------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,                # pointers
    N, H,                                        # runtime dims
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # ----- program IDs -----
    pid_i = tl.program_id(0)       # block index over output rows (i)
    pid_j = tl.program_id(1)       # block index over output cols (j)
    pid_z = tl.program_id(2)       # combined batch * hidden dimension

    batch = pid_z // H
    hid   = pid_z % H

    # ----- offsets within the block -----
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for this (batch, hidden) slice
    left_base  = left_ptr  + batch * stride_left_b  + hid * stride_left_h
    right_base = right_ptr + batch * stride_right_b + hid * stride_right_h
    out_base   = out_ptr   + batch * stride_out_b   + hid * stride_out_h

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of k‑blocks
    num_k_blocks = tl.cdiv(N, BLOCK_K)

    for kb in range(num_k_blocks):
        k_start = kb * BLOCK_K
        k_offset = k_start + offs_k
        mask_k = k_offset < N

        # ---- left block: shape (M, K) ----
        left_ptrs = left_base \
                    + offs_i[:, None] * stride_left_i \
                    + k_offset[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # ---- right block transposed: shape (K, N) ----
        # we load as (K, N) so that tl.dot(left, right) works out‑of‑the‑box
        right_ptrs = right_base \
                     + k_offset[:, None] * stride_right_k \
                     + offs_j[None, :] * stride_right_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # ----- dot‑product accumulation -----
        acc += tl.dot(left, right)   # (M,K)·(K,N) -> (M,N)

    # ----- write back -----
    out_ptrs = out_base \
               + offs_i[:, None] * stride_out_i \
               + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# -------------------------------------------------------------
# Helper that wraps the Triton kernel
# -------------------------------------------------------------
def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the contraction out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    for all batches and hidden dimensions using Triton.
    """
    assert left.shape == right.shape, "left and right must have identical shapes"
    B, N, _, H = left.shape
    # Ensure contiguity for stride correctness
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)

    # Strides (int64)
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_rj, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # Tunable block sizes (work well for N up to 1024, H up to 384)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H
    )

    tri_mul_kernel[grid](
        left, right, out,
        N, H,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_rj, s_rk, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4  # modest number of warps; Triton will auto‑tune
    )
    return out


# -------------------------------------------------------------
# Main entry point
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # -----------------------------------------------------------------
    # Unpack configuration
    # -----------------------------------------------------------------
    dim = config["dim"]               # input and output channel size
    hidden_dim = config["hidden_dim"] # projection size

    # -----------------------------------------------------------------
    # LayerNorm on the input
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # -----------------------------------------------------------------
    # Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]  # (hidden_dim, dim)

    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight) # [B,N,N,hidden_dim]

    # -----------------------------------------------------------------
    # Optional masking
    # -----------------------------------------------------------------
    if mask_tensor is not None:
        # mask shape: [B, N, N] -> broadcast to hidden dim
        mask = mask_tensor.to(dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # Gating signals (sigmoid)
    # -----------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)) # [B,N,N,hidden_dim]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))   # [B,N,N,hidden_dim]

    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # Heavy O(N^3) contraction via Triton
    # -----------------------------------------------------------------
    out_hidden = tri_mul_triton(left, right)   # [B,N,N,hidden_dim]

    # -----------------------------------------------------------------
    # Post‑processing: LayerNorm, gating, final linear
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out_hidden_norm = F.layer_norm(out_hidden, (hidden_dim,),
                                   weight=to_out_norm_weight,
                                   bias=to_out_norm_bias)

    out_hidden_norm = out_hidden_norm * out_gate

    to_out_weight = weights["to_out.weight"]   # (dim, hidden_dim)
    output = F.linear(out_hidden_norm, to_out_weight)  # [B,N,N,dim]

    return output
