def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    
    # Get device
    device = input_tensor.device
    dtype = input_tensor.dtype
    
    # Normalize
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # using torch.nn.functional.layer_norm
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    
    # Projections
    left_proj = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    
    # mask handling
    if mask_tensor is None:
        mask = torch.ones_like(input_tensor[...,0], dtype=x_norm.dtype, device=device)  # shape [B,N,N]
    else:
        mask = mask_tensor
    mask = mask.unsqueeze(-1)   # shape [B,N,N,1]
    
    left = left_proj * mask
    right = right_proj * mask
    
    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))
    
    left = left * left_gate
    right = right * right_gate
    
    # cast to float16 for kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)
    
    B, N, _, _ = left_h.shape  # B,N,N,hidden
    # Reshape to (B*hidden, N, N)
    left_perm = left_h.permute(0,3,1,2).contiguous()
    right_perm = right_h.permute(0,3,1,2).contiguous()
    left_flat = left_perm.view(-1, N, N)
    right_flat = right_perm.view(-1, N, N)
    BH = left_flat.shape[0]
    # allocate output
    out_flat = torch.empty_like(left_flat)   # same dtype float16
    
    # compute strides
    stride_l_bh, stride_l_m, stride_l_k = left_flat.stride()
    stride_r_bh, stride_r_m, stride_r_k = right_flat.stride()
    stride_o_bh, stride_o_m, stride_o_n = out_flat.stride()
    
    # Triton kernel parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    
    grid = (BH, (N + BLOCK_M - 1)//BLOCK_M, (N + BLOCK_N - 1)//BLOCK_N)
    # Launch kernel
    trmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_l_bh, stride_l_m, stride_l_k,
        stride_r_bh, stride_r_m, stride_r_k,
        stride_o_bh, stride_o_m, stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    
    # Reshape back to [B,N,N,hidden]
    out = out_flat.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()  # shape [B,N,N,hidden]
    # Cast back to float32
    out = out.to(torch.float32)
    
    # Output normalization and gating
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out_norm * out_gate
    
    # Final projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    
    return out

"""
TriMul (outgoing) forward pass with a fused batched GEMM kernel in Triton.

Algorithm
---------
1. Layer‑norm the input tensor X ∈ ℝ^{B×N×N×C}.
2. Linear projections (no bias) to hidden dimension H:
   L = X·W_left , R = X·W_right .
3. Apply mask (if any) and element‑wise gates:
   L = L * mask * sigmoid(X·W_left_gate)
   R = R * mask * sigmoid(X·W_right_gate)
4. Compute the “tri‑mul”:
   out_{b,i,j,:} = Σ_k L_{b,i,k,:} ⊙ R_{b,j,k,:},
   which is a batched matrix multiplication for each hidden slice.
   We reshape (B,H,N,N) → (B·H,N,N) and launch a Triton kernel that
   computes C = A @ Bᵀ for every batch‑hidden slice.
5. Layer‑norm over the hidden dimension, apply the output gate
   sigmoid(X·W_out_gate) and a final linear projection to the original
   channel dimension C.
All heavy linear work (step 4) is performed in a custom Triton kernel
with float‑16 loads and float‑32 accumulation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trmul_kernel(
    left_ptr,      # *fp16  [BH, M, K]
    right_ptr,     # *fp16  [BH, M, K]   (will be read transposed)
    out_ptr,       # *fp16  [BH, M, N]
    M, N, K,       # matrix sizes (int)
    stride_l_bh, stride_l_m, stride_l_k,
    stride_r_bh, stride_r_m, stride_r_k,
    stride_o_bh, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(0)                     # batch * hidden index
    pid_m  = tl.program_id(1)                     # block row
    pid_n  = tl.program_id(2)                     # block col

    # offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # tile of the accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        cur_k = k_start + offs_k
        k_mask = cur_k < K

        # left tile  (M x K)
        a_ptrs = (left_ptr
                  + pid_bh * stride_l_bh
                  + offs_m[:, None] * stride_l_m
                  + cur_k[None, :] * stride_l_k)
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & k_mask[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # right tile transposed (K x N)  -> we read original right[j,k]
        b_ptrs = (right_ptr
                  + pid_bh * stride_r_bh
                  + cur_k[:, None] * stride_r_k        # row = k
                  + offs_n[None, :] * stride_r_m)       # col = j
        b = tl.load(b_ptrs,
                    mask=k_mask[:, None] & (offs_n[None, :] < N),
                    other=0.0,
                    dtype=tl.float16)

        # dot‑product accumulation
        acc += tl.dot(a, b)

        # advance pointers (implicit via loop index)
    # write back
    c = acc.to(tl.float16)
    out_ptrs = (out_ptr
                + pid_bh * stride_o_bh
                + offs_m[:, None] * stride_o_m
                + offs_n[None, :] * stride_o_n)
    mask_out = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(out_ptrs, c, mask=mask_out)


def custom_kernel(data):
    """
    Triton‑accelerated forward of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on the same device).
    """
    # unpack inputs
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    # --------------------------------------------------------------------- #
    # 1) LayerNorm on the last dimension (dim)
    # --------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # --------------------------------------------------------------------- #
    # 2) Linear projections (no bias)
    # --------------------------------------------------------------------- #
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------------- #
    # 3) Mask handling
    # --------------------------------------------------------------------- #
    if mask_tensor is None:
        mask = torch.ones(x.shape[:-1], dtype=x.dtype, device=device)
    else:
        mask = mask_tensor
    mask = mask.unsqueeze(-1)                     # shape [B,N,N,1]
    left = left * mask
    right = right * mask

    # --------------------------------------------------------------------- #
    # 4) Gates (sigmoid of linear projections)
    # --------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # Cast to half for the heavy GEMM
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape          # B, N, N, hidden_dim

    # --------------------------------------------------------------------- #
    # 5) Prepare tensors for the batched GEMM:
    #    (B, hidden, N, N) → (BH, N, N) where BH = B * hidden
    # --------------------------------------------------------------------- #
    left_h  = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_h = right_h.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_flat  = left_h.view(-1, N, N)   # [BH, N, N]
    right_flat = right_h.view(-1, N, N)  # [BH, N, N]

    BH = left_flat.shape[0]

    # --------------------------------------------------------------------- #
    # 6) Allocate output buffer for the GEMM result (half precision)
    # --------------------------------------------------------------------- #
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # Strides (in elements)
    stride_l_bh, stride_l_m, stride_l_k = left_flat.stride()
    stride_r_bh, stride_r_m, stride_r_k = right_flat.stride()
    stride_o_bh, stride_o_m, stride_o_n = out_flat.stride()

    # --------------------------------------------------------------------- #
    # 7) Launch the Triton kernel
    # --------------------------------------------------------------------- #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (BH,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _trmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_l_bh, stride_l_m, stride_l_k,
        stride_r_bh, stride_r_m, stride_r_k,
        stride_o_bh, stride_o_m, stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------------- #
    # 8) Reshape back to [B, N, N, hidden_dim] and convert to float32
    # --------------------------------------------------------------------- #
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # --------------------------------------------------------------------- #
    # 9) Output LayerNorm, gating, and final linear projection
    # --------------------------------------------------------------------- #
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
