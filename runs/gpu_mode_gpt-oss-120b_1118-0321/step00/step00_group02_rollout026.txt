@triton.jit
def mask_gate_kernel(
    left_ptr, right_ptr,
    mask_ptr,
    left_gate_ptr, right_gate_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    total = B * N * N * H
    base = pid * BLOCK_SIZE
    offs = base + tl.arange(0, BLOCK_SIZE)
    mask = offs < total
    # compute multi-dim indices

    j = idx % N

    right = tl.load(right_ptr + right_offset, mask=mask, other=0.0)

        BLOCK_SIZE=BLOCK_SIZE,
    )
    # At this point left/right have been masked and gated

    # Reshape to (B*hidden_dim, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous()
    right_mat = right.permute(0, 3, 1, 2).contiguous()

        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # Apply out_gate (broadcast)
    out = out * out_gate

    # Final linear projection back to original dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask multiplication and gating for left/right tensors
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_kernel(
    left_ptr, right_ptr,
    mask_ptr,
    left_gate_ptr, right_gate_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place: left = left * mask * left_gate ; right = right * mask * right_gate."""
    pid = tl.program_id(0)
    total = B * N * N * H
    base = pid * BLOCK_SIZE
    offs = base + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    # ---- decode linear index to (b,i,j,h) ----
    idx = offs
    h = idx % H
    idx = idx // H
    j = idx % N
    idx = idx // N
    i = idx % N
    b = idx // N

    # ---- compute pointers ----
    left_offset = (
        b * stride_left_b
        + i * stride_left_i
        + j * stride_left_j
        + h * stride_left_h
    )
    right_offset = (
        b * stride_right_b
        + i * stride_right_i
        + j * stride_right_j
        + h * stride_right_h
    )
    mask_offset = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    lgate_offset = (
        b * stride_lgate_b
        + i * stride_lgate_i
        + j * stride_lgate_j
        + h * stride_lgate_h
    )
    rgate_offset = (
        b * stride_rgate_b
        + i * stride_rgate_i
        + j * stride_rgate_j
        + h * stride_rgate_h
    )

    left  = tl.load(left_ptr  + left_offset,  mask=mask, other=0.0)
    right = tl.load(right_ptr + right_offset, mask=mask, other=0.0)
    m     = tl.load(mask_ptr  + mask_offset,  mask=mask, other=1.0)
    lgate = tl.load(left_gate_ptr  + lgate_offset,  mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + rgate_offset, mask=mask, other=0.0)

    left  = left  * m * lgate
    right = right * m * rgate

    tl.store(left_ptr  + left_offset,  left,  mask=mask)
    tl.store(right_ptr + right_offset, right, mask=mask)

# Entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul operator.
    Steps:
      1. LayerNorm on the input.
      2. Linear projections (left/right) without bias.
      3. Compute sigmoid gates.
      4. Fuse mask and gates with a Triton kernel.
      5. Batched matrix multiplication (left @ rightᵀ) per hidden channel.
      6. LayerNorm, out‑gate scaling and final linear projection.
    Returns a tensor of shape [bs, seq_len, seq_len, dim].
    """
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # the inner dimension (c_z)
    device = input_tensor.device

    # --------------------------------------------------------------
    # 1. LayerNorm on the 4‑D input
    # --------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left  = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Gate values (sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    # --------------------------------------------------------------
    # 4. Optional mask handling (if not supplied, treat as all‑ones)
    # --------------------------------------------------------------
    if mask is None:
        mask = torch.ones(input_tensor.shape[:3], dtype=x_norm.dtype, device=device)
    mask = mask.to(x_norm.dtype)  # shape [B, N, N]

    # --------------------------------------------------------------
    # 5. Fuse mask & gates using Triton (in‑place on left/right)
    # --------------------------------------------------------------
    B, N, _, _ = left.shape
    # Strides for all tensors
    ls = left.stride()          # (B,N,N,H)
    rs = right.stride()
    ms = mask.stride()          # (B,N,N)
    lgs = left_gate.stride()
    rgs = right_gate.stride()

    BLOCK_SIZE = 1024
    total_elements = B * N * N * hidden_dim
    grid = ((total_elements + BLOCK_SIZE - 1) // BLOCK_SIZE,)

    mask_gate_kernel[grid](
        left, right,
        mask,
        left_gate, right_gate,
        B, N, hidden_dim,
        ls[0], ls[1], ls[2], ls[3],
        rs[0], rs[1], rs[2], rs[3],
        ms[0], ms[1], ms[2],
        lgs[0], lgs[1], lgs[2], lgs[3],
        rgs[0], rgs[1], rgs[2], rgs[3],
        BLOCK_SIZE=BLOCK_SIZE,
    )
    # left/right now contain the masked & gated values

    # --------------------------------------------------------------
    # 6. Core TriMul: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched matmul: (left) @ (right)^T per hidden channel.
    # --------------------------------------------------------------
    # reshape to (B, H, N, N) → (B*H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous()
    right_mat = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left_mat.shape
    left_mat  = left_mat.view(B * H, N, N)
    right_mat = right_mat.view(B * H, N, N)

    # Batched matrix multiplication (each channel independent)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # Back to shape [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate scaling and final linear projection
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
