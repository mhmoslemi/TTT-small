Define kernel:

```
@triton.jit
def triton_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_N: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    num_blocks_i = tl.cdiv(N, BLOCK_N)
    num_blocks_j = tl.cdiv(N, BLOCK_N) # use same block size for both dims
    # Compute batch and hidden offset: we compress (b, h) into a single dimension
    bh = pid // (num_blocks_i * num_blocks_j)
    block_idx = pid % (num_blocks_i * num_blocks_j)
    block_i = block_idx // num_blocks_j
    block_j = block_idx % num_blocks_j

    b = bh // H
    h = bh % H

    i_start = block_i * BLOCK_N
    j_start = block_j * BLOCK_N

    # offsets for output
    offs_i = i_start + tl.arange(0, BLOCK_N)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointers for out
    out_ptrs = out_ptr + b * stride_out_b + h * stride_out_h + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j

    acc = tl.zeros((BLOCK_N, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile: shape (BLOCK_N, BLOCK_K)
        left_idxs = (b * stride_left_b + h * stride_left_h
                     + offs_i[:, None] * stride_left_i
                     + offs_k[None, :] * stride_left_k)
        left = tl.load(left_ptr + left_idxs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right tile: shape (BLOCK_N, BLOCK_K) but we need right[j,k] transpose.
        right_idxs = (b * stride_right_b + h * stride_right_h
                      + offs_j[:, None] * stride_right_i   # treat i dimension of right as j? Actually right shape (B,N,N,H) where indices are (b, j, k, h)
                      + offs_k[None, :] * stride_right_k)
        right = tl.load(right_ptr + right_idxs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        # dot product across k dimension: accumulate left * right (elementwise across K)
        acc += tl.dot(left, right)  # shape (BLOCK_N, BLOCK_N)
    # store output
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_i[:, None] & mask_j[None, :])

```
import torch
import triton
import triton.language as tl

@triton.jit
def triton_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # compute dims
    num_blocks_i = tl.cdiv(N, BLOCK_N)
    num_blocks_j = tl.cdiv(N, BLOCK_N)
    tiles_per_batch_h = num_blocks_i * num_blocks_j
    bh = pid // tiles_per_batch_h   # flatten over (B, H)
    tile_id = pid % tiles_per_batch_h
    block_i = tile_id // num_blocks_j
    block_j = tile_id % num_blocks_j

    b = bh // H
    h = bh % H

    # safe guard for out-of-range batch/h
    if b >= B or h >= H:
        return

    i0 = block_i * BLOCK_N
    j0 = block_j * BLOCK_N

    # offsets
    offs_i = i0 + tl.arange(0, BLOCK_N)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers for out
    out_ptrs = out_ptr + b * stride_out_b + h * stride_out_h + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j

    acc = tl.zeros((BLOCK_N, BLOCK_N), dtype=tl.float32)

    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k_blocks):
        k0 = kk * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile (i,k)
        left_ptrs = left_ptr + b * stride_left_b + h * stride_left_h + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k

```
"""
TriMul outgoing kernel: Performs the O(N^3) part of AlphaFold3 triangle multiplicative update.
Given left and right tensors after projection and gating (shape [B,N,N,H]),
computes out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h].
Implemented as a tiled batched GEMM in Triton.
"""

import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # tiling indices
    num_blocks_i = tl.cdiv(N, BLOCK_N)
    num_blocks_j = tl.cdiv(N, BLOCK_N)
    tiles_per_batch_h = num_blocks_i * num_blocks_j
    bh = pid // tiles_per_batch_h        # combined batch & hidden dim index
    tile_id = pid % tiles_per_batch_h
    block_i = tile_id // num_blocks_j
    block_j = tile_id % num_blocks_j

    b = bh // H
    h = bh % H

    i0 = block_i * BLOCK_N
    j0 = block_j * BLOCK_N

    offs_i = i0 + tl.arange(0, BLOCK_N)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # output pointer
    out_ptrs = out_ptr + b * stride_out_b + h * stride_out_h + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j

    acc = tl.zeros((BLOCK_N, BLOCK_N), dtype=tl.float32)

    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k_blocks):
        k0 = kk * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile (i,k)
        left_ptrs = left_ptr + b * stride_left_b + h * stride_left_h + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right tile transposed (k,j)
        right_ptrs = right_ptr + b * stride_right_b + h * stride_right_h + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # accumulate product over k
        acc += tl.dot(left, right)

    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def tri_mul_outgoing_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the outgoing triangle multiplicative update using Triton.
    left/right: tensors of shape [B, N, N, H] (float32)
    Returns: tensor of same shape.
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    BLOCK_N = 32  # tile size for i and j
    BLOCK_K = 32  # tile size for reduction dimension

    grid_i = (N + BLOCK_N - 1) // BLOCK_N
    grid_j = (N + BLOCK_N - 1) // BLOCK_N
    grid = (B * H * grid_i * grid_j,)

    tri_mul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_left_b=left.stride(0),
        stride_left_i=left.stride(1),
        stride_left_k=left.stride(2),
        stride_left_h=left.stride(3),
        stride_right_b=right.stride(0),
        stride_right_i=right.stride(1),
        stride_right_k=right.stride(2),
        stride_right_h=right.stride(3),
        stride_out_b=out.stride(0),
        stride_out_i=out.stride(1),
        stride_out_j=out.stride(2),
        stride_out_h=out.stride(3),
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the outgoing Triangle
    Multiplicative Update (TriMul) used in AlphaFold3.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # LayerNorm over the feature dimension
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # Linear projections (no bias)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask if present
    if not config.get("nomask", False):
        mask_exp = mask.unsqueeze(-1).to(dtype)  # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # Gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous layout for Triton
    left = left.contiguous()
    right = right.contiguous()

    # Core O(N^3) multiplication via Triton
    out = tri_mul_outgoing_triton(left, right)  # shape [B,N,N,hidden_dim]

    # (Optional) LayerNorm on hidden dimension
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate

    # Final linear projection back to original dimension
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul outgoing kernel (AlphaFold3).

This implementation follows the reference PyTorch TriMul module but off‑loads the
quadratic‑over‑sequence (O(N³)) part to a custom Triton kernel.  After
normalisation and linear projections, the heavy step computes:

    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

for every batch `b`, hidden channel `h` and pair of sequence positions
`i, j`.  The kernel treats the (batch, hidden) dimensions as a combined
batch dimension and performs a tiled batched GEMM using `tl.dot`.  The rest
of the forward pass (layer‑norms, gates, final linear) is performed with
standard PyTorch ops.

The kernel tiles the `i` and `j` axes with `BLOCK_N` (default 32) threads,
and the reduction over `k` uses tiles of size `BLOCK_K` (default 32).  Each
CUDA thread computes one output element, accumulating a float32 accumulator.
All operations are performed in float32, keeping the final output identical
to the reference implementation.
"""

import torch
import triton
import triton.language as tl


# ---------------------------------------------------------------------------
# Triton kernel for the O(N³) part of the outgoing TriMul update
# ---------------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # ---- tiling over i, j -------------------------------------------------
    num_blocks_i = tl.cdiv(N, BLOCK_N)
    num_blocks_j = tl.cdiv(N, BLOCK_N)
    tiles_per_batch_h = num_blocks_i * num_blocks_j          # (i,j) tiles for one (b,h)
    bh = pid // tiles_per_batch_h                            # flattened (b,h) index
    tile_id = pid % tiles_per_batch_h
    block_i = tile_id // num_blocks_j
    block_j = tile_id % num_blocks_j

    b = bh // H
    h = bh % H

    i0 = block_i * BLOCK_N
    j0 = block_j * BLOCK_N

    offs_i = i0 + tl.arange(0, BLOCK_N)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointer to the output tile (shape BLOCK_N x BLOCK_N)
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + h * stride_out_h
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    acc = tl.zeros((BLOCK_N, BLOCK_N), dtype=tl.float32)

    # ---- reduction over the k dimension -----------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k_blocks):
        k0 = kk * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i, k, h]
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + h * stride_left_h
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right[j, k, h]  (transposed to shape (k, j) for dot)
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + h * stride_right_h
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_i
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # accumulate Σ_k left[i,k] * right[j,k]
        acc += tl.dot(left, right)   # left: (BLOCK_N, BLOCK_K), right: (BLOCK_K, BLOCK_N)

    # ---- write back --------------------------------------------------------
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def tri_mul_outgoing_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Triton‑accelerated outgoing triangle multiplicative update.

    Args:
        left, right: tensors of shape [B, N, N, H] (float32, contiguous)

    Returns:
        Tensor of shape [B, N, N, H] containing the summed products.
    """
    assert left.shape == right.shape, "left and right must have identical shapes"
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # Tile sizes – chosen to hit the 1024‑thread per block limit.
    BLOCK_N = 32
    BLOCK_K = 32

    # Grid dimensions: one program per (b, h, i‑tile, j‑tile)
    grid_i = (N + BLOCK_N - 1) // BLOCK_N
    grid_j = (N + BLOCK_N - 1) // BLOCK_N
    grid = (B * H * grid_i * grid_j,)

    tri_mul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_left_b=left.stride(0),
        stride_left_i=left.stride(1),
        stride_left_k=left.stride(2),
        stride_left_h=left.stride(3),
        stride_right_b=right.stride(0),
        stride_right_i=right.stride(1),
        stride_right_k=right.stride(2),
        stride_right_h=right.stride(3),
        stride_out_b=out.stride(0),
        stride_out_i=out.stride(1),
        stride_out_j=out.stride(2),
        stride_out_h=out.stride(3),
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # reasonable default for B200
    )
    return out


# ---------------------------------------------------------------------------
# Public entry point required by the task
# ---------------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing Triangle Multiplicative
    Update (TriMul) used in AlphaFold3‑style protein models.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] (may be ignored)
        - weights      : dict of torch.Tensors containing all model parameters
        - config       : dict with keys "dim", "hidden_dim", "nomask" ...

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim]
    """
    # Unpack arguments
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------- #
    # 1. LayerNorm over the feature dimension
    # ------------------------------------------------------------------- #
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------- #
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------- #
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3. Optional mask (applied before gating)
    # ------------------------------------------------------------------- #
    if not config.get("nomask", False):
        # mask shape: [B, N, N] → [B, N, N, 1]
        mask_exp = mask.unsqueeze(-1).to(dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------- #
    # 4. Gating (sigmoid of linear projections on the *original* x)
    # ------------------------------------------------------------------- #
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5. Core O(N³) multiplication via Triton
    # ------------------------------------------------------------------- #
    # Ensure contiguous layout for the kernel
    left = left.contiguous()
    right = right.contiguous()
    out = tri_mul_outgoing_triton(left, right)          # shape [B, N, N, hidden_dim]

    # ------------------------------------------------------------------- #
    # 6. Post‑processing (LayerNorm + final linear)
    # ------------------------------------------------------------------- #
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
