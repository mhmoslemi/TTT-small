"""
TriMul "outgoing" kernel implementation.

The heavy part of TriMul is the Einstein summation:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

We fuse this computation into a Triton kernel that treats each (batch, hidden)
pair as an independent matrix multiplication:
    out_{bh} = left_{bh} @ right_{bh}.T
where left_{bh}, right_{bh} ∈ ℝ^{N×N} and bh = b * hidden + h.

All other operations (LayerNorms, linear projections, gating, mask handling)
are performed with PyTorch. The kernel works on contiguous
[batch*hidden, N, N] tensors and writes the result back in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,         # torch.float32 *
    right_ptr,        # torch.float32 *
    out_ptr,          # torch.float32 *
    N: tl.constexpr,  # sequence length
    stride_batch,     # N * N
    stride_i,         # N
    stride_k,         # 1
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out = left @ right.T for many (batch*hidden) matrices."""
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col
    pid_bh = tl.program_id(2) # batch*hidden index

    # Base offsets for this matrix
    batch_offset = pid_bh * stride_batch

    # Starting coordinates of the tile
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K

        # Index ranges
        offs_i = i0 + tl.arange(0, BLOCK_M)
        offs_j = j0 + tl.arange(0, BLOCK_N)
        offs_k = k0 + tl.arange(0, BLOCK_K)

        # Bounds checking
        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N

        # Load A tile: left[bh, i, k]  -> (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + batch_offset + offs_i[:, None] * stride_i + offs_k[None, :] * stride_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile: right[bh, j, k] -> (BLOCK_N, BLOCK_K)
        b_ptrs = right_ptr + batch_offset + offs_j[:, None] * stride_i + offs_k[None, :] * stride_k
        b = tl.load(b_ptrs,
                    mask=mask_j[:, None] & mask_k[None, :],
                    other=0.0)

        # Accumulate a @ b.T  (dot interprets last dim of a and first of b,
        # trans_b=True swaps axes of b)
        acc += tl.dot(a, b, trans_b=True)

    # Store results
    o_ptrs = out_ptr + batch_offset + offs_i[:, None] * stride_i + offs_j[None, :] * stride_k
    tl.store(o_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul "outgoing" operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (may be ignored if config["nomask"] is True)
            - weights: dict of named weight tensors
            - config: dict containing "dim", "hidden_dim", and "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # Unpack config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]      # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]    # (hidden_dim, dim)
    left_gate_weight = weights["left_gate.weight"]      # (hidden_dim, dim)
    right_gate_weight = weights["right_gate.weight"]    # (hidden_dim, dim)
    out_gate_weight = weights["out_gate.weight"]        # (hidden_dim, dim)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]            # (dim, hidden_dim)

    B, N, _, _ = input_tensor.shape

    # ---------- LayerNorm over last dim ----------
    x_norm = F.layer_norm(input_tensor,
                         normalized_shape=(dim,),
                         weight=norm_weight,
                         bias=norm_bias)

    # ---------- Linear projections ----------
    left = F.linear(x_norm, left_proj_weight)   # (B,N,N,hidden_dim)
    right = F.linear(x_norm, right_proj_weight)

    # ---------- Optional mask ----------
    if not nomask:
        # mask: [B,N,N] -> [B,N,N,1] broadcast to hidden dim
        mask_f = mask.to(dtype=dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ---------- Gating ----------
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ---------- Triton kernel: pairwise multiply‑sum ----------
    # Rearrange to (B, hidden, N, N) and flatten batch*hidden dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_mat = left_perm.view(-1, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(-1, N, N) # (B*H, N, N)

    out_mat = torch.empty_like(left_mat)  # (B*H, N, N), float32

    # Kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    # Strides for contiguous [batch_h, N, N] layout
    stride_batch = N * N
    stride_i = N
    stride_k = 1

    # Launch
    tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,
        stride_batch,
        stride_i,
        stride_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Ensure kernel completion before next ops
    torch.cuda.synchronize(device)

    # Reshape back to (B, N, N, hidden_dim)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------- Output LayerNorm ----------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    # ---------- Output gating ----------
    out = out * out_gate

    # ---------- Final linear projection ----------
    output = F.linear(out, to_out_weight)   # (B,N,N,dim)

    return output
