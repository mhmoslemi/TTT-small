@triton.jit
def triton_trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_j, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    hidden_dim,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # Determine batch and hidden dimension
    pid = tl.program_id(0)
    batch = pid // hidden_dim
    h = pid % hidden_dim

    # Compute offsets for i and j tiles
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this batch and hidden slice
    left_base = left_ptr + batch * stride_left_batch + h * stride_left_h
    right_base = right_ptr + batch * stride_right_batch + h * stride_right_h
    out_base = out_ptr + batch * stride_out_batch + h * stride_out_h

    accumulator = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i,k] shape (M,K)
        left_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right_T[k,j] shape (K,N) using right[j,k] memory
        right_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Accumulate matmul
        accumulator += tl.dot(left, right)

    # Store result
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, accumulator, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # parse config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, N, _, _ = input_tensor.shape
    # Could trust bs,N,N,dim shape.

    # LayerNorm over dim
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    eps = 1e-5

    # Compute mean and var across last dim
    x = input_tensor
    mean = x.mean(-1, keepdim=True)
    var = ((x - mean) ** 2).mean(-1, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    if norm_weight is not None:
        x_norm = x_norm * norm_weight + norm_bias

    # Proj linear transforms
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # mask
    if mask is not None:
        mask_ = mask.unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Compute out using Triton kernel
    out = torch.empty((bs, N, N, hidden_dim), dtype=torch.float32, device=input_tensor.device)

    # get strides
    # left stride order: (B,N,N,H)
    stride_lb = left.stride(0)
    stride_li = left.stride(1)
    stride_lk = left.stride(2)
    stride_lh = left.stride(3)

    stride_rb = right.stride(0)
    stride_rj = right.stride(1)
    stride_rk = right.stride(2)
    stride_rh = right.stride(3)

    stride_ob = out.stride(0)
    stride_oi = out.stride(1)
    stride_oj = out.stride(2)
    stride_oh = out.stride(3)

    # define block sizes: tune
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (bs * hidden_dim,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    triton_trimul_gemm_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        N,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        hidden_dim,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K)

    # After kernel, out shape [B,N,N,hidden_dim]
    # Apply to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    eps_norm = 1e-5
    mean_out = out.mean(-1, keepdim=True)
    var_out = ((out - mean_out) ** 2).mean(-1, keepdim=True)
    out_norm = (out - mean_out) / torch.sqrt(var_out + eps_norm)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # gating and final linear
    out_norm = out_norm * out_gate

    out_final = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return out_final

"""
TriMul forward (outgoing) implementation.

The heavy cubic part
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed by a custom Triton GEMM kernel that treats each hidden
dimension `d` independently (one kernel instance per (batch, d)).
All other operations (LayerNorms, linear projections, gating,
final LayerNorm & linear) are done with PyTorch for clarity.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ----------------------------------------------------------------------
# Triton kernel: GEMM for a single hidden slice.
# ----------------------------------------------------------------------
@triton.jit
def _tritermul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,            # pointers
    N,                                      # sequence length
    stride_lb, stride_li, stride_lk, stride_lh,  # left strides
    stride_rb, stride_rj, stride_rk, stride_rh,  # right strides
    stride_ob, stride_oi, stride_oj, stride_oh,  # out strides
    hidden_dim,                               # total hidden dim (runtime const)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    where each program instance works on a single (b,h) pair.
    """
    # -----------------------------------------
    # Identify (batch, hidden) for this program
    # -----------------------------------------
    pid = tl.program_id(0)
    batch = pid // hidden_dim          # batch index
    h = pid % hidden_dim               # hidden-channel index

    # Grid indices for the (i,j) matrix tile
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Offsets for the current tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for this (batch, hidden) slice
    left_base  = left_ptr  + batch * stride_lb + h * stride_lh
    right_base = right_ptr + batch * stride_rb + h * stride_rh
    out_base   = out_ptr   + batch * stride_ob + h * stride_oh

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i,k]  -> shape (M, K)
        left_ptrs = left_base + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right_T[k,j] = right[j,k]   -> shape (K, N)
        right_ptrs = right_base + offs_k[:, None] * stride_rk + offs_j[None, :] * stride_rj
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # GEMM accumulation
        acc += tl.dot(left, right)

    # Store the result
    out_ptrs = out_base + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    # Configuration
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Compute mean/var over the last dimension
    mean = input_tensor.mean(-1, keepdim=True)
    var = ((input_tensor - mean) ** 2).mean(-1, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_weight + norm_bias  # broadcast over the last dim

    # ------------------------------------------------------------------
    # Linear projections
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Apply mask (if provided)
    # ------------------------------------------------------------------
    if mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triton kernel for the cubic einsum
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides (int64)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # Block sizes – chosen to give good occupancy on typical GPUs.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,                       # one program per (batch, hidden)
        (N + BLOCK_M - 1) // BLOCK_M,        # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,        # j‑tiles
    )

    _tritermul_gemm_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        N,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,  # reasonable default
    )
    # Ensure kernel completion before next ops
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # LayerNorm on hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    mean_out = out.mean(-1, keepdim=True)
    var_out = ((out - mean_out) ** 2).mean(-1, keepdim=True)
    out = (out - mean_out) / torch.sqrt(var_out + eps)
    out = out * to_out_norm_weight + to_out_norm_bias

    # ------------------------------------------------------------------
    # Apply output gate and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
