    # Projections
    left = torch.nn.functional.linear(x, left_proj_weight) # shape [bs,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    left = left.contiguous()
    right = right.contiguous()

    if mask is not None:
        mask = mask.to(x.dtype)
        mask = mask.unsqueeze(-1)  # shape [bs,N,N,1]
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Convert left and right to half for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Output tensor
    out = torch.empty((bs, N, N, hidden_dim), dtype=torch.float32, device=x.device)

    # Extract strides
    stride_left_bs, stride_left_i, stride_left_k, stride_left_h = left_fp16.stride()
    stride_right_bs, stride_right_i, stride_right_k, stride_right_h = right_fp16.stride()
    stride_out_bs, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Kernel launch config.
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 64

    # Adjust block sizes if N is small (e.g., 256). Ensure block size not exceed dimension.
    # Not necessary.

    num_j = (N + BLOCK_J - 1) // BLOCK_J
    num_bh = bs * hidden_dim

        right_fp16,

        stride_right_bs, stride_right_i, stride_right_k, stride_right_h,
        stride_out_bs, stride_out_i, stride_out_j, stride_out_h,
        bs, N, hidden_dim,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K
    )

    out_norm = out_norm * out_gate

    output = torch.nn.functional.linear(out_norm, to_out_weight)

"""
TriMul "outgoing" implementation with Triton acceleration.

The heavy O(N³) part (the pairwise multiplication over the third axis) is
implemented as a custom Triton kernel that computes, for each batch and hidden
channel d:

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix multiplication
`left[..., :, :, d] @ right[..., :, :, d].transpose(-2, -1)`.
All other operations (LayerNorms, linear projections, gates) are performed
with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: computes out = left @ rightᵀ over the third dimension.
# ----------------------------------------------------------------------
@triton.jit
def triton_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    # strides (in elements) for the three tensors
    stride_left_bs, stride_left_i, stride_left_k, stride_left_h,
    stride_right_bs, stride_right_i, stride_right_k, stride_right_h,
    stride_out_bs, stride_out_i, stride_out_j, stride_out_h,
    # runtime dimensions
    BS, N, H,
    # compile‑time block sizes
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    The kernel is tiled over i, j and the (batch, hidden) pair.
    """
    pid_i = tl.program_id(0)   # tile over rows   (i)
    pid_j = tl.program_id(1)   # tile over cols   (j)
    pid_bh = tl.program_id(2)  # combined batch & hidden channel

    # ------------------------------------------------------------------
    # Decode indices
    # ------------------------------------------------------------------
    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    # batch b and hidden channel h from the fused dimension
    b = pid_bh // H
    h = pid_bh % H

    mask_i = i < N
    mask_j = j < N

    # Pointer to the output tile
    out_off = (
        b * stride_out_bs
        + i[:, None] * stride_out_i
        + j[None, :] * stride_out_j
        + h * stride_out_h
    )
    out_ptrs = out_ptr + out_off

    # Accumulator in float32 for precision
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction axis k
    # ------------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k):
        k_base = kk * BLOCK_K
        k = k_base + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # left[b, i, k, h]  -> shape (BLOCK_I, BLOCK_K)
        left_off = (
            b * stride_left_bs
            + i[:, None] * stride_left_i
            + k[None, :] * stride_left_k
            + h * stride_left_h
        )
        left = tl.load(left_ptr + left_off,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right[b, j, k, h] transposed to (k, j) -> shape (BLOCK_K, BLOCK_J)
        # original right indices: (b, j, k, h)
        # we swap k and j in memory order to obtain rightᵀ.
        right_off = (
            b * stride_right_bs
            + k[:, None] * stride_right_k   # k uses the second‑N stride
            + j[None, :] * stride_right_i   # j uses the first‑N stride
            + h * stride_right_h
        )
        right = tl.load(right_ptr + right_off,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # Convert to fp32 for the dot‑product
        left_f = left.to(tl.float32)
        right_f = right.to(tl.float32)

        # Matrix multiplication for this k‑tile
        acc += tl.dot(left_f, right_f)   # (BLOCK_I, BLOCK_K) @ (BLOCK_K, BLOCK_J)

    # ------------------------------------------------------------------
    # Write the result
    # ------------------------------------------------------------------
    tl.store(out_ptrs, acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point called by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul forward (outgoing) with Triton‑accelerated pairwise multiplication.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)

        - input_tensor  : [B, N, N, C]        (float32)
        - mask          : [B, N, N] (optional, broadcastable)
        - weights       : dict of parameter tensors
        - config        : dict with keys "dim", "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    eps = 1e-5

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Load weight tensors
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # LayerNorm on the input (last dimension = dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight,
                     bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # Ensure contiguous layout for the subsequent Triton kernel
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # Optional masking
    # ------------------------------------------------------------------
    if mask is not None:
        # Convert mask to the same dtype as the tensors and broadcast
        mask_f = mask.to(dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gating pathways
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triton kernel: batch‑wise multiplication over the third axis
    # ------------------------------------------------------------------
    # Cast to fp16 for higher throughput; accumulation stays fp32.
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out = torch.empty((bs, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides in elements (not bytes) – required by the kernel.
    stride_left_bs, stride_left_i, stride_left_k, stride_left_h = left_fp16.stride()
    stride_right_bs, stride_right_i, stride_right_k, stride_right_h = right_fp16.stride()
    stride_out_bs, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Tile size – tuned for typical NVIDIA B200 memory limits.
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 64

    # Grid dimensions
    num_i = (N + BLOCK_I - 1) // BLOCK_I
    num_j = (N + BLOCK_J - 1) // BLOCK_J
    num_bh = bs * hidden_dim

    # Launch
    triton_tri_mul_kernel[(num_i, num_j, num_bh)](
        left_fp16,
        right_fp16,
        out,
        stride_left_bs, stride_left_i, stride_left_k, stride_left_h,
        stride_right_bs, stride_right_i, stride_right_k, stride_right_h,
        stride_out_bs, stride_out_i, stride_out_j, stride_out_h,
        bs, N, hidden_dim,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        num_warps=4,                # typical value for fp16 matmul
    )

    # ------------------------------------------------------------------
    # Post‑processing: LayerNorm, output gate, final linear
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                            bias=to_out_norm_bias, eps=eps)

    out_norm = out_norm * out_gate               # element‑wise gating

    # Final projection back to original dimension (no bias)
    output = F.linear(out_norm, to_out_weight)   # [B, N, N, dim]

    return output
