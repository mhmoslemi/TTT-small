BLOCK_D = tl.constexpr

@triton.jit
def mask_gate_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr,
                     stride_proj_batch, stride_proj_i, stride_proj_j, stride_proj_d,
                     stride_gate_batch, stride_gate_i, stride_gate_j, stride_gate_d,
                     stride_mask_batch, stride_mask_i, stride_mask_j,
                     stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
                     D, BLOCK_D: tl.constexpr):
    b = tl.program_id(0)
    i = tl.program_id(1)
    j = tl.program_id(2)
    # define range over d
    offs_d = tl.arange(0, BLOCK_D)
    # compute mask offset (scalar)
    mask_offset = b * stride_mask_batch + i * stride_mask_i + j * stride_mask_j
    mask = tl.load(mask_ptr + mask_offset, mask= (offs_d == 0))  # broadcast scalar
    # load proj and gate:
    proj_offset = b * stride_proj_batch + i * stride_proj_i + j * stride_proj_j + offs_d * stride_proj_d
    gate_offset = b * stride_gate_batch + i * stride_gate_i + j * stride_gate_j + offs_d * stride_gate_d
    proj = tl.load(proj_ptr + proj_offset, mask=offs_d < D, other=0.0)
    gate = tl.load(gate_ptr + gate_offset, mask=offs_d < D, other=0.0)

    out = proj * mask * gate
    # store
    out_offset = b * stride_out_batch + i * stride_out_i + j * stride_out_j + offs_d * stride_out_d
    tl.store(out_ptr + out_offset, out, mask=offs_d < D)

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    device = input_tensor.device

    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]

    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Projections
    left_proj = torch.nn.functional.linear(x, left_proj_weight)  # (B,N,N,hidden_dim)
    right_proj = torch.nn.functional.linear(x, right_proj_weight)

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    B, N, _, _ = left_proj.shape

        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        mask_s = mask.stride()

            gate_s[0], gate_s[1], gate_s[2], gate_s[3],

"""
TriMul forward kernel (outgoing version).

Algorithm
---------
1. Layer‑norm the input tensor x ∈ ℝ^{B×N×N×C}.
2. Linear projections to hidden dimension H:
       left_proj  = x·W_left   (B,N,N,H)
       right_proj = x·W_right  (B,N,N,H)
3. Compute three gates (sigmoid):
       left_gate  = σ(x·W_left_gate)
       right_gate = σ(x·W_right_gate)
       out_gate   = σ(x·W_out_gate)
4. Fuse mask (if present) and gates with a small Triton kernel:
       left  = left_proj  * left_gate  * mask
       right = right_proj * right_gate * mask
5. Compute the core “TriMul’’ operation:
       out_{b,i,j,:} = Σ_k left_{b,i,k,:} * right_{b,j,k,:}
   This is a batch of H independent matrix‑multiplications:
       out = (left_perm) @ (right_perm)^T
   where left_perm = left.permute(0,3,1,2).reshape(B·H,N,N)
   and right_perm is analogous.
6. Apply a second LayerNorm over the H dimension.
7. Scale by out_gate and a final linear projection back to C.

Only the mask‑gate fusion (step 4) is implemented in Triton; the
O(N³) accumulation is performed by a highly‑optimized batched GEMM
(torch.bmm), which on modern GPUs far out‑performs a naïve Triton
loop.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _apply_mask_gate_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    # strides for proj (B,N,N,H)
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    # strides for gate (B,N,N,H)
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    # strides for mask (B,N,N)
    stride_mask_b, stride_mask_i, stride_mask_j,
    # strides for output (B,N,N,H)
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    H: tl.constexpr,                   # hidden dimension (runtime)
    BLOCK_H: tl.constexpr,             # compile‑time tile size
):
    b = tl.program_id(0)
    i = tl.program_id(1)
    j = tl.program_id(2)

    # ------------------------------------------------------------
    # Load scalar mask (broadcasted later)
    # ------------------------------------------------------------
    mask_offset = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    mask_val = tl.load(mask_ptr + mask_offset)

    # ------------------------------------------------------------
    # Vector offsets for the hidden dimension
    # ------------------------------------------------------------
    offs_h = tl.arange(0, BLOCK_H)

    # Load projection and gate (masked by H)
    proj_offset = (
        b * stride_proj_b + i * stride_proj_i + j * stride_proj_j
        + offs_h * stride_proj_h
    )
    gate_offset = (
        b * stride_gate_b + i * stride_gate_i + j * stride_gate_j
        + offs_h * stride_gate_h
    )
    proj = tl.load(proj_ptr + proj_offset, mask=offs_h < H, other=0.0)
    gate = tl.load(gate_ptr + gate_offset, mask=offs_h < H, other=0.0)

    # Fuse everything
    out = proj * gate * mask_val

    # Store result
    out_offset = (
        b * stride_out_b + i * stride_out_i + j * stride_out_j
        + offs_h * stride_out_h
    )
    tl.store(out_ptr + out_offset, out, mask=offs_h < H)


def custom_kernel(data):
    """
    Args:
        data: Tuple (input_tensor, mask, weights, config)
              - input_tensor: torch.Tensor [B, N, N, C]
              - mask: torch.Tensor [B, N, N] (float or bool)
              - weights: dict of model parameters (torch.Tensors)
              - config: dict with keys 'dim', 'hidden_dim', 'nomask' (bool)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # -----------------------------------------------------------------
    # Unpack weights
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm over the channel dimension C
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # (B, N, N, C)

    # -----------------------------------------------------------------
    # 2️⃣ Linear projections to hidden_dim H
    # -----------------------------------------------------------------
    left_proj = F.linear(x, left_proj_weight)   # (B, N, N, H)
    right_proj = F.linear(x, right_proj_weight)  # (B, N, N, H)

    # -----------------------------------------------------------------
    # 3️⃣ Compute gates (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    B, N, _, _ = left_proj.shape

    # -----------------------------------------------------------------
    # 4️⃣ Fuse mask and gates with Triton (if mask is active)
    # -----------------------------------------------------------------
    if not config.get("nomask", False):
        # ensure mask dtype matches computation dtype
        if mask.dtype != dtype:
            mask = mask.to(dtype)

        # allocate output buffers for gated tensors
        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        # stride tuples (B, N, N, H) for all tensors
        lp_s = left_proj.stride()
        lg_s = left_gate.stride()
        rp_s = right_proj.stride()
        rg_s = right_gate.stride()
        m_s = mask.stride()
        lo_s = left.stride()
        ro_s = right.stride()

        # block size for hidden dimension (tuned for 128‑dim hidden)
        BLOCK_H = 64

        grid = (B, N, N)  # (program_id0, program_id1, program_id2)

        # left side
        _apply_mask_gate_kernel[grid](
            left_proj,
            left_gate,
            mask,
            left,
            lp_s[0],
            lp_s[1],
            lp_s[2],
            lp_s[3],
            lg_s[0],
            lg_s[1],
            lg_s[2],
            lg_s[3],
            m_s[0],
            m_s[1],
            m_s[2],
            lo_s[0],
            lo_s[1],
            lo_s[2],
            lo_s[3],
            H=hidden_dim,
            BLOCK_H=BLOCK_H,
        )

        # right side
        _apply_mask_gate_kernel[grid](
            right_proj,
            right_gate,
            mask,
            right,
            rp_s[0],
            rp_s[1],
            rp_s[2],
            rp_s[3],
            rg_s[0],
            rg_s[1],
            rg_s[2],
            rg_s[3],
            m_s[0],
            m_s[1],
            m_s[2],
            ro_s[0],
            ro_s[1],
            ro_s[2],
            ro_s[3],
            H=hidden_dim,
            BLOCK_H=BLOCK_H,
        )
    else:
        # No mask → simple elementwise multiplication with gates
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 5️⃣ Core TriMul: sum_k left[b,i,k,:] * right[b,j,k,:]
    #     Implemented as a batched GEMM across the hidden dimension.
    # -----------------------------------------------------------------
    # bring hidden dimension to the batch axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)          # (B*H, N, N)
    right_flat = right_perm.view(B * H, N, N)        # (B*H, N, N)

    # batched matrix multiplication → (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6️⃣ Second LayerNorm over H
    # -----------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 7️⃣ Apply out gate and final linear projection back to C
    # -----------------------------------------------------------------
    out_norm = out_norm * out_gate
    out_final = F.linear(out_norm, to_out_weight)   # (B, N, N, C)

    return out_final
