def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype # typically float32

    # Get dims
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm: x_norm = (x - mean)/sqrt(var+eps) * weight + bias
    # Using torch.nn.functional.layer_norm
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # Linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Mask (optional). If mask is None or mask.shape is (0?) else use.
    if mask is not None:
        mask_ = mask.unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_
        right = right * mask_
    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Cast to half for kernel
    left_h = left.to(torch.float16)

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute per-channel matrix multiplication out = left @ right^T.
    left, right: [B,N,N,H] tensors (half precision). out: [B,N,N,H] (float32).
    Each program computes a block (BLOCK_M, BLOCK_N) of the output matrix for a given batch and hidden channel.
    """
    pid = tl.program_id(0)
    b = pid // H
    h = pid % H
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Compute offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # Load left tile: shape (BLOCK_M, BLOCK_K)
        L_ptrs = left_ptr + b * stride_l_b + offs_i[:, None] * stride_l_i + k_offset[None, :] * stride_l_k + h * stride_l_h
        L = tl.load(L_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=tl.float16(0.0))

        # Load right tile transposed: shape (BLOCK_K, BLOCK_N)
        R_ptrs = right_ptr + b * stride_r_b + k_offset[:, None] * stride_r_k + offs_j[None, :] * stride_r_i + h * stride_r_h
        R = tl.load(R_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=tl.float16(0.0))

        # Accumulate
        acc += tl.dot(L, R)  # (BLOCK_M, BLOCK_N) -> float32

    # Store results
    O_ptrs = out_ptr + b * stride_o_b + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j + h * stride_o_h
    tl.store(O_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
tri_mul (outgoing) – Triton implementation

The core of the TriMul module is the tensor contraction
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
where `left` and `right` are the projected and gated 4‑D tensors
(shape [B, N, N, H]) and the sum runs over the intermediate sequence index `k`.
This is equivalent to a batched matrix multiplication
    out = left @ right.transpose(-1, -2)
for each (batch, hidden) pair.

The kernel below computes this batched GEMM with Triton, tiling the
output (i, j) plane with blocks of size BLOCK_M × BLOCK_N and processing
the reduction dimension `k` in tiles of size BLOCK_K.
All heavy arithmetic is performed in float16 with float32 accumulation.
The surrounding linear layers, layer‑norms and gating are handled in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,               # half *
    right_ptr,              # half *
    out_ptr,                # float *
    B, N, H,                # int32 scalars
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,  # tile size in i
    BLOCK_N: tl.constexpr,  # tile size in j
    BLOCK_K: tl.constexpr,  # tile size in k
):
    """Compute out = left @ right^T for each (batch, hidden) pair.

    Arguments
    ---------
    left_ptr, right_ptr, out_ptr : pointers to the tensors
        left/right: [B, N, N, H] (half precision)
        out       : [B, N, N, H] (float32)
    B, N, H : problem dimensions
    stride_* : element strides for the corresponding tensors
    BLOCK_* : compile‑time tile sizes
    """
    pid = tl.program_id(0)                # encodes (batch, hidden)
    b = pid // H
    h = pid % H

    pid_i = tl.program_id(1)               # tile index for i dimension
    pid_j = tl.program_id(2)               # tile index for j dimension

    # Offsets for the output tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[tile] : shape (BLOCK_M, BLOCK_K)  -> indices (i, k)
        L_ptrs = left_ptr + b * stride_l_b \
                 + offs_i[:, None] * stride_l_i \
                 + offs_k[None, :] * stride_l_k \
                 + h * stride_l_h
        L = tl.load(L_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=tl.float16(0.0))

        # right[tile] : we need right^T, i.e. shape (BLOCK_K, BLOCK_N) with indices (k, j)
        R_ptrs = right_ptr + b * stride_r_b \
                 + offs_k[:, None] * stride_r_k \
                 + offs_j[None, :] * stride_r_i \
                 + h * stride_r_h
        R = tl.load(R_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=tl.float16(0.0))

        # Accumulate dot product; result is (BLOCK_M, BLOCK_N) in float32
        acc += tl.dot(L, R)

    # Store the computed block
    O_ptrs = out_ptr + b * stride_o_b \
             + offs_i[:, None] * stride_o_i \
             + offs_j[None, :] * stride_o_j \
             + h * stride_o_h
    tl.store(O_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.FloatTensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] (or None)
        - weights      : dict of model parameters (see problem statement)
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.FloatTensor
        Tensor of shape [B, N, N, dim] after the TriMul forward pass.
    """
    torch.set_grad_enabled(False)

    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------- LayerNorm on the input ------------------- #
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------- Linear projections ------------------- #
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ------------------- Optional mask ------------------- #
    if mask is not None and mask.numel() > 0:
        mask_ = mask.unsqueeze(-1).to(left.dtype)   # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------- Gating ------------------- #
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------- Cast to half for the heavy kernel ------------------- #
    left_h = left.contiguous().to(torch.float16)
    right_h = right.contiguous().to(torch.float16)

    # Output container for the matmul, float32 for later ops
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # ------------------- Strides (in elements) ------------------- #
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_h.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right_h.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # ------------------- Triton launch configuration ------------------- #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    # Adjust block size for very small sequences (optional)
    if N <= 32:
        BLOCK_M = BLOCK_N = BLOCK_K = 32

    grid = (
        B * hidden_dim,                # program id 0 : (batch, hidden)
        (N + BLOCK_M - 1) // BLOCK_M,  # program id 1 : rows i
        (N + BLOCK_N - 1) // BLOCK_N   # program id 2 : cols j
    )

    # Launch the kernel
    tri_mul_kernel[grid](
        left_h,
        right_h,
        out,
        B,
        N,
        hidden_dim,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_i,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------- Final layernorm, gating and projection ------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])  # no bias as per reference

    return output
```
