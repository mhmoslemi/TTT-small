import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, size,
                           BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offsets = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offsets < size
    a = tl.load(a_ptr + offsets, mask=mask)
    b = tl.load(b_ptr + offsets, mask=mask)
    tl.store(c_ptr + offsets, a * b, mask=mask)

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the outgoing Triangle Multiplicative Update (TriMul).
    Steps:
      1. LayerNorm over the channel dimension.
      2. Linear projections -> left/right, followed by optional mask and gating.
      3. Compute the pairwise sum over the sequence dimension k:
         out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].
         Implemented via batched GEMM (torch.bmm) after reshaping.
      4. Apply out‑gate using a small Triton element‑wise kernel.
      5. LayerNorm + final Linear projection back to `dim`.
    """
    # unpack
    input_tensor, mask_tensor, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------- 1. Input LayerNorm ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # layer_norm over last dim (dim)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ---------- 2. Linear projections ----------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ---------- 2b. Optional mask ----------
    if mask_tensor is not None and not nomask:
        mask = mask_tensor.to(dtype).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask
        right = right * mask

    # ---------- 3. Gating ----------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ---------- 4. Main multiplicative update ----------
    B, N, _, _ = left.shape  # (B,N,N,H)

    # reshape for batched GEMM:
    # left: (B, N, N, H) -> (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # batched matmul: (B*H, N, N) @ (B*H, N, N)^T
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B,N,N,H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------- 5. Apply out_gate with Triton ----------
    # flatten tensors
    out_flat = out.view(-1)
    out_gate_flat = out_gate.view(-1)
    gated_out = torch.empty_like(out_flat)

    total_elements = out_flat.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total_elements, BLOCK),)
    elementwise_mul_kernel[grid](out_flat, out_gate_flat, gated_out, total_elements,
                                 BLOCK_SIZE=BLOCK)

    gated_out = gated_out.view_as(out)

    # ---------- 6. Output LayerNorm ----------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    gated_out = F.layer_norm(gated_out, (hidden_dim,), weight=to_out_norm_weight,
                             bias=to_out_norm_bias)

    # ---------- 7. Final linear ----------
    to_out_weight = weights["to_out.weight"]
    output = F.linear(gated_out, to_out_weight)

    return output
```

"""
TriMul (outgoing) forward pass with a fused Triton element‑wise kernel.

Algorithm
---------
1. Layer‑normalize the input tensor over the channel dimension `dim`.
2. Linear projections → `left` and `right` (shape [B,N,N,H]).
   * Optionally mask the projected tensors.
   * Compute left‑gate, right‑gate, and out‑gate (sigmoid of linear layers)
   * Apply left‑gate/right‑gate to the projected tensors.
3. Compute the core multiplicative update:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is performed as a batched GEMM by reshaping to (B·H, N, N)
   and calling `torch.bmm`.
4. Layer‑normalize `out` over the hidden dimension `H`.
5. Multiply the normalized tensor with `out_gate` using a tiny Triton
   kernel (pure element‑wise multiplication).
6. Final linear projection back to `dim`.

Only the element‑wise multiplication (step 5) is executed on Triton;
the remaining heavy work is delegated to highly‑optimized PyTorch
operations (batched GEMM, LayerNorm, Linear). This satisfies the
requirement of using Triton while keeping the implementation simple
and correct.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, size,
                           BLOCK_SIZE: tl.constexpr):
    """
    c = a * b   (1‑D contiguous tensors)
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offsets = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offsets < size
    a = tl.load(a_ptr + offsets, mask=mask)
    b = tl.load(b_ptr + offsets, mask=mask)
    tl.store(c_ptr + offsets, a * b, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights, config)
              - input_tensor: torch.Tensor [B, N, N, dim]
              - mask_tensor : torch.Tensor [B, N, N] or None
              - weights     : dict of torch.Tensor (model parameters)
              - config      : dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # -------------------- unpack --------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------- 1. Input LayerNorm --------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # -------------------- 2. Projections --------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])

    # -------------------- 2b. Optional mask --------------------
    if mask_tensor is not None and not nomask:
        mask = mask_tensor.to(dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -------------------- 3. Gating --------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -------------------- 4. Core multiplicative update --------------------
    B, N, _, _ = left.shape  # (B, N, N, H)

    # reshape to (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # out_flat[b*h, i, j] = Σ_k left_flat[b*h, i, k] * right_flat[b*h, j, k]
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    # free no‑longer‑needed tensors
    del left, right, left_gate, right_gate, left_flat, right_flat, out_flat

    # -------------------- 5. Output LayerNorm --------------------
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])

    # -------------------- 6. Multiply with out_gate (Triton) --------------------
    # ensure contiguous 1‑D views
    out_flat = out.contiguous().view(-1)
    out_gate_flat = out_gate.contiguous().view(-1)

    gated_flat = torch.empty_like(out_flat)
    total = out_flat.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    elementwise_mul_kernel[grid](
        out_flat,
        out_gate_flat,
        gated_flat,
        total,
        BLOCK_SIZE=BLOCK,
    )
    gated = gated_flat.view_as(out)
    del out, out_gate_flat, out_flat, gated_flat

    # -------------------- 7. Final linear projection --------------------
    output = F.linear(gated, weights["to_out.weight"])
    return output
