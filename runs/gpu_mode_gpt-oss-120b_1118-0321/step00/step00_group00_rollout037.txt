def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # maybe config contains "nomask" not needed
    eps = 1e-5

    # LayerNorm on input
    x_norm = torch.nn.functional.layer_norm(input_tensor,
                                            normalized_shape=(dim,),
                                            weight=weights["norm.weight"],
                                            bias=weights["norm.bias"],
                                            eps=eps)

    # Linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # apply mask if present and mask != None
    if mask is not None:
        # mask shape (B,N,N)
        mask = mask.to(input_tensor.dtype).unsqueeze(-1)  # to broadcast
        left = left * mask
        right = right * mask

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # Prepare for kernel
    B, N, _, _ = left.shape

    # Permute to (B, hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # flatten batch*hidden dimension
    total_batch = B * hidden_dim
    left_flat = left_perm.view(total_batch, N, N)
    right_flat = right_perm.view(total_batch, N, N)

    # allocate output
    out_flat = torch.empty_like(left_flat)

    # stride arguments
    stride_am = left_flat.stride(0)
    stride_ak = left_flat.stride(1)
    stride_ah = left_flat.stride(2)

    stride_br = right_flat.stride(0)
    stride_bk = right_flat.stride(1)  # original stride for K dimension (N)
    stride_bn = right_flat.stride(2)  # original stride for N dimension (1)

    # transposed strides
    stride_bk_t = stride_bn   # 1
    stride_bn_t = stride_bk   # N

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    # Choose block sizes based on N
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    # Compute grid
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (total_batch * num_blocks_m * num_blocks_n,)

    # Launch
    tri_mul_kernel[grid](
        left_flat, right_flat, out_flat,
        total_batch, N, N, N,
        stride_am, stride_ak, stride_ah,
        stride_br, stride_bk_t, stride_bn_t,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # shape (B,N,N,hidden)

    # to_out_norm layer norm
    out = torch.nn.functional.layer_norm(out,
                                         normalized_shape=(hidden_dim,),
                                         weight=weights["to_out_norm.weight"],
                                         bias=weights["to_out_norm.bias"],
                                         eps=eps)
    # multiply out_gate
    out = out * out_gate

    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul forward implementation (outgoing update) using Triton for the heavy N^3 reduction.

The module consists of:
1. LayerNorm on the input features.
2. Linear projections to a hidden dimension followed by optional masking.
3. Gating (sigmoid) applied to the projected tensors.
4. The core O(N³) update:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication left @ rightᵀ for each hidden slice.
   This reduction is performed by a custom Triton kernel.
5. LayerNorm on the hidden result, gated by out_gate, and a final linear projection
   back to the original feature dimension.

All operations are done in float32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_sz, M, N, K,
    stride_a_batch, stride_ai, stride_ak,
    stride_b_batch, stride_bk, stride_bn,
    stride_c_batch, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM kernel computing C = A @ Bᵀ where A,B ∈ (batch_sz, M, K)"""
    pid = tl.program_id(0)

    # number of tiles in each dimension
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    # decode 3‑D program id (batch, tile_i, tile_j)
    pid_batch = pid // (num_pid_m * num_pid_n)
    pid_rem   = pid % (num_pid_m * num_pid_n)
    pid_m = pid_rem // num_pid_n
    pid_n = pid_rem % num_pid_n

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointer to the start of the current batch
    a_batch_ptr = a_ptr + pid_batch * stride_a_batch
    b_batch_ptr = b_ptr + pid_batch * stride_b_batch
    c_batch_ptr = c_ptr + pid_batch * stride_c_batch

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = tl.minimum(BLOCK_K, K - k * BLOCK_K)

        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers for the current K‑tile
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
        b_ptrs = b_batch_ptr + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # load tiles with masking for out‑of‑bounds elements
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # write back the result
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask         : torch.Tensor or None, shape [B, N, N] (optional)
        - weights      : dict of torch.Tensor with layer‑norm and linear weights
        - config       : dict containing at least "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] after the TriMul update.
    """
    # unpack arguments
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------- 1. LayerNorm on input ----------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ---------- 2. Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])          # [B, N, N, hidden]
    right = F.linear(x, weights["right_proj.weight"])        # [B, N, N, hidden]

    # ---------- 3. Optional masking ----------
    if mask is not None and not config.get("nomask", False):
        mask_f = mask.to(x.dtype).unsqueeze(-1)             # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ---------- 4. Gating ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, hidden]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # [B, N, N, hidden]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))     # [B, N, N, hidden]

    left = left * left_gate
    right = right * right_gate

    # ---------- 5. Core N³ reduction via Triton ----------
    B, N, _, _ = left.shape

    # bring hidden dimension to the second axis and make it contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, hidden, N, N]

    # flatten batch & hidden into one leading dimension
    total_batch = B * hidden_dim
    left_flat = left_perm.reshape(total_batch, N, N)
    right_flat = right_perm.reshape(total_batch, N, N)

    # allocate output buffer
    out_flat = torch.empty_like(left_flat)

    # strides for the flattened tensors (all are contiguous in row‑major order)
    stride_a_batch = left_flat.stride(0)   # N*N
    stride_ai      = left_flat.stride(1)   # N
    stride_ak      = left_flat.stride(2)   # 1

    stride_b_batch = right_flat.stride(0)  # N*N
    # original right stride (K dimension = N, N dimension = 1)
    stride_bk_orig = right_flat.stride(1)  # N
    stride_bn_orig = right_flat.stride(2)  # 1
    # we need Bᵀ, so swap the inner strides
    stride_bk = stride_bn_orig                # 1
    stride_bn = stride_bk_orig                # N

    stride_c_batch = out_flat.stride(0)    # N*N
    stride_ci      = out_flat.stride(1)    # N
    stride_cj      = out_flat.stride(2)    # 1

    # block size (tuned for typical A200 GPUs)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # launch configuration: one program per (batch, tile_i, tile_j)
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (total_batch * num_blocks_m * num_blocks_n,)

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        total_batch,
        N,
        N,
        N,
        stride_a_batch,
        stride_ai,
        stride_ak,
        stride_b_batch,
        stride_bk,
        stride_bn,
        stride_c_batch,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, hidden]

    # ---------- 6. Post‑processing ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])  # final projection to dim

    return out
