"""
TriMul "outgoing" forward pass implementation.

Algorithm
---------
1. Layer‑norm over the last channel dimension (dim) using the provided
   `norm.weight` / `norm.bias`.
2. Linear projections (no bias) to a hidden dimension:
      left  = x @ left_proj.weight.T
      right = x @ right_proj.weight.T
3. Optional masking (if `config["nomask"]` is False) by element‑wise
   multiplication with `mask`.
4. Compute three learned gates (sigmoid of a linear projection) and
   apply them to `left`, `right` and later to the final output.
5. The heavy part: for each hidden channel `h` compute

        out[..., i, j, h] = Σ_k left[..., i, k, h] * right[..., j, k, h]

   This is a batched matrix multiplication `left_h @ right_h.T` for each
   hidden channel.  We implement this with a custom Triton kernel that
   processes the `(batch * hidden_dim)` batches in parallel using
   block‑wise GEMM (float16 inputs, float32 accumulation).
6. Layer‑norm over the hidden dimension, apply the output gate, and a
   final linear projection back to `dim`.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for batched matmul: C = A @ B.T
# A : (B, M, K)   (float16)
# B : (B, N, K)   (float16)   – we pass B already transposed (K, N)
# C : (B, M, N)   (float32)
# ----------------------------------------------------------------------
@triton.jit
def triton_batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    # matrix sizes
    M, N, K,
    # strides
    stride_ab, stride_am, stride_ak,   # A: batch, row, col
    stride_bb, stride_bk, stride_bn,   # B: batch, row(K), col(N)
    stride_cb, stride_cm, stride_cn,   # C: batch, row, col
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m   = tl.program_id(1)
    pid_n   = tl.program_id(2)

    # ------------------------------------------------------------------
    # Offsets within the tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Bounds checking
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers for this batch
    # ------------------------------------------------------------------
    a_batch_ptr = A_ptr + batch_id * stride_ab
    b_batch_ptr = B_ptr + batch_id * stride_bb
    c_batch_ptr = C_ptr + batch_id * stride_cb

    # ------------------------------------------------------------------
    # Initialize accumulator (fp32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load A block (M x K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k[None, :] < K),
                    other=0.0).to(tl.float32)

        # Load B block (K x N)  – note that B is already transposed,
        # so its leading dimension is K.
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # Accumulate
        acc += tl.dot(a, b)

        # Advance pointers to next K‑block
        a_batch_ptr += BLOCK_K * stride_ak
        b_batch_ptr += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Helper: launch the Triton kernel for a given input pair (left, right)
# ----------------------------------------------------------------------
def batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : (B, N, N, H)  float16
    returns out : (B, N, N, H)   float32
    Computes out[..., i, j, h] = sum_k left[..., i, k, h] * right[..., j, k, h]
    """
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_r = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # transpose the second operand on the last two dimensions (K,N)
    right_r = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_T = right_r.transpose(1, 2).contiguous()

    # Allocate output
    out = torch.empty_like(left_r, dtype=torch.float32)

    # Strides (in elements)
    stride_ab = left_r.stride(0)
    stride_am = left_r.stride(1)
    stride_ak = left_r.stride(2)

    stride_bb = right_T.stride(0)
    stride_bk = right_T.stride(1)
    stride_bn = right_T.stride(2)

    stride_cb = out.stride(0)
    stride_cm = out.stride(1)
    stride_cn = out.stride(2)

    # Tile sizes – tuned for typical B200 memory hierarchy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    triton_batched_matmul_kernel[grid](
        left_r,
        right_T,
        out,
        M=N,
        N=N,
        K=N,
        stride_ab=stride_ab,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bb=stride_bb,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cb=stride_cb,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # reshape back to (B, N, N, H)
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul "outgoing" operator.
    Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype
    eps = config.get("eps", 1e-5)

    # ------------------------------------------------------------------
    # Extract shapes & config
    # ------------------------------------------------------------------
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]  # called H in code

    # ------------------------------------------------------------------
    # Load weights (all are stored as torch tensors on the correct device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]  # (hidden_dim, dim)
    left_gate_weight  = weights["left_gate.weight"]   # (hidden_dim, dim)
    right_gate_weight = weights["right_gate.weight"]  # (hidden_dim, dim)
    out_gate_weight   = weights["out_gate.weight"]    # (hidden_dim, dim)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]          # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over the last channel dimension
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # shape [B, N, N, dim], float32

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    # torch.nn.functional.linear expects weight of shape (out_features, in_features)
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight) # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3) Optional mask
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask shape: [B, N, N] -> broadcast over hidden dim
        mask = mask_tensor.to(dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Compute gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Apply the gates to the projected tensors
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Heavy computation: batched matmul via Triton
    # ------------------------------------------------------------------
    # Cast to half for the kernel (speed) – keep a copy in fp32 for later
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Compute out[..., i, j, h] = Σ_k left[..., i, k, h] * right[..., j, k, h]
    out_hidden = batched_matmul_triton(left_fp16, right_fp16)  # [B,N,N,hidden_dim], float32

    # ------------------------------------------------------------------
    # 6) Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 7) Apply final output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8) Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_weight)  # [B,N,N,dim]

    return out
