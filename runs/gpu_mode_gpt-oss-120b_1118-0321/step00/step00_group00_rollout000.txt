def custom_kernel(data):
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype
    B,N,N_,dim = x.shape
    assert N==N_, "square"

    # Extract config
    hidden_dim = config["hidden_dim"]
    eps = 1e-5 # for layernorm

    # Perform LayerNorm (using torch.nn.functional)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)
    
    # Linear projection
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])  # weight shape (hidden_dim, dim)
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Mask
    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask
        right = right * mask
    else:
        # if nomask (==True in config?), ignore
        pass

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Compute core reduction via Triton
    import triton
    import triton.language as tl

    # ensure left,right contiguous
    left = left.contiguous()
    right = right.contiguous()
    # allocate output buffer hidden_dim shape
    out_hidden = torch.empty((B, N, N, hidden_dim), dtype=x.dtype, device=device)

    # Provide strides
    stride_bl, stride_il, stride_kl, stride_dl = left.stride()
    stride_br, stride_jr, stride_kr, stride_dr = right.stride()
    stride_ob, stride_io, stride_jo, stride_do = out_hidden.stride()

    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 64

    grid = (B * hidden_dim, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)

    @triton.jit
    def tri_mul_kernel(
        left_ptr, right_ptr, out_ptr,
        B, N, H,
        stride_bl, stride_il, stride_kl, stride_dl,
        stride_br, stride_jr, stride_kr, stride_dr,
        stride_ob, stride_io, stride_jo, stride_do,
        BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
    ):
        pid0 = tl.program_id(0)  # combined batch*channel
        pid1 = tl.program_id(1)  # i block
        pid2 = tl.program_id(2)  # j block

        # compute batch index b and channel d from pid0
        b = pid0 // H
        d = pid0 % H

        # compute offsets
        i_start = pid1 * BLOCK_I
        j_start = pid2 * BLOCK_J

        # allocate accumulator
        acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

        # loop over K dimension
        for k_start in range(0, N, BLOCK_K):
            # compute actual ranges
            i_range = i_start + tl.arange(0, BLOCK_I)
            j_range = j_start + tl.arange(0, BLOCK_J)
            k_range = k_start + tl.arange(0, BLOCK_K)
            # mask for bounds
            i_mask = i_range < N
            j_mask = j_range < N
            k_mask = k_range < N

            # compute pointers
            # left: (i,k) slice for fixed b,d
            left_offset = b * stride_bl + d * stride_dl + i_range[:, None] * stride_il + k_range[None, :] * stride_kl
            left_tile = tl.load(left_ptr + left_offset, mask=i_mask[:, None] & k_mask[None, :], other=0.0)

            # right: (j,k) slice for fixed b,d
            right_offset = b * stride_br + d * stride_dr + j_range[:, None] * stride_jr + k_range[None, :] * stride_kr
            right_tile = tl.load(right_ptr + right_offset, mask=j_mask[:, None] & k_mask[None, :], other=0.0)

            # compute block matmul: left_tile (I x K) * right_tile (J x K). Need transpose of right_tile to (K x J)
            # Use tl.dot
            acc += tl.dot(left_tile, right_tile.T)  # result shape (I,J)
        # store output
        # compute out offset (i,j)
        out_offset = b * stride_ob + d * stride_do + i_range[:, None] * stride_io + j_range[None, :] * stride_jo
        tl.store(out_ptr + out_offset, acc, mask=i_mask[:, None] & j_mask[None, :])
    # launch
    tri_mul_kernel[grid](
        left, right, out_hidden,
        B, N, hidden_dim,
        stride_bl, stride_il, stride_kl, stride_dl,
        stride_br, stride_jr, stride_kr, stride_dr,
        stride_ob, stride_io, stride_jo, stride_do,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K
    )
    # Continue forward passes: out_norm, out_gate, to_out linear
    out_norm = torch.nn.functional.layer_norm(out_hidden, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)
    out = out_norm * out_gate  # broadcasting over channel dimension hidden_dim? out_gate shape B,N,N,hidden_dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

"""
TriMul forward (outgoing) implementation with a Triton kernel for the
core O(N³) reduction.

Workflow
---------
1. Layer‑norm the input tensor `x` (shape B×N×N×C).
2. Linear projections → `left`/`right` (B×N×N×H) using model weights.
3. (Optional) mask the projections.
4. Compute element‑wise gates (sigmoid of linear projections) and apply them.
5. Compute the expensive contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   using a custom Triton kernel that performs a blocked matrix‑multiply:
        out_d = left_d @ right_dᵀ   for each hidden channel d.
6. Layer‑norm the hidden result, multiply by `out_gate`, and project back to
   the original dimension with a final linear layer.

Only the O(N³) contraction is performed in Triton; all other operations use
standard PyTorch for simplicity and numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

def custom_kernel(data):
    """
    Args:
        data: tuple (x, mask, weights, config)
            x       – torch.Tensor of shape [B, N, N, C] (float32)
            mask    – torch.Tensor of shape [B, N, N] (bool/float) or None
            weights – dict with all model parameters (torch.Tensor)
            config  – dict containing at least
                      "dim" (C), "hidden_dim" (H), and optionally "nomask".
    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------ #
    # Unpack inputs
    # ------------------------------------------------------------------ #
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    B, N, N2, dim = x.shape
    assert N == N2, "input must be square on the two sequence axes"
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------ #
    # 1) Layer‑norm on the input
    # ------------------------------------------------------------------ #
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------ #
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------ #
    left = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x_norm, weights["right_proj.weight"]) # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------ #
    # 3) Optional mask (mask is broadcast over the hidden dimension)
    # ------------------------------------------------------------------ #
    if not config.get("nomask", False) and mask is not None:
        # Cast mask to the same dtype and add a trailing dim for broadcasting
        mask_f = mask.to(dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------ #
    # 4) Compute gates and apply them
    # ------------------------------------------------------------------ #
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"])) # [B,N,N,hidden_dim]
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))     # [B,N,N,hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ #
    # 5) Core O(N³) contraction via Triton
    # ------------------------------------------------------------------ #
    # Ensure contiguous tensors for deterministic stride handling
    left = left.contiguous()
    right = right.contiguous()
    # Allocate output in hidden space
    out_hidden = torch.empty((B, N, N, hidden_dim), dtype=dtype, device=device)

    # Strides (in elements, not bytes)
    stride_bl, stride_il, stride_kl, stride_dl = left.stride()   # B, i, k, d
    stride_br, stride_jr, stride_kr, stride_dr = right.stride()  # B, j, k, d
    stride_ob, stride_io, stride_jo, stride_do = out_hidden.stride()  # B, i, j, d

    # Tunable tile sizes – 64 works well for the target hardware
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 64

    # Grid: (batch*channel, i‑tiles, j‑tiles)
    grid = (
        B * hidden_dim,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )

    @triton.jit
    def tri_mul_kernel(
        left_ptr, right_ptr, out_ptr,
        B, N, H,
        stride_bl, stride_il, stride_kl, stride_dl,
        stride_br, stride_jr, stride_kr, stride_dr,
        stride_ob, stride_io, stride_jo, stride_do,
        BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
    ):
        # ------------------------------------------------------------------ #
        # Program IDs
        # ------------------------------------------------------------------ #
        pid0 = tl.program_id(0)  # combined batch * hidden channel
        pid1 = tl.program_id(1)  # i‑tile index
        pid2 = tl.program_id(2)  # j‑tile index

        # Decode batch and hidden channel
        b = pid0 // H
        d = pid0 % H

        i0 = pid1 * BLOCK_I
        j0 = pid2 * BLOCK_J

        # ------------------------------------------------------------------ #
        # Accumulator (float32 for numerical safety)
        # ------------------------------------------------------------------ #
        acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

        # ------------------------------------------------------------------ #
        # Loop over the reduction dimension k
        # ------------------------------------------------------------------ #
        for k0 in range(0, N, BLOCK_K):
            # Ranges
            i_range = i0 + tl.arange(0, BLOCK_I)
            j_range = j0 + tl.arange(0, BLOCK_J)
            k_range = k0 + tl.arange(0, BLOCK_K)

            # Masks for out‑of‑bounds
            i_mask = i_range < N
            j_mask = j_range < N
            k_mask = k_range < N

            # Load left tile: shape (BLOCK_I, BLOCK_K)
            left_offset = (
                b * stride_bl
                + d * stride_dl
                + i_range[:, None] * stride_il
                + k_range[None, :] * stride_kl
            )
            left_tile = tl.load(
                left_ptr + left_offset,
                mask=i_mask[:, None] & k_mask[None, :],
                other=0.0,
            )

            # Load right tile: shape (BLOCK_J, BLOCK_K)
            right_offset = (
                b * stride_br
                + d * stride_dr
                + j_range[:, None] * stride_jr
                + k_range[None, :] * stride_kr
            )
            right_tile = tl.load(
                right_ptr + right_offset,
                mask=j_mask[:, None] & k_mask[None, :],
                other=0.0,
            )

            # Blocked matrix multiplication: (I x K) @ (K x J)
            acc += tl.dot(left_tile, right_tile.T, out_dtype=tl.float32)

        # ------------------------------------------------------------------ #
        # Store the result
        # ------------------------------------------------------------------ #
        out_offset = (
            b * stride_ob
            + d * stride_do
            + i_range[:, None] * stride_io
            + j_range[None, :] * stride_jo
        )
        tl.store(
            out_ptr + out_offset,
            acc,
            mask=i_mask[:, None] & j_mask[None, :],
        )

    # Launch kernel
    tri_mul_kernel[grid](
        left,
        right,
        out_hidden,
        B,
        N,
        hidden_dim,
        stride_bl,
        stride_il,
        stride_kl,
        stride_dl,
        stride_br,
        stride_jr,
        stride_kr,
        stride_dr,
        stride_ob,
        stride_io,
        stride_jo,
        stride_do,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------ #
    # 6) Post‑processing: hidden‑space LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------ #
    out_hidden_norm = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # Apply the output gate (broadcast over the hidden dimension)
    out_gated = out_hidden_norm * out_gate

    # Final linear back to original dimension
    out = F.linear(out_gated, weights["to_out.weight"])

    return out
