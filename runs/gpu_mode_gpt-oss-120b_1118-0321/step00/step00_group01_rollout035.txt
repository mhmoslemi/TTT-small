    hidden_dim = config["hidden_dim"]
    # load weights: torch tensors
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    # layernorm on input
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)
    # compute projections and gates
    # convert to float16 for memory and speed
    left_proj = torch.nn.functional.linear(x_norm, left_proj_weight).to(torch.float16)
    right_proj = torch.nn.functional.linear(x_norm, right_proj_weight).to(torch.float16)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight)).to(torch.float16)
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight)).to(torch.float16)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight)).to(torch.float16)
    left = left_proj * left_gate
    right = right_proj * right_gate
    # mask
    if mask is not None:
        mask = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask
        right = right * mask
    # Permute for batched matmul
    B, N, _, H = left.shape
    left_reshape = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_reshape = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)
    # Allocate output
    out_reshape = torch.empty((B * H, N, N), dtype=torch.float32, device=input_tensor.device)
    # define block sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    # compute grid
    grid = (
        (B * H),
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    # launch kernel
    matmul_batch_kernel[grid](
        left_reshape,
        right_reshape,
        out_reshape,
        N,
        N,

        left_reshape.stride(2),   # stride_ak: col stride

        right_reshape.stride(2),  # stride_bn (col stride, corresponds to n)

        right_reshape.stride(0),  # batch stride b
        out_reshape.stride(0),    # batch stride c
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape output back to [B,N,N,H]
    out = out_reshape.view(B, H, N, N).permute(0, 2, 3, 1)  # shape [B,N,N,H] float32

import triton.language as tl

@triton.jit
def matmul_batch_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_ab, stride_bb, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets within batch
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch

    b_batch_ptr = b_ptr + pid_batch * stride_bb
    c_batch_ptr = c_ptr + pid_batch * stride_cb

"""
TriMul (outgoing) forward pass implemented with a custom Triton GEMM kernel.

The core operation is:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

where
    left  = (x @ W_left)   * sigmoid(x @ W_left_gate)   * mask
    right = (x @ W_right)  * sigmoid(x @ W_right_gate)  * mask
    x     = LayerNorm(x, norm_w, norm_b)

All linear layers have no bias. The heavy N³ reduction over the intermediate
dimension `k` is performed by a batched matrix‑multiply kernel written in
Triton. The kernel treats the combined batch‑and‑hidden dimension
`B * H` as a batch index and computes a standard GEMM `A @ B` for each
slice, which is mathematically equivalent to the einsum above.

After the GEMM we apply a second LayerNorm, an output gate, and a final
linear projection back to the original feature dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------
# Triton batched GEMM kernel (A @ B) used for the TriMul einsum
# ------------------------------------------------------------
@triton.jit
def _triton_matmul_batch_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix sizes (all equal to seq_len)
    stride_am, stride_ak,         # a: (batch, M, K) strides
    stride_bk, stride_bn,         # b: (batch, K, N) strides
    stride_cm, stride_cn,         # c: (batch, M, N) strides
    stride_ab, stride_bb, stride_cb,  # batch strides for a, b, c
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ B for each batch element (including hidden dim)."""
    pid_batch = tl.program_id(0)          # combined batch * hidden index
    pid_m = tl.program_id(1)              # block row
    pid_n = tl.program_id(2)              # block column

    # --- Offsets within the current block ---------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # --- Base pointers for this batch --------------------------------------
    a_batch = a_ptr + pid_batch * stride_ab
    b_batch = b_ptr + pid_batch * stride_bb
    c_batch = c_ptr + pid_batch * stride_cb

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --- Loop over K dimension --------------------------------------------
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers for the current K‑tile
        a_tile_ptr = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_tile_ptr = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # load with masking (values outside the matrix are taken as 0)
        a = tl.load(a_tile_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)          # dtype = float16 (from the tensor)
        b = tl.load(b_tile_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)          # dtype = float16

        # dot product (float16 x float16 -> float16), then promote to float32
        acc += tl.dot(a, b).to(tl.float32)

    # --- Write back ---------------------------------------------------------
    c_tile_ptr = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_tile_ptr,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

# ------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Arguments:
        data = (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor [B, N, N, dim] (float32)
            mask         : torch.Tensor [B, N, N]  (bool/float, optional)
            weights      : dict of torch.Tensors containing all model weights
            config       : dict with keys "dim" and "hidden_dim"

    Returns:
        out : torch.Tensor [B, N, N, dim] (float32)
    """
    # unpack inputs ---------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # -------------------------------------------------------------------------
    # 1. LayerNorm on the input (no need for a custom kernel here)
    # -------------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_w,
                     bias=norm_b,
                     eps=1e-5)               # [B, N, N, dim]  float32

    # -------------------------------------------------------------------------
    # 2. Linear projections and gates (use fp16 for speed & memory)
    # -------------------------------------------------------------------------
    # Helper to cast weight to correct dtype (float16)
    def linear_fp16(inp, w):
        return F.linear(inp, w).to(torch.float16)

    left_proj = linear_fp16(x, weights["left_proj.weight"])          # [B, N, N, hidden_dim]
    right_proj = linear_fp16(x, weights["right_proj.weight"])

    left_gate = torch.sigmoid(linear_fp16(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear_fp16(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(linear_fp16(x, weights["out_gate.weight"]))

    left = left_proj * left_gate
    right = right_proj * right_gate

    # -------------------------------------------------------------------------
    # 3. Optional mask (broadcasted over the hidden dimension)
    # -------------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_b = mask.unsqueeze(-1).to(left.dtype)   # [B, N, N, 1]
        left = left * mask_b
        right = right * mask_b

    # -------------------------------------------------------------------------
    # 4. Prepare tensors for the batched GEMM (A @ B)
    # -------------------------------------------------------------------------
    B, N, _, H = left.shape               # H == hidden_dim
    # shape to (B*H, N, N) where:
    #   A  -> left  [i, k] per hidden slice
    #   B  -> right [k, j] per hidden slice
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (batch*hidden, M, K)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N) # (batch*hidden, K, N)

    # output buffer (float32 for stable accumulation)
    out_mat = torch.empty((B * H, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------
    # 5. Launch the Triton kernel
    # ------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,                                 # batch * hidden dimension
        (N + BLOCK_M - 1) // BLOCK_M,          # number of row tiles
        (N + BLOCK_N - 1) // BLOCK_N,          # number of column tiles
    )

    _triton_matmul_batch_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,                              # M, N, K
        left_mat.stride(1),  # stride_am (row stride)
        left_mat.stride(2),  # stride_ak (col stride)
        right_mat.stride(1), # stride_bk (row stride, i.e. k)
        right_mat.stride(2), # stride_bn (col stride, i.e. n)
        out_mat.stride(1),   # stride_cm
        out_mat.stride(2),   # stride_cn
        left_mat.stride(0),  # batch stride A
        right_mat.stride(0), # batch stride B
        out_mat.stride(0),   # batch stride C
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -------------------------------------------------------------------------
    # 6. Reshape back to [B, N, N, hidden_dim] (float32)
    # -------------------------------------------------------------------------
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)   # [B, N, N, hidden_dim]

    # -------------------------------------------------------------------------
    # 7. Second LayerNorm, output gate, and final linear projection
    # -------------------------------------------------------------------------
    out_norm = F.layer_norm(out,
                            normalized_shape=(hidden_dim,),
                            weight=weights["to_out_norm.weight"],
                            bias=weights["to_out_norm.bias"],
                            eps=1e-5)               # still float32

    out = out_norm * out_gate.to(out_norm.dtype)          # gating
    out = F.linear(out, weights["to_out.weight"])        # final projection to dim

    return out
```
