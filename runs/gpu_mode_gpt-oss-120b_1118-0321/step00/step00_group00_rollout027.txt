def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,   # M = #matrices, N=output dimension (rows), K=dim? Actually both N and K are same (seq_len)
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Load within bounds
    # Use masks for boundary
    # Compute accumulation
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # pointer to block of A: offsets (batch_id, offs_m, k_offset)
        a_ptrs = A_ptr + batch_id * stride_a_batch + offs_m[:, None] * stride_a_row + k_offset[None, :] * stride_a_col

@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_az, stride_ai, stride_ak,
    stride_bz, stride_bi, stride_bk,
    stride_cz, stride_ci, stride_ck,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # M = number of matrices (batch*hidden)
    # N = output rows (i)
    # K = input columns (k) = output cols (j) (since B transposed)
    # pointer arithmetic.
    # program IDs:
    pid_z = tl.program_id(0) # matrix index
    pid_m = tl.program_id(1) # row block
    pid_n = tl.program_id(2) # col block

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Determine mask for out-of-bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # Compute pointers for A block: (pid_z, offs_m, k_offset)
        a_ptrs = A_ptr + pid_z * stride_az + offs_m[:, None] * stride_ai + k_offset[None, :] * stride_ak
        b_ptrs = B_ptr + pid_z * stride_bz + k_offset[:, None] * stride_bi + offs_n[None, :] * stride_bk

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)  # works for float32 accum

    # Store result
    c_ptrs = C_ptr + pid_z * stride_cz + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_ck
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # dims
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # cast dims maybe int
    device = input_tensor.device
    dtype = input_tensor.dtype

    # If mask not provided or None, create mask of ones
    if mask is None:
        mask = torch.ones(bs, N, N, dtype=dtype, device=device)
    else:
        # ensure dtype match
        mask = mask.to(dtype)

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x_norm, left_proj_weight)
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask
    mask_unsq = mask.unsqueeze(-1)
    left = left * mask_unsq
    right = right * mask_unsq

    # Gating
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()
    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 for GEMM
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Transpose right for GEMM
    right_T = right.permute(0, 2, 1, 3).contiguous()

    # Permute left and right_T to bring hidden dim as 2nd dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right_T.permute(0, 3, 1, 2).contiguous()

    # Flatten batch*hidden
    M = bs * hidden_dim
    left_flat = left_perm.view(M, N, N)
    right_flat = right_perm.view(M, N, N)

    # Prepare output container
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)  # allocate FP32

    # Strides
    stride_a_batch, stride_a_row, stride_a_col = left_flat.stride()
    stride_b_batch, stride_b_row, stride_b_col = right_flat.stride()
    stride_c_batch, stride_c_row, stride_c_col = out_flat.stride()

    # Determine block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Compute grid
    grid = (M,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    # Launch Triton kernel
    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M,
        N,
        N,
        stride_a_batch,
        stride_a_row,
        stride_a_col,
        stride_b_batch,
        stride_b_row,
        stride_b_col,
        stride_c_batch,
        stride_c_row,
        stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (bs, N, N, hidden_dim)
    out = out_flat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # shape (bs,N,N,hidden_dim)

    # Output LayerNorm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Apply out_gate
    out_gated = out_norm * out_gate

    # Final linear to dim
    to_out_weight = weights["to_out.weight"]
    out_final = torch.nn.functional.linear(out_gated, to_out_weight)

    return out_final

"""
TriMul “outgoing” forward pass implemented with Triton.

Algorithm
---------
1. Layer‑norm the input tensor (shape [B, N, N, C]).
2. Project the normalized tensor into a hidden space (size H) with two
   linear layers: left_proj and right_proj.
3. Apply an optional binary mask (broadcast on the hidden dimension).
4. Compute three sigmoid gates (left, right, out) from the normalized input
   and apply the left/right gates to the projected tensors.
5. Compute the core “triangle” operation:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication where, for each hidden
   channel h, we need left (i × k) @ rightᵀ (k × j).
   The multiplication is performed by a custom Triton kernel that loops
   over the K dimension in tiles.
6. Layer‑norm the result of step 5, multiply by the out‑gate and finally
   project back to the original channel size with a linear layer.
All heavy work (the N³ reduction) is executed on the GPU via Triton,
while surrounding linear layers, gates and normalisations are handled
by PyTorch.

The kernel works for any batch size (≤2), sequence length N up to 1024,
and hidden dimension H = 128. It uses float32 throughout for numerical
accuracy.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_az, stride_ai, stride_ak,
    stride_bz, stride_bi, stride_bk,
    stride_cz, stride_ci, stride_ck,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Performs many independent GEMMs:
        C[m] = A[m] @ B[m].T      (A/B shape: N × K, B stored as K × N)
    where m ∈ [0, M). Each GEMM computes an N×N matrix.
    """
    pid_z = tl.program_id(0)      # which independent matrix (batch * hidden)
    pid_m = tl.program_id(1)      # tile row
    pid_n = tl.program_id(2)      # tile column

    # off‑sets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension (shared across A and B)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers to the current tile of A (shape BLOCK_M × BLOCK_K)
        a_ptrs = (A_ptr
                  + pid_z * stride_az
                  + offs_m[:, None] * stride_ai
                  + offs_k[None, :] * stride_ak)
        # pointers to the current tile of B (shape BLOCK_K × BLOCK_N)
        # B is stored already as (K × N) because we permuted it beforehand.
        b_ptrs = (B_ptr
                  + pid_z * stride_bz
                  + offs_k[:, None] * stride_bi
                  + offs_n[None, :] * stride_bk)

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0, dtype=tl.float32)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0, dtype=tl.float32)

        acc += tl.dot(a, b)      # (BLOCK_M, BLOCK_N)

    # store the result
    c_ptrs = (C_ptr
              + pid_z * stride_cz
              + offs_m[:, None] * stride_ci
              + offs_n[None, :] * stride_ck)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Executes the TriMul forward pass (outgoing variant) using a Triton kernel.

    Args:
        data: tuple
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of torch.Tensors containing all model weights
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1. LayerNorm over the channel dimension
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # --------------------------------------------------------------
    # 2. Linear projections
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # (B,N,N,hidden_dim)
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask (broadcasted on hidden dimension)
    # --------------------------------------------------------------
    if mask is not None:
        mask = mask.to(dtype).unsqueeze(-1)                 # (B,N,N,1)
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Gating
    # --------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Batched GEMM performed by Triton
    # --------------------------------------------------------------
    #   left  : (B, N, N, H) -> rows=i, cols=k
    #   right : (B, N, N, H) -> rows=j, cols=k
    #   We need left @ rightᵀ, i.e. sum_k left[i,k] * right[j,k].
    #   After permuting, both operands become (B, H, N, N) with
    #   right already transposed (k as row, j as column).
    #
    #   Shape after permutation: (B, H, N, N)
    #   Flatten batch & hidden: (M, N, N) where M = B * H.
    # --------------------------------------------------------------

    # move hidden dimension to the 2nd axis and apply the required transpose
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)  rows=i, cols=k
    right_perm = right.permute(0, 3, 2, 1).contiguous()        # (B, H, N, N)  rows=k, cols=j

    M = bs * hidden_dim
    left_flat = left_perm.view(M, N, N)
    right_flat = right_perm.view(M, N, N)

    # output buffer (float32)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # strides for the kernel
    stride_az, stride_ai, stride_ak = left_flat.stride()
    stride_bz, stride_bi, stride_bk = right_flat.stride()
    stride_cz, stride_ci, stride_ck = out_flat.stride()

    # tile sizes – 64 works well for the tested dimensions
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        M,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M,
        N,
        N,
        stride_az,
        stride_ai,
        stride_ak,
        stride_bz,
        stride_bi,
        stride_bk,
        stride_cz,
        stride_ci,
        stride_ck,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output normalisation, out‑gate and final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
