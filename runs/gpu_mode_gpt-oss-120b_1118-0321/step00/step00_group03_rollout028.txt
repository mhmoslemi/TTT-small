def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    # extract dimensions
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # Device
    device = input_tensor.device
    dtype = input_tensor.dtype

    # LayerNorm parameters
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Linear weight tensors
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Normalize input
    x = torch.nn.functional.layer_norm(input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias)

    # Projections
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Mask handling
    if mask_tensor is None or config.get("nomask", True):
        # use ones mask
        mask = torch.ones(batch_size, seq_len, seq_len, dtype=x.dtype, device=device)
    else:
        mask = mask_tensor
    # Ensure mask dtype matches x
    mask = mask.to(x.dtype)
    mask = mask.unsqueeze(-1)  # shape [B,N,N,1]
    left = left * mask
    right = right * mask

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Prepare for gemm
    # Permute to [B, hidden, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # transpose right on the last two dims
    right_T = right_perm.transpose(-2, -1).contiguous()

    B, D, N, _ = left_perm.shape
    # reshape to [batch*hidden, N, N]
    left_flat = left_perm.view(B * D, N, N).contiguous()
    right_flat = right_T.view(B * D, N, N).contiguous()

    # Cast to half for speed
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)

    # Output accumulator in float32
    out_flat = torch.empty((B * D, N, N), dtype=torch.float32, device=device)

    # Determine block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32  # might be 32 

    # Compute strides
    stride_am = left_fp16.stride(1)
    stride_ak = left_fp16.stride(2)
    stride_ab = left_fp16.stride(0)
    stride_bk = right_fp16.stride(1)
    stride_bn = right_fp16.stride(2)
    stride_bb = right_fp16.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)
    stride_cb = out_flat.stride(0)

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * D)
    batched_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_flat,
        N,
        N,
        N,
        stride_am,
        stride_ak,
        stride_ab,
        stride_bk,
        stride_bn,
        stride_bb,
        stride_cm,
        stride_cn,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back
    out = out_flat.view(B, D, N, N).permute(0, 2, 3, 1).contiguous()  # shape [B,N,N,hidden]

    # Final layernorm and gating
    out = torch.nn.functional.layer_norm(out, normalized_shape=(hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

"""
TriMul outgoing kernel (AlphaFold3)
----------------------------------

This implementation mirrors the PyTorch TriMul module but off‑loads the
O(N³·C_hidden) “einsum” (left @ rightᵀ) to a custom Triton kernel.
The heavy computation is expressed as a batched GEMM where each batch
corresponds to one hidden channel of one sample.  The kernel uses
tiling (BLOCK_M × BLOCK_N) with a K‑tile of size BLOCK_K and accumulates
in float32 while operating on fp16 inputs for higher throughput.

The surrounding operations (LayerNorm, linear projections, gating,
masking, final LayerNorm and output projection) are performed with
standard PyTorch ops.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ B   (A: [batch, M, K], B: [batch, K, N])
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    M, N, K,                            # matrix sizes
    stride_am, stride_ak, stride_ab,    # A strides (batch, M, K)
    stride_bk, stride_bn, stride_bb,    # B strides (batch, K, N)
    stride_cm, stride_cn, stride_cb,    # C strides (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication with fp16 inputs and fp32 accumulation."""
    pid_b = tl.program_id(2)  # batch id (merged batch*hidden)
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block column

    # Offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds handling
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + pid_b * stride_ab
    b_batch = b_ptr + pid_b * stride_bb
    c_batch = c_ptr + pid_b * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k):
        cur_k = k * BLOCK_K
        # Pointers to the current K‑tile
        a_ptrs = a_batch + offs_m[:, None] * stride_am + (cur_k + offs_k)[None, :] * stride_ak
        b_ptrs = b_batch + (cur_k + offs_k)[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # K‑dimension masks
        valid_k = (cur_k + offs_k) < K
        mask_a = mask_m[:, None] & valid_k[None, :]
        mask_b = valid_k[:, None] & mask_n[None, :]

        # Load tiles (fp16); out‑of‑range elements are zeroed
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # Accumulate in fp32
        acc += tl.dot(a, b).to(tl.float32)

    # Write results
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=mask_c)


# ----------------------------------------------------------------------
# Entry point expected by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.
    Arguments:
        data : tuple (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [batch, seq_len, seq_len, dim] (float32).
    """
    input_tensor, mask_tensor, weights, config = data

    # ------------------------------------------------------------------
    # Extract dimensions and configuration
    # ------------------------------------------------------------------
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # Load model parameters (all are weight‑only Linear modules)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # LayerNorm over the feature dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # Apply mask (if present)
    # ------------------------------------------------------------------
    if mask_tensor is None or config.get("nomask", True):
        mask = torch.ones(B, N, N, dtype=dtype, device=device)
    else:
        mask = mask_tensor.to(dtype)
    mask = mask.unsqueeze(-1)               # [B, N, N, 1]
    left = left * mask
    right = right * mask

    # ------------------------------------------------------------------
    # Gating mechanisms
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for the GEMM kernel
    #   left:  [B, N, N, hidden] -> [B, hidden, N, N]
    #   right: same then transpose on the last two axes (k <-> j)
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()        # [B, H, N, N]
    right_T = right_perm.transpose(-2, -1).contiguous()      # [B, H, N, N] (k ↔ j)

    # Merge batch and channel dimensions for a single GEMM launch
    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N).contiguous()
    right_flat = right_T.view(BH, N, N).contiguous()

    # Cast to half‑precision for the kernel (inputs) and allocate fp32 output
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)
    out_fp32 = torch.empty((BH, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # Kernel launch parameters
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    stride_am = left_fp16.stride(1)   # stride for rows (M dimension)
    stride_ak = left_fp16.stride(2)   # stride for columns (K dimension)
    stride_ab = left_fp16.stride(0)   # batch stride

    stride_bk = right_fp16.stride(1)  # stride for rows (K dimension)
    stride_bn = right_fp16.stride(2)  # stride for columns (N dimension)
    stride_bb = right_fp16.stride(0)  # batch stride

    stride_cm = out_fp32.stride(1)    # stride for rows (M dimension)
    stride_cn = out_fp32.stride(2)    # stride for columns (N dimension)
    stride_cb = out_fp32.stride(0)    # batch stride

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        BH,
    )

    batched_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        N,               # M
        N,               # N
        N,               # K
        stride_am,
        stride_ak,
        stride_ab,
        stride_bk,
        stride_bn,
        stride_bb,
        stride_cm,
        stride_cn,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, hidden]
    # ------------------------------------------------------------------
    out = out_fp32.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Final normalization, output gating and projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
