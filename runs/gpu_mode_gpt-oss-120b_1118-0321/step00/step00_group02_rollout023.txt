@triton.jit
def apply_mask_gate_kernel(
    left_ptr, right_ptr,
    mask_ptr,
    left_gate_ptr, right_gate_ptr,
    out_left_ptr, out_right_ptr,
    B, N, H,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    mask_offs = offs // H
    valid = offs < total
    # Load values
    left_val = tl.load(left_ptr + offs, mask=valid, other=0.0)
    right_val = tl.load(right_ptr + offs, mask=valid, other=0.0)
    mask_val = tl.load(mask_ptr + mask_offs, mask=valid, other=0.0) # broadcast

    right_gate_val = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    out_left = left_val * mask_val * left_gate_val
    out_right = right_val * mask_val * right_gate_val

    tl.store(out_left_ptr + offs, out_left, mask=valid)
    tl.store(out_right_ptr + offs, out_right, mask=valid)

    device = input_tensor.device
    dtype = input_tensor.dtype
    # dims
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B,N,N2,dim_check = input_tensor.shape
    assert N == N2
    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    norm_weight = norm_weight.to(dtype)
    norm_bias = norm_bias.to(dtype)

    right_gate = torch.nn.functional.linear(x16, right_gate_weight).sigmoid()

"""
TriMul (outgoing) forward implementation.

The kernel fuses three element‑wise steps:
    1) Apply the optional pairwise mask,
    2) Apply the left & right gating sigmoids,
    3) Write the gated, masked tensors back in‑place.

The heavy O(N³·C) contraction is carried out by a batched GEMM using
torch.bmm (one GEMM for each hidden channel) after the masking/gating
fusion.  Intermediate calculations are performed in float16 to keep the
memory‑footprint low; the final result is returned as float32.
"""
import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: in‑place mask + gating
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    left_ptr,            # *float16   (B,N,N,hidden)
    right_ptr,           # *float16   (B,N,N,hidden)
    mask_ptr,            # *float16   (B,N,N)   broadcast over hidden
    left_gate_ptr,       # *float16   (B,N,N,hidden)
    right_gate_ptr,      # *float16   (B,N,N,hidden)
    B, N, H,             # int64 scalars
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse mask and gate multiplication into the projected tensors."""
    offs = tl.program_id(0) * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    active = offs < total

    # index of the corresponding mask element (mask has no hidden dim)
    mask_offs = offs // H

    # load
    left  = tl.load(left_ptr + offs,        mask=active, other=0.0)
    right = tl.load(right_ptr + offs,       mask=active, other=0.0)
    mask  = tl.load(mask_ptr + mask_offs,  mask=active, other=0.0)
    lgate = tl.load(left_gate_ptr + offs,  mask=active, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    # fused multiplication
    left  = left  * mask * lgate
    right = right * mask * rgate

    # store in‑place
    tl.store(left_ptr + offs,  left,  mask=active)
    tl.store(right_ptr + offs, right, mask=active)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of torch.Tensor with keys matching the reference module
            - config: dict containing "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    eps = 1e-12
    norm_w = weights["norm.weight"].to(input_tensor.dtype).to(device)
    norm_b = weights["norm.bias"].to(input_tensor.dtype).to(device)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # Cast to float16 for the heavy part
    # ------------------------------------------------------------------
    compute_dtype = torch.float16
    x16 = x.to(compute_dtype)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(compute_dtype).to(device)
    right_proj_w = weights["right_proj.weight"].to(compute_dtype).to(device)

    left = torch.nn.functional.linear(x16, left_proj_w)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x16, right_proj_w)

    # ------------------------------------------------------------------
    # Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(compute_dtype).to(device)
    right_gate_w = weights["right_gate.weight"].to(compute_dtype).to(device)
    out_gate_w = weights["out_gate.weight"].to(compute_dtype).to(device)

    left_gate = torch.nn.functional.linear(x16, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x16, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x16, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # Prepare mask (broadcast already handled in the kernel)
    # ------------------------------------------------------------------
    if mask is None or nomask:
        mask_tensor = torch.ones((B, N, N), dtype=compute_dtype, device=device)
    else:
        mask_tensor = mask.to(compute_dtype).to(device)

    # ------------------------------------------------------------------
    # Fuse mask & gates with a Triton kernel (in‑place on left/right)
    # ------------------------------------------------------------------
    total_elements = left.numel()                 # B * N * N * hidden
    BLOCK_SIZE = 16384                            # reasonable for B200
    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

    _mask_gate_kernel[grid](
        left,
        right,
        mask_tensor,
        left_gate,
        right_gate,
        B, N, hidden_dim,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    # left & right now contain the masked & gated values

    # ------------------------------------------------------------------
    # Batched GEMM: for each hidden channel compute L @ Rᵀ
    # ------------------------------------------------------------------
    # reshape to (B*hidden, N, N) for torch.bmm
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # GEMM in half precision (fast on H100/B200)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, hidden]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # Post‑processing: LayerNorm, out‑gate, final linear
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(compute_dtype).to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(compute_dtype).to(device)

    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )

    # apply the output gate (broadcast over hidden dim)
    out = out * out_gate

    # final linear back to original dimensionality
    to_out_w = weights["to_out.weight"].to(compute_dtype).to(device)
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim]

    # Cast back to float32 for the caller
    return out.to(torch.float32)
