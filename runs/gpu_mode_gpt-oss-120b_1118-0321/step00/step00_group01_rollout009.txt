def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; eps = config.get("layernorm_eps", 1e-6)
    # We'll use torch for LN and linear:
    # Input LN:
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
    input_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Projections:
    left_proj = torch.nn.functional.linear(input_norm, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(input_norm, weights["right_proj.weight"])

    # mask
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left_proj.dtype)
        left_proj = left_proj * mask_
        right_proj = right_proj * mask_
    else:
        mask_ = None

    # gates
    left_gate = torch.nn.functional.linear(input_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(input_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(input_norm, weights["out_gate.weight"]).sigmoid()

    left = left_proj * left_gate
    right = right_proj * right_gate

    # Core TriMul via Triton:
    out_core = triton_tri_mul(left, right)

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(out_core, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Apply out_gate
    out_norm = out_norm * out_gate

    # Final linear projection
    out = torch.nn.functional.linear(out_norm, weights["to_out.weight"])
    return out

import torch
import triton
import triton.language as tl
import math

def triton_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
    where left,right have shape [B,N,N,H].
    This is the core of the Triangle Multiplicative Update (outgoing mode).
    """
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # Decode batch and head (hidden) index
    b = pid_bh // H
    h = pid_bh % H

    # Coordinates of the tile we are responsible for
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Bounds
    mask_i = i < N
    mask_j = j < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension in blocks
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(num_k_blocks):
        k_base = k * BLOCK_K
        k_idx = k_base + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N

        # Pointers for left[b,i,k,h]
        left_ptrs = left_ptr + \
            b * stride_left_b + h * stride_left_h + \
            i[:, None] * stride_left_i + k_idx[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        left = left.to(tl.float32)

        # Pointers for right[b,j,k,h]
        right_ptrs = right_ptr + \
            b * stride_right_b + h * stride_right_h + \
            j[:, None] * stride_right_j + k_idx[None, :] * stride_right_k
        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)
        right = right.to(tl.float32)

        # Compute dot product over k: left (M,K) * right.T (K,N)
        acc += tl.dot(left, tl.trans(right))

    # Store result
    out_ptrs = out_ptr + \
        b * stride_out_b + h * stride_out_h + \
        i[:, None] * stride_out_i + j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def triton_tri_mul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Wrapper that launches `triton_tri_mul_kernel` to compute the outgoing
    triangle multiplicative update.
    left,right: shape [B, N, N, H] (float32, contiguous, on CUDA)
    Returns tensor of same shape and dtype.
    """
    assert left.is_cuda and right.is_cuda
    B, N, N2, H = left.shape
    assert N == N2
    # Ensure contiguous layout
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)

    # Strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = ( (N + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N,
             B * H )

    triton_tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4
    )
    return out

def custom_kernel(data):
    """
    Implements the outgoing Triangle Multiplicative Update (TriMul) forward
    pass using a custom Triton kernel for the core N^3 einsum.
    Expects `data` = (input_tensor, mask, weights, config) where:
      - input_tensor : [B, N, N, dim] (float32)
      - mask         : [B, N, N] (float32 or bool) or None
      - weights      : dict with all Linear/LayerNorm parameters
      - config       : dict containing at least "dim" and "hidden_dim"
    Returns:
      - output tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-6)

    # ----- LayerNorm on input (last dimension = dim) -----
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps
    )

    # ----- Linear projections (no bias) -----
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ----- Optional mask -----
    if mask is not None:
        mask_f = mask.unsqueeze(-1).to(left.dtype)   # shape [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ----- Gating signals -----
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ----- Core TriMul via Triton -----
    out_core = triton_tri_mul(left, right)  # shape [B,N,N,hidden_dim]

    # ----- Output LayerNorm (over hidden_dim) -----
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(
        out_core,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps
    )

    # Apply output gate
    out_norm = out_norm * out_gate

    # ----- Final linear projection back to dim -----
    out = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return out

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-6)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the channel dimension `dim`
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Masking (if provided) – applied before gating
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating signals (sigmoid of linear transforms of the *normalized* input)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply element‑wise gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul: sum_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented with a custom Triton kernel for speed.
    # ------------------------------------------------------------------
    out_core = triton_tri_mul(left, right)                     # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 6. Output LayerNorm over hidden_dim
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(
        out_core,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 7. Apply output gate
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 8. Final projection back to `dim`
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return out
```
