@triton.jit
def tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                   B, N, H,
                   stride_la_batch, stride_la_i, stride_la_k, stride_la_h,
                   stride_ra_batch, stride_ra_i, stride_ra_k, stride_ra_h,
                   stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    bh_idx = tl.program_id(0)
    block_i = tl.program_id(1)
    block_j = tl.program_id(2)
    # Compute b and h indexes
    b = bh_idx // H
    h = bh_idx % H

    # compute offsets
    i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out-of-bound
    i_mask = i < N
    j_mask = j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # load left tile
        left_ptrs = left_ptr + b * stride_la_batch + i[:, None] * stride_la_i + k_offset[None, :] * stride_la_k + h * stride_la_h
        left = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0).to(tl.float32)

        # load right_T tile (right transposed)
        # we need (k, j) indices corresponding to right_T[ k, j ] = right[ j, k ]
        right_ptrs = right_ptr + b * stride_ra_batch + k_offset[:, None] * stride_ra_i + j[None, :] * stride_ra_k + h * stride_ra_h
        # Explanation: stride_ra_i is original stride over i (rows), stride_ra_k is original stride over k (cols)
        right = tl.load(right_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=0.0).to(tl.float32)

        # dot
        acc += tl.dot(left, right)

    # store
    out_ptrs = out_ptr + b * stride_out_batch + i[:, None] * stride_out_i + j[None, :] * stride_out_j + h * stride_out_h
    tl.store(out_ptrs, acc.to(tl.float16), mask=i_mask[:, None] & j_mask[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # step 1: layernorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    # step 2: linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)
    right = torch.nn.functional.linear(x, right_proj_weight)
    # mask
    if mask is not None and mask.numel() > 0:
        # unsqueeze and dtype conversion
        mask_ = mask.to(left.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_
        right = right * mask_
    # gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()
    left = left * left_gate
    right = right * right_gate
    # compute out via Triton
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)
    out_f16 = torch.empty_like(left_f16)  # shape [B,N,N,hidden_dim]
    # compute strides

@triton.jit
def tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                   B, N, H,
                   stride_la_batch, stride_la_i, stride_la_k, stride_la_h,
                   stride_ra_batch, stride_ra_i, stride_ra_k, stride_ra_h,
                   stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    bh_idx = tl.program_id(0)
    block_i = tl.program_id(1)
    block_j = tl.program_id(2)

    # unpack batch and channel
    b = bh_idx // H
    h = bh_idx % H

    # offsets in global dimension
    i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks
    i_valid = i < N
    j_valid = j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        k_valid = k_offset < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + b * stride_la_batch + i[:, None] * stride_la_i + k_offset[None, :] * stride_la_k + h * stride_la_h
        left = tl.load(left_ptrs, mask=i_valid[:, None] & k_valid[None, :], other=0.0).to(tl.float32)

        # right transposed tile: shape (BLOCK_K, BLOCK_N)

def custom_kernel(data):
    import torch
    import triton
    import triton.language as tl

    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, N2, dim = input_tensor.shape
    assert N == N2
    hidden_dim = config["hidden_dim"]
    # ensure contiguous
    x = input_tensor.contiguous()
    # layernorm
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # left, right projections
    left_proj_weight = weights["left_proj.weight"].to(device)
    right_proj_weight = weights["right_proj.weight"].to(device)
    left = torch.nn.functional.linear(x_norm, left_proj_weight)
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # mask
    if mask is not None and mask.numel() > 0:
        # assume mask dtype bool or float.
        mask_ = mask.to(device).to(left.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # gates
    left_gate_weight = weights["left_gate.weight"].to(device)
    right_gate_weight = weights["right_gate.weight"].to(device)
    out_gate_weight = weights["out_gate.weight"].to(device)
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    # cast gates to same dtype as left
    left_gate_f16 = left_gate.to(torch.float16)
    right_gate_f16 = right_gate.to(torch.float16)

    # apply gates
    left = left.to(torch.float16) * left_gate_f16
    right = right.to(torch.float16) * right_gate_f16

    # allocate output
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # compute strides
    stride_la_b, stride_la_i, stride_la_k, stride_la_h = left.stride()
    stride_ra_b, stride_ra_i, stride_ra_k, stride_ra_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Triton kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    tri_mul_kernel[(B * hidden_dim, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)](
        left, right, out,
        B, N, hidden_dim,
        stride_la_b, stride_la_i, stride_la_k, stride_la_h,
        stride_ra_b, stride_ra_i, stride_ra_k, stride_ra_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # cast to float32 for remaining ops
    out_fp32 = out.to(torch.float32)

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device)
    out_norm = torch.nn.functional.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # apply out_gate
    out = out_norm * out_gate

    # final linear projection
    to_out_weight = weights["to_out.weight"].to(device)
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

"""
TriMul forward implementation with a Triton kernel.

The heavy part of the TriMul operation is the tensor contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
which is equivalent to a batched matrix multiplication between a
(N×N) matrix and the transpose of another (N×N) matrix for every
pair (batch, hidden). The kernel tiles the multiplication
with BLOCK_M × BLOCK_N tiles and accumulates in fp32 while the
input matrices are stored in fp16 to save memory bandwidth. The
remaining LayerNorms, gate computations and final linear projection
are performed with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,               # *fp16   [B, N, N, H]
    right_ptr,              # *fp16   [B, N, N, H]
    out_ptr,                # *fp32   [B, N, N, H]
    B, N, H,
    stride_la_batch, stride_la_i, stride_la_k, stride_la_h,
    stride_ra_batch, stride_ra_i, stride_ra_k, stride_ra_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute `out = left @ right.T` for each (batch, hidden) pair.
    left shape : (B, N, N, H)  -> accessed as (i, k)
    right shape: (B, N, N, H)  -> accessed as (j, k)
    The kernel treats each (b, h) as an independent matmul of
    size N×N.
    """
    bh_idx = tl.program_id(0)               # linear index for (batch, hidden)
    block_i = tl.program_id(1)              # M‑tile index
    block_j = tl.program_id(2)              # N‑tile index

    # -----------------------------------------------------------------
    # decode batch and hidden indices
    b = bh_idx // H
    h = bh_idx % H

    # -----------------------------------------------------------------
    # compute global row/col indices of the output tile
    i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_valid = i < N
    j_valid = j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # loop over the reduction dimension k
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        k_valid = k < N

        # -------- left tile (M x K) ---------------------------------
        left_offset = (
            b * stride_la_batch
            + i[:, None] * stride_la_i
            + k[None, :] * stride_la_k
            + h * stride_la_h
        )
        left = tl.load(left_ptr + left_offset,
                       mask=i_valid[:, None] & k_valid[None, :],
                       other=tl.float16(0))
        left = left.to(tl.float32)       # up‑cast for accumulation

        # -------- right tile (K x N) – we need right[j, k] = right_T[k, j]
        #   Using transposed strides: row‑stride = stride_ra_k, col‑stride = stride_ra_i
        right_offset = (
            b * stride_ra_batch
            + k[:, None] * stride_ra_k      # k as rows (after transpose)
            + j[None, :] * stride_ra_i      # j as columns (original row index)
            + h * stride_ra_h
        )
        right = tl.load(right_ptr + right_offset,
                        mask=k_valid[:, None] & j_valid[None, :],
                        other=tl.float16(0))
        right = right.to(tl.float32)     # up‑cast

        # -------- accumulate -------------------------------------------------
        acc += tl.dot(left, right)       # (M,K) @ (K,N) → (M,N)

    # -----------------------------------------------------------------
    # write the result (fp32) back to global memory
    out_offset = (
        b * stride_out_batch
        + i[:, None] * stride_out_i
        + j[None, :] * stride_out_j
        + h * stride_out_h
    )
    tl.store(out_ptr + out_offset, acc,
             mask=i_valid[:, None] & j_valid[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul module.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool or float) or None
        - weights      : dict of model parameters (already on the correct device)
        - config       : dict containing at least ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # unpack arguments
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, N2, dim = input_tensor.shape
    assert N == N2, "expected square pairwise tensor"
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # -----------------------------------------------------------------
    # 1) LayerNorm on the channel dimension
    norm_weight = weights["norm.weight"].to(device)
    norm_bias   = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # -----------------------------------------------------------------
    # 2) Linear projections (float32)
    left_proj_weight  = weights["left_proj.weight"].to(device)
    right_proj_weight = weights["right_proj.weight"].to(device)
    left = F.linear(x, left_proj_weight)    # [B,N,N,H]
    right = F.linear(x, right_proj_weight)

    # -----------------------------------------------------------------
    # 3) Optional mask (broadcast over the hidden dimension)
    if mask is not None and mask.numel() > 0:
        mask_f = mask.to(device).to(torch.float32).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4) Gating values (float32)
    left_gate_weight  = weights["left_gate.weight"].to(device)
    right_gate_weight = weights["right_gate.weight"].to(device)
    out_gate_weight   = weights["out_gate.weight"].to(device)

    left_gate  = F.linear(x, left_gate_weight).sigmoid()   # [B,N,N,H]
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate   = F.linear(x, out_gate_weight).sigmoid()    # final gate

    # -----------------------------------------------------------------
    # 5) Apply gates and cast to fp16 for the heavy multiplication
    left = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    # -----------------------------------------------------------------
    # 6) Triton kernel: out_fp32[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    out_fp32 = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides for the three tensors
    s_la_b, s_la_i, s_la_k, s_la_h = left.stride()
    s_ra_b, s_ra_i, s_ra_k, s_ra_h = right.stride()
    s_out_b, s_out_i, s_out_j, s_out_h = out_fp32.stride()

    # tile sizes – tuned for A200 GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                      # one program per (batch, hidden)
        (N + BLOCK_M - 1) // BLOCK_M,       # tiles along i
        (N + BLOCK_N - 1) // BLOCK_N        # tiles along j
    )

    tri_mul_kernel[grid](
        left, right, out_fp32,
        B, N, hidden_dim,
        s_la_b, s_la_i, s_la_k, s_la_h,
        s_ra_b, s_ra_i, s_ra_k, s_ra_h,
        s_out_b, s_out_i, s_out_j, s_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # -----------------------------------------------------------------
    # 7) LayerNorm on the hidden dimension
    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device)
    out = F.layer_norm(out_fp32, (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    # -----------------------------------------------------------------
    # 8) Apply the output gate (float32)
    out = out * out_gate

    # -----------------------------------------------------------------
    # 9) Final linear projection back to original dimension
    to_out_weight = weights["to_out.weight"].to(device)
    out = F.linear(out, to_out_weight)   # [B,N,N,dim]

    return out
