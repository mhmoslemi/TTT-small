def custom_kernel(data):
    input_tensor, mask, weights, config = data

    dim=config["dim"]; hidden_dim=config["hidden_dim"]; batch_size, seq_len = input_tensor.shape[0], input_tensor.shape[1]
    # layernorm on last dim
    # weight & bias for norm: shape [dim] each.
    # compute mean and var per element across dim
    # we can use torch.nn.functional.layer_norm directly.

    # Compute linear projections and gates
    # Using torch.nn.functional.linear for each projection.
    # Input: shape (..., dim)
    # weight shapes: [hidden_dim, dim]

    left = torch.nn.functional.linear(input_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(input_norm, weights["right_proj.weight"])
    left_gate = torch.sigmoid(torch.nn.functional.linear(input_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(input_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(input_norm, weights["out_gate.weight"]))

    # apply mask: mask shape [B,N,N]; unsqueeze dim
    if mask is None:
        mask = torch.ones_like(input_tensor[...,0]) # shape B,N,N
    mask = mask[..., None].to(input_norm.dtype)

    # Fuse gating and mask using Triton kernel
    left, right = fuse_mask_gate(left, right, mask, left_gate, right_gate)
    # out_gate will be used later.

    # Compute out via batched matmul per channel
    # reshape for bmm
    B, N, _, H = left.shape
    left_reshape = left.permute(0,3,1,2).reshape(B*H, N, N)  # shape (B*H,N,N)
    right_reshape = right.permute(0,3,1,2).reshape(B*H, N, N) # same
    out = torch.bmm(left_reshape, right_reshape.transpose(1,2))  # (B*H,N,N)
    out = out.reshape(B, H, N, N).permute(0,2,3,1)  # B,N,N,H

    # apply to_out_norm
    out = layer_norm_triton(out, weights["to_out_norm.weight"], weights["to_out_norm.bias"])
    # optional: combine gating: out = out * out_gate
    out = out * out_gate

    # final linear to original dim (dim)
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

def fuse_mask_gate_kernel(
    left_ptr, right_ptr, mask_ptr, left_gate_ptr, right_gate_ptr,
    left_out_ptr, right_out_ptr,
    N, H, B,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_..., # but left and right have same strides
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_left_gate_...,  # same as left stride
    stride_right_gate..., # same as right stride
    stride_left_out_...,
    stride_right_out_...,

    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offsets = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offsets < (B * N * N * H)
    # compute indices: we need each offset to map to (b,i,j,h). Use division.
    # We'll compute idx = offsets[mask] then compute b = idx // (N*N*H)
    # ... we can use tl.div/mod but need compile-time values or use integer ops.
    # Since tl.int64 is supported, we can do:
    idx = offsets
    # compute b = idx // (N*N*H)
    # Use indexing: total_per_batch = N*N*H
    # h_idx = idx % H
    # tmp = idx // H
    # j_idx = tmp % N
    # i_idx = tmp // N
    # for stride: we compute pointer offsets.
    total_per_batch = N * N * H
    b = idx // total_per_batch
    tmp = idx % total_per_batch
    h = tmp % H
    tmp = tmp // H
    j = tmp % N
    i = tmp // N

    # compute address offsets:
    left_offset = b*stride_left_b + i*stride_left_i + j*stride_left_j + h*stride_left_h
    right_offset similarly
    mask_offset = b*stride_mask_b + i*stride_mask_i + j*stride_mask_j # mask no h dimension -> broadcast
    left_gate_offset = left_offset (same as left) etc.

    left_val = tl.load(left_ptr + left_offset, mask=mask, other=0.0)
    right_val = tl.load(right_ptr + right_offset, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr + mask_offset, mask=mask, other=1.0) # mask default 1
    left_gate_val = tl.load(left_gate_ptr + left_offset, mask=mask, other=0.0)

    tl.store(left_out_ptr + left_offset, left_out_val, mask=mask)
    tl.store(right_out_ptr + right_offset, right_out_val, mask=mask)

@triton.jit
def fuse_mask_gate_kernel(
    left_ptr, right_ptr, mask_ptr,
    left_gate_ptr, right_gate_ptr,
    left_out_ptr, right_out_ptr,
    N, H, B,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offsets = block_start + tl.arange(0, BLOCK_SIZE)
    # mask to avoid OOB
    mask = offsets < (B * N * N * H)

    # compute 3D indices
    idx = offsets
    total_per_batch = N * N * H
    b = idx // total_per_batch
    rem = idx % total_per_batch
    i = rem // (N * H)
    rem2 = rem % (N * H)
    j = rem2 // H
    h = rem2 % H

    # compute offsets for each pointer
    left_offset = b * stride_left_b + i * stride_left_i + j * stride_left_j + h * stride_left_h
    right_offset = left_offset  # same stride layout
    mask_offset = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    left_gate_offset = b * stride_lgate_b + i * stride_lgate_i + j * stride_lgate_j + h * stride_lgate_h
    right_gate_offset = b * stride_rgate_b + i * stride_rgate_i + j * stride_rgate_j + h * stride_rgate_h

    left_val = tl.load(left_ptr + left_offset, mask=mask, other=0.0)
    right_val = tl.load(right_ptr + right_offset, mask=mask, other=0.0)
    # mask default 1 if out-of-range
    mask_val = tl.load(mask_ptr + mask_offset, mask=mask, other=1.0)
    left_gate_val = tl.load(left_gate_ptr + left_gate_offset, mask=mask, other=0.0)

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # Ensure mask is present; if config["nomask"] True maybe mask not needed
    if mask is None:
        mask = torch.ones(input_tensor.shape[:3], dtype=input_tensor.dtype, device=input_tensor.device)
    else:
        # ensure same dtype as input
        mask = mask.to(dtype=input_tensor.dtype, device=input_tensor.device)

    # layernorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Using torch.nn.functional.layer_norm
    x_norm = torch.nn.functional.layer_norm(input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape B,N,N,hidden_dim
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Gates pre-activations
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate_pre = torch.nn.functional.linear(x_norm, left_gate_weight)
    right_gate_pre = torch.nn.functional.linear(x_norm, right_gate_weight)
    out_gate_pre = torch.nn.functional.linear(x_norm, out_gate_weight)

    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # Apply mask and gates fused in Triton kernel
    left_fused, right_fused = fuse_mask_gate(left, right, mask, left_gate, right_gate)

    # Compute out = left_fused @ right_fused^T per hidden_dim
    B, N, _, H = left_fused.shape
    # reshape for batch matmul
    left_mat = left_fused.permute(0,3,1,2).reshape(B*H, N, N)
    right_mat = right_fused.permute(0,3,1,2).reshape(B*H, N, N)

    # right matrix transposed on last two dims
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))  # shape (B*H,N,N)
    out = out_mat.reshape(B, H, N, N).permute(0,2,3,1)  # shape (B,N,N,H)

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, normalized_shape=(hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # out gate multiplication
    out = out * out_gate

    # final linear to original dim
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

"""
TriMul (outgoing) forward implementation.

The operation follows the AlphaFold3 TriMul module:
1. LayerNorm on the input tensor.
2. Linear projections `left` and `right` (dim → hidden_dim) and three gates
   (`left_gate`, `right_gate`, `out_gate`) produced by additional linear layers
   followed by a sigmoid.
3. A Triton kernel fuses mask multiplication and gating:
   left  = left  * mask * left_gate
   right = right * mask * right_gate
4. The core update is a batched matrix multiplication:
   out[b, i, j, d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is equivalent to `left @ rightᵀ` performed independently for each
   hidden dimension `d`.  This is realized via `torch.bmm` on reshaped tensors.
5. LayerNorm on the hidden dimension, multiplication by `out_gate`, and a final
   linear projection back to the original channel size.

Only the element‑wise mask/gate fusion is performed in Triton, satisfying the
requirement to have part of the computation in a custom kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def fuse_mask_gate_kernel(
    left_ptr, right_ptr, mask_ptr,
    left_gate_ptr, right_gate_ptr,
    left_out_ptr, right_out_ptr,
    N, H, B,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    BLOCK_SIZE: tl.constexpr
):
    """Fused kernel:
       left_out  = left  * mask * left_gate
       right_out = right * mask * right_gate
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offsets = block_start + tl.arange(0, BLOCK_SIZE)

    total = B * N * N * H
    valid = offsets < total

    idx = offsets
    total_per_batch = N * N * H
    b = idx // total_per_batch
    rem = idx % total_per_batch
    i = rem // (N * H)
    rem2 = rem % (N * H)
    j = rem2 // H
    h = rem2 % H

    # compute flat offsets for each tensor
    left_off  = b * stride_left_b  + i * stride_left_i  + j * stride_left_j  + h * stride_left_h
    right_off = left_off
    mask_off  = b * stride_mask_b  + i * stride_mask_i  + j * stride_mask_j
    lgate_off = b * stride_lgate_b + i * stride_lgate_i + j * stride_lgate_j + h * stride_lgate_h
    rgate_off = b * stride_rgate_b + i * stride_rgate_i + j * stride_rgate_j + h * stride_rgate_h

    left_val      = tl.load(left_ptr      + left_off,   mask=valid, other=0.0)
    right_val     = tl.load(right_ptr     + right_off,  mask=valid, other=0.0)
    mask_val      = tl.load(mask_ptr      + mask_off,   mask=valid, other=1.0)
    left_gate_val = tl.load(left_gate_ptr + lgate_off,  mask=valid, other=0.0)
    right_gate_val= tl.load(right_gate_ptr+ rgate_off, mask=valid, other=0.0)

    left_out_val  = left_val  * mask_val * left_gate_val
    right_out_val = right_val * mask_val * right_gate_val

    tl.store(left_out_ptr  + left_off,  left_out_val,  mask=valid)
    tl.store(right_out_ptr + right_off, right_out_val, mask=valid)

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (may be all ones)
            - weights: dict of torch.Tensors containing all linear and norm parameters
            - config: dict with keys "dim" and "hidden_dim" (and possibly "nomask")

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # If mask is None or nomask flag is True, use a mask of ones.
    if mask is None or config.get("nomask", False):
        mask = torch.ones((B, N, N), dtype=dtype, device=device)
    else:
        mask = mask.to(dtype=dtype, device=device)

    # 1. Input LayerNorm
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # 2. Linear projections
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # 3. Gates (pre‑sigmoid)
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # 4. Fuse mask and gate application via Triton
    left, right = _fuse_mask_gate(left, right, mask, left_gate, right_gate)

    # 5. Core TriMul via batched GEMM (left @ rightᵀ per hidden channel)
    # reshape to (B*H, N, N) for bmm
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # 6. Output LayerNorm on hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    # 7. Apply output gate and final linear projection
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
