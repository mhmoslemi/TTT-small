def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    B, N, _, _ = input_tensor.shape

    # LayerNorm
    norm_w = weights['norm.weight']
    norm_b = weights['norm.bias']
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Projections
    left_proj_w = weights['left_proj.weight']
    right_proj_w = weights['right_proj.weight']
    left = torch.nn.functional.linear(x, left_proj_w)  # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_w)

    # Apply mask
    if mask_tensor is not None:
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)  # broadcast
        left = left * mask
        right = right * mask

    # Gates
    left_gate_w = weights['left_gate.weight']
    right_gate_w = weights['right_gate.weight']
    out_gate_w = weights['out_gate.weight']
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Now compute out = sum_k left[..., i, k, d] * right[..., j, k, d]
    # Use tri_mul_matmul_kernel

    # left, right are float32 maybe; cast to float16 for speed
    # Use contiguous memory ordering.
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, H = left.shape
    # allocate output
    out = torch.empty((B, N, N, H), dtype=left.dtype, device=left.device)

    # Get strides
    # left.stride(): tuple length 4. For left shape [B,N,N,H]
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Choose block sizes: M,N,K minimal 32? Tuning
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    # compute grid dimensions
    num_i = (N + BLOCK_M - 1) // BLOCK_M
    num_j = (N + BLOCK_N - 1) // BLOCK_N
    num_batches_hidden = B * H

    grid = (num_i, num_j, num_batches_hidden)

    tri_mul_matmul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B= B,
        N= N,
        H=H,
        stride_l_b=stride_l_b,
        stride_l_i=stride_l_i,
        stride_l_k=stride_l_k,
        stride_l_h=stride_l_h,
        stride_r_b=stride_r_b,
        stride_r_j=stride_r_j,
        stride_r_k=stride_r_k,
        stride_r_h=stride_r_h,
        stride_o_b=stride_o_b,
        stride_o_i=stride_o_i,
        stride_o_j=stride_o_j,
        stride_o_h=stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Apply to_out_norm
    to_out_norm_w = weights['to_out_norm.weight']
    to_out_norm_b = weights['to_out_norm.bias']
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Multiply by out_gate
    out = out * out_gate

    # Final linear
    to_out_w = weights['to_out.weight']
    out = torch.nn.functional.linear(out, to_out_w)  # weight shape [dim, hidden_dim]
    return out

"""
TriMul forward kernel (outgoing) for AlphaFold3 style pairwise update.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Linear projections `left` and `right` to hidden dimension.
3. (Optional) mask the projected tensors.
4. Compute three sigmoid gates from the normalized input and apply them to
   `left`, `right` and later to the output.
5. The core operation is
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication:
        left_perm[b,d] (N×N) @ right_perm[b,d].T (N×N)
   for every hidden slice `d`.  A custom Triton kernel performs the
   reduction over `k` with tiling (BLOCK_M×BLOCK_N×BLOCK_K).
6. Layer‑norm the result, apply the output gate and a final linear projection
   back to the original dimension.

The Triton kernel computes a batched GEMM where each batch corresponds to a
pair (batch_index, hidden_index).  Input matrices are stored as FP16 to reduce
memory traffic; accumulation is performed in FP32 for numerical stability.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ---------------------------------------------------------
# Triton kernel: batched GEMM accumulating A(i,k) * B(j,k) over k
# ---------------------------------------------------------
@triton.jit
def batched_tri_mul_kernel(
    A, B, C,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col
    pid_batch = tl.program_id(2)      # batch * hidden index

    # Offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this (batch, hidden) slice
    a_batch_ptr = A + pid_batch * stride_a_batch
    b_batch_ptr = B + pid_batch * stride_b_batch
    c_batch_ptr = C + pid_batch * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over reduction dimension K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_a_m) + (offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load B tile transposed (K x N) -> original B is (N x K)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # Accumulate A @ B_T
        acc += tl.dot(a, b)

    # Store the result tile
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs,
              acc,
              mask=mask_m[:, None] & mask_n[None, :])

# ---------------------------------------------------------
# Entry point expected by the evaluator
# ---------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (dtype= float32)
    """
    # Unpack arguments
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------
    # LayerNorm (input)
    # -----------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # -----------------
    # Projections
    # -----------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)          # [B, N, N, hidden]
    right = F.linear(x, right_proj_weight)        # [B, N, N, hidden]

    # -----------------
    # Optional masking before gating (as in reference)
    # -----------------
    if not nomask:
        # mask shape [B, N, N] -> broadcast to hidden dimension
        mask = mask_tensor.to(dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # -----------------
    # Gating networks (sigmoid)
    # -----------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # Prepare tensors for the Triton kernel (FP16 for bandwidth)
    # -------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Permute to bring hidden dimension to batch axis: [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape

    # Collapse batch & hidden into a single leading dimension
    A = left_perm.view(B * H, N, N)
    B_ = right_perm.view(B * H, N, N)

    # Output buffer (FP32 accumulation)
    C = torch.empty_like(A, dtype=torch.float32, device=device)

    # Strides for the kernel (in elements, not bytes)
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_n, stride_b_k = B_.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tiling parameters (tuned for A200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid dimensions
    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (grid_m, grid_n, B * H)

    # Launch Triton kernel
    batched_tri_mul_kernel[grid](
        A, B_, C,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_n, stride_b_k,
        stride_c_batch, stride_c_m, stride_c_n,
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden]
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------
    # Output LayerNorm, gating and final projection
    # -----------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    out = out * out_gate  # element‑wise gating

    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)   # back to original `dim`

    return out
