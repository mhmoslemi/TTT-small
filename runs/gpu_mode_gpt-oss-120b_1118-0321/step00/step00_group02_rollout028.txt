def layer_norm(x, weight, bias, eps=1e-5):
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    if weight is not None:
        x_norm = x_norm * weight
    if bias is not None:
        x_norm = x_norm + bias

def tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_ah, stride_ai, stride_ak,
    stride_bh, stride_bk, stride_bj,
    stride_ch, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_b = tl.program_id(0)   # batch*hidden
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    num_k = tl.cdiv(K, BLOCK_K)

def custom_kernel(data):
    x, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    eps = 1e-5

    # LayerNorm
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'], eps=1e-5)

    # Projections
    left = torch.nn.functional.linear(x_norm, weights['left_proj.weight'])
    right = torch.nn.functional.linear(x_norm, weights['right_proj.weight'])

    # mask handling
    if mask is None:
        mask = torch.ones(x.shape[:3], device=x.device, dtype=x.dtype)
    mask = mask.unsqueeze(-1)  # [B,N,N,1]
    left = left * mask
    right = right * mask

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, weights['left_gate.weight']).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights['right_gate.weight']).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights['out_gate.weight']).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Permute for kernel
    # left shape (B,N,N,hidden) -> permute to (B, hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Transpose right for multiplication
    right_T = right_perm.transpose(-2, -1).contiguous()  # (B, hidden, N, N) with swapped i/k

    B, H, N, _ = left_perm.shape
    batch_h = B * H

    # Flatten batch dimension
    left_flat = left_perm.view(batch_h, N, N)
    right_flat = right_T.view(batch_h, N, N)

    out_flat = torch.empty_like(left_flat)

    # Get strides
    stride_ah = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bh = right_flat.stride(0)
    stride_bk = right_flat.stride(1)  # K dimension (original k)
    stride_bj = right_flat.stride(2)  # j dimension (output column)

    stride_ch = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    # Launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (batch_h, (N + BLOCK_M - 1)//BLOCK_M, (N + BLOCK_N - 1)//BLOCK_N)

        stride_bh, stride_bk, stride_bj,
        stride_ch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K

"""
TriMul outgoing update (AlphaFold3) – custom forward pass.

Algorithm
---------
1. Layer‑norm the input tensor `x` over the last channel dimension.
2. Linear projections (`left_proj`, `right_proj`) produce two tensors of shape
   [B, N, N, H] (H = hidden_dim).
3. Optional binary mask (broadcast to channel dim) zeros out entries of
   `left` and `right`.
4. Gating signals (`left_gate`, `right_gate`, `out_gate`) are obtained by
   linear layers on the normalized input followed by a sigmoid.
   The gates are multiplied element‑wise into `left` and `right`.
5. The core operation:
       out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   is a batched matrix multiplication for each hidden channel.  By flattening
   the (batch, hidden) dimensions we obtain a GEMM of shape
   (BH, N, N) @ (BH, N, N)ᵀ where BH = B * H.  This is implemented in Triton
   (`tri_mul_matmul_kernel`) with tiling over i, j and k.
6. The result is reshaped back to [B, N, N, H] and layer‑normed over H.
7. The final gated scaling (`out_gate`) and linear projection back to the
   original channel dimension produce the output tensor of shape
   [B, N, N, dim].

Only the heavy O(N³·H) multiplication is performed on the GPU via Triton;
all surrounding ops use PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    batch, M, N, K,                # dimensions (batch = B*H, M=N, N=N, K=N)
    stride_ah, stride_ai, stride_ak,   # strides for A (batch, i, k)
    stride_bh, stride_bk, stride_bj,   # strides for B (batch, k, j)
    stride_ch, stride_ci, stride_cj,   # strides for C (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr
):
    """Batched matrix multiplication C = A @ Bᵀ for the TriMul core.

    A : (batch, M, K)
    B : (batch, K, N)   (this is the transposed right tensor)
    C : (batch, M, N)
    """
    pid_b = tl.program_id(0)        # batch*hidden index
    pid_m = tl.program_id(1)        # block row index (i)
    pid_n = tl.program_id(2)        # block col index (j)

    # --------------------------------------------------------------
    # Offsets for the output block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the reduction dimension K
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers + offsets for A
        a_ptrs = (a_ptr + pid_b * stride_ah
                  + (offs_m[:, None] * stride_ai)
                  + (offs_k[None, :] * stride_ak))
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Pointers + offsets for B (B is already transposed, so its
        # first reduction dim is K and the column dim is j)
        b_ptrs = (b_ptr + pid_b * stride_bh
                  + (offs_k[:, None] * stride_bk)
                  + (offs_n[None, :] * stride_bj))
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Compute block dot product
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write back the result block
    c_ptrs = (c_ptr + pid_b * stride_ch
              + (offs_m[:, None] * stride_ci)
              + (offs_n[None, :] * stride_cj))
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [bs, seq_len, seq_len, dim]
    """
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input layer‑norm
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections
    left = F.linear(x_norm, weights["left_proj.weight"])   # (B,N,N,H)
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask
    if mask is None:
        mask = torch.ones(x.shape[:3], device=device, dtype=dtype)
    mask = mask.to(dtype).unsqueeze(-1)      # (B,N,N,1)
    left = left * mask
    right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating signals (computed on the normalized input)
    left_gate = F.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core TriMul computation via Triton
    #    Reshape to (B, H, N, N) -> flatten batch & hidden for batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()      # (B,H,N,N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()    # (B,H,N,N)

    # transpose right on the last two dims (k ↔ j) to obtain Bᵀ
    right_T = right_perm.transpose(-2, -1).contiguous()   # (B,H,N,N)

    B, H, N, _ = left_perm.shape
    batch_h = B * H

    # View as (batch_h, N, N)
    left_flat = left_perm.view(batch_h, N, N)
    right_flat = right_T.view(batch_h, N, N)

    # Allocate output
    out_flat = torch.empty_like(left_flat)

    # Strides (as Python integers)
    stride_ah = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bh = right_flat.stride(0)
    stride_bk = right_flat.stride(1)   # K dimension (original k)
    stride_bj = right_flat.stride(2)   # j dimension (output column)

    stride_ch = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    # Tiling parameters – chosen to fit in registers on B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch_h,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_h,
        N,
        N,
        N,                # M=N, N=N, K=N
        stride_ah,
        stride_ai,
        stride_ak,
        stride_bh,
        stride_bk,
        stride_bj,
        stride_ch,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Post‑process: layer‑norm over hidden dimension, gating, final linear
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out_norm * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
