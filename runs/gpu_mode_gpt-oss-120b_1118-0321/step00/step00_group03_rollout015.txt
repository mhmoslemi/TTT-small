def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, hidden,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_bh = tl.program_id(0)  # batch*hidden dimension index
    pid_i = tl.program_id(1)  # row block index
    pid_j = tl.program_id(2)  # col block index

    bh = pid_bh // hidden
    d = pid_bh % hidden

    # compute start indices
    i_start = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_start = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out-of-bounds
    mask_i = i_start < N
    mask_j = j_start < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(0, num_k):
        k_start = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_start < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + \
            bh * stride_left_b + \
            i_start[:, None] * stride_left_i + \
            k_start[None, :] * stride_left_k + \
            d * stride_left_d

        left = tl.load(left_ptrs, mask=(mask_i[:, None] & mask_k[None, :]), other=0.0).to(tl.float32)

        # right tile: shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + \
            bh * stride_right_b + \
            j_start[None, :] * stride_right_j + \
            k_start[:, None] * stride_right_k + \
            d * stride_right_d

        right = tl.load(right_ptrs, mask=(mask_k[:, None] & mask_j[None, :]), other=0.0).to(tl.float32)

        acc += tl.dot(left, right)
    # store
    out_ptrs = out_ptr + \
        bh * stride_out_b + \
        i_start[:, None] * stride_out_i + \
        j_start[None, :] * stride_out_j + \
        d * stride_out_d

"""
TriMul (outgoing) forward pass implemented with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape [B, N, N, C]).
2. Apply four linear projections (left, right, left_gate, right_gate, out_gate)
   and sigmoid gates where required.
3. Mask the projected tensors if a mask is supplied.
4. Fuse the core 4‑D tensor contraction
       out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   using a custom Triton kernel that performs a batched GEMM for every
   hidden channel `h`.  The kernel tiles the (i, j) output plane (BLOCK_M×BLOCK_N)
   while looping over k in BLOCK_K sized fragments.  Each program instance
   processes a specific (batch, hidden) pair, giving a grid of
   (B*H, ceil(N/BM), ceil(N/BN)) blocks.
5. Apply a second LayerNorm on the hidden dimension, multiply by `out_gate`,
   and a final linear projection back to the original channel dimension.

The heavy lifting (step 4) is executed on GPU with Triton, while the surrounding
operations rely on PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_matmul_kernel(
    left_ptr,          # const float*
    right_ptr,         # const float*
    out_ptr,           # float*
    N,                 # seq_len (int)
    H,                 # hidden_dim (int)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # ---------------------------------------------------------------
    #  Indexing
    # ---------------------------------------------------------------
    pid_bh = tl.program_id(0)          # enumerates batch*hidden
    pid_i  = tl.program_id(1)          # block row (i)
    pid_j  = tl.program_id(2)          # block col (j)

    b = pid_bh // H                     # batch index
    h = pid_bh %  H                     # hidden channel

    # tile start positions
    i_start = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_start = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_start < N
    mask_j = j_start < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for k_iter in range(0, num_k):
        k_start = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_start < N

        # --------------------------------------------------------------------
        # Load left tile  (M x K)
        # --------------------------------------------------------------------
        left_ptrs = left_ptr + \
            b   * stride_left_b + \
            i_start[:, None] * stride_left_i + \
            k_start[None, :] * stride_left_k + \
            h   * stride_left_h

        left = tl.load(left_ptrs,
                       mask=(mask_i[:, None] & mask_k[None, :]),
                       other=0.0).to(tl.float32)

        # --------------------------------------------------------------------
        # Load right tile (K x N)  – note the transpose so that we can use tl.dot
        # --------------------------------------------------------------------
        right_ptrs = right_ptr + \
            b   * stride_right_b + \
            j_start[None, :] * stride_right_j + \
            k_start[:, None] * stride_right_k + \
            h   * stride_right_h

        right = tl.load(right_ptrs,
                        mask=(mask_k[:, None] & mask_j[None, :]),
                        other=0.0).to(tl.float32)

        # --------------------------------------------------------------------
        # Matrix multiply: (M x K) * (K x N) -> (M x N)
        # --------------------------------------------------------------------
        acc += tl.dot(left, right)

    # --------------------------------------------------------------------
    # Store result
    # --------------------------------------------------------------------
    out_ptrs = out_ptr + \
        b   * stride_out_b + \
        i_start[:, None] * stride_out_i + \
        j_start[None, :] * stride_out_j + \
        h   * stride_out_h

    tl.store(out_ptrs,
             acc,
             mask=(mask_i[:, None] & mask_j[None, :]))

# ----------------------------------------------------------------------
# Entry point expected by the grader
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, C = input_tensor.shape
    hidden = config["hidden_dim"]
    dim = config["dim"]

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     (C,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # ------------------------------------------------------------------
    # 2) Linear projections
    # ------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # no bias in these linear layers
    left = F.linear(x, left_proj_weight)        # [B,N,N,hidden]
    right = F.linear(x, right_proj_weight)      # [B,N,N,hidden]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 3) Apply mask (if provided) before gating, as in reference code
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(dtype)   # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gating
    # ------------------------------------------------------------------
    left  = left * left_gate
    right = right * right_gate

    # Ensure tensors are contiguous for Triton
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 5) Core contraction via Triton
    # ------------------------------------------------------------------
    out = torch.empty_like(left)   # shape [B,N,N,hidden]

    # strides (in elements) for Triton pointer arithmetic
    slb, sli, slk, slh = left.stride()
    srb, srj, srk, srh = right.stride()
    sob, soi, soj, soh = out.stride()

    # Tunable block sizes (experimentally good for N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden,                     # program id 0 : batch * hidden
        (N + BLOCK_M - 1) // BLOCK_M,   # program id 1 : row tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # program id 2 : col tiles
    )

    tri_mul_matmul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        N=N,
        H=hidden,
        stride_left_b=slb,
        stride_left_i=sli,
        stride_left_k=slk,
        stride_left_h=slh,
        stride_right_b=srb,
        stride_right_j=srj,
        stride_right_k=srk,
        stride_right_h=srh,
        stride_out_b=sob,
        stride_out_i=soi,
        stride_out_j=soj,
        stride_out_h=soh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Post‑processing: LayerNorm, out_gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       (hidden,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)

    out = out * out_gate
    out = F.linear(out, to_out_weight)   # [B,N,N,dim]

    return out
