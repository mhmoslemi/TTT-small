   pid0 = tl.program_id(0) # corresponds to batch*hidden
   pid1 = tl.program_id(1) # block row
   pid2 = tl.program_id(2) # block col

   # Compute batch and hidden dimension from pid0
   b = pid0 // H
   d = pid0 % H

   # Offsets
   i = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
   j = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)

   # Mask computations: within range
   i_mask = i < N
   j_mask = j < N

   # Initialize accumulator
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   # k loop

def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Program ID
    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)
    pid2 = tl.program_id(2)

    # decode batch and hidden
    b = pid0 // H
    d = pid0 % H

    # block offsets
    i_base = pid1 * BLOCK_M
    j_base = pid2 * BLOCK_N

    # ranges
    i_range = tl.arange(0, BLOCK_M)
    j_range = tl.arange(0, BLOCK_N)
    k_range = tl.arange(0, BLOCK_K)

    i = i_base + i_range
    j = j_base + j_range

    mask_i = i < N
    mask_j = j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k blocks
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k_idx in range(num_k_blocks):
        k_base = k_idx * BLOCK_K
        k = k_base + k_range
        mask_k = k < N

        a_ptr = left_ptr + b * stride_left_b + i[:, None] * stride_left_i + k[None, :] * stride_left_k + d * stride_left_h

"""
TriMul (outgoing) fused implementation.

The module performs the following steps:

1. Layer‑norm the input tensor x ∈ ℝ^{B×N×N×C}.
2. Linear projections to a hidden dimension H:
      left  = x @ W_left   (no bias)
      right = x @ W_right
3. Optional pairwise mask (broadcasted over H) is applied.
4. Gating:
      left_gate  = sigmoid(x @ W_left_gate)
      right_gate = sigmoid(x @ W_right_gate)
      out_gate   = sigmoid(x @ W_out_gate)
   The gates are multiplied element‑wise with the projected tensors.
5. Core “triangular” reduction:
      out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]      (H‑dim vector)
   This is implemented as a fused batched GEMM in Triton:
      For each hidden channel h we compute (A_h @ B_hᵀ) where
          A_h[b] = left[b,:,:,h]   ∈ ℝ^{N×N}
          B_h[b] = right[b,:,:,h]  ∈ ℝ^{N×N}
6. Layer‑norm over the hidden dimension, gating by out_gate,
   and final linear projection back to C dimensions.

Only the O(N³) reduction (step 5) is executed on the GPU via a Triton
kernel; all other linear / normalization steps use PyTorch primitives.
The kernel works with half‑precision inputs (fp16) while accumulating in
fp32 for numerical stability.
"""

import triton.language as tl

    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                    # tensor sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out = sum_k left[:,i,k,:] * right[:,j,k,:] for each hidden channel."""
    pid0 = tl.program_id(0)   # combined batch * hidden
    pid1 = tl.program_id(1)   # block row (i)
    pid2 = tl.program_id(2)   # block col (j)

    # decode batch index and hidden index
    b = pid0 // H
    h = pid0 % H

    # base indices for the tile
    i_base = pid1 * BLOCK_M
    j_base = pid2 * BLOCK_N

    # tile coordinate vectors
    i = i_base + tl.arange(0, BLOCK_M)
    j = j_base + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k‑tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)

    for tk in range(num_k_tiles):
        k_base = tk * BLOCK_K
        k = k_base + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # ------------------- load A = left[b,i,k,h] (M×K) -----------------
        a_ptrs = (
            left_ptr
            + b * stride_left_b
            + i[:, None] * stride_left_i
            + k[None, :] * stride_left_k
            + h * stride_left_h
        )
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # ------------------- load Bᵀ = right[b,j,k,h] as (K×N) ------------
        # we load the transpose so that we can use tl.dot directly
        b_ptrs = (
            right_ptr
            + b * stride_right_b
            + k[:, None] * stride_right_k   # k as row dimension
            + j[None, :] * stride_right_i   # j as column dimension
            + h * stride_right_h
        )
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # ------------------- accumulate -----------------
        acc = acc + tl.dot(a, b)   # fp32 accumulator, a/b are fp16

    # ------------------- store result -----------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + i[:, None] * stride_out_i
        + j[None, :] * stride_out_j
        + h * stride_out_h
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Arguments
    ---------
    data: tuple
        (input_tensor, mask, weights, config)
        - input_tensor: [B, N, N, C]   (float32)
        - mask:         [B, N, N]      (bool or float)
        - weights: dict of model parameters
        - config:  dict containing "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ----------------------------------------------------------------------
    # LayerNorm on the input
    # ----------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B,N,N,dim]

    # ----------------------------------------------------------------------
    # Linear projections (no bias)
    # ----------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ----------------------------------------------------------------------
    # Optional mask (broadcast over hidden dimension)
    # ----------------------------------------------------------------------
    if mask is None:
        mask = torch.ones(input_tensor.shape[:3], dtype=dtype, device=device)
    mask = mask.to(dtype).unsqueeze(-1)   # [B,N,N,1]
    left = left * mask
    right = right * mask

    # ----------------------------------------------------------------------
    # Gating logits and sigmoid gates
    # ----------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()   # [B,N,N,hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------------
    # Cast to fp16 for the heavy O(N³) reduction
    # ----------------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape
    H = hidden_dim

    # Allocate output buffer (fp32 accumulator)
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # ----------------------------------------------------------------------
    # Extract strides (in elements, not bytes)
    # ----------------------------------------------------------------------
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left_h.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right_h.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # ----------------------------------------------------------------------
    # Triton kernel launch parameters
    # ----------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                                   # batch * hidden dimension
        (N + BLOCK_M - 1) // BLOCK_M,            # rows
        (N + BLOCK_N - 1) // BLOCK_N,            # columns
    )

    tri_mul_kernel[grid](
        left_h,
        right_h,
        out,
        B,
        N,
        H,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_i,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----------------------------------------------------------------------
    # Post‑processing: hidden‑dim LayerNorm, out_gate, final projection
    # ----------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )                     # [B,N,N,hidden_dim]

    out = out * out_gate   # gating

    to_out_weight = weights["to_out.weight"]
    output = F.linear(out, to_out_weight)   # [B,N,N,dim]

    return output
