"""
TriMul (outgoing) forward pass implemented with Triton kernels.

Algorithm
---------
1. Layer‑norm over the input channel dimension (dim) using a Triton kernel.
2. Linear projections (left/right) and gates (left_gate/right_gate/out_gate) are
   performed with torch.nn.functional.linear followed by sigmoid.
3. Mask (if present) is applied to the left/right projections.
4. Gate values are multiplied into the projections.
5. Core \"einsum\" is realized as a batch of GEMMs:
   for each hidden channel `h` we compute
       out[..., i, j, h] = Σ_k left[..., i, k, h] * right[..., j, k, h]
   which is equivalent to a batched matmul of shape (B*H, N, N) @
   (B*H, N, N).T.
6. The result is layer‑normed over the hidden dimension, multiplied by the
   out‑gate, and finally linearly projected back to the original channel size.
   The second layer‑norm + gating is performed with another Triton kernel.
"""

import torch
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernels
# --------------------------------------------------------------

@triton.jit
def layernorm_kernel(
    x_ptr,                     # const * float32, input
    out_ptr,                   # * float32, output
    w_ptr,                     # const * float32, weight (size C)
    b_ptr,                     # const * float32, bias   (size C)
    B, N, C, eps,              # int32/int32/int32/float32
    BLOCK_SIZE: tl.constexpr   # compile‑time block size for the channel dim
):
    """LayerNorm over the last dimension (size C) of a tensor [B,N,N,C]."""
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # Base offset for the vector belonging to this (b,i,j)
    base = pid * C

    # ------------------------------------------------------------------
    # First pass: compute mean and variance
    # ------------------------------------------------------------------
    sum_val = tl.float32(0.0)
    sum_sq = tl.float32(0.0)
    for start in range(0, C, BLOCK_SIZE):
        offs = start + tl.arange(0, BLOCK_SIZE)
        mask = offs < C
        x = tl.load(x_ptr + base + offs, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)

    mean = sum_val / C
    var = sum_sq / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalized values (with weight & bias)
    # ------------------------------------------------------------------
    for start in range(0, C, BLOCK_SIZE):
        offs = start + tl.arange(0, BLOCK_SIZE)
        mask = offs < C
        x = tl.load(x_ptr + base + offs, mask=mask, other=0.0).to(tl.float32)
        w = tl.load(w_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        b = tl.load(b_ptr + offs, mask=mask, other=0.0).to(tl.float32)

        y = (x - mean) * inv_std          # normalize
        y = y * w + b                     # weight + bias
        tl.store(out_ptr + base + offs, y, mask=mask)


@triton.jit
def to_out_norm_gate_kernel(
    x_ptr,                     # const * float32, input tensor (B,N,N,H)
    gate_ptr,                  # const * float32, out_gate (B,N,N,H)
    out_ptr,                   # * float32, output after norm&gate
    w_ptr,                     # const * float32, to_out_norm.weight (H)
    b_ptr,                     # const * float32, to_out_norm.bias   (H)
    B, N, H, eps,              # int32/int32/int32/float32
    BLOCK_SIZE: tl.constexpr   # compile‑time block size for hidden dim
):
    """LayerNorm over hidden dimension H, followed by multiplication with out_gate."""
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    base = pid * H

    # ------------------------------------------------------------------
    # Compute mean & variance over the hidden dimension
    # ------------------------------------------------------------------
    sum_val = tl.float32(0.0)
    sum_sq = tl.float32(0.0)
    for start in range(0, H, BLOCK_SIZE):
        offs = start + tl.arange(0, BLOCK_SIZE)
        mask = offs < H
        x = tl.load(x_ptr + base + offs, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Normalize, apply weight/bias, then multiply by out_gate
    # ------------------------------------------------------------------
    for start in range(0, H, BLOCK_SIZE):
        offs = start + tl.arange(0, BLOCK_SIZE)
        mask = offs < H
        x = tl.load(x_ptr + base + offs, mask=mask, other=0.0).to(tl.float32)
        g = tl.load(gate_ptr + base + offs, mask=mask, other=0.0).to(tl.float32)
        w = tl.load(w_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        b = tl.load(b_ptr + offs, mask=mask, other=0.0).to(tl.float32)

        y = (x - mean) * inv_std      # normalize
        y = y * w + b                 # apply learned ln weight/bias
        y = y * g                     # multiply by out_gate
        tl.store(out_ptr + base + offs, y, mask=mask)


# --------------------------------------------------------------
# Custom kernel entry point
# --------------------------------------------------------------

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        torch.Tensor of shape [bs, seq_len, seq_len, dim]
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # --------------------------------------------------------------
    # 1) First LayerNorm (over dim)
    # --------------------------------------------------------------
    x_norm = torch.empty_like(input_tensor, dtype=torch.float32, device=device)

    total_elem = bs * N * N
    grid = (total_elem,)
    layernorm_kernel[grid](
        input_tensor,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        bs, N, dim, eps,
        BLOCK_SIZE=32,
    )

    # --------------------------------------------------------------
    # 2) Linear projections & gates (using torch for the dense matmul)
    # --------------------------------------------------------------
    # Linear: out = x @ weight.T  (bias is None)
    left_proj = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["out_gate.weight"])
    )

    # --------------------------------------------------------------
    # 3) Apply mask (if present) and gating
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        mask = mask_tensor.to(dtype).unsqueeze(-1)      # [B,N,N,1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    left = left_proj * left_gate
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # 4) Core TriMul (batched GEMM over the hidden dimension)
    # --------------------------------------------------------------
    # Reshape tensors to (B*H, N, N) for batched matmul
    left = left.contiguous()
    right = right.contiguous()
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # B, H, N, N
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_mat = left_perm.view(bs * hidden_dim, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(bs * hidden_dim, N, N)

    # out_mat[b*h, i, j] = Σ_k left_mat[b*h, i, k] * right_mat[b*h, j, k]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # B,N,N,H

    # --------------------------------------------------------------
    # 5) Second LayerNorm + out_gate (Triton)
    # --------------------------------------------------------------
    out_normed = torch.empty_like(out, dtype=torch.float32, device=device)

    total_elem2 = bs * N * N
    grid2 = (total_elem2,)
    to_out_norm_gate_kernel[grid2](
        out,
        out_gate,
        out_normed,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        bs, N, hidden_dim, eps,
        BLOCK_SIZE=32,
    )

    # --------------------------------------------------------------
    # 6) Final linear projection back to original dimension
    # --------------------------------------------------------------
    output = torch.nn.functional.linear(out_normed, weights["to_out.weight"])

    return output
