@triton.jit
def trmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    batch_stride_a, stride_a_m, stride_a_k,
    batch_stride_b, stride_b_k, stride_b_n,
    batch_stride_c, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    # compute indices
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    batch = pid // (num_blocks_m * num_blocks_n)
    tile = pid % (num_blocks_m * num_blocks_n)
    pid_m = tile // num_blocks_n
    pid_n = tile % num_blocks_n

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # compute pointer base for current batch
    a_base = a_ptr + batch * batch_stride_a
    b_base = b_ptr + batch * batch_stride_b
    c_base = c_ptr + batch * batch_stride_c

    # load tiles
    a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
    b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (offs_k < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & (offs_n < N)[None, :], other=0.0)
        a = a.to(tl.float16)
        b = b.to(tl.float16)
        acc += tl.dot(a, b, out_dtype=tl.float32)

        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # store
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc, mask=mask)

def custom_kernel(data):
    import torch
    import triton
    import triton.language as tl

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # LayerNorm on input
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask if present
    if mask is not None and not config.get("nomask", False):
        mask = mask.to(left.dtype).unsqueeze(-1)  # shape B,N,N,1
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Prepare for Triton matmul: A @ B^T per channel
    B, N, _, _ = left.shape

    # Permute so that channel dimension becomes second, making rows contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Convert to half for compute
    left_fp16 = left_perm.to(torch.float16)
    right_fp16 = right_perm.to(torch.float16)

    # Output buffer
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # View as batch-of-matrices for kernel
    A = left_fp16.view(-1, N, N)
    B_mat = right_fp16.view(-1, N, N)
    C = out.view(-1, N, N)

    # Strides (in elements)
    stride_a_batch = A.stride(0)
    stride_a_m = A.stride(1)
    stride_a_k = A.stride(2)

    stride_b_batch = B_mat.stride(0)
    stride_b_k = B_mat.stride(2)   # column stride = 1
    stride_b_n = B_mat.stride(1)   # row stride = N

    stride_c_batch = C.stride(0)
    stride_c_m = C.stride(1)
    stride_c_n = C.stride(2)

    total_batches = A.shape[0]
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = total_batches * num_blocks_m * num_blocks_n

    @triton.jit
    def trmul_kernel(
        a_ptr, b_ptr, c_ptr,
        M, N, K,
        batch_stride_a, stride_a_m, stride_a_k,
        batch_stride_b, stride_b_k, stride_b_n,
        batch_stride_c, stride_c_m, stride_c_n,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
    ):
        pid = tl.program_id(0)

        # compute indices for batch, tiles
        num_blocks_m = tl.cdiv(M, BLOCK_M)
        num_blocks_n = tl.cdiv(N, BLOCK_N)

        batch_idx = pid // (num_blocks_m * num_blocks_n)
        tile_idx = pid % (num_blocks_m * num_blocks_n)
        pid_m = tile_idx // num_blocks_n
        pid_n = tile_idx % num_blocks_n

        # offsets inside the tile
        offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
        offs_k = tl.arange(0, BLOCK_K)

        # base pointers for this batch
        a_base = a_ptr + batch_idx * batch_stride_a
        b_base = b_ptr + batch_idx * batch_stride_b
        c_base = c_ptr + batch_idx * batch_stride_c

        # pointers to tiles
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

        # accumulator
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # loop over K dimension
        for _ in range(0, tl.cdiv(K, BLOCK_K)):
            a = tl.load(a_ptrs,
                        mask=(offs_m < M)[:, None] & (offs_k < K)[None, :],
                        other=0.0,
                        dtype=tl.float16)
            b = tl.load(b_ptrs,
                        mask=(offs_k < K)[:, None] & (offs_n < N)[None, :],
                        other=0.0,
                        dtype=tl.float16)
            acc += tl.dot(a, b, out_dtype=tl.float32)

            a_ptrs += BLOCK_K * stride_a_k
            b_ptrs += BLOCK_K * stride_b_k

        # store result
        c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
        mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
        tl.store(c_ptrs, acc, mask=mask)

    # launch kernel
    trmul_kernel[grid](
        A, B_mat, C,
        M=N, N=N, K=N,
        batch_stride_a=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        batch_stride_b=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        batch_stride_c=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # out now holds the result of left @ right^T per channel
    # Apply final LayerNorm, gating and linear projection
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    # Multiply by out_gate (already sigmoid)
    out = out * out_gate

    # Final linear to original dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

"""
Implementation of the outgoing TriMul forward pass.

The core of TriMul is the tensor contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` are the projected and gated versions of the
input tensor `x`.  For each hidden dimension `d` (and each batch) this
contraction is exactly a matrix multiplication of the form
`L @ Rᵀ` with `L,R ∈ ℝ^{N×N}`.  The O(N³·C) work is therefore expressed
as a batched matmul over `C = batch_size·hidden_dim` independent
matrices.  The heavy lifting is performed by a custom Triton kernel that
processes each (batch,channel) pair as a separate matmul, using
half‑precision arithmetic for the loads and fp32 accumulation for
numerical stability.

All other operations (layer‑norms, linear projections, gating,
final projection) are carried out with PyTorch primitives.
"""
import torch
import triton
import triton.language as tl

def custom_kernel(data):
    # unpack input tuple
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask = mask.to(left.dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton kernel: batched matmul left @ rightᵀ per hidden channel
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # bring channel dimension to second axis so rows are contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # half‑precision for the compute‑intensive part
    left_fp16 = left_perm.to(torch.float16)
    right_fp16 = right_perm.to(torch.float16)

    # output buffer
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # flatten batch & channel dimensions → (B*H, N, N)
    A = left_fp16.view(-1, N, N)
    B_mat = right_fp16.view(-1, N, N)
    C = out.view(-1, N, N)

    # strides in element units
    stride_a_batch = A.stride(0)
    stride_a_m = A.stride(1)   # row stride (N)
    stride_a_k = A.stride(2)   # col stride (1)

    stride_b_batch = B_mat.stride(0)
    stride_b_k = B_mat.stride(2)   # col stride = 1
    stride_b_n = B_mat.stride(1)   # row stride = N

    stride_c_batch = C.stride(0)
    stride_c_m = C.stride(1)   # row stride (N)
    stride_c_n = C.stride(2)   # col stride (1)

    total_batches = A.shape[0]                # B * hidden_dim
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = total_batches * num_blocks_m * num_blocks_n

    @triton.jit
    def trmul_kernel(
        a_ptr, b_ptr, c_ptr,
        M, N, K,
        batch_stride_a, stride_a_m, stride_a_k,
        batch_stride_b, stride_b_k, stride_b_n,
        batch_stride_c, stride_c_m, stride_c_n,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
    ):
        pid = tl.program_id(0)

        # compute which batch and which tile this program processes
        num_blocks_m = tl.cdiv(M, BLOCK_M)
        num_blocks_n = tl.cdiv(N, BLOCK_N)

        batch_idx = pid // (num_blocks_m * num_blocks_n)
        tile_idx = pid % (num_blocks_m * num_blocks_n)
        pid_m = tile_idx // num_blocks_n
        pid_n = tile_idx % num_blocks_n

        # tile offsets
        offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
        offs_k = tl.arange(0, BLOCK_K)

        # base pointers for the selected batch
        a_base = a_ptr + batch_idx * batch_stride_a
        b_base = b_ptr + batch_idx * batch_stride_b
        c_base = c_ptr + batch_idx * batch_stride_c

        # pointers to the tiles (A is [M,K], Bᵀ is [K,N])
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # loop over K dimension in blocks
        for _ in range(0, tl.cdiv(K, BLOCK_K)):
            a = tl.load(a_ptrs,
                        mask=(offs_m < M)[:, None] & (offs_k < K)[None, :],
                        other=0.0,
                        dtype=tl.float16)
            b = tl.load(b_ptrs,
                        mask=(offs_k < K)[:, None] & (offs_n < N)[None, :],
                        other=0.0,
                        dtype=tl.float16)
            acc += tl.dot(a, b, out_dtype=tl.float32)

            a_ptrs += BLOCK_K * stride_a_k
            b_ptrs += BLOCK_K * stride_b_k

        # write back the result
        c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
        mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
        tl.store(c_ptrs, acc, mask=mask)

    # launch the kernel
    trmul_kernel[grid](
        A, B_mat, C,
        M=N, N=N, K=N,
        batch_stride_a=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        batch_stride_b=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        batch_stride_c=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6. Post‑processing: LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
