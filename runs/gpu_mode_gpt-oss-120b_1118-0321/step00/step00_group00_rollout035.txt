@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    BATCH,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)  # flatten across batch and blocks
    # Compute batch ID
    batch_stride_m = tl.cdiv(M, BLOCK_M)
    batch_stride_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = batch_stride_m * batch_stride_n
    batch = pid // blocks_per_batch
    block_id = pid % blocks_per_batch
    pid_m = block_id // batch_stride_n
    pid_n = block_id % batch_stride_n

    # Use offsets similar to example.
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Compute pointer offsets with batch offset
    a_ptrs = a_ptr + batch * stride_am * M * K + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
    b_ptrs = b_ptr + batch * stride_bk * N * K + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)
    # accumulation
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, K, BLOCK_K):
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (offs_k[None, :] < min(K - k, BLOCK_K)), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < min(K - k, BLOCK_K)) & (offs_n[None, :] < N), other=0.0)
        acc += tl.dot(a, b)
        # increment pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk
    # Store
    c_ptrs = c_ptr + batch * stride_cm * M * N + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))

def bmm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_aa0, stride_aa1, stride_aa2,  # a: batch, M, K
    stride_bb0, stride_bb1, stride_bb2,  # b: batch, N, K (if transposed)
    stride_cc0, stride_cc1, stride_cc2,
    BATCH,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    batch_size = BATCH
    # Compute block indices
    num_m_blocks = tl.cdiv(M, BLOCK_M)
    num_n_blocks = tl.cdiv(N, BLOCK_N)
    num_blocks = num_m_blocks * num_n_blocks
    batch = pid // num_blocks
    block_id = pid % num_blocks
    pid_m = block_id // num_n_blocks
    pid_n = block_id % num_n_blocks

    # Offsets within batch
    off_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    off_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    off_k = tl.arange(0, BLOCK_K)

    # Base pointers for this batch
    a_batch_ptr = a_ptr + batch * stride_aa0
    b_batch_ptr = b_ptr + batch * stride_bb0
    c_batch_ptr = c_ptr + batch * stride_cc0

    # Pointers for A and B tiles
    a_ptrs = a_batch_ptr + off_m[:, None] * stride_aa1 + off_k[None, :] * stride_aa2
    b_ptrs = b_batch_ptr + off_k[:, None] * stride_bb1 + off_n[None, :] * stride_bb2

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        a = tl.load(a_ptrs, mask=(off_m[:, None] < M) & (off_k[None, :] < min(K - k, BLOCK_K)), other=0.0)
        b = tl.load(b_ptrs, mask=(off_k[:, None] < min(K - k, BLOCK_K)) & (off_n[None, :] < N), other=0.0)
        acc += tl.dot(a, b)
        a_ptrs += BLOCK_K * stride_aa2
        b_ptrs += BLOCK_K * stride_bb1
    # Store
    c_ptrs = c_batch_ptr + off_m[:, None] * stride_cc1 + off_n[None, :] * stride_cc2
    tl.store(c_ptrs, acc.to(tl.float16), mask=(off_m[:, None] < M) & (off_n[None, :] < N))

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # LayerNorm on input
    eps = 1e-5
    # Using torch.nn.functional.layer_norm
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps
    ).to(torch.float16)  # cast to fp16 for heavy compute

    # Linear projections
    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)

    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape B,N,N,hidden_dim
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # mask handling
    if mask is None:
        mask_expand = torch.ones_like(left, dtype=left.dtype)
    else:
        mask_expand = mask.unsqueeze(-1).to(left.dtype)
    left = left * mask_expand
    right = right * mask_expand

    # gates
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Prepare for batched matmul
    B, N, _, _ = left.shape
    # Permute to bring hidden dim to batch dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    batch = B * hidden_dim
    left_mat = left_perm.view(batch, N, N)  # (batch, M=N, K=N)
    right_mat = right_perm.view(batch, N, N)  # (batch, N, N)
    # Transpose right for dot
    right_mat_t = right_mat.transpose(-2, -1).contiguous()

    # allocate output
    out_mat = torch.empty(batch, N, N, dtype=torch.float16, device=device)

    # launch kernel
    import triton
    import triton.language as tl

    # define kernel (we'll define it outside of function or inside caching global).

    # compute grid
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    num_m_blocks = (N + BLOCK_M - 1) // BLOCK_M
    num_n_blocks = (N + BLOCK_N - 1) // BLOCK_N
    grid = batch * num_m_blocks * num_n_blocks

    # get strides
    a_batch_stride = left_mat.stride(0)
    a_m_stride = left_mat.stride(1)
    a_k_stride = left_mat.stride(2)

    b_batch_stride = right_mat_t.stride(0)
    b_k_stride = right_mat_t.stride(1)
    b_n_stride = right_mat_t.stride(2)

    c_batch_stride = out_mat.stride(0)
    c_m_stride = out_mat.stride(1)
    c_n_stride = out_mat.stride(2)

    bmm_kernel[(grid,)](
        left_mat,
        right_mat_t,
        out_mat,
        N, N, N,
        a_batch_stride, a_m_stride, a_k_stride,
        b_batch_stride, b_k_stride, b_n_stride,
        c_batch_stride, c_m_stride, c_n_stride,
        batch,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # Synchronize with torch.cuda.synchronize?
    torch.cuda.synchronize()

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # Apply to_out_norm
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    # Multiply by out_gate (cast to fp32)
    out_gate_fp32 = out_gate.to(torch.float32)
    out = out * out_gate_fp32

    # Final linear
    to_out_weight = weights["to_out.weight"]
    output = torch.nn.functional.linear(out, to_out_weight)  # shape [B,N,N,dim]

    return output

def bmm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_aa0, stride_aa1, stride_aa2,
    stride_bb0, stride_bb1, stride_bb2,
    stride_cc0, stride_cc1, stride_cc2,
    BATCH,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    batch = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    block_id = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_m = block_id // tl.cdiv(N, BLOCK_N)
    pid_n = block_id % tl.cdiv(N, BLOCK_N)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Compute base pointers with batch offset
    a_batch_ptr = a_ptr + batch * stride_aa0
    b_batch_ptr = b_ptr + batch * stride_bb0
    c_batch_ptr = c_ptr + batch * stride_cc0

    a_ptrs = a_batch_ptr + offs_m[:, None] * stride_aa1 + offs_k[None, :] * stride_aa2
    b_ptrs = b_batch_ptr + offs_k[:, None] * stride_bb1 + offs_n[None, :] * stride_bb2

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, K, BLOCK_K):

        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] + k < K)

"""
TriMul outgoing kernel: Implements the forward pass of the TriMul module
using a fused Triton batched matrix multiplication for the cubic
`einsum(... i k d, ... j k d -> ... i j d)`.
All other operations (LayerNorm, linear projections, gating, mask) are
performed with PyTorch.  The heavy N³ reduction is executed in a custom
Triton kernel that computes, for each hidden channel, C = A @ Bᵀ where
A, B have shape (batch*hidden, N, N).  The kernel works in FP16 for the
input matrices but accumulates in FP32, and writes the result directly in
FP32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BATCH,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    # number of blocks in each dimension
    num_m_blocks = tl.cdiv(M, BLOCK_M)
    num_n_blocks = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_m_blocks * num_n_blocks

    batch = pid // blocks_per_batch
    block_id = pid % blocks_per_batch
    pid_m = block_id // num_n_blocks
    pid_n = block_id % num_n_blocks

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # base pointers for this batch
    a_batch = a_ptr + batch * stride_a_batch
    b_batch = b_ptr + batch * stride_b_batch
    c_batch = c_ptr + batch * stride_c_batch

    # pointers to the beginning of the tile
    a_ptrs = a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
    b_ptrs = b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k_start in range(0, K, BLOCK_K):
        cur_k = k_start
        k_left = K - cur_k
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < k_left)
        mask_b = (offs_k[:, None] < k_left) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        acc += tl.dot(a, b)   # fp32 accumulation

        # advance pointers by BLOCK_K along K dimension
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # write back
    c_ptrs = c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim] float32
            - mask: [B, N, N] (bool or float)
            - weights: dict with all linear and LayerNorm parameters
            - config: dict containing 'dim' and 'hidden_dim'

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------- 1. LayerNorm over the last dimension ----------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # fp32

    # ---------- 2. Cast to fp16 for the heavy matrix ops ----------
    x_norm_fp16 = x_norm.to(torch.float16)

    # ---------- 3. Linear projections ----------
    left = F.linear(x_norm_fp16, weights["left_proj.weight"].to(torch.float16))   # [B,N,N,hd]
    right = F.linear(x_norm_fp16, weights["right_proj.weight"].to(torch.float16))

    # ---------- 4. Optional mask ----------
    if mask is None:
        mask_exp = torch.ones_like(left, dtype=left.dtype, device=device)
    else:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
    left = left * mask_exp
    right = right * mask_exp

    # ---------- 5. Gating ----------
    left_gate = F.linear(x_norm_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_norm_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_norm, weights["out_gate.weight"]).sigmoid()   # keep out_gate in fp32 for stability

    left = left * left_gate
    right = right * right_gate

    # ---------- 6. Cubic reduction via batched GEMM (Triton) ----------
    B, N, _, _ = left.shape
    # bring hidden dimension into the batch axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # B, hd, N, N
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    batch = B * hidden_dim
    left_mat = left_perm.view(batch, N, N)          # (batch, M=N, K=N)   fp16
    right_mat = right_perm.view(batch, N, N)        # (batch, N, N)       fp16
    right_mat_T = right_mat.transpose(-2, -1).contiguous()   # (batch, K=N, N)

    out_mat = torch.empty(batch, N, N, dtype=torch.float32, device=device)

    # Gather strides
    stride_a_batch = left_mat.stride(0)
    stride_a_m = left_mat.stride(1)
    stride_a_k = left_mat.stride(2)

    stride_b_batch = right_mat_T.stride(0)
    stride_b_k = right_mat_T.stride(1)
    stride_b_n = right_mat_T.stride(2)

    stride_c_batch = out_mat.stride(0)
    stride_c_m = out_mat.stride(1)
    stride_c_n = out_mat.stride(2)

    # Tile sizes; these were tuned for typical N up to 1024
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_m_blocks = (N + BLOCK_M - 1) // BLOCK_M
    num_n_blocks = (N + BLOCK_N - 1) // BLOCK_N
    grid = batch * num_m_blocks * num_n_blocks

    batched_matmul_kernel[
        (grid,)
    ](
        left_mat,
        right_mat_T,
        out_mat,
        N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Ensure kernel completion
    torch.cuda.synchronize()

    # ---------- 7. Reshape back to [B,N,N,hidden_dim] ----------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # fp32

    # ---------- 8. Output norm, gate and final linear ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate   # both fp32

    output = F.linear(out, weights["to_out.weight"])   # shape [B,N,N,dim]
    return output
