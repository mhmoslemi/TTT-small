"""
custom_kernel:  TriMul “outgoing” implementation (AlphaFold3).
---------------------------------------------------------------
Algorithm
1. LayerNorm on the input tensor.
2. Linear projections to a hidden dimension (left/right) and three gating
   projections (left_gate, right_gate, out_gate).  Sigmoid is applied to the
   gates.
3. Apply mask and the two gates element‑wise to the projected tensors.
   This step is fused into a tiny Triton kernel (tensor * gate * mask).
4. Perform the core O(N³·D) contraction:
        out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
   which is exactly a batched matrix‑multiply of shape
        (B·D, N, N) @ (B·D, N, N)ᵀ.
   Using torch.bmm after reshaping gives the same result with a highly
   optimized cuBLAS kernel.
5. LayerNorm + out_gate gating on the result.
6. Final linear projection back to the original dimension.

Only step 3 is performed in Triton; all heavy linear / matmul work
leverages PyTorch’s fast CUDA kernels.
"""

import triton
import triton.language as tl
from typing import Tuple, Dict

# Triton kernel: elementwise (tensor * gate * mask)  -->  out
# ----------------------------------------------------------------------

def _apply_gate_mask_kernel(

    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_d,

    j = tl.program_id(2)

    )

    tensor_val = tl.load(tensor_ptr + tensor_offset, mask=d_mask, other=0.0)
    gate_val   = tl.load(gate_ptr   + gate_offset,   mask=d_mask, other=0.0)

    # --------------------------------------------------------------
    # Fuse the three multiplicands
    # --------------------------------------------------------------
    out_val = tensor_val * gate_val * mask_val

    tl.store(out_ptr + tensor_offset, out_val, mask=d_mask)


# ----------------------------------------------------------------------
# Main entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]  (float32)
        - mask         : [B, N, N] (float32 or bool). May be None.
        - weights      : dict of model parameters (see code body)
        - config       : dict containing at least ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, N, N2, dim = input_tensor.shape
    assert N == N2, "Input must be square in the sequence dimensions"

    hidden_dim = config["hidden_dim"]
    assert hidden_dim == 128, "Current implementation assumes hidden_dim = 128"

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    # ------------------------------------------------------------------
    # 4. Optional mask (if nomask, use all‑ones)
    # ------------------------------------------------------------------
    if mask is None:
        mask = torch.ones(bs, N, N, device=device, dtype=dtype)
    else:
        # ensure float, same device
        mask = mask.to(dtype=dtype, device=device)

    # ------------------------------------------------------------------
    # 5. Fuse mask & gates into left & right via Triton
    # ------------------------------------------------------------------
    # Make everything contiguous (required for correct stride handling)
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    out_gate = out_gate.contiguous()
    mask = mask.contiguous()

    # strides (in element units)
    stride_left_b, stride_left_i, stride_left_j, stride_left_d = left.stride()
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_d = left_gate.stride()
    stride_mask_b, stride_mask_i, stride_mask_j = mask.stride()
    stride_right_b, stride_right_i, stride_right_j, stride_right_d = right.stride()
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_d = right_gate.stride()

    # launch grid: (batch, seq_len, seq_len)
    grid = (bs, N, N)

    # left = left * left_gate * mask
    _apply_gate_mask_kernel[grid](
        left.data_ptr(),
        left_gate.data_ptr(),
        mask.data_ptr(),
        left.data_ptr(),
        B=bs,
        N=N,
        D=hidden_dim,
        stride_tensor_b=stride_left_b,
        stride_tensor_i=stride_left_i,
        stride_tensor_j=stride_left_j,
        stride_tensor_d=stride_left_d,
        stride_gate_b=stride_gate_b,
        stride_gate_i=stride_gate_i,
        stride_gate_j=stride_gate_j,
        stride_gate_d=stride_gate_d,
        stride_mask_b=stride_mask_b,
        stride_mask_i=stride_mask_i,
        stride_mask_j=stride_mask_j,
        BLOCK_D=hidden_dim,
    )

    # right = right * right_gate * mask
    _apply_gate_mask_kernel[grid](
        right.data_ptr(),
        right_gate.data_ptr(),
        mask.data_ptr(),
        right.data_ptr(),
        B=bs,
        N=N,
        D=hidden_dim,
        stride_tensor_b=stride_right_b,
        stride_tensor_i=stride_right_i,
        stride_tensor_j=stride_right_j,
        stride_tensor_d=stride_right_d,
        stride_gate_b=stride_rgate_b,
        stride_gate_i=stride_rgate_i,
        stride_gate_j=stride_rgate_j,
        stride_gate_d=stride_rgate_d,
        stride_mask_b=stride_mask_b,
        stride_mask_i=stride_mask_i,
        stride_mask_j=stride_mask_j,
        BLOCK_D=hidden_dim,
    )

    # ------------------------------------------------------------------
    # 6. Core contraction: einsum → batched matmul
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # Rearrange to (B, D, N, N) then merge B and D for torch.bmm
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, D, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, D, N, N)

    left_mat = left_perm.view(bs * hidden_dim, N, N)    # (B*D, N, N)
    right_mat = right_perm.view(bs * hidden_dim, N, N)  # (B*D, N, N)

    # Batched matrix multiply with right transposed (k ↔ j)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*D, N, N)

    # Restore original layout
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1)  # (B, N, N, D)

    # ------------------------------------------------------------------
    # 7. Output LayerNorm + out_gate
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
