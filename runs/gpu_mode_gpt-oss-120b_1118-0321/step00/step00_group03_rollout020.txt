Define kernel:

@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    B, M, N, K, D,
    stride_Ab, stride_Ai, stride_Ak, stride_Ad,
    stride_Bb, stride_Bk, stride_Bj, stride_Bd,
    stride_Cb, stride_Ci, stride_Cj, stride_Cd,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute batch and hidden indices
    b_idx = batch_idx // D
    d_idx = batch_idx - b_idx * D

    # base pointers for this batch
    a_batch_ptr = a_ptr + b_idx * stride_Ab + d_idx * stride_Ad
    b_batch_ptr = b_ptr + b_idx * stride_Bb + d_idx * stride_Bd
    c_batch_ptr = c_ptr + b_idx * stride_Cb + d_idx * stride_Cd

    # mask out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        # load A and B tiles, with masking
        a = tl.load(
            a_batch_ptr + offs_m[:, None] * stride_Ai + offs_k[None, :] * stride_Ak,
            mask=mask_m[:, None] & (offs_k < K),
            other=0.0
        )
        b = tl.load(
            b_batch_ptr + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bj,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0
        )
        # dot product
        acc += tl.dot(a, b)

    # write back results, with masks for M,N
    c = acc.to(tl.float16)
    tl.store(
        c_batch_ptr + offs_m[:, None] * stride_Ci + offs_n[None, :] * stride_Cj,
        c,
        mask=mask_m[:, None] & mask_n[None, :]
    )

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    # dims
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=1e-5)

    # Linear proj
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Mask
    if not config.get("nomask", False) and mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)  # shape [bs,N,N,1]
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Permute right for matmul
    right_T_fp16 = right_fp16.permute(0, 2, 1, 3).contiguous()  # ensure contiguous maybe not needed

    # Output tensor (fp16)
    out_fp16 = torch.empty((bs, N, N, hidden_dim), dtype=torch.float16, device=input_tensor.device)

    # Strides
    a_ptr = left_fp16
    b_ptr = right_T_fp16
    c_ptr = out_fp16

    stride_Ab = left_fp16.stride(0)
    stride_Ai = left_fp16.stride(1)
    stride_Ak = left_fp16.stride(2)
    stride_Ad = left_fp16.stride(3)

    stride_Bb = right_T_fp16.stride(0)
    stride_Bk = right_T_fp16.stride(1)
    stride_Bj = right_T_fp16.stride(2)
    stride_Bd = right_T_fp16.stride(3)

    stride_Cb = out_fp16.stride(0)
    stride_Ci = out_fp16.stride(1)
    stride_Cj = out_fp16.stride(2)
    stride_Cd = out_fp16.stride(3)

    total_batches = bs * hidden_dim

    # Choose block sizes based on N
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (total_batches, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

    batched_gemm_kernel[grid](
        a_ptr,
        b_ptr,
        c_ptr,
        N,  # M
        N,  # N
        N,  # K
        hidden_dim,
        stride_Ab, stride_Ai, stride_Ak, stride_Ad,
        stride_Bb, stride_Bk, stride_Bj, stride_Bd,
        stride_Cb, stride_Ci, stride_Cj, stride_Cd,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert output back to fp32
    out = out_fp16.to(torch.float32)

    # to_out_norm and out_gate (already fp32)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=1e-5)

    out = out * out_gate

    out = torch.nn.functional.linear(out, weights["to_out.weight"])

@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K, D,
    stride_Ab, stride_Ai, stride_Ak, stride_Ad,
    stride_Bb, stride_Bk, stride_Bj, stride_Bd,
    stride_Cb, stride_Ci, stride_Cj, stride_Cd,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets for block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out-of-range rows/cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Calculate batch and hidden indices
    d_idx = batch_idx % D
    b_idx = batch_idx // D

    # Base pointers for this batch+hidden
    a_batch = a_ptr + b_idx * stride_Ab + d_idx * stride_Ad
    b_batch = b_ptr + b_idx * stride_Bb + d_idx * stride_Bd
    c_batch = c_ptr + b_idx * stride_Cb + d_idx * stride_Cd

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension

"""
TriMul (outgoing) forward pass with a custom Triton kernel.
The heavy O(N³·D) multiplication is implemented as a batched
matrix multiplication per head (batch * hidden_dim) using
fp16 tensor‑core arithmetic, while the surrounding LayerNorm,
linear projections, and gates are handled in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K, D,
    stride_Ab, stride_Ai, stride_Ak, stride_Ad,
    stride_Bb, stride_Bk, stride_Bj, stride_Bd,
    stride_Cb, stride_Ci, stride_Cj, stride_Cd,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C[b,d] = A[b,d] @ B[b,d].  A and B are (N,N) matrices,
    C is (N,N).  The batch dimension here is (batch * hidden_dim)."""
    batch_idx = tl.program_id(0)          # combined (batch, hidden)
    pid_m = tl.program_id(1)              # row tile id
    pid_n = tl.program_id(2)              # col tile id

    # Offsets for the current output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Predicates for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Decompose the combined batch index into (batch, hidden)
    d_idx = batch_idx % D
    b_idx = batch_idx // D

    # Base pointers for the selected (batch, hidden) slice
    a_batch = a_ptr + b_idx * stride_Ab + d_idx * stride_Ad
    b_batch = b_ptr + b_idx * stride_Bb + d_idx * stride_Bd
    c_batch = c_ptr + b_idx * stride_Cb + d_idx * stride_Cd

    # Accumulator in float32 (tensor‑core accumulation)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension (the shared sequence axis)
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_batch + offs_m[:, None] * stride_Ai + k_off[None, :] * stride_Ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile: shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_batch + k_off[:, None] * stride_Bk + offs_n[None, :] * stride_Bj,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Tensor‑core dot (fp16 inputs → fp32 accumulation)
        acc += tl.dot(a, b)

    # Store result back as fp16
    c = acc.to(tl.float16)
    tl.store(
        c_batch + offs_m[:, None] * stride_Ci + offs_n[None, :] * stride_Cj,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.
    Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    # ------------------------------------------------------------------
    # 1. Extract dimensions / configuration
    # ------------------------------------------------------------------
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]          # D in the paper

    # ------------------------------------------------------------------
    # 2. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 3. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 4. Optional mask (skip if config["nomask"] is True)
    # ------------------------------------------------------------------
    if not config.get("nomask", False) and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)           # [bs, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5. Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Core O(N³·D) multiplication – Triton batched GEMM
    # ------------------------------------------------------------------
    # Cast to fp16 for tensor‑core speed
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Right operand needs to be transposed on the two sequence axes
    right_T_fp16 = right_fp16.permute(0, 2, 1, 3).contiguous()

    # Output buffer
    out_fp16 = torch.empty((bs, N, N, hidden_dim), dtype=torch.float16, device=input_tensor.device)

    # Strides (in elements)
    stride_Ab, stride_Ai, stride_Ak, stride_Ad = left_fp16.stride()
    stride_Bb, stride_Bk, stride_Bj, stride_Bd = right_T_fp16.stride()
    stride_Cb, stride_Ci, stride_Cj, stride_Cd = out_fp16.stride()

    total_batches = bs * hidden_dim

    # Tunable tile sizes – work well for N up to 1024
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        total_batches,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_gemm_kernel[grid](
        left_fp16,
        right_T_fp16,
        out_fp16,
        N,          # M
        N,          # N
        N,          # K (shared sequence length)
        hidden_dim, # D (hidden)
        stride_Ab,
        stride_Ai,
        stride_Ak,
        stride_Ad,
        stride_Bb,
        stride_Bk,
        stride_Bj,
        stride_Bd,
        stride_Cb,
        stride_Ci,
        stride_Cj,
        stride_Cd,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Cast back to fp32 for the remaining ops
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 7. Final normalization, output gating and projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
