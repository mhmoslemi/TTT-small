import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_bhd = tl.program_id(0)  # batch_hidden index
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute batch and hidden from pid_bhd
    hidden = hidden_dim
    b = pid_bhd // hidden
    d = pid_bhd % hidden

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for valid rows/cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers for A (left) tile (M,K)
        a_ptrs = left_ptr + b * stride_left_b + d * stride_left_d \
                 + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k

        # pointers for B_T tile (K,N) (using right transposed)
        b_ptrs = right_ptr + b * stride_right_b + d * stride_right_d \
                 + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_i

        # load
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # dot
        acc += tl.dot(a, b)

    # store
    c_ptrs = out_ptr + b * stride_out_b + d * stride_out_d \
             + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j

    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched matrix multiplication for the TriMul operation.
    Computes, for each batch b and hidden index d:
        out[b, i, j, d] = sum_k left[b, i, k, d] * right[b, j, k, d]
    using tiled GEMM with float32 accumulation.
    """

    pid_bhd = tl.program_id(0)   # combined batch * hidden index
    pid_m = tl.program_id(1)     # block row
    pid_n = tl.program_id(2)     # block column

    # Recover batch and hidden indices
    b = pid_bhd // hidden_dim
    d = pid_bhd % hidden_dim

    # Offsets for the output tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile: left[b, i, k, d]
        a_ptrs = left_ptr \
                 + b * stride_left_b + d * stride_left_d \
                 + offs_i[:, None] * stride_left_i \
                 + offs_k[None, :] * stride_left_k

        # Load B tile transposed: right[b, j, k, d] -> we need right[..., k, j]
        b_ptrs = right_ptr \
                 + b * stride_right_b + d * stride_right_d \
                 + offs_k[:, None] * stride_right_k \
                 + offs_j[None, :] * stride_right_i

        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Store the result
    c_ptrs = out_ptr \
             + b * stride_out_b + d * stride_out_d \
             + offs_i[:, None] * stride_out_i \
             + offs_j[None, :] * stride_out_j

    tl.store(c_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Forward pass of the TriMul operator (outgoing version).
    Implements the heavy N^3 reduction with a Triton kernel and the remaining
    linear / LayerNorm / gating operations with PyTorch.
    """
    # unpack inputs
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    # configuration parameters
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm on the last dimension (dim)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # shape [B, N, N, dim]

    # Linear projections (no bias)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask if present
    if not nomask and mask_tensor is not None:
        # mask shape [B, N, N] - broadcast to last dim
        mask = mask_tensor.to(dtype=left.dtype, device=device).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # Gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Cast to half precision for the heavy GEMM
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Output buffer for the intermediate result (float32 for accumulation)
    out_intermediate = torch.empty(
        (input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2], hidden_dim),
        dtype=torch.float32,
        device=device,
    )

    # Strides for the tensors (in elements, not bytes)
    s_left = left_fp16.stride()
    s_right = right_fp16.stride()
    s_out = out_intermediate.stride()

    # Tile sizes – chosen to balance occupancy and shared memory usage
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    B, N, _, _ = left_fp16.shape
    # grid: (batch*hidden, n_blocks_m, n_blocks_n)
    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Launch the kernel
    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_intermediate,
        B,
        N,
        hidden_dim,
        s_left[0],
        s_left[1],
        s_left[2],
        s_left[3],
        s_right[0],
        s_right[1],
        s_right[2],
        s_right[3],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Apply final LayerNorm, out_gate and projection back to dim
    out_norm = torch.nn.functional.layer_norm(
        out_intermediate,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out_norm = out_norm * out_gate
    out = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return out

    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched GEMM for the TriMul reduction.
    Computes, for each (b, d):
        out[b, i, j, d] = Σₖ left[b, i, k, d] * right[b, j, k, d]
    using tiled matrix multiplication with float32 accumulation.
    """
    pid_bhd = tl.program_id(0)   # combined batch * hidden index
    pid_m = tl.program_id(1)     # output row block
    pid_n = tl.program_id(2)     # output col block

    # decode batch and hidden indices
    b = pid_bhd // hidden_dim
    d = pid_bhd % hidden_dim

    # tile offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: left[b, i, k, d]
        a_ptrs = left_ptr \
                 + b * stride_left_b + d * stride_left_d \
                 + offs_i[:, None] * stride_left_i \
                 + offs_k[None, :] * stride_left_k

        # right tile transposed: we need right[b, j, k, d] → right_T[k, j, d]
        b_ptrs = right_ptr \
                 + b * stride_right_b + d * stride_right_d \
                 + offs_k[:, None] * stride_right_k \
                 + offs_j[None, :] * stride_right_i

        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # write out the tile
    c_ptrs = out_ptr \
             + b * stride_out_b + d * stride_out_d \
             + offs_i[:, None] * stride_out_i \
             + offs_j[None, :] * stride_out_j

    tl.store(c_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    The N³ reduction is executed on GPU via `tri_mul_kernel`; all other
    linear, LayerNorm, and gating operations use PyTorch.
    Input:
        data = (input_tensor, mask_tensor, weights, config)
    Output:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcasted on the feature dimension)
    # -----------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(dtype=left.dtype, device=device).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid of linear maps)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Heavy N³ reduction via Triton GEMM kernel
    # -----------------------------------------------------------------
    # Cast to fp16 for better bandwidth, kernel accumulates in fp32.
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape
    out_intermediate = torch.empty(
        (B, N, N, hidden_dim), dtype=torch.float32, device=device
    )

    # Tensor strides (in elements)
    s_left = left_fp16.stride()
    s_right = right_fp16.stride()
    s_out = out_intermediate.stride()

    # Tile sizes – modest values that fit in shared memory on B200.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_intermediate,
        B,
        N,
        hidden_dim,
        s_left[0],
        s_left[1],
        s_left[2],
        s_left[3],
        s_right[0],
        s_right[1],
        s_right[2],
        s_right[3],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating and final projection back to dim
    # -----------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out_intermediate,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out_norm = out_norm * out_gate
    out = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return out
