import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          M, N, K,
                          stride_am, stride_ai, stride_ak,
                          stride_bb, stride_bi, stride_bk,
                          stride_cm, stride_ci, stride_cj,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch_idx = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # compute mask for bounds
    mask_i = offs_i < M
    mask_j = offs_j < N
    mask_k = offs_k < K

    # Pointers for A and B
    A_ptrs = A_ptr + batch_idx * stride_am + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak
    B_ptrs = B_ptr + batch_idx * stride_bb + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bi

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, K, BLOCK_K):
        # load A block
        a = tl.load(A_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        # load B block (transposed)
        b = tl.load(B_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        # accumulate
        acc += tl.dot(a, b)
        # advance pointers
        A_ptrs += BLOCK_K * stride_ak

"""
TriMul forward pass (outgoing) with a fused Triton batched matmul.

Algorithm
---------
1. Layer‑norm the 4‑D input [B, N, N, C] using the provided
   `norm.weight` / `norm.bias`.
2. Linear projections (no bias) to the hidden dimension `H`:
       left  = x_norm @ left_proj_weight.T
       right = x_norm @ right_proj_weight.T
3. Optional pairwise mask (broadcast over `H`).
4. Gating:
       left_gate  = sigmoid(x_norm @ left_gate_weight.T)
       right_gate = sigmoid(x_norm @ right_gate_weight.T)
       out_gate   = sigmoid(x_norm @ out_gate_weight.T)
   Apply gates to `left` and `right`.
5. Core TriMul:
   For every batch `b` and hidden channel `h`,
   compute   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
   This is a batched matrix multiplication:
       out_flat[b*h] = left_flat[b*h] @ right_flat[b*h].T
   The heavy reduction over `k` is implemented in Triton
   (`batched_matmul_kernel`) using tiling over the i, j and k axes.
6. Layer‑norm over the hidden dimension (`to_out_norm`) and
   scale by `out_gate`.
7. Final linear projection back to `dim`:
       out = out_norm @ to_out_weight.T

The Triton kernel processes `B*H` independent GEMMs in parallel,
each of size `N×N`.  Blocking parameters are chosen to give good
occupancy on the B200 GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bb, stride_bi, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """C[b,i,j] = Σ_k A[b,i,k] * B[b,j,k]   (B is accessed as B[b,k,j])"""
    batch_idx = tl.program_id(0)      # 0‑th axis : batch * hidden
    pid_i = tl.program_id(1)          # 1‑st axis : i‑block
    pid_j = tl.program_id(2)          # 2‑nd axis : j‑block

    # Offsets within the block
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds loads
    mask_i = offs_i < M
    mask_j = offs_j < N
    mask_k = offs_k < K

    # Pointers for the current block of A and transposed B
    a_ptrs = A_ptr + batch_idx * stride_am + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak
    b_ptrs = B_ptr + batch_idx * stride_bb + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bi

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k‑dimension tiles
    for _ in range(0, K, BLOCK_K):
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        acc += tl.dot(a, b)                     # (M,K)·(K,N) → (M,N)
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Store the block result
    c_ptrs = C_ptr + batch_idx * stride_cm + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask        : torch.Tensor [B, N, N] (may be None)
        - weights     : dict of torch.Tensors containing model weights
        - config      : dict with at least ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # Extract config
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Extract weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Layer‑norm over the last dimension
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections to hidden_dim
    # ------------------------------------------------------------------
    # torch.nn.functional.linear does: out = input @ weight.T + bias
    left = F.linear(x_norm, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x_norm, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Apply optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core TriMul via Triton batched matmul
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # Move hidden_dim to a separate batch dimension for the kernel
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Collapse batch and hidden_dim into one dimension
    batch_h = B * hidden_dim
    left_flat = left_perm.view(batch_h, N, N)
    right_flat = right_perm.view(batch_h, N, N)

    # Allocate output buffer
    out_flat = torch.empty_like(left_flat, device=device)

    # Triton launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        batch_h,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output layer‑norm and gating
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 7) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    output = F.linear(out_norm, to_out_weight)   # [B, N, N, dim]

    return output
