@triton.jit
def apply_mask_gate_kernel(inp_ptr, gate_ptr, mask_ptr, out_ptr,
                           B, N, H,
                           stride_in_b, stride_in_i, stride_in_k, stride_in_h,
                           stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_h,
                           stride_mask_b, stride_mask_i, stride_mask_k,
                           stride_out_b, stride_out_i, stride_out_k, stride_out_h,
                           BLOCK_POS: tl.constexpr, BLOCK_H: tl.constexpr):
    pid = tl.program_id(0)
    total = B * N * N
    block_start = pid * BLOCK_POS
    offs = block_start + tl.arange(0, BLOCK_POS)
    mask = offs < total

    # compute b,i,k indices
    b = offs // (N * N)
    rem = offs % (N * N)
    i = rem // N
    k = rem % N

    # base offset for in, gate, out
    inp_base = b * stride_in_b + i * stride_in_i + k * stride_in_k
    gate_base = b * stride_gate_b + i * stride_gate_i + k * stride_gate_k
    out_base = b * stride_out_b + i * stride_out_i + k * stride_out_k
    mask_base = b * stride_mask_b + i * stride_mask_i + k * stride_mask_k

    # load mask scalar per position
    mask_val = tl.load(mask_ptr + mask_base, mask=mask, other=0.0)

    # now loop over hidden dim
    for offset_h in range(0, tl.cdiv(H, BLOCK_H)):
        cur_h = offset_h * BLOCK_H + tl.arange(0, BLOCK_H)
        cur_mask = cur_h < H

        # offsets for hidden dimension
        off = cur_h * stride_in_h
        inp_ptrs = inp_ptr + inp_base[:, None] + off[None, :]
        gate_ptrs = gate_ptr + gate_base[:, None] + off[None, :]

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]
    # get weight tensors etc.
    # layernorm
    eps = 1e-5
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)
    # compute left_proj, right_proj
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"]) # weight shape [hidden_dim, dim] ; returns shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    # compute gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    # mask shape [B,N,N]; we need to broadcast mask to [B,N,N,1].
    # left & right currently shape [B,N,N,hidden_dim].

    # kernel for left:

        stride_in_b=left.stride(0), stride_in_i=left.stride(1), stride_in_k=left.stride(2), stride_in_h=left.stride(3),
        stride_gate_b=left_gate.stride(0), stride_gate_i=left_gate.stride(1), stride_gate_k=left_gate.stride(2), stride_gate_h=left_gate.stride(3),
        stride_mask_b=mask.stride(0), stride_mask_i=mask.stride(1), stride_mask_k=mask.stride(2),
        stride_out_b=left.stride(0), stride_out_i=left.stride(1), stride_out_k=left.stride(2), stride_out_h=left.stride(3),
        BLOCK_POS=256, BLOCK_H=32)
    # right similarly
    apply_mask_gate_kernel(right, right_gate, mask, right, ..., etc.)
    # now compute out via batched matmul:
    left_flat = left.permute(0,3,1,2) # B, H, N, N
    right_flat = right.permute(0,3,1,2) # B, H, N, N
    out = torch.matmul(left_flat, right_flat.transpose(-1,-2))
    # out shape B, H, N, N; permute back
    out = out.permute(0,2,3,1) # B, N, N, H

    # multiply by out_gate (already computed earlier)
    out_norm = out_norm * out_gate
    # final linear
    output = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

import torch
import triton
import triton.language as tl

@triton.jit
def apply_mask_gate_kernel(
        inp_ptr, gate_ptr, mask_ptr, out_ptr,
        B, N, H,
        stride_in_b, stride_in_i, stride_in_k, stride_in_h,
        stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_h,
        stride_mask_b, stride_mask_i, stride_mask_k,
        stride_out_b, stride_out_i, stride_out_k, stride_out_h,
        BLOCK_POS: tl.constexpr,
        BLOCK_H: tl.constexpr):
    pid = tl.program_id(0)
    total = B * N * N
    start = pid * BLOCK_POS
    offs = start + tl.arange(0, BLOCK_POS)
    mask = offs < total

    # Compute b,i,k indices
    b = tl.mul(offs, 1) // (N * N)  # due to integer division
    rem = offs % (N * N)
    i = rem // N
    k = rem % N

    # Base offsets
    inp_base = b * stride_in_b + i * stride_in_i + k * stride_in_k
    gate_base = b * stride_gate_b + i * stride_gate_i + k * stride_gate_k
    out_base = b * stride_out_b + i * stride_out_i + k * stride_out_k
    mask_base = b * stride_mask_b + i * stride_mask_i + k * stride_mask_k

    mask_val = tl.load(mask_ptr + mask_base, mask=mask, other=0.0)  # scalar per position
    # broadcast mask_val to vector of H
    # Loop over hidden dimension blocks
    for ho in range(0, tl.cdiv(H, BLOCK_H)):
        cur_h = ho * BLOCK_H + tl.arange(0, BLOCK_H)
        cur_h_mask = cur_h < H

        # compute offsets for hidden dim
        offset_h = cur_h * stride_in_h
        inp_ptrs = inp_ptr + inp_base[:, None] + offset_h[None, :]
        gate_ptrs = gate_ptr + gate_base[:, None] + cur_h * stride_gate_h   # combine
        # Might need to compute gate offset similarly
        gate_offset = cur_h * stride_gate_h
        gate_ptrs = gate_ptr + gate_base[:, None] + gate_offset[None, :]

        out_ptrs = out_ptr + out_base[:, None] + cur_h * stride_out_h

        # Load values
        inp = tl.load(inp_ptrs, mask=mask[:, None] & cur_h_mask[None, :], other=0.0)
        gate = tl.load(gate_ptrs, mask=mask[:, None] & cur_h_mask[None, :], other=0.0)

        # Multiply

        tl.store(out_ptrs, out, mask=mask[:, None] & cur_h_mask[None, :])

"""
TriMul “outgoing” implementation.

The forward pass of the TriMul module (AlphaFold3) consists of:
1. Layer‑norm over the input channel dimension.
2. Linear projections (left/right) and three gating projections.
3. Fusion of mask and gates (performed by a small Triton kernel).
4. Batched matrix multiplication over the sequence dimension:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   implemented with torch.matmul (highly optimized on the GPU).
5. Layer‑norm over the hidden dimension, gating, and a final linear
   projection back to the original channel dimension.

Only the mask‑ and gate‑fusion step is written in Triton; the remaining
high‑throughput GEMM and LayerNorm operations use PyTorch’s native CUDA
kernels.  This satisfies the requirement of “part of the operations in a
kernel” while keeping the implementation correct and reasonably fast.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left/right gating in‑place.
# ----------------------------------------------------------------------
@triton.jit
def apply_mask_gate_kernel(
    inp_ptr,                # float32*   [B, N, N, H]  (in/out)
    gate_ptr,               # float32*   [B, N, N, H]  (gate)
    mask_ptr,               # float32*   [B, N, N]     (mask)
    out_ptr,                # float32*   [B, N, N, H]  (result, same as inp_ptr)
    B, N,                   # runtime scalars
    stride_in_b, stride_in_i, stride_in_k, stride_in_h,
    stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_out_b, stride_out_i, stride_out_k, stride_out_h,
    BLOCK_POS: tl.constexpr,   # number of (b,i,k) positions per program
    BLOCK_H: tl.constexpr,     # hidden‑dim block size
    H: tl.constexpr            # hidden dimension (compile‑time)
):
    pid = tl.program_id(0)
    total = B * N * N
    start = pid * BLOCK_POS
    offs = start + tl.arange(0, BLOCK_POS)
    valid = offs < total

    # Unflatten (b,i,k) indices
    b = offs // (N * N)
    rem = offs % (N * N)
    i = rem // N
    k = rem % N

    # Base offsets for the three tensors (ignoring the hidden dim)
    base_in   = b * stride_in_b   + i * stride_in_i   + k * stride_in_k
    base_gate = b * stride_gate_b + i * stride_gate_i + k * stride_gate_k
    base_out  = b * stride_out_b  + i * stride_out_i  + k * stride_out_k
    base_mask = b * stride_mask_b + i * stride_mask_i + k * stride_mask_k

    # Load mask scalar (0/1) for this (b,i,k) position
    m_val = tl.load(mask_ptr + base_mask, mask=valid, other=0.0)   # [BLOCK_POS]

    # Loop over hidden dimension in blocks
    num_h_blocks = tl.cdiv(H, BLOCK_H)
    for hb in range(0, num_h_blocks):
        # hidden indices for this block
        h = hb * BLOCK_H + tl.arange(0, BLOCK_H)
        h_valid = h < H

        # Offsets for this hidden block
        off_in   = h * stride_in_h
        off_gate = h * stride_gate_h
        off_out  = h * stride_out_h

        # Build 2‑D pointer tensors: (pos, hidden)
        inp_ptrs   = inp_ptr   + base_in[:, None]   + off_in[None, :]
        gate_ptrs  = gate_ptr  + base_gate[:, None] + off_gate[None, :]
        out_ptrs   = out_ptr   + base_out[:, None]  + off_out[None, :]

        # Load values
        load_mask = valid[:, None] & h_valid[None, :]
        inp  = tl.load(inp_ptrs,  mask=load_mask, other=0.0)
        gate = tl.load(gate_ptrs, mask=load_mask, other=0.0)

        # Fuse: out = inp * gate * mask
        out = inp * gate * m_val[:, None]

        # Store result back (in‑place)
        tl.store(out_ptrs, out, mask=load_mask)


# ----------------------------------------------------------------------
# Python helper that launches the Triton kernel
# ----------------------------------------------------------------------
def _apply_mask_gate(tensor: torch.Tensor,
                     gate: torch.Tensor,
                     mask: torch.Tensor) -> torch.Tensor:
    """
    In‑place multiplication:
        tensor = tensor * gate * mask
    All tensors are on the same CUDA device and have dtype torch.float32.
    """
    B, N, _, H = tensor.shape
    total = B * N * N
    BLOCK_POS = 256                     # positions per program
    BLOCK_H = 32                        # hidden‑dim block size (fits registers)

    grid = (triton.cdiv(total, BLOCK_POS),)   # 1‑D grid

    _apply_mask_gate_kernel[grid](
        tensor,
        gate,
        mask,
        tensor,                         # write back into tensor (in‑place)

        B, N,
        tensor.stride(0), tensor.stride(1), tensor.stride(2), tensor.stride(3),
        gate.stride(0),   gate.stride(1),   gate.stride(2),   gate.stride(3),
        mask.stride(0),   mask.stride(1),   mask.stride(2),
        tensor.stride(0), tensor.stride(1), tensor.stride(2), tensor.stride(3),

        BLOCK_POS=BLOCK_POS,
        BLOCK_H=BLOCK_H,
        H=H,                            # compile‑time hidden dimension
    )
    return tensor

# ----------------------------------------------------------------------

        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor, shape [B, N, N, dim]
            - mask        : torch.Tensor, shape [B, N, N] (float32 or bool)
            - weights     : dict of model parameters (see problem statement)
            - config      : dict containing 'dim', 'hidden_dim', 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # Unpack arguments
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # Config parameters
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Ensure mask is float32 on the correct device
    if nomask or mask is None:
        mask_float = torch.ones_like(mask, dtype=dtype, device=device) if mask is not None else torch.ones(
            input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2],
            dtype=dtype, device=device)
    else:
        mask_float = mask.to(dtype=dtype, device=device)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    eps = 1e-5
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # ------------------------------------------------------------------
    # 2. Projections (no bias) and gating
    # ------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    # Linear projections
    left  = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # Gates (sigmoid)
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3. Fuse mask and gate using Triton (in‑place)
    # ------------------------------------------------------------------
    left  = _apply_mask_gate(left,  left_gate,  mask_float)
    right = _apply_mask_gate(right, right_gate, mask_float)
    # After this, left = left * mask * left_gate, right = right * mask * right_gate

    # ------------------------------------------------------------------
    # 4. Pairwise multiplication over the sequence dimension (einsum)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented with a batched matrix multiplication.
    # ------------------------------------------------------------------
    # Reshape to (B, hidden_dim, N, N) for matmul
    left_mat  = left.permute(0, 3, 1, 2)      # [B, H, N, N]
    right_mat = right.permute(0, 3, 1, 2)     # [B, H, N, N]

    # Compute batch‑wise matmul: left_mat @ right_mat^T over the K dimension
    out_mat = torch.matmul(left_mat, right_mat.transpose(-1, -2))   # [B, H, N, N]

    # Back to original layout [B, N, N, H]
    out = out_mat.permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 5. Output LayerNorm, final gating, and linear projection back to dim
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    out_norm = F.layer_norm(out,
                            normalized_shape=(hidden_dim,),
                            weight=to_out_norm_weight,
                            bias=to_out_norm_bias,
                            eps=eps)

    # Apply output gate (element‑wise)
    out_gated = out_norm * out_gate

    # Final linear projection (hidden -> dim)
    output = F.linear(out_gated, to_out_weight)   # [B, N, N, dim]

    return output
```
