def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, N, H,
    stride_ab, stride_ai, stride_ak, stride_ad,
    stride_bb, stride_bi, stride_bk, stride_bd,
    stride_cb, stride_ci, stride_cj, stride_cd,
    # perhaps not needed: but we can use same stride for c as a.
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)

    num_blocks_m = tl.cdiv(N, BLOCK_M)

    # Compute block_i and batch_hidden
    block_i = pid0 % num_blocks_m
    batch_hidden = pid0 // num_blocks_m
    batch_idx = batch_hidden // H
    d_idx = batch_hidden % H

    block_j = pid1

    # offsets
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers for a: shape (B,N,N,H) where second dim is i, third dim k
    a_ptrs = a_ptr + batch_idx * stride_ab + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak + d_idx * stride_ad
    # shape (M,K)

    # pointers for b: we need transpose k x j
    b_ptrs = b_ptr + batch_idx * stride_bb + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bi + d_idx * stride_bd

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k tiles
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias)

    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Mask
    if not nomask:
        # mask may be boolean but we treat as float
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)  # shape B,N,N,1
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    batch, N, _, _ = left_fp16.shape  # N,N,hidden_dim

    # Allocate output
    out_fp16 = torch.empty(batch, N, N, hidden_dim, dtype=torch.float16, device=device)

    # Get strides
    stride_ab = left_fp16.stride(0)
    stride_ai = left_fp16.stride(1)
    stride_ak = left_fp16.stride(2)
    stride_ad = left_fp16.stride(3)

    stride_bb = right_fp16.stride(0)
    stride_bi = right_fp16.stride(1)   # j dimension
    stride_bk = right_fp16.stride(2)
    stride_bd = right_fp16.stride(3)

    stride_cb = out_fp16.stride(0)
    stride_ci = out_fp16.stride(1)
    stride_cj = out_fp16.stride(2)
    stride_cd = out_fp16.stride(3)

    # Block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Compute grid dims
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    # total batch*hidden_dim* num_blocks_m
    grid0 = batch * hidden_dim * num_blocks_m
    grid = (grid0, num_blocks_n)

    # Launch kernel
    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        batch,
        N,
        hidden_dim,
        stride_ab, stride_ai, stride_ak, stride_ad,
        stride_bb, stride_bi, stride_bk, stride_bd,
        stride_cb, stride_ci, stride_cj, stride_cd,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert back to float32
    out = out_fp16.to(torch.float32)

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, normalized_shape=(hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Multiply by out_gate
    out = out * out_gate

    # Final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, N, H,
    stride_ab, stride_ai, stride_ak, stride_ad,  # A: B,N,K,H
    stride_bb, stride_bi, stride_bk, stride_bd,
    stride_cb, stride_ci, stride_cj, stride_cd,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)

    # Number of blocks along M dimension
    num_blocks_m = tl.cdiv(N, BLOCK_M)

    # Determine i block and batch*hidden index
    block_i = pid0 % num_blocks_m
    batch_hidden = pid0 // num_blocks_m
    batch_idx = batch_hidden // H
    d_idx = batch_hidden % H

    block_j = pid1

    # Offsets for i, j, k
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for i,j within bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

"""
TriMul (outgoing) forward pass implemented with Triton.

Algorithm:
1. Layer‑norm the input tensor over the feature dimension.
2. Linear projections (left/right) and gating functions are computed with
   standard torch.nn.functional calls (no gradients needed).
3. Apply the optional pairwise mask to the projected values.
4. Fuse the expensive N³ reduction:
   out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication of A (i×k) with Bᵀ (k×j) for each
   hidden dimension `d`.  The kernel processes a tile of the output
   (BLOCK_M × BLOCK_N) while iterating over k in blocks of size BLOCK_K.
   All arithmetic is performed in float16, accumulation in float32.
5. Layer‑norm over the hidden dimension, apply the output gate and final
   linear projection back to `dim`.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, N, H,
    stride_ab, stride_ai, stride_ak, stride_ad,   # A: (B, i, k, H)
    stride_bb, stride_bi, stride_bk, stride_bd,   # B: (B, j, k, H)
    stride_cb, stride_ci, stride_cj, stride_cd,   # C: (B, i, j, H)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid0 = tl.program_id(0)          # encodes (batch, d, block_i)
    pid1 = tl.program_id(1)          # encodes block_j

    # Number of tiles along the i‑dimension
    num_blocks_m = tl.cdiv(N, BLOCK_M)

    # Decode program ids
    block_i = pid0 % num_blocks_m
    batch_hidden = pid0 // num_blocks_m
    batch_idx = batch_hidden // H
    d_idx = batch_hidden % H
    block_j = pid1

    # Offsets for the current tile
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A tile: shape (M, K)  -> left[b, i, k, d]
        a_ptrs = a_ptr + batch_idx * stride_ab \
                       + offs_i[:, None] * stride_ai \
                       + offs_k[None, :] * stride_ak \
                       + d_idx * stride_ad
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # B tile transposed: shape (K, N) -> right[b, j, k, d] accessed as (k, j)
        b_ptrs = b_ptr + batch_idx * stride_bb \
                       + offs_k[:, None] * stride_bk \
                       + offs_j[None, :] * stride_bi \
                       + d_idx * stride_bd
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # Accumulate dot product over k
        acc += tl.dot(a, b)

    # Store the result (float16)
    c_ptrs = c_ptr + batch_idx * stride_cb \
                     + offs_i[:, None] * stride_ci \
                     + offs_j[None, :] * stride_cj \
                     + d_idx * stride_cd
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        torch.Tensor of shape [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Layer norm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # shape: [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (applied after projection, before gating)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core N³ reduction via Triton (fp16 for bandwidth)
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape

    out_fp16 = torch.empty(B, N, N, hidden_dim, dtype=torch.float16, device=input_tensor.device)

    # Strides (in elements) for the three tensors
    stride_al, stride_ai, stride_ak, stride_ad = left_fp16.stride()
    stride_bl, stride_bi, stride_bk, stride_bd = right_fp16.stride()
    stride_cl, stride_ci, stride_cj, stride_cd = out_fp16.stride()

    # Tile sizes – fine‑tuned for typical B200 capabilities
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N

    # The first launch dimension encodes (batch, hidden_dim, block_i)
    grid0 = B * hidden_dim * num_blocks_m
    grid = (grid0, num_blocks_n)

    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        stride_al, stride_ai, stride_ak, stride_ad,
        stride_bl, stride_bi, stride_bk, stride_bd,
        stride_cl, stride_ci, stride_cj, stride_cd,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    out = out_fp16.to(torch.float32)  # back to FP32 for the remaining ops

    # ------------------------------------------------------------------
    # 6) Output layer‑norm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
