    B, N, C, eps,

    i = rem // N
    j = rem % N

        mask = c_offset < C
        src = tl.load(input_base + c_offset, mask=mask, other=0.0)
        sum += tl.sum(src, axis=0)
        sum_sq += tl.sum(src * src, axis=0)

    inv_std = 1.0 / tl.sqrt(var + eps)

    for c in range(0, C, BLOCK_C):

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    B,N,N2,dim = input_tensor.shape
    assert N == N2
    hidden_dim = config['hidden_dim']
    eps = 1e-5

    # LayerNorm on input: use Triton kernel
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    x = layernorm_triton(input_tensor, norm_weight, norm_bias, eps)

    # Linear projections
    left_proj_weight = weights['left_proj.weight'] # shape hidden_dim x dim
    right_proj_weight = weights['right_proj.weight']
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape B,N,N,hidden_dim
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask after projection if not nomask (maybe mask is provided)
    if mask is not None:
        mask_exp = mask.to(x.dtype).unsqueeze(-1)  # shape B,N,N,1
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate_weight = weights['left_gate.weight']
    right_gate_weight = weights['right_gate.weight']
    out_gate_weight = weights['out_gate.weight']
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Core multiplication: compute out = left @ right^T per hidden dim
    # We'll permute to combine B and hidden_dim dimensions as separate batch for matmul
    # Option: use torch.matmul directly with broadcasting: (B,N,N,H) * (B,N,N,H) -> (B,N,N,H) by einsum, but we aim for efficient matmul.
    # Use permutes
    left_perm = left.permute(0, 3, 1, 2)   # B,H,N,N
    right_perm = right.permute(0, 3, 1, 2) # B,H,N,N
    # Compute matmul: left_perm @ right_perm^T (swap last two dims)
    # right_perm_trans = right_perm.transpose(-1, -2)  # B,H,N,N
    out_perm = torch.matmul(left_perm, right_perm.transpose(-1, -2))
    out = out_perm.permute(0, 2, 3, 1)  # B,N,N,H

    # to_out_norm
    to_out_norm_weight = weights['to_out_norm.weight']
    to_out_norm_bias = weights['to_out_norm.bias']
    out_norm = layernorm_triton(out, to_out_norm_weight, to_out_norm_bias, eps)

    # Multiply by out_gate
    out_norm = out_norm * out_gate

    # Final linear projection to dim
    to_out_weight = weights['to_out.weight']
    output = torch.nn.functional.linear(out_norm, to_out_weight)

    return output

def layernorm_kernel(
    input_ptr, output_ptr,
    weight_ptr, bias_ptr,
    B, N, C, eps,
    stride_b, stride_i, stride_j,
    BLOCK_C: tl.constexpr
):
    pid = tl.program_id(0)
    total_pixels = B * N * N
    if pid >= total_pixels:
        return

    # compute 3D indices
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # base pointer offset for the start of channel dimension
    offset = b * stride_b + i * stride_i + j * stride_j

    input_base = input_ptr + offset
    output_base = output_ptr + offset

    # Initialize accumulators
    sum = tl.zeros([1], dtype=tl.float32)
    sum_sq = tl.zeros([1], dtype=tl.float32)

    # loop over channel dimension
    c = 0
    while c < C:
        cur = c + tl.arange(0, BLOCK_C)
        mask = cur < C
        x = tl.load(input_base + cur, mask=mask, other=0.0).to(tl.float32)
        sum += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)
        c += BLOCK_C

    mean = sum / C
    var = sum_sq / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # second pass: write normalized values
    c = 0
    while c < C:
        cur = c + tl.arange(0, BLOCK_C)
        mask = cur < C
        x = tl.load(input_base + cur, mask=mask, other=0.0).to(tl.float32)
        normalized = (x - mean) * inv_std
        w = tl.load(weight_ptr + cur, mask=mask, other=0.0)
        b_ = tl.load(bias_ptr + cur, mask=mask, other=0.0)
        out = normalized * w + b_
        tl.store(output_base + cur, out, mask=mask)
        c += BLOCK_C

"""
TriMul forward (outgoing) implementation with Triton‑accelerated LayerNorm.

Algorithm
---------
1. Layer‑normalize the input tensor `x` over its channel dimension `dim`
   using a custom Triton kernel (no sync needed across blocks).
2. Linear projections:
   left  = x @ W_left   (no bias)
   right = x @ W_right
3. Optional mask (broadcasted on the channel dimension) is applied to
   `left` and `right`.
4. Compute three gates from the normalized input:
   left_gate  = sigmoid(x @ W_left_gate)
   right_gate = sigmoid(x @ W_right_gate)
   out_gate   = sigmoid(x @ W_out_gate)
   Apply the first two gates element‑wise to `left` and `right`.
5. Core multiplicative update:
   For each hidden channel `d`,
       out[..., d] = left[..., :, d] @ right[..., :, d].T
   Implemented as a batched matrix multiplication by permuting the
   tensors to shape (B, H, N, N) and using torch.matmul.
6. Layer‑normalize the result over the hidden dimension using the same
   Triton kernel.
7. Apply `out_gate` and a final linear projection back to `dim`.

Only the LayerNorm operations are written in Triton; the remaining
dense/element‑wise work uses optimized PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _layernorm_kernel(
    inp_ptr, out_ptr,
    weight_ptr, bias_ptr,
    B, N, C, eps,
    stride_b, stride_i, stride_j,
    BLOCK_C: tl.constexpr
):
    """LayerNorm over the last dimension (C) of a [B, N, N, C] tensor."""
    pid = tl.program_id(0)
    total_pixels = B * N * N
    if pid >= total_pixels:
        return

    # decode (b, i, j) from pid
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # base offset for the beginning of the C‑axis
    off = b * stride_b + i * stride_i + j * stride_j
    inp_base = inp_ptr + off
    out_base = out_ptr + off

    # ---------- first pass: mean & variance ----------
    sum_val = tl.zeros([1], dtype=tl.float32)
    sum_sq = tl.zeros([1], dtype=tl.float32)

    c = 0
    while c < C:
        cur = c + tl.arange(0, BLOCK_C)
        mask = cur < C
        x = tl.load(inp_base + cur, mask=mask, other=0.0)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)
        c += BLOCK_C

    mean = sum_val / C
    var = sum_sq / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    c = 0
    while c < C:
        cur = c + tl.arange(0, BLOCK_C)
        mask = cur < C
        x = tl.load(inp_base + cur, mask=mask, other=0.0)
        norm = (x - mean) * inv_std
        w = tl.load(weight_ptr + cur, mask=mask, other=0.0)
        b_ = tl.load(bias_ptr + cur, mask=mask, other=0.0)
        out = norm * w + b_
        tl.store(out_base + cur, out, mask=mask)
        c += BLOCK_C


def _layernorm_triton(tensor: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5) -> torch.Tensor:
    """Apply LayerNorm using the Triton kernel above."""
    B, N, N2, C = tensor.shape
    assert N == N2, "Input must be square in the sequence dimensions."
    out = torch.empty_like(tensor)

    stride_b, stride_i, stride_j, _ = tensor.stride()
    total = B * N * N
    # Choose a block size that works for all supported channel sizes.
    BLOCK_C = 128

    grid = (total,)
    _layernorm_kernel[grid](
        tensor,
        out,
        weight,
        bias,
        B, N, C, eps,
        stride_b, stride_i, stride_j,
        BLOCK_C=BLOCK_C,
        num_warps=4,
    )
    return out


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing Triangle Multiplicative
    Update (TriMul) module.

    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, N2, dim = input_tensor.shape
    assert N == N2, "Sequence length dimensions must be equal."

    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. LayerNorm on the input (dim)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = _layernorm_triton(input_tensor, norm_weight, norm_bias, eps)

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)      # shape [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)    # shape [B,N,N,hidden_dim]

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden_dim)
    # -----------------------------------------------------------------
    if mask is not None and mask.shape != torch.Size([0]):
        mask_f = mask.to(dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4. Gates (sigmoid of linear transforms of the normalized input)
    # -----------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core multiplicative update: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched matmul after permuting dimensions.
    # -----------------------------------------------------------------
    # Shape: (B, N, N, H) -> (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2)    # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2)  # [B, H, N, N]

    # Matrix multiplication: (B, H, N, N) @ (B, H, N, N).transpose(-1, -2)
    out_perm = torch.matmul(left_perm, right_perm.transpose(-1, -2))  # [B, H, N, N]
    out = out_perm.permute(0, 2, 3, 1)    # [B, N, N, H]

    # -----------------------------------------------------------------
    # 6. LayerNorm over the hidden dimension (H)
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = _layernorm_triton(out, to_out_norm_weight, to_out_norm_bias, eps)

    # -----------------------------------------------------------------
    # 7. Apply output gate and final linear projection back to `dim`
    # -----------------------------------------------------------------
    out_norm = out_norm * out_gate
    to_out_weight = weights["to_out.weight"]   # shape [dim, hidden_dim]

    output = F.linear(out_norm, to_out_weight)  # [B, N, N, dim]
    return output
