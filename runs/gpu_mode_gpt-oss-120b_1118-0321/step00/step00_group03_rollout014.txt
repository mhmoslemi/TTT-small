def tri_mul_forward(left: torch.Tensor, right: torch.Tensor):
    B, N, _, H = left.shape
    # The third dimension is also N but we can assume shape (B,N,N,H). That's consistent.
    # allocate output
    out = torch.empty((B, N, N, H), dtype=left.dtype, device=left.device)
    # strides
    # left shape (B, I, K, H)
    stride_bl, stride_bi, stride_bk, stride_bh = left.stride()
    stride_rl, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ol, stride_oi, stride_oj, stride_oh = out.stride()

    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32  # each tile size for reduction.

    # Determine grid
    num_blocks_i = (N + BLOCK_I - 1) // BLOCK_I
    num_blocks_j = (N + BLOCK_J - 1) // BLOCK_J
    # total programs = B * H * num_blocks_i * num_blocks_j
    total_blocks = B * H * num_blocks_i * num_blocks_j

    tri_mul_forward_kernel[(total_blocks,)](
        left, right, out,
        B, N, H,
        stride_bl, stride_bi, stride_bk, stride_bh,
        stride_rl, stride_rj, stride_rk, stride_rh,
        stride_ol, stride_oi, stride_oj, stride_oh,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K,
    )

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# TriMul kernel
@triton.jit
def tri_mul_forward_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_bi, stride_bk, stride_bh,
    stride_rl, stride_rj, stride_rk, stride_rh,
    stride_ol, stride_oi, stride_oj, stride_oh,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]"""
    pid = tl.program_id(0)

    # Compute block indices
    num_blocks_i = tl.cdiv(N, BLOCK_I)
    num_blocks_j = tl.cdiv(N, BLOCK_J)
    num_blocks_h = H
    num_blocks_b = B

    # Decompose pid into batch, hidden, i-block, j-block
    b = pid // (num_blocks_h * num_blocks_i * num_blocks_j)
    tmp = pid % (num_blocks_h * num_blocks_i * num_blocks_j)
    h = tmp // (num_blocks_i * num_blocks_j)
    tmp = tmp % (num_blocks_i * num_blocks_j)
    ib = tmp // num_blocks_j
    jb = tmp % num_blocks_j

    i_base = ib * BLOCK_I
    j_base = jb * BLOCK_J

    # Range indices
    offs_i = i_base + tl.arange(0, BLOCK_I)
    offs_j = j_base + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # Loop over k dimension
    for k_start in range(0, N, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile (i, k)
        left_ptrs = left_ptr + b * stride_bl + offs_i[:, None] * stride_bi + offs_k[None, :] * stride_bk + h * stride_bh
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right tile transposed (k, j)
        right_ptrs = right_ptr + b * stride_rl + offs_k[:, None] * stride_rk + offs_j[None, :] * stride_rj + h * stride_rh
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Dot-product over k
        acc += tl.dot(left, right)

    # Store result
    out_ptrs = out_ptr + b * stride_ol + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj + h * stride_oh
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def tri_mul_forward(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """Launch Triton kernel to compute the triangle multiplicative update.
    left/right: [B, N, N, H] tensors (float32)"""
    assert left.shape == right.shape, "Left and right must have same shape"
    B, N, _, H = left.shape
    # Ensure contiguity
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty((B, N, N, H), dtype=left.dtype, device=left.device)

    # Strides (in elements)
    stride_bl, stride_bi, stride_bk, stride_bh = left.stride()
    stride_rl, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ol, stride_oi, stride_oj, stride_oh = out.stride()

    # Block sizes (tuned for typical N up to 1024)
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32

    num_blocks_i = (N + BLOCK_I - 1) // BLOCK_I
    num_blocks_j = (N + BLOCK_J - 1) // BLOCK_J
    total_blocks = B * H * num_blocks_i * num_blocks_j

    # Launch
    tri_mul_forward_kernel[(total_blocks,)](
        left, right, out,
        B, N, H,
        stride_bl, stride_bi, stride_bk, stride_bh,
        stride_rl, stride_rj, stride_rk, stride_rh,
        stride_ol, stride_oi, stride_oj, stride_oh,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K,
    )
    return out

def custom_kernel(data):
    """
    Triton-accelerated forward pass for the TriMul (outgoing) operator.

    Arguments:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim]
            - mask: [B, N, N] (may be None or all-ones if config["nomask"]==True)
            - weights: dict of model parameters
            - config: dict with keys "dim", "hidden_dim", possibly "nomask"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # -------------------- LayerNorm on input --------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # -------------------- Linear projections --------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B,N,N,hidden_dim]

    # -------------------- Optional mask --------------------
    if mask is not None and not config.get("nomask", False):
        mask_ = mask.unsqueeze(-1).to(dtype=dtype)   # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # -------------------- Gates --------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # -------------------- Triangular multiplicative update --------------------
    # Compute out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
    out = tri_mul_forward(left, right)  # [B,N,N,hidden_dim]

    # -------------------- Output layernorm, gating, final linear --------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    out = out * out_gate

    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)   # [B,N,N,dim]

    return out

"""
Triton‑accelerated forward pass of the outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension.
2. Project the normalized tensor with two linear layers (left/right) → shape [B,N,N,H].
3. (Optional) Apply a binary mask to the projected tensors.
4. Compute three independent sigmoid gates and multiply them into the projections.
5. Core operation:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is implemented as a tiled batched GEMM using a Triton kernel where each
   program instance processes a (i‑block, j‑block, hidden‑dim) tile.
6. Layer‑norm the accumulated tensor, apply the output gate and a final linear
   projection back to the original channel dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_forward_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_bi, stride_bk, stride_bh,
    stride_rl, stride_rj, stride_rk, stride_rh,
    stride_ol, stride_oi, stride_oj, stride_oh,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""

    pid = tl.program_id(0)

    # Number of tiles in each dimension
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)

    # Decompose the 1‑D pid into (batch, hidden, i‑tile, j‑tile)
    b = pid // (H * num_i * num_j)
    tmp = pid % (H * num_i * num_j)
    h = tmp // (num_i * num_j)
    tmp = tmp % (num_i * num_j)
    ib = tmp // num_j
    jb = tmp % num_j

    i0 = ib * BLOCK_I
    j0 = jb * BLOCK_J

    # Index vectors for this tile
    offs_i = i0 + tl.arange(0, BLOCK_I)
    offs_j = j0 + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator for the I×J tile
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k0 in range(0, N, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left tile (i, k) -----
        left_ptrs = (
            left_ptr
            + b * stride_bl
            + offs_i[:, None] * stride_bi
            + offs_k[None, :] * stride_bk
            + h * stride_bh
        )
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # ----- load right tile transposed (k, j) -----
        right_ptrs = (
            right_ptr
            + b * stride_rl
            + offs_k[:, None] * stride_rk
            + offs_j[None, :] * stride_rj
            + h * stride_rh
        )
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # ----- accumulate -----
        acc += tl.dot(left, right)   # (I,K) @ (K,J) -> (I,J)

    # Store the result tile
    out_ptrs = (
        out_ptr
        + b * stride_ol
        + offs_i[:, None] * stride_oi
        + offs_j[None, :] * stride_oj
        + h * stride_oh
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def tri_mul_forward(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Launcher for the Triton kernel computing the triangle multiplicative update.
    `left` and `right` have shape [B, N, N, H] and are assumed to be contiguous.
    Returns a tensor of the same shape.
    """
    assert left.shape == right.shape, "Left and right must share shape"
    B, N, _, H = left.shape

    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty((B, N, N, H), dtype=left.dtype, device=left.device)

    # Strides expressed in element counts
    sb, si, sk, sh = left.stride()
    rb, rj, rk, rh = right.stride()
    ob, oi, oj, oh = out.stride()

    # Tile sizes (tuned for typical N up to 1024)
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32

    num_i = (N + BLOCK_I - 1) // BLOCK_I
    num_j = (N + BLOCK_J - 1) // BLOCK_J
    total_progs = B * H * num_i * num_j

    tri_mul_forward_kernel[(total_progs,)](
        left,
        right,
        out,
        B,
        N,
        H,
        sb,
        si,
        sk,
        sh,
        rb,
        rj,
        rk,
        rh,
        ob,
        oi,
        oj,
        oh,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )
    return out


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the TriMul (outgoing) operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (may be None or ignored when config["nomask"] is True)
            - weights: dict containing all model weights
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # --------------------------------------------------------------
    # 1) Input LayerNorm
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # --------------------------------------------------------------
    # 2) Linear projections (left / right)
    # --------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)    # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # 3) Optional mask
    # --------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask_ = mask.unsqueeze(-1).to(dtype=dtype)   # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4) Gating
    # --------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) Triangle multiplicative update (core O(N³) work)
    # --------------------------------------------------------------
    out = tri_mul_forward(left, right)   # [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # 6) Output LayerNorm, output gate, final projection
    # --------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate

    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)   # back to [B,N,N,dim]

    return out
