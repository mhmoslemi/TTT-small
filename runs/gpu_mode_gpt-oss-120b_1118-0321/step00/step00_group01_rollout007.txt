def tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                   B, N, H,
                   stride_bl, stride_il, stride_kl, stride_hl,
                   stride_br, stride_ir, stride_kr, stride_hr,
                   stride_bo, stride_io, stride_jo, stride_ho,
                   BLOCK_K: tl.constexpr,
                   ):
    pid = tl.program_id(0)
    num_i = tl.cdiv(N, 1)   # per element
    num_j = tl.cdiv(N, 1)
    total_per_batch = num_i * num_j
    b = pid // total_per_batch
    pid_batch = pid % total_per_batch
    i = pid_batch // num_j
    j = pid_batch % num_j

    # bound check for batch
    if b >= B:
        return

    mask_i = i < N
    mask_j = j < N
    if not (mask_i and mask_j):
        return

    # accumulator per hidden dimension
    acc = tl.zeros([H], dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        k_offset = k_base + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # compute ptrs
        left_ptrs = left_ptr + b * stride_bl + i * stride_il + k_offset[:, None] * stride_kl + tl.arange(0, H)[None, :] * stride_hl
        right_ptrs = right_ptr + b * stride_br + j * stride_ir + k_offset[:, None] * stride_kr + tl.arange(0, H)[None, :] * stride_hr

        left = tl.load(left_ptrs, mask=mask_k[:, None], other=0.0).to(tl.float32)

    # write output
    out_ptrs = out_ptr + b * stride_bo + i * stride_io + j * stride_jo + tl.arange(0, H) * stride_ho

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # get dims
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # layernorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )
    # projections
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()
    # mask handling
    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left_proj * mask * left_gate
        right = right_proj * mask * right_gate
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # cast to float16 for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape # shape: [B,N,N,hidden_dim]

    # allocate output
    out_tensor = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left_fp16.device)

    # strides
    stride_bl, stride_il, stride_kl, stride_hl = left_fp16.stride()
    stride_br, stride_ir, stride_kr, stride_hr = right_fp16.stride()

    grid = (total,)

    # Launch kernel
    BLOCK_K = 32
    num_warps = 4

        left_fp16,
        right_fp16,
        out_tensor,
        B,
        N,
        stride_bl,
        stride_il,
        stride_kl,

        stride_br,

"""
TriMul forward pass (outgoing version) with a custom Triton kernel.

The heavy part of the computation is:
    out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]

We implement this O(N³·H) reduction in a Triton kernel where each program
computes a single (b,i,j) element and accumulates a vector of length H
(hidden_dim) over the k dimension in blocks.  The rest of the module
(LayerNorms, linear projections, gating, and final linear) is performed
with PyTorch for simplicity.
"""

import math
import torch
import triton
import triton.language as tl
from torch.nn import functional as F


@triton.jit
def tri_mul_kernel(
    left_ptr,          # *fp16 left tensor [B,N,N,H]
    right_ptr,         # *fp16 right tensor [B,N,N,H]
    out_ptr,           # *fp32 output tensor [B,N,N,H]
    B: tl.int32,
    N: tl.int32,
    # strides for left
    stride_bl: tl.int32,
    stride_il: tl.int32,
    stride_kl: tl.int32,
    stride_hl: tl.int32,
    # strides for right
    stride_br: tl.int32,
    stride_ir: tl.int32,
    stride_kr: tl.int32,
    stride_hr: tl.int32,
    # strides for out
    stride_bo: tl.int32,
    stride_io: tl.int32,
    stride_jo: tl.int32,
    stride_ho: tl.int32,
    # compile‑time constants
    BLOCK_K: tl.constexpr,     # tile size along the summed k‑dimension
    H: tl.constexpr,           # hidden dimension (channel)
):
    # ------------------------------------------------------------------
    #  Mapping from a 1‑D program id to (b,i,j)
    # ------------------------------------------------------------------
    pid = tl.program_id(0)
    total_ij = N * N                     # per‑batch programs
    b = pid // total_ij
    pid_ij = pid % total_ij
    i = pid_ij // N
    j = pid_ij % N

    # bounds check (grid may be slightly oversized)
    if b >= B or i >= N or j >= N:
        return

    # ------------------------------------------------------------------
    #  Accumulator for the H‑dimensional output vector
    # ------------------------------------------------------------------
    acc = tl.zeros([H], dtype=tl.float32)

    # pre‑compute base offsets for the batch‑i and batch‑j slices
    base_left  = b * stride_bl + i * stride_il
    base_right = b * stride_br + j * stride_ir

    # hidden‑dimension offsets (contiguous, stride_hl/stride_hr == 1)
    h_off = tl.arange(0, H)

    # ------------------------------------------------------------------
    #  Loop over k in blocks of size BLOCK_K
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_k_blocks):
        k_base = kb * BLOCK_K
        k_off = k_base + tl.arange(0, BLOCK_K)               # (BLOCK_K,)
        mask_k = k_off < N

        # offsets for left[b,i,k,:] and right[b,j,k,:] (shape BLOCK_K x H)
        off_left  = base_left  + k_off[:, None] * stride_kl + h_off[None, :] * stride_hl
        off_right = base_right + k_off[:, None] * stride_kr + h_off[None, :] * stride_hr

        # load tiles, cast to fp32 for accumulation
        left_tile  = tl.load(left_ptr + off_left,
                             mask=mask_k[:, None],
                             other=0.0).to(tl.float32)   # (BLOCK_K, H)
        right_tile = tl.load(right_ptr + off_right,
                             mask=mask_k[:, None],
                             other=0.0).to(tl.float32)   # (BLOCK_K, H)

        # element‑wise product and reduction over k
        prod = left_tile * right_tile                     # (BLOCK_K, H)
        acc += tl.sum(prod, axis=0)                       # (H,)

    # ------------------------------------------------------------------
    #  Write the result
    # ------------------------------------------------------------------
    out_off = b * stride_bo + i * stride_io + j * stride_jo + h_off * stride_ho
    tl.store(out_ptr + out_off, acc)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/int/float) or None
            - weights: dict of torch.Tensor containing model parameters
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    #  LayerNorm on the input (last dimension = dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    #  Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])    # [B,N,N,hidden_dim]
    right_proj = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    #  Gating values (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    #  Apply mask (if any) and gating
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left_proj * mask_f * left_gate
        right = right_proj * mask_f * right_gate
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    #  Cast to fp16 for the heavy O(N³) kernel (memory‑bandwidth saving)
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape  # shape = [B, N, N, hidden_dim]

    # Output tensor of the reduction (fp32 for stability)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides (in elements, not bytes) for pointer arithmetic
    stride_bl, stride_il, stride_kl, stride_hl = left_fp16.stride()
    stride_br, stride_ir, stride_kr, stride_hr = right_fp16.stride()
    stride_bo, stride_io, stride_jo, stride_ho = out.stride()

    # ------------------------------------------------------------------
    #  Launch the Triton kernel: one program per (b,i,j) element
    # ------------------------------------------------------------------
    total_programs = B * N * N
    grid = (total_programs,)

    BLOCK_K = 32               # empirical choice; fits well in shared memory
    num_warps = 4

    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B,
        N,
        stride_bl,
        stride_il,
        stride_kl,
        stride_hl,
        stride_br,
        stride_ir,
        stride_kr,
        stride_hr,
        stride_bo,
        stride_io,
        stride_jo,
        stride_ho,
        BLOCK_K=BLOCK_K,
        H=hidden_dim,
        num_warps=num_warps,
    )

    # ------------------------------------------------------------------
    #  Post‑processing: to_out_norm, gating and final linear projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # Apply the output gate (broadcast over hidden_dim)
    out_norm = out_norm * out_gate

    # Final linear projection back to original dim
    final_out = F.linear(out_norm, weights["to_out.weight"])

    return final_out
