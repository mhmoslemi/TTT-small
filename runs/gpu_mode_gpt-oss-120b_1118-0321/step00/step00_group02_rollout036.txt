"""
Trident matrix multiplication kernel for the TriMul ("outgoing") operation.

The kernel computes, for each batch element and hidden dimension `h`,
    out[b, i, j, h] = sum_k left[b, i, k, h] * right[b, j, k, h]
which is equivalent to a batched matrix multiplication of
    left[b, h] (N×N) with the transpose of right[b, h] (N×N).

All heavy lifting (the N×N matmul) is performed in Triton.
Other operations (LayerNorm, linear projections, gating, mask handling,
final LayerNorm and output projection) are done with PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dimensions (M=N=Nseq, K=Nseq)
    stride_ab, stride_am, stride_ak,   # A: batch, row, col strides
    stride_bb, stride_bk, stride_bn,   # B (treated as transposed): batch, row(k), col(n) strides
    stride_cb, stride_cm, stride_cn,   # C: batch, row, col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)      # output tile row
    pid_n = tl.program_id(1)      # output tile col
    pid_b = tl.program_id(2)      # batch*hidden index

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + pid_b * stride_ab
    b_batch = b_ptr + pid_b * stride_bb
    c_batch = c_ptr + pid_b * stride_cb

    # Row offset for A and column offset for B (since B is logically transposed)
    a_base = a_batch + offs_m[:, None] * stride_am
    b_base = b_batch + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_blocks):
        k_offset = k * BLOCK_K
        cur_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Pointers to the current K‑tile
        a_ptrs = a_base + cur_k[None, :] * stride_ak   # (BLOCK_M, BLOCK_K)
        b_ptrs = b_base + cur_k[:, None] * stride_bk   # (BLOCK_K, BLOCK_N)

        # Load with bounds checking
        mask_a = mask_m[:, None] & mask_k[None, :]
        mask_b = mask_n[None, :] & mask_k[:, None]

        a = tl.load(a_ptrs, mask=mask_a, other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0).to(tl.float32)

        # Accumulate block matrix product
        acc += tl.dot(a, b)

    # Write back the result tile
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

    # unpack arguments
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------- LayerNorm on the input ----------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Gating signals ----------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ---------- Optional masking ----------
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)            # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ---------- Apply gates ----------
    left = left * left_gate
    right = right * right_gate

    # ---------- Prepare for batched matmul ----------
    # Move hidden dimension to the 2nd axis and flatten batch*hidden
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left.shape
    batch_flat = B * H

    left_flat = left.view(batch_flat, N, N)        # [B*H, N, N]
    right_flat = right.view(batch_flat, N, N)

    # Allocate output buffer
    out_flat = torch.empty_like(left_flat)

    # ---------- Launch Triton batched matmul ----------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch_flat,
    )

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        # For B we treat the input as transposed: rows -> original rows, cols -> original cols.
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    torch.cuda.synchronize()

    # ---------- Reshape back to original layout ----------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()      # [B,N,N,hidden_dim]

    # ---------- Output LayerNorm, gating, and final projection ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
```
