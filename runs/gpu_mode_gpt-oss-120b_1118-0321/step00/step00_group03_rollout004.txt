"""
TriMul (outgoing) forward implementation using Triton for the
core O(N³·H) contraction.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor (B,N,N,D) using the provided
   `norm.weight` / `norm.bias`.
2. Project the normalized tensor into the hidden dimension H with two
   linear layers (`left_proj`, `right_proj`).
3. Optionally mask the projected tensors.
4. Compute three sigmoid gates and apply them to `left`, `right` and the
   later output.
5. Compute the contracted tensor
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is mathematically a batch of matrix‑multiplications
   `L_h @ R_hᵀ` for every hidden channel `h`.  This step is performed
   by a custom Triton kernel that tiles the i‑ and j‑axes and loops
   over k in shared blocks, using fp16 arithmetic with fp32 accumulation.
6. Apply a second LayerNorm (`to_out_norm`), multiply by the output gate
   and finally linearly project back to the original dimension D.

Only the contraction (step 5) is executed on the GPU via Triton; all
other operations use regular PyTorch primitives for simplicity and
numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # -------------------------------------------------------------
    # Grid layout:
    #   program_id(0) -> i‑tile
    #   program_id(1) -> j‑tile
    #   program_id(2) -> (batch, hidden) pair
    # -------------------------------------------------------------
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # Compute the global indices for i, j
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Decode batch and hidden index from pid_bh
    bh = pid_bh
    b = bh // H          # batch index
    h = bh % H           # hidden channel index

    # Bounds checking masks (bool tensors)
    mask_i = i < N
    mask_j = j < N

    # Replace out‑of‑range indices with 0 to keep addresses in bounds
    i_safe = tl.where(mask_i, i, 0)
    j_safe = tl.where(mask_j, j, 0)

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of k‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for k in range(0, num_k):
        k_base = k * BLOCK_K
        k = k_base + tl.arange(0, BLOCK_K)
        mask_k = k < N
        k_safe = tl.where(mask_k, k, 0)

        # -----------------------------------------------------------------
        # Load left tile : shape (BLOCK_M, BLOCK_K)
        #   left[b, i, k, h]
        # -----------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + b * stride_lb
            + i_safe[:, None] * stride_li
            + k_safe[None, :] * stride_lk
            + h * stride_ld
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # -----------------------------------------------------------------
        # Load right tile transposed: shape (BLOCK_K, BLOCK_N)
        #   right[b, j, k, h]  → we want (k, j) ordering for dot
        # -----------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + b * stride_rb
            + k_safe[:, None] * stride_rk
            + j_safe[None, :] * stride_rj
            + h * stride_rd
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # Accumulate dot product (M×K)·(K×N) → (M×N)
        acc += tl.dot(left, right)  # fp16·fp16 → fp32 accumulation

    # -----------------------------------------------------------------
    # Store the result tile to out[b, i, j, h]
    # -----------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + i_safe[:, None] * stride_oi
        + j_safe[None, :] * stride_oj
        + h * stride_od
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, D] (float32)
            - mask_tensor : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict : mapping of weight names to tensors
            - config_dict  : contains "dim" (D) and "hidden_dim" (H)

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # unpack
    input_tensor, mask_tensor, weights, config = data
    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. LayerNorm over the channel dimension D
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (D,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 2. Linear projections to hidden dimension H
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"])  # (B,N,N,H)

    # -----------------------------------------------------------------
    # 3. Optional mask (mask is broadcasted on the hidden dim)
    # -----------------------------------------------------------------
    if mask_tensor is not None and not config.get("nomask", False):
        mask = mask_tensor.to(input_tensor.dtype).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4. Gates (sigmoid activated linear layers)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction using Triton: out = Σ_k left[...,k,:] * right[...,k,:]
    # -----------------------------------------------------------------
    # Cast to fp16 for the kernel (GPU tensor cores)
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp16 = torch.empty_like(left_fp16)

    # Strides (in elements)
    stride_lb, stride_li, stride_lk, stride_ld = left_fp16.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right_fp16.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out_fp16.stride()

    # Choose tiling parameters (tuned for B200, fp16)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    if N < BLOCK_M:
        BLOCK_M = max(16, N)
    if N < BLOCK_N:
        BLOCK_N = max(16, N)
    if N < BLOCK_K:
        BLOCK_K = max(8, N)

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        H,
        stride_lb,
        stride_li,
        stride_lk,
        stride_ld,
        stride_rb,
        stride_rj,
        stride_rk,
        stride_rd,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Back to fp32 for the remaining ops
    out = out_fp16.to(torch.float32)

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gate, and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
