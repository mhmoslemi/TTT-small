@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_ri, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    # Compute pid: program id for output tile
    pid = tl.program_id(axis=0)
    # total number of tiles = B * ceil(N/BLOCK_I)*ceil(N/BLOCK_J)
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    # compute batch index, i tile, j tile
    b = pid // (num_i * num_j)
    pid_tile = pid % (num_i * num_j)
    i_tile = pid_tile // num_j
    j_tile = pid_tile % num_j

    # compute start offsets
    i_offset = i_tile * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = j_tile * BLOCK_J + tl.arange(0, BLOCK_J)

    # mask out-of-range positions
    i_mask = i_offset < N
    j_mask = j_offset < N

    # Initialize accumulator for hidden dim
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # iterate over k dimension
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # load left: shape (BLOCK_I, BLOCK_K, BLOCK_H)
        left_ptrs = left_ptr + b * stride_lb + i_offset[:, None, None] * stride_li + k_offset[None, :, None] * stride_lk + tl.arange(0, BLOCK_H)[None, None, :] * stride_lh
        left = tl.load(left_ptrs, mask=i_mask[:, None, None] & k_mask[None, :, None], other=0.0)  # shape (BLOCK_I, BLOCK_K, BLOCK_H)
        # load right: shape (BLOCK_J, BLOCK_K, BLOCK_H)
        right_ptrs = right_ptr + b * stride_rb + j_offset[:, None, None] * stride_ri + k_offset[None, :, None] * stride_rk + tl.arange(0, BLOCK_H)[None, None, :] * stride_rh
        right = tl.load(right_ptrs, mask=j_mask[:, None, None] & k_mask[None, :, None], other=0.0)

        # compute elementwise multiply and sum over k
        # left: (BLOCK_I, BLOCK_K, BLOCK_H)
        # right: (BLOCK_J, BLOCK_K, BLOCK_H)
        # we need to sum over k dimension: for each i,j, sum over k left[i,k,:] * right[j,k,:]. So we need to broadcast left over j and right over i.
        # left_expand: (BLOCK_I, 1, BLOCK_K, BLOCK_H)
        # right_expand: (1, BLOCK_J, BLOCK_K, BLOCK_H)
        # produce product shape (BLOCK_I, BLOCK_J, BLOCK_K, BLOCK_H). Sum over k -> (BLOCK_I,BLOCK_J,BLOCK_H).
        left_exp = left[:, None, :, :] # (BLOCK_I,1,BLOCK_K,BLOCK_H)
        right_exp = right[None, :, :, :] # (1,BLOCK_J,BLOCK_K,BLOCK_H)
        # Multiply and sum over k
        acc += tl.sum(left_exp * right_exp, axis=2) # sum over k dimension

    # Store result
    out_ptrs = out_ptr + b * stride_ob + i_offset[:, None, None] * stride_oi + j_offset[None, :, None] * stride_oj + tl.arange(0, BLOCK_H)[None, None, :] * stride_oh
    tl.store(out_ptrs, acc.to(tl.float32), mask=i_mask[:, None, None] & j_mask[None, :, None])

@triton.jit
def tri_mul_channel_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_li, stride_lk,
    stride_ri, stride_rk,
    stride_oi, stride_oj,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    num_tiles_i = tl.cdiv(N, BLOCK_I)
    i_tile = pid // num_tiles_i
    j_tile = pid % num_tiles_i

    i_offset = i_tile * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = j_tile * BLOCK_J + tl.arange(0, BLOCK_J)

    i_mask = i_offset < N
    j_mask = j_offset < N

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        left_ptrs = left_ptr + i_offset[:, None] * stride_li + k_offset[None, :] * stride_lk
        left = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)

        right_ptrs = right_ptr + j_offset[:, None] * stride_ri + k_offset[None, :] * stride_rk
        right = tl.load(right_ptrs, mask=j_mask[:, None] & k_mask[None, :], other=0.0)

        # compute dot product accumulation
        acc += tl.dot(left, right, trans_b=True)

    out_ptrs = out_ptr + i_offset[:, None] * stride_oi + j_offset[None, :] * stride_oj
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])

@triton.jit
def tri_mul_channel_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_li, stride_lk,
    stride_ri, stride_rk,
    stride_oi, stride_oj,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    # compute i and j tile indices
    num_i_tiles = tl.cdiv(N, BLOCK_I)
    i_tile = pid // num_i_tiles
    j_tile = pid % num_i_tiles

    i_offset = i_tile * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = j_tile * BLOCK_J + tl.arange(0, BLOCK_J)

    i_mask = i_offset < N
    j_mask = j_offset < N

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # Load left tile
        left_ptrs = left_ptr + i_offset[:, None] * stride_li + k_offset[None, :] * stride_lk
        left = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)

        # Load right tile (right[j,k])
        right_ptrs = right_ptr + j_offset[:, None] * stride_ri + k_offset[None, :] * stride_rk
        right = tl.load(right_ptrs, mask=j_mask[:, None] & k_mask[None, :], other=0.0)

        # multiply accumulate over k: sum_k left[i,k] * right[j,k]
        acc += tl.dot(left, right, trans_b=True)  # sum over k

    # Store results
    out_ptrs = out_ptr + i_offset[:, None] * stride_oi + j_offset[None, :] * stride_oj

def custom_kernel(data):
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Step 1: layer norm
    # weight and bias from weights dict.
    # use torch.nn.functional.layer_norm.
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # step 2: projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # step 3: mask
    if not nomask and mask is not None:
        mask_f = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_f
        right = right * mask_f

    # step 4: gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # apply gates to left/right
    left = left * left_gate
    right = right * right_gate

    # step 5: triple product using Triton kernel
    B, N, _, _ = left.shape
    # create out tensor
    out_hidden = torch.empty_like(left)  # shape [B,N,N,hidden]

    # get strides
    left_stride_b, left_stride_i, left_stride_k, left_stride_h = left.stride()
    right_stride_b, right_stride_i, right_stride_k, right_stride_h = right.stride()
    out_stride_b, out_stride_i, out_stride_j, out_stride_h = out_hidden.stride()

    # kernel block sizes
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32

    # compute grid (tiles)
    num_i_tiles = (N + BLOCK_I - 1) // BLOCK_I
    num_j_tiles = (N + BLOCK_J - 1) // BLOCK_J
    grid = (num_i_tiles * num_j_tiles,)

    # loop over batch and channel
    for b in range(B):
        for d in range(hidden_dim):
            left_ptr = left.data_ptr() + (b * left_stride_b + d * left_stride_h) * left.element_size()
            right_ptr = right.data_ptr() + (b * right_stride_b + d * right_stride_h) * right.element_size()
            out_ptr = out_hidden.data_ptr() + (b * out_stride_b + d * out_stride_h) * out_hidden.element_size()
            tri_mul_channel_kernel[grid](
                left_ptr,
                right_ptr,
                out_ptr,
                N,
                left_stride_i,
                left_stride_k,
                right_stride_i,
                right_stride_k,
                out_stride_i,
                out_stride_j,
                BLOCK_I=BLOCK_I,
                BLOCK_J=BLOCK_J,
                BLOCK_K=BLOCK_K,
                num_warps=4,
                # optionally enable accumulating to float32? Already using float32.
            )

    # out_hidden now contains result shaped [B,N,N,hidden]
    # step 6: to_out_norm layernorm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out_hidden, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)

    # step 7: out_gate scaling
    out = out * out_gate

    # step 8: final linear
    to_out_weight = weights["to_out.weight"]
    output = torch.nn.functional.linear(out, to_out_weight)

    return output

"""
TriMul (outgoing) forward pass with a Triton kernel for the O(N³) reduction.

Algorithm
---------
1. Layer‑norm the input tensor `x` over the channel dimension `dim`.
2. Linear projections to a hidden dimension `hidden_dim`:
   * left  = Linear(x, left_proj_weight)
   * right = Linear(x, right_proj_weight)
3. Optional pair mask is applied to the projected tensors.
4. Gated tensors:
   left  = left  * sigmoid(Linear(x, left_gate_weight))
   right = right * sigmoid(Linear(x, right_gate_weight))
5. Triple‑product `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`.
   This O(N³·hidden_dim) reduction is performed by a Triton kernel that
   computes a batched matrix multiplication for a single hidden channel.
   The kernel tiles the `(i, j)` plane (BLOCK_I×BLOCK_J) and iterates over
   the reduction axis `k` in blocks (BLOCK_K), using `tl.dot` for the
   inner product.
6. Layer‑norm the hidden result, multiply by `out_gate` (sigmoid of a linear)
   and project back to the original dimension with `to_out`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: compute one hidden channel of the outgoing triangular update
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_channel_kernel(
    left_ptr, right_ptr, out_ptr,      # pointers to tensors of shape [N, N, hidden]
    N,                                 # sequence length (int)
    stride_li, stride_lk,              # strides (in elements) for left: i, k
    stride_ri, stride_rk,              # strides for right: i, k
    stride_oi, stride_oj,              # strides for out  : i, j
    ch_offset,                         # which hidden channel to compute (runtime int)
    BLOCK_I: tl.constexpr,             # tile size along i
    BLOCK_J: tl.constexpr,             # tile size along j
    BLOCK_K: tl.constexpr,             # tile size along k
):
    pid = tl.program_id(0)

    # number of tiles along each dimension
    num_i_tiles = tl.cdiv(N, BLOCK_I)
    num_j_tiles = tl.cdiv(N, BLOCK_J)

    i_tile = pid // num_j_tiles
    j_tile = pid % num_j_tiles

    # tile start indices
    i_offset = i_tile * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = j_tile * BLOCK_J + tl.arange(0, BLOCK_J)

    i_mask = i_offset < N
    j_mask = j_offset < N

    # accumulator for the output tile
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # reduction over k
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        k_offset = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # left[i, k, d]   (shape BLOCK_I × BLOCK_K)
        left_ptrs = left_ptr + \
            i_offset[:, None] * stride_li + \
            k_offset[None, :] * stride_lk + \
            ch_offset
        left = tl.load(
            left_ptrs,
            mask=i_mask[:, None] & k_mask[None, :],
            other=0.0,
        )

        # right[j, k, d]
        right_ptrs = right_ptr + \
            j_offset[:, None] * stride_ri + \
            k_offset[None, :] * stride_rk + \
            ch_offset
        right = tl.load(
            right_ptrs,
            mask=j_mask[:, None] & k_mask[None, :],
            other=0.0,
        )

        # acc[i,j] += Σ_k left[i,k] * right[j,k]
        acc += tl.dot(left, right, trans_b=True)   # (BLOCK_I, BLOCK_J)

    # write the tile back
    out_ptrs = out_ptr + \
        i_offset[:, None] * stride_oi + \
        j_offset[None, :] * stride_oj + \
        ch_offset
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])


# ----------------------------------------------------------------------
# Entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (may be omitted if config["nomask"] is True)
            - weights: dict of model weights (see code for expected keys)
            - config: dict containing at least "dim", "hidden_dim", and optional "nomask"

    Returns:
        output: torch.Tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Layer‑norm the pair representation
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.unsqueeze(-1).to(left.dtype)          # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gating (sigmoid of linear projections of x)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triangular multiplicative reduction via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    device = left.device

    # Output tensor in the hidden space
    out_hidden = torch.empty_like(left)

    # Tunable tile sizes (chosen to fit typical B200 resources)
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32

    # Number of tiles in the (i, j) plane
    num_i_tiles = (N + BLOCK_I - 1) // BLOCK_I
    num_j_tiles = (N + BLOCK_J - 1) // BLOCK_J
    grid = (num_i_tiles * num_j_tiles,)

    # Loop over batch and hidden channels; each launch handles a single channel
    for b in range(B):
        # views for the current batch – shape [N, N, hidden]
        left_b = left[b]
        right_b = right[b]
        out_b = out_hidden[b]

        # Strides (in elements) for the 3‑D tensors [i, k, hidden]
        stride_li = left_b.stride(0)   # i‑stride = N * hidden
        stride_lk = left_b.stride(1)   # k‑stride = hidden
        stride_ri = right_b.stride(0)
        stride_rk = right_b.stride(1)
        stride_oi = out_b.stride(0)
        stride_oj = out_b.stride(1)

        for d in range(hidden_dim):
            tri_mul_channel_kernel[grid](
                left_b,                     # base pointer for this batch
                right_b,
                out_b,
                N,
                stride_li, stride_lk,
                stride_ri, stride_rk,
                stride_oi, stride_oj,
                d,                          # channel offset
                BLOCK_I=BLOCK_I,
                BLOCK_J=BLOCK_J,
                BLOCK_K=BLOCK_K,
                num_warps=4,
            )

    # ------------------------------------------------------------------
    # Post‑processing: hidden‑space layer‑norm, output gating, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])

    return output
