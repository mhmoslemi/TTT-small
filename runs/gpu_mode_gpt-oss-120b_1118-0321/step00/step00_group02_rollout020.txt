@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids: pid_m, pid_n, pid_hb
    pid_m = tl.program_id(0)  # i block
    pid_n = tl.program_id(1)  # j block
    pid_hb = tl.program_id(2)  # combined batch*hidden location

    # compute batch and hidden
    batch_idx = pid_hb // H
    hidden_idx = pid_hb % H

    # compute offsets
    i_offset = pid_m * BLOCK_M
    j_offset = pid_n * BLOCK_N

    # block offsets for load; if i_offset or j_offset exceed N, mask will zero.

    # Create range indices
    offs_i = i_offset + tl.arange(0, BLOCK_M)
    offs_j = j_offset + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < N

    # compute base offsets for this batch and hidden
    batch_offset_l = batch_idx * stride_left_b
    hidden_offset_l = hidden_idx * stride_left_h

    batch_offset_r = batch_idx * stride_right_b
    hidden_offset_r = hidden_idx * stride_right_h

    batch_offset_o = batch_idx * stride_out_b
    hidden_offset_o = hidden_idx * stride_out_h

    # pointers:
    a_ptrs = left_ptr + batch_offset_l + hidden_offset_l + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
    b_ptrs = right_ptr + batch_offset_r + hidden_offset_r + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute actual k offset
        k_offset = k * BLOCK_K
        # compute current offset for k ranges
        cur_k_offset = k_offset
        # Effective offsets:
        cur_offs_k = cur_k_offset + offs_k
        # recompute a_ptrs, b_ptrs using k offset
        a_ptrs = left_ptr + batch_offset_l + hidden_offset_l + offs_i[:, None] * stride_left_i + cur_offs_k[None, :] * stride_left_k
        b_ptrs = right_ptr + batch_offset_r + hidden_offset_r + cur_offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j

        # mask within bounds
        mask_a = mask_i[:, None] & (cur_offs_k[None, :] < N)
        mask_b = mask_j[None, :] & (cur_offs_k[:, None] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0).to(tl.float32)

        # compute dot
        acc += tl.dot(a, b)  # both shape (BLOCK_M, BLOCK_K) and (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
    # store result
    # compute out pointer offsets:
    out_ptrs = out_ptr + batch_offset_o + hidden_offset_o + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j

    mask_o = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=mask_o)

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Layernorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = input_tensor
    # compute mean and var across last dim
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias

    # Projections
    left_proj = torch.matmul(x_norm, weights["left_proj.weight"].t())
    right_proj = torch.matmul(x_norm, weights["right_proj.weight"].t())

    # apply mask
    if mask is not None:
        mask_f = mask.unsqueeze(-1).to(x.dtype)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # Gates
    left_gate = torch.sigmoid(torch.matmul(x_norm, weights["left_gate.weight"].t()))
    right_gate = torch.sigmoid(torch.matmul(x_norm, weights["right_gate.weight"].t()))
    out_gate = torch.sigmoid(torch.matmul(x_norm, weights["out_gate.weight"].t()))

    left = left_proj * left_gate
    right = right_proj * right_gate

    left = left.contiguous()
    right = right.contiguous()

    to_out_norm_bias = weights["to_out_norm.bias"]

    out_norm = out

    to_out_weight = weights["to_out.weight"]
    # shape [dim, hidden]
    out_final = torch.matmul(out_gated, to_out_weight.t())

"""
TriMul (outgoing) forward implementation.

The heavy part is the contraction
    out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
which is a batched matrix multiplication for each hidden slice.
We implement this contraction with a Triton kernel that processes
a (i, j) tile for a given (batch, hidden) pair. The rest of the
operations (LayerNorm, linear projections, gating, final projection)
are performed with PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: computes C = A @ B^T where A,B have shape
#   (batch, N, N, hidden) and the contraction is over the second N.
# Each kernel instance works on a tile of (i, j) for a single
# (batch, hidden) pair.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # dimensions
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)      # i‑tile
    pid_n = tl.program_id(1)      # j‑tile
    pid_hb = tl.program_id(2)     # combined batch*hidden

    # decode batch and hidden dimension
    batch_idx = pid_hb // H
    hidden_idx = pid_hb % H

    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base offsets for this (batch, hidden) pair
    base_left = batch_idx * stride_left_b + hidden_idx * stride_left_h
    base_right = batch_idx * stride_right_b + hidden_idx * stride_right_h
    base_out = batch_idx * stride_out_b + hidden_idx * stride_out_h

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over the reduction dimension k
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        k0 = k * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + base_left \
                 + offs_i[:, None] * stride_left_i \
                 + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # B tile (transposed): shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_ptr + base_right \
                 + offs_k[:, None] * stride_right_k \
                 + offs_j[None, :] * stride_right_j
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # matrix‑multiply the tile
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # store result
    out_ptrs = out_ptr + base_out \
               + offs_i[:, None] * stride_out_i \
               + offs_j[None, :] * stride_out_j
    mask_store = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=mask_store)


# ----------------------------------------------------------------------
# Entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module.

    Arguments
    ---------
    data : tuple
        (input, mask, weights, config)
        - input  : torch.Tensor, shape [B, N, N, dim]
        - mask   : torch.Tensor, shape [B, N, N] (bool/float)
        - weights: dict of torch.Tensors with all linear / LN params
        - config : dict containing at least `dim` and `hidden_dim`

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # LayerNorm on the input tensor (dim -> dim)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # compute mean/var over last dim
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * norm_w + norm_b

    # ------------------------------------------------------------------
    # Linear projections (dim -> hidden_dim)
    # ------------------------------------------------------------------
    left_proj = torch.matmul(x_norm, weights["left_proj.weight"].t())
    right_proj = torch.matmul(x_norm, weights["right_proj.weight"].t())

    # ------------------------------------------------------------------
    # Optional mask (broadcast to hidden dim)
    # ------------------------------------------------------------------
    if mask is None or config.get("nomask", False):
        mask_f = torch.ones((B, N, N, 1), dtype=x.dtype, device=device)
    else:
        mask_f = mask.unsqueeze(-1).to(x.dtype)
    left_proj = left_proj * mask_f
    right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # Gating scalars (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.matmul(x_norm, weights["left_gate.weight"].t()))
    right_gate = torch.sigmoid(torch.matmul(x_norm, weights["right_gate.weight"].t()))
    out_gate = torch.sigmoid(torch.matmul(x_norm, weights["out_gate.weight"].t()))

    # Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # Triton contraction: out = sum_k left[...,k,:] * right[...,k,:]
    # ------------------------------------------------------------------
    # Ensure contiguous layout for the kernel
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides (in elements) for each tensor
    s_left = left.stride()   # (b,i,k,h)
    s_right = right.stride()
    s_out = out.stride()

    # Choose block sizes (empirically reasonable for N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        ( (N + BLOCK_M - 1) // BLOCK_M,
          (N + BLOCK_N - 1) // BLOCK_N,
          B * hidden_dim ),
    )

    tri_mul_kernel[grid](
        left_ptr=left.data_ptr(),
        right_ptr=right.data_ptr(),
        out_ptr=out.data_ptr(),
        B=B,
        N=N,
        H=hidden_dim,
        stride_left_b=s_left[0],
        stride_left_i=s_left[1],
        stride_left_k=s_left[2],
        stride_left_h=s_left[3],
        stride_right_b=s_right[0],
        stride_right_j=s_right[1],
        stride_right_k=s_right[2],
        stride_right_h=s_right[3],
        stride_out_b=s_out[0],
        stride_out_i=s_out[1],
        stride_out_j=s_out[2],
        stride_out_h=s_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # LayerNorm on the hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) * torch.rsqrt(var_o + eps)
    out_norm = out_norm * to_out_norm_w + to_out_norm_b

    # ------------------------------------------------------------------
    # Apply output gate and final linear projection back to `dim`
    # ------------------------------------------------------------------
    out_gated = out_norm * out_gate
    to_out_w = weights["to_out.weight"]          # shape [dim, hidden_dim]
    result = torch.matmul(out_gated, to_out_w.t())

    return result
