                   stride_r_batch, stride_r_j, stride_r_k, stride_r_d,

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_d,
    stride_r_batch, stride_r_j, stride_r_k, stride_r_d,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_K: tl.constexpr
):
    # program_id dimensions
    i = tl.program_id(0)
    j = tl.program_id(1)
    bd = tl.program_id(2)

    # extract batch and hidden indices
    d = bd % D
    b = bd // D

    # bounds check
    if i >= N or j >= N or b >= B or d >= D:
        return

    # accumulator
    acc = tl.float32(0.0)

    # loop over k blocks
    # compute number of blocks
    num_k_blocks = (N + BLOCK_K - 1) // BLOCK_K
    for idx in range(num_k_blocks):
        k_start = idx * BLOCK_K
        # compute offset pointer for left and right
        # k offsets
        k_off = k_start + tl.arange(0, BLOCK_K)
        mask = k_off < N

        left_offset = b * stride_l_batch + i * stride_l_i + k_off * stride_l_k + d * stride_l_d
        right_offset = b * stride_r_batch + j * stride_r_j + k_off * stride_r_k + d * stride_r_d

        left_vals = tl.load(left_ptr + left_offset, mask=mask, other=0.0)
        right_vals = tl.load(right_ptr + right_offset, mask=mask, other=0.0)

        # dot product across K dimension
        acc = acc + tl.dot(left_vals, right_vals)

    # store result
    out_offset = b * stride_out_batch + i * stride_out_i + j * stride_out_j + d * stride_out_d
    tl.store(out_ptr + out_offset, acc)

import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                   B, N, D,
                   stride_l_batch, stride_l_i, stride_l_k, stride_l_d,
                   stride_r_batch, stride_r_j, stride_r_k, stride_r_d,
                   stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
                   BLOCK_K: tl.constexpr):
    i = tl.program_id(0)
    j = tl.program_id(1)
    bd = tl.program_id(2)
    # derive batch and hidden dimension index
    d = bd % D
    b = bd // D

    # bounds check (early exit)
    if i >= N or j >= N or b >= B or d >= D:
        return

    # accumulate
    acc = tl.float32(0.0)

    k = tl.int32(0)
    while k < N:
        cur_k = k + tl.arange(0, BLOCK_K)  # [BLOCK_K]
        mask = cur_k < N

        # compute pointer offsets
        left_offset = b * stride_l_batch + i * stride_l_i + cur_k * stride_l_k + d * stride_l_d
        right_offset = b * stride_r_batch + j * stride_r_j + cur_k * stride_r_k + d * stride_r_d

        left_vals = tl.load(left_ptr + left_offset, mask=mask, other=tl.float32(0.0))
        right_vals = tl.load(right_ptr + right_offset, mask=mask, other=tl.float32(0.0))

        acc = acc + tl.dot(left_vals, right_vals)

        k += BLOCK_K

    # write the result
    out_offset = b * stride_out_batch + i * stride_out_i + j * stride_out_j + d * stride_out_d
    tl.store(out_ptr + out_offset, acc)

def custom_kernel(data):
    """
    TriMul “outgoing” forward pass implemented with a fused Triton kernel for the
    core O(N³) reduction. All other ops (LayerNorm, linear projections, and
    gating) are performed with PyTorch for clarity.
    """
    x, mask, weights, config = data

    device = x.device
    dtype = x.dtype
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # LayerNorm over channel dim (dim)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        x, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias, eps=eps
    )

    # Linear projections (no bias)
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask if provided (mask shape [B,N,N])
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(dtype)   # shape [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Allocate output tensor for the O(N³) reduction
    out = torch.empty_like(left)   # shape [B,N,N,hidden_dim]

    # Strides for the Triton kernel
    B, N, _, HD = left.shape
    stride_l = left.stride()   # (stride_b, stride_i, stride_k, stride_d)
    stride_r = right.stride()
    stride_o = out.stride()

    # Flatten batch and hidden dimension into a single program-id dimension
    # grid: (N, N, B*HD)
    grid = (N, N, B * HD)

    # Launch Triton kernel
    tri_mul_kernel[grid](
        left, right, out,
        B, N, HD,
        stride_l[0], stride_l[1], stride_l[2], stride_l[3],
        stride_r[0], stride_r[1], stride_r[2], stride_r[3],
        stride_o[0], stride_o[1], stride_o[2], stride_o[3],
        BLOCK_K=32,
    )

    # Apply to_out_norm (LayerNorm on hidden dimension)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out, normalized_shape=(hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps
    )

    # Output gating
    out = out * out_gate

    # Final linear projection back to dim
    to_out_weight = weights["to_out.weight"]
    output = torch.nn.functional.linear(out, to_out_weight)   # shape [B,N,N,dim]

    return output

    j = tl.program_id(1)   # 0 <= j < N

"""
TriMul “outgoing” forward pass.

The heavy O(N³) reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented with a custom Triton kernel. All other
operations (LayerNorm, linear projections, gating and final
projection) are performed with PyTorch for clarity.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,         # const * left  [B,N,N,HD]
    right_ptr,        # const * right [B,N,N,HD]
    out_ptr,          # * out          [B,N,N,HD]
    B, N, HD,         # int scalars
    stride_l_batch, stride_l_i, stride_l_k, stride_l_d,
    stride_r_batch, stride_r_j, stride_r_k, stride_r_d,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_K: tl.constexpr,
):
    # program ids index the output element (i, j, batch*hd)
    i = tl.program_id(0)            # 0 <= i < N
    j = tl.program_id(1)            # 0 <= j < N
    bd = tl.program_id(2)           # 0 <= bd < B*HD

    # decode batch index and hidden dimension index
    d = bd % HD
    b = bd // HD

    # accumulator (fp32)
    acc = tl.float32(0.0)

    # loop over k in BLOCK_K sized chunks
    k = tl.int32(0)
    while k < N:
        cur_k = k + tl.arange(0, BLOCK_K)               # [BLOCK_K]
        mask = cur_k < N                                 # tail handling

        # left[b, i, cur_k, d]
        left_off = b * stride_l_batch + i * stride_l_i + cur_k * stride_l_k + d * stride_l_d
        left_vals = tl.load(left_ptr + left_off, mask=mask, other=tl.float32(0.0))

        # right[b, j, cur_k, d]
        right_off = b * stride_r_batch + j * stride_r_j + cur_k * stride_r_k + d * stride_r_d
        right_vals = tl.load(right_ptr + right_off, mask=mask, other=tl.float32(0.0))

        # dot product over the BLOCK_K element vector
        acc = acc + tl.dot(left_vals, right_vals)

        k += BLOCK_K

    # write the result
    out_off = b * stride_o_batch + i * stride_o_i + j * stride_o_j + d * stride_o_d
    tl.store(out_ptr + out_off, acc)


def custom_kernel(data):
    """
    Forward pass of the Triangle Multiplicative Update (outgoing mode).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] (broadcastable)
        - weights      : dict of torch.Tensor parameters (see code)
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim]
    """
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # LayerNorm over the input channel dimension
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # -----------------------------------------------------------------
    # Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # [B,N,N,HD]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # -----------------------------------------------------------------
    # Apply mask (if present)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # Gating signals (sigmoid)
    # -----------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    left = (left * left_gate).contiguous()
    right = (right * right_gate).contiguous()

    # -----------------------------------------------------------------
    # Core O(N³) reduction via Triton
    # -----------------------------------------------------------------
    B, N, _, HD = left.shape
    out = torch.empty_like(left)  # [B,N,N,HD]

    # Strides in element units (not bytes)
    stride_l = left.stride()   # (batch, i, k, d)
    stride_r = right.stride()
    stride_o = out.stride()

    # Grid: one program per output element (i, j, batch*hd)
    grid = (N, N, B * HD)

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        HD,
        stride_l[0],
        stride_l[1],
        stride_l[2],
        stride_l[3],
        stride_r[0],
        stride_r[1],
        stride_r[2],
        stride_r[3],
        stride_o[0],
        stride_o[1],
        stride_o[2],
        stride_o[3],
        BLOCK_K=32,  # reasonable block size for the reduction loop
    )

    # -----------------------------------------------------------------
    # LayerNorm on hidden dimension
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # -----------------------------------------------------------------
    # Output gating and final projection back to original dim
    # -----------------------------------------------------------------
    out = out * out_gate
    to_out_weight = weights["to_out.weight"]
    output = torch.nn.functional.linear(out, to_out_weight)   # [B,N,N,dim]

    return output
