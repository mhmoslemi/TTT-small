import triton
import triton.language as tl

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb0, stride_lb1, stride_lb2, stride_lb3,
    stride_rb0, stride_rb1, stride_rb2, stride_rb3,
    stride_ob0, stride_ob1, stride_ob2, stride_ob3,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    pid_d = tl.program_id(3)

    # offsets
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)  # used inside loop
    d = pid_d * BLOCK_D + tl.arange(0, BLOCK_D)

    # masks for out-of-range
    mask_i = i < N
    mask_j = j < N
    mask_d = d < H

    # pointers to output start
    out_ptrs = out_ptr + pid_b * stride_ob0 + i[:, None, None] * stride_ob1 + j[None, :, None] * stride_ob2 + d[None, None, :] * stride_ob3

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # loop over K dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = kk * BLOCK_K + k
        mask_k = k_offset < N

        # left block pointer: shape (M, K, D)
        left_ptrs = left_ptr + pid_b * stride_lb0 + i[:, None, None] * stride_lb1 + k_offset[None, :, None] * stride_lb2 + d[None, None, :] * stride_lb3
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :] & mask_d[None, None, :], other=0.0).to(tl.float32)

        # right block pointer: shape (K, N, D)
        right_ptrs = right_ptr + pid_b * stride_rb0 + k_offset[:, None, None] * stride_rb2 + j[None, :, None] * stride_rb1 + d[None, None, :] * stride_rb3
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :] & mask_d[None, None, :], other=0.0).to(tl.float32)

        # compute gemm per d
        # left: (M,K,D), right: (K,N,D)
        # compute for each d
        for _d in range(0, BLOCK_D):
            # note: only compute where mask_d is true
            # but loops also compute for all; we will mask later in store.
            # tl dot: left_slice: (M,K), right_slice: (K,N)
            left_slice = left[:, :, _d]
            right_slice = right[:, :, _d]
            # accumulate
            acc[:, :, _d] = acc[:, :, _d] + tl.dot(left_slice, right_slice)
    # store result with mask
    tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # config: contains dim, hidden_dim, maybe bs? but we can compute from input tensor shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    # Handle mask
    if nomask or mask is None:
        mask = torch.ones(input_tensor.shape[:3], dtype=input_tensor.dtype, device=input_tensor.device)
    # Ensure mask shape matches input
    mask = mask.to(input_tensor.dtype)
    # Normalize input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask
    mask_expand = mask.unsqueeze(-1)  # shape [B,N,N,1]
    left = left * mask_expand
    right = right * mask_expand

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Core triple product via Triton
    out_mul = trimul_outgoing_kernel_launcher(left, right)

    # Normalize and output gating
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_mul = torch.nn.functional.layer_norm(out_mul, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    out_mul = out_mul * out_gate  # elementwise

    # Final linear projection
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out_mul, to_out_weight)  # shape [B,N,N,dim]

    return out

import torch
import triton
import triton.language as tl

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb0, stride_lb1, stride_lb2, stride_lb3,
    stride_rb0, stride_rb1, stride_rb2, stride_rb3,
    stride_ob0, stride_ob1, stride_ob2, stride_ob3,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    pid_d = tl.program_id(3)

    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    d = pid_d * BLOCK_D + tl.arange(0, BLOCK_D)

    mask_i = i < N
    mask_j = j < N
    mask_d = d < H

    # accumulator for this tile
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # iterate over k dimension
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # left block: shape (M, K, D)
        left_ptrs = left_ptr + \
            pid_b * stride_lb0 + \
            i[:, None, None] * stride_lb1 + \
            k[None, :, None] * stride_lb2 + \
            d[None, None, :] * stride_lb3
        left_mask = mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :]
        left = tl.load(left_ptrs, mask=left_mask, other=0.0).to(tl.float32)

        # right block: shape (K, N, D)
        right_ptrs = right_ptr + \
            pid_b * stride_rb0 + \
            k[:, None, None] * stride_rb2 + \
            j[None, :, None] * stride_rb1 + \
            d[None, None, :] * stride_rb3
        right_mask = mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
        right = tl.load(right_ptrs, mask=right_mask, other=0.0).to(tl.float32)

        # perform GEMM for each d
        for d_idx in range(0, BLOCK_D):
            # skip d if out of range (optional)
            # compute dot product left_slice (M,K) * right_slice (K,N)
            left_slice = left[:, :, d_idx]   # (M, K)
            right_slice = right[:, :, d_idx] # (K, N)
            acc[:, :, d_idx] = acc[:, :, d_idx] + tl.dot(left_slice, right_slice)
    # store result
    out_ptrs = out_ptr + \
        pid_b * stride_ob0 + \
        i[:, None, None] * stride_ob1 + \
        j[None, :, None] * stride_ob2 + \
        d[None, None, :] * stride_ob3
    out_mask = mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
    tl.store(out_ptrs, acc, mask=out_mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Ensure tensors are on same device
    device = input_tensor.device
    dtype = input_tensor.dtype

    # handle mask
    if nomask or mask is None:
        mask = torch.ones((input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]), dtype=dtype, device=device)
    else:
        mask = mask.to(dtype)

    # LayerNorm on input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # apply mask
    mask_exp = mask.unsqueeze(-1)  # [B,N,N,1]
    left = left * mask_exp
    right = right * mask_exp

    # gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # core multiplication via Triton kernel
    out = trimul_outgoing_kernel_launcher(left, right)

    # output normalization
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # apply out gate
    out = out * out_gate

    # final linear
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

    B, N, _, H = left.shape

    # strides as ints

    stride_rb0, stride_rb1, stride_rb2, stride_rb3 = right.stride()

            (N + BLOCK_N - 1) // BLOCK_N,

        stride_ob0, stride_ob1, stride_ob2, stride_ob3,

"""
TriMul (outgoing) implementation with a Triton kernel for the heavy
pairwise multiplication (einsum) step.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Linear projections (left/right) and three gating linear layers.
3. Apply mask (if provided) and element‑wise gates.
4. Compute out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d] with a custom
   Triton kernel. The kernel tiles the (i,j) output plane (BLOCK_M×BLOCK_N)
   and the channel dimension (BLOCK_D) while looping over the sequence
   dimension k in blocks (BLOCK_K).  Accumulation is done in FP32.
5. Apply a second Layer‑norm over the hidden dimension, multiply by the
   out‑gate, and finish with a linear projection back to the original
   dimension.

The kernel handles arbitrary batch size (B∈{1,2}), sequence length N
(up to 1024) and hidden dimension H (typically 128).  It falls back to
PyTorch for the inexpensive linear and LayerNorm operations, ensuring
correctness while the expensive O(N³·H) work is done in Triton.
"""

import triton
import triton.language as tl
import torch.nn.functional as F

def _trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,

    stride_lb0, stride_lb1, stride_lb2, stride_lb3,
    stride_rb0, stride_rb1, stride_rb2, stride_rb3,
    stride_ob0, stride_ob1, stride_ob2, stride_ob3,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]"""
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    pid_d = tl.program_id(3)

    # tile offsets
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    d = pid_d * BLOCK_D + tl.arange(0, BLOCK_D)

    mask_i = i < N
    mask_j = j < N
    mask_d = d < H

    # accumulator for this (i,j,d) tile
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # loop over the reduction axis k
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # load left block: shape (M, K, D)
        left_ptrs = (
            left_ptr
            + pid_b * stride_lb0
            + i[:, None, None] * stride_lb1
            + k[None, :, None] * stride_lb2
            + d[None, None, :] * stride_lb3
        )
        left_mask = mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :]
        left = tl.load(left_ptrs, mask=left_mask, other=0.0).to(tl.float32)

        # load right block transposed: shape (K, N, D)
        right_ptrs = (
            right_ptr
            + pid_b * stride_rb0
            + k[:, None, None] * stride_rb2
            + j[None, :, None] * stride_rb1
            + d[None, None, :] * stride_rb3
        )
        right_mask = mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
        right = tl.load(right_ptrs, mask=right_mask, other=0.0).to(tl.float32)

        # GEMM for each d slice
        for d_idx in range(0, BLOCK_D):
            # left_slice: (M, K), right_slice: (K, N)
            left_slice = left[:, :, d_idx]
            right_slice = right[:, :, d_idx]
            acc[:, :, d_idx] = acc[:, :, d_idx] + tl.dot(left_slice, right_slice)

    # write the result
    out_ptrs = (
        out_ptr
        + pid_b * stride_ob0
        + i[:, None, None] * stride_ob1
        + j[None, :, None] * stride_ob2
        + d[None, None, :] * stride_ob3
    )
    out_mask = mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
    tl.store(out_ptrs, acc, mask=out_mask)


def _trimul_outgoing_kernel_launcher(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """Launch the Triton kernel.  left/right shape: [B,N,N,H] (contiguous)."""
    assert left.shape == right.shape
    B, N, _, H = left.shape
    out = torch.empty_like(left)  # same dtype & device

    # strides in elements
    lb0, lb1, lb2, lb3 = left.stride()
    rb0, rb1, rb2, rb3 = right.stride()
    ob0, ob1, ob2, ob3 = out.stride()

    # Tunable tile sizes (work well for N≤1024, H=128)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 64
    BLOCK_D = 32

    grid = (
        B,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        (H + BLOCK_D - 1) // BLOCK_D,
    )
    _trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        lb0,
        lb1,
        lb2,
        lb3,
        rb0,
        rb1,
        rb2,
        rb3,
        ob0,
        ob1,
        ob2,
        ob3,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the "outgoing" TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim] (torch.Tensor)
            - mask: [B, N, N] (torch.Tensor) – may be ignored if config["nomask"] is True
            - weights: dict of torch.Tensors with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict containing at least "dim", "hidden_dim", and "nomask"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # Mask handling
    # ------------------------------------------------------------------
    if nomask or mask is None:
        mask = torch.ones(
            (input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]),
            dtype=dtype,
            device=device,
        )
    else:
        mask = mask.to(dtype)

    # ------------------------------------------------------------------
    # First LayerNorm (over the last dimension = dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # Linear projections
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left = F.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # Apply mask (broadcasted over hidden_dim)
    # ------------------------------------------------------------------
    mask_exp = mask.unsqueeze(-1)     # [B,N,N,1]
    left = left * mask_exp
    right = right * mask_exp

    # ------------------------------------------------------------------
    # Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Heavy pairwise multiplication via Triton
    # ------------------------------------------------------------------
    out_mul = _trimul_outgoing_kernel_launcher(left, right)  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # Second LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_mul = F.layer_norm(
        out_mul, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b
    )
    out_mul = out_mul * out_gate

    to_out_w = weights["to_out.weight"]
    out = F.linear(out_mul, to_out_w)  # final shape [B,N,N,dim]

    return out
