"""
TriMul (outgoing) implementation with a custom Triton kernel.

The heavy part of the forward pass is the triple sum
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
which is a batch of matrix multiplications L @ Rᵀ for each hidden channel `h`.
A Triton GEMM kernel computes this term in mixed precision (FP16 inputs,
FP32 accumulation). The remaining operations (layer‑norms, linear layers,
gates) are performed with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,                      # *fp16   left tensor
    right_ptr,                     # *fp16   right tensor
    out_ptr,                       # *fp32   result tensor
    B, N, H,                       # runtime constants
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute out = left @ rightᵀ for each (batch, hidden) pair.

    left  : [B, N, N, H]   (i, k)
    right : [B, N, N, H]   (j, k)   -> loaded as (k, j) to get Rᵀ
    out   : [B, N, N, H]   (i, j)
    """
    pid_bc = tl.program_id(0)   # combined batch * hidden
    pid_m = tl.program_id(1)    # block row (i)
    pid_n = tl.program_id(2)    # block col (j)

    # derive batch and hidden indices
    b = pid_bc // H
    h = pid_bc % H

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension in tiles
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- load left tile (M x K) ----
        left_ptrs = (
            left_ptr
            + b * stride_l_b
            + (offs_m[:, None]) * stride_l_i
            + (offs_k[None, :]) * stride_l_k
            + h * stride_l_h
        )
        left_tile = tl.load(
            left_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # ---- load right tile transposed (K x N) ----
        # right is indexed as (j, k); we load (k, j) to get Rᵀ.
        right_ptrs = (
            right_ptr
            + b * stride_r_b
            + (offs_k[:, None]) * stride_r_k
            + (offs_n[None, :]) * stride_r_j
            + h * stride_r_h
        )
        right_tile = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        # accumulate the dot product
        acc += tl.dot(left_tile, right_tile)   # (M,K) @ (K,N) -> (M,N)

    # ---- write result ----
    out_ptrs = (
        out_ptr
        + b * stride_o_b
        + (offs_m[:, None]) * stride_o_i
        + (offs_n[None, :]) * stride_o_j
        + h * stride_o_h
    )
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    # Config
    dim = config["dim"]            # input / output channel dimension
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Initial LayerNorm over the last dimension
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left = F.linear(x, left_proj_w)      # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(left.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to FP16 for the heavy GEMM
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape
    # Output tensor after the triple sum
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides (in elements)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_fp16.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right_fp16.stride()
    # For the right tensor we treat the first N dimension as "j"
    stride_r_j = stride_r_i

    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # ------------------------------------------------------------------
    # 6. Launch Triton kernel (batched GEMM per hidden channel)
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                 # program ids for (batch, hidden)
        triton.cdiv(N, BLOCK_M),       # rows
        triton.cdiv(N, BLOCK_N),       # cols
    )

    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B,
        N,
        hidden_dim,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_j,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7. Post‑processing: LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b, eps=1e-5
    )
    out = out_norm * out_gate

    to_out_w = weights["to_out.weight"]
    output = F.linear(out, to_out_w)   # shape [B,N,N,dim]

    return output
