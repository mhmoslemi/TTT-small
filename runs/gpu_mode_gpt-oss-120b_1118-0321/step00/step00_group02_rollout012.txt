def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_bi, stride_bk, stride_bh,
    stride_rl, stride_ri, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch_hidden = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # decode batch and hidden
    b = pid_batch_hidden // H
    h = pid_batch_hidden % H

    # compute base pointers for this b and h
    left_base = left_ptr + b * stride_bl + h * stride_bh
    right_base = right_ptr + b * stride_rl + h * stride_rh
    out_base = out_ptr + b * stride_ob + h * stride_oh

    # ranges
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for i, j bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for start_k in range(0, N, BLOCK_K):
        cur_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # load left tile [M,K]
        left_ptrs = left_base + (offs_i[:, None] * stride_bi) + (cur_k[None, :] * stride_bk)

"""
TriMul (outgoing) implementation with a fused Triton kernel.

The kernel computes the core O(N³) update:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

where left/right are the projected and gated pairwise representations.
The reduction over k is performed as a batched matrix multiplication using
tiling (BLOCK_M×BLOCK_N tiles) and accumulation in fp32 while loading in fp16.
All other operations (layer‑norm, linear projections, gating, final projection)
are done with PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM over the third sequence dimension (k)
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # batch, seq_len, hidden_dim
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,   # left strides
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,   # right strides
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out = Σ_k left[...,k,:] * right[...,k,:] using tiled GEMM."""
    pid_batch_hidden = tl.program_id(0)      # combined batch * hidden
    pid_i = tl.program_id(1)                # tile index along i
    pid_j = tl.program_id(2)                # tile index along j

    # Decode batch and hidden indices
    b = pid_batch_hidden // H
    h = pid_batch_hidden % H

    # Base pointers for this (b, h)
    left_base = left_ptr + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base = out_ptr + b * stride_o_b + h * stride_o_h

    # Offsets for the current tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension in BLOCK_K sized chunks
    for start_k in range(0, N, BLOCK_K):
        cur_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # ----- load left tile [M, K] -----
        # left[b, i, k, h]
        left_ptrs = (
            left_base
            + offs_i[:, None] * stride_l_i
            + cur_k[None, :] * stride_l_k
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # ----- load right tile [K, N] (transposed view) -----
        # right[b, j, k, h]  -> we treat as (K, N) with k as rows, j as cols
        right_ptrs = (
            right_base
            + cur_k[:, None] * stride_r_k
            + offs_j[None, :] * stride_r_i
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # Cast to fp32 and perform dot product
        left = left.to(tl.float32)
        right = right.to(tl.float32)
        acc += tl.dot(left, right)   # (M,K) @ (K,N) -> (M,N)

    # Store the result
    out_ptrs = (
        out_base
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# --------------------------------------------------------------------------- #
# Main entry point: custom_kernel
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (float32 or bool) or None
        - weights      : dict of model parameters
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------- #
    # LayerNorm on the input (over channel dim)
    # ------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # shape [B, N, N, dim]

    # ------------------------------------------------------------------- #
    # Linear projections
    # ------------------------------------------------------------------- #
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ------------------------------------------------------------------- #
    # Gating pathways
    # ------------------------------------------------------------------- #
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # ------------------------------------------------------------------- #
    # Apply mask (if present) and gating
    # ------------------------------------------------------------------- #
    if mask is None or config.get("nomask", False):
        mask_f = torch.ones_like(x[..., :1])
    else:
        mask_f = mask.unsqueeze(-1).to(x.dtype)  # [B,N,N,1]
    left = left * mask_f * left_gate
    right = right * mask_f * right_gate

    # ------------------------------------------------------------------- #
    # Cast to fp16 for the Triton kernel (reduces memory traffic)
    # ------------------------------------------------------------------- #
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, H = left_h.shape  # B, seq_len, seq_len, hidden_dim

    # Allocate output for the hidden representation
    out_h = torch.empty_like(left_h)  # [B,N,N,hidden] in fp16

    # Strides in elements
    s_l_b, s_l_i, s_l_k, s_l_h = left_h.stride()
    s_r_b, s_r_i, s_r_k, s_r_h = right_h.stride()
    s_o_b, s_o_i, s_o_j, s_o_h = out_h.stride()

    # Tile sizes – feel free to tune; 64 works well on modern GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid: (batch*hidden, i_tiles, j_tiles)
    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Launch the Triton kernel
    tri_mul_kernel[grid](
        left_h,
        right_h,
        out_h,
        B,
        N,
        H,
        s_l_b,
        s_l_i,
        s_l_k,
        s_l_h,
        s_r_b,
        s_r_i,
        s_r_k,
        s_r_h,
        s_o_b,
        s_o_i,
        s_o_j,
        s_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Ensure completion (useful for timing)
    torch.cuda.synchronize()

    # ------------------------------------------------------------------- #
    # Rest of the pipeline (LayerNorm, gating, final linear)
    # ------------------------------------------------------------------- #
    out_fp32 = out_h.to(torch.float32)  # convert back to fp32

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_normed = torch.nn.functional.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # Apply final gate and output projection
    out_normed = out_normed * out_gate
    to_out_weight = weights["to_out.weight"]
    output = torch.nn.functional.linear(out_normed, to_out_weight)  # [B,N,N,dim]

    return output
