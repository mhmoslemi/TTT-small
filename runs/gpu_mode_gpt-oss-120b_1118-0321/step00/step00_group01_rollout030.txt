    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'], eps=eps)

    # Linear projections
    left = torch.nn.functional.linear(x, weights['left_proj.weight'])
    right = torch.nn.functional.linear(x, weights['right_proj.weight'])
    # left/right shape (B,N,N,hidden)

    # mask
    if mask is not None:
        left = left * mask.unsqueeze(-1)
        right = right * mask.unsqueeze(-1)

    # gates
    left_gate = torch.nn.functional.linear(x, weights['left_gate.weight']).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights['right_gate.weight']).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights['out_gate.weight']).sigmoid()
    left = left * left_gate
    right = right * right_gate

    # Compute core einsum via Triton
    # Permute and flatten
    left_perm = left.permute(0,3,1,2).contiguous()  # B, hidden, N, N
    right_perm = right.permute(0,3,1,2).contiguous()

    B_hidden = left_perm.shape[0] * left_perm.shape[1]   # B*hidden

"""
TriMul (outgoing) forward kernel.

The implementation follows the AlphaFold3 reference:
1. LayerNorm over the input channel dimension.
2. Linear projections (left/right) and gating (left_gate/right_gate/out_gate).
3. Optional mask application.
4. Core triple multiplication:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM (left @ rightᵀ) using a tiled Triton kernel.
5. LayerNorm over the hidden dimension, gating with out_gate, and final linear
   projection back to the original dim.

Only the O(N³) multiplication is performed in Triton; all other operations
use PyTorch which already provides highly‑optimized kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    A, B, C,                     # pointers to input and output tensors
    N,                           # sequence length
    stride_aa, stride_am, stride_ak,   # A: batch, row(i), col(k) strides
    stride_bb, stride_bm, stride_bn,   # B: batch, row(k), col(j) strides
    stride_cc, stride_cm, stride_cn,   # C: batch, row(i), col(j) strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)   # batch*hidden index
    pid_m = tl.program_id(1)       # i‑block
    pid_n = tl.program_id(2)       # j‑block

    # -------------------- offsets & masks --------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this batch
    a_base = A + pid_batch * stride_aa
    b_base = B + pid_batch * stride_bb
    c_base = C + pid_batch * stride_cc

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------- reduction over k --------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(num_k_blocks):
        cur_k = k * BLOCK_K
        cur_k_offs = cur_k + offs_k          # absolute k indices
        mask_k = cur_k_offs < N

        # Load A block (i,k)
        a_ptr = a_base + offs_m[:, None] * stride_am + cur_k_offs[None, :] * stride_ak
        a = tl.load(a_ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B block (k,j) – note B is already transposed (row = k, col = j)
        b_ptr = b_base + cur_k_offs[:, None] * stride_bm + offs_n[None, :] * stride_bn
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # -------------------- write back --------------------
    c_ptr = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of TriMul (outgoing) using a fused Triton kernel for the O(N³) step.
    Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [batch, seq_len, seq_len, dim] (float32, on the same device).
    """
    # unpack
    x, mask, weights, config = data
    device = x.device
    eps = 1e-5

    B, N, _, dim = x.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (flatten -> Linear -> reshape)
    # ------------------------------------------------------------------
    x_flat = x.view(-1, dim)                     # (B*N*N, dim)

    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden)

    # ------------------------------------------------------------------
    # 3. Optional mask
    # ------------------------------------------------------------------
    if not config.get("nomask", True) and mask is not None:
        mask_exp = mask[..., None]               # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating (project -> sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden).sigmoid()
    right_gate = F.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden).sigmoid()
    out_gate = F.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core multiplication: einsum('... i k d, ... j k d -> ... i j d')
    #    Implemented as batched GEMM with Triton.
    # ------------------------------------------------------------------
    # Permute to (B, hidden, N, N) and make right transposed (k as rows)
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()        # (B, H, N, N)
    right_T = right_perm.transpose(-2, -1).contiguous()       # (B, H, N, N) where rows = k

    # Flatten batch*hidden for the kernel
    BH = B * hidden
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_T.view(BH, N, N)
    out_flat = torch.empty_like(left_flat)

    # Block sizes – chosen to give good occupancy on B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        left_flat.stride(0),   # batch stride (N*N)
        left_flat.stride(1),   # row (i) stride (N)
        left_flat.stride(2),   # col (k) stride (1)
        right_flat.stride(0),  # batch stride (N*N)
        right_flat.stride(1),  # row (k) stride (N)
        right_flat.stride(2),  # col (j) stride (1)
        out_flat.stride(0),    # batch stride (N*N)
        out_flat.stride(1),    # row (i) stride (N)
        out_flat.stride(2),    # col (j) stride (1)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, hidden, N, N) then permute to (B, N, N, hidden)
    out_perm = out_flat.view(B, hidden, N, N)
    out = out_perm.permute(0, 2, 3, 1)   # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating & final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    out = out * out_gate

    # final linear projection back to dim
    out_flat = out.view(-1, hidden)
    out = F.linear(out_flat, weights["to_out.weight"]).view(B, N, N, dim)

    return out
