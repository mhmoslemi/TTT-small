@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, EPS: tl.constexpr = 1e-5,
):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # compute masks for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # a_ptr base
    a_ptrs = a_ptr + batch * batch_stride_a + offs_m[:,None] * stride_am + offs_k[None,:] * stride_ak
    b_ptrs = b_ptr + batch * batch_stride_b + offs_k[:,None] * stride_bk + offs_n[None,:] * stride_bn

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute offsets within block
        cur_k = k * BLOCK_K
        cur_ak = cur_k * stride_ak
        cur_bk = cur_k * stride_bk

        a = tl.load(a_ptrs + cur_ak, mask=mask_m[:,None] & (offs_k[None,:] < K - cur_k), other=0.0)
        b = tl.load(b_ptrs + cur_bk, mask=(offs_k[:,None] < K - cur_k) & mask_n[None,:], other=0.0)

        # dot
        acc += tl.dot(a, b)

        # advance pointers
        # Not needed because we recompute pointers each iteration? use offset inside loop

    # apply mask for output bounds
    mask_out = mask_m[:,None] & mask_n[None,:]

    # store
    c_ptrs = c_ptr + batch * batch_stride_c + offs_m[:,None] * stride_cm + offs_n[None,:] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float32), mask=mask_out)

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_batch_a, stride_m_a, stride_k_a,
    stride_batch_b, stride_k_b, stride_n_b,
    stride_batch_c, stride_m_c, stride_n_c,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # pointer increments
    a_ptrs = a_ptr + batch_id * stride_batch_a + offs_m[:, None] * stride_m_a + offs_k[None, :] * stride_k_a
    b_ptrs = b_ptr + batch_id * stride_batch_b + offs_k[:, None] * stride_k_b + offs_n[None, :] * stride_n_b

    # masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        a = tl.load(a_ptrs + k_offset * stride_k_a,
                    mask=mask_m[:, None] & (offs_k + k_offset < K),
                    other=0.0)
        b = tl.load(b_ptrs + k_offset * stride_k_b,
                    mask=(offs_k + k_offset < K)[:, None] & mask_n[None, :],
                    other=0.0)
        # compute dot product of shape (BLOCK_M, BLOCK_K) * (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)

    # write-out
    c_ptrs = c_ptr + batch_id * stride_batch_c + offs_m[:, None] * stride_m_c + offs_n[None, :] * stride_n_c
    mask_out = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float32), mask=mask_out)
```

"""
TriMul outgoing version (AlphaFold3) implemented with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension using the provided
   `norm.weight` and `norm.bias`.
2. Linear projections (no bias) produce `left` and `right` of shape
   [B, N, N, H] where H = hidden_dim.
3. Apply the optional mask (broadcast on the last dim) to both `left` and
   `right`.
4. Compute three gates (sigmoid of linear projections of the normalized input):
   `left_gate`, `right_gate`, `out_gate`.  Multiply `left` and `right` by the
   corresponding gates.
5. Core update: for each batch and hidden channel compute
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is equivalent to a batched matrix multiplication
        out_h = left_h @ right_h.T
   for every hidden slice `h`.  The heavy O(N³·H) computation is performed
   by a custom Triton kernel (`batched_matmul_kernel`) that processes all
   hidden slices as independent batch items.
6. Layer‑norm the result over the hidden dimension using `to_out_norm.weight`
   and `to_out_norm.bias`.
7. Multiply by `out_gate` (element‑wise) and apply the final linear projection
   (`to_out.weight`) back to the original channel dimension.
8. Return a tensor of shape [B, N, N, dim].

The kernel implements a conventional tiled GEMM (BLOCK_M × BLOCK_N) with an
inner reduction over BLOCK_K, accumulating in float32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_batch_a, stride_m_a, stride_k_a,
    stride_batch_b, stride_k_b, stride_n_b,
    stride_batch_c, stride_m_c, stride_n_c,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication:
        C[b, i, j] = Σ_k A[b, i, k] * B[b, k, j]
    A : (batch, M, K)
    B : (batch, K, N)   (note B is stored in the original layout (batch, N, K)
                         but we pass strides such that it is interpreted as (K,N))
    C : (batch, M, N)

    The kernel is tiled with blocks of size (BLOCK_M, BLOCK_N, BLOCK_K) and
    accumulates in float32.
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for start_k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = start_k * BLOCK_K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + batch_id * stride_batch_a
            + offs_m[:, None] * stride_m_a
            + (offs_k + cur_k)[None, :] * stride_k_a,
            mask=mask_m[:, None] & ((offs_k + cur_k) < K),
            other=0.0,
        )
        # Load B tile interpreted as (K, BLOCK_N)
        b = tl.load(
            b_ptr
            + batch_id * stride_batch_b
            + (offs_k + cur_k)[:, None] * stride_k_b
            + offs_n[None, :] * stride_n_b,
            mask=((offs_k + cur_k) < K)[:, None] & mask_n[None, :],
            other=0.0,
        )

        acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_N)

    # Write the result
    c_ptrs = (
        c_ptr
        + batch_id * stride_batch_c
        + offs_m[:, None] * stride_m_c
        + offs_n[None, :] * stride_n_c
    )
    mask_out = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=mask_out)


def custom_kernel(data):
    """
    Forward pass of the TriMul outgoing operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor of shape [B, N, N, dim]
        mask         : torch.Tensor of shape [B, N, N] (float 0/1)
        weights      : dict of torch.Tensors (model weights)
        config       : dict containing at least "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    eps = 1e-5

    # Configuration
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # Load weights and cast to correct device / dtype
    # ------------------------------------------------------------------
    def _to(t):
        return t.to(device=device, dtype=dtype)

    norm_weight = _to(weights["norm.weight"])
    norm_bias = _to(weights["norm.bias"])
    left_proj_weight = _to(weights["left_proj.weight"])
    right_proj_weight = _to(weights["right_proj.weight"])
    left_gate_weight = _to(weights["left_gate.weight"])
    right_gate_weight = _to(weights["right_gate.weight"])
    out_gate_weight = _to(weights["out_gate.weight"])
    to_out_norm_weight = _to(weights["to_out_norm.weight"])
    to_out_norm_bias = _to(weights["to_out_norm.bias"])
    to_out_weight = _to(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1) LayerNorm over the channel dimension (dim)
    # ------------------------------------------------------------------
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = ((input_tensor - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (input_tensor - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias  # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_norm, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x_norm, right_proj_weight)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3) Apply mask (if any)
    # ------------------------------------------------------------------
    mask_f = mask.unsqueeze(-1).to(dtype=dtype)  # broadcast over hidden dim
    left = left * mask_f
    right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))  # [B,N,N,H]
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))      # [B,N,N,H]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core multiplicative update: batched matmul (left @ right^T) per hidden slice
    #    We reshape (B, H, N, N) → (B*H, N, N) and launch a Triton kernel.
    # ------------------------------------------------------------------
    # Permute to bring hidden dim next to batch for easy flattening
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Collapse batch & hidden dimension
    batch_H = B * hidden_dim
    left_flat = left_perm.view(batch_H, N, N)   # [B*H, N, N]
    right_flat = right_perm.view(batch_H, N, N) # [B*H, N, N]

    # Output buffer
    out_flat = torch.empty_like(left_flat, dtype=left_flat.dtype, device=device)

    # Extract strides (in elements)
    a_batch_stride, a_m_stride, a_k_stride = left_flat.stride()
    b_batch_stride, b_k_stride, b_n_stride = right_flat.stride()
    c_batch_stride, c_m_stride, c_n_stride = out_flat.stride()

    # Choose block sizes (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch_H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        a_batch_stride,
        a_m_stride,
        a_k_stride,
        b_batch_stride,
        b_k_stride,
        b_n_stride,
        c_batch_stride,
        c_m_stride,
        c_n_stride,
        N,
        N,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6) Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    mean_out = out.mean(dim=-1, keepdim=True)
    var_out = ((out - mean_out) ** 2).mean(dim=-1, keepdim=True)
    out_norm = (out - mean_out) / torch.sqrt(var_out + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # ------------------------------------------------------------------
    # 7) Apply output gate
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 8) Final linear projection back to original dim
    # ------------------------------------------------------------------
    final = F.linear(out_norm, to_out_weight)  # [B, N, N, dim]

    return final
