def triton_einsum_kernel(left_ptr, right_ptr, out_ptr,
                         B, N, hidden,
                         stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
                         stride_right_batch, stride_right_i, stride_right_k, stride_right_h,
                         stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
                         BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0) and (1) new dims for grid.
    # Determine block row and col
    pid_i = tl.program_id(0)  # block row = i
    pid_j = tl.program_id(1)  # block col = j
    pid_bh = tl.program_id(2)
    batch_idx = pid_bh // hidden
    hidden_idx = pid_bh % hidden
    # compute offsets
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    k_offset = tl.arange(0, BLOCK_K)

    # mask out-of-range
    i_mask = i_offset < N
    j_mask = j_offset < N

    # load acc
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute start offset for this k block
        k_base = k * BLOCK_K
        # Offsets for left: i, k, hidden
        a_ptrs = left_ptr + batch_idx * stride_left_batch + hidden_idx * stride_left_h + \
                 (i_offset[:, None] * stride_left_i) + ((k_base + k_offset)[None, :] * stride_left_k)
        # Offsets for right transposed: k, j
        b_ptrs = right_ptr + batch_idx * stride_right_batch + hidden_idx * stride_right_h + \
                 ((k_base + k_offset)[:, None] * stride_right_k) + (j_offset[None, :] * stride_right_i)
        a = tl.load(a_ptrs, mask=(i_mask[:, None] & (k_base + k_offset < N)), other=0.0)
        b = tl.load(b_ptrs, mask=((k_base + k_offset)[:, None] < N) & j_mask[None, :], other=0.0)
        # compute dot
        acc += tl.dot(a, b)
    # store result
    out_ptrs = out_ptr + batch_idx * stride_out_batch + hidden_idx * stride_out_h + \
               (i_offset[:, None] * stride_out_i) + (j_offset[None, :] * stride_out_j)
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # extract shapes
    B,N,N2,dim = input_tensor.shape
    assert N == N2
    hidden_dim = config["hidden_dim"]

    # Retrieve weight tensors
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]; right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]; right_gate_weight = weights["right_gate.weight"]; out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]; to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    eps = 1e-5

    # Compute LayerNorm
    # compute mean and var over dim dimension
    # both are shape [B,N,N,1]; but we can use torch.mean and torch.var
    # Might be more efficient to use torch.nn.functional.layer_norm
    x_normalized = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Compute projections
    left = torch.nn.functional.linear(x_normalized, left_proj_weight)  # [B,N,N,hidden]
    right = torch.nn.functional.linear(x_normalized, right_proj_weight)

    # Apply mask: mask shape [B,N,N] to [B,N,N,1] cast to same dtype
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x_normalized.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # Compute gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_normalized, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_normalized, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_normalized, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Compute out using custom Triton kernel
    # We'll allocate output tensor shape [B,N,N,hidden], dtype float32
    out = torch.empty_like(left)  # same shape and dtype
    # Use contiguous memory; ensure left,right,out contiguous.
    left_contig = left.contiguous()
    right_contig = right.contiguous()
    out_contig = out  # already allocate

    # Get strides (in element units)
    stride_left = left_contig.stride()  # tuple of 4 ints: (stride_batch, stride_i, stride_k, stride_h)
    stride_right = right_contig.stride()
    stride_out = out_contig.stride()

    # Launch Triton kernel
    # Determine grid dimensions: M,N,... blocks
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64  # can be tuned
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )
    triton_einsum_kernel[grid](
        left_contig, right_contig, out_contig,
        B, N, hidden_dim,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Now out shape [B,N,N,hidden]
    # Apply to_out_norm: layer norm over hidden dimension
    # Use torch.nn.functional.layer_norm with normalized_shape=hidden_dim
    out_norm = torch.nn.functional.layer_norm(out_contig, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Apply out_gate
    out_norm = out_norm * out_gate

    # Final linear to dim
    out_final = torch.nn.functional.linear(out_norm, to_out_weight)  # shape [B,N,N,dim]

    return out_final
```

TriMul forward (outgoing version) implemented with a custom Triton kernel.

Algorithm:
1. Layer‑norm the input tensor on the last dimension.
2. Two linear projections (left/right) produce tensors of shape [B,N,N,H].
3. Optional mask is applied element‑wise.
4. Gated multiplicative updates: left_gate/right_gate (sigmoid) are computed from
   the normalized input and multiplied into left/right.
5. Core operation:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication per hidden slice (i.e. for each
   hidden dimension h). Implemented as a Triton kernel that loops over the
   reduction dimension k and accumulates results in float32.
6. Layer‑norm over the hidden dimension, gated by out_gate and projected back
   to the original model dimension.
"""

from torch.nn import functional as F

# Triton kernel: batched GEMM for each hidden slice (outgoing TriMul)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                    # batch, seq_len, hidden
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_i = tl.program_id(0)          # i‑block
    pid_j = tl.program_id(1)          # j‑block
    pid_bh = tl.program_id(2)         # combined batch * hidden

    # decode batch and hidden indices
    batch_idx = pid_bh // H
    hidden_idx = pid_bh % H

    # offsets for i and j within the tile
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i < N
    j_mask = j < N

    batch_offset_left  = batch_idx * stride_left_b
    hidden_offset_left = hidden_idx * stride_left_h
    batch_offset_right = batch_idx * stride_right_b
    hidden_offset_right = hidden_idx * stride_right_h
    batch_offset_out   = batch_idx * stride_out_b
    hidden_offset_out  = hidden_idx * stride_out_h

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction axis k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm on the last dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    nomask = config.get("nomask", False)
    if not nomask and mask is not None:
        # ensure correct dtype and broadcast
        mask_f = mask.to(dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul einsum via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # left shape [B,N,N,H]

    # allocate output tensor
    out = torch.empty_like(left)

    # ensure contiguity for pointer arithmetic
    left_c = left.contiguous()
    right_c = right.contiguous()
    out_c = out

    # strides (in element units) - order: batch, i, k, hidden
    s_left = left_c.stride()   # (sb, si, sk, sh)
    s_right = right_c.stride()
    s_out = out_c.stride()

    # block sizes (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # grid: (i tiles, j tiles, batch*hidden)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    # launch Triton kernel
    tri_mul_kernel[grid](
        left_c,
        right_c,
        out_c,
        B,
        N,
        hidden_dim,
        s_left[0],
        s_left[1],
        s_left[2],
        s_left[3],
        s_right[0],
        s_right[1],
        s_right[2],
        s_right[3],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out_c,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out_norm = out_norm * out_gate

    final = F.linear(out_norm, weights["to_out.weight"])

    return final
