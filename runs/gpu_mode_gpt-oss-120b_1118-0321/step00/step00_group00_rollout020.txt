def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    
    # LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    
    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    
    # mask expand
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)
        left = left * mask_exp
        right = right * mask_exp
    else:
        # no mask: treat as all ones
        pass
    
    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()
    
    left = left * left_gate
    right = right * right_gate
    
    # Prepare for batched matmul
    B, N, _, _ = left.shape
    assert left.shape == (B,N,N,hidden_dim)
    # Permute to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    
    # Flatten batch and hidden_dim into single batch dimension
    left_flat = left_perm.reshape(-1, N, N)
    right_flat = right_perm.reshape(-1, N, N)
    
    # Cast to fp16 for compute
    left_flat_fp16 = left_flat.to(torch.float16)
    right_flat_fp16 = right_flat.to(torch.float16)
    out_flat_fp16 = torch.empty_like(left_flat_fp16)

    A, B, C,
    M, N, K,
    stride_ab, stride_am, stride_ak,

    stride_cb, stride_cm, stride_cn,

    pid_m = tl.program_id(0)

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

"""
custom_kernel for TriMul (outgoing) – AlphaFold3 core operation.

Algorithm (forward only):
1. Layer‑norm the input tensor over the last dimension.
2. Project the normalized tensor to a hidden dimension (left/right) via two
   weight‑only linear layers.
3. Optionally mask the projected tensors.
4. Compute per‑position gates (sigmoid of three linear layers) and apply them
   to the projections.
5. For every hidden channel compute a batched matrix multiplication:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   The implementation fuses all (batch * hidden) multiplications into a single
   Triton kernel (batched GEMM).  The kernel works on FP32 data, accumulates in
   FP32 and stores the result in FP32.
6. Apply a second layer‑norm over the hidden dimension, multiply by the output
   gate and project back to the original channel dimension.
The heavy O(N³) contraction (step 5) is completely handled by Triton, providing
significant speed‑up over a pure PyTorch loop/einsum.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the einsum '... i k d, ... j k d -> ... i j d'
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A, B, C,
    M, N, K,                     # matrix dimensions (all = seq_len)
    stride_ab, stride_am, stride_ak,   # A: batch, row(i), col(k)
    stride_bb, stride_bk, stride_bn,   # B_T: batch, row(k), col(j)  (B is read transposed)
    stride_cb, stride_cm, stride_cn,   # C: batch, row(i), col(j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(2)      # batch * hidden index
    pid_m = tl.program_id(0)          # output row block
    pid_n = tl.program_id(1)          # output col block

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # initialise accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # pointers for A (row i, col k)
        a_ptrs = A + pid_batch * stride_ab \
                   + offs_m[:, None] * stride_am \
                   + (k_offset[None, :] * stride_ak)
        # pointers for B_T (row k, col j) --> original B read transposed
        b_ptrs = B + pid_batch * stride_bb \
                   + (k_offset[:, None] * stride_bk) \
                   + offs_n[None, :] * stride_bn

        # load tiles with bounds checking
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & (offs_n[None, :] < N),
                    other=0.0)

        # dot product (FP32 accumulation)
        acc += tl.dot(a, b)

    # store the result
    c_ptrs = C + pid_batch * stride_cb \
                + offs_m[:, None] * stride_cm \
                + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc,
             mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (may be None)
            - weights: dict of weight tensors (see code)
            - config: dict containing "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) First LayerNorm over channel dimension `dim`
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if mask is None or nomask:
        mask_f = torch.ones((x.shape[0], x.shape[1], x.shape[2]), dtype=dtype, device=device)
    else:
        mask_f = mask.to(dtype)
    mask_f = mask_f.unsqueeze(-1)            # shape [B, N, N, 1]
    left = left * mask_f
    right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gate computations (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Batched matrix multiplication using Triton
    #    Compute: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N_seq, _, _ = left.shape
    assert N_seq == input_tensor.shape[1]  # sanity

    # Rearrange to (batch, hidden, N, N) and flatten batch*hidden
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    total_batches = B * hidden_dim
    left_flat = left_perm.view(total_batches, N_seq, N_seq)
    right_flat = right_perm.view(total_batches, N_seq, N_seq)

    # Output tensor
    out_flat = torch.empty_like(left_flat)

    # Strides for A (left)
    stride_ab, stride_am, stride_ak = left_flat.stride()   # (batch, row, col)
    # Strides for B (right) – we will read it transposed
    stride_bb, stride_br, stride_bc = right_flat.stride()
    stride_bk = stride_bc          # stride for the K dimension after transpose (original col)
    stride_bn = stride_br          # stride for the N dimension after transpose (original row)

    # Strides for C (output) – same layout as A
    stride_cb, stride_cm, stride_cn = out_flat.stride()

    # Block sizes – chosen to fit in shared memory and give good occupancy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N_seq, BLOCK_M),   # rows
        triton.cdiv(N_seq, BLOCK_N),   # cols
        total_batches                  # batch * hidden
    )

    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N_seq, N_seq, N_seq,
        stride_ab, stride_am, stride_ak,
        stride_bb, stride_bk, stride_bn,
        stride_cb, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # ------------------------------------------------------------------
    # 6) Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N_seq, N_seq).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over hidden dimension, apply out_gate and final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
