def tr_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_rowk, stride_b_colj,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Compute masks for out-of-bounds rows/cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Compute base pointers for this batch
    a_batch_offset = batch * stride_a_batch
    b_batch_offset = batch * stride_b_batch
    c_batch_offset = batch * stride_c_batch

    # Pointers to start of tile
    a_ptrs = a_ptr + a_batch_offset + offs_m[:, None] * stride_a_row + offs_k[None, :] * stride_a_col
    b_ptrs = b_ptr + b_batch_offset + offs_k[:, None] * stride_b_rowk + offs_n[None, :] * stride_b_colj

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Iterate over K
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        # Load a and b
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)
        # Compute dot product
        acc += tl.dot(a, b)
        # Increment pointers to next K block
        a_ptrs += BLOCK_K * stride_a_col
        b_ptrs += BLOCK_K * stride_b_rowk

    # Write back results
    c_ptrs = c_ptr + c_batch_offset + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    # Apply mask for out-of-bounds c
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
custom_kernel: fast “outgoing” TriMul implementation.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor (B, N, N, D) using the provided
   weight/bias (no gradient needed).
2. Linear projections to a hidden dimension H (left_proj, right_proj) and
   three gate projections (left_gate, right_gate, out_gate).  Apply an
   optional mask after the linear projections and then multiply by the
   gates (sigmoid activation).
3. The core heavy operation  out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is a batched matrix‑multiply where each hidden channel acts as an
   independent batch element:
        C = A @ Bᵀ   with A,B ∈ (BH, N, N)   (BH = B·H)
   This is executed by a custom Triton kernel that multiplies tiles of
   size (BLOCK_M × BLOCK_K) and (BLOCK_K × BLOCK_N) and accumulates in
   FP32.  The kernel works for arbitrary N, handling edge tiles with
   proper masking.
4. Reshape the result back to (B, N, N, H) and apply a second LayerNorm
   (to_out_norm), the out‑gate, and a final linear projection back to
   the original hidden dimension D.
5. Return the final tensor (B, N, N, D) on the same device.

Only the cubic‑complexity part (step 3) is performed in Triton; the rest
uses standard PyTorch ops for simplicity and numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tr_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_rowk, stride_b_colj,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matmul C = A @ Bᵀ  with A,B shape (batch, M, K) and (batch, N, K)."""
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds rows / columns
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k_iter in range(num_k_blocks):
        k_offset = k_iter * BLOCK_K
        k_idx = k_offset + offs_k                     # absolute k indices
        mask_k = k_idx < K

        # Load A tile (M × K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_row + k_idx[None, :] * stride_a_col
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load Bᵀ tile (K × N) – note the swapped strides:
        #   Bᵀ[k, j] corresponds to original B[j, k]
        b_ptrs = b_base + k_idx[:, None] * stride_b_rowk + offs_n[None, :] * stride_b_colj
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # Write back the tile
    c_ptrs = c_base + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a Triton kernel
    for the heavy cubic‑time matrix multiplication.
    """
    # unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------- LayerNorm on the input -------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)                       # [B, N, N, D]

    # ------- Linear projections -------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)                 # [B, N, N, H]
    right = F.linear(x, right_proj_weight)               # [B, N, N, H]

    # ------- Optional mask -------
    if mask is not None:
        mask = mask.to(dtype).unsqueeze(-1)               # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------- Gates -------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------- Prepare tensors for Triton kernel -------
    # Move hidden dim to the batch dimension: (B, H, N, N) -> (BH, N, N)
    left_bt = left.permute(0, 3, 1, 2).contiguous()
    right_bt = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left_bt.shape
    batch_total = B * H
    left_flat = left_bt.view(batch_total, N, N)
    right_flat = right_bt.view(batch_total, N, N)

    # Allocate output
    out_flat = torch.empty_like(left_flat)

    # Strides (int64)
    stride_a_batch, stride_a_row, stride_a_col = left_flat.stride()
    stride_b_batch, stride_b_row, stride_b_col = right_flat.stride()
    stride_c_batch, stride_c_row, stride_c_col = out_flat.stride()

    # For Bᵀ we need row‑stride = original column stride, col‑stride = original row stride
    stride_b_rowk = stride_b_col   # stride over k (original column)
    stride_b_colj = stride_b_row   # stride over j (original row)

    # ------- Triton launch -------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32   # fairly small to keep shared memory low

    grid = (
        batch_total,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tr_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_a_batch, stride_a_row, stride_a_col,
        stride_b_batch, stride_b_rowk, stride_b_colj,
        stride_c_batch, stride_c_row, stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------- Second LayerNorm, out‑gate, and final projection -------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)                                 # [B, N, N, H]

    out = out * out_gate                                              # gating

    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)                               # [B, N, N, D]

    return out
```
