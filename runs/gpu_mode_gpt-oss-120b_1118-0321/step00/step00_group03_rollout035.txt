    right_proj_weight = weights["right_proj.weight"]

        pass
    mask = mask.unsqueeze(-1)  # shape [B,N,N,1]

    right = right * mask

    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))

"""
TriMul (outgoing) forward pass with a custom Triton kernel.

The core O(N³) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched matrix multiplication:
    L_d = left[..., :, :, d]   (shape B×N×N)
    R_d = right[..., :, :, d]  (shape B×N×N)
    out[..., :, :, d] = L_d @ R_dᵀ
A Triton kernel performs the batched GEMM for all hidden dimensions
simultaneously (batch dimension = B·hidden_dim).  All other
operations (LayerNorm, linear projections, sigmoid gates) are done with
standard PyTorch for simplicity and numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # *float32 pointers
    M, N, K,                           # matrix sizes (int)
    stride_a_batch, stride_a_h, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_h, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_h, stride_c_m, stride_c_n,
    hidden_dim,                        # total hidden channels (int)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute for each (batch, hidden) slice:
        C = A @ Bᵀ   where
        A : (M, K)   B : (N, K)  (B appears in the stride of the third axis)
    The hidden dimension is folded into the program ID.
    """
    pid = tl.program_id(0)          # combined batch*hidden index
    pid_m = tl.program_id(1)        # tile row index
    pid_n = tl.program_id(2)        # tile col index

    batch_idx = pid // hidden_dim
    hidden_idx = pid % hidden_dim

    # Tile start coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Pointers with the correct stride offsets
        a_ptrs = (
            a_ptr
            + batch_idx * stride_a_batch
            + hidden_idx * stride_a_h
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        b_ptrs = (
            b_ptr
            + batch_idx * stride_b_batch
            + hidden_idx * stride_b_h
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n
        )

        # Masks for out‑of‑bounds loads
        a_mask = (offs_m < M)[:, None] & (offs_k < K)[None, :]
        b_mask = (offs_k < K)[:, None] & (offs_n < N)[None, :]

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # (BLOCK_K, BLOCK_N)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # Store the result
    c_ptrs = (
        c_ptr
        + batch_idx * stride_c_batch
        + hidden_idx * stride_c_h
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    c_mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple):
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of weight tensors
            - config: dict containing at least "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (dtype float32)
    """
    # unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    torch_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # -----------------------------------------------------------------
    # 1. LayerNorm on the input
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2. Linear projections
    # -----------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcasted over hidden dim)
    # -----------------------------------------------------------------
    if mask is None:
        mask = torch.ones(x.shape[:3], dtype=torch_dtype, device=device)
    mask = mask.to(torch_dtype).unsqueeze(-1)  # [B, N, N, 1]
    left = left * mask
    right = right * mask

    # -----------------------------------------------------------------
    # 4. Gating
    # -----------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate               # [B, N, N, hidden_dim]
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction via Triton batched GEMM
    #    Compute out_einsum_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # -----------------------------------------------------------------
    # Rearrange to [B, hidden, N, N] and transpose right for Bᵀ
    left_perm = left.permute(0, 3, 1, 2)          # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2)        # [B, H, N, N]
    right_t = right_perm.transpose(-2, -1)       # [B, H, N, N] with swapped axes

    # Output buffer for the matmul (same layout as left_perm)
    out_mat = torch.empty_like(left_perm)

    # Strides in element units
    a_strides = left_perm.stride()   # (batch, hidden, i, k)
    b_strides = right_t.stride()    # (batch, hidden, k, j)
    c_strides = out_mat.stride()    # (batch, hidden, i, j)

    B, H, N, _ = left_perm.shape
    M = N
    K = N

    # Tiling parameters (tuned for Hopper/AMPERE, fits in ~64KB shared memory)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (B * H, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    _batched_matmul_kernel[grid](
        left_perm,
        right_t,
        out_mat,
        M,
        N,
        K,
        a_strides[0],
        a_strides[1],
        a_strides[2],
        a_strides[3],
        b_strides[0],
        b_strides[1],
        b_strides[2],
        b_strides[3],
        c_strides[0],
        c_strides[1],
        c_strides[2],
        c_strides[3],
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Back‑to‑original layout: [B, N, N, hidden_dim]
    out_einsum = out_mat.permute(0, 2, 3, 1)

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, final gating and linear projection
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(
        out_einsum,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out_norm = out_norm * out_gate

    to_out_weight = weights["to_out.weight"]
    output = F.linear(out_norm, to_out_weight)   # [B, N, N, dim]

    return output
