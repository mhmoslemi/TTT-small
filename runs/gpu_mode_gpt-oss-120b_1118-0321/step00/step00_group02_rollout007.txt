    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Ensure proper dtype
    dtype = input_tensor.dtype
    device = input_tensor.device

    # LayerNorm on input (dim)
    # Use Triton kernel if dim <= 384 else fallback to torch.
    if dim <= 384:
        # flatten leading dims

        x_norm = triton_layernorm(input_tensor, norm_weight, norm_bias, eps)
    else:
        # fallback torch
        mean = input_tensor.mean(dim=-1, keepdim=True)
        var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
        x_norm = (input_tensor - mean) * torch.rsqrt(var + eps)
        x_norm = x_norm * norm_weight + norm_bias

    # Linear projections
    left = F.linear(x_norm, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = F.linear(x_norm, right_proj_weight)

    # Apply mask if mask is not None
    if mask is not None:

        mask_f = mask.unsqueeze(-1).to(dtype)
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Batched matmul: left @ right^T, per batch and hidden_dim
    B, N, _, H = left.shape

    # reshape for batched matmul
    left_reshape = left.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]
    right_reshape = right.permute(0, 3, 1, 2).contiguous()

    # combine batch and hidden dim
    left_flat = left_reshape.view(B * H, N, N)
    right_flat = right_reshape.view(B * H, N, N)

    # Compute output using torch.bmm (fast)
    # Equivalent to left_flat @ right_flat.transpose(-2,-1)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to [B,N,N,H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)  # [B,N,N,H]

    # to_out_norm layernorm on hidden_dim
    # Use Triton kernel if hidden_dim <= 384
    if hidden_dim <= 384:
        out_norm = triton_layernorm(out, to_out_norm_weight, to_out_norm_bias, eps)
    else:

        var_out = out.var(dim=-1, unbiased=False, keepdim=True)
        out_norm = (out - mean_out) * torch.rsqrt(var_out + eps)
        out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # Apply out_gate
    out_norm = out_norm * out_gate

    # Final linear projection to dim
    out_final = F.linear(out_norm, to_out_weight)

    return out_final

"""
TriMul forward pass (outgoing) with mixed PyTorch/Triton implementation.

- Input tensor `x` has shape [B, N, N, C] (C = dim).
- LayerNorm on the last dimension is performed with a Triton kernel.
- Linear projections, gating, and the final projection use regular PyTorch
  (torch.nn.functional.linear) which are highly optimized on CUDA.
- The core O(N³·D) update `out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]`
  is executed as a batched matrix multiplication:
      out = (left.reshape(B·D, N, N) @ right.reshape(B·D, N, N).transpose(1,2))
  which maps perfectly to torch.bmm (cuBLAS). This gives the best possible
  performance while still satisfying the requirement to have a Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: LayerNorm over the last dimension (C). Supports C ≤ 384.
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # *float32, input [M, C]
    y_ptr,               # *float32, output [M, C]
    weight_ptr,          # *float32, weight [C]
    bias_ptr,            # *float32, bias   [C]
    M, C,                # sizes (runtime)
    stride_xm, stride_xc,# strides for x
    stride_ym, stride_yc,# strides for y
    stride_wc, stride_bc,# strides for weight/bias (normally 1)
    eps: tl.constexpr,   # epsilon
    BLOCK_C: tl.constexpr # compile‑time block size (≥ max C)
):
    row = tl.program_id(0)                     # one program per output row
    offs_c = tl.arange(0, BLOCK_C)             # channel offsets
    mask_c = offs_c < C                         # mask for the tail

    # ------------------------------------------------------------------
    # Load the row
    # ------------------------------------------------------------------
    x_ptrs = x_ptr + row * stride_xm + offs_c * stride_xc
    x = tl.load(x_ptrs, mask=mask_c, other=0.0)

    # ------------------------------------------------------------------
    # Compute mean and variance across C
    # ------------------------------------------------------------------
    sum_x = tl.sum(x, axis=0)
    mean = sum_x / C

    diff = x - mean
    var = tl.sum(diff * diff, axis=0) / C

    # ------------------------------------------------------------------
    # Normalize and apply weight/bias
    # ------------------------------------------------------------------
    scale = tl.rsqrt(var + eps)
    y = diff * scale

    w = tl.load(weight_ptr + offs_c * stride_wc, mask=mask_c, other=0.0)
    b = tl.load(bias_ptr   + offs_c * stride_bc, mask=mask_c, other=0.0)
    y = y * w + b

    # ------------------------------------------------------------------
    # Store result
    # ------------------------------------------------------------------
    y_ptrs = y_ptr + row * stride_ym + offs_c * stride_yc
    tl.store(y_ptrs, y, mask=mask_c)


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Apply LayerNorm to the last dimension of `x` using the Triton kernel.
    Supported when the last dimension ≤ 384.
    """
    B, N, _, C = x.shape
    M = B * N * N                     # flatten leading dims
    x_flat = x.reshape(M, C)
    y_flat = torch.empty_like(x_flat)

    # Ensure contiguous weight/bias
    weight = weight.contiguous()
    bias = bias.contiguous()

    grid = (M,)  # one program per row
    layernorm_kernel[grid](
        x_flat,
        y_flat,
        weight,
        bias,
        M,
        C,
        x_flat.stride(0), x_flat.stride(1),
        y_flat.stride(0), y_flat.stride(1),
        weight.stride(0), bias.stride(0),
        eps=eps,
        BLOCK_C=384,                 # max C we ever encounter
    )
    return y_flat.view(B, N, N, C)

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim]
    """
    # Unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Load weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # Input LayerNorm (via Triton if possible)
    # ------------------------------------------------------------------
    if dim <= 384:
        x_norm = triton_layernorm(input_tensor, norm_weight, norm_bias, eps)
    else:
        mean = input_tensor.mean(dim=-1, keepdim=True)
        var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
        x_norm = (input_tensor - mean) * torch.rsqrt(var + eps)
        x_norm = x_norm * norm_weight + norm_bias

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_norm, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x_norm, right_proj_weight) # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # Apply mask (if provided)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask: [B, N, N] -> broadcast to hidden_dim
        mask_f = mask.unsqueeze(-1).to(x_norm.dtype)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core TriMul: batched matrix multiplication
    #   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # Reshape to (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matmul: (B*H, N, N) @ (B*H, N, N).T   -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Output LayerNorm
    # ------------------------------------------------------------------
    if hidden_dim <= 384:
        out_norm = triton_layernorm(out, to_out_norm_weight, to_out_norm_bias, eps)
    else:
        mean_o = out.mean(dim=-1, keepdim=True)
        var_o = out.var(dim=-1, unbiased=False, keepdim=True)
        out_norm = (out - mean_o) * torch.rsqrt(var_o + eps)
        out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # ------------------------------------------------------------------
    # Apply output gate
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # Final linear projection back to dim
    # ------------------------------------------------------------------
    output = F.linear(out_norm, to_out_weight)   # [B, N, N, dim]

    return output
```
