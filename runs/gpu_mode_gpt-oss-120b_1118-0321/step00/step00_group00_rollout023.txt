def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # Get weight/tensors:
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    
    eps = 1e-5
    # LayerNorm using torch.nn.functional
    # normalized_shape = dim
    # This supports weight and bias.
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps
    )
    # compute left projection
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape (B,N,N,H)
    right = torch.nn.functional.linear(x, right_proj_weight)
    # mask expansion
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)  # shape (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp
    # gates
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()
    # apply gates
    left = left * left_gate
    right = right * right_gate
    # free x, left_gate, right_gate to release memory
    del left_gate, right_gate, x
    torch.cuda.empty_cache()
    # Cast to float16 for kernel (to reduce memory and increase speed)
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    # output placeholder in float32
    out = torch.empty_like(left_fp16, dtype=torch.float32)  # shape (B,N,N,H)
    # Define the kernel
    # grid dims: (B * num_i_tiles, num_j_tiles)
    # We'll need to compute strides.
    B, N, _, H = left_fp16.shape
    # Compute strides (in elements)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_fp16.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right_fp16.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()
    # Determine block sizes:
    BLOCK_I = 16
    BLOCK_J = 16
    BLOCK_K = 16
    BLOCK_H = H  # full hidden dim

    # Compute grid
    num_i_tiles = (N + BLOCK_I - 1) // BLOCK_I
    num_j_tiles = (N + BLOCK_J - 1) // BLOCK_J
    grid = (B * num_i_tiles, num_j_tiles)
    # Launch kernel
    triton_tri_mul_kernel[grid](
        left_fp16, right_fp16, out,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H
    )
    # now out is float32 (computed from half inputs)
    # Apply to_out_norm: LayerNorm over last dimension hidden_dim
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps
    )
    # multiply with out_gate (original computed as float32)
    out_norm = out_norm * out_gate
    # final projection to dim
    output = torch.nn.functional.linear(out_norm, to_out_weight)
    return output

@triton.jit
def triton_tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                          B, N, H,
                          stride_l_b, stride_l_i, stride_l_k, stride_l_h,
                          stride_r_b, stride_r_i, stride_r_k, stride_r_h,
                          stride_o_b, stride_o_i, stride_o_j, stride_o_h,
                          BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr):
    # compute program IDs
    pid0 = tl.program_id(0)  # combined b and i_tile
    pid1 = tl.program_id(1)  # j_tile

    # Compute num_i_tiles
    num_i_tiles = tl.cdiv(N, BLOCK_I)
    # Determine batch index and i tile
    b = pid0 // num_i_tiles
    i_tile = pid0 % num_i_tiles
    j_tile = pid1

    # Compute offsets
    i_start = i_tile * BLOCK_I
    j_start = j_tile * BLOCK_J

    # Offsets within block
    i_offsets = i_start + tl.arange(0, BLOCK_I)
    j_offsets = j_start + tl.arange(0, BLOCK_J)
    # Create masks for out of bounds
    i_mask = i_offsets < N
    j_mask = j_offsets < N

    # Hidden dim offsets (full)
    d_offsets = tl.arange(0, BLOCK_H)

    # Broadcast for 2D thread grid
    # i_idx shape (BLOCK_I,1)
    # j_idx shape (1,BLOCK_J)
    i_idx = i_offsets[:, None]  # shape (BLOCK_I,1)
    j_idx = j_offsets[None, :]  # shape (1,BLOCK_J)

    # Initialize accumulator with zeros (float32)
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # Loop over k dimension in steps of BLOCK_K
    for k_start in range(0, N, BLOCK_K):
        k_offsets = k_start + tl.arange(0, BLOCK_K)
        k_mask = k_offsets < N

        # Compute left pointer offsets: shape (BLOCK_I, BLOCK_K, BLOCK_H)
        # left_ptr shape: left_ptr + b*stride_l_b + i_idx*stride_l_i + k_offsets[None,:,None]*stride_l_k + d_offsets[None,None,:]*stride_l_h
        left_ptrs = left_ptr + b * stride_l_b + i_idx * stride_l_i + k_offsets[None, :, None] * stride_l_k + d_offsets[None, None, :] * stride_l_h
        # Load left (float16) but accumulate as float32
        left = tl.load(left_ptrs, mask = (i_mask[:, None, None] & k_mask[None, :, None]), other=0.0).to(tl.float32)

        # Compute right pointer offsets: shape (BLOCK_J, BLOCK_K, BLOCK_H)
        right_ptrs = right_ptr + b * stride_r_b + j_idx * stride_r_i + k_offsets[None, :, None] * stride_r_k + d_offsets[None, None, :] * stride_r_h
        right = tl.load(right_ptrs, mask = (j_mask[None, :, None] & k_mask[None, :, None]), other=0.0).to(tl.float32)

        # Now left shape (BLOCK_I, BLOCK_K, BLOCK_H), right shape (BLOCK_J, BLOCK_K, BLOCK_H)
        # Expand dimensions for broadcasting:
        # left_expand shape (BLOCK_I, 1, BLOCK_K, BLOCK_H)
        # right_expand shape (1, BLOCK_J, BLOCK_K, BLOCK_H)
        left_expand = left[:, None, :, :]      # (BLOCK_I,1,BLOCK_K,BLOCK_H)
        right_expand = right[None, :, :, :]    # (1,BLOCK_J,BLOCK_K,BLOCK_H)

        # elementwise multiply and sum over k (axis=2)
        # product shape (BLOCK_I, BLOCK_J, BLOCK_K, BLOCK_H)
        prod = left_expand * right_expand
        acc += tl.sum(prod, axis=2)  # sum over k ending shape (BLOCK_I, BLOCK_J, BLOCK_H)

    # After loop over k, store acc to out
    # Compute output pointer offsets: shape (BLOCK_I, BLOCK_J, BLOCK_H)
    out_ptrs = out_ptr + b * stride_o_b + i_idx * stride_o_i + j_idx * stride_o_j + d_offsets[None, None, :] * stride_o_h
    # Use mask for i,j bounds
    mask_store = i_mask[:, None, None] & j_mask[None, :, None]
    tl.store(out_ptrs, acc, mask=mask_store)

"""
TriMul (outgoing) forward implementation.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Linear projections → left, right (hidden_dim) and three gating
   linear layers (sigmoid).  Apply mask and gates.
3. Main O(N³·C) reduction:
   out[b,i,j,c] = Σₖ left[b,i,k,c] * right[b,j,k,c]
   Implemented as a custom Triton kernel that processes a tile of (i,j)
   pairs per program, accumulates over the k‑dimension and stores the
   result (float32).  Left/right are cast to float16 to reduce memory
   bandwidth.
4. Layer‑norm on the hidden dimension, apply out‑gate and final linear
   projection back to the original feature dimension.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------------- #
# Triton kernel for the triple‑product reduction
# ----------------------------------------------------------------------------- #
@triton.jit
def _tri_mul_kernel(
    left_ptr,          # const float16 *
    right_ptr,         # const float16 *
    out_ptr,           # float32 *
    B, N, H,           # int32 scalars
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid0 = tl.program_id(0)          # combined batch and i‑tile
    pid1 = tl.program_id(1)          # j‑tile

    # Number of tiles covering the i dimension
    num_i_tiles = tl.cdiv(N, BLOCK_I)

    # Decompose pid0 into batch index and i‑tile index
    b = pid0 // num_i_tiles
    i_tile = pid0 % num_i_tiles
    j_tile = pid1

    # Starting coordinates of the tile
    i_start = i_tile * BLOCK_I
    j_start = j_tile * BLOCK_J

    # Offsets inside the tile
    i_offset = i_start + tl.arange(0, BLOCK_I)
    j_offset = j_start + tl.arange(0, BLOCK_J)
    d_offset = tl.arange(0, BLOCK_H)               # hidden dimension

    # Masks for out‑of‑bounds rows / columns
    i_mask = i_offset < N
    j_mask = j_offset < N

    # Shape (BLOCK_I, 1) and (1, BLOCK_J) for broadcasting
    i_idx = i_offset[:, None]
    j_idx = j_offset[None, :]

    # Accumulator (float32) for the tile
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # --------------------------------------------------------------------- #
    # Loop over the reduction dimension k in blocks of size BLOCK_K
    # --------------------------------------------------------------------- #
    for k_start in range(0, N, BLOCK_K):
        k_offset = k_start + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # -----------------------------------------------------------------
        # Load left[b, i, k, d]   → shape (BLOCK_I, BLOCK_K, BLOCK_H)
        # -----------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + b * stride_l_b
            + i_idx * stride_l_i
            + k_offset[None, :, None] * stride_l_k
            + d_offset[None, None, :] * stride_l_h
        )
        left = tl.load(left_ptrs,
                       mask=(i_mask[:, None, None] & k_mask[None, :, None]),
                       other=0.0).to(tl.float32)   # cast to fp32 for accumulation

        # -----------------------------------------------------------------
        # Load right[b, j, k, d]  → shape (BLOCK_J, BLOCK_K, BLOCK_H)
        # -----------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + b * stride_r_b
            + j_idx * stride_r_i
            + k_offset[None, :, None] * stride_r_k
            + d_offset[None, None, :] * stride_r_h
        )
        right = tl.load(right_ptrs,
                        mask=(j_mask[None, :, None] & k_mask[None, :, None]),
                        other=0.0).to(tl.float32)

        # -----------------------------------------------------------------
        # Broadcast to (BLOCK_I, BLOCK_J, BLOCK_K, BLOCK_H) and sum over k
        # -----------------------------------------------------------------
        # left  : (I, 1, K, H)
        # right : (1, J, K, H)
        prod = left[:, None, :, :] * right[None, :, :, :]   # (I,J,K,H)
        acc += tl.sum(prod, axis=2)                        # sum over K → (I,J,H)

    # --------------------------------------------------------------------- #
    # Write out the accumulated results
    # --------------------------------------------------------------------- #
    out_ptrs = (
        out_ptr
        + b * stride_o_b
        + i_idx * stride_o_i
        + j_idx * stride_o_j
        + d_offset[None, None, :] * stride_o_h
    )
    store_mask = i_mask[:, None, None] & j_mask[None, :, None]
    tl.store(out_ptrs, acc, mask=store_mask)


# ----------------------------------------------------------------------------- #
# Public entry point used by the test harness
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32 or bool)
            - weights: dict with keys
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight", "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config: dict with entries "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------------- #
    # Unpack inputs
    # --------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------------- #
    # Layer‑norm on the input (dim)
    # --------------------------------------------------------------------- #
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # --------------------------------------------------------------------- #
    # Linear projections (no bias)
    # --------------------------------------------------------------------- #
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------------- #
    # Apply mask if present (mask broadcasted over the hidden dim)
    # --------------------------------------------------------------------- #
    if mask is not None:
        mask_f = mask.unsqueeze(-1).to(left.dtype)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------------- #
    # Gating (sigmoid on linear transforms of the *normalized* input)
    # --------------------------------------------------------------------- #
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply gates in‑place and release temporary tensors ASAP
    left = left * left_gate
    right = right * right_gate
    del left_gate, right_gate, x  # free memory

    # Convert to half‑precision for the heavy reduction
    left_hp = left.to(torch.float16).contiguous()
    right_hp = right.to(torch.float16).contiguous()
    del left, right  # free full‑precision buffers

    B, N, _, _ = left_hp.shape  # (B, N, N, hidden_dim)

    # --------------------------------------------------------------------- #
    # Prepare output buffer for the reduction
    # --------------------------------------------------------------------- #
    out_hp = torch.empty((B, N, N, hidden_dim), device=device, dtype=torch.float32)

    # --------------------------------------------------------------------- #
    # Strides (in elements) for pointer arithmetic in the kernel
    # --------------------------------------------------------------------- #
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_hp.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right_hp.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out_hp.stride()

    # --------------------------------------------------------------------- #
    # Triton launch configuration
    # --------------------------------------------------------------------- #
    BLOCK_I = 16
    BLOCK_J = 16
    BLOCK_K = 16
    BLOCK_H = hidden_dim  # compile‑time constant

    num_i_tiles = (N + BLOCK_I - 1) // BLOCK_I
    num_j_tiles = (N + BLOCK_J - 1) // BLOCK_J

    grid = (B * num_i_tiles, num_j_tiles)

    # Launch the kernel
    _tri_mul_kernel[grid](
        left_hp,
        right_hp,
        out_hp,
        B,
        N,
        hidden_dim,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_i,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # --------------------------------------------------------------------- #
    # Post‑processing: hidden‑dim layer‑norm, out‑gate, final linear
    # --------------------------------------------------------------------- #
    out_norm = torch.nn.functional.layer_norm(
        out_hp,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # Apply out‑gate (same hidden dimension)
    out_norm = out_norm * out_gate

    # Final projection back to the original feature dimension
    output = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return output
