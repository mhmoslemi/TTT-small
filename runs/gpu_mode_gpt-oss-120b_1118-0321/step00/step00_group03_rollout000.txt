    dim = config['dim']
    hidden_dim = config['hidden_dim']
    # Access weights
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    left_proj_weight = weights['left_proj.weight']
    right_proj_weight = weights['right_proj.weight']
    left_gate_weight = weights['left_gate.weight']
    right_gate_weight = weights['right_gate.weight']
    out_gate_weight = weights['out_gate.weight']
    to_out_norm_weight = weights['to_out_norm.weight']
    to_out_norm_bias = weights['to_out_norm.bias']
    to_out_weight = weights['to_out.weight']

    bs, N, _, _ = input_tensor.shape # dim stride may be present

    # LayerNorm:
    # x_norm = (x - mean) / sqrt(var + eps) * norm_weight + norm_bias
    # Compute along last dim (dim)
    eps = 1e-5
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = ((input_tensor - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (input_tensor - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias

    # left_proj = linear x_norm @ left_proj_weight.T (no bias)
    # Use torch.nn.functional.linear
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape [B,N,N,hidden]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    right = right * right_gate

@triton.jit
def tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                   batch_size, seq_len, hidden_dim,
                   stride_left_b, stride_left_i, stride_left_k, stride_left_h,
                   stride_right_b, stride_right_i, stride_right_k, stride_right_h,
                   stride_out_b, stride_out_i, stride_out_j, stride_out_h,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # Determine indices
    pid_m = tl.program_id(0) # tile i
    pid_n = tl.program_id(1) # tile j
    pid_z = tl.program_id(2) # combined batch*hidden

    # Compute batch and hidden indices
    b = pid_z // hidden_dim
    h = pid_z % hidden_dim

    # Starting offsets
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # Define range for rows and cols
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # Masks for bounds
    mask_i = offs_i < seq_len
    mask_j = offs_j < seq_len

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k_offset in range(0, tl.cdiv(seq_len, BLOCK_K)):
        k_start = k_offset * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < seq_len

        # Load left tile: shape (BLOCK_M, BLOCK_K)
        # base pointer = left_ptr + b*stride_left_b + h*stride_left_h
        left_base = left_ptr + b * stride_left_b + h * stride_left_h
        left_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        # load with mask
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right tile: shape (BLOCK_N, BLOCK_K)
        right_base = right_ptr + b * stride_right_b + h * stride_right_h
        # right indexing: j dimension is first, k second
        right_ptrs = right_base + offs_j[:, None] * stride_right_i + offs_k[None, :] * stride_right_k
        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        # accumulate dot
        acc += tl.dot(left, right)  # left (M,K), right (N,K) -> (M,N)

    # Write out result
    out_base = out_ptr + b * stride_out_b + h * stride_out_h
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch_size, seq_len, hidden_dim,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # pid for i tile, j tile, and combined batch*hidden
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)

    # decode batch and hidden dimension from pid_z
    b = pid_z // hidden_dim
    h = pid_z % hidden_dim

    # starting indices
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # compute the actual indices for this tile
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < seq_len
    mask_j = offs_j < seq_len

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # base pointers that include batch and hidden offset
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # loop over the reduction dimension k
    num_k_tiles = tl.cdiv(seq_len, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < seq_len

        # Load left[i,k] tile
        left_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right[j,k] tile
        right_ptrs = right_base + offs_j[:, None] * stride_right_i + offs_k[None, :] * stride_right_k
        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        # accumulate dot product over k
        acc += tl.dot(left, right)   # (M,K) x (N,K) -> (M,N)

    # Write the result
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Triton-accelerated implementation of the outgoing TriMul operator.
    Performs LayerNorm, linear projections, gating, a fused
    pairwise reduction over the sequence dimension (einsum), a second
    LayerNorm, gating, and final output projection.
    """
    import torch.nn.functional as F
    # unpack inputs
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------- LayerNorm ----------
    # norm over last dimension
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ---------- Linear projections ----------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = F.linear(x, right_proj_weight)

    # ---------- Mask ----------
    if mask is not None:
        mask = mask.to(left.dtype)
        mask = mask.unsqueeze(-1)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ---------- Gating ----------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------- Fusion core: einsum over k ----------
    # Ensure contiguous layout for efficient Triton access
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, _ = left.shape

    # output tensor for the reduced result (float32)
    out = torch.empty((B, N, N, hidden_dim), device=left.device, dtype=torch.float32)

    # Strides (in elements)
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    # Tile sizes (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid_z = B * hidden_dim

    tri_mul_kernel[(grid_m, grid_n, grid_z)](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---------- Second LayerNorm ----------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                            bias=to_out_norm_bias, eps=eps)

    # ---------- Output gating ----------
    out_norm = out_norm * out_gate

    # ---------- Final linear ----------
    to_out_weight = weights["to_out.weight"]
    out_final = F.linear(out_norm, to_out_weight)  # shape [B,N,N,dim]

    return out_final

    Triton‑accelerated outgoing TriMul implementation.
    Performs:
      * LayerNorm over the last dim (`dim`)
      * Linear projections (`left_proj`, `right_proj`)
      * Optional masking
      * Sigmoid gating (`left_gate`, `right_gate`, `out_gate`)
      * Core reduction  out = einsum('... i k d, ... j k d -> ... i j d')
      * LayerNorm over hidden dim
      * Output gating and final linear projection
    Returns a tensor of shape [B, N, N, dim] (float32).
    """
    import torch.nn.functional as F

    # Unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -------- LayerNorm ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,),
                     weight=norm_weight, bias=norm_bias, eps=eps)

    # -------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # -------- Mask ----------
    if mask is not None:
        mask = mask.to(left.dtype).unsqueeze(-1)            # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -------- Gating ----------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -------- Core einsum (Triton) ----------
    left = left.contiguous()
    right = right.contiguous()
    B, N, _, _ = left.shape

    # Output buffer (float32)
    out = torch.empty((B, N, N, hidden_dim),
                      device=left.device, dtype=torch.float32)

    # Strides in element units
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    # Tile sizes – tuned for typical N up to 1024

    BLOCK_N = 64
    BLOCK_K = 64

    grid_n = (N + BLOCK_N - 1) // BLOCK_N

    out_norm = out_norm * out_gate
