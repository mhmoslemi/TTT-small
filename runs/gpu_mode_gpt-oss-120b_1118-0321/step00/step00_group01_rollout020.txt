def custom_kernel(data):
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # cast mask
    if mask is None:
        mask = torch.ones(B, N, N, dtype=input_tensor.dtype, device=input_tensor.device)
    mask = mask.to(input_tensor.dtype).unsqueeze(-1)
    # Norm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    )
    # Projections
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"]) # weight shape (hidden_dim, dim)
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # Apply mask
    left_proj = left_proj * mask
    right_proj = right_proj * mask
    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()
    # Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate
    # Convert to half
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)
    # Compute out via Triton kernel
    out = tri_mul_kernel(left_h, right_h, B, N, hidden_dim)
    # out is float32
    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    # Multiply by out_gate
    out = out * out_gate
    # Final linear
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)
    return out

import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_k, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    # compute indices
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    total_h = H
    total_b = B

    # Compute b, h, i_tile, j_tile
    b = pid // (total_h * num_blocks_m * num_blocks_n)
    tmp = pid % (total_h * num_blocks_m * num_blocks_n)
    h = tmp // (num_blocks_m * num_blocks_n)
    tmp2 = tmp % (num_blocks_m * num_blocks_n)
    i_tile = tmp2 // num_blocks_n
    j_tile = tmp2 % num_blocks_n

    # base offsets
    i_offset = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)

    # Check for out-of-bound
    i_valid = i_offset < N
    j_valid = j_offset < N

    # Pointers start for left and right
    # Left: left[b, i, k, h] with k reduction dimension
    # Right: right[b, j, k, h] with same k dimension.

    # Compute pointer base for batch and hidden
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        k_valid = k_offset < N

        b_ = tl.load(right_ptrs, mask=j_valid[None, :] & k_valid[:, None], other=0.0)

        # dot product accumulation
        acc += tl.dot(a, b_)
    
    # Store result

"""
TriMul forward (outgoing) implementation.

The core of the operator is the tensor contraction
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
where left/right are the projected and gated pairwise representations.
We fuse this O(N³·D) operation with a Triton kernel that performs a
batched matrix multiplication for each hidden channel `d`.  The kernel
computes tiles of size (BLOCK_M, BLOCK_N) while iterating over the
reduction dimension `k` in blocks of size BLOCK_K, accumulating in
float32 (even though the inputs are stored as float16) for numeric
stability.  The surrounding linear layers, LayerNorms and gating are
handled in PyTorch for simplicity.

The kernel launch grid is (num_i_tiles, num_j_tiles, B*hidden_dim) so
each program processes one (i‑tile, j‑tile) for a specific batch and
hidden channel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,      # *float16, shape [B, N, N, H]
    right_ptr,     # *float16, shape [B, N, N, H]
    out_ptr,       # *float32, shape [B, N, N, H]
    B: tl.constexpr,
    N: tl.constexpr,
    H: tl.constexpr,
    # strides (in elements) for left
    stride_lb, stride_li, stride_lk, stride_lh,
    # strides for right
    stride_rb, stride_ri, stride_rk, stride_rh,
    # strides for out
    stride_ob, stride_oi, stride_ok, stride_oh,
    # tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # --------------------------------------------------------------
    # Program IDs
    pid_i = tl.program_id(0)          # tile row (i)
    pid_j = tl.program_id(1)          # tile col (j)
    pid_bh = tl.program_id(2)         # combined batch * hidden index

    # decode batch and hidden dimension from pid_bh
    b = pid_bh // H
    h = pid_bh % H

    # --------------------------------------------------------------
    # Offsets for the i and j dimensions
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i_offset < N
    j_mask = j_offset < N

    # Base pointers for the current batch and hidden channel
    left_base  = left_ptr  + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base   = out_ptr   + b * stride_ob + h * stride_oh

    # --------------------------------------------------------------
    # Accumulator (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # left[b, i, k, h] -> shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + i_offset[:, None] * stride_li + k_offset[None, :] * stride_lk
        a = tl.load(left_ptrs,
                    mask=i_mask[:, None] & k_mask[None, :],
                    other=0.0)

        # right[b, j, k, h] need shape (BLOCK_K, BLOCK_N)
        # we load as (k, j) to match tl.dot(A, B)
        right_ptrs = right_base + j_offset[None, :] * stride_ri + k_offset[:, None] * stride_rk
        b_tile = tl.load(right_ptrs,
                         mask=j_mask[None, :] & k_mask[:, None],
                         other=0.0)

        # Accumulate dot product over K
        acc += tl.dot(a, b_tile)   # (BLOCK_M, BLOCK_N)

    # --------------------------------------------------------------
    # Write out the result (float32)
    out_ptrs = out_base + i_offset[:, None] * stride_oi + j_offset[None, :] * stride_ok
    tl.store(out_ptrs,
             acc,
             mask=i_mask[:, None] & j_mask[None, :])


def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication for the TriMul contraction.
    left/right: [B, N, N, H]   (torch.float16)
    Returns:    out : [B, N, N, H] (torch.float32)
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, _, H = left.shape
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=left.device)

    # Tile size selection (tuned for B200, works for the given N range)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Compute strides in elements (torch returns strides in element units)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_ri, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_ok, stride_oh = out.stride()

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # pid_i dimension
        (N + BLOCK_N - 1) // BLOCK_N,   # pid_j dimension
        B * H,                          # pid_bh dimension (batch * hidden)
    )

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_ri, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_ok, stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,  # reasonable default
    )
    return out


def custom_kernel(data):
    """
    Entry point for the TriMul "outgoing" forward pass.

    data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        torch.Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data

    # ------------------------------------------------------------------
    # Configuration
    dim = config["dim"]               # input / output channel dimension
    hidden_dim = config["hidden_dim"] # internal hidden dimension

    # ------------------------------------------------------------------
    # Normalization (LayerNorm over the last dimension)
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    left_proj_weight  = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]  # (hidden_dim, dim)

    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # Apply mask if present (mask shape [B, N, N])
    if mask_tensor is None:
        mask = torch.ones_like(left[..., :1])
    else:
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)   # broadcast over hidden
    left = left * mask
    right = right * mask

    # ------------------------------------------------------------------
    # Gating logits + sigmoid
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to half for the heavy contraction
    left_half  = left.to(torch.float16)
    right_half = right.to(torch.float16)

    # Core TriMul contraction via Triton
    out = tri_mul_triton(left_half, right_half)   # [B,N,N,hidden] float32

    # ------------------------------------------------------------------
    # Output LayerNorm (over hidden dimension)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    # Multiply by output gate
    out = out * out_gate

    # Final linear projection back to model dimension
    to_out_weight = weights["to_out.weight"]   # (dim, hidden_dim)
    out = F.linear(out, to_out_weight)        # [B, N, N, dim]

    return out
