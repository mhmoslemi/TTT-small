import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
        a_ptr, b_ptr, c_ptr,
        M, N, K,
        stride_am, stride_ak, stride_ab,   # a: batch, M, K
        stride_bk, stride_bn, stride_bb,   # b: batch, K, N
        stride_cm, stride_cn, stride_cb,   # c: batch, M, N
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
        FP16: tl.constexpr):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets for the block.
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A and B tiles
        a_ptrs = a_ptr + pid_batch * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + pid_batch * stride_bb + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Compute matrix multiplication
        acc += tl.dot(a, b)

    # Write back C
    c_ptrs = c_ptr + pid_batch * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn

"""
TriMul (outgoing) forward pass with a fused Triton batched‑matmul.

Algorithm
---------
1. Layer‑norm the input tensor x ∈ [B, N, N, C].
2. Linear projections → left, right ∈ [B, N, N, H] (H = hidden_dim).
3. (optional) Apply the pairwise mask.
4. Compute gates (sigmoid of linear projections) and multiply them:
       left  = left  * left_gate
       right = right * right_gate
5. Fuse the O(N³·H) contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   by reshaping to a batch of matrix multiplies:
       A = left .permute(0,3,1,2) → (B·H, N, N)
       B = right.permute(0,3,1,2).transpose(-2,-1) → (B·H, N, N)
   and executing a custom Triton kernel that performs
   C = A @ B for every batch element.
6. Reshape the result back to [B, N, N, H] (float16 → float32).
7. Apply LayerNorm on the hidden dimension, multiply by out_gate,
   and finally a linear projection back to the original channel size C.

Only the O(N³) contraction is performed on the GPU via Triton,
all other operations use PyTorch (which is already highly optimized).
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton batched‑matmul kernel (C = A @ B) where A, B, C are FP16.
# A, B, C have shapes (batch, M, K), (batch, K, N), (batch, M, N).
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,                       # matrix dimensions (all equal to seq_len)
    stride_am, stride_ak, stride_ab,  # A: row‑stride, col‑stride, batch‑stride
    stride_bk, stride_bn, stride_bb,  # B: row‑stride (K), col‑stride (N), batch‑stride
    stride_cm, stride_cn, stride_cb,  # C: row‑stride, col‑stride, batch‑stride
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)      # batch index
    pid_m = tl.program_id(1)          # block row
    pid_n = tl.program_id(2)          # block col

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension
    for kk in range(0, K, BLOCK_K):
        offs_k = kk + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile [BLOCK_M, BLOCK_K]
        a_ptrs = (
            a_ptr
            + pid_batch * stride_ab
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )   # fp16 → will be up‑cast in dot

        # Load B tile [BLOCK_K, BLOCK_N]
        b_ptrs = (
            b_ptr
            + pid_batch * stride_bb
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )   # fp16

        # Accumulate the multiplication (A @ B)
        acc += tl.dot(a, b)   # result is FP32

    # Write back the result tile
    c_ptrs = (
        c_ptr
        + pid_batch * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """TriMul forward (outgoing) – pure‑forward (no gradients)."""
    # unpack data
    input_tensor, mask, weights, config = data
    dim = config["dim"]          # C (input & output channel dimension)
    hidden_dim = config["hidden_dim"]  # H = projection dimension
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Helper: LayerNorm (bias and weight are provided)
    # ------------------------------------------------------------------
    def layer_norm(x, weight, bias, eps=1e-5):
        return torch.nn.functional.layer_norm(
            x, (x.shape[-1],), weight=weight, bias=bias, eps=eps
        )

    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    # ------------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        norm_weight=weights["norm.weight"],
        norm_bias=weights["norm.bias"],
    )  # [B, N, N, C]

    # ------------------------------------------------------------------
    # 2. Linear projections (FP32 → later FP16)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)        # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    left = left * left_gate          # [B, N, N, H]
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to FP16 for the heavy contraction
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, H = left.shape

    # ------------------------------------------------------------------
    # 6. Reshape to a batch of (N×N) matrix multiplies: (B·H, N, N)
    # ------------------------------------------------------------------
    # (B, N, N, H) → (B, H, N, N) → contiguous → (B·H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Transpose the last two axes of right (so we compute A @ Bᵀ)
    right_t = right_perm.transpose(-2, -1).contiguous()

    batch = B * H
    left_flat = left_perm.view(batch, N, N)
    right_flat = right_t.view(batch, N, N)

    # Output buffer (FP16)
    out_flat = torch.empty((batch, N, N), dtype=torch.float16, device=input_tensor.device)

    # ------------------------------------------------------------------
    # 7. Launch Triton batched‑matmul kernel
    # ------------------------------------------------------------------
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 64
    grid = (
        batch,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                # M
        N,                # N
        N,                # K
        left_flat.stride(1),  # stride_am
        left_flat.stride(2),  # stride_ak
        left_flat.stride(0),  # stride_ab
        right_flat.stride(1), # stride_bk (row of B)
        right_flat.stride(2), # stride_bn (col of B)
        right_flat.stride(0), # stride_bb
        out_flat.stride(1),   # stride_cm
        out_flat.stride(2),   # stride_cn
        out_flat.stride(0),   # stride_cb
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Ensure kernel completion
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 8. Reshape back to [B, N, N, H] and convert to FP32
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 9. Output LayerNorm, gating, and final linear projection
    # ------------------------------------------------------------------
    out = layer_norm(
        out,
        norm_weight=weights["to_out_norm.weight"],
        norm_bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out
