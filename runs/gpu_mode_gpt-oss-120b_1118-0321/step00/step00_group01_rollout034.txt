def matmul_kernel(A_ptr, B_ptr, C_ptr,
                  M, N, K,
                  stride_am, stride_ak, stride_a_batch,
                  stride_bk, stride_bn, stride_b_batch,
                  stride_cm, stride_cn, stride_c_batch,
                  BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):

def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # flatten and ensure dtype is float32
    x = x.contiguous()
    if mask is None or nomask:
        mask_f = torch.ones(x.shape[:3], dtype=x.dtype, device=x.device).unsqueeze(-1)
    else:
        mask_f = mask.to(x.dtype).unsqueeze(-1)

    # layer norm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    eps = 1e-5
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # mask
    left = left * mask_f
    right = right * mask_f

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Prepare for matmul kernel
    B, N, _, _ = left.shape
    # reshape left: [B, hidden_dim, N, N]
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,2,1).contiguous()  # note transpose of last two dims

    left_flat = left_perm.view(B * hidden_dim, N, N)
    right_flat = right_perm.view(B * hidden_dim, N, N)

    # compute out via Triton matmul
    out_flat = triton_matmul(left_flat, right_flat)

def custom_kernel(data):
    """
    Triton-accelerated implementation of the outgoing Triangle Multiplicative Update (TriMul)
    as described in AlphaFold3. The heavy O(N^3) tensor contraction is performed by a
    fused batched matrix multiplication kernel written in Triton. All other
    linear, gating and layer‑norm operations are carried out with PyTorch.
    """
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ensure contiguous tensors
    x = x.contiguous()
    device = x.device
    dtype = x.dtype

    # ---- LayerNorm on input ----
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    eps = 1e-5
    x_norm = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ---- Linear projections ----
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ---- Mask handling ----
    if mask is None or nomask:
        mask_f = torch.ones(x.shape[:3], dtype=dtype, device=device).unsqueeze(-1)
    else:
        mask_f = mask.to(dtype).unsqueeze(-1)
    left = left * mask_f
    right = right * mask_f

    # ---- Gating ----
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---- Prepare tensors for the O(N^3) contraction ----
    B, N, _, _ = left.shape  # left shape: [B, N, N, hidden_dim]
    # reshape to [B, hidden_dim, N, N] and fuse batch & channel
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden_dim, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous() # transpose inner dimensions

    left_flat = left_perm.view(B * hidden_dim, N, N)     # [B*hidden_dim, N, N]
    right_flat = right_perm.view(B * hidden_dim, N, N)   # [B*hidden_dim, N, N] interpreted as (K,N)

    # ---- Triton batched matmul: out = left @ right_T  ----
    out_flat = triton_matmul(left_flat, right_flat)     # [B*hidden_dim, N, N]

    # reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ---- Output layer‑norm and gating ----
    out = F.layer_norm(out, (hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=eps)
    out = out * out_gate

    # ---- Final linear projection back to original dimension ----
    out = F.linear(out, weights["to_out.weight"])
    return out

"""
TriMul (outgoing) forward pass with a Triton‑accelerated O(N³) contraction.

The heavy reduction
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
is expressed as a batched matrix multiplication per hidden channel
(left @ rightᵀ) and computed by a custom Triton kernel.  All other
operations (layer‑norm, linear projections, gating) are performed with
standard PyTorch for simplicity and numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,   # A: (batch, M, K)
    stride_bk, stride_bn, stride_bb,   # B: (batch, K, N)
    stride_cm, stride_cn, stride_cb,   # C: (batch, M, N)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: C_i = A_i @ B_i for each i in batch."""
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_bc = tl.program_id(2)  # batch * hidden channel index

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A: (batch, M, K)
        a_ptrs = (
            A_ptr
            + pid_bc * stride_ab
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # B: (batch, K, N)
        b_ptrs = (
            B_ptr
            + pid_bc * stride_bb
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # Write back
    c_ptrs = (
        C_ptr
        + pid_bc * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def _triton_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication using the Triton kernel above.
    A: [batch, M, K], B: [batch, K, N]  →  C: [batch, M, N]
    """
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2, "Incompatible inner dimensions for matmul"

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
    grid = (tl.cdiv(M, BLOCK_M), tl.cdiv(N, BLOCK_N), batch)

    _matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        A.stride(1),  # stride_am
        A.stride(2),  # stride_ak
        A.stride(0),  # stride_ab
        B.stride(1),  # stride_bk
        B.stride(2),  # stride_bn
        B.stride(0),  # stride_bb
        C.stride(1),  # stride_cm
        C.stride(2),  # stride_cn
        C.stride(0),  # stride_cb
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


def custom_kernel(data):
    """
    TriMul (outgoing) forward pass.
    Arguments:
        data = (input_tensor, mask_tensor, weight_dict, config_dict)
    Returns:
        output tensor of shape [bs, seq_len, seq_len, dim]
    """
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Guarantees contiguous layout
    x = x.contiguous()
    device, dtype = x.device, x.dtype

    # -------------------- LayerNorm on input --------------------
    eps = 1e-5
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -------------------- Linear projections --------------------
    left = F.linear(x_norm, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x_norm, weights["right_proj.weight"])  # [B, N, N, H]

    # -------------------- Mask handling --------------------
    if mask is None or nomask:
        mask_f = torch.ones(x.shape[:3], dtype=dtype, device=device).unsqueeze(-1)
    else:
        mask_f = mask.to(dtype).unsqueeze(-1)               # [B, N, N, 1]
    left = left * mask_f
    right = right * mask_f

    # -------------------- Gating --------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -------------------- Prepare for O(N³) contraction --------------------
    B, N, _, _ = left.shape  # left.shape == [B, N, N, hidden_dim]

    # reshape to [B, hidden_dim, N, N] and fuse batch & channel
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()        # transpose inner dims → [B, H, N, N]

    left_flat = left_perm.view(B * hidden_dim, N, N)          # [B*H, N, N]   (M, K)
    right_flat = right_perm.view(B * hidden_dim, N, N)        # [B*H, K, N]

    # -------------------- Triton batched GEMM --------------------
    out_flat = _triton_matmul(left_flat, right_flat)          # [B*H, N, N]

    # Reshape back to original layout
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # -------------------- Output LayerNorm, gating and final projection --------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])  # back to [B, N, N, dim]

    return out
