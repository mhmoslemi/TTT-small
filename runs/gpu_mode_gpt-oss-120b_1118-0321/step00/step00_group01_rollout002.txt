def tri_mul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    B, N, H,
    stride_ab, stride_ai, stride_ak, stride_ah,
    stride_bb, stride_bi, stride_bk, stride_bh,
    stride_cb, stride_ci, stride_cj, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)  # combined batch*H
    # compute batch and hidden indices
    b = pid_bh // H
    h = pid_bh % H

    # offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # masking for out-of-bound
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-6

    # LayerNorm weight & bias
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    # apply layernorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # compute left, right, gates
    # left = x @ left_proj_weight.T
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape [B,N,N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # mask handling
    if mask is not None:
        # mask shape [B,N,N]; ensure float
        m = mask.to(x.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * m
        right = right * m

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Convert to fp16 for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape
    # allocate output FP16
    out_fp16 = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # stride values
    stride_a_b = left_fp16.stride(0)
    stride_a_i = left_fp16.stride(1)
    stride_a_k = left_fp16.stride(2)
    stride_a_h = left_fp16.stride(3)

    stride_b_b = right_fp16.stride(0)
    stride_b_i = right_fp16.stride(1)  # this corresponds to j dimension
    stride_b_k = right_fp16.stride(2)
    stride_b_h = right_fp16.stride(3)

    stride_c_b = out_fp16.stride(0)
    stride_c_i = out_fp16.stride(1)
    stride_c_j = out_fp16.stride(2)
    stride_c_h = out_fp16.stride(3)

    # Blocking parameters
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

        left_fp16, right_fp16, out_fp16,
        B, N, hidden_dim,

        stride_b_b, stride_b_i, stride_b_k, stride_b_h,
        stride_c_b, stride_c_i, stride_c_j, stride_c_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K

    out = out_fp16.to(torch.float32)

"""
TriMul outgoing update (AlphaFold3) implemented with a custom Triton kernel.

The kernel computes, for each batch `b` and hidden channel `h`,
the matrix product:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
i.e. a batched GEMM where the second operand is transposed.
All other operations (LayerNorm, Linear projections, gating, final
normalisation and projection) are performed in PyTorch.
The core O(N³) reduction is fused into a single Triton kernel to
fully exploit GPU parallelism.
"""

import torch
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: out = left @ right^T   (per hidden channel, per batch)
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_matmul_kernel(
    # pointers
    A_ptr, B_ptr, C_ptr,
    # problem size
    B, N, H,
    # strides for A (shape B,N,N,H)
    stride_ab, stride_ai, stride_ak, stride_ah,
    # strides for B (shape B,N,N,H)
    stride_bb, stride_bi, stride_bk, stride_bh,
    # strides for C (shape B,N,N,H)
    stride_cb, stride_ci, stride_cj, stride_ch,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)            # tile index in i dimension
    pid_j = tl.program_id(1)            # tile index in j dimension
    pid_bh = tl.program_id(2)           # combined batch * hidden index

    b = pid_bh // H                     # batch index
    h = pid_bh % H                      # hidden channel index

    # ----------------------- offsets & masks ----------------------- #
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # pointer base for the current batch & channel
    a_base = A_ptr + b * stride_ab + h * stride_ah
    b_base = B_ptr + b * stride_bb + h * stride_bh
    c_base = C_ptr + b * stride_cb + h * stride_ch

    # ----------------------- reduction over k --------------------- #
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + (offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load B block transposed so that its shape is (BLOCK_K, BLOCK_N)
        b_ptrs = b_base + (offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bi)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # Multiply‑accumulate
        acc += tl.dot(a, b)   # (M,K) @ (K,N) -> (M,N)

    # ----------------------- store result -------------------------- #
    c_ptrs = c_base + (offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj)
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# --------------------------------------------------------------------------- #
# Entry point expected by the autograder
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: Tuple containing
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask_tensor:  torch.Tensor [B, N, N] (may be all‑ones)
            - weights:      dict of parameter tensors
            - config:       dict with keys "dim", "hidden_dim", "nomask", ...

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, on the same device).
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------- #
    # 1. LayerNorm over the channel dimension (dim)
    # ------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------- #
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------- #
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------- #
    # 3. Optional masking (applied after projection, before gating)
    # ------------------------------------------------------------------- #
    if not config.get("nomask", False):
        # mask is float (0/1) after unsqueeze
        m = mask_tensor.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * m
        right = right * m

    # ------------------------------------------------------------------- #
    # 4. Gating (sigmoid of linear transformations of the normalized x)
    # ------------------------------------------------------------------- #
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5. Core O(N³) multiplication via Triton
    # ------------------------------------------------------------------- #
    # Cast to half for bandwidth, accumulation done in fp32 inside the kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape
    # Allocate output buffer (half)
    out_fp16 = torch.empty((B, N, N, hidden_dim),
                           dtype=torch.float16,
                           device=device)

    # Strides (in elements, not bytes)
    stride_a_b = left_fp16.stride(0)
    stride_a_i = left_fp16.stride(1)
    stride_a_k = left_fp16.stride(2)
    stride_a_h = left_fp16.stride(3)

    stride_b_b = right_fp16.stride(0)
    stride_b_i = right_fp16.stride(1)   # j dimension
    stride_b_k = right_fp16.stride(2)
    stride_b_h = right_fp16.stride(3)

    stride_c_b = out_fp16.stride(0)
    stride_c_i = out_fp16.stride(1)
    stride_c_j = out_fp16.stride(2)
    stride_c_h = out_fp16.stride(3)

    # Tuning parameters – these work well for the given N range.
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),      # i‑tiles
        triton.cdiv(N, BLOCK_N),      # j‑tiles
        B * hidden_dim,               # batch * channel
    )

    tri_mul_matmul_kernel[grid](
        left_fp16,                     # A_ptr
        right_fp16,                    # B_ptr
        out_fp16,                      # C_ptr
        B, N, hidden_dim,
        stride_a_b, stride_a_i, stride_a_k, stride_a_h,
        stride_b_b, stride_b_i, stride_b_k, stride_b_h,
        stride_c_b, stride_c_i, stride_c_j, stride_c_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Back to float32 for the remaining operations
    out = out_fp16.to(torch.float32)   # [B,N,N,hidden]

    # ------------------------------------------------------------------- #
    # 6. Output LayerNorm over hidden_dim
    # ------------------------------------------------------------------- #
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )

    # ------------------------------------------------------------------- #
    # 7. Apply the output gate (element‑wise)
    # ------------------------------------------------------------------- #
    out = out * out_gate

    # ------------------------------------------------------------------- #
    # 8. Final linear projection back to original dim
    # ------------------------------------------------------------------- #
    to_out_w = weights["to_out.weight"]
    output = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim]

    return output
