import triton
import triton.language as tl

@triton.jit
def trimul_kernel(left_ptr, right_ptr, out_ptr,
                  B, N, hidden_dim,
                  stride_l_b, stride_l_i, stride_l_k, stride_l_d,
                  stride_r_b, stride_r_i, stride_r_k, stride_r_d,
                  stride_o_b, stride_o_i, stride_o_j, stride_o_d,
                  BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_hd = tl.program_id(2)  # combined batch & hidden
    b = pid_hd // hidden_dim
    d = pid_hd % hidden_dim

    # Offsets for each tensor
    off_left = left_ptr + b * stride_l_b + d * stride_l_d
    off_right = right_ptr + b * stride_r_b + d * stride_r_d
    off_out = out_ptr + b * stride_o_b + d * stride_o_d

    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_offset < N
    mask_j = j_offset < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # left tile (M x K)
        left_ptrs = off_left + i_offset[:, None] * stride_l_i + k_offset[None, :] * stride_l_k
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right tile (K x N)
        right_ptrs = off_right + k_offset[:, None] * stride_r_k + j_offset[None, :] * stride_r_i
        right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(left_tile, right_tile)

    out_ptrs = off_out + i_offset[:, None] * stride_o_i + j_offset[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

    import torch
    import triton
    import triton.language as tl

    # kernel defined above

    # unpack
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # shape B,N,N,dim
    B, N, _, _ = input_tensor.shape

    # Normalize input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape B,N,N,hidden_dim
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask if needed
    if mask_tensor is not None and not config.get("nomask", False):

        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous
    left = left.contiguous()
    right = right.contiguous()

    # Allocate output
    out_hidden = torch.empty_like(left)  # same shape [B,N,N,hidden_dim]

    # Determine block sizes (tuned)

    BLOCK_N = 64
    BLOCK_K = 32

"""
TriMul "outgoing" forward pass.

The heavy O(N³·H) computation (einsum over the sequence dimension) is
implemented as a batched matrix multiplication in a custom Triton kernel.
For each batch b and hidden channel h we compute:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
which is equivalent to (left @ rightᵀ) per hidden channel.
All other operations (LayerNorms, Linear projections, gates, mask handling,
final LayerNorm and output projection) are performed with PyTorch.
"""
import torch
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel for the core TriMul operation.
# --------------------------------------------------------------------------- #
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                    # dimensions
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out = Σ_k left[...,k,:] * right[...,k,:]   (batched GEMM)

    Grid dimensions:
        program_id(0) -> tile row (i)
        program_id(1) -> tile column (j)
        program_id(2) -> combined batch * hidden channel index
    """
    pid_hd = tl.program_id(2)                # index over batch*hidden
    b = pid_hd // H                          # batch index
    h = pid_hd % H                           # hidden channel index

    # Pointers to the start of the (b,h) slice
    left_base  = left_ptr  + b * stride_l_b  + h * stride_l_h
    right_base = right_ptr + b * stride_r_b  + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b  + h * stride_o_h

    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # Offsets for the i and j tiles
    i_off = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_off < N
    mask_j = j_off < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over reduction dimension k
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # left[i,k] tile (M x K)
        left_ptrs = left_base + i_off[:, None] * stride_l_i + k_off[None, :] * stride_l_k
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0)

        # right[j,k] is accessed as right[k,j] for a K×N tile
        right_ptrs = right_base + k_off[:, None] * stride_r_k + j_off[None, :] * stride_r_i
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0)

        # Multiply‑accumulate
        acc += tl.dot(left_tile, right_tile)

    # Write the result tile
    out_ptrs = out_base + i_off[:, None] * stride_o_i + j_off[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

    data : tuple
        (input_tensor, mask_tensor, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask_tensor  : torch.Tensor of shape [B, N, N] (may be ignored)
        - weights      : dict of model weights
        - config       : dict containing at least "dim", "hidden_dim" and "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim]
    """
    import torch.nn.functional as F

    # ------------------------------------------------------------------- #
    # Unpack arguments
    # ------------------------------------------------------------------- #
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------- #
    # LayerNorm on the input (dim‑dimensional)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------- #
    # Linear projections (no bias)
    # ------------------------------------------------------------------- #
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # Optional mask (applied after projection, before gating)
    # ------------------------------------------------------------------- #
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------- #
    # Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # Core TriMul via Triton (batched GEMM over the sequence dimension)
    # ------------------------------------------------------------------- #
    left = left.contiguous()
    right = right.contiguous()

    out_hidden = torch.empty_like(left)  # [B, N, N, hidden_dim]

    # Tile sizes – tuned for typical B200 memory/occupancy limits
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),               # tiles over i
        triton.cdiv(N, BLOCK_N),               # tiles over j
        B * hidden_dim,                        # combined batch‑hidden dimension
    )

    trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out_hidden,
        B=B,
        N=N,
        H=hidden_dim,
        stride_l_b=left.stride(0),
        stride_l_i=left.stride(1),
        stride_l_k=left.stride(2),
        stride_l_h=left.stride(3),
        stride_r_b=right.stride(0),
        stride_r_i=right.stride(1),
        stride_r_k=right.stride(2),
        stride_r_h=right.stride(3),
        stride_o_b=out_hidden.stride(0),
        stride_o_i=out_hidden.stride(1),
        stride_o_j=out_hidden.stride(2),
        stride_o_h=out_hidden.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------- #
    # Post‑processing: LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate  # gating after to_out_norm

    # Final linear projection back to original dim
    out = F.linear(out, weights["to_out.weight"])

    return out
