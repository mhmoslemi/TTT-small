def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)  # default epsilon for layernorms.

    # Step1: LayerNorm across dim (last dimension)
    # norm weight shape [dim], bias shape [dim].

    # Use torch.nn.functional.layer_norm
    # Result shape: [bs,N,N,dim] same as input.

    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # Left and right projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Mask handling
    nomask = config.get("nomask", True)
    if not nomask:
        # mask shape [bs,N,N]; broadcast to dim
        mask = mask_tensor.to(x.dtype).unsqueeze(-1) # shape [bs,N,N,1]
        left = left * mask
        right = right * mask
    else:
        # If nomask True just ignore mask (i.e., treat all ones)
        pass

    # Gate computations
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Reshape for Triton kernel: combine batch and hidden_dim.

    bs, N, _, _ = x.shape  # input shape maybe [bs, N, N, dim]
    # left shape [bs, N, N, hidden_dim]; we need contiguous layout to reshape
    left = left.contiguous()
    right = right.contiguous()
    # Permute to bring hidden_dim to batch dimension
    left_ = left.permute(0, 3, 1, 2).contiguous()  # [bs, hidden_dim, N, N]
    right_ = right.permute(0, 3, 1, 2).contiguous()

    total_batch = bs * hidden_dim
    left_flat = left_.view(total_batch, N, N)  # [total_batch, N, N]
    right_flat = right_.view(total_batch, N, N)

    # Output tensor allocation for hidden_dim output before final transform
    out_flat = torch.empty_like(left_flat)  # same dtype/device

    # Triton kernel parameters: compute strides
    stride_left_batch, stride_left_i, stride_left_k = left_flat.stride()
    stride_right_batch, stride_right_i, stride_right_k = right_flat.stride()
    stride_out_batch, stride_out_i, stride_out_j = out_flat.stride()

    # Load kernel
    import triton
    import triton.language as tl

    @triton.jit
    def tri_mul_kernel(
        left_ptr, right_ptr, out_ptr,
        N: tl.constexpr,
        stride_l_batch, stride_l_i, stride_l_k,
        stride_r_batch, stride_r_i, stride_r_k,
        stride_o_batch, stride_o_i, stride_o_j,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    ):
        pid_batch = tl.program_id(0)
        pid_m = tl.program_id(1)
        pid_n = tl.program_id(2)

        # compute base offsets
        # offsets for rows and cols
        offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
        # mask for out-of-range
        mask_m = offs_m < N
        mask_n = offs_n < N

        # Initialize accumulator
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # Loop over K dimension
        for k in range(0, N, BLOCK_K):
            cur_k = k + tl.arange(0, BLOCK_K)
            mask_k = cur_k < N

            # Load left tile (M x K)
            left_ptrs = left_ptr + pid_batch * stride_l_batch + (offs_m[:, None] * stride_l_i) + (cur_k[None, :] * stride_l_k)
            l = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

            # Load right tile transposed (K x N)
            # We need right as (K x N) where rows are k and columns are n
            right_ptrs = right_ptr + pid_batch * stride_r_batch + (cur_k[:, None] * stride_r_i) + (offs_n[None, :] * stride_r_k)
            r = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

            # Compute block multiplication
            acc += tl.dot(l, r)

        # Write back
        out_ptrs = out_ptr + pid_batch * stride_o_batch + (offs_m[:, None] * stride_o_i) + (offs_n[None, :] * stride_o_j)
        tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

    # Determine block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (total_batch, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)
    tri_mul_kernel[grid](
        left_ptr=left_flat,
        right_ptr=right_flat,
        out_ptr=out_flat,
        N=N,
        stride_l_batch=stride_left_batch,
        stride_l_i=stride_left_i,
        stride_l_k=stride_left_k,
        stride_r_batch=stride_right_batch,
        stride_r_i=stride_right_i,
        stride_r_k=stride_right_k,
        stride_o_batch=stride_out_batch,
        stride_o_i=stride_out_i,
        stride_o_j=stride_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        # maybe enable prefetch? Not needed
    )

    # Reshape output back to [bs,N,N,hidden_dim]
    out = out_flat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Apply to_out_norm
    # LayerNorm over hidden_dim dimension
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Apply out_gate
    out = out * out_gate

    # Final linear to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

TriMul forward (outgoing) implementation with Triton.

The operation:

    (input_tensor, mask_tensor, weights_dict, config_dict)

`weights_dict` contains the parameters of the TriMul module:
    norm.weight, norm.bias,
    left_proj.weight, right_proj.weight,
    left_gate.weight, right_gate.weight, out_gate.weight,
    to_out_norm.weight, to_out_norm.bias,
    to_out.weight

The function returns a tensor of shape [B, N, N, dim] (float32).
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N: tl.constexpr,
    stride_l_batch, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_i, stride_r_k,
    stride_o_batch, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out = left @ right.T for a batch of matrices.
    left, right, out have shape [batch, N, N] (C‑contiguous).
    """
    pid_batch = tl.program_id(0)          # which (batch * hidden) matrix
    pid_m = tl.program_id(1)              # tile row index
    pid_n = tl.program_id(2)              # tile column index

    # Compute absolute offsets for the start of the tile.
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension.
    for k in range(0, N, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # left tile: (M x K)
        left_ptrs = (
            left_ptr
            + pid_batch * stride_l_batch
            + (offs_m[:, None] * stride_l_i)
            + (cur_k[None, :] * stride_l_k)
        )
        l = tl.load(
            left_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # right tile transposed: (K x N)
        # we read right as (row = k, col = n) which is the transpose of the
        # usual layout (row = j, col = k).
        right_ptrs = (
            right_ptr
            + pid_batch * stride_r_batch
            + (cur_k[:, None] * stride_r_i)
            + (offs_n[None, :] * stride_r_k)
        )
        r = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        # Block‑wise matrix multiplication.
        acc += tl.dot(l, r)  # (M x K) @ (K x N) -> (M x N)

    # Store the result.
    out_ptrs = (
        out_ptr
        + pid_batch * stride_o_batch
        + (offs_m[:, None] * stride_o_i)
        + (offs_n[None, :] * stride_o_j)
    )
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a Triton kernel for the
    O(N³·C) reduction.
    """
    # Unpack inputs
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)       # LayerNorm epsilon
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1) Input LayerNorm
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 2) Linear projections (left, right) and gating values
    # -----------------------------------------------------------------
    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask if required (mask broadcasted over the hidden dim)
    if not nomask:
        mask = mask_tensor.to(dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # Gates (sigmoid)
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 3) Triton kernel: batched matmul left @ right.T
    # -----------------------------------------------------------------
    bs, N, _, _ = x.shape  # N is sequence length

    # Rearrange tensors to bring hidden dimension into the batch axis.
    # Shape after permute/contiguous: [B, hidden, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    total_batch = bs * hidden_dim
    left_flat = left_perm.view(total_batch, N, N)    # [B*H, N, N]
    right_flat = right_perm.view(total_batch, N, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat)

    # Strides (in elements)
    stride_l_batch, stride_l_i, stride_l_k = left_flat.stride()
    stride_r_batch, stride_r_i, stride_r_k = right_flat.stride()
    stride_o_batch, stride_o_i, stride_o_j = out_flat.stride()

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        total_batch,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_kernel[grid](
        left_ptr=left_flat,
        right_ptr=right_flat,
        out_ptr=out_flat,
        N=N,
        stride_l_batch=int(stride_l_batch),
        stride_l_i=int(stride_l_i),
        stride_l_k=int(stride_l_k),
        stride_r_batch=int(stride_r_batch),
        stride_r_i=int(stride_r_i),
        stride_r_k=int(stride_r_k),
        stride_o_batch=int(stride_o_batch),
        stride_o_i=int(stride_o_i),
        stride_o_j=int(stride_o_j),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 4) Output LayerNorm, out_gate, final projection
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
