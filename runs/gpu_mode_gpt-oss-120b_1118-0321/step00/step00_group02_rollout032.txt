    input_tensor, mask_tensor, weights, config = data
    
    device = input_tensor.device
    
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]   # shape dim, hidden_dim
    
    # maybe ensure contiguous
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)
    
    # Projections
    left_proj = torch.nn.functional.linear(x, left_proj_weight)
    right_proj = torch.nn.functional.linear(x, right_proj_weight)
    
    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))
    
    # Mask handling
    if config.get("nomask", True) or mask_tensor is None:
        mask = torch.ones(input_tensor.shape[:3], dtype=x.dtype, device=device)  # shape [B,N,N]
    else:
        mask = mask_tensor
    mask = mask.unsqueeze(-1)   # [B,N,N,1]
    
    # Apply mask and gates
    left = left_proj * mask * left_gate
    right = right_proj * mask * right_gate
    
    B, N, _, _ = left.shape  # _ is hidden_dim
    
    # Prepare for triangular multiplication (outgoing)
    # Reshape to (B*hidden_dim, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous()
    right_flat = right.permute(0, 3, 1, 2).contiguous()
    batch_hidden = B * hidden_dim
    left_flat = left_flat.view(batch_hidden, N, N)
    right_flat = right_flat.view(batch_hidden, N, N)
    
    out_flat = torch.empty_like(left_flat)
    
    # Triton kernel parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), batch_hidden)
    
    tri_mul_kernel[grid](

"""
TriMul custom kernel for outgoing update.

Algorithm:
  1. Apply per‑channel LayerNorm to the input.
  2. Linear projections and sigmoid gates (left/right/out).
  3. Mask + gate are fused element‑wise.
  4. Compute the contracted Tensor:
       out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
     implemented as a batched GEMM (N×N) for each hidden channel using a
     Triton kernel that tiles the i‑j plane and iterates over k.
  5. Layer‑norm + out‑gate, final linear projection back to `dim`.

The kernel works for arbitrary batch size (≤2), sequence length up to
1024 and hidden dimensions up to 384.
"""
import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch_hidden, N,
    stride_l_bh, stride_l_i, stride_l_k,
    stride_r_bh, stride_r_i, stride_r_k,
    stride_o_bh, stride_o_i, stride_o_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    bh = tl.program_id(2)          # which batch*hidden slice we are processing
    pid_m = tl.program_id(0)       # tile index in the i dimension
    pid_n = tl.program_id(1)       # tile index in the j dimension

    # start offsets for the current tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i,k] – shape (BLOCK_M, BLOCK_K)
        left = tl.load(
            left_ptr + bh * stride_l_bh + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )
        # right[j,k] – shape (BLOCK_N, BLOCK_K)
        right = tl.load(
            right_ptr + bh * stride_r_bh + offs_j[:, None] * stride_r_i + offs_k[None, :] * stride_r_k,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0
        )
        # accumulate Σ_k left[i,k] * right[j,k]; trans_b=True treats right as (K,N)
        acc += tl.dot(left, right, trans_b=True)

    # write result
    out = acc
    out_ptrs = out_ptr + bh * stride_o_bh + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_k
    tl.store(out_ptrs, out, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Forward pass of the triangular multiplicative update (outgoing variant).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim].
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # ------------------------------------------------------------------
    # Layer‑norm on the input
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # Linear projections
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Gates (sigmoid after a linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # Mask handling – when `nomask` is True we treat the mask as all‑ones
    # ------------------------------------------------------------------
    if config.get("nomask", True) or mask_tensor is None:
        mask = torch.ones(
            (input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]),
            dtype=input_tensor.dtype,
            device=device,
        )
    else:
        mask = mask_tensor
    mask = mask.unsqueeze(-1)  # [B,N,N,1]

    # ------------------------------------------------------------------
    # Fuse mask and gates (this part could be a separate kernel, but the
    # heavy work – the N³ contraction – is already in Triton)
    # ------------------------------------------------------------------
    left = left_proj * mask * left_gate
    right = right_proj * mask * right_gate

    B, N, _, _ = left.shape
    # ------------------------------------------------------------------
    # Reshape to (B*hidden_dim, N, N) so each hidden channel becomes a
    # separate batch for the kernel.
    # ------------------------------------------------------------------
    left_flat = left.permute(0, 3, 1, 2).contiguous()
    right_flat = right.permute(0, 3, 1, 2).contiguous()
    batch_hidden = B * hidden_dim
    left_flat = left_flat.view(batch_hidden, N, N)
    right_flat = right_flat.view(batch_hidden, N, N)

    out_flat = torch.empty_like(left_flat)

    # ------------------------------------------------------------------
    # Triton kernel launch
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch_hidden,
    )
    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_hidden,
        N,
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        right_flat.stride(0),
        right_flat.stride(1),
        right_flat.stride(2),
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Make sure the kernel has finished before we continue
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Output layer‑norm, out‑gate and final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul custom kernel (outgoing update).

The kernel implements the heavy N³ contraction:
    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
by treating each hidden channel as an independent batch and performing a
batched matrix multiplication (N×N)×(N×N)ᵀ with a Triton kernel that
tiles the i‑j plane and iterates over k.  All other operations (LayerNorm,
linear projections, sigmoid gates, masking, final projection) are done
with PyTorch.

Supports batch size ≤2, sequence length up to 1024 and hidden dimensions
up to 384.  The kernel works in float32 and returns a float32 output.
"""

import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: batched GEMM for the contraction
# --------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch_hidden, N,
    stride_l_bh, stride_l_i, stride_l_k,
    stride_r_bh, stride_r_i, stride_r_k,
    stride_o_bh, stride_o_i, stride_o_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # 3‑D grid: (i‑tile, j‑tile, batch*hidden)
    bh      = tl.program_id(2)          # which (batch, hidden) slice
    pid_m   = tl.program_id(0)          # tile in i dimension
    pid_n   = tl.program_id(1)          # tile in j dimension

    # Offsets of the current tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator for the tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k (the contracted dimension)
    num_k = tl.cdiv(N, BLOCK_K)
