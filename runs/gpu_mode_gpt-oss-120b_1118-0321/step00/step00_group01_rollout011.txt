"""
Optimized forward pass for the "outgoing" TriMul operation.

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C) using the provided LN weights.
2. Linear projections (no bias) to obtain left/right tensors of shape
   (B,N,N,H).  Apply optional mask (broadcast on the last dimension).
3. Compute sigmoid gates for left, right and the final output using linear
   layers on the normalized input.
4. Fuse the gates with the projected tensors.
5. Perform the core O(N³) operation:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM of left (i,k) with rightᵀ (k,j) for every head * batch.
   Implemented with a Triton kernel that tiles over i, j and k.
6. Permute the result back to (B,N,N,H) and apply a second LayerNorm,
   the output gate, and the final linear projection (H → C).
All heavy arithmetic (step 5) runs on the GPU via Triton; the remaining
operations use PyTorch which already provides highly tuned kernels.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM of left (M×K) with rightᵀ (K×N)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_T_kernel(
    left_ptr,  # const float*  [B, H, N, N]
    right_ptr, # const float*  [B, H, N, N]
    out_ptr,   # float*        [B, H, N, N]

    B, H, N,                 # problem sizes
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_i, stride_r_k,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,

    BLOCK_M: tl.constexpr,   # tile size in i dimension
    BLOCK_N: tl.constexpr,   # tile size in j dimension
    BLOCK_K: tl.constexpr,   # tile size in k dimension
):
    pid_m = tl.program_id(0)          # tile row (i)
    pid_n = tl.program_id(1)          # tile col (j)
    pid_bh = tl.program_id(2)         # batch * head index

    # ------------------------------------------------------------------
    # Compute global coordinates
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Decode batch and head from combined index
    bh = pid_bh
    b = bh // H
    h = bh % H

    # Base pointers for this (b, h) slice
    left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    for kb in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: (M, K)  – indices (i, k)
        left_ptrs = left_base + (offs_m[:, None] * stride_l_i) + (offs_k[None, :] * stride_l_k)
        left = tl.load(
            left_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )

        # right block transposed: (K, N) where right_T[k, j] = right[j, k]
        # indices (k, j)
        right_ptrs = right_base + (offs_k[:, None] * stride_r_k) + (offs_n[None, :] * stride_r_i)
        right_T = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        )

        # GEMM accumulate
        acc += tl.dot(left, right_T)   # (M,K) x (K,N) -> (M,N)

    # ------------------------------------------------------------------
    # Write result
    # ------------------------------------------------------------------
    out_ptrs = out_base + (offs_m[:, None] * stride_o_i) + (offs_n[None, :] * stride_o_j)
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor of shape [B, N, N, C]
        mask          : torch.Tensor of shape [B, N, N] (may be None)
        weights       : dict of torch.Tensor model parameters
        config        : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    out : torch.Tensor of shape [B, N, N, C]
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # Helper to move weights to correct device/dtype
    # --------------------------------------------------------------
    def to_dev(t):
        return t.to(device=device, dtype=dtype)

    # LayerNorm on input
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=to_dev(weights["norm.weight"]),
        bias=to_dev(weights["norm.bias"])
    )  # (B,N,N,dim)

    # --------------------------------------------------------------
    # Linear projections (no bias)
    # --------------------------------------------------------------
    left = torch.nn.functional.linear(x, to_dev(weights["left_proj.weight"]))   # (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x, to_dev(weights["right_proj.weight"]))

    # --------------------------------------------------------------
    # Optional mask
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # ensure mask dtype matches, broadcast on hidden dim
        mask_f = mask.to(dtype=dtype).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # Gates (sigmoid of linear layers)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, to_dev(weights["left_gate.weight"])))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, to_dev(weights["right_gate.weight"])))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, to_dev(weights["out_gate.weight"])))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # Core O(N³) operation via Triton
    # --------------------------------------------------------------
    # Permute to (B, H, N, N) for easier batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    assert H == hidden_dim and _ == N, "Unexpected shape after permute."

    out_perm = torch.empty_like(left_perm)  # (B, H, N, N)

    # Strides (in elements)
    stride_l_b, stride_l_h, stride_l_i, stride_l_k = left_perm.stride()
    stride_r_b, stride_r_h, stride_r_i, stride_r_k = right_perm.stride()
    stride_o_b, stride_o_h, stride_o_i, stride_o_j = out_perm.stride()

    # Tile sizes – trade‑off between shared memory and occupancy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        ( (N + BLOCK_M - 1) // BLOCK_M,
          (N + BLOCK_N - 1) // BLOCK_N,
          B * H )
    )

    batched_gemm_T_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        B, H, N,
        stride_l_b, stride_l_h, stride_l_i, stride_l_k,
        stride_r_b, stride_r_h, stride_r_i, stride_r_k,
        stride_o_b, stride_o_h, stride_o_i, stride_o_j,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # Ensure kernel completion before next PyTorch ops
    torch.cuda.synchronize()

    # Restore original layout (B,N,N,H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # Second LayerNorm, output gate, final linear projection
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_dev(weights["to_out_norm.weight"]),
        bias=to_dev(weights["to_out_norm.bias"])
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_dev(weights["to_out.weight"]))  # (B,N,N,dim)

    return out
