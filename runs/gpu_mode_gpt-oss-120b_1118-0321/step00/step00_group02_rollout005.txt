"""
TriMul (outgoing) implementation with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor (dim = last dimension).
2. Linear projections to a hidden dimension: left = x·W_left, right = x·W_right.
3. (Optional) apply a pairwise mask.
4. Compute three sigmoid gates from the normalized input and multiply them
   into the projected tensors.
5. Perform the heavy N‑by‑N matrix multiplications:
       out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
   This is exactly C = A @ Bᵀ per hidden channel.  The kernel launches one
   program per (batch, hidden) head and tiles the matrices with
   BLOCK_M × BLOCK_N × BLOCK_K blocks.
6. Layer‑norm the resulting tensor over the hidden dimension,
   multiply by the output gate and a final linear projection back to `dim`.

Only the N³ reduction (step 5) is performed on the GPU via Triton; all
other operations use PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    B, N, H,                           # batch, seq_len, hidden_dim
    stride_ab, stride_ai, stride_ak, stride_ad,   # strides for A (left)
    stride_bb, stride_bi, stride_bk, stride_bd,   # strides for B (right)
    stride_cb, stride_ci, stride_cj, stride_cd,   # strides for C (out)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """C = A @ Bᵀ for many heads (batch * hidden) using block‑wise reduction."""
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col
    pid_h = tl.program_id(2)          # head = batch * hidden

    batch = pid_h // H                # batch index
    d = pid_h % H                     # hidden index

    # offsets within the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for output tile (handle tail)
    mask_m = offs_m < N
    mask_n = offs_n < N
    mask_out = mask_m[:, None] & mask_n[None, :]

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, N, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # ----- Load A tile (M x K) -----
        a_ptrs = (
            a_ptr
            + batch * stride_ab
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
            + d * stride_ad
        )
        a_mask = mask_m[:, None] & mask_k[None, :]
        a = tl.load(a_ptrs, mask=a_mask, other=0.0, dtype=tl.float16)

        # ----- Load B tile (K x N) -----
        # we need Bᵀ, i.e. element B[j, k] -> row k, col j
        b_ptrs = (
            b_ptr
            + batch * stride_bb
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bi
            + d * stride_bd
        )
        b_mask = mask_k[:, None] & mask_n[None, :]
        b = tl.load(b_ptrs, mask=b_mask, other=0.0, dtype=tl.float16)

        # dot product: (M x K) @ (K x N) -> (M x N)
        acc += tl.dot(a, b)

    # ----- Store the result -----
    c_ptrs = (
        c_ptr
        + batch * stride_cb
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
        + d * stride_cd
    )
    tl.store(c_ptrs, acc, mask=mask_out)


def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module implemented with a fused Triton kernel.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    eps = 1e-5

    # unpack config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, seq_len, _, _ = input_tensor.shape

    # ---------- LayerNorm ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # ---------- Linear projections ----------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)               # [B,N,N,H]
    right = F.linear(x, right_proj_weight)

    # ---------- Optional mask ----------
    if mask is not None:
        # mask shape: [B, N, N] → broadcast over hidden dim
        mask_f = mask.to(x.dtype).unsqueeze(-1)        # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ---------- Gates ----------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ---------- Cast to FP16 for the heavy kernel ----------
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    # ---------- Output buffer ----------
    out = torch.empty((bs, seq_len, seq_len, hidden_dim), dtype=torch.float32, device=device)

    # ---------- Strides (in elements) ----------
    stride_ab, stride_ai, stride_ak, stride_ad = left_fp16.stride()
    stride_bb, stride_bi, stride_bk, stride_bd = right_fp16.stride()
    stride_cb, stride_ci, stride_cj, stride_cd = out.stride()

    # ---------- Triton launch ----------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (seq_len + BLOCK_M - 1) // BLOCK_M,
        (seq_len + BLOCK_N - 1) // BLOCK_N,
        bs * hidden_dim,
    )
    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        bs,
        seq_len,
        hidden_dim,
        stride_ab,
        stride_ai,
        stride_ak,
        stride_ad,
        stride_bb,
        stride_bi,
        stride_bk,
        stride_bd,
        stride_cb,
        stride_ci,
        stride_cj,
        stride_cd,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )

    # ---------- Post‑processing ----------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate  # element‑wise gate

    to_out_weight = weights["to_out.weight"]
    output = F.linear(out, to_out_weight)  # final projection to `dim`

    return output
