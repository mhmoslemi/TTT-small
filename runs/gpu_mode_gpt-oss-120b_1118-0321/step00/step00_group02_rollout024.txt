"""
TriMul "outgoing" forward pass with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor (shape [B,N,N,C]).
2. Apply four linear projections (left, right, left_gate, right_gate, out_gate) using the
   supplied weight matrices.  Gates are passed through a sigmoid.
3. Fuse the element‑wise mask and gate multiplications in a Triton kernel:
       left   = left_proj   * left_gate   * mask
       right  = right_proj  * right_gate  * mask
   (mask is broadcast to the hidden dimension).
   The kernel works on a flattened view of the tensors and therefore
   works for any batch‑size / sequence‑length / hidden‑dim configuration.
4. The core “TriMul” operation is a batched matrix‑multiply over the
   sequence dimension:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is equivalent to a per‑channel matrix‑multiplication:
       out_{b,d} = left_{b,d} @ right_{b,d}ᵀ
   This is performed with a single torch.bmm call after reshaping the
   tensors to (B*D, N, N).
5. Apply a second LayerNorm over the hidden dimension, multiply by the
   out‑gate, and project back to the original channel dimension.
6. Return the final tensor of shape [B,N,N,C].

Only the element‑wise mask‑gate fusion is executed on the GPU via Triton,
fulfilling the requirement of using a custom kernel while keeping the
overall implementation simple and efficient.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# ----------------------------------------------------------------------
# Triton kernel: fused mask * gate * input (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_kernel(
    inp_ptr,          # const float*  input tensor (proj)
    gate_ptr,         # const float*  gate tensor (sigmoid output)
    mask_ptr,         # const float*  mask tensor broadcasted to hidden dim
    out_ptr,          # float*        output tensor (can be same as inp)
    n_elements,       # i32           total number of elements
    BLOCK_SIZE: tl.constexpr,   # compile‑time block size (threads per program)
):
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE
    idx = offset + tl.arange(0, BLOCK_SIZE)

    # bounds check
    mask = idx < n_elements

    # load values (masked loads return 0.0 for out‑of‑bounds)
    inp = tl.load(inp_ptr + idx, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + idx, mask=mask, other=0.0)
    m = tl.load(mask_ptr + idx, mask=mask, other=1.0)   # default mask = 1.0 (no masking)

    out = inp * gate * m
    tl.store(out_ptr + idx, out, mask=mask)


def custom_kernel(data):
    """
    Custom forward kernel for the TriMul "outgoing" operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [batch, seq_len, seq_len, dim]
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]            # input/output channel dimension C
    hidden_dim = config["hidden_dim"]  # hidden dimension used internally
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Grab weight tensors
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # Helper: linear projection without bias: out = x @ W^T
    # ------------------------------------------------------------------
    def linear_nobias(x, w):
        # x: (..., dim_in) , w: (dim_out, dim_in)
        return torch.matmul(x, w.t())

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension of the input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections and gate pre‑activations
    # ------------------------------------------------------------------
    left_proj = linear_nobias(x, left_proj_weight)          # [B,N,N,hidden_dim]
    right_proj = linear_nobias(x, right_proj_weight)        # [B,N,N,hidden_dim]

    left_gate_raw = linear_nobias(x, left_gate_weight)
    right_gate_raw = linear_nobias(x, right_gate_weight)
    out_gate_raw = linear_nobias(x, out_gate_weight)

    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # ------------------------------------------------------------------
    # 3. Prepare mask (broadcast to hidden_dim) and apply through Triton
    # ------------------------------------------------------------------
    if mask_tensor is None or nomask:
        mask = torch.ones(
            (input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]),
            dtype=dtype,
            device=device,
        )
    else:
        mask = mask_tensor.to(dtype)

    # broadcast mask to hidden dimension
    mask = mask.unsqueeze(-1)                 # [B,N,N,1]
    mask = mask.expand(-1, -1, -1, hidden_dim)  # [B,N,N,hidden_dim]
    mask = mask.contiguous()

    # flatten tensors for the kernel
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask = mask.contiguous()

    total_elements = left_proj.numel()   # B * N * N * hidden_dim
    BLOCK_SIZE = 16384                  # tunable; power‑of‑two works well
    num_blocks = (total_elements + BLOCK_SIZE - 1) // BLOCK_SIZE

    # ------------------------------------------------------------------
    # Triton launch: left = left_proj * left_gate * mask   (in‑place)
    # ------------------------------------------------------------------
    mask_gate_kernel[(num_blocks,)](
        left_proj.data_ptr(),
        left_gate.data_ptr(),
        mask.data_ptr(),
        left_proj.data_ptr(),          # output overwrites left_proj
        total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )
    # right = right_proj * right_gate * mask   (in‑place)
    mask_gate_kernel[(num_blocks,)](
        right_proj.data_ptr(),
        right_gate.data_ptr(),
        mask.data_ptr(),
        right_proj.data_ptr(),         # output overwrites right_proj
        total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 4. Core TriMul operation: batched matrix multiplication per hidden channel
    # ------------------------------------------------------------------
    B, N, _, _ = left_proj.shape

    # reshape to (B*hidden_dim, N, N) for torch.bmm
    left_mat = left_proj.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right_proj.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # right needs transposition on the last two axes
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N]

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Second LayerNorm over hidden dimension, apply out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate                           # element‑wise gating

    # final linear projection back to dim
    out = torch.matmul(out.view(-1, hidden_dim), to_out_weight.t())
    out = out.view(B, N, N, dim)

    return out
