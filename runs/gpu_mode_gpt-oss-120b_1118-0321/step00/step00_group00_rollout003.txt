@triton.jit
def layernorm_kernel(x_ptr, out_ptr, weight_ptr, bias_ptr,
                     B, N, dim,
                     stride_xb, stride_xn1, stride_xn2, stride_xd,
                     stride_outb, stride_outn1, stride_outn2, stride_outd,
                     BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    total = B * N * N
    # compute b,i,j coordinates
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # compute base offset for this (b,i,j)
    offset = b * stride_xb + i * stride_xn1 + j * stride_xn2
    # offsets for each dim
    offs_d = tl.arange(0, BLOCK_D)
    # will iterate over dim in steps of BLOCK_D
    sum_val = tl.zeros([1], dtype=tl.float32)
    sum_sq = tl.zeros([1], dtype=tl.float32)
    # compute sum and sum_sq
    for start in range(0, dim, BLOCK_D):
        cur_offs = offset + (start + offs_d) * stride_xd
        x = tl.load(x_ptr + cur_offs, mask=(start+offs_d)<dim, other=0.0)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)
    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    # compute rstd = rsqrt(var + eps)
    eps = 1e-5
    rstd = tl.rsqrt(var + eps)
    # Now load again and write normalized values
    # but we need weight and bias broadcast across positions.
    weight = tl.load(weight_ptr + offs_d, mask=(offs_d)<dim, other=0.0)
    bias = tl.load(bias_ptr + offs_d, mask=(offs_d)<dim, other=0.0)

    for start in range(0, dim, BLOCK_D):
        cur_offs = offset + (start + offs_d) * stride_xd
        x = tl.load(x_ptr + cur_offs, mask=(start+offs_d)<dim, other=0.0)
        y = (x - mean) * rstd
        y = y * weight + bias
        tl.store(out_ptr + cur_offs, y, mask=(start+offs_d)<dim)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def layernorm_kernel(
    x_ptr, out_ptr, weight_ptr, bias_ptr,
    B, N, D,
    stride_xb, stride_xn1, stride_xn2, stride_xd,
    stride_ob, stride_on1, stride_on2, stride_od,
    BLOCK_D: tl.constexpr,
    EPS: tl.constexpr = 1e-5
):
    pid = tl.program_id(0)
    # total positions = B*N*N
    total = B * N * N
    # map pid to (b,i,j)
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # compute base offsets for x and out
    base_x = b * stride_xb + i * stride_xn1 + j * stride_xn2
    base_o = b * stride_ob + i * stride_on1 + j * stride_on2

    # compute number of blocks over D
    num_blocks = tl.cdiv(D, BLOCK_D)

    # first pass: compute mean and variance
    sum_val = tl.zeros([1], dtype=tl.float32)
    sum_sq  = tl.zeros([1], dtype=tl.float32)
    for blk in range(num_blocks):
        d_start = blk * BLOCK_D
        offs_d = d_start + tl.arange(0, BLOCK_D)
        mask = offs_d < D
        x = tl.load(x_ptr + base_x + offs_d * stride_xd, mask=mask, other=0.0)
        sum_val += tl.sum(x, axis=0)
        sum_sq  += tl.sum(x * x, axis=0)

    mean = sum_val / D
    var = sum_sq / D - mean * mean
    rstd = tl.rsqrt(var + EPS)

    # load weight and bias once per block
    for blk in range(num_blocks):
        d_start = blk * BLOCK_D
        offs_d = d_start + tl.arange(0, BLOCK_D)
        mask = offs_d < D
        x = tl.load(x_ptr + base_x + offs_d * stride_xd, mask=mask, other=0.0)
        w = tl.load(weight_ptr + offs_d, mask=mask, other=0.0)
        b_ = tl.load(bias_ptr + offs_d, mask=mask, other=0.0)

        y = (x - mean) * rstd
        y = y * w + b_
        tl.store(out_ptr + base_o + offs_d * stride_od, y, mask=mask)


def layernorm_triton(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor) -> torch.Tensor:
    """
    Apply LayerNorm (with affine weight/bias) over the last dimension of ``x`` using a
    Triton kernel. ``x`` is expected to be contiguous and of shape (B, N, N, D).
    """
    assert x.is_cuda and weight.is_cuda and bias.is_cuda
    B, N, N2, D = x.shape
    assert N == N2
    out = torch.empty_like(x)

    # stride vectors (int64)
    stride_xb, stride_xn1, stride_xn2, stride_xd = x.stride()
    stride_ob, stride_on1, stride_on2, stride_od = out.stride()

    # launch kernel: one program per (b,i,j) position
    total = B * N * N
    grid = (total,)

    BLOCK_D = 64  # can be tuned, currently works for D up to 384

    layernorm_kernel[grid](
        x_ptr=x,
        out_ptr=out,
        weight_ptr=weight,
        bias_ptr=bias,
        B=B,
        N=N,
        D=D,
        stride_xb=stride_xb,
        stride_xn1=stride_xn1,
        stride_xn2=stride_xn2,
        stride_xd=stride_xd,
        stride_ob=stride_ob,
        stride_on1=stride_on1,
        stride_on2=stride_on2,
        stride_od=stride_od,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    return out

"""
TriMul (outgoing) forward pass with a Triton‑accelerated LayerNorm.

Algorithm
---------
1. **LayerNorm (last dimension)** – implemented with a Triton kernel that
   computes mean/variance per (batch, i, j) position and applies the learned
   affine parameters.
2. **Linear projections** – left/right projections and three gates are
   evaluated with `torch.nn.functional.linear`.  The gates are passed through
   `sigmoid`.
3. **Mask** – optional element‑wise multiplication with a (B,N,N) mask.
4. **Gated tensors** – left = left * left_gate, right = right * right_gate.
5. **Bilinear aggregation** – for each hidden channel `h`,
   `out[..., :, :, h] = left[..., :, :, h] @ right[..., :, :, h].T`.
   This is performed efficiently as a batched matrix multiplication using
   `torch.bmm`, after reshaping to (B*H, N, N).
6. **Output LayerNorm + gating** – layer‑norm over the hidden dimension,
   multiplied by `out_gate`.
7. **Final projection** – linear mapping back to the original `dim`.

Only the LayerNorm step is written in Triton; the remaining operations use
high‑performance PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_kernel(
    x_ptr, out_ptr, weight_ptr, bias_ptr,
    B, N, D,
    stride_xb, stride_xn1, stride_xn2, stride_xd,
    stride_ob, stride_on1, stride_on2, stride_od,
    BLOCK_D: tl.constexpr,
    EPS: tl.constexpr = 1e-5,
):
    """LayerNorm over the last dimension of a (B,N,N,D) tensor."""
    pid = tl.program_id(0)                     # one program per (b,i,j)
    # decode pid -> (b,i,j)
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # base pointers for this position
    base_x = b * stride_xb + i * stride_xn1 + j * stride_xn2
    base_o = b * stride_ob + i * stride_on1 + j * stride_on2

    # number of blocks needed to cover D dimensions
    num_blocks = tl.cdiv(D, BLOCK_D)

    # -------- first pass: compute mean & variance ----------
    sum_val = tl.zeros([1], dtype=tl.float32)
    sum_sq = tl.zeros([1], dtype=tl.float32)

    for blk in range(num_blocks):
        d_start = blk * BLOCK_D
        offs_d = d_start + tl.arange(0, BLOCK_D)
        mask = offs_d < D
        x = tl.load(x_ptr + base_x + offs_d * stride_xd,
                    mask=mask, other=0.0)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)

    mean = sum_val / D
    var = sum_sq / D - mean * mean
    rstd = tl.rsqrt(var + EPS)

    # -------- second pass: write normalized output ----------
    for blk in range(num_blocks):
        d_start = blk * BLOCK_D
        offs_d = d_start + tl.arange(0, BLOCK_D)
        mask = offs_d < D

        x = tl.load(x_ptr + base_x + offs_d * stride_xd,
                    mask=mask, other=0.0)
        w = tl.load(weight_ptr + offs_d, mask=mask, other=0.0)
        b_ = tl.load(bias_ptr + offs_d, mask=mask, other=0.0)

        y = (x - mean) * rstd
        y = y * w + b_
        tl.store(out_ptr + base_o + offs_d * stride_od,
                 y, mask=mask)


def layernorm_triton(x: torch.Tensor, weight: torch.Tensor,
                     bias: torch.Tensor) -> torch.Tensor:
    """Apply LayerNorm over the last dimension of ``x`` using Triton."""
    if not x.is_cuda:
        raise RuntimeError("layernorm_triton expects CUDA tensors")
    B, N, N2, D = x.shape
    assert N == N2, "Input must be square in the sequence dimensions"
    out = torch.empty_like(x)

    # Strides (in elements, not bytes)
    sx_b, sx_i, sx_j, sx_d = x.stride()
    so_b, so_i, so_j, so_d = out.stride()

    total = B * N * N
    grid = (total,)

    # BLOCK_D should be a divisor of typical D (<=384). 64 works for all cases.
    BLOCK_D = 64

    layernorm_kernel[grid](
        x_ptr=x,
        out_ptr=out,
        weight_ptr=weight,
        bias_ptr=bias,
        B=B,
        N=N,
        D=D,
        stride_xb=sx_b,
        stride_xn1=sx_i,
        stride_xn2=sx_j,
        stride_xd=sx_d,
        stride_ob=so_b,
        stride_on1=so_i,
        stride_on2=so_j,
        stride_od=so_d,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    return out
