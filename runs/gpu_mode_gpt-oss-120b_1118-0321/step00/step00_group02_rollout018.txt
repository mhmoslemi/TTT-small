def triton_matmul_T(a, b, BLOCK_M=64, BLOCK_N=64, BLOCK_K=64):
    assert a.is_contiguous()
    assert b.is_contiguous()
    B, M, K = a.shape
    _, Kb, N = b.shape
    assert K == Kb
    c = torch.empty((B, M, N), dtype=a.dtype, device=a.device)
    grid = (tl.cdiv(M, BLOCK_M), tl.cdiv(N, BLOCK_N), B)
    matmul_T_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=M,
        N=N,
        K=K,
        stride_a_batch=a.stride(0),
        stride_a_m=a.stride(1),

"""
TriMul (outgoing) forward implementation with a custom Triton kernel.

The heavy part of the TriMul forward is the tensor contraction
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
which is a batched matrix multiplication of shape (B*H, N, N) with the
second operand transposed.  This kernel computes C = A @ Bᵀ for each
(batch * hidden) pair using a classic GEMM tiling strategy and
stores the result back in FP32.  All other operations (LayerNorm,
linear projections, sigmoid gates, mask handling) are performed with
PyTorch for simplicity.

The kernel works for any batch size (<=2), sequence length up to 1024,
and hidden dimension up to 384.  It handles optional masks and works
with both float32 and float16 inputs (accumulating in float32).
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# ----------------------------------------------------------------------
# Triton GEMM kernel (C = A @ Bᵀ) where:
#   A : (B*H, M, K)   (M == N == K == seq_len)
#   B : (B*H, K, N)   (the transposed right tensor)
#   C : (B*H, M, N)
# ----------------------------------------------------------------------
@triton.jit
def _matmul_transpose_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)          # batch*hidden index
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block column

    # ------------------------------------------------------------------
    # Offsets within the block
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the A and B tiles for this block
    a_ptrs = (
        a_ptr
        + pid_b * stride_a_batch
        + offs_m[:, None] * stride_a_m
        + tl.arange(0, BLOCK_K)[None, :] * stride_a_k
    )
    b_ptrs = (
        b_ptr
        + pid_b * stride_b_batch
        + tl.arange(0, BLOCK_K)[:, None] * stride_b_k
        + offs_n[None, :] * stride_b_n
    )

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main reduction over K dimension
    # ------------------------------------------------------------------
    for _k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load tiles with bounds checking
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (tl.arange(0, BLOCK_K) < K)[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=(tl.arange(0, BLOCK_K) < K)[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Accumulate
        acc += tl.dot(a, b)

        # Advance to the next K‑tile
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # ------------------------------------------------------------------
    # Write the result back
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_b * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_transpose(a: torch.Tensor, b: torch.Tensor,
                              BLOCK_M=64, BLOCK_N=64, BLOCK_K=64) -> torch.Tensor:
    """
    Wrapper for the Triton kernel above.
    a : (B*H, M, K)  contiguous
    b : (B*H, K, N)  contiguous (already transposed)
    Returns c : (B*H, M, N)
    """
    assert a.is_contiguous() and b.is_contiguous()
    B_batch, M, K = a.shape
    _, Kb, N = b.shape
    assert K == Kb, "Inner dimensions must match"

    c = torch.empty((B_batch, M, N), dtype=a.dtype, device=a.device)

    grid = (
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B_batch,
    )
    _matmul_transpose_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=M,
        N=N,
        K=K,
        stride_a_batch=a.stride(0),
        stride_a_m=a.stride(1),
        stride_a_k=a.stride(2),
        stride_b_batch=b.stride(0),
        stride_b_k=b.stride(1),
        stride_b_n=b.stride(2),
        stride_c_batch=c.stride(0),
        stride_c_m=c.stride(1),
        stride_c_n=c.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return c


# ----------------------------------------------------------------------
# Main entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [batch, seq_len, seq_len, dim]
    """
    input_tensor, mask, weights, config = data

    # ------------------------------------------------------------------
    # Configuration and weight extraction
    # ------------------------------------------------------------------
    device = input_tensor.device
    dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm parameters
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    # Linear weights (no bias)
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # Optional masking handling
    # ------------------------------------------------------------------
    if mask is None or config.get("nomask", False):
        mask = torch.ones_like(input_tensor[..., 0], dtype=dtype, device=device)
    else:
        # Ensure mask is float for multiplication
        mask = mask.to(dtype)

    mask = mask.unsqueeze(-1)                     # [B, N, N, 1]

    # ------------------------------------------------------------------
    # Normalization and linear projections
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    left = F.linear(x, left_proj_weight)          # -> [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # Apply mask to the projected tensors
    left = left * mask
    right = right * mask

    # ------------------------------------------------------------------
    # Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Efficient Einstein summation via Triton (left @ rightᵀ)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # batch, seq_len, seq_len, hidden_dim

    # Move hidden dimension to the batch axis: (B, H, N, N) -> (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Transpose the second operand so the kernel sees (K, N)
    right_flat_T = right_flat.transpose(1, 2).contiguous()

    # Triton‑accelerated batched matrix multiplication
    out_flat = _batched_matmul_transpose(
        left_flat,
        right_flat_T,
        BLOCK_M=64,
        BLOCK_N=64,
        BLOCK_K=64,
    )  # shape (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Post‑processing (norm, out‑gate, final linear)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)           # -> [B, N, N, dim]

    return out
