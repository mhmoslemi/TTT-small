def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = x.device
    B, N, _, _ = x.shape  # N is seq_len

    # LayerNorm
    x_norm = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Linear projections
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # if mask is None or config['nomask']: create ones mask
    if mask is None or config.get("nomask", False):
        mask_tensor = torch.ones(B, N, N, dtype=x.dtype, device=device)
    else:
        mask_tensor = mask.to(x.dtype)

    # Gates
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # Fuse mask and gate using Triton kernel (if mask exists)
    # Ensure leaf tensors contiguous.
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_tensor = mask_tensor.contiguous()

    # launch kernel
    total = B * N * N * hidden_dim
    BLOCK = 128
    grid = lambda meta: (triton.cdiv(total, BLOCK),)
    fuse_mask_gate_kernel[grid](
        left, right, mask_tensor, left_gate, right_gate,
        B, N, hidden_dim,
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        mask_tensor.stride(0), mask_tensor.stride(1), mask_tensor.stride(2),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
    )

    # core einsum via batched matmul
    # reshape to (B*hidden_dim, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # matrix multiplication left @ right^T
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    # reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # to_out_norm
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])

    # multiply by out gate
    out = out * out_gate

    # final linear to dim
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
TriMul forward (outgoing) implementation.

Algorithm
---------
1. Layer‑norm the input tensor over the last dimension (dim).
2. Project the normalized tensor to a hidden dimension H with two linear layers:
   left = x_norm @ W_left,   right = x_norm @ W_right.
3. Compute three sigmoid‑gates (left, right, out) from the same normalized tensor.
4. Fuse mask (if present) and the two gates into the projected tensors using a
   Triton kernel (single pass over the [B,N,N,H] memory).
5. Compute the core einsum
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is equivalent to a batched matrix multiplication:
      out_h = left_h @ right_hᵀ   for each hidden channel h.
   This is performed with torch.bmm on tensors reshaped to (B*H, N, N).
6. Apply a second LayerNorm over the hidden dimension, multiply by the out‑gate,
   and finally linearly project back to the original dimension.

The Triton kernel only handles step 4, satisfying the requirement that part
of the computation be performed in custom Triton code while the bulk of the
heavy O(N³) work uses highly‑optimized Torch batched matmul.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: fuse mask and the two elementwise gates into left/right.
# --------------------------------------------------------------------------- #
@triton.jit
def _fuse_mask_gate_kernel(
    L_ptr, R_ptr, M_ptr, LG_ptr, RG_ptr,
    B, N, H,
    stride_Lb, stride_Ln1, stride_Ln2, stride_Lh,
    stride_Rb, stride_Rn1, stride_Rn2, stride_Rh,
    stride_Mb, stride_Mn1, stride_Mn2,
    stride_LGb, stride_LGn1, stride_LGn2, stride_LGh,
    stride_RGb, stride_RGn1, stride_RGn2, stride_RGh,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    total = B * N * N * H
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < total

    # Linear index -> (b,i,j,h)
    b = offs // (N * N * H)
    rem = offs % (N * N * H)
    i = rem // (N * H)
    rem2 = rem % (N * H)
    j = rem2 // H
    h = rem2 % H

    # Compute pointer offsets
    L_off = b * stride_Lb + i * stride_Ln1 + j * stride_Ln2 + h * stride_Lh
    R_off = b * stride_Rb + i * stride_Rn1 + j * stride_Rn2 + h * stride_Rh
    M_off = b * stride_Mb + i * stride_Mn1 + j * stride_Mn2                # mask has no H dim
    LG_off = b * stride_LGb + i * stride_LGn1 + j * stride_LGn2 + h * stride_LGh
    RG_off = b * stride_RGb + i * stride_RGn1 + j * stride_RGn2 + h * stride_RGh

    # Load values (use 0.0 for missing entries, 1.0 for missing mask)
    L_val = tl.load(L_ptr + L_off, mask=mask, other=0.0)
    R_val = tl.load(R_ptr + R_off, mask=mask, other=0.0)
    M_val = tl.load(M_ptr + M_off, mask=mask, other=1.0)   # default: no masking
    LG_val = tl.load(LG_ptr + LG_off, mask=mask, other=0.0)
    RG_val = tl.load(RG_ptr + RG_off, mask=mask, other=0.0)

    # Fuse mask and gates
    L_val = L_val * M_val * LG_val
    R_val = R_val * M_val * RG_val

    # Store results back in‑place
    tl.store(L_ptr + L_off, L_val, mask=mask)
    tl.store(R_ptr + R_off, R_val, mask=mask)


# --------------------------------------------------------------------------- #
# Custom kernel entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) operator.

    Args:
        data: Tuple (input, mask, weights, config)
            - input: torch.Tensor of shape [B, N, N, dim]
            - mask:  torch.Tensor of shape [B, N, N] (may be None)
            - weights: dict of model parameters (tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------- #
    # Unpack inputs
    # ------------------------------------------------------------------- #
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = x.device
    B, N, _, _ = x.shape

    # ------------------------------------------------------------------- #
    # 1. LayerNorm over the feature dimension
    # ------------------------------------------------------------------- #
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=weights.get("norm.weight"),
        bias=weights.get("norm.bias"),
    )

    # ------------------------------------------------------------------- #
    # 2. Linear projections (no bias) to hidden dimension
    # ------------------------------------------------------------------- #
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3. Prepare mask (all‑ones if none / nomask)
    # ------------------------------------------------------------------- #
    if config.get("nomask", False) or mask is None:
        mask_tensor = torch.ones(B, N, N, dtype=x.dtype, device=device)
    else:
        mask_tensor = mask.to(dtype=x.dtype, device=device)

    # ------------------------------------------------------------------- #
    # 4. Compute sigmoid gates
    # ------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # ------------------------------------------------------------------- #
    # 5. Fuse mask and the two gates into the projected tensors via Triton
    # ------------------------------------------------------------------- #
    # Ensure contiguous layout for all tensors
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_tensor = mask_tensor.contiguous()

    BLOCK_SIZE = 128
    total_elems = B * N * N * hidden_dim
    grid = (triton.cdiv(total_elems, BLOCK_SIZE),)

    _fuse_mask_gate_kernel[grid](
        left,
        right,
        mask_tensor,
        left_gate,
        right_gate,
        B,
        N,
        hidden_dim,
        left.stride(0),
        left.stride(1),
        left.stride(2),
        left.stride(3),
        right.stride(0),
        right.stride(1),
        right.stride(2),
        right.stride(3),
        mask_tensor.stride(0),
        mask_tensor.stride(1),
        mask_tensor.stride(2),
        left_gate.stride(0),
        left_gate.stride(1),
        left_gate.stride(2),
        left_gate.stride(3),
        right_gate.stride(0),
        right_gate.stride(1),
        right_gate.stride(2),
        right_gate.stride(3),
        BLOCK_SIZE,
    )

    # ------------------------------------------------------------------- #
    # 6. Core einsum: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched matmul per hidden channel.
    # ------------------------------------------------------------------- #
    # Reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched matrix multiplication with transposed right matrix
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 7. Second LayerNorm over hidden dimension
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights.get("to_out_norm.weight"),
        bias=weights.get("to_out_norm.bias"),
    )

    # ------------------------------------------------------------------- #
    # 8. Apply output gate and final linear projection back to dim
    # ------------------------------------------------------------------- #
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
