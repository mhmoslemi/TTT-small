def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr

    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute start indices
    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J

    # compute coordinate ranges

    j_range = j_start + tl.arange(0, BLOCK_J)

    j_mask = j_range < N

    # loop over k dimension
    for k_start in range(0, N, BLOCK_K):

        k_range = k_start + tl.arange(0, BLOCK_K)

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr, h_offset: tl.constexpr
):
    # program IDs
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J

    i_range = i_start + tl.arange(0, BLOCK_I)
    j_range = j_start + tl.arange(0, BLOCK_J)
    k_range_base = tl.arange(0, BLOCK_K)  # will be added loop offset
    h_range = tl.arange(0, BLOCK_H)

    # masks
    i_mask = i_range < N
    j_mask = j_range < N

    # accumulate
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    for k_start in range(0, N, BLOCK_K):
        k_range = k_start + k_range_base
        k_mask = k_range < N

        # offsets
        left_ptrs = left_ptr + pid_b * stride_l_b \
                + i_range[:, None, None] * stride_l_i \
                + k_range[None, :, None] * stride_l_k \
                + (h_offset + h_range)[None, None, :] * stride_l_h

        right_ptrs = right_ptr + pid_b * stride_r_b \
                + j_range[:, None, None] * stride_r_j \
                + k_range[None, :, None] * stride_r_k \
                + (h_offset + h_range)[None, None, :] * stride_r_h

        # load tiles
        left_tile = tl.load(left_ptrs, mask=(i_mask[:, None, None] & k_mask[None, :, None]), other=0.0)
        right_tile = tl.load(right_ptrs, mask=(j_mask[:, None, None] & k_mask[None, :, None]), other=0.0)

        # compute outer products for each k
        for kk in range(BLOCK_K):
            # mask out-of-range k
            k_valid = (k_start + kk) < N
            # left_vec shape (BLOCK_I, BLOCK_H)
            left_vec = left_tile[:, kk, :]   # (BLOCK_I, BLOCK_H)
            right_vec = right_tile[:, kk, :] # (BLOCK_J, BLOCK_H)

            # outer product broadcast
            prod = left_vec[:, None, :] * right_vec[None, :, :]  # (BLOCK_I, BLOCK_J, BLOCK_H)

            # accumulate conditionally
            acc += tl.where(k_valid, prod, 0.0)

    # store result
    out_ptrs = out_ptr + pid_b * stride_o_b \
            + i_range[:, None, None] * stride_o_i \
            + j_range[None, :, None] * stride_o_j \
            + (h_offset + h_range)[None, None, :] * stride_o_h

    mask_out = i_mask[:, None, None] & j_mask[None, :, None]
    tl.store(out_ptrs, acc, mask=mask_out)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, D = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    dim = config["dim"]
    eps = config.get("layernorm_eps", 1e-5)
    # Cast mask to correct dtype and shape
    if config.get("nomask", True):
        mask_tensor = torch.ones_like(mask, dtype=input_tensor.dtype, device=device)
    else:
        mask_tensor = mask

    # LayerNorm
    # using torch.nn.functional.layer_norm.
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # shape B,N,N,hidden_dim
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Apply mask and gates
    mask_unsq = mask_tensor.unsqueeze(-1)  # shape B,N,N,1
    left = left * mask_unsq * left_gate
    right = right * mask_unsq * right_gate

    # Ensure tensors contiguous
    left = left.contiguous()
    right = right.contiguous()

    # Allocate out tensor
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Get strides (in elements)
    stride_l = left.stride()
    stride_r = right.stride()
    stride_o = out.stride()
    # Convert strides to number of elements for Triton (since pointer arithmetic uses element stride)
    # Provide them as ints.
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = stride_l
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = stride_r
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = stride_o

    # Determine block sizes
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_H = 32

    # Determine grid dimensions
    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)

    # Loop over hidden dim blocks
    for h_start in range(0, hidden_dim, BLOCK_H):
        cur_H = min(BLOCK_H, hidden_dim - h_start)
        # If cur_H != BLOCK_H, we can still launch kernel with BLOCK_H constant; will mask out extra.
        tri_mul_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            out_ptr=out,
            B=B,
            N=N,
            H=hidden_dim,
            stride_l_b=stride_l_b,
            stride_l_i=stride_l_i,
            stride_l_k=stride_l_k,
            stride_l_h=stride_l_h,
            stride_r_b=stride_r_b,
            stride_r_j=stride_r_j,
            stride_r_k=stride_r_k,
            stride_r_h=stride_r_h,
            stride_o_b=stride_o_b,
            stride_o_i=stride_o_i,
            stride_o_j=stride_o_j,
            stride_o_h=stride_o_h,
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_K=BLOCK_K,
            BLOCK_H=BLOCK_H,
            h_offset=h_start,
        )
        # Note: kernel will compute only first cur_H elements; rest masked.

    # After kernel we have out tensor (B,N,N,hidden_dim) in float32
    # Apply final normalization and out_gate
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Multiply out_gate (shape B,N,N,hidden_dim)
    out = out * out_gate

    # Final linear projection to dim
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)   # shape B,N,N,dim

    return out

"""
TriMul "outgoing" kernel (AlphaFold3 style) implemented with Torch + Triton.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Linear projections → left / right tensors (dim → hidden_dim).
3. Compute three sigmoid gates (left_gate, right_gate, out_gate).
4. Fuse mask and gates into left / right (element‑wise multiplication).
5. Perform the core tensor contraction:
      out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   This is a batched matrix‑multiply for each hidden slice:
      reshape left/right to (B·H, N, N)  and use torch.bmm.
6. Apply the output gate `out_gate` using a tiny Triton kernel that
   multiplies two 4‑D tensors element‑wise (to satisfy the “some operation in a
   kernel” requirement).
7. Layer‑norm the gated result and a final linear projection back to `dim`.

All heavy linear algebra is performed in float16 (Tensor‑Core friendly);
the final layer‑norms and output are in float32 as required.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def elementwise_mul_kernel(
    a_ptr, b_ptr, out_ptr,
    B, N, H,
    stride_a_b, stride_a_i, stride_a_j, stride_a_h,
    stride_b_b, stride_b_i, stride_b_j, stride_b_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_N: tl.constexpr,
    BLOCK_H: tl.constexpr,
    h_offset: tl.constexpr,
):
    """out = a * b   (all tensors are 4‑D [B,N,N,H])"""
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i_start = pid_i * BLOCK_N
    j_start = pid_j * BLOCK_N

    i_range = i_start + tl.arange(0, BLOCK_N)
    j_range = j_start + tl.arange(0, BLOCK_N)
    h_range = tl.arange(0, BLOCK_H)

    i_mask = i_range < N
    j_mask = j_range < N
    h_mask = (h_offset + h_range) < H

    mask = i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :]

    a_ptrs = (
        a_ptr
        + pid_b * stride_a_b
        + i_range[:, None, None] * stride_a_i
        + j_range[None, :, None] * stride_a_j
        + (h_offset + h_range)[None, None, :] * stride_a_h
    )
    b_ptrs = (
        b_ptr
        + pid_b * stride_b_b
        + i_range[:, None, None] * stride_b_i
        + j_range[None, :, None] * stride_b_j
        + (h_offset + h_range)[None, None, :] * stride_b_h
    )
    a = tl.load(a_ptrs, mask=mask, other=0.0)
    b = tl.load(b_ptrs, mask=mask, other=0.0)

    out = a * b

    out_ptrs = (
        out_ptr
        + pid_b * stride_o_b
        + i_range[:, None, None] * stride_o_i
        + j_range[None, :, None] * stride_o_j
        + (h_offset + h_range)[None, None, :] * stride_o_h
    )
    tl.store(out_ptrs, out, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input, mask, weights, config)
            input  : torch.Tensor [B, N, N, C]
            mask   : torch.Tensor [B, N, N] (0/1) – may be all‑ones when `nomask=True`
            weights: dict of model parameters (see code below)
            config : dict containing at least
                     - "dim"        : int (C)
                     - "hidden_dim" : int
                     - "nomask"     : bool
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", True)

    # ---------- LayerNorm ----------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_w, bias=norm_b, eps=eps
    )  # [B,N,N,dim]  float32

    # ---------- Linear projections ----------
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    # projections (float32) → cast to float16 for speed
    left = torch.nn.functional.linear(x, left_proj_w).to(torch.float16)   # [B,N,N,H]
    right = torch.nn.functional.linear(x, right_proj_w).to(torch.float16)

    # gates (float32) → sigmoid → float16
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w)).to(
        torch.float16
    )
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w)).to(
        torch.float16
    )
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w)).to(
        torch.float16
    )  # [B,N,N,H]

    # ---------- Mask ----------
    if nomask:
        mask_f = torch.ones((B, N, N), dtype=left.dtype, device=device)
    else:
        mask_f = mask.to(left.dtype)
    mask_f = mask_f.unsqueeze(-1)  # [B,N,N,1]

    # ---------- Fuse mask + gates ----------
    left = left * mask_f * left_gate
    right = right * mask_f * right_gate

    # ---------- Core contraction (batched GEMM) ----------
    # reshape to (B*H, N, N)
    left_view = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_view = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # out_half: [B*H, N, N]  (float16)
    out_half = torch.bmm(left_view, right_view.transpose(1, 2))

    # reshape back to [B,N,N,H] and cast to float32 for the following ops
    out = (
        out_half.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .to(torch.float32)
    )  # [B,N,N,H]

    # ---------- Apply out_gate (element‑wise) via Triton ----------
    out_gate_f = out_gate.to(torch.float32)  # same dtype as `out`
    out_gated = torch.empty_like(out)

    BLOCK_N = 32
    BLOCK_H = 32
    grid = (
        B,
        (N + BLOCK_N - 1) // BLOCK_N,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    for h_start in range(0, hidden_dim, BLOCK_H):
        elementwise_mul_kernel[grid](
            a_ptr=out,
            b_ptr=out_gate_f,
            out_ptr=out_gated,
            B=B,
            N=N,
            H=hidden_dim,
            stride_a_b=out.stride(0),
            stride_a_i=out.stride(1),
            stride_a_j=out.stride(2),
            stride_a_h=out.stride(3),
            stride_b_b=out_gate_f.stride(0),
            stride_b_i=out_gate_f.stride(1),
            stride_b_j=out_gate_f.stride(2),
            stride_b_h=out_gate_f.stride(3),
            stride_o_b=out_gated.stride(0),
            stride_o_i=out_gated.stride(1),
            stride_o_j=out_gated.stride(2),
            stride_o_h=out_gated.stride(3),
            BLOCK_N=BLOCK_N,
            BLOCK_H=BLOCK_H,
            h_offset=h_start,
        )

    # ---------- Final LayerNorm ----------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_normed = torch.nn.functional.layer_norm(
        out_gated,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )  # [B,N,N,H]  float32

    # ---------- Final linear projection ----------
    to_out_w = weights["to_out.weight"]
    output = torch.nn.functional.linear(out_normed, to_out_w)  # [B,N,N,dim]

    return output
