@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    pid_i = pid // num_j
    pid_j = pid % num_j

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Loop over batches
    for b in range(B):
        # Initialize accumulator per h block
        for h_start in range(0, H, BLOCK_H):
            # acc shape (BLOCK_I, BLOCK_J, BLOCK_H) -> use tl.zeros per hidden tile
            acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)
            # iterate k tiles
            for k_start in range(0, N, BLOCK_K):
                cur_k = k_start + tl.arange(0, BLOCK_K)
                mask_k = cur_k < N

                # load left tile shape (BLOCK_I, BLOCK_K, BLOCK_H)
                left_ptrs = left_ptr + b*stride_left_b + offs_i[:, None, None] * stride_left_i + cur_k[None, :, None] * stride_left_k + (h_start + tl.arange(0, BLOCK_H))[None, None, :] * stride_left_h
                left_tile = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & ( (h_start + tl.arange(0, BLOCK_H)) < H ), other=0.0).to(tl.float32)

                # load right tile shape (BLOCK_J, BLOCK_K, BLOCK_H)
                right_ptrs = right_ptr + b*stride_right_b + offs_j[:, None, None] * stride_right_i + cur_k[None, :, None] * stride_right_k + (h_start + tl.arange(0, BLOCK_H))[None, None, :] * stride_right_h
                right_tile = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None] & ( (h_start+tl.arange(0, BLOCK_H)) < H ), other=0.0).to(tl.float32)

                # compute for each h
                for hh in range(BLOCK_H):
                    # check if (h_start+hh)<H maybe handle mask.
                    # left_slice: (BLOCK_I, BLOCK_K)
                    left_slice = left_tile[:, :, hh]  # shape (I, K)
                    # right_slice: (BLOCK_J, BLOCK_K) need to transpose for dot
                    right_slice = right_tile[:, :, hh]  # shape (J, K)
                    # compute dot (I,J) = left_slice @ right_slice.T
                    c = tl.dot(left_slice, right_slice, trans_b=True)  # shape (I,J)
                    acc[:, :, hh] += c

            # Write out accumulator to global memory
            out_ptrs = out_ptr + b*stride_out_b + offs_i[:, None, None] * stride_out_i + offs_j[None, :, None] * stride_out_j + (h_start + tl.arange(0, BLOCK_H))[None, None, :] * stride_out_h
            # mask for out store
            mask_out = mask_i[:, None, None] & mask_j[None, :, None] & ((h_start + tl.arange(0, BLOCK_H)) < H)

```
import triton
import triton.language as tl

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)  # over i,j tiles
    b = tl.program_id(1)    # batch index
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    pid_i = pid // num_j
    pid_j = pid % num_j

    # coordinates
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for this batch
    left_batch_ptr = left_ptr + b * stride_left_b
    right_batch_ptr = right_ptr + b * stride_right_b
    out_batch_ptr = out_ptr + b * stride_out_b

    # loop over hidden dimension blocks
    for h_start in range(0, H, BLOCK_H):
        h_range = h_start + tl.arange(0, BLOCK_H)
        mask_h = h_range < H

        # accumulator per hidden block
        acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

        # loop over k dimension
        for k_start in range(0, N, BLOCK_K):
            k_range = k_start + tl.arange(0, BLOCK_K)
            mask_k = k_range < N

            # left tile load (I, K, H)
            left_ptrs = left_batch_ptr \
                + offs_i[:, None, None] * stride_left_i \
                + k_range[None, :, None] * stride_left_k \
                + h_range[None, None, :] * stride_left_h
            left = tl.load(left_ptrs,
                mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                other=0.0).to(tl.float32)

            # right tile load (J, K, H)
            right_ptrs = right_batch_ptr \
                + offs_j[:, None, None] * stride_right_i \
                + k_range[None, :, None] * stride_right_k \
                + h_range[None, None, :] * stride_right_h
            right = tl.load(right_ptrs,
                mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                other=0.0).to(tl.float32)

            # compute dot for each h in block
            for hh in range(BLOCK_H):
                # no need to mask inside as zero values already present for out-of-bound h
                # slice shapes (I,K) and (J,K)
                a = left[:, :, hh]
                b_ = right[:, :, hh]
                # dot (I,J)
                acc[:, :, hh] += tl.dot(a, b_, trans_b=True)

        # store result
        out_ptrs = out_batch_ptr \
            + offs_i[:, None, None] * stride_out_i \
            + offs_j[None, :, None] * stride_out_j \
            + h_range[None, None, :] * stride_out_h
        out_mask = mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :]
        tl.store(out_ptrs, acc, mask=out_mask)
```

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
    using a tiled Triton kernel. The kernel works on tiles of size
    (BLOCK_I, BLOCK_J) over the i/j dimensions and processes the hidden
    dimension in blocks of size BLOCK_H. The k dimension is iterated in
    blocks of size BLOCK_K.
    """
    pid = tl.program_id(0)      # tile over i and j
    b = tl.program_id(1)        # batch index

    # compute row/col tile indices
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    pid_i = pid // num_j
    pid_j = pid % num_j

    # thread-local coordinates inside the tile
    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = i < N
    mask_j = j < N

    for h_start in range(0, H, BLOCK_H):
        h = h_start + tl.arange(0, BLOCK_H)
        mask_h = h < H

        # accumulator for the hidden block
        acc = tl.zeros((BLOCK_H,), dtype=tl.float32)

        # loop over the k dimension
        for k_start in range(0, N, BLOCK_K):
            k = k_start + tl.arange(0, BLOCK_K)
            mask_k = k < N

            # pointers for the left and right tiles
            left_ptrs = left_ptr \
                + b * stride_left_b \
                + i[:, None, None] * stride_left_i \
                + k[None, :, None] * stride_left_k \
                + h[None, None, :] * stride_left_h

            right_ptrs = right_ptr \
                + b * stride_right_b \
                + j[:, None, None] * stride_right_i \
                + k[None, :, None] * stride_right_k \
                + h[None, None, :] * stride_right_h

            left = tl.load(left_ptrs,
                mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                other=0.0)
            right = tl.load(right_ptrs,
                mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                other=0.0)

            # accumulate product over k
            # left*right has shape (BLOCK_I, BLOCK_K, BLOCK_H); summing over K yields (BLOCK_I, BLOCK_H)
            # after indexing by the thread's i-coordinate we obtain a vector of length BLOCK_H.
            acc += tl.sum(left * right, axis=1)

        # write out the hidden block
        out_ptrs = out_ptr \
            + b * stride_out_b \
            + i[:, None, None] * stride_out_i \
            + j[None, :, None] * stride_out_j \
            + h[None, None, :] * stride_out_h

        out_mask = mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :]
        # each thread stores a vector of length BLOCK_H
        tl.store(out_ptrs, acc, mask=out_mask)

def _trimul_einsum_torch(left: torch.Tensor, right: torch.Tensor, hidden_dim: int):
    """
    Wrapper that launches the Triton kernel for the einsum
    out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h].
    """
    B, N, _, H = left.shape
    assert H == hidden_dim
    out = torch.empty_like(left)

    # strides (in elements) of the tensors
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    # block sizes – chosen to fit into registers on B200
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 64
    BLOCK_H = 8   # hidden dim is processed in chunks of 8

    grid = ( ( (N + BLOCK_I - 1) // BLOCK_I ) * ( (N + BLOCK_J - 1) // BLOCK_J ), B )
    trimul_outgoing_kernel[grid](
        left, right, out,
        B, N, H,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H,
        num_warps=4, num_stages=2
    )
    return out

def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.
    Arguments:
        data: tuple (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # -------------------- LayerNorm on the input --------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # -------------------- Linear projections --------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])    # [B,N,N,H]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -------------------- Optional mask --------------------
    if mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)      # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # -------------------- Gates --------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -------------------- Einstein sum (kernel) --------------------
    out = _trimul_einsum_torch(left, right, hidden_dim)

    # -------------------- Output normalisation and final linear --------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
Triton implementation of the “outgoing” TriMul operator (AlphaFold3).

The kernel computes:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

where `left` and `right` are the gated and masked projections of the input.
The kernel tiles the i/j dimensions (BLOCK_I×BLOCK_J) and processes the
hidden dimension in blocks of size BLOCK_H.  The k‐dimension is streamed
in blocks of size BLOCK_K, accumulating the per‑hidden‑dim dot product with
`tl.sum`.  The rest of the module (LayerNorms, linear projections, gating,
and final projection) is performed with PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a tiled implementation.  The kernel processes a tile of i/j
    (size BLOCK_I×BLOCK_J) and a hidden sub‑block of size BLOCK_H.
    """
    pid = tl.program_id(0)          # tile over (i,j)
    b = tl.program_id(1)            # batch index

    # tile coordinates
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    pid_i = pid // num_j
    pid_j = pid % num_j

    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = i < N
    mask_j = j < N

    # ------------------------------------------------------------------
    # loop over hidden dimension in blocks of BLOCK_H
    # ------------------------------------------------------------------
    for h_start in range(0, H, BLOCK_H):

            mask_k = k < N

            left = tl.load(
                left_ptrs,
                mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                other=0.0,

                right_ptrs,
                mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                other=0.0,
            )

            # accumulate over k → shape after sum is (BLOCK_H) for this thread

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Arguments:
        data: tuple (input_tensor, mask, weights, config)
              - input_tensor: [B, N, N, dim] (float32, CUDA)
              - mask:        [B, N, N]      (bool or float, same device)
              - weights: dict with all required Linear/LayerNorm parameters
              - config: dict with keys "dim" and "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # --------------------------------------------------------------
    # LayerNorm on the input
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # Linear projections
    # --------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # Optional mask (mask may be None → no masking)
    # --------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # Gating
    # --------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # Core tensor contraction – Triton kernel
    # --------------------------------------------------------------
    out = _trimul_einsum_torch(left, right, hidden_dim)   # [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # Output normalization, gating and final projection
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out
