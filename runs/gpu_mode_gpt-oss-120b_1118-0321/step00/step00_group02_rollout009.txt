"""
custom_kernel: Triton‑accelerated forward pass of the “outgoing” TriMul module.

Algorithm
---------
1. Layer‑norm the input tensor (last dimension).
2. Project the normalized tensor to a hidden dimension with two linear layers
   (left_proj, right_proj) and compute three gating scalars (left_gate,
   right_gate, out_gate) from the same normalized tensor.
3. Optionally mask the left/right tensors and apply the gates.
4. Compute the core O(N³·H) contraction
       out[i, j, h] = Σ_k left[i, k, h] * right[j, k, h]
   via a Triton batched GEMM where each (batch, hidden) pair is treated as
   an independent matrix‑multiply:  (N×N) × (N×N)ᵀ.
5. Layer‑norm the result, apply the output gate, and finally a linear
   projection back to the original dimension.

The heavy contraction (step 4) is performed by a custom Triton kernel that
processes tiles of the output matrix and accumulates using `tl.dot` with
`trans_b=True`.  All other operations are expressed with standard PyTorch
functions.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel: batched matrix multiply C = A @ Bᵀ
# ------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_ab, stride_am, stride_ak,
    stride_bb, stride_bn, stride_bk,
    stride_cb, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_b = tl.program_id(2)   # batch index (B * hidden)
    pid_m = tl.program_id(0)   # tile along M dimension (i)
    pid_n = tl.program_id(1)   # tile along N dimension (j)

    # ---- tile offsets ----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ---- base pointers for this batch ----
    a_batch = a_ptr + pid_b * stride_ab
    b_batch = b_ptr + pid_b * stride_bb
    c_batch = c_ptr + pid_b * stride_cb

    # ---- compute pointers for the current tile ----
    a_tile_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_tile_ptrs = b_batch + offs_n[:, None] * stride_bn + offs_k[None, :] * stride_bk

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K
        k_mask = (cur_k + offs_k) < K

        a = tl.load(a_tile_ptrs,
                    mask=mask_m[:, None] & k_mask[None, :],
                    other=0.0)
        b = tl.load(b_tile_ptrs,
                    mask=mask_n[:, None] & k_mask[None, :],
                    other=0.0)

        # tl.dot performs (M,K) @ (K,N); we need Bᵀ, so set trans_b=True
        acc += tl.dot(a, b, trans_b=True, out_dtype=tl.float32)

        # advance pointers to the next K‑tile
        a_tile_ptrs += BLOCK_K * stride_ak
        b_tile_ptrs += BLOCK_K * stride_bk

    # ---- write out the tile ----
    c_tile_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_tile_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix product C = A @ Bᵀ where A,B have shape
    (batch, M, K) and (batch, N, K) respectively.
    Returns C of shape (batch, M, N).
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, N, _ = B.shape

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    grid = (
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch,
    )

    _batched_matmul_kernel[grid](
        A, B, C,
        A.stride(0), A.stride(1), A.stride(2),   # batch, M, K strides for A
        B.stride(0), B.stride(1), B.stride(2),   # batch, N, K strides for B
        C.stride(0), C.stride(1), C.stride(2),   # batch, M, N strides for C
        M, N, K,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return C

    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections and gates (no bias in any Linear)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])      # [B, N, N, hidden]
    right = F.linear(x, weights["right_proj.weight"])    # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcasted on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # ensure mask dtype matches tensors
        mask = mask.to(input_tensor.dtype)
        mask = mask.unsqueeze(-1)           # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core contraction via Triton batched GEMM
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # reshape to (B*hidden, N, N) for the batched gemm
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous() # [B, hidden, N, N]

    batch_hidden = B * hidden_dim
    A = left_t.reshape(batch_hidden, N, N)
    Bmat = right_t.reshape(batch_hidden, N, N)

    # C = A @ Bᵀ  (shape: [batch_hidden, N, N])
    C = _batched_matmul(A, Bmat)

    # reshape back to [B, N, N, hidden]
    out = C.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output layer‑norm, output gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
```
