def trilu_mul_outgoing(left, right, BLOCK_M=64, BLOCK_N=64, BLOCK_K=64):
    B, N, _, H = left.shape
    # flatten batch*H
    BH = B * H
    # compute strides
    # Compute pointer offsets:
    stride_left_b = left.stride(0) # stride for batch dimension
    stride_left_i = left.stride(1) # stride for i dimension
    stride_left_k = left.stride(2) # stride for k dimension
    stride_left_h = left.stride(3) # stride for hidden dimension
    similarly for right and out.

    # we need to flatten b and h. striding across combined dimension:
    # pointer offset: b * stride_left_b + h * stride_left_h
    # In kernel we will compute both using b and h from bh_idx.
    # We can compute base_offset = (b * stride_left_b) + (h * stride_left_h)
    # Then within kernel for each i/j/k we add i_offset*stride_left_i + k_offset*stride_left_k

    # Grid sizes
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (BH * num_blocks_m * num_blocks_n, )  # flatten to 1D

    # kernel:
    @triton.jit
    def trimul_kernel(
        left_ptr, right_ptr, out_ptr,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        N, BH,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
        pid = tl.program_id(0)
        # compute BH index and block_i, block_j
        total_blocks = ( (N + BLOCK_M - 1)//BLOCK_M ) * ((N + BLOCK_N - 1)//BLOCK_N)
        bh_idx = pid // total_blocks
        block_idx = pid % total_blocks
        pid_m = block_idx // ((N + BLOCK_N - 1)//BLOCK_N)
        pid_n = block_idx % ((N + BLOCK_N - 1)//BLOCK_N)
        # compute start coordinates
        i0 = pid_m * BLOCK_M
        j0 = pid_n * BLOCK_N
        # compute block sizes
        i_size = tl.minimum(BLOCK_M, N - i0)
        j_size = tl.minimum(BLOCK_N, N - j0)
        # compute b, h from bh_idx
        b = bh_idx // H
        h = bh_idx % H
        # compute base offsets:
        left_base = left_ptr + b * stride_lb + h * stride_lh
        right_base = right_ptr + b * stride_rb + h * stride_rh
        out_base = out_ptr + b * stride_ob + h * stride_oh

        # create a 0-init accumulator
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # loop over k tiles
        for k0 in range(0, N, BLOCK_K):
            k_size = tl.minimum(BLOCK_K, N - k0)
            # i offsets for left
            i_range = tl.arange(0, BLOCK_M)[:i_size] + i0
            k_range = tl.arange(0, BLOCK_K)[:k_size] + k0
            j_range = tl.arange(0, BLOCK_N)[:j_size] + j0

            # pointers for left: shape (i_range, k_range)
            left_ptrs = left_base + i_range[:, None] * stride_li + k_range[None, :] * stride_lk
            left_block = tl.load(left_ptrs, mask=(i_range[:, None] < N) & (k_range[None, :] < N), other=0.0)

            # pointers for right: we need shape (k_range, j_range)
            # stride for rows (k) is stride_rk; for columns (j) stride is stride_rj
            right_ptrs = right_base + k_range[:, None] * stride_rk + j_range[None, :] * stride_rj

            # dot product

            # Use tl.dot for each row (i) and each column (j) (full block)
            acc = acc + tl.dot(left_block, right_block)

        # store acc to out
        out_ptrs = out_base + i_range[:, None] * stride_oi + j_range[None, :] * stride_oj
        tl.store(out_ptrs, acc, mask=(i_range[:, None] < N) & (j_range[None, :] < N))

def trimul_einsum(left, right):
    assert left.is_contiguous() and right.is_contiguous()
    B,N,N,H = left.shape
    out = torch.empty_like(left) # same shape
    # kernel call
    # compute stride values as ints
    stride_lb = left.stride(0)
    stride_li = left.stride(1)
    stride_lk = left.stride(2)
    stride_lh = left.stride(3)

    # for right
    stride_rb = right.stride(0)
    stride_rj = right.stride(1)
    stride_rk = right.stride(2)
    stride_rh = right.stride(3)

    # out same as left
    stride_ob = out.stride(0)
    stride_oi = out.stride(1)
    stride_oj = out.stride(2)
    stride_oh = out.stride(3)

    # compute grid
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1)//BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1)//BLOCK_N
    total_blocks = num_blocks_m * num_blocks_n
    bh = B * H
    grid = (bh * total_blocks,)

    trimul_kernel[grid](
        left, right, out,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        N, H,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    return out

"""
TriMul (outgoing) implementation using Triton for the heavy
pairwise einsum reduction.

The kernel computes, for each batch `b` and hidden channel `h`:

    out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]

which is equivalent to a batched matrix multiplication
L (N×N) @ Rᵀ (N×N) for every hidden channel.
All other operations (LayerNorm, projections, gating, final
linear) are performed with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------
# Triton kernel for the core einsum
# --------------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    left_ptr, right_ptr, out_ptr,           # pointers
    stride_lb, stride_li, stride_lk, stride_lh,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rh,   # right strides
    stride_ob, stride_oi, stride_oj, stride_oh,   # out strides
    N, H,                                    # sequence length, hidden dim
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # 1) decompose pid into (bh, block_i, block_j)
    # --------------------------------------------------------------
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_hb = num_blocks_m * num_blocks_n       # number of (i,j) tiles per (b,h)

    bh_idx = pid // blocks_per_hb                      # combined batch*hidden index
    block_id = pid % blocks_per_hb
    pid_m = block_id // num_blocks_n                  # tile row index
    pid_n = block_id % num_blocks_n                   # tile col index

    # recover original batch and hidden indices
    b = bh_idx // H
    h = bh_idx % H

    # --------------------------------------------------------------
    # 2) compute absolute offsets for this tile
    # --------------------------------------------------------------
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # coordinate vectors
    i_range = tl.arange(0, BLOCK_M) + i0
    j_range = tl.arange(0, BLOCK_N) + j0

    # base pointers for the selected (b,h)
    left_base  = left_ptr  + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base   = out_ptr   + b * stride_ob + h * stride_oh

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # 3) loop over k dimension in tiles
    # --------------------------------------------------------------
    for k0 in range(0, N, BLOCK_K):
        k_range = tl.arange(0, BLOCK_K) + k0

        # ------- load left tile (M x K) -------
        left_ptrs = left_base + i_range[:, None] * stride_li + k_range[None, :] * stride_lk
        left_mask = (i_range[:, None] < N) & (k_range[None, :] < N)
        left_tile = tl.load(left_ptrs, mask=left_mask, other=0.0)

        # ------- load right tile (K x N) -------
        # note: for the right matrix we need the (k, j) ordering,
        # i.e. rows correspond to k and columns to j.
        right_ptrs = right_base + k_range[:, None] * stride_rk + j_range[None, :] * stride_rj
        right_mask = (k_range[:, None] < N) & (j_range[None, :] < N)
        right_tile = tl.load(right_ptrs, mask=right_mask, other=0.0)

        # ------- block‐wise dot product -------
        acc += tl.dot(left_tile, right_tile)   # (M,K)·(K,N) -> (M,N)

    # --------------------------------------------------------------
    # 4) write the result
    # --------------------------------------------------------------
    out_ptrs = out_base + i_range[:, None] * stride_oi + j_range[None, :] * stride_oj
    out_mask = (i_range[:, None] < N) & (j_range[None, :] < N)
    tl.store(out_ptrs, acc, mask=out_mask)


def _trimul_einsum(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute ``out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]``
    using the Triton kernel defined above.
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    # ensure contiguous for pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)

    # extract strides (as Python ints)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_tiles = num_blocks_m * num_blocks_n
    total_programs = B * H * total_tiles

    grid = (total_programs,)

    _trimul_kernel[grid](
        left,
        right,
        out,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        N, H,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# --------------------------------------------------------------------
# Entry point required by the test harness
# --------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Expected ``data`` layout:
      (input_tensor, mask, weights, config)

    Returns:
      torch.Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # unpack arguments
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # ----------------------------------------------------------------
    # 1) LayerNorm on the input
    # ----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B,N,N,dim]

    # ----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ----------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ----------------------------------------------------------------
    # 3) Optional mask (applied before gating)
    # ----------------------------------------------------------------
    if not config.get("nomask", False) and mask is not None:
        mask = mask.to(dtype=left.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ----------------------------------------------------------------
    # 4) Gating (sigmoid on linear transforms of x)
    # ----------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------
    # 5) Core trigram multiplication via Triton
    # ----------------------------------------------------------------
    out = _trimul_einsum(left, right)       # [B,N,N,hidden_dim]

    # ----------------------------------------------------------------
    # 6) Output LayerNorm, gating and final linear projection
    # ----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate                     # elementwise gating
    out = F.linear(out, to_out_weight)       # final projection to `dim`

    return out
