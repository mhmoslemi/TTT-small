def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # Extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm on input
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections (no bias)
    left_proj = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right_proj = torch.nn.functional.linear(x, right_proj_weight)

    # mask handling
    if mask is None or config.get("nomask", True):
        mask_tensor = torch.ones(x.shape[:3], dtype=x.dtype, device=device)
    else:
        mask_tensor = mask.to(x.dtype)
    mask_exp = mask_tensor.unsqueeze(-1)    # [B,N,N,1]

    left_proj = left_proj * mask_exp
    right_proj = right_proj * mask_exp

    # Gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Cast to half for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Permute to bring hidden dim to second axis and flatten batch*hidden
    B, N, _, _ = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()
    batch = B * hidden_dim
    left_flat = left.view(batch, N, N)
    right_flat = right.view(batch, N, N)

    # Core TriMul via batched matmul kernel
    out_hidden = batched_matmul(left_flat, right_flat, BLOCK_M=64, BLOCK_N=64, BLOCK_K=64)

    # Reshape back
    out_hidden = out_hidden.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # Convert back to float32 for subsequent ops
    out_hidden = out_hidden.to(torch.float32)

    # to_out_norm
    out_norm = torch.nn.functional.layer_norm(out_hidden, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # gating
    out_norm = out_norm * out_gate

    # final linear projection
    out = torch.nn.functional.linear(out_norm, to_out_weight)

    return out

"""
Custom Triton-powered TriMul (outgoing) implementation.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Linear projections for left/right tensors and three gating tensors.
3. Apply the optional pairwise mask, then multiply by the left/right gates.
4. Convert left/right to float16, permute to bring the hidden dimension to the
   batch axis and flatten: shape → (B*H, N, N).
5. Compute the core O(N³) reduction using a fused batched GEMM kernel:
      out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   which is a batched mat‑mul of left and the transpose of right.
6. Reshape back to (B, N, N, H) and cast to float32.
7. Layer‑norm over the hidden dimension, apply the output gate and final linear
   projection back to the original channel dimension.
"""

import torch.nn.functional as F

def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH, M, N, K,
    stride_Abatch, stride_Am, stride_Ak,
    stride_Bbatch, stride_Bn, stride_Bk,
    stride_Cbatch, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):

    # compute block indices
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    batch_idx = pid // (num_blocks_m * num_blocks_n)
    block_idx = pid % (num_blocks_m * num_blocks_n)
    pid_m = block_idx // num_blocks_n
    pid_n = block_idx % num_blocks_n

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        k_range = cur_k + tl.arange(0, BLOCK_K)

        # A tile: (batch, m, k)
        a_ptrs = (
            A_ptr
            + batch_idx * stride_Abatch
            + offs_m[:, None] * stride_Am
            + k_range[None, :] * stride_Ak
        )
        # B tile (transposed on‑the‑fly): (batch, k, n)
        b_ptrs = (
            B_ptr
            + batch_idx * stride_Bbatch
            + k_range[:, None] * stride_Bk
            + offs_n[None, :] * stride_Bn
        )

        # masks for out‑of‑bounds loads
        mask_a = (offs_m[:, None] < M) & (k_range[None, :] < K)
        mask_b = (k_range[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        acc += tl.dot(a, b)               # (BLOCK_M, BLOCK_K) × (BLOCK_K, BLOCK_N)

    # write the result tile
    c_ptrs = (
        C_ptr
        + batch_idx * stride_Cbatch
        + offs_m[:, None] * stride_Cm
        + offs_n[None, :] * stride_Cn
    )
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def _batched_matmul(A: torch.Tensor, B: torch.Tensor,
                    BLOCK_M=64, BLOCK_N=64, BLOCK_K=64) -> torch.Tensor:
    """
    Batched matrix multiplication:
        C[b, i, j] = Σ_k A[b, i, k] * B[b, j, k]   (i‑j pairwise product)
    A, B : (batch, M, K) and (batch, N, K)   with K == N == M in our use‑case.
    Returns C of shape (batch, M, N) with dtype matching A/B (float16).
    """
    assert A.shape[0] == B.shape[0]
    batch, M, K = A.shape
    _, N, _ = B.shape
    # output buffer
    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # strides in elements (not bytes)
    stride_Abatch, stride_Am, stride_Ak = A.stride()
    stride_Bbatch, stride_Bn, stride_Bk = B.stride()
    stride_Cbatch, stride_Cm, stride_Cn = C.stride()

    # launch grid
    num_blocks_m = (M + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = batch * num_blocks_m * num_blocks_n
    grid = (total_blocks,)

    _batched_matmul_kernel[grid](
        A, B, C,
        batch, M, N, K,
        stride_Abatch, stride_Am, stride_Ak,
        stride_Bbatch, stride_Bn, stride_Bk,
        stride_Cbatch, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return C


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module, heavily accelerated by a
    Triton kernel for the O(N³) reduction.
    """
    # unpack inputs
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ----- weights -----
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ----- layer‑norm on input -----
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ----- linear projections (no bias) -----
    left_proj = F.linear(x, left_proj_weight)   # (B, N, N, H)
    right_proj = F.linear(x, right_proj_weight)

    # ----- mask handling -----
    nomask = config.get("nomask", True)
    if mask is None or nomask:
        mask_tensor = torch.ones(x.shape[:3], dtype=x.dtype, device=device)
    else:
        mask_tensor = mask.to(dtype=x.dtype, device=device)
    mask_expand = mask_tensor.unsqueeze(-1)      # (B, N, N, 1)
    left_proj = left_proj * mask_expand
    right_proj = right_proj * mask_expand

    # ----- gating -----
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left_proj * left_gate
    right = right_proj * right_gate

    # ----- cast to half for the heavy kernel -----
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ----- prepare for batched GEMM (flatten batch*hidden) -----
    B, N, _, _ = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    batch = B * hidden_dim
    left_flat = left.view(batch, N, N)   # (B*H, N, N)
    right_flat = right.view(batch, N, N)

    # ----- core O(N³) reduction via Triton -----
    out_hidden = _batched_matmul(left_flat, right_flat,
                                 BLOCK_M=64, BLOCK_N=64, BLOCK_K=64)  # (B*H, N, N)

    # ----- reshape back to (B, N, N, H) -----
    out_hidden = out_hidden.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out_hidden = out_hidden.to(torch.float32)

    # ----- post‑processing -----
    out_norm = F.layer_norm(out_hidden, (hidden_dim,),
                            weight=to_out_norm_weight,
                            bias=to_out_norm_bias)
    out_norm = out_norm * out_gate
    out = F.linear(out_norm, to_out_weight)   # (B, N, N, dim)

    return out
