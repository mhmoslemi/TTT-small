def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # maybe config also includes "nomask" to indicate mask content may be None
    # but we can just handle mask if mask is None or not.

    # Extra config may have noise distribution etc; ignore.

    # Extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # x norm
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # linear projections - using functional linear
    left = torch.nn.functional.linear(x_norm, left_proj_weight)
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    if mask is not None:
        mask_ = mask.unsqueeze(-1)  # shape (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Batched matmul using Triton
    # reshape
    B, N, _, _ = left.shape  # _ = hidden_dim
    # Permute to bring hidden_dim to batch dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # flatten batch dimension
    batch_flat = B * hidden_dim
    left_flat = left_perm.view(batch_flat, N, N)  # (B*H, N, N)
    # For right, we need shape (batch, N, K) where K = N as we treat k dimension last.
    # Right is shape (B*H, N, N) same as left.
    right_flat = right_perm.view(batch_flat, N, N)

    # Compute matmul: left @ right^T (right^T shape (batch, K, N)??)
    # In kernel we interpret B's dimensions as (batch, N, K) and we use stride_bk = stride of last dim (K), stride_bn = stride of first dim (N).
    # We'll pass right_flat as is (batch, N, K) i.e., N dimension first, K second, where K = N.
    # Our kernel computes C[i,j] = sum_k A[i,k] * B[j,k]
    # So B should be (batch, N, K). That's right_flat shape matches: first dim after batch is N (j), second dim K (k). Good.

    # Use batched_matmul_triton 
    out_flat = batched_matmul_triton(left_flat, right_flat)

    # reshape
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # to_out_norm
    out_normed = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # gate
    out_normed = out_normed * out_gate

    # final linear
    output = torch.nn.functional.linear(out_normed, to_out_weight)

    return output

@triton.jit
def batched_matmul_kernel(a_ptr, b_ptr, c_ptr,
                          stride_batch_a, stride_batch_b, stride_batch_c,
                          stride_am, stride_ak, stride_bk, stride_bn, stride_cm, stride_cn,
                          M, N, K,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    """Compute C = A @ B^T where A: (M,K), B: (N,K) per batch.
    Input:
    - A: shape (batch, M, K)
    - B: shape (batch, N, K)
    - C: shape (batch, M, N)
    This kernel computes per batch: C[i,j] = sum_k A[i,k] * B[j,k].
    """

    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # block start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Compute pointers for this batch
    a_batch_ptr = a_ptr + batch_idx * stride_batch_a
    b_batch_ptr = b_ptr + batch_idx * stride_batch_b
    c_batch_ptr = c_ptr + batch_idx * stride_batch_c

    # Mask for out-of-range
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M,K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B tile (K,N) ; B dimensions are (N, K) thus we load transposed as (K,N)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Compute dot product: (M,K) x (K,N) => (M,N)
        acc += tl.dot(a, b)

    # Store result
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def batched_matmul_triton(A, B, block_m=128, block_n=128, block_k=32):
    """Batched matmul: for each batch b,
    out[b,i,j] = sum_k A[b,i,k] * B[b,j,k]"""

    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb
    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # Compute grid
    grid = (batch, (M + block_m - 1) // block_m, (N + block_n - 1) // block_n)

    # Compute strides
    stride_batch_a = A.stride(0)
    stride_batch_b = B.stride(0)
    stride_batch_c = C.stride(0)
    stride_am = A.stride(1)
    stride_ak = A.stride(2)
    stride_bk = B.stride(2)  # stride across k (last dim)
    stride_bn = B.stride(1)  # stride across N (first dim)
    stride_cm = C.stride(1)
    stride_cn = C.stride(2)

    batched_matmul_kernel[grid](
        A, B, C,
        stride_batch_a, stride_batch_b, stride_batch_c,
        stride_am, stride_ak, stride_bk, stride_bn, stride_cm, stride_cn,
        M, N, K,
        BLOCK_M=block_m, BLOCK_N=block_n, BLOCK_K=block_k
    )
    return C

"""
custom_kernel: Triton‑accelerated implementation of the “outgoing” TriMul operator.

The heavy computation is the pairwise multiplicative update:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

For each hidden dimension `h` this is a matrix multiplication
    left_h @ right_h.T  (both of size N×N).

We fuse this N×N×N reduction across all batches and hidden dimensions
with a Triton kernel that treats (batch * hidden_dim) as the batch axis.
All other operations (LayerNorm, Linear projections, gating, final linear)
are performed with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_batch_a, stride_batch_b, stride_batch_c,
    stride_am, stride_ak, stride_bk, stride_bn, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """C = Σ_k A[:,i,k] * B[:,j,k]  (A: (M,K), B: (N,K) per batch).

    Shapes per batch:
        A : (M, K)
        B : (N, K)   (note the K dimension is the *last* one)
        C : (M, N)
    """
    batch_id = tl.program_id(0)          # batch index (batch * hidden_dim)
    pid_m = tl.program_id(1)            # block row
    pid_n = tl.program_id(2)            # block column

    # Block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers for this batch
    a_batch = a_ptr + batch_id * stride_batch_a
    b_batch = b_ptr + batch_id * stride_batch_b
    c_batch = c_ptr + batch_id * stride_batch_c

    # Masks for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M, K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_am
                            + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load B tile transposed to (K, N)
        # B is stored as (N, K); we want (K, N) for the dot product.
        b_ptrs = b_batch + (offs_k[:, None] * stride_bk
                            + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # Accumulate dot product over K
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm
                        + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor,
                           BLOCK_M=128, BLOCK_N=128, BLOCK_K=32) -> torch.Tensor:
    """
    Batched GEMM where for each batch:
        C[i, j] = Σ_k A[i, k] * B[j, k]

    A: (B, M, K)   contiguous
    B: (B, N, K)   contiguous (note K is last)
    Returns C: (B, M, N)   float32
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, "Inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # Compute strides (in elements)
    stride_batch_a = A.stride(0)
    stride_am = A.stride(1)
    stride_ak = A.stride(2)

    stride_batch_b = B.stride(0)
    stride_bk = B.stride(2)   # stride of K (last dim)
    stride_bn = B.stride(1)   # stride of N (first dim after batch)

    stride_batch_c = C.stride(0)
    stride_cm = C.stride(1)
    stride_cn = C.stride(2)

    grid = (batch,
            (M + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _tri_mul_gemm_kernel[grid](
        A, B, C,
        stride_batch_a, stride_batch_b, stride_batch_c,
        stride_am, stride_ak, stride_bk, stride_bn, stride_cm, stride_cn,
        M, N, K,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return C


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor, shape [B, N, N, dim]
            mask: torch.Tensor or None, shape [B, N, N] (float or bool)
            weights: dict of torch.Tensor with keys:
                'norm.weight', 'norm.bias',
                'left_proj.weight', 'right_proj.weight',
                'left_gate.weight', 'right_gate.weight', 'out_gate.weight',
                'to_out_norm.weight', 'to_out_norm.bias',
                'to_out.weight'
            config: dict with at least keys 'dim' (int) and 'hidden_dim' (int)

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ----- LayerNorm on input -----
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # ----- Linear projections (no bias) -----
    left = F.linear(x, weights["left_proj.weight"])    # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, hidden_dim]

    # ----- Optional mask -----
    if mask is not None:
        if mask.dtype != left.dtype:
            mask = mask.to(left.dtype)
        mask = mask.unsqueeze(-1)            # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ----- Gating signals (sigmoid) -----
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ----- Pairwise multiplicative update (batched GEMM) -----
    # Bring hidden dimension into batch dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    batch_flat = B * H

    left_flat = left_perm.view(batch_flat, N, N)   # (batch, M=N, K=N)
    right_flat = right_perm.view(batch_flat, N, N) # (batch, N, K=N)

    # Compute out_flat[b, i, j] = Σ_k left_flat[b, i, k] * right_flat[b, j, k]
    out_flat = _batched_matmul_triton(left_flat, right_flat)

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden_dim]

    # ----- Output normalization and final linear -----
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_norm = out_norm * out_gate
    output = F.linear(out_norm, weights["to_out.weight"])  # [B, N, N, dim]

    return output
