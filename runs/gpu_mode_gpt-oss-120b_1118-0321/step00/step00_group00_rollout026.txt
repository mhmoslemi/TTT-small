@triton.jit
def batch_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Am, stride_An, stride_Ah,
    stride_Bm, stride_Bn, stride_Bh,
    stride_Cm, stride_Cn, stride_Ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    # total batch dimension: total_batches = B * H (we treat as a leading axis)
    batch = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    block_idx = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_m = block_idx // tl.cdiv(N, BLOCK_N)  # row block
    pid_n = block_idx % tl.cdiv(N, BLOCK_N)  # col block

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for A and B: they have shape (batch, M, K) and (batch, K, N) respectively.
    A_ptrs = A_ptr + batch * stride_Ab + offs_m[:, None] * stride_Am + tl.arange(0, BLOCK_K)[None, :] * stride_An
    B_ptrs = B_ptr + batch * stride_Bb + tl.arange(0, BLOCK_K)[:, None] * stride_Bm + offs_n[None, :] * stride_Bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, K, BLOCK_K):
        a = tl.load(A_ptrs, mask=mask_m[:, None] & (k + tl.arange(0, BLOCK_K) < K), other=0.0)
        b = tl.load(B_ptrs, mask=(k + tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)  # yields (BLOCK_M, BLOCK_N)
        # increment pointers for next tile
        A_ptrs += BLOCK_K * stride_An
        B_ptrs += BLOCK_K * stride_Bm

    # write back C
    C_ptrs = C_ptr + batch * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(C_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
```

def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_ptr = A_ptr + batch_idx * stride_Ab
    b_ptr = B_ptr + batch_idx * stride_Bb
    c_ptr = C_ptr + batch_idx * stride_Cb

    a_ptrs = a_ptr + offs_m[:, None] * stride_Am + tl.arange(0, BLOCK_K)[None, :] * stride_Ak

    norm_bias = weights["norm.bias"]

    # Linear projections
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    right = right_proj * right_gate

"""
custom_kernel: High‑performance TriMul (outgoing) forward pass.

Algorithm
---------
1. Layer‑norm the input tensor (last dimension = dim).
2. Compute four linear projections (left/right) and three gating
   projections (left_gate, right_gate, out_gate) using the supplied
   weights and a sigmoid non‑linearity for the gates.
3. Apply the optional mask to the left/right projections, then multiply
   by their respective gates → tensors L,R of shape [B,N,N,H].
4. Core O(N³) reduction:
   out[b,i,j,d] = Σₖ L[b,i,k,d] * R[b,j,k,d].
   This is a batched matrix multiplication for each hidden channel:
   for each (b,d) treat L₍b,d₎ ∈ ℝ^{N×N} and R₍b,d₎ᵀ ∈ ℝ^{N×N},
   compute C₍b,d₎ = L₍b,d₎ @ R₍b,d₎ᵀ.
   We implement this with a custom Triton kernel that processes a
   batch of (B·H) independent GEMMs in mixed‑precision (FP16
   accumulation in FP32).
5. Reshape the result back to [B,N,N,H] and apply a second
   Layer‑norm (to_out_norm) followed by the out‑gate (sigmoid) and a
   final linear projection to the original dimension.
6. Return the tensor of shape [B, N, N, dim].

The kernel fuses the heavy N³ reduction and provides a large speed‑up
over the naïve PyTorch `einsum`. All other lightweight operations are
performed with PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    A_ptr,                     # FP16 pointer, shape (B*H, M, K)
    B_ptr,                     # FP16 pointer, shape (B*H, K, N)
    C_ptr,                     # FP32 pointer, shape (B*H, M, N)
    M, N, K,                   # matrix dimensions (int)
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM C = A @ B   (A: (M,K), B: (K,N)) for many batches."""
    batch_id = tl.program_id(0)          # which (b,d) we are processing
    pid_m = tl.program_id(1)             # block row
    pid_n = tl.program_id(2)             # block col

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_base = A_ptr + batch_id * stride_Ab
    b_base = B_ptr + batch_id * stride_Bb
    c_base = C_ptr + batch_id * stride_Cb

    # Pointers to the start of the current tile
    a_ptrs = a_base + offs_m[:, None] * stride_Am + tl.arange(0, BLOCK_K)[None, :] * stride_Ak
    b_ptrs = b_base + tl.arange(0, BLOCK_K)[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, K, BLOCK_K):
        # compute mask for the possibly partial K‑tile
        k_range = tl.arange(0, BLOCK_K) + k
        mask_k = k_range < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # FP16×FP16 → FP32 accumulation

        # advance to next K‑tile
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # Store C
    c_ptrs = c_base + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """Entry point expected by the evaluator."""
    import torch
    import triton

    # unpack arguments
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections and gates
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 3. Optional mask (if provided)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)                 # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 4. Apply gates
    # ------------------------------------------------------------------
    left = left_proj * left_gate       # [B, N, N, H]
    right = right_proj * right_gate    # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5. Core TriMul (batched GEMM) using Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # bring hidden dimension to the second axis for convenient batching
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # transpose the second N‑axis of right (k ↔ j)
    right_t = right_perm.transpose(-2, -1).contiguous()  # [B, H, N, N]

    # collapse batch & hidden into a single batch dimension
    batch_h = B * hidden_dim
    A = left_perm.view(batch_h, N, N)        # (B*H, M=N, K=N)
    Bmat = right_t.view(batch_h, N, N)       # (B*H, K=N, N)

    # mixed‑precision: compute in FP16, accumulate in FP32
    A_fp16 = A.to(torch.float16)
    B_fp16 = Bmat.to(torch.float16)
    C_fp32 = torch.empty((batch_h, N, N), dtype=torch.float32, device=device)

    # Choose block sizes (tuned for N up to 1024, H≤128)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    if N <= 64:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 16

    grid = (
        batch_h,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_matmul_kernel[grid](
        A_ptr=A_fp16,
        B_ptr=B_fp16,
        C_ptr=C_fp32,
        M=N,
        N=N,
        K=N,
        stride_Ab=A_fp16.stride(0),
        stride_Am=A_fp16.stride(1),
        stride_Ak=A_fp16.stride(2),
        stride_Bb=B_fp16.stride(0),
        stride_Bk=B_fp16.stride(1),
        stride_Bn=B_fp16.stride(2),
        stride_Cb=C_fp32.stride(0),
        stride_Cm=C_fp32.stride(1),
        stride_Cn=C_fp32.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out = C_fp32.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output normalisation, gating, and final projection
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out_norm * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
