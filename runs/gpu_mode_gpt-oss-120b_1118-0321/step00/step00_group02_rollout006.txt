"""
TriMul (outgoing) forward pass implemented with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input over the last dimension (dim).
2. Compute linear projections:
   left_proj, right_proj          :  x_norm @ W (dim → hidden_dim)
   left_gate, right_gate, out_gate:  x_norm @ W (dim → hidden_dim)  followed by sigmoid.
3. Fuse mask, projection and gate:
   left  = left_proj  * left_gate  * mask
   right = right_proj * right_gate * mask
   The multiplication is performed by a small Triton kernel that
   reads three contiguous tensors and writes the element‑wise product.
4. For each head (hidden_dim) compute the pairwise “outer‑product” update:
   out[...,h] = left[...,h] @ right[...,h].T
   Implemented as a batched GEMM: reshape the tensors to
   (B*hidden_dim, N, N) and call `torch.bmm`.
5. Layer‑norm the result over the hidden dimension and apply `out_gate`.
6. Final linear projection back to `dim`.

Only the element‑wise gated‑mask multiplication is carried out with Triton;
all heavy matrix multiplications use PyTorch’s highly‑optimized cuBLAS kernels.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fused element‑wise multiplication (proj * gate * mask)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # float*   (output of projection)
    gate_ptr,          # float*   (gate after sigmoid)
    mask_ptr,          # float*   (broadcasted mask, 0/1)
    out_ptr,           # float*   (result)
    total_elements: tl.int32,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + offs, mask=mask, other=0.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor) -> torch.Tensor:
    """
    proj, gate, mask must be contiguous tensors of identical shape
    (float32). Returns `proj * gate * mask`.
    """
    assert proj.shape == gate.shape == mask.shape
    out = torch.empty_like(proj)

    total = proj.numel()
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)

    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        BLOCK_SIZE=BLOCK,
    )
    return out


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
        input_tensor : [B, N, N, dim]  (float32)
        mask         : [B, N, N] or None
        weights      : dict with the following keys (all float32 on the same device)
                       - "norm.weight", "norm.bias"
                       - "left_proj.weight", "right_proj.weight"
                       - "left_gate.weight", "right_gate.weight", "out_gate.weight"
                       - "to_out_norm.weight", "to_out_norm.bias"
                       - "to_out.weight"
        config       : dict with keys "dim" and "hidden_dim"
    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1) LayerNorm over the last dimension
    # --------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # --------------------------------------------------------------
    # 2) Linear projections and gates (no bias)
    # --------------------------------------------------------------
    left_proj   = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right_proj  = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    left_gate_lin  = torch.nn.functional.linear(x_norm, weights["left_gate.weight"])
    right_gate_lin = torch.nn.functional.linear(x_norm, weights["right_gate.weight"])
    out_gate_lin   = torch.nn.functional.linear(x_norm, weights["out_gate.weight"])

    left_gate  = torch.sigmoid(left_gate_lin)
    right_gate = torch.sigmoid(right_gate_lin)
    out_gate   = torch.sigmoid(out_gate_lin)

    # --------------------------------------------------------------
    # 3) Mask handling (broadcast to hidden_dim)
    # --------------------------------------------------------------
    if mask is None:
        mask_tensor = torch.ones((bs, N, N), device=device, dtype=dtype)
    else:
        mask_tensor = mask.to(dtype=dtype, device=device)

    # Expand mask to (bs, N, N, hidden_dim) – produce a contiguous tensor.
    mask_expanded = mask_tensor.unsqueeze(-1).repeat(1, 1, 1, hidden_dim)

    # --------------------------------------------------------------
    # 4) Fuse projection, gate and mask with Triton
    # --------------------------------------------------------------
    left  = _apply_gate_and_mask(left_proj, left_gate, mask_expanded)
    right = _apply_gate_and_mask(right_proj, right_gate, mask_expanded)

    # --------------------------------------------------------------
    # 5) Pairwise outer‑product update: out = left @ right^T per head
    # --------------------------------------------------------------
    # shape: (bs, N, N, hidden_dim) -> (bs, hidden_dim, N, N)
    L = left.permute(0, 3, 1, 2).contiguous()
    R = right.permute(0, 3, 1, 2).contiguous()

    BHH = bs * hidden_dim
    L_flat = L.reshape(BHH, N, N)
    R_flat = R.reshape(BHH, N, N)

    # Batched matrix multiplication (GEMM)
    out_flat = torch.bmm(L_flat, R_flat.transpose(1, 2))

    # Restore original layout
    out = out_flat.reshape(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6) Final LayerNorm, out‑gate and linear projection
    # --------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    out_norm = out_norm * out_gate

    out_final = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return out_final
