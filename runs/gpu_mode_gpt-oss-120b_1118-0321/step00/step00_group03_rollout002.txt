    B_D, M, N, K,

    batch_id = tl.program_id(0)
    
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    
    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

"""
TriMul forward kernel (outgoing version)

The heavy part is the tensor contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
where left/right are the projected and gated pairwise tensors.
We implement this contraction with a custom Triton batched GEMM kernel that
computes, for each (batch, hidden) pair, a matrix multiplication
    left_mat (N×N) @ right_mat.T (N×N)   ->   out_mat (N×N)
using FP16 arithmetic and FP32 accumulation. The rest of the
TriMul pipeline (LayerNorms, Linear projections, gating) is performed
with PyTorch for simplicity.

The kernel processes a flattened batch dimension (batch * hidden_dim) and
uses three‑dimensional grid (batch, tile_i, tile_j) with tiling
BLOCK_M × BLOCK_N tiles and a BLOCK_K inner reduction.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul where the second operand is transposed.
# Computes: C = A @ B.T  for each batch.
# A, B, C are stored as (batch, M, K), (batch, N, K), (batch, M, N)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_batch_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # A block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_batch * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(
            a_ptrs,
            mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
            other=0.0,
        )

        # B_T block (transposed): shape (BLOCK_K, BLOCK_N)
        b_ptrs = (
            B_ptr
            + pid_batch * stride_b_batch
            + offs_k[:, None] * stride_b_k   # row moves along original K (col)
            + offs_n[None, :] * stride_b_n   # column moves along original N (row)
        )
        b_T = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
            other=0.0,
        )

        # matrix multiply (M×K)·(K×N) → (M×N)
        acc += tl.dot(a, b_T)

    # store result
    c_ptrs = (
        C_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=(offs_m[:, None] < M) & (offs_n[None, :] < N),
    )


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # unpack
    input_tensor, mask_tensor, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype
    eps = 1e-5

    # ------------------------------------------------------------------
    # Config / shape
    # ------------------------------------------------------------------
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # LayerNorm on the input (dim dimension)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # ensure correct dtype / device
    norm_weight = norm_weight.to(device=device, dtype=dtype)
    norm_bias = norm_bias.to(device=device, dtype=dtype)

    # mean / var over the last dimension
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = ((input_tensor - mean) ** 2).mean(dim=-1, keepdim=True)
    inv_std = torch.rsqrt(var + eps)
    x = (input_tensor - mean) * inv_std
    x = x * norm_weight + norm_bias  # (..., dim)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"].to(device=device, dtype=dtype)
    right_proj_weight = weights["right_proj.weight"].to(device=device, dtype=dtype)
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # Mask handling
    # ------------------------------------------------------------------
    # If mask is omitted or config indicates "nomask", treat as all ones
    if config.get("nomask", True):
        mask = torch.ones((bs, N, N), dtype=dtype, device=device)
    else:
        mask = mask_tensor.to(device=device, dtype=dtype)
    mask = mask.unsqueeze(-1)  # broadcast over hidden dim

    left = left * mask
    right = right * mask

    # ------------------------------------------------------------------
    # Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"].to(device=device, dtype=dtype)
    right_gate_weight = weights["right_gate.weight"].to(device=device, dtype=dtype)
    out_gate_weight = weights["out_gate.weight"].to(device=device, dtype=dtype)

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # Reshape to (batch*hidden, N, N) and cast to fp16 for the kernel
    left_t = left.permute(0, 3, 1, 2).contiguous()   # B, H, N, N
    right_t = right.permute(0, 3, 1, 2).contiguous()
    B, H, _, _ = left_t.shape
    flat_batch = B * H

    left_flat = left_t.view(flat_batch, N, N).to(torch.float16)
    right_flat = right_t.view(flat_batch, N, N).to(torch.float16)

    out_flat = torch.empty_like(left_flat, dtype=torch.float16, device=device)

    # strides (in elements) for Triton
    stride_a_batch = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_batch = right_flat.stride(0)
    stride_b_n = right_flat.stride(1)  # rows (N)
    stride_b_k = right_flat.stride(2)  # columns (K)

    stride_c_batch = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    # Tile sizes – chosen to fit well on modern GPUs.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        flat_batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    tri_mul_batch_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        flat_batch,
        N,
        N,
        N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_n,
        stride_b_k,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert back to fp32 and reshape to original layout
    out = out_flat.to(dtype).view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # Output LayerNorm
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(device=device, dtype=dtype)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device=device, dtype=dtype)

    mean_out = out.mean(dim=-1, keepdim=True)
    var_out = ((out - mean_out) ** 2).mean(dim=-1, keepdim=True)
    out_norm = (out - mean_out) * torch.rsqrt(var_out + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # ------------------------------------------------------------------
    # Apply output gate and final linear projection
    # ------------------------------------------------------------------
    out = out_norm * out_gate

    to_out_weight = weights["to_out.weight"].to(device=device, dtype=dtype)
    output = F.linear(out, to_out_weight)  # [B, N, N, dim]

    return output
