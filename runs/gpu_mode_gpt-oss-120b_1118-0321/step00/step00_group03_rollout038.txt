def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    N, D,
    stride_left_i, stride_left_k, stride_left_d,
    stride_right_j, stride_right_k, stride_right_d,
    stride_out_i, stride_out_j, stride_out_d,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):

    # Compute tile indices
    num_pid_n = tl.cdiv(N, BLOCK_J)  # Oops careful: we need number of tiles in J dimension.
    # Actually we defined total tiles as num_i * num_j.
    num_pid_i = tl.cdiv(N, BLOCK_I)
    num_pid_j = tl.cdiv(N, BLOCK_J)
    pid_i = pid // num_pid_j
    pid_j = pid % num_pid_j

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Outer loop over D blocks
    for d_start in range(0, D, BLOCK_D):
        offs_d = d_start + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

        # Loop over K dimension
        for k_start in range(0, N, BLOCK_K):
            offs_k = k_start + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # Load left block (I, K, D)
            left_ptrs = left_ptr + (
                offs_i[:, None, None] * stride_left_i +
                offs_k[None, :, None] * stride_left_k +
                offs_d[None, None, :] * stride_left_d
            )
            left = tl.load(left_ptrs,
                mask=mask_i[:, None, None] & mask_k[None, :, None],
                other=0.0
            )   # shape (I, K, D)

            # Load right block (K, J, D)
            right_ptrs = right_ptr + (
                offs_k[:, None, None] * stride_right_k +
                offs_j[None, :, None] * stride_right_j +
                offs_d[None, None, :] * stride_right_d
            )
            right = tl.load(right_ptrs,
                mask=mask_k[:, None, None] & mask_j[None, :, None],
                other=0.0
            )   # shape (K, J, D)

            # left: (I, K, D) -> (I, K, 1, D)
            # right: (K, J, D) -> (1, K, J, D)
            prod = left[:, :, None, :] * right[None, :, :, :]   # shape (I, K, J, D)
            acc += tl.sum(prod, axis=1)   # sum across K -> (I, J, D)

        # Store accumulator to output

            offs_i[:, None, None] * stride_out_i +
            offs_j[None, :, None] * stride_out_j +
            offs_d[None, None, :] * stride_out_d

    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)  # not used

    # Extract weight tensors
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Input shape
    B, N, _, _ = input_tensor.shape

    x = F.layer_norm(input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]

        right = right * mask_

    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    right = right * right_gate

Define `mask_k_vec = mask_k[k_idx]` (bool). Convert to float mask: `mask_k_f = tl.where(mask_k_vec, 1.0, 0.0)`. @triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    N, D,
    stride_left_i, stride_left_k, stride_left_d,
    stride_right_i, stride_right_k, stride_right_d,
    stride_out_i, stride_out_j, stride_out_d,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    num_pid_j = tl.cdiv(N, BLOCK_J)
    pid_i = pid // num_pid_j
    pid_j = pid % num_pid_j

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    for d_start in tl.range(0, D, BLOCK_D):
        d_offs = d_start + tl.arange(0, BLOCK_D)
        mask_d = d_offs < D

        # accumulator in float32
        acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

        # Loop over k blocks
        for k_start in tl.range(0, N, BLOCK_K):
            k_offs = k_start + tl.arange(0, BLOCK_K)
            mask_k = k_offs < N

            # Load left block: shape (BLOCK_I, BLOCK_K, BLOCK_D)
            left_ptrs = left_ptr + (
                offs_i[:, None, None] * stride_left_i +
                k_offs[None, :, None] * stride_left_k +
                d_offs[None, None, :] * stride_left_d
            )
            left = tl.load(left_ptrs,
                mask=mask_i[:, None, None] & mask_k[None, :, None],
                other=0.0
            ).to(tl.float32)

            # Load right block: shape (BLOCK_K, BLOCK_J, BLOCK_D)
            right_ptrs = right_ptr + (
                k_offs[:, None, None] * stride_right_k +
                offs_j[None, :, None] * stride_right_i +
                d_offs[None, None, :] * stride_right_d
            )
            right = tl.load(right_ptrs,
                mask=mask_k[:, None, None] & mask_j[None, :, None],
                other=0.0
            ).to(tl.float32)

            # Loop over k within the block
            for kk in range(0, BLOCK_K):
                # scalar mask for this kk - optional, but loads already zeroed out.
                # get slices
                left_kk = left[:, kk, :]          # (BLOCK_I, BLOCK_D)
                right_kk = right[kk, :, :]        # (BLOCK_J, BLOCK_D)
                # broadcast and accumulate
                acc += left_kk[:, None, :] * right_kk[None, :, :]

        # Store result to out
        out_ptrs = out_ptr + (
            offs_i[:, None, None] * stride_out_i +
            offs_j[None, :, None] * stride_out_j +
            d_offs[None, None, :] * stride_out_d
        )
        tl.store(out_ptrs,
            acc,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
        )

"""
TriMul "outgoing" forward pass optimized with a Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Apply linear projections for the left/right paths and their gates.
3. Mask (if provided), then multiply by the gates.
4. Compute the core tensor contraction
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   using a custom Triton kernel that tiles over the i, j and hidden
   dimensions and iterates over k in blocks.
5. Layer‑norm the result, apply the output gate and a final linear
   projection back to the original feature dimension.

The Triton kernel processes a tile of (BLOCK_I, BLOCK_J, BLOCK_D) output
elements.  For each tile it loads blocks of size (BLOCK_I, BLOCK_K, BLOCK_D)
and (BLOCK_K, BLOCK_J, BLOCK_D) from left and right, respectively, and
accumulates the outer‑product for each k within the block.  Accumulation
is performed in float32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------------- #
# Triton kernel
# ----------------------------------------------------------------------------- #
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    N, D,
    stride_left_i, stride_left_k, stride_left_d,
    stride_right_i, stride_right_k, stride_right_d,
    stride_out_i, stride_out_j, stride_out_d,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    num_pid_j = tl.cdiv(N, BLOCK_J)               # tiles along the j‑axis
    pid_i = pid // num_pid_j
    pid_j = pid %  num_pid_j

    # Global i / j coordinates for this tile
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------------- #
    # Loop over hidden dimension in small sub‑blocks (BLOCK_D)
    # --------------------------------------------------------------------- #
    for d_start in range(0, D, BLOCK_D):
        offs_d = d_start + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        # accumulator (float32)
        acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

        # ----------------------------------------------------------------- #
        # Loop over the sequence dimension k in blocks
        # ----------------------------------------------------------------- #
        for k_start in range(0, N, BLOCK_K):
            offs_k = k_start + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # left block:  (i, k, d)
            left_ptrs = left_ptr + (
                offs_i[:, None, None] * stride_left_i +
                offs_k[None, :, None] * stride_left_k +
                offs_d[None, None, :] * stride_left_d
            )
            left = tl.load(
                left_ptrs,
                mask=mask_i[:, None, None] & mask_k[None, :, None],
                other=0.0
            ).to(tl.float32)                       # shape (BLOCK_I, BLOCK_K, BLOCK_D)

            # right block: (k, j, d)
            right_ptrs = right_ptr + (
                offs_k[:, None, None] * stride_right_k +
                offs_j[None, :, None] * stride_right_i +
                offs_d[None, None, :] * stride_right_d
            )
            right = tl.load(
                right_ptrs,
                mask=mask_k[:, None, None] & mask_j[None, :, None],
                other=0.0
            ).to(tl.float32)                       # shape (BLOCK_K, BLOCK_J, BLOCK_D)

            # Accumulate over the k‑dimension inside the block
            for kk in range(0, BLOCK_K):
                # left[i, kk, d] : (BLOCK_I, BLOCK_D)
                # right[kk, j, d]: (BLOCK_J, BLOCK_D)
                left_kk  = left[:, kk, :]          # (BLOCK_I, BLOCK_D)
                right_kk = right[kk, :, :]         # (BLOCK_J, BLOCK_D)
                # broadcast to (i, j, d) and accumulate
                acc += left_kk[:, None, :] * right_kk[None, :, :]

        # ----------------------------------------------------------------- #
        # Write the tile back to global memory
        # ----------------------------------------------------------------- #
        out_ptrs = out_ptr + (
            offs_i[:, None, None] * stride_out_i +
            offs_j[None, :, None] * stride_out_j +
            offs_d[None, None, :] * stride_out_d
        )
        tl.store(
            out_ptrs,
            acc,
            mask=mask_i[:, None, None] &
                 mask_j[None, :, None] &
                 mask_d[None, None, :]
        )


# ----------------------------------------------------------------------------- #
# Python entry point
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul "outgoing" operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, C]
        - mask         : torch.Tensor [B, N, N] (or None)
        - weights      : dict of model parameters
        - config       : dict containing "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------------- #
    # Extract weights (all tensors are assumed to be on the correct device)
    # --------------------------------------------------------------------- #
    norm_weight          = weights["norm.weight"]
    norm_bias            = weights["norm.bias"]
    left_proj_weight     = weights["left_proj.weight"]
    right_proj_weight    = weights["right_proj.weight"]
    left_gate_weight     = weights["left_gate.weight"]
    right_gate_weight    = weights["right_gate.weight"]
    out_gate_weight      = weights["out_gate.weight"]
    to_out_norm_weight   = weights["to_out_norm.weight"]
    to_out_norm_bias     = weights["to_out_norm.bias"]
    to_out_weight        = weights["to_out.weight"]

    # --------------------------------------------------------------------- #
    # 1) Layer‑norm on input
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]

    # --------------------------------------------------------------------- #
    # 2) Linear projections & gates
    # --------------------------------------------------------------------- #
    left  = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)          # [B, N, N, 1]
        left  = left * mask_
        right = right * mask_

    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left  = left * left_gate
    right = right * right_gate

    B, N, _, _ = left.shape

    # --------------------------------------------------------------------- #
    # 3) Core contraction via Triton
    # --------------------------------------------------------------------- #
    out = torch.empty_like(left)  # will hold [B, N, N, hidden_dim]

    # Tiling parameters – chosen to fit into registers on an A100/B200.
    BLOCK_I = 16   # Tile size along the i axis
    BLOCK_J = 16   # Tile size along the j axis
    BLOCK_K = 32   # Block size for the reduction axis k
    BLOCK_D = 8    # Sub‑block size for the hidden dimension

    num_i = (N + BLOCK_I - 1) // BLOCK_I
    num_j = (N + BLOCK_J - 1) // BLOCK_J
    grid = (num_i * num_j,)

    for b in range(B):
        # Slice the current batch – these tensors are views with the same
        # strides, which we pass directly to the kernel.
        left_batch  = left[b]   # shape [N, N, hidden_dim]
        right_batch = right[b]
        out_batch   = out[b]

        trimul_outgoing_kernel[grid](
            left_batch,
            right_batch,
            out_batch,
            N,
            hidden_dim,
            left_batch.stride(0),   # i‑stride
            left_batch.stride(1),   # k‑stride
            left_batch.stride(2),   # d‑stride
            right_batch.stride(0),  # j‑stride (first N)
            right_batch.stride(1),  # k‑stride
            right_batch.stride(2),  # d‑stride
            out_batch.stride(0),    # i‑stride
            out_batch.stride(1),    # j‑stride
            out_batch.stride(2),    # d‑stride
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_K=BLOCK_K,
            BLOCK_D=BLOCK_D,
        )

    # --------------------------------------------------------------------- #
    # 4) Final Layer‑norm, gating and output projection
    # --------------------------------------------------------------------- #
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out_norm = out_norm * out_gate
    output = F.linear(out_norm, to_out_weight)  # [B, N, N, dim]

    return output
