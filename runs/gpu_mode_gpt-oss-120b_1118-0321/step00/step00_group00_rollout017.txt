def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # GPU device
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # norm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # linear projections
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # mask handling
    if mask is None or config.get("nomask", False):
        mask_tensor = torch.ones((bs, N, N), dtype=x.dtype, device=device)
    else:
        mask_tensor = mask.to(dtype=x.dtype)  # ensure dtype

    # allocate output tensors
    left_fused = torch.empty_like(left_proj)
    right_fused = torch.empty_like(right_proj)

    # launch kernel
    total_elements = left_proj.numel()  # B*N*N*hidden_dim
    BLOCK_SIZE = 256
    num_blocks = (total_elements + BLOCK_SIZE - 1) // BLOCK_SIZE
    apply_mask_gate_kernel[(num_blocks,)](
        left_proj, right_proj,
        mask_tensor,
        left_gate, right_gate,
        left_fused, right_fused,
        bs, N, hidden_dim,
        left_proj.stride(0), left_proj.stride(1), left_proj.stride(2), left_proj.stride(3),
        right_proj.stride(0), right_proj.stride(1), right_proj.stride(2), right_proj.stride(3),
        mask_tensor.stride(0), mask_tensor.stride(1), mask_tensor.stride(2),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        BLOCK_SIZE=BLOCK_SIZE,
    )
    # Now compute pairwise multiplication via batched gemm
    # reshape for batch matmul
    # shape [B,N,N,hidden_dim] -> [B, hidden_dim, N, N]
    left_perm = left_fused.permute(0, 3, 1, 2).contiguous()
    right_perm = right_fused.permute(0, 3, 1, 2).contiguous()

    # flatten batch and hidden_dim
    B, D, N, N_ = left_perm.shape
    assert N == N_, "non-square? should be equal"

    left_flat = left_perm.view(B * D, N, N)
    right_flat = right_perm.view(B * D, N, N)

    # compute out = left * right^T across k dimension
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    # reshape back
    out = out_flat.view(B, D, N, N).permute(0, 2, 3, 1)

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # gating
    out = out * out_gate

    # final linear projection
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

@triton.jit
def apply_mask_gate_kernel(left_ptr, right_ptr, mask_ptr,
                           left_gate_ptr, right_gate_ptr,
                           out_left_ptr, out_right_ptr,
                           B, N, D,
                           stride_left_b, stride_left_i, stride_left_k, stride_left_d,
                           stride_right_b, stride_right_i, stride_right_k, stride_right_d,
                           stride_mask_b, stride_mask_i, stride_mask_k,
                           stride_left_gate_b, stride_left_gate_i, stride_left_gate_k, stride_left_gate_d,
                           stride_right_gate_b, stride_right_gate_i, stride_right_gate_k, stride_right_gate_d,
                           BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    total = B * N * N * D
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    # compute coordinates
    b = offs // (N * N * D)
    rem = offs % (N * N * D)
    i = rem // (N * D)
    rem2 = rem % (N * D)
    k = rem2 // D
    d = rem2 % D

    # Offsets
    off_left = b * stride_left_b + i * stride_left_i + k * stride_left_k + d * stride_left_d
    off_right = b * stride_right_b + i * stride_right_i + k * stride_right_k + d * stride_right_d
    off_mask = b * stride_mask_b + i * stride_mask_i + k * stride_mask_k
    off_lgate = b * stride_left_gate_b + i * stride_left_gate_i + k * stride_left_gate_k + d * stride_left_gate_d
    off_rgate = b * stride_right_gate_b + i * stride_right_gate_i + k * stride_right_gate_k + d * stride_right_gate_d

    left = tl.load(left_ptr + off_left, mask=mask, other=0.0)
    right = tl.load(right_ptr + off_right, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr + off_mask, mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr + off_lgate, mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + off_rgate, mask=mask, other=0.0)

    out_l = left * mask_val * lgate
    out_r = right * mask_val * rgate

    tl.store(out_left_ptr + off_left, out_l, mask=mask)
    tl.store(out_right_ptr + off_right, out_r, mask=mask)

"""
TriMul (outgoing) forward pass with a fused mask‑gate Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor `x` over the last dimension.
2. Linear projections for left/right values and for left/right/out gates.
   Gate values are passed through a sigmoid.
3. Fuse the element‑wise mask, left‑gate and right‑gate multiplications in a
   custom Triton kernel.  The kernel processes the flattened tensor
   `[B, N, N, hidden_dim]` and writes gated tensors `left_fused`,
   `right_fused`.
4. Compute the pairwise inner product over the sequence dimension `k`:
   `out_{b,i,j,d} = Σ_k left_fused[b,i,k,d] * right_fused[b,j,k,d]`.
   This is performed efficiently with a batched GEMM:
   - reshape to `[B*hidden_dim, N, N]`,
   - `out = left @ right^T` via `torch.bmm`,
   - reshape back to `[B, N, N, hidden_dim]`.
5. Apply a second LayerNorm over the hidden dimension, multiply by the
   `out_gate`, and finally project back to the original channel dimension
   with a linear layer.
6. Return the resulting tensor of shape `[B, N, N, dim]`.

Only the mask‑gate fusion (step 3) is implemented in Triton; all other
operations use PyTorch’s highly‑optimized kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def apply_mask_gate_kernel(
    left_ptr, right_ptr, mask_ptr,
    left_gate_ptr, right_gate_ptr,
    out_left_ptr, out_right_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_lgate_b, stride_lgate_i, stride_lgate_k, stride_lgate_d,
    stride_rgate_b, stride_rgate_i, stride_rgate_k, stride_rgate_d,
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse mask and gate values into left/right tensors."""
    pid = tl.program_id(0)
    total = B * N * N * D
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    # Decode flat index into (b, i, k, d)
    b = offs // (N * N * D)
    rem = offs % (N * N * D)
    i = rem // (N * D)
    rem2 = rem % (N * D)
    k = rem2 // D
    d = rem2 % D

    # Compute element offsets for each tensor
    off_left  = b * stride_left_b  + i * stride_left_i  + k * stride_left_k  + d * stride_left_d
    off_right = b * stride_right_b + i * stride_right_i + k * stride_right_k + d * stride_right_d
    off_mask  = b * stride_mask_b  + i * stride_mask_i  + k * stride_mask_k
    off_lgate = b * stride_lgate_b + i * stride_lgate_i + k * stride_lgate_k + d * stride_lgate_d
    off_rgate = b * stride_rgate_b + i * stride_rgate_i + k * stride_rgate_k + d * stride_rgate_d

    # Load values
    left_val  = tl.load(left_ptr + off_left,  mask=mask, other=0.0)
    right_val = tl.load(right_ptr + off_right, mask=mask, other=0.0)
    mask_val  = tl.load(mask_ptr + off_mask,  mask=mask, other=0.0)
    lgate_val = tl.load(left_gate_ptr + off_lgate, mask=mask, other=0.0)
    rgate_val = tl.load(right_gate_ptr + off_rgate, mask=mask, other=0.0)

    # Fuse mask & gates
    out_left  = left_val  * mask_val * lgate_val
    out_right = right_val * mask_val * rgate_val

    # Store results
    tl.store(out_left_ptr + off_left,  out_left,  mask=mask)
    tl.store(out_right_ptr + off_right, out_right, mask=mask)


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    input_tensor, mask_tensor, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2. Linear projections and gate values
    # ------------------------------------------------------------------
    left_proj   = F.linear(x, weights["left_proj.weight"])
    right_proj  = F.linear(x, weights["right_proj.weight"])
    left_gate   = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate  = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate    = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 3. Mask handling & fused mask‑gate kernel
    # ------------------------------------------------------------------
    if mask_tensor is None or config.get("nomask", False):
        mask = torch.ones((bs, N, N), dtype=dtype, device=device)
    else:
        mask = mask_tensor.to(dtype=dtype)

    # Allocate output buffers for the fused tensors
    left_fused  = torch.empty_like(left_proj)
    right_fused = torch.empty_like(right_proj)

    total_el = left_proj.numel()                # B * N * N * hidden_dim
    BLOCK_SIZE = 256
    num_blocks = (total_el + BLOCK_SIZE - 1) // BLOCK_SIZE

    # launch the Triton kernel
    apply_mask_gate_kernel[
        (num_blocks,)
    ](
        left_proj, right_proj, mask,
        left_gate, right_gate,
        left_fused, right_fused,
        bs, N, hidden_dim,
        left_proj.stride(0), left_proj.stride(1), left_proj.stride(2), left_proj.stride(3),
        right_proj.stride(0), right_proj.stride(1), right_proj.stride(2), right_proj.stride(3),
        mask.stride(0), mask.stride(1), mask.stride(2),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # ------------------------------------------------------------------
    # 4. Pairwise inner‑product over sequence dimension (einsum)
    #    Implemented via a batched GEMM for efficiency.
    # ------------------------------------------------------------------
    # reshape to [B, hidden_dim, N, N] → flatten batch & channel
    left_perm = left_fused.permute(0, 3, 1, 2).contiguous()   # [B, D, N, N]
    right_perm = right_fused.permute(0, 3, 1, 2).contiguous() # [B, D, N, N]

    B, D, N, _ = left_perm.shape
    left_flat  = left_perm.view(B * D, N, N)                # (B*D, N, N)
    right_flat = right_perm.view(B * D, N, N)               # (B*D, N, N)

    # Batched matrix multiplication: left @ right^T (sum over k)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Reshape back to [B, N, N, D]
    out = out_flat.view(B, D, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Second layer‑norm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    out = out * out_gate    # broadcast over the hidden dimension

    out = F.linear(out, weights["to_out.weight"])  # project back to `dim`

    return out
