def layernorm_kernel(x_ptr, out_ptr, weight_ptr, bias_ptr, N_TOKENS, C, epsilon,
                     stride_x_token, stride_y_token,
                     stride_x_c, stride_y_c,
                     BLOCK_SIZE: tl.constexpr):
    token_id = tl.program_id(0)
    if token_id >= N_TOKENS: return
    # Base offset for token start
    token_off = token_id * stride_x_token  # stride token = C maybe
    # Compute sum and sumsq for this token
    sum_val = tl.zeros((1,), dtype=tl.float32)
    sumsq_val = tl.zeros((1,), dtype=tl.float32)
    c = tl.arange(0, BLOCK_SIZE)
    for offset in range(0, C, BLOCK_SIZE):
        c_off = offset + c
        mask = c_off < C
        x = tl.load(x_ptr + token_off + c_off * stride_x_c,
                    mask=mask, other=0.0)
        sum_val += tl.sum(x)
        sumsq_val += tl.sum(x * x)
    mean = sum_val / C
    var = sumsq_val / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + epsilon)
    # Second pass to write normalized values
    for offset in range(0, C, BLOCK_SIZE):
        c_off = offset + c
        mask = c_off < C

                    mask=mask, other=0.0)

        if weight_ptr is not None:

        tl.store(out_ptr + token_off + c_off * stride_y_c, y, mask=mask)

def layernorm_kernel(x_ptr, y_ptr, weight_ptr, bias_ptr,
                     N_TOKENS: tl.constexpr,
                     C: tl.constexpr,
                     epsilon: tl.constexpr,
                     BLOCK_SIZE: tl.constexpr):
    token_idx = tl.program_id(0)
    if token_idx >= N_TOKENS:
        return
    # compute base offset:
    token_offset = token_idx * C

    # Initialize sum and sumsq
    sum_val = tl.zeros((1,), dtype=tl.float32)
    sumsq_val = tl.zeros((1,), dtype=tl.float32)

    # offsets for channel
    offs = tl.arange(0, BLOCK_SIZE)
    num_iters = (C + BLOCK_SIZE - 1) // BLOCK_SIZE

    for i in range(num_iters):
        c = i * BLOCK_SIZE + offs
        mask = c < C
        x = tl.load(x_ptr + token_offset + c, mask=mask, other=0.0)
        sum_val += tl.sum(x, axis=0)
        sumsq_val += tl.sum(x * x, axis=0)

    mean = sum_val / C
    var = sumsq_val / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + epsilon)

    # Normalize and write
    for i in range(num_iters):
        c = i * BLOCK_SIZE + offs
        mask = c < C
        x = tl.load(x_ptr + token_offset + c, mask=mask, other=0.0)

@triton.jit
def layernorm_kernel(x_ptr, y_ptr, weight_ptr, bias_ptr,
                     N_TOKENS, C, eps,
                     BLOCK_SIZE: tl.constexpr):
    token_idx = tl.program_id(0)
    if token_idx >= N_TOKENS:
        return

    token_offset = token_idx * C
    # sum over channels
    sum_val = tl.float32(0.0)
    sumsq_val = tl.float32(0.0)

    offs = tl.arange(0, BLOCK_SIZE)
    num_iters = (C + BLOCK_SIZE - 1) // BLOCK_SIZE

    for i in range(num_iters):
        c = i * BLOCK_SIZE + offs
        mask = c < C
        x = tl.load(x_ptr + token_offset + c, mask=mask, other=0.0)
        sum_val += tl.sum(x)
        sumsq_val += tl.sum(x * x)

    mean = sum_val / C
    var = sumsq_val / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    for i in range(num_iters):
        c = i * BLOCK_SIZE + offs
        mask = c < C
        x = tl.load(x_ptr + token_offset + c, mask=mask, other=0.0)
        y = (x - mean) * inv_std
        w = tl.load(weight_ptr + c, mask=mask, other=1.0)
        b = tl.load(bias_ptr + c, mask=mask, other=0.0)
        y = y * w + b
        tl.store(y_ptr + token_offset + c, y, mask=mask)
```

"""
TriMul (outgoing) forward pass accelerated with a custom Triton LayerNorm kernel.

Algorithm
---------
1. Layer‑norm the input tensor (shape [B,N,N,C]) using a Triton kernel.
2. Linear projections (left/right) from C → hidden_dim.
3. Optional mask multiplication (if config["nomask"] is False).
4. Gated linear layers: sigmoid(left_gate), sigmoid(right_gate), sigmoid(out_gate).
5. Apply the gates to left/right.
6. Core triangular multiplicative update:
   For each channel d, compute out[...,i,j,d] = Σ_k left[...,i,k,d] * right[...,j,k,d],
   which is a batched matrix multiplication:
   out = left.permute(0,3,1,2) @ right.permute(0,3,2,1).transpose(-2,-1)
7. Layer‑norm the intermediate result (hidden_dim) with another Triton kernel.
8. Multiply by out_gate and project back to the original dimension C.
9. Return the final tensor [B,N,N,C].

Only the LayerNorm steps are implemented in Triton; the heavy linear and
matrix‑multiplication work is delegated to PyTorch/CuBLAS for maximum
throughput.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------------- #
# Triton LayerNorm kernel (operates on a flattened [tokens, C] tensor)
# ----------------------------------------------------------------------------- #
@triton.jit
def layernorm_kernel(
    x_ptr,               # *float32, input [tokens, C]
    y_ptr,               # *float32, output [tokens, C]
    weight_ptr,          # *float32, LayerNorm weight (C)
    bias_ptr,            # *float32, LayerNorm bias   (C)
    tokens,              # i32, number of tokens = B*N*N
    C,                   # i32, channel dimension
    eps,                 # f32, epsilon
    BLOCK_SIZE: tl.constexpr,  # compile‑time block size for channel loops
):
    # One program per token (i.e. a single (b,i,j) slice)
    token_idx = tl.program_id(0)
    if token_idx >= tokens:
        return

    token_off = token_idx * C                     # offset of the token in memory

    # --------------------------------------------------------------------- #
    # 1) Compute mean and variance across C
    # --------------------------------------------------------------------- #
    sum_val = tl.float32(0.0)
    sumsq_val = tl.float32(0.0)

    offs = tl.arange(0, BLOCK_SIZE)               # [0 .. BLOCK_SIZE)
    num_iters = (C + BLOCK_SIZE - 1) // BLOCK_SIZE

    for it in range(num_iters):
        c = it * BLOCK_SIZE + offs                 # channel indices for this iteration
        mask = c < C                               # out‑of‑range guard
        x = tl.load(x_ptr + token_off + c, mask=mask, other=0.0)
        sum_val += tl.sum(x)                       # scalar reduction
        sumsq_val += tl.sum(x * x)

    mean = sum_val / C
    var = sumsq_val / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # --------------------------------------------------------------------- #
    # 2) Normalize, apply weight & bias, write back
    # --------------------------------------------------------------------- #
    for it in range(num_iters):
        c = it * BLOCK_SIZE + offs
        mask = c < C
        x = tl.load(x_ptr + token_off + c, mask=mask, other=0.0)
        y = (x - mean) * inv_std
        w = tl.load(weight_ptr + c, mask=mask, other=1.0)
        b = tl.load(bias_ptr + c, mask=mask, other=0.0)
        y = y * w + b
        tl.store(y_ptr + token_off + c, y, mask=mask)


# ----------------------------------------------------------------------------- #
# Custom kernel entry point
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask: torch.Tensor [B, N, N] (optional, float or bool)
            - weights: dict of model weights (all torch.Tensors on CUDA)
            - config: dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------------- #
    # Extract weights
    # --------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # --------------------------------------------------------------------- #
    # 1) LayerNorm on the input (C = dim)
    # --------------------------------------------------------------------- #
    B, N, _, _ = input_tensor.shape
    total_tokens = B * N * N
    x_flat = input_tensor.contiguous().view(total_tokens, dim)
    y_flat = torch.empty_like(x_flat)

    BLOCK_C = 64  # works for dim up to 384 (multiple iterations)
    layernorm_kernel[(total_tokens,)](
        x_flat,
        y_flat,
        norm_weight,
        norm_bias,
        total_tokens,
        dim,
        eps,
        BLOCK_SIZE=BLOCK_C,
        num_warps=4,
    )
    x_norm = y_flat.view(B, N, N, dim)

    # --------------------------------------------------------------------- #
    # 2) Linear projections (no bias)
    # --------------------------------------------------------------------- #
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # --------------------------------------------------------------------- #
    # 3) Optional mask (applied after projection, before gating)
    # --------------------------------------------------------------------- #
    if not config.get("nomask", True):
        # mask: [B,N,N] -> [B,N,N,1] broadcast
        mask_f = mask.to(dtype=x_norm.dtype, device=device).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------------- #
    # 4) Gating signals (sigmoid of linear transforms of x_norm)
    # --------------------------------------------------------------------- #
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 5) Core triangular multiplicative update via batched matmul
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Equivalent to: out = left @ right^T per channel.
    # --------------------------------------------------------------------- #
    # Permute to bring channel dimension into batch for bmm
    left_perm = left.permute(0, 3, 1, 2)      # [B, hidden, i, k]
    right_perm = right.permute(0, 3, 2, 1)    # [B, hidden, k, i] (transpose of right)

    # Merge B and hidden into a single batch dimension for torch.bmm
    left_2d = left_perm.reshape(B * hidden_dim, N, N)
    right_2d = right_perm.reshape(B * hidden_dim, N, N)

    # Batched matrix multiplication
    out_2d = torch.bmm(left_2d, right_2d)    # [B*hidden, N, N]

    # Reshape back to [B, N, N, hidden]
    out = out_2d.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B,N,N,hidden]

    # --------------------------------------------------------------------- #
    # 6) LayerNorm on hidden dimension
    # --------------------------------------------------------------------- #
    out_flat = out.contiguous().view(B * N * N, hidden_dim)
    out_norm_flat = torch.empty_like(out_flat)

    layernorm_kernel[(out_flat.shape[0],)](
        out_flat,
        out_norm_flat,
        to_out_norm_weight,
        to_out_norm_bias,
        out_flat.shape[0],
        hidden_dim,
        eps,
        BLOCK_SIZE=BLOCK_C,
        num_warps=4,
    )
    out_norm = out_norm_flat.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------------- #
    # 7) Final gating and projection back to original dim
    # --------------------------------------------------------------------- #
    out = out_norm * out_gate
    output = torch.nn.functional.linear(out, to_out_weight)  # [B,N,N,dim]

    return output
