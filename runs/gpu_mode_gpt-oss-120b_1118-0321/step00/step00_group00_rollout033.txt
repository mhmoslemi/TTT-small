@triton.jit
def layernorm_kernel(
    inp_ptr, out_ptr,
    weight_ptr, bias_ptr,
    M, D,
    stride_in_m, stride_in_d,
    stride_out_m, stride_out_d,
    BLOCK_D: tl.constexpr,
    EPS: tl.constexpr = 1e-5,
):
    row = tl.program_id(0)
    # compute base offset for row start
    inp_row_ptr = inp_ptr + row * stride_in_m
    out_row_ptr = out_ptr + row * stride_out_m

    # Initialize sum and sumsq
    sum_val = tl.zeros([1], dtype=tl.float32)
    sum_sq = tl.zeros([1], dtype=tl.float32)

    # Iterate over columns
    for off in range(0, D, BLOCK_D):
        cur_off = off + tl.arange(0, BLOCK_D)
        mask = cur_off < D
        x = tl.load(inp_row_ptr + cur_off * stride_in_d, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)

    mean = sum_val / D
    var = sum_sq / D - mean * mean
    rstd = tl.rsqrt(var + EPS)

    # Apply normalization and scale+bias
    for off in range(0, D, BLOCK_D):
        cur_off = off + tl.arange(0, BLOCK_D)
        mask = cur_off < D
        x = tl.load(inp_row_ptr + cur_off * stride_in_d, mask=mask, other=0.0).to(tl.float32)
        y = (x - mean) * rstd
        w = tl.load(weight_ptr + cur_off, mask=mask, other=0.0)
        b = tl.load(bias_ptr + cur_off, mask=mask, other=0.0)
        y = y * w + b
        tl.store(out_row_ptr + cur_off * stride_out_d, y.to(tl.float32), mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # Norm

    norm_bias = weights["norm.bias"]
    x_norm = layernorm_triton(input_tensor, norm_weight, norm_bias)  # shape [bs,N,N,dim]

    # Linear projections
    # flatten
    x_flat = x_norm.reshape(-1, dim)   # shape [bs*N*N, dim]
    left = torch.nn.functional.linear(x_flat, weights["left_proj.weight"]).reshape(bs,N,N,hidden_dim)
    right = torch.nn.functional.linear(x_flat, weights["right_proj.weight"]).reshape(bs,N,N,hidden_dim)

    # Mask
    if mask is not None and mask.numel() > 0:
        mask_expand = mask.unsqueeze(-1)  # shape [bs,N,N,1]
        left = left * mask_expand
        right = right * mask_expand

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, weights["left_gate.weight"])).reshape(bs,N,N,hidden_dim)
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, weights["right_gate.weight"])).reshape(bs,N,N,hidden_dim)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, weights["out_gate.weight"])).reshape(bs,N,N,hidden_dim)

    left = left * left_gate
    right = right * right_gate

    # Core multiplication
    # reshape to (B*hidden_dim, N, N)
    L = left.permute(0, 3, 1, 2).contiguous().reshape(bs * hidden_dim, N, N)
    R = right.permute(0, 3, 1, 2).contiguous().reshape(bs * hidden_dim, N, N)  # we need transpose of k dimension?
    # We need to transpose second dimension (k) to align with j. Actually we need right_T shape (bs*hidden_dim, N, N) where second axis is k and third axis is j.
    R_T = right.permute(0, 3, 2, 1).contiguous().reshape(bs * hidden_dim, N, N)
    # Now compute batched matmul:
    out_mat = torch.bmm(L, R_T)   # (B*hidden_dim,N,N)
    out = out_mat.view(bs, hidden_dim, N, N).permute(0,2,3,1).contiguous()  # shape (bs,N,N,hidden_dim)

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = layernorm_triton(out, to_out_norm_weight, to_out_norm_bias)  # shape (bs,N,N,hidden_dim)

    # gating
    out_norm = out_norm * out_gate

    # final linear
    out_flat = out_norm.reshape(-1, hidden_dim)   # shape (bs*N*N, hidden_dim)
    output = torch.nn.functional.linear(out_flat, weights["to_out.weight"]).reshape(bs, N, N, dim)

    return output

"""
TriMul (outgoing) forward implementation with Triton‑accelerated LayerNorm.

The kernel normalizes each vector (last dimension) of a 2‑D tensor
[M, D] (where M = B·N·N) using the standard LayerNorm formula:
    y = (x - mean) / sqrt(var + eps) * weight + bias

The main TriMul computation follows the reference implementation:

    x = LayerNorm(x)
    left  = Linear(x, left_proj)   * left_gate(x).sigmoid()
    right = Linear(x, right_proj)  * right_gate(x).sigmoid()
    left/right = left/right * mask   (if mask is present)
    out = left @ rightᵀ   (batched matmul over the K dimension)
    out = LayerNorm(out) * out_gate(x).sigmoid()
    out = Linear(out, to_out)

Only the LayerNorms are performed in Triton; the remaining
operations use highly‑optimized PyTorch kernels (linear, bmm,
sigmoid). This satisfies the requirement to implement *part* of the
operation in Triton while keeping the whole forward pass correct.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton LayerNorm kernel
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    inp_ptr,                # *const float*  [M, D] input
    out_ptr,                # *float*        [M, D] output
    weight_ptr,             # *const float*  [D]    gamma
    bias_ptr,               # *const float*  [D]    beta
    M,                      # total rows (runtime)
    stride_in_m, stride_in_d,   # input strides (elements)
    stride_out_m, stride_out_d, # output strides (elements)
    D: tl.constexpr,            # size of the normalized dimension (compile‑time)
    BLOCK_D: tl.constexpr,      # tile size for D
    EPS: tl.constexpr = 1e-5,   # epsilon
):
    row = tl.program_id(0)
    if row >= M:
        return

    # Base pointers for the current row
    inp_row_ptr = inp_ptr + row * stride_in_m
    out_row_ptr = out_ptr + row * stride_out_m

    # ---------- first pass: compute mean and variance ----------
    sum_val = tl.float32(0.0)
    sum_sq  = tl.float32(0.0)

    for off in range(0, D, BLOCK_D):
        col = off + tl.arange(0, BLOCK_D)
        mask = col < D
        x = tl.load(inp_row_ptr + col * stride_in_d, mask=mask).to(tl.float32)
        sum_val += tl.sum(x, axis=0)
        sum_sq  += tl.sum(x * x, axis=0)

    mean = sum_val / D
    var  = sum_sq / D - mean * mean
    rstd = tl.rsqrt(var + EPS)

    # ---------- second pass: write normalized values ----------
    for off in range(0, D, BLOCK_D):
        col = off + tl.arange(0, BLOCK_D)
        mask = col < D
        x = tl.load(inp_row_ptr + col * stride_in_d, mask=mask).to(tl.float32)
        y = (x - mean) * rstd
        w = tl.load(weight_ptr + col, mask=mask)
        b = tl.load(bias_ptr + col, mask=mask)
        y = y * w + b
        tl.store(out_row_ptr + col * stride_out_d, y, mask=mask)


def layernorm_triton(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    """
    Apply LayerNorm (γ=weight, β=bias) to the last dimension of `x`
    using the Triton kernel above.  `x` must be a 2‑D contiguous
    tensor of shape (M, D). Returns a tensor of the same shape and
    dtype (float32).
    """
    assert x.dim() == 2, "LayerNorm kernel expects a 2‑D tensor"
    M, D = x.shape
    out = torch.empty_like(x)

    # Strides in element units (not bytes)
    stride_in_m = x.stride(0)
    stride_in_d = x.stride(1)
    stride_out_m = out.stride(0)
    stride_out_d = out.stride(1)

    # Launch one program per row
    grid = (M,)
    layernorm_kernel[grid](
        x,
        out,
        weight,
        bias,
        M,
        stride_in_m,
        stride_in_d,
        stride_out_m,
        stride_out_d,
        D=D,
        BLOCK_D=128,               # block size for the feature dimension
        EPS=eps,
        num_warps=1,
    )
    return out


# ----------------------------------------------------------------------
# Custom TriMul forward entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (γ, β)
    # ------------------------------------------------------------------
    # flatten to (M, D) where M = bs * N * N
    x_flat = input_tensor.view(-1, dim)  # (M, dim)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = layernorm_triton(x_flat, norm_weight, norm_bias).view(bs, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Linear projections and gates (no bias)
    # ------------------------------------------------------------------
    x_flat = x_norm.reshape(-1, dim)  # reuse flattened view

    # Projections
    left_proj = F.linear(x_flat, weights["left_proj.weight"]).view(bs, N, N, hidden_dim)
    right_proj = F.linear(x_flat, weights["right_proj.weight"]).view(bs, N, N, hidden_dim)

    # Optional mask
    if not config.get("nomask", False):
        # mask: [bs, N, N] -> [bs, N, N, 1] and cast to same dtype
        mask_f = mask.to(dtype).unsqueeze(-1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # Gates (sigmoid)
    left_gate = torch.sigmoid(
        F.linear(x_flat, weights["left_gate.weight"]).view(bs, N, N, hidden_dim)
    )
    right_gate = torch.sigmoid(
        F.linear(x_flat, weights["right_gate.weight"]).view(bs, N, N, hidden_dim)
    )
    out_gate = torch.sigmoid(
        F.linear(x_flat, weights["out_gate.weight"]).view(bs, N, N, hidden_dim)
    )

    # Apply gating
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Free memory we no longer need
    del left_proj, right_proj, left_gate, right_gate, x_flat, x_norm

    # ------------------------------------------------------------------
    # 3) Core multiplicative update: out = left @ rightᵀ (batched GEMM)
    # ------------------------------------------------------------------
    # Reshape to (B*hidden_dim, N, N) for batched matmul
    L = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)   # (B, D, i, k)
    R = right.permute(0, 3, 2, 1).contiguous().view(bs * hidden_dim, N, N)  # (B, D, k, j)
    out_mat = torch.bmm(L, R)  # (B*D, N, N)
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, D)

    # ------------------------------------------------------------------
    # 4) Output LayerNorm and final gating
    # ------------------------------------------------------------------
    out_norm_weight = weights["to_out_norm.weight"]
    out_norm_bias = weights["to_out_norm.bias"]
    # Flatten for Triton LayerNorm
    out_flat = out.view(-1, hidden_dim)
    out_norm = layernorm_triton(out_flat, out_norm_weight, out_norm_bias).view(bs, N, N, hidden_dim)

    # Apply output gate
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 5) Final linear projection back to dim
    # ------------------------------------------------------------------
    out_norm_flat = out_norm.reshape(-1, hidden_dim)
    output = F.linear(out_norm_flat, weights["to_out.weight"]).view(bs, N, N, dim)

    return output
```
