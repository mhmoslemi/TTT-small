    stride_xb, stride_xi, stride_xj, stride_xc,

"""
TriMul forward (outgoing) implementation.

- LayerNorm on the input tensor (bs, N, N, dim) is performed by a
  custom Triton kernel (`layernorm_kernel`). This computes per‑vector
  mean/variance, applies the learned weight and bias and writes the
  normalized result.
- The remaining linear projections, sigmoidal gates and the final
  LayerNorm are carried out with PyTorch ops (which are already highly
  optimized). The core quadratic‑cubic reduction
  `out[i,j] = Σ_k left[i,k] * right[j,k]` is expressed as a batched
  matrix multiplication (`torch.bmm`) across the hidden dimension,
  which is equivalent to the original Einstein summation.
- The function `custom_kernel` follows the required signature and
  returns the processed tensor of shape (bs, N, N, dim).
"""

import torch
import triton
import triton.language as tl

@triton.jit
def layernorm_kernel(
    x_ptr, out_ptr,               # pointers to input and output tensors
    weight_ptr, bias_ptr,         # layernorm weight and bias (1‑D)
    stride_xb, stride_xi, stride_xj, stride_xc,  # strides for the 4‑D tensor
    EPS: tl.constexpr,            # epsilon for numerical stability
    BLOCK_C: tl.constexpr,        # size of the channel dimension (dim)
):
    # program IDs map to (batch, i, j)
    b = tl.program_id(0)
    i = tl.program_id(1)
    j = tl.program_id(2)

    # base offset for the vector x[b, i, j, :]
    base = b * stride_xb + i * stride_xi + j * stride_xj

    # offsets in the channel dimension
    offs = tl.arange(0, BLOCK_C)

    # ----- load the input vector -----
    x = tl.load(x_ptr + base + offs * stride_xc,
                mask=offs < BLOCK_C,
                other=0.0)                     # shape: [BLOCK_C]

    # ----- compute mean and variance -----
    sum_x = tl.sum(x, axis=0)                     # scalar
    mean = sum_x / BLOCK_C

    sum_x2 = tl.sum(x * x, axis=0)
    var = sum_x2 / BLOCK_C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # ----- load weight & bias (they are 1‑D) -----
    w = tl.load(weight_ptr + offs,
                mask=offs < BLOCK_C,
                other=0.0)
    b_val = tl.load(bias_ptr + offs,
                    mask=offs < BLOCK_C,
                    other=0.0)

    # ----- normalize, then apply affine transform -----
    y = (x - mean) * inv_std
    y = y * w + b_val

    # ----- store result -----
    tl.store(out_ptr + base + offs * stride_xc, y,
             mask=offs < BLOCK_C)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weight_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # unpack arguments
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    epsilon = config.get("eps", 1e-5)

    # -----------------------------------------------------------------
    # 1) LayerNorm on the input (implemented via Triton)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # allocate output tensor for the normalized input
    x_norm = torch.empty_like(input_tensor)

    # strides for the (bs, N, N, dim) layout
    stride_b, stride_i, stride_j, stride_c = input_tensor.stride()

    # launch the kernel: one program per (b, i, j) vector
    grid = (bs, N, N)
    layernorm_kernel[grid](
        input_tensor,
        x_norm,
        norm_weight,
        norm_bias,
        stride_b,
        stride_i,
        stride_j,
        stride_c,
        EPS=epsilon,
        BLOCK_C=dim,
        num_warps=8,
    )

    # -----------------------------------------------------------------
    # 2) Linear projections and gates (pure PyTorch)
    # -----------------------------------------------------------------
    # projection weights
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # projections
    left_proj = torch.nn.functional.linear(x_norm, left_proj_w)               # (bs,N,N,hidden_dim)
    right_proj = torch.nn.functional.linear(x_norm, right_proj_w)

    # gates
    left_gate = torch.nn.functional.linear(x_norm, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_w).sigmoid()

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------
    if mask is None or config.get("nomask", False):
        mask_tensor = torch.ones((bs, N, N), dtype=dtype, device=device)
    else:
        mask_tensor = mask.to(dtype)

    mask_tensor = mask_tensor.unsqueeze(-1)  # (bs, N, N, 1)

    left = left_proj * mask_tensor
    right = right_proj * mask_tensor

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 4) Core O(N³) reduction via batched matrix multiplication
    # -----------------------------------------------------------------
    # reshape to (bs, hidden_dim, N, N) for batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left_perm.view(bs * hidden_dim, N, N)                     # (B*H, N, N)
    right_flat = right_perm.view(bs * hidden_dim, N, N)                   # (B*H, N, N)

    # matrix multiplication with right side transposed on the K dimension
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))          # (B*H, N, N)

    # reshape back to (bs, N, N, hidden_dim)
    out = out_flat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 5) Final LayerNorm over the hidden dimension
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=epsilon,
    )

    # -----------------------------------------------------------------
    # 6) Apply output gate and final linear projection back to dim
    # -----------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)  # (bs, N, N, dim)

    return out
