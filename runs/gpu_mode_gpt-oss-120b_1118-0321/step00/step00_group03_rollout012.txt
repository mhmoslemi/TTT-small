"""
TriMul forward pass with a Triton kernel for the heavy N³ term.

Algorithm
---------
1. Layer‑norm the input (last dimension).
2. Linear projections: left, right (no bias).
3. Optional pairwise mask multiplication (broadcast on hidden dim).
4. Compute three gates (sigmoid of linear layers) and apply them.
5. Core operation:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a set of independent matrix multiplies per (batch, hidden) pair:
      out_d = left_d @ right_dᵀ
   Implemented with a Triton kernel that processes one (batch, hidden) pair
   per program and tiles the N×N output.
6. Layer‑norm over hidden dimension, gating, and final linear projection
   back to the original channel dimension.

The kernel performs the O(N³·C) part in fp16/fp32 mixed precision and
contributes the required “part in Triton”.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the batched per‑channel matrix multiplication:
#   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,               # *float32  [B, N, N, C]
    right_ptr,              # *float32  [B, N, N, C]
    out_ptr,                # *float32  [B, N, N, C]
    B, N,                   # batch size, sequence length
    stride_left_b, stride_left_i, stride_left_k, stride_left_c,
    stride_right_b, stride_right_i, stride_right_k, stride_right_c,
    stride_out_b, stride_out_i, stride_out_j, stride_out_c,
    BLOCK_M: tl.constexpr,  # tile size in i dimension
    BLOCK_N: tl.constexpr,  # tile size in j dimension
    BLOCK_K: tl.constexpr,  # tile size in k dimension
    HIDDEN_DIM: tl.constexpr # hidden dimension (C)
):
    pid_z = tl.program_id(2)                # combined batch and channel index
    batch = pid_z // HIDDEN_DIM               # batch index
    hid   = pid_z %  HIDDEN_DIM               # hidden channel index

    pid_x = tl.program_id(0)                # row tile id
    pid_y = tl.program_id(1)                # col tile id

    # --------------------------------------------------------------
    # Offsets for the output tile (i, j)
    # --------------------------------------------------------------
    offs_i = pid_x * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_y * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator for the output tile (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over k dimension
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_idx in range(0, num_k_tiles):
        k_base = k_idx * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left tile (i, k) -----
        left_ptrs = (
            left_ptr
            + batch * stride_left_b
            + (offs_i[:, None] * stride_left_i)
            + (offs_k[None, :] * stride_left_k)
            + hid * stride_left_c
        )
        left = tl.load(
            left_ptrs,
            mask=(mask_i[:, None] & mask_k[None, :]),
            other=0.0,
            dtype=tl.float32,
        )

        # ----- load right tile (k, j)  (transposed view of right) -----
        # right shape is (batch, j, k, C); we need (k, j)
        right_ptrs = (
            right_ptr
            + batch * stride_right_b
            + (offs_j[None, :] * stride_right_i)   # j dimension as columns
            + (offs_k[:, None] * stride_right_k)   # k dimension as rows
            + hid * stride_right_c
        )
        right = tl.load(
            right_ptrs,
            mask=(mask_k[:, None] & mask_j[None, :]),
            other=0.0,
            dtype=tl.float32,
        )

        # ----- dot‑product accumulation over the k tile -----
        acc += tl.dot(left, right)   # (M, K) @ (K, N) -> (M, N)

    # --------------------------------------------------------------
    # Store the completed output tile
    # --------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + batch * stride_out_b
        + (offs_i[:, None] * stride_out_i)
        + (offs_j[None, :] * stride_out_j)
        + hid * stride_out_c
    )
    tl.store(out_ptrs, acc, mask=(mask_i[:, None] & mask_j[None, :]))


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul module with a Triton kernel for the
    cubic pairwise multiplication.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N] or None
        - weights      : dict of torch.Tensors containing all model parameters
        - config       : dict with at least ``dim`` and ``hidden_dim`` keys

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper to fetch weight tensors
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight   = weights["left_proj.weight"]
    right_proj_weight  = weights["right_proj.weight"]
    left_gate_weight   = weights["left_gate.weight"]
    right_gate_weight  = weights["right_gate.weight"]
    out_gate_weight    = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Optional mask (mask is float, 0/1)
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask_f = mask.to(dtype).unsqueeze(-1)          # [B, N, N, 1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core cubic multiplication via Triton
    # ------------------------------------------------------------------
    # Ensure contiguous tensors for consistent stride extraction
    left  = left.contiguous()
    right = right.contiguous()
    out   = torch.empty((B, N, N, hidden_dim), dtype=dtype, device=device)

    # Strides (in elements)
    lsb, lsi, lsk, lsc = left.stride()
    rsb, rsi, rsk, rsc = right.stride()
    osb, osi, osj, osc = out.stride()

    # Tunable block sizes (fit well on typical GPUs)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        lsb,
        lsi,
        lsk,
        lsc,
        rsb,
        rsi,
        rsk,
        rsc,
        osb,
        osi,
        osj,
        osc,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HIDDEN_DIM=hidden_dim,
    )

    # ------------------------------------------------------------------
    # 6. Post‑processing: LayerNorm, gating, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    output = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return output
