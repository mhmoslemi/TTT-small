def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k,
    stride_r_b, stride_r_i, stride_r_k,
    stride_o_b, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    b = pid_bh // H
    h = pid_bh % H

    # offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base ptr for batch and hidden
    a_ptr_base = left_ptr + b * stride_l_b + h
    b_ptr_base = right_ptr + b * stride_r_b + h
    c_ptr_base = out_ptr + b * stride_o_b + h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load A tile (M,K)
        a_ptrs = a_ptr_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load B tile (N,K)
        b_ptrs = b_ptr_base + offs_j[:, None] * stride_r_i + offs_k[None, :] * stride_r_k
        b = tl.load(b_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        # b is shape (N,K). Need (K,N) for dot.
        acc += tl.dot(a, tl.trans(b))

    # store result
    c_ptrs = c_ptr_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
Triton implementation of the outgoing TriMul operation (AlphaFold3).

The forward pass consists of:
1. LayerNorm on the input tensor.
2. Two linear projections (left/right) followed by optional masking.
3. Gating with sigmoid‑activated linear layers.
4. A fused reduction over the sequence dimension:
   out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   The reduction is performed by a custom Triton kernel that executes a
   batched matrix multiplication for each hidden channel h.
5. LayerNorm on the hidden dimension, output gating and a final linear
   projection back to the original channel dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k,
    stride_r_b, stride_r_i, stride_r_k,
    stride_o_b, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute, for a single hidden channel, the matrix product
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].

    Grid dimensions:
      axis 0 -> i‑tile index
      axis 1 -> j‑tile index
      axis 2 -> combined batch*hidden index (b * H + h)
    """
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    b = pid_bh // H            # batch index
    h = pid_bh % H             # hidden‑channel index

    # Offsets for the i and j dimensions of this tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for the current batch and hidden channel
    a_ptr_base = left_ptr + b * stride_l_b + h                # left[b, :, :, h]
    b_ptr_base = right_ptr + b * stride_r_b + h               # right[b, :, :, h]
    c_ptr_base = out_ptr + b * stride_o_b + h                 # out[b, :, :, h]

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load a tile of shape (BLOCK_M, BLOCK_K) from left
        a_ptrs = a_ptr_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # Load a tile of shape (BLOCK_N, BLOCK_K) from right
        b_ptrs = b_ptr_base + offs_j[:, None] * stride_r_i + offs_k[None, :] * stride_r_k
        b = tl.load(b_ptrs,
                    mask=mask_j[:, None] & mask_k[None, :],
                    other=0.0)

        # b is (N, K); transpose to (K, N) for tl.dot
        acc += tl.dot(a, tl.trans(b))

    # Store the result tile (BLOCK_M, BLOCK_N)
    c_ptrs = c_ptr_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(c_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a Triton kernel for the
    core N³ reduction.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (left/right) and optional masking
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)      # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 3) Gating (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4) Triton kernel: compute Σ_k left[...,k,:] * right[...,k,:]
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    B, N, _, _ = left.shape

    out = torch.empty((B, N, N, hidden_dim),
                      dtype=left.dtype,
                      device=left.device)

    # Strides in number of elements (not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Block sizes (tuned for typical seq lengths)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid_i = (N + BLOCK_M - 1) // BLOCK_M
    grid_j = (N + BLOCK_N - 1) // BLOCK_N
    grid_bh = B * hidden_dim

    tri_mul_matmul_kernel[(grid_i, grid_j, grid_bh)](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_r_b,
        stride_r_i,
        stride_r_k,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 5) Second LayerNorm on hidden_dim, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
