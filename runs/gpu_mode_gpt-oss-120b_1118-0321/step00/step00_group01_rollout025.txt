@triton.jit
def triton_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr
):
    # Axis mapping
    bh_idx = tl.program_id(0)  # batch * hidden
    i_tile = tl.program_id(1)
    j_tile = tl.program_id(2)

    # compute batch and hidden index
    b = bh_idx // H
    h = bh_idx % H

    # compute start indices
    i_start = i_tile * BLOCK_I
    j_start = j_tile * BLOCK_J

    # range indices
    i_range = i_start + tl.arange(0, BLOCK_I)
    j_range = j_start + tl.arange(0, BLOCK_J)

    # mask for base ranges in bounds
    i_mask = i_range < N
    j_mask = j_range < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # Loop over k dimension
    for k_start in range(0, N, BLOCK_K):
        # k range
        k_range = k_start + tl.arange(0, BLOCK_K)
        k_mask = k_range < N

        # compute addresses for left block: shape (BLOCK_I, BLOCK_K)
        # left address = base + i*stride_i + k*stride_k + h*stride_h
        # base offset includes batch offset
        left_base = left_ptr + b * stride_left_b + h * stride_left_h

        # left offsets matrix
        left_offsets = i_range[:,None] * stride_left_i + k_range[None,:] * stride_left_k

        left_ptrs = left_base + left_offsets

        left = tl.load(left_ptrs, mask = (i_mask[:,None] & k_mask[None,:]), other=0.0)

        # right block: shape (BLOCK_J, BLOCK_K)
        right_base = right_ptr + b * stride_right_b + h * stride_right_h

        right_offsets = j_range[:,None] * stride_right_j + k_range[None,:] * stride_right_k

        right_ptrs = right_base + right_offsets

        # load and mask
        right = tl.load(right_ptrs, mask = (j_mask[:,None] & k_mask[None,:]), other=0.0)

        # Compute matmul contribution. left: (I,K), right: (J,K) -> we need to dot left with right^T
        # Use trans_b=True
        acc += tl.dot(left, right, trans_b=True)

    # Store result
    # out pointer
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    out_offsets = i_range[:,None] * stride_out_i + j_range[None,:] * stride_out_j
    out_ptrs = out_base + out_offsets

    tl.store(out_ptrs, acc, mask = (i_mask[:,None] & j_mask[None,:]))

"""
Triton-accelerated TriMul forward (outgoing) implementation.
The kernel computes the tensor contraction:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
where left/right are the projected and gated pairwise representations.
The rest of the TriMul pipeline (layernorms, linear projections and gates)
is performed in PyTorch; only the O(N³·H) contraction is off‑loaded to
Triton to achieve high throughput.
"""

import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    # 3‑D grid: (batch*hidden, i‑tile, j‑tile)
    bh_idx = tl.program_id(0)
    i_tile = tl.program_id(1)
    j_tile = tl.program_id(2)

    # map linear id to batch and hidden dimension
    b = bh_idx // H
    h = bh_idx % H

    # start indices of the current tile
    i_start = i_tile * BLOCK_I
    j_start = j_tile * BLOCK_J

    # range vectors for the tile
    i_range = i_start + tl.arange(0, BLOCK_I)
    j_range = j_start + tl.arange(0, BLOCK_J)

    i_mask = i_range < N
    j_mask = j_range < N

    # accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # base pointers for this (b, h) slice
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # loop over the reduction dimension k
    for k_start in range(0, N, BLOCK_K):
        k_range = k_start + tl.arange(0, BLOCK_K)
        k_mask = k_range < N

        # load left tile   shape (BLOCK_I, BLOCK_K)
        left_off = i_range[:, None] * stride_left_i + k_range[None, :] * stride_left_k
        left = tl.load(left_base + left_off,
                       mask=(i_mask[:, None] & k_mask[None, :]),
                       other=0.0)

        # load right tile  shape (BLOCK_J, BLOCK_K)
        right_off = j_range[:, None] * stride_right_i + k_range[None, :] * stride_right_k
        right = tl.load(right_base + right_off,
                        mask=(j_mask[:, None] & k_mask[None, :]),
                        other=0.0)

        # matmul: (I,K) @ (K,J)ᵀ  → (I,J)
        acc += tl.dot(left, right, trans_b=True)

    # store the result
    out_off = i_range[:, None] * stride_out_i + j_range[None, :] * stride_out_j
    tl.store(out_base + out_off,
             acc,
             mask=(i_mask[:, None] & j_mask[None, :]))


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module, using a Triton kernel for the
    O(N³·hidden) contraction and PyTorch for LayerNorm / Linear / Sigmoid.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N] (0/1)
        - weights      : dict of torch.Tensor containing all model parameters
        - config       : dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, on the same CUDA device).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. LayerNorm over the channel dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gating values (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )  # shape [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask is [B,N,N]; make it float and add a trailing dim
        mask_ = mask.unsqueeze(-1).to(left_proj.dtype)
        left_proj = left_proj * mask_
        right_proj = right_proj * mask_

    # ------------------------------------------------------------------
    # 5. Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left_proj * left_gate   # [B,N,N,hidden_dim]
    right = right_proj * right_gate

    # clean up temporaries that are no longer needed to free GPU memory
    del left_proj, right_proj, left_gate, right_gate, x

    # ------------------------------------------------------------------
    # 6. Core O(N³·H) contraction via Triton
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    B, N1, N2, H = left.shape
    assert N1 == N2 == N, "pairwise tensor must be square"
    assert H == hidden_dim, "hidden dimension mismatch"

    out = torch.empty_like(left)   # [B,N,N,hidden_dim]

    # strides in element units
    s_left = left.stride()
    s_right = right.stride()
    s_out = out.stride()
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = s_left
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = s_right
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = s_out

    # Block sizes – tuned for the B200 GPU
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )

    tri_mul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out_normed = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # Apply the output gate
    out_gated = out_normed * out_gate

    # Final linear projection back to `dim`
    to_out_weight = weights["to_out.weight"]
    output = torch.nn.functional.linear(out_gated, to_out_weight)

    return output

"""
Triton‑accelerated TriMul forward (outgoing) implementation.

The heavy O(N³·H) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed in a custom Triton kernel.  The surrounding
LayerNorms, linear projections and sigmoid gates are handled with
standard PyTorch ops.  This layout matches the reference AlphaFold3
TriMul module while keeping the kernel fused and memory‑coalesced.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel that computes the einsum contraction.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    # 3‑D grid: (batch*hidden, i‑tile, j‑tile)
    bh_idx = tl.program_id(0)          # index for (batch, hidden)
    i_tile = tl.program_id(1)          # tile along i dimension
    j_tile = tl.program_id(2)          # tile along j dimension

    # map linear id to batch and hidden index
    b = bh_idx // H
    h = bh_idx % H

    # start offsets of the current tile
    i_start = i_tile * BLOCK_I
    j_start = j_tile * BLOCK_J

    # 1‑D index vectors for the tile
    i_range = i_start + tl.arange(0, BLOCK_I)
    j_range = j_start + tl.arange(0, BLOCK_J)

    i_mask = i_range < N
    j_mask = j_range < N

    # accumulator for the partial result
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # base pointers for this (b, h) slice
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # reduction over the k dimension
    for k_start in range(0, N, BLOCK_K):
        k_range = k_start + tl.arange(0, BLOCK_K)
        k_mask = k_range < N

        # ---- load left tile (I, K) ------------------------------------
        left_off = i_range[:, None] * stride_left_i + k_range[None, :] * stride_left_k
        left = tl.load(left_base + left_off,
                       mask=(i_mask[:, None] & k_mask[None, :]),
                       other=0.0)

        # ---- load right tile (J, K) -----------------------------------
        right_off = j_range[:, None] * stride_right_i + k_range[None, :] * stride_right_k
        right = tl.load(right_base + right_off,
                        mask=(j_mask[:, None] & k_mask[None, :]),
                        other=0.0)

        # ---- matrix multiply: (I,K) @ (K,J).T -> (I,J) ----------------
        # trans_b=True transposes the second operand, turning (J,K) into (K,J)
        acc += tl.dot(left, right, trans_b=True)

    # ---- write the result ------------------------------------------------
    out_off = i_range[:, None] * stride_out_i + j_range[None, :] * stride_out_j
    tl.store(out_base + out_off,
             acc,
             mask=(i_mask[:, None] & j_mask[None, :]))


# ----------------------------------------------------------------------
# Public entry‑point used by the evaluator.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N] (0/1)
        - weights      : dict of torch.Tensor containing all model parameters
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional
                         ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, on the same CUDA device).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm over the channel dimension (dim)
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2. Linear projections (bias‑less)
    # --------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Gates (sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )  # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 4. Optional mask (broadcast on hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        mask_f = mask.unsqueeze(-1).to(left_proj.dtype)   # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # --------------------------------------------------------------
    # 5. Apply gates
    # --------------------------------------------------------------
    left = left_proj * left_gate      # [B, N, N, hidden_dim]
    right = right_proj * right_gate

    # free temporaries that are no longer needed
    del left_proj, right_proj, left_gate, right_gate, x

    # --------------------------------------------------------------
    # 6. Core O(N³·H) contraction via Triton
    # --------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    B, N1, N2, H = left.shape
    assert N1 == N2 == N, "pairwise dimensions must be square"
    assert H == hidden_dim, "hidden dimension mismatch"

    out = torch.empty_like(left)   # [B, N, N, hidden_dim]

    # strides in element units
    s_left = left.stride()
    s_right = right.stride()
    s_out = out.stride()
    (stride_left_b, stride_left_i, stride_left_k, stride_left_h) = s_left
    (stride_right_b, stride_right_i, stride_right_k, stride_right_h) = s_right
    (stride_out_b, stride_out_i, stride_out_j, stride_out_h) = s_out

    # Tuned block sizes for the target GPU
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                     # programs over (batch, hidden)
        (N + BLOCK_I - 1) // BLOCK_I,       # i‑tiles
        (N + BLOCK_J - 1) // BLOCK_J        # j‑tiles
    )

    tri_mul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate and final projection back to dim
    # --------------------------------------------------------------
    out_normed = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out_gated = out_normed * out_gate

    output = torch.nn.functional.linear(out_gated, weights["to_out.weight"])

    return output
