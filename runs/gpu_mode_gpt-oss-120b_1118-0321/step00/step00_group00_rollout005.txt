"""
TriMul forward (outgoing) implementation.

The heavy N³ reduction is performed by a custom Triton kernel that computes,
for each batch‑hidden index, the matrix product
    out[b·H + h] = left[b·H + h] @ right[b·H + h].T
where left/right are the gated projections of the input tensor.
All other operations (LayerNorm, linear projections, sigmoids,
masking and final projection) are done in PyTorch.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix sizes (M=N=seq_len, K=seq_len)
    stride_ab, stride_am, stride_ak,   # strides for A (batch, M, K)
    stride_bb, stride_bk, stride_bn,   # strides for B (batch, K, N)   (B is stored transposed)
    stride_cb, stride_cm, stride_cn,   # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched half‑precision matrix multiplication with accumulation in fp32."""
    batch_id = tl.program_id(0)       # which (batch, hidden) pair
    pid_m = tl.program_id(1)          # block row index
    pid_n = tl.program_id(2)          # block col index

    # offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # start pointers for this batch
    a_batch_ptr = A_ptr + batch_id * stride_ab
    b_batch_ptr = B_ptr + batch_id * stride_bb
    c_batch_ptr = C_ptr + batch_id * stride_cb

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # tile pointers
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

        # masks for out‑of‑bounds loads
        a_mask = (offs_m < M)[:, None] & (offs_k < K)[None, :]
        b_mask = (offs_k < K)[:, None] & (offs_n < N)[None, :]

        a = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.float32)   # (BM, BK)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0).to(tl.float32)   # (BK, BN)

        acc += tl.dot(a, b)

    # store result (back to fp16)
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    c_mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor: torch.Tensor [B, N, N, dim]
        - mask: torch.Tensor [B, N, N] (or None)
        - weights: dict of torch.Tensor containing all linear / norm parameters
        - config: dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim]
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype_fp16 = torch.float16

    # ------------------------------------------------------------------
    # Unpack configuration and weights
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Normalization parameters
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)

    # Linear projection weights
    left_proj_w = weights["left_proj.weight"].to(device)
    right_proj_w = weights["right_proj.weight"].to(device)

    # Gate weights
    left_gate_w = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w = weights["out_gate.weight"].to(device)

    # Output normalization & projection
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    to_out_w = weights["to_out.weight"].to(device)

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dimension
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim] in fp32

    # ------------------------------------------------------------------
    # 2) Linear projections + optional mask
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x_norm, left_proj_w).to(dtype_fp16)   # [B,N,N,hd]
    right = torch.nn.functional.linear(x_norm, right_proj_w).to(dtype_fp16) # [B,N,N,hd]

    if not config.get("nomask", True):
        # mask is bool/float; broadcast to feature dim
        mask = mask_tensor.to(dtype=left.dtype, device=device).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 3) Gates (sigmoid) and application
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_w)).to(dtype_fp16)
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_w)).to(dtype_fp16)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_w))    # keep fp32 for later

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4) Prepare tensors for batched matmul (A @ B.T)
    # ------------------------------------------------------------------
    # Rearrange to [B, hidden_dim, N, N] then merge batch & hidden into one dimension
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, hd, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, hd, N, N]

    B, hd, N, _ = left.shape
    batch_dim = B * hd

    left_flat = left.view(batch_dim, N, N)   # [B*hd, N, N]
    right_flat = right.view(batch_dim, N, N) # [B*hd, N, N]

    # Output buffer (half‑precision)
    out_flat = torch.empty_like(left_flat, dtype=dtype_fp16)

    # ------------------------------------------------------------------
    # 5) Triton batched matrix multiplication
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (batch_dim,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,  # M, N, K
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Reshape back to [B, N, N, hidden_dim] (float32 for following ops)
    # ------------------------------------------------------------------
    out = out_flat.view(B, hd, N, N).permute(0, 2, 3, 1).to(torch.float32)  # [B,N,N,hd]

    # ------------------------------------------------------------------
    # 7) Output normalization, gating and final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)  # [B,N,N,dim]

    return out
```
