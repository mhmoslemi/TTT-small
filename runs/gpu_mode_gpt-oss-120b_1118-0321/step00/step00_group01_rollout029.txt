    N, H,
    stride_left_i, stride_left_k, stride_left_h,
    stride_right_i, stride_right_k, stride_right_h,

"""
TriMul (outgoing) forward pass implemented with a fused Triton kernel.

The expensive part of the TriMul operation is the contraction
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
for each batch b, sequence indices i, j and hidden dimension d.
We implement this contraction as a Triton kernel where each program
instance (i, j) computes one output vector of length `hidden_dim`
by looping over the `k` dimension in blocks.  The rest of the
TriMul pipeline (LayerNorm, linear projections, gating, final
LayerNorm, output gating and linear projection) is performed with
standard PyTorch primitives.

The kernel loads a (BLOCK_K, hidden_dim) tile from `left` and the
corresponding tile from `right`, multiplies element‑wise, reduces over
the K dimension and accumulates the result.  After all K‑tiles have
been processed, the hidden‑dim vector is stored to the output tensor.
This approach avoids any explicit three‑nested Python loops and
leverages the GPU’s high arithmetic throughput.

The `custom_kernel` entry point follows the required signature:
    data = (input_tensor, mask_tensor, weights_dict, config_dict)
and returns the processed tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,                     # sequence length (runtime)
    stride_left_i, stride_left_k, stride_left_h,
    stride_right_i, stride_right_k, stride_right_h,
    stride_out_i, stride_out_j, stride_out_h,
    BLOCK_K: tl.constexpr,  # tile size along K
    H: tl.constexpr,        # hidden dimension (compile‑time)
):
    # program IDs give the i, j coordinates
    i = tl.program_id(0)
    j = tl.program_id(1)

    # bounds check (grid is sized exactly to N, but guard for safety)
    if i >= N or j >= N:
        return

    # offsets for the two rows we are processing
    off_i = i * stride_left_i          # offset for left[i, :, :]
    off_j = j * stride_right_i         # offset for right[j, :, :]

    # hidden‑dim index range
    h_range = tl.arange(0, H)

    # accumulator for the hidden‑dim vector (float32 accumulator)
    acc = tl.zeros([H], dtype=tl.float32)

    # number of K‑tiles to cover the sequence dimension
    num_tiles = tl.cdiv(N, BLOCK_K)

    for tile_idx in range(num_tiles):
        k_start = tile_idx * BLOCK_K
        k_offset = k_start + tl.arange(0, BLOCK_K)          # shape (BLOCK_K,)

        # mask for out‑of‑bounds k (only needed for the last tile)
        k_mask = k_offset < N                               # (BLOCK_K,)

        # pointers for the current K‑tile
        left_tile_ptr  = left_ptr + off_i + k_offset[:, None] * stride_left_k + h_range[None, :] * stride_left_h
        right_tile_ptr = right_ptr + off_j + k_offset[:, None] * stride_right_k + h_range[None, :] * stride_right_h

        # load the tiles; values outside bounds are zeroed
        left_tile  = tl.load(left_tile_ptr,  mask=k_mask[:, None], other=0.0)   # (BLOCK_K, H)
        right_tile = tl.load(right_tile_ptr, mask=k_mask[:, None], other=0.0)   # (BLOCK_K, H)

        # element‑wise multiply and reduce over K dimension
        acc += tl.sum(left_tile * right_tile, axis=0)       # (H,)

    # store the resulting hidden‑dim vector
    out_ptrs = out_ptr + i * stride_out_i + j * stride_out_j + h_range * stride_out_h
    tl.store(out_ptrs, acc)


def _tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the TriMul contraction using the Triton kernel.
    Arguments:
        left, right: tensors of shape [B, N, N, H] (float32, contiguous)
    Returns:
        out: tensor of shape [B, N, N, H] (float32)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # Kernel launch configuration
    BLOCK_K = 64                 # can be tuned; works well for the given sizes
    grid = (N, N)                # one program per (i, j)

    for b in range(B):
        left_b  = left[b].contiguous()
        right_b = right[b].contiguous()
        out_b   = out[b].contiguous()

        # strides in elements (not bytes)
        l_stride_i, l_stride_k, l_stride_h = left_b.stride()
        r_stride_i, r_stride_k, r_stride_h = right_b.stride()
        o_stride_i, o_stride_j, o_stride_h = out_b.stride()

        tri_mul_kernel[grid](
            left_b, right_b, out_b,
            N,
            l_stride_i, l_stride_k, l_stride_h,
            r_stride_i, r_stride_k, r_stride_h,
            o_stride_i, o_stride_j, o_stride_h,
            BLOCK_K=BLOCK_K,
            H=H,
            num_warps=4,          # typical choice; can be increased on larger GPUs
        )
        # write back into the output tensor
        out[b] = out_b

    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask_tensor: torch.Tensor of shape [B, N, N] or None
            - weights_dict: mapping from weight names to tensors
            - config_dict: contains 'dim', 'hidden_dim', and 'nomask' flag

    Returns:
        output: torch.Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----- LayerNorm over the channel dimension -----
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # ----- Linear projections (no bias) -----
    left_proj_weight  = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]  # (hidden_dim, dim)

    left  = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B, N, N, hidden_dim]

    # ----- Optional masking -----
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ----- Gating values -----
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # [B,N,N,hidden_dim]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))    # [B,N,N,hidden_dim]

    left  = left * left_gate
    right = right * right_gate

    # ----- Core contraction using Triton -----
    out = _tri_mul_triton(left, right)   # [B, N, N, hidden_dim]

    # ----- Final LayerNorm, output gating and linear projection -----
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)

    out = out * out_gate                     # output gating
    to_out_weight = weights["to_out.weight"] # (dim, hidden_dim)
    output = F.linear(out, to_out_weight)    # final projection to [B, N, N, dim]

    return output
