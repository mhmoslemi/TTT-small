@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_b, stride_i, stride_k, stride_d,
    N, hidden_dim,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_bc = tl.program_id(0)  # batch*hidden
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    total_bc = tl.num_programs(0) # B*hidden

    # Compute batch and channel indices
    batch_idx = pid_bc // hidden_dim
    channel_idx = pid_bc % hidden_dim

    # Compute base offsets
    offset_batch = batch_idx * stride_b
    offset_channel = channel_idx * stride_d

    # Compute tile start indices
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # Create rangelists
    i_range = i_start + tl.arange(0, BLOCK_M)
    j_range = j_start + tl.arange(0, BLOCK_N)
    k_range = tl.arange(0, BLOCK_K)  # will be added in loop

    # Create mask for i and j against N
    mask_i = i_range < N
    mask_j = j_range < N

    # Offsets for pointers (assuming left_ptr base points to start of tensor)
    # left: dims [B,N,N,D]; stride_i: stride of first N (i), stride_k: stride of second N (k)
    a_ptrs = left_ptr + offset_batch + offset_channel + i_range[:, None] * stride_i + k_range[None, :] * stride_k
    # right: we need (k,j) layout
    b_ptrs = right_ptr + offset_batch + offset_channel + k_range[:, None] * stride_k + j_range[None, :] * stride_i

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k_tile * BLOCK_K
        k_idx = k_offset + k_range
        mask_k = k_idx < N

        # Load A tile one
        a = tl.load(a_ptrs + k_offset * stride_k, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        # Load B tile
        b = tl.load(b_ptrs + k_offset * stride_k, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Compute dot product (M x K) * (K x N) -> (M x N)
        acc += tl.dot(a, b)

    # Store output
    c_ptrs = out_ptr + offset_batch + offset_channel + i_range[:, None] * stride_i + j_range[None, :] * stride_i
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

```
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_b, stride_i, stride_k, stride_d,
    N, hidden_dim,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_bc = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Compute batch and channel indices
    batch_idx = pid_bc // hidden_dim
    channel_idx = pid_bc % hidden_dim

    # Offsets
    offset_batch = batch_idx * stride_b
    offset_channel = channel_idx * stride_d

    # Tile start indices
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # Ranges
    i_range = i_start + tl.arange(0, BLOCK_M)
    j_range = j_start + tl.arange(0, BLOCK_N)

    # Masks for boundaries
    mask_i = i_range < N
    mask_j = j_range < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        k_range = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # Compute pointers
        a_ptrs = left_ptr + offset_batch + offset_channel + i_range[:, None] * stride_i + k_range[None, :] * stride_k
        b_ptrs = right_ptr + offset_batch + offset_channel + k_range[:, None] * stride_k + j_range[None, :] * stride_i

        # Load tiles with masking
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # dot
        acc += tl.dot(a, b)

    # Store results
    c_ptrs = out_ptr + offset_batch + offset_channel + i_range[:, None] * stride_i + j_range[None, :] * stride_i
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                   stride_b, stride_i, stride_k, stride_d,
                   N, hidden_dim,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    """
    Compute out[b,i,j,d] = sum_{k=0}^{N-1} left[b,i,k,d] * right[b,j,k,d]
    using tiled matmul (A @ B^T) for every d.
    left, right, out are stored as [B, N, N, D], where D = hidden_dim.
    """
    pid_bc = tl.program_id(0)  # batch * hidden_dim index
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Recover batch and channel
    batch_idx = pid_bc // hidden_dim
    channel_idx = pid_bc % hidden_dim

    # Base offsets for this (batch, channel)
    offset_batch = batch_idx * stride_b
    offset_channel = channel_idx * stride_d

    # Tile start coordinates
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # Index ranges within the tile
    i_range = i_start + tl.arange(0, BLOCK_M)
    j_range = j_start + tl.arange(0, BLOCK_N)

    # Boundary masks for i and j
    mask_i = i_range < N
    mask_j = j_range < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        k_range = k_start + tl.arange(0, BLOCK_K)

"""
TriMul (outgoing) forward pass implemented with a custom Triton kernel.

Algorithm
---------
1. Apply LayerNorm on the input tensor.
2. Linear projections produce `left` and `right` of shape [B, N, N, H].
3. (Optional) mask these tensors.
4. Compute gating tensors (sigmoid of linear projections) and multiply
   with `left`/`right`.
5. Core operation: for every batch `b` and hidden channel `h`
   compute  out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] .
   This is a batch of matrix multiplications `A @ Bᵀ` where
   A = left[..., h] and B = right[..., h].
   The computation is tiled with a Triton kernel (BLOCK_M × BLOCK_N tiles,
   reduction over K in BLOCK_K chunks) and accumulated in float32.
6. Layer‑norm on the hidden dimension, multiply by the output gate and
   project back to the original channel dimension.

Only the O(N³) reduction is performed in Triton; all other operations are
handled by PyTorch to keep the implementation concise.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,
    right_ptr,
    out_ptr,
    stride_b,
    stride_i,
    stride_k,
    stride_d,
    N,
    hidden_dim,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    for a single hidden channel `h`.  The kernel processes all batches
    and all hidden channels by iterating over a combined index
    `pid_bc = batch * hidden_dim + h`.
    """
    pid_bc = tl.program_id(0)   # batch * hidden_dim
    pid_i = tl.program_id(1)    # tile index in the i (row) dimension
    pid_j = tl.program_id(2)    # tile index in the j (col) dimension

    # Recover batch and hidden‑channel indices
    batch_idx = pid_bc // hidden_dim
    chan_idx = pid_bc % hidden_dim

    # Base offsets for this (batch, channel) pair
    offset_batch = batch_idx * stride_b
    offset_chan = chan_idx * stride_d

    # Tile start positions
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # Index vectors for this tile
    i_idx = i_start + tl.arange(0, BLOCK_M)
    j_idx = j_start + tl.arange(0, BLOCK_N)

    # Masks for the boundaries of i and j
    mask_i = i_idx < N
    mask_j = j_idx < N

    # Accumulator for the tile (float32 for precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        k_idx = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N

        # Pointers to the current A and B tiles
        a_ptrs = (
            left_ptr
            + offset_batch
            + offset_chan
            + i_idx[:, None] * stride_i
            + k_idx[None, :] * stride_k
        )
        b_ptrs = (
            right_ptr
            + offset_batch
            + offset_chan
            + k_idx[:, None] * stride_k
            + j_idx[None, :] * stride_i
        )

        # Load tiles with appropriate masking; out‑of‑bounds values are zero
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Accumulate dot product over the K dimension
        acc += tl.dot(a, b)

    # Store the result tile
    c_ptrs = (
        out_ptr
        + offset_batch
        + offset_chan
        + i_idx[:, None] * stride_i
        + j_idx[None, :] * stride_i
    )
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Entry point called by the evaluator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C]
        - mask         : torch.Tensor of shape [B, N, N] (may be None)
        - weights      : dict of parameter tensors
        - config       : dict with keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32, on the same device).
    """
    # Unpack arguments
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Layer‑norm on the input
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        m = mask.unsqueeze(-1).to(dtype)  # shape [B, N, N, 1]
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4. Gating tensors (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(
        x, weights["left_gate.weight"]
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x, weights["right_gate.weight"]
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x, weights["out_gate.weight"]
    ).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul:  out = Σ_k left[...,k,:] * right[...,k,:]
    #    Implemented with Triton for speed.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # Cast to half‑precision for the kernel (better bandwidth)
    left_f16 = left.to(torch.float16).contiguous()
    right_f16 = right.to(torch.float16).contiguous()

    # Allocate the output tensor (float32)
    out_hidden = torch.empty(
        (B, N, N, hidden_dim), dtype=torch.float32, device=device
    )

    # Strides (in element units, not bytes)
    stride_b = left_f16.stride(0)
    stride_i = left_f16.stride(1)  # same for the second N dimension
    stride_k = left_f16.stride(2)
    stride_d = left_f16.stride(3)

    total_bc = B * hidden_dim
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        total_bc,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_kernel[grid](
        left_f16,
        right_f16,
        out_hidden,
        stride_b,
        stride_i,
        stride_k,
        stride_d,
        N,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Final processing: hidden‑dim LayerNorm, gating, linear projection
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    out = out_norm * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
