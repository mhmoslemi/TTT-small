    # Determine device = inp.device
    dtype = inp.dtype

    x = torch.nn.functional.layer_norm(inp, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Linear projections
    left = torch.nn.functional.linear(x, left_proj_weight)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask if provided and not nomask
    if mask is not None:
        # mask shape [bs, seq_len, seq_len], broadcast to hidden_dim
        mask_bc = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_bc
        right = right * mask_bc

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Convert to fp16 for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Allocate output tensor (fp16) shape [B,N,N,hidden_dim], maybe bmm
    out_fp16 = torch.empty_like(left_fp16)  # same shape

    # Setup kernel launch:
    B, N, _, D = left_fp16.shape
    # Compute strides (in elements) - we need to pass to kernel as int64
    # For each tensor, compute stride for each dimension as left_fp16.stride()
    stride_bl, stride_il, stride_jl, stride_dl = left_fp16.stride() # B,N,N,D
    # For right: same shape, but we treat indices similarly
    stride_br, stride_ir, stride_jr, stride_dr = right_fp16.stride()
    stride_bo, stride_io, stride_jo, stride_do = out_fp16.stride()

    # Define block sizes:
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_D = 32  # D tile for vectorization but we will use full D in accumulator; for now we keep D = hidden_dim

    # Compute grid dimensions:
    grid_i = (N + BLOCK_M - 1) // BLOCK_M
    grid_j = (N + BLOCK_N - 1) // BLOCK_N
    grid = (grid_i, grid_j, B) # B in third dimension

    # Define kernel as triton.jit function
    trmul_kernel[grid](
        left_fp16, right_fp16, out_fp16,
        B, N, D,
        stride_bl, stride_il, stride_jl, stride_dl,
        stride_br, stride_ir, stride_jr, stride_dr,
        stride_bo, stride_io, stride_jo, stride_do,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, BLOCK_D=BLOCK_D,
    )

    # Convert to fp32
    out = out_fp16.to(torch.float32)

    out = torch.nn.functional.layer_norm(out, normalized_shape=(hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)

    out = out * out_gate

    # Final linear projection to dim
    out = torch.nn.functional.linear(out, to_out_weight)

def trmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_bl, stride_il, stride_jl, stride_dl,
    stride_br, stride_ir, stride_jr, stride_dr,
    stride_bo, stride_io, stride_jo, stride_do,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_b = tl.program_id(2)

    # compute start offsets
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N
    b = pid_b

    # range vectors
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)  # will be offset inside loop
    offs_d = tl.arange(0, BLOCK_D)

    # masks for boundary
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < D

    # allocate accumulator shape (BLOCK_M, BLOCK_N, BLOCK_D) but we store separate for each d? Use a 2D per d loop.

    # create accumulator as 2D array of shape (BLOCK_M, BLOCK_N, BLOCK_D)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # main loop over k
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k_tiles):
        k_base = kk * BLOCK_K
        k_vals = k_base + offs_k
        mask_k = k_vals < N

        # compute pointers for left for each i,k,d
        left_ptrs = left_ptr + \
            b * stride_bl + \

            k_vals[None, :, None] * stride_jl + \

Implement TriMul outgoing update using Triton for the core O(N^3 * hidden_dim) reduction.
The kernel computes out_{b,i,j,d} = sum_k left_{b,i,k,d} * right_{b,j,k,d}
where left and right are the gated and masked linear projections of the input.
The kernel is tiled over the (i,j) output dimensions and the hidden dimension d.
"""

import torch
import triton
import triton.language as tl

@triton.jit
def trmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_bl, stride_il, stride_jl, stride_dl,
    stride_br, stride_ir, stride_jr, stride_dr,
    stride_bo, stride_io, stride_jo, stride_do,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid_i = tl.program_id(0)  # tile row
    pid_j = tl.program_id(1)  # tile col
    pid_b = tl.program_id(2)  # batch

    # base offsets for this tile
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    b = pid_b

    # coordinate vectors
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # number of tiles
    num_k = tl.cdiv(N, BLOCK_K)
    num_d = tl.cdiv(D, BLOCK_D)

    for d_tile in range(num_d):
        d0 = d_tile * BLOCK_D
        offs_d = d0 + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        # accumulator for this d tile
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

        for k_tile in range(num_k):
            k0 = k_tile * BLOCK_K
            offs_k = k0 + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # pointers for left and right tiles
            left_ptrs = left_ptr + \
                b * stride_bl + \
                offs_i[:, None, None] * stride_il + \
                offs_k[None, :, None] * stride_jl + \
                offs_d[None, None, :] * stride_dl

                b * stride_br + \
                offs_j[:, None, None] * stride_ir + \
                offs_k[None, :, None] * stride_jr + \
                offs_d[None, None, :] * stride_dr

            left_tile = tl.load(
                left_ptrs,

    Forward pass of the outgoing Triangle Multiplicative Update.
    Performs LayerNorm, linear projections, gating, the O(N³·Cₕ) reduction
    using a Triton kernel, followed by a second LayerNorm and a final linear.
    """
    input_tensor, mask, weights, config = data
    # unpack config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ---------- LayerNorm ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # shape [B,N,N,dim]

    # ---------- Linear projections ----------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight) # [B,N,N,hidden_dim]

    # ---------- Optional masking ----------
    # If a mask is provided and we are not in a nomask configuration, apply it.
    if not config.get("nomask", True):
        # mask shape [B, N, N]; broadcast to hidden_dim
        mask_bc = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_bc
        right = right * mask_bc

    # ---------- Gating ----------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ---------- Triton kernel for the O(N³·Cₕ) reduction ----------
    # Convert to half precision before the kernel to reduce bandwidth.
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, D = left_fp16.shape  # D == hidden_dim

    # output tensor placeholder (half precision)
    out_fp16 = torch.empty_like(left_fp16, dtype=torch.float16)

    # strides in elements
    stride_bl, stride_il, stride_jl, stride_dl = left_fp16.stride()
    stride_br, stride_ir, stride_jr, stride_dr = right_fp16.stride()
    stride_bo, stride_io, stride_jo, stride_do = out_fp16.stride()

    # tile sizes – chosen to keep register pressure reasonable
    BLOCK_M = 16
    BLOCK_N = 16
    BLOCK_K = 16
    BLOCK_D = 8  # hidden_dim is tiled, 8 works for hidden_dim up to 128

    grid_i = (N + BLOCK_M - 1) // BLOCK_M
    grid_j = (N + BLOCK_N - 1) // BLOCK_N
    grid = (grid_i, grid_j, B)

    trmul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        D,
        stride_bl,
        stride_il,
        stride_jl,
        stride_dl,
        stride_br,
        stride_ir,
        stride_jr,
        stride_dr,
        stride_bo,
        stride_io,
        stride_jo,
        stride_do,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # Back to fp32 for the remaining layers
    out = out_fp16.to(torch.float32)

    # ---------- Second LayerNorm ----------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # Apply the output gate
    out = out * out_gate

    # ---------- Final linear projection ----------
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

    4. Gating (sigmoid) and element‑wise multiplication.
    5. Triton kernel computes the O(N³·hidden_dim) reduction.
    6. Second LayerNorm, output gating and final linear projection back to dim.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ---------- 1️⃣ LayerNorm ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # shape: [B, N, N, dim]

    # ---------- 2️⃣ Linear projections ----------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)  # [B, N, N, hidden_dim]

    # ---------- 3️⃣ Optional mask ----------
    if not config.get("nomask", True):
        # mask: [B, N, N] → broadcast over hidden_dim
        mask_bc = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_bc
        right = right * mask_bc

    # ---------- 4️⃣ Gating ----------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ---------- 5️⃣ Triton kernel (core O(N³·hidden_dim) reduction) ----------
    # Use half‑precision inside the kernel to reduce bandwidth.
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, D = left_fp16.shape  # D == hidden_dim
    out_fp16 = torch.empty_like(left_fp16, dtype=torch.float16)

    # strides in elements (as returned by .stride())
    stride_bl, stride_il, stride_jl, stride_dl = left_fp16.stride()
    stride_br, stride_ir, stride_jr, stride_dr = right_fp16.stride()
    stride_bo, stride_io, stride_jo, stride_do = out_fp16.stride()

    # Tile sizes – chosen to keep register usage reasonable.
    BLOCK_M = 16
    BLOCK_N = 16
    BLOCK_K = 16
    BLOCK_D = 8   # hidden_dim is tiled; 8 works for hidden_dim up to 128

    grid_i = (N + BLOCK_M - 1) // BLOCK_M
    grid_j = (N + BLOCK_N - 1) // BLOCK_N
    grid = (grid_i, grid_j, B)

    trmul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        D,
        stride_bl,
        stride_il,
        stride_jl,
        stride_dl,
        stride_br,
        stride_ir,
        stride_jr,
        stride_dr,
        stride_bo,
        stride_io,
        stride_jo,
        stride_do,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # Back to fp32 for the remaining layers
    out = out_fp16.to(torch.float32)

    # ---------- 6️⃣ Second LayerNorm, output gate and final projection ----------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    out = out * out_gate

    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)  # back to [B, N, N, dim]

    return out
