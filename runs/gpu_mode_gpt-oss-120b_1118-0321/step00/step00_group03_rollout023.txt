@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    BH, N,
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_j,
    stride_c_batch, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_bh = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    a_ptrs_base = a_ptr + pid_bh * stride_a_batch
    b_ptrs_base = b_ptr + pid_bh * stride_b_batch
    c_ptrs_base = c_ptr + pid_bh * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load tile of A: shape (BLOCK_M,BLOCK_K)
        a_ptrs = a_ptrs_base + (offs_m[:, None] * stride_a_i) + (offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load tile of B_T: shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_ptrs_base + (offs_k[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_j)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # accumulate
        acc += tl.dot(a, b)

    # store result
    c_ptrs = c_ptrs_base + (offs_m[:, None] * stride_c_i) + (offs_n[None, :] * stride_c_j)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm
    x = torch.nn.functional.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Mask
    if not config.get("nomask", False):
        # mask shape: [B,N,N]; unsqueeze last dim for broadcasting
        mask_exp = mask.unsqueeze(-1).to(x.dtype)  # broadcast
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Permute + flatten for kernel
    B, N, _, _ = left.shape

    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    BH = B * hidden_dim

    left_flat = left_perm.view(BH, N, N)
    right_flat = right_perm.view(BH, N, N)

    out_flat = torch.empty_like(left_flat)

    # Strides
    stride_a_batch = left_flat.stride(0)
    stride_a_i = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_batch = right_flat.stride(0)
    stride_b_k = right_flat.stride(2)   # original k
    stride_b_j = right_flat.stride(1)   # original j (transposed)

    stride_c_batch = out_flat.stride(0)
    stride_c_i = out_flat.stride(1)
    stride_c_j = out_flat.stride(2)

    # Kernel launch
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (BH,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)
    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        BH, N,
        stride_a_batch, stride_a_i, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_j,
        stride_c_batch, stride_c_i, stride_c_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Reshape output back
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # shape [B,N,N,hidden_dim]

    # Output norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)

    # Apply out_gate
    out = out * out_gate

    # Final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul forward (outgoing) kernel.

The operation consists of:
1. Layer‑norm the 4‑D input.
2. Linear projections to a hidden dimension (`left_proj`, `right_proj`).
3. Optional pair‑wise mask.
4. Gate vectors (sigmoid on separate linear layers) applied to the projections.
5. Core TriMul: for each batch and hidden channel h,
   compute  out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] .
   This is a batch of N×N matrix multiplications (one per hidden channel)
   and is implemented as a Triton GEMM kernel with the hidden dimension folded
   into the batch dimension.
6. Layer‑norm on the hidden dimension, gate, and final linear projection back
   to the original feature dimension.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication A @ B.T
#   A : (batch_h, M, K)   where M = N, K = N
#   B : (batch_h, N, K)   (the same data as Right, we treat it as transposed)
#   C : (batch_h, M, N)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,                    # input/output pointers
    batch_h, N,                              # dimensions
    stride_a_batch, stride_a_i, stride_a_k, # A strides
    stride_b_batch, stride_b_k, stride_b_j, # B (transposed) strides
    stride_c_batch, stride_c_i, stride_c_j, # C strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_bh = tl.program_id(0)   # batch * hidden index
    pid_m = tl.program_id(1)    # tile row (i)
    pid_n = tl.program_id(2)    # tile col (j)

    # Offsets for the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers for this (batch_h) slice
    a_batch = a_ptr + pid_bh * stride_a_batch
    b_batch = b_ptr + pid_bh * stride_b_batch
    c_batch = c_ptr + pid_bh * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction axis K (the original sequence dimension)
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_offset = k_tile * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A block [M, K]
        a_ptrs = a_batch + offs_m[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B block transposed => B.T has shape [K, N]
        b_ptrs = b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_j
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) x (BLOCK_K, BLOCK_N)

    # Store the result
    c_ptrs = c_batch + offs_m[:, None] * stride_c_i + offs_n[None, :] * stride_c_j
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack arguments
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not config.get("nomask", False) and mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)  # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid on separate linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul using Triton GEMM kernel
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # Permute to bring hidden dim next to batch for easy flattening
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Merge batch and hidden dimensions: shape (B*H, N, N)
    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_perm.view(BH, N, N)

    # Allocate output buffer
    out_flat = torch.empty_like(left_flat)

    # Strides (in elements)
    stride_a_batch = left_flat.stride(0)
    stride_a_i = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_batch = right_flat.stride(0)
    # We need B transposed: (k, j) => stride over k is original last dim,
    # stride over j is original middle dim.
    stride_b_k = right_flat.stride(2)   # original k stride
    stride_b_j = right_flat.stride(1)   # original j stride (to be used as column)

    stride_c_batch = out_flat.stride(0)
    stride_c_i = out_flat.stride(1)
    stride_c_j = out_flat.stride(2)

    # Tiling parameters – chosen to fit typical B200 SMs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        BH,
        N,
        stride_a_batch,
        stride_a_i,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_j,
        stride_c_batch,
        stride_c_i,
        stride_c_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gate, and final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
