@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    B, N, D,
    stride_xb, stride_xi, stride_xj, stride_xd,
    stride_yb, stride_yi, stride_yj, stride_yd,
    stride_w, stride_b,
    epsilon: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0) # combined pid over B*N*N
    # compute b,i,j indices
    total = B * N * N
    pid = tl.program_id(0)
    # compute indices:
    b = pid // (N * N)
    tmp = pid % (N * N)
    i = tmp // N
    j = tmp % N

    # load weight and bias: they are 1D length D
    # We'll compute mean and var per (b,i,j) using loop over D.

    offs_d = tl.arange(0, BLOCK_D)
    mask = offs_d < D

    # compute sum and sum_sq
    sum_val = tl.zeros([1], dtype=tl.float32)
    sum_sq = tl.zeros([1], dtype=tl.float32)

    # loop over d dimension tiles
    for offset in range(0, D, BLOCK_D):
        cur_offs = offset + offs_d
        cur_mask = cur_offs < D
        x = tl.load(x_ptr + b*stride_xb + i*stride_xi + j*stride_xj + cur_offs*stride_xd,
                     mask=cur_mask, other=0.0)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)
    # compute mean, var
    mean = sum_val / D
    var = sum_sq / D - mean * mean
    inv_std = 1.0 / tl.sqrt(var + epsilon)
    # second pass to write normalized values
    for offset in range(0, D, BLOCK_D):
        cur_offs = offset + offs_d
        cur_mask = cur_offs < D
        x = tl.load(x_ptr + b*stride_xb + i*stride_xi + j*stride_xj + cur_offs*stride_xd,
                    mask=cur_mask, other=0.0)
        norm = (x - mean) * inv_std
        w = tl.load(weight_ptr + cur_offs, mask=cur_mask, other=0.0)
        b_ = tl.load(bias_ptr + cur_offs, mask=cur_mask, other=0.0)
        y = norm * w + b_
        tl.store(y_ptr + b*stride_yb + i*stride_yi + j*stride_yj + cur_offs*stride_yd,
                 y, mask=cur_mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # LayerNorm over dim
    norm_weight = weights["norm.weight"]  # shape [dim]
    norm_bias = weights["norm.bias"]  # shape [dim]
    eps = 1e-5

    # call triton layernorm kernel
    normed = torch.empty_like(input_tensor)
    launch_layernorm(input_tensor, normed, norm_weight, norm_bias, eps)

    # Linear projections
    left = torch.nn.functional.linear(normed, weights["left_proj.weight"])
    right = torch.nn.functional.linear(normed, weights["right_proj.weight"])

    # mask
    if not config.get("nomask", False):
        # mask shape [B,N,N], broadcast to last dim
        mask_f = mask.to(dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # Gate linear
    left_gate = torch.nn.functional.linear(normed, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(normed, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(normed, weights["out_gate.weight"]).sigmoid()

    # Apply gates to left/right
    left = left * left_gate
    right = right * right_gate

    # Compute out via batched matmul
    # Permute for batched GEMM
    # left: [B,N,N,hidden_dim] -> (B*hidden_dim,N,N)
    left_flat = left.permute(0,3,1,2).contiguous().view(batch_size*hidden_dim, seq_len, seq_len)
    right_flat = right.permute(0,3,2,1).contiguous().view(batch_size*hidden_dim, seq_len, seq_len)
    out_flat = torch.bmm(left_flat, right_flat)  # shape (B*hidden_dim, N, N)
    # Reshape back
    out = out_flat.view(batch_size, hidden_dim, seq_len, seq_len).permute(0,2,3,1)  # (B,N,N,hidden_dim)

    # to_out_norm layernorm over hidden_dim
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = torch.empty_like(out)  # allocate
    launch_layernorm(out, out_norm, to_out_norm_weight, to_out_norm_bias, eps)

    # multiply by out_gate (gate shape [B,N,N,hidden_dim])
    out_norm = out_norm * out_gate

    # final linear
    final = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return final

def layernorm_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    B, N, D,
    stride_xb, stride_xi, stride_xj, stride_xd,
    stride_yb, stride_yi, stride_yj, stride_yd,
    epsilon,
    BLOCK_D: tl.constexpr,
):
    # compute combined program id: pid = tl.program_id(0)
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # compute indices
    b = pid // (N * N)
    tmp = pid % (N * N)
    i = tmp // N
    j = tmp % N

    # Offsets for feature dimension usage
    offs_d = tl.arange(0, BLOCK_D)
    # load weight and bias later per block

    # compute sum & sum_sq in float32
    sum_val = tl.zeros([1], dtype=tl.float32)
    sum_sq = tl.zeros([1], dtype=tl.float32)

    # Loop
    for k in range(0, D, BLOCK_D):
        cur_offs = k + offs_d
        mask = cur_offs < D
        x = tl.load(x_ptr + b*stride_xb + i*stride_xi + j*stride_xj + cur_offs*stride_xd,
                    mask=mask, other=0.0)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)
    mean = sum_val / D
    var = sum_sq / D - mean * mean
    inv_std = 1.0 / tl.sqrt(var + epsilon)

    # second loop to write normalized output
    for k in range(0, D, BLOCK_D):
        cur_offs = k + offs_d
        mask = cur_offs < D
        x = tl.load(x_ptr + b*stride_xb + i*stride_xi + j*stride_xj + cur_offs*stride_xd,
                    mask=mask, other=0.0)
        w = tl.load(weight_ptr + cur_offs, mask=mask, other=0.0)
        b_ = tl.load(bias_ptr + cur_offs, mask=mask, other=0.0)
        norm = (x - mean) * inv_std
        y = norm * w + b_
        tl.store(y_ptr + b*stride_yb + i*stride_yi + j*stride_yj + cur_offs*stride_yd,
                 y, mask=mask)

Now we need to define launch_layernorm function:

def launch_layernorm(x, y, weight, bias, eps=1e-5, BLOCK_D=64):
    B, N, N2, D = x.shape
    assert N2 == N
    # ensure contiguous
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()
    stride_xb, stride_xi, stride_xj, stride_xd = x.stride()
    stride_yb, stride_yi, stride_yj, stride_yd = y.stride()
    # weight and bias stride is 1
    grid = (B * N * N,)
    layernorm_kernel[grid](
        x_ptr=x.data_ptr(),
        y_ptr=y.data_ptr(),
        weight_ptr=weight.data_ptr(),
        bias_ptr=bias.data_ptr(),
        B=B,
        N=N,
        D=D,
        stride_xb=stride_xb,
        stride_xi=stride_xi,
        stride_xj=stride_xj,
        stride_xd=stride_xd,
        stride_yb=stride_yb,
        stride_yi=stride_yi,
        stride_yj=stride_yj,
        stride_yd=stride_yd,
        epsilon=eps,
        BLOCK_D=BLOCK_D,
    )
    return y

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # dims
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # layernorm over dim
    x_norm = _layernorm_triton(input_tensor, weights["norm.weight"], weights["norm.bias"], eps=eps)

    # Projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Apply mask if present and not nomask
    if not config.get("nomask", False):
        # mask shape B,N,N
        mask_f = mask.to(dtype).unsqueeze(-1)  # shape B,N,N,1
        left = left * mask_f
        right = right * mask_f

    # Gating
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Compute out via batched GEMM
    # reshape for batch GEMM
    left_flat = left.permute(0,3,1,2).contiguous().view(B*hidden_dim, N, N)
    right_flat = right.permute(0,3,2,1).contiguous().view(B*hidden_dim, N, N)
    out_flat = torch.bmm(left_flat, right_flat)
    out = out_flat.view(B, hidden_dim, N, N).permute(0,2,3,1)  # back to B,N,N,hidden_dim

    # Norm hidden dim
    out_norm = _layernorm_triton(out, weights["to_out_norm.weight"], weights["to_out_norm.bias"], eps=eps)

    # Apply out gate
    out_norm = out_norm * out_gate

    # Final linear
    final = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return final

"""
TriMul (outgoing) implementation with Triton LayerNorm.

The operator works on a tensor x of shape [B, N, N, C] and performs:
1. LayerNorm over the feature dimension C.
2. Linear projections to a hidden dimension H (left/right) and gating
   (left_gate, right_gate, out_gate) using sigmoid.
3. Optional masking of left/right.
4. Multiplicative update: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
   This is a batched matrix multiplication (B*H independent GEMMs of size N×N).
5. LayerNorm over H, gated by out_gate, and a final linear back to C.

The LayerNorm in steps 1 and 5 is implemented in Triton to demonstrate a
custom kernel; the heavy N³ multiplication uses torch.bmm which maps to
high‑performance cuBLAS batched GEMM.
"""

import torch
import triton
import triton.language as tl

# --------------------------------------------------------------------
# Triton LayerNorm kernel (normalizes over the last dimension)
# --------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # *const float*   pointer to input
    y_ptr,               # *float*         pointer to output
    weight_ptr,          # *const float*   LayerNorm weight (gamma)
    bias_ptr,            # *const float*   LayerNorm bias (beta)
    B, N, D,             # compile‑time constants: batch, seq_len, feature dim
    stride_xb, stride_xi, stride_xj, stride_xd,   # strides of x
    stride_yb, stride_yi, stride_yj, stride_yd,   # strides of y
    epsilon: tl.constexpr,                       # epsilon for variance
    BLOCK_D: tl.constexpr,                       # tile size for the feature dim
):
    pid = tl.program_id(0)                     # each program processes one (b,i,j)
    total = B * N * N
    if pid >= total:
        return

    # -------------------- index calculation --------------------
    b = pid // (N * N)
    tmp = pid % (N * N)
    i = tmp // N
    j = tmp % N

    # base offsets for this (b,i,j) element (pointing to the first feature)
    base_x = b * stride_xb + i * stride_xi + j * stride_xj
    base_y = b * stride_yb + i * stride_yi + j * stride_yj

    # -------------------- first pass: compute mean & var --------------------
    offs_d = tl.arange(0, BLOCK_D)                     # [0, BLOCK_D)
    sum_val = tl.float32(0.0)
    sum_sq = tl.float32(0.0)

    for off in range(0, D, BLOCK_D):
        cur_off = off + offs_d
        mask = cur_off < D
        x = tl.load(x_ptr + base_x + cur_off * stride_xd,
                    mask=mask, other=0.0, dtype=tl.float32)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)

    mean = sum_val / D
    var = sum_sq / D - mean * mean
    inv_std = 1.0 / tl.sqrt(var + epsilon)

    # -------------------- second pass: write normalized output --------------------
    for off in range(0, D, BLOCK_D):
        cur_off = off + offs_d
        mask = cur_off < D
        x = tl.load(x_ptr + base_x + cur_off * stride_xd,
                    mask=mask, other=0.0, dtype=tl.float32)
        w = tl.load(weight_ptr + cur_off, mask=mask, other=0.0, dtype=tl.float32)
        b_ = tl.load(bias_ptr + cur_off, mask=mask, other=0.0, dtype=tl.float32)

        norm = (x - mean) * inv_std
        y = norm * w + b_

        tl.store(y_ptr + base_y + cur_off * stride_yd,
                 y, mask=mask)


def _layernorm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5,
                      BLOCK_D: int = 32) -> torch.Tensor:
    """
    Apply LayerNorm over the last dimension of `x` using a Triton kernel.
    `weight` and `bias` must be 1‑D tensors of size `C`.
    """
    assert x.is_contiguous()
    B, N, N2, D = x.shape
    assert N == N2 and D == weight.shape[0] == bias.shape[0]

    y = torch.empty_like(x)

    # make sure everything is on the same device & dtype
    weight = weight.to(x.dtype).contiguous()
    bias = bias.to(x.dtype).contiguous()

    # strides (number of elements, not bytes)
    stride_xb, stride_xi, stride_xj, stride_xd = x.stride()
    stride_yb, stride_yi, stride_yj, stride_yd = y.stride()

    grid = (B * N * N,)

    layernorm_kernel[grid](
        x_ptr=x.data_ptr(),
        y_ptr=y.data_ptr(),
        weight_ptr=weight.data_ptr(),
        bias_ptr=bias.data_ptr(),
        B=B,
        N=N,
        D=D,
        stride_xb=stride_xb,
        stride_xi=stride_xi,
        stride_xj=stride_xj,
        stride_xd=stride_xd,
        stride_yb=stride_yb,
        stride_yi=stride_yi,
        stride_yj=stride_yj,
        stride_yd=stride_yd,
        epsilon=eps,
        BLOCK_D=BLOCK_D,
    )
    return y


# --------------------------------------------------------------------
# Public entry point
# --------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, C]
        - mask         : torch.Tensor or None, shape [B, N, N] (float or bool)
        - weights      : dict of layer weights (all tensors on the same CUDA device)
        - config       : dict containing at least ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # unpack arguments
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, C = input_tensor.shape
    H = config["hidden_dim"]                     # hidden dimension (H)

    # ----------------------------------------------------------------
    # Step 1: LayerNorm over channel dimension C (Triton)
    # ----------------------------------------------------------------
    x_norm = _layernorm_triton(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
        eps=1e-5,
        BLOCK_D=32,
    )

    # ----------------------------------------------------------------
    # Step 2: Linear projections (no bias) and optional masking
    # ----------------------------------------------------------------
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Apply mask if present and not disabled via config
    if not config.get("nomask", False) and mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ----------------------------------------------------------------
    # Step 3: Gates (sigmoid)
    # ----------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------
    # Step 4: Multiplicative update via batched GEMM
    # out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # This is a batch of B*H independent N×N matrix multiplications.
    # ----------------------------------------------------------------
    # reshape to (B*H, N, N) for left and (B*H, N, N) for right^T
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # right needs transposition of the two sequence axes (j,k)
    right_flat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)

    # Use torch.bmm (batched GEMM). Autocast can give fp16 speedup.
    with torch.cuda.amp.autocast(enabled=True):
        out_flat = torch.bmm(left_flat, right_flat)   # (B*H, N, N)

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------------------
    # Step 5: LayerNorm over hidden dimension H (Triton)
    # ----------------------------------------------------------------
    out_norm = _layernorm_triton(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps=1e-5,
        BLOCK_D=32,
    )

    # ----------------------------------------------------------------
    # Step 6: Final gating and linear projection back to C
    # ----------------------------------------------------------------
    out_norm = out_norm * out_gate
    final = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return final
