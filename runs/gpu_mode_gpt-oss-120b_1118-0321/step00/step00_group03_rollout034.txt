def custom_kernel(data):
    input_tensor, mask, weights, config = data

    # Extract config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply mask if needed
    if not nomask:
        # mask shape [B,N,N], cast to same dtype as left[right]
        mask_unsq = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_unsq
        right = right * mask_unsq

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Cast to half for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Prepare output tensor
    batch = left_fp16.shape[0]
    N = left_fp16.shape[1]
    # out_hidden shape
    out_hidden = torch.empty((batch, N, N, hidden_dim), dtype=torch.float32, device=left_fp16.device)

    # Get strides as ints
    stride_a_batch = left_fp16.stride(0)
    stride_a_i = left_fp16.stride(1)
    stride_a_k = left_fp16.stride(2)
    stride_a_d = left_fp16.stride(3)  # 1

    stride_b_batch = right_fp16.stride(0)
    stride_b_k = right_fp16.stride(2)  # k dimension
    stride_b_n = right_fp16.stride(1)  # j dimension
    stride_b_d = right_fp16.stride(3)

    stride_c_batch = out_hidden.stride(0)
    stride_c_i = out_hidden.stride(1)
    stride_c_j = out_hidden.stride(2)
    stride_c_d = out_hidden.stride(3)

    # Determine block sizes (tune)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # grid
    batch_hidden = batch * hidden_dim
    grid = (batch_hidden, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

    # Launch kernel
    trimul_matmul_kernel[grid](
        left_fp16, right_fp16, out_hidden,
        batch, N, N, hidden_dim,
        stride_a_batch, stride_a_i, stride_a_k, stride_a_d,
        stride_b_batch, stride_b_k, stride_b_n, stride_b_d,
        stride_c_batch, stride_c_i, stride_c_j, stride_c_d,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # After kernel we have out_hidden (float32)

    # Apply to_out_norm (layernorm over hidden_dim)
    out_hidden_norm = torch.nn.functional.layer_norm(out_hidden, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=1e-5)

    # Multiply by out_gate (already float)
    out_hidden_norm = out_hidden_norm * out_gate

    # final linear projection to dim
    output = torch.nn.functional.linear(out_hidden_norm, weights["to_out.weight"])

    return output

def trimul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, hidden_dim,
    stride_a_batch, stride_a_i, stride_a_k, stride_a_d,
    stride_b_batch, stride_b_k, stride_b_n, stride_b_d,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # get ids
    batch_hidden = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute b and d
    b = batch_hidden // hidden_dim

"""Triton implementation of the core Tensor Multiplication (TriMul) operation.

The heavy part of TriMul is the einsum
    out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix multiplication for each hidden channel d.
This kernel computes that product in Triton, while all other
linear/LayerNorm/gating steps are performed with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    batch, M, N, hidden_dim,                  # dimensions
    stride_a_batch, stride_a_i, stride_a_k, stride_a_d,
    stride_b_batch, stride_b_k, stride_b_n, stride_b_d,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]"""

    # Program IDs: (batch_hidden, block_i, block_j)
    batch_hidden = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # De‑compose batch_hidden into batch index and hidden‑channel index
    b = batch_hidden // hidden_dim          # batch index
    d = batch_hidden % hidden_dim           # channel index

    # Offsets for the i and j dimensions
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < M
    mask_j = offs_j < N

    # Base pointers for this (batch, channel) slice
    a_base = a_ptr + b * stride_a_batch + d * stride_a_d
    b_base = b_ptr + b * stride_b_batch + d * stride_b_d
    c_base = c_ptr + b * stride_c_batch + d * stride_c_d

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction axis k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Pointers for the current tile of A (left) and B (right)
        a_ptrs = a_base + offs_i[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_j[None, :] * stride_b_n

        # Load tiles, applying bounds masks
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # Matrix‑multiply the tiles and accumulate
        acc += tl.dot(a, b)   # a,b are fp16 → acc is fp32

    # Write the result back to C
    c_ptrs = c_base + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul module (no gradient).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (may be ignored if config["nomask"] is True)
            - weights: dict of model parameters
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)   # True -> no mask applied

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections and gating
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])        # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    if not nomask:
        # mask: [B, N, N] -> broadcast to hidden dim
        mask_broadcast = mask.to(left.dtype).unsqueeze(-1)
        left = left * mask_broadcast
        right = right * mask_broadcast

    # Apply element‑wise gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. Core TriMul: batched matmul over the K dimension (handled by Triton)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # Use FP16 inside the kernel for bandwidth, but keep tensors in FP32 for
    # exactness; Triton can load FP32 directly.
    left_t = left.contiguous()
    right_t = right.contiguous()

    # Output tensor after the einsum, shape [B, N, N, hidden_dim]
    out_hidden = torch.empty((B, N, N, hidden_dim),
                             dtype=torch.float32,
                             device=left.device)

    # Strides (in elements)
    stride_a_batch = left_t.stride(0)
    stride_a_i = left_t.stride(1)
    stride_a_k = left_t.stride(2)
    stride_a_d = left_t.stride(3)

    stride_b_batch = right_t.stride(0)
    stride_b_k = right_t.stride(2)   # k dimension (third)
    stride_b_n = right_t.stride(1)   # j dimension (second)
    stride_b_d = right_t.stride(3)

    stride_c_batch = out_hidden.stride(0)
    stride_c_i = out_hidden.stride(1)
    stride_c_j = out_hidden.stride(2)
    stride_c_d = out_hidden.stride(3)

    # Tunable block size (empirically works for the target shapes)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid: (batch*hidden_dim, n_blocks_M, n_blocks_N)
    batch_hidden = B * hidden_dim
    grid = (
        batch_hidden,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    trimul_matmul_kernel[grid](
        left_t,
        right_t,
        out_hidden,
        B,
        N,
        N,
        hidden_dim,
        stride_a_batch,
        stride_a_i,
        stride_a_k,
        stride_a_d,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_b_d,
        stride_c_batch,
        stride_c_i,
        stride_c_j,
        stride_c_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 4. Post‑processing: LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out_hidden = out_hidden * out_gate  # element‑wise gating

    # Final linear projection back to original dim
    output = F.linear(out_hidden, weights["to_out.weight"])

    return output
