    M, N, K,

    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,

    pid_m = tl.program_id(1)

def custom_kernel(data):
    """TriMul outgoing implementation using Triton batch matmul.
    This function follows the AlphaFold3 TriMul forward pass:
    - LayerNorm on input.
    - Linear projections (left/right) and gating.
    - Compute `out_{i,j}` = sum_k left_{i,k} * right_{j,k} per batch and hidden channel.
    - Apply output LayerNorm, gating and final linear projection.
    The heavy contraction is performed by a fused Triton kernel that
    treats each (batch, hidden) pair as an independent matrix‑matrix
    multiplication A @ B^T.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nominal_nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. LayerNorm on the input
    # --------------------------------------------------------------
    # weight and bias are provided as 1‑D tensors of shape (dim,)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask
    # --------------------------------------------------------------
    if not nominal_nomask:
        # mask shape: [B, N, N]; broadcast over hidden dimension
        mask_ = mask.unsqueeze(-1)
        left = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4. Gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    right = right * right_gate

    # 5. Heavy contraction – Triton batched matmul
    #    out_{i,j,d} = Σ_k left_{i,k,d} * right_{j,k,d}
    # --------------------------------------------------------------
    B, N, _, _ = left.shape

    # Move hidden dimension to be the second axis and make matrices contiguous
    left_mat = left.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N]
    right_mat = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    right_t = right_mat.transpose(-2, -1).contiguous()  # [B, H, N, N]

"""
TriMul (outgoing) forward pass implemented with a fused Triton kernel.

The kernel performs the heavy contraction
    out_{i,j,h} = Σ_k left_{i,k,h} * right_{j,k,h}
for every batch and hidden channel h.  By merging the batch and hidden
dimensions, each (batch, h) pair becomes an independent matrix‑matrix
multiply A @ B (where B is the transposed right tensor).  The Triton
kernel tiles the multiplication (BLOCK_M × BLOCK_N) and accumulates
in float32 for numerical stability.  All other operations (LayerNorm,
Linear projections, gating) are carried out with regular PyTorch
routines.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matrix multiplication C = A @ B with A:[M,K], B:[K,N].

    The batch dimension is encoded in program_id(0); each batch is a
    separate matrix pair.  The kernel is a straightforward tiling
    implementation using tl.dot.
    """
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + pid_batch * batch_stride_a
    b_batch = b_ptr + pid_batch * batch_stride_b
    c_batch = c_ptr + pid_batch * batch_stride_c

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # tile over the K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_am
                            + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # B tile: (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch + (offs_k[:, None] * stride_bk
                            + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

    # store the result
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm
                        + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


def _triton_batch_matmul(a: torch.Tensor, b: torch.Tensor, M: int, N: int, K: int,
                         BLOCK_M: int = 64, BLOCK_N: int = 64, BLOCK_K: int = 32) -> torch.Tensor:
    """
    Wrapper around the Triton kernel performing batched matmul.
    a: (B, M, K) contiguous
    b: (B, K, N) contiguous
    Returns: (B, M, N) contiguous tensor.
    """
    assert a.is_contiguous() and b.is_contiguous()
    batch = a.shape[0]
    out = torch.empty((batch, M, N), dtype=a.dtype, device=a.device)

    # strides in number of elements
    stride_am, stride_ak = a.stride(1), a.stride(2)
    stride_bk, stride_bn = b.stride(1), b.stride(2)
    stride_cm, stride_cn = out.stride(1), out.stride(2)

    batch_stride_a = a.stride(0)
    batch_stride_b = b.stride(0)
    batch_stride_c = out.stride(0)

    grid = (
        batch,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batch_matmul_kernel[grid](
        a, b, out,
        M, N, K,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        batch_stride_a, batch_stride_b, batch_stride_c,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out


def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor: [B, N, N, dim] float32 tensor.
        - mask        : [B, N, N] (may be None if config["nomask"] is True).
        - weights     : dict of parameters (LayerNorm weights/biases, Linear weights, etc.).
        - config      : dict containing at least "dim" and "hidden_dim".
    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] after the TriMul forward pass.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (over last dimension)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_ = mask.unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of bias‑less linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via Triton (out_{i,j,h} = Σ_k left_{i,k,h} * right_{j,k,h})
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # Move hidden dimension next to batch and make matrices contiguous
    left_mat = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_mat = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Transpose right on the last two axes so that we can use A @ B
    right_t = right_mat.transpose(-2, -1).contiguous()  # [B, H, N, N]

    # Collapse batch & hidden into a single batch dimension
    left_flat = left_mat.view(-1, N, N)    # [B*H, N, N]
    right_flat = right_t.view(-1, N, N)   # [B*H, N, N]

    # Perform batched matmul on the flattened tensors
    out_flat = _triton_batch_matmul(left_flat, right_flat, N, N, N)

    # Restore original layout: [B, N, N, hidden]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
