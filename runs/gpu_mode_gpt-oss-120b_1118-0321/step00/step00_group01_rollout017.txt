import triton
import triton.language as tl

@triton.jit
def triton_trimul_einsum_kernel(

    left_ptr, right_ptr, out_ptr,
    B, N, HD,
    stride_left_bs, stride_left_i, stride_left_k, stride_left_d,
    stride_right_bs, stride_right_i, stride_right_k, stride_right_d,
    stride_out_bs, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    ### The kernel
    pid_z = tl.program_id(0)  # combined batch*hidden
    pid_m = tl.program_id(1)  # i
    pid_n = tl.program_id(2)  # j

    # Determine batch and hidden index
    b = pid_z // HD
    d = pid_z % HD

    # Compute base pointers for slice
    left_base = left_ptr + b * stride_left_bs + d * stride_left_d
    right_base = right_ptr + b * stride_right_bs + d * stride_right_d
    out_base = out_ptr + b * stride_out_bs + d * stride_out_d

    # tile start.
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # Offsets vectors
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i,k]: row i offset along i dimension, col offset along k dimension
        left_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right_T[k,j] which is right[j,k] transposed
        # row offset k uses stride_right_k, column offset j uses stride_right_i
        right_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        right_T = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # dot product over K dimension (inner)
        acc += tl.dot(left, right_T)

    # Store result
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

The kernel is done.

"""
custom_kernel: Triton‑accelerated forward pass of the TriMul module (AlphaFold3).

Algorithm
---------
1. Layer‑normalize the input tensor over the channel dimension `dim`.
2. Linear projections to a hidden dimension `hidden_dim` for left/right tensors
   and three gating tensors (left, right, out).  All linear layers are bias‑free.
3. Apply optional mask (broadcast on the hidden dimension) and element‑wise
   sigmoid gates.
4. Core operation – compute for each batch `b` and hidden channel `d`

        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

   This is implemented as a blocked matrix‑multiply (`left @ right.T`) in a
   Triton kernel.  The kernel iterates over tiles of the (i,j) output plane,
   loads blocks of `left` and the transposed block of `right`, performs a
   `tl.dot` accumulation and writes the result.
5. Layer‑normalize the intermediate tensor over the hidden dimension,
   apply the out‑gate, and a final linear projection back to `dim`.

The kernel works for arbitrary batch size (1‑2), sequence length up to 1024,
and hidden dimensions up to 384.  All computations are performed in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel for the O(N³) einsum: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
# --------------------------------------------------------------------------- #
@triton.jit
def triton_trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, HD,
    stride_left_bs, stride_left_i, stride_left_k, stride_left_d,
    stride_right_bs, stride_right_i, stride_right_k, stride_right_d,
    stride_out_bs, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program IDs
    pid_z = tl.program_id(0)   # combined batch * hidden dimension
    pid_m = tl.program_id(1)   # i tile
    pid_n = tl.program_id(2)   # j tile

    # Decode batch and hidden indices
    b = pid_z // HD
    d = pid_z % HD

    # Base pointers for this (b, d) slice
    left_base  = left_ptr  + b * stride_left_bs  + d * stride_left_d
    right_base = right_ptr + b * stride_right_bs + d * stride_right_d
    out_base   = out_ptr   + b * stride_out_bs   + d * stride_out_d

    # Tile offsets
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension in blocks of size BLOCK_K
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i, k]  -> shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right_T[k, j] where right_T is right transposed (row = k, col = j)
        # right[j, k] lives at (j * stride_i + k * stride_k)
        right_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        right_T = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Multiply‑accumulate
        acc += tl.dot(left, right_T)

    # Store the result tile
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

    nomask = config.get("nomask", True)   # True => ignore mask

    # ------------------------------------------------------------------- #
    # LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    eps = 1e-5
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------- #
    # Linear projections (bias‑free)
    # ------------------------------------------------------------------- #
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)          # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------- #
    # Gating tensors (sigmoid)
    # ------------------------------------------------------------------- #
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------- #
    # Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------- #
    if not nomask:
        mask = mask.to(dtype=dtype)
        mask = mask.unsqueeze(-1)                     # [B,N,N,1]
        left = left * mask
        right = right * mask

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous tensors for predictable strides
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------- #
    # Triton kernel: compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------- #
    B, N, _, _ = left.shape
    out = torch.empty_like(left)   # [B,N,N,hidden_dim], same dtype/device

    # Strides in number of elements (not bytes)
    stride_left_bs, stride_left_i, stride_left_k, stride_left_d = left.stride()
    stride_right_bs, stride_right_i, stride_right_k, stride_right_d = right.stride()
    stride_out_bs, stride_out_i, stride_out_j, stride_out_d = out.stride()

    # Block sizes – chosen to fit in registers and shared memory
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                       # z dimension: batch * hidden
        (N + BLOCK_M - 1) // BLOCK_M,         # i tiles
        (N + BLOCK_N - 1) // BLOCK_N,         # j tiles
    )

    triton_trimul_einsum_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, hidden_dim,
        stride_left_bs, stride_left_i, stride_left_k, stride_left_d,
        stride_right_bs, stride_right_i, stride_right_k, stride_right_d,
        stride_out_bs, stride_out_i, stride_out_j, stride_out_d,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,                         # reasonable default
    )

    # ------------------------------------------------------------------- #
    # Post‑processing: LayerNorm over hidden_dim, out‑gate, final linear
    # ------------------------------------------------------------------- #
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]   # shape (dim, hidden_dim)

    # LayerNorm over the hidden dimension
    out_norm = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                            bias=to_out_norm_bias, eps=eps)

    # Apply out‑gate
    out_gate = out_gate  # already computed on shape [B,N,N,hidden_dim]
    out_norm = out_norm * out_gate

    # Final linear projection back to original channel dimension
    output = F.linear(out_norm, to_out_weight)   # shape [B,N,N,dim]

    return output
```
