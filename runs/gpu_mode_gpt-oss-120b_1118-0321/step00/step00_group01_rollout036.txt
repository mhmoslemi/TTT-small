@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_ri, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # base offsets
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # ranges
    i_off = i_start + tl.arange(0, BLOCK_M)
    j_off = j_start + tl.arange(0, BLOCK_N)
    # masks for out-of-bounds
    mask_i = i_off < N
    mask_j = j_off < N

    # allocate accumulator across hidden dim
    # We'll loop over d in steps of BLOCK_D
    for d_start in range(0, hidden_dim, BLOCK_D):
        d_off = d_start + tl.arange(0, BLOCK_D)
        mask_d = d_off < hidden_dim

        # accumulator for this d block
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

        # loop over k dimension
        for k_start in range(0, N, BLOCK_K):
            k_off = k_start + tl.arange(0, BLOCK_K)
            mask_k = k_off < N

            # offsets for left: shape (M,K,D)
            left_ptrs = left_ptr \
                + pid_b * stride_lb \
                + i_off[:, None, None] * stride_li \
                + k_off[None, :, None] * stride_lk \
                + d_off[None, None, :] * stride_ld

            left = tl.load(left_ptrs,
                mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                other=0.0)

            # offsets for right (transposed): shape (K,N,D)
            right_ptrs = right_ptr \
                + pid_b * stride_rb \
                + k_off[:, None, None] * stride_rk \
                + j_off[None, :, None] * stride_ri \
                + d_off[None, None, :] * stride_rd

            right = tl.load(right_ptrs,
                mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
                other=0.0)

            # compute dot for each hidden slice
            for d in range(BLOCK_D):
                left_slice = left[:, :, d]   # (M, K)
                right_slice = right[:, :, d] # (K, N)
                acc[:, :, d] += tl.dot(left_slice, right_slice)

        # store result for this hidden block
        out_ptrs = out_ptr \
            + pid_b * stride_ob \
            + i_off[:, None, None] * stride_oi \
            + j_off[None, :, None] * stride_oj \
            + d_off[None, None, :] * stride_od

        tl.store(out_ptrs,
            acc,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # Cast mask if not None
    if mask is None or config.get("nomask", False):
        mask_tensor = torch.ones((input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]), dtype=dtype, device=device)
    else:
        mask_tensor = mask.to(dtype)

    # Layernorm: using weight/bias
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape [bs,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask
    mask_unsqueezed = mask_tensor.unsqueeze(-1)  # shape [bs,N,N,1]
    left = left * mask_unsqueezed
    right = right * mask_unsqueezed

    # Gate computations
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous
    left = left.contiguous()
    right = right.contiguous()

    # Allocate output for core computation
    batch, N, _, _ = left.shape  # shape [bs, N, N, hidden_dim]
    out_mul = torch.empty_like(left)  # shape [bs,N,N,hidden_dim]

    # get strides
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_rb, stride_ri, stride_rk, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out_mul.stride()

    # compile and launch kernel
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_D = 8  # hidden_dim multiple

    grid = (
        batch,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out_mul,
        B=batch,
        N=N,
        hidden_dim=hidden_dim,
        stride_lb=stride_lb,
        stride_li=stride_li,
        stride_lk=stride_lk,
        stride_ld=stride_ld,
        stride_rb=stride_rb,
        stride_ri=stride_ri,
        stride_rk=stride_rk,
        stride_rd=stride_rd,
        stride_ob=stride_ob,
        stride_oi=stride_oi,
        stride_oj=stride_oj,
        stride_od=stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # Continue with to_out_norm and gating
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out_mul, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Apply output gate
    out = out * out_gate

    # Final linear projection
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

    left_ptr, right_ptr, out_ptr,
    B, N,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_ri, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute offsets
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    i_range = i_base + tl.arange(0, BLOCK_M)
    j_range = j_base + tl.arange(0, BLOCK_N)

    mask_i = i_range < N
    mask_j = j_range < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k_blocks):
        k_base = kk * BLOCK_K
        k_range = k_base + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # left

            + i_range[:, None, None] * stride_li \
            + k_range[None, :, None] * stride_lk \
            + tl.arange(0, BLOCK_D)[None, None, :] * stride_ld
        left = tl.load(left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None],
            other=0.0)

            + k_range[:, None, None] * stride_rk \
            + j_range[None, :, None] * stride_ri \
            + tl.arange(0, BLOCK_D)[None, None, :] * stride_rd
        right = tl.load(right_ptrs,
            mask=mask_k[:, None, None] & mask_j[None, :, None],
            other=0.0)

            right_d = right[:, :, d] # (K, N)
            acc[:, :, d] += tl.dot(left_d, right_d)

        + j_range[None, :, None] * stride_oj \
        + tl.arange(0, BLOCK_D)[None, None, :] * stride_od

        mask=mask_i[:, None, None] & mask_j[None, :, None])

    BLOCK_N: tl.constexpr,  # tile size along j
    BLOCK_K: tl.constexpr,  # tile size along k
    BLOCK_D: tl.constexpr,  # hidden‑channel block size
):
    pid_b = tl.program_id(0)                     # batch dimension
    pid_i = tl.program_id(1)                     # tile row (i)
    pid_j = tl.program_id(2)                     # tile col (j)

    # ------------------------------------------------------------------
    # Tile coordinates and out‑of‑bounds masks
    # ------------------------------------------------------------------
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    i_range = i_start + tl.arange(0, BLOCK_M)
    j_range = j_start + tl.arange(0, BLOCK_N)

    mask_i = i_range < N
    mask_j = j_range < N

    # accumulator for this (i, j) tile and hidden‑block
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k in tiles
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(num_k_blocks):
        k_start = kb * BLOCK_K
        k_range = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # --------------------------------------------------------------
        # Load left[b, i, k, d]   -> shape (BLOCK_M, BLOCK_K, BLOCK_D)
        # --------------------------------------------------------------
        left_ptrs = left_ptr \
            + pid_b * stride_lb \
            + i_range[:, None, None] * stride_li \
            + k_range[None, :, None] * stride_lk \
            + tl.arange(0, BLOCK_D)[None, None, :] * stride_ld
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None],
            other=0.0,
        )

        # --------------------------------------------------------------
        # Load right[b, j, k, d] transposed to (k, j, d)
        # --------------------------------------------------------------
        right_ptrs = right_ptr \
            + pid_b * stride_rb \
            + k_range[:, None, None] * stride_rk \
            + j_range[None, :, None] * stride_ri \
            + tl.arange(0, BLOCK_D)[None, None, :] * stride_rd
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None, None] & mask_j[None, :, None],
            other=0.0,
        )

        # --------------------------------------------------------------
        # Compute dot products for each hidden channel in the block
        # --------------------------------------------------------------
        for d in range(BLOCK_D):
            left_d = left[:, :, d]      # (BLOCK_M, BLOCK_K)
            right_d = right[:, :, d]    # (BLOCK_K, BLOCK_N)
            acc[:, :, d] += tl.dot(left_d, right_d)

    # ------------------------------------------------------------------
    # Write the result back
    # ------------------------------------------------------------------
    out_ptrs = out_ptr \
        + pid_b * stride_ob \
        + i_range[:, None, None] * stride_oi \
        + j_range[None, :, None] * stride_oj \
        + tl.arange(0, BLOCK_D)[None, None, :] * stride_od

    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None],
    )


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------- Mask handling ---------------------------
    if config.get("nomask", False) or mask_tensor is None:
        mask = torch.ones(bs, seq_len, seq_len, dtype=dtype, device=device)
    else:
        mask = mask_tensor.to(dtype)

    # --------------------------- LayerNorm ------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # --------------------------- Linear projections -----------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # [bs, N, N, hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)  # same shape

    # --------------------------- Apply mask ------------------------------
    mask_unsq = mask.unsqueeze(-1)                 # [bs, N, N, 1]
    left = left * mask_unsq
    right = right * mask_unsq

    # --------------------------- Gates ----------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous tensors for Triton
    left = left.contiguous()
    right = right.contiguous()

    # --------------------------- Core TriMul (Triton) -------------------
    out_mul = torch.empty_like(left)   # shape [bs, N, N, hidden_dim]

    # Block sizes – tuned for typical B200 memory limits
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_D = 8                      # hidden_dim divisor (128, 384 are multiples)

    assert hidden_dim % BLOCK_D == 0, "hidden_dim must be a multiple of BLOCK_D"

    # Process hidden dimension in slices to keep register pressure low
    for d_start in range(0, hidden_dim, BLOCK_D):
        # Slice views (no copy)
        left_slice = left[..., d_start : d_start + BLOCK_D]
        right_slice = right[..., d_start : d_start + BLOCK_D]
        out_slice = out_mul[..., d_start : d_start + BLOCK_D]

        # Make sure slices are contiguous (they already are)
        left_slice = left_slice.contiguous()
        right_slice = right_slice.contiguous()
        out_slice = out_slice.contiguous()

        # Strides for the sliced tensors
        stride_lb, stride_li, stride_lk, stride_ld = left_slice.stride()
        stride_rb, stride_ri, stride_rk, stride_rd = right_slice.stride()
        stride_ob, stride_oi, stride_oj, stride_od = out_slice.stride()

        grid = (
            bs,
            (seq_len + BLOCK_M - 1) // BLOCK_M,
            (seq_len + BLOCK_N - 1) // BLOCK_N,
        )

        tri_mul_outgoing_kernel[grid](
            left_slice,
            right_slice,
            out_slice,
            bs,
            seq_len,
            stride_lb,
            stride_li,
            stride_lk,
            stride_ld,
            stride_rb,
            stride_ri,
            stride_rk,
            stride_rd,
            stride_ob,
            stride_oi,
            stride_oj,
            stride_od,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            BLOCK_D=BLOCK_D,
        )

    # --------------------------- Post‑processing ------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out_mul,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate

    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)   # [bs, N, N, dim]

    return out
