def triton_matmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b0, stride_left_b1, stride_left_b2, stride_left_b3,
    stride_right_b0, stride_right_b1, stride_right_b2, stride_right_b3,
    stride_out_b0, stride_out_b1, stride_out_b2, stride_out_b3,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # indexes
    slice_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # decode batch and channel
    b = slice_idx // H
    h = slice_idx % H

    # compute block start coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask within bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    for k in range(0, N, BLOCK_K):
        # compute k block offsets
        offs_k = k + tl.arange(0, BLOCK_K)
        # mask for k dimension
        mask_k = offs_k < N

        # compute pointers for left tile (M x K)
        # left: shape (B,N,N,H) indexing (b, i, k, h)
        a_ptrs = left_ptr + \
                 b * stride_left_b0 + \

"""
TriMul (outgoing) implementation.

The heavy O(N³·C) operation is the batched matrix multiplication
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
where `left` and `right` are already masked and gated.
We compute this step with a custom Triton kernel that processes each
(batch, hidden) slice independently (i.e. a separate matmul of shape
N×N). The surrounding LayerNorms, linear projections and gates are
performed with PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul for a single (batch, hidden) slice.
# ----------------------------------------------------------------------
@triton.jit
def _triton_trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b0, stride_left_b1, stride_left_b2, stride_left_b3,
    stride_right_b0, stride_right_b1, stride_right_b2, stride_right_b3,
    stride_out_b0, stride_out_b1, stride_out_b2, stride_out_b3,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    slice_idx = tl.program_id(0)          # encodes (batch, hidden)
    pid_m     = tl.program_id(1)          # block row
    pid_n     = tl.program_id(2)          # block col

    # decode batch and hidden channel
    b = slice_idx // H
    h = slice_idx % H

    # block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile:   shape (M, K)   from left[b, i, k, h]
        a_ptrs = left_ptr + \
                 b * stride_left_b0 + \
                 h * stride_left_b3 + \
                 offs_m[:, None] * stride_left_b1 + \
                 offs_k[None, :] * stride_left_b2

        # right tile transposed: shape (K, N) from right[b, j, k, h]
        # we need right indexed as (k, j) => swap axes 1 and 2
        b_ptrs = right_ptr + \
                 b * stride_right_b0 + \
                 h * stride_right_b3 + \
                 offs_k[:, None] * stride_right_b2 + \
                 offs_n[None, :] * stride_right_b1

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # write back
    out_ptrs = out_ptr + \
               b * stride_out_b0 + \
               h * stride_out_b3 + \
               offs_m[:, None] * stride_out_b1 + \
               offs_n[None, :] * stride_out_b2

    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Compute the TriMul (outgoing) forward pass.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor: torch.Tensor of shape [B, N, N, D]
        - mask        : torch.Tensor of shape [B, N, N] (may be ignored)
        - weights     : dict of torch.Tensor containing all linear / LN params
        - config      : dict containing at least:
            * "dim"          : int, input feature dimension D
            * "hidden_dim"   : int, hidden dimension H
            * "nomask"       : bool, True → ignore mask (use all‑ones)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype
    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]
    # ------------------------------------------------------------------
    # 1. LayerNorm (input)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (D,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, D]

    # ------------------------------------------------------------------
    # 2. Linear projections
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4. Apply mask (if present)
    # ------------------------------------------------------------------
    if config.get("nomask", False):
        mask_float = torch.ones((B, N, N), dtype=dtype, device=device)
    else:
        # ensure mask is float and broadcastable
        mask_float = mask.to(dtype)

    mask_broadcast = mask_float.unsqueeze(-1)   # [B,N,N,1]
    left = left_proj * mask_broadcast
    right = right_proj * mask_broadcast

    # ------------------------------------------------------------------
    # 5. Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Core TriMul via Triton (batched matmul over the K axis)
    # ------------------------------------------------------------------
    left_f16  = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    out_f16 = torch.empty((B, N, N, H), dtype=torch.float16, device=device)

    # strides for the 4‑D tensors (batch, dim1, dim2, channel)
    s_l = left_f16.stride()
    s_r = right_f16.stride()
    s_o = out_f16.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                               # one program per (batch, hidden)
        (N + BLOCK_M - 1) // BLOCK_M,        # block rows
        (N + BLOCK_N - 1) // BLOCK_N,        # block cols
    )

    _triton_trimul_outgoing_kernel[grid](
        left_f16,
        right_f16,
        out_f16,
        B,
        N,
        H,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_o[0], s_o[1], s_o[2], s_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # back to float32 for the remaining layers
    out = out_f16.to(torch.float32)

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, out_gate, final linear
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])  # [B,N,N,D]

    return out
