def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Get weight tensors
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]; right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]; right_gate_weight = weights["right_gate.weight"]; out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]; to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    eps = 1e-5

    # LayerNorm input
    # Compute mean and var along last dim
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = ((input_tensor - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (input_tensor - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias

    # Projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # shape [B,N,N,H]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask if provided and not nominal
    if mask is not None:
        # mask shape [B,N,N]
        # Ensure mask dtype matches left dtype
        mask = mask.to(left.dtype)
        # Use Triton kernel to apply mask to left and right
        # flatten left and right for mask kernel
        B, N, _, H = left.shape
        total = B*N*N*H
        BLOCK_SIZE = 1024
        grid = ( (total + BLOCK_SIZE - 1) // BLOCK_SIZE, )
        # Flatten pointers
        left_ptr = left.data_ptr()
        right_ptr = right.data_ptr()
        mask_ptr = mask.data_ptr()
        # Get strides as ints
        stride_left = left.stride()
        stride_right = right.stride()
        stride_mask = mask.stride()
        # Use triton c wrapper?
        # We'll use torch.utils.dlpack perhaps? But triton expects pointers as integers.
        # We can pass via left_ptr as torch tensor? We can pass left, right, mask tensors directly to kernel.
        mask_apply_kernel[grid](
            left, right, mask,
            B, N, H,
            stride_left[0], stride_left[1], stride_left[2], stride_left[3],
            stride_right[0], stride_right[1], stride_right[2], stride_right[3],
            stride_mask[0], stride_mask[1], stride_mask[2],

    # else mask is all ones; skip

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Core TriMul using batched matmul
    # left: [B,N,N,H] -> permute to [B,H,N,N] where (i,k)
    # right: [B,N,N,H] -> permute to [B,H,N,N] with (k,j)

    # Multiply by out_gate
    out = out * out_gate

    # Final linear
    out = torch.nn.functional.linear(out, to_out_weight)  # shape [B,N,N,dim]

    return out

"""
TriMul (outgoing) forward pass implemented with fused operations.
The heavy O(N³) interaction is computed via a batched GEMM (leveraging cuBLAS)
and a small Triton kernel is used to apply the pairwise mask efficiently.
All other linear / layer‑norm / gating steps are performed with PyTorch
operations for simplicity and maximum performance.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _apply_mask_kernel(
    left_ptr,               # float*  left tensor
    right_ptr,              # float*  right tensor
    mask_ptr,               # float*  mask tensor (broadcasted over channel dim)
    B, N, H,                # runtime arguments
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    BLOCK: tl.constexpr,
):
    """
    In‑place multiply left and right tensors by a binary mask.
    The tensors are 4‑D [B, N, N, H] and [B, N, N, H]; mask is 3‑D [B, N, N].
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    # Linear index decomposition: idx = b*(N*N*H) + i*(N*H) + k*H + h
    idx = tl.where(mask, offs, 0)

    NNH = N * N * H        # N * N * H
    b = idx // NNH
    rem = idx % NNH
    NH = N * H
    i = rem // NH
    rem2 = rem % NH
    k = rem2 // H
    h = rem2 % H

    # Compute flat offsets for each tensor
    left_off = (
        b * stride_left_b
        + i * stride_left_i
        + k * stride_left_k
        + h * stride_left_h
    )
    right_off = (
        b * stride_right_b
        + i * stride_right_i
        + k * stride_right_k
        + h * stride_right_h
    )
    mask_off = b * stride_mask_b + i * stride_mask_i + k * stride_mask_k

    # Load, multiply, store
    left = tl.load(left_ptr + left_off, mask=mask, other=0.0)
    right = tl.load(right_ptr + right_off, mask=mask, other=0.0)
    mval = tl.load(mask_ptr + mask_off, mask=mask, other=1.0)

    left = left * mval
    right = right * mval

    tl.store(left_ptr + left_off, left, mask=mask)
    tl.store(right_ptr + right_off, right, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask_tensor: torch.Tensor [B, N, N] (or None for no mask)
            - weights_dict: mapping str -> torch.Tensor (model parameters)
            - config_dict: contains at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Grab model parameters
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # LayerNorm on the input (dim‑wise)
    # ------------------------------------------------------------------
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = ((input_tensor - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (input_tensor - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias   # broadcasting over last dim

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_norm, left_proj_weight)   # (B, N, N, hidden_dim)
    right = F.linear(x_norm, right_proj_weight)

    # ------------------------------------------------------------------
    # Apply mask in‑place via a tiny Triton kernel (if a mask is supplied)
    # ------------------------------------------------------------------
    if mask is not None:
        # Ensure correct dtype for multiplication
        mask = mask.to(dtype)

        # Prepare strides for the kernel
        stride_left = left.stride()
        stride_right = right.stride()
        stride_mask = mask.stride()

        total_elems = B * N * N * hidden_dim
        BLOCK = 1024
        grid = ( (total_elems + BLOCK - 1) // BLOCK, )

        _apply_mask_kernel[grid](
            left,
            right,
            mask,
            B,
            N,
            hidden_dim,
            stride_left[0],
            stride_left[1],
            stride_left[2],
            stride_left[3],
            stride_right[0],
            stride_right[1],
            stride_right[2],
            stride_right[3],
            stride_mask[0],
            stride_mask[1],
            stride_mask[2],
            BLOCK=BLOCK,
        )
    # ------------------------------------------------------------------
    # Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core multiplicative update: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # This is a batched GEMM across the hidden dimension.
    # ------------------------------------------------------------------
    # Re‑order tensors to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2)          # (B, H, N, K)
    right_perm = right.permute(0, 3, 2, 1)        # (B, H, K, N)

    # Collapse batch & channel for a strided batched matmul
    BH = B * hidden_dim
    left_mat = left_perm.reshape(BH, N, N)       # (B*H, N, K)
    right_mat = right_perm.reshape(BH, N, N)     # (B*H, K, N)

    out_mat = torch.bmm(left_mat, right_mat)     # (B*H, N, N)

    # Restore original layout
    out = out_mat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # Output layer‑norm + gating + final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # (B, N, N, dim)

    return out
