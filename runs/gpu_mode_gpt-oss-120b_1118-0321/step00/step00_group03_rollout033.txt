```
@triton.jit
def proj_gate_kernel(
    x_ptr, w_proj_ptr, w_gate_ptr, mask_ptr, out_ptr,
    B, N, D, H,
    stride_x_b, stride_x_i, stride_x_j, stride_x_d,
    stride_w_h, stride_w_d,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # offsets
    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    h = tl.arange(0, BLOCK_H)

    # masks
    i_mask = i < N
    j_mask = j < N
    h_mask = h < H

    # mask value
    mask_ptrs = mask_ptr + pid_b * stride_mask_b + i[:, None] * stride_mask_i + j[None, :] * stride_mask_j
    mask = tl.load(mask_ptrs, mask=(i_mask[:, None] & j_mask[None, :]), other=0.0)  # shape (BLOCK_I, BLOCK_J)

    # accumulate
    acc_proj = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)
    acc_gate = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # loop over D in chunks
    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        cur_D = tl.minimum(BLOCK_D, D - d_off * BLOCK_D)

        d = tl.arange(0, BLOCK_D)
        # mask for d within cur_D
        d_mask = d < cur_D

        # load x slice
        x_ptrs = x_ptr + pid_b * stride_x_b \
                 + i[:, None, None] * stride_x_i \
                 + j[None, :, None] * stride_x_j \
                 + (d_off*BLOCK_D + d)[None, None, :] * stride_x_d
        x = tl.load(x_ptrs, mask=(i_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :]), other=0.0)  # shape (BLOCK_I, BLOCK_J, BLOCK_D)

        # load weight proj slice (H, D)
        w_proj_ptrs = w_proj_ptr + h[:, None] * stride_w_h + (d_off*BLOCK_D + d)[None, :] * stride_w_d
        w_gate_ptrs = w_gate_ptr + h[:, None] * stride_w_h + (d_off*BLOCK_D + d)[None, :] * stride_w_d

        w_proj = tl.load(w_proj_ptrs, mask=(h_mask[:, None] & d_mask[None, :]), other=0.0)  # shape (BLOCK_H, BLOCK_D)
        w_gate = tl.load(w_gate_ptrs, mask=(h_mask[:, None] & d_mask[None, :]), other=0.0)

        # accumulate dot product
        # x: (BLOCK_I, BLOCK_J, cur_D)
        # w: (BLOCK_H, cur_D)
        # broadcasting:
        # Multiply x[..., None, :] * w[None, None, :, :] -> shape (BLOCK_I, BLOCK_J, BLOCK_H, cur_D)
        prod_proj = x[..., None, :] * w_proj[None, None, :, :]
        prod_gate = x[..., None, :] * w_gate[None, None, :, :]
        # sum over d dimension (last dim)
        acc_proj += tl.sum(prod_proj, axis=3)
        acc_gate += tl.sum(prod_gate, axis=3)

    # sigmoid gate
    gate = tl.sigmoid(acc_gate)

    # apply mask broadcast
    mask_broadcast = mask[:, :, None]  # shape (BLOCK_I, BLOCK_J, 1)
    out = acc_proj * gate * mask_broadcast

    # store out
    out_ptrs = out_ptr + pid_b * stride_out_b \
               + i[:, None, None] * stride_out_i \
               + j[None, :, None] * stride_out_j \
               + h[None, None, :] * stride_out_h
    tl.store(out_ptrs, out, mask=(i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :]))
```

"""
TriMul (outgoing) forward implementation with a fused Triton kernel.

The heavy part of TriMul is the triple product:
    out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]

We compute the layer‑norm and the final linear projection with PyTorch
(because they are cheap relative to the O(N³) work) and accelerate the
projection + gating + mask step for the left and right tensors with a
custom Triton kernel.  The triple product itself is expressed as a
batched GEMM per hidden dimension (torch.bmm) which uses cuBLAS and is
far faster than a naïve Python loop.

The kernel `proj_gate_kernel` fuses:
    proj = x @ W_proj.T
    gate = sigmoid(x @ W_gate.T)
    out  = proj * gate * mask

where `x` has shape (B,N,N,D) and the result has shape (B,N,N,H).
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: fused projection + gate + mask
# --------------------------------------------------------------
@triton.jit
def proj_gate_kernel(
    x_ptr,               # *float32  (B,N,N,D)
    w_proj_ptr,          # *float32  (H,D)
    w_gate_ptr,          # *float32  (H,D)
    mask_ptr,            # *float32  (B,N,N)
    out_ptr,             # *float32  (B,N,N,H)

    B, N, D, H,          # runtime sizes
    stride_x_b, stride_x_i, stride_x_j, stride_x_d,
    stride_w_h, stride_w_d,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,

    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # ------------------------------------------------------------------
    # 1) compute the tile offsets
    # ------------------------------------------------------------------
    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    h = tl.arange(0, BLOCK_H)

    i_mask = i < N
    j_mask = j < N
    h_mask = h < H

    # ------------------------------------------------------------------
    # 2) load mask (broadcasted over hidden)
    # ------------------------------------------------------------------
    mask_off = pid_b * stride_mask_b + i[:, None] * stride_mask_i + j[None, :] * stride_mask_j
    mask = tl.load(mask_off, mask=(i_mask[:, None] & j_mask[None, :]), other=0.0)   # (BLOCK_I, BLOCK_J)

    # ------------------------------------------------------------------
    # 3) accumulate projection and gate contributions over D
    # ------------------------------------------------------------------
    acc_proj = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)
    acc_gate = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # loop over D in chunks of BLOCK_D
    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        d = tl.arange(0, BLOCK_D)
        d_idx = d_off * BLOCK_D + d          # absolute index in D
        d_mask = d_idx < D

        # ----- load x slice -------------------------------------------------
        x_off = (pid_b * stride_x_b
                 + i[:, None, None] * stride_x_i
                 + j[None, :, None] * stride_x_j
                 + d_idx[None, None, :] * stride_x_d)
        x = tl.load(
            x_off,
            mask=(i_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :]),
            other=0.0,
        )   # (BLOCK_I, BLOCK_J, BLOCK_D)

        # ----- load weight slices -------------------------------------------
        w_proj_off = h[:, None] * stride_w_h + d_idx[None, :] * stride_w_d
        w_gate_off = h[:, None] * stride_w_h + d_idx[None, :] * stride_w_d

        w_proj = tl.load(
            w_proj_ptr + w_proj_off,
            mask=(h_mask[:, None] & d_mask[None, :]),
            other=0.0,
        )   # (BLOCK_H, BLOCK_D)

        w_gate = tl.load(
            w_gate_ptr + w_gate_off,
            mask=(h_mask[:, None] & d_mask[None, :]),
            other=0.0,
        )   # (BLOCK_H, BLOCK_D)

        # ----- accumulate dot products ---------------------------------------
        # x: (i, j, d)   -> (BLOCK_I, BLOCK_J, 1, BLOCK_D)
        # w: (h, d)      -> (1, 1, BLOCK_H, BLOCK_D)
        prod_proj = x[..., None, :] * w_proj[None, None, :, :]   # (BI,BJ,BH,BD)
        prod_gate = x[..., None, :] * w_gate[None, None, :, :]

        acc_proj += tl.sum(prod_proj, axis=3)   # sum over D
        acc_gate += tl.sum(prod_gate, axis=3)

    # ------------------------------------------------------------------
    # 4) apply sigmoid gate and mask, store result
    # ------------------------------------------------------------------
    gate = tl.sigmoid(acc_gate)                     # (BI,BJ,BH)
    mask_b = mask[:, :, None]                       # (BI,BJ,1)
    out_val = acc_proj * gate * mask_b              # (BI,BJ,BH)

    out_off = (pid_b * stride_out_b
               + i[:, None, None] * stride_out_i
               + j[None, :, None] * stride_out_j
               + h[None, None, :] * stride_out_h)

    tl.store(
        out_off,
        out_val,
        mask=(i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :]),
    )
# --------------------------------------------------------------


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, D] (float32, on CUDA)
        - mask         : torch.Tensor[B, N, N] (bool/float32) or None
        - weights      : dict of parameter tensors
        - config       : dict with at least ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor[B, N, N, D]
        The updated pair representation.
    """
    # unpack
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # (B,N,N,dim)

    B, N, _, _ = x_norm.shape

    # ------------------------------------------------------------------
    # 2) Prepare mask (float32, broadcastable)
    # ------------------------------------------------------------------
    if mask is None:
        mask_f = torch.ones((B, N, N), dtype=dtype, device=device)
    else:
        mask_f = mask.to(dtype)

    # ------------------------------------------------------------------
    # 3) Allocate tensors for left and right updates
    # ------------------------------------------------------------------
    left = torch.empty((B, N, N, hidden_dim), dtype=dtype, device=device)
    right = torch.empty((B, N, N, hidden_dim), dtype=dtype, device=device)

    # ------------------------------------------------------------------
    # 4) Launch the Triton kernel for left and right branches
    # ------------------------------------------------------------------
    BLOCK_I, BLOCK_J, BLOCK_H, BLOCK_D = 16, 16, 32, 32

    # strides for the input tensor
    sx_b, sx_i, sx_j, sx_d = x_norm.stride()
    # strides for mask
    sm_b, sm_i, sm_j = mask_f.stride()
    # strides for output tensors (left/right)
    slo_b, slo_i, slo_j, slo_h = left.stride()

    # left branch
    proj_gate_kernel[
        (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)
    ](
        x_norm,
        weights["left_proj.weight"],
        weights["left_gate.weight"],
        mask_f,
        left,
        B,
        N,
        dim,
        hidden_dim,
        sx_b,
        sx_i,
        sx_j,
        sx_d,
        # weight strides (H, D) -> (stride_h, stride_d)
        weights["left_proj.weight"].stride(0),
        weights["left_proj.weight"].stride(1),
        # mask strides
        sm_b,
        sm_i,
        sm_j,
        # output strides
        slo_b,
        slo_i,
        slo_j,
        slo_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
    )

    # right branch (reuse same block config)
    proj_gate_kernel[
        (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)
    ](
        x_norm,
        weights["right_proj.weight"],
        weights["right_gate.weight"],
        mask_f,
        right,
        B,
        N,
        dim,
        hidden_dim,
        sx_b,
        sx_i,
        sx_j,
        sx_d,
        weights["right_proj.weight"].stride(0),
        weights["right_proj.weight"].stride(1),
        sm_b,
        sm_i,
        sm_j,
        slo_b,
        slo_i,
        slo_j,
        slo_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 5) Compute the output gate (global, same for all hidden dims)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"], None))  # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 6) Triple‑product via batched GEMM per hidden dimension
    # ------------------------------------------------------------------
    # change layout to (B, H, N, N) for efficient bmm
    left_t = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_t.shape
    left_flat = left_t.view(B * H, N, N)            # (B*H, N, N)
    right_flat = right_t.view(B * H, N, N)          # (B*H, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B*H, N, N)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)           # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7) Final LayerNorm, apply output gate, and project back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"], None)   # (B,N,N,dim)

    return out
