@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    pid_batch_hidden = tl.program_id(1)

    # Determine batch and hidden indices
    b = pid_batch_hidden // H
    h = pid_batch_hidden % H

    # compute tile indices for i (M) and j (N)
    num_m = tl.cdiv(N, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_n
    pid_n = pid % num_n

    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        # Offsets for k
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile
        left_ptrs = left_ptr + b * stride_left_b + h * stride_left_h + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # right tile transposed
        right_ptrs = right_ptr + b * stride_right_b + h * stride_right_h + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_j
        right_t = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(left, right_t)  # shape (BLOCK_M, BLOCK_N)

    # write results
    out_ptrs = out_ptr + b * stride_out_b + h * stride_out_h + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

    input_tensor, mask, weights, config = data
    B,N,N,C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm x
    # compute mean var across last dim
    # Use torch.nn.functional.layer_norm
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # Projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Mask
    if mask is not None and not config.get("nomask", False):

        left = left * mask_uns
        right = right * mask_uns

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    right = right * right_gate

    # allocate output
    out = torch.empty_like(left)  # shape B,N,N,hidden_dim
    # ensure contiguous
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # launch kernel
    import triton
    import triton.language as tl
    # kernel defined earlier

    # Compute block sizes based on hidden_dim and N, heuristics.

"""
TriMul (outgoing) forward pass implemented with a custom Triton kernel.

The kernel computes the core O(N³) operation:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is equivalent to a batched matrix multiplication
    left[b,d] @ right[b,d].T
for each hidden dimension d. The surrounding operations (LayerNorm,
linear projections, gating, final LayerNorm and Linear) are performed
with PyTorch for simplicity and correctness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel for the O(N³) triangle multiplicative update.
# -------------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,                     # tensor pointers
    B, H, N,                                         # batch, hidden, seq_len
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out = Σ_k left[...,k,:] * right[...,k,:]  (i.e. left @ rightᵀ)
    across the sequence dimension.

    Each program instance processes a tile (BLOCK_M x BLOCK_N) of the
    output matrix for a given (batch, hidden) pair.
    """
    pid = tl.program_id(0)          # tile index over i/j
    pid_bh = tl.program_id(1)       # combined batch*hidden index

    # -------------------------------------------------
    # Derive batch (b) and hidden (h) indices
    # -------------------------------------------------
    b = pid_bh // H
    h = pid_bh % H

    # -------------------------------------------------
    # Tile coordinates for the output matrix (i, j)
    # -------------------------------------------------
    num_m = tl.cdiv(N, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_n          # row tile index
    pid_n = pid % num_n           # col tile index

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # -------------------------------------------------
    # Accumulator for the output tile
    # -------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------
    # Loop over k dimension in BLOCK_K sized chunks
    # -------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- Load left tile: shape (BLOCK_M, BLOCK_K) -----
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + h * stride_left_h
            + offs_m[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        left = tl.load(
            left_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # ----- Load right tile transposed: shape (BLOCK_K, BLOCK_N) -----
        # right is indexed as (b, j, k, h); we need (k, j) order.
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + h * stride_right_h
            + offs_k[:, None] * stride_right_k
            + offs_n[None, :] * stride_right_j
        )
        right_t = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        # ----- Matrix‑multiply the tile and accumulate -----
        acc += tl.dot(left, right_t)      # (BLOCK_M, BLOCK_N)

    # -------------------------------------------------
    # Store the result tile
    # -------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + h * stride_out_h
        + offs_m[:, None] * stride_out_i
        + offs_n[None, :] * stride_out_j
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )

# -------------------------------------------------------------------------
# Entry point used by the evaluation harness.
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Perform the TriMul forward pass.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim]
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # Configuration
    dim = config["dim"]                # input channel dimension
    hidden_dim = config["hidden_dim"]  # projection dimension
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1) LayerNorm over the channel dimension
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # shape [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Optional mask application
    # -----------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        # Convert mask to the same dtype and broadcast
        mask_f = mask.to(dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4) Gating (sigmoid on linear projections of the normalized input)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Core TriMul operation via Triton kernel
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    # Ensure contiguous tensors for the kernel
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)   # same shape [B, N, N, hidden_dim]

    # Block sizes – chosen for typical GPU sweet spots
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    # Grid: (tiles over i/j, total batch*hidden)
    grid_x = ( (N + BLOCK_M - 1) // BLOCK_M ) * ( (N + BLOCK_N - 1) // BLOCK_N )
    grid = (grid_x, B * hidden_dim)

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        hidden_dim,
        N,
        left.stride(0),   # batch stride
        left.stride(1),   # i stride
        left.stride(2),   # k stride
        left.stride(3),   # hidden stride
        right.stride(0),  # batch stride
        right.stride(1),  # j stride (used as column after transpose)
        right.stride(2),  # k stride (used as row after transpose)
        right.stride(3),  # hidden stride
        out.stride(0),    # batch stride
        out.stride(1),    # i stride
        out.stride(2),    # j stride
        out.stride(3),    # hidden stride
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # -----------------------------------------------------------------
    # 6) Output layer norm, gating, and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                     # apply final gate
    out = F.linear(out, weights["to_out.weight"])   # back to dim

    return out
