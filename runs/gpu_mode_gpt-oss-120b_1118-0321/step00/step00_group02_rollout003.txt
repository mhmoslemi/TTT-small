import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def fused_mul3_kernel(a_ptr, b_ptr, c_ptr, out_ptr, total_elems,
                      BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    a = tl.load(a_ptr + offs, mask=mask).to(tl.float32)
    b = tl.load(b_ptr + offs, mask=mask).to(tl.float32)
    c = tl.load(c_ptr + offs, mask=mask).to(tl.float32)
    out = a * b * c
    tl.store(out_ptr + offs, out, mask=mask)

def fused_mul3(a, b, c):
    assert a.shape == b.shape == c.shape
    out = torch.empty_like(a)
    total = a.numel()
    BLOCK_SIZE = 8192
    grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)
    fused_mul3_kernel[grid](a_ptr=a.data_ptr(),
                            b_ptr=b.data_ptr(),
                            c_ptr=c.data_ptr(),
                            out_ptr=out.data_ptr(),
                            total_elems=total,
                            BLOCK_SIZE=BLOCK_SIZE)
    return out

"""
TriMul (outgoing) forward pass.

The implementation follows the reference PyTorch code but off‑loads the
element‑wise multiplications (projections × gates × mask and the final
gate multiplication) to tiny Triton kernels.  The heavy O(N³·D) part
(the einsum) is performed as a batched matrix multiplication using
cuBLAS (torch.bmm), which is already highly optimized on the target
B200 GPU.

Algorithm
---------
1. Layer‑norm the input tensor (last dimension = dim).
2. Linear projections: left_proj, right_proj = x·W_left/right.
3. Compute three gates (sigmoid of linear projections of x):
   left_gate, right_gate, out_gate.
4. Fuse element‑wise masks and gates with Triton kernels:
   left  = left_proj  * left_gate  * mask   (if mask present)
   right = right_proj * right_gate * mask   (if mask present)
5. Reshape to (B·D, N, N) and compute
   out = left @ rightᵀ  for each hidden dimension D
   using torch.bmm (batched GEMM).
6. Layer‑norm the result over the hidden dimension.
7. Apply the output gate with a Triton kernel.
8. Final linear projection back to the original dimension.

All tensors stay on CUDA; the kernels operate on contiguous memory and
support both float16 and float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernels
# --------------------------------------------------------------------------- #

@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, out_ptr, total_elems,
                           BLOCK_SIZE: tl.constexpr):
    """out = a * b  (1‑D flattened tensors)"""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


@triton.jit
def elementwise_mul3_kernel(a_ptr, b_ptr, c_ptr, out_ptr, total_elems,
                            BLOCK_SIZE: tl.constexpr):
    """out = a * b * c  (1‑D flattened tensors)"""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    c = tl.load(c_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b * c, mask=mask)


def elementwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Element‑wise multiplication via Triton."""
    assert a.shape == b.shape, "shape mismatch for elementwise_mul"
    out = torch.empty_like(a)
    total = a.numel()
    BLOCK_SIZE = 8192  # must be a multiple of 32
    grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)
    elementwise_mul_kernel[grid](
        a_ptr=a.data_ptr(),
        b_ptr=b.data_ptr(),
        out_ptr=out.data_ptr(),
        total_elems=total,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    return out


def elementwise_mul3(a: torch.Tensor, b: torch.Tensor, c: torch.Tensor) -> torch.Tensor:
    """Element‑wise multiplication of three tensors via Triton."""
    assert a.shape == b.shape == c.shape, "shape mismatch for elementwise_mul3"
    out = torch.empty_like(a)
    total = a.numel()
    BLOCK_SIZE = 8192
    grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)
    elementwise_mul3_kernel[grid](
        a_ptr=a.data_ptr(),
        b_ptr=b.data_ptr(),
        c_ptr=c.data_ptr(),
        out_ptr=out.data_ptr(),
        total_elems=total,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    return out


# --------------------------------------------------------------------------- #
# Custom kernel entry point
# --------------------------------------------------------------------------- #

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask        : torch.Tensor [B, N, N] or None
        - weights     : dict of torch.Tensor model weights
        - config      : dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------- #
    # Unpack arguments
    # ------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # Dimensions
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------- #
    # Load weights
    # ------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]

    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------- #
    # 1) LayerNorm on the input
    # ------------------------------------------------------------------- #
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # ------------------------------------------------------------------- #
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------- #
    left_proj  = F.linear(x_norm, left_proj_weight)   # [B,N,N,hidden_dim]
    right_proj = F.linear(x_norm, right_proj_weight)

    # ------------------------------------------------------------------- #
    # 3) Gates (sigmoid applied)
    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    # ------------------------------------------------------------------- #
    # 4) Apply mask and gates (element‑wise Triton kernels)
    # ------------------------------------------------------------------- #
    # fuse left_proj * left_gate (and mask if present)
    left = elementwise_mul(left_proj, left_gate)
    right = elementwise_mul(right_proj, right_gate)

    if not nomask and mask is not None:
        # Expand mask to hidden_dim and make it contiguous for the kernel
        mask_exp = mask.to(dtype).unsqueeze(-1).expand(-1, -1, -1, hidden_dim).contiguous()
        left = elementwise_mul(left, mask_exp)
        right = elementwise_mul(right, mask_exp)

    # ------------------------------------------------------------------- #
    # 5) Core O(N³·D) operation via batched GEMM (cuBLAS)
    # ------------------------------------------------------------------- #
    # reshape to (B*hidden_dim, N, N) for torch.bmm
    left_b = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_b = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # out_b shape: (B*hidden_dim, N, N)
    out_b = torch.bmm(left_b, right_b.transpose(1, 2))

    # reshape back to [B, N, N, hidden_dim]
    out = out_b.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 6) Output LayerNorm (over hidden_dim)
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # ------------------------------------------------------------------- #
    # 7) Apply the final gate (Triton)
    # ------------------------------------------------------------------- #
    out = elementwise_mul(out, out_gate)

    # ------------------------------------------------------------------- #
    # 8) Final linear projection back to dim
    # ------------------------------------------------------------------- #
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
