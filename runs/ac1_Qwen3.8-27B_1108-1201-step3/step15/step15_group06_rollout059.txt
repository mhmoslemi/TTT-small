def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def normalize_sum(a, target):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) * target / len(a)
    return a * target / s


def build_constraint_matrix(a, active_idx):
    n = len(a)
    size = 2 * n - 1
    m = len(active_idx)
    T = np.zeros((m, n))
    for row, k in enumerate(active_idx):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                T[row, j] = a[idx]
    return T


def solve_minmax_lp(a_fixed, solver, active_idx=None, normalize_sum_target=1.0, max_iters_hint=200):
    n = len(a_fixed)
    size = 2 * n - 1
    b = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    if active_idx is None or len(active_idx) >= size:
        T = build_constraint_matrix(a_fixed, np.arange(size))
    else:
        T = build_constraint_matrix(a_fixed, active_idx)
    cons = [T @ b <= t, cp.sum(b) == normalize_sum_target, b >= 0]
    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    t_val = float(t.value) if t.value is not None else float('inf')
    if val.sum() < 1e-15:
        return None, None
    return val, t_val


def solve_maxsum_lp(a_fixed, M, solver, active_idx=None):
    n = len(a_fixed)
    size = 2 * n - 1
    b = cp.Variable(n, nonneg=True)
    if active_idx is None or len(active_idx) >= size:
        T = build_constraint_matrix(a_fixed, np.arange(size))
    else:
        T = build_constraint_matrix(a_fixed, active_idx)
    cons = [T @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if val.sum() < 1e-15:
        return None
    return val


def get_subgradient(a, k_star):
    n = len(a)
    g = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            if 2 * j == k_star:
                g[j] = a[idx]
            else:
                g[j] = 2.0 * a[idx]
    return g


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, seed=0):
    rng = np.random.default_rng(seed)
    n = len(a0)
    size = 2 * n - 1
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    a_mom = a.copy()
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    stall_threshold = 60
    dual_momentum = np.zeros(size)
    iteration = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        iteration = t

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive active set with dual momentum
        if active_set is None or t % 30 == 0 or stall_count > stall_threshold // 2:
            # Use dual momentum to weight constraint importance
            tol_base = 0.02 + 0.03 * np.exp(-t / 200.0)
            threshold = M_curr * (1.0 - tol_base)
            candidate_active = np.where(conv_y >= threshold)[0]
            # Boost constraints with high dual momentum
            if dual_momentum.max() > 1e-10:
                weighted = conv_y + 0.1 * dual_momentum
                boost_active = np.where(weighted >= weighted.max() * (1.0 - tol_base * 1.5))[0]
                candidate_active = np.union1d(candidate_active, boost_active)
            if len(candidate_active) < 10:
                candidate_active = np.where(conv_y >= M_curr * 0.6)[0]
            if len(candidate_active) > 0.85 * size:
                candidate_active = np.where(conv_y >= M_curr * 0.85)[0]
            active_set = candidate_active

        # Solve min-max LP
        b_mm, t_mm = solve_minmax_lp(y, solver, active_set, normalize_sum_target=norm)

        # Also solve max-sum LP for blending
        b_ms = solve_maxsum_lp(y, M_curr, solver, active_set)

        # Determine best direction
        if b_mm is not None and b_ms is not None:
            g_mm = b_mm * norm / b_mm.sum()
            g_ms = b_ms * norm / b_ms.sum()
            # Blend: weight by quality
            mm_score = t_mm
            ms_score = M_curr
            w_mm = 1.0 / (mm_score + 1e-10)
            w_ms = 1.0 / (ms_score + 1e-10)
            w_total = w_mm + w_ms
            g = (w_mm * g_mm + w_ms * g_ms) / w_total
        elif b_mm is not None:
            g = b_mm * norm / b_mm.sum()
        elif b_ms is not None:
            g = b_ms * norm / b_ms.sum()
        else:
            # Fallback: subgradient step
            k_star = np.argmax(conv_y)
            g_sub = get_subgradient(y, k_star)
            g_sub = normalize_sum(y - 0.01 * g_sub / max(g_sub.max(), 1e-10), norm)
            g = g_sub

        # Check for constraint violations and expand active set
        if len(active_set) < size:
            inactive = np.setdiff1d(np.arange(size), active_set)
            if len(inactive) > 0 and len(inactive) < size - len(active_set):
                T_check = build_constraint_matrix(y, inactive[:min(len(inactive), 500)])
                viol = T_check @ g
                viol_idx = np.where(viol > M_curr * (1.0 + 1e-5))[0]
                if len(viol_idx) > 0:
                    active_set = np.union1d(active_set, inactive[viol_idx])

        # Nesterov update with restart
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_t) * y + alpha_t * g
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize_sum(y_new, norm)

        # Momentum step
        if t == 1:
            a_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            a_new = y_new + beta * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_sum(a_new, norm)

        prev_y = y
        y = y_new
        a_mom = a_new

        # Evaluate and track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a_mom).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a_mom.copy()
            stall_count = 0

        # Update dual momentum
        k_star = np.argmax(conv_y)
        k_star_new = np.argmax(fast_autocorr(y))
        dual_momentum = 0.95 * dual_momentum
        dual_momentum[k_star_new] += 1.0

        # Stall-based restart
        if stall_count > stall_threshold:
            y = best_a.copy()
            a_mom = best_a.copy()
            dual_momentum = np.zeros(size)
            active_set = None
            stall_count = 0
            # Small perturbation to escape
            pert = 1.0 + 0.02 * rng.standard_normal(n)
            pert = np.maximum(pert, 0.5)
            y = normalize_sum(y * pert, norm)

        # Periodic subgradient diversification
        if t % 100 == 50 and time.monotonic() - t0 < time_limit * 0.9:
            conv_cur = fast_autocorr(y)
            k_peak = np.argmax(conv_cur)
            g_sub = get_subgradient(y, k_peak)
            g_sub_norm = g_sub / max(g_sub.sum(), 1e-10)
            # Take a step in the subgradient direction (blended with LP direction)
            y_sg = normalize_sum(y - 0.005 * g_sub_norm * norm, norm)
            sc_sg = fast_autocorr(y_sg).max()
            if sc_sg < best_score:
                best_score = sc_sg
                best_a = y_sg.copy()
                y = y_sg.copy()

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.10f}, stall={stall_count}")

    return best_a, best_score


def subgradient_nesterov(a0, time_limit, seed=0):
    """Pure subgradient method with Nesterov acceleration (no LP)."""
    rng = np.random.default_rng(seed)
    n = len(a0)

    best_score = fast_autocorr(a).max()
    y = a.copy()
    a_mom = a.copy()

    stall_count = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()

        # Get subgradient from top-k active positions
        top_k = min(5, 2 * n - 1)
        top_idx = np.argsort(conv_y)[-top_k:]
        weights = np.zeros(2 * n - 1)
        weights[top_idx] = conv_y[top_idx] ** 2
        weights /= weights.sum()

        g = np.zeros(n)
        for k in top_idx:
            g_k = get_subgradient(y, k)
            g += weights[0] * 0  # placeholder
        # Proper weighted subgradient
        g = np.zeros(n)
        for i, k in enumerate(top_idx):
            w = conv_y[k] ** 2
            g_k = get_subgradient(y, k)
            g += w * g_k
        g /= max(g.sum(), 1e-10)

        # Nesterov step
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_t) * y + alpha_t * g
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize_sum(y_new, norm)

        if t == 1:
            a_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            a_new = y_new + beta * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_sum(a_new, norm)

        y = y_new
        a_mom = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:

def coordinate_descent(a0, time_limit, rng, verbose=True):
    """Coordinate descent with golden section on important coordinates."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_sum(a, norm)

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0
    phi = 0.6180339887498949

    # Identify important coordinates (those in active constraints)
    M_curr = conv.max()
    active_k = np.where(conv >= M_curr * 0.8)[0]
    important_coords = set()
    for k in active_k:
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        important_coords.update(range(lo, hi))
    if not important_coords:
        important_coords = set(range(n))

    def obj_val(a_vec):
        c = fast_autocorr(a_vec)
        s = a_vec.sum()
        if s < 1e-15:
            return float('inf')
        return c.max()

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False

        # Focus on important coordinates first
        coords = list(important_coords)
        rng.shuffle(coords)
        # Add some random coordinates
        extra = list(rng.choice(n, size=min(n // 5, 200), replace=False))
        coords.extend(extra)

        for i in coords:
            if time.monotonic() - t0 > time_limit:
                break

            ai = a[i]
            if ai < 1e-10 * norm:
                continue

            lo = 0.0
            hi = ai * 2.5 + norm * 0.05

            c1 = lo + (1 - phi) * (hi - lo)
            c2 = lo + phi * (hi - lo)

            a_test1 = a.copy()
            a_test1[i] = c1
            a_test1 = normalize_sum(a_test1, norm)
            f1 = obj_val(a_test1)

            a_test2 = a.copy()
            a_test2[i] = c2
            a_test2 = normalize_sum(a_test2, norm)
            f2 = obj_val(a_test2)

            for _ in range(25):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + (1 - phi) * (hi - lo)
                    a_test1 = a.copy()
                    a_test1[i] = c1
                    a_test1 = normalize_sum(a_test1, norm)
                    f1 = obj_val(a_test1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + phi * (hi - lo)
                    a_test2 = a.copy()
                    a_test2[i] = c2
                    a_test2 = normalize_sum(a_test2, norm)
                    f2 = obj_val(a_test2)

            optimal_val = min(f1, f2)
            a_test_opt = a.copy()
            a_test_opt[i] = c1 if f1 < f2 else c2
            a_test_opt = normalize_sum(a_test_opt, norm)
            sc = obj_val(a_test_opt)

            if sc < best_score - 1e-12:
                a = a_test_opt
                best_score = sc
                best_a = a.copy()
                pass_improved = True
                no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break

        # Refresh important coords every 3 passes
        if n_passes % 3 == 0:
            conv = fast_autocorr(a)
            M_curr = conv.max()
            active_k = np.where(conv >= M_curr * 0.7)[0]
            important_coords = set()
            for k in active_k:
                lo = max(0, k - n + 1)
                hi = min(n, k + 1)
                important_coords.update(range(lo, hi))

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def perturb_seq(a, rng, scale=0.05):
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.05)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s


def symmetric_perturb(a, rng, scale=0.03):
    """Perturb in a way that maintains some asymmetry (key for this problem)."""
    n = len(a)
    # Add asymmetric noise: more on one side
    side_bias = rng.uniform(0.3, 0.7)
    x = np.linspace(0, 1, n)
    noise = 1.0 + scale * rng.standard_normal(n)
    # Amplify noise on one side based on bias
    asym = 1.0 + 0.5 * (x - side_bias)
    noise = noise * asym
    noise = np.maximum(noise, 0.1)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s
