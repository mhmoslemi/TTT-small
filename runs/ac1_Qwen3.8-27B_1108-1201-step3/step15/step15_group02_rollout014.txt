from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(y):
    """Build the (2n-1) x n matrix T where T[k,i] = y[k-i] if 0<=k-i<n else 0"""
    n = len(y)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        # T[k, i] = y[k-i] for k = i, i+1, ..., i+n-1 (clipped to [0, size-1])
        k_start = i
        k_end = min(i + n, size)
        T[k_start:k_end, i] = y[:k_end - k_start]
    return T

def solve_lp_best_response(y, M, verbose=False):
    """Solve: max 1^T b s.t. T@b <= M, b >= 0
    Returns the optimal b."""
    n = len(y)
    size = 2 * n - 1
    T = build_conv_matrix(y)
    
    c = -np.ones(n)  # minimize -sum(b)
    A_ub = T
    b_ub = np.full(size, M)

    if res.status != 0:
        return None
    return np.maximum(res.x, 0)

import numpy as np

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def build_conv_matrix(y):
    n = len(y)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        k_end = min(i + n, size)
        T[i:k_end, i] = y[:k_end - i]
    return T


def solve_lp_best_response(y, M):
    n = len(y)
    size = 2 * n - 1
    T = build_conv_matrix(y)
    res = linprog(-np.ones(n), A_ub=T, b_ub=np.full(size, M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True,
                           'dual_feasibility_tolerance': 1e-12,
                           'primal_feasibility_tolerance': 1e-12})
    if res.status != 0:
        return None
    return np.maximum(res.x, 0.0)

            a = a * np.exp(-eta_s * pressure)
            a = np.maximum(a, 1e-14)
            a = a * norm / a.sum()

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


def quartic_smooth(a0, time_limit, verbose=False):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s

    best_a = a.copy()
    best_max = fast_conv_self(a).max()
    t0 = time.monotonic()

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    grad_fft_size = 1 << (3 * n - 2).bit_length()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0
    lr = 0.05 / np.sqrt(n)

    n_steps = int(time_limit * 50)
    for it in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        t_adam += 1

        A_pad = rfft(a, grad_fft_size)
        C_fft = A_pad * A_pad
        C_full = irfft(C_fft, grad_fft_size)

        c_trunc = C_full[:size]

        C_grad_fft = rfft(np.concatenate([c_trunc, np.zeros(grad_fft_size - size)]), grad_fft_size)
        A_grad_fft = rfft(np.concatenate([a, np.zeros(grad_fft_size - n)]), grad_fft_size)

        grad_full = 4.0 * irfft(C_grad_fft * np.conj(A_grad_fft), grad_fft_size)
        grad = grad_full[n:2 * n]

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t_adam)
        v_hat = v / (1 - beta2 ** t_adam)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new * norm / s_new
        a = a_new

        sc = fast_conv_self(a).max()
        if sc < best_max:
            best_max = sc
            best_a = a.copy()

        if verbose and it % 200 == 0:
            print(f"    Quartic step {it} ({time.monotonic()-t0:.0f}s): max={sc:.8f}")

    return best_a, best_max


def pair_cd_optimize(a0, time_limit, rng, verbose=False):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            peak_lags = np.where(conv >= cur_max * 0.9)[0]

            j_candidates = set()
            for _ in range(5):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:8]:
                j_candidates.add(int((kl - i + n) % n))
                j_candidates.add(int((kl + i) % n))
            j_candidates.discard(i)

            best_delta = 0.0
            best_v = cur_max
            best_j = -1

            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]

                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                d_max = min(ai, aj) * 0.9
                if d_max < 1e-12:
                    continue
                n_samples = 15
                deltas = np.linspace(0.02, d_max, n_samples)

                for d in deltas:
                    new_max = cur_max
                    peak_region = conv >= cur_max * 0.95
                    new_conv_peaks = conv[peak_region] + d * g[peak_region]
                    new_max = max(new_max, new_conv_peaks.max())
                    if 2 * i < size:
                        new_val = conv[2 * i] - 2 * ai * d + d * d
                        new_max = max(new_max, new_val)
                    if 2 * j < size:
                        new_val = conv[2 * j] + 2 * aj * d + d * d
                        new_max = max(new_max, new_val)
                    if new_max < best_v:
                        best_v = new_max
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0 and best_v < best_score - 1e-12:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = a * norm / a.sum()
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
                    total_improvements += 1
                else:
                    a[best_j] -= best_delta
                    a[i] += best_delta
                    a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 8:
                n_perturb = max(5, n // 60)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.1 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                a = a * norm / a.sum()
                if no_improve >= 15:
                    break
        else:
            no_improve = 0

        if verbose and total_improvements % 300 == 0 and total_improvements > 0:
            print(f"    PCD improv={total_improvements} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)

    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        f1 = rng.integers(1, 6)
        f2 = rng.integers(6, 20)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * f1 * x) + 0.2 * np.cos(2 * np.pi * f2 * x)
    elif choice == 3:
        c1 = rng.uniform(0.2, 0.5)
        c2 = rng.uniform(0.5, 0.8)
        a = np.exp(-30.0 * (x - c1) ** 2) + np.exp(-30.0 * (x - c2) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.5
    elif choice == 6:
        a = np.exp(-5.0 * x) + 0.4
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        n_peaks = rng.integers(3, 10)
        centers = rng.uniform(0.05, 0.95, n_peaks)
        a = np.zeros(n)
        for c in centers:
            width = rng.uniform(0.02, 0.1)
            a += np.exp(-(x - c) ** 2 / (2 * width ** 2))
    elif choice == 9:
        a = np.sinc(x - 0.5) ** 2 + 0.1
    elif choice == 10:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * x) + 0.2 * np.sin(2 * np.pi * 7 * x)
    else:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a = a + 0.005 * np.abs(rng.standard_normal(n))
    a = a / a.sum()
    return a


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def symmetrize(a):
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    s = a_sym.sum()
    if s < 1e-15:
        return a
    return a_sym * (a.sum() / s)


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1680.0)
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            old_best = best_score
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (was {old_best:.6f}, n={len(a)}, t={elapsed:.0f}s)")

    # Phase 1: LSE-Adam warm start (fast, ~80s)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 80.0)
    print(f"\n=== Phase 1: LSE-Adam n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 10:
        a_out, _ = lse_adam_optimize(best_seq, t1, verbose=True)
        check_and_update(a_out, f"lse1_n{len(a_out)}")

    # Phase 2: Main LP Nesterov with HiGHS (~500s)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.35, 500.0)
    print(f"\n=== Phase 2: LP Nesterov (HiGHS) n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, verbose=True, patience=120)
        check_and_update(a_out, f"lp1_n{len(a_out)}")

    # Phase 3: Vegas multiplicative (~100s)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.08, 100.0)
    print(f"\n=== Phase 3: Vegas n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_out, _ = vegas_optimize(best_seq, t3, verbose=False)
        check_and_update(a_out, f"vegas_n{len(a_out)}")

    # Phase 4: LP Nesterov round 2 from improved point (~300s)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.22, 300.0)
    print(f"\n=== Phase 4: LP Nesterov round 2 n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t4, verbose=True, patience=80)
        check_and_update(a_out, f"lp2_n{len(a_out)}")

    # Phase 5: Quartic smoothing (~80s)
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.06, 80.0)
    print(f"\n=== Phase 5: Quartic smooth n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_out, _ = quartic_smooth(best_seq, t5, verbose=True)
        check_and_update(a_out, f"quart_n{len(a_out)}")

    # Phase 6: Pair CD polish (~150s)
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 6: Pair CD n={len(best_seq)} (budget={t6:.0f}s) ===")
    if t6 > 15:
        a_out, _ = pair_cd_optimize(best_seq, t6, rng, verbose=True)
        check_and_update(a_out, f"pcd1_n{len(a_out)}")

    # Phase 7: LP Nesterov round 3 (~150s)
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 7: LP Nesterov round 3 n={len(best_seq)} (budget={t7:.0f}s) ===")
    if t7 > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t7, verbose=True, patience=60)
        check_and_update(a_out, f"lp3_n{len(a_out)}")

    # Phase 8: Symmetrization + final PCD
    remaining = time_budget - (time.monotonic() - t_global)
    a_sym = symmetrize(best_seq)
    check_and_update(a_sym, f"sym_n{len(a_sym)}")

    if remaining > 40:
        t8 = min(remaining * 0.15, 80.0)
        print(f"\n=== Phase 8: Final PCD n={len(best_seq)} (budget={t8:.0f}s) ===")
        a_out, _ = pair_cd_optimize(best_seq, t8, rng, verbose=False)
        check_and_update(a_out, f"pcd2_n{len(a_out)}")

    # Phase 9: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        lengths_to_try = [n_base + 200, n_base - 200, n_base + 400, n_base + 600]
        lengths_to_try = [x for x in lengths_to_try if 200 <= x <= 4000]
        for n_try in lengths_to_try:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 50:
                break
            t_try = min(remaining * 0.3, 50.0)
            print(f"\n=== Multi-length: n={n_try} (budget={t_try:.0f}s) ===")
            init_try = interpolate_seq(best_seq, n_try)
            a_out, _ = lse_adam_optimize(init_try, min(t_try, 25.0), verbose=False)
            check_and_update(a_out, f"lse_ml_n{n_try}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                t_lp = min(remaining * 0.4, 25.0)
                a_out, _ = nesterov_lp_optimize(a_out, t_lp, verbose=False, patience=50)
                check_and_update(a_out, f"lp_ml_n{n_try}")

    # Phase 10: Fresh starts
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Fresh starts (remaining={remaining:.0f}s) ===")
        for attempt in range(2):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            n_fresh = n_base + int(rng.integers(-200, 400))
            if n_fresh < 300:
                n_fresh = 800
            a_fresh = structured_init(n_fresh, rng)
            t_fresh = min(remaining * 0.3, 20.0)
            a_out, _ = lse_adam_optimize(a_fresh, t_fresh, verbose=False)
            check_and_update(a_out, f"fresh{attempt}_n{n_fresh}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20:
                t_lp = min(remaining * 0.4, 20.0)
                a_out, _ = nesterov_lp_optimize(a_out, t_lp, verbose=False, patience=40)
                check_and_update(a_out, f"freshlp{attempt}_n{n_fresh}")

    # Final polish with remaining time
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Final LP polish (remaining={remaining:.0f}s) ===")
        t_final = min(remaining * 0.7, 60.0)
        a_out, _ = nesterov_lp_optimize(best_seq, t_final, verbose=False, patience=50)
        check_and_update(a_out, f"final_lp_n{len(a_out)}")

    # Final evaluation
    if best_seq is not None:
        a_final = np.maximum(best_seq, 0.0)
        a_final = a_final / a_final.sum()
        final_score = eval_fn(a_final)
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
            best_seq = a_final
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
