            a_new = norm_seq(a_new, norm)
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_max:
                best_max = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best={best_max:.8f}")

    return best_a, best_max


def lp_solve_fast(a, M, solver, active_indices, warm_start=None):
    """Solve max sum(b) s.t. T[active]@b <= M, b >= 0, optionally warm-started."""
    n = len(a)
    size = 2 * n - 1

    b_var = cp.Variable(n, nonneg=True)

    # Build constraint rows for active indices
    # T[k, j] = a[k-j] if 0 <= k-j < n
    rows = []
    for k in active_indices:
        row = np.zeros(n)
        # T[k,j] = a[k-j] for j such that 0 <= k-j < n, i.e., k-n+1 <= j <= k
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_hi >= j_lo:
            row[j_lo:j_hi+1] = a[k - np.arange(j_lo, j_hi+1)]
        rows.append(row)

    T_active = np.array(rows) if rows else np.zeros((1, n))

    constraints = [T_active @ b_var <= M]

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

    # Warm start
    if warm_start is not None:
        try:
            prob.solve(solver=solver, verbose=False, **{f"{solver}_ARGS": {"warm_start": True}})
        except Exception:
            prob.solve(solver=solver, verbose=False)
    else:
        prob.solve(solver=solver, verbose=False)

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)


def lp_nesterov_optimize(a0, time_limit, solver, verbose=True):
    """Main optimization: LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = norm_seq(a, norm)

    best_a = a.copy()
    best_max = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    y = a.copy()  # Nesterov "lookahead" point
    prev_b = None

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Adaptive constraint screening
        # Start with fewer constraints, add more as t increases
        frac = min(1.0, 0.2 + 0.8 * t / 200.0)
        threshold = M * (1.0 - 0.15 * frac)
        active = np.where(conv_y >= threshold)[0]

        if len(active) < 50:
            threshold2 = M * 0.5
            active = np.where(conv_y >= threshold2)[0]

        b = lp_solve_fast(y, M, solver, active, warm_start=prev_b)
        if b is None:
            # Fallback: full constraint set
            active_full = np.where(conv_y >= M * 0.85)[0]
            b = lp_solve_fast(y, M, solver, active_full)
        if b is None:
            b = lp_solve_fast(y, M, solver, np.arange(size))
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Verify dropped constraints (every 10 iterations)
        if t % 10 == 0 and len(active) < size:
            check = fast_conv_cross(y, g)
            if check.max() > M * 1.01:
                active_full = np.where(conv_y >= M * 0.8)[0]
                b2 = lp_solve_fast(y, M, solver, active_full)
                if b2 is not None and b2.sum() > 1e-15:
                    g = b2 * norm / b2.sum()
                    b = b2

        prev_b = b

        # Nesterov with over-momentum for first half, standard for second half
        total_iters = 2000
        if t < total_iters // 2:
            alpha = 2.0 / (t + 2.0)
            momentum = 0.92  # over-momentum
        else:
            alpha = 2.0 / (t + 2.0)
            momentum = (t - 1.0) / (t + 2.0)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = norm_seq(y_new, norm)

        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        a_new = norm_seq(a_new, norm)

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_max:
            best_max = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_max:
            best_max = sc_a
            best_a = a.copy()

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Ne t={t} ({elapsed:.0f}s): best={best_max:.8f}, M={M:.6f}, n_active={len(active)}")

    return best_a, best_max


def dual_mwu_polish(a0, time_limit, verbose=True):
    """Use dual MWU to identify support, then project."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = norm_seq(a, norm)
    best_a = a.copy()
    best_max = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    M = fast_conv_self(a).max()
    eta = 0.5 / np.sqrt(n)
    y = np.ones(size) / size

    # Averaged dual for primal recovery
    y_avg = np.zeros(size)

    n_iter = 200
    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute T^T y
        Ty = cross_corr(a, y, fft_size)

        # Find most violated
        j_star = np.argmin(Ty)

        # Multiplicative update
        lo = j_star
        hi = min(size, j_star + n)
        a_seg = a[:hi - lo]
        y[lo:hi] *= (1.0 - eta * a_seg)
        y = np.maximum(y, 1e-20)

        y_avg += y

        if verbose and it % 50 == 0:
            print(f"    Dual MWU it={it}: min(Ty)={Ty.min():.4f}")

    # From averaged dual, identify support
    Ty_avg = cross_corr(a, y_avg, fft_size)
    # b_j should be positive where Ty_avg_j is close to 1 (complementary slackness)
    # Use soft thresholding
    threshold = np.percentile(Ty_avg, 60)
    support = np.where(Ty_avg <= threshold * 1.1)[0]

    # Project a onto the identified support
    if len(support) > n // 4:
        a_proj = np.zeros(n)
        a_proj[support] = a[support]
        a_proj = norm_seq(a_proj, norm)
        sc = fast_conv_self(a_proj).max()
        if sc < best_max:
            best_max = sc
            best_a = a_proj.copy()

    return best_a, best_max


def targeted_cd_polish(a0, time_limit, rng, verbose=True):
    """Targeted coordinate descent: adjust pairs of coordinates to reduce peak."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = norm_seq(a, norm)
    best_a = a.copy()
    best_max = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        # Find peak locations
        peak_thresh = cur_max * 0.93
        peaks = np.where(conv >= peak_thresh)[0]

        # Compute "pressure" for each coordinate (how much it contributes to peaks)
        pressure = np.zeros(n)
        for pk in peaks[:min(30, len(peaks))]:
            for j in range(n):
                k_minus_j = pk - j
                if 0 <= k_minus_j < n:
                    pressure[j] += a[k_minus_j]

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-8:
                continue

            # Try moving mass from i to j
            # Pick j with low pressure
            j_candidates = np.argsort(pressure)[:20]
            j_candidates = j_candidates[j_candidates != i]
            if len(j_candidates) == 0:
                continue

            best_val = cur_max
            best_delta = 0.0
            best_j = -1

            for j in j_candidates[:12]:
                aj = a[j]

                # Compute delta_conv for moving d from i to j
                # new_conv = conv + d * (gain_j - loss_i)
                # gain_j[k] = a[k-j] + a[j-k] (for valid indices)  ... actually:
                # d/d a_j conv[k] = a[k-j] + a[j-k] (symmetric part)
                # More precisely: conv[k] = sum_i a[i]*a[k-i]
                # d/d a_j conv[k] = a[k-j] + a[j+k-(2j)] ... 
                # Actually: d/d a_j [sum_i a[i]a[k-i]] = a[k-j] + a[j] (if k-j valid) + ... 
                # Let me just use: if we add d to a[j] and subtract d from a[i],
                # delta_conv[k] = d * (a[k-j] + a[j-k] - a[k-i] - a[i-k]) (with bounds)
                # Plus the second-order terms at k=2i, k=2j, k=i+j, k=i-j

                # For simplicity, use the linear approximation:
                # delta_conv[k] ≈ d * (a[k-j] + a[j-k] - a[k-i] - a[i-k])
                # where a[x] = 0 if x not in [0, n-1]

                g_j = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                g_j[lo:hi] += a[0:hi-lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g_j[lo2:hi2] += a[j + np.arange(lo2, hi2) - np.arange(lo2, hi2)]
                    # Actually a[j-k] for k in [lo2, hi2): a[j-lo2], ..., a[j-hi2+1]
                    # j-k ranges from j-lo2 to j-hi2+1
                    # We need a[index] where index = j-k, k from lo2 to hi2-1
                    # index from j-lo2 to j-(hi2-1) = j-hi2+1
                    # So g_j[lo2:hi2] += a[j-lo2 : j-hi2+2][::-1]
                    g_j[lo2:hi2] = a[j - np.arange(lo2, hi2)][::-1]

                g_i = np.zeros(size)
                lo = max(0, i)
                hi = min(size, i + n)
                g_i[lo:hi] += a[0:hi-lo]
                lo2 = max(0, i - n + 1)
                hi2 = min(size, i + 1)
                g_i[lo2:hi2] = a[i - np.arange(lo2, hi2)][::-1]

                delta_g = g_j - g_i

                # Binary search for best delta
                d_max = ai * 0.9
                n_trials = 15
                deltas = np.linspace(0, d_max, n_trials)
                for d in deltas:
                    if d < 1e-12:
                        continue
                    new_conv = conv + d * delta_g
                    # Second order corrections
                    if 2*i < size:
                        new_conv[2*i] -= 2*ai*d - d*d
                    if 2*j < size:
                        new_conv[2*j] += 2*aj*d + d*d
                    if (i+j) < size and i != j:
                        new_conv[i+j] += 2*d*(ai - aj) if i < j else 2*d*(aj - ai)
                        # Actually the cross term: d*a[j]*a[i] - d*a[i]*a[j] = 0 (they cancel in the linear part)
                        # The second order at k=i+j: +d*ai + d*aj - d*ai - d*aj = 0? No...
                        # Let me be more careful.
                        # Original conv[i+j] includes terms a[i]*a[j] and a[j]*a[i] = 2*a[i]*a[j]
                        # After: (a[i]-d)*a[j] + a[j]*(a[i]-d) + ... = 2*a[i]*a[j] - 2*d*a[j] + ...
                        # And at j: a[j]+d contributes (a[j]+d)*a[i] + ... 
                        # This is getting complicated. Let me just do a full recompute for the final candidate.
                    v = new_conv.max()
                    if v < best_val:
                        best_val = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = norm_seq(a, norm)
                new_max = fast_conv_self(a).max()
                if new_max < best_max - 1e-12:
                    best_max = new_max
                    best_a = a.copy()
                    pass_improved = True

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
            # Perturb
            n_pert = max(10, n // 25)
            idx_pert = rng.choice(n, n_pert, replace=False)
            a[idx_pert] *= (1.0 + 0.12 * rng.standard_normal(n_pert))
            a = np.maximum(a, 0.0)
            a = norm_seq(a, norm)
        else:
            no_improve = 0

        if verbose and (idx + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass ({elapsed:.0f}s): best={best_max:.8f}")

    return best_a, best_max


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    r = rng.random()

    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** (1.0 + r)
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-5.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-5.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** (1.0 + r)
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-3.0 * x) * 0.6 + 0.4 + 0.2 * np.sin(3 * np.pi * x)
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        a = np.sinc(2.5 * (x - 0.5)) ** 2 + 0.1
    elif choice == 9:
        # Asymmetric: steep rise, gentle fall
        a = np.where(x < 0.4, x / 0.4, (1-x) / 0.6 ** 0.8)
    elif choice == 10:
        # Periodic with phase
        phase = rng.random() * 2 * np.pi
        a = 0.5 + 0.3 * np.sin(6 * np.pi * x + phase) + 0.1 * np.sin(10 * np.pi * x)
    else:
        a = np.exp(-2.0 * (x - 0.5) ** 2 / (0.1 + 0.15 * r))

    a = np.maximum(a, 0.0)
    a = a + 0.005 * rng.random(n)
    return norm_seq(a, 1.0)


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def eval_local(a):
    """Local evaluation matching the external evaluate_sequence."""
    try:
        n = len(a)
        if n == 0:
            return float('inf')
        if np.any(a < 0) or not np.all(np.isfinite(a)):
            return float('inf')
        b_sequence = np.convolve(a, a)
        max_b = max(b_sequence)
        sum_a = np.sum(a)
        if sum_a < 0.01:
            return float('inf')
        return float(2 * n * max_b / (sum_a ** 2))
    except Exception:
        return float('inf')

    rng = np.random.default_rng(seed)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}, solver={solver}")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_local(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):

        if a is None:

        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
