def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = np.asarray(f, dtype=np.float64)
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def spectral_shaping(a0, time_limit, verbose=True):
    """
    BLIMP-inspired spectral shaping to reduce peak autocorrelation.
    Works in frequency domain to flatten power spectrum.
    """
    a = norm_sum1(a0)
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    alpha = 0.3
    min_alpha = 0.001
    decay = 0.9995
    stall = 0
    max_stall = 300
    
    for it in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if alpha < min_alpha:
            # Reset with smaller steps
            alpha = 0.1
            a = best_a.copy()
        
        A = np.fft.rfft(a, N)
        P = A.real ** 2 + A.imag ** 2
        
        # Compute autocorrelation from spectrum
        r = np.fft.irfft(P, N)
        r_lin = r[:2 * n - 1]
        
        # Find top peaks to target
        top_k = min(10, 2 * n - 1)
        peak_indices = np.argsort(r_lin)[::-1][:top_k]
        peak_vals = r_lin[peak_indices]

        freqs = np.fft.rfftfreq(N)
        influence = np.zeros(len(freqs))

def targeted_coordinate_descent(a0, time_limit, verbose=True, rng_seed=42):

        ac = fast_autocorr(a)
        k_peak = int(np.argmax(ac))
        peak_val = ac[k_peak]
        
        # Find the "peak region": positions that contribute to the top peaks
        top_peaks = np.argsort(ac)[::-1][:min(20, len(ac))]

        sensitivity = np.zeros(n)
        for k in top_peaks[:10]:
            k_i = int(k)
            weight = ac[k_i] / max(ac[top_peaks[0]], 1e-30)
            i_start = max(0, k_i - n + 1)
            i_end = min(n - 1, k_i)
            for i in range(i_start, i_end + 1):
                j = k_i - i
                if 0 <= j < n:
                    sensitivity[i] += weight * a[j]
        
        # Find high-sensitivity (source) and low-sensitivity (destination) positions
        src_candidates = np.where(sensitivity > np.median(sensitivity))[0]
        dst_candidates = np.where(sensitivity < np.median(sensitivity))[0]
        
        if len(src_candidates) == 0 or len(dst_candidates) == 0:
            src_candidates = np.arange(n)
            dst_candidates = np.arange(n)
        
        improved = False

            if time.monotonic() - t0 > time_limit:
                break
            
            src = src_candidates[rng.integers(len(src_candidates))]
            dst = dst_candidates[rng.integers(len(dst_candidates))]
            
            if src == dst or a[src] < 1e-15:
                continue
            
            # Try different fractions
            for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_M - 1e-15:
                    a = a_trial
                    best_M = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall % 50 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                while j == i:
                    j = int(rng.integers(n))
                if a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.1)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_M:
                        best_M = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
    
    J = 2.0 * n * best_M
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    Coord ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_M

            break
        
        # Build constraint matrix T: (T @ b)[k] = (y * b)[k]
        # T has shape (2n-1, n)
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = y
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        iter_count += 1
        
        my = fast_autocorr(y).max()
        if my < best_M:
            best_M = my
            best_a = y.copy()
        
        mx = fast_autocorr(x).max()
        if mx < best_M:
            best_M = mx
            best_a = x.copy()
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_M


def accelerated_mirror_descent(a0, time_limit, top_k=50, beta=5.0, c0=3.0, verbose=True):
    """Nesterov-accelerated mirror descent with soft subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(54321)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    max_stall = 4000
    cur_top_k = top_k
    cur_beta = beta
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        ac = fast_autocorr(y)
        # Soft subgradient from top peaks
        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]
        
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_M - 1e-14:
            best_M = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > max_stall:
            stall_count = 0
            perturbation = rng.normal(0, 0.008, n)
            x_restarted = best_a + perturbation * best_a
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted
            cur_top_k = min(cur_top_k + 10, 80)
            cur_beta = max(cur_beta - 0.5, 2.0)
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    AMD restart t={t} ({elapsed:.1f}s): top_k={cur_top_k}, beta={cur_beta:.1f}")
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    AMD t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_M


def spectral_shaping_v2(a0, time_limit, verbose=True):
    """
    Improved spectral shaping with adaptive peak targeting and
    multi-scale frequency updates.
    """
    a = norm_sum1(a0)
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Multi-phase: start aggressive, end gentle
    phases = [
        ('aggressive', 0.3, 0.05, 3000),
        ('moderate', 0.15, 0.02, 5000),
        ('gentle', 0.05, 0.005, 10000),
    ]
    
    for phase_name, alpha_init, alpha_min, max_iters in phases:
        if time.monotonic() - t0 > time_limit:
            break
        alpha = alpha_init
        stall = 0
        max_stall = 200
        
        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                break
            if alpha < alpha_min:
                break
            if stall > max_stall:
                a = best_a.copy()
                alpha = max(alpha * 0.8, alpha_min)
                stall = 0
            
            A = np.fft.rfft(a, N)
            P = A.real ** 2 + A.imag ** 2
            P_max = P.max()
            if P_max < 1e-30:
                break
            
            r = np.fft.irfft(P, N)
            r_lin = r[:2 * n - 1]
            
            # Target top 5 peaks with emphasis on the global max
            n_peaks = min(5, len(r_lin))
            peak_idx = np.argsort(r_lin)[::-1][:n_peaks]
            peak_vals = r_lin[peak_idx]
            
            freqs = np.fft.rfftfreq(N)
            influence = np.zeros(len(freqs))
            for pk, pv in zip(peak_idx, peak_vals):
                weight = (pv / peak_vals[0]) ** 3  # Heavy emphasis on global max
                influence += weight * np.cos(2 * np.pi * freqs * pk)
            
            # Smooth the influence to avoid overly aggressive updates
            # Apply a lowpass to the influence (temporal smoothness in freq domain)
            if len(influence) > 4:
                kernel = np.ones(3) / 3.0
                influence_smooth = np.convolve(influence, kernel, mode='same')
                influence = 0.7 * influence + 0.3 * influence_smooth
            
            # Update
            infl_max = np.abs(influence).max()
            if infl_max < 1e-15:
                break
            influence = influence / infl_max  # Normalize
            
            scale = 1.0 - alpha * np.maximum(influence, 0) + alpha * 0.2 * np.maximum(-influence, 0)
            scale = np.maximum(scale, 0.05)
            
            magnitude = np.sqrt(P)
            new_mag = magnitude * np.sqrt(scale)
            phase = np.angle(A)
            A_new = new_mag * np.exp(1j * phase)
            
            a_new = np.fft.irfft(A_new, N)[:n]
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s_new
            
            M_new = fast_autocorr(a_new).max()
            if M_new < best_M - 1e-14:
                best_M = M_new
                best_a = a_new.copy()
                stall = 0
                a = a_new
            else:
                a = a_new
                stall += 1
            
            alpha *= 0.9998
        
        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    Spectral v2 [{phase_name}] ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_M


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator combining spectral shaping, coordinate descent,
    LP Frank-Wolfe, and AMD for minimizing the autocorrelation peak metric.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        p = np.maximum(p, 0.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence([float(x) for x in parent]))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"  Warning: could not load parent: {e}")
    
    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = float(evaluate_sequence([float(x) for x in best_seq]))
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        try:
            sc = float(evaluate_sequence([float(x) for x in a_norm]))
        except Exception:
            return
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")
    
    # === Phase 1: Spectral shaping (novel method) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Spectral Shaping v2, n={n0} ===")
    t1 = min(remaining * 0.20, 300.0)
    if t1 > 30:
        a_out, _ = spectral_shaping_v2(norm_sum1(best_seq), t1, verbose=True)
        try_update(a_out, f"spec2_n{n0}")
    
    # === Phase 2: AMD from current best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: AMD, n={n_best} ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 30:
        a_out, _ = accelerated_mirror_descent(norm_sum1(best_seq), t2, top_k=50, beta=5.0, c0=3.5, verbose=True)
        try_update(a_out, f"amd_n{n_best}")
    
    # === Phase 3: LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP Frank-Wolfe, n={n_best} ===")
    t3 = min(remaining * 0.20, 280.0)
    if t3 > 30:
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t3, solver, verbose=True)
        try_update(a_out, f"lp_n{n_best}")
    
    # === Phase 4: Targeted coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coordinated CD, n={n_best} ===")
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 20:
        a_out, _ = targeted_coordinate_descent(norm_sum1(best_seq), t4, verbose=True)
        try_update(a_out, f"coord_n{n_best}")
    
    # === Phase 5: Spectral shaping again (different phase of the optimization) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Spectral Shaping round 2, n={n_best} ===")
    t5 = min(remaining * 0.12, 180.0)
    if t5 > 20:
        a_out, _ = spectral_shaping(norm_sum1(best_seq), t5, verbose=True)
        try_update(a_out, f"spec1_n{n_best}")
    
    # === Phase 6: Try a larger length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    n_large = min(n_best + 1200, 3600)
    if n_large > n_best and remaining > 150:
        print(f"\n=== Phase 6: Scale to n={n_large} ===")
        a_long = interpolate_sequence(best_seq, n_large)
        # Spectral shape at larger length
        t6a = min(remaining * 0.15, 200.0)
        if t6a > 30:
            a_out, _ = spectral_shaping_v2(norm_sum1(a_long), t6a, verbose=True)
            try_update(a_out, f"spec2_n{n_large}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t6b = min(remaining * 0.10, 100.0)
            if t6b > 20:
                a_out, _ = accelerated_mirror_descent(norm_sum1(a_long if best_seq is None else a_out), t6b, top_k=40, beta=5.0, c0=3.0, verbose=False)
                try_update(a_out, f"amd_n{n_large}")
    
    # === Phase 7: LP polish at best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if n_best <= 2800 and remaining > 80:
        print(f"\n=== Phase 7: LP polish, n={n_best} ===")
        t7 = min(remaining * 0.15, 200.0)
        if t7 > 30:
            a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t7, solver, verbose=True)
            try_update(a_out, f"lp2_n{n_best}")
    
    # === Phase 8: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final CD, n={n_best} ===")
    t8 = min(remaining * 0.15, 180.0)
    if t8 > 20:
        a_out, _ = targeted_coordinate_descent(norm_sum1(best_seq), t8, verbose=True, rng_seed=777)
        try_update(a_out, f"coord_final_n{n_best}")
    
    # === Phase 9: Final spectral + coord combo ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 60:
        print(f"\n=== Phase 9: Final spectral, n={n_best} ===")
        t9 = min(remaining * 0.4, 200.0)
        if t9 > 30:
            a_out, _ = spectral_shaping_v2(norm_sum1(best_seq), t9, verbose=True)
            try_update(a_out, f"spec_final_n{n_best}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t9b = min(remaining * 0.6, 150.0)
            if t9b > 20:
                a_out, _ = targeted_coordinate_descent(norm_sum1(best_seq), t9b, verbose=True, rng_seed=1234)
                try_update(a_out, f"coord_very_final_n{n_best}")
    
    # Final validation
    if best_seq is not None:
        try:
            fs = float(evaluate_sequence([float(x) for x in best_seq]))
            if np.isfinite(fs):
                best_score = fs
        except Exception:
            pass
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
