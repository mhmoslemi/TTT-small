def freq_smooth(a, alpha):
    """Apply frequency-domain smoothing to reduce high-frequency content."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    # Taper high frequencies
    freqs = np.arange(fft_size // 2 + 1)
    max_freq = fft_size // 2
    # Gaussian taper centered at low frequencies
    taper = np.exp(-alpha * (freqs / max_freq)**2)
    A_tapered = A * taper
    a_new = irfft(A_tapered, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    return a_new / s * a.sum()

        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.15 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0.1)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-3.0 * x) + 0.5 * np.exp(-3.0 * (1 - x))
    elif choice == 7:
        a = 0.5 * np.exp(-5.0 * (x - 0.25) ** 2) + 0.3 * np.exp(-5.0 * (x - 0.6) ** 2) + 0.2 * np.exp(-5.0 * (x - 0.9) ** 2)
    elif choice == 8:
        a = np.abs(np.sin(2.5 * np.pi * x)) ** 0.7
    elif choice == 9:
        a = 0.4 + 0.3 * np.cos(5 * np.pi * x) + 0.2 * np.cos(13 * np.pi * x)
        a = np.maximum(a, 0.05)
    elif choice == 10:
        a = np.power(np.sin(np.pi * x), 0.5) * (1 + 0.2 * np.cos(8 * np.pi * x))
        a = np.maximum(a, 0.0)
    else:
        # Random smooth function
        coeffs = rng.standard_normal(8) * 0.3
        a = np.ones(n)
        for k, c in enumerate(coeffs):
            a += c * np.cos(2 * np.pi * (k + 1) * x)
        a = np.maximum(a, 0.01)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def adaptive_length_search(best_seq, best_score, time_budget, solver, rng, verbose=True):
    """Try slightly different lengths to see if there's improvement."""
    n_base = len(best_seq)
    candidates = []
    
    # Try n_base +/- adjustments
    for dn in [-200, -100, 100, 200, 300]:
        n_try = n_base + dn
        if n_try < 100 or n_try > 5000:
            continue
        init = interpolate_seq(best_seq, n_try)
        sc = compute_score(init)
        candidates.append((sc, n_try, init))
    
    candidates.sort(key=lambda x: x[0])
    results = []
    
    for sc, n_try, init in candidates[:3]:
        remaining = time_budget - (time.monotonic() - time_budget + time_budget)
        # Simple quick optimization
        t_alloc = min(time_budget * 0.15, 120.0)
        if t_alloc > 20:
            a_out, out_sc = lp_nesterov_optimize(init, t_alloc, solver, verbose=False)
            final_sc = compute_score(a_out)
            results.append((final_sc, a_out))
            if verbose:
                print(f"    Length {n_try}: score={final_sc:.6f}")
    
    return results
