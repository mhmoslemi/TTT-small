Let me think about a **block coordinate** approach instead:
- Divide the sequence into blocks of size b (e.g., b=8 or b=16)
- For each block, optimize the block values while keeping the rest fixed
- This is a lower-dimensional subproblem

For a block of size b at position p, the contribution to the autocorrelation at lag k from this block is:
`sum_{i=0}^{b-1} a[p+i] * a[k-(p+i)]` (for valid indices)

The block optimization would be: minimize `max_k (existing_k + block_contribution_k)` over the b block variables, subject to them being non-negative.

This is a small LP (b variables, 2n-1 constraints) that can be solved quickly!

Actually, this is a nice hybrid: LP for small blocks, gradient for the overall direction.

The primal is: minimize `max_k (a*a)[k]` s.t. `a >= 0`, `sum(a) = sqrt(2n)`.

The dual involves finding a probability distribution over lags that "witnesses" a lower bound. Specifically, if we can find weights w_k >= 0, sum w_k = 1, such that for all i: `sum_k w_k * 2*a[k-i] <= c` (where c is the dual bound), then c is a lower bound on the optimal value.

This is related to the LP approach in the previous code.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based subgradient descent with Nesterov momentum and adaptive active sets
2. Block LP optimization (small-scale convex subproblems per block)
3. Spectral peak shaving in frequency domain
4. Multi-scale coordinate descent polishing
5. Multi-restart with perturbation and cross-pollination
6. Adaptive length exploration
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft

# --- Global solver cache ---
_solver_cache = None

def _get_solver():

    if _solver_cache is not None:
        return _solver_cache
    for s in [cp.CLARABEL, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver_cache = s
                print(f"  Solver: {s}")

        except Exception:

    _solver_cache = cp.SCS
    return _solver_cache


def _fast_autocorr(a):
    """Full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def _score(a):
    """Compute objective: 2n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    conv = _fast_autocorr(a)
    return 2.0 * n * conv.max() / (s * s)


def _norm_to(a, target=None):
    """Normalize a to target sum (default sqrt(2n))."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    s = a.sum()
    if s < 1e-12:
        return np.full(len(a), target / len(a))
    return a * target / s


def _grad_at_lags(a, lags):
    """Subgradient of max autocorrelation: for each active lag k, grad[i] += 2*a[k-i]."""
    n = len(a)
    grad = np.zeros(n)
    for k in lags:
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo >= hi:
            continue
        # grad[lo:hi] += 2 * a[k-hi+1 : k-lo+1][::-1]
        start = k - hi + 1
        end = k - lo + 1
        grad[lo:hi] += 2.0 * a[start:end][::-1]
    return grad


def _perturb(a, rng, scale=0.05):
    """Multiplicative Gaussian perturbation."""
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.01)
    ap = a * noise
    return _norm_to(ap)


def _interpolate_to(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    s = f.sum()
    if s < 1e-12:
        return np.full(new_n, 1.0 / new_n)
    fn = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-12:
        return np.full(new_n, 1.0 / new_n)
    return fi / ss


def _subgrad_nesterov(a0, time_limit, seed=0, K=30, verbose=False):
    """Subgradient descent with Nesterov momentum."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = _norm_to(a0.copy(), norm)
    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    
    y = a.copy()
    v = np.zeros(n)
    momentum = 0.85
    lr = 0.02
    lr_min = 1e-5
    lr_max = 0.5
    t0 = time.monotonic()
    stall = 0
    n_steps = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = _fast_autocorr(y)
        M = conv_y.max()
        
        # Adaptive active set: top K lags that are close to max
        tol = 0.02 + 0.08 * min(1.0, 200.0 / (t + 1.0))
        threshold = M * (1.0 - tol)
        active = np.where(conv_y >= threshold)[0]
        if len(active) < 5:
            active = np.argsort(conv_y)[::-1][:10]
        if len(active) > K:
            top_order = np.argsort(-conv_y, kind='stable')
            active = top_order[:K]
        
        # Subgradient
        g = _grad_at_lags(y, active)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        g = g / g_norm
        
        # Momentum update
        v = momentum * v + lr * g
        a_new = y - v
        a_new = np.maximum(a_new, 0.0)
        a_new = _norm_to(a_new, norm)
        
        # Evaluate
        score_new = _fast_autocorr(a_new).max()
        n_steps += 1
        
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            stall = 0
            # Slightly increase lr
            lr = min(lr * 1.05, lr_max)
        else:
            stall += 1
            if stall > 30:
                lr = max(lr * 0.7, lr_min)
                stall = 0
            if stall > 200:
                # Reset
                y = best_a.copy()
                v = np.zeros(n)
                lr = 0.02
                stall = 0
                break
        
        y = a_new
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SubGrad t={t} ({elapsed:.0f}s): best={best_score:.10f}, lr={lr:.4f}")

def _block_lp_refine(a0, time_limit, solver, block_size=16, verbose=False):
    """Optimize blocks of the sequence using small LPs."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = _norm_to(a0.copy(), norm)
    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    size = 2 * n - 1
    t0 = time.monotonic()
    
    # Precompute "other" contributions for each lag
    # For block [p, p+b), the contribution to lag k is:
    # sum_{i=0}^{b-1} a[p+i] * a[k-(p+i)]
    # The "other" part is conv(a,a)[k] - block_contribution[k]
    
    n_blocks = n // block_size
    if n_blocks == 0:
        return best_a, best_score
    
    rng = np.random.default_rng(123)
    pass_count = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if no_improve > 3:
            break
        pass_count += 1
        pass_improved = False
        
        # Shuffle block order
        block_order = rng.permutation(n_blocks)
        
        for bi in block_order:
            if time.monotonic() - t0 > time_limit:
                break
            
            p = bi * block_size
            b = min(block_size, n - p)
            if b < 2:
                continue
            
            # Current block values
            block = a[p:p+b].copy()
            
            # Compute current autocorrelation
            conv = _fast_autocorr(a)
            M_curr = conv.max()
            
            # For this block, compute the "other" contribution to each lag
            # other[k] = conv[k] - sum_{i=0}^{b-1} a[p+i]*a[k-(p+i)]
            # We'll build the LP: minimize t s.t. other[k] + block_contrib[k] <= t for all k
            # where block_contrib[k] = sum_i x_i * a[k-p-i] (for valid indices)
            
            # Build constraint matrix: for each lag k, the coefficient of x_i in block_contrib[k]
            # is a[k - p - i] (if 0 <= k-p-i < n)
            # This is a (size x b) matrix, but we only need the lags near the current max
            
            # For efficiency, only constrain the top lags
            top_lags = np.argsort(-conv)[:100]
            
            # Build T matrix: T[k_idx, i] = a[k - p - i] for valid (k, i)
            T = np.zeros((len(top_lags), b))
            other_vals = np.zeros(len(top_lags))
            for idx, k in enumerate(top_lags):
                for i in range(b):
                    j = k - p - i
                    if 0 <= j < n:
                        T[idx, i] = a[j]
                # other = conv[k] - block's current contribution
                contrib = 0.0
                for i in range(b):
                    j = k - p - i
                    if 0 <= j < n:
                        contrib += a[p + i] * a[j]
                other_vals[idx] = conv[k] - contrib
            
            # LP: minimize t
            # s.t. T @ x + other_vals <= t
            #      x >= 0
            #      sum(x) = sum(block)  (maintain normalization locally)
            
            x = cp.Variable(b, nonneg=True)
            t_var = cp.Variable()
            cons = [T @ x + other_vals <= t_var, cp.sum(x) == block.sum(), x >= 0]
            prob = cp.Problem(cp.Minimize(t_var), cons)
            try:
                prob.solve(solver=solver, verbose=False, max_iter=500)
            except Exception:
                continue
            
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                continue
            
            if x.value is None:
                continue
            
            x_new = np.maximum(np.asarray(x.value, dtype=np.float64), 0.0)
            if x_new.sum() < 1e-12:
                continue
            
            # Apply
            a_candidate = a.copy()
            a_candidate[p:p+b] = x_new
            a_candidate = _norm_to(a_candidate, norm)
            
            sc = _fast_autocorr(a_candidate).max()
            if sc < best_score - 1e-10:
                a = a_candidate
                best_a = a.copy()
                best_score = sc
                pass_improved = True
        
        if not pass_improved:
            no_improve += 1
        else:
            no_improve = 0
        
        if verbose and pass_count % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      BlockLP pass={pass_count} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score


def _spectral_shave(a0, time_limit, n_shaves=100, verbose=False):
    """Iteratively shave spectral peaks and project back to a>=0."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = _norm_to(a0.copy(), norm)
    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    
    fft_size = 1 << (n - 1).bit_length()
    
    for step in range(n_shaves):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = rfft(a, fft_size)
        mag = np.abs(A).copy()
        mag[0] = 0  # Don't touch DC
        
        # Find top bins to reduce
        n_top = max(10, n // 20)
        top_idx = np.argsort(mag)[::-1][:n_top]
        
        # Reduce them
        strength = 0.1 + 0.1 * step / n_shaves
        for idx in top_idx:
            if idx < len(A):
                A[idx] *= (1.0 - strength)
        
        a_new = irfft(A, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        a_new = _norm_to(a_new, norm)
        
        sc = _fast_autocorr(a_new).max()
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Try with less aggressive reduction
            A2 = rfft(a, fft_size)
            for idx in top_idx[:n_top//2]:
                if idx < len(A2):
                    A2[idx] *= (1.0 - strength * 0.5)
            a_new2 = irfft(A2, fft_size)[:n]
            a_new2 = np.maximum(a_new2, 0.0)
            a_new2 = _norm_to(a_new2, norm)
            sc2 = _fast_autocorr(a_new2).max()
            if sc2 < best_score - 1e-10:
                best_score = sc2
                best_a = a_new2.copy()
                a = a_new2
    
    if verbose:
        print(f"      Spectral: best={best_score:.10f}")
    
    return best_a, best_score

def _multi_gradient_descent(a0, time_limit, seed=42, verbose=False):
    """Slower but more careful gradient descent with line search."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = _norm_to(a0.copy(), norm)
    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    
    v = np.zeros(n)
    beta = 0.9
    base_lr = 0.01
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = _fast_autocorr(a)
        M = conv.max()
        
        # Top lags
        n_lags = min(50, n)
        active = np.argsort(-conv)[:n_lags]
        
        # Weight by how close to max
        weights = np.maximum(conv[active] - M * 0.5, 0.0)
        if weights.sum() < 1e-10:
            weights = np.ones(len(active))
        weights = weights / weights.sum()
        
        # Weighted gradient
        g = np.zeros(n)
        for w, k in zip(weights, active):
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            start = k - hi + 1
            end = k - lo + 1
            g[lo:hi] += w * 2.0 * a[start:end][::-1]
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        g = g / g_norm
        
        # Momentum
        v = beta * v + g
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-12:
            break
        v = v / v_norm
        
        # Backtracking line search
        lr = base_lr
        for _ in range(20):
            a_try = a - lr * v
            a_try = np.maximum(a_try, 0.0)
            a_try = _norm_to(a_try, norm)
            sc = _fast_autocorr(a_try).max()
            if sc < M - lr * 0.1 * g_norm:  # sufficient decrease
                a = a_try
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                break
            lr *= 0.5
        else:
            # No improvement possible with this direction
            beta = max(beta * 0.95, 0.5)
            base_lr = max(base_lr * 0.9, 1e-5)
            if base_lr < 1e-5:
                break
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MGDescent t={t} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score


def _dual_lower_bound(a, n_lags=200):
    """
    Compute a dual lower bound on the optimal max autocorrelation.
    The dual: find w_k >= 0, sum w_k = 1, maximizing min_i sum_k w_k * 2*a[k-i]
    (over valid i for each k).
    This gives a lower bound on the optimal value.
    We approximate by solving a zero-sum game (fictitious play).
    """
    n = len(a)
    size = 2 * n - 1
    
    # Fictitious play: 
    # Player 1 chooses lags (row strategy), Player 2 chooses indices (column strategy)
    # Payoff matrix: A[k, i] = 2*a[k-i] if valid, else 0
    # We want to find the value of this game (lower bound on optimal max_conv)
    
    # For efficiency, use only top lags
    conv = _fast_autocorr(a)
    M = conv.max()
    active_lags = np.where(conv >= M * 0.7)[0]
    if len(active_lags) > n_lags:
        active_lags = np.argsort(-conv)[:n_lags]
    
    K = len(active_lags)
    
    # Fictitious play
    p = np.full(K, 1.0 / K)  # Player 1 strategy (over lags)
    q = np.full(n, 1.0 / n)   # Player 2 strategy (over indices)
    
    for _ in range(200):
        # Player 1's best response to q: choose lag k maximizing sum_i q[i] * A[k,i]
        # = sum_i q[i] * 2*a[k-i] (for valid i)
        # This is 2 * conv_q(k) where conv_q = convolve(q, a) at lag k ... 
        # Actually it's the cross-correlation: sum_i q[i] * a[k-i]
        # = (q * a)[k] in convolution sense (with appropriate indices)
        
        # Compute: for each k in active_lags, payoff_k = sum_i q[i] * a[k-i] (valid)
        payoffs_1 = np.zeros(K)
        for idx, k in enumerate(active_lags):
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            start = k - hi + 1
            end = k - lo + 1
            payoffs_1[idx] = np.dot(q[lo:hi], a[start:end][::-1])
        
        # Best response: put all mass on best lag
        best_k = np.argmax(payoffs_1)
        p = 0.9 * p + 0.1 * np.zeros(K)
        p[best_k] += 0.1
        
        # Player 2's best response to p: choose i minimizing sum_k p[k] * A[k,i]
        # = sum_k p[k] * 2*a[k-i] (for valid k)
        # For each i, payoff_i = sum_k p[k] * a[k-i]
        # This is like: for i in 0..n-1, sum over k where k-i is valid
        
        payoffs_2 = np.zeros(n)
        for idx, k in enumerate(active_lags):
            if p[idx] < 1e-10:
                continue
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            start = k - hi + 1
            end = k - lo + 1
            payoffs_2[lo:hi] += p[idx] * a[start:end][::-1]
        
        best_i = np.argmin(payoffs_2)
        q = 0.9 * q + 0.1 * np.zeros(n)
        q[best_i] += 0.1
        
        p /= p.sum()
        q /= q.sum()
    
    # Lower bound: min_i sum_k p[k] * A[k,i] = 2 * min(payoffs_2)
    lb = 2.0 * np.min(payoffs_2)
    return lb, payoffs_2
