    ss = f_interp.sum()

def exact_subgradient(a, ac):

        if 0 <= j < n:
            if 2 * i == k_star:
                grad[i] = a[j]
            else:
                grad[i] = 2 * a[j]
    return grad

"""

1. Multi-length exploration with structured initializations (monotone, geometric, interpolated parent)
2. FISTA with exact/soft subgradients for the max-autocorrelation objective
3. Two-stage optimization: soft subgradient for global structure, hard subgradient for local refinement
4. Aggressive coordinate descent with multi-peak sensitivity
5. Adaptive time allocation across phases

import numpy as np
import time


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_hard_subgradient(a, ac, top_k=1):
    """Compute subgradient using only the top_k autocorrelation peaks."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals / top_vals.sum()

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        for i in range(n):
            j = k_int - i
            if 0 <= j < n:
                if 2 * i == k_int:
                    grad[i] += wi * a[j]
                else:
                    grad[i] += wi * 2.0 * a[j]
    return grad


def compute_soft_subgradient(a, ac, top_k=20, beta=8.0):
    """Compute soft subgradient with exponential weighting over top_k peaks."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        for i in range(n):
            j = k_int - i
            if 0 <= j < n:
                if 2 * i == k_int:
                    grad[i] += wi * a[j]
                else:
                    grad[i] += wi * 2.0 * a[j]
    return grad


def fista_optimize(a0, time_limit, subgrad_type='soft', top_k=1, beta=8.0,
                   step_size=None, verbose=False):
    """FISTA with configurable subgradient for max-autocorrelation minimization."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    x = a.copy()
    z = a.copy()
    t_mom = 1.0

    if step_size is None:
        step_size = 0.5

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(z)
        if subgrad_type == 'hard':
            grad = compute_hard_subgradient(z, ac, top_k=top_k)
        else:
            grad = compute_soft_subgradient(z, ac, top_k=top_k, beta=beta)

        x_new = z - step_size * grad
        x_new = np.maximum(x_new, 0.0)
        s = x_new.sum()
        if s > 1e-15:
            x_new = x_new / s
        else:
            x_new = z.copy()

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        mom_coeff = (t_mom - 1.0) / t_new
        z_new = x_new + mom_coeff * (x_new - x)
        z_new = np.maximum(z_new, 0.0)
        sz = z_new.sum()
        if sz > 1e-15:
            z_new = z_new / sz
        else:
            z_new = x_new.copy()

        x = x_new
        z = z_new
        t_mom = t_new

        m_x = fast_autocorr(x).max()
        if m_x < best_m:
            best_m = m_x
            best_a = x.copy()

        m_z = fast_autocorr(z).max()
        if m_z < best_m:
            best_m = m_z
            best_a = z.copy()

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      FISTA t={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def mirror_descent_optimize(a0, time_limit, subgrad_type='soft', top_k=1,
                            beta=8.0, c0=2.0, verbose=False):
    """Mirror descent with exponential family for simplex constraint."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    x = a.copy()
    x_prev = a.copy()
    gamma = c0

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        if subgrad_type == 'hard':
            g = compute_hard_subgradient(y, ac, top_k=top_k)
        else:
            g = compute_soft_subgradient(y, ac, top_k=top_k, beta=beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m:
            best_m = M_new
            best_a = x.copy()

        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      MD t={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m

            j_idx = k_int - idx

                    d2 = dst - k2i
                    add = 0.0

                    if 0 <= d2 < n and d2 != d1:

                rem_sens = 0.0

                    if 0 <= d1 < n:

def geometric_init(n, alpha=3.0):
    """Create a monotonically decreasing exponential initialization."""
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-alpha * x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def power_init(n, p=2.0):
    """Create a power-law decreasing initialization."""
    x = np.linspace(0.0, 1.0, n)
    a = (1.0 - x) ** p + 1e-6
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def sine_init(n):
    """Sine squared initialization."""
    x = np.linspace(0.0, 1.0, n)
    a = np.sin(np.pi * x) ** 2
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

        start = a_in if sc_in <= sc_geo else a_geo

        t2 = min(t_per_n, time_budget - 100 - (time.monotonic() - t_global))
        if t2 > 20:
            a_out, _ = fista_optimize(start, t2, subgrad_type='hard', top_k=1,
                                       step_size=0.3, verbose=False)
            try_update(a_out, f"fista_hard_n{n_opt}")

    # === Phase 3: Mirror descent with soft subgradient ===
    print(f"\n=== Phase 3: Mirror descent soft ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.25, 300.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent_optimize(a_in, t3, subgrad_type='soft',
                                            top_k=30, beta=6.0, c0=3.0, verbose=False)
        try_update(a_out, f"md_soft_n{n_best}")

        # Also try hard subgradient version
        remaining = time_budget - (time.monotonic() - t_global)
        t3b = min(remaining * 0.15, 150.0)
        if t3b > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = mirror_descent_optimize(a_in, t3b, subgrad_type='hard',
                                                top_k=1, c0=2.5, verbose=False)
            try_update(a_out, f"md_hard_n{len(best_seq)}")

    # === Phase 4: FISTA with soft subgradient (different params) ===
    print(f"\n=== Phase 4: FISTA soft subgradient ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = fista_optimize(a_in, t4, subgrad_type='soft', top_k=10,
                                   beta=6.0, step_size=0.4, verbose=False)
        try_update(a_out, f"fista_soft_n{n_best}")

    # === Phase 5: Coordinate descent polish ===
    print(f"\n=== Phase 5: Coordinate polish ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.20, 200.0)
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t5, verbose=True, rng_seed=42)
        try_update(a_out, f"coord1_n{n_best}")

    # === Phase 6: Try different initializations at best n ===
    print(f"\n=== Phase 6: Alternative inits ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 100:
        inits = []
        # Geometric with different alphas
        inits.append(geometric_init(n_best, alpha=1.5))
        inits.append(geometric_init(n_best, alpha=4.0))
        inits.append(power_init(n_best, p=3.0))
        inits.append(power_init(n_best, p=1.5))
        # Blend with current best
        blend1 = 0.7 * norm_sum1(best_seq) + 0.3 * geometric_init(n_best, 2.0)
        inits.append(norm_sum1(blend1))

        t_per = min(remaining * 0.15, 60.0)
        for idx, init in enumerate(inits[:4]):
            if time.monotonic() - t_global > time_budget - 50:
                break
            t6 = min(t_per, time_budget - 50 - (time.monotonic() - t_global))
            if t6 > 15:
                a_out, _ = fista_optimize(init, t6, subgrad_type='hard', top_k=1,
                                           step_size=0.3, verbose=False)
                try_update(a_out, f"alt{idx}_fista_n{len(a_out)}")

    # === Phase 7: Scale up if current n is small ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if n_best <= 2800 and remaining > 80:
        n_upgrade = min(n_best + 400, 3600)
        print(f"\n=== Phase 7: Upgrade to n={n_upgrade} ===")
        a_long = interpolate_sequence(best_seq, n_upgrade)
        t7 = min(remaining * 0.20, 150.0)
        if t7 > 20:
            a_out, _ = fista_optimize(a_long, t7, subgrad_type='hard', top_k=1,
                                       step_size=0.3, verbose=False)
            try_update(a_out, f"upgrade_n{n_upgrade}")

    # === Phase 8: Final hard FISTA polish ===
    print(f"\n=== Phase 8: Final FISTA hard polish ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t8 = min(remaining * 0.25, 200.0)
    if t8 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = fista_optimize(a_in, t8, subgrad_type='hard', top_k=2,
                                   step_size=0.15, verbose=False)
        try_update(a_out, f"fista_final_n{n_best}")

    # === Phase 9: Final coordinate polish ===
    print(f"\n=== Phase 9: Final coord polish ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t9 = min(remaining * 0.30, 150.0)
    if t9 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t9, verbose=True, rng_seed=777)
        try_update(a_out, f"coord_final_n{n_best}")

    # === Phase 10: Mirror descent with very hard subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        t10 = min(remaining * 0.40, 120.0)
        if t10 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = mirror_descent_optimize(a_in, t10, subgrad_type='hard',
                                                top_k=1, c0=1.5, verbose=False)
            try_update(a_out, f"md_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_sequence([float(x) for x in best_seq])
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        best_seq = np.ones(2400) / 2400

    return [float(x) for x in best_seq]
