Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

2. Nesterov-accelerated smooth gradient descent with FFT (log-sum-exp)

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Compute full autocorrelation (convolution of a with itself) using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n + 1) > (css - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad(a, w):
    """Compute gradient of T*log(sum exp(c/T)) where w = softmax(c/T)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    cross = irfft(W * np.conj(A), fft_size)[:n]
    return 2.0 * cross


def cross_corr_uv(u, v):
    """Compute cross-correlation (u * v)[i] = sum_k u[k]*v[k+i] using FFT."""
    n = len(u)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    U = rfft(u, fft_size)
    V = rfft(v, fft_size)
    return irfft(U * np.conj(V), fft_size)[:n]

        if remaining < 1.0:
            break

        # Allocate more steps to early (high T) phases
        frac_left = sum(1.0 / t for t in temps[ti:])
        frac_total = sum(1.0 / t for t in temps)
        budget = remaining * (1.0 / T) / frac_left * 0.7
        n_steps = max(20, min(300, int(budget)))

        y = a.copy()
        momentum_buf = np.zeros(n)

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            c = fast_autocorr(y)
            c_max = c.max()
            # Softmax weights with temperature
            exp_c = np.exp((c - c_max) / T)
            w = exp_c / exp_c.sum()

            grad = compute_smooth_grad(y, w)

            # Adaptive step size
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-12:
                break
            step = T / (grad_norm + 1e-10) * 0.5
            step = min(step, norm / n * 0.1)

            # Gradient step
            a_new = y - step * grad
            a_new = project_simplex(a_new, norm)

            # Nesterov momentum
            beta = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1) ** 2)) / 2.0
            beta = min(beta, 2.0)
            momentum_buf = (it / (it + 1.0)) * momentum_buf + (a_new - y) / (it + 1.0)
            y_new = a_new + momentum_buf
            y_new = project_simplex(y_new, norm)

            y = y_new
            a = a_new

            # Track best
            score = fast_autocorr(a).max()
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()

            score_y = fast_autocorr(y).max()
            if score_y < best_score - 1e-12:
                best_score = score_y
                best_a = y.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"      Smooth T={T:.3f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def active_constraint_step(a, n_active_max=30, verbose=False):
    """Take a gradient step to reduce the most active constraints."""
    n = len(a)
    S = a.sum()
    c = fast_autocorr(a)
    M = c.max()

    # Select active lags
    threshold = 0.92 * M
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.argmax(c)
        active = np.array([active])
    if len(active) > n_active_max:
        top_idx = np.argsort(c)[::-1][:n_active_max]
        active = top_idx

    # Build indicator of active lags
    lambda_vec = np.zeros(2 * n - 1)
    for j in active:
        lambda_vec[j] = 1.0

    # Gradient direction: reduce active constraints
    # d(c[j])/d(a[i]) = 2*a[j-i] for valid j-i
    # Total gradient for active: sum_j lambda_j * 2*a[j-i] = 2 * cross_corr(lambda, a)[i]
    grad = 2.0 * cross_corr_uv(lambda_vec, a)

    # Project to sum-zero (preserve sum constraint)
    grad_centered = grad - grad.mean()
    gnorm = np.linalg.norm(grad_centered)
    if gnorm < 1e-12:
        return a

    # Step size: try to reduce M by a fraction
    step = 0.1 * M / (gnorm + 1e-10)
    step = min(step, 0.05 * norm_val)

    a_new = a - step * grad_centered
    a_new = project_simplex(a_new, S)
    return a_new


def focused_pair_cd(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent focusing on peak-contributing positions."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        conv = fast_autocorr(a)
        M = conv.max()

        # Find peak lags
        peak_threshold = 0.97 * M
        peak_lags = np.where(conv >= peak_threshold)[0]
        if len(peak_lags) == 0:
            peak_lags = np.array([np.argmax(conv)])

        # For each peak lag, identify top contributing positions
        candidate_moves = set()
        for lag in peak_lags[:8]:
            # c[lag] = sum_i a[i]*a[lag-i]
            # Find pairs (i, lag-i) with large product
            max_i = min(n - 1, lag)
            min_i = max(0, lag - (n - 1))
            if max_i < min_i:
                continue
            for i in range(min_i, max_i + 1):
                j = lag - i
                prod = a[i] * a[j]
                if prod > 0.3 * M / max(1, (max_i - min_i + 1)):
                    candidate_moves.add(i)
                    candidate_moves.add(j)

        # Also add positions with large values
        top_vals = np.argsort(a)[::-1][:min(50, n // 10)]
        for i in top_vals:
            candidate_moves.add(i)

        candidates = list(candidate_moves)
        rng.shuffle(candidates)

        for i in candidates:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < norm * 1e-8:
                continue

            # Try moving mass from i to several target positions
            # Target: positions that contribute least to peak lags
            # Use positions far from peak-contributing positions
            n_targets = min(12, n // 10)
            # Score each position by how much it contributes to peaks
            target_scores = np.zeros(n)
            for lag in peak_lags[:5]:
                max_i2 = min(n - 1, lag)
                min_i2 = max(0, lag - (n - 1))
                for k in range(min_i2, max_i2 + 1):
                    j2 = lag - k
                    target_scores[k] += a[j2]
                    target_scores[j2] += a[k]

            # Prefer targets with low peak contribution
            # Exclude i itself
            target_scores[i] = np.inf
            targets = np.argsort(target_scores)[:n_targets]

            for j in targets:
                if j == i or a[i] < norm * 1e-8:
                    continue

                # Try several transfer fractions
                for frac in [0.1, 0.3, 0.5]:
                    d = a[i] * frac
                    if d < 1e-12:
                        break
                    a_test = a.copy()
                    a_test[i] -= d
                    a_test[j] += d
                    s_test = a_test.sum()
                    if s_test < 1e-10:
                        continue
                    a_test = a_test * norm / s_test
                    conv_test = fast_autocorr(a_test)
                    if conv_test.max() < best_score - 1e-12:
                        a = a_test
                        conv = conv_test
                        best_score = conv.max()
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0
                    elif conv_test.max() < conv.max() - 1e-12:
                        # Local improvement (not global best)
                        a = a_test
                        conv = conv_test
                        pass_improved = True
                        no_improve = 0

        if not pass_improved:
            no_improve += 1
            # Try random pairs as fallback
            for _ in range(min(50, n // 10)):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < norm * 1e-8:
                    continue
                d = a[i] * 0.2
                a_test = a.copy()
                a_test[i] -= d
                a_test[j] += d
                a_test = np.maximum(a_test, 0)
                s_test = a_test.sum()
                if s_test < 1e-10:
                    continue
                a_test = a_test * norm / s_test
                conv_test = fast_autocorr(a_test)
                if conv_test.max() < best_score - 1e-12:
                    a = a_test
                    conv = conv_test
                    best_score = conv.max()
                    best_a = a.copy()
                    pass_improved = True
                    no_improve = 0

            if no_improve >= 5:
                break

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"      FocusPCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def random_pair_cd(a0, time_limit, rng, verbose=True):
    """Broader random pair coordinate descent with line search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        # Try many random pairs
        n_tries = min(200, n // 5)
        for _ in range(n_tries):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j or a[i] < norm * 1e-8:
                continue

            # Try a few transfer amounts
            for frac in [0.15, 0.35, 0.6]:
                d = a[i] * frac
                if d < 1e-12:
                    break
                a_test = a.copy()
                a_test[i] -= d
                a_test[j] += d
                a_test = np.maximum(a_test, 0.0)
                s_test = a_test.sum()
                if s_test < 1e-10:
                    continue
                a_test = a_test * norm / s_test
                conv_test = fast_autocorr(a_test)
                if conv_test.max() < conv.max() - 1e-12:
                    a = a_test
                    conv = conv_test
                    improved = True
                    if conv.max() < best_score - 1e-12:
                        best_score = conv.max()
                        best_a = a.copy()
                    no_improve = 0

        if not improved:
            no_improve += 1
            if no_improve >= 8:
                break
            # Perturb to escape local minimum
            idx = rng.integers(n)
            if a[idx] > norm * 1e-6:
                d = a[idx] * 0.1
                j = rng.integers(n)
                a[idx] -= d
                a[j] += d
                a = np.maximum(a, 0.0)
                a = a * norm / a.sum()
                conv = fast_autocorr(a)

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving the normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    return f_interp * np.sqrt(2.0 * new_n) / ss


def generate_initializations(n, base_seq, rng, n_init=8):
    """Generate diverse initializations."""
    inits = []
    norm = np.sqrt(2.0 * n)

    # 1. Base sequence itself
    if base_seq is not None and len(base_seq) == n:
        b = np.maximum(np.asarray(base_seq, dtype=np.float64), 0.0)
        s = b.sum()
        if s > 1e-10:
            inits.append(b * norm / s)

    # 2. Cosine-modulated uniform: a[k] = 1 + alpha*cos(2*pi*f*k/n)
    for alpha, f in [(0.3, 1), (0.3, 2), (0.5, 3), (0.4, 5), (0.2, 7), (0.6, 1), (0.35, 4)]:
        k = np.arange(n)
        a = 1.0 + alpha * np.cos(2.0 * np.pi * f * k / n)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            inits.append(a * norm / s)
        if len(inits) >= n_init:
            break

    # 3. Noisy version of base
    if base_seq is not None and len(base_seq) == n:
        b = np.maximum(np.asarray(base_seq, dtype=np.float64), 0.0)
        s = b.sum()
        if s > 1e-10:
            b = b * norm / s
            noise_scale = 0.05 * b.std()
            for ns in [noise_scale, 0.15 * b.std()]:
                a_noisy = b + rng.normal(0, ns, n)
                a_noisy = np.maximum(a_noisy, 0.0)
                s2 = a_noisy.sum()
                if s2 > 1e-10:
                    inits.append(a_noisy * norm / s2)
                if len(inits) >= n_init + 4:
                    break

    # 4. Two-level sequence
    p = rng.uniform(0.3, 0.7)
    a_two = np.where(rng.random(n) < p, 1.0, 0.0)
    s = a_two.sum()
    if s > n * 0.1:
        inits.append(a_two * norm / s)

    return inits[:n_init + 4]


def active_set_refine(a0, time_limit, verbose=True):
    """Iteratively refine using active constraint gradients."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    n_iters = 0

    # Adaptive step size
    step_scale = 0.3

    while time.monotonic() - t0 < time_limit:
        n_iters += 1
        conv = fast_autocorr(a)
        M = conv.max()

        # Select active lags (top few)
        n_active = min(20, 2 * n - 1)
        top_idx = np.argsort(conv)[::-1][:n_active]
        # Only use those close to max
        active_idx = top_idx[conv[top_idx] >= 0.9 * M]
        if len(active_idx) == 0:
            active_idx = top_idx[:3]

        # Build lambda
        lambda_vec = np.zeros(2 * n - 1)
        for j in active_idx:
            lambda_vec[j] = 1.0

        # Gradient: sum over active of d(c[j])/d(a[i]) = sum active 2*a[j-i]
        # = 2 * cross_corr(lambda, a)
        grad = 2.0 * cross_corr_uv(lambda_vec, a)
        grad_centered = grad - grad.mean()
        gnorm = np.linalg.norm(grad_centered)

        if gnorm < 1e-12:
            break

        # Backtracking line search
        alpha = step_scale
        for _ in range(20):
            a_try = a - alpha * grad_centered / gnorm
            a_try = project_simplex(a_try, norm)
            conv_try = fast_autocorr(a_try)
            if conv_try.max() < M - 1e-12:
                a = a_try
                conv = conv_try
                step_scale = min(alpha * 1.5, 1.0)
                break
            alpha *= 0.5
        else:
            # No improvement found, try smaller
            alpha = step_scale * 0.1
            a_try = a - alpha * grad_centered / gnorm
            a_try = project_simplex(a_try, norm)
            conv_try = fast_autocorr(a_try)
            if conv_try.max() < M - 1e-15:
                a = a_try
                conv = conv_try
            step_scale = max(alpha * 0.5, 0.01)

        if conv.max() < best_score - 1e-12:
            best_score = conv.max()
            best_a = a.copy()

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      ActiveSet iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function. Returns best sequence found within time budget."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            best_seq = parent.copy()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
        parent = best_seq.copy()

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-10:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Strategy: Multi-start with diverse inits, then refine the best ===
    n_base = len(best_seq)

    # Generate initializations at base length
    inits = generate_initializations(n_base, best_seq, rng, n_init=6)
    print(f"Generated {len(inits)} initializations at n={n_base}")

    # Also try a few different lengths
    alt_lengths = [n_base - 100, n_base + 100, n_base + 200]
    for n_alt in alt_lengths:
        if n_alt < 200:
            continue
        alt_init = interpolate_sequence(best_seq, n_alt)
        inits.append(alt_init)

    # Phase 1: Quick smooth GD on all inits to find the best starting point
    remaining = time_budget - (time.monotonic() - t_global)
    t_quickscore = remaining * 0.05  # 5% time for quick evaluation
    print(f"\n=== Phase 1: Quick scoring ({t_quickscore:.0f}s) ===")

    scored_inits = []
    for ii, init in enumerate(inits):
        t_sub = t_quickscore / max(1, len(inits))
        if time.monotonic() - t_global > time_budget - t_sub:
            break
        a_norm = init * np.sqrt(2.0 * len(init)) / init.sum()
        score = fast_autocorr(a_norm).max()
        scored_inits.append((score, init, len(init)))
        print(f"    Init {ii}: n={len(init)}, score={score:.6f}")

    scored_inits.sort(key=lambda x: x[0])
    print(f"    Best init score: {scored_inits[0][0]:.6f}")

    # Phase 2: Full optimization on top inits (70% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    top_inits = scored_inits[:min(4, len(scored_inits))]
    t_per_init = remaining * 0.70 / len(top_inits)

    for ii, (init_score, init, n_init) in enumerate(top_inits):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        t_alloc = min(t_per_init, remaining * 0.5)
        print(f"\n=== Phase 2.{ii}: Optimize init {ii} (n={n_init}, score={init_score:.6f}, budget={t_alloc:.0f}s) ===")

        a_work = init * np.sqrt(2.0 * n_init) / init.sum()

        # Step A: Smooth gradient descent (50% of allocation)
        t_smooth = t_alloc * 0.5
        if t_smooth > 5:
            a_smooth, _ = smooth_gradient_descent(a_work, t_smooth, verbose=(ii < 2))
            do_try_update(a_smooth, f"smooth_{ii}_n{n_init}")

        # Step B: Focused pair CD (30% of allocation)
        remaining_sub = time_budget - (time.monotonic() - t_global)
        t_pcd = min(t_alloc * 0.3, remaining_sub * 0.4)
        if t_pcd > 5 and best_seq is not None and len(best_seq) == n_init:
            a_pcd_init = best_seq * np.sqrt(2.0 * n_init) / best_seq.sum()
        elif t_pcd > 5:
            a_pcd_init = a_smooth.copy()
        else:
            a_pcd_init = a_work.copy()
        if t_pcd > 5:
            a_pcd, _ = focused_pair_cd(a_pcd_init, t_pcd, rng, verbose=(ii < 2))
            do_try_update(a_pcd, f"focuspcd_{ii}_n{n_init}")

        # Step C: Active set refinement (15% of allocation)
        remaining_sub = time_budget - (time.monotonic() - t_global)
        t_active = min(t_alloc * 0.15, remaining_sub * 0.3)
        if t_active > 5:
            a_active = (best_seq if best_seq is not None and len(best_seq) == n_init else a_pcd) * np.sqrt(2.0 * n_init) / (best_seq if best_seq is not None and len(best_seq) == n_init else a_pcd).sum()
            a_act_out, _ = active_set_refine(a_active, t_active, verbose=(ii < 2))
            do_try_update(a_act_out, f"active_{ii}_n{n_init}")

        # Step D: More smooth GD with lower temperature (5%)
        remaining_sub = time_budget - (time.monotonic() - t_global)
        t_smooth2 = min(t_alloc * 0.05, remaining_sub * 0.2)
        if t_smooth2 > 5:
            a_refine = (best_seq if best_seq is not None and len(best_seq) == n_init else a_act_out)
            a_refine = a_refine * np.sqrt(2.0 * n_init) / a_refine.sum()
            a_out2, _ = smooth_gradient_descent(a_refine, t_smooth2, verbose=False)
            do_try_update(a_out2, f"smooth2_{ii}_n{n_init}")

    # Phase 3: Random pair CD on best (15% of remaining time)
    remaining = time_budget - (time.monotonic() - t_global)
    t_rand = remaining * 0.15
    print(f"\n=== Phase 3: Random PCD (budget={t_rand:.0f}s) ===")
    if t_rand > 10 and best_seq is not None:
        n_best = len(best_seq)
        a_rand = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_rand_out, _ = random_pair_cd(a_rand, t_rand, rng, verbose=True)
        do_try_update(a_rand_out, "rand_pcd")

    # Phase 4: Final polishing (15% of remaining time)
    remaining = time_budget - (time.monotonic() - t_global)
    t_final = remaining * 0.3
    print(f"\n=== Phase 4: Final polish (budget={t_final:.0f}s) ===")
    if t_final > 10 and best_seq is not None:
        n_best = len(best_seq)
        a_final = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        # Alternate smooth GD and focused PCD
        t_split = t_final / 2
        a_f1, _ = smooth_gradient_descent(a_final, t_split, verbose=False)
        do_try_update(a_f1, "final_smooth")
        remaining_sub = time_budget - (time.monotonic() - t_global)
        if remaining_sub > 5 and best_seq is not None:
            n_b2 = len(best_seq)
            a_f2 = best_seq * np.sqrt(2.0 * n_b2) / best_seq.sum()
            a_f2_out, _ = focused_pair_cd(a_f2, min(remaining_sub, t_split), rng, verbose=False)
            do_try_update(a_f2_out, "final_pcd")

    # Phase 5: Try a couple more lengths if time permits
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60 and best_seq is not None:
        n_best = len(best_seq)
        for n_try in [n_best + 50, n_best - 50, n_best + 300]:
            if n_try < 200 or time.monotonic() - t_global > time_budget - 30:
                continue
            t_sub = min(remaining * 0.2, 60.0)
            print(f"\n=== Alt length n={n_try} (budget={t_sub:.0f}s) ===")
            a_alt = interpolate_sequence(best_seq, n_try)
            a_alt_out, _ = smooth_gradient_descent(a_alt, t_sub * 0.6, verbose=False)
            do_try_update(a_alt_out, f"alt_n{n_try}")
            remaining_sub = time_budget - (time.monotonic() - t_global)
            if remaining_sub > 10:
                a_alt2 = (best_seq if best_seq is not None and len(best_seq) == n_try else a_alt_out)
                a_alt2 = a_alt2 * np.sqrt(2.0 * n_try) / a_alt2.sum()
                a_alt2_out, _ = focused_pair_cd(a_alt2, min(remaining_sub, t_sub * 0.4), rng, verbose=False)
                do_try_update(a_alt2_out, f"alt_pcd_n{n_try}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\n{'='*60}")
        print(f"FINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
        print(f"{'='*60}")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
