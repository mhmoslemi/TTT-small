import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp

def fast_convolution(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_lse_obj_and_grad(conv, a, temp):
    """Compute log-sum-exp objective and its gradient w.r.t. a."""
    n = len(a)
    size = len(conv)
    scaled = conv / temp
    m = scaled.max()
    exp_terms = np.exp(scaled - m)
    sum_exp = exp_terms.sum()
    obj = m * temp + temp * np.log(sum_exp)  # This is temp * log(sum(exp(conv/temp)))
    # Actually, the smooth max is: temp * log(sum(exp(conv/temp)))
    # And its gradient w.r.t. conv is: exp(conv/temp) / sum(exp(conv/temp)) = softmax(conv/temp)
    
    weights = exp_terms / sum_exp  # softmax(conv/temp)
    
    # Gradient w.r.t. a:
    # d/d a_i of [temp * log(sum_k exp(conv_k / temp))]
    # = sum_k weights_k * d(conv_k)/d(a_i)
    # = sum_k weights_k * 2 * a_{k-i} (if 0 <= k-i < n)
    # = 2 * (cross-correlation of weights and a) [i]
    
    N = 1 << (3 * n - 1).bit_length()
    W = rfft(weights, N)
    A = rfft(a, N)
    grad = 2.0 * irfft(W * np.conj(A), N)[:n]
    
    return obj, grad

def smooth_max_descent(a0, time_limit, verbose=True):
    """Optimize using smoothed max (log-sum-exp) with adaptive temperature."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = fast_convolution(a).max()
    
    # Adam parameters
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    
    # Temperature schedule
    temp = 1.0 / (n * best_score)  # Initial temperature
    temp_decay = 0.999
    
    t0 = time.monotonic()
    n_iters = 0
    patience = 0
    
    for it in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolution(a)
        obj, grad = compute_lse_obj_and_grad(conv, a, temp)
        
        # Adam update
        n_iters += 1
        m_hat = beta1 * m_hat + (1 - beta1) * grad
        v_hat = beta2 * v_hat + (1 - beta2) * grad**2
        m_hat_hat = m_hat / (1 - beta1**n_iters)
        v_hat_hat = v_hat / (1 - beta2**n_iters)
        
        a_new = a - lr * m_hat_hat / (np.sqrt(v_hat_hat) + eps)
        
        # Project onto simplex (sum = 1, a >= 0)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        a = a_new
        temp *= temp_decay
        
        # Check improvement
        conv_new = fast_convolution(a)
        score = conv_new.max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > 500:
            # Reset temperature and try a different learning rate
            temp = 1.0 / (n * best_score)
            lr *= 0.5
            patience = 0
        
        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SMD iter={it} ({elapsed:.0f}s): best={best_score:.8f}, temp={temp:.6f}, lr={lr:.6f}")
    
    return best_a, best_score

        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, restarts={restart_count}")

    return best_a, best_score


def compute_lse_gradient(conv, a, temp):
    """Compute gradient of temp*log(sum(exp(conv/temp))) w.r.t. a using FFT."""
    n = len(a)
    scaled = conv / temp
    m = scaled.max()
    exp_terms = np.exp(scaled - m)
    weights = exp_terms / exp_terms.sum()

    # grad_i = 2 * sum_{j=0}^{n-1} weights[i+j] * a[j]
    N = 1 << (3 * n).bit_length()
    W = rfft(weights, N)
    A = rfft(a, N)
    grad = 2.0 * irfft(W * np.conj(A), N)[:n]
    return grad


def smooth_gradient_optimize(a0, time_limit, verbose=True):
    """Optimize using log-sum-exp smoothed objective with Adam."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-10)
    a = a / a.sum()

    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()

    # Adam
    lr = 0.005
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)

    # Temperature: start reasonable, decay slowly
    conv_init = fast_conv(a)
    temp = conv_init.max() / 5.0
    temp_min = conv_init.max() / 100.0

    patience = 0
    rng = np.random.default_rng(777)

    for it in range(50000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv(a)
        grad = compute_lse_gradient(conv, a, temp)

        # Normalize gradient by the scale of the problem
        grad_scale = np.abs(grad).mean()
        if grad_scale > 1e-15:
            grad = grad / grad_scale

        # Adam update
        m_hat = beta1 * m_hat + (1 - beta1) * grad
        v_hat = beta2 * v_hat + (1 - beta2) * grad ** 2
        m_hat_hat = m_hat / (1 - beta1 ** (it + 1))
        v_hat_hat = v_hat / (1 - beta2 ** (it + 1))

        a_new = a - lr * m_hat_hat / (np.sqrt(v_hat_hat) + eps)

        # Project onto simplex
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        a = a_new
        temp = max(temp * 0.9995, temp_min)

        # Periodic evaluation
        if (it + 1) % 50 == 0:
            score = fast_conv(a).max()
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                patience = 0
            else:
                patience += 1

            if patience > 100:
                # Perturb
                if patience % 200 == 0:
                    noise = rng.laplace(0, 0.01, n)
                    a_pert = a + noise
                    a_pert = np.maximum(a_pert, 0)
                    s_pert = a_pert.sum()
                    if s_pert > 1e-15:
                        a = a_pert / s_pert
                    patience = 0
                    lr *= 0.8

        if verbose and (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            cur = fast_conv(a).max()
            print(f"    SGM iter={it} ({elapsed:.0f}s): cur={cur:.8f}, best={best_score:.8f}, temp={temp:.6f}")

    return best_a, best_score


def active_set_cd(a0, time_limit, verbose=True):
    """Coordinate descent targeting active lags with smart mass transfer."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv(a)
        size = len(conv)
        max_val = conv.max()
        # Identify top lags
        top_k = min(15, size)
        active_idx = np.argsort(conv)[-top_k:]

        improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-10:
                continue

            # For each active lag, find partners to shift mass to
            for lag in active_idx:
                j = int(lag) - i
                if j == i or j < 0 or j >= n:
                    j2 = int(lag) - (n - 1 - i)
                    if 0 <= j2 < n and j2 != i:
                        j = j2
                    else:
                        continue

                # Try moving a small amount from i to j
                for frac in [0.1, 0.2, 0.3, 0.5]:
                    d = a[i] * frac
                    if d < norm * 1e-10:
                        break
                    a[i] -= d
                    a[j] += d
                    new_score = fast_conv(a).max()
                    if new_score < best_score - 1e-14:
                        best_score = new_score
                        best_a = a.copy()
                        improved = True
                        no_improve = 0
                        break
                    else:
                        a[i] += d
                        a[j] -= d

        if not improved:
            no_improve += 1
            if no_improve >= 10:
                # Perturb
                noise = rng.normal(0, norm * 0.003, n)
                a_pert = a + noise
                a_pert = np.maximum(a_pert, 0)
                s_pert = a_pert.sum()
                if s_pert > 1e-15:
                    a = a_pert * norm / s_pert
                no_improve = 0
                if no_improve >= 50:
                    break

        if verbose and (idx == n - 1) and (n_pass := (it // n)) % 20 == 0:
            pass  # simplified

    return best_a, best_score


def pair_cd_polish(a0, time_limit, verbose=True):
    """Pair coordinate descent with delta computation via FFT."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    conv = fast_conv(a)
    best_score = conv.max()
    size = 2 * n - 1
    t0 = time.monotonic()
    rng = np.random.default_rng(31415)
    no_improve = 0

    def compute_delta(i, j):
        """Compute delta vector when moving epsilon from i to j."""
        d = np.zeros(size)
        # +epsilon at j: conv contribution is +epsilon * a shifted to position j
        # -epsilon at i: conv contribution is -epsilon * a shifted to position i
        # conv(epsilon_ej, a)[k] = epsilon * a[k-j] (if valid)
        # conv(epsilon_ei, a)[k] = epsilon * a[k-i] (if valid)
        lo = j
        hi = min(size, j + n)
        d[lo:hi] += a[0:hi - lo]
        lo = i
        hi = min(size, i + n)
        d[lo:hi] -= a[0:hi - lo]
        # Also the symmetric part: conv(a, epsilon_ej)[k] = epsilon * a[j] * delta_{k,j}... 
        # Actually conv(a, a)[k] = sum_i a[i]*a[k-i]
        # d/d(eps) of conv(a+eps*(e_j-e_i), a+eps*(e_j-e_i)) at eps=0:
        # = conv(e_j - e_i, a) + conv(a, e_j - e_i)
        # = conv(e_j, a) - conv(e_i, a) + conv(a, e_j) - conv(a, e_i)
        # conv(e_j, a)[k] = a[k-j] (if 0<=k-j<n)
        # conv(a, e_j)[k] = a[j] * delta... no, conv(a, e_j)[k] = sum_i a[i] * e_j[k-i] = a[k-j] (if valid)
        # So conv(e_j, a)[k] = conv(a, e_j)[k] = a[k-j] (if 0<=k-j<n)
        # Thus total delta = 2*(conv(e_j, a) - conv(e_i, a))
        # Plus second order: +eps^2*(conv(e_j-e_i, e_j-e_i)) which we ignore for small eps
        return 2.0 * d

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv(a)
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-10:
                continue

            for offset in range(1, min(25, n)):
                for j in [(i + offset) % n, (i - offset) % n]:
                    if j == i or j < 0 or j >= n:
                        continue
                    if a[i] < norm * 1e-10:
                        break

                    delta = compute_delta(i, j)
                    best_d = 0.0
                    best_val = conv.max()
                    n_samples = 8
                    for d_val in np.linspace(a[i] * 0.1, a[i] * 0.8, n_samples):
                        new_conv = conv + d_val * delta
                        # Correct for the quadratic terms
                        new_conv[2 * i] = conv[2 * i] - 2 * a[i] * d_val + d_val * d_val
                        if 2 * j < size:
                            new_conv[2 * j] = conv[2 * j] + 2 * a[j] * d_val + d_val * d_val
                        if i == j:
                            pass  # shouldn't happen
                        nm = new_conv.max()
                        if nm < best_val - 1e-14:
                            best_val = nm
                            best_d = d_val

                    if best_d > 0:
                        a[i] -= best_d
                        a[j] += best_d
                        s = a.sum()
                        if s < 1e-15:
                            a[i] += best_d
                            a[j] -= best_d
                            continue
                        a = a * norm / s
                        conv = fast_conv(a)
                        cur = conv.max()
                        if cur < best_score - 1e-14:
                            best_score = cur
                            best_a = a.copy()
                            pass_improved = True
                            no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
            if no_improve >= 3:
                noise = rng.normal(0, norm * 0.002, n)
                a_pert = a + noise
                a_pert = np.maximum(a_pert, 0)
                s_pert = a_pert.sum()
                if s_pert > 1e-15:
                    a = a_pert * norm / s_pert
                no_improve = 0

        if verbose and idx == n - 1:
            if int((time.monotonic() - t0)) % 10 == 0:
                pass

    return best_a, best_score


def random_restart_lp(a0, time_limit, solver, n_restarts=3, verbose=True):
    """Random perturbations followed by quick LP optimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    best_a = np.asarray(a0, dtype=np.float64).copy()
    best_score = fast_conv(best_a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(9999)

    for r in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        scale = norm * 0.03 * (r + 1)
        noise = rng.laplace(0, scale, n)
        a_pert = best_a + noise
        a_pert = np.maximum(a_pert, 0)
        s = a_pert.sum()
        if s < 1e-15:
            continue
        a_pert = a_pert * norm / s

        t_remaining = time_limit - (time.monotonic() - t0)
        t_sub = min(time_limit / max(1, n_restarts - r) * 0.8, t_remaining * 0.85)
        if t_sub < 5:
            break
        a_out, sc = nesterov_lp_optimize(a_pert, t_sub, solver, verbose=False)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_out.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    RR restart {r}: improved to {best_score:.8f} ({elapsed:.0f}s)")

    return best_a, best_score


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def final_polish(a0, time_limit, solver, verbose=True):
    """Final polishing: alternating LP steps and pair CD."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv(a).max()
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()

    # Do a few LP iterations
    while time.monotonic() - t0 < time_limit * 0.6:
        M = fast_conv(a).max()
        b, _ = lp_best_response(a, M, solver)
        if b is None:
            break
        b = b * norm / b.sum()
        a = 0.5 * a + 0.5 * b
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            break
        a = a * norm / s
        sc = fast_conv(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # Pair CD on the result
    t_remaining = time_limit - (time.monotonic() - t0)
    if t_remaining > 5:
        a_out, sc = pair_cd_polish(best_a, t_remaining, verbose=False)
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1600
        a_init = np.ones(n_base) / n_base
        best_seq = a_init.copy()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Primary LP Nesterov (40% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = n_base
    t1 = remaining * 0.40
    print(f"\n=== Phase 1: LP Nesterov n={n1} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n1) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp1_n{n1}")

    # Phase 2: Smooth gradient optimization (20% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = remaining * 0.20
    print(f"\n=== Phase 2: Smooth Gradient n={n_best} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = smooth_gradient_optimize(best_seq, t2, verbose=True)
        do_try_update(a_out, f"sgm_n{n_best}")

    # Phase 3: LP round 2 (15% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = remaining * 0.15
    print(f"\n=== Phase 3: LP Nesterov R2 n={n_best} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True)
        do_try_update(a_out, f"lp2_n{n_best}")

    # Phase 4: Pair CD polish (10% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = remaining * 0.10
    print(f"\n=== Phase 4: Pair CD n={n_best} (budget={t4:.0f}s) ===")
    if t4 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_cd_polish(a_norm, t4, verbose=True)
        do_try_update(a_out, f"pcd_n{n_best}")

    # Phase 5: Random restarts (10% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        t5 = remaining * 0.50
        print(f"\n=== Phase 5: Random Restarts n={n_best} (budget={t5:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = random_restart_lp(a_norm, t5, solver, n_restarts=2, verbose=True)
        do_try_update(a_out, f"rr_n{n_best}")

    # Phase 6: Final polish with LP + CD (remaining time)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_best = len(best_seq)
        print(f"\n=== Phase 6: Final Polish n={n_best} (budget={remaining:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = final_polish(a_norm, remaining * 0.9, solver, verbose=True)
        do_try_update(a_out, f"polish_n{n_best}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
