```python
    n = len(a)
    size = 2 * n - 1

    # This is the cross-correlation of a and y at offset i
    # Using FFT: correlate(a, y) gives sum_m a[m+i] * y[m] ... hmm

    # So T' @ y is the "adjoint" operation
    # (T'b)[k] = conv(a, b)[k] = sum_i a[i] b[k-i]
    # (T'y)[i] = sum_k T[k,i] y[k] = sum_k a[k-i] y[k]
    # Let j = k-i: = sum_j a[j] y[j+i]
    # This is the cross-correlation of a and y with y shifted by i
    
    # Using FFT:
    # Rev_a = a[::-1]
    # conv(Rev_a, y) gives sum_j Rev_a[j] y[k-j] = sum_j a[n-1-j] y[k-j]
    # Let m = n-1-j: = sum_m a[m] y[k-n+1+m]
    # So conv(Rev_a, y)[k] = sum_m a[m] y[k-n+1+m]
    # We want (T'y)[i] = sum_m a[m] y[m+i]
    # So (T'y)[i] = conv(Rev_a, y)[i + n - 1]
```

```python
def fast_best_response(a, M, n_iter=50):
    """
    Approximate: b* ≈ argmax sum(b) s.t. conv(a, b) <= M, b >= 0
    Using multiplicative weights / mirror descent.
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    
    # Start with uniform b
    b = np.ones(n)
    
    for it in range(n_iter):
        B = rfft(b, fft_size)
        c = irfft(A * B, fft_size)[:size]
        
        # Multiplicative weight update
        # b_i *= exp(-eta * sum_k T[k,i] * max(0, c[k] - M))
        # T[k,i] = a[k-i] for valid k-i
        # sum_k T[k,i] * v[k] = (T'v)[i] where v = max(0, c-M)
        v = np.maximum(0, c - M)
        if v.max() < 1e-10:
            break
        
        # Compute T'v using FFT
        V = rfft(v, fft_size)
        corr = irfft(rfft(a[::-1], fft_size) * V, fft_size)
        Tv = corr[n-1:2*n-1]
        
        # Update: b *= exp(-eta * Tv)
        eta = 1.0 / (Tv.max() + 1e-10)
        b = b * np.exp(-eta * Tv)
        
        # Renormalize to maintain sum
        b = b / b.sum() * n  # keep sum around n
    
    # Final feasibility check
    B = rfft(b, fft_size)
    c = irfft(A * B, fft_size)[:size]
    max_c = c.max()
    if max_c > M:
        b = b * M / max_c
    
    return b
```

```python
def nesterov_lp(a0, time_limit, solver, use_custom_lp=False, verbose=False):

    t0 = time.monotonic()
    y = a.copy()

    # Track LP solve time to adaptively switch strategies
    lp_time_total = 0.0
    lp_count = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Solve LP (use custom if avg LP time is too high)
        lp_start = time.monotonic()
        if use_custom_lp or (lp_count > 10 and lp_time_total / lp_count > 0.5):
            b = custom_best_response(y, M, n_iter=40)
        else:
            b = cvxpy_best_response(y, M, solver)
        lp_time = time.monotonic() - lp_start
        lp_time_total += lp_time
        lp_count += 1

        if b is None or b.sum() < 1e-15:
            b = custom_best_response(y, M, n_iter=30)

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}, lp_avg={lp_time_total/lp_count:.3f}s")

    return best_a, best_score


def pair_cd_fast(a0, time_limit, rng, verbose=False):
    """
    Improved pair coordinate descent with ternary search for optimal transfer.
    Key improvements: ternary search instead of linear scan, more j candidates,
    and vectorized gain computation.
    """
    n = len(a0)
    norm = math.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv_self(a).max()
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    def compute_gain(i, j):
        """Compute gain vector for transferring mass from i to j."""
        g = np.zeros(size)
        # Adding mass to j: conv increases by a shifted by j
        lo = j
        hi = min(size, j + n)
        g[lo:hi] += a[0:hi - lo]
        # Subtracting mass from i: conv decreases by a shifted by i
        lo_i = i
        hi_i = min(size, i + n)
        g[lo_i:hi_i] -= a[0:hi_i - lo_i]
        # Cross terms
        # The diagonal term at position 2i: -2*a[i]*d + d^2
        # The diagonal term at position 2j: +2*a[j]*d + d^2
        # The cross term at positions i+j and |i-j|: these are handled in the gain
        return g

    def ternary_search_delta(i, j, conv_base, g, ai, aj):
        """Find optimal delta in [0, ai*0.95] that minimizes max(conv + delta*g)."""
        lo_d = 0.0
        hi_d = ai * 0.95
        if hi_d < 1e-12:
            return 0.0, conv_base.max()

        # Also check if moving mass away from i and to j creates issues
        # The modified conv at position 2i: conv[2i] - 2*ai*d + d^2
        # The modified conv at position 2j: conv[2j] + 2*aj*d + d^2

        for _ in range(25):  # 25 iterations of ternary search
            m1 = lo_d + (hi_d - lo_d) / 3.0
            m2 = hi_d - (hi_d - lo_d) / 3.0

            def eval_at(d):
                if d < 1e-14:
                    return conv_base.max()
                # Approximate: conv + d*g, plus diagonal corrections
                new_max = conv_base.max()
                # Check the diagonal positions
                if 2 * i < size:
                    val_2i = conv_base[2 * i] - 2 * ai * d + d * d
                    new_max = max(new_max, val_2i)
                if 2 * j < size:
                    val_2j = conv_base[2 * j] + 2 * aj * d + d * d
                    new_max = max(new_max, val_2j)
                # Check gain-peaked positions
                top_idx = np.argsort(g)[-10:]
                for idx in top_idx:
                    val = conv_base[idx] + d * g[idx]
                    if val > new_max:
                        new_max = val
                return new_max

            v1 = eval_at(m1)
            v2 = eval_at(m2)
            if v1 < v2:
                hi_d = m2
            else:
                lo_d = m1

        best_d = (lo_d + hi_d) / 2.0
        # Evaluate at best_d and 0
        v0 = conv_base.max()
        v_best = eval_at(best_d)
        if v_best < v0:
            return best_d, v_best
        return 0.0, v0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        # Find peak lags for structured candidate selection
        peak_threshold = cur_max * 0.93
        peak_lags = np.where(conv >= peak_threshold)[0]

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            # Generate j candidates
            j_candidates = set()
            # Random candidates
            for _ in range(8):
                j_candidates.add(int(rng.integers(n)))
            # Structured candidates from peak lags
            for kl in peak_lags[:5]:
                j_s1 = (kl - i) % n
                j_s2 = (kl + i) % n
                j_s3 = int(kl) % n
                j_candidates.add(j_s1)
                j_candidates.add(j_s2)
                j_candidates.add(j_s3)
            # Nearby candidates
            for offset in [-3, -2, -1, 1, 2, 3]:
                j_neigh = (i + offset) % n
                j_candidates.add(j_neigh)
            j_candidates.discard(i)
            j_list = list(j_candidates)[:15]

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in j_list:
                if j == i:
                    continue
                aj = a[j]
                if aj < 1e-15 and ai < 1e-15:
                    continue

                g = compute_gain(i, j)
                delta, val = ternary_search_delta(i, j, conv, g, ai, aj)

                if val < best_val - 1e-12:
                    best_val = val
                    best_delta = delta
                    best_j = j

            if best_j >= 0 and best_delta > 1e-12:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new

                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
                    total_improvements += 1

                if s_new > 1e-15:

            elapsed = time.monotonic() - t0
            print(f"    PairCD improvements={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

        sorted_peaks = np.argsort(conv)[::-1]
        top_k = min(10, size)
        top_peak_lags = sorted_peaks[:top_k]
        peak_threshold = conv[top_peak_lags[-1]]

        # Compute pressure: how much each coordinate contributes to top peaks
        pressure = np.zeros(n)
        for kl in top_peak_lags:
            for i in range(n):
                # a[i] contributes to conv[kl] via a[i]*a[kl-i] and a[kl]*a[i-kl]
                idx1 = kl - i
                if 0 <= idx1 < n:
                    pressure[i] += a[idx1]
                idx2 = i - kl
                if 0 <= idx2 < n:
                    pressure[i] += a[idx2]

        # Normalize pressure
        p_max = pressure.max()
        if p_max > 0:
            pressure = pressure / p_max

        # Try to reduce pressure on high-pressure coordinates
        # by transferring to low-pressure coordinates that don't contribute to peaks
        ordered = np.argsort(pressure)  # lowest pressure first (best targets)
        pass_improved = False

        for i in range(n - 1, -1, -1):  # highest pressure first (best sources)
            if time.monotonic() - t0 > time_limit:
                break
            src = perm_i = i  # source is the i-th highest pressure
            src = (n - 1 - i) if i < n else i
            # Actually let's just iterate in order of decreasing pressure
            src = np.argsort(pressure)[::-1][i] if i < n else 0
            src = int(np.argsort(pressure)[::-1][i])
            ai = a[src]
            if ai < norm * 1e-8:
                continue

            # Try transferring to a few low-pressure targets
            for target_rank in range(min(5, n)):
                tgt = int(ordered[target_rank])
                if tgt == src:
                    continue
                aj = a[tgt]

                # Small transfer
                delta = ai * 0.1
                a_new = a.copy()
                a_new[src] -= delta
                a_new[tgt] += delta
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    continue
                a_new = a_new * norm / s_new

                new_max = fast_conv_self(a_new).max()
                if new_max < best_score - 1e-12:
                    a = a_new
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
                    break

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
            # Perturb
            n_pert = max(5, n // 50)
            idx_p = rng.choice(n, size=n_pert, replace=False)
            a[idx_p] *= (1.0 + 0.08 * rng.standard_normal(n_pert))
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a * norm / s
        else:
            no_improve = 0

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    PeakFlood ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def subgradient_optimize(a0, time_limit, verbose=False):
    """Momentum subgradient with adaptive temperature."""
    n = len(a0)
    norm = math.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    momentum = np.zeros(n)
    beta = 0.85
    lr_scale = 1.0

    for it in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        max_c = conv.max()
        if < 1e-15:
            break

        # Adaptive temperature: decrease over time
        temperature = max(0.05, 50.0 / (1.0 + 0.005 * it))
        shifted = (conv - max_c) / temperature
        shifted = np.maximum(shifted, -60.0)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-15:
            break
        w = w / w_sum

        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        g1 = irfft(np.conj(A) * W, fft_size)[:n]
        g2 = irfft(A * W, fft_size)[:n]
        grad = 2.0 * (g1 + g2)

        # Polyak step
        lr = lr_scale / (1.0 + 0.002 * it)
        momentum = beta * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        a = a_new

        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.1 * np.cos(2 * np.pi * 7 * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-8.0 * (x - 0.75) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2.0
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.8
    elif choice == 6:
        a = np.exp(-3.0 * x) + 0.3 + 0.2 * np.sin(4 * np.pi * x)
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(6 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        a = np.sinc(2.5 * (x - 0.5)) ** 2 + 0.05
    elif choice == 9:
        a = np.exp(-3.0 * (x - 0.5) ** 2 / 0.1)
    elif choice == 10:
        a = 0.4 + 0.4 * np.cos(2 * np.pi * x * rng.integers(2, 6)) + 0.1 * rng.random(n)
    else:
        a = 1.0 + 0.3 * rng.standard_normal(n)
    a = np.maximum(a, 0.0)
    a = a + 0.005 * rng.random(n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def eval_fn(a):
    """Evaluate using the provided evaluate_sequence function."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        if len(a_list) == 0:
            return float('inf')
        for val in a_list:
            if val < 0 or not np.isfinite(val):
                return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Tries multiple strategies and lengths,
    returning the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}, cvxpy={HAS_CVXPY}")

    best_seq = None
    best_score = float('inf')

    # Load parent if available
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    base_n = len(best_seq)

    # === Phase 1: LSE GD warmup ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 1: LSE GD n={base_n} (budget={t1:.0f}s) ===")
    if t1 > 15:
        a_out, _ = lse_gradient_descent(best_seq, t1, verbose=True)
        try_update(a_out, "lse1")

    # === Phase 2: Nesterov LP (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 2: Nesterov LP n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = nesterov_lp(best_seq, t2, solver, use_custom_lp=False, verbose=True)
        try_update(a_out, "lp1")

    # === Phase 3: Pair CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.18, 280.0)
    print(f"\n=== Phase 3: Pair CD n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_out, _ = pair_cd_fast(best_seq, t3, rng, verbose=True)
        try_update(a_out, "pcd1")

    # === Phase 4: Nesterov LP round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 4: Nesterov LP round 2 (budget={t4:.0f}s) ===")
    if t4 > 15:
        a_out, _ = nesterov_lp(best_seq, t4, solver, use_custom_lp=True, verbose=True)
        try_update(a_out, "lp2_custom")

    # === Phase 5: Peak flooding ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 5: Peak Flood n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_out, _ = peak_flood(best_seq, t5, rng, verbose=True)
        try_update(a_out, "peakflood")

    # === Phase 6: Pair CD round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 6: Pair CD round 2 (budget={t6:.0f}s) ===")
    if t6 > 10:
        a_out, _ = pair_cd_fast(best_seq, t6, rng, verbose=True)
        try_update(a_out, "pcd2")

    # === Phase 7: Try longer length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = base_n + 400
    t7 = min(remaining * 0.15, 180.0)
    print(f"\n=== Phase 7: Longer n={n_long} (budget={t7:.0f}s) ===")
    if t7 > 20 and remaining > 60:
        init_long = interpolate_seq(best_seq, n_long)
        t7a = min(t7 * 0.5, 90.0)
        a_out, _ = nesterov_lp(init_long, t7a, solver, use_custom_lp=True, verbose=False)
        try_update(a_out, f"lp_long{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t7b = min(remaining * 0.5, 60.0)
            a_out2, _ = pair_cd_fast(best_seq, t7b, rng, verbose=False)
            try_update(a_out2, "pcd_long")

    # === Phase 8: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final polish (budget={remaining:.0f}s) ===")
    if remaining > 20:
        a_out, _ = lse_gradient_descent(best_seq, min(remaining * 0.3, 40.0), verbose=False)
        try_update(a_out, "lse_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_out, _ = pair_cd_fast(best_seq, min(remaining * 0.7, 50.0), rng, verbose=False)
            try_update(a_out, "pcd_final")

    # === Phase 9: Try alternate length (shorter) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_short = max(1200, base_n - 300)
    if remaining > 30 and n_short != base_n:
        print(f"\n=== Phase 9: Shorter n={n_short} (budget={remaining:.0f}s) ===")
        init_short = interpolate_seq(best_seq, n_short)
        a_out, _ = nesterov_lp(init_short, min(remaining * 0.6, 80.0), solver,
                               use_custom_lp=True, verbose=False)
        try_update(a_out, f"lp_short{n_short}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            a_out2, _ = pair_cd_fast(best_seq, min(remaining, 20.0), rng, verbose=False)
            try_update(a_out2, "pcd_short")

    # Final
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
```
