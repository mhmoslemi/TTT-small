def fast_autocorr(a):
    """Compute full (linear) autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    return float(evaluate_sequence(a_list))

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n + 1) > (css - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

Let me also think about the **FTRL (Follow The Regularized Leader)** approach for the game iteration:

FTRL update:
`a_t = argmin_{a in C} <g_1 + ... + g_t, a> + R(a_t, g_1, ..., g_t)`

where C is the simplex and R is a regularizer.

For the "online learning" version of the game:
- Player 1 (us) chooses `a_t`
- Player 2 (adversary) chooses a lag `k_t`
- We observe the loss `L(a_t, k_t) = (a_t * a_t)[k_t]`
- We update using FTRL

The FTRL with log-régularization (which gives multiplicative weights) has the update:
`a_{t+1}[i] = argmin_{a in simplex} <sum_{s=1}^t g_s, a> + sum_i a[i] * (log a[i] - log a_{init}[i])`

This gives: `a_{t+1}[i] ∝ a_init[i] * exp(-sum_{s=1}^t g_s[i])`

This is exactly multiplicative weights! And it's equivalent to mirror descent.

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v.copy()
    if S <= 0:
        return np.zeros(n)
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n + 1) > (css - S))[0]
    if len(rho) == 0:
        return np.full(n, S / n)
    rho = rho[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def normalize_to(a, target_sum):
    """Normalize a to have sum = target_sum, preserving non-negativity."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), target_sum / len(a))
    return a * target_sum / s

def build_conv_rows(a, row_indices):
    """Build rows of the convolution matrix.
    Returns T such that (T @ b)[idx] = (a * b)[row_indices[idx]]
    """
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((len(row_indices), n))
    for ii, k in enumerate(row_indices):
        # (a*b)[k] = sum_i a[i] * b[k-i] where 0 <= k-i < n
        # i ranges from max(0, k-n+1) to min(k, n-1)
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        if i_lo <= i_hi:
            T[ii, i_lo:i_hi+1] = a[i_lo:i_hi+1]
    return T

def nesterov_game_iter(a0, time_limit, solver, verbose=True, n_top=300):
    """LP best-response game iteration with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_to(np.asarray(a0, dtype=np.float64), norm)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    stuck = 0
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-14:
            break
        
        b = lp_best_response(y, M_curr, solver, n_top=n_top)
        if b is None:
            stuck += 1
            if stuck > 50:
                break
            continue
        
        stuck = 0
        g = normalize_to(b, norm)
        n_iters += 1
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = normalize_to(y_new, norm)
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = normalize_to(a_new, norm)
        
        y = y_new
        a = a_new
        
        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        # Running average
        avg_a += a
        n_avg += 1
        if n_avg % 10 == 0:
            avg_check = normalize_to(avg_a, norm)
            score_avg = fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
        
        if verbose and n_iters % 30 == 0:
            elapsed = time.monotonic() - t0
            avg_score = fast_autocorr(normalize_to(avg_a, norm)).max()
            print(f"    Game iter={n_iters} ({elapsed:.0f}s): "
                  f"y={score_y:.7f}, a={score_a:.7f}, avg={avg_score:.7f}, best={best_score:.7f}")
    
    return best_a, best_score

def mirror_descent(a0, time_limit, verbose=True):
    """Mirror descent (multiplicative weights) with adaptive temperature."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_to(np.asarray(a0, dtype=np.float64), norm)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    t0 = time.monotonic()
    
    # Temperature schedule
    temps = [50.0, 25.0, 15.0, 10.0, 7.0, 5.0, 3.5, 2.5, 1.8, 1.2, 0.8, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02]
    
    log_a = np.log(a + 1e-30)
    
    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        n_steps = max(50, min(500, int(remaining * 5)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = fast_autocorr(a)
            c_max = conv.max()
            exp_c = np.exp((conv - c_max) / T)
            w = exp_c / exp_c.sum()
            
            # Gradient: grad[i] = 2 * sum_k w[k] * a[k-i] (with bounds)
            # Using FFT: this is 2 * (w cross-correlate a)[i]
            fft_size = 1 << (2 * n - 1 - 1).bit_length()
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            cross = irfft(W * np.conj(A), fft_size)[:n]
            grad = 2.0 * cross
            
            # Multiplicative weights update
            eta = 1.0 / (norm * 2.0)
            log_a -= eta * grad
            # Subtract max for numerical stability
            log_a -= log_a.max()
            a = np.exp(log_a)
            a = normalize_to(a, norm)
            
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
            
            if verbose and it % 50 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Mirror T={T:.2f} step={it} ({elapsed:.0f}s): best={best_score:.7f}")
    
    return best_a, best_score

def smooth_gradient_adam(a0, time_limit, verbose=True):
    """Smooth log-sum-exp minimization with Adam optimizer."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_to(np.asarray(a0, dtype=np.float64), norm)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    
    # Adam hyperparameters
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    m = np.zeros(n)
    v = np.zeros(n)
    
    temps = [20.0, 10.0, 5.0, 3.0, 2.0, 1.2, 0.8, 0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01]
    
    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        n_steps = max(50, min(300, int(remaining * 4)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = fast_autocorr(a)
            c_max = conv.max()
            exp_c = np.exp((conv - c_max) / T)
            w = exp_c / exp_c.sum()
            
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            cross = irfft(W * np.conj(A), fft_size)[:n]
            grad = 2.0 * cross
            
            # Adam update
            t_adam = it + 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**t_adam)
            v_hat = v / (1 - beta2**t_adam)
            
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = project_simplex(a_new, norm)
            a = a_new
            
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
            
            # Reduce learning rate
            if it % 50 == 0 and it > 0:
                lr *= 0.7
        
        # Reset Adam state for new temperature
        m = np.zeros(n)
        v = np.zeros(n)
    
    return best_a, best_score

def pair_cd(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent with greedy selection and line search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_to(np.asarray(a0, dtype=np.float64), norm)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    conv = fast_autocorr(a)
    
    t0 = time.monotonic()
    n_passes = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-12:
                continue
            
            # Try moving mass from i to nearby positions
            for offset in range(1, 30):
                for sign in [1, -1]:
                    j = (i + sign * offset) % n
                    if j == i:
                        continue
                    
                    # Line search: try a few amounts
                    for frac in [0.5, 0.25, 0.75, 0.1]:
                        d = a[i] * frac
                        if d < norm * 1e-12:
                            continue
                        a_t = a.copy()
                        a_t[i] -= d
                        a_t[j] += d
                        a_t = normalize_to(a_t, norm)
                        conv_t = fast_autocorr(a_t)
                        if conv_t.max() < best_score - 1e-12:
                            a = a_t
                            conv = conv_t
                            best_score = conv_t.max()
                            best_a = a.copy()
                            pass_improved = True
                            break
                    else:
                        continue
                    break  # Found improvement for this i, move to next
            else:
                continue
            # Found improvement, break inner loop
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 2:
                break
            no_improve = 0
        
        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PairCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.7f}")
    
    return best_a, best_score

def make_initializations(n, rng):
    """Create diverse initializations."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Ramp (linear)

    inits.append(ramp / ramp.sum())
    
    # Triangular (peaked in middle)
    mid = n // 2
    tri = np.abs(np.arange(n) - mid).astype(np.float64)
    tri = (tri.max() + 1 - tri)
    inits.append(tri / tri.sum())
    
    # Two peaks
    two = np.ones(n) * 0.1
    two[n//4] = 1.0
    two[3*n//4] = 1.0
    inits.append(two / two.sum())
    
    # Gaussian
    x = np.arange(n) / n - 0.5
    gauss = np.exp(-20 * x**2)
    inits.append(gauss / gauss.sum())
    
    # Comb (periodic)
    comb = np.zeros(n)
    comb[::3] = 1.0
    inits.append(comb / comb.sum())
    
    # Random
    rnd = rng.exponential(1.0, n)
    inits.append(rnd / rnd.sum())
    
    return inits
