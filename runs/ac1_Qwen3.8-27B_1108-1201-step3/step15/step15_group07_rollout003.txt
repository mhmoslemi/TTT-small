import numpy as np

from scipy.fft import fft, ifft, rfft, irfft

def get_fft_size(n):
    """Get next power of 2 >= 2n-1."""
    size = 2 * n - 1
    return 1 << (size - 1).bit_length()

def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    fft_size = get_fft_size(n)
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:2*n-1]

def eval_score(a):

def subgradient_descent(a0, time_limit, f_star=1.5, verbose=True):
    """
    Subgradient descent on the normalized problem:
    minimize max(conv(a,a)) subject to sum(a) = sqrt(2n), a >= 0
    Using Polyak step size.
    """
    n = len(a0)
    target_sum = np.sqrt(2.0 * n)
    a = normalize(a0)
    
    best_a = a.copy()
    conv = fast_autocorr(a)
    best_score = conv.max()
    t0 = time.monotonic()
    n_iters = 0
    patience = 0
    rng = np.random.default_rng(2024)
    
    # Running average of iterates (for Polyak averaging)
    average_a = a.copy()
    weight_sum = 1.0
    
    for it in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_autocorr(a)
        f_curr = conv.max()
        
        # Find active lags (where conv is close to max)
        threshold = f_curr - 0.01 * f_curr
        active_mask = conv >= threshold
        active_idx = np.where(active_mask)[0]
        
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(conv)])
        
        # Compute subgradient
        # d max(conv) / d a_i = sum_{k in active} d conv[k] / d a_i
        # conv[k] = sum_j a[j] * a[k-j]
        # d conv[k] / d a_i = a[k-i] + a[i-k] (for valid indices)
        grad = np.zeros(n)
        for k in active_idx:
            for i in range(n):
                j1 = k - i
                j2 = i - k
                if 0 <= j1 < n:
                    grad[i] += a[j1]
                if 0 <= j2 < n:
                    grad[i] += a[j2]
        
        # Project gradient onto the tangent space of {sum(a) = target_sum}
        # The constraint is sum(a) = target_sum, so project by subtracting mean
        grad = grad - np.mean(grad)
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Polyak step size
        gap = f_curr - f_star
        if gap < 1e-6:
            break
        eta = gap / (grad_norm ** 2)
        eta = min(eta, 1.0 / np.sqrt(n * (it + 1)))  # cap the step
        
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        
        # Renormalize
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * target_sum / s
        
        # Update running average
        weight_sum += 1.0
        average_a = (weight_sum - 1) * average_a / weight_sum + a_new / weight_sum
        
        a = a_new
        n_iters += 1
        
        # Check improvement
        conv_new = fast_autocorr(a)
        score_new = conv_new.max()
        
        conv_avg = fast_autocorr(average_a)
        score_avg = conv_avg.max()
        
        if min(score_new, score_avg) < best_score - 1e-12:
            if score_new < score_avg:
                best_score = score_new
                best_a = a_new.copy()
            else:
                best_score = score_avg
                best_a = average_a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > 500:
            # Perturb
            noise = rng.laplace(0, target_sum * 0.01, n)
            a_pert = a + noise
            a_pert = np.maximum(a_pert, 0)
            s = a_pert.sum()
            if s > 1e-15:
                a = a_pert * target_sum / s
                average_a = a.copy()
                weight_sum = 1.0
            patience = 0
        
        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  SD iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, f_curr={f_curr:.8f}")
    
    return best_a, best_score

def simulated_annealing(a0, time_limit, verbose=True):
    """Simulated annealing with block perturbations."""
    n = len(a0)
    target_sum = np.sqrt(2.0 * n)
    a = normalize(a0)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    T_start = target_sum * 0.1
    T_end = target_sum * 0.001
    rng = np.random.default_rng(777)
    
    n_total_iters = 50000
    start_time = time.monotonic()
    
    for it in range(n_total_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Cooling schedule
        progress = it / n_total_iters
        T = T_start * (T_end / T_start) ** progress
        
        # Generate a random move: pick 2-4 positions and redistribute
        n_move = rng.integers(2, 5)
        indices = rng.choice(n, size=n_move, replace=False)
        
        # Store original values
        orig = a[indices].copy()
        
        # Randomly redistribute (Dirichlet-like)
        weights = rng.dirichlet(np.ones(n_move) * 2.0)
        total = orig.sum()
        a[indices] = weights * total
        
        # Renormalize
        s = a.sum()
        if s < 1e-15:
            a[indices] = orig
            continue
        a_temp = a * target_sum / s
        
        conv_temp = fast_autocorr(a_temp)
        new_score = conv_temp.max()
        old_score = best_score
        
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_temp.copy()
            a = a_temp
        else:
            # Accept with probability exp(-(new - old) / T) if worse
            delta = new_score - eval_score(a)  # use current score, not best
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                a = a_temp
            else:
                a[indices] = orig
                s = a.sum()
                if s > 1e-15:
                    a = a * target_sum / s
    
    return best_a, best_score

            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Try to load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            sc = eval_score(parent)
            print(f"Parent: n={len(parent)}, score={sc:.6f}")
            best_seq = parent.copy()
            best_score = sc
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if parent is None:
        # Generate initial sequence
        n = 1600
        parent = spectral_moments_init(n, rng)
        best_seq = parent.copy()
        best_score = eval_score(best_seq)
        print(f"Generated init: n={n}, score={best_score:.6f}")
    
    n_best = len(best_seq)
    print(f"\nStarting search with n={n_best}, score={best_score:.6f}")
    
    # Phase 1: Frequency-domain equalization (40% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.40
    print(f"\n=== Phase 1: FDE n={n_best} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = frequency_domain_equalize(best_seq, t1, alpha=0.5, verbose=True)
        try_update(a_out, "fde1")
    
    # Phase 2: Subgradient descent (20% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = remaining * 0.20
    print(f"\n=== Phase 2: Subgradient n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = subgradient_descent(best_seq, t2, f_star=1.4, verbose=True)
        try_update(a_out, "sd")
    
    # Phase 3: Block coordinate descent (20% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = remaining * 0.20
    print(f"\n=== Phase 3: BCD n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_out, _ = block_coordinate_descent(best_seq, t3, verbose=True)
        try_update(a_out, "bcd")
    
    # Phase 4: Simulated annealing (10% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = remaining * 0.10
    print(f"\n=== Phase 4: SA n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 10:
        a_out, _ = simulated_annealing(best_seq, t4, verbose=True)
        try_update(a_out, "sa")
    
    # Phase 5: Final FDE polish (10% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = remaining * 0.10
    print(f"\n=== Phase 5: FDE polish n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 10:
        a_out, _ = frequency_domain_equalize(best_seq, t5, alpha=0.3, verbose=False)
        try_update(a_out, "fde2")
    
    # Final: try a few different n values
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        for n_try in [n_best // 2, n_best * 2]:
            if n_try < 10 or n_try > 10000:
                continue
            t_sub = min(remaining * 0.3, 120)
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            print(f"\n=== Trying n={n_try} (budget={t_sub:.0f}s) ===")
            a_init = spectral_moments_init(n_try, np.random.default_rng(n_try))
            a_out, _ = frequency_domain_equalize(a_init, t_sub, alpha=0.5, verbose=False)
            try_update(a_out, f"n{n_try}")
    
    # Final verification
    if best_seq is not None:
        fs = eval_score(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
