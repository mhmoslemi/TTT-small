def fast_autocorr(a):

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

    return mat

def fast_build_T(a, k_indices, n):
    """Faster build of constraint matrix using vectorized ops."""
    k_indices = np.asarray(k_indices)
    size = len(k_indices)
    mat = np.zeros((size, n))
    for idx in range(size):
        k = k_indices[idx]
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            mat[idx, lo:hi] = a[k - (hi-1):k - lo][::-1]
    return mat

def lp_solve_best_response(a_fixed, M, active_set, solver=None):
    """Solve LP: max sum(b) s.t. (a_fixed*b)[k] <= M, b >= 0."""
    n = len(a_fixed)
    k_indices = np.asarray(active_set)
    if len(k_indices) == 0:
        return None, None
    
    T = fast_build_T(a_fixed, k_indices, n)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    
    if solver is None:
        solver = cp.CLARABEL
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        # Try fallback solvers
        for s in [cp.PDLP, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=s, verbose=False)
                break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    return b_val, None

def project_simplex(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(len(v)) * target_sum / len(v)
    return v * target_sum / s

def subgradient_at_peak(a, k):
    """Compute subgradient of (a*a)[k] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    for j in range(n):
        v = 0.0
        j1 = k - j
        if 0 <= j1 < n:
            v += a[j1]
        j2 = j - k
        if 0 <= j2 < n:
            v += a[j2]
        grad[j] = v
    return grad

def subgradient_fast(a, k):
    """Faster subgradient computation using convolution."""
    n = len(a)
    # grad[j] = a[k-j] (if valid) + a[j-k] (if valid)
    # This is (reverse(a) * a)[j] evaluated at specific positions
    # Actually: grad[j] = (a reversed, padded) convolved with a, evaluated at j+k and j-k
    # Simpler: just use numpy
    grad = np.zeros(n)
    # a[k-j] for j where 0 <= k-j < n, i.e., max(0, k-n+1) <= j <= min(n-1, k)
    lo1 = max(0, k - n + 1)
    hi1 = min(n, k + 1)
    if lo1 < hi1:
        grad[lo1:hi1] += a[k - (hi1-1):k - lo1 + 1][::-1]
    # a[j-k] for j where 0 <= j-k < n, i.e., max(0, k) <= j <= min(n-1, k+n-1)
    lo2 = max(0, k)
    hi2 = min(n, k + n)
    if lo2 < hi2:
        grad[lo2:hi2] += a[0:hi2-lo2]
    return grad

def best_response_gradient(a_fixed, M, active_set, n_inner=10, eta=None):
    """
    Approximate the best response using projected gradient descent.
    Solve: max sum(b) s.t. (a_fixed*b)[k] <= M for k in active_set, b >= 0.
    Use subgradient of the constraint violations.
    """
    n = len(a_fixed)
    target_sum = np.sqrt(2.0 * n)
    b = np.ones(n) * target_sum / n
    
    if eta is None:
        eta = target_sum / (n * M + 1e-10)
    
    k_indices = np.asarray(active_set)
    
    for _ in range(n_inner):
        # Compute constraint violations
        # (a_fixed * b)[k] for k in active_set
        conv_ab = fast_autocorr(np.concatenate([a_fixed, np.zeros(n-1)]))  # wrong
        # Actually, we need (a_fixed * b), not (b * b)
        # Let's compute it directly
        # (a_fixed * b)[k] = sum_j a_fixed[k-j] * b[j]
        # Use FFT: convolve a_fixed and b
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        A = rfft(a_fixed, fft_size)
        B = rfft(b, fft_size)
        conv_ab = irfft(A * B, fft_size)[:size]
        
        violations = conv_ab[k_indices] - M
        max_viol = violations.max()
        
        if max_viol < 1e-10:
            break
        
        # Subgradient of max violation w.r.t. b
        k_star = k_indices[np.argmax(violations)]
        grad = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n, k_star + 1)
        grad[lo:hi] = a_fixed[k_star - (hi-1):k_star - lo + 1][::-1]
        
        # Also want to maximize sum(b), so add -1 to gradient
        grad -= 1.0 / n  # normalize
        
        # Update
        b_new = b - eta * grad
        b = np.maximum(b_new, 0.0)
        s = b.sum()
        if s > 1e-15:
            b = b * target_sum / s
        else:
            b = np.ones(n) * target_sum / n
    
    return b

def fast_best_response_iter(a0, time_limit, verbose=False, seed=0):
    """
    Fast best-response using projected gradient (no LP).
    Much faster per iteration, allows more iterations.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = get_score(a)
    t0 = time.monotonic()
    
    # We'll alternate: fix a, optimize b; fix b, optimize a
    # But since the problem is symmetric, we can just do:
    # a_new = project(best_response(a))
    
    step = 1.0 / n
    n_iters = 0
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_autocorr(a)
        M_curr = conv.max()
        
        # Active set: top peaks
        n_peaks = max(5, min(20, n // 50))
        top_indices = np.argsort(conv)[-n_peaks:]
        
        # Compute "force" on each coordinate j
        # We want to decrease (a*a)[k] for active k
        # The gradient of (a*a)[k] w.r.t. a[j] is: a[k-j] + a[j-k] (with bounds)
        grad = np.zeros(n)
        for k in top_indices:
            # Add contribution from this peak
            grad += subgradient_fast(a, k)
        
        # Move in direction that reduces the peaks
        # But we also want to maintain sum(a) = norm
        # So: a_new = a - step * grad + step * mean(grad) * ones (to preserve sum)
        grad_centered = grad - grad.mean()
        a_new = a - step * grad_centered
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        score_new = get_score(a_new)
        if score_new < best_score:
            a = a_new
            best_score = score_new
            best_a = a_new.copy()
            step = min(step * 1.01, 5.0)
        else:
            step *= 0.95
            if step < 1e-10:
                break
        
        n_iters += 1
        if verbose and n_iters % 1000 == 0:
            elapsed = time.monotonic() - t0

def coordinate_descent_fast(a0, time_limit, verbose=False):

    Uses the fact that changing a[j] only affects specific lags of the autocorrelation.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()

        return a, float('inf')

    size = 2 * n - 1
    
    n_passes = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            aj = a[j]
            s_others = a.sum() - aj
            
            # Try several values of a[j]
            # The score depends on a[j] through:
            # - sum: s_others + x
            # - conv[k] for k where j is involved: conv[k] += 2*x*a[k-j] + x^2 (at k=2j)
            
            # Key insight: only lags k in [j, 2j] and [j, j+n-1] are affected
            # Specifically, conv[k] = old_conv[k] + 2*x*(a[k-j] if valid) - 2*aj*(a[k-j] if valid) + ...
            # This is complex, so let's just try a few values and evaluate directly
            
            candidates = [0.0, aj * 0.5, aj * 0.75, aj, aj * 1.25, aj * 1.5,
                         s_others * 0.05, s_others * 0.1, s_others * 0.2]
            candidates = [x for x in candidates if 0 <= x <= 2 * norm]
            
            best_x = aj
            best_local_score = best_score
            
            for x in candidates:
                if abs(x - aj) < 1e-10:
                    continue
                a[j] = x
                s = a.sum()
                if s < 1e-15:
                    a[j] = aj
                    continue
                a_norm = a * norm / s
                sc = get_score(a_norm)
                if sc < best_local_score:
                    best_local_score = sc
                    best_x = x
                a[j] = aj
            
            if best_x != aj and abs(best_x - aj) > 1e-10:
                a[j] = best_x
                s = a.sum()
                if s > 1e-15:
                    a = a * norm / s
                    new_score = get_score(a)
                    if new_score < best_score - 1e-12:

                        best_a = a.copy()
                        improved = True

                        a[j] = aj
                        a = a * norm / a.sum()
        
        if not improved:

            if no_improve >= 3:
                break
        else:
            no_improve = 0
        
        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score

import time

def normalize_a(a, target_sum=None):
    s = a.sum()
    if s < 1e-15:
        ts = target_sum if target_sum else np.sqrt(2.0 * len(a))
        return np.ones(len(a)) * ts / len(a)
    if target_sum is None:
        target_sum = np.sqrt(2.0 * len(a))
    return a * target_sum / s

def build_T_fast(a, k_indices, n):
    size = len(k_indices)
    mat = np.zeros((size, n))
    for idx in range(size):
        k = k_indices[idx]
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            mat[idx, lo:hi] = a[k - (hi - 1):k - lo + 1][::-1]
    return mat

def lp_best_response(a_fixed, M, active_set, solver=None):
    n = len(a_fixed)
    k_indices = np.asarray(active_set)
    if len(k_indices) == 0:
        return None
    T = build_T_fast(a_fixed, k_indices, n)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    if solver is None:
        solver = cp.CLARABEL
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        for s in [cp.PDLP, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=s, verbose=False)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def exact_coordinate_descent(a0, time_limit, verbose=False):
    """
    Exact coordinate descent: for each coordinate j, compute the exact effect
    of changing a[j] and do a 1D minimization via golden section or grid.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    best_a = a.copy()
    best_score = get_score(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False
        conv = fast_autocorr(a)
        current_max = conv.max()
        # Background max: max of conv at lags NOT affected by any single coordinate
        # For coordinate j, affected lags are: k where 0<=k-j<n (i.e., j<=k<=j+n-1) and k=2j
        # Non-affected lags for j: k where k<j or k>j+n-1 (and k!=2j)
        # But since we're changing one coordinate at a time, let's just compute directly

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            aj = a[j]
            s_others = a.sum() - aj

            # Affected lags: k where j is "visible"
            # (a*a)[k] involves a[j]*a[k-j] if 0<=k-j<n, i.e., j<=k<=j+n-1
            # And a[k-j]*a[j] if 0<=k-j<n, same condition
            # Plus the diagonal: k=2j gives a[j]*a[j]
            # So affected range: k in [j, j+n-1] (clipped to [0, 2n-2])
            k_lo = max(0, j)
            k_hi = min(2 * n - 2, j + n - 1)

            # Background: max of conv[k] for k NOT in [k_lo, k_hi]
            if k_lo == 0 and k_hi == 2 * n - 2:
                bg_max = 0.0
            elif k_lo > 0 and k_hi < 2 * n - 2:
                bg_left = conv[:k_lo]
                bg_right = conv[k_hi + 1:]
                bg_max = max(bg_left.max() if len(bg_left) > 0 else 0.0,
                            bg_right.max() if len(bg_right) > 0 else 0.0)
            elif k_lo > 0:
                bg_max = conv[:k_lo].max() if k_lo > 0 else 0.0
            else:
                bg_max = conv[k_hi + 1:].max() if k_hi < 2 * n - 2 else 0.0

            # For affected k in [k_lo, k_hi]:
            # conv_new[k] = conv[k] + 2*(x-aj)*a[k-j] (if k!=2j) or conv[k] + x^2-aj^2 (if k==2j)
            # Note: a[k-j] is valid because k-j is in [k_lo-j, k_hi-j] = [max(0,j)-j, min(2n-2,j+n-1)-j]
            # = [max(0,j)-j, min(n-2, n-1)] ... let me just clip properly
            # k-j ranges from k_lo-j to k_hi-j
            # k_lo = max(0,j), so k_lo-j = max(0,j)-j = 0 if j>=0, which is always true
            # k_hi = min(2n-2, j+n-1), so k_hi-j = min(n-2, n-1) = n-2 if j+n-1 <= 2n-2, i.e., j<=n-2
            # So k-j ranges from 0 to min(n-2, k_hi-j)
            # Actually the valid range for k-j is [0, n-1], and we need k-j <= k_hi-j
            # Let me just compute the vector of a[k-j] for affected k

            affected_k = np.arange(k_lo, k_hi + 1)
            kj = affected_k - j  # these should be in [0, n-1]
            kj_valid = (kj >= 0) & (kj < n)
            a_vals = np.zeros(len(affected_k))
            a_vals[kj_valid] = a[kj[kj_valid]]

            # The change in conv at affected k:
            # delta[k] = 2*(x-aj)*a_vals[k] for k != 2j
            # delta[2j] = x^2 - aj^2 (special case)
            # conv_new[k] = conv[k] + delta[k]
            # Score = 2*n * max(bg_max, max(conv_new[affected])) / (s_others + x)^2

            # Find optimal x in [0, 2*norm]
            # Use grid + refinement
            x_lo = 0.0
            x_hi = 2.0 * norm
            s_min = s_others  # minimum sum (x=0)
            # Grid search
            n_grid = 30
            x_grid = np.linspace(x_lo, x_hi, n_grid)
            # Also add aj and nearby values
            x_grid = np.unique(np.concatenate([x_grid, [aj, aj * 0.5, aj * 1.5, s_others * 0.1]]))
            x_grid = x_grid[(x_grid >= 0) & (x_grid <= x_hi)]

            best_x = aj
            best_local = get_score(a)  # current score

            for x in x_grid:
                if abs(x - aj) < 1e-10:
                    continue
                s_new = s_others + x
                if s_new < 1e-10:
                    continue
                # Compute max conv
                delta = 2.0 * (x - aj) * a_vals
                # Handle diagonal
                diag_idx = np.where(affected_k == 2 * j)[0]
                if len(diag_idx) > 0:
                    # At k=2j: conv[2j] changes by x^2 - aj^2 (not 2*(x-aj)*a[j])
                    # We already added 2*(x-aj)*a[j] for this k, need to correct
                    delta[diag_idx] = x * x - aj * aj
                conv_new_affected = conv[affected_k] + delta
                max_new = max(bg_max, conv_new_affected.max())
                score_candidate = 2.0 * n * max_new / (s_new * s_new)
                if score_candidate < best_local:
                    best_local = score_candidate
                    best_x = x

            # Refine with local search around best_x
            if best_x != aj:
                ref_lo = max(0.0, best_x * 0.8)
                ref_hi = min(x_hi, best_x * 1.2)
                n_ref = 20
                x_ref = np.linspace(ref_lo, ref_hi, n_ref)
                for x in x_ref:
                    s_new = s_others + x
                    if s_new < 1e-10:
                        continue
                    delta = 2.0 * (x - aj) * a_vals
                    diag_idx = np.where(affected_k == 2 * j)[0]
                    if len(diag_idx) > 0:
                        delta[diag_idx] = x * x - aj * aj
                    conv_new_affected = conv[affected_k] + delta
                    max_new = max(bg_max, conv_new_affected.max())
                    score_candidate = 2.0 * n * max_new / (s_new * s_new)
                    if score_candidate < best_local:
                        best_local = score_candidate
                        best_x = x

            if abs(best_x - aj) > 1e-10:
                a[j] = best_x
                s = a.sum()
                if s > 1e-15:
                    a = a * norm / s
                    new_score = get_score(a)
                    if new_score < best_score - 1e-12:
                        best_score = new_score
                        best_a = a.copy()
                        improved = True
                        conv = fast_autocorr(a)
                    else:
                        a[j] = aj
                        a = a * norm / a.sum()

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")
    return best_a, best_score

def subgradient_multi(a0, time_limit, verbose=False):
    """Subgradient method using combined gradient from top peaks."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    best_a = a.copy()
    best_score = get_score(a)
    t0 = time.monotonic()
    step = 1.0 / (n ** 0.4)
    patience = 0
    max_patience = 800

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break
        conv = fast_autocrr(a) if False else fast_autocorr(a)
        M_curr = conv.max()
        # Get top peaks
        n_peaks = max(3, min(15, n // 50))
        top_idx = np.argsort(conv)[-n_peaks:]
        # Weighted gradient
        weights = np.exp(-10.0 * (M_curr - conv[top_idx]) / max(M_curr * 0.05, 1e-10))
        weights = weights / weights.sum()
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            # grad contribution from peak k
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            if lo < hi:
                grad[lo:hi] += w * a[k - (hi - 1):k - lo + 1][::-1]
            lo2 = max(0, k)
            hi2 = min(n, k + n)
            if lo2 < hi2:
                grad[lo2:hi2] += w * a[0:hi2 - lo2]
        # Center gradient to preserve sum
        grad = grad - grad.mean()
        a_new = a - step * grad / np.linalg.norm(grad + 1e-15)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new * norm / s_new
            sc = get_score(a_new)
            if sc < best_score - 1e-13:
                a = a_new
                best_score = sc
                best_a = a.copy()
                patience = 0
                step = min(step * 1.02, 3.0)
            else:
                patience += 1
                step *= 0.97
                if step < 1e-10:
                    break
    return best_a, best_score

            if lo2 < hi2:
                grad[lo2:hi2] += a[0:hi2 - lo2]
        # Multiplicative update: a_new = a * exp(-eta * grad / norm)
        gn = np.linalg.norm(grad) + 1e-15
        a_new = a * np.exp(-eta * grad / gn)
        a_new = np.maximum(a_new, eps)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new * norm / s_new
            sc = get_score(a_new)
            if sc < best_score - 1e-14:
                a = a_new
                best_score = sc
                best_a = a.copy()
                patience = 0
                eta = min(eta * 1.01, 10.0)
            else:
                patience += 1
                eta *= 0.98
                if eta < 1e-8:
                    break
    return best_a, best_score

def cosine_init(n, rng, n_terms=5):
    x = np.arange(n) / n
    a = np.ones(n) * 0.8
    for _ in range(n_terms):
        freq = rng.integers(1, 15)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.4)
        a += amp * np.cos(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.01)
    return a / a.sum()

def exp_init(n, rng):
    alpha = rng.uniform(0.5, 2.5)
    x = np.arange(n) / n
    a = np.exp(-alpha * x) + 0.1 * np.abs(rng.standard_normal(n))
    a = np.maximum(a, 0.01)
    return a / a.sum()

def symmetric_init(n, rng):
    half = n // 2
    a_half = 1.0 + 0.3 * np.cos(np.pi * np.arange(half) / half * rng.uniform(1, 4))
    a_half = np.maximum(a_half, 0.01)
    a = np.concatenate([a_half, a_half[::-1]])
    if len(a) < n:
        a = np.concatenate([a, np.ones(n - len(a)) * a_half.min()])
    a = a[:n]
    return a / a.sum()

def interpolate_to(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    out = np.interp(new_x, old_x, f_norm)
    out = np.maximum(out, 0.0)
    ss = out.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return out / ss
