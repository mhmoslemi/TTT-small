    for s in candidates:

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_active_constraints(a, active_idx):
    n = len(a)
    m = len(active_idx)
    T = np.zeros((m, n))
    for row, k in enumerate(active_idx):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                T[row, j] = a[idx]
    return T


def lp_max_sum(a, M, solver, active_idx=None):
    n = len(a)
    size = 2 * n - 1
    b = cp.Variable(n, nonneg=True)
    if active_idx is None or len(active_idx) >= size:
        T = build_active_constraints(a, np.arange(size))
    else:
        T = build_active_constraints(a, active_idx)
    cons = [T @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if val.sum() < 1e-15:
        return None
    return val


def nesterov_lp(a0, time_limit, solver, verbose=True, seed=0):
    """Nesterov-accelerated LP best-response with improved active set management."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    size = 2 * n - 1
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    stall_reset = 120
    alpha_min = 0.06
    erg_a = a.copy()
    erg_count = 1

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Improved adaptive active set
        if active_set is None or t % 25 == 0:
            # Start with wider tolerance, narrow over time
            tol = max(0.015, 0.06 * np.exp(-t / 400.0))
            threshold = M_curr * (1.0 - tol)
            active_set = np.where(conv_y >= threshold)[0]
            if len(active_set) < 12:
                active_set = np.where(conv_y >= M_curr * 0.55)[0]
            if len(active_set) > 0.9 * size:
                active_set = np.where(conv_y >= M_curr * 0.95)[0]

        # LP best response
        b_max = lp_max_sum(y, M_curr, solver, active_set)

        if b_max is None:
            if t % 50 == 0:
                active_set = None
                b_max = lp_max_sum(y, M_curr, solver, None)
            if b_max is None:
                continue

        b_sum = b_max.sum()
        g = b_max * norm / b_sum

        # Check violations on inactive constraints (crucial for correctness)
        if len(active_set) < size:
            inactive = np.setdiff1d(np.arange(size), active_set)
            if 0 < len(inactive) < size:
                T_check = build_active_constraints(y, inactive)
                viol = T_check @ g
                viol_idx = np.where(viol > M_curr * (1.0 + 1e-7))[0]
                if len(viol_idx) > 0:
                    active_set = np.union1d(active_set, inactive[viol_idx])

        # Nesterov update with improved schedule
        alpha_t = max(2.0 / (t + 2.0), alpha_min)
        y_new = (1.0 - alpha_t) * y + alpha_t * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.96)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y
        y = y_new
        a = a_new

        # Ergodic average
        erg_a = (erg_count * erg_a + a) / (erg_count + 1)
        erg_count += 1

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score - 1e-11:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Check ergodic average periodically
        if t % 100 == 0:
            s_erg = erg_a.sum()
            if s_erg > 1e-15:
                erg_check = erg_a * norm / s_erg
                sc_erg = fast_autocorr(erg_check).max()
                if sc_erg < best_score - 1e-11:
                    best_score = sc_erg
                    best_a = erg_check.copy()
                    stall_count = 0

        # Restart if stalled
        if stall_count > stall_reset:
            y = best_a.copy()
            a = best_a.copy()
            erg_a = a.copy()
            erg_count = 1
            active_set = None
            stall_count = 0

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.10f}, |A|={len(active_set)}")

    return best_a, best_score


def subgradient_polish(a0, time_limit, verbose=True, max_iter=100000):
    """FFT-based subgradient descent on the smoothed max-conv objective."""
    n = len(a0)
    size = 2 * n - 1
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Precompute FFT sizes
    N_fft = 1 << (3 * n - 2 - 1).bit_length()

    for t in range(1, max_iter + 1):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(a)
        M = conv.max()
        if M < 1e-12:
            break

        # Adaptive temperature for soft-max
        epsilon = M * max(0.005, 0.03 * np.exp(-t / 300.0))

        # Softmax weights
        exps = np.exp((conv - M) / epsilon)
        weights = exps / exps.sum()

        # Compute subgradient g_i = 2 * sum_k weights[k] * a[k-i] via FFT
        W = rfft(weights, N_fft)
        A = rfft(a, N_fft)
        g = 2.0 * irfft(W * np.conj(A), N_fft)[:n]

        # Normalize step size (Polyak-type)
        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-15:
            break
        step = 2.0 * best_score / (g_norm_sq / norm**2 + 1e-10)
        step = min(step, 0.05)

        a_new = a - step * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        score = fast_autocorr(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()

        a = a_new

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG t={t} ({elapsed:.0f}s): best={best_score:.10f}, eps={epsilon:.4f}")

    return best_a, best_score


def coord_descent(a0, time_limit, rng, verbose=True):
    """Coordinate descent with golden-section search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0
    phi = 0.6180339887498949

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break

            ai = a[i]
            if ai < 1e-12 * norm:
                continue

            lo = 0.0
            hi = ai * 2.5 + norm * 0.05

            c1 = lo + (1 - phi) * (hi - lo)
            c2 = lo + phi * (hi - lo)

            a_test1 = a.copy()
            a_test1[i] = c1
            a_test1 = a_test1 / a_test1.sum() * norm
            f1 = fast_autocorr(a_test1).max()

            a_test2 = a.copy()
            a_test2[i] = c2
            a_test2 = a_test2 / a_test2.sum() * norm
            f2 = fast_autocorr(a_test2).max()

            for _ in range(22):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + (1 - phi) * (hi - lo)
                    a_test1 = a.copy()
                    a_test1[i] = c1
                    a_test1 = a_test1 / a_test1.sum() * norm
                    f1 = fast_autocorr(a_test1).max()
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + phi * (hi - lo)
                    a_test2 = a.copy()
                    a_test2[i] = c2
                    a_test2 = a_test2 / a_test2.sum() * norm
                    f2 = fast_autocorr(a_test2).max()

            optimal_val = c1 if f1 < f2 else c2
            a_test_opt = a.copy()
            a_test_opt[i] = optimal_val
            a_test_opt = a_test_opt / a_test_opt.sum() * norm
            sc = fast_autocorr(a_test_opt).max()

            if sc < best_score - 1e-13:
                a = a_test_opt
                best_score = sc
                best_a = a.copy()
                pass_improved = True
                no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def pair_swap_cd(a0, time_limit, rng, verbose=True):
    """Pair swap coordinate descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_score = fast_autocorr(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(0, n, 2):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx % n]
            j = perm[(idx + 1) % n]
            if i == j:
                continue
            ai = a[i]
            aj = a[j]
            if ai < 1e-12 * norm and aj < 1e-12 * norm:
                continue

            best_val = fast_autocorr(a).max()
            best_delta = 0.0

            max_d = max(ai, 0)
            for frac in [0.05, 0.15, 0.3, 0.5, 0.7, 0.9]:
                d = max_d * frac
                if d <= 0 or d > ai:
                    continue
                a_try = a.copy()
                a_try[i] = ai - d
                a_try[j] = aj + d
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try * norm / s_try
                val = fast_autocorr(a_try).max()
                if val < best_val - 1e-14:
                    best_val = val
                    best_delta = d

            max_d2 = max(aj, 0)
            for frac in [0.05, 0.15, 0.3, 0.5, 0.7, 0.9]:
                d = max_d2 * frac
                if d <= 0 or d > aj:
                    continue
                a_try = a.copy()
                a_try[i] = ai + d
                a_try[j] = aj - d
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try * norm / s_try
                val = fast_autocorr(a_try).max()
                if val < best_val - 1e-14:
                    best_val = val
                    best_delta = -d

            if abs(best_delta) > 1e-15:
                a_try = a.copy()
                if best_delta > 0:
                    a_try[i] -= best_delta
                    a_try[j] += best_delta
                else:
                    a_try[i] += abs(best_delta)
                    a_try[j] -= abs(best_delta)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try * norm / s_try
                    sc = fast_autocorr(a_try).max()
                    if sc < best_score - 1e-14:
                        a = a_try
                        best_score = sc
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break

        if verbose and n_passes % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def structured_init(n, rng, style=0):
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif style == 1:
        a = np.sin(np.pi * x) ** 1.2
    elif style == 2:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(8 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif style == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif style == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif style == 5:
        a = 0.4 + 0.6 * np.sin(np.pi * x) ** 0.8
    elif style == 6:
        a = 1.0 + 0.25 * np.cos(6 * np.pi * x) + 0.1 * np.cos(12 * np.pi * x)
    elif style == 7:
        # Monotone increasing - minimizes center lag
        a = x ** 1.5
    elif style == 8:
        # Bi-modal with gap
        a = np.exp(-30.0 * (x - 0.2) ** 2) + np.exp(-30.0 * (x - 0.8) ** 2)
    else:
        a = np.sin(np.pi * x) ** 0.6 * (1.0 + 0.15 * np.cos(10 * np.pi * x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def perturb_seq(a, rng, scale=0.05):
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.05)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s

        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            old_score = best_score
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s, delta={old_score - sc:.6f})")

    remaining = time_budget - (time.monotonic() - t_global)
    n_base = len(best_seq)

    # === Phase 1: LP Nesterov on parent (40% time) ===
    print(f"\n=== Phase 1: LP Nesterov n={n_base} ===")
    t1 = min(remaining * 0.40, 600.0)
    if t1 > 80:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t1, solver, verbose=True, seed=0)
        try_update(a_out, f"lp1_n{n_base}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 2: Subgradient polish (15% time) ===
    print(f"\n=== Phase 2: Subgradient polish n={len(best_seq)} ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 30:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = subgradient_polish(a_norm, t2, verbose=True)
        try_update(a_out, f"sg1_n{n_cur}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 3: Coordinate descent (10% time) ===
    print(f"\n=== Phase 3: Coord Descent n={len(best_seq)} ===")
    t3 = min(remaining * 0.10, 120.0)
    if t3 > 20:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coord_descent(a_norm, t3, rng, verbose=True)
        try_update(a_out, f"cd1_n{n_cur}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 4: Pair swap CD (5% time) ===
    print(f"\n=== Phase 4: Pair CD n={len(best_seq)} ===")
    t4 = min(remaining * 0.05, 60.0)
    if t4 > 15:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = pair_swap_cd(a_norm, t4, rng, verbose=False)
        try_update(a_out, f"pcd1_n{n_cur}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 5: Perturbation + LP restart (15% time) ===
    print(f"\n=== Phase 5: Perturbation restarts ===")
    t5_total = min(remaining * 0.15, 200.0)
    n_restarts = 3
    t5_each = t5_total / n_restarts
    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        t5 = min(t5_each, remaining - 40)
        if t5 < 30:
            break
        n_cur = len(best_seq)
        scale = 0.02 + 0.02 * r
        pert = perturb_seq(best_seq, rng, scale)
        a_norm = pert * np.sqrt(2.0 * n_cur) / pert.sum()
        a_out, _ = nesterov_lp(a_norm, t5, solver, verbose=False, seed=200 + r * 13)
        try_update(a_out, f"pert{r}_n{n_cur}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 6: Subgradient on perturbed (5% time) ===
    if remaining > 60:
        print(f"\n=== Phase 6: Subgradient on best ===")
        t6 = min(remaining * 0.05, 80.0)
        if t6 > 20:
            n_cur = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = subgradient_polish(a_norm, t6, verbose=False)
            try_update(a_out, f"sg2_n{n_cur}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 7: Try longer length (10% time) ===
    n_long = n_base + 300
    print(f"\n=== Phase 7: LP Nesterov n={n_long} ===")
    t7 = min(remaining * 0.10, 120.0)
    if t7 > 30:
        init_long = interpolate_seq(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = nesterov_lp(a_norm, t7, solver, verbose=True, seed=77)
        try_update(a_out, f"lp_long_n{n_long}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 8: Try shorter length ===
    n_short = max(800, n_base - 200)
    if n_short != n_base and remaining > 80:
        print(f"\n=== Phase 8: LP Nesterov n={n_short} ===")
        t8 = min(remaining * 0.08, 100.0)
        if t8 > 20:
            init_short = interpolate_seq(best_seq, n_short)
            a_norm = init_short * np.sqrt(2.0 * n_short) / init_short.sum()
            a_out, _ = nesterov_lp(a_norm, t8, solver, verbose=False, seed=33)
            try_update(a_out, f"lp_short_n{n_short}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 9: Final LP polish on best (use remaining) ===
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final LP n={n_best} ===")
    t9 = max(remaining * 0.45, 60.0)
    t9 = min(t9, remaining - 30)
    if t9 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t9, solver, verbose=True, seed=1234)
        try_update(a_out, f"lp_final_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 10: Final subgradient ===
    if remaining > 40:
        n_best = len(best_seq)
        print(f"\n=== Phase 10: Final SG n={n_best} ===")
        t10 = min(remaining * 0.3, 60.0)
        if t10 > 15:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = subgradient_polish(a_norm, t10, verbose=False)
            try_update(a_out, f"sg_final_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 11: Final CD ===
    if remaining > 30:
        n_best = len(best_seq)
        print(f"\n=== Phase 11: Final CD n={n_best} ===")
        t11 = min(remaining * 0.5, 50.0)
        if t11 > 10:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = coord_descent(a_norm, t11, rng, verbose=False)
            try_update(a_out, f"cd_final_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 12: Final pair CD ===
    if remaining > 20:
        n_best = len(best_seq)
        t12 = min(remaining * 0.6, 30.0)
        if t12 > 10:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = pair_swap_cd(a_norm, t12, rng, verbose=False)
            try_update(a_out, f"pcd_final_n{n_best}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
