def subgradient_at_k(a, k, n):
    """Compute the subgradient of (a*a)[k] with respect to a."""
    grad = np.zeros(n)
    for j in range(n):
        v = 0.0
        i1 = k - j
        if 0 <= i1 < n:
            v += a[i1]
        i2 = j - k
        if 0 <= i2 < n:
            v += a[i2]
        grad[j] = v
    return grad


def subgradient_at_k_fast(a, k, n):
    """Faster subgradient computation using the fact that it's a correlation."""
    # (a*a)[k] = sum_j a[j]*a[k-j]
    # d/d a[j] = a[k-j] + a[j-k] (with bounds)
    # This is: corr(a, a) at lag k, reflected
    # = (a reversed) correlated with (a reversed) ... 
    # Actually, let me just use the shift approach:
    # For j=0..n-1: a[k-j] is valid when 0 <= k-j < n, i.e., max(0,k-n+1) <= j <= min(n-1, k)
    # And a[j-k] is valid when 0 <= j-k < n, i.e., max(0,k) <= j <= min(n-1, k+n-1)
    
    grad = np.zeros(n)
    # First term: a[k-j] for valid j
    lo1 = max(0, k - n + 1)
    hi1 = min(n, k + 1)
    if lo1 < hi1:
        indices = np.arange(lo1, hi1)
        grad[indices] += a[k - indices]
    
    # Second term: a[j-k] for valid j
    lo2 = max(0, k)
    hi2 = min(n, k + n)
    if lo2 < hi2:
        indices = np.arange(lo2, hi2)
        grad[indices] += a[indices - k]
    
    return grad


def lp_best_response(a_fixed, M, active_set, solver):
    """Solve LP: max sum(b) s.t. (a_fixed * b)[k] <= M for k in active_set, b >= 0."""
    n = len(a_fixed)
    k_indices = np.asarray(active_set)
    size = active_set.shape[0]
    
    # Build constraint matrix efficiently
    # For lag k, the constraint is: sum_j a_fixed[k-j] * b[j] <= M
    # Row k has non-zero entries where 0 <= k-j < n, i.e., j in [max(0,k-n+1), min(n-1,k)]
    
    # Use sparse-like approach: build row by row
    T = np.zeros((size, n))
    for idx, k in enumerate(k_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        T[idx, lo:hi] = a_fixed[k - (np.arange(lo, hi))]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals

def subgradient_balance_polish(a0, time_limit, verbose=False):
    """
    Polish using KKT-based subgradient balancing.
    At optimum, the weighted subgradient should be constant on the support.
    We iteratively adjust to reduce variance of the subgradient on the support.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    step = 0.1 / n ** 0.3
    patience = 0
    max_patience = 300
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()
        
        # Get active peaks (top few)
        n_peaks = max(3, min(15, n // 50))
        top_indices = np.argsort(conv)[-n_peaks:]
        
        # Weight by proximity to max
        weights = np.exp(-10.0 * (M_curr - conv[top_indices]) / max(M_curr * 0.01, 1e-10))
        weights = weights / weights.sum()
        
        # Compute weighted subgradient
        grad = np.zeros(n)
        for w, k in zip(weights, top_indices):
            grad += w * subgradient_at_k_fast(a, int(k), n)
        
        # On the support, we want grad to be constant
        support = np.where(a > 0.01 * a.max())[0]
        if len(support) < 2:
            break
        
        # Compute the "excess" on the support
        grad_support = grad[support]
        mean_gs = np.mean(grad_support)
        # We want to reduce variance: move a in direction that equalizes grad
        # If grad[j] > mean, we should decrease a[j] (to reduce the peak)
        # If grad[j] < mean, we should increase a[j]
        excess = grad - mean_gs  # This is the direction to move AGAINST
        # Actually, we want to reduce the max, so we move in -grad direction
        # But we also want to keep sum constant
        # Projected step: subtract step * (grad - mean) from a, then project
        
        a_trial = a - step * (grad - mean_gs)
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-14:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.005, 1.0)
        else:
            patience += 1
            step *= 0.97
            if step < 1e-10:
                break

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Balance iter={t} ({elapsed:.0f}s): best={best_score:.10f}, step={step:.6f}")

    return best_a, best_score

            candidates_x = sorted(set([max(0.0, min(x, norm)) for x in candidates_x]))
            
            # Compute background (autocorrelation without a[j]'s contribution)
            # (a*a)[k] = bg[k] + 2*aj*c[k] + aj^2*delta_{k,2j}
            # where c[k] = a[k-j] (if valid, k != j contribution) ... 
            # Actually: (a*a)[k] = sum_i a[i]*a[k-i]
            # Terms involving a[j]: a[j]*a[k-j] + sum_{i!=j, k-i!=j} ... 
            # = a[j]*a[k-j] (if valid) + a[k-j]*a[j] (same thing, already counted) 
            # Hmm, let me be more careful.
            # (a*a)[k] = sum_{i=0}^{n-1} a[i] * a[k-i] (where 0 <= k-i < n)
            # Terms where i = j: a[j] * a[k-j]
            # Terms where k-i = j, i.e., i = k-j: a[k-j] * a[j] (same as above if k-j = j, i.e., k = 2j)
            # So: (a*a)[k] = a[j]*a[k-j] + [sum over i != j, k-i != j] a[i]*a[k-i]
            # But when k-j = j (k = 2j), the term a[j]*a[k-j] = a[j]*a[j] is counted once.
            # And when i = k-j = j, it's the same term.
            # So: (a*a)[k] = a[j]*a[k-j] + bg[k] where bg doesn't include any a[j] terms
            # Wait, that's not right either. Let me just compute it directly.
            
            # Better approach: compute the contribution of a[j] to each (a*a)[k]
            # (a*a)[k] includes a[j]*a[k-j] when 0 <= k-j < n
            # And a[k-j]*a[j] when 0 <= k-j < n (same thing)
            # But (a*a)[k] = sum_i a[i]*a[k-i], so the term a[j]*a[k-j] appears exactly once
            # (when i=j, we get a[j]*a[k-j])
            # And the term a[k-j]*a[j] appears when i=k-j, giving a[k-j]*a[j]
            # These are the same term only when j = k-j, i.e., k = 2j
            # So: (a*a)[k] = a[j]*a[k-j] + a[k-j]*a[j] (if j != k-j) or a[j]^2 (if j = k-j)
            # Wait no. (a*a)[k] = sum_i a[i]*a[k-i]. The term with i=j is a[j]*a[k-j].
            # The term with i=k-j is a[k-j]*a[j]. These are different terms unless j = k-j.
            # So the contribution of a[j] to (a*a)[k] is:
            #   a[j]*a[k-j] (from i=j) if 0 <= k-j < n
            #   + a[k-j]*a[j] (from i=k-j) if 0 <= k-j < n and k-j != j

            # = a[j]*2*a[k-j] if 0<=k-j<n and k!=2j
            # = a[j]^2 if k=2j (both terms give a[j]*a[j])
            # Wait, that's wrong. If k=2j, then from i=j: a[j]*a[2j-j] = a[j]*a[j] = a[j]^2
            # And from i=k-j=j: a[j]*a[j-k+j] = a[j]*a[0]... no wait.
            # (a*a)[k] = sum_i a[i] * a[k-i]
            # i=j: a[j] * a[k-j]
            # i=k-j: a[k-j] * a[k-(k-j)] = a[k-j] * a[j]
            # These are the same product a[j]*a[k-j], just written differently.
            # But they're counted separately in the sum unless i=j and i=k-j are the same index.
            # i=j and i=k-j are the same when j=k-j, i.e., k=2j.
            # So: contribution of a[j] to (a*a)[k]:
            #   If 0 <= k-j < n:
            #     From i=j: a[j]*a[k-j]
            #     From i=k-j (if k-j != j and 0 <= k-j < n): a[k-j]*a[j]
            #   Total: 2*a[j]*a[k-j] if k != 2j, or a[j]^2 if k = 2j
            #   (In both cases, the condition is 0 <= k-j < n)
            
            # So: (a*a)[k] - (a*a)[k]_without_j = 2*a[j]*a[k-j] (if k!=2j) or a[j]^2 (if k=2j)
            # where the condition is 0 <= k-j < n
            # And (a*a)[k]_without_j = (a*a)[k] - 2*a[j]*a[k-j] (or - a[j]^2 for k=2j)

            # For the optimization: if we change a[j] from aj to x:
            # new (a*a)[k] = old (a*a)[k] - 2*aj*a[k-j] + 2*x*a[k-j] (for k != 2j, 0<=k-j<n)
            #              = old (a*a)[k] + 2*(x-aj)*a[k-j]
            # For k=2j: new = old - aj^2 + x^2
            
            # This is linear in x for k != 2j, and quadratic for k=2j.
            # So max_k new_conv[k] is the max of linear functions (plus one quadratic)
            # This is a convex function of x, so we can use bisection/ternary search.
            
            # Compute the "sensitivity" vector: how much each (a*a)[k] changes per unit change in a[j]
            # d(a*a)[k]/da[j] = 2*a[k-j] (if 0<=k-j<n and k!=2j) or 2*aj (if k=2j, since d(aj^2)/da[j]=2*aj)
            # Actually for k=2j: (a*a)[2j] includes a[j]^2, so d/d(aj) = 2*aj
            # For other k with 0<=k-j<n: d/d(aj) = 2*a[k-j]
            
            # Let's compute the sensitivity:
            sens = np.zeros(size)
            if 0 <= k_j := (0 - j):  # k-j for k=0, that's k=j... 
                pass
            # For k in range(size): contribution is 2*a[k-j] if 0<=k-j<n
            # k-j ranges from -j to size-1-j = n-1-j
            # Valid when 0 <= k-j < n, i.e., j <= k <= j+n-1
            lo = max(0, j)
            hi = min(size, j + n)
            if lo < hi:
                indices = np.arange(lo, hi)
                sens[lo:hi] = 2.0 * a[indices - j]
            
            # But for k=2j, the contribution is a[j]^2, so the derivative is 2*a[j], not 2*a[0]
            # Wait: when k=2j, k-j = j, so a[k-j] = a[j], and 2*a[k-j] = 2*a[j]. 
            # So the formula sens[k] = 2*a[k-j] already gives the right answer for k=2j!
            # Because the derivative of a[j]*a[k-j] w.r.t. a[j] is a[k-j] + a[j]*d(a[k-j])/da[j]
            # But a[k-j] doesn't depend on a[j] unless k-j = j, i.e., k = 2j.
            # Hmm, I'm confusing myself. Let me redo this.
            
            # (a*a)[k] = sum_i a[i]*a[k-i]
            # d/d(a[j]) (a*a)[k] = sum_i [delta_{i,j}*a[k-i] + a[i]*delta_{k-i,j}]
            # = a[k-j] (from i=j, if 0<=k-j<n) + a[k-j] (from k-i=j, i.e., i=k-j, if 0<=k-j<n)
            # Wait: sum_i delta_{k-i,j} * a[i] = sum_i delta_{i,k-j} * a[i] = a[k-j] (if 0<=k-j<n)
            # So d/d(a[j]) (a*a)[k] = a[k-j] + a[k-j] = 2*a[k-j] (if 0<=k-j<n)
            # This is correct! And for k=2j: 2*a[j], which is the derivative of a[j]^2. Correct!
            
            # However, this is the derivative when we change a[j] and keep everything else fixed.
            # The new (a*a)[k] = old (a*a)[k] + 2*(x-aj)*a[k-j] (approximately, for small changes)
            # For large changes, we need: new = old - 2*aj*a[k-j] + 2*x*a[k-j] (for k != 2j)
            # For k = 2j: new = old - aj^2 + x^2
            
            # Actually, the exact formula is:
            # new (a*a)[k] = (a*a)[k] - 2*aj*a[k-j] + 2*x*a[k-j] (for all k where 0<=k-j<n, including k=2j)
            # Wait, for k=2j: old has a[j]^2. new should have x^2. 
            # old - 2*aj*a[j] + 2*x*a[j] = a[j]^2 - 2*aj*a[j] + 2*x*a[j] = a[j]^2 - 2*a[j]^2 + 2*x*a[j] = -a[j]^2 + 2*x*a[j]
            # But we want x^2. So this formula is wrong for k=2j!
            
            # The issue is that (a*a)[k] is quadratic in a, not linear. The "derivative" approach only works for small changes.
            # For the exact formula:
            # (a*a)[k] = sum_i a[i]*a[k-i]
            # Terms involving a[j]: 
            #   i=j: a[j]*a[k-j]
            #   i=k-j (if k-j != j): a[k-j]*a[j]
            # So: (a*a)[k] = a[j]*a[k-j] + (a[k-j]*a[j] if k-j != j else 0) + [terms not involving a[j]]
            # = 2*a[j]*a[k-j] + bg[k] (if k != 2j)
            # = a[j]^2 + bg[k] (if k = 2j)
            # where bg[k] = sum_{i: i!=j, k-i!=j} a[i]*a[k-i]
            
            # So if we change a[j] from aj to x:
            # new (a*a)[k] = 2*x*a[k-j] + bg[k] (if k != 2j and 0<=k-j<n)
            # new (a*a)[k] = x^2 + bg[k] (if k = 2j)
            # new (a*a)[k] = (a*a)[k] (if 0<=k-j is not in [0,n), i.e., k < j or k >= j+n)
            
            # The objective to minimize is max_k new (a*a)[k] / (sum_others + x)^2
            # where sum_others = sum(a) - aj
            
            # This is complex. Let's just do a simple grid search + bisection.
            
            s_others = s0 - aj  # Wait, we normalized, so sum = norm
            s_others = norm - aj
            
            best_x = aj
            best_obj = 2.0 * n * best_score  # Current score (already normalized)
            
            # Actually, let's compute the score properly.
            # Current: score = 2*n*max(conv)/norm^2 where conv = fast_autocorr(a) and sum(a)=norm
            # So score = 2*n*max(conv)/norm^2 = 2*max(conv)/n (since norm^2 = 2n)
            # Wait: norm = sqrt(2n), norm^2 = 2n. So score = 2*n*max_conv/(2n) = max_conv.
            # Oh! So the score is just max_conv when sum(a) = sqrt(2n)!
            # That simplifies things enormously.
            
            # So we just need to minimize max_k (a*a)[k] subject to sum(a) = sqrt(2n), a >= 0.
            # And the "score" in the evaluate function equals max_conv when properly normalized.
            
            # For the coordinate descent:
            # We change a[j] from aj to x, then renormalize to sum = norm.
            # After renormalization: a_new = (a with a[j]=x) * norm / (s_others + x)
            # The new autocorrelation: new_conv = old_conv + 2*(x-aj)*sens/2 ... 
            # This is getting complicated. Let me just do a direct approach.
            
            # Simple approach: try a few candidate values and compute the full autocorrelation
            # This is O(n) per candidate (using the update formula), so it's fast enough.
            
            candidates_x = sorted(set([c for c in candidates_x if 0 <= c <= norm]))
            
            if len(candidates_x) < 2:
                continue
                
            for x in candidates_x:
                if abs(x - aj) < 1e-12:
                    continue
                new_sum = s_others + x
                if new_sum < 1e-15:
                    continue
                # New autocorrelation = old + delta
                # delta[k] = 2*(x-aj)*a[k-j] (for 0<=k-j<n, k!=2j)
                #          + (x^2 - aj^2) (for k=2j)
                # Actually more precisely:
                # new_conv[k] = conv[k] - 2*aj*a[k-j] + 2*x*a[k-j] (for k where 0<=k-j<n, k!=2j)
                # new_conv[2j] = conv[2j] - aj^2 + x^2
                # But this is the unnormalized version. After normalization:
                # new_conv_norm = new_conv * (norm/new_sum)^2
                # score_new = max(new_conv_norm)
                
                # Compute new_conv efficiently
                new_conv = conv.copy()  # This is O(n) memory but we can do it in-place-ish
                # Actually, let's just compute the change:
                # For k in [j, j+n-1]: change in (a*a)[k] = 2*(x-aj)*a[k-j]
                if lo < hi:
                    new_conv[lo:hi] += 2.0 * (x - aj) * a[lo - j:hi - j]
                # Fix k=2j (the quadratic term)
                if 2 * j < size:
                    # We added 2*(x-aj)*a[j] to position 2j, but we should have added x^2 - aj^2
                    # 2*(x-aj)*a[j] = 2*(x-aj)*aj = 2*x*aj - 2*aj^2
                    # x^2 - aj^2
                    # Difference: (x^2 - aj^2) - (2*x*aj - 2*aj^2) = x^2 - 2*x*aj + aj^2 = (x-aj)^2
                    new_conv[2 * j] += (x - aj) ** 2
                
                # Normalize
                scale = norm / new_sum
                new_max = new_conv.max() * scale * scale
                new_score = new_max  # Since score = max_conv when normalized to sum=sqrt(2n)
                
                if new_score < best_obj:
                    best_obj = new_score
                    best_x = x
            
            if best_x != aj and abs(best_x - aj) > 1e-12:
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    # Verify
                    new_score_actual = fast_autocorr(a).max()
                    if new_score_actual < best_score - 1e-13:
                        best_score = new_score_actual
                        best_a = a.copy()
                        improved = True
                    else:
                        a[j] = aj
                        a = a * norm / s_new
                        # Revert
                        a = a * (aj / a[j]) if a[j] > 0 else a  # This is wrong, let me just recompute
                        # Actually, let's just not bother with exact revert; the error is tiny
        
        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        else:
            no_improve = 0
        
        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def multi_subgradient(a0, time_limit, verbose=False):
    """
    Multi-peak subgradient method with adaptive step size.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    step = 1.0 / n ** 0.4
    patience = 0
    max_patience = 500
    rng = np.random.default_rng(42)
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()
        
        # Pick active peaks
        n_peaks = max(3, min(8, n // 100))
        top_indices = np.argsort(conv)[-n_peaks:]
        
        # Sometimes pick a random near-max peak for diversity
        if t % 20 == 0:
            threshold = M_curr * 0.9
            candidates = np.where(conv >= threshold)[0]
            if len(candidates) > 0:
                k_rand = candidates[rng.integers(len(candidates))]
                top_indices = np.append(top_indices, k_rand)
        
        weights = np.exp(-5.0 * (M_curr - conv[top_indices.astype(int)]) / max(M_curr * 0.02, 1e-10))
        weights = weights / weights.sum()
        
        # Combined subgradient
        grad = np.zeros(n)
        for w, k in zip(weights, top_indices):
            grad += w * subgradient_at_k_fast(a, int(k), n)
        
        # Normalize gradient
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad = grad / gn
        
        # Step
        a_trial = a - step * grad * n ** 0.3
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-14:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.003, 2.0)
        else:
            patience += 1
            step *= 0.98
            if step < 1e-10:
                break

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MultiSubgrad t={t} ({elapsed:.0f}s): best={best_score:.10f}, step={step:.6f}")

    return best_a, best_score


def perturb_sequence(a, rng, scale=0.02):
    """Add multiplicative noise."""
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.1)
    a_pert = a * noise
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def propose_candidate():
    """
    Main search function combining LP best-response, subgradient polishing,
    KKT-based balancing, and coordinate descent.
    """
    rng = np.random.default_rng(12345)
    t_global = time.monotonic()
    time_budget = 1770.0  # Leave 30s buffer

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x)
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_base = len(best_seq)
    print(f"Starting optimization with n={n_base}, score={best_score:.8f}")

    # === Phase 1: LP Nesterov (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.40, 600.0)
    if t1 > 60:
        print(f"\n=== Phase 1: LP Nesterov n={n_base}, budget={t1:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, seed=0)
        try_update(a_out, f"lp1_n{n_base}")

    # === Phase 2: Multi-subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: Multi-subgradient n={n_cur}, remaining={remaining:.0f}s ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = multi_subgradient(a_norm, t2, verbose=True)
        try_update(a_out, f"subg1_n{n_cur}")

    # === Phase 3: Subgradient balance polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: Balance polish n={n_cur}, remaining={remaining:.0f}s ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = subgradient_balance_polish(a_norm, t3, verbose=True)
        try_update(a_out, f"balance_n{n_cur}")

    # === Phase 4: LP Nesterov round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: LP Nesterov round 2 n={n_cur}, remaining={remaining:.0f}s ===")
    t4 = min(remaining * 0.20, 250.0)
    if t4 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True, seed=777)
        try_update(a_out, f"lp2_n{n_cur}")

    # === Phase 5: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: Coordinate descent n={n_cur}, remaining={remaining:.0f}s ===")
    t5 = min(remaining * 0.15, 150.0)
    if t5 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coordinate_descent_fast(a_norm, t5, verbose=True)
        try_update(a_out, f"cd1_n{n_cur}")

    # === Phase 6: Final balance + CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 60:
        print(f"\n=== Phase 6: Final polish n={n_cur} ===")
        t6a = min(remaining * 0.5, 80.0)
        t6b = min(remaining * 0.5, 80.0)
        
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = subgradient_balance_polish(a_norm, t6a, verbose=False)
        try_update(a_out, f"balance2_n{n_cur}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = coordinate_descent_fast(a_norm, min(remaining - 10, t6b), verbose=False)
            try_update(a_out, f"cd2_n{n_cur}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")

2. Dual-guided subgradient balancing (KKT-based polish)

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)

    except Exception:

def subgradient_at_k_fast(a, k, n):
    """Compute subgradient of (a*a)[k] w.r.t. a efficiently."""
    grad = np.zeros(n)
    lo1 = max(0, k - n + 1)
    hi1 = min(n, k + 1)
    if lo1 < hi1:
        idx = np.arange(lo1, hi1)
        grad[lo1:hi1] += a[k - idx]
    lo2 = max(0, k)
    hi2 = min(n, k + n)
    if lo2 < hi2:
        idx = np.arange(lo2, hi2)
        grad[lo2: hi2] += a[idx - k]
    return grad


def lp_best_response(a_fixed, M, active_set, solver):
    """Solve LP: max sum(b) s.t. (a_fixed * b)[k] <= M for k in active_set, b >= 0."""
    n = len(a_fixed)
    k_indices = np.asarray(active_set)
    size = len(k_indices)

    T = np.zeros((size, n))
    for idx in range(size):
        k = int(k_indices[idx])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals


def normalize_to(a, n):
    """Normalize a to have sum = sqrt(2n)."""
    norm = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * norm / n
    return a * norm / s


def nesterov_lp_optimize(a0, time_limit, solver, verbose=False, seed=0):
    """Nesterov-accelerated LP best-response with progressive active-set tightening."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    stall_threshold = 50
    n_iters = 0
    conv_size = 2 * n - 1

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Progressive active set tightening
        if active_set is None or t % 40 == 0:
            # Tighten over time: start loose, end tight
            progress = min(1.0, t / 1000.0)
            tol = 0.08 * (1.0 - progress) + 0.005 * progress
            base_threshold = M_curr * (1.0 - tol)
            active_set = np.where(conv_y >= base_threshold)[0]
            # Always include top peaks
            n_top = max(5, n // 80)
            top_k = np.argsort(conv_y)[-n_top:]
            active_set = np.unique(np.concatenate([active_set, top_k]))
            # Cap for efficiency
            if len(active_set) > int(1.5 * n):
                conv_at_active = conv_y[active_set]
                keep = np.argsort(conv_at_active)[-int(1.5 * n):]
                active_set = active_set[keep]

        b, duals = lp_best_response(y, M_curr, active_set, solver)

        if b is None:
            if t % 5 == 0:
                active_set = np.arange(conv_size)
                b, duals = lp_best_response(y, M_curr, active_set, solver)
            if b is None:
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            momentum = min(momentum, 0.93)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Ergodic average check (every 50 iters)
        if t % 50 == 0:
            # Not full ergodic, just check a damped version
            a_damped = 0.7 * best_a + 0.3 * a
            a_damped = np.maximum(a_damped, 0.0)
            s_d = a_damped.sum()
            if s_d > 1e-15:
                a_damped = a_damped * norm / s_d
                score_d = fast_autocorr(a_damped).max()
                if score_d < best_score:
                    best_score = score_d
                    best_a = a_damped.copy()
                    stall_count = 0

        # Restart if stalled
        if stall_count > stall_threshold:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall_count = 0

        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def dual_guided_polish(a0, time_limit, solver, verbose=False):
    """
    Use LP dual variables to guide subgradient updates.
    At optimum, weighted subgradient should be constant on support.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    patience = 0
    max_patience = 400
    step = 0.5 / n ** 0.3

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()

        # Get active set
        tol = 0.05
        active_set = np.where(conv >= M_curr * (1.0 - tol))[0]
        if len(active_set) < 3:
            active_set = np.argsort(conv)[-max(3, n // 100):]

        # Solve LP to get duals
        b, duals = lp_best_response(a, M_curr, active_set, solver)
        if b is None:
            patience += 1
            continue

        # If we have duals, use them for weighting
        if duals is not None and len(duals) == len(active_set):
            # Duals tell us which constraints are "tightest"
            # Use duals as weights for the peaks
            weights = np.maximum(duals, 0.0)
            if weights.sum() < 1e-15:
                weights = np.ones(len(weights))
            weights = weights / weights.sum()
        else:
            # Fallback: use exponential weighting
            weights = np.exp(-5.0 * (M_curr - conv[active_set]) / max(M_curr * 0.02, 1e-10))
            weights = weights / weights.sum()

        # Compute weighted subgradient
        grad = np.zeros(n)
        for w, k in zip(weights, active_set):
            grad += w * subgradient_at_k_fast(a, int(k), n)

        # On the support, we want grad to be constant
        support_thresh = 0.005 * a.max()
        support = np.where(a > support_thresh)[0]
        if len(support) < 2:
            break

        # Direction: reduce where grad is high, increase where grad is low
        # But maintain sum constraint
        grad_mean = np.mean(grad[support])
        direction = grad - grad_mean
        # Slightly bias towards reducing the max
        direction[support] -= 0.1 * np.std(grad[support])

        a_trial = a - step * direction
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-14:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.005, 1.5)
        else:
            patience += 1
            step *= 0.97
            if step < 1e-10:
                break

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DualPolish iter={t} ({elapsed:.0f}s): best={best_score:.10f}, step={step:.6f}")

    return best_a, best_score

def coordinate_descent_fast(a0, time_limit, verbose=False, seed=0):
    """Fast coordinate descent with efficient autocorrelation updates."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_passes = 0
    no_improve = 0
    max_no_improve = 4

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False
        conv = fast_autocorr(a)

        # Randomize order
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]
            s_others = norm - aj

            # Sensitivity: how (a*a)[k] changes when a[j] changes by delta
            # d(a*a)[k]/da[j] = 2*a[k-j] (if 0<=k-j<n)
            lo = max(0, j)
            hi = min(size, j + n)
            if lo >= hi:
                continue

            # Try candidate values
            candidates = [0.0, aj * 0.3, aj * 0.6, aj * 1.0, aj * 1.4, aj * 1.8, aj * 2.5]
            typical = norm / n
            candidates += [typical * 0.5, typical, typical * 1.5, typical * 3.0]
            candidates = sorted(set([max(0.0, min(c, norm * 0.5)) for c in candidates]))

            best_x = aj
            best_obj = fast_autocorr(a).max()

            for x in candidates:
                if abs(x - aj) < 1e-12:
                    continue
                new_sum = s_others + x
                if new_sum < 1e-15:
                    continue

                # Update autocorrelation
                # (a*a)[k] changes by 2*(x-aj)*a[k-j] for k in [j, j+n-1]
                # Except at k=2j where we need x^2-aj^2 instead of 2*(x-aj)*a[j]
                new_conv = conv.copy()
                delta = x - aj
                new_conv[lo:hi] += 2.0 * delta * a[lo - j:hi - j]
                if 2 * j < size:
                    # Correction: we added 2*delta*a[j] but should add x^2-aj^2
                    # Difference: x^2-aj^2 - 2*delta*a[j] = (x-aj)^2
                    new_conv[2 * j] += delta * delta

                scale = norm / new_sum
                new_max = new_conv.max() * scale * scale

                if new_max < best_obj:
                    best_obj = new_max
                    best_x = x

            if abs(best_x - aj) > 1e-12:
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    # Verify with full computation
                    new_score = fast_autocorr(a).max()
                    if new_score < best_score - 1e-13:
                        best_score = new_score
                        best_a = a.copy()
                        conv = new_score * np.ones_like(conv)  # Invalidate cached conv
                        conv = fast_autocorr(a)
                        improved = True
                    else:
                        # Revert
                        a = best_a.copy()
                        conv = fast_autocorr(a)

        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

def try_update(best_seq, best_score, a, label, t_global):

    n_cur = len(best_seq)
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 20:
        print(f"\n=== Phase 3: Multi-subgradient n={n_cur}, budget={t3:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = multi_subgradient(a_norm, t3, verbose=True, seed=1)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"subg_n{n_cur}", t_global)

    # === Phase 4: LP Nesterov round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.18, 250.0)
    if t4 > 30:
        print(f"\n=== Phase 4: LP Nesterov r2 n={n_cur}, budget={t4:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True, seed=999)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"lp2_n{n_cur}", t_global)

    # === Phase 5: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 15:
        print(f"\n=== Phase 5: CD n={n_cur}, budget={t5:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = coordinate_descent_fast(a_norm, t5, verbose=True, seed=5)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"cd_n{n_cur}", t_global)

    # === Phase 6: Dual-guided polish again ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t6 = min(remaining * 0.10, 100.0)
    if t6 > 20:
        print(f"\n=== Phase 6: Dual polish 2 n={n_cur} ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = dual_guided_polish(a_norm, t6, solver, verbose=False)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"dual2_n{n_cur}", t_global)

    # === Phase 7: Perturbation restarts ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    n_restarts = 2
    t7_total = min(remaining * 0.30, 200.0)
    t7_each = t7_total / n_restarts
    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        t7 = min(t7_each, remaining - 20)
        if t7 < 20:
            break
        print(f"\n=== Phase 7r{r}: Perturbation restart n={n_cur}, budget={t7:.0f}s ===")
        a_out = perturb_and_optimize(best_seq, t7, solver, rng, seed=500 + r)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"pert{r}_n{n_cur}", t_global)

    # === Phase 8: Final CD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 8: Final CD n={n_cur} ===")
        t8 = min(remaining * 0.6, 80.0)
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = coordinate_descent_fast(a_norm, t8, verbose=False, seed=77)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"cdFinal_n{n_cur}", t_global)

    # Final subgradient micro-polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 20:
        t9 = min(remaining * 0.5, 40.0)
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = multi_subgradient(a_norm, t9, verbose=False, seed=314)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"subgFinal_n{n_cur}", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
