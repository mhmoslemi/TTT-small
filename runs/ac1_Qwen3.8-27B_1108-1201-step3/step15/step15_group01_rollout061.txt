    for s in [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.2])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = s
                return s
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_weighted_grad(a, w):
    """Compute 2 * cross_correlation(w, a) where cross[m] = sum_j w[m+j]*a[j]."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    cross = irfft(W * np.conj(A), fft_size)[:n]
    return 2.0 * cross

def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v.copy()
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n + 1) > (css - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def normalize_to_sqrt2n(a):
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * np.sqrt(2.0 * len(a)) / len(a)
    return a * np.sqrt(2.0 * len(a)) / s

def evaluate_score(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a_list = [float(x) for x in a]
    try:
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def power_sum_optimize(a0, time_limit, p_schedule=None, verbose=True):
    """Power-sum gradient descent with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0  # normalize to sum=1 for optimization
    
    best_a = a.copy()
    best_c = fast_autocorr(a)
    best_max_c = best_c.max()
    
    if p_schedule is None:
        p_schedule = [4, 8, 16, 32, 64, 128, 256]
    
    t0 = time.monotonic()
    
    for p in p_schedule:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        # Nesterov setup
        y = a.copy()
        n_steps = min(2000, max(50, int(remaining * 2)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            c = fast_autocorr(y)
            c_nonneg = np.maximum(c, 1e-300)  # avoid 0^(p-1) issues
            S_p = np.sum(c_nonneg**p)
            if S_p < 1e-300:
                break
            PS_p = S_p ** (1.0/p)
            
            # Weights: w[k] = c[k]^(p-1) / PS_p^(p-1)
            w = c_nonneg**(p-1) / (PS_p**(p-1))
            
            # Gradient
            grad = compute_weighted_grad(y, w)
            
            # Step size (adaptive)
            step = 1.0 / (n * (1.0 + 0.5 * it / n_steps))
            
            a_new = y - step * grad
            a_new = project_simplex(a_new, 1.0)
            
            # Nesterov momentum
            alpha = 2.0 / (it + 2.0)
            y_new = (1.0 - alpha) * y + alpha * a_new
            y_new = project_simplex(y_new, 1.0)
            
            if it > 0:
                momentum = (it) / (it + 2.0)
                a_curr = y_new + momentum * (y_new - y)
                a_curr = project_simplex(a_curr, 1.0)
            else:
                a_curr = a_new.copy()
            
            y = y_new
            a = a_curr
            
            # Check improvement
            c_a = fast_autocorr(a)
            max_a = c_a.max()
            if max_a < best_max_c:
                best_max_c = max_a
                best_a = a.copy()
            
            c_y = fast_autocorr(y)
            if c_y.max() < best_max_c:
                best_max_c = c_y.max()
                best_a = y.copy()
            
            if verbose and it % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"    PS-p={p} step={it} ({elapsed:.0f}s): max_conv={best_max_c:.8f}")
    
    return best_a, best_max_c

def logsumexp_optimize(a0, time_limit, verbose=True):
    """Log-sum-exp smooth optimization with Nesterov acceleration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0
    
    best_a = a.copy()
    best_max_c = fast_autocorr(a).max()
    
    t0 = time.monotonic()
    temps = [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015, 0.0075, 0.004, 0.002, 0.001]
    
    for T in temps:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        y = a.copy()
        n_steps = min(1500, max(30, int(remaining * 2)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            c = fast_autocorr(y)
            c_max = c.max()
            exp_c = np.exp((c - c_max) / T)
            w = exp_c / exp_c.sum()
            
            grad = compute_weighted_grad(y, w)
            
            step = 1.0 / (n * (1.0 + 0.3 * it / n_steps))
            
            a_new = y - step * grad
            a_new = project_simplex(a_new, 1.0)
            
            alpha = 2.0 / (it + 2.0)
            y_new = (1.0 - alpha) * y + alpha * a_new
            y_new = project_simplex(y_new, 1.0)
            
            if it > 0:
                momentum = it / (it + 2.0)
                a_curr = y_new + momentum * (y_new - y)
                a_curr = project_simplex(a_curr, 1.0)
            else:
                a_curr = a_new.copy()
            
            y = y_new
            a = a_curr
            
            c_a = fast_autocorr(a)
            if c_a.max() < best_max_c:
                best_max_c = c_a.max()
                best_a = a.copy()
            
            if verbose and it % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LSE T={T:.4f} step={it} ({elapsed:.0f}s): max_conv={best_max_c:.8f}")
    
    return best_a, best_max_c

def direct_max_subgrad(a0, time_limit, verbose=True):
    """Direct max subgradient with Nesterov acceleration and averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0
    
    best_a = a.copy()
    best_max_c = fast_autocorr(a).max()
    
    t0 = time.monotonic()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(y)
        k_max = np.argmax(c)
        M = c[k_max]
        
        # Subgradient of max at y is the gradient of c[k_max]
        # grad[m] = 2 * a[k_max - m] if 0 <= k_max - m < n, else 0
        grad = np.zeros(n)
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        for m in range(lo, hi + 1):
            grad[m] = 2.0 * a[k_max - m]
        
        # Wait, I need to be more careful. The subgradient of max_k c_k(a) at a
        # where c_{k_max}(a) is the unique max is just dc_{k_max}/da.
        # But c_k is quadratic in a, so the gradient is 2 * a shifted.
        
        # Actually, let me reconsider. c[k] = sum_i a[i]*a[k-i].
        # dc[k]/da[m] = 2*a[k-m] if 0 <= k-m < n, else 0.
        # But this is the gradient of c[k] at the current a.
        
        # For the subgradient of max, we use the gradient of the active constraint.
        # If multiple k are active (tie), we could average their gradients.
        
        # Step size
        step = 1.0 / (n * (1.0 + 0.5 * it / 1000))
        
        a_new = y - step * grad
        a_new = project_simplex(a_new, 1.0)
        
        alpha = 2.0 / (it + 2.0)
        y_new = (1.0 - alpha) * y + alpha * a_new
        y_new = project_simplex(y_new, 1.0)
        
        if it > 0:
            momentum = it / (it + 2.0)
            a_curr = y_new + momentum * (y_new - y)
            a_curr = project_simplex(a_curr, 1.0)
        else:
            a_curr = a_new.copy()
        
        y = y_new
        a = a_curr
        it += 1
        
        avg_a += a
        n_avg += 1
        
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            check_sum = avg_check.sum()
            if check_sum > 1e-15:
                avg_check = avg_check / check_sum
                c_avg = fast_autocorr(avg_check)
                if c_avg.max() < best_max_c:
                    best_max_c = c_avg.max()
                    best_a = avg_check.copy()
        
        if it % 200 == 0:
            c_a = fast_autocorr(a)
            if c_a.max() < best_max_c:
                best_max_c = c_a.max()
                best_a = a.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    SubGrad iter={it} ({elapsed:.0f}s): max_conv={best_max_c:.8f}")
    
    return best_a, best_max_c

def lp_best_response_full(a, M, solver):
    """Solve max sum(b) s.t. conv(a,b) <= M, b >= 0. Uses full constraints."""
    n = len(a)
    size = 2 * n - 1
    
    # Build the convolution matrix: T[k, i] = a[k-i] if 0 <= k-i < n
    # conv(a, b)[k] = sum_i a[i] * b[k-i] = sum_i T[k,i] * b[i]
    # where T[k,i] = a[k-i] if 0 <= k-i < n, else 0
    # Also: conv(a,b)[k] = sum_i a[i] * b[k-i], so T[k,i] = a[i] 
    # Wait, let me re-derive.
    # np.convolve(a, b)[k] = sum_i a[i] * b[k-i]
    # = sum_i a[i] * b[k-i] where b is only non-zero for 0 <= k-i < n
    # = sum_i a[i] * b[k-i] for i in [max(0, k-n+1), min(n-1, k)]
    # Let j = k-i, then i = k-j:
    # = sum_j a[k-j] * b[j] for j in [max(0, k-n+1), min(n-1, k)]
    # So conv(a,b)[k] = sum_j T[k,j] * b[j] where T[k,j] = a[k-j] if 0 <= k-j < n
    
    # Build T as a dense matrix (size x n)
    T = np.zeros((size, n))
    for k in range(size):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                T[k, j] = a[idx]
    
    # Actually, this is O(n * 2n) = O(n^2) which for n=1600 is 5M entries.
    # That's a 3199 x 1600 matrix, about 20MB. Should be fine.
    
    # Better way to build T:
    T = np.zeros((size, n))
    for j in range(n):
        # T[k, j] = a[k-j] for 0 <= k-j < n, i.e., j <= k < j+n
        T[j:j+n, j] = a
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = b_var.value
    if b_val is None:
        return None
    b_val = np.maximum(np.asarray(b_val, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def lp_nesterov_full(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov, using full constraint set."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    t0 = time.monotonic()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    it = 0
    stuck = 0
    
    while time.monotonic() - t0 < time_limit:
        c_y = fast_autocorr(y)
        M = c_y.max()
        if M < 1e-12:
            break
        
        b = lp_best_response_full(y, M, solver)
        if b is None:
            stuck += 1
            if stuck > 
50:
                break
            continue
        stuck = 0
        
        b_sum = b.sum()
        g = b * norm / b_sum
        
        alpha = 2.0 / (it + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if it == 0:
            a_new = g.copy()
        else:
            momentum = it / (it + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        it += 1
        
        avg_a += a
        n_avg += 1
        
        if n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                c_avg = fast_autocorr(avg_check)
                if c_avg.max() < best_score:
                    best_score = c_avg.max()
                    best_a = avg_check.copy()
        
        c_y_new = fast_autocorr(y)
        if c_y_new.max() < best_score:
            best_score = c_y_new.max()
            best_a = y.copy()
        
        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Full iter={it} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def lp_best_response_active(a, M, solver, top_fraction=0.3):
    """LP with only top constraints for speed."""
    n = len(a)
    size = 2 * n - 1
    c = fast_autocorr(a)
    
    # Select top constraints
    threshold_idx = int(size * (1 - top_fraction))
    sorted_c = np.sort(c)[::-1]
    threshold = sorted_c[min(threshold_idx, size-1)]
    active = np.where(c >= threshold)[0]
    
    T = np.zeros((len(active), n))
    for idx, k in enumerate(active):
        for j in range(n):
            ii = k - j
            if 0 <= ii < n:
                T[idx, j] = a[ii]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = b_var.value
    if b_val is None:
        return None
    b_val = np.maximum(np.asarray(b_val, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def structured_perturb(a0, rng, alpha=0.1):
    """Perturb a sequence in a structured way based on its convolution profile."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return a
    
    c = fast_autocorr(a)
    c_max = c.max()
    
    # Find peak positions
    peak_threshold = 0.8 * c_max
    peak_positions = np.where(c >= peak_threshold)[0]
    
    # For each peak position k, the contribution is sum_i a[i]*a[k-i]
    # Reduce the pairs that contribute most to the peak
    for k in peak_positions[:min(20, len(peak_positions))]:
        # Find i that maximizes a[i]*a[k-i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        pairs = a[lo:hi+1] * a[k-hi:k-lo+1][::-1]
        if len(pairs) == 0:
            continue
        i_max = lo + np.argmax(pairs)
        j_max = k - i_max
        # Reduce a[i_max] and increase a[j_max] slightly (or redistribute)
        # Actually, reduce the larger one
        if a[i_max] > a[j_max]:
            transfer = a[i_max] * alpha * rng.uniform(0.1, 0.5)
            a[i_max] -= transfer
            # Distribute to random positions
            targets = rng.choice(n, size=3, replace=False)
            for t in targets:
                a[t] += transfer / 3
        else:
            transfer = a[j_max] * alpha * rng.uniform(0.1, 0.5)
            a[j_max] -= transfer
            targets = rng.choice(n, size=3, replace=False)
            for t in targets:
                a[t] += transfer / 3
    
    a = np.maximum(a, 0)
    s_new = a.sum()
    if s_new < 1e-15:
        return a0
    return a * s / s_new

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Uses power-sum gradient descent, LP best-response,
    log-sum-exp optimization, and structured perturbation to minimize
    2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1650.0)
    
    print(f"Time budget: {time_budget:.0f}s")
    
    # Load parent sequence
    best_seq = None
    best_score = float('inf')
    
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_score(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass
    
    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_score(a_norm)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_score(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    
    n_current = len(best_seq)
    solver = get_lp_solver()
    
    # === Phase 1: Power-sum optimization (25% time) ===
    remaining = time_budget - (time.monotonic() - t_start)
    t1 = remaining * 0.25
    print(f"\n=== Phase 1: Power-sum opt n={n_current} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_init = best_seq.copy()  # already normalized to sum=1
        a_out, _ = power_sum_optimize(a_init, t1, p_schedule=[4, 8, 16, 32, 64, 128, 256], verbose=True)
        update_best(a_out, "power_sum")
    
    # === Phase 2: LP Nesterov full constraints (35% time) ===
    remaining = time_budget - (time.monotonic() - t_start)
    t2 = remaining * 0.35
    print(f"\n=== Phase 2: LP Nesterov full n={n_current} (budget={t2:.0f}s) ===")
    if t2 > 60 and solver is not None:
        a_init = best_seq * np.sqrt(2.0 * n_current) / best_seq.sum()
        a_out, _ = lp_nesterov_full(a_init, t2, solver, verbose=True)
        update_best(a_out, "lp_full")
    
    # === Phase 3: Log-sum-exp optimization (20
