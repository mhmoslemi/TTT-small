def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n + 1) > (css - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    t0 = time.monotonic()
    size = 2 * n - 1

    T = 50.0

    a_curr = a.copy()
    it = 0
    steps_since_improve = 0

        if it % 5 == 0:

            c_max = c.max()
            if c_max > best_score * 1.05:  # If getting worse, cool faster
                T = max(min_T, T * 0.9)
            # If improving, keep temperature
        else:

            c_max = c.max()
            exp_c = np.exp((c - c_max) / T)
            w = exp_c / exp_c.sum()

        # Compute gradient

def smooth_optimize_v3(a0, time_limit, verbose=True):
    """Smooth optimization with multiple temperature restarts."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Multiple temperature phases with restarts
    temps = [30.0, 15.0, 8.0, 4.0, 2.0, 1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001]

    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        # Determine steps for this phase
        n_steps = max(50, min(800, int(remaining * 2)))

        y = a.copy()
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            c = fast_autocorr(y)
            c_max = c.max()
            exp_c = np.exp((c - c_max) / T)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            cross = irfft(W * np.conj(A), fft_size)[:n]
            grad = 2.0 * cross

            # Step size with annealing
            step = norm / (n * (T + 0.05))
            step *= (1.0 - 0.3 * it / n_steps)
            step = max(step, norm / (n * 50.0))

            a_new = a - step * grad
            a_new = project_simplex(a_new, norm)

            # Nesterov
            alpha = 2.0 / (it + 2.0)
            y_new = (1.0 - alpha) * y + alpha * a_new
            y_new = project_simplex(y_new, norm)

            if it < 5:
                a_curr = a_new.copy()
            else:
                momentum = (it - 1.0 + 1.0) / (it + 2.0)
                a_curr = y_new + momentum * (y_new - y)
                a_curr = project_simplex(a_curr, norm)

            y = y_new
            a = a_curr

            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

            score_y = fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    SmoothV3 T={T:.3f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def pair_cd_peak_aware(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent that targets peak-contributing positions."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0

    while time.monotonic() - t0 < time_limit and no_improve < 5:
        n_passes += 1
        pass_improved = False

        # Identify peak positions
        c_max = conv.max()
        peak_thresh = 0.9 * c_max
        peak_positions = np.where(conv >= peak_thresh)[0]

        # For each peak position k, the pairs (i, k-i) contribute
        # Build a priority list of positions that contribute to peaks
        contrib_score = np.zeros(n)
        for k in peak_positions:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    contrib_score[i] += a[i] * a[j]
                j2 = i - k
                if 0 <= j2 < n:
                    contrib_score[i] += a[i] * a[j2]

        # Process positions in order of contribution
        order = np.argsort(-contrib_score)

        for idx in order:
            if time.monotonic() - t0 > time_limit:
                break
            i = idx
            if a[i] < norm * 1e-12:
                continue

            # Try moving mass from i to nearby positions
            # Use a window that adapts to the problem
            window = min(n // 4, 80)
            best_delta = 0.0
            best_new_score = conv.max()

            for offset in range(1, window + 1):
                for sign in [1, -1]:
                    j = (i + sign * offset) % n
                    if j == i:
                        continue

                    # Binary search for best amount to move
                    lo, hi = 0.0, a[i] * 0.5
                    best_d = 0.0
                    best_v = best_new_score

                    for _ in range(6):
                        mid = (lo + hi) / 2
                        if mid < 1e-10:
                            break
                        a_t = a.copy()
                        a_t[i] -= mid
                        a_t[j] += mid
                        s_t = a_t.sum()
                        if s_t < 1e-15:
                            break
                        a_t = a_t * norm / s_t
                        conv_t = fast_autocorr(a_t)
                        if conv_t.max() < best_v - 1e-12:
                            best_v = conv_t.max()
                            best_d = mid
                            lo = mid
                        else:
                            hi = mid

                    if best_d > 1e-10 and best_v < best_new_score - 1e-12:
                        best_delta = best_d
                        best_j = j
                        best_new_score = best_v

            if best_delta > 1e-10:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = a * norm / a.sum()
                conv = fast_autocorr(a)
                if conv.max() < best_score - 1e-12:
                    best_score = conv.max()
                    best_a = a.copy()
                    pass_improved = True

        if not pass_improved:
            no_improve += 1
        else:
            no_improve = 0

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD-Peak pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def perturbation_refine_cycle(a0, time_limit, rng, verbose=True):
    """Repeatedly perturb and refine the solution."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    best_a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = best_a.sum()
    if s < 1e-15:
        best_a = np.ones(n) * norm / n
        s = norm
    best_a = best_a * norm / s
    best_score = fast_autocorr(best_a).max()

    t0 = time.monotonic()
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        iteration += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        # Create perturbation
        # Mix of global and local perturbations
        scale = max(0.05, 0.3 * (1.0 - iteration / 50.0))
        
        if rng.random() < 0.3:
            # Global perturbation
            noise = rng.normal(0, scale * best_a.std(), n)
            a_pert = best_a + noise
        else:
            # Local perturbation: move mass between random pairs
            a_pert = best_a.copy()
            n_moves = max(5, n // 20)
            for _ in range(n_moves):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i != j:
                    d = rng.uniform(0.01, 0.1) * a_pert[i]
                    a_pert[i] -= d
                    a_pert[j] += d

        a_pert = np.maximum(a_pert, 0.0)
        s_p = a_pert.sum()
        if s_p < 1e-15:
            continue
        a_pert = a_pert * norm / s_p

        # Quick smooth optimization on perturbed version
        t_refine = min(remaining * 0.7, 30.0)
        a_out, _ = smooth_optimize_v3(a_pert, t_refine, verbose=False)

        score_out = fast_autocorr(a_out).max()
        if score_out < best_score - 1e-10:
            best_score = score_out
            best_a = a_out.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    PerturbRefine iter={iteration} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score
