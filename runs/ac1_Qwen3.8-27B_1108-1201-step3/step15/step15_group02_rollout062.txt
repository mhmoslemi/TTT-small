    # Let me use the following identity:
    # sum_j a[j] * w[i+j] = [a * w_flip][i + n - 1]
    # where w_flip[k] = w[n-1-k] (reversal of w, but w has length 2n-1, not n)
    
    # Hmm, w has length 2n-1. Let me define w_flip[k] = w[2n-2-k] for k=0,...,2n-2.
    # Then (a * w_flip)[k] = sum_j a[j] * w_flip[k-j] = sum_j a[j] * w[2n-2-k+j]
    # We want sum_j a[j] * w[i+j]. So we need 2n-2-k+j = i+j → k = 2n-2-i.
    # So g[i] = (a * w_flip)[2n-2-i]
    
    # But (a * w_flip) has length n + (2n-1) - 1 = 3n-2.
    # And 2n-2-i ranges from 2n-2-(n-1) = n-1 to 2n-2-0 = 2n-2.
    
    # So g[i] = conv_full(a, w_flip)[2n-2-i] for i=0,...,n-1
    
    # Let me just compute this with FFT:
    A_full = rfft(a, fft_size)
    w_flip = w[::-1]
    W_flip = rfft(w_flip, fft_size)
    conv_aw = irfft(A_full * W_flip, fft_size)
    
    # g[i] = conv_aw[2n-2-i]
    # So g = conv_aw[2n-2 : n-2 : -1] = conv_aw[n-2 : 2n-1][::-1]
    g = conv_aw[n-1 : 2*n-1][::-1]  # indices n-1, n, ..., 2n-2, reversed
    
    # Wait, let me recheck. 2n-2-i for i=0 gives 2n-2, for i=n-1 gives n-1.
    # So g[0] = conv_aw[2n-2], g[1] = conv_aw[2n-3], ..., g[n-1] = conv_aw[n-1]
    # g = conv_aw[2n-2 : n-1 : -1] 
    # In numpy: conv_aw[n-1:2n-1][::-1] gives conv_aw[2n-2], conv_aw[2n-3], ..., conv_aw[n-1]
    # Yes! That's g[0], g[1], ..., g[n-1]. ✓
    
    return 2.0 * g, conv, w

def lse_adam_v2(a0, time_limit, verbose=True):
    """LSE-Adam optimizer with adaptive temperature."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_adam = 0
    
    # Adaptive temperature: start high, decrease based on improvement
    T = 10.0 / best_score if best_score > 0 else 1.0
    T_min = 0.01 / best_score if best_score > 0 else 0.001
    T_max = 50.0 / best_score if best_score > 0 else 5.0
    T_decay = 0.995
    
    while time.monotonic() - t0 < time_limit:
        t_adam += 1
        
        grad, conv, w = lse_gradient(a, T)
        
        # Check effective support of w
        entropy = -(w * np.log(w + 1e-30)).sum()
        eff_support = np.exp(entropy)
        
        # Adjust temperature to maintain effective support around 20-50
        if eff_support < 15:
            T *= 1.3
        elif eff_support > 100:
            T *= 0.7
        T = np.clip(T, T_min, T_max)
        T *= T_decay  # slow decay
        
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t_adam)
        v_hat = v / (1.0 - beta2 ** t_adam)
        
        # Adaptive learning rate
        lr = 1.0 / (np.sqrt(t_adam + 1) * np.sqrt(n) * 0.1)
        # Backtracking line search
        alpha = lr
        best_a_trial = a.copy()
        best_score_trial = best_score
        
        for ls in range(10):
            a_trial = a - alpha * m_hat / (np.sqrt(v_hat) + eps)
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                break
            a_trial = a_trial * norm / s_trial
            sc_trial = fast_conv_self(a_trial).max()
            if sc_trial < best_score_trial - 1e-10:
                best_score_trial = sc_trial
                best_a_trial = a_trial.copy()
                a = a_trial
                break
            else:
                alpha *= 0.5
        
        if t_adam % 10 == 0:
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                if verbose and t_adam % 100 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"    LSE2 t={t_adam} ({elapsed:.0f}s): M={best_score:.6f}, T={T:.4f}")
    
    return best_a, best_score

def topk_subgradient_update(a, k, lr):
    """
    Update using top-k subgradient of max(conv(a,a)).
    Moves mass from positions that contribute most to the top-k convolution values.
    """
    n = len(a)
    norm = np.sqrt(2.0 * n)
    conv = fast_conv_self(a)
    
    # Get top-k indices
    top_k_idx = np.argsort(conv)[-k:]
    top_k_weights = np.zeros(2 * n - 1)
    top_k_weights[top_k_idx] = 1.0 / k
    
    # Compute gradient contribution
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(top_k_weights, fft_size)
    
    # grad[i] = 2 * sum_j top_k_weights[i+j] * a[j]
    w_flip = top_k_weights[::-1]
    W_flip = rfft(w_flip, fft_size)
    conv_aw = irfft(A * W_flip, fft_size)
    grad = 2.0 * conv_aw[n-1:2*n-1][::-1]
    
    # Multiplicative update
    a_new = a * np.exp(-lr * grad / (grad.max() + 1e-30))
    a_new = np.maximum(a_new, 1e-15)
    a_new = a_new * norm / a_new.sum()
    return a_new, fast_conv_self(a_new).max()

def vegas_v2(a0, time_limit, verbose=True):
    """Vegas multiplicative updates with adaptive parameters."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    
    eta = 5.0 / best_score if best_score > 0 else 5.0
    min_eta = 0.01 / best_score if best_score > 0 else 0.001
    max_eta = 20.0 / best_score if best_score > 0 else 20.0
    
    stall_count = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        M = conv.max()
        
        # Pressure: how much each above-average convolution value "pushes"
        # Use a soft threshold
        threshold = M * 0.85
        w = np.maximum(conv - threshold, 0.0)
        ww = w / (w.sum() + 1e-30)
        
        W = rfft(ww, fft_size)
        A = rfft(a, fft_size)
        
        # grad[i] = 2 * sum_j ww[i+j] * a[j]
        w_flip = ww[::-1]
        W_flip = rfft(w_flip, fft_size)
        conv_aw = irfft(A * W_flip, fft_size)
        grad = 2.0 * conv_aw[n-1:2*n-1][::-1]
        
        # Multiplicative update
        a = a * np.exp(-eta * grad / (M + 1e-30))
        a = np.maximum(a, 1e-15)
        a = a * norm / a.sum()
        
        sc = fast_conv_self(a).max()
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 50:
                # Reduce learning rate
                eta *= 0.8
                if eta < min_eta:
                    eta = min_eta
                    # Random perturbation
                    n_pert = max(5, n // 50)
                    idx = np.random.randint(0, n, n_pert)
                    a[idx] *= (1.0 + 0.1 * np.random.randn(n_pert))
                    a = np.maximum(a, 0.0)
                    a = a * norm / a.sum()
                    stall_count = 0
        
        eta = np.clip(eta, min_eta, max_eta)
        eta *= 0.9995  # very slow decay
        
        if verbose and stall_count % 200 == 0 and stall_count > 0:
            elapsed = time.monotonic() - t0
            print(f"    Vegas2 ({elapsed:.0f}s): M={best_score:.6f}, eta={eta:.4f}")
    
    return best_a, best_score

def pair_cd_fast(a0, time_limit, verbose=True):
    """Fast pair coordinate descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    
    rng = np.random.default_rng(12345)
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue
            
            # Try moving mass from i to a few random j's
            # and to j's that reduce the max convolution
            j_candidates = set(rng.integers(n, size=5))
            # Add j's near the peak of convolution
            peak_idx = np.argmax(conv)
            j_candidates.add(int((peak_idx - i + n) % n))
            j_candidates.discard(i)
            
            best_delta = 0.0
            best_v = cur_max
            best_j = -1
            
            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]
                
                # Compute the change in convolution
                # c_new[k] = c_old[k] + d*(a[k-j mod stuff] + a[k-(j) ... ])
                # Simplified: g[k] = change in c[k] when moving d from i to j
                g = np.zeros(size)
                
                # Contribution from a[j] increasing by d:
                # c[k] += d * a[k-j] for valid k-j, and d * a[k-(j)] ... 
                # Actually: c[k] = sum_m a[m]*a[k-m]
                # If a[j] += d: c[k] += d*a[k-j] + d*a[k-j] = 2d*a[k-j] (when k-j valid)
                # But if k-j = j, i.e., k=2j, then it's d*a[j]+d*a[j]+d*d... 
                # Let me be more careful:
                # c_new = conv(a_new) where a_new[j] = a[j]+d, a_new[i] = a[i]-d
                # c_new[k] = c_old[k] + d*(a[k-j] + a[k-j]) - d*(a[k-i] + a[k-i]) + d*corrections
                # Wait: c[k] = sum_m a[m]*a[k-m]
                # delta_c[k] from a[j]+=d: d*a[k-j] (m=j term) + d*a[k-j] (k-m=j term) = 2d*a[k-j]
                #   BUT when m=j AND k-m=j (i.e., k=2j): the term is a[j]*a[j], change is (a[j]+d)^2 - a[j]^2 = 2*a[j]*d + d^2
                #   So for k=2j: delta = 2*a[j]*d + d^2 (not 2*a[j]*d)
                #   Extra: +d^2
                # Similarly from a[i]-=d: -2*d*a[k-i], and for k=2i: extra -d^2 + ... 
                #   (a[i]-d)^2 - a[i]^2 = -2*a[i]*d + d^2
                #   So delta from i: -2*d*a[k-i] + d^2 for k=2i (extra -d^2 from the -2d*a[k-i] ... 
                #   Actually: -2*a[i]*d + d^2 vs -2*d*a[i], extra is +d^2
            
                # Let me just build g properly:
                # g[k] = delta_c[k] / d (for small d, ignoring d^2 terms)
                # g[k] = 2*a[k-j] (if 0<=k-j<n) - 2*a[k-i] (if 0<=k-i<n)
                
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += 2.0 * a[0:hi-lo]
                
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += 2.0 * a[j - np.arange(lo2, hi2)]
                
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= 2.0 * a[0:hi3-lo3]
                
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= 2.0 * a[i - np.arange(lo4, hi4)]
                
                # Find best d
                d_max = ai * 0.9
                for d in np.linspace(0.05, d_max, 10):
                    new_conv = conv + d * g
                    # Add d^2 corrections
                    if 2*i < size:
                        new_conv[2*i] += d * d  # from (a[i]-d)^2
                    if 2*j < size:
                        new_conv[2*j] += d * d  # from (a[j]+d)^2
                    # Wait, I need to be more careful about the d^2 terms
                    # For k=2i: the term a[i]*a[i] becomes (a[i]-d)^2 = a[i]^2 - 2*a[i]*d + d^2
                    # Original contribution to c[2i] from this: a[i]^2
                    # The -2*a[i]*d is already in g[2i] (as part of -2*a[k-i] with k=2i, k-i=i)
                    # So the extra is +d^2
                    # For k=2j: (a[j]+d)^2 = a[j]^2 + 2*a[j]*d + d^2
                    # The +2*a[j]*d is in g[2j]
                    # Extra: +d^2
                    # But wait, there might be cross terms: a[i]*a[j] appears in c[i+j] and c[|i-j|]
                    # When we change a[i] by -d and a[j] by +d:
                    # c[i+j]: the term a[i]*a[j] becomes (a[i]-d)*(a[j]+d) = a[i]*a[j] + a[i]*d - a[j]*d - d^2
                    # So delta = d*(a[i]-a[j]) - d^2
                    # This is already captured by the linear terms in g (the +g from j and -g from i)
                    # The -d^2 is an extra term
                    # Similarly c[|i-j|]: same cross term
                    if i != j:
                        cross1 = i + j
                        if cross1 < size:
                            new_conv[cross1] -= d * d
                        cross2 = abs(i - j)
                        if cross2 < size and cross2 != cross1:
                            new_conv[cross2] -= d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_delta = d
                        best_j = j
            
            if best_j >= 0 and best_delta > 0 and best_v < best_score - 1e-12:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 8:
                # Perturb
                n_pert = max(3, n // 100)
                idx_p = rng.choice(n, size=n_pert, replace=False)
                a[idx_p] *= (1.0 + 0.1 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0.0)
                a = a * norm / a.sum()
            if no_improve >= 20:
                if verbose:
                    print(f"    PCD stopped: no improvement for 20 passes")
                break
        else:
            no_improve = 0
            
            if verbose and pass_improved:
                elapsed = time.monotonic() - t0
                if elapsed % 50 < 2:  # print occasionally
                    print(f"    PCD ({elapsed:.0f}s): M={best_score:.6f}")
    
    return best_a, best_score

def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)
    
    if choice == 0:
        # Raised cosine (Hann window)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif choice == 1:
        # Sine power
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        # Multiple peaks
        n_peaks = rng.integers(3, 12)
        centers = rng.uniform(0.05, 0.95, n_peaks)
        a = np.zeros(n)
        for c in centers:
            width = rng.uniform(0.03, 0.15)
            a += np.exp(-(x - c) ** 2 / (2 * width ** 2))
    elif choice == 3:
        # Discrete cosine
        k1 = rng.integers(1, 5)
        k2 = rng.integers(5, 15)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * k1 * x) + 0.2 * np.cos(2 * np.pi * k2 * x)
    elif choice == 4:
        # Tapered
        t = rng.uniform(0.3, 0.7)
        a = np.where(x < t, np.sin(np.pi * x / (2*t)), np.sin(np.pi * (1-x) / (2*(1-t))))
    elif choice == 5:
        # Exponential decay pair
        a = np.exp(-x / 0.1) + np.exp(-(1-x) / 0.1)
    elif choice == 6:
        # Flat with noise
        a = np.ones(n) + 0.3 * rng.standard_normal(n)
    elif choice == 7:
        # Sinc-like
        a = np.sinc(x - 0.5) ** 2 + 0.05
    elif choice == 8:
        # Power law taper
        p = rng.uniform(0.5, 2.0)
        a = (1 - np.abs(2*x - 1)) ** p
    else:
        # Random smooth
        a = np.exp(-rng.uniform(2, 8) * x) + np.exp(-rng.uniform(2, 8) * (1-x)) + 0.3
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1720.0)
    
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)
    
    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            old_best = best_score
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (was {old_best:.6f}, n={len(a)}, t={elapsed:.0f}s)")
    
    # Phase 1: LSE-Adam (fast, initial progress)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.20, 250.0)
    print(f"\n=== Phase 1: LSE-Adam v2 n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_out, _ = lse_adam_v2(best_seq, t1, verbose=True)
        check_and_update(a_out, f"lse1_n{n_base}")
    
    # Phase 2: Vegas v2
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.15, 180.0)
    print(f"\n=== Phase 2: Vegas v2 n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = vegas_v2(best_seq, t2, verbose=True)
        check_and_update(a_out, f"vegas_n{len(best_seq)}")
    
    # Phase 3: LSE-Adam round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 3: LSE-Adam round 2 n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = lse_adam_v2(best_seq, t3, verbose=False)
        check_and_update(a_out, f"lse2_n{len(best_seq)}")
    
    # Phase 4: Pair CD (polish)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.25, 300.0)
    print(f"\n=== Phase 4: Pair CD n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = pair_cd_fast(best_seq, t4, verbose=True)
        check_and_update(a_out, f"pcd1_n{len(best_seq)}")
    
    # Phase 5: Vegas round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 100.0)
    print(f"\n=== Phase 5: Vegas round 2 n={len(best_seq)} (budget={t5:.0f}s) ===")

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adaptive LSE (Log-Sum-Exp) gradient descent with entropy-based temperature control
2. Top-k subgradient multiplicative updates for focused refinement
3. Vegas-style pressure-based multiplicative updates
4. Pair coordinate descent with analytic convolution update for final polish
5. Multi-length exploration and diverse random restarts
All operations use FFT for O(n log n) convolution/gradient computation.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_conv_self(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size <<= 1
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_lse_gradient(a, T):
    """
    Compute gradient of LSE-smoothed max of self-convolution.
    f(a) = T * log(sum_k exp(c_k/T)), grad = 2 * sum_k w_k * a_{k-i}
    Returns (gradient, convolution, weights).
    """
    n = len(a)
    size = 2 * n - 1
    # Need FFT size >= 3n-2 for the cross-convolution
    fft_size = 1
    while fft_size < 3 * n - 2:
        fft_size <<= 1

    conv = fast_conv_self(a)
    max_c = conv.max()

    # LSE softmax weights
    shifted = (conv - max_c) / T
    shifted = np.maximum(shifted, -50.0)
    exp_c = np.exp(shifted)
    w = exp_c / exp_c.sum()

    # grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    # = 2 * (w_flip * a)[2n-2-i] where w_flip = w[::-1]
    w_flip = w[::-1]
    A = rfft(a, fft_size)
    Wf = rfft(w_flip, fft_size)
    conv_aw = irfft(A * Wf, fft_size)
    grad = 2.0 * conv_aw[n - 1:2 * n - 1][::-1]

    return grad, conv, w


def compute_topk_gradient(a, k):
    """
    Compute top-k subgradient: average subgradient over top-k convolution values.
    Returns (gradient, convolution, max_val).
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < 3 * n - 2:
        fft_size <<= 1

    conv = fast_conv_self(a)
    max_c = conv.max()

    # Top-k indicator
    if k >= size:
        w = np.ones(size) / size
    else:
        top_idx = np.argsort(conv)[-k:]
        w = np.zeros(size)
        w[top_idx] = 1.0 / k

    w_flip = w[::-1]
    A = rfft(a, fft_size)
    Wf = rfft(w_flip, fft_size)
    conv_aw = irfft(A * Wf, fft_size)
    grad = 2.0 * conv_aw[n - 1:2 * n - 1][::-1]

    return grad, conv, max_c


def lse_adam_optimize(a0, time_limit, verbose=True):
    """LSE-Adam with adaptive temperature."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s

    best_a = a.copy()
    best_M = fast_conv_self(a).max()
    t0 = time.monotonic()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    t_adam = 0

    # Adaptive temperature
    T = 5.0 / max(best_M, 1e-10)
    T_min = 0.02 / max(best_M, 1e-10)
    T_max = 30.0 / max(best_M, 1e-10)

    prev_score = best_M
    stall = 0

    while time.monotonic() - t0 < time_limit:
        t_adam += 1
        grad, conv, w = compute_lse_gradient(a, T)

        # Adapt temperature based on effective support
        entropy = -(w * np.log(w + 1e-30)).sum()
        eff_support = np.exp(entropy)
        if eff_support < 10:
            T *= 1.5
        elif eff_support < 30:
            T *= 1.1
        elif eff_support > 200:
            T *= 0.6
        elif eff_support > 80:
            T *= 0.9
        T = np.clip(T, T_min, T_max)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t_adam)
        v_hat = v / (1.0 - beta2 ** t_adam)

        # Learning rate with backtracking
        lr = 2.0 / (np.sqrt(t_adam + 1) * np.sqrt(n))
        alpha = lr
        improved = False
        for _ in range(8):
            a_trial = a - alpha * m_hat / (np.sqrt(v_hat) + eps)
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                break
            a_trial = a_trial * norm / s_t
            sc = fast_conv_self(a_trial).max()
            if sc < best_M - 1e-12:
                best_M = sc
                best_a = a_trial.copy()
                a = a_trial
                improved = True
                break
            alpha *= 0.5

        if not improved and t_adam % 50 == 0:
            # Small random perturbation
            n_p = max(3, n // 50)
            idx = np.random.randint(0, n, n_p)
            a[idx] *= (1.0 + 0.05 * np.random.randn(n_p))
            a = np.maximum(a, 0.0)
            a = a * norm / a.sum()

        if t_adam % 100 == 0:
            sc = fast_conv_self(a).max()
            if sc < best_M:
                best_M = sc
                best_a = a.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    LSE t={t_adam} ({elapsed:.0f}s): M={best_M:.7f}, T={T:.5f}")

    return best_a, best_M


def topk_vegas_optimize(a0, time_limit, k_schedule, verbose=True):
    """Top-k subgradient with multiplicative (Vegas) update."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s

    best_a = a.copy()
    best_M = fast_conv_self(a).max()
    t0 = time.monotonic()

    eta = 3.0 / max(best_M, 1e-10)
    eta_min = 0.01 / max(best_M, 1e-10)
    eta_max = 15.0 / max(best_M, 1e-10)
    stall = 0
    step = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        k = min(k_schedule[min(step // 50, len(k_schedule) - 1)], 2 * n - 1)
        grad, conv, M = compute_topk_gradient(a, k)

        # Multiplicative update
        a = a * np.exp(-eta * grad / (M + 1e-30))
        a = np.maximum(a, 1e-14)
        a = a * norm / a.sum()

        sc = fast_conv_self(a).max()
        if sc < best_M - 1e-12:
            best_M = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 30:
                eta *= 0.7
                if eta < eta_min:
                    # Perturb
                    n_p = max(5, n // 40)
                    idx = np.random.randint(0, n, n_p)
                    a[idx] *= (1.0 + 0.1 * np.random.randn(n_p))
                    a = np.maximum(a, 0.0)
                    a = a * norm / a.sum()
                    eta = max(eta, 1.0 / max(best_M, 1e-10))
                stall = 0

        eta = np.clip(eta, eta_min, eta_max)
        eta *= 0.9997

        if verbose and step % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    TopK-Vegas step={step} ({elapsed:.0f}s): M={best_M:.7f}, eta={eta:.4f}, k={k}")

    return best_a, best_M


def pressure_vegas_optimize(a0, time_limit, verbose=True):
    """Pressure-based multiplicative update (Vegas)."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s

    best_a = a.copy()
    best_M = fast_conv_self(a).max()
    t0 = time.monotonic()

    fft_size = 1
    while fft_size < 3 * n - 2:
        fft_size <<= 1

    eta = 5.0 / max(best_M, 1e-10)
    eta_min = 0.005 / max(best_M, 1e-10)
    eta_max = 20.0 / max(best_M, 1e-10)
    stall = 0
    step = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        conv = fast_conv_self(a)
        M = conv.max()

        # Pressure: soft threshold
        threshold = M * 0.80
        w = np.maximum(conv - threshold, 0.0)
        ww = w / (w.sum() + 1e-30)

        w_flip = ww[::-1]
        A = rfft(a, fft_size)
        Wf = rfft(w_flip, fft_size)
        conv_aw = irfft(A * Wf, fft_size)
        grad = 2.0 * conv_aw[n - 1:2 * n - 1][::-1]

        a = a * np.exp(-eta * grad / (M + 1e-30))
        a = np.maximum(a, 1e-14)
        a = a * norm / a.sum()

        sc = fast_conv_self(a).max()
        if sc < best_M - 1e-12:
            best_M = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 40:
                eta *= 0.75
                if eta < eta_min:
                    n_p = max(5, n // 40)
                    idx = np.random.randint(0, n, n_p)
                    a[idx] *= (1.0 + 0.1 * np.random.randn(n_p))
                    a = np.maximum(a, 0.0)
                    a = a * norm / a.sum()
                    eta = max(eta, 2.0 / max(best_M, 1e-10))
                stall = 0

        eta = np.clip(eta, eta_min, eta_max)
        eta *= 0.9995

        if verbose and step % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Vegas step={step} ({elapsed:.0f}s): M={best_M:.7f}, eta={eta:.4f}")

    return best_a, best_M


def pair_cd_optimize(a0, time_limit, verbose=True):
    """Fast pair coordinate descent with analytic convolution updates."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s

    best_a = a.copy()
    best_M = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    rng = np.random.default_rng(98765)
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            # Candidate j's: random + near convolution peaks
            j_candidates = list(set(rng.integers(n, size=4).tolist()))
            peak_positions = np.where(conv >= cur_max * 0.9)[0]
            for pk in peak_positions[:5]:
                j_candidates.append(int((pk - i) % n))
                j_candidates.append(int((pk + i) % n))
            j_candidates = [j for j in j_candidates if j != i][:6]

            best_delta = 0.0
            best_v = cur_max
            best_j = -1

            for j in j_candidates:
                aj = a[j]

                # Analytic change in convolution when moving d from i to j
                g = np.zeros(size)

                # a[j] += d contributes: 2*d*a[k-j] for valid k-j
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += 2.0 * a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += 2.0 * a[j - np.arange(lo2, hi2)]

                # a[i] -= d contributes: -2*d*a[k-i] for valid k-i
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= 2.0 * a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= 2.0 * a[i - np.arange(lo4, hi4)]

                # Try several d values
                d_max = ai * 0.85
                d_vals = np.linspace(0.02, d_max, 8)

                for d in d_vals:
                    new_conv = conv + d * g
                    # d^2 corrections
                    if 2 * i < size:
                        new_conv[2 * i] += d * d
                    if 2 * j < size:
                        new_conv[2 * j] += d * d
                    if i != j:
                        c1 = i + j
                        if c1 < size:
                            new_conv[c1] -= d * d
                        c2 = abs(i - j)
                        if c2 < size and c2 != c1:
                            new_conv[c2] -= d * d

                    if v < best_v:
                        best_v = v
                        best_delta = d
                        best_j = j
