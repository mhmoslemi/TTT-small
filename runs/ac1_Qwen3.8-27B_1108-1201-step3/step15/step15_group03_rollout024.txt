import numpy as np

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    return _solver


def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_toeplitz(a):
    """Build sparse Toeplitz matrix for convolution with a."""
    n = len(a)
    size = 2 * n - 1
    # T[k, i] = a[k-i] if 0 <= k-i < n, else 0
    # T has size (2n-1) x n, banded with bandwidth n
    rows = []
    cols = []
    vals = []
    for i in range(n):
        for k in range(i, i + n):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    from scipy.sparse import csr_matrix
    T = csr_matrix((vals, (rows, cols)), shape=(size, n))
    return T


def lp_solve(a, M, solver, active_indices=None):
    """Solve LP: max sum(b) s.t. T_a @ b <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    
    if active_indices is not None and 0 < len(active_indices) < size:
        T = build_toeplitz(a)
        constraints = [T[active_indices, :] @ b_var <= M]
    else:
        T = build_toeplitz(a)
        constraints = [T @ b_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    
    # Extract dual variables
    dual = None
    try:
        if active_indices is not None and 0 < len(active_indices) < size:
            dual_val = np.asarray(prob.dual_values[0], dtype=np.float64)
            dual = np.zeros(size)
            dual[active_indices] = dual_val
        else:
            dual_val = np.asarray(prob.dual_values[0], dtype=np.float64)
            dual = dual_val
    except Exception:
        dual = None
    
    return val, dual


def lse_gradient_descent(a0, time_limit, verbose=True):
    """LSE smooth gradient descent with adaptive temperature."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

    else:
        a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Multi-stage temperature schedule
    n_stages = 15
    temps = np.geomspace(100.0, 0.1, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        n_steps = max(50, min(500, int(remaining * 8)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -80.0)
            exp_c = np.exp(shifted)
            exp_sum = exp_c.sum()
            if exp_sum < 1e-30:
                break
            w = exp_c / exp_sum

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)

            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break

            # Adaptive step size
            lr = 1.0 / (T_val + 1.0) * np.sqrt(n / 10.0)
            lr = min(lr, 0.5)
            
            a_new = a - lr * grad / grad_norm * np.sqrt(max_c / n)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:

                best_a = a.copy()

            if dual_new is not None and dual_new.sum() > 1e-15:
                dual = np.maximum(dual_new, 0)
                dual = dual / dual.sum()

        # Compute dual-weighted gradient
        # grad = 2 * (IDFT(A * D) + IDFT(A * conj(D)))[0:n] + correction
        A = rfft(a, fft_size)
        D = rfft(dual, fft_size)
        
        g1 = irfft(np.conj(A) * D, fft_size)[:n]
        g2 = irfft(A * D, fft_size)[:n]
        grad = 2.0 * (g1 + g2)

        # Add the diagonal correction (a[i]^2 terms)
        # For lag k=2i, the gradient includes 2*a[i]*dual[2i]
        # This is already partially captured by the convolution, but let's add it
        for i in range(n):
            if 2 * i < size:
                grad[i] += 2.0 * a[i] * dual[2 * i]

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Adaptive step size with backtracking
        lr = lr_base / (1.0 + 0.01 * it)
        conv_before = fast_conv_self(a).max()
        a_trial = a - lr * grad / grad_norm * np.sqrt(conv_before / n)
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial > 1e-15:
            a_trial = a_trial * norm / s_trial
            conv_after = fast_conv_self(a_trial).max()
            if conv_after > conv_before * 1.5:
                lr *= 0.1
                a_trial = a - lr * grad / grad_norm * np.sqrt(conv_before / n)
                a_trial = np.maximum(a_trial, 0.0)
                s_trial = a_trial.sum()
                if s_trial > 1e-15:
                    a_trial = a_trial * norm / s_trial
        a = a_trial

        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DualGuided it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_shaving_cd(a0, time_limit, rng, verbose=True):
    """Targeted peak-shaving coordinate descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv_self(a).max()
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_passes = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        # Focus on top peaks
        peak_threshold = cur_max * 0.93
        peak_lags = np.where(conv >= peak_threshold)[0]
        
        if len(peak_lags) == 0:
            peak_lags = np.array([np.argmax(conv)])

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-9:
                continue

            # Find transfer candidates based on peak structure
            j_candidates = set()
            for kl in peak_lags[:5]:
                j1 = (kl - i) % n
                j2 = (kl + i) % n
                if j1 != i:
                    j_candidates.add(j1)
                if j2 != i:
                    j_candidates.add(j2)
            for _ in range(8):
                j_rand = int(rng.integers(n))
                if j_rand != i:
                    j_candidates.add(j_rand)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in list(j_candidates)[:12]:
                if j == i:
                    continue
                aj = a[j]

                # Compute the change in convolutions when moving delta from i to j
                # new_conv[k] = conv[k] + delta*(contrib_j[k] - contrib_i[k])
                # where contrib_x[k] = a[k-x] (if valid) + a[x-k] (if valid)
                
                # Vectorized computation of contribution vectors
                h_i = np.zeros(size)
                h_j = np.zeros(size)
                
                # Contribution of a[i] to conv[k]: a[k-i] for k-i in [0,n-1]
                lo = 0
                hi = n
                k_range = np.arange(lo, hi) + i
                valid = (k_range >= 0) & (k_range < size)
                h_i[k_range[valid]] += a[k_range[valid] - i]
                
                # Contribution of a[j] to conv[k]: a[k-j] for k-j in [0,n-1]
                lo = 0
                hi = n
                k_range = np.arange(lo, hi) + j
                valid = (k_range >= 0) & (k_range < size)
                h_j[k_range[valid]] += a[k_range[valid] - j]

                g = h_j - h_i  # change per unit delta

                # Try several deltas
                n_samples = 15
                deltas = np.linspace(0.01 * ai, 0.95 * ai, n_samples)
                for d in deltas:
                    # new_conv = conv + d * g + quadratic corrections
                    new_conv = conv + d * g
                    # Quadratic term at k=2i: -2*ai*d + d^2
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    # Quadratic term at k=2j: +2*aj*d + d^2
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    # Cross term: -d^2 at k where both i and j contribute (k = i and k = j if they overlap)
                    # Actually the h vectors already handle the linear terms correctly.
                    # The only quadratic terms are at k=2i (a[i]^2) and k=2j (a[j]^2)
                    
                    v = new_conv.max()
                    if v < best_val - 1e-10:
                        best_val = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 1e-10:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-10:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
                    # Update conv for next iteration
                    conv = fast_conv_self(a)
                    cur_max = conv.max()

        total_passes += 1
        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break
            # Perturb to escape
            n_perturb = max(5, n // 40)
            idx_perturb = rng.choice(n, size=n_perturb, replace=False)
            a[idx_perturb] *= (1.0 + 0.08 * rng.standard_normal(n_perturb))
            a = np.maximum(a, 0.0)
            s_new = a.sum()
            if s_new > 1e-15:
                a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and total_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakCD pass={total_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-4.0 * x) + 0.4 + 0.2 * np.sin(3 * np.pi * x)
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        a = np.sinc(3.0 * (x - 0.5)) ** 2 + 0.1
    elif choice == 9:
        a = np.exp(-2.0 * (x - 0.5) ** 2 / 0.2)
    elif choice == 10:
        # Multi-peak structure
        a = sum(np.exp(-20.0 * (x - p) ** 2) for p in [0.2, 0.5, 0.8])
    else:
        # Random smooth
        a = np.exp(-rng.random() * 3 * (x - 0.5) ** 2 / (0.1 + 0.2 * rng.random()))
    a = np.maximum(a, 0.0)
    a = a + 0.005 * rng.random(n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def eval_fn(a):
    """Evaluate a sequence using the provided evaluate_sequence function."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        if len(a_list) == 0:
            return float('inf')
        for val in a_list:
            if val < 0 or not np.isfinite(val):
                return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function that tries multiple strategies to minimize the score."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}, solver={solver}")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LSE gradient descent warmup
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 90.0)
    print(f"\n=== Phase 1: LSE GD n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 15:
        a_work = best_seq.copy()
        a_out, _ = lse_gradient_descent(a_work, t1, verbose=True)
        try_update(a_out, "lse_warmup")

    # Phase 2: LP Nesterov (main workhorse, longer run)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.30, 500.0)
    print(f"\n=== Phase 2: LP Nesterov n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, solver, verbose=True)
        try_update(a_out, "lp_nesterov1")

    # Phase 3: Dual-guided polish
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 3: Dual-guided polish n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = dual_guided_polish(best_seq, t3, solver, verbose=True)
        try_update(a_out, "dual_polish")

    # Phase 4: Peak shaving CD
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 4: Peak Shaving CD n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = peak_shaving_cd(best_seq, t4, rng, verbose=True)
        try_update(a_out, "peak_cd1")

    # Phase 5: LP Nesterov round 2 (with potentially different length)
    remaining = time_budget - (time.monotonic() - t_global)
    # Try a slightly longer length
    n_try = len(best_seq) + 200
    t5_total = min(remaining * 0.20, 250.0)
    print(f"\n=== Phase 5: LP Nesterov n={n_try} (budget={t5_total:.0f}s) ===")
    if t5_total > 30:
        init_long = interpolate_seq(best_seq, n_try)
        t5 = min(t5_total * 0.7, 180.0)
        a_out, _ = nesterov_lp_optimize(init_long, t5, solver, verbose=True)
        try_update(a_out, f"lp_n{n_try}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t5b = min(remaining * 0.4, 60.0)
            a_out, _ = peak_shaving_cd(a_out if a_out is not None else best_seq, t5b, rng, verbose=False)
            try_update(a_out, f"pcd_n{n_try}")

    # Phase 6: LP Nesterov on original length with extended time
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 6: LP Nesterov extended n={len(best_seq)} (budget={t6:.0f}s) ===")
    if t6 > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t6, solver, verbose=True)
        try_update(a_out, "lp_ext")

    # Phase 7: Final peak CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.5, 200.0)
    print(f"\n=== Phase 7: Final Peak CD n={len(best_seq)} (budget={t7:.0f}s) ===")
    if t7 > 15:
        a_out, _ = peak_shaving_cd(best_seq, t7, rng, verbose=True)
        try_update(a_out, "peak_cd_final")

    # Phase 8: Very short LSE + LP for final kick
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t8a = min(remaining * 0.3, 40.0)
        t8b = min(remaining - t8a - 10, 150.0)
        print(f"\n=== Phase 8: Final LSE+LP n={len(best_seq)} (budget={t8a+t8b:.0f}s) ===")
        a_out, _ = lse_gradient_descent(best_seq, t8a, verbose=False)
        try_update(a_out, "lse_final")
        if t8b > 20:
            a_out, _ = nesterov_lp_optimize(best_seq, t8b, solver, verbose=False)
            try_update(a_out, "lp_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
