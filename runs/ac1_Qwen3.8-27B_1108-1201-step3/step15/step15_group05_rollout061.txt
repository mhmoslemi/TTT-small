        M = R.max()
        
        # Find peaks that should be active
        new_tol = 0.003
        new_peaks = np.where(R >= M * (1 - new_tol))[0]
        A_new = np.union1d(A, new_peaks)
        
        # Cap active set size
        if len(A_new) > 150:
            A_new = np.argsort(R)[-100:]
        elif len(A_new) < 10:
            A_new = np.argsort(R)[-20:]
        
        A_changed = not np.array_equal(np.sort(A), np.sort(A_new))
        A = A_new
        
        # Update support
        S = np.where(a > max(1e-12, 1e-8 * a.max()))[0]
        
        # Reinitialize lambda if A changed
        if A_changed:
            # Try to preserve old lambda values where possible
            old_A_set = set(A.tolist())
            lam = np.ones(len(A)) / len(A)
        
        # Check KKT conditions
        R_check = fast_autocorr(a)
        score_val = 2.0 * n * R_check.max() / (a.sum()**2)
        if score_val < best_score:
            best_score = score_val
            best_a = a.copy()
        
        # Convergence check
        if it > 10:
            max_viol = R_check.max() - np.min(R_check[A]) if len(A) > 0 else 0
            # Check stationarity
            if len(S) > 0 and len(A) > 0:
                # 2 * sum_k lam_k * (Q_k a)[i] should be constant on S
                stat_vals = np.zeros(s_idx)
                for ki in range(m_idx):
                    k_val = A[ki]
                    diff = k_val - S
                    valid_s = (diff >= 0) & (diff < n)
                    stat_vals[valid_s] += lam[ki] * 2.0 * a[diff[valid_s]]
                if len(stat_vals) > 1:
                    stat_range = stat_vals.max() - stat_vals.min()
                    if stat_range / max(stat_vals.max(), 1e-15) < 1e-8 and max_viol < 1e-10:
                        if verbose:
                            print(f"    KKT converged at iter {it}, score={best_score:.10f}")
                        break
        
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    KKT iter={it} ({elapsed:.1f}s): |S|={len(S)}, |A|={len(A)}, "
                  f"step={step_norm:.2e}, score={best_score:.10f}")

    a = np.maximum(best_a, 0.0)
    s_final = a.sum()
    if s_final < 1e-15:
        return None
    a = a / s_final
    return a, compute_score(a)

def subgradient_refine(a0, time_limit=30.0, verbose=False):
    """Subgradient method for final refinement."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return None
    a = a / s
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    step = 0.01
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        R = fast_autocorr(a)
        M = R.max()
        k_star = int(np.argmax(R))
        
        # Subgradient: dR[k_star]/da[j] = a[k_star-j] + a[j-k_star] (with bounds)
        grad = np.zeros(n)
        for j in range(n):
            v = 0.0
            j1 = k_star - j
            if 0 <= j1 < n:
                v += a[j1]
            j2 = j - k_star
            if 0 <= j2 < n:
                v += a[j2]
            grad[j] = v
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm
        
        a_trial = a - step * grad
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            break
        a_trial = a_trial / s_trial
        
        score_trial = compute_score(a_trial)
        if score_trial < best_score - 1e-12:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            step = min(step * 1.02, 0.5)
        else:
            step *= 0.95
            if step < 1e-10:
                break
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate():
    """
    Main search: LP best-response + KKT Newton polishing + multi-length exploration.
    """
    t_global = time.monotonic()
    time_budget = 1750.0
    solver = get_solver()
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = compute_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = compute_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x)
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = compute_score(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")
    
    n_base = len(best_seq)
    print(f"Starting: n={n_base}, score={best_score:.8f}")
    
    # === Phase 1: LP Best-Response (main initialization) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 400.0)
    if t1 > 60:
        print(f"\n=== Phase 1: LP Nesterov n={n_base}, budget={t1:.0f}s ===")
        a_norm = best_seq
        a_out, _ = nesterov_lp(a_norm, t1, solver, verbose=True, seed=42)
        try_update(a_out, f"lp1_n{n_base}")
    
    # === Phase 2: KKT Newton Polish on base n ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.10, 80.0)
    if t2 > 10:
        print(f"\n=== Phase 2: KKT Newton n={n_cur}, budget={t2:.0f}s ===")
        print(f"  Before KKT: score={best_score:.10f}")
        result = kkt_newton_polish(best_seq, t2, verbose=True)

            try_update(a_out, f"kkt1_n{n_cur}")

    # === Phase 3: Subgradient refine ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t3 = min(remaining * 0.05, 40.0)
        print(f"\n=== Phase 3: Subgradient n={len(best_seq)}, budget={t3:.0f}s ===")
        result = subgradient_refine(best_seq, t3, verbose=False)

    # === Phase 4: Try other n values ===
    remaining = time_budget - (time.monotonic() - t_global)

    # Try a range of n values

    n_candidates.sort(key=lambda x: abs(x - n_base))
    
    n_per = max(20, (remaining - 60) / max(len(n_candidates), 1))

            try_update(a_out, f"subg_final_n{len(best_seq)}")

    remaining = time_budget - (time.monotonic() - t_global)
