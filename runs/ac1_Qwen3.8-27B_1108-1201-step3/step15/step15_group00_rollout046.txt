Since ||g||_∞ <= max(a) <= 1 (for normalized a), L <= 1.
So η_t ≈ sqrt(2*log(n)/t) ≈ sqrt(2*7.8/t) ≈ sqrt(15.6/t) for n=2400.

The previous code uses γ/sqrt(t) with γ=4, which gives η_t ≈ 4/sqrt(t). The theoretical value would be sqrt(15.6/t) ≈ 3.95/sqrt(t). So γ=4 is about right!

But we can do better with adaptive step sizing. Let's use a backtracking approach:
- Try η_t, compute the new point
- If the objective increased, halve η_t and try again
- This gives a practical upper bound on the step size

Actually, for the entropic mirror descent, the "objective increase" test is natural:
If f(x_{t+1}) > f(y_t) where y_t is the mirror-mapped point, reduce the step.

But this is expensive. A simpler approach: just use η_t = γ/sqrt(t) with γ slightly above the theoretical value, and add a "restart" mechanism when progress stalls.

def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum=1, ensure non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_exact_subgradient(a, ac, n_active=30, rel_tol=0.02):
    """
    Compute exact subgradient of max_k ac_k.
    Finds all lags k where ac_k >= (1 - rel_tol) * max(ac),
    then averages the individual gradients.
    """
    n = len(a)
    max_val = ac.max()
    if max_val < 1e-15:
        return np.zeros(n)
    threshold = max_val * (1.0 - rel_tol)
    active = np.where(ac >= threshold)[0]
    # Keep at most n_active (the largest ones)
    if len(active) > n_active:
        active = active[np.argsort(ac[active])[::-1][:n_active]]
    n_act = len(active)
    g = np.zeros(n)
    for k in active:
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        g[idx] += a[k - idx]
    g /= n_act
    return g


def compute_soft_subgradient(a, ac, top_k=40, beta=5.0):
    """Weighted soft subgradient focusing on top peaks."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        g[idx] += 2.0 * wi * a[k_int - idx]
    return g


def amd_refined(a0, time_limit, solver_gamma=4.0, verbose=True):
    """
    Mirror descent with exact subgradient, Nesterov momentum,
    and adaptive restarts.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    x = a.copy()
    x_prev = a.copy()
    gamma = solver_gamma
    stall = 0
    max_stall = 8000
    phase = 0  # 0=exact, 1=soft

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        mt = (t - 1.0) / (t + 2.0)
        y = x + mt * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)

        # Adaptive subgradient: start exact, switch to soft if stuck
        if phase == 0:
            g = compute_exact_subgradient(y, ac, n_active=35, rel_tol=0.015)
        else:
            g = compute_soft_subgradient(y, ac, top_k=50, beta=4.0)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive restart
        if stall > max_stall:
            stall = 0
            phase = 1 - phase
            # Perturb from best
            noise_scale = 0.005 + 0.005 * (stall % 3)
            pert = rng.normal(0, noise_scale, n)
            x_re = best_a * (1.0 + pert)
            x_re = np.maximum(x_re, 1e-10)
            x_re = x_re / x_re.sum()
            x = x_re
            x_prev = x_re
            # Adjust gamma
            if gamma < 8.0:
                gamma += 0.5
            elif gamma > 2.0:
                gamma -= 0.5
            if verbose:
                el = time.monotonic() - t0
                print("    AMD restart t=" + str(t) + " (" + format(el, ".1f") + "s) phase=" + str(phase) + " gamma=" + format(gamma, ".1f"))

        if verbose and t % 5000 == 0:
            el = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    AMD t=" + str(t) + " (" + format(el, ".1f") + "s): J=" + format(J, ".6f") + " stall=" + str(stall))

    return best_a, best_m

                    if st > 1e-15:

                            stall = 0
                        else:

                            a[j] -= amt
                            a = np.maximum(a, 0.0)

        y = y / y.sum()
        ac = fast_autocorr(y)
        g = compute_exact_subgradient(y, ac, n_active=25, rel_tol=0.02)
        eta = 3.0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        x_prev = x
        x = a_new
        M_new = fast_autocorr(x).max()
        if M_new < best_m:
            best_m = M_new
            best_a = x.copy()
        if verbose and t % 2000 == 0:
            el = time.monotonic() - t0
            print("    Polish t=" + str(t) + " (" + format(el, ".1f") + "s): J=" + format(2.0*n*best_m, ".6f"))

    return best_a


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator. Multi-phase optimization with constraint generation LP,
    exact subgradient AMD, and coordinate polish.
    """
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 150.0, 1650.0)

    solver = _get_lp_solver()
    print("Time budget: " + format(time_budget, ".0f") + "s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print("Parent: n=" + str(len(parent)) + " score=" + format(p_score, ".6f"))
            best_seq = parent / parent.sum()
            best_score = p_score
    except Exception:
        pass

    def _try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            el = time.monotonic() - t_global
            print("  *** NEW BEST [" + label + "]: " + format(sc, ".6f") + " (n=" + str(len(a)) + " t=" + format(el, ".0f") + "s)")

    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(n_base) + " score=" + format(best_score, ".6f"))

    n0 = len(best_seq)

    # === Phase 1: AMD with exact subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: AMD exact n=" + str(n0) + " ===")
    t1 = min(remaining * 0.30, 450.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_refined(a_in, t1, solver_gamma=4.0, verbose=True)
        _try_update(a_out, "amd1_n" + str(n0))

    # === Phase 2: LP constraint generation ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== Phase 2: LP CG n=" + str(n_cur) + " ===")
    t2 = min(remaining * 0.25, 350.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out = lp_fw_congen(a_in, t2, solver, verbose=True)
        _try_update(a_out, "lp_cg_n" + str(n_cur))

    # === Phase 3: AMD polish after LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== Phase 3: AMD polish n=" + str(n_cur) + " ===")
    t3 = min(remaining * 0.10, 120.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_refined(a_in, t3, solver_gamma=3.0, verbose=False)
        _try_update(a_out, "amd_polish_n" + str(n_cur))

    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== Phase 4: Coord n=" + str(n_cur) + " ===")
    t4 = min(remaining * 0.12, 130.0)
    if t4 > 15:
        a_in = norm_sum1(best_seq)
        a_out = coord_polish_v2(a_in, t4, verbose=True, rng_seed=42)
        _try_update(a_out, "coord_n" + str(n_cur))

    # === Phase 5: Try longer length ===
    n1 = min(n_cur + 600, 3000)
    remaining = time_budget - (time.monotonic() - t_global)
    if n1 > n_cur and remaining > 120:
        print("\n=== Phase 5: Scale to n=" + str(n1) + " ===")
        a_long = interpolate_seq(best_seq, n1)
        t5a = min(remaining * 0.12, 100.0)
        if t5a > 20:
            a_out, _ = amd_refined(a_long, t5a, solver_gamma=3.5, verbose=True)
            _try_update(a_out, "amd_long_n" + str(n1))
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t5b = min(remaining * 0.10, 80.0)
            if t5b > 20:
                a_out = lp_fw_congen(a_long, t5b, solver, verbose=False)
                _try_update(a_out, "lp_long_n" + str(n1))

    # === Phase 6: Short AMD at best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== Phase 6: Short AMD n=" + str(n_cur) + " ===")
    t6 = min(remaining * 0.08, 80.0)
    if t6 > 15:
        a_in = norm_sum1(best_seq)
        a_out = short_amd_polish(a_in, t6, verbose=False)
        _try_update(a_out, "short_amd_n" + str(n_cur))

    # === Phase 7: Try shorter length (might be optimal) ===
    n2 = max(n_cur - 400, 1600)
    remaining = time_budget - (time.monotonic() - t_global)
    if n2 < n_cur and remaining > 80:
        print("\n=== Phase 7: Try n=" + str(n2) + " ===")
        a_short = interpolate_seq(best_seq, n2)
        t7 = min(remaining * 0.15, 120.0)
        if t7 > 20:
            a_out, _ = amd_refined(a_short, t7, solver_gamma=3.0, verbose=False)
            _try_update(a_out, "amd_short_n" + str(n2))

    # === Phase 8: Final LP at best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if n_cur <= 2500 and remaining > 50:
        print("\n=== Phase 8: Final LP n=" + str(n_cur) + " ===")
        t8 = min(remaining * 0.30, 180.0)
        if t8 > 20:
            a_in = norm_sum1(best_seq)
            a_out = lp_fw_congen(a_in, t8, solver, verbose=True)
            _try_update(a_out, "lp_final_n" + str(n_cur))

    # === Phase 9: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== Phase 9: Final coord n=" + str(n_cur) + " ===")
    t9 = min(remaining * 0.50, 150.0)
    if t9 > 15:
        a_in = norm_sum1(best_seq)
        a_out = coord_polish_v2(a_in, t9, verbose=True, rng_seed=999)
        _try_update(a_out, "coord_final_n" + str(n_cur))

    # === Phase 10: One more quick AMD with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_cur = len(best_seq)
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_refined(a_in, min(remaining * 0.5, 60.0), solver_gamma=5.0, verbose=False)
        _try_update(a_out, "amd_final_n" + str(n_cur))

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print("\nFINAL: score=" + format(best_score, ".6f") + " n=" + str(len(best_seq)) + " time=" + format(time.monotonic() - t_global, ".0f") + "s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
