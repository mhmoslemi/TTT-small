from scipy.fft import rfft, irfft

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print(f"  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver):
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp(a0, time_limit, solver, verbose=True):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = evaluate_safe(a)

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    patience = 0
    restart_count = 0
    momentum_t = 1.0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            patience += 1
            if patience > 50:
                break
            continue

        patience = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b / b_sum
        n_iters += 1

        alpha = 2.0 / (momentum_t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y

        if momentum_t <= 1.0:
            a_new = g.copy()
        else:
            momentum = (momentum_t - 1.0) / (momentum_t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a

        y = y_new
        a = a_new
        momentum_t += 1.0

        score_a = evaluate_safe(a)
        if np.isfinite(score_a) and score_a < best_score - 1e-9:
            best_score = score_a
            best_a = a.copy()
            patience = 0

        score_y = evaluate_safe(y)
        if np.isfinite(score_y) and score_y < best_score - 1e-9:
            best_score = score_y
            best_a = y.copy()
            patience = 0

        if patience > 100 and n_iters % 50 == 0:
            y = best_a.copy()
            a = best_a.copy()
            momentum_t = 1.0
            patience = 0
            restart_count += 1
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    NLP restart {restart_count} at iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

        if verbose and n_iters % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, restarts={restart_count}")

    return best_a, best_score


def flat_loss_grad(a):
    """Compute gradient of sum_k (conv(a,a)[k] - target)^2 using FFT."""
    n = len(a)
    size = 2 * n - 1
    conv = fast_autocorr(a)
    target = (a.sum() ** 2) / size
    d = 2.0 * (conv - target)

    a_rev = a[::-1]
    fft_size = 1 << (3 * n - 1).bit_length()
    D = rfft(d, fft_size)
    A = rfft(a, fft_size)
    AR = rfft(a_rev, fft_size)

    conv_da = irfft(D * A, fft_size)
    conv_dar = irfft(D * AR, fft_size)

    g = np.zeros(n)
    g[:] = conv_dar[n - 1:n - 1 + n] + conv_da[0:n]
    return g


def flat_loss_descent(a0, time_limit, verbose=True):
    """Gradient descent on the flat-autocorrelation loss with backtracking."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = evaluate_safe(a)
    t0 = time.monotonic()
    eta = 0.5 / n
    patience = 0
    n_iters = 0
    rng = np.random.default_rng(77)

    for it in range(5000000):
        if time.monotonic() - t0 > time_limit:
            break

        g = flat_loss_grad(a)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        # Backtracking line search
        a_new = a
        accepted = False
        for trial in range(15):
            a_try = a - eta * g
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 0.01:
                eta *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = evaluate_safe(a_try)
            if np.isfinite(sc_try) and sc_try < best_score + 1e-8:
                a_new = a_try
                accepted = True
                break
            eta *= 0.6

        if not accepted:
            # Try a very small step or perturb
            if eta < 1e-10:
                noise = rng.normal(0, a.max() * 0.01, n)
                a = np.maximum(a + noise, 0.0)
                a = a / a.sum()
                eta = 0.5 / n
                patience += 1
                if patience > 200:
                    break
            continue

        a = a_new
        n_iters += 1
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-10:
            best_score = sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FL iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, eta={eta:.2e}")

    return best_a, best_score


def pairwise_cd(a0, time_limit, window=40, verbose=True):
    """Try all pairwise mass transfers within a window."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = evaluate_safe(a)
    conv = fast_autocorr(a)
    max_conv = conv.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(555)
    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        if no_improve > 5:
            break

        pass_improved = False
        conv = fast_autocorr(a)
        max_val = conv.max()

        # Focus on positions that contribute to peak lags
        top_lags = np.argsort(conv)[-10:]
        hot_positions = set()
        for k in top_lags:
            k = int(k)
            for j in range(n):
                if 0 <= k - j < n:
                    hot_positions.add(j)

        hot_list = list(hot_positions)
        if len(hot_list) > 200:
            hot_list = list(rng.choice(hot_list, 200, replace=False))

        for i in hot_list:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-12:
                continue
            for offset in range(1, window + 1):
                for j in [(i + offset) % n, (i - offset) % n]:
                    if j == i:
                        continue
                    # Try transferring mass from i to j
                    for frac in [0.3, 0.6, 0.9]:
                        d = a[i] * frac
                        a[i] -= d
                        a[j] += d
                        s_new = a.sum()
                        if s_new < 0.01:
                            a[i] += d
                            a[j] -= d
                            continue
                        a_normed = a / s_new * s_new  # keep same sum
                        sc_new = evaluate_safe(a)
                        if np.isfinite(sc_new) and sc_new < best_score - 1e-11:
                            best_score = sc_new
                            best_a = a.copy()
                            pass_improved = True
                            no_improve = 0
                            break
                        else:
                            a[i] += d
                            a[j] -= d

        if not pass_improved:
            no_improve += 1
            # Periodic perturbation
            if no_improve % 3 == 0:
                noise = rng.normal(0, a.max() * 0.003, n)
                a = np.maximum(a + noise, 0.0)
                a = a / a.sum()

        if verbose and n_passes % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def micro_polish(a0, time_limit, verbose=True):
    """Very fine-grained local search: tiny mass transfers."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = evaluate_safe(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(31337)
    n_iters = 0

    # Find top contributing pairs
    conv = fast_autocorr(a)
    top_lags = np.argsort(conv)[-5:]

    while time.monotonic() - t0 < time_limit:
        n_iters += 1
        # Pick a random hot position and a random nearby target
        k = top_lags[rng.integers(len(top_lags))]
        k = int(k)
        valid_j = [j for j in range(n) if 0 <= k - j < n and a[j] > 1e-10]
        if not valid_j:
            continue
        i = valid_j[rng.integers(len(valid_j))]
        j = (i + rng.integers(1, 30)) % n

        # Try tiny transfer
        d = a[i] * rng.uniform(0.01, 0.1)
        a[i] -= d
        a[j] += d
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
        else:
            a[i] += d
            a[j] -= d

        # Occasionally recompute top lags
        if n_iters % 500 == 0:
            conv = fast_autocorr(a)
            top_lags = np.argsort(conv)[-5:]

        if verbose and n_iters % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-10:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 1: LP Nesterov (55% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.55
    print(f"\n=== Phase 1: LP Nesterov n={n} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = nesterov_lp(best_seq, t1, solver, verbose=True)
        try_update(a_out, "lp1")

    # Phase 2: Flat-loss gradient descent (20% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = remaining * 0.20
    print(f"\n=== Phase 2: Flat-Loss GD n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = flat_loss_descent(best_seq, t2, verbose=True)
        try_update(a_out, "flat_gd")

    # Phase 3: Another LP round (15% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = remaining * 0.15
    print(f"\n=== Phase 3: LP Round 2 n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = nesterov_lp(best_seq, t3, solver, verbose=True)
        try_update(a_out, "lp2")

    # Phase 4: Pairwise CD (7% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = remaining * 0.07
    print(f"\n=== Phase 4: Pairwise CD n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 10:
        a_out, _ = pairwise_cd(best_seq, t4, window=30, verbose=True)
        try_update(a_out, "cd")

    # Phase 5: Flat-loss GD round 2 (remaining time * 0.5)
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = remaining * 0.5
    print(f"\n=== Phase 5: Flat-Loss GD R2 n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 10:
        a_out, _ = flat_loss_descent(best_seq, t5, verbose=False)
        try_update(a_out, "flat_gd2")

    # Phase 6: Micro polish (remaining time)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        print(f"\n=== Phase 6: Micro Polish n={len(best_seq)} (budget={remaining:.0f}s) ===")
        a_out, _ = micro_polish(best_seq, remaining * 0.9, verbose=False)
        try_update(a_out, "micro")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
