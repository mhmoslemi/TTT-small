def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def build_constraint_matrix(a, k_indices, n):
    """
    Build the constraint matrix for (a * b)[k] <= M.
    For fixed a, (a*b)[k] = sum_j a[k-j] * b[j] where 0 <= k-j < n.
    So the row for lag k has entries: row[j] = a[k-j] if 0 <= k-j < n, else 0.
    """
    size = len(k_indices)
    mat = np.zeros((size, n))
    for idx, k in enumerate(k_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            mat[idx, j] = a[k - j]
    return mat


def lp_best_response(a_fixed, M, active_set, solver):
    """
    Solve LP: max sum(b) s.t. (a_fixed * b)[k] <= M for k in active_set, b >= 0.
    Returns (b, duals) or (None, None).
    """
    n = len(a_fixed)
    k_indices = np.asarray(active_set)
    T = build_constraint_matrix(a_fixed, k_indices, n)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals

def subgradient_polish(a0, time_limit, verbose=False):
    """
    Subgradient method polishing: minimize max_k (a*a)[k] subject to sum(a)=S, a>=0.
    Uses the subgradient at the active peak(s).
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Adaptive step size
    step = 1.0 / (n ** 0.5)
    patience = 0
    max_patience = 500

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()
        k_star = int(np.argmax(conv))

        # Compute subgradient at k_star
        # d/d a[j] of (a*a)[k_star] = a[k_star - j] + a[j - k_star] (with bounds)
        grad = np.zeros(n)
        for j in range(n):
            v = 0.0
            j1 = k_star - j
            if 0 <= j1 < n:
                v += a[j1]
            j2 = j - k_star
            if 0 <= j2 < n:
                v += a[j2]
            grad[j] = v

        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        # Projected gradient step
        # We want to decrease M, so move in direction -grad
        # But we need to maintain sum(a) = norm and a >= 0
        # Projected step: a_new = proj_simplex(a - step * grad)
        a_trial = a - step * grad
        # Project onto {x >= 0, sum(x) = norm}
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-12:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            # Slowly increase step
            step = min(step * 1.01, 5.0)
        else:
            patience += 1
            step *= 0.95
            if step < 1e-10:
                break

        # Try a few random k_star for diversity
        if t % 50 == 0:
            conv = fast_autocorr(a)
            # Pick a random high-value lag
            threshold = conv.max() * 0.95
            candidates = np.where(conv >= threshold)[0]
            if len(candidates) > 1:
                k_rand = candidates[np.random.randint(len(candidates))]
                grad_r = np.zeros(n)
                for j in range(n):
                    v = 0.0
                    j1 = k_rand - j
                    if 0 <= j1 < n:
                        v += a[j1]
                    j2 = j - k_rand
                    if 0 <= j2 < n:
                        v += a[j2]
                    grad_r[j] = v
                grad_r_norm = np.linalg.norm(grad_r)
                if grad_r_norm > 1e-15:
                    grad_r = grad_r / grad_r_norm
                    a_trial2 = a - step * 0.5 * grad_r
                    a_trial2 = np.maximum(a_trial2, 0.0)
                    s2 = a_trial2.sum()
                    if s2 > 1e-15:
                        a_trial2 = a_trial2 * norm / s2
                        score2 = fast_autocorr(a_trial2).max()
                        if score2 < best_score - 1e-12:
                            a = a_trial2
                            best_score = score2
                            best_a = a_trial2.copy()
                            patience = 0

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad iter={t} ({elapsed:.0f}s): best={best_score:.10f}, step={step:.6f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, rng, verbose=False):
    """
    Systematic coordinate descent with exact 1D line search.
    For each coordinate j, find optimal a[j] given all others fixed.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_passes = 0
    no_improve = 0
    max_no_improve = 3

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]

            # Compute the "background" autocorrelation (contributions not involving a[j])
            # (a*a)[k] = bg[k] + 2*a[j]*contrib[k] + a[j]^2 * delta
            # where contrib[k] = a[k-j] (if valid) + a[j-k] (if valid) [excluding a[j] itself]
            # Actually: (a*a)[k] includes terms a[i]*a[k-i]. Terms involving a[j]:
            #   a[j]*a[k-j] (when k-j is valid, i.e., 0 <= k-j < n) -> gives a[k-j]
            #   a[i]*a[j] (when i = k-j, same as above)
            #   a[j]*a[j] (when j = k-j, i.e., k = 2j) -> gives a[j]^2
            # So: (a*a)[k] = bg[k] + 2*a[j]*c[k] + (a[j]^2 if k==2j else 0)
            # where c[k] = a[k-j] (if 0<=k-j<n and k-j != j) + a[j-k] (if 0<=j-k<n and j-k != j)

            # Precompute c[k] for all k
            c = np.zeros(size)
            for k in range(size):
                val = 0.0
                i1 = k - j
                if 0 <= i1 < n and i1 != j:
                    val += a[i1]
                i2 = j - k
                if 0 <= i2 < n and i2 != j:
                    val += a[i2]
                c[k] = val

            # bg[k] = (a*a)[k] - 2*aj*c[k] - (aj^2 if k==2j else 0)
            conv = fast_autocorr(a)
            bg = conv - 2.0 * aj * c
            if 2 * j < size:
                bg[2 * j] -= aj * aj

            # Now minimize max_k (bg[k] + 2*x*c[k] + (x^2 if k==2j else 0)) over x in [0, norm]
            # Subject to: sum(a) - aj + x = norm, so x = norm - (sum(a) - aj)
            # Wait, we're changing a[j] from aj to x, so new sum = sum(a) - aj + x
            # We need new sum = norm, so x = norm - (sum(a) - aj) = aj (no change in sum!)
            # 
            # Actually, since we normalize after each step, let's think of it differently:
            # We fix all a[i] for i != j, and optimize a[j] = x, then renormalize.
            # After renormalization, the function is homogeneous, so we can just minimize
            # max_k (bg[k] + 2*x*c[k] + x^2*delta_{k,2j}) / (sum others + x)^2 * (norm^2)
            # But this is complicated. Let's just do a simple grid search.

            # Simple approach: try a few values of x
            s_others = a.sum() - aj
            candidates_x = [0.0, aj * 0.5, aj * 0.75, aj * 1.0, aj * 1.25, aj * 1.5,
                           s_others * 0.1, s_others * 0.5, s_others]
            candidates_x = [x for x in candidates_x if 0 <= x <= norm]

            best_x = aj
            best_obj = float('inf')

            for x in candidates_x:
                new_sum = s_others + x
                if new_sum < 1e-15:
                    continue
                # Compute new autocorrelation
                new_conv = bg + 2.0 * x * c
                if 2 * j < size:
                    new_conv[2 * j] += x * x
                obj = new_conv.max() / (new_sum * new_sum) * (2 * n)
                # Actually the score is 2*n*max_conv/sum^2, and we want to minimize this
                # But since we'll renormalize, the score is 2*n*max_conv_at_norm
                # Let's just compute: after setting a[j]=x and normalizing to sum=norm:
                # a_scaled = a * (norm / new_sum)
                # conv_scaled = conv * (norm/new_sum)^2
                # score = 2*n*max(conv_scaled)/norm^2 = 2*n*max_conv/(new_sum^2)
                if obj < best_obj:
                    best_obj = obj
                    best_x = x

            # Also try bisection between best candidates
            if best_x != aj and abs(best_x - aj) > 1e-10:
                lo = min(aj, best_x)
                hi = max(aj, best_x)
                for _ in range(8):
                    mid = (lo + hi) / 2.0
                    new_sum = s_others + mid
                    if new_sum < 1e-15:
                        break
                    new_conv = bg + 2.0 * mid * c
                    if 2 * j < size:
                        new_conv[2 * j] += mid * mid
                    obj = 2.0 * n * new_conv.max() / (new_sum * new_sum)
                    aj_score = 2.0 * n * conv.max() / (a.sum() ** 2)
                    if obj < aj_score:
                        lo = mid
                    else:
                        hi = mid
                best_x = (lo + hi) / 2.0

            if best_x != aj and abs(best_x - aj) > 1e-12:
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_score = fast_autocorr(a).max()
                    if new_score < best_score - 1e-13:
                        best_score = new_score
                        best_a = a.copy()
                        improved = True
                    else:
                        a[j] = aj
                        a = a * norm / a.sum()

        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def projected_subgradient(a0, time_limit, verbose=False):
    """
    More aggressive subgradient method with multiple active peaks.
    Uses the top few peaks to compute a combined gradient.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_passed = 0
    max_patience = 2000
    step = 0.5 / n ** 0.3

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if n_passed > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()

        # Get top peaks
        n_peaks = max(3, min(10, n // 100))
        top_indices = np.argsort(conv)[-n_peaks:]
        # Weight by how close to maximum
        weights = np.exp(-5.0 * (M_curr - conv[top_indices]) / max(M_curr * 0.01, 1e-10))
        weights = weights / weights.sum()

        # Combined gradient
        grad = np.zeros(n)
        for w, k in zip(weights, top_indices):
            for j in range(n):
                v = 0.0
                j1 = k - j
                if 0 <= j1 < n:
                    v += a[j1]
                j2 = j - k
                if 0 <= j2 < n:
                    v += a[j2]
                grad[j] += w * v

        # Projected step
        a_trial = a - step * grad / (n ** 0.5)
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-13:
            n_passed = 0
            a = a_trial

def symmetrize(a):
    """Symmetrize the sequence: a = (a + reverse(a)) / 2."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    s = a_sym.sum()

def perturb_sequence(a, rng, scale=0.03):
    """Add multiplicative noise."""
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.1)
    a_pert = a * noise
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def interpolate_sequence(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss
