import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver

def fast_autocorr(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build the matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def lp_best_response(a, M, solver):
    """Solve max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration and adaptive restarts."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stuck_count = 0
    patience = 0
    restart_count = 0
    momentum_t = 1.0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            stuck_count += 1
            if stuck_count > 50:
                break
            continue

        stuck_count = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov step
        alpha = 2.0 / (momentum_t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if momentum_t < 2.0:
            a_new = g.copy()
        else:
            momentum = (momentum_t - 1.0) / (momentum_t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new
        momentum_t += 1.0

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            patience = 0
        else:
            patience += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            patience = 0

        # Restart if stuck
        if patience > 60 and n_iters % 30 == 0:
            y = best_a.copy()
            a = best_a.copy()
            momentum_t = 1.0
            patience = 0
            restart_count += 1
            if verbose and restart_count <= 10:
                elapsed = time.monotonic() - t0
                print(f"    NLP restart {restart_count} at iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, restarts={restart_count}")

    return best_a, best_score

def spectral_equalize(a, n_iters=50, verbose=False):
    """Gerchberg-Saxton style spectral equalization to flatten the convolution spectrum."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    S = a.sum()
    a_work = a.copy().astype(np.float64)
    best_a = a_work.copy()
    best_score = fast_autocorr(a_work).max()
    
    for it in range(n_iters):
        A = np.fft.rfft(a_work, fft_size)
        power = A.real**2 + A.imag**2
        # Target flat power spectrum
        target_power = power.mean()
        # New magnitude = sqrt(target_power), keep phase
        phase = np.angle(A)
        new_A = np.sqrt(target_power) * np.cos(phase) + 1j * np.sqrt(target_power) * np.sin(phase)
        # Preserve DC
        new_A[0] = A[0]
        a_new = np.fft.irfft(new_A, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new * S / s
        score = fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a_work = a_new
    
    return best_a

def mirror_descent(a0, time_limit, top_k=30, verbose=True):
    """Mirror descent with entropy-regularized subgradient on the simplex."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    eta = 1.0 / np.sqrt(n)
    n_iters = 0
    patience = 0
    rng = np.random.default_rng(12345)

    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(a)
        max_val = conv.max()

        k_actual = min(top_k, size)
        # Use weighted active set: weights proportional to (conv[k] / max_val)^10
        weights = np.zeros(size)
        active_mask = conv > max_val * 0.95
        if not active_mask.any():
            active_mask = np.zeros(size, dtype=bool)
            active_mask[np.argmax(conv)] = True
        weights[active_mask] = (conv[active_mask] / max_val) ** 5.0
        w_sum = weights.sum()
        if w_sum < 1e-15:
            break
        weights /= w_sum

        # Compute weighted subgradient
        grad = np.zeros(n)
        for k in range(size):
            if weights[k] < 1e-10:
                continue
            i_lo = max(0, int(k) - n + 1)
            i_hi = min(int(k), n - 1)
            if i_hi >= i_lo:
                j_lo = int(k) - i_hi
                j_hi = int(k) - i_lo
                grad[i_lo:i_hi + 1] += 2.0 * weights[k] * a[j_lo:j_hi + 1][::-1]
                # Handle the diagonal (k = 2i) case: the gradient is 2*a[i] not a[i]
                # Actually for the convolution c[k] = sum_j a[j]*a[k-j], the gradient w.r.t. a[i]
                # is 2*a[k-i] (when k != 2i) or 2*a[i] (when k=2i)
                # But since we're summing over all j, the term a[i] appears in c[k] 
                # when j=i (giving a[i]*a[k-i]) and when j=k-i (giving a[k-i]*a[i])
                # So the gradient is a[k-i] + a[k-i] = 2*a[k-i] for k != 2i
                # and 2*a[i] for k = 2i (which is the same formula since k-i = i)
                # So actually the formula 2*a[k-i] works for all cases!
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        # Mirror descent step with backtracking
        log_a = np.log(a + 1e-30)
        best_a_try = a.copy()
        best_conv_max = max_val
        for factor in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
            eta_try = eta * factor
            log_a_try = log_a - eta_try * grad
            a_try = np.exp(log_a_try - log_a_try.max())
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try * norm / s_try
            conv_try = fast_autocorr(a_try).max()
            if conv_try < best_conv_max:
                best_conv_max = conv_try
                best_a_try = a_try
                eta = max(eta_try, eta * 0.8)

        a = best_a_try
        n_iters += 1

        if best_conv_max < best_score - 1e-12:
            best_score = best_conv_max
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience % 50 == 0:
                eta *= 0.9

        if patience > 200:
            noise = rng.laplace(0, norm * 0.01, n)
            a_pert = a + noise
            a_pert = np.maximum(a_pert, 0)
            s_pert = a_pert.sum()
            if s_pert > 1e-15:
                a = a_pert * norm / s_pert
            patience = 0
            eta = 1.0 / np.sqrt(n)

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, eta={eta:.4f}")

    conv = fast_autocorr(a)
    best_score = conv.max()
    t0 = time.monotonic()
    size = 2 * n - 1
    rng = np.random.default_rng(999)
    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        if no_improve > 100:
            break
        
        conv = fast_autocorr(a)
        max_val = conv.max()
        
        # Compute gradient of max (subgradient using top active lags)
        top_k = min(20, size)
        active_idx = np.argsort(conv)[-top_k:]
        active_weights = np.zeros(size)
        for k in active_idx:
            active_weights[k] = (conv[k] / max_val) ** 10.0
        w_sum = active_weights.sum()
        if w_sum > 0:
            active_weights /= w_sum
        
        grad = np.zeros(n)
        for k in active_idx:
            if active_weights[k] < 1e-10:
                continue
            i_lo = max(0, int(k) - n + 1)
            i_hi = min(int(k), n - 1)
            if i_hi >= i_lo:
                j_lo = int(k) - i_hi
                j_hi = int(k) - i_lo
                grad[i_lo:i_hi + 1] += 2.0 * active_weights[k] * a[j_lo:j_hi + 1][::-1]
        
        # Project gradient to tangent space of simplex (subtract mean)
        grad_tangent = grad - grad.mean()
        
        # Try moving in -grad_tangent direction
        improved = False
        for step_scale in [1.0, 0.5, 0.25, 0.1, 0.05]:
            eta = step_scale * norm / (np.linalg.norm(grad_tangent) + 1e-15)
            a_try = a - eta * grad_tangent
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try * norm / s_try
            conv_try = fast_autocorr(a_try)
            max_try = conv_try.max()
            if max_try < best_score * (1 - 1e-10):
                best_score = max_try
                best_a = a_try.copy()
                a = a_try.copy()
                conv = conv_try
                improved = True
                break
        
        if not improved:
            no_improve += 1
            # Try random coordinate shifts
            for _ in range(50):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j or a[i] < 1e-10:
                    continue
                for frac in [0.1, 0.2, 0.3, 0.5]:
                    d = a[i] * frac
                    a[i] -= d
                    a[j] += d
                    s = a.sum()
                    a = a * norm / s
                    conv_new = fast_autocorr(a)
                    if conv_new.max() < best_score * (1 - 1e-10):
                        best_score = conv_new.max()
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[i] += d
                        a[j] -= d
                        a = a * norm / s
                if improved:
                    no_improve = 0
                    break
        else:
            no_improve = 0

        if verbose and n_passes % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def truncate_sequence(a, max_iterations=100):
    """Try to remove small elements to reduce n and improve score."""
    n = len(a)
    best_a = a.copy()
    best_score = 2 * n * fast_autocorr(a).max() / a.sum()**2
    a_work = a.copy()
    
    for _ in range(max_iterations):
        n_curr = len(a_work)
        if n_curr < 50:
            break
        # Find smallest element
        min_idx = np.argmin(a_work)
        if a_work[min_idx] < best_score * 1e-10:  # relatively small
            # Try removing it
            a_trial = np.delete(a_work, min_idx)
            if a_trial.sum() < 0.01:
                break
            n_trial = len(a_trial)
            score_trial = 2 * n_trial * fast_autocorr(a_trial).max() / a_trial.sum()**2
            if score_trial < best_score - 1e-10:
                best_score = score_trial
                best_a = a_trial.copy()
                a_work = a_trial.copy()
            else:
                break
        else:
            break
    
    return best_a, best_score
