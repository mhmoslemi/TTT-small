Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LSE-Adam gradient descent for initialization
2. Fast LP best-response (HiGHS) with Nesterov acceleration
3. Pair coordinate descent for polishing
4. Multi-length exploration

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_T(a):
    """Build convolution matrix T where T[k,i] = a[k-i] if 0<=k-i<n."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for k in range(n):
        T[k, :k+1] = a[k::-1]
    for k in range(n, size):
        lo = k - n + 1
        T[k, lo:n] = a[n-1:lo-1:-1]
    return T

def solve_lp(T, M, n):
    """Solve max sum(g) s.t. T@g <= M, g >= 0 using HiGHS."""
    c = -np.ones(n)
    A_ub = T
    b_ub = np.full(T.shape[0], M)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs',
                  options={'presolve': True})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def lse_adam(a0, time_limit, verbose=False):
    """LSE-Adam optimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t_adam = 0
    
    n_stages = 15
    temps = np.geomspace(50.0, 0.1, n_stages)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        n_steps = max(20, min(100, int(remaining * 8)))
        lr = 0.3 / np.sqrt(n)
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            t_adam += 1
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = np.maximum((conv - max_c) / T_val, -80.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            ca = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (ca + aw)
            
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_adam)
            v_hat = v / (1.0 - beta2 ** t_adam)
            
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    return best_a, best_score

def pair_cd(a0, time_limit, rng, verbose=False):
    """Pair coordinate descent for polishing."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue
            
            peak_lags = np.where(conv >= cur_max * 0.8)[0]
            j_candidates = set()
            for _ in range(6):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:8]:
                j_candidates.add(int((kl - i + n) % n))
            
            best_delta = 0.0
            best_v = cur_max
            best_j = -1
            
            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]
                
                # Compute change in conv when moving delta from i to j
                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                g[lo:hi] += a[0:hi-lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                g[lo3:hi3] -= a[0:hi3-lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]
                
                d_max = ai * 0.9
                n_samples = 15
                deltas = np.linspace(0.01, d_max, n_samples)
                
                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2*i] = conv[2*i] - 2*ai*d + d*d
                    if 2 * j < size:
                        new_conv[2*j] = conv[2*j] + 2*aj*d + d*d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_delta = d
                        best_j = j
            
            if best_j >= 0 and best_delta > 0 and best_v < best_score - 1e-10:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-10:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()
        
        if not pass_improved:

            if no_improve >= 8:

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LSE-Adam gradient descent for quick initialization
2. Full-constraint LP best-response (via HiGHS) with Nesterov acceleration
3. Pair coordinate descent for fine polishing
4. Multi-length exploration
Key improvement: much faster LP solves via scipy HiGHS (vs cvxpy), allowing
many more Nesterov iterations within the time budget, plus full constraint set.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_T(a):
    """Build convolution matrix T where (T@g)[k] = conv(a,g)[k]."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for k in range(n):
        T[k, :k + 1] = a[k::-1]
    for k in range(n, size):
        lo = k - n + 1
        T[k, lo:n] = a[n - 1:lo - 1:-1]
    return T


def solve_lp_best_response(T, M, n):
    """Solve max sum(g) s.t. T@g <= M, g >= 0 using HiGHS."""
    c = -np.ones(n)
    b_ub = np.full(T.shape[0], M)
    res = linprog(c, A_ub=T, b_ub=b_ub, method='highs',
                  options={'presolve': True})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def lse_adam(a0, time_limit, verbose=False):
    """LSE-Adam optimization for smooth approximation of max-conv."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t_adam = 0

    n_stages = 18
    temps = np.geomspace(60.0, 0.05, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        n_steps = max(15, min(120, int(remaining * 10)))
        lr = 0.4 / np.sqrt(n)

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            t_adam += 1
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = np.maximum((conv - max_c) / T_val, -80.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            ca = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (ca + aw)

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_adam)
            v_hat = v / (1.0 - beta2 ** t_adam)

            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose and si % 4 == 0:
            print(f"    LSE stage {si} T={T_val:.3f}: best={best_score:.8f}")

    return best_a, best_score


def lp_nesterov(a0, time_limit, verbose=True):
    """LP best-response with Nesterov acceleration, full constraint set."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    no_improve = 0
    it_count = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        T = build_T(y)
        g = solve_lp_best_response(T, M, n)

        if g is None or g.sum() < 1e-15:
            no_improve += 10
            if no_improve > 500:
                break
            continue

        it_count += 1
        g_norm = g * norm / g.sum()

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_norm
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g_norm.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score - 1e-13:
            best_score = sc_y
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score - 1e-13:
            best_score = sc_a
            best_a = a.copy()
            no_improve = 0

        if no_improve > 800:
            break

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M:.6f}, iters={it_count}")

    return best_a, best_score


def pair_cd(a0, time_limit, rng, verbose=False):
    """Pair coordinate descent for polishing."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            peak_lags = np.where(conv >= cur_max * 0.8)[0]
            j_candidates = set()
            for _ in range(6):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:8]:
                j_candidates.add(int((kl - i + n) % n))
                j_candidates.add(int((kl + i) % n))

            best_delta = 0.0
            best_v = cur_max
            best_j = -1

            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]

                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                d_max = ai * 0.9
                n_samples = 20
                deltas = np.linspace(0.005, d_max, n_samples)

                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0 and best_v < best_score - 1e-10:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-10:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 8:
                n_pert = max(5, n // 80)
                idx_p = rng.choice(n, size=n_pert, replace=False)
                a[idx_p] *= (1.0 + 0.12 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                if no_improve >= 15:
                    break
        else:
            no_improve = 0

        if verbose and total_improvements % 300 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD imp={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        f1 = rng.integers(1, 5)
        f2 = rng.integers(5, 15)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * f1 * x) + 0.2 * np.cos(2 * np.pi * f2 * x)
    elif choice == 3:
        c1 = rng.uniform(0.2, 0.5)
        c2 = rng.uniform(0.5, 0.8)
        a = np.exp(-30.0 * (x - c1) ** 2) + np.exp(-30.0 * (x - c2) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.5
    elif choice == 6:
        n_peaks = rng.integers(3, 8)
        centers = rng.uniform(0.1, 0.9, n_peaks)
        a = np.zeros(n)
        for c in centers:
            width = rng.uniform(0.03, 0.08)
            a += np.exp(-(x - c) ** 2 / (2 * width ** 2))
    elif choice == 7:
        a = np.sinc(x - 0.5) ** 2 + 0.1
    elif choice == 8:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x)
    else:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * x) + 0.2 * np.sin(2 * np.pi * 7 * x)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a = a + 0.004 * np.abs(rng.standard_normal(n))
    a = a / a.sum()
    return a


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def check_update(best_seq, best_score, a, label, t_global):
    """Check if a improves best. Returns (best_seq, best_score)."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    sc = eval_fn(a)
    if np.isfinite(sc) and sc < best_score:
        old = best_score
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (was {old:.6f}, n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1680.0)
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LSE-Adam (quick start)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.05, 70.0)
    print(f"\n=== Phase 1: LSE-Adam n={n_base} ({t1:.0f}s) ===")
    if t1 > 10:
        a_out, _ = lse_adam(best_seq, t1)
        best_seq, best_score = check_update(best_seq, best_score, a_out, "lse1", t_global)

    # Phase 2: LP Nesterov (main workhorse - full constraints, HiGHS)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.55, 850.0)
    print(f"\n=== Phase 2: LP Nesterov n={len(best_seq)} ({t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = lp_nesterov(best_seq, t2, verbose=True)
        best_seq, best_score = check_update(best_seq, best_score, a_out, "lp1", t_global)

    # Phase 3: LSE-Adam polish from LP result
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 3: LSE polish ({t3:.0f}s) ===")
    if t3 > 10:
        a_out, _ = lse_adam(best_seq, t3)
        best_seq, best_score = check_update(best_seq, best_score, a_out, "lse2", t_global)

    # Phase 4: LP Nesterov round 2 (refine from improved point)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 280.0)
    print(f"\n=== Phase 4: LP Nesterov r2 ({t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = lp_nesterov(best_seq, t4, verbose=True)
        best_seq, best_score = check_update(best_seq, best_score, a_out, "lp2", t_global)

    # Phase 5: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 5: Pair CD ({t5:.0f}s) ===")
    if t5 > 10:
        a_out, _ = pair_cd(best_seq, t5, rng, verbose=True)
        best_seq, best_score = check_update(best_seq, best_score, a_out, "pcd", t_global)

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_candidates = [n_base + 200, n_base - 200, n_base + 400, n_base - 400]
        for n_try in n_candidates:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 40:
                break
            if n_try < 200 or n_try > 4000:
                continue
            t_try = min(remaining * 0.3, 35.0)
            print(f"\n=== Multi-length n={n_try} ({t_try:.0f}s) ===")
            init_try = interpolate_seq(best_seq, n_try)
            a_out, _ = lse_adam(init_try, min(t_try, 20.0))
            best_seq, best_score = check_update(best_seq, best_score, a_out, f"ml_n{n_try}", t_global)
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20:
                a_out, _ = pair_cd(a_out, min(remaining * 0.4, 15.0), rng)
                best_seq, best_score = check_update(best_seq, best_score, a_out, f"ml_pcd_n{n_try}", t_global)

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
