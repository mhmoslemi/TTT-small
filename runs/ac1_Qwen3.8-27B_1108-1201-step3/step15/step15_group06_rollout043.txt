def _build_T(a, idx):
    n = len(a)
    m = len(idx)
    T = np.zeros((m, n))
    for row, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            p = k - j
            if 0 <= p < n:
                T[row, j] = a[p]
    return T


def _lp_max_sum(a, M, solver, active_idx):

    try:

        return None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)

            avg = (orig_a + (t - 1) * y) / t  # rough average
            # Actually use a running average
            pass

        # Restart on stall
        if stall > 60:
            y = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall = 0
            # Add perturbation
            pert = 1.0 + 0.05 * rng.standard_normal(n)
            pert = np.maximum(pert, 0.1)
            y = y * pert
            y = y * norm / y.sum()

        if verbose and t % 300 == 0:
            el = time.monotonic() - t0
            print(f"    NLP t={t} ({el:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def _multiplicative_descent(a0, time_limit, seed=0, verbose=False):
    """Mirror descent with Fisher information geometry."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    size = 2 * n - 1
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-10)
    a = a * norm / a.sum()

    best_a = a.copy()
    best_score = _fast_conv(a).max()
    t0 = time.monotonic()
    lr = 0.1
    epoch = 0

    while time.monotonic() - t0 < time_limit:
        epoch += 1
        conv = _fast_conv(a)
        M = conv.max()
        avg_c = (a.sum() ** 2) / size

        # Subgradient of max conv: weight active lags
        active = np.where(conv >= M * 0.95)[0]
        weights = np.zeros(size)
        weights[active] = 1.0 / len(active)

        # Gradient of c[k] w.r.t. a[j]: a[k-j] (if valid) + a[j+k] (if valid, for symmetry)
        # Actually d/dc: d(a*a)_k / da_j = a[k-j] (if 0<=k-j<n) + a[k-j] (same term counted once)
        # Wait: c[k] = sum_j a[j]*a[k-j]. d/dc_k / da[j] = a[k-j] + a[j+k] (if both valid)
        # But j+k might be out of range. Let me be careful.
        # c[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j] a[k-j]
        # d/d a[j0]: term a[j0]*a[k-j0] gives a[k-j0] (if k-j0 in [0,n-1])
        #            term a[k-j0]*a[j0] gives a[k-j0] (if k-j0 in [0,n-1])
        # So d/d a[j0] = 2*a[k-j0] if 0<=k-j0<n, else a[k-j0] if k-j0==j0 (i.e., 2j0==k)
        # Actually more carefully:
        # c[k] = sum_j a[j] a[k-j]
        # d/d a[j0]: if j0 appears as the first factor: a[k-j0] (when k-j0 in range)
        #            if j0 appears as the second factor: a[j0'] where k-j0'=j0, i.e., j0'=k-j0
        #            So: a[k-j0] (from first) + a[k-j0] (from second, when k-j0 in range)
        #            = 2*a[k-j0] if k-j0 in [0,n-1]
        #            But if j0 = k-j0 (i.e., 2*j0=k), then it's just a[j0] (counted once)
        #            = a[j0] if 2*j0 == k, else 2*a[k-j0]

        # Compute gradient
        grad = np.zeros(n)
        for k in active:
            for j in range(max(0, k - n + 1), min(n, k + 1)):
                p = k - j
                if 0 <= p < n:
                    if j == p:
                        grad[j] += weights[k] * a[p]
                    else:
                        grad[j] += weights[k] * a[p]
                        grad[p] += weights[k] * a[j]

        # Multiplicative update: a <- a * exp(-lr * grad / M)
        update = np.exp(-lr * grad / (M + 1e-15))
        a_new = a * update
        a_new = np.maximum(a_new, 1e-15)
        sa = a_new.sum()
        a_new = a_new * norm / sa

        sc = _fast_conv(a_new).max()
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.05, 0.5)
        else:
            lr *= 0.9
            if lr < 1e-4:
                lr = 0.05
                # Perturb
                a = a * (1.0 + 0.03 * rng.standard_normal(n))
                a = np.maximum(a, 1e-10)
                a = a * norm / a.sum()

        if verbose and epoch % 200 == 0:
            el = time.monotonic() - t0
            print(f"    MD epoch={epoch} ({el:.0f}s): best={best_score:.10f}, lr={lr:.4f}")

    return best_a, best_score


def _coord_descent(a0, time_limit, seed=0, verbose=False):
    """Block coordinate descent with golden section."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_conv(a).max()
    t0 = time.monotonic()
    no_improve = 0
    block_size = min(4, n)

    def _eval(a_test):
        c = _fast_conv(a_test)
        return c.max()

    while time.monotonic() - t0 < time_limit:
        improved = False
        # Process in blocks
        start = 0
        while start < n:
            if time.monotonic() - t0 > time_limit:
                break
            end = min(start + block_size, n)
            idx = list(range(start, end))
            block_val = a[idx].copy()

            # Try small perturbations of the block
            best_block = block_val.copy()
            best_bv = _eval(a)

            for trial in range(20):
                if time.monotonic() - t0 > time_limit:
                    break
                perm = list(idx)
                rng.shuffle(perm)
                scale = 0.1 * norm / n
                delta = scale * rng.standard_normal(len(perm))
                cand = a.copy()
                for ii, j in enumerate(perm):
                    cand[j] = max(0.0, a[j] + delta[ii])
                sc = cand.sum()
                if sc < 1e-15:
                    continue
                cand = cand * norm / sc
                bv = _eval(cand)
                if bv < best_bv - 1e-13:
                    best_bv = bv
                    best_block = cand[idx].copy()

            if best_bv < _eval(a) - 1e-13:
                a[idx] = best_block
                a = np.maximum(a, 0.0)
                a = a * norm / a.sum()
                sc = _fast_conv(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                improved = True

            start = end

        if not improved:
            no_improve += 1
            if no_improve >= 4:
                break
        else:
            no_improve = 0

        if verbose:
            el = time.monotonic() - t0
            print(f"    CD ({el:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def _flatten_postprocess(a0, time_limit, seed=0, verbose=False):
    """Try to flatten the convolution by moving mass from peak to trough contributors."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    size = 2 * n - 1
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_conv(a).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        conv = _fast_conv(a)
        M = conv.max()
        avg_c = (a.sum() ** 2) / size
        peak_lags = np.where(conv >= M * 0.98)[0]
        trough_lags = np.where(conv <= avg_c * 0.5)[0]

        if len(peak_lags) == 0 or len(trough_lags) == 0:
            break

        # For each peak lag, find which coordinates contribute most
        # and try to reduce them while increasing coordinates that help troughs
        improved = False
        for k in peak_lags[:5]:
            if time.monotonic() - t0 > time_limit:
                break
            # Coordinates contributing to c[k]
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            contributors = list(range(lo, hi))
            if len(contributors) < 2:
                continue

            # Find the largest contributor
            contribs = [(a[j] * a[k - j], j) for j in contributors]
            contribs.sort(reverse=True)
            j_big = contribs[0][1]

            # Try to move mass from j_big to a coordinate that doesn't contribute to peaks
            for k2 in trough_lags[:3]:
                lo2 = max(0, k2 - n + 1)
                hi2 = min(n, k2 + 1)
                # Find a coordinate NOT in contributors that IS in contributors of k2
                for j2 in range(lo2, hi2):
                    if j2 in contributors:
                        continue
                    if a[j2] < a[j_big] * 0.5:
                        # Move small amount from j_big to j2
                        for frac in [0.01, 0.05, 0.1, 0.2]:
                            amt = a[j_big] * frac
                            a_try = a.copy()
                            a_try[j_big] -= amt
                            a_try[j2] += amt
                            a_try = np.maximum(a_try, 0.0)
                            sc = a_try.sum()
                            if sc < 1e-15:
                                continue
                            a_try = a_try * norm / sc
                            bv = _fast_conv(a_try).max()
                            if bv < best_score - 1e-12:
                                a = a_try
                                best_score = bv
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break
                if improved:
                    break

        if not improved:
            # Try random perturbation
            a_pert = a * (1.0 + 0.02 * rng.standard_normal(n))
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert * norm / a_pert.sum()
            bv = _fast_conv(a_pert).max()
            if bv < best_score:
                a = a_pert
                best_score = bv
                best_a = a.copy()
            else:
                break

        if verbose:
            el = time.monotonic() - t0
            print(f"    Flatten ({el:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def _init_ramp(n, direction=1):
    if direction == 1:
        a = np.arange(1, n + 1, dtype=np.float64)
    else:
        a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()
