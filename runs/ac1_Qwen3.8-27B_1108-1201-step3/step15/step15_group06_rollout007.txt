def fast_autocorr(a):

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_autocorr(a):
    """Compute full linear autocorrelation of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_subgrad_fft(a, active_set):
    """Compute subgradient of max autocorrelation at point a using FFT.
    g[i] = (1/|S|) * sum_{k in S} a[k-i] (with a[j]=0 for j out of [0,n-1])
    """
    n = len(a)
    size = 2 * n - 1
    # Need cross-correlation of h (length size) and a (length n)
    # Result length: size + n - 1
    fft_size = 1 << (size + n).bit_length()
    
    h = np.zeros(size)
    h[active_set] = 1.0 / len(active_set)
    
    F_a = rfft(a, fft_size)
    F_h = rfft(h, fft_size)
    
    # Cross-correlation: (h convolve-with-conj a)[i] = sum_k h[k] * a[k-i]
    # ifft(fft(h) * conj(fft(a))) gives sum_k h[k] * a[k-i] (circular, but zero-padded)
    g_full = irfft(F_h * np.conjugate(F_a), fft_size)
    
    return g_full[:n]

def block_coord_descent(a0, time_limit, rng, verbose=True, block_size=8):
    """Block coordinate descent with simple line search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_seq(a0, norm)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0
    
    n_blocks = (n + block_size - 1) // block_size
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n_blocks)
        
        for b in perm:
            if time.monotonic() - t0 > time_limit:
                break
            
            start = b * block_size
            end = min(start + block_size, n)
            block_idx = np.arange(start, end)
            block_val = a[block_idx].copy()
            
            if block_val.sum() < 1e-12:
                continue
            
            # Try redistributing mass within the block
            # The idea: try to make the block more "uniform" or "less peaky"
            # Parameterize: a_block = block_val * (1 + alpha * noise)
            # Or: try moving mass from high to low within the block
            
            best_local = a.copy()
            best_local_score = best_score
            
            # Try several perturbations of the block
            for trial in range(5):
                # Random perturbation of the block
                noise = rng.standard_normal(end - start)
                alpha = 0.1 * (trial + 1) / 5.0
                new_block = block_val * (1.0 + alpha * noise)
                new_block = np.maximum(new_block, 0.0)
                s_block = new_block.sum()
                if s_block < 1e-15:
                    continue
                # Scale to maintain total sum
                new_block = new_block * block_val.sum() / s_block
                
                a_try = a.copy()
                a_try[block_idx] = new_block
                a_try = normalize_seq(a_try, norm)
                
                sc = fast_autocorr(a_try).max()
                if sc < best_local_score - 1e-10:
                    best_local_score = sc
                    best_local = a_try.copy()
            
            if best_local_score < best_score - 1e-12:
                a = best_local
                best_score = best_local_score
                best_a = a.copy()
                pass_improved = True
                no_improve = 0
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 8:
                break
        
        if verbose and n_passes % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"      BCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score

                a_try = normalize_seq(a_try, norm)
                sc = fast_autocorr(a_try).max()
                if sc < best_sc - 1e-10:
                    best_sc = sc
                    best_factor = factor
            
            if best_factor != 1.0:
                a[i] = a[i] * best_factor
                a = normalize_seq(a, norm)
                if best_sc < best_score:
                    best_score = best_sc
                    best_a = a.copy()
                    pass_improved = True
                    no_improve = 0
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
        
        if verbose and n_passes % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score


def structured_init(n, style=0):
    """Create a structured initial sequence."""
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif style == 1:
        a = np.sin(np.pi * x) ** 1.2
    elif style == 2:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(8 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif style == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif style == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif style == 5:
        a = 0.4 + 0.6 * np.sin(np.pi * x) ** 0.8
    elif style == 6:
        a = 1.0 + 0.25 * np.cos(6 * np.pi * x) + 0.1 * np.cos(12 * np.pi * x)
    else:
        a = np.sin(np.pi * x) ** 0.6 * (1.0 + 0.15 * np.cos(10 * np.pi * x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def perturb_seq(a, rng, scale=0.05):
    """Perturb a sequence with multiplicative noise."""
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.05)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s


def annealed_mirror_descent(a0, time_limit, verbose=True, seed=0):
    """Mirror descent with simulated annealing restarts."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    
    a = normalize_seq(a0, norm)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    t0 = time.monotonic()
    
    # Annealing schedule
    T_start = 0.1
    T_end = 0.01
    n_restarts = 5
    
    for restart in range(n_restarts):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 30:
            break
        
        # Temperature for this phase
        frac = restart / max(n_restarts - 1, 1)
        T_cur = T_start * (T_end / T_start) ** frac
        
        # Run mirror descent for a chunk of time
        chunk_time = remaining * 0.7
        a_out, sc = mirror_descent_nesterov(a, chunk_time, verbose=False, seed=seed + restart * 100)
        
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()
        
        # Annealing jump: perturb the current best
        if restart < n_restarts - 1:
            jump_scale = T_cur
            a = perturb_seq(best_a, rng, jump_scale)
            
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"      Anneal restart={restart} ({elapsed:.0f}s): best={best_score:.10f}, T={T_cur:.4f}")
    
    return best_a, best_score
