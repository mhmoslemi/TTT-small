def build_active_constraints(a, active_idx):
    n = len(a)
    m = len(active_idx)
    T = np.zeros((m, n))
    for row, k in enumerate(active_idx):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                T[row, j] = a[idx]
    return T


def compute_subgrad(a, k_star, n):
    sg = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            sg[j] = 2.0 * a[idx]
    return sg


def soft_subgrad(a, n, temperature=0.05):
    c = fast_autocorr(a)
    c_max = c.max()
    if c_max < 1e-15:
        return np.zeros(n)
    weights = np.exp((c - c_max) / (temperature * c_max))
    weights = weights / weights.sum()
    n_active = np.where(weights > 1e-6)[0]
    sg = np.zeros(n)
    for w, k in zip(weights[n_active], n_active):
        if w < 1e-8:
            continue
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                sg[j] += w * 2.0 * a[idx]
    return sg

def lp_max_sum(a, M, solver, active_idx=None):

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)

def lp_min_max(a, solver, active_idx=None, normalize_sum=1.0):

    t_val = float(t.value) if t.value is not None else float('inf')

def smooth_sequence(a, kernel_width=3):
    n = len(a)
    if kernel_width <= 1:
        return a.copy()
    half = kernel_width // 2
    kernel = np.ones(kernel_width) / kernel_width
    padded = np.pad(a, half, mode='edge')
    smoothed = np.convolve(padded, kernel, mode='valid')
    s = smoothed.sum()
    if s < 1e-15:
        return a.copy()
    return smoothed / s

            a_test2 = a.copy()
            a_test2[i] = c2
            a_test2 = a_test2 * norm / a_test2.sum()
            f2 = fast_autocorr(a_test2).max()

            for _ in range(20):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + (1 - phi) * (hi - lo)
                    a_test1 = a.copy()
                    a_test1[i] = c1
                    a_test1 = a_test1 * norm / a_test1.sum()
                    f1 = fast_autocorr(a_test1).max()
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + phi * (hi - lo)
                    a_test2 = a.copy()
                    a_test2[i] = c2
                    a_test2 = a_test2 * norm / a_test2.sum()
                    f2 = fast_autocorr(a_test2).max()

            opt_val = c1 if f1 < f2 else c2
            a_test_opt = a.copy()
            a_test_opt[i] = opt_val
            a_test_opt = a_test_opt * norm / a_test_opt.sum()
            sc = fast_autocorr(a_test_opt).max()

            if sc < best_score - 1e-12:
                a = a_test_opt
                best_score = sc
                best_a = a.copy()
                pass_improved = True
                no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def structured_init(n, style=0):
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif style == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif style == 2:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(8 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif style == 3:
        a = np.exp(-6.0 * (x - 0.35) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.65) ** 2)
    elif style == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif style == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 1.0
    elif style == 6:
        a = 1.0 + 0.2 * np.cos(5 * np.pi * x) + 0.08 * np.cos(10 * np.pi * x)
    elif style == 7:
        a = np.sin(np.pi * x) ** 0.8 * (1.0 + 0.1 * np.cos(8 * np.pi * x))
    elif style == 8:
        a = np.cos(np.pi * x / 2.0) ** 1.5
    elif style == 9:
        a = 1.0 / (1.0 + 2.0 * np.sin(np.pi * x / 2.0) ** 2)
    else:
        a = np.sin(np.pi * x) ** 1.2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def perturb_seq(a, rng, scale=0.03):
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.05)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s


def propose_candidate():
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = 1740.0

    solver = get_best_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_init = 1600
        for sty in range(10):
            cand = structured_init(n_init, sty)
            sc = evaluate_safe(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand
        print(f"Best init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            old_score = best_score
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s, delta={old_score - sc:.2e})")

    def remaining():
        return time_budget - (time.monotonic() - t_global)

    # === Phase 0: Quick subgradient pre-processing ===
    n_cur = len(best_seq)
    print(f"\n=== Phase 0: Fast subgradient n={n_cur} ===")
    t0 = remaining()
    t_sub = min(t0 * 0.08, 120.0)
    if t_sub > 30:
        a_work = best_seq.copy()
        a_out, _ = fast_subgrad(a_work, 8000, alpha0=1.0, temperature=0.04, verbose=True, t0=time.monotonic())
        # Respect time
        try_update(a_out, f"subgrad_n{n_cur}")

    # === Phase 1: LP Nesterov main (largest allocation) ===
    n_cur = len(best_seq)
    print(f"\n=== Phase 1: LP Nesterov n={n_cur} ===")
    t1 = min(remaining() * 0.30, 450.0)
    if t1 > 60:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t1, solver, verbose=True, seed=0)
        try_update(a_out, f"lp1_n{n_cur}")

    # === Phase 2: Smoothing + LP ===
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: Smooth+LP n={n_cur} ===")
    t2 = min(remaining() * 0.12, 180.0)
    if t2 > 40:
        a_smooth = smooth_sequence(best_seq, 5)
        a_sm_score = evaluate_safe(a_smooth)
        if a_sm_score < best_score:
            best_seq = a_smooth.copy()
            best_score = a_sm_score
            print(f"  Smoothing improved: {best_score:.8f}")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t2, solver, verbose=True, seed=123, use_smoothing=True, smooth_every=300)
        try_update(a_out, f"smoothlp_n{n_cur}")

    # === Phase 3: Subgradient + LP hybrid on perturbed ===
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: Perturb+Subgrad+LP n={n_cur} ===")
    t3_total = min(remaining() * 0.12, 180.0)
    if t3_total > 60:
        t3_each = t3_total / 2
        for r in range(2):

            a_pert = perturb_seq(best_seq, rng, scale=0.04 + 0.02 * r)

                try_update(a_sg, f"pert{r}_sg")
            if remaining() > 50 and t_lp > 20:

                try_update(a_out, f"pert{r}_lp")

    # === Phase 4: Length exploration ===
    n_base = len(best_seq)
    print(f"\n=== Phase 4: Length exploration ===")
    lengths_to_try = [n_base + 300, n_base + 600, n_base + 1000, n_base - 200]

        if remaining() < 150:
            break
        alloc = min(remaining() * 0.12, 150.0)
        if alloc < 50:
            break
        print(f"  Trying n={n_try} (alloc={alloc:.0f}s)")
        init_try = interpolate_seq(best_seq, n_try)
        a_sg, _ = fast_subgrad(init_try, 3000, alpha0=1.0, temperature=0.05, verbose=False, t0=time.monotonic())
        sc_sg = evaluate_safe(a_sg)
        if np.isfinite(sc_sg):
            try_update(a_sg, f"subgrad_n{n_try}")
        a_work = best_seq if best_score <= sc_sg else a_sg
        a_work = interpolate_seq(a_work, n_try)

    # === Phase 5: CD polish on best ===

        a_out, _ = coord_descent(a_norm, t5, rng, verbose=True)

    # === Phase 6: Final LP on best ===
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final LP n={n_best} ===")
    t6 = min(remaining() * 0.30, 250.0)

        a_out, _ = nesterov_lp(a_norm, t6, solver, verbose=True, seed=999, alpha_min=0.05)
        try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 7: Final CD ===
    n_best = len(best_seq)
    if remaining() > 40:
        print(f"\n=== Phase 7: Final CD n={n_best} ===")
        t7 = min(remaining() * 0.5, 80.0)
        if t7 > 15:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = coord_descent(a_norm, t7, rng, verbose=False)
            try_update(a_out, f"cd_final_n{n_best}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
