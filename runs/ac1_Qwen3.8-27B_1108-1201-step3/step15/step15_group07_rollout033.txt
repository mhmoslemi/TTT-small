**Reduced workspace**: Instead of solving the full LP, maintain a working set of active constraints. At each iteration, only add new constraints that are violated and remove constraints that are no longer active. This is the "column generation" or "constraint generation" approach.

The constraint generation approach:
1. Start with a small set of constraints (e.g., just the one at the current peak)
2. Solve the LP with these constraints
3. Check if the solution violates any other constraints
4. If yes, add the violated constraint and re-solve
5. If no, the solution is optimal

This could be much faster than solving the full LP, especially when the number of active constraints is small (which is typical for LP optima).

```python
def lp_best_response_cg(a, M, solver, max_inner_iters=50):
    """LP best response using constraint generation."""
    n = len(a)
    size = 2 * n - 1
    
    # Start with just the center constraint (lag 0)
    active_constraints = [n - 1]  # Position n-1 in the full convolution
    
    T = build_T(a)  # (size, n)
    
    for _ in range(max_inner_iters):
        # Build reduced LP
        T_reduced = T[active_constraints, :]  # (|active|, n)
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_reduced @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        prob.solve(solver=solver, verbose=False)
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        
        # Check if all constraints are satisfied
        conv_val = T @ b_val  # (size,)
        violated = np.where(conv_val > M + 1e-10)[0]
        
        if len(violated) == 0:
            return b_val
        
        # Add violated constraints (limit to avoid too many)
```

```python
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP best response with HiGHS solver + Nesterov acceleration (primary)
2. Constraint generation for accelerated LP solves
3. Multiple n exploration (800, 1200, 1600)
4. Diverse restarts with structured initializations
5. Subgradient + coordinate descent polish

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def lp_best_response(a, M, use_cg=False):
    """Solve LP: max sum(b) s.t. T_a @ b <= M, b >= 0"""
    n = len(a)
    size = 2 * n - 1
    
    if use_cg:
        return lp_best_response_cg(a, M)
    else:
        T = build_T(a)
        c = -np.ones(n)
        b_ub = np.full(size, M)
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
        if res.status != 0 or res.x is None:
            return None
        return np.maximum(res.x, 0.0)

def lp_best_response_cg(a, M, max_iter=100, max_active=100):
    """Constraint generation LP best response."""
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    
    # Start with the center constraint
    active = [n - 1]
    
    for _ in range(max_iter):
        T_a = T[active, :]
        c = -np.ones(n)
        b_ub = np.full(len(active), M)
        res = linprog(c, A_ub=T_a, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
        if res.status != 0 or res.x is None:
            return None
        b_val = np.maximum(res.x, 0.0)
        
        conv_val = T @ b_val
        tol = 1e-10 * M
        violated = np.where(conv_val > M + tol)[0]
        
        if len(violated) == 0:
            return b_val
        
        # Add most violated constraints
        viol_idx = violated[np.argsort(conv_val[violated])[::-1][:min(10, len(violated))]]
        for idx in viol_idx:
            if idx not in active:
                active.append(idx)
        
        if len(active) > max_active:
            # Reduce to most important constraints
            conv_active = T[active, :] @ b_val
            keep = np.argsort(conv_active)[::-1][:max_active]
            active = [active[i] for i in keep]
    
    return None

def nesterov_lp_optimize(a0, time_limit, use_cg=False, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    momentum_t = 1.0
    patience = 0
    stuck_count = 0
    
    for t in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-15:
            break
        
        b = lp_best_response(y, M_curr, use_cg=use_cg)
        if b is None:
            stuck_count += 1
            if stuck_count > 50:
                break
            continue
        
        stuck_count = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        alpha = 2.0 / (momentum_t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if momentum_t < 3.0:
            a_new = y_new.copy()
        else:
            momentum = (momentum_t - 1.0) / (momentum_t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        momentum_t += 1.0
        
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            patience = 0
        else:
            patience += 1
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            patience = 0
        
        if patience > 100 and n_iters % 50 == 0:
            y = best_a.copy()
            a = best_a.copy()
            momentum_t = 1.0
            patience = 0
            
        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score
```

Wait, I realize there's an issue with the subgradient step. The projection onto {x >= 0, sum(x) = norm} is not simply "clip to non-negative and rescale". The proper projection onto the non-negative simplex is more involved (it requires solving for the threshold).

The Euclidean projection onto {x >= 0, sum(x) = S} is:
- Find tau such that sum(max(x_i - tau, 0)) = S
- Return max(x_i - tau, 0)

This can be found by bisection on tau.

But for our purposes, the simple "clip and rescale" projection is a valid (if non-Euclidean) projection that preserves non-negativity and the sum constraint. It's not the Euclidean projection, but it works for a "projected subgradient" method in a non-Euclidean sense.

Actually, for the mirror descent / multiplicative weights approach, the "clip and rescale" is the right projection (it's the Bregman projection for the entropy regularizer). So it's consistent.

Actually, a simpler approach: use the log-sum-exp smoothing:
`f_smooth(a) = tau * log(sum_k exp(conv[k]/tau))`

The gradient of this is:
`grad[j] = sum_k (exp(conv[k]/tau) / sum_l exp(conv[l]/tau)) * 2 * a[k-j]` (for valid k-j)

This is a weighted average of the subgradients, with weights proportional to exp(conv[k]/tau). For small tau, it concentrates on the maximum; for large tau, it's more spread out.

This gives a smooth optimization problem that we can solve with gradient descent.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP best response via HiGHS + Nesterov acceleration (primary)
2. Multiple n exploration with diverse initializations  
3. Top-k smoothed subgradient for polishing
4. Targeted coordinate descent for final refinement
Key: Fast LP via scipy HiGHS, constraint generation option,
adaptive restarts, and multi-phase time allocation.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_T(a):
    """Build convolution matrix T where (T@b)[k] = (a*b)[k]."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def lp_solve_full(a, M):
    """Solve LP: max sum(b) s.t. T_a @ b <= M, b >= 0 using HiGHS."""
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    c = -np.ones(n)
    b_ub = np.full(size, M)
    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
        if res.status == 0 and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def lp_solve_cg(a, M, max_iter=200, max_active=150):
    """Constraint generation LP solve."""
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    active = [n - 1]
    b_val = None
    
    for _ in range(max_iter):
        T_a = T[active, :]
        c = -np.ones(n)
        b_ub = np.full(len(active), M)
        try:
            res = linprog(c, A_ub=T_a, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
            if res.status != 0 or res.x is None:
                break
            b_val = np.maximum(res.x, 0.0)
        except Exception:
            break
        
        if b_val is None:
            break
        
        conv_val = T @ b_val
        tol = 1e-10 * max(M, 1e-10)
        violated = np.where(conv_val > M + tol)[0]
        
        if len(violated) == 0:
            return b_val
        
        # Add most violated
        viol_sorted = violated[np.argsort(conv_val[violated])[::-1]]
        add_count = min(20, len(viol_sorted))
        for idx in viol_sorted[:add_count]:
            if idx not in active:
                active.append(idx)
        
        if len(active) > max_active:
            conv_active = T[active, :] @ b_val
            keep = np.argsort(conv_active)[::-1][:max_active]
            active = [active[i] for i in keep]
    
    return b_val

def nesterov_lp(a0, time_limit, use_cg=False, verbose=True):
    """Nesterov-accelerated LP best response."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    momentum_t = 1.0
    patience = 0
    stuck = 0
    
    while time.monotonic() - t0 < time_limit:
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-15:
            break
        
        if use_cg:
            b = lp_solve_cg(y, M_curr)
        else:
            b = lp_solve_full(y, M_curr)
        
        if b is None:
            stuck += 1
            if stuck > 100:
                break
            continue
        
        stuck = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        alpha = 2.0 / (momentum_t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        y_new = y_new * norm / s_y if s_y > 1e-15 else y.copy()
        
        if momentum_t < 3.0:
            a_new = y_new.copy()
        else:
            momentum = (momentum_t - 1.0) / (momentum_t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            a_new = a_new * norm / s_a if s_a > 1e-15 else y_new.copy()
        
        y = y_new
        a = a_new
        momentum_t += 1.0
        
        sy = fast_autocorr(y).max()
        if sy < best_score:
            best_score = sy
            best_a = y.copy()
            patience = 0
        else:
            patience += 1
        
        sa = fast_autocorr(a).max()
        if sa < best_score:
            best_score = sa
            best_a = a.copy()
            patience = 0
        
        if patience > 150 and n_iters % 75 == 0:
            y = best_a.copy()
            a = best_a.copy()
            momentum_t = 1.0
            patience = 0
        
        if verbose and n_iters % 200 == 0:
            print(f"    NLP {n_iters} ({time.monotonic()-t0:.0f}s): {best_score:.8f}")
    
    return best_a, best_score

def smoothed_subgrad(a0, time_limit, top_k=50, tau=0.05, verbose=True):
    """Top-k smoothed subgradient method."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    eta = 2.0 / n
    n_iters = 0
    patience = 0
    rng = np.random.default_rng(54321)
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_autocorr(a)
        k_actual = min(top_k, size)
        active_idx = np.argsort(conv)[-k_actual:]
        active_conv = conv[active_idx]
        
        weights = np.exp(active_conv / tau - np.max(active_conv) / tau)
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for idx, k in enumerate(active_idx):
            w = weights[idx]
            if w < 1e-12:
                continue
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            grad[j_start:j_end+1] += 2.0 * w * a[k - j_start:k - j_end + 1][::-1]
        
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) * norm / n
        else:
            a_new = a_new * norm / s
        
        score = fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
            if patience > 100 and patience % 50 == 0:
                eta *= 0.9
        
        if patience > 500:
            noise = rng.laplace(0, norm * 0.01, n)
            a_new = np.maximum(a_new + noise, 0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new * norm / s
            patience = 0
            eta = 2.0 / n
        
        a = a_new
        n_iters += 1
        
        if verbose and n_iters % 500 == 0:
            print(f"    SG {n_iters} ({time.monotonic()-t0:.0f}s): {best_score:.8f}")
    
    return best_a, best_score

def cd_polish(a0, time_limit, window=20, verbose=True):
    """Coordinate descent polish."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    
    conv = fast_autocorr(a)
    best_a = a.copy()
    best_score = conv.max()
    t0 = time.monotonic()
    size = 2 * n - 1
    rng = np.random.default_rng(777)
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < 5:
        pass_improved = False
        perm = rng.permutation(n)
        
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < norm * 1e-10:
                continue
            
            for off in range(1, window + 1):
                for j in [(i + off) % n, (i - off) % n]:
                    if j == i or a[i] < norm * 1e-10:
                        continue
                    
                    # Quick evaluation: try moving a fraction from i to j
                    for frac in [0.5, 0.7, 0.9]:
                        d = a[i] * frac
                        a[i] -= d
                        a[j] += d
                        new_conv = fast_autocorr(a)
                        new_max = new_conv.max()
                        if new_max < best_score - 1e-13:
                            best_score = new_max
                            best_a = a.copy()
                            conv = new_conv
                            pass_improved = True
                            no_improve = 0
                            break
                        else:
                            a[i] += d
                            a[j] -= d
        
        if not pass_improved:
            no_improve += 1
        
        if verbose:
            print(f"    CD ({time.monotonic()-t0:.0f}s): {best_score:.8f}")
    
    return best_a, best_score
