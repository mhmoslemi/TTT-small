def normalize(a, target_sum):
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * target_sum / len(a)

def build_constraint_matrix(a, k_indices, n):
    size = len(k_indices)
    mat = np.zeros((size, n))
    for idx, k in enumerate(k_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            mat[idx, j] = a[k - j]
    return mat


def build_constraint_matrix_fast(a, k_indices, n):
    """Vectorized version of constraint matrix construction."""
    mat = np.zeros((len(k_indices), n))
    for idx, k in enumerate(k_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        j_range = np.arange(lo, hi)
        mat[idx, j_range] = a[k - j_range]
    return mat


def lp_best_response(a_fixed, M, active_set, solver, n):
    k_indices = np.asarray(active_set)
    if len(k_indices) > n * 2:  # Too many constraints, subsample
        # Keep the most binding ones

        binding = np.where(conv >= M * 0.99)[0]
        k_indices = binding

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def peak_equalization_step(a, n_peaks=5, step_scale=0.1):
    """
    Try to equalize the top peaks by adjusting a in the direction
    that reduces the spread of the top peaks.
    """
    n = len(a)
    norm = np.sqrt(2.0 * n)
    conv = fast_autocorr(a)
    M = conv.max()
    
    # Get top peaks
    n_peaks = min(n_peaks, 2*n-1)
    top_idx = np.argsort(conv)[-n_peaks:]
    top_vals = conv[top_idx]
    
    # Spread of top peaks
    spread = top_vals.max() - top_vals.min()
    if spread < 1e-8:
        return a.copy()
    
    # For each top peak, compute gradient direction
    # We want to reduce the spread: decrease high peaks, increase low peaks
    # Use a weighted combination of gradients
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        weight = (top_vals[i] - top_vals.mean())  # positive for high peaks, negative for low
        # Gradient of (a*a)[k] w.r.t. a
        for j in range(n):
            j1 = k - j
            if 0 <= j1 < n:
                grad[j] += 2.0 * weight * a[j1]
    
    grad_norm = np.linalg.norm(grad)
    if grad_norm < 1e-15:
        return a.copy()
    
    # Try a few step sizes
    best_a = a.copy()
    best_score = M
    for scale in [step_scale * 0.1, step_scale * 0.3, step_scale * 0.5, step_scale, step_scale * 2]:
        a_trial = a - scale * grad / grad_norm
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            continue
        a_trial = a_trial * norm / s_trial
        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score:
            best_score = score_trial
            best_a = a_trial.copy()
    
    return best_a


def coordinate_descent_golden(a0, time_limit, verbose=False, max_passes=50):
    """

    For each coordinate j, find optimal a[j] given all others fixed.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    golden = 0.6180339887498949

    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit and n_passes < max_passes:
        n_passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]
            s_others = a.sum() - aj
            
            # Compute contributions
            # (a*a)[k] = bg[k] + 2*x*c[k] + x^2 * delta(k, 2j)
            c = np.zeros(size)
            for k in range(size):
                i1 = k - j
                if 0 <= i1 < n and i1 != j:
                    c[k] += a[i1]
                i2 = j - k
                if 0 <= i2 < n and i2 != j:
                    c[k] += a[i2]

            conv_full = fast_autocorr(a)
            bg = conv_full - 2.0 * aj * c
            if 2 * j < size:
                bg[2 * j] -= aj * aj

            # Objective for variable x = a[j]:
            # f(x) = 2*n*max_k(bg[k] + 2*x*c[k] + x^2*delta(k,2j)) / (s_others + x)^2
            # We want to minimize this.
            # Use golden section search on [0, 2*aj + norm/n]
            
            def obj_func(x):
                new_conv = bg + 2.0 * x * c
                if 2 * j < size:
                    new_conv[2 * j] += x * x
                new_sum = s_others + x
                if new_sum < 1e-15:
                    return float('inf')
                return 2.0 * n * new_conv.max() / (new_sum * new_sum)

            # Golden section search
            lo = 0.0
            hi = 2.0 * aj + norm / n
            c1 = hi - golden * (hi - lo)
            c2 = lo + golden * (hi - lo)
            f_c1 = obj_func(c1)
            f_c2 = obj_func(c2)
            
            for _ in range(40):
                if f_c1 < f_c2:
                    hi = c2
                    c2 = c1
                    f_c2 = f_c1
                    c1 = hi - golden * (hi - lo)
                    f_c1 = obj_func(c1)
                else:
                    lo = c1
                    c1 = c2
                    f_c1 = f_c2
                    c2 = lo + golden * (hi - lo)
                    f_c2 = obj_func(c2)
            
            x_opt = (lo + hi) / 2.0
            f_opt = obj_func(x_opt)
            f_curr = obj_func(aj)
            
            if f_opt < f_curr - 1e-12:
                a[j] = x_opt
                a = a * norm / a.sum()
                new_score = fast_autocorr(a).max()
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
                    improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def subgradient_polish(a0, time_limit, verbose=False):
    """Subgradient method with adaptive step size and multiple peaks."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    patience = 0
    max_patience = 300
    step = 0.5

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()

        # Use top peaks with weights
        n_peaks = max(3, min(8, n // 200))
        top_idx = np.argsort(conv)[-n_peaks:]
        weights = np.exp(-10.0 * (M_curr - conv[top_idx]) / max(M_curr * 0.01, 1e-10))
        weights = weights / weights.sum()

        # Combined gradient
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            for j in range(n):
                j1 = k - j
                if 0 <= j1 < n:
                    grad[j] += 2.0 * w * a[j1]

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        a_trial = a - step * grad
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-12:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.01, 3.0)
        else:
            patience += 1
            step *= 0.97
            if step < 1e-10:
                break

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def perturb_sequence(a, rng, scale=0.03):
    n = len(a)

    noise = np.maximum(noise, 0.05)
    a_pert = a * noise

    if s < 1e-15:
        return a.copy()
    return a_pert / s

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP best-response with Nesterov acceleration and adaptive active-set
2. Softmax-approximation gradient descent for smooth fine optimization
3. Peak equalization via targeted gradient steps
4. Golden-section coordinate descent for final refinement
5. Subgradient polishing and perturbation restarts
Key improvements: tighter active-set, smooth softmax objective, better time allocation.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate the score."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def normalize_to(a, target_sum):
    """Normalize array to have the given sum."""
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * target_sum / len(a)
    return a * target_sum / s


def build_constraint_matrix_fast(a, k_indices, n):
    """Build constraint matrix for (a*b)[k] <= M."""
    mat = np.zeros((len(k_indices), n))
    for idx, k in enumerate(k_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        j_range = np.arange(lo, hi)
        mat[idx, j_range] = a[k - j_range]
    return mat


def lp_best_response(a_fixed, M, active_set, solver, n):
    """Solve LP: max sum(b) s.t. (a_fixed * b)[k] <= M for k in active_set, b >= 0."""
    k_indices = np.asarray(active_set)
    if len(k_indices) == 0:
        return None
    T = build_constraint_matrix_fast(a_fixed, k_indices, n)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=False, seed=0):
    """Nesterov-accelerated LP best-response iteration with improved active set."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    stall_threshold = 35
    n_iters = 0
    last_improve_iter = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive active set with progressive tightening
        if active_set is None or t % 25 == 0:
            # Tighten tolerance over time
            tol_frac = max(0.003, 0.04 * np.exp(-t / 150.0))
            active_set = np.where(conv_y >= M_curr * (1.0 - tol_frac))[0]
            # Always include top peaks
            n_top = max(5, min(30, n // 50))
            top_k = np.argsort(conv_y)[-n_top:]
            active_set = np.unique(np.concatenate([active_set, top_k]))
            # Cap constraints for solver efficiency
            if len(active_set) > 150:
                # Keep the most binding
                vals = conv_y[active_set]
                keep_thresh = np.percentile(vals, 10)
                active_set = active_set[vals >= keep_thresh]
                if len(active_set) > 150:
                    active_set = np.argsort(vals)[-150:] + active_set[np.argsort(vals)[-150:]]
                    active_set = np.unique(active_set[:150])

        b = lp_best_response(y, M_curr, active_set, solver, n)

        if b is None:
            # Retry with broader active set
            active_set = np.where(conv_y >= M_curr * 0.95)[0]
            if len(active_set) > 200:
                active_set = np.argsort(conv_y)[-200:]
            b = lp_best_response(y, M_curr, active_set, solver, n)
            if b is None:
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update with adaptive momentum
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Momentum (slightly damped to avoid oscillation)
        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.92)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score - 1e-14:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
            last_improve_iter = t
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score - 1e-14:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
            last_improve_iter = t

        # Restart if stalled
        if stall_count > stall_threshold:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall_count = 0
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    LP restart at iter {t} ({elapsed:.0f}s), score={best_score:.10f}")

        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def softmax_gradient_descent(a0, time_limit, verbose=False):
    """
    Smooth optimization using softmax approximation of the max function.
    Minimizes soft_max_k (a*a)[k] with gradually increasing beta.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    beta = 50.0  # Start with a reasonably high beta for accuracy
    lr = 0.05
    lr_decay = 0.9995
    patience = 0
    max_patience = 8000
    n_active_thresh = 1e-8

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        max_conv = conv.max()

        # Soft max weights (numerically stable)
        scaled = beta * (conv - max_conv)
        # Clip to avoid underflow
        scaled = np.maximum(scaled, -50.0)
        exp_conv = np.exp(scaled)
        weights = exp_conv / exp_conv.sum()

        # Compute gradient: d/d a[j] of sum_k w_k * (a*a)[k]
        # = 2 * sum_k w_k * a[k-j] (for valid k-j)
        # This is a cross-correlation, computable via FFT
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(weights, fft_size)
        A = rfft(a, fft_size)
        # Cross-correlation: sum_k w[k] * a[k-j]
        # This is the circular cross-correlation (padded)
        grad_raw = irfft(W * np.conj(A), fft_size)[:size]
        # grad_raw[k] = sum_j weights[j] * a[j-k] (circular)
        # We need: sum_k weights[k] * a[k-j] for each j
        # Which is: grad[j] = sum_k weights[k] * a[k-j]
        # This is the cross-correlation of weights and a at lag j
        # In FFT: ifft(fft(weights) * conj(fft(a))) gives sum_j weights[j] * a[j-k]
        # We need sum_k weights[k] * a[k-j] = sum_k weights[k] * a[-(j-k)]
        # Actually let's just do it directly for the active weights

        # Direct computation for efficiency (only nonzero weights matter)
        active_w = np.where(weights > n_active_thresh)[0]
        if len(active_w) == 0:
            active_w = np.array([int(np.argmax(weights))])
        if len(active_w) <= 200:
            grad = np.zeros(n)
            for k in active_w:
                wk = weights[k]
                # d(a*a)[k]/da[j] = 2*a[k-j] if 0<=k-j<n
                lo = max(0, k - n + 1)
                hi = min(n, k + 1)
                if lo < hi:
                    grad[lo:hi] += 2.0 * wk * a[k - lo:k - hi + 1][::-1]
        else:
            # Use FFT-based approach
            grad = np.zeros(n)
            for k in range(size):
                if weights[k] < 1e-10:
                    continue
                lo = max(0, k - n + 1)
                hi = min(n, k + 1)
                if lo < hi:
                    j_range = np.arange(lo, hi)
                    grad[j_range] += 2.0 * weights[k] * a[k - j_range]

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Projected gradient step
        a_trial = a - lr * grad / n
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-13:
            best_score = score_trial
            best_a = a_trial.copy()
            a = a_trial
            patience = 0
            # Slightly increase lr if improving consistently
            lr = min(lr * 1.002, 0.5)
        else:
            patience += 1
            lr *= lr_decay
            if lr < 1e-8:
                break

        # Gradually increase beta for sharper max approximation
        if t % 2000 == 0 and beta < 200.0:
            beta *= 1.3

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SoftMaxGD t={t} ({elapsed:.0f}s): best={best_score:.10f}, beta={beta:.1f}, lr={lr:.6f}")

    return best_a, best_score


def peak_equalization(a0, time_limit, verbose=False):
    """
    Target the top peaks and try to equalize them using gradient steps.
    Reduces the spread of top autocorrelation values.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_peaks = max(3, min(10, n // 200))
    lr = 0.1
    patience = 0
    max_patience = 2000

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M = conv.max()

        # Get top peaks
        top_idx = np.argsort(connv)[-n_peaks:] if False else np.argsort(conv)[-n_peaks:]
        top_vals = conv[top_idx]
        spread = top_vals.max() - top_vals.min()

        if spread < 1e-10:
            break

        # Weight by deviation from mean of top peaks
        mean_top = top_vals.mean()
        weights = top_vals - mean_top  # positive for high, negative for low

        # Effective step for equalization should be small
        eff_lr = lr * spread / max(M, 1e-10)

        # Compute gradient direction to reduce spread
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            if lo < hi:
                j_range = np.arange(lo, hi)
                grad[j_range] += 2.0 * w * a[k - j_range]

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        a_trial = a - eff_lr * grad / grad_norm
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            continue
        a_trial = a_trial * norm / s_trial

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-13:
            best_score = score_trial
            best_a = a_trial.copy()
            a = a_trial
            patience = 0
            lr = min(lr * 1.01, 1.0)
        else:
            patience += 1
            lr *= 0.98
            if lr < 1e-8:
                break

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakEq t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=False, max_passes=20):
    """
    Coordinate descent with bisection line search.
    Optimizes each coordinate independently.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit and n_passes < max_passes:
        n_passes += 1
        improved = False
        pass_improvement = 0.0

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]
            s_others = a.sum() - aj

            # Precompute the "cross-term" c[k]
            c = np.zeros(size)
            for k in range(size):
                i1 = k - j
                if 0 <= i1 < n and i1 != j:
                    c[k] += a[i1]
                i2 = j - k
                if 0 <= i2 < n and i2 != j:
                    c[k] += a[i2]

            conv_full = fast_autocorr(a)
            bg = conv_full - 2.0 * aj * c
            if 2 * j < size:
                bg[2 * j] -= aj * aj

            # Bisection to find optimal x for a[j]
            # f(x) = 2*n*max(bg + 2*x*c + x^2*delta(2j)) / (s_others + x)^2
            lo = 0.0
            hi = 2.0 * aj + norm / n

            # Quick check: is reducing a[j] helpful?
            f_at_aj = 2.0 * n * conv_full.max() / (a.sum() ** 2)

            # Try a few strategic points first
            candidates = [0.0, aj * 0.5, aj * 0.7, aj * 1.0, aj * 1.3, aj * 1.5, s_others * 0.3]
            candidates = [x for x in candidates if 0 <= x <= hi]

            best_x = aj
            best_f = f_at_aj
            for x in candidates:
                new_conv = bg + 2.0 * x * c
                if 2 * j < size:
                    new_conv[2 * j] += x * x
                new_sum = s_others + x
                if new_sum < 1e-15:
                    continue
                f_val = 2.0 * n * new_conv.max() / (new_sum * new_sum)
                if f_val < best_f:
                    best_f = f_val
                    best_x = x

            # Bisection refinement
            if best_x != aj and best_f < f_at_aj - 1e-10:
                lo = min(aj, best_x)
                hi = max(aj, best_x)
                for _ in range(12):
                    mid = (lo + hi) / 2.0
                    new_conv = bg + 2.0 * mid * c
                    if 2 * j < size:
                        new_conv[2 * j] += mid * mid
                    new_sum = s_others + mid
                    if new_sum < 1e-15:
                        break
                    f_mid = 2.0 * n * new_conv.max() / (new_sum * new_sum)
                    f_lo_new = 2.0 * n * (bg + 2.0 * lo * c + (lo * lo if 2 * j < size else 0)).max() / ((s_others + lo) ** 2)
                    if f_mid < f_lo_new:
                        lo = mid
                    else:
                        hi = mid
                best_x = (lo + hi) / 2.0

            if abs(best_x - aj) > 1e-10:
                a[j] = best_x
                a = a * norm / a.sum()
                new_score = fast_autocorr(a).max()
                if new_score < best_score - 1e-13:
                    best_score = new_score
                    best_a = a.copy()
                    improved = True
                    pass_improvement += best_a is not None
                else:
                    a[j] = aj
                    a = a * norm / a.sum()

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def subgradient_polish(a0, time_limit, verbose=False):
    """Subgradient method with adaptive step and multi-peak weighting."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    patience = 0
    max_patience = 500
    step = 0.3
    n_peaks = max(3, min(8, n // 200))

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()

        top_idx = np.argsort(conv)[-n_peaks:]
        weights = np.exp(-10.0 * (M_curr - conv[top_idx]) / max(M_curr * 0.01, 1e-10))
        weights = weights / weights.sum()

        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            if lo < hi:
                j_range = np.arange(lo, hi)
                grad[j_range] += 2.0 * w * a[k - j_range]

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        a_trial = a - step * grad / np.sqrt(n)
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-13:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.005, 2.0)
        else:
            patience += 1
            step *= 0.99
            if step < 1e-10:
                break

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def perturb_sequence(a, rng, scale=0.03):
    """Add multiplicative noise."""
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.05)
    a_pert = a * noise
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s

    print(f"Starting with n={n_base}, score={best_score:.8f}")

        t3 = min(remaining * 0.10, 120.0)
        print(f"\n=== Phase 3: Peak equalization n={n_cur}, budget={t3:.0f}s ===")
