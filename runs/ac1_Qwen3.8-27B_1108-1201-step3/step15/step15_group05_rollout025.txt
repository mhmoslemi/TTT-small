Let me think more carefully about the LP approach:

The LP best-response method works as follows:
- Given current `a`, solve: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0
- This gives the "best response" `b` that maximizes total mass while keeping the cross-convolution bounded
- Update: a_new = (1-t)*a + t*b

This is essentially a variant of the multiplicative weights / best response dynamics for the zero-sum game where one player chooses `a` and the other chooses `b`, and the payoff is `max_k (a*b)[k]`.

The key insight is that at equilibrium, both players achieve the same value, which equals the minimum possible `max_k (a*a)[k]` (by minimax theorem).

def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def build_constraint_matrix(a, n, all_k=None):
    """
    Build the constraint matrix for (a * b)[k] <= M.
    For lag k: (a*b)[k] = sum_j a[k-j] * b[j] where 0 <= k-j < n.
    So row k, col j: a[k-j] if 0 <= k-j < n, else 0.
    """
    if all_k is None:
        all_k = np.arange(2 * n - 1)
    size = len(all_k)
    mat = np.zeros((size, n))
    for idx, k in enumerate(all_k):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            mat[idx, lo:hi] = a[k - np.arange(lo, hi)]
    return mat


def lp_best_response_full(a_fixed, M, solver, n=None):
    """
    Solve LP: max sum(b) s.t. (a_fixed * b)[k] <= M for ALL k, b >= 0.
    Returns b or None.
    """
    if n is None:
        n = len(a_fixed)
    
    b_var = cp.Variable(n, nonneg=True)
    # Build full constraint matrix
    T = build_constraint_matrix(a_fixed, n)
    constraints = [T @ b_var <= M, b_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception as e:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=False, seed=0):
    """
    Nesterov-accelerated LP best-response with full constraint set.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n * norm
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    lp_time_total = 0.0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Solve full LP
        t_lp_start = time.monotonic()
        b = lp_best_response_full(y, M_curr, solver, n)
        lp_time_total += time.monotonic() - t_lp_start

        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Heavy-ball momentum
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            momentum = min(momentum, 0.95)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}, M={M_curr:.6f}")

    return best_a, best_score


def subgradient_polish(a0, time_limit, verbose=False):
    """
    Subgradient method: minimize max_k (a*a)[k] s.t. sum(a)=norm, a>=0.
    Uses top peaks for combined gradient.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    patience = 0
    max_patience = 5000
    step = 0.1

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()

        # Get top peaks with exponential weighting
        n_peaks = max(5, min(20, n // 50))
        top_idx = np.argsort(conv)[-n_peaks:]
        weights = np.exp(-10.0 * (M_curr - conv[top_idx]) / max(M_curr * 0.01, 1e-10))
        weights = weights / weights.sum()

        # Combined gradient (vectorized)
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            # grad[j] += w * (a[k-j] if valid + a[j-k] if valid)
            for j in range(n):
                v = 0.0
                j1 = k - j
                if 0 <= j1 < n:
                    v += a[j1]
                j2 = j - k
                if 0 <= j2 < n:
                    v += a[j2]
                grad[j] += w * v

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        a_trial = a - step * grad
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-14:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.005, 1.0)
        else:
            patience += 1
            step *= 0.99
            if step < 1e-12:
                break

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad iter={t} ({elapsed:.0f}s): best={best_score:.10f}, step={step:.6f}")

    return best_a, best_score

def perturb_sequence(a, rng, scale=0.03):

    s = a_pert.sum()

    return a_pert / s


def symmetrize(a):

    return a_sym

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Full-constraint LP best-response with Nesterov acceleration (exact best responses)
2. Adaptive momentum and duality-gap-based convergence detection
3. Subgradient polishing at weighted active peaks
4. Coordinate descent for final refinement
5. Multi-restart with structured and random perturbations
6. Length exploration (1600 and nearby)
Key improvement: using ALL 2n-1 constraints in the LP (not just active set)
gives exact best responses, leading to faster and more reliable convergence.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.2])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.CLARABEL
    print(f"  LP solver (default): CLARABEL")
    return _global_solver


def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate the score via the provided function."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_T(a, n):
    """
    Build constraint matrix T where T[k,j] = a[k-j] if 0<=k-j<n else 0.
    Shape: (2n-1, n)
    """
    size = 2 * n - 1
    T = np.zeros((size, n))
    for k in range(size):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[k, lo:hi] = a[k - np.arange(lo, hi)]
    return T


def lp_full(a_fixed, M, solver, n):
    """
    Solve exact LP: max sum(b) s.t. T @ b <= M, b >= 0.
    Returns b vector or None on failure.
    """
    T = build_T(a_fixed, n)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M, b_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b.sum() < 1e-15:
        return None
    return b


def nesterov_lp(a0, time_limit, solver, verbose=False, seed=0):
    """
    Nesterov-accelerated LP best-response iteration with full constraint set.
    Uses adaptive momentum and tracks convergence via score improvement.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    stall_limit = 60
    best_erg = np.zeros(n)
    erg_cnt = 0

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-14:
            break

        b = lp_full(y, M_curr, solver, n)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Adaptive momentum (heavy ball with decay)
        if t == 1:
            a_new = g.copy()
        else:
            mu = min((t - 1.0) / (t + 2.0), 0.92)
            a_new = y_new + mu * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y.copy()
        y = y_new
        a = a_new

        # Track best from y and a
        sy = fast_autocorr(y).max()
        if sy < best_score:
            best_score = sy
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        sa = fast_autocorr(a).max()
        if sa < best_score:
            best_score = sa
            best_a = a.copy()
            stall = 0

        # Ergodic average check every 50 iters
        best_erg += a
        erg_cnt += 1
        if erg_cnt % 50 == 0:
            erg = best_erg / erg_cnt
            s_erg = erg.sum()
            if s_erg > 1e-15:
                erg_n = erg * norm / s_erg
                s_erg_score = fast_autocorr(erg_n).max()
                if s_erg_score < best_score:
                    best_score = s_erg_score
                    best_a = erg_n.copy()
                    stall = 0

        # Restart if stalled
        if stall > stall_limit:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            stall = 0
            if verbose:
                print(f"      Restart at iter {n_iters} ({time.monotonic()-t0:.0f}s)")

        if verbose and n_iters % 300 == 0:
            print(f"      LP iter={n_iters} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def subgrad_polish(a0, time_limit, verbose=False):
    """
    Subgradient polishing using weighted combination of top peak gradients.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    step = 0.05
    patience = 0
    max_pat = 8000

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_pat:
            break

        conv = fast_autocorr(a)
        M_c = conv.max()

        # Top peaks with exponential weighting
        n_p = max(3, min(15, n // 100))
        top = np.argsort(conv)[-n_p:]
        wts = np.exp(-8.0 * (M_c - conv[top]) / max(M_c * 0.005, 1e-10))
        wts = wts / wts.sum()

        # Compute combined gradient (vectorized per peak)
        grad = np.zeros(n)
        for w, k in zip(wts, top):
            # grad[j] += w * (a[k-j] + a[j-k]) with bounds
            # Using convolution trick: the gradient at peak k is the "reversed" a
            # shifted by k, plus a shifted by -k
            # Equivalent to: for each j, grad[j] += w * (a[(k-j)%n if valid] + a[(j-k)%n if valid])
            lo1 = max(0, k - n + 1)
            hi1 = min(n, k + 1)
            for j in range(lo1, hi1):
                grad[j] += w * a[k - j]
            lo2 = max(0, -k)
            hi2 = min(n, n - k)
            for j in range(lo2, hi2):
                grad[j] += w * a[j - k]

        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad = grad / gn

        a_trial = a - step * grad
        a_trial = np.maximum(a_trial, 0.0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / st

        sc = fast_autocorr(a_trial).max()
        if sc < best_score - 1e-15:
            a = a_trial
            best_score = sc
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.003, 0.5)
        else:
            patience += 1
            step *= 0.995
            if step < 1e-13:
                break

        if verbose and t % 5000 == 0:
            print(f"      Subgrad t={t} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def coord_desc(a0, time_limit, verbose=False):
    """
    Coordinate descent: sweep through coordinates, try to improve each one.
    Uses a subset of candidate values for speed.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Step size for sweeping (skip some coords for speed)
    stride = max(1, n // 400)
    no_improve_passes = 0

    for p in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve_passes > 4:
            break

        improved = False
        for j in range(0, n, stride):
            if time.monotonic() - t0 > time_limit:
                break
            aj = a[j]
            # Try: reduce, keep, increase
            for frac in [0.3, 0.6, 1.4, 2.0]:
                x = aj * frac
                if x < 0 or x > norm:
                    continue
                a[j] = x
                s_new = a.sum()
                if s_new < 1e-15:
                    a[j] = aj
                    continue
                a = a * norm / s_new
                sc = fast_autocorr(a).max()
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    aj = x  # update current for next fractions
                else:
                    a[j] = aj
                    a = a * norm / a.sum()

        if not improved:
            no_improve_passes += 1
        else:
            no_improve_passes = 0

        if verbose and p % 20 == 0:
            print(f"      CD pass={p} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def perturb(a, rng, scale=0.02):
    """Multiplicative perturbation."""
    noise = 1.0 + scale * rng.standard_normal(len(a))
    noise = np.maximum(noise, 0.0)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s


def symmetrize(a):
    """Symmetrize sequence."""
    asy = 0.5 * (a + a[::-1])
    s = asy.sum()
    if s > 1e-15:
        asy = asy / s
    return asy
