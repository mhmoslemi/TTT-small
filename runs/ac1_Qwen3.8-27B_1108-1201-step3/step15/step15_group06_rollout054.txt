            best_score = score
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        v = a_new - a
        a = a_new

        # Adaptive parameters
        if stall > 500:
            momentum = max(momentum * 0.995, 0.5)
            lr *= 0.95
            top_k = max(50, int(top_k * 0.9))

        if stall > 3000:
            # Restart from best with perturbation
            a = best_a * (1.0 + 0.02 * rng.standard_normal(n))
            a = np.maximum(a, 0.0)
            a = a * S / a.sum()
            v = np.zeros(n)
            momentum = 0.85
            lr = 0.2
            top_k = 150
            stall = 0

        if verbose and t % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NGD t={t//1000}k ({elapsed:.0f}s): best={best_score:.10f}, lr={lr:.4f}, mom={momentum:.3f}, top_k={top_k}")

    return best_a, best_score


def build_active_constraints(a, active_idx):
    n = len(a)
    m = len(active_idx)
    T = np.zeros((m, n))
    for row, k in enumerate(active_idx):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                T[row, j] = a[idx]
    return T


def lp_max_sum(a, M, solver, active_idx=None):
    n = len(a)
    size = 2 * n - 1
    b = cp.Variable(n, nonneg=True)
    if active_idx is None or len(active_idx) >= size:
        T = build_active_constraints(a, np.arange(size))
    else:
        T = build_active_constraints(a, active_idx)
    cons = [T @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if b.value is None:
        return None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if val.sum() < 1e-15:
        return None
    return val


def lp_min_max(a, solver, active_idx=None, normalize_sum=1.0):
    n = len(a)
    size = 2 * n - 1
    b = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    if active_idx is None or len(active_idx) >= size:
        T = build_active_constraints(a, np.arange(size))
    else:
        T = build_active_constraints(a, active_idx)
    cons = [T @ b <= t, cp.sum(b) == normalize_sum]
    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    if b.value is None:
        return None, None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    t_val = float(t.value) if t.value is not None else float('inf')
    if val.sum() < 1e-15:
        return None, None
    return val, t_val


def nesterov_lp(a0, time_limit, solver, verbose=True, seed=0, beta_blend=0.3):
    """Nesterov-accelerated LP iteration with dual LP blending."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    size = 2 * n - 1
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    stall_reset = 60

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive active set with multiple thresholds
        if active_set is None or t % 30 == 0:
            # Use a wider active set
            tol = 0.05 + 0.03 * np.exp(-t / 200.0)
            threshold = M_curr * (1.0 - tol)
            active_set = np.where(conv_y >= threshold)[0]
            if len(active_set) < 10:
                active_set = np.where(conv_y >= M_curr * 0.6)[0]
            # Also always include lag 0
            if 0 not in active_set:
                active_set = np.concatenate([active_set, [0]])

        # Max-sum LP
        b_max = lp_max_sum(y, M_curr, solver, active_set)

        # Min-max LP for blending
        b_min = None
        if t % 5 == 0:
            b_min, _ = lp_min_max(y, solver, active_set, normalize_sum=norm)

        if b_max is None:
            if t % 20 == 0:
                active_set = None
                b_max = lp_max_sum(y, M_curr, solver, None)
            if b_max is None:
                continue

        b_sum = b_max.sum()
        g = b_max * norm / b_sum

        if b_min is not None and b_min.sum() > 1e-10:
            g_min_norm = b_min * norm / b_min.sum()
            g = (1.0 - beta_blend) * g + beta_blend * g_min_norm

        # Verify constraint violations and expand active set
        if len(active_set) < size:
            inactive = np.setdiff1d(np.arange(size), active_set)
            if len(inactive) > 0:
                T_check = build_active_constraints(y, inactive)
                viol = T_check @ g
                viol_idx = np.where(viol > M_curr * (1.0 + 1e-5))[0]
                if len(viol_idx) > 0:
                    active_set = np.union1d(active_set, inactive[viol_idx])

        # Nesterov update
        alpha_t = max(2.0 / (t + 2.0), 0.05)
        y_new = (1.0 - alpha_t) * y + alpha_t * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.9)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Restart if stalled
        if stall_count > stall_reset:
            y = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall_count = 0

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def fast_coord_descent(a0, time_limit, rng, verbose=True, n_steps_per_coord=5):
    """Fast coordinate descent: try a few fixed step sizes per coordinate."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_score = fast_autocorr(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0

    # Precompute: which autocorrelation lags does each coordinate affect?
    # coord i affects conv[k] for k in [i, i+n-1] (shifted by i)
    # The peak is conv.max(), so we focus on coords that affect the peak

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)
        conv = fast_autocorr(a)
        peak_k = conv.argmax()

        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break

            # Only try coordinates that affect the current peak
            if not (peak_k - n + 1 <= i <= min(n - 1, peak_k)):
                # Also try if coordinate affects near-peak
                if abs(i - peak_k) > n:
                    continue

            ai = a[i]
            if ai < 1e-10 * norm:
                # Try adding mass
                for frac in [0.01, 0.05, 0.1, 0.2, 0.5]:
                    d = norm * frac
                    a_try = a.copy()
                    a_try[i] = d
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        continue
                    a_try = a_try * norm / s_try
                    cv = fast_autocorr(a_try).max()
                    if cv < best_score - 1e-12:
                        a = a_try
                        best_score = cv
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0
                        conv = fast_autocorr(a)
                        peak_k = conv.argmax()
                        break
                continue

            # Try reducing and redistributing
            for frac in [0.1, 0.25, 0.4, 0.6, 0.8]:
                d = ai * frac
                if d <= 0:
                    continue
                a_try = a.copy()
                a_try[i] = ai - d
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                # Redistribute to random other coordinate
                j = perm[(i + 1) % n]
                a_try[j] += d
                a_try = a_try * norm / a_try.sum()
                cv = fast_autocorr(a_try).max()
                if cv < best_score - 1e-12:
                    a = a_try
                    best_score = cv
                    best_a = a.copy()
                    pass_improved = True
                    no_improve = 0
                    conv = fast_autocorr(a)
                    peak_k = conv.argmax()
                    break

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def pair_cd(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent with adaptive step sizes."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        pass_improved = False
        perm = rng.permutation(n)
        conv_cur = fast_autocorr(a)
        peak_k = conv_cur.argmax()

        for idx in range(0, n - 1, 2):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            j = perm[idx + 1]
            ai = a[i]
            aj = a[j]
            if ai < 1e-12 * norm and aj < 1e-12 * norm:
                continue

            # Try moving mass between i and j
            max_d = max(ai, aj)
            if max_d < 1e-12:
                continue

            best_val = best_score
            best_i_val = ai
            best_j_val = aj

            for frac in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                d = max_d * frac
                # Direction: i -> j
                if d <= ai:
                    a_try = a.copy()
                    a_try[i] = ai - d
                    a_try[j] = aj + d
                    a_try = a_try * norm / a_try.sum()
                    cv = fast_autocorr(a_try).max()
                    if cv < best_val - 1e-12:
                        best_val = cv
                        best_i_val = ai - d
                        best_j_val = aj + d
                # Direction: j -> i
                if d <= aj:
                    a_try = a.copy()
                    a_try[j] = aj - d
                    a_try[i] = ai + d
                    a_try = a_try * norm / a_try.sum()
                    cv = fast_autocorr(a_try).max()
                    if cv < best_val - 1e-12:
                        best_val = cv
                        best_i_val = ai + d
                        best_j_val = aj - d

            if best_val < best_score - 1e-12:
                a[i] = best_i_val
                a[j] = best_j_val
                a = a * norm / a.sum()
                best_score = fast_autocorr(a).max()
                best_a = a.copy()
                pass_improved = True
                no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    PCD ({elapsed:.0f}s): best={best_score:.10f}, no_improve={no_improve}")

    return best_a, best_score


def structured_init(n, rng, style=0):
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif style == 1:
        a = np.sin(np.pi * x) ** 1.2
    elif style == 2:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(8 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif style == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif style == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif style == 5:
        a = 0.4 + 0.6 * np.sin(np.pi * x) ** 0.8
    elif style == 6:
        a = 1.0 + 0.25 * np.cos(6 * np.pi * x) + 0.1 * np.cos(12 * np.pi * x)
    else:
        a = np.sin(np.pi * x) ** 0.6 * (1.0 + 0.15 * np.cos(10 * np.pi * x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def perturb_seq(a, rng, scale=0.05):
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.05)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s
