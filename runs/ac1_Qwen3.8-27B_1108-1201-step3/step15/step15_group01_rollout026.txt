def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_fast(a):
    """Fast evaluation without the overhead of the full evaluator."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    n = len(a)
    if n == 0:
        return float('inf')
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_autocorr(a)
    return float(2.0 * n * conv.max() / (s * s))


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n + 1) > (css - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_best_response(a, M, solver, active_indices=None):
    """Solve max sum(b) s.t. conv(a,b) <= M (only active constraints), b >= 0."""
    n = len(a)
    size = 2 * n - 1
    # Build constraint matrix: (size x n) where T[k, i] = a[k-i] if valid
    # For efficiency, use only active constraints
    if active_indices is not None:
        T = np.zeros((len(active_indices), n))
        for idx, k in enumerate(active_indices):
            # (a*b)[k] = sum_i a[k-i]*b[i] for valid k-i
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    T[idx, i] = a[j]
    else:
        T = np.zeros((size, n))
        for i in range(n):
            T[i:i+n, i] = a
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lp_best_response_fast(a, M, solver, top_k=50):

def nesterov_lp(a0, time_limit, solver, verbose=True, top_k_list=None):
    """LP best-response Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stuck_count = 0
    avg_a = np.zeros(n)
    n_avg = 0
    
    if top_k_list is None:
        top_k_list = [50, 100, 150, 200, 300, 400]
    
    # Adaptive top_k: start small, increase over time
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        # Adaptive constraint generation
        progress = min(t / 200.0, 1.0)
        top_k = top_k_list[min(int(progress * len(top_k_list)), len(top_k_list) - 1)]
        
        conv_y = fast_autocorr(y)
        # Use top-k + threshold
        threshold_idx = np.where(conv_y >= 0.7 * M_curr)[0]
        top_idx = np.argsort(conv_y)[::-1][:top_k]
        active = np.unique(np.concatenate([threshold_idx, top_idx]))
        
        b = lp_best_response(y, M_curr, solver, active_indices=active)
        if b is None:
            # Fallback: all constraints (might be slow for large n)
            if time.monotonic() - t0 + 5 < time_limit:
                b = lp_best_response(y, M_curr, solver, active_indices=None)
            if b is None:
                stuck_count += 1
                if stuck_count > 50:
                    break
                continue
        
        stuck_count = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Averaged iterate
        avg_a += a
        n_avg += 1
        if n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    # Check averaged iterate one more time
    if n_avg > 0:
        avg_check = avg_a / n_avg
        s_avg = avg_check.sum()
        if s_avg > 1e-15:
            avg_check = avg_check * norm / s_avg
            score_avg = fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

    return best_a, best_score


def smooth_optimize(a0, time_limit, verbose=True, initial_T=None):
    """Minimize smooth approximation using Nesterov-accelerated projected gradient."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    t0 = time.monotonic()
    
    if initial_T is None:
        initial_T = 10.0  # Start with moderate temperature
        temps = np.array([10.0, 5.0, 2.5, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002])
    else:
        temps = np.array([initial_T, initial_T/2, initial_T/4, initial_T/8, initial_T/16, initial_T/32, initial_T/64, initial_T/128])

    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        n_steps = max(20, min(300, int(remaining * 4)))

        y = a.copy()
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            c = fast_autocorr(y)
            c_max = c.max()
            # Numerically stable softmax
            exp_c = np.exp((c - c_max) / T)
            w = exp_c / exp_c.sum()

            # Gradient: grad_i = 2 * sum_k w_{k+i} * a_k (cross-correlation)
            # This is the cross-correlation of w and a, evaluated at lag i
            size = 2 * n - 1
            fft_size = 1 << (size - 1).bit_length()
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            cross = irfft(W * np.conj(A), fft_size)[:n]
            grad = 2.0 * cross

            # Adaptive step size
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            step = norm / (n * 0.1)  # Fixed step based on problem scale
            # Scale by inverse gradient magnitude for normalization
            step = min(step, 0.3 * norm / max(grad_norm, 1e-10))
            
            a_new = y - step * grad
            a_new = project_simplex(a_new, norm)

            # Nesterov update
            beta = (1.0 + np.sqrt(1.0 + 4.0 * ((it + 1) / (it + 2))**2)) / 2.0
            alpha = (1.0 + 1.0/(it + 2)) / beta
            t_coef = (it + 1) / beta
            
            if it == 0:
                y_new = a_new
                a_curr = a_new
            else:
                y_new = (1.0 - t_coef) * y + t_coef * (a_new + t_coef * (a_new - y))
                y_new = project_simplex(y_new, norm)
                a_curr = y_new

            y = y_new
            a = a_curr

            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

            score_y = fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()

            if verbose and it % 50 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Smooth T={T:.3f} step={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_aware_cd(a0, time_limit, rng, verbose=True, max_window=80):
    """Coordinate descent that focuses on reducing the current peak."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 10

    while time.monotonic() - t0 < time_limit:

def pair_cd(a0, time_limit, rng, verbose=True, window=50):
    """Standard pair coordinate descent with adaptive step."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve >= 5:
            break
        improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-9:
                continue

            # Try moving to nearby positions
            offsets = rng.choice(window, size=min(10, window), replace=False) + 1
            for offset in offsets:
                j1 = (i + offset) % n
                j2 = (i - offset) % n
                for j in [j1, j2]:
                    if j == i or a[i] < norm * 1e-9:
                        continue
                    
                    # Try a moderate step
                    for frac in [0.3, 0.15, 0.05]:
                        delta = a[i] * frac
                        a_test = a.copy()
                        a_test[i] -= delta
                        a_test[j] += delta
                        s_test = a_test.sum()
                        a_test = a_test * norm / s_test
                        conv_test = fast_autocorr(a_test)

                            no_improve = 0
                            break
                if improved:
                    break

                continue

            no_improve += 1

        if verbose and no_improve % 2 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PairCD ({elapsed:.0f}s): best={best_score:.8f}, no_improve={no_improve}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving the sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    return f_interp * np.sqrt(2.0 * new_n) / ss


def make_initializations(n, rng):
    """Generate diverse initializations."""
    inits = []
    norm = np.sqrt(2.0 * n)
    
    # Uniform
    a = np.ones(n) * norm / n
    inits.append(('uniform', a))
    
    # Binary random
    a_bin = rng.choice([0.0, 1.0], size=n)
    if a_bin.sum() > 0:
        a_bin = a_bin * norm / a_bin.sum()
        inits.append(('binary_random', a_bin))
    
    # Binary with 50% density
    a_bin2 = (rng.random(n) < 0.5).astype(float)
    if a_bin2.sum() > 0:
        a_bin2 = a_bin2 * norm / a_bin2.sum()
        inits.append(('binary_50', a_bin2))
    
    # Sinusoidal
    x = np.arange(n) / n
    a_sin = 1.0 + 0.5 * np.cos(2 * np.pi * x)
    a_sin = np.maximum(a_sin, 0.0)
    if a_sin.sum() > 0:
        a_sin = a_sin * norm / a_sin.sum()
        inits.append(('sinusoidal', a_sin))
    
    # Gaussian-like
    a_gauss = np.exp(-0.5 * ((x - 0.5) / 0.2) ** 2)
    a_gauss = a_gauss * norm / a_gauss.sum()
    inits.append(('gaussian', a_gauss))
    
    # Triangular
    a_tri = np.minimum(x, 1 - x) * 2 + 0.1
    a_tri = a_tri * norm / a_tri.sum()
    inits.append(('triangular', a_tri))
    
    return inits

    """Compute full autocorrelation via FFT."""
    n = len(a)

    """Fast evaluation of the objective (mimics evaluate_sequence)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return float('inf')
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_autocorr(a)
    return float(2.0 * n * conv.max() / (s * s))


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n + 1) > (css - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_rows(a, indices):
    """Build rows of the convolution matrix T where (T @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    T = np.zeros((len(indices), n))
    for idx, k in enumerate(indices):
        i_min = max(0, k - n + 1)
        i_max = min(n, k + 1)
        for i in range(i_min, i_max):
            T[idx, i] = a[k - i]
    return T


def lp_best_response(a, M, solver, active_indices):
    """Solve max sum(b) s.t. conv(a,b) <= M (active constraints only), b >= 0."""
    n = len(a)
    T = build_conv_rows(a, active_indices)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp(a0, time_limit, solver, verbose=True, top_k_schedule=None):
    """
    LP best-response Frank-Wolfe with Nesterov acceleration.
    Uses adaptive constraint generation: starts with few constraints, increases over time.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stuck_count = 0
    avg_a = np.zeros(n)
    n_avg = 0

    if top_k_schedule is None:
        top_k_schedule = [20, 40, 60, 100, 150, 200, 300, 400, 600]

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        # Adaptive constraint generation
        progress = min(t / 500.0, 1.0)
        k_idx = min(int(progress * len(top_k_schedule)), len(top_k_schedule) - 1)
        top_k = top_k_schedule[k_idx]

        conv_y = fast_autocorr(y)
        # Combine threshold-based and top-k selection
        threshold_mask = conv_y >= 0.6 * M_curr
        top_k_mask = np.zeros(len(conv_y), dtype=bool)
        top_idx = np.argsort(conv_y)[::-1][:top_k]
        top_k_mask[top_idx] = True
        active = np.where(threshold_mask | top_k_mask)[0]

        # Ensure we always include the center
        if (n - 1) not in active:
            active = np.append(active, n - 1)

        b = lp_best_response(y, M_curr, solver, active)
        if b is None:
            # Try with fewer constraints
            fewer_k = max(10, top_k // 2)
            fewer_idx = np.argsort(conv_y)[::-1][:fewer_k]
            if (n - 1) not in fewer_idx:
                fewer_idx = np.append(fewer_idx, n - 1)
            b = lp_best_response(y, M_curr, solver, fewer_idx)
            if b is None:
                stuck_count += 1
                if stuck_count > 100:
                    break
                continue

        stuck_count = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        # Track best
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Averaged iterate (Polyak average)
        avg_a += a
        n_avg += 1
        if n_avg % 5 == 0 or n_avg < 20:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    # Final check of average
    if n_avg > 0:
        avg_check = avg_a / n_avg
        s_avg = avg_check.sum()
        if s_avg > 1e-15:
            avg_check = avg_check * norm / s_avg
            score_avg = fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

    return best_a, best_score


def smooth_optimize(a0, time_limit, verbose=True, initial_T=5.0):
    """Minimize smooth approximation with Nesterov-accelerated projected gradient."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    t0 = time.monotonic()

    # Adaptive temperature schedule
    temps = np.concatenate([
        np.array([initial_T, initial_T / 2, initial_T / 4, initial_T / 8]),
        np.array([0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001])
    ])
    # Remove duplicates and sort
    temps = np.unique(np.sort(temps)[::-1])

    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        n_steps = max(20, min(400, int(remaining * 3.5)))

        y = a.copy()
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            c = fast_autocorr(y)
            c_max = c.max()
            exp_c = np.exp((c - c_max) / T)
            w = exp_c / exp_c.sum()

            # Gradient computation via cross-correlation
            size = 2 * n - 1
            fft_size = 1 << (size - 1).bit_length()
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            cross = irfft(W * np.conj(A), fft_size)[:n]
            grad = 2.0 * cross

            # Adaptive step size
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            step = 0.5 * norm / (n * 0.1)
            step = min(step, 0.5 * norm / max(grad_norm, 1e-10))

            a_new = y - step * grad
            a_new = project_simplex(a_new, norm)

            # Nesterov momentum
            if it == 0:
                y_new = a_new
                a_curr = a_new
            else:
                beta = (1.0 + np.sqrt(1.0 + 4.0 * ((it) / (it + 1.0)) ** 2)) / 2.0
                t_old = it / beta
                y_new = (1.0 - (it + 1.0) / beta) * y + ((it + 1.0) / beta) * (a_new + t_old * (a_new - y))
                y_new = project_simplex(y_new, norm)
                a_curr = y_new

            y = y_new
            a = a_curr

            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

            score_y = fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()

            if verbose and it % 50 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Smooth T={T:.4f} step={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_aware_cd(a0, time_limit, rng, verbose=True):
    """Coordinate descent focused on reducing the current peak."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve >= 8:
            break

        # Find peak and its contributors
        peak_pos = np.argmax(conv)
        peak_val = conv[peak_pos]

        # Indices contributing to peak: i where both i and peak_pos-i are in [0,n)
        valid_i = np.arange(max(0, peak_pos - n + 1), min(n, peak_pos + 1))
        if len(valid_i) == 0:
            break

        # Sort by contribution a[i]*a[peak_pos-i]
        contribs = a[valid_i] * a[peak_pos - valid_i]
        sorted_valid = valid_i[np.argsort(contribs)[::-1]]

        # Non-contributing positions
        all_idx = set(range(n))
        contrib_set = set(valid_i.tolist())
        non_contrib = np.array([i for i in all_idx if i not in contrib_set])

        if len(non_contrib) == 0:
            break

        improved = False
        # Try moving from top contributors to random non-contributors
        for i in sorted_valid[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < norm * 1e-8:
                continue

            n_try = min(8, len(non_contrib))
            j_candidates = non_contrib[rng.choice(len(non_contrib), size=n_try, replace=False)]

            for j in j_candidates:
                for frac in [0.4, 0.2, 0.1, 0.05]:
                    delta = a[i] * frac
                    a_test = a.copy()
                    a_test[i] -= delta
                    a_test[j] += delta
                    s_test = a_test.sum()
                    a_test = a_test * norm / s_test
                    conv_test = fast_autocorr(a_test)
                    if conv_test.max() < best_score - 1e-10:
                        a = a_test
                        conv = conv_test
                        best_score = conv.max()
                        best_a = a.copy()
                        improved = True
                        no_improve = 0
                        break
                if improved:
                    break
            if improved:
                continue

        if not improved:
            no_improve += 1

        if verbose and no_improve % 2 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PACD ({elapsed:.0f}s): best={best_score:.8f}, no_improve={no_improve}")

    return best_a, best_score


def pair_cd(a0, time_limit, rng, verbose=True, window=60):
    """Pair coordinate descent with random offsets."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve >= 6:
            break
        improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-9:
                continue

            n_offsets = min(15, window)
            offsets = rng.integers(1, window + 1, size=n_offsets)
            for offset in offsets:
                for j in [(i + offset) % n, (i - offset) % n]:
                    if j == i or a[i] < norm * 1e-9:
                        continue
                    for frac in [0.25, 0.1, 0.04]:
                        delta = a[i] * frac
                        a_test = a.copy()
                        a_test[i] -= delta
                        a_test[j] += delta
                        s_test = a_test.sum()
                        a_test = a_test * norm / s_test
                        conv_test = fast_autocorr(a_test)
                        if conv_test.max() < conv.max() - 1e-12:
                            a = a_test
                            conv = conv_test
                            if conv.max() < best_score:
                                best_score = conv.max()
                                best_a = a.copy()
                            improved = True
                            no_improve = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                continue

        if not improved:
            no_improve += 1

        if verbose and no_improve % 2 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD ({elapsed:.0f}s): best={best_score:.8f}, no_improve={no_improve}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    return f_interp * np.sqrt(2.0 * new_n) / ss


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function. Returns best sequence found within time budget."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_fast(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_fast(a)
        if np.isfinite(sc) and sc < best_score - 1e-10:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: score={sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_fast(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: LP Nesterov on parent (35% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.35
    n_c = len(best_seq)
    print(f"\n=== Phase 1: LP Nesterov (n={n_c}, budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t1, solver, verbose=True,
                               top_k_schedule=[20, 40, 60, 100, 150, 250, 400, 600, 800])
        update_best(a_out, f"lp1_n{n_c}")

    # === Phase 2: LP Nesterov on binary init (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = remaining * 0.15
    n_c = len(best_seq)
    print(f"\n=== Phase 2: LP Nesterov (binary, n={n_c}, budget={t2:.0f}s) ===")
    if t2 > 20:
        a_bin = (rng.random(n_c) < 0.5).astype(float)
        a_bin = a_bin * np.sqrt(2.0 * n_c) / a_bin.sum()
        a_out, _ = nesterov_lp(a_bin, t2, solver, verbose=True,
                               top_k_schedule=[20, 50, 100, 200, 400])
        update_best(a_out, f"lp_bin_n{n_c}")

    # === Phase 3: Smooth gradient (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = remaining * 0.15
    n_c = len(best_seq)
    print(f"\n=== Phase 3: Smooth grad (n={n_c}, budget={t3:.0f}s) ===")
    if t3 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = smooth_optimize(a_norm, t3, verbose=True, initial_T=5.0)
        update_best(a_out, f"smooth_n{n_c}")

    # === Phase 4: LP Nesterov round 2 (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = remaining * 0.20
    n_c = len(best_seq)
    print(f"\n=== Phase 4: LP Nesterov r2 (n={n_c}, budget={t4:.0f}s) ===")
    if t4 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t4, solver, verbose=True,
                               top_k_schedule=[50, 100, 200, 400, 600, 1000])
        update_best(a_out, f"lp2_n{n_c}")

    # === Phase 5: Peak-aware CD + Smooth polish (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = remaining * 0.10
    n_c = len(best_seq)
    print(f"\n=== Phase 5: Polish (n={n_c}, budget={t5:.0f}s) ===")
    if t5 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        t5a = t5 * 0.5
        a_out, _ = peak_aware_cd(a_norm, t5a, rng, verbose=True)
        update_best(a_out, f"pacd_n{n_c}")

        remaining5 = time_budget - (time.monotonic() - t_global)
        if remaining5 > 10:
            n_c = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
            a_out, _ = smooth_optimize(a_norm, min(remaining5 * 0.6, 60.0), verbose=False, initial_T=1.0)
            update_best(a_out, "polish_smooth")

    # === Phase 6: Final CD + length exploration (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30 and best_seq is not None:
        n_cur = len(best_seq)
        # Try alternate lengths
        alt_lengths = [n_cur + 100, n_cur - 100, n_cur + 200]
        alt_lengths = [l for l in alt_lengths if l >= 256 and l != n_cur]
        for n_alt in alt_lengths[:2]:
            remaining_sub = time_budget - (time.monotonic() - t_global)
            if remaining_sub < 20:
                break
            t_alt = min(remaining_sub * 0.4, 90.0)
            print(f"\n=== Alt length n={n_alt} (budget={t_alt:.0f}s) ===")
            init_alt = interpolate_sequence(best_seq, n_alt)
            a_out, _ = nesterov_lp(init_alt, t_alt, solver, verbose=False,
                                   top_k_schedule=[40, 80, 150, 300])
            update_best(a_out, f"alt_n{n_alt}")

        # Final pair CD
        remaining_sub = time_budget - (time.monotonic() - t_global)
        if remaining_sub > 15:
            n_c = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
            a_out, _ = pair_cd(a_norm, min(remaining_sub * 0.7, 45.0), rng, verbose=False)
            update_best(a_out, "final_pcd")

    # Final verification with actual evaluator
    if best_seq is not None:
        try:
            final_score = evaluate_sequence([float(x) for x in best_seq])
            if np.isfinite(final_score):
                best_score = final_score
        except Exception:
            pass
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
