def optimal_transfer_1d(a, i, j, conv, d_max):
    """Find optimal d to transfer from position i to position j."""
    n = len(a)
    size = 2 * n - 1
    
    # Compute g[k] = 2*(a[k-j] - a[k-i]) with boundary handling
    g = np.zeros(size)
    for k in range(size):
        val = 0.0
        if 0 <= k - j < n:
            val += a[k - j]
        if 0 <= k - i < n:
            val -= a[k - i]
        g[k] = 2.0 * val
    
    # Quadratic corrections
    q_coeffs = {}  # k -> coefficient of d^2
    if 2 * i < size:
        q_coeffs[2 * i] = q_coeffs.get(2 * i, 0) + 1.0
    if 2 * j < size:
        q_coeffs[2 * j] = q_coeffs.get(2 * j, 0) + 1.0
    if (i + j) < size and i != j:
        q_coeffs[i + j] = q_coeffs.get(i + j, 0) - 2.0
    
    # We want to minimize max_k (conv[k] + d*g[k] + q_coeff[k]*d^2) for d in [0, d_max]
    # This is convex in d (max of convex functions)
    # Use golden section search
    
    def f(d):
        vals = conv + d * g
        for k, qc in q_coeffs.items():
            vals[k] += qc * d * d
        return vals.max()
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2  # golden ratio
    lo, hi = 0.0, d_max
    c = hi - gr * (hi - lo)
    d2 = lo + gr * (hi - lo)
    fc = f(c)
    fd2 = f(d2)
    
    for _ in range(50):
        if fc < fd2:
            hi = d2
            d2 = c
            fd2 = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d2
            fc = fd2
            d2 = lo + gr * (hi - lo)
            fd2 = f(d2)
    
    d_opt = (lo + hi) / 2
    return d_opt, f(d_opt)

def compute_transfer_gradient(a, i, j):
    """Compute g[k] = 2*(a[k-j] - a[k-i]) for transfer from i to j."""
    n = len(a)
    size = 2 * n - 1
    g = np.zeros(size)
    lo_j = j
    hi_j = min(j + n, size)
    g[lo_j:hi_j] += a[:hi_j - lo_j]
    g[lo_j:hi_j] *= 2.0
    lo_i = i
    hi_i = min(i + n, size)
    g[lo_i:hi_i] -= a[:hi_i - lo_i]
    g[lo_i:hi_i] *= 2.0
    return g


def golden_section_1d(conv, g, q_indices, q_coeffs, d_max, n_iter=25):
    """Minimize max(conv + d*g + sum q_coeffs[k]*d^2) over d in [0, d_max]."""
    if d_max <= 0:
        return 0.0, conv.max()

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = d_max
    c = hi - gr * (hi - lo)
    d2 = lo + gr * (hi - lo)

    vals_c = conv + c * g
    for k, qc in zip(q_indices, q_coeffs):
        vals_c[k] += qc * c * c
    fc = vals_c.max()

    vals_d2 = conv + d2 * g
    for k, qc in zip(q_indices, q_coeffs):
        vals_d2[k] += qc * d2 * d2
    fd2 = vals_d2.max()

    for _ in range(n_iter):
        if fc < fd2:
            hi = d2
            d2 = c
            fd2 = fc
            c = hi - gr * (hi - lo)
            vals_c = conv + c * g
            for k, qc in zip(q_indices, q_coeffs):
                vals_c[k] += qc * c * c
            fc = vals_c.max()
        else:
            lo = c
            c = d2
            fc = fd2
            d2 = lo + gr * (hi - lo)
            vals_d2 = conv + d2 * g
            for k, qc in zip(q_indices, q_coeffs):
                vals_d2[k] += qc * d2 * d2
            fd2 = vals_d2.max()

    d_opt = (lo + hi) / 2.0
    vals_opt = conv + d_opt * g
    for k, qc in zip(q_indices, q_coeffs):
        vals_opt[k] += qc * d_opt * d_opt
    return d_opt, vals_opt.max()


def pair_cd_refine(a0, time_limit, rng, verbose=False):
    """Pair coordinate descent with exact 1D golden-section optimization."""
    n = len(a0)

    total_improvements = 0

        pass_improved = False

        # Focus on positions that contribute to peaks
        peak_threshold = cur_max * 0.90

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-12:
                continue

            # Candidate targets: random + structured
            j_candidates = set()
            for _ in range(8):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:5]:
                j1 = (kl - i + n) % n
                j2 = (kl + i) % n
                if j1 != i:
                    j_candidates.add(j1)
                if j2 != i:
                    j_candidates.add(j2)
            j_candidates.discard(i)

            best_d = 0.0
            best_val = cur_max
            best_j = -1

                if j == i:

                aj = a[j]
                if aj < 1e-15:
                    continue

                g = compute_transfer_gradient(a, i, j)

                # Quadratic corrections
                q_indices = []
                q_coeffs = []
                if 2 * i < size:
                    q_indices.append(2 * i)
                    q_coeffs.append(1.0)
                if 2 * j < size:
                    q_indices.append(2 * j)
                    q_coeffs.append(1.0)
                ij_sum = i + j
                if ij_sum < size and i != j:
                    q_indices.append(ij_sum)
                    q_coeffs.append(-2.0)

                d_max = ai * 0.95
                d_opt, v_opt = golden_section_1d(conv, g, q_indices, q_coeffs, d_max, n_iter=20)

                if v_opt < best_val - 1e-10 and d_opt > 1e-12:
                    best_val = v_opt
                    best_d = d_opt
                    best_j = j

            if best_j >= 0 and best_d > 1e-12:
                a[i] -= best_d
                a[best_j] += best_d
                s_new = a.sum()
                a = a * norm / s_new
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True

                conv = fast_conv_self(a)
                cur_max = conv.max()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 6:
                break
            if no_improve >= 2:
                n_perturb = max(10, n // 25)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.2 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and total_improvements % 500 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD imp={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

        cur_max = conv.max()

        # Find the single worst peak
        worst_lag = conv.argmax()
        # Find positions that contribute most to this peak
        contrib = np.zeros(n)
        for m in range(n):
            if 0 <= worst_lag - m < n:
                contrib[m] += a[worst_lag - m]
            if 0 <= m - worst_lag < n:
                contrib[m] += a[m - worst_lag]

        # Sort by contribution (highest first)
        order = np.argsort(contrib)[::-1]

        pass_improved = False
        for idx in range(min(200, n)):
            if time.monotonic() - t0 > time_limit:
                break
            i = order[idx]
            ai = a[i]
            if ai < num * 1e-12:
                continue

            # Find best target with lowest contribution to peak
            j_candidates = np.argsort(contrib)[:50]
            j_candidates = j_candidates[j_candidates != i]
            if len(j_candidates) == 0:
                continue
            j_candidates = j_candidates[:15]

            best_d = 0.0
            best_val = cur_max
            best_j = -1

            for j in j_candidates:
                if a[j] > ai * 0.9:
                    continue
                g = compute_transfer_gradient(a, i, j)
                q_indices = []
                q_coeffs = []
                if 2 * i < size:
                    q_indices.append(2 * i)
                    q_coeffs.append(1.0)
                if 2 * j < size:
                    q_indices.append(2 * j)
                    q_coeffs.append(1.0)
                ij_sum = i + j
                if ij_sum < size and i != j:
                    q_indices.append(ij_sum)
                    q_coeffs.append(-2.0)

                d_max = min(ai * 0.9, (cur_max - conv.min()) / max(1.0, np.abs(g).max() + 1e-15))
                d_max = min(d_max, ai * 0.9)
                if d_max <= 0:
                    continue
                d_opt, v_opt = golden_section_1d(conv, g, q_indices, q_coeffs, d_max, n_iter=20)
                if v_opt < best_val - 1e-10 and d_opt > 1e-12:
                    best_val = v_opt
                    best_d = d_opt
                    best_j = j

            if best_j >= 0 and best_d > 1e-12:
                a[i] -= best_d
                a[best_j] += best_d
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * num / s_new
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
            if no_improve >= 2:
                n_perturb = max(5, n // 50)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.15 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * num / s_new
        else:
            no_improve = 0

        if verbose and (idx + 1) % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakDir ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.exp(-8.0 * x) + 0.3
    elif choice == 1:
        a = np.exp(-8.0 * (1.0 - x)) + 0.3
    elif choice == 2:
        a = np.exp(-5.0 * x) * np.exp(-5.0 * (1.0 - x)) + 0.1
    elif choice == 3:
        a = np.cos(np.pi * x * 0.7) ** 2 + 0.1
    elif choice == 4:
        a = np.sin(np.pi * x * 0.4) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-12.0 * (x - 0.2) ** 2) + 0.5 * np.exp(-8.0 * (x - 0.8) ** 2)
    elif choice == 6:
        a = 0.4 + 0.6 * np.exp(-6.0 * x)
    elif choice == 7:
        a = 0.4 + 0.6 * np.exp(-6.0 * (1.0 - x))
    elif choice == 8:
        a = np.sinc(4.0 * (x - 0.3)) ** 2 + 0.1
    elif choice == 9:
        a = 1.0 / (1.0 + np.exp(-10.0 * (x - 0.5)))
    elif choice == 10:
        a = np.exp(-3.0 * (x - 0.3) ** 2 / 0.05) + 0.3 * np.exp(-3.0 * (x - 0.7) ** 2 / 0.03)
    else:
        a = 0.3 + 0.7 * np.exp(-4.0 * (1.0 - x))
    a = np.maximum(a, 0.0)
    a = a + 0.005 * rng.random(n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        if len(a_list) == 0:
            return float('inf')
        for val in a_list:
            if val < 0 or not np.isfinite(val):
                return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LSE GD warmup
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.04, 60.0)
    print(f"\n=== Phase 1: LSE GD n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 10:
        a_work = best_seq.copy()
        a_out, _ = lse_gd(a_work, t1, verbose=True)
        try_update(a_out, "lse1")

    # Phase 2: LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq)
    t2 = min(remaining * 0.40, 650.0)
    print(f"\n=== Phase 2: LP Nesterov n={n2} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, solver, verbose=True)
        try_update(a_out, "lp_main")

    # Phase 3: Pair CD with exact 1D
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = len(best_seq)
    t3 = min(remaining * 0.18, 280.0)
    print(f"\n=== Phase 3: Pair CD n={n3} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = pair_cd_refine(best_seq, t3, rng, verbose=True)
        try_update(a_out, "pcd1")

    # Phase 4: Peak-directed refinement
    remaining = time_budget - (time.monotonic() - t_global)
    n4 = len(best_seq)
    t4 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 4: Peak Directed n={n4} (budget={t4:.0f}s) ===")
    if t4 > 15:
        a_out, _ = peak_directed_refine(best_seq, t4, rng, verbose=False)
        try_update(a_out, "peakdir")

    # Phase 5: LP Nesterov round 2
    remaining = time_budget - (time.monotonic() - t_global)
    n5 = len(best_seq)
    t5 = min(remaining * 0.18, 280.0)
    print(f"\n=== Phase 5: LP Nesterov R2 n={n5} (budget={t5:.0f}s) ===")
    if t5 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t5, solver, verbose=True)
        try_update(a_out, "lp_r2")

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Multi-length (budget={remaining:.0f}s) ===")
    if remaining > 150:
        lengths_to_try = []
        # Try slightly larger and slightly smaller
        lengths_to_try.append(n_base + 300)
        lengths_to_try.append(n_base + 600)
        if n_base > 200:
            lengths_to_try.append(n_base - 200)

        for n_try in lengths_to_try:
            if remaining < 60:
                break
            if n_try < 100 or n_try > 5000:
                continue
            t_alloc = min(remaining * 0.35, 120.0)
            if t_alloc < 20:
                continue
            init_try = interpolate_seq(best_seq, n_try)
            # Quick LSE warmup
            t_warm = min(t_alloc * 0.3, 30.0)
            a_warm, _ = lse_gd(init_try, t_warm, verbose=False)
            # LP optimization
            t_lp = min(t_alloc - t_warm - 5, t_alloc * 0.5)
            if t_lp > 15:
                a_out, _ = nesterov_lp_optimize(a_warm, t_lp, solver, verbose=False)
                try_update(a_out, f"lp_n{n_try}")
            # Quick CD
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20:
                t_cd = min(remaining * 0.1, 20.0)
                if t_cd > 5:
                    a_out2, _ = pair_cd_refine(a_warm, t_cd, rng, verbose=False)
                    try_update(a_out2, f"cd_n{n_try}")

    # Phase 7: Final aggressive polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 7: Final Polish n={n_final} (budget={remaining:.0f}s) ===")
    if remaining > 30:
        # Pair CD
        t_cd_final = min(remaining * 0.5, 150.0)
        if t_cd_final > 10:
            a_out, _ = pair_cd_refine(best_seq, t_cd_final, rng, verbose=True)
            try_update(a_out, "pcd_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            # LSE refinement
            t_lse_final = min(remaining * 0.4, 60.0)
            if t_lse_final > 10:
                a_out, _ = lse_gd(best_seq, t_lse_final, verbose=False)
                try_update(a_out, "lse_final")

            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 15:
                a_out, _ = pair_cd_refine(best_seq, remaining * 0.5, rng, verbose=False)
                try_update(a_out, "pcd_final2")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
