def fast_conv_self(a):
    """Compute self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1

    A = rfft(a, fft_size)

def lse_adam_optimize(a0, time_limit, verbose=True):
    """LSE-smoothed Adam optimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

    n_stages = 25
    temps = np.geomspace(100.0, 0.02, n_stages)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:

            break

        lr = 0.3 / np.sqrt(n)
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            t_adam += 1
            
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            
            # LSE smoothing
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)

            w = exp_c / exp_c.sum()

            A = rfft(a, fft_size)
            
            # Gradient of LSE(conv) w.r.t. a
            ca = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (ca + aw)
            
            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_adam)
            v_hat = v / (1.0 - beta2 ** t_adam)
            
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            if s_new < 1e-15:

            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

def build_constraint_matrix(a, active_indices):
    """Build the constraint matrix T where T @ b = conv(a, b) at active indices."""
    n = len(a)
    T = np.zeros((len(active_indices), n))
    for idx, k in enumerate(active_indices):
        # conv(a,b)[k] = sum_i a[i] * b[k-i] where 0<=i<n and 0<=k-i<n
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            T[idx, i] = a[k - i]
    return T
