"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed gradient descent with FFT-based O(n log n) gradient computation
2. Nesterov acceleration with momentum schedule
3. Temperature annealing from smooth (softmax) to hard max
4. Multiple restarts with diverse structured initializations
5. Local refinement via coordinate descent and pair swaps
6. Multi-length exploration with interpolation
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_smoothed_grad(a, weights):
    """Compute gradient of smoothed max autocorrelation via FFT.
    
    grad[j] = 2 * sum_{m=0}^{n-1} a[m] * weights[j+m]
    This is 2 * cross_correlation(a, weights)[j]
    """
    n = len(a)
    # Need FFT size >= n + len(weights) - 1 = n + (2n-1) - 1 = 3n-2
    N = 1 << (3 * n - 2).bit_length()
    A = rfft(a, N)
    W = rfft(weights, N)
    # Linear cross-correlation via FFT
    corr = irfft(A * np.conj(W), N)
    return 2.0 * corr[:n]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    nz = np.nonzero(cond)[0]
    if len(nz) == 0:
        return np.ones(n) / n
    rho = nz[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_weights(conv, temperature):
    """Compute softmax weights over convolution values with numerical stability."""
    shifted = conv - conv.max()
    # Scale to avoid overflow
    max_shifted = shifted.max()
    if max_shifted > 50:
        shifted = shifted - max_shifted
        exp_vals = np.exp(shifted / temperature)
    else:
        exp_vals = np.exp(shifted / temperature)
    s = exp_vals.sum()
    if s < 1e-300:
        return np.full_like(conv, 1.0 / len(conv))
    return exp_vals / s


def evaluate_local(a):
    """Fast local evaluation (no security checks)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_autocorr(a)
    return float(2.0 * len(a) * conv.max() / (s * s))

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss
