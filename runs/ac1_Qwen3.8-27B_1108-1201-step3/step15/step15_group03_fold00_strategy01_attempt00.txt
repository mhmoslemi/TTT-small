def fast_conv(sequence):
    n = len(sequence)
    fft_size = 1 << (2 * n).bit_length()
    A = np.fft.rfft(sequence, fft_size)
    # Auto-correlation: [0, 2*n-2]
    conv_ = np.fft.irfft(A * A, fft_size)[:2*n-1]
    # Only real parts are finite (we assume)
    return conv_
