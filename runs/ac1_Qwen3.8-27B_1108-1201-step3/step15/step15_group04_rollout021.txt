def build_toeplitz(a, active_indices):
    """Build T matrix where T[idx, i] = a[k-i] if valid, for k = active_indices[idx]."""
    n = len(a)
    T = np.zeros((len(active_indices), n))

        # T[idx, i] = a[k - i] if 0 <= k-i < n
        lo = max(0, k - (n - 1))
        hi = min(n, k + 1)
        if hi > lo:
            i_arr = np.arange(lo, hi)
            a_idx = k - i_arr
            T[idx, i_arr] = a[a_idx]
    return T

            # For our purpose, g1 and g2 above should give us what we need.
            # Let me verify: the subgradient of max(a*a) at the active set S* is:
            # grad_i = 2 * sum_{k in S*} a[k-i] (with a_m = 0 for invalid m)
            # With the smooth weight w (approximating uniform on S*), we get:
            # g_i = 2 * sum_k w[k] * a[k-i]
            
            # Using FFT: c[i] = sum_k w[k] * a[k-i]
            # Pad w to length fft_size, pad a to length fft_size
            # c = IFFT(FFT(w_padded) * conj(FFT(a_padded)))... 
            # Actually the standard cross-correlation formula:
            # If c[i] = sum_m x[m] * y[m-i], then C = IFFT(X * conj(Y))
            # So c[i] = sum_m w[m] * a[m-i] => C = IFFT(W * conj(A))
            
            c = irfft(W * np.conj(A), fft_size)[:n]
            grad = 2.0 * c
            
            # Project gradient onto feasible directions (nonnegativity + sum constraint)
            # Simple approach: subtract mean, then take positive part, renormalize
            grad_centered = grad - grad.mean()
            
            # Adaptive step size
            grad_norm = np.linalg.norm(grad_centered)
            if grad_norm < 1e-15:
                break
            step = 0.5 / (1.0 + 0.01 * it)
            
            for trial in range(10):
                a_new = a - step * grad_centered
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    step *= 0.5
                    continue
                a_new = a_new * norm / s_new
                sc = fast_conv_self(a_new).max()
                if sc <= max_c + 0.01 * T_val:
                    break
                step *= 0.5
            
            a = np.maximum(a - step * grad_centered, 0.0)
            s_new = a.sum()
            if s_new < 1e-15:
                a = best_a.copy()
            else:
                a = a * norm / s_new
            
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    PSF stage {si} T={T_val:.1f}: best={best_score:.8f}")

    return best_a, best_score


def lp_nesterov_optimize(a0, time_limit, verbose=True, use_adaptive_active=True):
    """
    LP best-response with Nesterov acceleration and adaptive active set.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    y = a.copy()  # Nesterov lookahead
    no_improve = 0
    best_duals = None

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Adaptive active set
        if use_adaptive_active:
            # Start with top constraints, grow if needed
            if t < 20:
                threshold = 0.3
            elif t < 100:
                threshold = 0.5
            elif t < 500:
                threshold = 0.65
            else:
                threshold = 0.75
            active = np.where(conv_y > M * threshold)[0]
            if len(active) < 20:
                active = np.where(conv_y > M * 0.2)[0]
        else:
            active = np.arange(size)

        b, duals = lp_best_response(y, M, active)
        if b is None:
            # Fallback: full constraint set
            b, duals = lp_best_response(y, M, np.arange(size))
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Verify: check if dropped constraints are violated
        if use_adaptive_active and len(active) < size:
            check_conv = fast_conv_cross(y, g)
            max_viol = check_conv.max()
            if max_viol > M * (1 + 0.01):
                # Re-solve with more constraints
                threshold2 = max(0.1, (max_viol - M) / max_viol - 0.1)
                active2 = np.where(conv_y > M * threshold2)[0]
                if len(active2) < size:
                    active2 = np.arange(size)
                b2, duals2 = lp_best_response(y, M, active2)
                if b2 is not None and b2.sum() > 1e-15:
                    b = b2
                    g = b * norm / b.sum()
                    if duals2 is not None:
                        duals = duals2

        if duals is not None:
            best_duals = duals

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Nesterov extrapolation for a
        if t == 1:
            a_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            a_new = y_new + beta * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        # Track best
        sc_y = fast_conv_self(y).max()
        improved = False
        if sc_y < best_score - 1e-12:
            best_score = sc_y
            best_a = y.copy()
            improved = True

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score - 1e-12:
            best_score = sc_a
            best_a = a.copy()
            improved = True

        if not improved:
            no_improve += 1
        else:
            no_improve = 0

        # Perturbation to escape plateaus
        if no_improve == 150:
            rng_p = np.random.default_rng(t)
            n_pert = max(20, n // 10)
            idx_p = rng_p.choice(n, size=n_pert, replace=False)
            y[idx_p] *= (1.0 + 0.3 * rng_p.standard_normal(n_pert))
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y * norm / s_y
            no_improve = 0

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M:.6f}")

    return best_a, best_score


def dual_guided_cd(a0, time_limit, duals=None, verbose=True):
    """
    Pair coordinate descent guided by LP dual variables.
    Focus on pairs that affect the most-tight constraints.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_score(a)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    rng = np.random.default_rng(12345)
    no_improve_passes = 0
    total_moves = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        peak_lags = np.where(conv >= cur_max * 0.85)[0]

        # Select candidate i's based on dual weights
        if duals is not None and duals.sum() > 0:
            # Weight for index i: sum of duals at lags k where a[k-i] is large
            w_i = np.zeros(n)
            for k in peak_lags[:20]:
                for i in range(n):
                    if 0 <= k - i < n:
                        w_i[i] += duals[k] * a[k - i]
            # Pick i's with high weight (they contribute most to tight constraints)
            top_i = np.where(w_i > 0.5 * w_i.max())[0]
        else:
            top_i = np.arange(n)

        if len(top_i) == 0:
            top_i = np.arange(n)

        perm = rng.permutation(len(top_i))
        pass_improved = False

        for idx in perm:
            if time.monotonic() - t0 > time_limit:
                break
            i = top_i[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            # Find best j to transfer mass to
            # Prioritize j's that reduce the peak lags
            candidates = set()
            for _ in range(15):
                candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:10]:
                j1 = (kl - i) % n
                j2 = (kl + i) % n
                if 0 <= j1 < n:
                    candidates.add(j1)
                if 0 <= j2 < n:
                    candidates.add(j2)
            candidates.discard(i)

            best_delta = 0.0
            best_conv_max = cur_max
            best_j = -1

            for j in candidates:
                if j == i:
                    continue
                aj = a[j]

                # Gain vector: change in conv when moving delta from i to j
                g = np.zeros(size)
                # Removing from i: conv[k] -= ai * a[k-i] and a[i] * a[k-i]
                # Adding to j: conv[k] += ai * a[k-j] and a[j] * a[k-j]... 
                # Wait, we're moving delta from i to j:
                # a[i] -> a[i] - delta, a[j] -> a[j] + delta
                # conv_new[k] = conv[k] - delta*(a[k-i] + a[i-k]) + delta*(a[k-j] + a[j-k]) + higher order
                # (ignoring delta^2 terms for small delta)
                
                for kl in range(size):
                    val = 0.0
                    if 0 <= kl - i < n:
                        val -= a[kl - i]
                    if 0 <= i - kl < n:
                        val -= a[i - kl]
                    if 0 <= kl - j < n:
                        val += a[kl - j]
                    if 0 <= j - kl < n:
                        val += a[j - kl]
                    g[kl] = delta_placeholder if False else val  # placeholder

                # Actually let me compute g properly:
                # d(conv[k])/d(delta) at delta=0 when moving from i to j:
                # = -(a[k-i] + a[i-k]) + (a[k-j] + a[j-k])
                # where a[m] = 0 if m < 0 or m >= n
                g_val = np.zeros(size)
                for kl in range(size):
                    v = 0.0
                    ki = kl - i
                    if 0 <= ki < n:
                        v -= a[ki]
                    ik = i - kl
                    if 0 <= ik < n:
                        v -= a[ik]
                    kj = kl - j
                    if 0 <= kj < n:
                        v += a[kj]
                    jk = j - kl
                    if 0 <= jk < n:
                        v += a[jk]
                    g_val[kl] = v

                # Find optimal delta via bisection/line search
                # conv_new[k] = conv[k] + delta * g_val[k] + (delta^2 terms)
                # The delta^2 terms are: +delta^2 at k=2j (from a[j+delta]^2) 
                # and -delta^2 at k=2i (from (a[i]-delta)^2 - a[i]^2 = -2*a[i]*delta + delta^2)
                # and cross terms... Let's just do a simple line search.
                
                n_samples = 15
                deltas = np.linspace(0.005, 0.9, n_samples) * ai
                for d in deltas:
                    new_conv = conv + d * g_val
                    # Second order corrections
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_conv_max:
                        best_conv_max = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-14:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_moves += 1
                    else:
                        # Revert
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        s_new = a.sum()
                        if s_new > 1e-15:
                            a = a * norm / s_new

        if not pass_improved:
            no_improve_passes += 1
            if no_improve_passes >= 6:
                break
            if no_improve_passes == 3:
                n_pert = max(20, n // 15)
                idx_pert = rng.choice(n, size=n_pert, replace=False)
                a[idx_pert] *= (1.0 + 0.2 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve_passes = 0

        if verbose and total_moves % 500 == 0 and total_moves > 0:
            elapsed = time.monotonic() - t0
            print(f"    CD moves={total_moves} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def smooth_max_refine(a0, time_limit, n_iters=500, verbose=True):
    """
    Direct optimization using smooth max (log-sum-exp) with project
ed gradient.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        max_c = conv.max()

        # Log-sum-exp temperature
        T = max(0.1, 5.0 * np.exp(-it / 50.0))
        shifted = (conv - max_c) / T
        shifted = np.maximum(shifted, -100.0)
        log_sum = np.log(np.sum(np.exp(shifted))) + max_c / T
        w = np.exp(shifted) / np.exp(shifted).sum()

        # Gradient: g_i = 2 * sum_k w[k] * a[k-i]
        A = rfft(a, fft_size)
        W = rfft(w, fft_size)
        c = irfft(W * np.conj(A), fft_size)[:n]
        grad = 2.0 * c

        # Center the gradient (for the sum constraint)
        grad = grad - grad.mean()

        # Backtracking line search
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        step = 1.0 / (1.0 + 0.05 * it)
        
        for trial in range(20):
            a_new = a - step * grad / np.sqrt(n)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                step *= 0.5
                continue
            a_new = a_new * norm / s_new
            sc = fast_conv_self(a_new).max()
            if sc <= max_c:
                break
            step *= 0.5

        a_new = np.maximum(a - step * grad / np.sqrt(n), 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        a = a_new

        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SmoothMax it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def structured_init(n, rng):
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.15 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0.1)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-6.0 * (x - 0.75) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-2.0 * x) + 0.5 * np.exp(-2.0 * (1 - x))
    elif choice == 7:
        a = 0.5 * np.exp(-4.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-4.0 * (x - 0.7) ** 2)
        a = a * (1.0 + 0.3 * np.cos(20 * np.pi * x))
        a = np.maximum(a, 0.01)
    elif choice == 8:
        a = np.abs(np.sin(3 * np.pi * x)) ** 0.8
    elif choice == 9:
        a = 0.4 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(11 * np.pi * x)
        a = np.maximum(a, 0.05)
    elif choice == 10:
        # Triangle with modulation
        tri = 1.0 - np.abs(2.0 * x - 1.0)
        a = tri * (1.0 + 0.2 * np.cos(30 * np.pi * x))
        a = np.maximum(a, 0.01)
    else:
        # Multi-scale
        a = 0.3 * np.sin(2 * np.pi * x) ** 2 + 0.5 * np.sin(4 * np.pi * x) ** 2 + 0.2 * np.sin(8 * np.pi * x) ** 2
        a = np.maximum(a, 0.01)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    print(f"\n=== Phase 5: PSF refine n={n5} (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_out, _ = power_spectrum_gradient_descent(best_seq, t5, verbose=False)
        try_update(a_out, f"psf_n{n5}")

    # Phase 6: Final LP + CD
    remaining = time_budget - (time.monotonic() - t_global)
    n6 = len(best_seq)
    if remaining > 80:
        t6a = min(remaining * 0.4, 120.0)
        t6b = min(remaining * 0.4, 120.0) - 10
        print(f"\n=== Phase 6: Final LP ({t6a:.0f}s) + CD ({t6b:.0f}s) ===")
        if t6a > 20:
            a_out, _ = lp_nesterov_optimize(best_seq, t6a, verbose=False)
            try_update(a_out, "lp_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t_cd = min(remaining * 0.7, 120.0)
            a_out, _ = dual_guided_cd(best_seq, t_cd, duals=None, verbose=False)
            try_update(a_out, "cd_final")

    # Phase 7: Try longer sequence
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 400
    print(f"\n=== Phase 7: Length extension n={n_long} (remaining={remaining:.0f}s) ===")
    if remaining > 120:
        init_long = interpolate_seq(best_seq, n_long)
        t_long_lp = min(remaining * 0.4, 150.0)
        a_out, _ = lp_nesterov_optimize(init_long, t_long_lp, verbose=False)
        try_update(a_out, f"lp_long_{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and len(best_seq) == n_long:
            t_long_cd = min(remaining * 0.5, 80.0)
            a_out2, _ = dual_guided_cd(best_seq, t_long_cd, duals=None, verbose=False)
            try_update(a_out2, f"cd_long_{n_long}")

    # Phase 8: Also try shorter sequence (sometimes shorter is better per the ratio)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100 and n_base > 1200:
        n_short = max(1000, n_base - 400)
        print(f"\n=== Phase 8: Shorter sequence n={n_short} (remaining={remaining:.0f}s) ===")
        init_short = interpolate_seq(best_seq, n_short)
        t_short = min(remaining * 0.5, 150.0)
        a_out, _ = lp_nesterov_optimize(init_short, t_short, verbose=False)
        try_update(a_out, f"lp_short_{n_short}")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
