def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    A = rfft(a, fft_size)
    B = rfft(b, fft_size)

def lp_best_response_full(a, M, solver):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0. Uses ALL constraints."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    
    # Build constraint matrix T where T[k, i] = a[k-i] if 0 <= k-i < n, else 0
    # This gives (a*b)[k] = sum_i T[k,i] * b[i]
    # For efficiency, use banded structure
    
    # Actually for cvxpy, let's just build the constraint directly
    # (a*b)[k] = sum_i a[max(0,k-n+1):min(n,k+1)] * b[... corresponding range]
    # Better to build the Toeplitz matrix
    
    # T is size x n Toeplitz
    # T[k, i] = a[k-i] if 0 <= k-i < n else 0
    
    # For large n, building the full matrix is O(n^2) which is fine for n=1600
    T = np.zeros((size, n))
    for i in range(n):
        # T[k, i] = a[k-i] for k-i in [0, n-1], i.e., k in [i, i+n-1]
        start = i
        end = min(i + n, size)
        T[start:end, i] = a[:end - start]
    
    constraints = [T @ b_var <= M, b_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception as e:
        print(f"    LP error: {e}")
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)

def nesterov_full_lp(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP alternating method with full constraints."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    
    best_a = a.copy()
    best_score = compute_max_conv(a)
    t0 = time.monotonic()
    
    y = a.copy()  # Nesterov's y
    print(f"    Init: score={best_score:.8f}")
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute M from y
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break
        
        # Solve LP with full constraints
        b = lp_best_response_full(y, M, solver)
        
        if b is None:
            # Fallback: use uniform direction
            b = np.ones(n) / n
            b = b * M * n  # Scale so that (a*b) has max around M
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            b = np.ones(n) / n
            b_sum = 1.0
        
        # Normalize b to same "scale" as a
        g = b * norm / b_sum
        
        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        
        # Update y
        y_new = (1.0 - alpha_t) * y + alpha_t * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        # Update a with momentum
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        # Check improvement
        sc_y = compute_max_conv(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        sc_a = compute_max_conv(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        
        if verbose and t % 10 == 0:

            print(f"    LP-Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M:.6f}")

                best_a = a.copy()
        
        if verbose and (si % 5 == 0 or si == n_stages - 1):
            print(f"    Adam-LSE stage {si} T={T_val:.3f}: best={best_score:.8f}")
    
    return best_a, best_score

def pair_cd_exhaustive(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent with smart candidate selection."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    
    best_a = a.copy()
    best_score = compute_max_conv(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        
        # Find positions with high convolution contribution
        high_pressure = np.where(conv >= cur_max * 0.7)[0]
        
        perm = rng.permutation(n)
        pass_improved = False
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-8:
                continue
            
            # Generate candidate j's based on pressure
            j_candidates = set()
            
            # Random candidates
            for _ in range(5):
                j_candidates.add(int(rng.integers(n)))
            
            # Pressure-based: j such that moving mass from i to j reduces peak
            # The peak is at positions where conv is high
            # Moving mass from i reduces conv at positions i+j for all j
            # So we want to move mass to j where the conv at i+j is low
            for kl in high_pressure[:15]:
                # If we add mass at j, it increases conv at i+j
                # We want conv[i+j] to be low
                j1 = int((kl - i) % n)
                j2 = int((kl + i) % n)  # approximate
                j_candidates.add(j1)
                j_candidates.add(j2)
            
            j_candidates.discard(i)
            
            best_delta = 0.0
            best_v = cur_max
            best_j = -1
            
            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]
                
                # Compute the change in conv when moving delta from i to j
                # delta_conv[k] = delta * (a[k-j] - a[k-i]) + delta^2 * (delta_kj - delta_ki)
                # where delta_kj = 1 if k=j else 0 (for the self-term)
                # Actually: conv_new = conv + delta*(contrib_j - contrib_i) + delta^2*(correction)
                
                # contrib_i[k] = a[k-i] (if valid) -- this is what gets removed
                # contrib_j[k] = a[k-j] (if valid) -- this is what gets added
                
                # For efficiency, compute the max of (conv + delta*(c_j - c_i)) 
                # The correction term delta^2 only affects positions 2i and 2j
                
                # Build the difference vector
                diff = np.zeros(size)
                
                # a[k-j] for k-j in [0, n-1], i.e., k in [j, j+n-1]
                lo1 = j
                hi1 = min(j + n, size)
                if hi1 > lo1:
                    diff[lo1:hi1] += a[:hi1 - lo1]
                
                # a[i - k] ... wait, for self-conv: conv[k] = sum_m a[m]*a[k-m]
                # d(conv[k])/da[i] = a[k-i] + a[i-k+n-1]... 
                # Actually d(conv[k])/da[i] = sum_m delta_{m,i} * a[k-m] + a[m] * delta_{m,i}
                # = a[k-i] + a[i-(k-n+1)]... 
                # For full conv: conv[k] = sum_{m: 0<=m<n, 0<=k-m<n} a[m] * a[k-m]
                # d(conv[k])/da[i] = (if 0<=k-i<n) a[k-i] + (if 0<=i-(k-n+1)<n... ) 
                # Simpler: d(conv[k])/da[i] = a[k-i] (if 0<=k-i<n) + a[n-1-k+i] (if 0<=n-1-k+i<n)
                # Hmm, let me just use the direct formula.
                
                # Moving delta from i to j:
                # new_conv[k] = conv[k] - delta*a[k-i] - delta*a[i-k+n-1]... 
                # This is getting complicated. Let me just compute it directly for the candidates.
                
                # Simpler approach: compute new_conv = old_conv + delta * (g_j - g_i)
                # where g_i[k] = d(conv[k])/da[i] = a[k-i] (valid) + a[k-i+n-1-reversed]...
                
                # For self-convolution with full mode:
                # conv[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i] * a[k-i]
                # d(conv[k])/da[i] = a[k-i] (if k-i in [0,n-1]) + a[k-i] ... 
                # Wait: conv[k] = sum_i a[i]*a[k-i]
                # d(conv[k])/da[i] = a[k-i] + a[i] * d(a[k-i])/da[i]
                # But a[k-i] is a fixed value (not a variable), so:
                # d(conv[k])/da[i] = a[k-i] (from the first term) + [k-i == i ? a[i] : 0]... no
                # conv[k] = sum_j a[j] * a[k-j]
                # d(conv[k])/da[i] = a[k-i] * (if i is in the first factor) + a[i] * (if k-i == i, i.e., k=2i) * 2
                # Actually: d(conv[k])/da[i] = a[k-i] + a[k-i] if k=2i... no
                # Let j be the index: conv[k] = sum_j a[j] * a[k-j]
                # d/d a[i] = sum_j [delta_{j,i} * a[k-j] + a[j] * delta_{k-j,i}]
                # = a[k-i] + a[i] * delta_{k-i,i}... no
                # = a[k-i] + (k-i == i ? a[i] : 0)... this is wrong
                # d/d a[i] of sum_j a[j]*a[k-j]:
                # The term where j=i gives: a[k-i]
                # The term where k-j=i, i.e., j=k-i gives: a[k-i]
                # If i == k-i (i.e., k=2i), these are the same term, so we get 2*a[i]
                # So: d(conv[k])/da[i] = a[k-i] (if 0<=k-i<n) + a[k-i] (if i==k-i)
                # = a[k-i] * (1 + delta_{k,2i}) ... no that's not right either.
                
                # OK let me just be explicit:
                # d(conv[k])/da[i] = 
                #   a[k-i] if 0 <= k-i < n (j=i term)
                # + a[k-i] if k-i == i, i.e., k == 2i (j=k-i term, which equals i)
                # Wait no. Let me redo:
                # conv[k] = sum_{j: valid} a[j] * a[k-j]
                # where valid means 0<=j<n and 0<=k-j<n.
                # d(conv[k])/da[i] = sum_{j:valid} [delta_{j,i} * a[k-j] + a[j] * delta_{k-j,i}]
                # = a[k-i] (if 0<=k-i<n, from j=i) + a[k-i] (if i<=k<... from k-j=i, i.e., j=k-i, need 0<=k-i<n)
                # So d(conv[k])/da[i] = a[k-i] * (number of times i appears)
                # i appears as j=i (if valid) and as k-j=i meaning j=k-i (if valid and j!=i)
                # If k-i == i (k=2i): i appears once (j=i=k-i), coefficient is 2*a[i]
                # If k-i != i: i appears as j=i and as j=k-i (two different positions)
                # So: d(conv[k])/da[i] = a[k-i] + a[k-i] = 2*a[k-i] if k-i != i
                # = 2*a[i] if k = 2i (which is the same as 2*a[k-i] since k-i=i)
                # Wait that gives 2*a[k-i] in both cases? Let me check with a simple example.
                # a = [a0, a1], k=1: conv[1] = a0*a1 + a1*a0 = 2*a0*a1
                # d(conv[1])/da[0] = 2*a1. Using formula: a[k-i] = a[1-0] = a1, and since k=1 != 2*0=0,
                # we get a1 + a1 = 2*a1. ✓
                # k=0: conv[0] = a0*a0 = a0^2
                # d(conv[0])/da[0] = 2*a0. Formula: a[0-0]=a0, k=0=2*0, so 2*a0. ✓
                # k=2: conv[2] = a1*a1 = a1^2
                # d(conv[2])/da[1] = 2*a1. Formula: a[2-1]=a1, k=2=2*1, so 2*a1. ✓
                # d(conv[2])/da[0]: k=2, i=0: k-i=2 which is not < n=2, so 0. ✓
                
                # So the formula is: d(conv[k])/da[i] = 2*a[k-i] if 0<=k-i<n, else 0
                # Wait, that gives 2*a[k-i] always (when valid), including when k=2i.
                # Let me recheck k=0, i=0: 2*a[0-0] = 2*a0 ✓
                # k=1, i=0: 2*a[1] ✓
                # k=1, i=1: 2*a[0] ✓
                # Great! So d(conv[k])/da[i] = 2*a[k-i] if 0<=k-i<n else 0.
                
                # This simplifies things! The gradient is just 2 * cross-correlation.
                
                # So moving delta from i to j:
                # new_conv[k] = conv[k] + delta * 2 * (a[k-j] - a[k-i]) [for valid indices]
                # Plus the delta^2 correction at k=i+j (from a[i]*delta and delta*a[j] terms)
                # Actually the delta^2 terms: when we change a[i] by -delta and a[j] by +delta,
                # the second-order terms are at positions where both i and j appear.
                # conv_new[2i] = conv[2i] - 2*delta*a[i] + delta^2 (removing from i)
                # conv_new[2j] = conv[2j] + 2*delta*a[j] + delta^2 (adding to j)
                # conv_new[i+j] = conv[i+j] + delta*(a[j]... ) 
                
                # Hmm this is getting complex. Let me just use the linear approximation
                # and add the quadratic correction for the self-terms.
                
                # Linear part: delta * 2 * (a[k-j] - a[k-i]) for valid k
                # The vector g = 2*(shift(a,j) - shift(a,i)) where shift(a,m)[k] = a[k-m] if valid
                
                g = np.zeros(size)
                
                # Add: a[k-j] for k-j in [0,n-1], i.e., k in [j, j+n-1]
                lo1 = j
                hi1 = min(j + n, size)
                if hi1 > lo1:
                    g[lo1:hi1] += a[:hi1 - lo1]
                
                # Subtract: a[k-i] for k-i in [0,n-1], i.e., k in [i, i+n-1]
                lo2 = i
                hi2 = min(i + n, size)
                if hi2 > lo2:
                    g[lo2:hi2] -= a[:hi2 - lo2]
                
                g *= 2.0  # The factor of 2
                
                # Now find the best delta in [0, ai]
                # new_conv ≈ conv + delta * g
                # We want to minimize max(conv + delta * g)
                # This is a 1D problem: minimize max_k (conv[k] + delta * g[k])
                # The max is achieved at some k that either:
                # - was already at the peak (conv[k] = max, and g[k] < 0 helps)
                # - is a point where conv[k] + delta * g[k] = max (crossing point)
                
                # Simple approach: binary search or sample
                d_max = ai  # Can't move more than ai
                if d_max < 1e-15:
                    continue
                
                # Sample some deltas
                n_samples = 15
                deltas = np.linspace(0.0, d_max, n_samples)
                
                best_v_local = cur_max
                best_d_local = 0.0
                
                for d in deltas:
                    nc = conv + d * g
                    # Add quadratic correction
                    if 2 * i < size:
                        nc[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size and j != i:
                        nc[2 * j] = conv[2 * j] + 2 * a[j] * d + d * d
                    # Cross term at i+j
                    if i + j < size:
                        nc[i + j] = conv[i + j] + d * (a[j] + ai) + 2 * d * 0  # approximate
                        # Actually the cross term: a[i-d + d]*a[j-d+d]... 
                        # The change in conv[i+j] from the linear part is already in g
                        # The quadratic correction: (-d)*(+d) at position i+j = -d^2
                        # And also from the i=j... no, i != j here
                        # conv[i+j] includes the term a[i]*a[j] which changes to (a[i]-d)*(a[j]+d)
                        # = a[i]*a[j] + d*a[i] - d*a[j] - d^2
                        # So the quadratic correction is -d^2
                        nc[i + j] -= d * d
                    
                    v = nc.max()
                    if v < best_v_local:
                        best_v_local = v
                        best_d_local = d
                
                if best_d_local > 0 and best_v_local < best_score - 1e-10:
                    a[i] -= best_d_local
                    a[j] += best_d_local
                    s_new = a.sum()
                    if s_new > 1e-15:
                        a = a * norm / s_new
                        new_max = compute_max_conv(a)
                        if new_max < best_score - 1e-10:
                            best_score = new_max
                            best_a = a.copy()
                            pass_improved = True
                        else:
                            # Revert
                            a[j] -= best_d_local
                            a[i] += best_d_local
                            a = a * norm / a.sum()
                    else:
                        a[j] -= best_d_local
                        a[i] += best_d_local
                
                if best_j >= 0:
                    pass  # already handled above
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 8:
                # Perturb
                n_perturb = max(10, n // 50)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.1 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                if no_improve >= 15:
                    break
        else:
            no_improve = 0
    
    return best_a, best_score

def vegas_optimize(a0, time_limit, verbose=True):
    """Vegas (multiplicative weight) optimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    
    best_a = a.copy()
    best_score = compute_max_conv(a)
    t0 = time.monotonic()
    fft_size = 1 << ((2*n - 2) - 1 + 1).bit_length()
    
    n_stages = 30
    etas = np.geomspace(0.3, 0.001, n_stages)
    
    for si, eta_s in enumerate(etas):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(15, min(60, int(remaining * 6)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = fast_conv_self(a)
            M = conv.max()
            if M < 1e-15:
                break
            
            # Weight: positions at the peak get high weight
            w = np.maximum(conv - M * 0.85, 0.0)
            ww = w / (w.sum() + 1e-15)
            
            W = rfft(ww, fft_size)
            A = rfft(a, fft_size)
            
            # Gradient (pressure) on a
            pressure = irfft(W * A.conj(), fft_size)[:n]
            pressure = np.maximum(pressure, 0)
            
            # Multiplicative update
            a = a * np.exp(-eta_s * pressure / (pressure.max() + 1e-15))
            a = np.maximum(a, 1e-10)
            a = a * norm / a.sum()
            
            sc = compute_max_conv(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose and si % 6 == 0:
            print(f"    Vegas stage {si} eta={eta_s:.4f}: best={best_score:.8f}")
    
    return best_a, best_score

def structured_init(n, rng):
    """Generate a structured initial sequence."""
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)
    
    if choice == 0:
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    elif choice == 2:
        f1 = rng.integers(1, 5)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * f1 * x) + 0.2 * np.cos(2 * np.pi * f1 * 2 * x)
    elif choice == 3:
        c = rng.uniform(0.3, 0.7)
        a = np.exp(-50.0 * (x - c) ** 2) + 0.1
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        n_peaks = rng.integers(3, 8)
        centers = rng.uniform(0.1, 0.9, n_peaks)
        a = np.zeros(n)
        for c in centers:
            w = rng.uniform(0.03, 0.08)
            a += np.exp(-(x - c) ** 2 / (2 * w ** 2))
        a += 0.05
    elif choice == 6:
        a = np.sin(np.pi * x) ** 0.5
    elif choice == 7:
        a = 1.0 + 0.2 * np.sin(4 * np.pi * x) + 0.1 * np.cos(6 * np.pi * x)
    elif choice == 8:
        a = np.sinc(3 * (x - 0.5)) ** 2 + 0.1
    else:
        a = np.ones(n) * (1.0 + 0.3 * np.cos(2 * np.pi * x))
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def symmetrize(a):
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    s = a_sym.sum()
    if s < 1e-15:
        return a
    return a_sym / s * (a.sum() / s)

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function combining LP-based alternating, Adam gradient descent,
    Vegas multiplicative updates, and pair coordinate descent.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}, solver={solver}")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p.copy()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)
    
    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            old_best = best_score
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (was {old_best:.6f}, n={len(a)}, t={elapsed:.0f}s)")
    
    # Phase 1: Vegas (fast, good for initial progress)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 1: Vegas n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_out, _ = vegas_optimize(best_seq, t1, verbose=True)
        check_and_update(a_out, f"vegas1_n{n_base}")
    
    # Phase 2: Adam LSE
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 2: Adam-LSE n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = adam_smoothed_max(best_seq, t2, verbose=True)
        check_and_update(a_out, f"adam1_n{len(best_seq)}")
    
    # Phase 3: LP Nesterov (full constraints - main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.30, 500.0)
    print(f"\n=== Phase 3: LP-Nesterov (full) n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = nesterov_full_lp(best_seq, t3, solver, verbose=True)
        check_and_update(a_out, f"lp1_n{len(best_seq)}")
    
    # Phase 4: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 4: Pair CD n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = pair_cd_exhaustive(best_seq, t4, rng, verbose=True)
        check_and_update(a_out, f"pcd1_n{len(best_seq)}")
    
    # Phase 5: Vegas round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.08, 80.0)
    print(f"\n=== Phase 5: Vegas round 2 n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_out, _ = vegas_optimize(best_seq, t5, verbose=False)
        check_and_update(a_out, f"vegas2_n{len(best_seq)}")
    
    # Phase 6: LP Nesterov round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 6: LP-Nesterov round 2 n={len(best_seq)} (budget={t6:.0f}s) ===")
    if t6 > 30:
        a_out, _ = nesterov_full_lp(best_seq, t6, solver, verbose=True)
        check_and_update(a_out, f"lp2_n{len(best_seq)}")
    
    # Phase 7: Symmetrize + final PCD

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1

    A = rfft(a, fft_size)

                    row_start = max(0, k - n + 1)
                    row_end = min(n, k + 1)
                    for i in range(row_start, row_end):
                        T_sub2[idx, i] = y[k - i]
                prob2 = cp.Problem(cp.Maximize(cp.sum(b_var)), [T_sub2 @ b_var <= M])
                try:
                    prob2.solve(solver=solver, verbose=False)
                    if prob2.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                        b = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
                except Exception:
                    pass

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_t) * y + alpha_t * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = compute_max_conv(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        sc_a = compute_max_conv(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if verbose and t % 8 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Active t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M:.6f}, |A|={len(active_set)}")

    return best_a, best_score


def adam_lse_optimize(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = compute_max_conv(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    step = 0

    n_stages = 25
    temps = np.geomspace(60.0, 0.03, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(30, min(120, int(remaining * 8)))
        lr = 0.8 / np.sqrt(n)

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            step += 1

            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.clip(shifted, -100.0, 0.0)
            w = np.exp(shifted)
            w = w / w.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            grad = 2.0 * irfft(W * A.conj(), fft_size)[:n]

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** step)
            v_hat = v / (1.0 - beta2 ** step)

            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new

            a = a_new
            sc = compute_max_conv(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose and si % 5 == 0:
            print(f"    Adam-LSE stage {si} T={T_val:.3f}: best={best_score:.8f}")

    return best_a, best_score


def vegas_optimize(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = compute_max_conv(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 30
    etas = np.geomspace(0.5, 0.002, n_stages)

    for si, eta_s in enumerate(etas):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(15, min(50, int(remaining * 5)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_conv_self(a)
            M = conv.max()
            if M < 1e-15:
                break

            w = np.maximum(conv - M * 0.8, 0.0)
            w_sum = w.sum()
            if w_sum < 1e-15:
                w = np.ones(size) / size
            else:
                w = w / w_sum

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            pressure = irfft(W * A.conj(), fft_size)[:n]
            pressure = np.maximum(pressure, 0.0)
            p_max = pressure.max()
            if p_max < 1e-15:
                break

            a = a * np.exp(-eta_s * pressure / p_max)
            a = np.maximum(a, 1e-10)
            a = a * norm / a.sum()

            sc = compute_max_conv(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose and si % 5 == 0:
            print(f"    Vegas stage {si} eta={eta_s:.4f}: best={best_score:.8f}")

    return best_a, best_score


def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = compute_max_conv(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-9:
                continue

            peak_lags = np.where(conv >= cur_max * 0.75)[0]

            j_candidates = set()
            for _ in range(6):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:12]:
                j_candidates.add(int((kl - i + n) % n))
                j_candidates.add(int((kl + i) % n))
            j_candidates.discard(i)

            best_delta = 0.0
            best_v = cur_max
            best_j = -1

            for j in j_candidates:
                if j == i:
                    continue

                g = np.zeros(size)
                lo1 = j
                hi1 = min(j + n, size)
                if hi1 > lo1:
                    g[lo1:hi1] += a[:hi1 - lo1]
                lo2 = i
                hi2 = min(i + n, size)
                if hi2 > lo2:
                    g[lo2:hi2] -= a[:hi2 - lo2]
                g *= 2.0

                d_max = ai
                if d_max < 1e-12:
                    continue
                n_samples = 20
                deltas = np.linspace(0.0, d_max, n_samples)

                for d in deltas:
                    if d == 0.0:
                        continue
                    nc = conv + d * g
                    if 2 * i < size:
                        nc[2 * i] = conv[2 * i] - 2.0 * ai * d + d * d
                    if 2 * j < size and j != i:
                        nc[2 * j] = conv[2 * j] + 2.0 * a[j] * d + d * d
                    if i + j < size and i != j:
                        nc[i + j] -= d * d
                    v = nc.max()
                    if v < best_v:
                        best_v = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0 and best_v < best_score - 1e-10:
                a[best_j] += best_delta
                a[i] -= best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = compute_max_conv(a)
                    if new_max < best_score - 1e-10:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[i] += best_delta
                        a[best_j] -= best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 6:
                n_perturb = max(8, n // 60)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.12 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                if no_improve >= 12:
                    break
        else:
            no_improve = 0

        if verbose and total_improvements % 300 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD improv={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def subgradient_optimize(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = compute_max_conv(a)
    t0 = time.monotonic()
    size = 2 * n - 1

    n_restarts = 5
    for restart in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        n_steps = max(50, int(remaining * 15))

        for step in range(1, n_steps + 1):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_conv_self(a)
            k_star = conv.argmax()
            M = conv[k_star]
            if M < 1e-15:
                break

            grad = np.zeros(n)
            r_start = max(0, k_star - n + 1)
            r_end = min(n, k_star + 1)
            for i in range(r_start, r_end):
                grad[i] = 2.0 * a[k_star - i]

            lr = norm / (step + 10.0)
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = compute_max_conv(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    Subgrad restart {restart}: best={best_score:.8f}")

        a[::2] *= (1.0 + 0.05 * np.random.standard_normal(len(a[::2])))
        a = np.maximum(a, 0.0)
        a = a * norm / a.sum()

    return best_a, best_score
