```python
def peak_shave(p, n_peaks=5, eta=0.01):
    """Reduce the top peaks of the auto-convolution."""
    b = fast_conv_self(p)
    S = p.sum()
    n = len(p)
    
    # Find top peaks
    top_k = np.argsort(b)[-n_peaks:]
    
    # For each peak, compute gradient contribution
    grad = np.zeros(n)
    for k in top_k:
        # b[k] = sum_i p[i]*p[k-i]
        # db[k]/dp[i] = p[k-i] + p[i-k] (with bounds)
        for i in range(n):
            ki = k - i
            ik = i - k
            if 0 <= ki < n:
                grad[i] += p[ki]
            if 0 <= ik < n:
                grad[i] += p[ik]
    
    # Normalize gradient
    grad_max = grad.max()
    if grad_max > 0:
        grad = grad / grad_max
    
    # Update: reduce high-gradient positions
    p_new = p * np.exp(-eta * grad)
    p_new = np.maximum(p_new, 1e-12)
    p_new = p_new / p_new.sum()
    
    return p_new
```

```python
def fast_conv_self(a):

def lse_gradient(a, T, fft_size):
    """Compute gradient of LSE_T(a*a) using FFT."""
    conv = fast_conv_self(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros_like(a), 0.0
    shifted = (conv - max_c) / T
    shifted = np.maximum(shifted, -80.0)
    exp_c = np.exp(shifted)
    w = exp_c / exp_c.sum()
    n = len(a)
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    g1 = irfft(np.conj(W) * A, fft_size)[:n]
    g2 = irfft(W * A, fft_size)[:n]
    return 2.0 * (g1 + g2), w

    n_stages = 30

        np.geomspace(60.0, 5.0, 10),

        np.geomspace(0.5, 0.02, 10)
    ])

    # Nesterov momentum
    y = a.copy()

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        n_steps = max(15, min(100, int(remaining * 8)))
        lr = 1.0 / np.sqrt(n)

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            t_adam += 1

            # Evaluate at Nesterov lookahead point
            a_eval = y
            conv = fast_conv_self(a_eval)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -80.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A_eval = rfft(a_eval, fft_size)
            g1 = irfft(np.conj(W) * A_eval, fft_size)[:n]
            g2 = irfft(W * A_eval, fft_size)[:n]

            # Clip gradient
            grad_norm = np.linalg.norm(grad)
            if grad_norm > 100.0:

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_adam)
            v_hat = v / (1.0 - beta2 ** t_adam)

            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

def lp_best_response(a, M, solver, active_set=None):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for k in active_set, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)

    if active_set is not None and 0 < len(active_set) < 2 * n - 1:
        constraints = []
        for k in active_set:
            row_start = max(0, k - n + 1)
            row_end = min(n, k + 1)
            coef = np.zeros(n)
            for i in range(row_start, row_end):
                coef[i] = a[k - i]
            if coef.max() > 0:
                constraints.append(coef @ b_var <= M)
        if not constraints:
            constraints = [cp.sum(b_var) <= M * n / a.max() if a.max() > 0 else cp.sum(b_var) <= 1]
    else:
        constraints = []
        for k in range(2 * n - 1):
            row_start = max(0, k - n + 1)
            row_end = min(n, k + 1)
            coef = np.zeros(n)
            for i in range(row_start, row_end):
                coef[i] = a[k - i]
            if coef.max() > 0:
                constraints.append(coef @ b_var <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)

                # Check if increasing delta helps
                test_conv = conv + 0.1 * d_max * g
                if 2 * i < size:
                    test_conv[2 * i] = conv[2 * i] - 2 * ai * 0.1 * d_max + (0.1 * d_max) ** 2
                if 2 * j < size:
                    test_conv[2 * j] = conv[2 * j] + 2 * aj * 0.1 * d_max + (0.1 * d_max) ** 2
                if test_conv.max() >= cur_max:
                    continue

                for _ in range(12):
                    mid = (lo_d + hi_d) / 2
                    new_conv = conv + mid * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * mid + mid * mid
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * mid + mid * mid
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_delta = mid
                        best_j = j
                    if v <= cur_max:
                        lo_d = mid
                    else:
                        hi_d = mid

            if best_j >= 0 and best_delta > 0 and best_v < best_score - 1e-12:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        s_new2 = a.sum()
                        if s_new2 > 1e-15:
                            a = a * norm / s_new2

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                n_perturb = max(10, n // 40)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.1 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                if no_improve >= 8:
                    break
        else:
            no_improve = 0

        if verbose and total_improvements > 0 and total_improvements % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD imp={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)

    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        f1 = rng.integers(1, 6)
        f2 = rng.integers(6, 20)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * f1 * x) + 0.2 * np.cos(2 * np.pi * f2 * x)
    elif choice == 3:
        c1 = rng.uniform(0.2, 0.5)
        c2 = rng.uniform(0.5, 0.8)
        a = np.exp(-30.0 * (x - c1) ** 2) + np.exp(-30.0 * (x - c2) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 4
    elif choice == 5:
        n_peaks = rng.integers(3, 8)
        centers = rng.uniform(0.1, 0.9, n_peaks)
        a = np.zeros(n)
        for c in centers:
            width = rng.uniform(0.05, 0.15)
            a += np.exp(-(x - c) ** 2 / (2 * width ** 2))
    elif choice == 6:
        a = 0.4 + 0.4 * np.sin(np.pi * x) ** 0.7 + 0.2 * np.sin(2 * np.pi * x) ** 2
    elif choice == 7:
        a = np.exp(-3.0 * (x - 0.5) ** 2 / (0.15 ** 2)) + 0.3 * np.exp(-3.0 * (x - 0.1) ** 2 / (0.05 ** 2)) + 0.3 * np.exp(-3.0 * (x - 0.9) ** 2 / (0.05 ** 2))
    elif choice == 8:
        a = 1.0 + 0.2 * np.sin(2 * np.pi * 2 * x) + 0.1 * np.cos(2 * np.pi * 5 * x)
    else:
        a = np.ones(n) * (0.5 + 0.3 * np.cos(np.pi * x))

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a = a + 0.003 * np.abs(rng.standard_normal(n))
    a = a / a.sum()
    return a

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def symmetrize(a):
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    s = a_sym.sum()
    if s < 1e-15:
        return a
    return a_sym / s * (a.sum() / s)

def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function: hybrid optimization to minimize the evaluation score."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1680.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            old_best = best_score
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (was {old_best:.6f}, n={len(a)}, t={elapsed:.0f}s)")

    # Phase 1: Quick LSE-Adam warmup
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 1: LSE-Adam warmup n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_out, _ = lse_adam_optimize(best_seq, t1, verbose=True)
        check_and_update(a_out, f"lse1_n{len(a_out)}")

    # Phase 2: LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.35, 550.0)
    print(f"\n=== Phase 2: LP Nesterov n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, solver, verbose=True)
        check_and_update(a_out, f"lp1_n{len(a_out)}")

    # Phase 3: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 3: Peak Shaving n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = peak_shave_optimize(best_seq, t3, rng, verbose=True)
        check_and_update(a_out, f"ps1_n{len(a_out)}")

    # Phase 4: LP Nesterov round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 4: LP Nesterov round 2 n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t4, solver, verbose=True)
        check_and_update(a_out, f"lp2_n{len(a_out)}")

    # Phase 5: LSE-Adam refinement
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.12, 200.0)
    print(f"\n=== Phase 5: LSE-Adam refinement n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = lse_adam_optimize(best_seq, t5, verbose=True)
        check_and_update(a_out, f"lse2_n{len(a_out)}")

    # Phase 6: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.12, 200.0)
    print(f"\n=== Phase 6: Pair CD n={len(best_seq)} (budget={t6:.0f}s) ===")
    if t6 > 15:
        a_out, _ = pair_cd_optimize(best_seq, t6, rng, verbose=True)
        check_and_update(a_out, f"pcd_n{len(a_out)}")

    # Phase 7: Symmetrization + sparsification attempts
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Symmetry/Sparsity (remaining={remaining:.0f}s) ===")
    a_sym = symmetrize(best_seq)
    check_and_update(a_sym, f"sym_n{len(a_sym)}")

    # Try slight truncation
    for trim in [10, 20, 50]:
        if len(best_seq) > trim * 2 + 100:
            a_trim = best_seq[trim:len(best_seq) - trim]
            check_and_update(a_trim, f"trim{trim}_n{len(a_trim)}")

    # Phase 8: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 8: Multi-length exploration (remaining={remaining:.0f}s) ===")
        lengths_to_try = sorted(set([
            len(best_seq) + 200,
            len(best_seq) + 400,
            len(best_seq) - 200,
            len(best_seq) + 600,
            len(best_seq) - 400,
        ]))
        for n_try in lengths_to_try:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 50:
                break
            if n_try < 200 or n_try > 5000:
                continue
            t_try = min(remaining * 0.4, 50.0)
            print(f"  Trying n={n_try} (budget={t_try:.0f}s)")
            init_try = interpolate_seq(best_seq, n_try)
            a_out, _ = lse_adam_optimize(init_try, min(t_try, 25.0), verbose=False)
            check_and_update(a_out, f"ml_lse_n{n_try}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 25:
                a_out, _ = pair_cd_optimize(a_out, min(remaining * 0.4, 20.0), rng, verbose=False)
                check_and_update(a_out, f"ml_pcd_n{n_try}")

    # Phase 9: Final peak shaving + PCD on best
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 9: Final polish (remaining={remaining:.0f}s) ===")
        t9a = min(remaining * 0.5, 80.0)
        if t9a > 15:
            a_out, _ = peak_shave_optimize(best_seq, t9a, rng, verbose=False)
            check_and_update(a_out, "final_ps")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out, _ = pair_cd_optimize(best_seq, min(remaining * 0.8, 60.0), rng, verbose=False)
            check_and_update(a_out, "final_pcd")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
```
