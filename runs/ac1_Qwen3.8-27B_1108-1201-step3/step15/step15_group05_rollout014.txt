    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size, worker=1)
    return np.real(ifft(A * A, fft_size, worker=1)[:size])

def subgradient_step(a, k_star, lr):
    """One subgradient step using the correct gradient formula."""
    n = len(a)
    # Gradient: grad[j] = 2 * a[k_star - j] if 0 <= k_star - j < n else 0
    grad = np.zeros(n)

    j_lo = max(0, k_star - n + 1)
    j_hi = min(k_star, n - 1)
    for j in range(j_lo, j_hi + 1):
        grad[j] = 2.0 * a[k_star - j]
    return grad

def subgradient_descent(a0, S, time_limit, verbose=False):
    """Subgradient descent with adaptive step size."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a * S / s

    best_max = fast_autocorr(a).max()
    t0 = time.time()
    
    lr = 1.0 / np.sqrt(n)
    patience = 0
    max_patience = 50000

            best_a = a.copy()
            patience = 0
        
        k_star = int(np.argmax(C))
        grad = subgradient_step(a, k_star, lr)
        
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * S / s_new

        M_new = C_new.max()

            a = a_new
            lr = min(lr * 1.01, 10.0)
        else:

            lr *= 0.99
            if lr < 1e-8:
                break
    
    return best_a, best_max

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LP best-response with adaptive active set (using scipy linprog)
2. Nesterov-accelerated iteration
3. Subgradient polishing with correct gradient
4. Coordinate descent with 1D line search
5. Multi-length exploration
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def eval_score(a):
    """Compute the score 2*n*max(conv)/sum^2."""
    n = len(a)
    C = fast_autocorr(a)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    return 2.0 * n * C.max() / (S * S)

def build_A_matrix(a, k_indices, n):
    """Build constraint matrix for (a*b)[k] <= M.
    Row k has: A[k, j] = a[k-j] if 0 <= k-j < n, else 0.
    """
    A = np.zeros((len(k_indices), n))
    for idx, k in enumerate(k_indices):
        j_lo = max(0, k - n + 1)
        j_hi = min(k + 1, n)
        for j in range(j_lo, j_hi):
            A[idx, j] = a[k - j]
    return A

def solve_lp(a, M, active_k, n):
    """Solve: max sum(b) s.t. (a*b)[k] <= M for k in active_k, b >= 0."""
    A = build_A_matrix(a, active_k, n)
    c = -np.ones(n)  # minimize -sum(b)
    res = linprog(c, A_ub=A, b_ub=np.full(len(active_k), M), bounds=[(0, None)]*n, method='highs')
    if not res.success:
        return None
    b = np.maximum(res.x, 0)
    if b.sum() < 1e-15:
        return None
    return b

    rng = np.random.default_rng(seed)
    n = len(a0)

    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    
    y = a.copy()
    y_prev = a.copy()
    t0 = time.time()
    active_set = None
    stall = 0
    stall_thresh = 60
    
    for t in range(1, 10000000):
        elapsed = time.time() - t0
        if elapsed > time_limit:
            break
        
        C_y = fast_autocorr(y)
        M_curr = C_y.max()
        
        # Adaptive active set
        if active_set is None or t % 30 == 0:
            tol = max(0.02, 0.08 * np.exp(-t / 200.0))
            thresholds = M_curr * (1.0 - tol)
            active_set = np.where(C_y >= thresholds)[0]
            # Always include top peaks
            top_k = np.argsort(C_y)[-max(3, n // 500):]
            active_set = np.unique(np.concatenate([active_set, top_k]))
        
        b = solve_lp(y, M_curr, active_set, n)
        if b is None:
            # Expand active set
            active_set = np.arange(2*n-1)
            b = solve_lp(y, M_curr, active_set, n)
            if b is None:
                continue
        
        # Verify and expand
        C_check = fast_autocorr(y) * 0 + 1  # placeholder
        # Actually verify: check if (y*b)[k] > M for k not in active_set

                if b is not None:
                    active_set = new_active
                if b is None:
                    continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * S / b_sum
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * S / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t-1.0)/(t+2.0), 0.97)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * S / s_a
        
        y_prev = y
        y = y_new
        a = a_new
        
        # Track best
        M_y = fast_autocorr(y).max()
        if M_y < best_M:
            best_M = M_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        M_a = fast_autocorr(a).max()
        if M_a < best_M:
            best_M = M_a
            best_a = a.copy()
            stall = 0
        
        if stall > stall_thresh:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall = 0
            if verbose:
                print(f"    LP restart at iter {t} ({elapsed:.0f}s), M={best_M:.8f}")
        
        if verbose and t % 500 == 0:
            print(f"    LP iter={t} ({elapsed:.0f}s): best_M={best_M:.8f}, score={2*n*best_M/S**2:.6f}")
    
    return best_a, best_M

def subgradient_polish(a0, time_limit, verbose=False):
    """Subgradient with correct gradient formula and adaptive step."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a * S / s
    
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.time()
    
    lr = 0.1
    patience = 0
    
    for it in range(1, 100000000):
        if time.time() - t0 > time_limit:
            break
        if patience > 100000:
            break
        
        C = fast_autocorr(a)
        M = C.max()
        k_star = int(np.argmax(C))
        
        # Gradient: grad[j] = 2*a[k_star - j] if 0 <= k_star - j < n
        j_lo = max(0, k_star - n + 1)
        j_hi = min(k_star, n - 1)
        a_new = a.copy()
        a_new[j_lo:j_hi+1] -= lr * 2.0 * a[k_star - j_lo:k_star - j_lo + (j_hi - j_lo + 1)]
        # Wait, I need to be more careful.
        # grad[j] = 2 * a[k_star - j] for j in [max(0, k_star-n+1), min(k_star, n-1)]
        # So for j in that range, a[k_star - j] ranges from a[k_star - j_hi] to a[k_star - j_lo]
        # a[k_star - j_lo] ... a[k_star - j_hi] = a[k_star - min(k_star, n-1)] ... a[k_star - max(0, k_star-n+1)]
        
        # Let me just compute it directly for clarity
        grad = np.zeros(n)
        for j in range(j_lo, j_hi + 1):
            grad[j] = 2.0 * a[k_star - j]
        
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * S / s_new
        
        M_new = fast_autocorr(a_new).max()
        if M_new < best_M - 1e-14:
            a = a_new
            best_M = M_new
            best_a = a_new.copy()
            patience = 0
            lr = min(lr * 1.005, 5.0)
        else:
            patience += 1

def coordinate_descent(a0, time_limit, verbose=False):
    """Coordinate descent with 1D mini-max optimization."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a * S / s
    
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.time()
    size = 2 * n - 1
    
    C = fast_autocorr(a)  # current full convolution
    n_passes = 0
    no_improve = 0
    
    while time.time() - t0 < time_limit:
        n_passes += 1
        improved = False
        
        for j in range(n):
            if time.time() - t0 > time_limit:
                break
            
            x_old = a[j]
            if x_old < 1e-15:
                continue
            
            # Compute g[k] = 2*a[k-j] for 0 <= k-j < n
            # k ranges from j to j+n-1
            k_lo = j
            k_hi = j + n - 1
            # g[k] = 2*a[k-j] for k in [k_lo, k_hi]
            # c[k] = 1 if k == 2j else 0
            
            # C_bg[k] = C[k] - x_old * g[k] - x_old^2 * c[k] for affected k
            # For k not in [k_lo, k_hi]: C_bg[k] = C[k]
            
            # We want to minimize: max_k (C_bg[k] + x*g[k] + x^2*c[k]) / (S - x_old + x)^2
            # = 2n * max_k (...) / (S - x_old + x)^2
            # Since 2n is constant, minimize max_k (...) / (S - x_old + x)^2
            
            # The max is achieved at some k. For k in [k_lo, k_hi]:
            #   C_bg[k] + x*g[k] + x^2*c[k] = C[k] - x_old*g[k] - x_old^2*c[k] + x*g[k] + x^2*c[k]
            #   = C[k] + (x - x_old)*g[k] + (x^2 - x_old^2)*c[k]
            # For k not in [k_lo, k_hi]: value is C[k] (unchanged)
            
            # So we need: max( max_{k outside [k_lo,k_hi]} C[k],  max_{k in [k_lo,k_hi]} (C[k] + (x-x_old)*g[k] + (x^2-x_old^2)*c[k]) )
            
            # Let M_out = max C[k] for k outside [k_lo, k_hi]
            M_out = 0.0
            for k in range(size):
                if k < k_lo or k > k_hi:
                    if C[k] > M_out:
                        M_out = C[k]
            
            # For k in [k_lo, k_hi], the value is C[k] + (x-x_old)*g[k] + (x^2-x_old^2)*c[k]
            # where g[k] = 2*a[k-j] and c[k] = 1 if k==2j else 0
            
            # We want to minimize: max(M_out, max_{k in [k_lo,k_hi]} (C[k] + (x-x_old)*2*a[k-j] + (x^2-x_old^2)*[k==2j])) / (S-x_old+x)^2
            
            # Try a grid of x values
            s_others = S - x_old
            # x should be in [0, S] (since sum must be positive)
            # Good candidates: 0, x_old, and values that balance terms
            
            candidates = [0.0, x_old * 0.5, x_old * 0.8, x_old, x_old * 1.2, x_old * 1.5, s_others * 0.5]
            candidates = [x for x in candidates if 0 <= x <= S]
            
            def obj(x):
                new_sum = s_others + x
                if new_sum < 1e-15:
                    return float('inf')
                val = M_out
                for k in range(k_lo, k_hi + 1):
                    v = C[k] + (x - x_old) * 2.0 * a[k - j]
                    if k == 2 * j:
                        v += (x * x - x_old * x_old)
                    if v > val:
                        val = v
                return val / (new_sum * new_sum)
            
            best_x = x_old
            best_obj = obj(x_old)
            
            for x in candidates:
                o = obj(x)
                if o < best_obj:
                    best_obj = o
                    best_x = x
            
            # Refine with bisection
            if abs(best_x - x_old) > 1e-10:
                lo = min(best_x, x_old)
                hi = max(best_x, x_old)
                for _ in range(30):
                    mid = (lo + hi) / 2.0
                    if obj(mid) < obj(hi):
                        hi = mid
                    else:
                        lo = mid
                best_x = (lo + hi) / 2.0
            
            if abs(best_x - x_old) > 1e-12:
                # Update a[j] and C
                a[j] = best_x
                new_sum = a.sum()
                # Rescale
                a = a * S / new_sum
                C = fast_autocorr(a)
                new_score = 2.0 * n * C.max() / (S * S)
                if new_score < best_score - 1e-14:
                    best_score = new_score
                    best_a = a.copy()
                    improved = True
                else:
                    a[j] = x_old
                    C = fast_autocorr(a)
        
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        
        if verbose and n_passes % 2 == 0:
            print(f"    CD pass={n_passes} ({time.time()-t0:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score

Let me optimize: instead of looping over k for each j, I can precompute the max more efficiently.

Actually, the key insight is that for a given j, the affected lags are k in [j, j+n-1]. The values at these lags are:
C[k] + (x-x_old)*2*a[k-j] + (x^2-x_old^2)*[k==2j]

The max over these is the max of (C[k] - 2*x_old*a[k-j] + 2*x*a[k-j] + (x^2-x_old^2)*[k==2j]) for k in [j, j+n-1].

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP best-response with Nesterov acceleration (HiGHS solver via scipy)
2. Adaptive active-set management
3. Multi-peak subgradient polishing
4. Short coordinate descent for final refinement
5. Multi-length exploration

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return irfft(A * A, n=fft_size)[:size]

def eval_score(a):
    n = len(a)
    C = fast_autocorr(a)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    return 2.0 * n * C.max() / (S * S)

def normalized_eval(a):
    """Evaluate with sum(a) = sqrt(2n) normalization."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    a_norm = a * np.sqrt(2.0 * n) / S
    C = fast_autocorr(a_norm)
    return C.max()

def build_constraint_rows(a, k_vals, n):
    """For each k in k_vals, build the row of the constraint (a*b)[k] <= M.
    Row for k: has a[k-j] at position j, for valid j.
    Returns a 2D array of shape (len(k_vals), n).
    """
    m = len(k_vals)
    # For sparse construction
    rows = np.zeros((m, n))
    for idx, k in enumerate(k_vals):
        j_lo = max(0, k - n + 1)
        j_hi = min(k + 1, n)
        # positions j_lo to j_hi-1 have values a[k-j]
        for j in range(j_lo, j_hi):
            rows[idx, j] = a[k - j]
    return rows

def solve_best_response(a, M, k_active, n):
    """Solve max sum(b) s.t. (a*b)[k] <= M for k in k_active, b >= 0."""
    A_ub = build_constraint_rows(a, k_active, n)
    c = -np.ones(n)
    b_ub = np.full(len(k_active), M)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', 
                  options={'presolve': True, 'time_limit': 1.0})
    if not res.success:
        return None
    b = np.maximum(res.x, 0.0)
    if b.sum() < 1e-12:
        return None
    return b
