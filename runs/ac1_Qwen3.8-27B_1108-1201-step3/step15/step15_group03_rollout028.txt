        active_indicator = np.zeros(2*n-1)
        active_indicator[active] = 1.0

        # = 2 * cross_correlation(active_indicator, a)[i]
        N = 1 << (2*n).bit_length()
        AI = rfft(active_indicator, N)
        A = rfft(a, N)
        sg = 2.0 * irfft(AI * np.conj(A), N)[:n]
        
        # Projected subgradient step
        lr = 1.0 / (np.sqrt(2.0 * t) * M)  # Polyak step size (rough)
        g = a - lr * sg
        g = np.maximum(g, 0.0)
        g = g / g.sum()
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        
        # Extrapolation
        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()
        
        y = y_new
        a = a_new
        
        sc = fast_conv_self(a).max()
        if sc < best_max:
            best_max = sc
            best_a = a.copy()
    
    return best_a, best_max

def l2_spectral(a0, time_limit, verbose=True):
    """Minimize sum(conv^2) via gradient descent to smooth the convolution."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size + n - 1 - 1).bit_length()

    for it in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        # Gradient of sum(conv^2) w.r.t. a:
        # grad[i] = 4 * sum_k conv[k] * a[k-i] (cross-correlation)
        C = rfft(conv, fft_size)
        A = rfft(a, fft_size)
        cross = irfft(C * np.conj(A), fft_size)[:n]
        grad = 4.0 * cross

        # Also subtract the mean to focus on reducing variance
        # (the mean of conv is fixed = (sum a)^2, so minimizing sum conv^2
        #  is equivalent to minimizing variance)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Adaptive learning rate
        lr = 0.05 * norm / (grad_norm * (1.0 + 0.005 * it))
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        a = a_new

        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    L2 it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score
