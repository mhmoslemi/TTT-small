def fast_autocorr(a):

def lp_best_response(a, M, solver):
    """Solve max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1

        T[idx, j] = a[idx - j]

def compute_smooth_gradient(a, c, w, tau):
    """Compute gradient of tau*log(sum(exp(c/tau))) w.r.t. a."""
    n = len(a)
    size = 2 * n - 1
    # g[j] = (2/tau) * sum_{m=0}^{n-1} a[m] * w[j+m] (w=0 outside [0, size-1])
    # Use FFT: g_unscaled[j] = conv(a, reversed_w)[size-1-j]
    w_rev = w[::-1]  # reverse w
    # conv(a, w_rev) has length n + size - 1 = n + 2n - 1 - 1 = 3n - 2
    fft_size = 1 << (3 * n - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w_rev, fft_size)
    conv_result = irfft(A * W, fft_size)[:3 * n - 2]
    # g_unscaled[j] = conv_result[size-1-j]
    g_unscaled = conv_result[size - 1 :: -1][:n]  # conv_result[size-1], conv_result[size-2], ..., conv_result[size-n]
    g = (2.0 / tau) * g_unscaled
    return g


def smooth_gradient_descent(a0, time_limit, verbose=True):
    """Smooth gradient descent using log-sum-exp approximation."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = normalize_seq(a0, n)
    size = 2 * n - 1
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    t0 = time.monotonic()
    # Start with large tau, decrease over time
    tau_init = best_score * 0.5
    tau_min = best_score * 0.02
    n_iters = 0
    patience = 0
    
    # Learning rate schedule
    eta = 1.0 / (tau_init * n * 0.1)
    eta_min = 1e-10
    
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        elapsed = time.monotonic() - t0
        total_time = time_limit
        
        # Progressive tau schedule: tau decreases from tau_init to tau_min
        frac = min(1.0, elapsed / (total_time * 0.7))
        tau = tau_init * np.exp(np.log(tau_min / tau_init) * frac)
        tau = max(tau, tau_min)
        
        c = fast_autocorr(a)
        c_max = c.max()
        
        # Compute softmax weights (numerically stable)
        c_shifted = c - c.max()
        exp_c = np.exp(c_shifted / tau)
        w = exp_c / exp_c.sum()
        
        # Compute gradient
        g = compute_smooth_gradient(a, c, w, tau)
        
        # Adaptive learning rate: scale by gradient magnitude
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        # Line search-style step
        best_a_try = a.copy()
        best_conv = c_max
        for factor in [0.1, 0.3, 1.0, 3.0]:
            eta_try = eta * factor
            a_try = a - eta_try * g
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try * norm_val / s_try
            conv_try = fast_autocorr(a_try).max()
            if conv_try < best_conv - 1e-12:
                best_conv = conv_try
                best_a_try = a_try
                eta = max(eta_try, eta_min)
        
        a = best_a_try
        n_iters += 1
        
        if best_conv < best_score - 1e-10:
            best_score = best_conv
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience % 50 == 0:
                eta *= 0.8
        
        if patience > 200:
            # Perturb
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
            noise = rng.normal(0, norm_val * 0.01, n)
            a_pert = a + noise
            a_pert = np.maximum(a_pert, 0)
            s_pert = a_pert.sum()
            if s_pert > 1e-15:
                a = a_pert * norm_val / s_pert
            patience = 0
            eta = 1.0 / (tau * n * 0.1)
        
        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, tau={tau:.4f}, eta={eta:.6f}")
    
    return best_a, best_score


def targeted_cd(a0, time_limit, top_lags=10, verbose=True):
    """Coordinate descent targeting the top autocorrelation lags."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = normalize_seq(a0, n)
    
    best_a = a.copy()
    conv = fast_autocorr(a)
    best_score = conv.max()
    t0 = time.monotonic()
    size = 2 * n - 1
    rng = np.random.default_rng(999)
    no_improve = 0
    n_passes = 0
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        if no_improve > 300:
            break
        
        conv = fast_autocorr(a)
        max_val = conv.max()
        active_idx = np.argsort(conv)[-top_lags:]
        
        improved = False
        perm = rng.permutation(n)
        
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < norm_val * 1e-10:
                continue
            
            # Try shifting mass from i to j
            offsets = [1, -1, 2, -2, 3, -3, 5, -5, 7, -7, 10, -10]
            for off in offsets:
                j = (i + off) % n
                if j == i:
                    continue
                
                for frac in [0.2, 0.4, 0.6]:
                    d = a[i] * frac
                    if d < 1e-15:
                        continue
                    a[i] -= d
                    a[j] += d
                    new_conv = fast_autocorr(a)
                    new_max = new_conv.max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        no_improve = 0
                        break
                    else:
                        a[i] += d
                        a[j] -= d
                else:
                    continue
                break
        
        if not improved:
            no_improve += 1
            if no_improve >= 50:
                noise = rng.laplace(0, norm_val * 0.005, n)
                a_pert = a + noise
                a_pert = np.maximum(a_pert, 0)
                s_pert = a_pert.sum()
                if s_pert > 1e-15:
                    a = a_pert * norm_val / s_pert
                else:
                    break
                no_improve = 0
        
        if verbose and n_passes % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score
