import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate the score using the provided evaluate_sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_T(a, k_indices, n):
    """
    Build constraint matrix T where T[i,j] = a[k_indices[i]-j] if 0<=k_indices[i]-j<n else 0.
    Uses vectorized slice assignment for speed.
    """
    T = np.zeros((len(k_indices), n))
    for idx in range(len(k_indices)):
        k = k_indices[idx]
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo >= hi:
            continue
        # T[idx, lo:hi] = a[k-hi+1 : k-lo+1][::-1]
        T[idx, lo:hi] = a[k - hi + 1 : k - lo + 1][::-1]
    return T


def lp_solve(a, M, active_set, solver, n):
    """
    Solve LP: max sum(b) s.t. (a*b)[k] <= M for k in active_set, b >= 0.
    Returns (b, duals) or (None, None).
    """
    k_indices = np.asarray(active_set, dtype=int)
    if len(k_indices) == 0:
        return None, None
    T = build_T(a, k_indices, n)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
            duals = np.maximum(duals, 0.0)
    except Exception:
        pass
    return b_val, duals


def nesterov_lp(a0, time_limit, solver, verbose=False, seed=42):
    """
    Nesterov-accelerated LP best-response iteration.
    Minimizes max_k (a*a)[k] subject to sum(a) = sqrt(2n), a >= 0.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t_start = time.monotonic()

    y = a.copy()
    y_prev = a.copy()
    active_set = None
    stall_count = 0
    stall_threshold = 50
    n_iters = 0

    # Ergodic average tracking
    erg_window = 30
    erg_buf = [a.copy()]

    for t in range(1, 20000000):
        if time.monotonic() - t_start > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive active set: use tolerance that tightens over time
        need_new_active = (active_set is None) or (t % 30 == 0) or (stall_count > 0 and t % 5 == 0)
        if need_new_active:
            tol = max(0.005, 0.04 * np.exp(-t / 200.0))
            active_set = np.where(conv_y >= M_curr * (1.0 - tol))[0]
            # Always include top peaks
            n_top = max(5, min(20, n // 80))
            top_k = np.argsort(conv_y)[-n_top:]
            active_set = np.unique(np.concatenate([active_set, top_k]))
            # Cap active set size for speed
            if len(active_set) > 500:
                # Keep only the most important ones
                active_set = np.argsort(conv_y[active_set])[-500:]
                active_set = active_set[np.argsort(active_set)]

        b, duals = lp_solve(y, M_curr, active_set, solver, n)

        if b is None:
            # Try with full active set
            if t % 10 == 0:
                active_set = np.arange(2 * n - 1)
                b, duals = lp_solve(y, M_curr, active_set, solver, n)
            if b is None:
                continue

        # Check for constraint violations and expand if needed
        size = 2 * n - 1
        complement = np.setdiff1d(np.arange(size), active_set)
        if len(complement) > 0 and len(complement) < size:
            T_comp = build_T(y, complement, n)
            viol = T_comp @ b
            viol_idx = np.where(viol > M_curr * 1.001)[0]
            if len(viol_idx) > 0:
                new_active = np.union1d(active_set, complement[viol_idx])
                if len(new_active) > 0.85 * size:
                    new_active = np.arange(size)
                b, duals = lp_solve(y, M_curr, new_active, solver, n)
                if b is not None:
                    active_set = new_active
                else:
                    continue
            else:
                # No violations - could potentially shrink active set
                if len(active_set) > 50 and t % 100 == 0:
                    # Try removing low-dual constraints
                    if duals is not None and len(duals) == len(active_set):
                        keep = duals > np.percentile(duals, 50)
                        if keep.sum() >= 10:
                            active_set = active_set[keep]

        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * S / b_sum
        n_iters += 1

        # Nesterov update with adaptive momentum
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * S / s_y

        # Heavy ball momentum
        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.95)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * S / s_a

        y_prev = y
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Ergodic average
        erg_buf.append(a.copy())
        if len(erg_buf) > erg_window:
            erg_buf.pop(0)
        if len(erg_buf) == erg_window and t % 10 == 0:
            erg_avg = np.mean(erg_buf, axis=0)
            s_erg = erg_avg.sum()
            if s_erg > 1e-15:
                erg_avg = erg_avg * S / s_erg
                score_erg = fast_autocorr(erg_avg).max()
                if score_erg < best_score:
                    best_score = score_erg
                    best_a = erg_avg.copy()
                    stall_count = 0

        # Restart if stalled
        if stall_count > stall_threshold:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall_count = 0
            erg_buf = [a.copy()]
            if verbose and (t // 1000) % 5 == 0:
                elapsed = time.monotonic() - t_start
                print(f"    LP restart iter={t} ({elapsed:.0f}s) score={best_score:.10f}")

        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def coordinate_descent_fast(a0, time_limit, verbose=False):
    """
    Fast coordinate descent: for each coordinate, try several delta values
    and keep the best. Uses FFT to recompute autocorrelation each time.
    """
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * S / s0

    conv = fast_autocorr(a)
    best_a = a.copy()
    best_score = conv.max()
    t0 = time.monotonic()

    size = 2 * n - 1
    passes = 0
    no_improve = 0
    max_no_improve = 4

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]
            if aj < 1e-15:
                continue

            # Try several delta values
            deltas = np.array([
                -aj * 0.5, -aj * 0.3, -aj * 0.1,
                aj * 0.1, aj * 0.3, aj * 0.5, aj * 1.0
            ])

            best_delta = 0.0
            best_obj = conv.max()

            for d in deltas:
                if aj + d < -1e-12:
                    continue
                a_trial = a.copy()
                a_trial[j] = aj + d
                s_new = a_trial.sum()
                if s_new < 1e-15:
                    continue
                a_trial = a_trial * S / s_new
                conv_trial = fast_autocorr(a_trial)
                obj = conv_trial.max()
                if obj < best_obj - 1e-13:
                    best_obj = obj
                    best_delta = d

            if abs(best_delta) > 1e-12:
                a[j] = aj + best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[j] = aj
                    continue
                a = a * S / s_new
                conv = fast_autocorr(a)
                if conv.max() < best_score - 1e-13:
                    best_score = conv.max()
                    best_a = a.copy()
                    improved = True
                else:
                    a[j] = aj
                    a = a * S / a.sum()
                    conv = fast_autocorr(a)

        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        else:
            no_improve = 0

        if verbose and passes % 2 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def subgradient_polish(a0, time_limit, verbose=False):
    """
    Subgradient method: minimize max_k (a*a)[k] using gradient at active peaks.
    Gradient of (a*a)[k] w.r.t. a[j] = 2*a[k-j] if 0<=k-j<n else 0.
    """
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * S / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    step = 0.5 / n ** 0.4
    patience = 0
    max_patience = 3000
    size = 2 * n - 1

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()

        # Get top peaks for combined gradient
        n_peaks = max(3, min(15, n // 100))
        top_indices = np.argsort(conv)[-n_peaks:]
        # Weight by proximity to max
        weights = np.exp(-10.0 * (M_curr - conv[top_indices]) / max(M_curr * 0.01, 1e-10))
        weights = weights / weights.sum()

        # Combined gradient using the formula: grad[k][j] = 2*a[k-j] if 0<=k-j<n else 0
        grad = np.zeros(n)
        for w, k in zip(weights, top_indices):
            # grad += w * 2 * a[max(0, k-n+1) : min(n, k+1)]: shifted
            # The indices j where 0<=k-j<n are j in [max(0,k-n+1), min(n-1,k)]
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            # grad[j] += w * 2 * a[k-j] for j in [lo, hi)
            # a[k-j] for j=lo,...,hi-1: a[k-lo], a[k-lo-1], ..., a[k-hi+1]
            grad[lo:hi] += w * 2.0 * a[k - hi + 1 : k - lo + 1][::-1]

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        # Projected gradient step with normalization
        a_trial = a - step * grad
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * S / s_trial

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-13:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.01, 3.0)
        else:
            patience += 1
            step *= 0.97
            if step < 1e-10:
                break

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def try_symmetrize(a):
    """Try symmetrization: a = (a + reverse(a))/2."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    s = a_sym.sum()
    if s < 1e-15:
        return None
    return a_sym


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length, preserving shape."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss


def spectral_init(n, rng, mode=0):
    """Generate initial sequences with flat spectral properties."""
    if mode == 0:
        # Random with cosine modulation
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.0)
    elif mode == 1:
        # Gaussian bump
        x = np.linspace(0, 1, n)
        center = 0.5
        width = 0.25 + 0.15 * rng.uniform(0, 1)
        a = np.exp(-0.5 * ((x - center) / width) ** 2)
    elif mode == 2:
        # Flat with small random perturbation
        a = np.ones(n) * (1.0 + 0.1 * rng.standard_normal(n))
        a = np.maximum(a, 0.0)
    elif mode == 3:
        # Double bump
        x = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((x - 0.25) / 0.15) ** 2) + np.exp(-0.5 * ((x - 0.75) / 0.15) ** 2)
    else:
        a = np.ones(n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

        S = np.sqrt(2.0 * n_cur)
        a_norm = best_seq * S / best_seq.sum()
        a_out, _ = coordinate_descent_fast(a_norm, t2, verbose=True)
        try_update(a_out, f"cd1_n{n_cur}")

    # === Phase 3: Subgradient polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 20:
        print(f"\n=== Phase 3: Subgrad n={n_cur}, budget={t3:.0f}s ===")
        S = np.sqrt(2.0 * n_cur)
        a_norm = best_seq * S / best_seq.sum()
        a_out, _ = subgradient_polish(a_norm, t3, verbose=True)
        try_update(a_out, f"subg1_n{n_cur}")

    # === Phase 4: LP round 2 from improved point ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 40:
        print(f"\n=== Phase 4: LP round 2 n={n_cur}, budget={t4:.0f}s ===")
        S = np.sqrt(2.0 * n_cur)
        a_norm = best_seq * S / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t4, solver, verbose=True, seed=123)
        try_update(a_out, f"lp2_n{n_cur}")

    # === Phase 5: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        print(f"\n=== Phase 5: Multi-length exploration, remaining={remaining:.0f}s ===")
        n_base = len(best_seq)
        target_lengths = []
        for dl in [-100, -50, 50, 100, 200]:
            nl = n_base + dl
            if 800 <= nl <= 3000:
                target_lengths.append(nl)

        n_trials = min(3, len(target_lengths))
        t_each = (remaining * 0.30) / max(n_trials, 1)

        for i in range(n_trials):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            nl = target_lengths[i]
            t_trial = min(t_each, remaining - 30)
            if t_trial < 30:
                break

            print(f"  Trying n={nl} (budget={t_trial:.0f}s)")
            a_interp = interpolate_to_length(best_seq, nl)
            S = np.sqrt(2.0 * nl)
            a_interp = a_interp * S / a_interp.sum()

            # Quick LP
            t_lp = t_trial * 0.7
            a_out, _ = nesterov_lp(a_interp, t_lp, solver, verbose=False, seed=500 + i)
            try_update(a_out, f"len{nl}")

            # Quick CD
            remaining_trial = time_budget - (time.monotonic() - t_global)
            if remaining_trial < 30:
                break
            t_cd = min(t_trial * 0.25, remaining_trial - 20)
            if t_cd > 10:
                a_out2, _ = coordinate_descent_fast(a_out, t_cd, verbose=False)
                try_update(a_out2, f"cd_len{nl}")

    # === Phase 6: Perturbation + LP restart ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        print(f"\n=== Phase 6: Perturbation restarts, remaining={remaining:.0f}s ===")
        n_cur = len(best_seq)
        n_restarts = 2
        t_each = (remaining * 0.25) / n_restarts

        for r in range(n_restarts):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 50:
                break
            t_trial = min(t_each, remaining - 20)
            if t_trial < 20:
                break

            scale = 0.02 + 0.01 * r
            noise = 1.0 + scale * rng.standard_normal(n_cur)
            noise = np.maximum(noise, 0.1)
            pert = best_seq * noise
            s_p = pert.sum()
            if s_p < 0.01:
                continue
            S = np.sqrt(2.0 * n_cur)
            pert = pert * S / s_p
            a_out, _ = nesterov_lp(pert, t_trial, solver, verbose=False, seed=800 + r)
            try_update(a_out, f"pert{r}_n{n_cur}")

    # === Phase 7: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 7: Final CD, remaining={remaining:.0f}s ===")
        n_cur = len(best_seq)
        t7 = min(remaining * 0.30, 120.0)
        S = np.sqrt(2.0 * n_cur)
        a_norm = best_seq * S / best_seq.sum()
        a_out, _ = coordinate_descent_fast(a_norm, t7, verbose=True)
        try_update(a_out, f"cd_final")

    # === Phase 8: Final subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 8: Final subgrad, remaining={remaining:.0f}s ===")
        n_cur = len(best_seq)
        t8 = min(remaining * 0.4, 80.0)
        S = np.sqrt(2.0 * n_cur)
        a_norm = best_seq * S / best_seq.sum()
        a_out, _ = subgradient_polish(a_norm, t8, verbose=False)
        try_update(a_out, "subg_final")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0 / 1600] * 1600

    return [float(x) for x in best_seq]
