def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    v = np.maximum(v, 0)
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n + 1) > (css - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def nesterov_lp_improved(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stuck_count = 0
    avg_a = np.zeros(n)
    n_avg = 0
    best_avg = a.copy()
    best_avg_score = best_score

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-14:
            break

        # Dynamic constraint selection: tighten threshold over time
        progress = min(1.0, n_iters / 500.0)
        threshold = 0.80 + 0.15 * progress
        active = np.where(conv_y >= threshold * M_curr)[0]

        # Constraint generation: iteratively add violated constraints
        max_active = 150
        for cg_iter in range(20):
            if len(active) >= max_active:
                break
            if len(active) < 5:
                active = np.where(conv_y >= 0.5 * M_curr)[0]
                if len(active) < 3:
                    active = np.arange(len(conv_y))
                break

            b = lp_best_response(y, M_curr, solver, active)
            if b is None:
                break

            # Check if b violates non-active constraints
            conv_b = fast_autocorr(b)
            M_b = M_curr  # We need conv(y, b) <= M_curr, not conv(b, b) <= M_curr
            # Actually the constraint is conv(y, b)_k <= M_curr for all k
            conv_yb = np.convolve(y, b, mode='full')
            violations = np.where(conv_yb > M_curr * (1 + 1e-6))[0]
            non_active_violations = [v for v in violations if v not in set(active.tolist())]
            if non_active_violations:
                # Add the most violated non-active constraint
                worst = max(non_active_violations, key=lambda k: conv_yb[k])
                active = np.append(active, worst)
            else:
                break

        if b is None:
            # Fallback: use all constraints
            if len(active) < len(conv_y):
                active = np.arange(len(conv_y))
                b = lp_best_response(y, M_curr, solver, active)
            if b is None:
                stuck_count += 1
                if stuck_count > 50:
                    break
                continue

        stuck_count = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update with adaptive momentum
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        # Running average (Miranomski trick)
        avg_a += a
        n_avg += 1
        if n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_avg_score:
                    best_avg_score = score_avg
                    best_avg = avg_check.copy()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, avg={best_avg_score:.8f}")

    # Return the better of best_a and best_avg
    if best_avg_score < best_score:
        return best_avg, best_avg_score
    return best_a, best_score


def smooth_power_mean_optimize(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Power mean schedule: increase p for sharper max approximation
    p_schedule = [4, 6, 8, 12, 16, 24, 32, 48, 64, 96, 128, 192]

    for p in p_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        n_steps = max(20, min(300, int(remaining * 2)))

        y = a.copy()
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            c = fast_autocorr(y)
            c_max = c.max() + 1e-30
            # Power mean: f = (sum c^p)^(1/p)
            c_p = np.maximum(c, 0) ** p
            sum_cp = c_p.sum()
            if sum_cp < 1e-30:
                break
            wm = c_p / sum_cp  # Weight distribution

            # Gradient: d/dy_i [f] = (1/p) * (sum c^p)^(1/p - 1) * p * c^p-1 * dc/dy_i
            # = (sum c^p)^(1-1/p) * c^{p-1} * dc/dy_i
            # dc/dy_i at position k = y[k-i] + y[i-k] (for valid indices)
            # Using cross-correlation: dc/dy_i = 2 * cross_corr(w, y)[i] (approximately)
            size = 2 * n - 1
            fft_size = 1 << (size - 1).bit_length()
            W = rfft(wm, fft_size)
            Y = rfft(y, fft_size)
            cross = irfft(W * np.conj(Y), fft_size)[:n]

            # Scale factor for power mean gradient
            scale = sum_cp ** (1.0 - 1.0 / p)
            grad = scale * (np.maximum(c, 0) ** (p - 1))  # This is for the smooth function
            # Actually we need: grad_i = sum_k wm[k] * (T@e_i)[k] where T is conv matrix
            # = sum_k wm[k] * (conv(y, e_i))[k] = cross_corr(wm, y)[i] (approximately)
            # Let me use a simpler formulation:
            # f(y) = (sum_k c_k^p)^(1/p), c = conv(y,y)
            # df/dy_i = (1/p)(sum c^p)^(1/p - 1) * sum_k p*c_k^{p-1} * dc_k/dy_i
            # dc_k/dy_i = y[k-i] + y[i-k] (for linear conv)
            # So df/dy_i = (sum c^p)^(1/p-1) * sum_k c_k^{p-1} * (y[k-i] + y[i-k])
            # = (sum c^p)^(1/p-1) * 2 * (c^{p-1} * y-corr)[i] -- this is getting complex
            # Let's use the simpler weight-based gradient:
            grad = 2.0 * cross

            # Adaptive step size
            step = norm / (n * p * 0.5 * (1.0 + 0.02 * it))
            step = min(step, norm / (n * 0.3))

            a_new = y - step * grad
            a_new = project_simplex(a_new, norm)

            # Nesterov
            alpha_nt = 2.0 / (it + 2.0)
            y_new = (1.0 - alpha_nt) * y + alpha_nt * a_new
            y_new = project_simplex(y_new, norm)

            if it == 0:
                a_curr = a_new.copy()
            else:
                momentum = (it - 1.0 + 1.0) / (it + 2.0)
                a_curr = y_new + momentum * (y_new - y)
                a_curr = project_simplex(a_curr, norm)

            y = y_new
            a = a_curr

            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

            score_y = fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    PowerMean p={p} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_cd_wide(a0, time_limit, rng, window=80, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-12:
                continue

            # Try moving mass to nearby positions
            max_offset = min(window, n - 1)
            for offset in range(1, max_offset + 1):
                for sign in [1, -1]:
                    j = (i + sign * offset) % n
                    if j == i:
                        continue
                    if a[i] < norm * 1e-12:
                        break

                    # Try multiple step sizes
                    for frac in [0.5, 0.25, 0.1]:
                        test_d = a[i] * frac
                        a_test = a.copy()
                        a_test[i] -= test_d
                        a_test[j] += test_d
                        s_test = a_test.sum()
                        if s_test < 1e-15:
                            continue
                        a_test = a_test * norm / s_test
                        conv_test = fast_autocorr(a_test)
                        if conv_test.max() < conv.max() - 1e-14:
                            a = a_test
                            conv = conv_test
                            if conv.max() < best_score - 1e-14:
                                best_score = conv.max()
                                best_a = a.copy()
                            pass_improved = True
                            no_improve = 0
                            break  #Break fraction loop, try next offset

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD-wide pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_cd_focused(a0, time_limit, rng, verbose=True):
    """Focus on reducing the peak specifically by redistributing mass from peak-contributing positions."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    n_iters = 0

    while time.monotonic() - t0 < time_limit:
        n_iters += 1
        conv = fast_autocorr(a)
        peak_val = conv.max()
        peak_pos = np.argmax(conv)

        # Find which a[i] contribute most to the peak
        # conv[peak_pos] = sum_i a[i] * a[peak_pos - i]
        contributions = np.zeros(n)
        i_start = max(0, peak_pos - n + 1)
        i_end = min(peak_pos, n - 1)
        for i in range(i_start, i_end + 1):
            contributions[i] = a[i] * a[peak_pos - i]

        # Pick the largest contributor and try to reduce it
        top_idx = np.argsort(contributions)[::-1][:5]

        improved = False
        for i in top_idx:
            if a[i] < 1e-15:
                continue
            # Try moving a small amount from position i to a "quiet" position
            # Find positions where a is small
            quiet_positions = np.where(a < 0.1 * a.max())[0]
            if len(quiet_positions) == 0:
                quiet_positions = np.argsort(a)[:n // 4]

            for j in rng.choice(quiet_positions, size=min(10, len(quiet_positions)), replace=False):
                if j == i:
                    continue
                for frac in [0.3, 0.15, 0.05]:
                    test_d = a[i] * frac
                    a_test = a.copy()
                    a_test[i] -= test_d
                    a_test[j] += test_d
                    s_test = a_test.sum()
                    if s_test < 1e-15:
                        continue
                    a_test = a_test * norm / s_test
                    conv_test = fast_autocorr(a_test)
                    if conv_test.max() < peak_val - 1e-14:
                        a = a_test
                        if conv_test.max() < best_score - 1e-14:
                            best_score = conv_test.max()
                            best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if verbose and n_iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD-focus iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def generate_initializations(n, rng):
    """Generate diverse structural initializations."""
    inits = []

    # 1. Uniform
    inits.append(('uniform', np.ones(n) / n))

    # 2. Linear ramp (decreasing)
    ramp = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(('ramp_dec', ramp / ramp.sum()))

    # 3. Linear ramp (increasing)
    ramp_inc = np.arange(1, n + 1, dtype=np.float64)
    inits.append(('ramp_inc', ramp_inc / ramp_inc.sum()))

    # 4. Quadratic bump
    x = np.linspace(0, 1, n)
    bump = (x * (1 - x)) ** 0.5
    inits.append(('quad_bump', bump / bump.sum()))

    # 5. Gaussian
    center = n / 2.0
    sigma = n / 6.0
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    inits.append(('gaussian', gauss / gauss.sum()))

    # 6. Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    inits.append(('raised_cos', rc / rc.sum()))

    # 7. Bimodal
    c1 = n * 0.3
    c2 = n * 0.7
    sigma_b = n / 12.0
    bimodal = np.exp(-0.5 * ((np.arange(n) - c1) / sigma_b) ** 2) + np.exp(-0.5 * ((np.arange(n) - c2) / sigma_b) ** 2)
    inits.append(('bimodal', bimodal / bimodal.sum()))

    # 8. Step function (half high, half low)
    step = np.ones(n)
    step[:n // 2] = 2.0
    step[n // 2:] = 1.0
    inits.append(('step', step / step.sum()))

    # 9. Power law ramp
    x = np.arange(n, 0, -1, dtype=np.float64)
    power = x ** 2
    inits.append(('power_ramp', power / power.sum()))

    # 10. Random smooth
    rng_init = np.random.default_rng(12345)
    random_smooth = np.abs(rng_init.standard_normal(n)).cumsum()
    random_smooth -= random_smooth.min()
    random_smooth += 0.1
    inits.append(('random_smooth', random_smooth / random_smooth.sum()))

    return inits


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    return f_interp * np.sqrt(2.0 * new_n) / ss


def refine_sequence(a, time_limit, solver, rng, verbose=True):
    """Multi-method refinement of a single sequence."""
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
        s = norm
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    def try_update(candidate, label):
        nonlocal best_a, best_score
        if candidate is None:
            return
        c = np.maximum(np.asarray(candidate, dtype=np.float64), 0.0)
        sc = c.sum()
        if sc < 1e-15:
            return
        c = c * norm / sc
        score = fast_autocorr(c).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = c.copy()
            if verbose:
                print(f"      [{label}] improved to {best_score:.8f}")

    # Sub-phase 1: LP Nesterov (50% of time)
    remaining = time_limit - (time.monotonic() - t0)
    t1 = remaining * 0.50
    if t1 > 10:
        out, _ = nesterov_lp_improved(best_a, t1, solver, verbose=False)
        try_update(out, "LP")

    # Sub-phase 2: Power mean smooth (25% of time)
    remaining = time_limit - (time.monotonic() - t0)
    t2 = remaining * 0.25
    if t2 > 5:
        out, _ = smooth_power_mean_optimize(best_a, t2, verbose=False)
        try_update(out, "PowerMean")

    # Sub-phase 3: Focused pair CD (15% of time)
    remaining = time_limit - (time.monotonic() - t0)
    t3 = remaining * 0.15
    if t3 > 5:
        out, _ = pair_cd_focused(best_a, t3, rng, verbose=False)
        try_update(out, "PCD-focus")

    # Sub-phase 4: Wide pair CD (10% of time)
    remaining = time_limit - (time.monotonic() - t0)
    t4 = remaining * 0.10
    if t4 > 5:
        out, _ = pair_cd_wide(best_a, t4, rng, window=60, verbose=False)
        try_update(out, "PCD-wide")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function. Returns best sequence found within time budget."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    total_budget = min(float(budget_s), 1780.0)

    solver = get_solver()

    best_seq = None
    best_score = float('inf')

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Try to load parent
    n_base = 1600
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    # Generate diverse initializations
    inits = generate_initializations(n_base, rng)

    # If we have a parent, add it as the first initialization (normalized)
    parent_norm = best_seq / best_seq.sum()
    inits.insert(0, ('parent', parent_norm))

    print(f"\n=== Phase 1: Multi-start LP refinement (n={n_base}) ===")
    remaining = total_budget - (time.monotonic() - t0
        if False else time.monotonic() - t_global)
    phase1_budget = remaining * 0.50
    n_inits = len(inits)
    per_init_time = phase1_budget / n_inits

    for idx, (name, init) in enumerate(inits):
        elapsed = time.monotonic() - t_global
        remaining = total_budget - elapsed
        if remaining < 30:
            break

        # Allocate time: give more time to parent and early inits
        if name == 'parent':
            t_alloc = min(per_init_time * 2.5, remaining * 0.4)
        elif idx < 3:
            t_alloc = min(per_init_time * 1.5, remaining * 0.25)
        else:
            t_alloc = min(per_init_time, remaining * 0.15)

        t_alloc = max(t_alloc, 5.0)

        # Convert to norm = sqrt(2n) for internal optimization
        n_c = len(init)
        a_norm = init * np.sqrt(2.0 * n_c) / init.sum()

        if name == 'parent':
            # For parent, use full refinement
            out, _ = refine_sequence(a_norm, t_alloc, solver, rng, verbose=True)
        else:
            # For other inits, just do LP Nesterov
            out, _ = nesterov_lp_improved(a_norm, t_alloc, solver, verbose=False)

        do_try_update(out, f"init_{name}")

    # Phase 2: Focus refinement on the best sequence
    print(f"\n=== Phase 2: Focused refinement (best_score={best_score:.8f}) ===")
    remaining = total_budget - (time.monotonic() - t_global)
    phase2_budget = remaining * 0.30

    if phase2_budget > 20 and best_seq is not None:
        n_c = len(best_seq)
        a_nom = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        out, _ = refine_sequence(a_nom, phase2_budget, solver, rng, verbose=True)
        do_try_update(out, "refined")

    # Phase 3: Power mean smooth on best
    print(f"\n=== Phase 3: Power mean smooth ===")
    remaining = total_budget - (time.monotonic() - t_global)
    phase3_budget = remaining * 0.25
    if phase3_budget > 10 and best_seq is not None:
        n_c = len(best_seq)
        a_nom = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        out, _ = smooth_power_mean_optimize(a_nom, phase3_budget, verbose=True)
        do_try_update(out, "power_mean")

    # Phase 4: Aggressive coordinate descent
    print(f"\n=== Phase 4: Aggressive PCD ===")
    remaining = total_budget - (time.monotonic() - t_global)
    phase4_budget = remaining * 0.20
    if phase4_budget > 10 and best_seq is not None:
        n_c = len(best_seq)
        a_nom = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        out, _ = pair_cd_wide(a_nom, phase4_budget, rng, window=100, verbose=True)
        do_try_update(out, "pcd_wide")
        remaining = total_budget - (time.monotonic() - t_global)
        if remaining > 10:
            a_nom = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
            out, _ = pair_cd_focused(a_nom, min(remaining, phase4_budget * 0.5), rng, verbose=True)
            do_try_update(out, "pcd_focus")

    # Phase 5: Try alternate lengths
    print(f"\n=== Phase 5: Alternate lengths ===")
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 60 and best_seq is not None:
        n_current = len(best_seq)
        alt_lengths = [n_current + 200, n_current - 200, n_current + 400]
        for n_alt in alt_lengths:
            if n_alt < 100:
                continue
            remaining = total_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            t_alt = min(remaining * 0.3, 60.0)
            init_alt = interpolate_sequence(best_seq, n_alt)
            a_nom = init_alt * np.sqrt(2.0 * n_alt) / init_alt.sum()
            out, _ = nesterov_lp_improved(a_nom, t_alt, solver, verbose=False)
            do_try_update(out, f"alt_n{n_alt}")

    # Final: one more LP pass on absolute best
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 30 and best_seq is not None:
        print(f"\n=== Final polish ===")
        n_c = len(best_seq)
        a_nom = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        out, _ = nesterov_lp_improved(a_nom, min(remaining, 120.0), solver, verbose=True)
        do_try_update(out, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        best_seq = np.ones(100) / 100

    # Safety: ensure return value is valid
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_check = evaluate_safe(result)
    if not np.isfinite(final_check):
        result = [1.0] * 100
    return result
