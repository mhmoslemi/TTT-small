If we let `s = sum(a)`, then the objective is `2n * max_b / s^2`. We can normalize `s = sqrt(2n)` WLOG (since scaling a by c scales max_b by c^2 and s^2 by c^2, so the ratio is invariant). So we want to minimize `max(conv(a,a))` subject to `sum(a) = sqrt(2n)` and `a >= 0`.

For a uniform sequence `a[i] = 1/sqrt(2n) * sqrt(2n) = 1` (wait, sum = n * a_i = sqrt(2n), so a_i = sqrt(2n)/n = sqrt(2/n)). The optimum should be lower than 2. The theoretical infimum for this type of problem (related to Sidon sets, Costas arrays, etc.) is related to the minimum possible peak of the autocorrelation.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LSE-smoothed gradient descent for fast convergence
2. LP best-response with Nesterov acceleration
3. Exact 1D line search pair coordinate descent
4. Adaptive length exploration

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def lse_gd(a0, time_limit, verbose=True):
    """LSE-smoothed gradient descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # More aggressive temperature schedule
    n_stages = 15
    temps = np.geomspace(100.0, 0.05, n_stages)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(50, min(200, int(remaining * 8)))
        
        # Momentum
        momentum = np.zeros(n)
        beta = 0.85
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)
            
            lr = 1.0 / (np.sqrt(n) * (1.0 + 0.02 * it))
            momentum = beta * momentum - lr * grad
            a_new = a + momentum
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new
            
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose:
            print(f"    LSE stage {si} T={T_val:.3f}: best={best_score:.8f}")
    
    return best_a, best_score

def lp_nesterov(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration and adaptive constraints."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break
        
        # Adaptive constraint selection: use top fraction
        # Start with tight (top 5%), relax to top 30% if few constraints
        top_frac = min(0.3, 0.05 + 0.01 * (t // 100))
        k = max(20, int(size * top_frac))
        top_idx = np.argsort(conv_y)[-k:]
        
        # Build constraint matrix for active constraints
        # T @ b <= M where T[m, i] = a[m-i] if 0<=m-i<n else 0 (for the "first" part)
        # Actually T is the full convolution matrix: (T @ b)[m] = sum_i T[m,i] * b[i]
        # where T[m,i] = a[m-i] if valid, which gives (a*b)[m]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [cp.sum(b_var) >= 1]  # Scale constraint
        
        # Build active constraint rows
        T_active = np.zeros((len(top_idx), n))
        for idx_m, m in enumerate(top_idx):
            for i in range(n):
                j = m - i
                if 0 <= j < n:
                    T_active[idx_m, i] = a[j]
        
        constraints.append(T_active @ b_var <= M)
        
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            # Fallback: solve with all constraints
            T_full = np.zeros((size, n))
            for m in range(size):
                for i in range(n):
                    j = m - i
                    if 0 <= j < n:
                        T_full[m, i] = a[j]
            prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [cp.sum(b_var) >= 1, T_full @ b_var <= M])
            try:
                prob.solve(solver=solver, verbose=False)
            except Exception:
                continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        val = np.asarray(b_var.value, dtype=np.float64)
        b = np.maximum(val, 0.0)
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def exact_1d_pair_cd(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent with exact 1D minimization via ternary search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv_self(a).max()
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue
            
            # Select candidate j values
            j_set = set()
            # Random candidates
            for _ in range(8):
                j_set.add(int(rng.integers(n)))
            # Structured: near peak lags
            peak_lags = np.where(conv >= cur_max * 0.93)[0]
            for kl in peak_lags[:5]:
                j_struct = (kl - i) % n
                j_set.add(j_struct)
                j_struct2 = (kl + i) % n
                j_set.add(j_struct2)
            j_set.discard(i)
            
            best_d = 0.0
            best_val = cur_max
            best_j = -1
            
            for j in list(j_set)[:12]:
                if j == i:
                    continue
                aj = a[j]
                
                # Compute gain vector g where new_conv[k] = conv[k] + d*g[k] + quadratic corrections
                # For transfer d from i to j:
                # a[i] -= d, a[j] += d
                # Change in conv[k] = sum_m a_new[m]*a_new[k-m] - sum_m a[m]*a[m+k]
                # = d * (sum_m [delta(m-j)*a[k-m] - delta(m-i)*a[k-m]]) 
                #   + d * (sum_m [a[m]*delta(k-m-j) - a[m]*delta(k-m-i)])
                #   - d^2 * delta(k-2i) + d^2 * delta(k-2j)
                # Wait, let me be more careful.
                # conv_new[k] = sum_m a_new[m] * a_new[k-m]
                # Let a_new[m] = a[m] + d*(delta(m-j) - delta(m-i))
                # conv_new[k] = sum_m (a[m] + d*(δmj-δmi)) * (a[k-m] + d*(δ(k-m-j)-δ(k-m-i)))
                # = conv[k] + d * sum_m (δmj-δmi)*a[k-m] + d * sum_m a[m]*(δ(k-m-j)-δ(k-m-i))
                #   + d^2 * sum_m (δmj-δmi)*(δ(k-m-j)-δ(k-m-i))
                # First correction: d * (a[k-j] - a[k-i]) [if indices valid]
                # Second correction: d * (a[j... wait, sum_m a[m]*δ(k-m-j) = a[k-j] if 0<=k-j<n, and sum_m a[m]*δ(k-m-i) = a[k-i] if 0<=k-i<n)
                # So second correction is also d * (a[k-j] - a[k-i])
                # Third (quadratic): d^2 * [(δmj)(δ(k-m-j)) - (δmj)(δ(k-m-i)) - (δmi)(δ(k-m-j)) + (δmi)(δ(k-m-i))]
                # = d^2 * [δ(k-2j) - δ(k-i-j) - δ(k-i-j) + δ(k-2i)]
                # = d^2 * [δ(k-2j) + δ(k-2i) - 2*δ(k-i-j)]
                
                # So: conv_new[k] = conv[k] + 2d*(a_valid[k-j] - a_valid[k-i]) + d^2*(δ(k-2j) + δ(k-2i) - 2*δ(k-i-j))
                # where a_valid[x] = a[x] if 0<=x<n else 0
                
                # For efficiency, compute the linear part
                lin = np.zeros(size)
                # a[k-j] for k in range(size), valid when 0<=k-j<n i.e. j<=k<j+n
                lo1 = max(0, j)
                hi1 = min(size, j + n)
                if hi1 > lo1:
                    lin[lo1:hi1] += a[0:hi1-lo1]
                # -a[k-i] for k in range(size), valid when 0<=k-i<n i.e. i<=k<i+n
                lo2 = max(0, i)
                hi2 = min(size, i + n)
                if hi2 > lo2:
                    lin[lo2:hi2] -= a[0:hi2-lo2]
                
                # Quadratic corrections at specific positions
                q_pos_2j = 2 * j
                q_pos_2i = 2 * i
                q_pos_ij = i + j  # where delta(k-i-j) is 1
                
                def score_at_d(d):
                    """Compute max of conv_new at given d."""
                    val = conv + 2.0 * d * lin
                    if 0 <= q_pos_2j < size:
                        val[q_pos_2j] += d * d
                    if 0 <= q_pos_2i < size:
                        val[q_pos_2i] += d * d
                    if 0 <= q_pos_ij < size:
                        val[q_pos_ij] -= 2.0 * d * d
                    return val.max()
                
                # Ternary search for optimal d in [0, ai*0.95]
                d_max = ai * 0.95
                a_lo = 0.0
                a_hi = d_max
                for _ in range(30):
                    m1 = a_lo + (a_hi - a_lo) / 3.0
                    m2 = a_hi - (a_hi - a_lo) / 3.0
                    v1 = score_at_d(m1)
                    v2 = score_at_d(m2)
                    if v1 < v2:
                        a_hi = m2
                    else:
                        a_lo = m1
                
                d_opt = (a_lo + a_hi) / 2.0
                v_opt = score_at_d(d_opt)
                
                if v_opt < best_val - 1e-12:
                    best_val = v_opt
                    best_d = d_opt
                    best_j = j
            
            if best_j >= 0 and best_d > 1e-10:
                a[i] -= best_d
                a[best_j] += best_d
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
            if no_improve >= 2:
                # Perturb to escape
                n_pert = max(10, n // 25)
                idx_p = rng.choice(n, size=n_pert, replace=False)
                a[idx_p] *= (1.0 + 0.12 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0
        
        if verbose and (idx + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PairCD ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def peak_shaving(a0, time_limit, rng, verbose=True):
    """Directly target the highest peaks by redistributing mass away from contributing positions."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv_self(a).max()
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        
        # Find the top peaks
        n_peaks = min(20, n // 10)
        peak_positions = np.argsort(conv)[-n_peaks:]
        peak_positions = peak_positions[peak_positions <= n - 1]  # Only consider valid lags (center)
        if len(peak_positions) == 0:
            n_peaks = min(10, n // 10)
            peak_positions = np.argsort(conv)[-n_peaks:]
        
        improved = False
        for pk in peak_positions:
            if time.monotonic() - t0 > time_limit:
                break
            
            # For peak at lag pk, the contribution is sum_m a[m]*a[m+pk-(n-1)]
            # Actually for np.convolve, the peak at index pk means:
            # conv[pk] = sum_m a[m] * a[pk - m] where 0<=pk-m<n
            # The "center" of the convolution is at n-1.
            # Positions that contribute to peak at pk: pairs (m, pk-m) where both in [0,n-1]
            
            # Compute per-position "responsibility" for this peak
            resp = np.zeros(n)
            for m in range(n):
                j = pk - m
                if 0 <= j < n:
                    resp[m] += a[j]
                    resp[j] += a[m]  # symmetric contribution
            
            # Find positions with high responsibility but low "importance" elsewhere
            # Reduce the highest-responsibility positions slightly, add to low-responsibility ones
            
            top_resp = np.argsort(resp)[-5:]
            low_resp = np.argsort(resp)[:5]
            
            for i in top_resp:
                if a[i] < 1e-10:
                    continue
                for j in low_resp:
                    if j == i:
                        continue
                    # Try small transfer
                    d = a[i] * 0.1
                    a[i] -= d
                    a[j] += d
                    s_new = a.sum()
                    if s_new > 1e-15:
                        a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        improved = True
                    else:
                        # Revert
                        a[i] += d
                        a[j] -= d
                        s_new = a.sum()
                        if s_new > 1e-15:
                            a = a * norm / s_new
        
        if not improved:
            no_improve += 1
            if no_improve >= 8:
                break
        else:
            no_improve = 0
    
    return best_a, best_score

def structured_init(n, rng):
    choice = rng.integers(0, 8)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        for val in a_list:
            if val < 0 or not np.isfinite(val):
                return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LSE-smoothed gradient descent with momentum for fast convergence
2. LP best-response with Nesterov acceleration and adaptive constraint screening
3. Exact 1D ternary search pair coordinate descent
4. Peak shaving for direct peak reduction
5. Adaptive length exploration for broader search
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_T_rows(a, indices):
    """Build rows of the convolution matrix T for given lag indices."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((len(indices), n))
    for idx_m, m in enumerate(indices):
        if m < 0 or m >= size:
            continue
        # T[m, i] = a[m - i] if 0 <= m-i < n
        lo_i = max(0, m - n + 1)
        hi_i = min(n, m + 1)
        for i in range(lo_i, hi_i):
            j = m - i
            T[idx_m, i] = a[j]
    return T


def lse_gd(a0, time_limit, verbose=True):
    """LSE-smoothed gradient descent with momentum."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 14
    temps = np.geomspace(80.0, 0.05, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(40, min(200, int(remaining * 7)))

        momentum = np.zeros(n)
        beta = 0.82

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -80.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            lr = 1.0 / (np.sqrt(n) * (1.0 + 0.015 * it))
            momentum = beta * momentum - lr * grad
            a_new = a + momentum
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.3f}: best={best_score:.8f}")

    return best_a, best_score


def lp_nesterov(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Adaptive constraint selection
        top_frac = min(0.25, 0.05 + 0.005 * (t // 50))
        k = max(30, int(size * top_frac))
        top_idx = np.argsort(conv_y)[-k:]

        # Build active constraint matrix
        T_active = build_T_rows(y, top_idx)

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_active @ b_var <= M, cp.sum(b_var) >= 1]

        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            # Fallback with fewer constraints
            k2 = max(20, k // 2)
            top_idx2 = np.argsort(conv_y)[-k2:]
            T_active2 = build_T_rows(y, top_idx2)
            b_var2 = cp.Variable(n, nonneg=True)
            constraints2 = [T_active2 @ b_var2 <= M, cp.sum(b_var2) >= 1]
            prob2 = cp.Problem(cp.Maximize(cp.sum(b_var2)), constraints2)
            try:
                prob2.solve(solver=solver, verbose=False)
            except Exception:
                continue
            if prob2.status not in ('optimal', 'optimal_inaccurate'):
                continue
            val = np.asarray(b_var2.value, dtype=np.float64)
        else:
            val = np.asarray(b_var.value, dtype=np.float64)

        b = np.maximum(val, 0.0)
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        momentum_coef = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum_coef * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def ternary_score(conv, lin, d, q2j, q2i, qij, size):
    """Compute max of conv + 2*d*lin + quadratic corrections at given d."""
    val = conv + 2.0 * d * lin
    if 0 <= q2j < size:
        val[q2j] += d * d
    if 0 <= q2i < size:
        val[q2i] += d * d
    if 0 <= qij < size:
        val[qij] -= 2.0 * d * d
    return val.max()


def exact_1d_pair_cd(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent with exact 1D ternary search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv_self(a).max()
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_imp = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        # Precompute peak lags
        peak_lags = np.where(conv >= cur_max * 0.94)[0]

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            # Candidate j values
            j_set = set()
            for _ in range(6):
                j_set.add(int(rng.integers(n)))
            for kl in peak_lags[:4]:
                j1 = (int(kl) - i) % n
                j2 = (int(kl) + i) % n
                j_set.add(j1)
                j_set.add(j2)
            j_set.discard(i)

            best_d = 0.0
            best_val = cur_max
            best_j = -1

            for j in list(j_set)[:10]:
                if j == i:
                    continue

                # Compute linear gain vector
                lin = np.zeros(size)
                lo1 = max(0, j)
                hi1 = min(size, j + n)
                if hi1 > lo1:
                    lin[lo1:hi1] += a[0:hi1 - lo1]
                lo2 = max(0, i)
                hi2 = min(size, i + n)
                if hi2 > lo2:
                    lin[lo2:hi2] -= a[0:hi2 - lo2]

                q2j = 2 * j
                q2i = 2 * i
                qij = i + j

                # Ternary search
                d_max = ai * 0.95
                a_lo = 0.0
                a_hi = d_max
                for _ in range(28):
                    m1 = a_lo + (a_hi - a_lo) / 3.0
                    m2 = a_hi - (a_hi - a_lo) / 3.0
                    v1 = ternary_score(conv, lin, m1, q2j, q2i, qij, size)
                    v2 = ternary_score(conv, lin, m2, q2j, q2i, qij, size)
                    if v1 < v2:
                        a_hi = m2
                    else:
                        a_lo = m1

                d_opt = (a_lo + a_hi) / 2.0
                v_opt = ternary_score(conv, lin, d_opt, q2j, q2i, qij, size)

                if v_opt < best_val - 1e-12:
                    best_val = v_opt
                    best_d = d_opt
                    best_j = j

            if best_j >= 0 and best_d > 1e-10:
                a[i] -= best_d
                a[best_j] += best_d
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
                    total_imp += 1

            if no_improve >= 5:
                break
            if no_improve >= 2:

                idx_p = rng.choice(n, size=n_pert, replace=False)
                a[idx_p] *= (1.0 + 0.1 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and idx % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PairCD ({elapsed:.0f}s): best={best_score:.8f}, imp={total_imp}")

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()

        # Focus on central region (lags near n-1)
        center = n - 1
        window = max(5, n // 20)
        lo_c = max(0, center - window)
        hi_c = min(size, center + window + 1)
        central_conv = conv[lo_c:hi_c]
        top_peaks = np.argsort(central_conv)[-min(15, len(central_conv)):]

        improved = False
        for pk_idx in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            pk = lo_c + int(pk_idx)

            # Compute responsibility for this peak
            # conv[pk] = sum_m a[m] * a[pk-m] for valid m
            resp = np.zeros(n)
            for m in range(n):
                j = pk - m
                if 0 <= j < n:
                    resp[m] += a[j]
                    resp[j] += a[m]

            top_resp = np.argsort(resp)[-4:]
            low_resp = np.argsort(resp)[:4]

            for i in top_resp:
                if a[i] < 1e-10:
                    continue
                for j in low_resp:
                    if j == i:
                        continue
                    # Try multiple transfer amounts
                    for frac in [0.05, 0.1, 0.15, 0.2]:
                        d = a[i] * frac
                        if d < 1e-12:
                            continue
                        a[i] -= d
                        a[j] += d
                        s_new = a.sum()
                        if s_new > 1e-15:
                            a = a * norm / s_new
                        new_max = fast_conv_self(a).max()
                        if new_max < best_score - 1e-12:
                            best_score = new_max
                            best_a = a.copy()
                            improved = True
                        else:
                            a[i] += d
                            a[j] -= d
                            s_new = a.sum()
                            if s_new > 1e-15:
                                a = a * norm / s_new

        if not improved:
            no_improve += 1
            if no_improve >= 10:
                break
        else:
            no_improve = 0

        if verbose and no_improve % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    choice = int(rng.integers(0, 10))
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.3
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.15 * np.cos(2 * np.pi * 7 * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-5.0 * x) + 0.4 + 0.2 * np.sin(3 * np.pi * x)
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        a = np.sinc(2.5 * (x - 0.5)) ** 2 + 0.08
    else:
        a = np.exp(-3.0 * (x - 0.5) ** 2 / 0.1)
    a = np.maximum(a, 0.0)
    a = a + 0.01 * rng.random(n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def eval_fn(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        if len(a_list) == 0:
            return float('inf')
        for val in a_list:
            if val < 0 or not np.isfinite(val):
                return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def get_solver():
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x_var = cp.Variable(4, nonneg=True)
            test_prob = cp.Problem(cp.Minimize(cp.sum(x_var)), [cp.sum(x_var) == 1, x_var <= 0.4])
            test_prob.solve(solver=s)
            if test_prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    print("  Using solver (fallback): ECOS")
    return cp.ECOS


def propose_candidate(seed=42, budget_s=1800.0):

    Multi-method optimization with exact 1D line search, adaptive length,
    and LP Nesterov for the autocorrelation peak minimization problem.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    # Phase 1: LSE GD warmup
    remaining = time_budget - (time.monotonic() - t_global)
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)
    t1 = min(remaining * 0.10, 120.0)
    print(f"\n=== P1: LSE GD n={n_base} ({t1:.0f}s) ===")
    if t1 > 10:
        a_out, _ = lse_gd(best_seq.copy(), t1, verbose=True)
        sc = eval_fn(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0.0)
            best_seq = best_seq / best_seq.sum()
            print(f"  *** LSE improved: {best_score:.6f}")

    remaining = time_budget - (time.monotonic() - t_global)

    if t2 > 20:

            best_seq = np.maximum(a_out, 0.0)
            best_seq = best_seq / best_seq.sum()

    # Phase 3: Exact 1D Pair CD

    # Phase 4: Peak Shaving

            best_score = sc

            print(f"  *** PeakShave improved: {best_score:.6f}")

    if t5 > 20:
        a_out, _ = lp_nesterov(best_seq.copy(), t5, solver, verbose=True)

        if np.isfinite(sc) and sc < best_score:

    # Phase 6: Pair CD round 2
