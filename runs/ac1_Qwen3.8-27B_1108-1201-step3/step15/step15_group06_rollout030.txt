import numpy as np
import time
from scipy.fft import fft, ifft
import cvxpy as cp

def fast_conv(a, b):
    """Full linear convolution using FFT."""
    n = len(a) + len(b) - 1
    N = 1 << (n - 1).bit_length()
    A = fft(a, N)
    B = fft(b, N)
    return ifft(A * B).real[:n]

def gradient_of_max_conv(a, c, epsilon):
    """Compute smoothed gradient of max(conv(a,a)) with respect to a."""
    n = len(a)
    # Compute weights (softmax of c/epsilon)
    c_shifted = c - c.max()
    exp_c = np.exp(c_shifted / epsilon)
    w = exp_c / exp_c.sum()
    b = 2.0 * w  # length 2n-1
    
    # Compute gradient: grad_i = sum_k b_k * a_{k-i}
    # = cross-correlation of a and b at lag i
    N = 1 << (3 * n - 1).bit_length()  # need N >= n + (2n-1) - 1
    A = fft(a, N)
    B = fft(b, N)
    grad = ifft(np.conj(A) * B).real[:n]
    return grad

def subgradient_descent(a0, time_limit, seed=0, verbose=True):
    """Subgradient descent with Nesterov momentum for minimizing max(conv(a,a))."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s  # normalize to sum = 1
    
    best_a = a.copy()
    best_c = fast_conv(a, a)
    best_max = best_c.max()
    
    # Nesterov variables
    y = a.copy()
    v = np.zeros(n)
    
    # Initial epsilon
    epsilon = best_max * 0.1
    epsilon_min = best_max * 0.001
    
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    lr = 0.01  # initial learning rate
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        
        # Compute convolution at y
        c = fast_conv(y, y)
        max_c = c.max()
        
        # Update epsilon (annealing)
        epsilon = max(epsilon * 0.999, epsilon_min)
        
        # Compute gradient
        grad = gradient_of_max_conv(y, c, epsilon)
        
        # Nesterov momentum
        t = iteration + 2
        beta = (t - 1) / (t + 2)  # Nesterov momentum
        y_new = y - lr * grad
        y_new = np.maximum(y_new, 0.0)
        s_new = y_new.sum()
        if s_new < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_new
        
        # Nesterov extrapolation
        a_new = y_new + beta * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new / s_a
        
        y = y_new
        a = a_new
        
        # Check improvement
        c_a = fast_conv(a, a)
        max_a = c_a.max()
        if max_a < best_max - 1e-10:
            best_max = max_a
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Backtracking: reduce learning rate if stuck
            if no_improve > 100:
                lr *= 0.95
                no_improve = 0
        
        # Also check y
        if max_c < best_max - 1e-10:
            best_max = max_c
            best_a = y.copy()
            no_improve = 0
        
        if verbose and iteration % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad iter={iteration} ({elapsed:.0f}s): best_max={best_max:.10f}, score={2*n*best_max:.8f}")
    
    return best_a, 2 * n * best_max

This makes sense: for the sequence [1,1,2,1], all the "mass" of the gradient goes into the direction that's constrained away (changing the overall scale). The actual shape can't be improved by a uniform perturbation.

So I need to use the projected gradient or work in a parameterization that automatically satisfies the constraint (like softmax).

Let me use the softmax parameterization:
- u ∈ R^n (free parameters)
- a = softmax(u) = exp(u) / sum(exp(u))
- Gradient in u-space: grad_u_i = a_i * (grad_a_i - sum_j a_j * grad_a_j)

This automatically handles the sum constraint. The non-negativity is also handled (softmax is always positive).

Let me verify: if grad_a = [1,1,1,1], then sum_j a_j * grad_a_j = sum_j a_j = 1 (since sum a = 1).
grad_u_i = a_i * (1 - 1) = 0. ✓ (stationary in u-space too)

Good. So the softmax parameterization correctly handles the constraint.

Now, what about the non-negativity constraint? With softmax, a_i is always positive, so we don't need to explicitly handle non-negativity. However, the optimal solution might have some a_i = 0 (sparse solutions). The softmax can't represent exactly zero, but it can get arbitrarily close.

In practice, for our problem, I don't think the optimal solution is sparse (we want to spread mass to flatten the convolution). So softmax should be fine.
