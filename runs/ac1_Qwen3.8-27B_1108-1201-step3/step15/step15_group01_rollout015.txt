def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def normalize_seq(a, target_sum=None):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    if target_sum is None:
        return a / s
    return a * target_sum / s


def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n + 1) > (css - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad(a, w):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    cross = irfft(W * np.conj(A), fft_size)[:n]
    return 2.0 * cross


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        T[i:i + n, i] = a
    return T


def lp_best_response(a, M, solver, active_idx=None):
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    if active_idx is not None:
        T_use = T[active_idx, :]
    else:
        T_use = T
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_use @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    # Try to get dual values for gap certification
    dual_val = None
    try:
        dual_val = np.asarray(prob.dual_values, dtype=np.float64)
    except Exception:
        pass
    return b_val, dual_val

def smooth_optimize_bt(a0, time_limit, verbose=True):
    """Smooth optimization with backtracking line search and adaptive temperature."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    t0 = time.monotonic()

    temps = [30.0, 15.0, 8.0, 4.0, 2.0, 1.2, 0.8, 0.5, 0.3, 0.2, 0.12, 0.08, 0.05, 0.03, 0.02, 0.01, 0.006, 0.003, 0.0015, 0.0008]

    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        n_steps = max(20, min(300, int(remaining * 4)))

        y = a.copy()
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            c = fast_autocorr(y)
            c_max = c.max()
            exp_arg = (c - c_max) / T
            exp_arg = np.clip(exp_arg, -50, 50)
            exp_c = np.exp(exp_arg)
            w = exp_c / exp_c.sum()

            grad = compute_smooth_grad(y, w)

            # Backtracking line search
            step = 1.0
            f_curr = T * np.log(np.sum(exp_c) + 1e-300) + c_max
            max_step = norm / (n * T * 0.1)
            step = min(step, max_step)
            accepted = False

            for _ls in range(10):
                a_trial = y - step * grad
                a_trial = project_simplex(a_trial, norm)
                c_trial = fast_autocorr(a_trial)
                c_trial_max = c_trial.max()
                exp_trial = np.exp(np.clip((c_trial - c_trial_max) / T, -50, 50))
                f_trial = T * np.log(np.sum(exp_trial) + 1e-300) + c_trial_max

                if f_trial < f_curr - 1e-10:
                    accepted = True
                    break
                step *= 0.5

            if not accepted:
                a_trial = project_simplex(y - step * grad, norm)

            # Nesterov extrapolation
            alpha = 2.0 / (it + 2.0)
            y_new = (1.0 - alpha) * y + alpha * a_trial
            y_new = project_simplex(y_new, norm)

            if it == 0:
                a_curr = a_trial.copy()
            else:
                momentum = (it - 1.0 + 1.0) / (it + 2.0)
                a_curr = y_new + momentum * (y_new - y)
                a_curr = project_simplex(a_curr, norm)

            y = y_new
            a = a_curr

            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

            score_y = fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    Smooth T={T:.3f} done ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_pair_cd(a0, time_limit, rng, verbose=True):
    """Peak-pair coordinate descent: directly optimize pairs contributing to the peak lag."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 5:
        conv = fast_autocorr(a)
        M = conv.max()
        # Find all near-peak positions
        peak_thresh = 0.9 * M
        peak_lags = np.where(conv >= peak_thresh)[0]

        improved = False

        for k in peak_lags:
            if time.monotonic() - t0 > time_limit:
                break

            # Find pairs (i, k-i) contributing to conv[k]
            # conv[k] = sum_i a[i]*a[k-i] for valid i
            if k < n:
                i_range = range(0, k + 1)
            else:
                i_range = range(k - n + 1, n)

            # Get contribution of each pair
            pairs = []
            for i in i_range:
                j = k - i
                if 0 <= j < n and j != i:
                    c = a[i] * a[j]
                    pairs.append((c, i, j))
                elif j == i and 0 <= j < n:
                    c = a[i] * a[i]
                    pairs.append((c, i, j))

            pairs.sort(reverse=True)

            # Try to reduce contributions from top pairs
            for c_val, i, j in pairs[:10]:
                if time.monotonic() - t0 > time_limit:
                    break
                if c_val < M * 1e-6:
                    break

                if i == j:
                    # Diagonal: reduce a[i] and redistribute
                    # Move mass from i to a position that doesn't contribute to lag k
                    # Position that contributes to lag k: i such that i + j = k
                    # We want to move to position m where m and k-m are both small
                    transfer = a[i] * 0.3
                    # Find a good destination
                    best_dest = -1
                    best_new_max = M
                    for m in range(n):
                        if m == i:
                            continue
                        a_test = a.copy()
                        a_test[i] -= transfer
                        a_test[m] += transfer
                        if a_test[i] < 0:
                            continue
                        a_test = a_test * norm / a_test.sum()
                        conv_test = fast_autocorr(a_test)
                        if conv_test.max() < best_new_max:
                            best_new_max = conv_test.max()
                            best_dest = m
                    if best_dest >= 0 and best_new_max < best_score - 1e-12:
                        a[best_dest] += transfer
                        a[i] -= transfer
                        a = a * norm / a.sum()
                        best_score = best_new_max
                        best_a = a.copy()
                        improved = True
                else:
                    # Off-diagonal: transfer from (i,j) to reduce their product
                    # Option 1: reduce a[i], increase some a[m]
                    # Option 2: reduce a[j], increase some a[m]
                    # Option 3: redistribute between i and j (keep i+j fixed for this lag)
                    # Actually, the best is to move mass AWAY from both i and j

                    # Try moving a fraction from i to a random position
                    for frac in [0.1, 0.3, 0.5]:
                        transfer = a[i] * frac
                        if transfer < norm * 1e-10:
                            continue
                        a_test = a.copy()
                        a_test[i] -= transfer
                        # Move to a position far from the peak
                        m = (i + n // 2) % n
                        a_test[m] += transfer
                        a_test = np.maximum(a_test, 0)
                        a_test = a_test * norm / a_test.sum()
                        conv_test = fast_autocorr(a_test)
                        if conv_test.max() < best_score - 1e-12:
                            a = a_test
                            best_score = conv_test.max()
                            best_a = a.copy()
                            improved = True
                            break

        if not improved:
            no_improve += 1
        else:
            no_improve = 0

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    PeakPairCD epoch={no_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def swap_cd(a0, time_limit, rng, verbose=True):
    """Swap-based coordinate descent: try all (i,j) swaps and keep improvements."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 3:
        improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            # Try moving to nearby and far positions
            offsets = list(range(1, 20)) + [n // 4, n // 3, n // 2, 2 * n // 3, 3 * n // 4]
            for off in offsets:
                for sign in [1, -1]:
                    j = (i + sign * off) % n
                    if j == i:
                        continue

                    # Binary search for optimal transfer amount
                    lo, hi = 0.0, ai * 0.9
                    best_d = 0.0
                    best_val = conv.max()

                    for _ in range(6):
                        mid = (lo + hi) / 2
                        if mid < 1e-15:
                            lo = mid
                            continue
                        a_t = a.copy()
                        a_t[i] -= mid
                        a_t[j] += mid
                        a_t = a_t * norm / a_t.sum()
                        conv_t = fast_autocorr(a_t)
                        if conv_t.max() < best_val - 1e-12:
                            best_val = conv_t.max()
                            best_d = mid
                            lo = mid
                        else:
                            hi = mid

                    if best_d > 1e-15:
                        a[i] -= best_d
                        a[j] += best_d
                        a = a * norm / a.sum()
                        conv = fast_autocorr(a)
                        if conv.max() < best_score - 1e-12:
                            best_score = conv.max()
                            best_a = a.copy()
                            improved = True

        if not improved:
            no_improve += 1
        else:
            no_improve = 0

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    SwapCD epoch={no_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def subgrad_euler_avg(a0, time_limit, verbose=True):
    """Subgradient method with Euler averaging (simpler but robust)."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Accumulator for averaging
    acc = np.zeros(n)
    total_weight = 0.0

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(a)
        M = conv.max()
        # Subgradient: at the peak lag k*, grad[i] = a[k*-i] + a[i-k*] (properly indexed)
        k_star = np.argmax(conv)

        grad = np.zeros(n)
        for i in range(n):
            j1 = k_star - i
            j2 = i - k_star
            if 0 <= j1 < n:
                grad[i] += a[j1]
            if 0 <= j2 < n:
                grad[i] += a[j2]

        # Step size: eta_t = c / sqrt(t)
        eta = norm / (np.sqrt(t) * M * 2.0)
        a_new = a - eta * grad
        a_new = project_simplex(a_new, norm)

        # Accumulate for averaging
        acc += a_new
        total_weight += 1.0

        a = a_new

        # Check both current and average
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if t % 50 == 0:
            avg_a = acc / total_weight
            avg_a = avg_a * norm / avg_a.sum()
            score_avg = fast_autocorr(avg_a).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_a.copy()

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng, method='triangle'):
    """Generate structured initializations."""
    if method == 'triangle':
        x = np.linspace(0, 1, n)
        f = 1.0 - np.abs(2 * x - 1)  # Triangle peaking at center
    elif method == 'edges':
        x = np.linspace(0, 1, n)
        f = np.abs(2 * x - 1)  # Peak at edges
    elif method == 'cosine':
        x = np.linspace(0, np.pi, n)
        f = np.cos(x / 2) ** 2
    elif method == 'bimodal':
        x = np.linspace(0, 1, n)
        f = np.exp(-10 * (x - 0.25) ** 2) + np.exp(-10 * (x - 0.75) ** 2)
    elif method == 'random':
        f = rng.uniform(0.1, 1.0, n)
    else:
        f = np.ones(n)

    f = np.maximum(f, 0.001)
    return f / f.sum()

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = y_new * norm / max(y_new.sum(), 1e-15)

        if t <= 2:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function. Returns best sequence found within time budget."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1650.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: LP Nesterov with duality (30% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.30
    print(f"\n=== Phase 1: LP+Dual n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp_duality(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp_dual_n{n_base}")

    # === Phase 2: Smooth optimization with backtracking (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = remaining * 0.20
    print(f"\n=== Phase 2: Smooth-BT n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 20:
        n_c = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = smooth_optimize_bt(a_norm, t2, verbose=True)
        do_try_update(a_out, f"smooth_bt_n{n_c}")

    # === Phase 3: Peak-pair CD (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = remaining * 0.15
    print(f"\n=== Phase 3: PeakPairCD n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 10:
        n_c = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = peak_pair_cd(a_norm, t3, rng, verbose=True)
        do_try_update(a_out, f"peakpair_n{n_c}")

    # === Phase 4: Swap CD (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = remaining * 0.15
    print(f"\n=== Phase 4: SwapCD n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 10:
        n_c = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = swap_cd(a_norm, t4, rng, verbose=True)
        do_try_update(a_out, f"swapcd_n{n_c}")

    # === Phase 5: Multi-start diversification (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = remaining * 0.10
    print(f"\n=== Phase 5: Multi-start (budget={t5:.0f}s) ===")
    if t5 > 60:
        n_c = len(best_seq)
        methods = ['edges', 'bimodal', 'cosine', 'random']
        per_method = t5 / (len(methods) + 1)

        for m in methods:
            remaining_sub = time_budget - (time.monotonic() - t_global)
            if remaining_sub < 15:
                break
            t_sub = min(per_method * 0.8, remaining_sub * 0.7)
            if t_sub < 10:
                break

            init = structured_init(n_c, rng, method=m)
            # Blend with current best
            blend = 0.3 * init + 0.7 * best_seq
            blend = np.maximum(blend, 0.0)
            blend = blend / blend.sum()
            a_norm = blend * np.sqrt(2.0 * n_c)

            a_out, _ = smooth_optimize_bt(a_norm, t_sub, verbose=False)
            do_try_update(a_out, f"ms_{m}_n{n_c}")

    # === Phase 6: Length exploration (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = remaining * 0.10
    print(f"\n=== Phase 6: Length exploration (budget={t6:.0f}s) ===")
    if t6 > 60:
        n_cur = len(best_seq)
        candidates_n = [n_cur - 200, n_cur + 200, n_cur - 400, n_cur + 400]
        candidates_n = [x for x in candidates_n if x >= 200]
        per_len = t6 / max(len(candidates_n), 1)

        for n_alt in candidates_n:
            remaining_sub = time_budget - (time.monotonic() - t_global)
            if remaining_sub < 20:
                break
            t_sub = min(per_len * 0.8, remaining_sub * 0.7)
            if t_sub < 15:
                break

            print(f"  Trying n={n_alt} (budget={t_sub:.0f}s)")
            init_alt = interpolate_sequence(best_seq, n_alt)
            a_norm = init_alt * np.sqrt(2.0 * n_alt)

            # Quick smooth optimization
            a_out, _ = smooth_optimize_bt(a_norm, t_sub * 0.5, verbose=False)
            do_try_update(a_out, f"len_{n_alt}_smooth")

            remaining_sub2 = time_budget - (time.monotonic() - t_global)
            if remaining_sub2 > 10:
                # LP polish
                a_out2 = polish_with_lp(a_out, solver, min(remaining_sub2 * 0.4, 30.0))
                do_try_update(a_out2, f"len_{n_alt}_lp")

    # === Phase 7: Final polish (5% remaining time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20 and best_seq is not None:
        n_c = len(best_seq)
        t7 = remaining * 0.7
        print(f"\n=== Phase 7: Final polish n={n_c} (budget={t7:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_c)
        a_out = polish_with_lp(a_norm, solver, t7)
        do_try_update(a_out, "final_lp")

        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 10:
            n_c2 = len(best_seq)
            a_norm2 = best_seq * np.sqrt(2.0 * n_c2)
            a_out2, _ = smooth_optimize_bt(a_norm2, remaining2 * 0.6, verbose=False)
            do_try_update(a_out2, "final_smooth")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
