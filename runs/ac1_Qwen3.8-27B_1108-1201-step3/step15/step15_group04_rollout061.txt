def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T

def lp_solve(a, M, solver, active=None):
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    if active is not None and 0 < len(active) < size:
        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    duals = None

        con = constraints[0]
        if con.dual_value is not None:
            if active is not None and len(active) < size:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP best-response with Nesterov acceleration and adaptive constraint screening
2. Dual-variable gradient refinement
3. Targeted coordinate descent guided by LP dual solutions

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print(f"  Using solver (fallback): ECOS")
    return _solver

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T

def lp_solve(a, M, solver, active=None):
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    if active is not None and 0 < len(active) < size:
        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    duals = None
    try:
        con = constraints[0]
        if con.dual_value is not None:
            if active is not None and len(active) < size:
                full_duals = np.zeros(size)
                full_duals[active] = np.asarray(con.dual_value, dtype=np.float64)
                duals = full_duals
            else:
                duals = np.asarray(con.dual_value, dtype=np.float64)
    except Exception:
        pass
    return val, duals

def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 10
    temps = np.geomspace(50.0, 0.5, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(30, min(200, int(remaining * 3)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -200.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)
            g2 = aw[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                step = 0.0
            else:
                step = norm / (T_val * np.sqrt(n)) / (1.0 + 0.03 * it)
                for trial in range(12):
                    a_new = a - step * grad
                    a_new = np.maximum(a_new, 0.0)
                    s_new = a_new.sum()
                    if s_new < 1e-15:
                        step *= 0.5
                        continue
                    a_new = a_new * norm / s_new
                    sc = fast_conv_self(a_new).max()
                    if sc <= conv.max() + 0.05 * T_val:
                        break
                    step *= 0.5

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.1f}: best={best_score:.8f}")

    return best_a, best_score

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, use_multiplicative=False):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1
    avg_a = np.zeros(n)
    n_avg = 0
    no_improve_count = 0
    prev_lp_obj = None

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        if t < 30:
            active = np.where(conv_y > M * 0.4)[0]
        elif t < 100:
            active = np.where(conv_y > M * 0.6)[0]
        elif t < 300:
            active = np.where(conv_y > M * 0.75)[0]
        else:
            active = np.where(conv_y > M * 0.85)[0]
        if len(active) < 20:
            active = np.where(conv_y > M * 0.3)[0]

        b, duals = lp_solve(y, M, solver, active)
        if b is None:
            b, duals = lp_solve(y, M, solver, None)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        if active is not None and len(active) < size:
            check_conv = fast_conv_cross(y, g)
            violated = np.where(check_conv > M * (1 + 5e-3))[0]
            if len(violated) > 0:
                b_full, _ = lp_solve(y, M, solver, None)
                if b_full is not None and b_full.sum() > 1e-15:
                    b = b_full
                    g = b * norm / b.sum()

        # Adaptive step size based on improvement ratio
        if prev_lp_obj is not None and prev_lp_obj > 1e-15:
            improvement_ratio = b_sum / (y.sum() * norm / np.sqrt(2.0 * n))
            alpha = 2.0 / (t + 2.0)
            alpha = alpha * min(2.0, max(0.3, improvement_ratio))
        else:
            alpha = 2.0 / (t + 2.0)

        prev_lp_obj = b_sum

        if use_multiplicative and t > 30:
            ratio = g / np.maximum(y, 1e-30)
            log_ratio = np.log(np.clip(ratio, 1e-10, 1e10))
            y_new = y * np.exp(alpha * log_ratio)
        else:
            y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min(1.0, (t - 1.0) / (t + 2.0))
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        improved = False
        if sc_y < best_score - 1e-10:
            best_score = sc_y
            best_a = y.copy()
            improved = True

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score - 1e-10:
            best_score = sc_a
            best_a = a.copy()
            improved = True

        if not improved:
            no_improve_count += 1
        else:
            no_improve_count = 0

        if no_improve_count == 80:
            n_perturb = max(15, n // 15)
            idx_perturb = np.random.default_rng(t * 7 + 3).choice(n, size=n_perturb, replace=False)
            noise_scale = 0.15 + 0.1 * (no_improve_count // 100)
            y[idx_perturb] *= (1.0 + noise_scale * np.random.default_rng(t * 7 + 4).standard_normal(n_perturb))
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y * norm / s_y
            no_improve_count = 0

        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc_avg = fast_conv_self(avg_check).max()
                if sc_avg < best_score:
                    best_score = sc_avg
                    best_a = avg_check.copy()

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def dual_gradient_refine(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_score(a)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    for it in range(300):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        M = conv.max()
        if M < 1e-12:
            break

        active = np.where(conv > M * 0.7)[0]
        b, duals = lp_solve(a, M, solver, active)
        if duals is None or b is None:
            b2, duals2 = lp_solve(a, M, solver, None)
            if duals2 is not None and b2 is not None:
                duals = duals2
                b = b2
            else:
                duals = np.zeros(size)
                duals[np.argmax(conv)] = 1.0

        fft_size = 1 << (size - 1).bit_length()
        A = rfft(a, fft_size)
        D = rfft(duals, fft_size)
        g1 = irfft(np.conj(A) * D, fft_size)[:n]
        g2 = irfft(A * D, fft_size)[:n]
        grad = 2.0 * (g1 + g2)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        step = 0.05
        for trial in range(20):
            a_new = a - step * grad / grad_norm
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                step *= 0.5
                continue
            a_new = a_new * norm / s_new
            sc = fast_conv_self(a_new).max()
            if sc < M - 1e-10:
                a = a_new
                sc = sc
                break
            step *= 0.5

        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose and it % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DualGrad it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def targeted_cd(a0, time_limit, solver, rng, verbose=True):
    """Coordinate descent guided by LP dual information."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_score(a)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    # Get initial dual information
    conv = fast_conv_self(a)
    M = conv.max()
    active = np.where(conv > M * 0.8)[0]
    b, duals = lp_solve(a, M, solver, active)

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        # Get current tight constraints
        peak_threshold = cur_max * 0.92
        peak_lags = np.where(conv >= peak_threshold)[0]

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            j_candidates = set()
            for _ in range(8):
                j_candidates.add(int(rng.integers(n)))

            for kl in peak_lags[:6]:
                j1 = (kl - i) % n
                j2 = (kl + i) % n
                j_candidates.add(j1)
                j_candidates.add(j2)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]

                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                n_samples = 15
                deltas = np.linspace(0.01, 0.9, n_samples) * ai
                best_d = 0.0
                best_v = cur_max
                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_d = d

                if best_d > 0 and best_v < best_val - 1e-12:
                    best_val = best_v
                    best_delta = best_d
                    best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 6:
                break
            if no_improve == 3:
                n_perturb = max(10, n // 25)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.1 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and total_improvements % 100 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    TCD improvements={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def compute_score(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    n = len(a)
    max_b = fast_conv_self(a).max()
    return 2.0 * n * max_b / (S * S)

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_init = None
        best_init_score = float('inf')
        for trial in range(8):
            cand = structured_init(n_base, rng)
            sc = eval_fn(cand)
            if sc < best_init_score:
                best_init_score = sc
                best_init = cand
        best_seq = best_init
        best_score = best_init_score
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Quick LSE to improve init
    remaining = time_budget - (time.monotonic() - t_global)
    t_init = min(remaining * 0.06, 90.0)
    if t_init > 15:
        a_out, _ = lse_gradient_descent(best_seq, t_init, verbose=False)
        try_update(a_out, "lse_init")

    # Phase 1: LP Nesterov (main workhorse) - first round
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = len(best_seq)
    t1 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 1: LP Nesterov n={n1} (budget={t1:.0f}s) ===")
    if t1 > 25:
        a_out, _ = nesterov_lp_optimize(best_seq, t1, solver, verbose=True, use_multiplicative=False)
        try_update(a_out, f"lp1_n{n1}")

    # Phase 2: Dual gradient refinement
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq)
    t2 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 2: Dual Gradient n={n2} (budget={t2:.0f}s) ===")
    if t2 > 15:
        a_out, _ = dual_gradient_refine(best_seq, t2, solver, verbose=True)
        try_update(a_out, f"dualgrad_n{n2}")

    # Phase 3: LP Nesterov round 2 with multiplicative updates
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = len(best_seq)
    t3 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 3: LP Nesterov round 2 (multiplicative) n={n3} (budget={t3:.0f}s) ===")
    if t3 > 25:
        a_out, _ = nesterov_lp_optimize(best_seq, t3, solver, verbose=True, use_multiplicative=True)
        try_update(a_out, f"lp2_n{n3}")

    # Phase 4: Targeted CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    n4 = len(best_seq)
    t4 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 4: Targeted CD n={n4} (budget={t4:.0f}s) ===")
    if t4 > 15:
        a_out, _ = targeted_cd(best_seq, t4, solver, rng, verbose=True)
        try_update(a_out, f"tcd_n{n4}")

    # Phase 5: LSE + LP alternating
    remaining = time_budget - (time.monotonic() - t_global)
    n5 = len(best_seq)
    print(f"\n=== Phase 5: LSE+LP alternate n={n5} (remaining={remaining:.0f}s) ===")
    if remaining > 50:
        t_half = remaining * 0.4
        a_out, _ = lse_gradient_descent(best_seq, min(t_half, 80.0), verbose=False)
        try_update(a_out, "lse_mid")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t_lp = min(remaining * 0.5, 100.0)
            a_out, _ = nesterov_lp_optimize(best_seq, t_lp, solver, verbose=False, use_multiplicative=False)
            try_update(a_out, "lp_mid")

    # Phase 6: Try longer sequence
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 400
    print(f"\n=== Phase 6: Length extension n={n_long} (remaining={remaining:.0f}s) ===")
    if remaining > 80:
        init_long = interpolate_seq(best_seq, n_long)
        t_long = min(remaining * 0.4, 120.0)
        a_out, _ = nesterov_lp_optimize(init_long, t_long, solver, verbose=False, use_multiplicative=False)
        try_update(a_out, f"lp_long_{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and len(best_seq) == n_long:
            t_long_cd = min(remaining * 0.5, 60.0)
            a_out2, _ = targeted_cd(best_seq, t_long_cd, solver, rng, verbose=False)
            try_update(a_out2, f"tcd_long_{n_long}")

    # Phase 7: Final polish on best sequence
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 7: Final polish n={n_final} (remaining={remaining:.0f}s) ===")
    if remaining > 30:
        t_f1 = min(remaining * 0.35, 50.0)
        a_out, _ = targeted_cd(best_seq, t_f1, solver, rng, verbose=False)
        try_update(a_out, "tcd_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 25:
            t_f2 = min(remaining * 0.5, 45.0)
            a_out, _ = nesterov_lp_optimize(best_seq, t_f2, solver, verbose=False, use_multiplicative=False)
            try_update(a_out, "lp_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t_f3 = min(remaining * 0.6, 30.0)
            a_out, _ = dual_gradient_refine(best_seq, t_f3, solver, verbose=False)
            try_update(a_out, "dualgrad_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
