    return 1 << (size - 1).bit_length()


def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fs = fft_size_for(n)
    A = rfft(a, fs)
    return irfft(A * A, fs)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fs = fft_size_for(n)
    A = rfft(a, fs)
    B = rfft(b, fs)
    return irfft(A * B, fs)[:size]


def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T


def lp_solve(a, M, solver, active=None, eps=1e-8):
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    if active is not None and 0 < len(active) < size:
        constraints = [T[active, :] @ b_var <= M + eps]
    else:
        constraints = [T @ b_var <= M + eps]
    obj = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)


def normalize(a, n):
    norm = np.sqrt(2.0 * n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n * norm
    return a * norm / s


def lse_gradient_descent(a0, time_limit, verbose=False):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(np.asarray(a0, dtype=np.float64), n)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fs = fft_size_for(n)

    n_stages = 16
    temps = np.geomspace(100.0, 0.1, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        n_steps = max(50, min(500, int(remaining * 8)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -200.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fs)
            A = rfft(a, fs)

            g1 = irfft(np.conj(A) * W, fs)[:n]
            g2 = irfft(A * W, fs)[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break

            step = norm / (T_val * np.sqrt(n)) * (1.0 / (1.0 + 0.03 * it))
            a_new = a - step * grad
            a_new = normalize(a_new, n)
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best={best_score:.8f}")

    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=False):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(np.asarray(a0, dtype=np.float64), n)

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1
    full_check_interval = 100

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        if t % full_check_interval == 0:
            active = None
        else:
            thresh_1 = M * 0.85
            active = np.where(conv_y >= thresh_1)[0]
            if len(active) < 20:
                thresh_2 = M * 0.55
                active = np.where(conv_y >= thresh_2)[0]
            if len(active) < 10:
                active = None

        b = lp_solve(y, M, solver, active)
        if b is None:
            b = lp_solve(y, M, solver, None)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        if active is not None and len(active) < size and t % 30 != 0:
            max_lag = 5
            check_vals = np.convolve(y, g)[:size]
            violated = np.where(check_vals > M * (1.0 + 1e-6))[0]
            if len(violated) > max_lag:
                b_full = lp_solve(y, M, solver, None)
                if b_full is not None:
                    b = b_full
                    b_sum = b.sum()
                    if b_sum > 1e-15:
                        g = b * norm / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = normalize(y_new, n)

        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = normalize(a_new, n)

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def exact_peak_cd(a0, time_limit, rng, verbose=False):
    """Peak-targeted coordinate descent using exact 1D minimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(np.asarray(a0, dtype=np.float64), n)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        peak_threshold = cur_max * 0.93
        peak_lags = np.where(conv >= peak_threshold)[0]

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            j_candidates = set()
            for _ in range(8):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:4]:
                j_s = (kl - i + n) % n
                if j_s != i:
                    j_candidates.add(j_s)
                j_s2 = (kl + i) % n
                if j_s2 != i:
                    j_candidates.add(j_s2)
            j_candidates.discard(i)

            if not j_candidates:
                continue

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in list(j_candidates)[:12]:
                if j == i:
                    continue
                aj = a[j]

                # Compute gain vector g[k] = 2*(a[k-j] - a[k-i])
                g = np.zeros(size)
                # Contribution from j: 2*a[k-j] for k where 0<=k-j<n
                lo_j = j
                hi_j = min(size, j + n)
                g[lo_j:hi_j] += 2.0 * a[0:hi_j - lo_j]
                # Contribution from i: -2*a[k-i] for k where 0<=k-i<n
                lo_i = i
                hi_i = min(size, i + n)
                g[lo_i:hi_i] -= 2.0 * a[0:hi_i - lo_i]

                # Quadratic corrections at specific lags
                # k=2j: +d^2
                # k=2i: +d^2
                # k=i+j: -2d^2
                q_lags = []
                if 2 * j < size:
                    q_lags.append((2 * j, 1.0))
                if 2 * i < size:
                    q_lags.append((2 * i, 1.0))
                if i + j < size:
                    q_lags.append((i + j, -2.0))

                # Find optimal d using ternary search on the max function
                # f(d) = max_k (conv[k] + d*g[k] + d^2*q[k])
                # This is convex in d (max of convex functions)
                d_lo = 0.0
                d_hi = ai * 0.99

                # Quick check: does any direction help?
                # f(d_hi) vs f(0) = cur_max
                test_conv = conv + d_hi * g
                for (k_q, q_val) in q_lags:
                    test_conv[k_q] += d_hi * d_hi * q_val
                if test_conv.max() >= cur_max:
                    # Try smaller range
                    d_hi = ai * 0.5
                    test_conv = conv + d_hi * g
                    for (k_q, q_val) in q_lags:
                        test_conv[k_q] += d_hi * d_hi * q_val
                    if test_conv.max() >= cur_max:
                        continue

                # Ternary search for convex function
                for _ in range(25):
                    d_m1 = d_lo + (d_hi - d_lo) / 3.0
                    d_m2 = d_hi - (d_hi - d_lo) / 3.0

                    c1 = conv + d_m1 * g
                    for (k_q, q_val) in q_lags:
                        c1[k_q] += d_m1 * d_m1 * q_val
                    v1 = c1.max()

                    c2 = conv + d_m2 * g
                    for (k_q, q_val) in q_lags:
                        c2[k_q] += d_m2 * d_m2 * q_val
                    v2 = c2.max()

                    if v1 < v2:
                        d_hi = d_m2
                    else:
                        d_lo = d_m1

                d_opt = (d_lo + d_hi) / 2.0
                c_final = conv + d_opt * g
                for (k_q, q_val) in q_lags:
                    c_final[k_q] += d_opt * d_opt * q_val
                v_final = c_final.max()

                if v_final < best_val - 1e-12 and d_opt > 1e-12:
                    best_val = v_final
                    best_delta = d_opt
                    best_j = j

            if best_j >= 0 and best_delta > 1e-12:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = normalize(a, n)
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
                    total_improvements += 1

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break
            if no_improve >= 2:
                n_perturb = max(10, n // 30)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.12 * rng.standard_normal(n_perturb))
                a = normalize(a, n)
        else:
            no_improve = 0

        if verbose and total_improvements % 100 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakCD improvements={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def aggressive_subgradient(a0, time_limit, verbose=False):
    """Subgradient with adaptive temperature and Polyak step size."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(np.asarray(a0, dtype=np.float64), n)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    fs = fft_size_for(n)

    momentum = np.zeros(n)
    beta = 0.85
    lr0 = 2.0 / np.sqrt(n)
    best_ever = fast_conv_self(a).max()
    target = best_ever * 0.95

    for it in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        max_c = conv.max()

        temp = max(0.05, (max_c - target) * 0.5 + 0.05)
        shifted = (conv - max_c) / temp
        shifted = np.maximum(shifted, -100.0)
        w = np.exp(shifted)
        w = w / w.sum()

        W = rfft(w, fs)
        A = rfft(a, fs)
        g1 = irfft(np.conj(A) * W, fs)[:n]
        g2 = irfft(A * W, fs)[:n]
        grad = 2.0 * (g1 + g2)

        gap = max_c - target
        if gap > 0:
            lr = min(lr0, gap / (np.linalg.norm(grad) ** 2 + 1e-10))
        else:
            lr = lr0 / (1.0 + 0.005 * it)

        momentum = beta * momentum - lr * grad
        a_new = a + momentum
        a_new = normalize(a_new, n)
        a = a_new

        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            target = max(best_score * 0.98, 0.5)

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.15 * np.cos(2 * np.pi * 7 * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.8
    elif choice == 6:
        a = np.exp(-3.0 * (x - 0.5) ** 2 / (0.05 + 0.1 * rng.random()))
    else:
        a = 0.4 + 0.4 * np.cos(2 * np.pi * x) * np.cos(4 * np.pi * x)
    a = np.maximum(a, 0.0)
    a = a + 0.02 * rng.random(n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        if len(a_list) == 0:
            return float('inf')
        for val in a_list:
            if val < 0 or not np.isfinite(val):
                return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LSE gradient descent warmup
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 80.0)
    print(f"\n=== Phase 1: LSE GD n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 15:
        a_out, _ = lse_gradient_descent(best_seq, t1, verbose=True)
        try_update(a_out, "lse1")

    # Phase 2: LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.30, 480.0)
    print(f"\n=== Phase 2: LP Nesterov n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, solver, verbose=True)
        try_update(a_out, "lp1")

    # Phase 3: Exact peak CD
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 3: Exact Peak CD n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = exact_peak_cd(best_seq, t3, rng, verbose=True)
        try_update(a_out, "peakcd1")

    # Phase 4: Aggressive subgradient
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 4: SubGradient n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 15:
        a_out, _ = aggressive_subgradient(best_seq, t4, verbose=True)
        try_update(a_out, "subgrad")

    # Phase 5: LP Nesterov round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 5: LP Nesterov r2 n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t5, solver, verbose=True)
        try_update(a_out, "lp2")

    # Phase 6: Peak CD round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 6: Peak CD r2 n={len(best_seq)} (budget={t6:.0f}s) ===")
    if t6 > 15:
        a_out, _ = exact_peak_cd(best_seq, t6, rng, verbose=True)
        try_update(a_out, "peakcd2")

    # Phase 7: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = len(best_seq) + 200
    t7 = min(remaining * 0.35, 250.0)
    print(f"\n=== Phase 7: Longer length n={n_long} (budget={t7:.0f}s) ===")
    if t7 > 40 and remaining > 100:
        init_long = interpolate_seq(best_seq, n_long)
        t7a = min(t7 * 0.45, 100.0)
        a_out, _ = nesterov_lp_optimize(init_long, t7a, solver, verbose=False)
        try_update(a_out, f"lp_n{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t7b = min(remaining * 0.35, 70.0)
            a_out2, _ = exact_peak_cd(best_seq if best_seq is not None else init_long, t7b, rng, verbose=False)
            try_update(a_out2, f"cd_n{n_long}")

    # Phase 8: Try even longer
    remaining = time_budget - (time.monotonic() - t_global)
    n_long2 = len(best_seq) + 400
    t8 = min(remaining * 0.25, 150.0)
    print(f"\n=== Phase 8: Longer n={n_long2} (budget={t8:.0f}s) ===")
    if t8 > 40 and remaining > 80:
        init_long2 = interpolate_seq(best_seq, n_long2)
        a_out, _ = nesterov_lp_optimize(init_long2, t8, solver, verbose=False)
        try_update(a_out, f"lp_n{n_long2}")

    # Phase 9: Final polish on best sequence
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 9: Final polish n={len(best_seq)} (budget={remaining:.0f}s) ===")
    if remaining > 20:
        t_half = remaining * 0.4
        a_out, _ = lse_gradient_descent(best_seq, min(t_half, 40.0), verbose=False)
        try_update(a_out, "lse_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_out, _ = exact_peak_cd(best_seq, min(remaining * 0.7, 50.0), rng, verbose=False)
            try_update(a_out, "cd_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out, _ = nesterov_lp_optimize(best_seq, min(remaining * 0.8, 60.0), solver, verbose=False)
            try_update(a_out, "lp_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = min(best_score, final_score)
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
