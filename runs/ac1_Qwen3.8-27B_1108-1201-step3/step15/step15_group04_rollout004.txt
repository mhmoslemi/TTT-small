_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(8, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.SCS
    print(f"  Using solver (fallback): SCS")
    return _solver


def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_T_fast(a, active):
    n = len(a)
    m_vals = np.asarray(active, dtype=np.int64)
    T = np.zeros((len(active), n))
    for idx in range(len(active)):
        m = int(m_vals[idx])
        js = max(0, m - n + 1)
        je = min(m, n - 1)
        if js <= je:
            T[idx, js:je + 1] = a[m - je:m - js + 1]
    return T


def lp_solve(a, M, solver, active=None):
    n = len(a)
    size = 2 * n - 1
    if active is None:
        active = np.arange(size)
    active = np.asarray(active, dtype=np.int64)
    if len(active) == 0:
        return None, None
    T = build_T_fast(a, active)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    duals = None
    try:
        con = constraints[0]
        if con.dual_value is not None:
            if len(active) < size:
                full_duals = np.zeros(size)
                full_duals[active] = np.asarray(con.dual_value, dtype=np.float64)
                duals = full_duals
            else:
                duals = np.asarray(con.dual_value, dtype=np.float64)
    except Exception:
        pass
    return val, duals


def normalize_a(a, norm_val=None):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    n = len(a)
    if norm_val is None:
        norm_val = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * norm_val / n
    return a * norm_val / s

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):

    norm = np.sqrt(2.0 * n)
    a = normalize_a(a0, norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1

    avg_y = np.zeros(n)
    n_avg = 0
    no_improve = 0
    active_set = np.where(fast_conv_self(y) > fast_conv_self(y).max() * 0.5)[0]

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        if t < 20:
            frac = 0.4
        elif t < 100:
            frac = 0.6
        elif t < 500:
            frac = 0.75
        else:
            frac = 0.85

        active_set = np.where(conv_y > M * frac)[0]
        if len(active_set) < 50:
            active_set = np.where(conv_y > M * 0.3)[0]
        if len(active_set) < 20:
            active_set = np.argsort(conv_y)[-200:]

        b, duals = lp_solve(y, M, solver, active_set)
        if b is None:
            b, duals = lp_solve(y, M, solver, None)
        if b is None:
            break

        b_sum = b.sum()
        if b_sum < 1e-15:
            b = np.ones(n) / n
            b_sum = 1.0
        g = b * norm / b_sum

        if len(active_set) < size:
            check = fast_conv_cross(y, g)
            violated = np.where(check > M * 1.001)[0]
            if len(violated) > 0:
                b_full, _ = lp_solve(y, M, solver, None)
                if b_full is not None and b_full.sum() > 1e-15:
                    b = b_full
                    g = b * norm / b.sum()

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        momentum = min(1.0, (t - 1.0) / (t + 2.0))
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        improved = False
        if sc_y < best_score - 1e-10:
            best_score = sc_y
            best_a = y.copy()
            improved = True

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score - 1e-10:
            best_score = sc_a
            best_a = a.copy()
            improved = True

        if not improved:
            no_improve += 1
        else:
            no_improve = 0

        if no_improve == 80:
            n_pert = max(20, n // 15)
            idx_pert = np.random.default_rng(t * 7 + 13).choice(n, size=n_pert, replace=False)
            scale = 0.3
            y[idx_pert] *= (1.0 + scale * np.random.default_rng(t * 7 + 14).standard_normal(n_pert))
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y * norm / s_y
            no_improve = 0

        avg_y += y
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_y / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:

                    best_a = avg_check.copy()

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest t={t} ({elapsed:.0f}s): best={best_score:.8f}, active={len(active_set)}")

    return best_a, best_score


def subgradient_optimize(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_a(a0, norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    avg_a = np.zeros(n)
    n_avg = 0
    f_star = best_score * 0.95

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        M = conv.max()
        argmax = np.where(conv >= M * 0.999)[0]
        lam = np.zeros(size)
        lam[argmax] = 1.0 / len(argmax)

        A = rfft(a, fft_size)
        L = rfft(lam, fft_size)
        g1 = irfft(np.conj(A) * L, fft_size)[:n]
        g2 = irfft(A * L, fft_size)[:n]
        grad = g1 + g2

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        gap = M - f_star
        if gap < 1e-6:
            f_star = M * 0.99
            gap = M - f_star
            if gap < 1e-6:
                gap = M * 0.01

        eta = gap / (grad_norm * grad_norm + 1e-15)
        eta = min(eta, 0.5 / grad_norm)

        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        a = a_new

        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc_avg = fast_conv_self(avg_check).max()
                if sc_avg < best_score:
                    best_score = sc_avg
                    best_a = avg_check.copy()

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_a(a0, norm)
    best_a = a.copy()

    t0 = time.monotonic()
    size = 2 * n - 1

    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-12:
                continue

            j_candidates = set()
            for _ in range(8):
                j_candidates.add(int(rng.integers(n)))

            peak_lags = np.where(conv >= cur_max * 0.85)[0]
            for kl in peak_lags[:6]:
                j1 = (int(kl) - i) % n
                j2 = (int(kl) + i) % n
                j_candidates.add(j1)
                j_candidates.add(j2)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]

                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                n_samples = 16
                deltas = np.linspace(0.02, 0.9, n_samples) * ai
                best_d = 0.0
                best_v = cur_max
                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_d = d

                if best_d > 0 and best_v < best_val - 1e-13:
                    best_val = best_v
                    best_delta = best_d
                    best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-13:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        s_tmp = a.sum()
                        if s_tmp > 1e-15:
                            a = a * norm / s_tmp

        if not pass_improved:
            no_improve += 1
            if no_improve >= 6:
                break
            if no_improve == 3:
                n_perturb = max(15, n // 25)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.2 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and total_improvements % 500 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD impr={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lse_gradient_descent(a0, time_limit, verbose=False):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_a(a0, norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 10
    temps = np.geomspace(80.0, 0.2, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(20, min(250, int(remaining * 3)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -200.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            A = rfft(a, fft_size)
            W = rfft(w, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break

            step = norm / (T_val * np.sqrt(n)) / (1.0 + 0.03 * it)
            for trial in range(12):
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    step *= 0.5
                    continue
                a_new = a_new * norm / s_new
                sc = fast_conv_self(a_new).max()
                if sc <= max_c + 0.03 * T_val:
                    break
                step *= 0.5

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.1f}: best={best_score:.8f}")

    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    n2 = len(best_seq)
    t2 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 2: Subgradient n={n2} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = subgradient_optimize(best_seq, t2, verbose=True)
        try_update(a_out, f"subgrad_n{n2}")

    # Phase 3: LP Nesterov round 2
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = len(best_seq)
    t3 = min(remaining * 0.22, 300.0)
    print(f"\n=== Phase 3: LP-Nesterov r2 n={n3} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t3, solver, verbose=True)
        try_update(a_out, f"lp2_n{n3}")

    # Phase 4: Pair CD
    remaining = time_budget - (time.monotonic() - t_global)
    n4 = len(best_seq)
    t4 = min(remaining * 0.22, 280.0)
    print(f"\n=== Phase 4: Pair CD n={n4} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = pair_cd_optimize(best_seq, t4, rng, verbose=True)
        try_update(a_out, f"pcd_n{n4}")

    # Phase 5: LSE + LP alternating
    remaining = time_budget - (time.monotonic() - t_global)
    n5 = len(best_seq)
    print(f"\n=== Phase 5: LSE+LP alt n={n5} (remaining={remaining:.0f}s) ===")
    if remaining > 80:
        t_half = min(remaining * 0.35, 100.0)
        a_out, _ = lse_gradient_descent(best_seq, t_half, verbose=False)
        try_update(a_out, "lse_mid")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50:
            t_lp = min(remaining * 0.5, 120.0)
            a_out, _ = nesterov_lp_optimize(best_seq, t_lp, solver, verbose=False)
            try_update(a_out, "lp_mid")

    # Phase 6: Multi-start diversification
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Multi-start (remaining={remaining:.0f}s) ===")
    if remaining > 100:
        n_starts = 3
        t_per = min(remaining / (n_starts + 1), 120.0)
        for ms in range(n_starts):
            if time.monotonic() - t_global > time_budget - 60:
                break
            n_ms = len(best_seq)
            perturbed = best_seq.copy()
            n_pert = max(50, n_ms // 10)
            idx_pert = rng.choice(n_ms, size=n_pert, replace=False)
            perturbed[idx_pert] *= (1.0 + 0.5 * rng.standard_normal(n_pert))
            perturbed = np.maximum(perturbed, 0.0)
            s_p = perturbed.sum()
            if s_p > 1e-15:
                perturbed = perturbed / s_p
            a_out, _ = nesterov_lp_optimize(perturbed, t_per, solver, verbose=False)
            try_update(a_out, f"multistart_{ms}")

    # Phase 7: Length extension
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 400
    print(f"\n=== Phase 7: Length ext n={n_long} (remaining={remaining:.0f}s) ===")
    if remaining > 120:
        init_long = interpolate_seq(best_seq, n_long)
        t_long = min(remaining * 0.5, 150.0)
        a_out, _ = nesterov_lp_optimize(init_long, t_long, solver, verbose=False)
        try_update(a_out, f"lp_long_{n_long}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50 and len(best_seq) == n_long:
            t_cd = min(remaining * 0.5, 60.0)
            a_out2, _ = pair_cd_optimize(best_seq, t_cd, rng, verbose=False)
            try_update(a_out2, f"pcd_long_{n_long}")

    # Phase 8: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 8: Final polish n={n_final} (remaining={remaining:.0f}s) ===")
    if remaining > 40:
        t_f = min(remaining * 0.4, 50.0)
        a_out, _ = lse_gradient_descent(best_seq, t_f, verbose=False)
        try_update(a_out, "lse_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t_f2 = min(remaining * 0.5, 50.0)
            a_out, _ = pair_cd_optimize(best_seq, t_f2, rng, verbose=False)
            try_update(a_out, "pcd_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 25:
            t_f3 = min(remaining * 0.6, 40.0)
            a_out, _ = subgradient_optimize(best_seq, t_f3, verbose=False)
            try_update(a_out, "subgrad_final")

    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
