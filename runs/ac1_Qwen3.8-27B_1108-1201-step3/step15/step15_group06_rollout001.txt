def build_conv_matrix(a, lags):
    """Build the convolution matrix T where T[k,j] = a[k-j] (if valid)."""
    n = len(a)
    m = len(lags)
    T = np.zeros((m, n))
    for row, k in enumerate(lags):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                T[row, j] = a[idx]
    return T

def lp_max_sum(a, M, solver, active_lags):
    """Maximize sum(b) s.t. conv(a,b)[k] <= M for k in active_lags, b >= 0."""
    n = len(a)
    T = build_conv_matrix(a, active_lags)
    b = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(b)), [T @ b <= M])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if val.sum() < 1e-15:
        return None, None
    # Get dual variables
    dual = None
    try:
        dual = np.asarray(prob.constraints[0].dual_value, dtype=np.float64)
    except Exception:
        pass
    return val, dual

def lp_min_max(a, solver, active_lags, target_sum):
    """Minimize t s.t. conv(a,b)[k] <= t, sum(b) = target_sum, b >= 0."""
    n = len(a)

    b = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    prob = cp.Problem(cp.Minimize(t), [T @ b <= t, cp.sum(b) == target_sum])

        return None, None

        return None, None

    t_val = float(t.value) if t.value is not None else float('inf')
    if val.sum() < 1e-15:
        return None, None
    return val, t_val
