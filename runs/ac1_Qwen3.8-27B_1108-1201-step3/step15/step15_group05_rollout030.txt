from scipy.fft import rfft, irfft

def fast_convolve(a):

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    """Compute gradient of sum_k (a*a)[k]^2 w.r.t. a.
    Result: 4 * cross_correlation(b, a) where b = a*a.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smooth gradient descent with adaptive temperature
2. LP best-response with active set management
3. Subgradient polishing at active peaks
4. Multi-length search (1600, 2000, 2400)
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def get_score(a):
    """Compute the score directly."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (S * S)

def normalize(a, target_sum=None):
    s = a.sum()
    if s < 1e-15:
        return a.copy()
    if target_sum is None:
        return a / s
    return a * target_sum / s

def logsumexp_gradient(a, tau):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau * (a*a)[k])) w.r.t. a.
    Returns (grad, f_value).
    """
    n = len(a)
    size = 2 * n - 1
    b = fast_convolve(a)
    
    # Compute weights w_k = exp(tau * b[k]) / sum exp(tau * b[k])
    # Use the shifted log-sum-exp for numerical stability
    b_max = b.max()
    shifted = tau * (b - b_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    
    # f_value = (1/tau) * (tau * b_max + log(sum exp_shifted))
    f_val = b_max + np.log(exp_shifted.sum()) / tau
    
    # grad[j] = (2/tau) * sum_k w[k] * a[k-j] * [0<=k-j<n]
    # For efficiency, only use significant weights
    threshold = 0.01 / len(w)  # adaptive threshold
    significant = np.where(w > threshold)[0]
    
    grad = np.zeros(n)
    for k in significant:
        wk = w[k]
        # a[k-j] is valid for j in [max(0, k-n+1), min(n-1, k)]
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        # a[k-j] for j in [lo, hi): indices are k-lo, k-lo+1, ..., k-(hi-1)
        # = k-lo, ..., k-hi+1
        # Which is a[k-hi+1 : k-lo+1]
        idx_lo = k - hi + 1
        idx_hi = k - lo + 1
        grad[lo:hi] += 2.0 / tau * wk * a[idx_lo:idx_hi]
    
    return grad, f_val

    Gradient descent on log-sum-exp of autocorrelation.

    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = normalize(np.maximum(np.asarray(a0, dtype=np.float64), 0.0), S)
    
    best_a = a.copy()
    best_score = get_score(a)
    t0 = time.monotonic()
    
    # Adaptive step size
    step = 0.1 / n

1. Log-sum-exp smooth gradient descent with adaptive temperature
2. Multi-length search (1600, 2000, 2400, 3200)
3. Subgradient polishing at active peaks
4. Exact coordinate descent for final refinement

import numpy as np

def fast_convolve(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def get_score(a):
    """Compute the score 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    max_b = b.max()
    return 2.0 * n * max_b / (S * S)


def normalize(a, target_sum=None):
    """Normalize a to have the given target sum (or unit sum)."""
    s = a.sum()
    if s < 1e-15:
        return a.copy()
    if target_sum is None:
        return a / s
    return a * target_sum / s


def eval_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, x)) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def logsumexp_grad(a, tau):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*(a*a)[k])).
    Returns (grad, f_value).
    """
    n = len(a)
    b = fast_convolve(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    exp_sum = exp_shifted.sum()
    if exp_sum < 1e-300:
        w = np.ones(len(b)) / len(b)
    else:
        w = exp_shifted / exp_sum
    f_val = b_max + np.log(max(exp_sum, 1e-300)) / tau

    # grad[j] = (2/tau) * sum_k w[k] * a[k-j] * [0<=k-j<n]
    # Use only significant weights for speed
    n_sig = max(10, min(len(b), int(0.5 * np.sqrt(n))))
    top_k = np.argsort(w)[-n_sig:]
    grad = np.zeros(n)
    for k in top_k:
        wk = w[k]
        if wk < 1e-10:
            continue
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        idx_lo = k - hi + 1
        idx_hi = k - lo + 1
        grad[lo:hi] += (2.0 / tau) * wk * a[idx_lo:idx_hi]
    return grad, f_val

def coordinate_descent(a0, time_limit, verbose=False):
    """
    Coordinate descent with exact 1D optimization per coordinate.
    For coordinate j, minimize max_k (base[k] + 2*x*a[k-j] + x^2*[k==2j])
    subject to the sum constraint (handled by renormalization).
    """
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = normalize(np.maximum(np.asarray(a0, dtype=np.float64), 0.0), S)

    best_a = a.copy()
    best_score = get_score(a)
    t0 = time.monotonic()
    size = 2 * n - 1

    n_passes = 0
    no_improve = 0
    max_no_improve = 4

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]
            s_others = a.sum() - aj

            # Compute background: autocorrelation with a[j] = 0
            # b[k] = sum_i a[i]*a[k-i], terms involving a[j]:
            #   i=j: a[j]*a[k-j] (if 0<=k-j<n)
            #   k-i=j: a[k-j]*a[j] (if 0<=k-j<n) -> same contribution
            #   So: b[k] = bg[k] + 2*a[j]*a[k-j] (if 0<=k-j<n)
            #   where bg[k] = b[k] - 2*aj*a[k-j] (if 0<=k-j<n)
            # But we also need bg to not include a[j]^2 at k=2j:
            #   At k=2j: b[2j] includes a[j]*a[j] = aj^2 (from i=j=k-i)
            #   So: bg[2j] = b[2j] - 2*aj*a[j] + aj^2 - aj^2 = b[2j] - 2*aj^2
            #   Wait: b[2j] = ... + a[j]*a[j] + ... = bg[2j] + 2*aj*a[j] ... no
            #   Actually b[k] = sum_i a[i]*a[k-i]. The terms with a[j]:
            #     i=j: term is a[j]*a[k-j]
            #     i=k-j: term is a[k-j]*a[j]
            #     If j=k-j (k=2j): both are a[j]*a[j], but it's ONE term, not two.
            #   So for k != 2j: contribution is 2*a[j]*a[k-j]
            #   For k = 2j: contribution is a[j]^2
            #   Therefore: bg[k] = b[k] - 2*aj*a[k-j] (k!=2j, if 0<=k-j<n)
            #               bg[2j] = b[2j] - aj^2 (if 2j < size)
            #   And new_b[k] = bg[k] + 2*x*a[k-j] (k!=2j) + x^2 (k=2j)

            b_cur = fast_convolve(a)
            bg = b_cur.copy()
            # Subtract contributions of a[j]
            lo = max(0, 2 * j - n + 1)
            hi = min(size, 2 * j + 1)
            for k in range(lo, hi):
                kj = k - j
                if 0 <= kj < n:
                    if k == 2 * j:
                        bg[k] -= aj * aj
                    else:
                        bg[k] -= 2.0 * aj * a[kj]

            # Now minimize max_k (bg[k] + 2*x*a[k-j] + x^2*[k==2j]) over x in [0, S]
            # Subject to: s_others + x <= S (to keep sum <= S, then we renormalize)
            # Actually we just need x >= 0, and we'll renormalize.
            # The objective after renormalization to sum=S:
            #   score(x) = 2n * max_k(bg[k] + 2*x*a[k-j] + x^2*[k==2j]) / (s_others+x)^2
            # We minimize this over x in [0, S*2] (generous upper bound).

            # Use ternary search / golden section on x
            lo_x = 0.0
            hi_x = 2.0 * S

            def obj_x(x):
                new_b = bg.copy()
                # Add 2*x*a[k-j] for valid k
                lo2 = max(0, -j)
                hi2 = min(size, n + n - j)  # k-j in [0,n) means k in [j, j+n)
                k_lo = max(0, j)
                k_hi = min(size, j + n)
                for k in range(k_lo, k_hi):
                    kj = k - j
                    new_b[k] += 2.0 * x * a[kj]
                if 2 * j < size:
                    new_b[2 * j] += x * x
                total = s_others + x
                if total < 1e-15:
                    return float('inf')
                return 2.0 * n * new_b.max() / (total * total)

            # Current objective
            obj_cur = obj_x(aj)

            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_g = lo_x + (1.0 - gr) * (hi_x - lo_x)
            b_g = lo_x + gr * (hi_x - lo_x)
            fa = obj_x(a_g)
            fb = obj_x(b_g)

            for _ in range(30):
                if fa < fb:
                    hi_x = b_g
                    b_g = a_g
                    fb = fa
                    a_g = lo_x + (1.0 - gr) * (hi_x - lo_x)
                    fa = obj_x(a_g)
                else:
                    lo_x = a_g
                    a_g = b_g
                    fa = fb
                    b_g = lo_x + gr * (hi_x - lo_x)
                    fb = obj_x(b_g)

            x_opt = (lo_x + hi_x) / 2.0
            obj_opt = obj_x(x_opt)

            if obj_opt < best_score - 1e-12 and abs(x_opt - aj) > 1e-10:
                a[j] = x_opt
                a = normalize(a, S)
                new_score = get_score(a)
                if new_score < best_score - 1e-13:
                    best_score = new_score
                    best_a = a.copy()
                    improved = True
                else:
                    a[j] = aj
                    a = normalize(a, S)

        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def lp_best_response(a_fixed, M, k_indices, n):
    """
    Solve LP: max sum(b) s.t. (a_fixed*b)[k] <= M for k in k_indices, b >= 0.
    """
    if not HAS_CP:
        return None

    k_indices = np.asarray(k_indices)
    size = len(k_indices)
    if size == 0:
        return None

    # Build constraint matrix
    mat = np.zeros((size, n))
    for idx, k in enumerate(k_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        idx_lo = k - hi + 1
        idx_hi = k - lo + 1
        mat[idx, lo:hi] = a_fixed[idx_lo:idx_hi]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=cp.CLARABEL, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except Exception:
            return None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp(a0, time_limit, verbose=False, seed=0):
    """Nesterov-accelerated LP best-response iteration."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = normalize(np.maximum(np.asarray(a0, dtype=np.float64), 0.0), S)

    best_a = a.copy()
    best_score = get_score(a)

    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    stall_count = 0
    stall_threshold = 60
    n_iters = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        b_y = fast_convolve(y)
        M_curr = b_y.max()
        if M_curr < 1e-12:
            break

        # Active set: lags within tolerance of max
        tol = max(0.005, 0.03 * np.exp(-t / 500.0))
        active_set = np.where(b_y >= M_curr * (1.0 - tol))[0]
        # Always include top few
        top_k = np.argsort(b_y)[-max(5, n // 100):]
        active_set = np.unique(np.concatenate([active_set, top_k]))

        b_resp = lp_best_response(y, M_curr, active_set, n)
        if b_resp is None:
            continue

        b_sum = b_resp.sum()
        if b_sum < 1e-15:
            continue
        g = b_resp * S / b_sum
        n_iters += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * S / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.95)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * S / s_a

        y_prev = y
        y = y_new
        a = a_new

        score_y = get_score(y)
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = get_score(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        if stall_count > stall_threshold:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            stall_count = 0

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_initialization(n, parent=None):
    """Create initial sequence of length n."""
    if parent is not None and len(parent) > 10:
        return interpolate_sequence(parent, n)
    # Fallback: raised cosine
    x = np.linspace(0.0, 1.0, n)
    a = 0.5 + 0.4 * np.cos(np.pi * (2.0 * x - 1.0)) + 0.1 * np.cos(2.0 * np.pi * (2.0 * x - 1.0))
    a = np.maximum(a, 0.01)
    return a / a.sum()


def propose_candidate():
    """
    Main search function. Uses smooth gradient descent, subgradient polishing,
    coordinate descent, and LP best-response to minimize the score.
    Tries multiple sequence lengths.
    """
    t_global = time.monotonic()
    BUDGET = 1750.0

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 10 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
    except Exception:
        pass

    # Determine candidate lengths
    lengths_to_try = []
    if parent is not None:
        n_par = len(parent)
        base_scores = {}
        for n_try in [n_par, int(n_par * 1.25), int(n_par * 1.5), int(n_par * 2.0)]:
            n_try = min(n_try, 4000)
            n_try = max(n_try, 400)
            if n_try not in base_scores:
                a_test = make_initialization(n_try, parent)
                sc = eval_safe(a_test)
                base_scores[n_try] = sc
                print(f"  Init n={n_try}: score={sc:.8f}")
        # Sort by score, take top 3
        sorted_lengths = sorted(base_scores.items(), key=lambda x: x[1])
        for n_try, sc in sorted_lengths[:3]:
            lengths_to_try.append(n_try)
    else:
        lengths_to_try = [1600, 2000, 2400]

    # Ensure parent length is included
    if parent is not None and len(parent) not in lengths_to_try:
        lengths_to_try.append(len(parent))
    lengths_to_try = lengths_to_try[:4]  # Max 4 lengths

    print(f"Trying lengths: {lengths_to_try}")

    best_overall = None
    best_overall_score = float('inf')

    def try_update(a, label):
        nonlocal best_overall, best_overall_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        sc = eval_safe(a)
        if np.isfinite(sc) and sc < best_overall_score:
            best_overall_score = sc
            best_overall = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize best with parent
    if parent is not None:
        try_update(parent, "parent")

    # Allocate time across lengths
    n_lengths = len(lengths_to_try)
    # Weight more time to shorter (likely better) lengths
    weights = [1.0 / (1.0 + 0.3 * i) for i in range(n_lengths)]
    total_w = sum(weights)
    weights = [w / total_w for w in weights]

    for li, n in enumerate(lengths_to_try):
        remaining = BUDGET - (time.monotonic() - t_global)
        if remaining < 80:
            break
        alloc = remaining * weights[li]
        print(f"\n=== Length n={n}, budget={alloc:.0f}s ===")

        a_init = make_initialization(n, parent)

        # Phase 1: Smooth gradient descent (main workhorse)
        t1 = min(alloc * 0.40, 400.0)
        if t1 > 30:
            remaining_check = BUDGET - (time.monotonic() - t_global)
            if remaining_check > 100:
                print(f"  Phase 1: SGD ({t1:.0f}s)")
                a1, s1 = smooth_gradient_descent(a_init, t1, verbose=True)
                try_update(a1, f"sgd_n{n}")

        # Phase 2: LP best-response
        remaining = BUDGET - (time.monotonic() - t_global)
        if remaining < 60:
            break
        t2 = min(alloc * 0.25, 250.0)
        if t2 > 30 and HAS_CP:
            print(f"  Phase 2: LP ({t2:.0f}s)")
            a2, s2 = nesterov_lp(a1, t2, verbose=True, seed=li * 100)
            try_update(a2, f"lp_n{n}")

        # Phase 3: Subgradient polish
        remaining = BUDGET - (time.monotonic() - t_global)
        if remaining < 50:
            break
        t3 = min(alloc * 0.20, 200.0)
        if t3 > 20:
            print(f"  Phase 3: Subgrad ({t3:.0f}s)")
            a3, s3 = subgradient_polish(a2, t3, n_peaks=7, verbose=True)
            try_update(a3, f"subg_n{n}")

        # Phase 4: Coordinate descent
        remaining = BUDGET - (time.monotonic() - t_global)
        if remaining < 40:
            break
        t4 = min(alloc * 0.15, 150.0)
        if t4 > 15:
            print(f"  Phase 4: CD ({t4:.0f}s)")
            a4, s4 = coordinate_descent(a3, t4, verbose=True)
            try_update(a4, f"cd_n{n}")

    # Final polishing on the best found
    remaining = BUDGET - (time.monotonic() - t_global)
    if best_overall is not None and remaining > 50:
        n_best = len(best_overall)
        print(f"\n=== Final polishing n={n_best}, remaining={remaining:.0f}s ===")

        # Final subgradient
        t_final1 = min(remaining * 0.4, 100.0)
        if t_final1 > 20:
            a_f1, _ = subgradient_polish(best_overall, t_final1, n_peaks=10, verbose=True)
            try_update(a_f1, "final_subg")

        # Final smooth GD
        remaining = BUDGET - (time.monotonic() - t_global)
        if remaining > 40:
            t_final2 = min(remaining * 0.4, 80.0)
            a_f2, _ = smooth_gradient_descent(best_overall, t_final2, verbose=False)
            try_update(a_f2, "final_sgd")

        # Final CD
        remaining = BUDGET - (time.monotonic() - t_global)
        if remaining > 30:
            t_final3 = min(remaining * 0.5, 60.0)
            a_f3, _ = coordinate_descent(best_overall, t_final3, verbose=False)
            try_update(a_f3, "final_cd")

    # Final evaluation
    if best_overall is not None:
        fs = eval_safe(best_overall)
        if np.isfinite(fs):
            best_overall_score = fs
        print(f"\nFINAL: score={best_overall_score:.8f}, n={len(best_overall)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        # Fallback
        n_fb = 1600
        x = np.linspace(0.0, 1.0, n_fb)
        a_fb = 0.5 + 0.3 * np.cos(2 * np.pi * x)
        a_fb = np.maximum(a_fb, 0.0)
        best_overall = a_fb / a_fb.sum()
        best_overall_score = eval_safe(best_overall)
        print(f"Fallback: score={best_overall_score:.8f}")

    return [float(max(0.0, min(1000.0, x))) for x in best_overall]
