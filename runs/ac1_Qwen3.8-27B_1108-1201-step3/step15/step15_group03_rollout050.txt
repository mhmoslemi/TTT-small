"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LSE smooth gradient descent for warmup
2. LP best-response with adaptive constraint screening and Nesterov acceleration
3. Penalty-based fast best-response for large n

6. Spectral flattening in frequency domain

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print("  Using solver (fallback): ECOS")
    return _solver


def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_T_cols(a, indices):
    """Build columns of T for given lag indices. T[k, i] = a[k-i] if valid."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((len(indices), n))
    for row, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if hi > lo:
            T[row, lo:hi] = a[k - np.arange(lo, hi)]
    return T


def lp_solve_with_constraints(a, M, solver, active_indices):
    """Solve LP: max sum(b) s.t. b>=0, conv(a,b)[k] <= M for k in active_indices."""
    n = len(a)
    T = build_T_cols(a, active_indices)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)


def adaptive_lp_solve(a, M, solver, max_active=300, max_iters=15):
    """Adaptive LP: start with few constraints, add violated ones iteratively."""
    n = len(a)
    size = 2 * n - 1
    conv_a = fast_conv_self(a)
    
    # Start with top K lags
    K = min(30, size)
    active = np.argsort(conv_a)[-K:]
    
    b = None
    for it in range(max_iters):
        b = lp_solve_with_constraints(a, M, solver, active)
        if b is None:
            return None
        
        # Check full convolution for violations
        conv_ab = fast_conv_cross(a, b)
        violated = np.where(conv_ab > M * 1.0005)[0]
        
        # Remove already-active
        mask = ~np.isin(violated, active)
        new_violated = violated[mask]
        
        if len(new_violated) == 0:
            return b
        
        # Add violated constraints (cap the size)
        active = np.union1d(active, new_violated)
        
        if len(active) > max_active:
            # Keep the most violated
            vals = conv_ab[active]
            keep_idx = np.argsort(vals)[-max_active:]
            active = active[keep_idx]
    
    return b


def penalty_best_response(a, M, n_iter=200, verbose=False):
    """Fast penalty-based approximation of the LP best response."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    
    b = np.ones(n) / n
    rho = 10.0
    lr = 1.0 / np.sqrt(n)
    
    for it in range(n_iter):
        B = rfft(b, fft_size)
        conv_ab = irfft(A * B, fft_size)[:size]
        
        # Violation amounts
        viol = np.maximum(conv_ab - M, 0.0)
        
        # Gradient of penalty: T^T @ viol
        V = rfft(viol, fft_size)
        grad_pen = irfft(np.conj(A) * V, fft_size)[:n] + irfft(A * V, fft_size)[:n]
        
        # Gradient of objective is ones
        grad = np.ones(n) - 2.0 * rho * grad_pen
        
        # Step with projection onto b >= 0
        b = np.maximum(b + lr * grad, 0.0)
        s = b.sum()
        if s < 1e-15:
            b = np.ones(n) / n
        else:
            # Don't normalize here; the LP doesn't have a sum constraint
            pass
        
        # Decay learning rate
        lr *= 0.995
        rho *= 1.01
    
    return np.maximum(b, 0.0)


def lse_gradient_descent(a0, time_limit, verbose=True):
    """Log-sum-exp smoothed gradient descent for fast initial progress."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 14
    temps = np.geomspace(60.0, 0.1, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        n_steps = max(40, min(400, int(remaining * 8)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)

            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            # Also account for the normalization constraint
            # d/da [max_conv / sum^2] -> gradient of max_conv (sum is fixed by renormalization)
            
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            
            # Adaptive step size
            step = norm / (T_val * np.sqrt(n)) * (1.0 / (1.0 + 0.03 * it))

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best={best_score:.8f}")

    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, adaptive_constraints=True):
    """Nesterov-accelerated LP best response with adaptive constraint management."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1

    # Nesterov momentum with periodic restarts
    last_best = best_score
    restart_counter = 0
    max_restart_gap = 200

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Solve LP with adaptive constraints
        if adaptive_constraints:
            b = adaptive_lp_solve(y, M, solver, max_active=200, max_iters=10)
        else:
            b = adaptive_lp_solve(y, M, solver, max_active=size, max_iters=1)

        if b is None:
            # Fallback to penalty method
            b = penalty_best_response(y, M, n_iter=50)
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Momentum
        if t > 1:
            momentum = (t - 1.0) / (t + 2.0)
            # Cap momentum to avoid instability
            momentum = min(momentum, 0.95)
            a_new = y_new + momentum * (y_new - y)
        else:
            a_new = y_new.copy()
        
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score - 1e-12:
            best_score = sc_y
            best_a = y.copy()
            restart_counter = 0

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score - 1e-12:
            best_score = sc_a
            best_a = a.copy()
            restart_counter = 0

        # Periodic restart check
        restart_counter += 1
        if restart_counter >= max_restart_gap and best_score > last_best * (1 - 1e-6):
            # No progress in last max_restart_gap iterations, restart momentum
            y = best_a.copy()
            a = best_a.copy()
            last_best = best_score
            restart_counter = 0
            max_restart_gap = max(100, max_restart_gap // 2)

        if verbose and t % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv_self(a).max()
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for it in range(200):
        if time.monotonic() - t0 > time_limit:
            break

        A = rfft(a, fft_size)
        power = np.abs(A) ** 2
        
        # Compute autocorrelation
        conv = fast_conv_self(a)
        cur_max = conv.max()
        
        # Identify peak lags
        peak_idx = np.where(conv >= cur_max * 0.92)[0]
        
        # For each peak lag k, we want to reduce its contribution
        # In frequency domain, lag k corresponds to phase factor exp(2πikω/size)
        # We can try to reduce power at frequencies that contribute to peaks
        
        # Simple approach: reduce power at frequencies where the interference is constructive
        # for the peak lags
        freqs = np.arange(fft_size // 2 + 1) * 2.0 * np.pi / fft_size
        
        power_new = power.copy()
        for kl in peak_idx[:10]:
            # Phase factor for lag kl
            phase = np.exp(-1j * kl * freqs)
            # Frequencies where phase is near 1 (constructive interference)
            constructive = np.abs(phase - 1.0) < 0.5
            power_new[constructive] *= 0.99
        
        # Reconstruct with new power (keep phase)
        phase_A = np.angle(A)
        mag_A = np.sqrt(power_new)
        A_new = mag_A * np.exp(1j * phase_A)
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new * norm / s_new
        
        sc = fast_conv_self(a_new).max()
        if sc < best_score - 1e-10:
            a = a_new
            best_score = sc
            best_a = a.copy()
        else:
            # Try a small random perturbation in frequency domain
            A_pert = A * (1.0 + 0.01 * rng.standard_normal(fft_size // 2 + 1))
            a_pert = irfft(A_pert, fft_size)[:n]
            a_pert = np.maximum(a_pert, 0.0)
            s_pert = a_pert.sum()
            if s_pert > 1e-15:
                a_pert = a_pert * norm / s_pert
                sc_pert = fast_conv_self(a_pert).max()
                if sc_pert < best_score - 1e-10:
                    a = a_pert
                    best_score = sc_pert
                    best_a = a.copy()

    if verbose:
        print(f"    Spectral flatten: best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    """Generate structured initial sequences."""
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.2
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.1 * np.cos(2 * np.pi * 7 * x)
    elif choice == 3:
        a = np.exp(-5.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-5.0 * (x - 0.75) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2.0
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.5
    elif choice == 6:
        a = np.exp(-3.0 * x) + 0.3 + 0.2 * np.sin(4 * np.pi * x)
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(6 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        a = np.sinc(4.0 * (x - 0.5)) ** 2 + 0.05
    elif choice == 9:
        a = np.exp(-1.5 * (x - 0.5) ** 2 / (0.1 + 0.05 * rng.random()))
    elif choice == 10:
        # Almost uniform with slight modulation
        a = 1.0 + 0.3 * np.sin(2 * np.pi * rng.integers(2, 8) * x)
    else:
        # Bimodal
        a = np.exp(-8.0 * (x - 0.3) ** 2) + np.exp(-8.0 * (x - 0.7) ** 2) + 0.1
    a = np.maximum(a, 0.0)
    a = a + 0.005 * rng.random(n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def eval_fn(a):
    """Evaluate using the provided evaluate_sequence function."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        if len(a_list) == 0:
            return float('inf')
        for val in a_list:
            if val < 0 or not np.isfinite(val):
                return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def momentum_subgradient(a0, time_limit, verbose=True):
    """Momentum subgradient with adaptive temperature."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    momentum = np.zeros(n)
    beta = 0.85
    lr = 5.0 / np.sqrt(n)

    for it in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        max_c = conv.max()

        # Adaptive temperature
        temperature = max(0.05, 0.5 / (1.0 + 0.005 * it))
        shifted = (conv - max_c) / temperature
        shifted = np.maximum(shifted, -80.0)
        w = np.exp(shifted)
        w = w / w.sum()

        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        g1 = irfft(np.conj(A) * W, fft_size)[:n]
        g2 = irfft(A * W, fft_size)[:n]
        grad = 2.0 * (g1 + g2)

        momentum = beta * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        a = a_new

        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Decay learning rate
        lr *= 0.9997

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function: multi-phase optimization with adaptive LP Nesterov."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LSE gradient descent warmup
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 100.0)
    print(f"\n=== Phase 1: LSE GD n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 15:
        a_out, _ = lse_gradient_descent(best_seq, t1, verbose=True)
        try_update(a_out, "lse1")

    # Phase 2: Main LP Nesterov
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.35, 550.0)
    print(f"\n=== Phase 2: LP Nesterov n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 50:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, solver, verbose=True, adaptive_constraints=True)
        try_update(a_out, "lp1")

    # Phase 3: Subgradient momentum
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.08, 100.0)
    print(f"\n=== Phase 3: SubGrad n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 10:
        a_out, _ = momentum_subgradient(best_seq, t3, verbose=True)
        try_update(a_out, "subgrad")

    # Phase 4: LP Nesterov round 2 (refined)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 4: LP Nesterov round 2 n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t4, solver, verbose=True, adaptive_constraints=True)
        try_update(a_out, "lp2")

    # Phase 5: Spectral flattening
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.06, 70.0)
    print(f"\n=== Phase 5: Spectral n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 10:
        a_out, _ = spectral_flatten(best_seq, t5, rng, verbose=True)
        try_update(a_out, "spectral")

    # Phase 6: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 6: Pair CD n={len(best_seq)} (budget={t6:.0f}s) ===")
    if t6 > 20:
        a_out, _ = pair_cd_optimize(best_seq, t6, rng, verbose=True)
        try_update(a_out, "cd1")

    # Phase 7: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Multi-length (budget={remaining:.0f}s) ===")
    if remaining > 100:
        # Try slightly longer
        n_long = n_base + 200
        t7a = min(remaining * 0.4, 200.0)
        init_long = interpolate_seq(best_seq, n_long)
        a_out, _ = nesterov_lp_optimize(init_long, min(t7a, 150.0), solver, verbose=False, adaptive_constraints=True)
        try_update(a_out, f"lp_long{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t7b = min(remaining * 0.4, 100.0)
            a_out2, _ = pair_cd_optimize(best_seq, t7b, rng, verbose=False)
            try_update(a_out2, "cd_long")

    # Phase 8: Final LP + CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 8: Final polish n={n_final} (budget={remaining:.0f}s) ===")
    if remaining > 40:
        t8a = min(remaining * 0.5, 100.0)
        a_out, _ = nesterov_lp_optimize(best_seq, t8a, solver, verbose=False, adaptive_constraints=True)
        try_update(a_out, "lp_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t8b = min(remaining * 0.7, 60.0)
            a_out2, _ = pair_cd_optimize(best_seq, t8b, rng, verbose=False)
            try_update(a_out2, "cd_final")

    # Final LSE polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        a_out, _ = lse_gradient_descent(best_seq, min(remaining, 30.0), verbose=False)
        try_update(a_out, "lse_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
