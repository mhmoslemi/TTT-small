def nesterov_lse_optimize(a0, time_limit, verbose=False):
    """Nesterov-accelerated LSE gradient descent with adaptive temperature."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    
    best_a = a.copy()
    best_max = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Nesterov momentum
    y = a.copy()
    t_nest = 1.0
    best_t = 0
    patience = 0
    
    # Temperature schedule with restarts
    T = 50.0
    T_min = 0.1
    T_decay = 0.92
    T_restart = 10.0
    
    lr_base = 0.3 / np.sqrt(n)
    
    while time.monotonic() - t0 < time_limit:
        # Nesterov step
        t_nest += 1.0
        momentum = (t_nest - 1.0) / (t_nest + 1.8)
        y_new = y + momentum * (y - a)
        
        # Compute convex at y_new
        conv = fast_conv_self(y_new)
        max_c = conv.max()
        if max_c < 1e-15:
            break
        
        # LSE approximation of max
        shifted = (conv - max_c) / T
        shifted = np.maximum(shifted, -100.0)
        exp_c = np.exp(shifted)
        w = exp_c / exp_c.sum()
        
        # Gradient: 2 * (convolve(conj(a), w) + convolve(a, w))[:n]
        W = rfft(w, fft_size)
        Y = rfft(y_new, fft_size)
        
        g1 = irfft(np.conj(Y) * W, fft_size)[:n]
        g2 = irfft(Y * W, fft_size)[:n]
        grad = 2.0 * (g1 + g2)
        
        # Step
        lr = lr_base / np.sqrt(t_nest + 1.0)
        a_new = y_new - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        y = y_new
        a = a_new
        
        # Track best
        sc = fast_conv_self(a).max()
        if sc < best_max - 1e-12:
            best_max = sc
            best_a = a.copy()
            best_t = int(t_nest)
            patience = 0
        else:
            patience += 1
        
        # Cooldown
        if patience > 500:
            if T > T_min * 2:
                T *= T_decay
            else:
                # Restart temperature
                T = T_restart
                # Perturb
                a = a * (1.0 + 0.05 * np.random.standard_normal(n))
                a = np.maximum(a, 0.0)
                a = a * norm / a.sum()
                y = a.copy()
                t_nest = 1.0
            patience = 0
        
        if verbose and int(t_nest) % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLSE t={int(t_nest)} T={T:.2f} best={best_max:.8f} ({elapsed:.0f}s)")
    
    return best_a, best_max

def adam_lse_optimize(a0, time_limit, verbose=False):
    """Adam-based LSE gradient descent with multiple temperature stages."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    
    best_a = a.copy()
    best_max = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    t_adam = 0
    
    # Multi-stage with restarts
    stages = [
        (100.0, 5.0),    # (T_start, T_end)
        (5.0, 1.0),
        (1.0, 0.2),
        (0.2, 0.05),
        (0.05, 0.01),
    ]
    
    for stage_idx, (T_start, T_end) in enumerate(stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        
        n_stages = 30
        temps = np.geomspace(T_start, T_end, n_stages)
        
        for si, T_val in enumerate(temps):
            if time.monotonic() - t0 > time_limit:
                break
            remaining = time_limit - (time.monotonic() - t0)
            if remaining < 5:
                break
            
            n_steps = max(20, min(100, int(remaining * 5)))
            lr = 0.4 / np.sqrt(n)
            
            for it in range(n_steps):
                if time.monotonic() - t0 > time_limit:
                    break
                t_adam += 1
                
                conv = fast_conv_self(a)
                max_c = conv.max()
                if max_c < 1e-15:
                    break
                
                shifted = (conv - max_c) / T_val
                shifted = np.maximum(shifted, -100.0)
                exp_c = np.exp(shifted)
                w = exp_c / exp_c.sum()
                
                W = rfft(w, fft_size)
                A = rfft(a, fft_size)
                
                g1 = irfft(np.conj(A) * W, fft_size)[:n]
                g2 = irfft(A * W, fft_size)[:n]
                grad = 2.0 * (g1 + g2)
                
                m = beta1 * m + (1.0 - beta1) * grad
                v = beta2 * v + (1.0 - beta2) * grad * grad
                m_hat = m / (1.0 - beta1 ** t_adam)
                v_hat = v / (1.0 - beta2 ** t_adam)
                
                a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    a_new = a.copy()
                else:
                    a_new = a_new * norm / s_new
                
                a = a_new
                sc = fast_conv_self(a).max()
                if sc < best_max - 1e-12:
                    best_max = sc
                    best_a = a.copy()
        
        if verbose:
            print(f"    Adam LSE stage {stage_idx}: best={best_max:.8f}")
    
    return best_a, best_max

def lp_best_response(a, M, solver, active_set=None, warm_start=None):
    """Solve LP: max sum(b) s.t. b>=0, conv(a,b)[k] <= M for active k."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    
    if warm_start is not None:
        b_var.value = warm_start
    
    constraints = []
    
    if active_set is not None and 0 < len(active_set) < size:
        # Build constraint matrix for active set
        T = np.zeros((len(active_set), n))
        for idx, k in enumerate(active_set):
            for i in range(max(0, k - n + 1), min(n, k + 1)):
                T[idx, i] = a[k - i]
        constraints = [T @ b_var <= M]
    else:
        # All constraints
        T = np.zeros((size, n))
        for k in range(size):
            for i in range(max(0, k - n + 1), min(n, k + 1)):
                T[k, i] = a[k - i]
        constraints = [T @ b_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None, None
    return np.maximum(val, 0.0), None

    y = a.copy()

    active_set = None
    n_lp_solved = 0

            break

        threshold = M * 0.8

            active_set = np.where(conv_y >= M * 0.5)[0]
        if len(active_set) > 300:

            top_k = 100
            active_set = np.argsort(conv_y)[-top_k:]

        b, _ = lp_best_response(y, M, solver, active_set, warm_b)

            active_set = None

            violated = np.where(check > M * 1.0001)[0]

                active_set = np.union1d(active_set, violated)
                b2, _ = lp_best_response(y, M, solver, active_set, b)
                if b2 is not None:

                warm_b = b
            else:
                warm_b = b
        else:
            warm_b = b

            y_new = y.copy()
        else:

            a_new = g.copy()

            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        sc_y = fast_conv_self(y).max()

def vegas_optimize(a0, time_limit, verbose=False):
    """Vegas multiplicative weight updates."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    
    best_a = a.copy()
    best_max = fast_conv_self(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (len(a) * 2 - 1).bit_length()
    
    # Adaptive eta
    eta = 0.05
    best_ever = best_max
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        M = conv.max()
        if M < 1e-15:
            break
        
        # Pressure: how much each element contributes to the peak
        w = np.maximum(conv - M * 0.85, 0.0)
        ww = w / (w.sum() + 1e-15)
        
        W = rfft(ww, fft_size)
        A = rfft(a, fft_size)
        
        pressure = irfft(np.conj(A) * W, fft_size)[:n] + irfft(A * W, fft_size)[:n]
        
        # Multiplicative update
        a = a * np.exp(-eta * pressure / max(pressure.max(), 1e-15))
        a = np.maximum(a, 1e-10)
        a = a * norm / a.sum()
        
        sc = fast_conv_self(a).max()
        if sc < best_max - 1e-12:
            best_max = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Adapt eta
        if stall > 200:
            eta *= 0.7
            stall = 0
        elif stall > 500:
            # Perturb
            a = a * (1.0 + 0.1 * np.random.standard_normal(n))
            a = np.maximum(a, 0.0)
            a = a * norm / a.sum()
            stall = 0
    
    return best_a, best_max

def pair_cd_optimize(a0, time_limit, rng, verbose=False):
    """Pair coordinate descent with gradient-informed moves."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_max = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue
            
            # Find high-pressure lags
            peak_lags = np.where(conv >= cur_max * 0.8)[0]
            
            # Candidate j's
            j_candidates = set()
            for _ in range(6):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:8]:
                j_candidates.add(int((kl - i + n) % n))
            j_candidates.discard(i)
            
            best_v = cur_max
            best_delta = 0.0
            best_j = -1
            
            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]
                
                # Compute delta conv for moving mass from i to j
                # new_conv[k] = conv[k] - ai*a[k-i] adjusted + aj*a[k-j] adjusted
                # This is approximate; use the gradient direction
                g = np.zeros(size)
                
                # Contribution of a[i]: for each k, a[i] appears as a[i]*a[k-i] and a[k-i]*a[i]
                for k in range(max(0, i), min(size, i + n)):
                    j_idx = k - i
                    if 0 <= j_idx < n:
                        g[k] += a[j_idx]
                for k in range(max(0, i - n + 1), min(size, i + 1)):
                    j_idx = k - i
                    if 0 <= j_idx < n:
                        # Already counted if k-i >= 0, so this handles k < i case
                        if k - i < 0:
                            g[k] += a[j_idx]
                
                # Hmm this is getting complicated. Let me use FFT instead.
                # grad_i = d(max)/da[i] approximated by 2*(conv(w, a) + conv(a, w)) at i
                # But for pair moves, we need the actual change.
                
                # Simpler: compute the change in conv for moving delta from i to j
                # conv_new[k] = conv[k] + delta * (a[k-j] - a[k-i]) [approximately, for the cross terms]
                # Plus second order: delta^2 at k=2j and k=2i
                
                # For now, just try a few delta values
                d_max = min(ai, norm * 0.3)
                delta = d_max * 0.5
                
                # Quick check: apply the move and evaluate
                a[i] -= delta
                a[j] += delta
                new_max = fast_conv_self(a).max()
                
                if new_max < best_v:
                    best_v = new_max
                    best_delta = delta
                    best_j = j
                else:
                    a[i] += delta
                    a[j] -= delta
            
            # Try half and quarter of best delta
            if best_j >= 0:
                for frac in [0.5, 0.25]:
                    d = best_delta * frac
                    a[i] -= d
                    a[best_j] += d
                    new_max = fast_conv_self(a).max()
                    if new_max < best_v:
                        best_v = new_max
                        best_delta = d
                    else:
                        a[i] += d
                        a[best_j] -= d
            
            if best_j >= 0 and best_delta > 0 and best_v < best_max - 1e-10:
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                new_max = fast_conv_self(a).max()
                if new_max < best_max - 1e-10:
                    best_max = new_max
                    best_a = a.copy()
                    pass_improved = True
                    total_improvements += 1
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                n_perturb = max(10, n // 50)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.2 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                a = a * norm / a.sum()
                if no_improve >= 8:
                    break
        else:
            no_improve = 0
    
    return best_a, best_max

def structured_init(n, rng, mode=0):
    """Generate structured initializations."""
    x = np.linspace(0.0, 1.0, n)
    choice = mode if mode >= 0 else rng.integers(0, 15)
    
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        f1 = rng.integers(1, 5)
        f2 = rng.integers(5, 15)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * f1 * x) + 0.2 * np.cos(2 * np.pi * f2 * x)
    elif choice == 3:
        c1 = rng.uniform(0.2, 0.4)
        c2 = rng.uniform(0.6, 0.8)
        a = np.exp(-40.0 * (x - c1) ** 2) + np.exp(-40.0 * (x - c2) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.5
    elif choice == 6:
        a = np.exp(-3.0 * x) + 0.5 * np.exp(-3.0 * (1-x))
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        n_peaks = rng.integers(3, 8)
        centers = rng.uniform(0.1, 0.9, n_peaks)
        a = np.zeros(n)
        for c in centers:
            width = rng.uniform(0.03, 0.1)
            a += np.exp(-(x - c) ** 2 / (2 * width ** 2))
    elif choice == 9:
        a = np.sinc(x - 0.5) ** 2 + 0.1
    elif choice == 10:
        # Periodic with varying duty cycle
        period = rng.integers(3, 10)
        duty = rng.uniform(0.3, 0.7)
        period_pattern = (np.linspace(0, 1, period) < duty).astype(float)
        a = np.tile(period_pattern, n // period + 1)[:n]
    elif choice == 11:
        # Binary random
        a = (rng.random(n) < rng.uniform(0.2, 0.5)).astype(float)
    elif choice == 12:
        # Geometric decay from both ends
        decay = rng.uniform(0.95, 0.999)
        a = np.maximum(decay ** np.arange(n), decay ** (n - 1 - np.arange(n)))
    elif choice == 13:
        # Sinc-squared window
        a = np.sinc((x - 0.5) * rng.uniform(3, 10)) ** 2

    s = a.sum()

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def sparsify(a, threshold_frac=0.001):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return a
    a_norm = a / s
    threshold = threshold_frac * np.max(a_norm)
    mask = a_norm > threshold
    if np.sum(mask) < n // 4:
        mask = a_norm > threshold * 0.1
    a_s = np.where(mask, a, 0.0)
    s_new = a_s.sum()
    if s_new < 1e-15:
        return a
    return a_s / s_new

def block_momentum_optimize(a0, time_limit, rng, verbose=False):
    """Block coordinate descent: update blocks of consecutive elements together."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    
    best_a = a.copy()
    best_max = fast_conv_self(a).max()
    t0 = time.monotonic()
    
    block_size = max(5, n // 100)
    n_blocks = n // block_size
    
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        for _ in range(n_blocks):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Pick random block
            b_start = int(rng.integers(0, max(1, n - block_size)))
            b_end = b_start + block_size
            
            # Try perturbing this block
            orig_block = a[b_start:b_end].copy()
            block_sum = orig_block.sum()
            
            if block_sum < 1e-15:
                continue
            
            # Try random redistribution
            for trial in range(3):
                if trial == 0:
                    # Smooth
                    new_block = orig_block + 0.1 * np.diff(np.concatenate([[orig_block[0]], orig_block, [orig_block[-1]]]))
                elif trial == 1:
                    # Peak in middle
                    center = len(orig_block) // 2
                    new_block = orig_block * (1.0 + 0.3 * np.exp(-np.arange(len(orig_block))**2 / (2*center**2)))
                else:
                    # Random
                    new_block = np.abs(rng.standard_normal(block_size)) + 0.1
                    new_block *= block_sum / new_block.sum()
                
                new_block = np.maximum(new_block, 0.0)
                new_block = new_block * block_sum / max(new_block.sum(), 1e-15)
                
                a[b_start:b_end] = new_block
                new_max = fast_conv_self(a).max()
                
                if new_max < best_max - 1e-12:
                    best_max = new_max
                    best_a = a.copy()
                    improved = True
                else:
                    a[b_start:b_end] = orig_block
        
        if not improved:
            no_improve += 1
            if no_improve >= 10:
                break
        else:
            no_improve = 0
    
    return best_a, best_max

                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        sc_y = fast_conv_self(y).max()
        if sc_y < best_max - 1e-12:
            best_max = sc_y
            best_a = y.copy()
        
        sc_a = fast_conv_self(a).max()
        if sc_a < best_max - 1e-12:
            best_max = sc_a
            best_a = a.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): best={best_max:.8f}, M={M:.6f}, lp={n_lp}")
    
    return best_a, best_max

def nesterov_lse_optimize(a0, time_limit, verbose=False):
    """Nesterov-accelerated LSE gradient descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    
    best_a = a.copy()
    best_max = fast_conv_self(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    
    y = a.copy()
    t_nest = 1.0
    T = 50.0
    T_min = 0.05
    patience = 0
    
    while time.monotonic() - t0 < time_limit:
        t_nest += 1.0
        momentum = (t_nest - 1.0) / (t_nest + 1.8)
        y_trial = y + momentum * (y - a)
        
        conv = fast_conv_self(y_trial)
        max_c = conv.max()
        if max_c < 1e-15:
            break
        
        shifted = (conv - max_c) / T
        shifted = np.maximum(shifted, -100.0)
        exp_c = np.exp(shifted)
        w = exp_c / exp_c.sum()
        
        W = rfft(w, fft_size)
        Y = rfft(y_trial, fft_size)
        
        g1 = irfft(np.conj(Y) * W, fft_size)[:n]
        g2 = irfft(Y * W, fft_size)[:n]
        grad = 2.0 * (g1 + g2)
        
        lr = 0.3 / (np.sqrt(n) * np.sqrt(t_nest + 1.0))
        a_new = y_trial - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        y = y_trial
        a = a_new
        
        sc = fast_conv_self(a).max()
        if sc < best_max - 1e-12:
            best_max = sc
            best_a = a.copy()
            patience = 0
            # Cool down
            T = max(T * 0.95, T_min)
        else:
            patience += 1
        
        if patience > 300:
            T = max(T * 0.7, T_min)
            patience = 0
        
        if patience > 1000:
            # Hard restart with perturbation
            a = best_a * (1.0 + 0.1 * np.random.standard_normal(n))
            a = np.maximum(a, 0.0)
            a = a * norm / a.sum()
            y = a.copy()
            t_nest = 1.0
            T = 20.0
            patience = 0
    
    return best_a, best_max

def adam_lse_optimize(a0, time_limit, verbose=False):
    """Adam LSE with multi-stage temperature."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    
    best_a = a.copy()
    best_max = fast_conv_self(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    t_adam = 0
    
    stages = [(80.0, 5.0), (5.0, 1.0), (1.0, 0.1)]
    
    for si, (T_start, T_end) in enumerate(stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        
        n_temps = 25
        temps = np.geomspace(T_start, T_end, n_temps)
        
        for T_val in temps:
            if time.monotonic() - t0 > time_limit:
                break
            remaining = time_limit - (time.monotonic() - t0)
            if remaining < 5:
                break
            
            n_steps = max(15, min(80, int(remaining * 4)))
            lr = 0.4 / np.sqrt(n)
            
            for it in range(n_steps):
                if time.monotonic() - t0 > time_limit:
                    break
                t_adam += 1
                
                conv = fast_conv_self(a)
                max_c = conv.max()
                if max_c < 1e-15:
                    break
                
                shifted = (conv - max_c) / T_val
                shifted = np.maximum(shifted, -100.0)
                exp_c = np.exp(shifted)
                w = exp_c / exp_c.sum()
                
                W = rfft(w, fft_size)
                A = rfft(a, fft_size)
                
                g1 = irfft(np.conj(A) * W, fft_size)[:n]
                g2 = irfft(A * W, fft_size)[:n]
                grad = 2.0 * (g1 + g2)
                
                m = beta1 * m + (1.0 - beta1) * grad
                v = beta2 * v + (1.0 - beta2) * grad * grad
                m_hat = m / (1.0 - beta1 ** t_adam)
                v_hat = v / (1.0 - beta2 ** t_adam)
                
                a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new < 1e-15:
                    a = best_a.copy()
                else:
                    a = a * norm / s_new
                
                sc = fast_conv_self(a).max()
                if sc < best_max - 1e-12:
                    best_max = sc
                    best_a = a.copy()
    
    return best_a, best_max
