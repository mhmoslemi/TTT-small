def fast_autocorr(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

def build_T_matrix(a, active_idx):
    """Build constraint matrix rows for given active indices."""
    n = len(a)
    m = len(active_idx)
    T = np.zeros((m, n))
    for row, k in enumerate(active_idx):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                T[row, j] = a[idx]
    return T

def solve_lp_max_sum(a, M, active_idx):
    """Solve: max sum(b) s.t. T@b <= M, b >= 0"""
    n = len(a)
    T = build_T_matrix(a, active_idx)
    c_obj = -np.ones(n)  # minimize -sum(b)
    A_ub = T
    b_ub = np.full(len(active_idx), M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status != 0:
        return None
    val = np.maximum(res.x, 0.0)
    if val.sum() < 1e-15:
        return None
    return val

def solve_lp_min_max(a, active_idx, target_sum):
    """Solve: min t s.t. T@b <= t, sum(b) = target_sum, b >= 0"""
    n = len(a)
    T = build_T_matrix(a, active_idx)
    m = len(active_idx)
    # Variables: b[0..n-1], t[n]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    # Constraints: T@b - t <= 0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = T
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    # Equality: sum(b) = target_sum
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0:
        return None, None
    val = np.maximum(res.x[:n], 0.0)
    t_val = float(res.x[n])
    if val.sum() < 1e-15:
        return None, None
    return val, t_val

def rng_choice(arr, k):
    """Simple random choice without numpy rng to avoid closure issues."""
    if k >= len(arr):
        return arr.copy()
    indices = np.random.choice(len(arr), size=k, replace=False)
    return arr[indices]

def pair_cd(a0, time_limit, verbose=True):
    """Pair coordinate descent with exact 1D optimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0
    size = 2 * n - 1

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = np.random.permutation(n)

        for idx in range(0, n, 2):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx % n]
            j = perm[(idx + 1) % n]
            if i == j:
                continue
            ai = a[i]
            aj = a[j]
            if ai < 1e-14 and aj < 1e-14:
                continue

            # Try moving mass: a[i] -> a[i]-d, a[j] -> a[j]+d
            # and reverse
            best_delta = 0.0
            best_val = conv.max()

            for sign in [1, -1]:
                max_d = (aj if sign == 1 else ai) * 0.99
                if max_d <= 0:
                    continue
                # Sample several delta values
                for frac in [0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 0.85, 1.0]:
                    d = max_d * frac
                    if d <= 0 or d > max_d:
                        continue
                    a_try = a.copy()
                    if sign == 1:
                        a_try[i] = ai - d
                        a_try[j] = aj + d
                    else:
                        a_try[i] = ai + d
                        a_try[j] = aj - d
                    if a_try[i] < 0 or a_try[j] < 0:
                        continue
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        continue
                    a_try = a_try * norm / s_try
                    cv = fast_autocorr(a_try)
                    val = cv.max()
                    if val < best_val - 1e-14:
                        best_val = val
                        best_delta = sign * d

            if abs(best_delta) > 1e-15:
                a_try = a.copy()
                if best_delta > 0:
                    a_try[i] -= best_delta
                    a_try[j] += best_delta
                else:
                    a_try[i] += abs(best_delta)
                    a_try[j] -= abs(best_delta)
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try * norm / s_try
                    cv = fast_autocorr(a_try)
                    if cv.max() < best_score - 1e-14:
                        a = a_try
                        conv = cv
                        best_score = cv.max()
                        best_a = a.copy()
                        pass_improved = True

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): score={best_score:.10f}")

    return best_a, best_score

def subgradient_descent(a0, time_limit, verbose=True, seed=123):

    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    step = norm / (n * best_score)  # initial step size
    n_iter = 0

    while time.monotonic() - t0 < time_limit:
        n_iter += 1
        conv = fast_autocorr(a)
        M = conv.max()

        # Subgradient: average over top peaks
        top_frac = 0.05
        threshold = M * (1.0 - top_frac)
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv)])

        # Compute subgradient
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for j in range(lo, hi):
                g[j] += a[k - j]
        g /= len(active)

        # Adaptive step size (Polyak-like)
        # We know the lower bound on the optimal score is related to 1/n
        # Use a simple decreasing step
        step = step * 0.9995 + 0.0005 * (norm / (n * (M + 1e-10)))

        # Update with projection onto simplex (sum = norm, a >= 0)
        a_new = a - step * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        score_new = fast_autocorr(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()

        a = a_new

        if verbose and n_iter % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG t={n_iter} ({elapsed:.0f}s): score={best_score:.10f}")

    return best_a, best_score

def structured_init(n, style=0):
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif style == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif style == 2:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(8 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif style == 3:
        a = np.exp(-6.0 * (x - 0.25) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.75) ** 2) + 0.3 * np.exp(-6.0 * (x - 0.5) ** 2)
    elif style == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif style == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif style == 6:
        a = 1.0 + 0.2 * np.cos(4 * np.pi * x) + 0.1 * np.cos(8 * np.pi * x) + 0.05 * np.cos(12 * np.pi * x)
    elif style == 7:
        # Edge-enhanced
        a = 0.2 + 0.8 * (x ** 0.5 + (1 - x) ** 0.5) / np.sqrt(2)
    else:
        a = np.sin(np.pi * x) ** 0.5 * (1.0 + 0.1 * np.cos(8 * np.pi * x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss

def peak_equalize(a0, time_limit, verbose=False):
    """Iteratively equalize the top peaks using LP."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    for iteration in range(5000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(a)
        M = conv.max()
        size = 2 * n - 1

        # Find top peaks to equalize
        top_k = 10
        idx_sorted = np.argsort(conv)[::-1]
        top_indices = idx_sorted[:top_k]

        # Build LP: maximize sum(b) s.t. 
        #   conv(a, b)[k] <= M for k in top_indices
        #   conv(a, b)[k] <= conv(a,b)[k*] for all other active k (to reduce max)
        # Simplified: just constrain the top peaks to be <= M
        T = build_T_matrix(a, top_indices)
        c_obj = -np.ones(n)
        res = linprog(c_obj, A_ub=T, b_ub=np.full(top_k, M),
                      bounds=[(0, None)] * n, method='highs')
        if res.status != 0:
            stall += 1
            if stall > 10:
                break
            continue

        b = np.maximum(res.x, 0.0)
        b_sum = b.sum()
        if b_sum < 1e-15:
            stall += 1
            continue
        b = b * norm / b_sum

        # Blend
        t_blend = 0.3 + 0.1 * np.sin(iteration * 0.1)  # oscillating blend
        a_new = (1.0 - t_blend) * a + t_blend * b
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new * norm / s_new
            score = fast_autocorr(a_new).max()
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                stall = 0
            a = a_new

        if verbose and iteration % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PE it={iteration} ({elapsed:.0f}s): score={best_score:.10f}")

    return best_a, best_score

def propose_candidate():
    """Main search function combining multiple optimization strategies."""
    t_start = time.monotonic()
    time_budget = 1740.0  # Reserve 60s buffer

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent / parent.sum()
                best_score = p_score
            else:
                best_seq = parent / parent.sum()
                best_score = evaluate_local(best_seq)
            print(f"Parent: n={len(parent)}, score={best_score:.8f}")
    except Exception as e:
        print(f"Parent load error: {e}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            old = best_score
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        elif a is not None and best_seq is None:
            best_seq = a.copy()
            best_score = sc if np.isfinite(sc) else evaluate_local(a)

    # If no parent, create initial
    if best_seq is None:
        n_init = 1600
        for sty in range(8):
            cand = structured_init(n_init, sty)
            sc = evaluate_local(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand
        print(f"Best init: n={len(best_seq)}, score={best_score:.8f}")

    n_base = len(best_seq)

    # === Phase 1: LP Nesterov (40% time) ===
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget {remaining*0.40:.0f}s) ===")
    t1 = remaining * 0.40
    if t1 > 60:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t1, verbose=True, seed=0)
        try_update(a_out, f"lp1_n{n_base}")

    # === Phase 2: Exact CD (15% time) ===
    remaining = time_budget - (time.monotonic() - t_start)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: Exact CD n={n_cur} (budget {remaining*0.15:.0f}s) ===")
    t2 = remaining * 0.15
    if t2 > 40:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = exact_coord_descent(a_norm, t2, verbose=True)
        try_update(a_out, f"cd1_n{n_cur}")

    # === Phase 3: Peak Equalization + LP (15% time) ===
    remaining = time_budget - (time.monotonic() - t_start)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: Peak Equalize n={n_cur} (budget {remaining*0.15:.0f}s) ===")
    t3 = remaining * 0.15
    if t3 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_equalize(a_norm, t3, verbose=True)
        try_update(a_out, f"pe1_n{n_cur}")

    # === Phase 4: Pair CD (10% time) ===
    remaining = time_budget - (time.monotonic() - t_start)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: Pair CD n={n_cur} (budget {remaining*0.10:.0f}s) ===")
    t4 = remaining * 0.10
    if t4 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = pair_cd(a_norm, t4, verbose=True)
        try_update(a_out, f"pcd1_n{n_cur}")

    # === Phase 5: Subgradient (10% time) ===
    remaining = time_budget - (time.monotonic() - t_start)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: Subgradient n={n_cur} (budget {remaining*0.10:.0f}s) ===")
    t5 = remaining * 0.10
    if t5 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = subgradient_descent(a_norm, t5, verbose=False, seed=42)
        try_update(a_out, f"sg1_n{n_cur}")

    # === Phase 6: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_start)
    if remaining > 120:
        print(f"\n=== Phase 6: Length exploration ===")
        # Try slightly longer
        n_long = len(best_seq) + 200
        t6a = min(remaining * 0.15, remaining - 60)
        if t6a > 30:
            print(f"  Trying n={n_long}...")
            init_long = interpolate_seq(best_seq, n_long)
            a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
            a_out, _ = nesterov_lp(a_norm, t6a, verbose=False, seed=77)
            try_update(a_out, f"lp_long_n{n_long}")

        remaining = time_budget - (time.monotonic() - t_start)
        if remaining > 60:
            # Try slightly shorter
            n_short = max(800, len(best_seq) - 200)
            t6b = min(remaining * 0.10, remaining - 40)
            if t6b > 20:
                print(f"  Trying n={n_short}...")
                init_short = interpolate_seq(best_seq, n_short)
                a_norm = init_short * np.sqrt(2.0 * n_short) / init_short.sum()
                a_out, _ = nesterov_lp(a_norm, t6b, verbose=False, seed=111)
                try_update(a_out, f"lp_short_n{n_short}")

    # === Phase 7: Final LP on best (remaining time * 0.35) ===
    remaining = time_budget - (time.monotonic() - t_start)
    n_best = len(best_seq)
    if remaining > 80:
        print(f"\n=== Phase 7: Final LP n={n_best} ===")
        t7 = remaining * 0.35
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t7, verbose=True, seed=999)
        try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 8: Final exact CD polish ===
    remaining = time_budget - (time.monotonic() - t_start)
    n_best = len(best_seq)
    if remaining > 40:
        print(f"\n=== Phase 8: Final CD n={n_best} ===")
        t8 = min(remaining * 0.40, remaining - 20)
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = exact_coord_descent(a_norm, t8, verbose=True)
        try_update(a_out, f"cd_final_n{n_best}")

    # === Phase 9: Final pair CD ===
    remaining = time_budget - (time.monotonic() - t_start)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 9: Final PCD n={n_best} ===")
        t9 = min(remaining * 0.50, remaining - 15)
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_cd(a_norm, t9, verbose=False)
        try_update(a_out, f"pcd_final_n{n_best}")

    # Final verification
    if best_seq is not None:
        final_score = evaluate_safe(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_start:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
