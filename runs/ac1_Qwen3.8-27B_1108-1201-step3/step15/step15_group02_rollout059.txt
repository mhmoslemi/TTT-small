        alpha = 2.0 / (t + 2.0)
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new

        sc = fast_conv_self(a).max()
        if sc < best_max_b:
            best_max_b = sc
            best_score = compute_score_fast(a)
            best_a = a.copy()

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def pair_cd(a0, time_limit, rng, verbose=True):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a / s
    best_a = a.copy()
    best_score = compute_score_fast(a)
    best_max_b = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < 1e-10:
                continue

            peak_lags = np.where(conv >= cur_max * 0.8)[0]
            j_candidates = set()
            for _ in range(6):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:8]:
                j1 = int((kl - i) % n)
                j2 = int((kl + i) % n)
                j_candidates.add(j1)
                j_candidates.add(j2)
            j_candidates.discard(i)

            best_delta = 0.0
            best_v = cur_max
            best_j = -1

            for j in list(j_candidates)[:12]:
                if j == i:
                    continue
                aj = a[j]

                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo1:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                d_max = min(ai, 0.5)
                if d_max < 1e-10:
                    continue
                n_samples = 20
                deltas = np.linspace(0.02, d_max, n_samples)

                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0 and best_v < cur_max - 1e-12:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_max_b - 1e-12:
                        best_max_b = new_max
                        best_score = compute_score_fast(a)
                        best_a = a.copy()
                        pass_improved = True
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                n_perturb = max(3, n // 100)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.1 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
            if no_improve >= 8:
                break
        else:
            no_improve = 0

        if verbose and (t // 100 > 0):
            elapsed = time.monotonic() - t0
            if elapsed % 10 < 0.1:
                print(f"    PCD ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def frangi_es(a0, time_limit, rng):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a / s
    best_a = a.copy()
    best_score = compute_score_fast(a)
    best_max_b = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    T_current = 0.1
    T_min = 0.005
    T_decay = 0.97
    step_size = 0.3 / np.sqrt(n)

    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        M = conv.max()

        w = softmax(conv, T_current)
        W = rfft(w, size)
        A = rfft(a, size)
        grad = 2.0 * (irfft(np.conj(A) * W, size)[:n] + irfft(A * W, size)[:n])
        grad = grad / (np.abs(grad).max() + 1e-15)

        a_new = a - step_size * grad
        noise = rng.standard_normal(n) * (step_size * 0.1)
        a_new = a_new + noise
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        sc_new = fast_conv_self(a_new).max()
        if sc_new < best_max_b:
            best_max_b = sc_new
            best_score = compute_score_fast(a_new)
            best_a = a_new.copy()
            a = a_new
        elif np.random.random() < np.exp(-(sc_new - M) / T_current):
            a = a_new

        if it % 50 == 0:
            T_current = max(T_min, T_current * T_decay)
            step_size *= (1.0 + 0.001)

    return best_a, best_score

def structured_init(n, rng):
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        f1 = rng.integers(1, 6)
        f2 = rng.integers(6, 20)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * f1 * x) + 0.2 * np.cos(2 * np.pi * f2 * x)
    elif choice == 3:
        c1 = rng.uniform(0.2, 0.5)
        c2 = rng.uniform(0.5, 0.8)
        a = np.exp(-30.0 * (x - c1) ** 2) + np.exp(-30.0 * (x - c2) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.5
    elif choice == 6:
        a = np.exp(-5.0 * x) + 0.4
    elif choice == 7:
        n_peaks = rng.integers(3, 10)
        centers = rng.uniform(0.05, 0.95, n_peaks)
        a = np.zeros(n)
        for c in centers:
            width = rng.uniform(0.02, 0.1)
            a += np.exp(-(x - c) ** 2 / (2 * width ** 2))
    elif choice == 8:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * x) + 0.2 * np.sin(2 * np.pi * 7 * x)
    else:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a = a + 0.003 * np.abs(rng.standard_normal(n))
    a = a / a.sum()
    return a

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1650.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            old_best = best_score
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (was {old_best:.6f}, n={len(a)}, t={elapsed:.0f}s)")

    # Phase 1: Nesterov Mirror Descent (fast initial progress)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.20, 280.0)
    print(f"\n=== Phase 1: Nesterov Mirror Descent n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = nesterov_mirror_descent(best_seq, t1, verbose=True)
        check_and_update(a_out, f"nmr1_n{n_base}")

    # Phase 2: Kaczmarz-style coordinate mirror descent
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 2: Kaczmarz Mirror n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = kaczmarz_mirror(best_seq, t2, verbose=True)
        check_and_update(a_out, f"kacz_n{len(best_seq)}")

    # Phase 3: LP-based refinement (main precision tool)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 3: LP Refinement n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = lp_refine(best_seq, t3, solver, verbose=True)
        check_and_update(a_out, f"lp1_n{len(best_seq)}")

    # Phase 4: Another round of mirror descent from improved point
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 4: Mirror Descent Round 2 n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = nesterov_mirror_descent(best_seq, t4, verbose=False)
        check_and_update(a_out, f"nmr2_n{len(best_seq)}")

    # Phase 5: Frangi-ES (simulated annealing with gradient)
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 5: Frangi-ES n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = frangi_es(best_seq, t5, rng)
        check_and_update(a_out, f"frangi_n{len(best_seq)}")

    # Phase 6: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 6: Pair CD n={len(best_seq)} (budget={t6:.0f}s) ===")
    if t6 > 20:
        a_out, _ = pair_cd(best_seq, t6, rng, verbose=True)
        check_and_update(a_out, f"pcd1_n{len(best_seq)}")

    # Phase 7: LP Round 2 from best
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 7: LP Round 2 n={len(best_seq)} (budget={t7:.0f}s) ===")
    if t7 > 20:
        a_out, _ = lp_refine(best_seq, t7, solver, verbose=True)
        check_and_update(a_out, f"lp2_n{len(best_seq)}")

    # Phase 8: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        lengths_to_try = [n_base + 200, n_base - 200, n_base + 400, n_base - 400]
        for n_try in lengths_to_try:
            if remaining < 40:
                break
            if n_try < 100 or n_try > 4000:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            t_try = min(remaining * 0.30, 50.0)
            print(f"\n=== Multi-length: n={n_try} (budget={t_try:.0f}s) ===")
            init_try = interpolate_seq(best_seq, n_try)
            a_out, _ = nesterov_mirror_descent(init_try, min(t_try, 25.0), verbose=False)
            check_and_update(a_out, f"ml_nmr_n{n_try}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 25 and best_seq is not None:
                a_out, _ = pair_cd(a_out, min(remaining * 0.4, 20.0), rng, verbose=False)
                check_and_update(a_out, f"ml_pcd_n{n_try}")

    # Final: one more LP pass on the best sequence
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30 and best_seq is not None:
        print(f"\n=== Final LP polish (remaining={remaining:.0f}s) ===")
        a_out, _ = lp_refine(best_seq, min(remaining * 0.7, 60.0), solver, verbose=False)
        check_and_update(a_out, "final_lp")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
