def build_active_constraints(a, active_idx):
    n = len(a)
    m = len(active_idx)
    T = np.zeros((m, n))
    for row, k in enumerate(active_idx):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                T[row, j] = a[idx]
    return T

def lp_oracle(a, M, solver, active_idx=None):
    n = len(a)
    size = 2 * n - 1
    b = cp.Variable(n, nonneg=True)
    if active_idx is None or len(active_idx) >= size:
        T = build_active_constraints(a, np.arange(size))
    else:
        T = build_active_constraints(a, active_idx)
    cons = [T @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if val.sum() < 1e-15:
        return None
    return val

def soft_minimax_optimize(a0, time_limit, beta_start=30, beta_end=300, 
                           lp_every=500, solver=None, verbose=True, seed=0):
    """Soft minimax with Nesterov acceleration and periodic LP corrections."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0  # Normalize to sum=1
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()  # Nesterov momentum
    a_prev = a.copy()
    t0 = time.monotonic()
    eta = 1.0 / np.sqrt(n)
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive beta
        progress = min(t / 5000.0, 1.0)
        beta = beta_start + (beta_end - beta_start) * progress
        
        # Compute autocorrelation
        c = fast_autocorr(y)
        c_max = c.max()
        
        # Softmax weights (numerically stable)
        shifted = beta * (c - c_max)
        exp_c = np.exp(shifted)
        w = exp_c / exp_c.sum()
        
        # Gradient
        g = soft_gradient(y, w)
        
        # Nesterov step
        if t == 1:
            y_new = simplex_proj(y - eta * g)
            a_new = y_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            y_new = a_new + momentum * (a_new - a_prev)
            y_new = simplex_proj(y_new)  # Ensure feasibility
            
            # Gradient step from y
            c_y = fast_autocorr(y_new)
            c_y_max = c_y.max()
            shifted_y = beta * (c_y - c_y_max)
            exp_y = np.exp(shifted_y)
            w_y = exp_y / exp_y.sum()
            g_y = soft_gradient(y_new, w_y)
            
            # Adaptive step size
            g_norm = np.linalg.norm(g_y)
            if g_norm > 1e-12:
                eta_t = 2.0 / (np.sqrt(n) * (t + 1.0))
            else:
                eta_t = eta
            
            a_new = simplex_proj(y_new - eta_t * g_y)
        
        a_prev = y  # Save for momentum
        y = a_new
        
        # Track best
        score = fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        # Periodic LP correction
        if solver is not None and t % lp_every == 0 and t > lp_every:
            M_curr = c_max
            active_idx = np.where(c >= M_curr * 0.7)[0]
            if len(active_idx) > 200:
                active_idx = np.where(c >= M_curr * 0.9)[0]
            b_lp = lp_oracle(a_new, M_curr, solver, active_idx)
            if b_lp is not None:
                b_norm = b_lp / b_lp.sum()
                # Blend
                alpha = 0.3
                a_blend = (1 - alpha) * a_new + alpha * b_norm
                a_blend = simplex_proj(a_blend)
                score_blend = fast_autocorr(a_blend).max()
                if score_blend < best_score:
                    best_score = score_blend
                    best_a = a_blend.copy()
                    a_new = a_blend
                    y = a_blend
                    a_prev = a_blend

            elapsed = time.monotonic() - t0
            print(f"  SM t={t} ({elapsed:.0f}s): best={best_score:.10f}, beta={beta:.0f}")
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss
