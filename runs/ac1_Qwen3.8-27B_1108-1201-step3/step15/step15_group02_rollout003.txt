def fast_conv_self(a):

def lse_adam_optimize(a0, time_limit, verbose=True):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_adam = 0

    n_stages = 25
    temps = np.geomspace(60.0, 0.03, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        n_steps = max(20, min(300, int(remaining * 8)))
        lr = 2.0 / np.sqrt(n)

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            t_adam += 1

            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            Z = exp_c.sum()
            if Z < 1e-30:
                w = np.zeros(size)
                w[np.argmax(conv)] = 1.0
            else:
                w = exp_c / Z

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)

            ca = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (ca + aw)
            grad -= grad.mean()

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_adam)
            v_hat = v / (1.0 - beta2 ** t_adam)

            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose and (si % 5 == 0 or si == n_stages - 1):
            print(f"    LSE-Adam stage {si} T={T_val:.3f}: best={best_score:.8f}")

    return best_a, best_score

def lp_best_response(a, M, solver, constraints_list):
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)

    if constraints_list is not None and 0 < len(constraints_list) < size:
        T = np.zeros((len(constraints_list), n))
        for idx, k in enumerate(constraints_list):
            row_start = max(0, k - n + 1)
            row_end = min(n, k + 1)
            for i in range(row_start, row_end):
                T[idx, i] = a[k - i]
        constraints = [T @ b_var <= M]
    else:
        T = np.zeros((size, n))
        for k in range(n):
            row_start = max(0, k - n + 1)
            row_end = min(n, k + 1)
            for i in range(row_start, row_end):
                T[k, i] = a[k - i]
        for k in range(n, size):
            row_start = max(0, k - n + 1)
            row_end = min(n, k + 1)
            for i in range(row_start, row_end):
                T[k, i] = a[k - i]
        constraints = [T @ b_var <= M]

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)

def lp_solve_with_cg(a, M, solver, max_iters=30):
    n = len(a)
    size = 2 * n - 1
    conv_y = fast_conv_self(a)

    top_k = min(200, size)
    active = np.argsort(conv_y)[-top_k:]

    for it in range(max_iters):
        b = lp_best_response(a, M, solver, active.tolist())
        if b is None or b.sum() < 1e-15:
            b = lp_best_response(a, M, solver, None)
            if b is None:
                return None
            return b

        cross = fast_conv_cross(a, b)
        violated = np.where(cross > M * 1.0001)[0]
        if len(violated) == 0:
            return b

        new_active = np.unique(np.concatenate([active, violated]))
        if len(new_active) > 1000:
            active = np.argsort(cross)[-1000:]
            b = lp_best_response(a, M, solver, active.tolist())
            if b is not None:
                return b
            b = lp_best_response(a, M, solver, None)
            return b
        else:
            active = new_active

    b = lp_best_response(a, M, solver, active.tolist())
    if b is None:
        b = lp_best_response(a, M, solver, None)
    return b

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    def check_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            old_best = best_score
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (was {old_best:.6f}, n={len(a)}, t={elapsed:.0f}s)")

    # Phase 1: LSE-Adam (fast initial)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 1: LSE-Adam n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_out, _ = lse_adam_optimize(best_seq, t1, verbose=True)
        check_update(a_out, f"lse1_n{n_base}")

    # Phase 2: Pressure equalization (new method)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.08, 120.0)
    print(f"\n=== Phase 2: Pressure-eq n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = pressure_equalize(best_seq, t2, verbose=False)
        check_update(a_out, f"pres_n{len(best_seq)}")

    # Phase 3: LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.30, 500.0)
    print(f"\n=== Phase 3: LP Nesterov n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t3, solver, verbose=True)
        check_update(a_out, f"lp1_n{len(best_seq)}")

    # Phase 4: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 4: Pair CD n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = pair_cd_optimize(best_seq, t4, rng, verbose=True)
        check_update(a_out, f"pcd1_n{len(best_seq)}")

    # Phase 5: LSE-Adam round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 5: LSE-Adam r2 n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_out, _ = lse_adam_optimize(best_seq, t5, verbose=False)
        check_update(a_out, f"lse2_n{len(best_seq)}")

    # Phase 6: LP round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 6: LP r2 n={len(best_seq)} (budget={t6:.0f}s) ===")
    if t6 > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t6, solver, verbose=False)
        check_update(a_out, f"lp2_n{len(best_seq)}")

    # Phase 7: Multi-n exploration
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        print(f"\n=== Phase 7: Multi-n exploration (remaining={remaining:.0f}s) ===")
        n_options = sorted(set([n_base - 200, n_base + 200, n_base + 400, n_base + 600, n_base + 1000]))
        n_options = [x for x in n_options if 400 <= x <= 5000]
        for n_try in n_options[:4]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 50:
                break
            t_try = min(remaining * 0.3, 50.0)
            print(f"  Trying n={n_try} (budget={t_try:.0f}s)")
            init_try = interpolate_seq(best_seq, n_try)
            a_out, _ = lse_adam_optimize(init_try, min(t_try * 0.5, 25.0), verbose=False)
            check_update(a_out, f"lse_ml_n{n_try}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20:
                a_out2, _ = pair_cd_optimize(a_out, min(remaining * 0.4, 20.0), rng, verbose=False)
                check_update(a_out2, f"pcd_ml_n{n_try}")

    # Phase 8: Final PCD polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 8: Final PCD (remaining={remaining:.0f}s) ===")
        t8 = min(remaining * 0.5, 60.0)
        a_out, _ = pair_cd_optimize(best_seq, t8, rng, verbose=True)
        check_update(a_out, "pcd_final")

    # Phase 9: Fresh start diversity
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        print(f"\n=== Phase 9: Fresh starts (remaining={remaining:.0f}s) ===")
        for attempt in range(4):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 15:
                break
            n_fresh = n_base + int(rng.integers(-200, 500))
            if n_fresh < 200:
                n_fresh = 800
            if n_fresh > 4000:
                n_fresh = 3000
            a_fresh = structured_init(n_fresh, rng)
            t_fresh = min(remaining * 0.25, 15.0)
            a_out, _ = lse_adam_optimize(a_fresh, t_fresh, verbose=False)
            check_update(a_out, f"fresh{attempt}_n{n_fresh}")

    # Final
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
