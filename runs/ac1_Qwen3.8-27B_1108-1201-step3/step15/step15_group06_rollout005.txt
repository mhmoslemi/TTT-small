import numpy as np

from scipy.fft import rfft, irfft


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(a, s):
    """Project a onto {x >= 0, sum(x) = s} via water-filling."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    if s <= 0:
        return np.zeros(n)
    sorted_a = np.sort(a)[::-1]
    lo = float(sorted_a[-1]) - 1.0
    hi = float(sorted_a[0]) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        val = np.sum(np.maximum(a - mid, 0.0))
        if val > s:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) * 0.5
    result = np.maximum(a - tau, 0.0)
    diff = s - result.sum()
    if abs(diff) > 1e-12:
        pos = result > 1e-15
        if pos.sum() > 0:
            result[pos] += diff / pos.sum()
            result = np.maximum(result, 0.0)
    return result


def compute_subgrad(a, k, n):
    """Compute subgradient of max autocorrelation at lag k."""
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    idx = np.arange(lo, hi)
    g[idx] = 2.0 * a[k - idx]
    return g


def subgrad_nesterov(a0, time_limit, verbose=True, seed=0, top_k=5, damping=0.4):
    """Subgradient descent with Nesterov momentum for min max-autocorrelation."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()

    else:
        a = a * norm / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    f_lower = 2.0 * n / (2.0 * n - 1.0)

    t0 = time.monotonic()
    a_prev = a.copy()
    stall = 0
    best_iter = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if t > 1:
            momentum = min((t - 1.0) / (t + 2.0), 0.95)
            a_nest = a + momentum * (a - a_prev)
        else:
            a_nest = a.copy()
        a_nest = np.maximum(a_nest, 0.0)
        s_nest = a_nest.sum()
        if s_nest > 1e-15:
            a_nest = a_nest * norm / s_nest
        else:
            a_nest = a.copy()

        # Compute autocorrelation
        c = fast_autocorr(a_nest)
        current_max = float(c.max())

        # Find top-k peaks
        k_all = np.argsort(c)[::-1]
        k_top = k_all[:min(top_k, len(k_all))]

        # Compute averaged subgradient
        g = np.zeros(n)
        for k in k_top:
            g += compute_subgrad(a_nest, int(k), n)
        g /= float(len(k_top))

        # Adaptive step size
        gap = current_max - f_lower
        if gap < 1e-6:

        g_norm_sq = float(np.dot(g, g))

            eta_trust = norm / max(s_nest, 1e-10) * 0.5
            eta = min(eta_polyak, eta_trust)

        # Update
        a_new = a_nest - eta * g
        a_new = project_simplex(a_new, norm)

        a_prev = a.copy()
        a = a_new

        # Track best

            a_prev = a.copy()
            stall = 0

def freq_smooth(a, sigma_freq):
    """Apply Gaussian smoothing in frequency domain (low-pass on phase)."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, fft_size)
    freqs = np.arange(len(A))
    mask = np.exp(-0.5 * (freqs / (sigma_freq * len(A))) ** 2)
    magnitude = np.abs(A)
    phase = np.angle(A)
    A_smooth = magnitude * mask * np.exp(1j * phase)
    a_smooth = irfft(A_smooth, fft_size)[:n]
    return np.maximum(a_smooth, 0.0)

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss

            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_init = 1600
        for sty in range(11):
            cand = structured_init(n_init, rng, sty)
            sc = evaluate_safe(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand
        print(f"Best init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            old_score = best_score
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s, delta={old_score - sc:.6f})")

    n_base = len(best_seq)

    # === Phase 1: Subgradient descent on parent (35% time) ===
    print(f"\n=== Phase 1: Subgradient SGD n={n_base} ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 600.0)
    if t1 > 60:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = subgrad_nesterov(a_norm, t1, verbose=True, seed=0, top_k=5, damping=0.4)
        try_update(a_out, f"sgd1_n{n_base}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 2: CD polish (12% time) ===
    print(f"\n=== Phase 2: CD polish n={len(best_seq)} ===")
    t2 = min(remaining * 0.12, 200.0)
    if t2 > 30:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coord_descent_polish(a_norm, t2, rng, verbose=True)
        try_update(a_out, f"cd1_n{n_cur}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 3: Pair CD polish (8% time) ===
    print(f"\n=== Phase 3: Pair CD n={len(best_seq)} ===")
    t3 = min(remaining * 0.08, 130.0)
    if t3 > 20:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = pair_cd_polish(a_norm, t3, rng, verbose=True)
        try_update(a_out, f"pcd1_n{n_cur}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 4: SGD with different top_k and damping (20% time) ===
    print(f"\n=== Phase 4: SGD variants ===")
    t4_total = min(remaining * 0.20, 350.0)
    configs = [
        (10, 0.3, 100),
        (3, 0.5, 200),
        (8, 0.35, 300),
    ]
    t4_each = t4_total / len(configs)
    for ci, (tk, damp, seed) in enumerate(configs):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        t4 = min(t4_each, remaining - 40)
        if t4 < 20:
            break
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = subgrad_nesterov(a_norm, t4, verbose=False, seed=seed, top_k=tk, damping=damp)
        try_update(a_out, f"sgd_tk{tk}_d{damp}_n{n_cur}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 5: Perturbation restarts (15% time) ===
    print(f"\n=== Phase 5: Perturbation restarts ===")
    t5_total = min(remaining * 0.15, 250.0)
    for r in range(4):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        t5 = min(t5_total / 4.0, remaining - 30)
        if t5 < 15:
            break
        n_cur = len(best_seq)
        scale = 0.02 + 0.015 * r
        noise = 1.0 + scale * rng.standard_normal(n_cur)
        noise = np.maximum(noise, 0.1)
        pert = best_seq * noise
        pert = pert / pert.sum()
        a_norm = pert * np.sqrt(2.0 * n_cur)
        a_out, _ = subgrad_nesterov(a_norm, t5, verbose=False, seed=500 + r * 13, top_k=5, damping=0.4)
        try_update(a_out, f"pert{r}_n{n_cur}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 6: Frequency smoothing escape (5% time) ===
    print(f"\n=== Phase 6: Freq smooth escape ===")
    t6 = min(remaining * 0.05, 80.0)
    if t6 > 15:
        n_cur = len(best_seq)
        a_smooth = freq_smooth(best_seq, 5.0)
        s_sm = a_smooth.sum()
        if s_sm > 1e-10:
            a_smooth = a_smooth * np.sqrt(2.0 * n_cur) / s_sm
            a_out, _ = subgrad_nesterov(a_smooth, t6, verbose=False, seed=777, top_k=10, damping=0.5)
            try_update(a_out, f"freq_n{n_cur}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 7: Try longer length (10% time) ===
    n_long = n_base + 200
    if remaining > 80:
        print(f"\n=== Phase 7: Longer n={n_long} ===")
        t7 = min(remaining * 0.10, 150.0)
        if t7 > 30:
            init_long = interpolate_seq(best_seq, n_long)
            a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
            a_out, _ = subgrad_nesterov(a_norm, t7, verbose=True, seed=42, top_k=5, damping=0.4)
            try_update(a_out, f"long_n{n_long}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 8: Final SGD polish with best params (remaining time * 40%) ===
    n_best = len(best_seq)
    if remaining > 60:
        print(f"\n=== Phase 8: Final SGD n={n_best} ===")
        t8 = min(remaining * 0.40, 200.0)
        if t8 > 20:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = subgrad_nesterov(a_norm, t8, verbose=True, seed=999, top_k=3, damping=0.3)
            try_update(a_out, f"sgd_final_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 9: Final CD polish ===
    n_best = len(best_seq)
    if remaining > 40:
        print(f"\n=== Phase 9: Final CD n={n_best} ===")
        t9 = min(remaining * 0.50, 120.0)
        if t9 > 15:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = coord_descent_polish(a_norm, t9, rng, verbose=True)
            try_update(a_out, f"cd_final_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 10: Final pair CD ===
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 10: Final PCD n={n_best} ===")
        t10 = min(remaining * 0.60, 80.0)
        if t10 > 10:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = pair_cd_polish(a_norm, t10, rng, verbose=True)
            try_update(a_out, f"pcd_final_n{n_best}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
