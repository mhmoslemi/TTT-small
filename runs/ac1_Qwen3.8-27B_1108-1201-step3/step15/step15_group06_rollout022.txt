        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                T[row, j] = a[idx]
    return T


def lp_best_response(a, M, solver, active_idx=None):
    n = len(a)
    size = 2 * n - 1
    b = cp.Variable(n, nonneg=True)
    if active_idx is None or len(active_idx) >= size:
        act = np.arange(size)
    else:
        act = active_idx
    T = build_active_constraints(a, act)
    cons = [T @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if val.sum() < 1e-15:
        return None
    return val


def lp_minmax(a, solver, active_idx=None):
    n = len(a)
    size = 2 * n - 1
    b = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    if active_idx is None or len(active_idx) >= size:
        act = np.arange(size)
    else:
        act = active_idx
    T = build_active_constraints(a, act)
    cons = [T @ b <= t, b >= 0]
    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    t_val = float(t.value) if t.value is not None else float('inf')
    if val.sum() < 1e-15:
        return None, t_val
    return val, t_val


def fista_subgrad(a0, time_limit, solver, verbose=True, seed=0, use_lp=True):
    rng = np.random.default_rng(seed)
    n = len(a0)
    size = 2 * n - 1
    norm = np.sqrt(2.0 * n)

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.full(n, norm / n)
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t_k = 1.0
    lr = 0.5
    active_tol = 0.04
    t0 = time.monotonic()
    stall = 0
    npass = 0

    while time.monotonic() - t0 < time_limit:
        npass += 1

        conv_y = fast_autocorr(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        threshold = M * (1.0 - active_tol)
        active = np.where(conv_y >= threshold)[0]
        if len(active) < 4:
            active = np.where(conv_y >= M * 0.5)[0]
        if len(active) > 0.7 * size:
            active = np.where(conv_y >= M * 0.85)[0]

        g = subgradient_compute(y, active)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = (t_k - 1.0) / t_new

        y_new = a + beta * (a - y)
        y_new = np.maximum(y_new, 0.0)
        s_yn = y_new.sum()
        if s_yn < 1e-15:
            y_new = a.copy()
        else:
            y_new = y_new * norm / s_yn

        a_trial = y_new - lr * g
        a_trial = np.maximum(a_trial, 0.0)
        s_tr = a_trial.sum()
        if s_tr < 1e-15:
            a_trial = y_new.copy()
        else:
            a_trial = a_trial * norm / s_tr

        score_trial = fast_autocorr(a_trial).max()

        bt = 0
        while score_trial > M * 1.02 and bt < 8:
            lr *= 0.5
            a_trial = y_new - lr * g
            a_trial = np.maximum(a_trial, 0.0)
            s_tr = a_trial.sum()
            if s_tr < 1e-15:
                a_trial = y_new.copy()
            else:
                a_trial = a_trial * norm / s_tr
            score_trial = fast_autocorr(a_trial).max()
            bt += 1

        a = a_trial
        y = y_new
        t_k = t_new

        if score_trial < best_score - 1e-13:
            best_score = score_trial
            best_a = a.copy()
            stall = 0
            lr = min(lr * 1.05, 5.0)
        else:
            stall += 1

        if stall > 20:
            scale = 0.03 * (1.0 + np.exp(-npass / 200.0))
            noise = 1.0 + scale * rng.standard_normal(n)
            noise = np.maximum(noise, 0.05)
            a_pert = a * noise
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert * norm / a_pert.sum()
            sc_pert = fast_autocorr(a_pert).max()
            if sc_pert < best_score * 1.1:
                a = a_pert
            y = a.copy()
            t_k = 1.0
            lr = 0.5
            stall = 0

        if use_lp and npass % 80 == 0 and time.monotonic() - t0 < time_limit - 5:
            b_lp = lp_best_response(y, M, solver, active)
            if b_lp is not None:
                b_lp = b_lp * norm / b_lp.sum()
                sc_lp = fast_autocorr(b_lp).max()
                if sc_lp < best_score:
                    best_score = sc_lp
                    best_a = b_lp.copy()
                a = b_lp
                y = a.copy()
                t_k = 1.0
                stall = 0

        if verbose and npass % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA it={npass} ({elapsed:.0f}s): best={best_score:.10f}, lr={lr:.4f}, |S|={len(active)}")

    return best_a, best_score


def nesterov_lp(a0, time_limit, solver, verbose=True, seed=0):
    rng = np.random.default_rng(seed)
    n = len(a0)
    size = 2 * n - 1
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_conv = fast_autocorr(a)
    best_score = best_conv.max()

    y = a.copy()
    t_k = 1.0
    t0 = time.monotonic()
    active_set = None
    stall_count = 0

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        if active_set is None or t % 40 == 0:
            tol = 0.03 + 0.01 * np.exp(-t / 500.0)
            threshold = M_curr * (1.0 - tol)
            active_set = np.where(conv_y >= threshold)[0]
            if len(active_set) < 6:
                active_set = np.where(conv_y >= M_curr * 0.6)[0]
            if len(active_set) > 0.85 * size:
                active_set = np.where(conv_y >= M_curr * 0.92)[0]

        b_max = lp_best_response(y, M_curr, solver, active_set)

        b_min = None
        if t % 4 == 0:
            b_min, _ = lp_minmax(y, solver, active_set)

        if b_max is None:
            if t % 30 == 0:
                active_set = None
                b_max = lp_best_response(y, M_curr, solver, None)
            if b_max is None:
                continue

        b_sum = b_max.sum()
        g = b_max * norm / b_sum

        if b_min is not None and b_min.sum() > 1e-10:
            g_min_norm = b_min * norm / b_min.sum()
            g = 0.7 * g + 0.3 * g_min_norm

        if len(active_set) < size:
            inactive = np.setdiff1d(np.arange(size), active_set)
            if len(inactive) > 0 and len(inactive) < 500:
                T_check = build_active_constraints(y, inactive)
                viol = T_check @ g
                viol_idx = np.where(viol > M_curr * (1.0 + 1e-5))[0]
                if len(viol_idx) > 0:
                    active_set = np.union1d(active_set, inactive[viol_idx])

        alpha_t = max(2.0 / (t + 2.0), 0.08)
        y_new = (1.0 - alpha_t) * y + alpha_t * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            momentum = (t_k - 1.0) / t_new
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
            t_k = t_new

        prev_y = y
        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score - 1e-14:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score - 1e-14:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        if stall_count > 60:
            y = best_a.copy()
            a = best_a.copy()
            active_set = None
            t_k = 1.0
            stall_count = 0
            pert = 1.0 + 0.02 * rng.standard_normal(n)
            pert = np.maximum(pert, 0.1)
            y = y * pert
            y = np.maximum(y, 0.0)
            y = y * norm / y.sum()

        if verbose and t % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def coord_descent_fine(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_score = fast_autocorr(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for i in perm[:min(n, 400)]:
            if time.monotonic() - t0 > time_limit:
                break

            ai = a[i]
            if ai < 1e-14 * norm:
                continue

            best_delta = 0.0
            best_val = best_score

            for frac in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                for direction in [-1, 1]:
                    d = ai * frac * direction
                    if d < 0 and abs(d) > ai + 1e-15:
                        continue
                    a_try = a.copy()
                    a_try[i] += d
                    a_try = np.maximum(a_try, 0.0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        continue
                    a_try = a_try * norm / s_try
                    cv = fast_autocorr(a_try)
                    val = cv.max()
                    if val < best_val - 1e-14:
                        best_val = val
                        best_delta = d

            if abs(best_delta) > 1e-16:
                a_try = a.copy()
                a_try[i] += best_delta
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try * norm / s_try
                    sc = fast_autocorr(a_try).max()
                    if sc < best_score - 1e-14:
                        a = a_try
                        best_score = sc
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def structured_init(n, rng, style=0):
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
        a = a ** 1.5
    elif style == 1:
        a = np.sin(np.pi * x) ** 0.8
    elif style == 2:
        mu = 0.5
        sigma = 0.2
        a = np.exp(-((x - mu) ** 2) / (2 * sigma ** 2))
    elif style == 3:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(8 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif style == 4:
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif style == 5:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif style == 6:
        a = np.cos(np.pi * x / 2.0) ** 1.2
    else:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.5 * (1.0 + 0.1 * np.cos(6 * np.pi * x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def perturb_seq(a, rng, scale=0.03):
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.05)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s
