def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix_sparse(a, active_indices):
    """Build only the rows of the convolution matrix corresponding to active indices."""
    n = len(a)
    size = 2 * n - 1
    rows = len(active_indices)
    T = np.zeros((rows, n))
    for row, k in enumerate(active_indices):
        i_lo = max(0, int(k) - n + 1)
        i_hi = min(int(k), n - 1)
        if i_hi >= i_lo:
            j_lo = int(k) - i_hi
            j_hi = int(k) - i_lo
            T[row, i_lo:i_hi+1] = a[j_lo:j_hi+1]
    return T


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T


def lp_best_response_sparse(a, M, solver, top_k=100):
    """LP best response using only top-k active constraints."""
    n = len(a)
    size = 2 * n - 1
    conv = fast_autocorr(a)
    
    # Get top-k indices
    k = min(top_k, size)
    active_idx = np.argsort(conv)[-k:][::-1]  # descending order
    
    T = build_conv_matrix_sparse(a, active_idx)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, use_sparse=True, top_k=80):
    """Nesterov-accelerated LP best-response optimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stuck_count = 0
    patience = 0
    restart_count = 0
    momentum_t = 1.0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        if use_sparse:
            b = lp_best_response_sparse(y, M_curr, solver, top_k=top_k)
        else:
            b = lp_best_response_full(y, M_curr, solver)
        
        if b is None:
            stuck_count += 1
            if stuck_count > 50:
                break
            continue

        stuck_count = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (momentum_t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if momentum_t < 2.0:
            a_new = g.copy()
        else:
            momentum = (momentum_t - 1.0) / (momentum_t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new
        momentum_t += 1.0

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            patience = 0
        else:
            patience += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            patience = 0

        if patience > 100 and n_iters % 50 == 0:
            y = best_a.copy()
            a = best_a.copy()
            momentum_t = 1.0
            patience = 0
            restart_count += 1
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    NLP restart {restart_count} at iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, restarts={restart_count}")

    return best_a, best_score


def subgradient_nesterov(a0, time_limit, top_k=50, verbose=True):
    """Subgradient method with Nesterov acceleration for max(a*a)."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    patience = 0
    momentum_t = 1.0
    eta = 1.0
    eta_decay = 0.999

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(y)
        max_val = conv.max()
        
        if max_val < 1e-12:
            break

        # Get subgradient: average of gradients at active indices
        k = min(top_k, 2 * n - 1)
        active_idx = np.argsort(conv)[-k:]
        
        grad = np.zeros(n)
        for ki in active_idx:
            ki = int(ki)
            i_lo = max(0, ki - n + 1)
            i_hi = min(ki, n - 1)
            if i_hi >= i_lo:
                j_lo = ki - i_hi
                j_hi = ki - i_lo
                grad[i_lo:i_hi+1] += a[j_lo:j_hi+1][::-1]
        grad /= (2.0 * k)  # factor of 2 from derivative of (a*a)_k w.r.t. a
        
        # Project gradient onto simplex-like constraint
        grad -= np.sum(grad) / n  # project onto sum=0 direction
        
        alpha = 2.0 / (momentum_t + 2.0)
        y_new = y - eta * alpha * grad
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if momentum_t < 2.0:
            a_new = y_new.copy()
        else:
            momentum = (momentum_t - 1.0) / (momentum_t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new
        momentum_t += 1.0
        n_iters += 1
        eta *= eta_decay

        score = fast_autocorr(a).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience > 200:
            # Reset momentum and increase eta slightly
            momentum_t = 1.0
            eta = 1.0
            patience = 0

        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, eta={eta:.4f}")

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, verbose=True):
    """Pair coordinate descent for polishing."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    rng = np.random.default_rng(12345)
    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        if n_passes % 50 == 0 and no_improve > 100:
            break
        
        pass_improved = False
        # Sample random pairs
        n_pairs = min(n, 500)
        pairs = rng.integers(0, n, size=(n_pairs, 2))
        
        for i, j in pairs:
            if time.monotonic() - t0 > time_limit:
                break
            if i == j:
                continue
            if a[i] < 1e-15:
                continue
            
            # Try moving mass from i to j
            # Compute the effect on convolution
            # (a*a)[k] changes by: -a[i]*a[k-i] - a[j-i]*a[i] + a[j]*a[k-j] + a[j-i]*a[j]
            # for each k... this is complex, so just try a few step sizes
            
            for frac in [0.1, 0.3, 0.5]:
                d = a[i] * frac
                a[i] -= d
                a[j] += d
                new_conv = fast_autocorr(a)
                new_max = new_conv.max()
                if new_max < best_score - 1e-13:
                    best_score = new_max
                    best_a = a.copy()
                    conv = new_conv
                    s = a.sum()
                    a = a * norm / s
                    conv = fast_autocorr(a)
                    best_score = conv.max()
                    best_a = a.copy()
                    pass_improved = True
                    no_improve = 0
                    break
                else:
                    a[i] += d
                    a[j] -= d
        
        if not pass_improved:
            no_improve += 1
        else:
            no_improve = 0

        if verbose and n_passes % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def mirror_descent_subgrad(a0, time_limit, top_k=40, verbose=True):
    """Mirror descent (multiplicative weights) with subgradient of max(a*a)."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0
    patience = 0
    eta = 2.0 / np.sqrt(n)
    rng = np.random.default_rng(54321)

    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(a)
        max_val = conv.max()

        k = min(top_k, 2 * n - 1)
        active_idx = np.argsort(conv)[-k:]

        # Compute subgradient
        grad = np.zeros(n)
        for ki in active_idx:
            ki = int(ki)
            i_lo = max(0, ki - n + 1)
            i_hi = min(ki, n - 1)
            if i_hi >= i_lo:
                j_lo = ki - i_hi
                j_hi = ki - i_lo
                grad[i_lo:i_hi + 1] += a[j_lo:j_hi + 1][::-1]
        grad /= (2.0 * k)

        # Project gradient to remove mean (for normalization invariance)
        grad -= np.mean(grad)

        # Line search over a few eta values
        log_a = np.log(a + 1e-30)
        best_a_try = a.copy()
        best_conv_max = max_val

        for factor in [0.3, 0.5, 0.8, 1.0, 1.5, 2.0]:
            eta_try = eta * factor
            log_a_try = log_a - eta_try * grad
            log_a_try -= log_a_try.max()
            a_try = np.exp(log_a_try)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try * norm / s_try
            conv_try = fast_autocorr(a_try).max()
            if conv_try < best_conv_max:
                best_conv_max = conv_try
                best_a_try = a_try

            best_score = best_conv_max

        else:

            a_pert = a + noise
            a_pert = np.maximum(a_pert, 0)

            if s_pert > 1e-15:

            print(f"    MD iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, eta={eta:.4f}")

def cd_polish(a0, time_limit, verbose=True):

    n = len(a0)
    norm = np.sqrt(2.0 * n)

    s0 = a.sum()

        a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(7777)
    n_passes = 0
    no_improve = 0
    size = 2 * n - 1

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        if n_passes % 100 == 0 and no_improve > 200:
            break

        # Focus on top active lags
        top_lags = np.argsort(conv)[-20:]
        
        # Generate candidate pairs: target high-convolution lags
        n_pairs = 300
        improved_this_pass = False

        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break

            # Choose i from positions that contribute to high-conv lags
            lag = top_lags[rng.integers(0, len(top_lags))]
            i = rng.integers(max(0, lag - n + 1), min(lag + 1, n))
            j = rng.integers(0, n)
            if i == j:
                continue
            if a[i] < norm * 1e-10:
                continue

            # Try transferring mass from i to j
            for frac in [0.2, 0.4, 0.6]:
                d = a[i] * frac
                if d < 1e-15:
                    continue
                a[i] -= d
                a[j] += d
                new_conv = fast_autocorr(a)
                new_max = new_conv.max()
                if new_max < best_score - 1e-14:
                    s = a.sum()
                    if s < 1e-15:
                        a[i] += d
                        a[j] -= d
                        break
                    a = a * norm / s
                    conv = fast_autocorr(a)
                    cur_max = conv.max()
                    best_score = cur_max
                    best_a = a.copy()
                    improved_this_pass = True
                    no_improve = 0
                    break
                else:
                    a[i] += d
                    a[j] -= d

        if not improved_this_pass:
            no_improve += 1
            if no_improve >= 30:
                # Gentle perturbation
                noise = rng.normal(0, norm * 0.003, n)
                a_pert = a + noise
                a_pert = np.maximum(a_pert, 0)
                s_pert = a_pert.sum()
                if s_pert > 1e-15:
                    a = a_pert * norm / s_pert
                    conv = fast_autocorr(a)
                no_improve = 0
        else:
            no_improve = 0

        if verbose and n_passes % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}, no_imp={no_improve}")

    return best_a, best_score

        a_init = np.ones(n_base) / n_base
        best_seq = a_init.copy()
        best_score = evaluate_safe(best_seq)

    n_best = len(best_seq)
    print(f"Starting optimization with n={n_best}")

    remaining = time_budget - (time.monotonic() - t_global)

    print(f"\n=== Phase 1: Sparse LP Nesterov n={n_best} (budget={t1:.0f}s, top_k=100) ===")
    if t1 > 60:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, sc = nesterov_lp(a_norm, t1, solver, verbose=True, use_sparse=True, top_k=100, full_every=30)

    remaining = time_budget - (time.monotonic() - t_global)

    print(f"\n=== Phase 2: Full LP Nesterov n={n_best} (budget={t2:.0f}s) ===")
    if t2 > 40:

        a_out, sc = nesterov_lp(a_norm, t2, solver, verbose=True, use_sparse=False, top_k=0, full_every=1)

            best_seq = np.maximum(a_out, 0.0).copy()

    # Phase 3: Mirror descent subgradient (15% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = remaining * 0.15

    print(f"\n=== Phase 4: CD Polish n={n_best} (budget={t4:.0f}s) ===")
    if t4 > 20:

    n_best = len(best_seq)
    t5 = remaining * 0.5
    print(f"\n=== Phase 5: Final LP n={n_best} (budget={t5:.0f}s, top_k=150) ===")
    if t5 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, sc = nesterov_lp(a_norm, t5, solver, verbose=True, use_sparse=True, top_k=150, full_every=10)

            best_score = sc_eval
            best_seq = np.maximum(a_out, 0.0).copy()
            print(f"  *** Phase 5 improved: {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        # Ensure all values are valid
        best_seq = np.maximum(np.asarray(best_seq, dtype=np.float64), 0.0)
        # Clamp to [0, 1000]
        best_seq = np.minimum(best_seq, 1000.0)
        s_final = best_seq.sum()
        if s_final < 0.01:
            best_seq = np.ones(1600) / 1600
            best_score = evaluate_safe(best_seq)
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0, 1.0, 1.0, 1.0]
