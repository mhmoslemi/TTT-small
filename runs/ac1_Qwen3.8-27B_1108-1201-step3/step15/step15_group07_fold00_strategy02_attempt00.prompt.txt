import numpy as np

def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def compute_subgradient(a, active_indices):
    n = len(a)
    grad = np.zeros(n)
    for k in active_indices:
        i_lo = max(0, int(k) - n + 1)
        i_hi = min(int(k), n - 1)
        if i_hi >= i_lo:
            j_lo = int(k) - i_hi
            j_hi = int(k) - i_lo
            grad[i_lo:i_hi + 1] += a[j_lo:j_hi + 1][::-1]
    return grad

def targeted_cd(a0, time_limit, top_lags=10, verbose=True):

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()

    no_improve = 0
    rng = np.random.default_rng(999)

        n_passes += 1
        if n_passes % 100 == 0 and no_improve > 200:
            break
        conv = fast_autocorr(a)
        max_val = conv.max()
        active_idx = np.argsort(conv)[-top_lags:]

        improved_this_pass = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-12:
                continue

            # Try moving mass from i to nearby positions
            offsets = [1, -1, 2, -2, 3, -3, 5, -5, 10, -10, 20, -20]
            for off in offsets:
                j = (i + off) % n
                if j == i or a[j] >= a[i]:
                    continue
                # Try a few fractions
                for frac in [0.3, 0.5, 0.7]:
                    d = a[i] * frac
                    a[i] -= d
                    a[j] += d
                    new_conv = fast_autocorr(a)
                    new_max = new_conv.max()
                    if new_max < best_score - 1e-13:
                        best_score = new_max
                        best_a = a.copy()
                        conv = new_conv
                        improved_this_pass = True
                        no_improve = 0
                        break
                    else:
                        a[i] += d
                        a[j] -= d

        if not improved_this_pass:
            no_improve += 1
            if no_improve >= 50:
                # Perturb and try again
                noise = rng.normal(0, norm * 0.005, n)
                a_pert = a + noise
                a_pert = np.maximum(a_pert, 0)
                s_pert = a_pert.sum()
                if s_pert > 1e-15:
                    a = a_pert * norm / s_pert
                no_improve = 0

        if verbose and n_passes % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    TCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def full_cd_window(a0, time_limit, window=20, verbose=True):
    """Full pair coordinate descent with window."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    rng = np.random.default_rng(777)
    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-10:
                continue

            for offset in range(1, window + 1):
                for j in [(i + offset) % n, (i - offset) % n]:
                    if j == i:
                        continue
                    if a[i] < norm * 1e-10:
                        break

                    # Compute delta in convolution
                    delta_vec = np.zeros(size)
                    lo1 = i
                    hi1 = min(size, i + n)
                    if hi1 > lo1:
                        delta_vec[lo1:hi1] += a[0:hi1 - lo1]
                    lo2 = max(0, n - 1 - i)
                    hi2 = min(size, n - 1 + i + 1)
                    if hi2 > lo2:
                        idx2 = np.arange(lo2, hi2)
                        src2 = n - 1 - idx2
                        valid = (src2 >= 0) & (src2 < n)
                        if valid.any():
                            delta_vec[lo2:hi2][valid] += a[src2[valid]]

                    lo3 = j
                    hi3 = min(size, j + n)
                    if hi3 > lo3:
                        delta_vec[lo3:hi3] -= a[0:hi3 - lo3]
                    lo4 = max(0, n - 1 - j)
                    hi4 = min(size, n - 1 + j + 1)
                    if hi4 > lo4:
                        idx4 = np.arange(lo4, hi4)
                        src4 = n - 1 - idx4
                        valid4 = (src4 >= 0) & (src4 < n)
                        if valid4.any():
                            delta_vec[lo4:hi4][valid4] -= a[src4[valid4]]

                    delta_vec *= 2.0

                    best_d = 0.0
                    best_val = conv.max()
                    n_samples = 10
                    deltas = np.linspace(0.0, a[i] * 0.95, n_samples)
                    for d in deltas:
                        if d <= 0:
                            continue
                        new_conv = conv + d * delta_vec
                        if 2 * i < size:
                            new_conv[2 * i] = conv[2 * i] - 2 * a[i] * d + d * d
                        if 2 * j < size:
                            new_conv[2 * j] = conv[2 * j] + 2 * a[j] * d + d * d
                        new_max = new_conv.max()
                        if new_max < best_val - 1e-13:
                            best_val = new_max
                            best_d = d

                    if best_d > 0:
                        a[i] -= best_d
                        a[j] += best_d
                        s = a.sum()
                        if s < 1e-15:
                            a[i] += best_d
                            a[j] -= best_d
                            continue
                        a = a * norm / s
                        conv = fast_autocorr(a)
                        cur_max = conv.max()
                        if cur_max < best_score - 1e-13:
                            best_score = cur_max
                            best_a = a.copy()
                            pass_improved = True
                            no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break

        if verbose and n_passes % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def random_restart_search(a0, time_limit, solver, n_restarts=3, verbose=True):
    """Try random perturbations + quick LP optimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    best_a = np.asarray(a0, dtype=np.float64).copy()
    best_score = fast_autocorr(best_a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(4242)
    time_per_restart = time_limit / max(1, n_restarts)

    for r in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        # Perturb
        scale = norm * 0.05 * (r + 1)
        noise = rng.laplace(0, scale, n)
        a_pert = best_a + noise
        a_pert = np.maximum(a_pert, 0)
        s = a_pert.sum()
        if s < 1e-15:
            continue
        a_pert = a_pert * norm / s

        # Quick LP optimization
        t_remaining = time_limit - (time.monotonic() - t0)
        t_sub = min(time_per_restart * 0.7, t_remaining * 0.8)
        if t_sub < 5:
            break
        a_out, sc = nesterov_lp_optimize(a_pert, t_sub, solver, verbose=False)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_out.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    RR restart {r}: improved to {best_score:.8f} ({elapsed:.0f}s)")

    return best_a, best_score
