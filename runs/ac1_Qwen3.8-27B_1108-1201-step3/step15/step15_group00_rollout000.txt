def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def build_convolution_matrix(a):
    """Build the matrix T where T[k,j] = a[j] if j <= k <= j+n-1, else 0."""
    n = len(a)
    num_rows = 2 * n - 1
    # T is (2n-1) x n, each column j has a[j] in rows j to j+n-1
    T = np.zeros((num_rows, n))
    for j in range(n):
        T[j:j+n, j] = a[j]
    return T


def lp_frank_wolfe_fast(a0, time_limit, solver_name, verbose=True, max_iters=100000):
    """LP-based Frank-Wolfe with Nesterov acceleration and warm-start."""
    import cvxpy as cp
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    x_prev = a.copy()
    iter_count = 0
    best_lp_b = None

    # Pre-compute T (changes when y changes, but we rebuild each iteration)
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        # Build constraint matrix
        T = build_convolution_matrix(y)

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver_name, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov-accelerated update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}, M={best_m:.8f}")

    return best_a, best_m


def soft_subgradient_descent(a0, time_limit, top_k=50, beta=5.0, verbose=True):
    """Nesterov-accelerated mirror descent with soft subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    x = a.copy()
    x_prev = a.copy()
    gamma = 4.0
    stall_count = 0
    max_stall = 3000

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > max_stall:
            stall_count = 0
            perturbation = rng.normal(0, 0.02, n)
            x_restarted = best_a + perturbation * best_a
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def peak_targeted_coordinate_descent(a0, time_limit, verbose=True, rng_seed=123):
    """Coordinate descent that specifically targets the peak autocorrelation values."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_idx = np.argmax(ac)

        # Find the indices contributing most to the peak
        k_int = int(peak_idx)
        js = max(0, k_int - n + 1)
        je = min(n - 1, k_int)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_int - idx]

        if contrib.max() < 1e-20:
            stall += 1
            continue

        # Try moving mass from high-contributing positions to low-impact positions
        src_ranking = np.argsort(contrib)[::-1]
        
        # Compute "sensitivity" for each position: how much it contributes to top peaks
        top_peaks = np.argsort(ac)[::-1][:10]
        sensitivity = np.zeros(n)
        for pk in top_peaks:
            pk_int = int(pk)
            for pos in range(n):
                d1 = pk_int - pos
                d2 = pos - pk_int
                if 0 <= d1 < n:
                    sensitivity[pos] += a[d1]
                if 0 <= d2 < n:
                    sensitivity[pos] += a[d2]

        improved_this_round = False
        for src_rank in src_ranking[:5]:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(idx[src_rank])

            # Find best destination: low sensitivity, not in peak region
            candidates = np.argsort(sensitivity)[:max(50, n // 10)]
            
            for dst in candidates:
                if dst == src:
                    continue
                for frac in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved_this_round = True
                        stall = 0
                        break
                if improved_this_round:
                    break

        if not improved_this_round:
            stall += 1
            # Small random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                if a.sum() > 1e-15:
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    CoordPD ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    if remaining > 40:
        print(f"\n=== Phase 1: AMD warmup n={n0} ===")
        t1 = min(remaining * 0.05, 30.0)
        a_out, _ = soft_subgradient_descent(norm_sum1(best_seq), t1, top_k=40, beta=5.0, verbose=False)
        try_update(a_out, "amd_warmup")

    # === Phase 2: LP-FW at parent length (main workhorse, ~50% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_cur = len(best_seq)
        print(f"\n=== Phase 2: LP-FW n={n_cur} (main) ===")
        t2 = min(remaining * 0.45, 700.0)
        a_out, _ = lp_frank_wolfe_fast(norm_sum1(best_seq), t2, solver, verbose=True)
        try_update(a_out, f"lp_main_n{n_cur}")

    # === Phase 3: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        n_cur = len(best_seq)
        print(f"\n=== Phase 3: Coord polish n={n_cur} ===")
        t3 = min(remaining * 0.10, 120.0)
        a_out, _ = peak_targeted_coordinate_descent(norm_sum1(best_seq), t3, verbose=True, rng_seed=42)
        try_update(a_out, f"coord_n{n_cur}")

    # === Phase 4: LP-FW after coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_cur = len(best_seq)
        print(f"\n=== Phase 4: LP-FW round 2 n={n_cur} ===")
        t4 = min(remaining * 0.25, 400.0)
        a_out, _ = lp_frank_wolfe_fast(norm_sum1(best_seq), t4, solver, verbose=True)
        try_update(a_out, f"lp_r2_n{n_cur}")

    # === Phase 5: Try larger n ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    n_target = n_cur + 400
    if n_target > n_cur and remaining > 120:
        print(f"\n=== Phase 5: Expand to n={n_target} ===")
        a_long = interpolate_sequence(best_seq, n_target)
        # Quick AMD at new length
        t5a = min(remaining * 0.08, 60.0)
        a_out, _ = soft_subgradient_descent(norm_sum1(a_long), t5a, top_k=40, beta=5.0, verbose=False)
        try_update(a_out, f"amd_n{n_target}")
        # LP-FW at new length
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t5b = min(remaining * 0.3, 250.0)
            a_out, _ = lp_frank_wolfe_fast(norm_sum1(best_seq), t5b, solver, verbose=True)
            try_update(a_out, f"lp_n{n_target}")

    # === Phase 6: Multi-n exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        print(f"\n=== Phase 6: Multi-n exploration ===")
        n_best = len(best_seq)
        test_ns = []
        for delta in [200, 400, 600, -200, -400]:
            nt = n_best + delta
            if 400 <= nt <= 3200 and nt != n_best:
                test_ns.append(nt)

        for nt in test_ns[:3]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 50:
                break
            print(f"  Trying n={nt}...")
            a_inter = interpolate_sequence(best_seq, nt)
            t_per = min(remaining * 0.25, 100.0)
            a_out, _ = lp_frank_wolfe_fast(norm_sum1(a_inter), t_per, solver, verbose=False)
            try_update(a_out, f"multin_n{nt}")

    # === Phase 7: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        print(f"\n=== Phase 7: Final coord polish n={n_best} ===")
        t7 = min(remaining * 0.4, 150.0)
        a_out, _ = peak_targeted_coordinate_descent(norm_sum1(best_seq), t7, verbose=True, rng_seed=777)
        try_update(a_out, f"coord_final_n{n_best}")

    # === Phase 8: Final LP if time remaining ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        print(f"\n=== Phase 8: Final LP n={n_best} ===")
        t8 = min(remaining * 0.7, 200.0)
        a_out, _ = lp_frank_wolfe_fast(norm_sum1(best_seq), t8, solver, verbose=True)
        try_update(a_out, f"lp_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
