def compute_smoothed_gradient(a, T):
    """Compute gradient of smoothed objective: 2n * LSE(conv/T) / (sum a)^2"""
    n = len(a)
    size = 2 * n - 1
    conv = fast_autocorr(a)
    
    # Softmax weights
    max_conv = conv.max()
    exp_conv = np.exp((conv - max_conv) / T)
    q = exp_conv / exp_conv.sum()
    
    # Gradient of LSE: sum_k q_k * G_k where G_k[j] = 2*a[k-j] if 0<=k-j<n
    # This is a cross-correlation: grad_lse[j] = sum_k q_k * 2*a[k-j]
    # = 2 * (q_reversed * a)[j] in the appropriate sense
    # Actually: sum_k q_k * a[k-j] for valid k,j
    # Let's compute via FFT: this is the cross-correlation of q and a
    fft_size = 1 << (size - 1).bit_length()
    Q = rfft(q, fft_size)
    A = rfft(a, fft_size)
    # Cross-correlation: sum_k q[k] * a[k-j] = ifft(conj(Q) * A) shifted
    # Actually, we want: for each j, sum_k q[k] * a[k-j]
    # This is (q * a_flip) at position (n-1-j)... let me think differently
    # 
    # sum_k q[k] * a[k-j] = (q cross-correlate a at lag j)
    # = ifft(conj(FFT(q)) * FFT(a)) at the right position
    # 
    # Let's use: corr[j] = sum_k q[k] * a[k-j]
    # This is the correlation of q with a at lag j
    # FFT approach: corr = ifft(conj(Q) .* A) gives cyclic correlation
    # For linear correlation, we need zero-padding (which we have via fft_size)
    
    corr = irfft(np.conj(Q) * A, fft_size)
    # corr[j] = sum_k q[k] * a[(k-j) mod fft_size] but we need linear
    # With zero padding, corr[j] for j=0,...,n-1 gives sum_k q[k]*a[k-j]
    # (where a is zero-padded)
    grad_lse = 2.0 * corr[:n]
    
    # Now the full gradient of F = 2n * L / (sum_a)^2
    # dF/da = 2n/(sum_a)^2 * grad_lse - 2n * L * 2/(sum_a)^3
    sum_a = a.sum()
    L = (max_conv + np.log(exp_conv.sum())) / T
    grad = 2.0 * n / (sum_a ** 2) * grad_lse - 4.0 * n * L / (sum_a ** 3)
    
    return grad, conv.max()

def smoothed_gradient_descent(a0, time_limit, verbose=False):

    # Adam parameters
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1
    
    # Temperature schedule: start high, decrease
    T_init = best_score * 0.5
    T_final = best_score * 0.01
    total_iters_est = 5000
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    n_iters = 0
    patience = 0
    rng = np.random.default_rng(777)

    for it in range(200000):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive temperature
        progress = min(1.0, it / total_iters_est)
        T = T_init * (T_final / T_init) ** progress
        T = max(T, best_score * 0.005)

        grad, cur_max = compute_smoothed_gradient(a, T)
        
        # Adam update
        n_iters += 1
        m_hat = beta1 * m_hat + (1 - beta1) * grad
        v_hat = beta2 * v_hat + (1 - beta2) * grad ** 2
        m_hat_hat = m_hat / (1 - beta1 ** n_iters)
        v_hat_hat = v_hat / (1 - beta2 ** n_iters)
        
        a_new = a - lr * m_hat_hat / (np.sqrt(v_hat_hat) + eps)

        noise = rng.laplace(0, scale, n)
        a_pert = best_a + noise
        a_pert = np.maximum(a_pert, 0)
        s = a_pert.sum()
        if s < 1e-15:
            continue
        a_pert = a_pert * norm / s

        t_remaining = time_limit - (time.monotonic() - t0)
        t_sub = min(time_per_restart * 0.7, t_remaining * 0.8)
        if t_sub < 5:
            break
        a_out, sc = nesterov_lp_optimize(a_pert, t_sub, solver, verbose=False)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_out.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    RR restart {r}: improved to {best_score:.8f} ({elapsed:.0f}s)")

    return best_a, best_score

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1600
        a_init = np.ones(n_base) / n_base
        best_seq = a_init.copy()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # === Phase 0: Multi-n exploration (8% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t_explore = remaining * 0.08
    print(f"\n=== Phase 0: Multi-n exploration (budget={t_explore:.0f}s) ===")
    
    n_candidates = [800, 1000, 1200, 1400, 1600, 1800, 2000, 2400]
    # Filter: only try n values close to parent size or different
    if parent is not None:
        n_parent = len(parent)
        n_candidates = [n for n in n_candidates if abs(n - n_parent) > 100 or n == n_parent]
        if n_parent not in n_candidates:
            n_candidates.append(n_parent)
    
    n_per_try = max(30, int(5000 / max(1, len(n_candidates))))
    
    for n_try in n_candidates:
        if time.monotonic() - t_global > time_budget - 100:
            break
        # Initialize: if parent exists and same n, use it; else random
        if parent is not None and len(parent) == n_try:
            a_init = parent.copy()
        else:
            # Use a structured initialization: smooth bump
            x = np.linspace(0, 1, n_try)
            a_init = np.exp(-5.0 * (x - 0.5) ** 2)
            a_init = a_init + 0.1 * rng.random(n_try)
            a_init = a_init / a_init.sum()
        
        a_out, sc = quick_lp_optimize(a_init, n_per_try, solver)
        if sc < best_score:
            # Compute actual score
            a_score = evaluate_safe(a_out)
            if np.isfinite(a_score) and a_score < best_score:
                best_score = a_score
                best_seq = a_out.copy()
                elapsed = time.monotonic() - t_global
                print(f"  Multi-n: n={n_try} -> score={a_score:.6f} (t={elapsed:.0f}s)")
    
    print(f"  After Phase 0: best={best_score:.6f}, n={len(best_seq)}")

    # === Phase 1: LP Nesterov at best n (35% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t1 = remaining * 0.35
    print(f"\n=== Phase 1: LP Nesterov n={n_best} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n_best}")

    # === Phase 2: Smoothed Gradient Descent (25% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = remaining * 0.25
    print(f"\n=== Phase 2: Smoothed GD n={n_best} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = smoothed_gradient_descent(a_norm, t2, verbose=True)
        do_try_update(a_out, f"sgd_n{n_best}")

    # === Phase 3: LP Nesterov round 2 (20% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = remaining * 0.20
    print(f"\n=== Phase 3: LP Nesterov R2 n={n_best} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True)
        do_try_update(a_out, f"lp2_n{n_best}")

    # === Phase 4: Targeted CD (10% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = remaining * 0.10
    print(f"\n=== Phase 4: Targeted CD n={n_best} (budget={t4:.0f}s) ===")
    if t4 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = targeted_cd(a_norm, t4, top_lags=8, verbose=True)
        do_try_update(a_out, f"tcd_n{n_best}")

    # === Phase 5: Random restarts (5% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_best = len(best_seq)
        t5 = remaining * 0.5
        print(f"\n=== Phase 5: Random Restarts n={n_best} (budget={t5:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = random_restart_search(a_norm, t5, solver, n_restarts=3, verbose=True)
        do_try_update(a_out, f"rr_n{n_best}")

    # === Phase 6: Final polish - LP + CD on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        n_best = len(best_seq)
        t6a = min(remaining * 0.5, 80.0)
        t6b = remaining - t6a - 10
        if t6a > 15:
            print(f"\n=== Phase 6a: Final LP n={n_best} (budget={t6a:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = nesterov_lp_optimize(a_norm, t6a, solver, verbose=False)
            do_try_update(a_out, f"lp3_n{n_best}")
        if t6b > 15:
            print(f"\n=== Phase 6b: Final CD n={n_best} (budget={t6b:.0f}s) ===")
            n_best = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = targeted_cd(a_norm, t6b, top_lags=12, verbose=False)
            do_try_update(a_out, f"cd3_n{n_best}")

    # Final: try smoothed GD one more time if time
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        t7 = remaining * 0.7
        print(f"\n=== Phase 7: Final SGD n={n_best} (budget={t7:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = smoothed_gradient_descent(a_norm, t7, verbose=False)
        do_try_update(a_out, f"sgd2_n{n_best}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
