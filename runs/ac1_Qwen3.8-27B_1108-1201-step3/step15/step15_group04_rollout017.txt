def gs_optimize(a0, n_iters=50):
    """Gerchberg-Saxton style spectral flattening."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    
    for it in range(n_iters):
        # Compute DFT
        A = fft(a)
        # Flatten the amplitude, keep phase
        amp = np.abs(A)
        phase = np.angle(A)
        target_amp = np.ones(n)
        # Partial flattening (interpolate between current and flat)
        blend = 0.3
        new_amp = (1 - blend) * amp + blend * target_amp
        # Reconstruct
        A_new = new_amp * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))
        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        sc = fast_conv_self(a_new).max()
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a

def lse_gradient_descent(a0, time_limit, verbose=True):
    """LSE (Log-Sum-Exp) smoothed gradient descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    n_stages = 10
    temps = np.geomspace(50.0, 0.5, n_stages)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(20, min(200, int(remaining * 3)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -200.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)
            g2 = aw[:n]
            grad = 2.0 * (g1 + g2)
            
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                step = 0.0
            else:
                step = norm / (T_val * np.sqrt(n)) / (1.0 + 0.1 * si)
            
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new
            
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose:

def lp_cutting_plane(y, M, solver, max_constraints=200, max_rounds=5):
    """Solve LP max sum(b) s.t. conv(y,b) <= M, b>=0 using cutting planes."""
    n = len(y)
    size = 2 * n - 1
    
    # Initialize active set with the most "important" constraints
    conv_y = fast_conv_self(y)
    # Constraints where conv_y is largest are most likely to be binding
    # Also include lag 0 (always binding)
    top_k = min(max_constraints, size)
    active = np.argsort(-conv_y)[:top_k]
    # Ensure lag 0 is included
    if 0 not in active:
        active[0] = 0
    
    b = None
    for round_i in range(max_rounds):
        T = build_T(y, active)
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            return None
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None
        
        b = np.asarray(b_var.value, dtype=np.float64)
        b = np.maximum(b, 0.0)
        
        # Check all constraints
        conv_yb = fast_conv_cross(y, b)
        violated = conv_yb > M * (1 + 1e-4)
        n_violated = np.sum(violated)
        
        if n_violated == 0:
            return b
        
        # Add most violated constraints
        if n_violated <= 50:
            new_active = np.where(violated)[0]
        else:
            # Add top 50 most violated
            viol_idx = np.where(violated)[0]
            viol_vals = conv_yb[viol_idx]
            top_viol = viol_idx[np.argsort(-viol_vals)[:50]]
            new_active = top_viol
        
        # Merge active sets, keeping the most important ones
        combined = np.unique(np.concatenate([active, new_active]))
        if len(combined) > max_constraints * 2:
            # Keep the most violated + original top
            keep = np.unique(np.concatenate([new_active, active[:max_constraints//2]]))
            combined = keep[:max_constraints * 2]
        active = combined
    
    return b

    y = a.copy()

    iteration = 0

        M = conv_y.max()
        if M < 1e-12:

        g = b * norm / b.sum()
        
        # Nesterov update
        alpha = 2.0 / (iteration + 2.0)

            if s_y > 1e-15:

    size = 2 * n - 1
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue
            
            # Find best j to transfer mass to
            j_candidates = set(int(rng.integers(n)) for _ in range(8))
            # Add structured candidates from peak lags
            peak_lags = np.where(conv >= cur_max * 0.85)[0]
            for kl in peak_lags[:6]:
                j_candidates.add(int(kl % n))
                j_candidates.add(int((kl + i) % n))
            j_candidates.discard(i)
            
            best_val = cur_max
            best_j = -1
            best_delta = 0.0
            
            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]
                
                # Compute change in conv when moving delta from i to j
                # conv_new[k] = conv[k] - delta*(a[k-i] + a[i-k]) + delta*(a[k-j] + a[j-k])
                # (simplified, ignoring the delta^2 terms for now)
                g = np.zeros(size)
                # Effect of reducing a[i] by delta
                for k in range(size):
                    ki = k - i
                    if 0 <= ki < n:
                        g[k] -= a[ki]
                    kj = k - (-i)  # a[i-k] contributes
                    # Actually, let me just compute the gradient of conv w.r.t. a[i]
                    pass
                
                # Simpler: just try a few delta values
                for frac in [0.1, 0.3, 0.5, 0.7]:
                    delta = frac * ai
                    a_trial = a.copy()
                    a_trial[i] = ai - delta
                    a_trial[j] = aj + delta
                    if a_trial[i] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    sc_t = fast_conv_self(a_trial).max()
                    if sc_t < best_val:
                        best_val = sc_t
                        best_j = j
                        best_delta = delta
            
            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 10:
                break
            if no_improve == 5:
                n_perturb = max(10, n // 25)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.2 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0
        
        if verbose and iteration % 50 == 0:
            print(f"    CD ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

    choice = rng.integers(0, 8)
    x = np.linspace(0.0, 1.0, n)
