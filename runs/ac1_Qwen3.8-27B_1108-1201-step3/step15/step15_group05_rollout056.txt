            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        if active_set is None or t % 30 == 0:
            tol = max(0.005, 0.04 * np.exp(-t / 200.0))
            active_set = np.where(conv_y >= M_curr * (1.0 - tol))[0]
            top_k = np.argsort(conv_y)[-max(3, n // 300):]
            active_set = np.unique(np.concatenate([active_set, top_k]))
            if len(active_set) > 0.5 * (2 * n - 1):
                active_set = top_k

        b, _ = lp_best_response(y, M_curr, active_set, solver)
        if b is None:
            active_set = np.argsort(conv_y)[-max(20, n // 50):]
            b, _ = lp_best_response(y, M_curr, active_set, solver)
            if b is None:
                continue

        # Verify and expand active set
        all_k = np.arange(2 * n - 1)
        complement = np.setdiff1d(all_k, active_set)
        if len(complement) > 0:
            T_comp = build_constraint_matrix(y, complement, n)
            violations = T_comp @ b
            viol_idx = np.where(violations > M_curr * (1 + 1e-5))[0]
            if len(viol_idx) > 0:
                new_active = np.union1d(active_set, complement[viol_idx])
                if len(new_active) > 0.7 * (2 * n - 1):
                    new_active = all_k
                b, _ = lp_best_response(y, M_curr, new_active, solver)
                if b is None:
                    continue
                active_set = new_active

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.97)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Line search refinement every 50 iterations
        if n_iters % 50 == 0:
            conv_cur = fast_autocorr(a)
            M_cur = conv_cur.max()
            # Try blending a with g
            for frac in [0.1, 0.2, 0.3, 0.5, 0.7]:
                blend = (1.0 - frac) * a + frac * g
                blend = np.maximum(blend, 0.0)
                s_b = blend.sum()
                if s_b > 1e-15:
                    blend = blend * norm / s_b
                    sc = fast_autocorr(blend).max()
                    if sc < best_score:
                        best_score = sc
                        best_a = blend.copy()
                        a = blend.copy()
                        y = blend.copy()
                        stall_count = 0

        if stall_count > stall_threshold:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall_count = 0

        if verbose and n_iters % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

    # Final line search
    if n_iters > 0:
        for frac in np.linspace(0, 1, 20):
            blend = (1.0 - frac) * best_a + frac * a
            blend = np.maximum(blend, 0.0)
            s_b = blend.sum()
            if s_b > 1e-15:
                blend = blend * norm / s_b
                sc = fast_autocorr(blend).max()
                if sc < best_score:
                    best_score = sc
                    best_a = blend.copy()

    return best_a, best_score


def mirror_descent_optimize(a0, time_limit, verbose=False, seed=0):
    """Multiplicative mirror descent for the peak minimization problem."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    step = 0.5 / np.sqrt(n)
    patience = 0
    max_patience = 3000

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()

        # Use top few peaks
        n_peaks = max(3, min(8, n // 200))
        top_indices = np.argsort(conv)[-n_peaks:]
        weights = np.exp(-8.0 * (M_curr - conv[top_indices]) / max(M_curr * 0.01, 1e-10))
        weights = weights / weights.sum()

        # Combined subgradient
        grad = np.zeros(n)
        for w, k in zip(weights, top_indices):
            for j in range(n):
                v = 0.0
                j1 = k - j
                if 0 <= j1 < n:
                    v += a[j1]
                j2 = j - k
                if 0 <= j2 < n:
                    v += a[j2]
                grad[j] += w * v

        # Multiplicative update
        grad_max = grad.max()
        if grad_max < 1e-15:
            break
        grad = grad / grad_max

        a_new = a * np.exp(-step * grad)
        a_new = np.maximum(a_new, 1e-15)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        score_new = fast_autocorr(a_new).max()
        if score_new < best_score - 1e-12:
            patience = 0
            a = a_new
            best_score = score_new
            best_a = a_new.copy()
            step = min(step * 1.03, 3.0)
        else:
            patience += 1
            step *= 0.97
            if step < 1e-10:
                break

        # Periodic restart from best
        if patience % 500 == 0:
            a = best_a.copy()
            step = 0.5 / np.sqrt(n)

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Mirror t={t} ({elapsed:.0f}s): best={best_score:.10f}, step={step:.6f}")

    return best_a, best_score


def coordinate_descent_fast(a0, time_limit, verbose=False):
    """Coordinate descent with incremental convolution updates."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_a = a.copy()
    best_score = conv.max()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_passes = 0
    max_no_improve = 4

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        no_improve = 0

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]
            if aj < 1e-15:
                continue

            # Compute effect of changing a[j]
            # (a*a)[k] change when a[j] -> x:
            # delta_k = 2*(x-aj)*a[k-j] for k != 2j (with bounds)
            # delta_{2j} = (x-aj)*(x+aj) = x^2 - aj^2

            # Find the current max and where it is
            M_cur = conv.max()

            # Try a few candidate values
            s_others = a.sum() - aj
            candidates = [aj * 0.3, aj * 0.6, aj * 0.8, aj * 1.2, aj * 1.5,
                          s_others * 0.05, s_others * 0.2, s_others * 0.5]
            candidates = [x for x in candidates if x >= 0 and x <= norm]
            candidates.append(0.0)
            candidates.append(aj)

            best_x = aj
            best_obj = 2.0 * n * M_cur / (a.sum() ** 2)

            for x in candidates:
                if abs(x - aj) < 1e-12:
                    continue
                new_sum = s_others + x
                if new_sum < 1e-15:
                    continue
                # Compute new max conv approximately
                delta = x - aj
                # The change in conv:
                # For k where k-j is valid (0 <= k-j < n): conv[k] += 2*delta*a[k-j]
                # For k = 2j: additional delta*(x+aj) = x^2 - aj^2 (already counted above as 2*delta*aj, so add delta^2)
                new_conv_max = M_cur
                # Quick estimate: check the top few lags
                top_lags = np.argsort(conv)[-20:]
                for k in top_lags:
                    val = conv[k]
                    kj = k - j
                    if 0 <= kj < n:
                        val += 2.0 * delta * a[kj]
                    if k == 2 * j:
                        val += delta * delta
                    new_conv_max = max(new_conv_max, val)
                obj = 2.0 * n * new_conv_max / (new_sum * new_sum)
                if obj < best_obj:
                    best_obj = obj
                    best_x = x

            if abs(best_x - aj) > 1e-12:
                # Update convolution incrementally
                delta = best_x - aj
                for k in range(size):
                    kj = k - j
                    if 0 <= kj < n:
                        conv[k] += 2.0 * delta * a[kj]
                conv[2 * j] += delta * delta  # additional for k=2j

                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-15:
                    scale = norm / s_new
                    if abs(scale - 1.0) > 1e-10:
                        a = a * scale
                        conv = conv * scale * scale

                new_score = conv.max()
                if new_score < best_score * (1 - 1e-10):
                    best_score = new_score
                    best_a = a.copy()
                    no_improve = 0
                else:
                    # Revert
                    a[j] = aj
                    for k in range(size):
                        kj = k - j
                        if 0 <= kj < n:
                            conv[k] -= 2.0 * delta * a[kj]
                    conv[2 * j] -= delta * delta
                    no_improve += 1
            else:
                no_improve += 1

            if no_improve > n // 2:
                break

        if n_passes % 5 == 1:
            # Full recompute to prevent drift
            conv = fast_autocorr(a)

        # Check if we should stop
        full_conv = fast_autocorr(a)
        if abs(full_conv.max() - best_score) < 1e-10 and n_passes > 3:
            break

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def subgradient_polish(a0, time_limit, verbose=False):
    """Subgradient method with adaptive step and multi-peak gradient."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    step = 2.0 / np.sqrt(n)
    patience = 0
    max_patience = 2000

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()
        n_peaks = max(2, min(6, n // 300))
        top_indices = np.argsort(conv)[-n_peaks:]
        weights = np.exp(-6.0 * (M_curr - conv[top_indices]) / max(M_curr * 0.02, 1e-10))
        weights = weights / weights.sum()

        grad = np.zeros(n)
        for w, k in zip(weights, top_indices):
            for j in range(n):
                v = 0.0
                j1 = k - j
                if 0 <= j1 < n:
                    v += a[j1]
                j2 = j - k
                if 0 <= j2 < n:
                    v += a[j2]
                grad[j] += w * v

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Projected gradient step with backtracking line search
        best_trial = None
        best_trial_score = best_score
        for frac in [step, step * 0.5, step * 0.25, step * 0.1]:
            trial = a - frac * grad / np.sqrt(n)
            trial = np.maximum(trial, 0.0)
            s_tr = trial.sum()
            if s_tr < 1e-15:
                continue
            trial = trial * norm / s_tr
            sc = fast_autocorr(trial).max()
            if sc < best_trial_score:
                best_trial_score = sc
                best_trial = trial

        if best_trial is not None and best_trial_score < best_score - 1e-12:
            patience = 0
            a = best_trial
            best_score = best_trial_score
            best_a = best_trial.copy()
            step = min(step * 1.02, 5.0)
        else:
            patience += 1
            step *= 0.96
            if step < 1e-10:
                break

        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def perturb_sequence(a, rng, scale=0.03):
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.05)
    a_pert = a * noise
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def symmetrize_and_check(a):
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    s = a_sym.sum()
    if s < 1e-15:
        return None, None
    a_sym = a_sym * np.sqrt(2.0 * n) / s
    sc = fast_autocorr(a_sym).max()
    return a_sym, sc


def propose_candidate():
    """
    Main search function combining LP best-response, mirror descent,
    coordinate descent, and subgradient polishing with multiple restarts.
    """
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = 1740.0

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    # Quick symmetrization check
    a_sym, sc_sym = symmetrize_and_check(best_seq)
    if sc_sym is not None and sc_sym < best_score:
        a_sym_norm = a_sym / a_sym.sum()
        sc_check = evaluate_safe(a_sym_norm)
        if sc_check < best_score:
            best_score = sc_check
            best_seq = a_sym_norm.copy()
            print(f"  *** Symmetrization: {sc_check:.8f}")

    n_base = len(best_seq)
    print(f"Starting: n={n_base}, score={best_score:.8f}")

    # === Phase 1: LP Nesterov (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 450.0)
    if t1 > 60:
        print(f"\n=== Phase 1: LP Nesterov n={n_base}, budget={t1:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, seed=0)
        try_update(a_out, f"lp1_n{n_base}")

    # === Phase 2: Mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.12, 120.0)
    if t2 > 30:
        print(f"\n=== Phase 2: Mirror descent n={n_cur}, budget={t2:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = mirror_descent_optimize(a_norm, t2, verbose=True, seed=1)
        try_update(a_out, f"mirror_n{n_cur}")

    # === Phase 3: LP Nesterov round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.20, 250.0)
    if t3 > 40:
        print(f"\n=== Phase 3: LP Nesterov round 2 n={n_cur}, budget={t3:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True, seed=77)
        try_update(a_out, f"lp2_n{n_cur}")

    # === Phase 4: Subgradient polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.10, 100.0)
    if t4 > 20:
        print(f"\n=== Phase 4: Subgradient n={n_cur}, budget={t4:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = subgradient_polish(a_norm, t4, verbose=True)
        try_update(a_out, f"subg_n{n_cur}")

    # === Phase 5: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = min(remaining * 0.15, 150.0)
    if t5 > 15:
        print(f"\n=== Phase 5: Coord descent n={n_cur}, budget={t5:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coordinate_descent_fast(a_norm, t5, verbose=True)
        try_update(a_out, f"cd_n{n_cur}")

    # === Phase 6: Perturbation + LP restart ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t6_total = min(remaining * 0.15, 120.0)
    if t6_total > 30:
        print(f"\n=== Phase 6: Perturbation restarts, budget={t6_total:.0f}s ===")
        n_restarts = 2
        t6_each = t6_total / n_restarts
        for r in range(n_restarts):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 40:
                break
            t6 = min(t6_each, remaining - 20)
            if t6 < 15:
                break
            pert = perturb_sequence(best_seq, rng, scale=0.015 + 0.01 * r)
            a_norm = pert * np.sqrt(2.0 * n_cur) / pert.sum()
            a_out, _ = nesterov_lp_optimize(a_norm, t6, solver, verbose=False, seed=500 + r * 100)
            try_update(a_out, f"pert{r}_n{n_cur}")

    # === Phase 7: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t7 = min(remaining * 0.20, 100.0)
    if t7 > 20:
        print(f"\n=== Phase 7: Final LP polish n={n_cur}, budget={t7:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t7, solver, verbose=False, seed=999)
        try_update(a_out, f"lp_final_n{n_cur}")

    # === Phase 8: Final subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 20:
        t8 = min(remaining * 0.3, 40.0)
        if t8 > 10:
            print(f"\n=== Phase 8: Final subgrad n={n_cur} ===")
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = subgradient_polish(a_norm, t8, verbose=False)
            try_update(a_out, f"subg_final_n{n_cur}")

    # === Phase 9: Final CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 15:
        t9 = min(remaining * 0.4, 30.0)
        if t9 > 5:
            print(f"\n=== Phase 9: Final CD n={n_cur} ===")
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = coordinate_descent_fast(a_norm, t9, verbose=False)
            try_update(a_out, f"cd_final_n{n_cur}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
