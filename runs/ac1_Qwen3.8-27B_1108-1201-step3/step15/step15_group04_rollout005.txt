def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def compute_obj(a):
    """Compute the objective value directly."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    n = len(a)
    conv = fast_conv_self(a)
    max_b = conv.max()
    return 2.0 * n * max_b / (S * S)

def multiplicative_weight_update(a, M, eta, n_iter=100, verbose=False):
    """
    Use multiplicative weight updates to find the best response b to a.
    Maximize sum(b) s.t. T(a) @ b <= M, b >= 0.
    """
    n = len(a)
    norm = np.sqrt(2.0 * n)
    b = a.copy() * norm / a.sum()  # Initialize with normalized a
    size = 2 * n - 1
    
    for it in range(n_iter):
        Tu_b = fast_conv_cross(a, b)
        violations = np.maximum(0.0, Tu_b - M)
        if violations.max() < 1e-10:
            break
        # Multiplicative weight update
        # Penalty for each b[i] is sum of violations it contributes to
        # T[k,i] = a[i] for k in [i, i+n-1] and a[k-i+n-1] for k in [0, i-1]
        # Actually, let's compute the gradient of the penalty w.r.t. b
        # penalty = sum_k max(0, (T@b)[k] - M)
        # d(penalty)/d(b[i]) = sum_k T[k,i] * 1[(T@b)[k] > M]
        # T[k,i] is: for the convolution, (a*b)[k] = sum_j a[j]*b[k-j]
        # So d(T@b)[k]/d(b[i]) = a[k-i] if k-i is in [0,n-1], else 0
        
        # Efficient computation: the gradient is the cross-correlation of a with the violation indicator
        indicator = (Tu_b > M).astype(float)
        # grad[i] = sum_k a[k-i] * indicator[k] for valid k
        # This is the cross-correlation of a and indicator
        fft_size = 1 << (size - 1).bit_length()
        A = rfft(a, fft_size)
        I = rfft(indicator, fft_size)
        grad = irfft(np.conj(A) * I, fft_size)[:n]
        # Also need the "other direction": d(T@b)[k]/d(b[i]) also involves a[i-k+n-1]... 
        # Actually for full convolution: (a*b)[k] = sum_j a[j] b[k-j]
        # d/d(b[i]) (a*b)[k] = a[k-i] (if k-i in [0,n-1])
        # So grad[i] = sum_{k: k-i in [0,n-1]} a[k-i] * indicator[k]
        # = sum_{j: 0<=j<n} a[j] * indicator[j+i]
        # This is a cross-correlation
        
        grad = np.maximum(grad, 0)
        b = b * np.exp(-eta * grad)
        b = np.maximum(b, 0)
        s = b.sum()
        if s < 1e-15:
            b = np.ones(n) / n
        b = b * norm / s
        
        if verbose and it % 20 == 0:
            print(f"      MWU it={it}, sum_b={b.sum():.4f}, max_viol={violations.max():.6f}")
    
    return b

def lp_solve_fast(a, M, solver, active_constraints=None, verbose=False):
    """Solve the LP: max sum(b) s.t. T(a)@b <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    
    # Build constraint matrix sparsely
    # T[k, i] contributes to (a*b)[k] = sum_j a[j]*b[k-j]
    # For fixed a, (a*b)[k] = sum_i a[k-i]*b[i] (where valid)
    # So the constraint is: for each k, sum_i T[k,i]*b[i] <= M
    # where T[k,i] = a[k-i] if 0<=k-i<n, else 0
    
    # This is a large constraint. For n=1600, we have 3199 constraints and 1600 variables.
    # Let's use the sparse structure.
    # Actually, for the full formulation, each constraint k involves at most n variables.
    # The total number of non-zeros is about n*(n) = n^2 (each column i has n non-zeros).
    
    # For large n, this LP is too big. Let's use a subset of constraints.
    if active_constraints is not None:
        idx = active_constraints
    else:
        idx = np.arange(size)
    
    # Build the submatrix
    # For constraint k: sum_i a[k-i]*b[i] where max(0, k-n+1) <= i <= min(n-1, k)
    rows = []
    cols = []
    vals = []
    for k in idx:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    
    from scipy.sparse import csr_matrix
    T_sub = csr_matrix((vals, (rows, cols)), shape=(len(idx), n))
    
    constraint = T_sub @ b_var <= M
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [constraint])
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    
    duals = None
    try:
        if constraint.dual_value is not None:
            full_duals = np.zeros(size)
            full_duals[idx] = np.asarray(constraint.dual_value, dtype=np.float64)
            duals = full_duals
    except Exception:
        pass
    
    return b_val, duals

def lse_smooth_descent(a0, time_budget, verbose=True):
    """Optimize using log-sum-exp smoothing of the max objective."""
    n = len(a0)

    s = a.sum()

    # Progressive temperature
    n_phases = 8

        if time.monotonic() - t0 > time_budget:
            break

        n_steps = max(20, min(200, int(remaining * 3)))

            grad = 2.0 * irfft(np.conj(A) * W, fft_size)[:n]

            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            
            # Backtracking line search
            step = lr

def accelerated_bcd(a0, time_budget, solver, verbose=True):

    Accelerated Block Coordinate Descent:

    y = a.copy()  # Nesterov iterate
    v = np.zeros(n)  # Momentum
    
    no_improve = 0

        if time.monotonic() - t0 > time_budget:
            break
        
        # Step 1: Compute M from y
        conv_y = fast_conv_self(y)
        M = conv_y.max()

        else:

            # Or simply: b = y (self-consistent approximation)

        if b is None or b.sum() < 1e-15:
            b = y.copy()
        
        g = b * norm / b.sum()  # Normalized best response
        
        # Step 3: Nesterov update
        alpha = 2.0 / (t + 2.0)
        v = alpha * (g - y) + (1 - alpha) * v
        y_new = y + v
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        # Step 4: Averaged iterate
        if t == 1:
            a = g.copy()
        else:
            a = (1 - 1.0/t) * a + (1.0/t) * y_new
            a = np.maximum(a, 0.0)
            s_a = a.sum()
            if s_a < 1e-15:
                a = y_new.copy()
            else:
                a = a * norm / s_a
        
        y = y_new
        
        # Check improvement
        m_y = fast_conv_self(y).max()
        improved = False
        if m_y < best_m - 1e-10:
            best_m = m_y
            best_a = y.copy()
            improved = True
        
        m_a = fast_conv_self(a).max()
        if m_a < best_m - 1e-10:
            best_m = m_a
            best_a = a.copy()
            improved = True
        
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
        
        # Perturbation when stuck
        if no_improve == 50:
            n_pert = max(20, n // 10)
            rng_p = np.random.default_rng(t)
            idx = rng_p.choice(n, size=n_pert, replace=False)
            y[idx] *= (1 + 0.3 * rng_p.standard_normal(n_pert))

            s_y = y.sum()
            if s_y > 1e-15:
                y = y * norm / s_y
            no_improve = 0
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ABCD t={t} ({elapsed:.0f}s): best_m={best_m:.8f}")

def interp_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    s = f.sum()
    f_norm = f / s
    if old_n == new_n:
        return f_norm.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f_norm)
    f_new = np.maximum(f_new, 0)
    s_new = f_new.sum()
    if s_new < 1e-15:
        return np.ones(new_n) / new_n
    return f_new / s_new
