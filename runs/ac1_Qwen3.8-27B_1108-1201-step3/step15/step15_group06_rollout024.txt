Actually, I realize I should think more carefully about what "peak shaving" means. If the max autocorrelation is at lag k*, then:
(a*a)[k*] = sum_j a[j]*a[k*-j]

To reduce this, we want to reduce the product a[j]*a[k*-j] for the pairs that contribute most. We can:
- Reduce a[j] for the j where a[k*-j] is large (and vice versa)
- But this increases other lags potentially
- A safe move: reduce a[j] and a[k*-j] simultaneously (which reduces (a*a)[k*] by 2*a[j]*a[k*-j]*delta - delta^2)

More precisely, if we set a[j] -= δ and a[k*-j] -= δ, the change in (a*a)[k*] is:
- The term a[j]*a[k*-j] becomes (a[j]-δ)(a[k*-j]-δ) = a[j]*a[k*-j] - a[j]*δ - a[k*-j]*δ + δ^2
- Change: -δ(a[j] + a[k*-j]) + δ^2

And the change in sum(a) is -2δ, so (sum(a))^2 changes by approximately -4δ*sum(a).

The net effect on the score depends on whether this lag was the unique max or not.

This is essentially what coordinate descent does, but focused on the peak lag.

            if p == q or q < 0 or q >= n:
                # Diagonal term: a[p]^2
                # Reduce a[p]
                best_delta = 0
                best_val = M
                for frac in [0.05, 0.1, 0.15, 0.2, 0.3]:
                    d = a[p] * frac
                    a_try = a.copy()
                    a_try[p] = max(0, a_try[p] - d)
                    a_try = normalize(a_try, norm)
                    cv = fast_autocorr(a_try)
                    val = cv.max()
                    if val < best_val - 1e-14:
                        best_val = val
                        best_delta = -d
                if best_delta != 0:
                    a[p] = max(0, a[p] + best_delta)
                    a = normalize(a, norm)
                    sc = fast_autocorr(a).max()
                    if sc < best_score - 1e-14:
                        best_score = sc
                        best_a = a.copy()
                        improved = True
            else:
                # Off-diagonal: reduce a[p] and/or a[q]
                for target in [p, q]:
                    best_delta = 0
                    best_val = M
                    for frac in [0.05, 0.1, 0.15, 0.2]:
                        d = a[target] * frac
                        a_try = a.copy()
                        a_try[target] = max(0, a_try[target] - d)
                        a_try = normalize(a_try, norm)
                        cv = fast_autocorr(a_try)
                        val = cv.max()
                        if val < best_val - 1e-14:
                            best_val = val
                            best_delta = -d
                    if best_delta != 0:
                        a[target] = max(0, a[target] + best_delta)
                        a = normalize(a, norm)
                        sc = fast_autocorr(a).max()
                        if sc < best_score - 1e-14:
                            best_score = sc
                            best_a = a.copy()
                            improved = True
        
        if not improved and round_num > 20:
            break
    
    return best_a, best_score

def coord_descent_simple(a0, time_limit, rng):
    """Simple coordinate descent: try adjusting each coordinate."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(a0, norm)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if no_improve > 20:
            break
        pass_improved = False
        perm = rng.permutation(n)
        
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-6 * norm:
                continue
            
            best_val = best_score
            best_ai = a[i]
            for frac in [0.5, 0.7, 1.0, 1.3, 1.5, 2.0]:
                ai_new = a[i] * frac
                a_try = a.copy()
                a_try[i] = ai_new
                a_try = normalize(a_try, norm)
                cv = fast_autocorr(a_try)
                val = cv.max()
                if val < best_val - 1e-14:
                    best_val = val
                    best_ai = ai_new
            
            if best_ai != a[i]:
                a[i] = best_ai
                a = normalize(a, norm)
                sc = fast_autocorr(a).max()
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a.copy()
                    pass_improved = True
        
        if pass_improved:
            no_improve = 0
        else:
            no_improve += 1
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss

def get_solver():
    candidates = [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    for s in candidates:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.SCS

def propose_candidate():

    t_global = time.monotonic()
    time_budget = 1750.0

            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            old = best_score
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
    
    if best_seq is None:
        # Generate initial
        n_init = 1600
        x = np.linspace(0, 1, n_init)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Initial: n={n_init}, score={best_score:.8f}")
    
    remaining = time_budget - (time.monotonic() - t_global)
    
    # === Phase 1: LP Nesterov on parent ===
    n_base = len(best_seq)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} ===")
    t1 = min(remaining * 0.30, 400.0)
    if t1 > 60:

    # === Phase 2: Peak shaving ===

    if t2 > 30:
        rng = np.random.default_rng(42)
        a_norm = normalize(best_seq, np.sqrt(2.0 * n_cur))
        a_out, _ = peak_shaving(a_norm, t2, rng)

    if t3 > 60:
        a_norm = normalize(best_seq, np.sqrt(2.0 * n_cur))
        a_out, _ = nesterov_iteration(a_norm, t3, solver, seed=123)

    # === Phase 6: Final peak shaving ===
    n_cur = len(best_seq)

        return [1.0]
