import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None


def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.PDLP
    print(f"  Solver (fallback): {cp.PDLP}")
    return _solver


def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_toeplitz(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T


def lp_best_response(a, M, solver, active_indices=None):
    """Solve max sum(b) s.t. T(a)@b <= M, b >= 0"""
    n = len(a)
    size = 2 * n - 1
    if active_indices is not None and 0 < len(active_indices) < size:
        T_sub = build_toeplitz_row_subset(a, active_indices)
    else:
        T_sub = build_toeplitz(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_sub @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    return val, prob


def build_toeplitz_row_subset(a, row_indices):
    """Build only the specified rows of the Toeplitz matrix."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((len(row_indices), n))
    a_full = np.zeros(size)
    a_full[:n] = a
    for row_idx, k in enumerate(row_indices):
        lo = max(0, k - n + 1)
        hi = min(k + 1, n)
        for i in range(lo, hi):
            T[row_idx, i] = a_full[k - i]
    return T


def lse_smooth_descent(a0, time_limit, verbose=True):
    """LSE-smoothed gradient descent for fast initial progress."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 10
    temps = np.geomspace(50.0, 0.3, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(20, min(200, int(remaining * 8)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -80.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            step = norm / (T_val * np.sqrt(n)) * (1.0 / (1.0 + 0.03 * it))

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.1f}: best={best_score:.8f}")

    return best_a, best_score


def nesterov_lp(a0, time_limit, solver, verbose=True, max_active_frac=0.6):
    """LP best-response with Nesterov acceleration and adaptive constraint screening."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1
    active_set = None

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-14:
            break

        # Adaptive constraint screening
        # Include constraints that are near-active (>= 85% of max) or top 30%
        threshold = M * 0.85
        active_candidates = np.where(conv_y >= threshold)[0]
        if len(active_candidates) < 50:
            threshold2 = M * 0.55
            active_candidates = np.where(conv_y >= threshold2)[0]
        # Also always include center and edges
        must_include = np.array([0, 1, 2, n - 2, n - 1, size - 2, size - 1])
        active_set = np.unique(np.concatenate([active_candidates, must_include]))
        # Cap the active set size
        max_active = int(size * max_active_frac)
        if len(active_set) > max_active:
            # Keep the most active ones
            vals = conv_y[active_set]
            keep = np.argsort(vals)[-max_active:]
            active_set = active_set[keep]

        b, prob = lp_best_response(y, M, solver, active_set)
        if b is None:
            b, prob = lp_best_response(y, M, solver, None)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Verify dropped constraints
        if len(active_set) < size:
            cross_conv = fast_conv_cross(y, g)
            violated = np.where(cross_conv > M * 1.005)[0]
            if len(violated) > 5:
                # Re-solve with all violated constraints added
                active_set2 = np.unique(np.concatenate([active_set, violated]))
                b2, _ = lp_best_response(y, M, solver, active_set2)
                if b2 is not None:
                    b = b2
                    b_sum = b.sum()
                    if b_sum < 1e-15:
                        continue
                    g = b * norm / b_sum

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M:.6f}")

    return best_a, best_score

def structured_init(n, rng, style=None):
    """Generate structured initial sequence with asymmetric properties."""
    x = np.linspace(0.0, 1.0, n)
    if style is None:
        style = rng.integers(0, 12)

    if style == 0:
        # Exponential decay
        a = np.exp(-3.0 * x)
    elif style == 1:
        # Power law decay
        a = (1.0 - x + 0.05) ** 2.5
    elif style == 2:
        # Asymmetric Gaussian
        a = np.exp(-5.0 * (x - 0.2) ** 2) + 0.3 * np.exp(-5.0 * (x - 0.8) ** 2)
    elif style == 3:
        # Cosine taper (asymmetric)
        a = 0.5 + 0.4 * np.cos(np.pi * (x - 0.3)) ** 2
    elif style == 4:
        # Linear decay with wobble
        a = (1.0 - 0.5 * x) + 0.15 * np.sin(4 * np.pi * x)
    elif style == 5:
        # Bump at edge
        a = np.exp(-8.0 * x) + 0.4 * (1.0 - x)
    elif style == 6:
        # Two-hump asymmetric
        a = np.exp(-10.0 * (x - 0.15) ** 2) + 0.6 * np.exp(-10.0 * (x - 0.7) ** 2)
    elif style == 7:
        # Sigmoid-like
        a = 1.0 / (1.0 + np.exp(6.0 * (x - 0.4)))
    elif style == 8:
        # Polynomial
        a = (1.0 - x) ** 1.5 + 0.1 * x
    elif style == 9:
        # Mixed
        a = 0.3 + 0.5 * np.exp(-4.0 * x) + 0.2 * np.cos(3 * np.pi * x) ** 2
    elif style == 10:
        # Uniform with small perturbation
        a = np.ones(n) + 0.1 * rng.standard_normal(n)
    else:
        # Pure noise (positive)
        a = np.exp(0.5 * rng.standard_normal(n))

    a = np.maximum(a, 0.0)
    a = a + 0.005 * rng.random(n)  # Ensure strictly positive
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def eval_fn(a):

            if val < 0 or not np.isfinite(val):

    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.12, 180.0)
    print(f"\n=== P3: Peak Shave ({t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = peak_shave(best_seq, t3, rng, verbose=True)
        try_update(a_out, "peak1")

    # === Phase 4: LP Nesterov round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.25, 350.0)
    print(f"\n=== P4: LP Nesterov r2 ({t4:.0f}s) ===")
    if t4 > 40:
        a_out, _ = nesterov_lp(best_seq, t4, solver, verbose=True, max_active_frac=0.7)
        try_update(a_out, "lp_r2")

    # === Phase 5: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 120.0)
    print(f"\n=== P5: Coord Descent ({t5:.0f}s) ===")
    if t5 > 15:
        a_out, _ = coord_descent(best_seq, t5, rng, verbose=True)
        try_update(a_out, "cd1")

    # === Phase 6: Try longer length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = min(n_base + 500, 3000)
    t6 = min(remaining * 0.20, 200.0)
    print(f"\n=== P6: Longer n={n_long} ({t6:.0f}s) ===")
    if t6 > 40 and remaining > 100:
        init_long = interpolate_seq(best_seq, n_long)
        t6a = min(t6 * 0.6, 120.0)
        a_out, _ = nesterov_lp(init_long, t6a, solver, verbose=False, max_active_frac=0.4)
        try_update(a_out, f"lp_long{n_long}")

    # === Phase 7: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== P7: Final polish ({remaining:.0f}s) ===")
    if remaining > 30:
        t7a = min(remaining * 0.4, 60.0)
        a_out, _ = lse_smooth_descent(best_seq, t7a, verbose=False)
        try_update(a_out, "lse_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t7b = min(remaining * 0.5, 60.0)
            a_out, _ = nesterov_lp(best_seq, t7b, solver, verbose=True, max_active_frac=0.8)
            try_update(a_out, "lp_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            a_out, _ = peak_shave(best_seq, min(remaining * 0.7, 30.0), rng, verbose=False)
            try_update(a_out, "peak_final")

    # Final
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
