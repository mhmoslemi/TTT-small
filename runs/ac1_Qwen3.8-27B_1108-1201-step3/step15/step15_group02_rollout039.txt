def topk_gradient_descent(a0, time_limit, verbose=True):

    S0 = a0.sum()
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    a = a / S0

    best_score = compute_score_fast(a)
    t0 = time.monotonic()

    n_stages = 30
    k_values = np.geomspace(200, 5, n_stages).astype(int)
    lr_init = 0.3 / np.sqrt(n)

    for si, k_val in enumerate(k_values):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        n_steps = max(20, min(150, int(remaining * 8)))
        lr = lr_init * (1.0 + 0.5 * np.cos(np.pi * si / n_stages))

            # Backtracking line search

                            new_max = max(new_max, new_vals.max())
                        if 2 * i < size and 2 * i != 2 * j:
                            val_2i = conv[2 * i] - 2 * d * ai + d * d
                            if j <= 2 * i < j + n:
                                val_2i += d * a[2 * i - j]
                            new_max = max(new_max, val_2i)
                        if 2 * j < size and 2 * j != 2 * i:
                            val_2j = conv[2 * j] + 2 * d * aj + d * d
                            if i <= 2 * j < i + n:
                                val_2j -= d * a[2 * j - i]
                            new_max = max(new_max, val_2j)
                        if new_max < best_new_max:
                            best_new_max = new_max
                            best_delta = d

                if best_delta > 1e-10 and best_new_max < best_score - 1e-10:
                    a[i] -= best_delta
                    a[j] += best_delta
                    a = np.maximum(a, 0.0)
                    s_new = a.sum()
                    if s_new > 1e-15:
                        a = a / s_new
                    new_real_max = fast_conv_self(a).max()
                    new_real_score = compute_score_fast(a)
                    if new_real_score < best_score:
                        best_score = new_real_score
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        # Revert
                        a[j] -= best_delta
                        a[i] += best_delta

        if not pass_improved:
            no_improve_passes += 1
            if no_improve_passes >= 8:
                # Perturb to escape
                n_pert = max(5, n // 50)
                idx_p = rng.choice(n, size=n_pert, replace=False)
                a[idx_p] *= (1.0 + 0.1 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                if no_improve_passes >= 15:
                    break
        else:
            no_improve_passes = 0

        if verbose and total_improvements % 200 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD improv={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def freq_domain_refine(a0, time_limit, rng, verbose=True):
    """Perturb in frequency domain to flatten the power spectrum."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score_fast(a)
    t0 = time.monotonic()

    fft_size = 1 << (2 * n - 1).bit_length()
    n_iters = 200

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        A = rfft(a, fft_size)
        # Flattening: reduce dynamic range of |A|
        mag = np.abs(A)
        phase = np.angle(A)
        mag_smooth = mag ** 0.9 * (mag.mean() / (mag ** 0.9).mean())
        A_new = mag_smooth * np.exp(1j * phase)
        a_new = np.real(irfft(A_new, fft_size)[:n])
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new

        # Mix with original
        t_mix = 0.1 + 0.3 * rng.random()
        a_trial = (1 - t_mix) * a + t_mix * a_new
        a_trial = np.maximum(a_trial, 0.0)
        a_trial = a_trial / a_trial.sum()

        sc = compute_score_fast(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial.copy()
        else:
            # Accept with some probability (simulated annealing)
            delta = sc - best_score
            prob = np.exp(-delta / (0.001 + 0.005 * (1 - it / n_iters)))
            if rng.random() < prob:
                a = a_trial.copy()

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        f1 = rng.integers(1, 6)
        f2 = rng.integers(6, 20)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * f1 * x) + 0.2 * np.cos(2 * np.pi * f2 * x)
    elif choice == 3:
        c1 = rng.uniform(0.3, 0.5)
        c2 = rng.uniform(0.5, 0.7)
        w = rng.uniform(0.05, 0.15)
        a = np.exp(-(x - c1) ** 2 / (2 * w ** 2)) + np.exp(-(x - c2) ** 2 / (2 * w ** 2))
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.5
    elif choice == 6:
        n_peaks = rng.integers(3, 8)
        centers = rng.uniform(0.1, 0.9, n_peaks)
        a = np.zeros(n)
        for c in centers:
            w = rng.uniform(0.03, 0.12)
            a += np.exp(-(x - c) ** 2 / (2 * w ** 2))
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        a = np.exp(-3.0 * x) * (1 - x) + 0.3
    else:
        a = 1.0 + 0.2 * np.sin(2 * np.pi * 2 * x) + 0.1 * np.cos(2 * np.pi * 5 * x)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def symmetrize(a):
    a_sym = (a + a[::-1]) / 2.0
    s = a_sym.sum()
    if s < 1e-15:
        return a.copy()
    return a_sym / s


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            old_best = best_score
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (was {old_best:.6f}, n={len(a)}, t={elapsed:.0f}s)")

    n_current = len(best_seq)

    # === Phase 1: TopK gradient descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 1: TopK-GD n={n_current} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = topk_gradient_descent(best_seq, t1, verbose=True)
        check_and_update(a_out, f"topk1_n{n_current}")

    # === Phase 2: Pair CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.30, 480.0)
    print(f"\n=== Phase 2: Pair CD n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = pair_coordinate_descent(best_seq, t2, rng, verbose=True)
        check_and_update(a_out, f"pcd1_n{len(best_seq)}")

    # === Phase 3: Freq domain refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 180.0)
    print(f"\n=== Phase 3: Freq Refine n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = freq_domain_refine(best_seq, t3, rng, verbose=False)
        check_and_update(a_out, f"freq_n{len(best_seq)}")

    # === Phase 4: TopK-GD round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 4: TopK-GD round 2 n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = topk_gradient_descent(best_seq, t4, verbose=False)
        check_and_update(a_out, f"topk2_n{len(best_seq)}")

    # === Phase 5: Pair CD round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 5: Pair CD round 2 n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 30:
        a_out, _ = pair_coordinate_descent(best_seq, t5, rng, verbose=True)
        check_and_update(a_out, f"pcd2_n{len(best_seq)}")

    # === Phase 6: Symmetrization and sparsification ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Sym/Sparse n={len(best_seq)} ===")
    a_sym = symmetrize(best_seq)
    check_and_update(a_sym, f"sym_n{len(best_seq)}")

    # Try mild sparsification
    for thresh in [0.005, 0.002, 0.001]:
        a_spar = best_seq.copy()
        a_norm = a_spar / a_spar.sum()
        a_spar[a_norm < thresh * a_norm.max()] = 0.0
        s_spar = a_spar.sum()
        if s_spar > 0.01:
            a_spar = a_spar / s_spar
            check_and_update(a_spar, f"sparse_{thresh}_n{len(best_seq)}")

    # === Phase 7: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 7: Multi-length (remaining={remaining:.0f}s) ===")
        lengths_to_try = [n_current + 200, n_current - 200, n_current + 400, n_current - 400]
        for n_try in lengths_to_try:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 40:
                break
            if n_try < 400 or n_try > 4000:
                continue
            t_try = min(remaining * 0.4, 45.0)
            print(f"  Trying n={n_try} (budget={t_try:.0f}s)")
            init_try = interpolate_seq(best_seq, n_try)
            a_out, _ = topk_gradient_descent(init_try, min(t_try * 0.6, 25.0), verbose=False)
            check_and_update(a_out, f"ml_topk_n{n_try}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20:
                a_out, _ = pair_coordinate_descent(a_out, min(remaining * 0.4, 15.0), rng, verbose=False)
                check_and_update(a_out, f"ml_pcd_n{n_try}")

    # === Phase 8: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 8: Final Polish (remaining={remaining:.0f}s) ===")
        t_final = min(remaining * 0.5, 60.0)
        a_out, _ = pair_coordinate_descent(best_seq, t_final, rng, verbose=False)
        check_and_update(a_out, "final_pcd")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_sym2 = symmetrize(best_seq)
            check_and_update(a_sym2, "final_sym")
            a_out, _ = topk_gradient_descent(best_seq, min(remaining * 0.5, 15.0), verbose=False)
            check_and_update(a_out, "final_topk")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
