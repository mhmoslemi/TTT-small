def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):

    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def lse_adam_optimize(a0, time_limit, verbose=True):
    """LSE (Log-Sum-Exp) smooth approximation of max with Adam optimizer."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_adam = 0
    
    # More stages with finer temperature control
    n_stages = 30
    temps = np.geomspace(100.0, 0.01, n_stages)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(50, min(300, int(remaining * 15)))
        lr = 0.3 / np.sqrt(n)
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            t_adam += 1
            
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            
            # LSE smooth max
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            
            # Gradient of LSE: 2*(a*w convolved, both directions)
            ca = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (ca + aw)
            
            # Normalize gradient
            grad_norm = np.linalg.norm(grad)
            if grad_norm > 1e-10:
                grad = grad / grad_norm
            
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_adam)
            v_hat = v / (1.0 - beta2 ** t_adam)
            
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            
            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    return best_a, best_score

def build_constraint_matrix(a, constraint_indices):
    """Build the constraint matrix for the LP using specific constraint indices."""
    n = len(a)
    T = np.zeros((len(constraint_indices), n))
    for idx, k in enumerate(constraint_indices):
        k = int(k)
        row_start = max(0, k - n + 1)
        row_end = min(n, k + 1)
        for i in range(row_start, row_end):
            T[idx, i] = a[k - i]
    return T

def lp_best_response(a, M, solver, active_set=None):
    """Solve the LP: max sum(b) s.t. b >= 0, T(a, active_set) @ b <= M."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    
    if active_set is not None and 0 < len(active_set) < size:
        T = build_constraint_matrix(a, active_set)
        constraints = [T @ b_var <= M]
    else:
        # Full constraint set
        T = build_constraint_matrix(a, np.arange(size))
        constraints = [T @ b_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best-response method."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1
    
    # Adaptive parameters
    nesterov_params = [
        (0.5, 0.7, 300),   # (threshold_frac, active_set_cap, max_iter)
        (0.6, 0.6, 300),
        (0.7, 0.5, 300),
        (0.8, 0.4, 300),
        (0.9, 0.3, 300),
    ]
    
    for param_set in nesterov_params:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 30:
            break
        thresh_frac, cap_frac, max_iter = param_set
        n_iters = min(max_iter, int(remaining * 2))
        
        for t in range(1, n_iters + 1):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv_y = fast_conv_self(y)
            M = conv_y.max()
            if M < 1e-12:
                break
            
            # Active set: top constraints
            threshold = M * thresh_frac
            active_set = np.where(conv_y > threshold)[0]
            
            # Cap the active set size
            if len(active_set) > int(size * cap_frac):
                # Keep only the top ones
                top_k = int(size * cap_frac)
                active_set = np.argsort(conv_y)[-top_k:]
            
            if len(active_set) < 5:
                active_set = None
            
            b = lp_best_response(y, M, solver, active_set)
            
            if b is None:
                b = lp_best_response(y, M, solver, None)
            
            if b is None:
                continue
            
            # Verify and add violated constraints
            if active_set is not None and len(active_set) < size:
                check = fast_conv_cross(y, b)
                violated = np.where(check > M * 1.001)[0]
                if len(violated) > 0:
                    active_set = np.union1d(active_set, violated)
                    if len(active_set) > int(size * 0.5):
                        active_set = np.argsort(check)[-int(size * 0.5):]
                    b2 = lp_best_response(y, M, solver, active_set)
                    if b2 is not None:
                        b = b2
            
            b_sum = b.sum()
            if b_sum < 1e-15:
                continue
            g = b * norm / b_sum
            
            # Nesterov step
            alpha = 2.0 / (t + 2.0)
            y_new = (1.0 - alpha) * y + alpha * g
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y
            
            if t == 1:
                a_new = g.copy()
            else:
                momentum = (t - 1.0) / (t + 2.0)
                a_new = y_new + momentum * (y_new - y)
                a_new = np.maximum(a_new, 0.0)
                s_a = a_new.sum()
                if s_a < 1e-15:
                    a_new = y_new.copy()
                else:
                    a_new = a_new * norm / s_a
            
            y = y_new
            a = a_new
            
            sc = fast_conv_self(y).max()
            if sc < best_score:
                best_score = sc
                best_a = y.copy()
            
            sc_a = fast_conv_self(a).max()
            if sc_a < best_score:
                best_score = sc_a
                best_a = a.copy()
            
            if verbose and t % 20 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M:.6f}")
    
    return best_a, best_score

                                    dk -= 2.0 * d * d
                                v = max(v, conv[check_k] + dk)
                        if v < best_v:
                            best_v = v
                            best_delta = d
                            best_j = j
            
            if best_j >= 0 and best_delta > 0 and best_v < best_score * 1.0001:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score * 1.0001:
                        if new_max < best_score:
                            best_score = new_max
                            best_a = a.copy()
                        pass_improved = True
                    else:
                        # Revert
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()
        # End for idx
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 8:
                # Perturb and restart
                n_perturb = max(10, n // 50)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.1 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                if no_improve >= 15:
                    break
        else:
            no_improve = 0
    
    return best_a, best_score

def vegas_optimize(a0, time_limit, verbose=True):
    """Vegas-style multiplicative updates targeting high-convolution regions."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    n_stages = 30
    etas = np.geomspace(0.2, 0.0005, n_stages)
    
    for si, eta_s in enumerate(etas):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(30, min(100, int(remaining * 5)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = fast_conv_self(a)
            M = conv.max()
            if M < 1e-15:
                break
            
            # Focus on the top portion of the convolution
            threshold = M * 0.85
            w = np.maximum(conv - threshold, 0.0)
            ww = w / (w.sum() + 1e-15)
            
            W = rfft(ww, fft_size)
            A = rfft(a, fft_size)
            
            ca = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)[:n]
            pressure = ca + aw
            
            # Multiplicative update with floor
            a = a * np.exp(-eta_s * pressure / (pressure.max() + 1e-10))
            a = np.maximum(a, 1e-10)
            a = a * norm / a.sum()
            
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    return best_a, best_score

def structured_init(n, rng):

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def symmetrize(a):
    """Make sequence symmetric."""
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    s = a_sym.sum()
    if s < 1e-15:
        return a
    return a_sym / s * (a.sum() / s)

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_adam)
            v_hat = v / (1.0 - beta2 ** t_adam)

            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new

            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose and (si % 7 == 0 or si == n_stages - 1):
            print(f"    LSE stage {si}/{n_stages} T={T_val:.4f}: best={best_score:.8f}")

    return best_a, best_score

def build_constraint_rows(a, indices):
    n = len(a)
    T = np.zeros((len(indices), n))
    for idx, k in enumerate(indices):
        k = int(k)
        row_start = max(0, k - n + 1)
        row_end = min(n, k + 1)
        if row_end > row_start:
            T[idx, row_start:row_end] = a[k - np.arange(row_start, row_end)]
    return T

def lp_solve(a, M, solver, active_indices):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_constraint_rows(a, active_indices)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1

    max_active = min(int(size * 0.4), 1200)
    min_active = 50

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Adaptive active set: start with top violators, grow over time
        frac = 0.5 + 0.4 * min(1.0, t / 200.0)
        threshold = M * (1.0 - frac)
        active = np.where(conv_y > threshold)[0]

        if len(active) > max_active:
            active = np.argsort(conv_y)[-max_active:]
        if len(active) < min_active:
            active = np.argsort(conv_y)[-min_active:]

        b = lp_solve(y, M, solver, active)

        if b is None:
            b = lp_solve(y, M, solver, np.argsort(conv_y)[-min_active:])

        if b is None:
            continue

        # Constraint verification and growth
        check = fast_conv_cross(y, b)
        violated = np.where(check > M * 1.0005)[0]
        if len(violated) > 0:
            new_active = np.union1d(active, violated)
            if len(new_active) > max_active:
                new_active = np.argsort(check)[-max_active:]
            b2 = lp_solve(y, M, solver, new_active)
            if b2 is not None:
                b = b2

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if verbose and t % 25 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M:.6f}, active={len(active)}")

    return best_a, best_score

def peak_cd_optimize(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        peak_pos = int(np.argmax(conv))
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-8:
                continue

            # Check if a[i] contributes to the peak
            j_peak = peak_pos - i
            if not (0 <= j_peak < n and a[j_peak] > 1e-10):
                j_peak2 = peak_pos - (i + n)
                if not (0 <= j_peak2 < n - i - 1 and a[j_peak2 + i + 1] > 1e-10):
                    continue

            # Find candidate j's that don't contribute to the peak
            j_candidates = []
            for _ in range(10):
                j = int(rng.integers(n))
                j_peak_j = peak_pos - j
                if not (0 <= j_peak_j < n and a[j_peak_j] > a.max() * 0.3):
                    j_candidates.append(j)

            # Also try neighbors of peak-free positions
            for off in range(-3, 4):
                j = (peak_pos + off) % n
                if j != i:
                    j_candidates.append(j)

            best_delta = 0.0
            best_v = cur_max
            best_j = -1

            for j in j_candidates:
                if j == i:
                    continue
                d_max = min(ai, norm * 0.1)
                if d_max < 1e-10:
                    continue
                n_samples = 12
                deltas = np.linspace(0.005, d_max, n_samples)

                for d in deltas:
                    # Compute change in conv at peak
                    dp = 0.0
                    if 0 <= peak_pos - j < n:
                        dp += a[peak_pos - j]
                    if 0 <= peak_pos - i < n:
                        dp -= a[peak_pos - i]
                    new_peak = cur_max + 2.0 * d * dp
                    if peak_pos == 2 * j:
                        new_peak += d * d
                    if peak_pos == 2 * i:
                        new_peak += d * d
                    if peak_pos == i + j:
                        new_peak -= 2.0 * d * d

                    if new_peak >= best_v:
                        continue

                    # Spot check a few other positions
                    v = new_peak
                    for ck in [peak_pos - 1, peak_pos + 1, 2 * i, 2 * j, i + j]:
                        if 0 <= ck < size:
                            dk = 0.0
                            if 0 <= ck - j < n:
                                dk += a[ck - j]
                            if 0 <= ck - i < n:
                                dk -= a[ck - i]
                            v = max(v, conv[ck] + 2.0 * d * dk)
                            if ck == 2 * j or ck == i + j:
                                pass
                    if v < best_v:
                        best_v = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 6:
                n_perturb = max(8, n // 60)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.12 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                if no_improve >= 12:
                    break
        else:
            no_improve = 0

        if verbose and (t0 and int(time.monotonic() - t0) % 30 == 0):
            elapsed = time.monotonic() - t0
            print(f"    PCD ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def vegas_optimize(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 30
    etas = np.geomspace(0.15, 0.0003, n_stages)

    for si, eta_s in enumerate(etas):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        n_steps = max(25, min(80, int(remaining * 4)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_conv_self(a)
            M = conv.max()
            if M < 1e-15:
                break

            threshold = M * 0.80
            w = np.maximum(conv - threshold, 0.0)
            ww = w / (w.sum() + 1e-15)

            W = rfft(ww, fft_size)
            A = rfft(a, fft_size)

            ca = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)[:n]
            pressure = ca + aw
            pmax = pressure.max()
            if pmax < 1e-10:
                break

            a = a * np.exp(-eta_s * pressure / pmax)
            a = np.maximum(a, 1e-10)
            a = a * norm / a.sum()

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score

def structured_init(n, rng):
    choice = rng.integers(0, 14)
    x = np.linspace(0.0, 1.0, n)

    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        f1 = rng.integers(1, 6)
        f2 = rng.integers(6, 20)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * f1 * x) + 0.2 * np.cos(2 * np.pi * f2 * x)
    elif choice == 3:
        c1 = rng.uniform(0.2, 0.5)
        c2 = rng.uniform(0.5, 0.8)
        a = np.exp(-30.0 * (x - c1) ** 2) + np.exp(-30.0 * (x - c2) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.5
    elif choice == 6:
        a = np.exp(-5.0 * x) + 0.4
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        n_peaks = rng.integers(3, 10)
        centers = rng.uniform(0.05, 0.95, n_peaks)
        a = np.zeros(n)
        for c in centers:
            width = rng.uniform(0.02, 0.1)
            a += np.exp(-(x - c) ** 2 / (2 * width ** 2))
    elif choice == 9:
        a = np.sinc(x - 0.5) ** 2 + 0.1
    elif choice == 10:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * x) + 0.2 * np.sin(2 * np.pi * 7 * x)
    elif choice == 11:
        a = np.exp(-20.0 * (x - 0.5) ** 2) + 0.2
    elif choice == 12:
        a = np.ones(n)
        for k in range(4):
            freq = rng.integers(2, 30)
            amp = rng.uniform(0.05, 0.15)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.sin(2 * np.pi * freq * x + phase)
    else:
        n_fourier = rng.integers(3, 8)
        coeffs = rng.standard_normal(n_fourier)
        freqs = rng.integers(1, 20, n_fourier)
        a = np.ones(n)
        for c, f in zip(coeffs, freqs):
            a += c * 0.1 * np.cos(2 * np.pi * f * x)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a = a + 0.003 * np.abs(rng.standard_normal(n))
    a = a / a.sum()
    return a

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def symmetrize(a):
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    s = a_sym.sum()
    if s < 1e-15:
        return a
    return a_sym / s * (a.sum() / s)

def propose_candidate(seed=42, budget_s=1800.0):
    """Main search: LSE warmup -> Vegas -> LP Nesterov (x2) -> Peak CD -> Multi-length -> Fresh starts."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            old_best = best_score
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (was {old_best:.6f}, n={len(a)}, t={elapsed:.0f}s)")

    # Phase 1: LSE-Adam warmup (fast, gets us into a good basin)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.12, 200.0)
    print(f"\n=== Phase 1: LSE-Adam n={n_base} ({t1:.0f}s) ===")
    if t1 > 15:
        a_out, _ = lse_adam_optimize(best_seq, t1, verbose=True)
        check_and_update(a_out, f"lse1_n{n_base}")

    # Phase 2: Vegas multiplicative (smooths out local structure)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.08, 120.0)
    print(f"\n=== Phase 2: Vegas n={len(best_seq)} ({t2:.0f}s) ===")
    if t2 > 15:
        a_out, _ = vegas_optimize(best_seq, t2, verbose=False)
        check_and_update(a_out, f"vegas_n{len(best_seq)}")

    # Phase 3: LP Nesterov - main optimizer (most time budget)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.38, 550.0)
    print(f"\n=== Phase 3: LP Nesterov n={len(best_seq)} ({t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t3, solver, verbose=True)
        check_and_update(a_out, f"lp1_n{len(best_seq)}")

    # Phase 4: Peak CD (targeted reduction of convolution peaks)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 4: Peak CD n={len(best_seq)} ({t4:.0f}s) ===")
    if t4 > 15:
        a_out, _ = peak_cd_optimize(best_seq, t4, rng, verbose=True)
        check_and_update(a_out, f"pcd1_n{len(best_seq)}")

    # Phase 5: LP Nesterov round 2 (refine from improved point)
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.20, 280.0)
    print(f"\n=== Phase 5: LP Nesterov r2 n={len(best_seq)} ({t5:.0f}s) ===")
    if t5 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t5, solver, verbose=True)
        check_and_update(a_out, f"lp2_n{len(best_seq)}")

    # Phase 6: Symmetrization + quick LSE
    a_sym = symmetrize(best_seq)
    check_and_update(a_sym, f"sym_n{len(best_seq)}")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        a_out, _ = lse_adam_optimize(a_sym, min(remaining * 0.3, 60.0), verbose=False)
        check_and_update(a_out, f"lse_sym_n{len(best_seq)}")

    # Phase 7: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        print(f"\n=== Phase 7: Multi-length ({remaining:.0f}s remaining) ===")
        offsets = [-150, -50, 50, 150, 300]
        for off in offsets:
            if remaining < 40:
                break
            n_try = n_base + off
            if n_try < 500 or n_try > 4000:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            t_try = min(remaining * 0.2, 40.0)
            print(f"  n={n_try} ({t_try:.0f}s)")
            init_try = interpolate_seq(best_seq, n_try)
            a_out, _ = lse_adam_optimize(init_try, min(t_try * 0.5, 20.0), verbose=False)
            check_and_update(a_out, f"ml_n{n_try}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20:
                a_out, _ = peak_cd_optimize(a_out, min(remaining * 0.3, 20.0), rng, verbose=False)
                check_and_update(a_out, f"mlpcd_n{n_try}")

    # Phase 8: Fresh starts
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 50:
        print(f"\n=== Phase 8: Fresh starts ({remaining:.0f}s) ===")
        for attempt in range(3):
            if remaining < 20:
                break
            n_fresh = n_base + int(rng.integers(-200, 300))
            if n_fresh < 600:
                n_fresh = 1000
            a_fresh = structured_init(n_fresh, rng)
            t_fresh = min(remaining * 0.25, 18.0)
            a_out, _ = lse_adam_optimize(a_fresh, t_fresh, verbose=False)
            check_and_update(a_out, f"fresh{attempt}_n{n_fresh}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 12:
                a_out, _ = peak_cd_optimize(a_out, min(remaining * 0.3, 12.0), rng, verbose=False)
                check_and_update(a_out, f"freshcd{attempt}_n{n_fresh}")

    # Final LP polish with remaining time
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Final LP ({remaining:.0f}s) ===")
        a_out, _ = nesterov_lp_optimize(best_seq, max(remaining - 5, 10), solver, verbose=True)
        check_and_update(a_out, "lp_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
