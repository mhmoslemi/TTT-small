    norm = np.sqrt(2.0 * n)
    a = np.maximum(a_init, 0)
    a = a * norm / a.sum()

    x = a.copy()

        alpha = 2.0 / (t + 2.0)
        y_new = (1 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0)
        y_new = y_new * norm / y_new.sum()
        
        x_new = y_new
        if t > 1:
            beta = (t - 1) / (t + 2)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0)
            x_new = x_new * norm / x_new.sum()

        x = x_new

        for candidate in [y, x]:
            s = compute_score(candidate)
            if s < best_score:
                best_score = s
                best = candidate.copy()
    
    return best, best_score

            # We want grad_i = d/da_i [sum_k w_k * conv_full[k]]

            # = sum_k w_k * [a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n, i.e., i>=k)]
            # 
            # First sum: sum_{k=i}^{i+n-1} w_k * a[k-i] (where k-i ranges 0 to n-1)

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
