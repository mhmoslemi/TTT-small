def fast_convolve(a):
    """Compute a * a (full convolution) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def smooth_objective_and_grad(a, beta):
    """
    Compute the smooth approximation of max(conv(a,a)) and its gradient.
    Returns (value, gradient).
    """
    n = len(a)
    conv = fast_convolve(a)
    x = beta * conv
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    sum_exp = exp_x.sum()
    value = (x_max + np.log(sum_exp)) / beta
    
    # Softmax weights
    p = exp_x / sum_exp
    
    # Gradient: grad_a[k] = 2 * sum_j p[k+j] * a[j] for k = 0,...,n-1
    # This is 2 * cross_correlation(p, a)[:n]
    N = 1 << (3 * n - 2 - 1).bit_length()
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    cross_corr = np.fft.irfft(np.conj(A) * P, N)
    grad = 2.0 * cross_corr[:n]
    
    return value, grad


def lbfgs_optimize(a0, beta, time_limit, maxiter=2000):
    """Run L-BFGS-B on the smooth objective with bounds a >= 0."""
    n = len(a0)
    a = norm_sum1(a0)
    t0 = time.monotonic()
    
    def fun(z):
        if time.monotonic() - t0 > time_limit:
            return 1e10  # Signal to stop
        z = np.maximum(z, 0)
        s = z.sum()
        if s < 1e-15:
            return 1e10
        z = z / s
        val, _ = smooth_objective_and_grad(z, beta)
        return val

        s = z.sum()
        if s < 1e-15:

        # Actually, since the objective is scale-invariant, the gradient
        # in the z-space is just the gradient in the a-space
        # (the scale direction is flat)
        return g

        fun, a, method='L-BFGS-B', jac=grad,

    )
    
    if result.success or result.fun < 1e9:

    return a, fun(a)
