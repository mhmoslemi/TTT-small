import numpy as np
import cvxpy as cp
import time

def fast_autocorr(a):
    """Compute linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def get_solver():
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS

def grad_at_lag(a, k):
    """Compute gradient of b_k with respect to a."""
    n = len(a)
    g = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n, k + 1)
    for i in range(i_start, i_end):
        j = k - i
        if 0 <= j < n:
            g[i] += 2.0 * a[j]
    return g

def amd_exact(a0, time_limit, c0=3.0, top_k_check=5, verbose=True):
    """Nesterov accelerated mirror descent with exact subgradient."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute autocorrelation and find top lags
        ac = fast_autocorr(y)
        
        # Use top-k lags for the subgradient (to reduce oscillation)
        k_max = min(top_k_check, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = ac[top_idx]
        
        # Weighted combination of gradients
        w = top_vals ** 8.0  # Sharp weighting
        w /= w.sum()
        
        g = np.zeros(n)
        for wi, k in zip(w, top_idx):
            g += wi * grad_at_lag(y, int(k))
        
        # Mirror descent step
        eta = c0 / (np.sqrt(t) + 1.0)
        log_ratio = -eta * g
        a_new = y * np.exp(log_ratio)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        # Check improvement
        M_new = fast_autocorr(x).max()
        if M_new < best_m:
            best_m = M_new
            best_a = x.copy()
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def lp_frank_wolfe_lin(a0, time_limit, solver, verbose=True):
    """
    Frank-Wolfe with linearized LP.
    At each step, linearize the autocorrelation at the current point,
    solve an LP to find the best direction, and take a line search step.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()  # Current iterate (average)
    x_nest = a.copy()  # Nesterov point
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute autocorrelation at Nesterov point
        ac = fast_autocorr(x_nest)
        M_curr = ac.max()
        
        # Identify active lags (within 5% of max)
        threshold = 0.95 * M_curr
        active_lags = np.where(ac >= threshold)[0]
        
        # Cap the number of active lags
        if len(active_lags) > 80:
            active_lags = np.argsort(ac)[::-1][:80]
        
        # Build the LP
        v_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        
        constraints = [
            cp.sum(v_var) == 1,
        ]
        
        for k in active_lags:
            k_int = int(k)
            # Linearized constraint: b_k(x_nest) + grad_k(x_nest) @ (v - x_nest) <= M
            # = b_k(x_nest) - 2*b_k(x_nest) + 2 * sum_j x_nest[k-j] * v[j] <= M
            # = -b_k(x_nest) + 2 * sum_j x_nest[k-j] * v[j] <= M
            grad = grad_at_lag(x_nest, k_int)
            constraints.append(cp.sum(grad * v_var) - M_var <= ac[k_int])
        
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov accelerated mirror descent with exact/top-k subgradient
2. Linearized LP-based Frank-Wolfe (new targeted approach)

import numpy as np
import cvxpy as cp
import time


def fast_autocorr(a):
    """Compute linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def get_solver():
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    print("  LP solver (fallback): ECOS")
    return cp.ECOS


def grad_at_lag(a, k):
    """Compute gradient of b_k(a) = sum_j a[j]*a[k-j] with respect to a."""
    n = len(a)
    g = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n, k + 1)
    for i in range(i_start, i_end):
        j = k - i
        if 0 <= j < n:
            g[i] += 2.0 * a[j]
    return g


def amd_exact(a0, time_limit, c0=3.0, top_k=3, beta_w=8.0, verbose=True):
    """Nesterov accelerated mirror descent with weighted top-k subgradient."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    x = a.copy()
    x_prev = a.copy()

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta_w
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

            g += wi * grad_at_lag(y, int(k))

        a_new = y * np.exp(log_ratio)

        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:min(15, len(ac))]
        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)

            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            n_pairs = i_end - i_start

            if n_pairs < 2:
                continue

            # Sample pairs to keep it fast
            if n_pairs > 300:
                indices = rng.choice(np.arange(i_start, i_end), size=300, replace=False)
            else:
                indices = np.arange(i_start, i_end)

            # Sort by product a[i]*a[k-i] descending (target largest products first)
            products = np.array([a[i] * a[k_int - i] for i in indices if 0 <= k_int - i < n and k_int - i != i])
            valid_indices = [i for i in indices if 0 <= k_int - i < n and k_int - i != i]
            if len(valid_indices) == 0:
                continue

            # Sort by product
            prod_arr = np.array([a[i] * a[k_int - i] for i in valid_indices])
            order = np.argsort(prod_arr)[::-1][:50]

            for idx in order:
                if time.monotonic() - t0 > time_limit:
                    break
                i = valid_indices[idx]
                j = k_int - i
                if j < 0 or j >= n or j == i:
                    continue

                ai = a[i]
                aj = a[j]
                if ai < 1e-12 or aj < 1e-12:
                    continue

                # Try moving mass from i to j (or j to i) to reduce the product
                # If ai > aj, move from i to j; else move from j to i
                if ai > aj:
                    src, dst = i, j
                    asrc, adst = ai, aj
                else:
                    src, dst = j, i
                    asrc, adst = aj, ai

                if asrc < 1e-12:
                    continue

                # Optimal: equalize a[src] and a[dst]
                # New values: (asrc+adst)/2 each, delta = (asrc-adst)/2
                delta_opt = (asrc - adst) / 2.0
                if delta_opt < 1e-14:
                    continue

                found = False
                for frac in [0.5, 1.0, 0.3, 0.7, 0.8, 0.2, 0.9]:
                    delta = delta_opt * frac
                    if delta < 1e-14:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= delta
                    a_trial[dst] += delta
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t

                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break

                if found:
                    break

            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.015)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                else:
                    a[i] += amt
                    a[j] -= amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()

        if verbose and (stall % 30 == 0 or (stall == 1 and t % 1 == 0)):
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.1f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def amd_restart(a0, time_limit, c0=3.0, top_k=5, beta_w=6.0, verbose=False):
    """AMD with periodic random restarts to escape local minima."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 8000

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta_w
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        g = np.zeros(n)
        for wi, k in zip(w, top_idx):
            g += wi * grad_at_lag(y, int(k))

        eta = c0 / (np.sqrt(t) + 1.0)
        log_ratio = -eta * g
        a_new = y * np.exp(log_ratio)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            stall = 0
            # Restart from best with perturbation
            pert = rng.normal(0, 0.005, n)
            x = best_a * (1 + pert)
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()

    return best_a, best_m


def try_update_state(state, a, label):
    """Update best state if a is better. state = [best_seq, best_score]."""
    if a is None or len(a) == 0:
        return
    a = normalize(a)
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < state[1]:
        state[1] = sc
        state[0] = a.copy()
        elapsed = time.monotonic() - state[2]
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    # State: [best_seq, best_score, t_global]
    state = [None, float('inf'), t_global]

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            state[0] = parent
            state[1] = p_score
    except Exception:
        pass

    if state[0] is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        init = np.sin(np.pi * x) ** 2
        init = init / init.sum()
        state[0] = init
        state[1] = evaluate_safe(init)
        print(f"Default init: n={n_base}, score={state[1]:.6f}")

    n0 = len(state[0])

    # === Phase 1: AMD exact (fast, gets close) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: AMD n={n0} ===")
    t1 = min(remaining * 0.22, 350.0)
    if t1 > 30:
        a_out, _ = amd_exact(state[0], t1, c0=3.5, top_k=3, beta_w=10.0, verbose=True)
        try_update_state(state, a_out, f"amd1_n{n0}")

    # === Phase 2: LP Frank-Wolfe linearized (targeted) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(state[0])
    print(f"\n=== Phase 2: LP-FW lin n={n_cur} ===")
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 30:
        a_out, _ = lp_frank_wolfe_lin(state[0], t2, solver, verbose=True, n_active_max=50)
        try_update_state(state, a_out, f"lpfw1_n{n_cur}")

    # === Phase 3: Pair-coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(state[0])
    print(f"\n=== Phase 3: Pair-coord n={n_cur} ===")
    t3 = min(remaining * 0.18, 280.0)
    if t3 > 20:
        a_out, _ = pair_coord_descent(state[0], t3, verbose=True, rng_seed=42)
        try_update_state(state, a_out, f"pair_n{n_cur}")

    # === Phase 4: AMD with restarts ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(state[0])
    print(f"\n=== Phase 4: AMD restart n={n_cur} ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 20:
        a_out, _ = amd_restart(state[0], t4, c0=2.5, top_k=5, beta_w=6.0, verbose=False)
        try_update_state(state, a_out, f"amdr_n{n_cur}")

    # === Phase 5: LP-FW second pass ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(state[0])
    if n_cur <= 3000 and remaining > 60:
        print(f"\n=== Phase 5: LP-FW 2nd n={n_cur} ===")
        t5 = min(remaining * 0.20, 300.0)
        if t5 > 30:
            a_out, _ = lp_frank_wolfe_lin(state[0], t5, solver, verbose=True, n_active_max=40)
            try_update_state(state, a_out, f"lpfw2_n{n_cur}")

    # === Phase 6: Final pair-coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(state[0])
    print(f"\n=== Phase 6: Final coord n={n_cur} ===")
    if remaining > 30:
        t6 = min(remaining * 0.85, 350.0)
        if t6 > 15:
            a_out, _ = pair_coord_descent(state[0], t6, verbose=True, rng_seed=777)
            try_update_state(state, a_out, f"coord_fin_n{n_cur}")

    # Final validation
    if state[0] is not None:
        fs = evaluate_safe(state[0])
        if np.isfinite(fs):
            state[1] = fs
        print(f"\nFINAL: score={state[1]:.6f}, n={len(state[0])}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in state[0]]
