# Global solver
_solver = None

def get_solver():

    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def normalize_sum(a, target):
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) * target / len(a)
    return a * target / s

def build_T(a):
    """Build the convolution matrix T where T[k, j] = a[k-j] if 0 <= k-j < n, else 0."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m+n, m] = a
    return T

def lp_best_response(a, M, solver, active_set=None):
    """Solve the best-response LP: max sum(b) s.t. T@b <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    if active_set is not None and 0 < len(active_set) < size:
        constraints = [T[active_set, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        if prob.status == 'infeasible':
            return None
        # Try again with all constraints
        constraints = [T @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

            prob.solve(solver=solver, verbose=False)
        except Exception:
            return None
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)

def lse_gd(a0, time_limit, n_stages=10, verbose=False):
    """LSE (Log-Sum-Exp) smoothed gradient descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n).bit_length()
    size = 2 * n - 1
    
    temps = np.geomspace(50.0, 0.3, n_stages)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        
        n_steps = max(20, min(200, int(remaining * 8)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -80.0)
            w = np.exp(shifted)
            w = w / w.sum()
            
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)
            
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            
            # Adaptive step size
            step = norm / (T_val * np.sqrt(n) * (1.0 + 0.02 * it))
            a = a - step * grad
            a = np.maximum(a, 0.0)
            a = normalize_sum(a, norm)
            
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    return best_a, best_score

    t0 = time.monotonic()

        # Adaptive constraint screening

        elif t < 500:

        active = np.where(conv_y >= M * frac)[0]

        if b is None:

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        
        # Verify dropped constraints

            check = fast_conv_cross(y, g)
            if check.max() > M * 1.01:
                b_full = lp_best_response(y, M, solver, None)

                    if b_sum > 1e-15:
                        g = b * norm / b_sum
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize_sum(y_new, norm)
        
        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_sum(a_new, norm)
        
        y = y_new
        a = a_new
        
        sc = fast_conv_self(y).max()
        if sc < best_score:
            best_score = sc
            best_a = y.copy()

            best_a = a.copy()
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP-Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def peak_equality_refine(a0, time_limit, verbose=False):
    """Refine by solving for perturbations that equalize top peaks."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_sum(a, norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    
    for iteration in range(200):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_conv_self(a)
        cur_max = conv.max()
        
        # Find top peaks
        top_k = min(30, size)
        peak_indices = np.argsort(conv)[-top_k:][::-1]
        peak_vals = conv[peak_indices]
        
        # Only work with peaks that are close to max
        threshold = peak_vals[0] * 0.90
        active_peaks = peak_indices[peak_vals >= threshold]
        if len(active_peaks) < 3:
            break
        
        # Build the linear system: we want to find delta (length n) such that
        # for each active peak k: sum_j (2 * a[j] * delta[k-j] + ...) = -epsilon
        # This is approximately: T[active_peaks, :] @ delta = -epsilon * ones
        # where T is the convolution matrix
        # But we also want ||delta||_2 to be small, so we add a regularization
        
        T_active = np.zeros((len(active_peaks), n))
        for ii, k in enumerate(active_peaks):
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    T_active[ii, j] = 2 * a[idx]
        
        if len(active_peaks) < n:
            # Underdetermined: use minimum norm solution (Moore-Penrose)
            # min ||delta|| s.t. T_active @ delta = -eps
            eps = cur_max * 0.005  # reduce peaks by 0.5%
            try:
                # Use SVD for minimum norm
                U, s, Vt = np.linalg.svd(T_active, full_matrices=False)
                target = -eps * np.ones(len(active_peaks))
                # Regularized solution
                reg = 0.01 * np.max(s) if len(s) > 0 else 0.01
                s_reg = s + reg
                delta = Vt.T @ ((U.T @ target) / s_reg)
            except Exception:
                delta = None
            
            if delta is not None:
                delta = np.maximum(-a * 0.5, np.minimum(delta, (1.0 - a) * 0.5))
                # Actually, allow both positive and negative changes but keep a >= 0
                a_new = a + delta
                a_new = np.maximum(a_new, 0.0)
                a_new = normalize_sum(a_new, norm)
                new_max = fast_conv_self(a_new).max()
                if new_max < best_score:
                    best_score = new_max
                    best_a = a_new.copy()
                    a = a_new
                elif new_max < cur_max:
                    a = a_new
        else:
            # Overdetermined or square: use least squares
            eps = cur_max * 0.005
            target = -eps * np.ones(len(active_peaks))
            try:
                delta, _, _, _ = np.linalg.lstsq(T_active, target, rcond=None)
            except Exception:
                break
            a_new = a + delta
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_sum(a_new, norm)
            new_max = fast_conv_self(a_new).max()
            if new_max < best_score:
                best_score = new_max
                best_a = a_new.copy()
                a = a_new
    
    return best_a, best_score

def coord_descent(a0, time_limit, rng, verbose=False):
    """Coordinate descent: try moving mass between pairs of coordinates."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_sum(a, norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_imp = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < 1e-12:
                continue
            
            # Try transferring to a few random targets
            best_delta = 0.0
            best_val = cur_max
            best_j = -1
            
            n_targets = 8
            targets = rng.integers(0, n, size=n_targets)
            
            for j in targets:
                if j == i:
                    continue
                # Compute the gain vector for moving mass from i to j
                # conv_new[k] = conv[k] + delta * (a[k-j] - a[k-i]) (approx, ignoring delta^2 terms at 2i, 2j)
                # More precisely: conv_new = conv + delta*(a shifted by j - a shifted by i)
                # At position 2i: conv[2i] changes by -2*ai*delta + delta^2
                # At position 2j: conv[2j] changes by +2*aj*delta + delta^2
                
                gain = np.zeros(size)
                # Effect of adding delta at position j
                for k in range(j, min(j + n, size)):
                    gain[k] += a[k - j]
                for k in range(max(0, j - n + 1), j + 1):
                    gain[k] += a[j - k]
                # Effect of removing delta from position i
                for k in range(i, min(i + n, size)):
                    gain[k] -= a[k - i]
                for k in range(max(0, i - n + 1), i + 1):
                    gain[k] -= a[i - k]
                
                # Sample several delta values
                max_d = min(ai * 0.9, a[j] + ai * 0.5)
                deltas = np.linspace(0, max_d, 10)
                for d in deltas:
                    if d < 1e-13:
                        continue
                    new_conv = conv + d * gain
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * a[j] * d + d * d
                    v = new_conv.max()
                    if v < best_val:
                        best_val = v
                        best_delta = d
                        best_j = j
            
            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = normalize_sum(a, norm)
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-13:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
                    total_imp += 1
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
            if no_improve >= 2:
                # Perturb to escape local minimum
                n_pert = max(5, n // 50)
                idx_p = rng.choice(n, size=n_pert, replace=False)
                a[idx_p] *= (1.0 + 0.1 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0.0)
                a = normalize_sum(a, norm)
        else:
            no_improve = 0
        
        if verbose and total_imp % 500 == 0 and total_imp > 0:
            elapsed = time.monotonic() - t0
            print(f"  CoordDescent imp={total_imp} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def structured_init(n, rng):
    """Generate a structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.1 * np.cos(2 * np.pi * 4 * x)
    elif choice == 3:
        a = np.exp(-5.0 * (x - 0.3) ** 2) + 0.6 * np.exp(-5.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.2
    elif choice == 5:
        a = 0.4 + 0.6 * np.sin(np.pi * x) ** 0.8
    elif choice == 6:
        a = np.exp(-3.0 * x) + 0.3 + 0.2 * np.sin(2 * np.pi * 3 * x)
    else:
        a = np.sinc(2.0 * (x - 0.5)) ** 2 + 0.1
    a = np.maximum(a, 0.0)
    a = a + 0.005 * rng.random(n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_extend(a, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0.0)
    s2 = a_new.sum()
    if s2 < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s2

def symmetrize(a):
    """Return the symmetrized version (a + reverse(a)) / 2."""
    return (a + a[::-1]) / 2.0

def eval_fn(a):
    """Evaluate using the provided evaluate_sequence function."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        if len(a_list) == 0:
            return float('inf')
        for val in a_list:
            if val < 0 or not np.isfinite(val):
                return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')
