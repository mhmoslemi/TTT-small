def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    conv = fast_autocorr(a)
    return float(2.0 * n * conv.max() / (S * S))

def coord_update_fast(a, conv, j, n, size):
    """
    Fast coordinate update. Returns (new_aj, new_conv, improved).
    """
    aj = a[j]
    S = a.sum()
    s_others = S - aj
    
    if s_others < 1e-15 or aj < 0:
        return aj, conv, False
    
    # Build c vector: c[k] = 2*a[k-j] for valid k, except k=2j
    c = np.zeros(size)
    k_start = j
    k_end = min(j + n, size)  # k ranges from j to j+n-1
    length = k_end - k_start
    c[k_start:k_end] = 2.0 * a[0:length]
    
    k2j = 2 * j
    has_e = (0 <= k2j < size)
    if has_e:
        c[k2j] = 0.0
    
    # bg = conv - aj*c - aj^2*e
    bg = conv - aj * c
    if has_e:
        bg[k2j] -= aj * aj
    
    # Current objective
    current_obj = conv.max() / (S * S)
    
    # Try candidates
    uniform_val = s_others / n
    candidates = np.array([0.0, aj*0.5, aj*0.75, aj*1.25, aj*1.5,
                           uniform_val, uniform_val*0.5, uniform_val*2.0])
    candidates = np.clip(candidates, 0.0, 10.0 * uniform_val + 1.0)
    # Filter out values too close to aj
    mask = np.abs(candidates - aj) > 1e-10
    candidates = candidates[mask]
    
    if len(candidates) == 0:
        return aj, conv, False
    
    best_x = aj
    best_obj = current_obj
    
    # Vectorized evaluation of candidates
    for x in candidates:
        new_conv = bg + x * c
        if has_e:
            new_conv[k2j] += x * x
        new_sum = s_others + x
        if new_sum < 1e-15:
            continue
        obj = new_conv.max() / (new_sum * new_sum)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_x = x
    
    if best_x == aj:
        return aj, conv, False
    
    # Update conv
    new_conv = bg + best_x * c
    if has_e:
        new_conv[k2j] += best_x * best_x
    
    return best_x, new_conv, True
