def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

def build_T_matrix(a, constraints_idx):
    n = len(a)
    size = 2 * n - 1
    m = len(constraints_idx)
    T = np.zeros((m, n))
    for row, k in enumerate(constraints_idx):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                T[row, j] = a[idx]
    return T


def solve_lp_maxsum(a, M, active_idx, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_T_matrix(a, active_idx)
    cons = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if val.sum() < 1e-15:
        return None
    return val


def solve_lp_minmax(a, active_idx, solver, norm_sum=1.0):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    T = build_T_matrix(a, active_idx)
    cons = [T @ b_var <= t_var, cp.sum(b_var) == norm_sum, b_var >= 0]
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    t_val = float(t_var.value) if t_var.value is not None else float('inf')
    if val.sum() < 1e-15:
        return None, None
    return val, t_val

        # This is the cross-correlation of weights with a
        # corr_full = ifft(fft(weights, long) * conj(fft(a, long)))
        # Then grad[i] = corr_full[i+n-1] (the linear cross-correlation)
        corr_full = irfft(rfft(weights, fft_size) * np.conj(rfft(a, fft_size)), fft_size)
        # Extract linear part
        for i in range(n):
            idx = (i + n - 1) % fft_size
            grad[i] = 2.0 * corr_full[idx]

        # Projected gradient step with sum constraint
        # minimize: log-sum-exp subject to sum(a) = norm, a >= 0
        # Use Lagrange multiplier for sum constraint
        # Simple approach: subtract mean, clip, renormalize
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        # Adaptive learning rate
        score_new = fast_autocorr(a_new).max()
        if score_new > best_score * 1.1:
            lr *= 0.5
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new * norm / s_new
            score_new = fast_autocorr(a_new).max()
        else:
            lr = min(lr * 1.05, 0.1)

        a = a_new

        if score_new < best_score:
            best_score = score_new
            best_a = a.copy()

        # Partial sum for averaging
        partial_sum += a
        n_ps += 1
        if n_ps % 50 == 0:
            avg_check = partial_sum / n_ps
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc = fast_autocorr(avg_check).max()
                if sc < best_score:
                    best_score = sc
                    best_a = avg_check.copy()

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE it={it} ({elapsed:.0f}s): best={best_score:.10f}, temp={temp:.4f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, seed=0, verbose=True):
    """Coordinate descent with golden-section search."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            ai = a[i]
            if ai < 1e-10:
                # Try setting to small positive
                a_try = a.copy()
                a_try[i] = norm * 0.001
                a_try = a_try * norm / a_try.sum()
                sc = fast_autocorr(a_try).max()
                if sc < best_score * (1 - 1e-8):
                    a = a_try
                    best_score = sc
                    best_a = a.copy()
                    pass_improved = True
                continue

            # Golden section on [0, hi]
            lo = 0.0
            hi = ai * 2.5 + norm * 0.05
            phi = 0.6180339887498949

            c1 = lo + (1 - phi) * (hi - lo)
            c2 = lo + phi * (hi - lo)

            a1 = a.copy(); a1[i] = c1; a1 = a1 * norm / a1.sum(); f1 = fast_autocorr(a1).max()
            a2 = a.copy(); a2[i] = c2; a2 = a2 * norm / a2.sum(); f2 = fast_autocorr(a2).max()

            for _ in range(25):
                if f1 < f2:
                    hi = c2; c2 = c1; f2 = f1
                    c1 = lo + (1 - phi) * (hi - lo)
                    a1 = a.copy(); a1[i] = c1; a1 = a1 * norm / a1.sum(); f1 = fast_autocorr(a1).max()
                else:
                    lo = c1; c1 = c2; f1 = f2
                    c2 = lo + phi * (hi - lo)
                    a2 = a.copy(); a2[i] = c2; a2 = a2 * norm / a2.sum(); f2 = fast_autocorr(a2).max()

            opt_val = c1 if f1 < f2 else c2
            a_opt = a.copy(); a_opt[i] = opt_val; a_opt = a_opt * norm / a_opt.sum()
            sc = fast_autocorr(a_opt).max()

            if sc < best_score - 1e-12:
                a = a_opt
                best_score = sc
                best_a = a.copy()
                pass_improved = True

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def structured_init(n, seed=0, style=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif style == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif style == 2:
        # Asymmetric ramp
        a = x ** 1.5 * (1 - x) ** 0.5
        a = np.maximum(a, 0)
    elif style == 3:
        # Asymmetric: fast rise, slow fall
        a = np.exp(-10 * (x - 0.4) ** 2) + 0.5 * np.exp(-6 * (x - 0.7) ** 2)
    elif style == 4:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif style == 5:
        # Multi-peak asymmetric
        a = 0.6 * np.exp(-15 * (x - 0.3) ** 2) + 0.4 * np.exp(-10 * (x - 0.65) ** 2)
    elif style == 6:
        # Linear ramp (explores non-palindromic)
        a = 0.3 + 0.7 * x
    elif style == 7:
        # Inverse-ramp
        a = 0.3 + 0.7 * (1 - x)
    else:
        a = np.sin(np.pi * x) * (1 + 0.2 * np.cos(2 * np.pi * x))
        a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def perturb_seq(a, seed=0, scale=0.03):
    rng = np.random.default_rng(seed)
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.1)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s

        pass

        if cand is None:
            return
        cand = np.maximum(np.asarray(cand, dtype=np.float64), 0.0)
        s = cand.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(cand)
        if np.isfinite(sc) and sc < best_score:
            old = best_score
            best_score = sc
            best_seq = cand.copy()
            elapsed = time.monotonic() - t_global
