# Global state

def fast_autocorr(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def project_to_simplex(a, target_sum):
    """Project a onto {x >= 0, sum(x) = target_sum}"""
    n = len(a)
    # Bisection on tau
    # sum(max(0, a_i - tau)) = target_sum
    lo = a.min() - 1
    hi = a.max() + 1
    for _ in range(200):
        mid = (lo + hi) / 2
        s = np.sum(np.maximum(0, a - mid))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2
    return np.maximum(0, a - tau)

def compute_active_subgradient(a, top_k=50):
    """Compute subgradient of max(convolve(a,a)) using top-k active set"""
    n = len(a)
    conv = fast_autocorr(a)
    size = 2*n - 1
