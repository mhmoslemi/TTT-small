def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_smooth_grad(a, w):
    """Gradient of T*log(sum exp(c/T)) where w = softmax(c/T)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # cross_corr(w, a)[i] = sum_k w[i+k] * a[k]
    cross = irfft(W * np.conj(A), fft_size)[:n]
    return 2.0 * cross


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v_arr = np.asarray(v, dtype=np.float64)
    v_sorted = np.sort(v_arr)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n + 1) > (css - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v_arr - theta, 0.0)


def fista_smooth(a0, time_limit, verbose=True):
    """FISTA on smoothed log-sum-exp objective."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()

        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Temperature schedule - more aggressive
    temps = [30.0, 15.0, 8.0, 4.0, 2.0, 1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.008, 0.004, 0.002]

    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        n_steps = max(50, min(1000, int(remaining * 4)))

        y = a.copy()
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            c = fast_autocorr(y)
            c_max = c.max()
            # Avoid overflow
            exp_c = np.exp((c - c_max) / T)
            w = exp_c / exp_c.sum()

            grad = compute_smooth_grad(y, w)

            # Step size: use 1/L where L is Lipschitz constant bound
            # L <= 2 * ||hessian||_2 <= 4 * max(a) * (2n-1) roughly
            # Use adaptive step
            step = norm / (n * 0.5)
            step = min(step, 2.0 / (n * T + 1.0))

            a_new = project_simplex(a_new, norm)

            # Nesterov momentum
            beta = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1) ** 2)) / 2.0
            y_new = a_new + ((it) / (it + 1 + beta - 1.0)) * (a_new - y) if it > 0 else a_new.copy()
            # Actually, standard FISTA: y_{k+1} = x_{k+1} + (t_k - 1)/(t_{k+1}) * (x_{k+1} - x_k)
            t_k = (1 + np.sqrt(1 + 4 * it * it)) / 2
            t_k1 = (1 + np.sqrt(1 + 4 * (it + 1) ** 2)) / 2
            if it > 0:
                y_new = a_new + ((t_k - 1) / t_k1) * (a_new - y)
            else:
                y_new = a_new.copy()
            y_new = project_simplex(y_new, norm)

            y = y_new
            a = a_new

            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
            score_y = fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()

            if verbose and it % 100 == 0:
                print(f"    FISTA T={T:.4f} step={it}: best={best_score:.8f}")

    return best_a, best_score


def lp_best_response_multipk(a, M_vals, k_indices, solver):
    """Solve max sum(b) s.t. (a*b)[k] <= M_k for k in k_indices, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = []
    
    for k, M_k in zip(k_indices, M_vals):
        # (a*b)[k] = sum_i a[i] * b[k-i] for valid i
        # = b^T @ (a shifted)
        # conv(a,b)[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i] * b[k-i]
        # So the constraint is: sum_i a[i] * b[k-i] <= M_k
        # This is a linear constraint on b: (a_reversed_shifted) @ b <= M_k
        # (a*b)[k] = sum_j b[j] * a[k-j] for 0 <= k-j < n
        # = sum_j b[j] * a[k-j]
        # The coefficient of b[j] in this sum is a[k-j] if 0 <= k-j < n, else 0
        
        coeff = np.zeros(n)
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                coeff[j] = a[idx]
        
        if coeff.max() > 1e-15:
            constraints.append(coeff @ b_var <= M_k)
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, np.pad(f_norm, (1, 1), mode='edge')[1:-1])
    # Actually just simple interp
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
