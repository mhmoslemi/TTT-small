            aw = irfft(A * W, fft_size)
            g2 = aw[:n]
            
            grad = 2.0 * (g1 + g2)
            
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_adam)
            v_hat = v / (1.0 - beta2 ** t_adam)
            
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            
            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose and (si % 5 == 0 or si == n_stages - 1):
            print(f"    LSE-Adam stage {si} T={T_val:.3f}: best={best_score:.8f}")
    
    return best_a, best_score

def lp_best_response_fast(a, M, constraints_indices):
    """Solve LP: max sum(b) s.t. b>=0, conv(a,b)[k] <= M for k in constraints_indices"""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    if constraints_indices is not None and len(constraints_indices) < 2 * n - 1:
        # Build constraint matrix for only the specified indices
        T = np.zeros((len(constraints_indices), n))
        for idx, k in enumerate(constraints_indices):
            # conv(a,b)[k] = sum_i a[i] * b[k-i] for valid i
            # = sum_j b[j] * a[k-j] for valid j
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for j in range(lo, hi):
                T[idx, j] = a[k - j]
        constraints = [T @ b_var <= M]
    else:
        # Full constraint matrix
        size = 2 * n - 1
        T = np.zeros((size, n))
        for k in range(size):
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for j in range(lo, hi):
                T[k, j] = a[k - j]
        constraints = [T @ b_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=_solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)

def nesterov_lp_optimize(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    
    max_iters = 20000
    max_iters = min(max_iters, int(time_limit * 50))  # rough estimate
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_conv_self(a)
        M = conv_y.max()
        if M < 1e-12:
            break
        
        # Adaptive active set: include constraints near the peak
        threshold_frac = 0.85
        if t < 50:
            threshold_frac = 0.7  # broader set initially
        elif t < 200:
            threshold_frac = 0.85
        else:
            threshold_frac = 0.92
        
        active_mask = conv_y > M * threshold_frac
        active_indices = np.where(active_mask)[0]
        
        if len(active_indices) < 5:
            active_indices = np.where(conv_y > M * 0.5)[0]
        if len(active_indices) > 800:
            active_indices = np.where(conv_y > M * 0.95)[0]
            if len(active_indices) < 5:
                active_indices = None
        else:
            active_indices = active_indices  # use as is
        
        b = lp_best_response_fast(a, M, active_indices)
        
        if b is None:
            b = lp_best_response_fast(a, M, None)
        
        if b is None:
            continue
        
        # Check if solution is truly feasible (for partial active set)
        if active_indices is not None and len(active_indices) < size:
            cross = fast_conv_cross(a, b)
            if cross.max() > M * 1.01:
                # Add violated constraints and re-solve
                violated = np.where(cross > M)[0]
                new_active = np.union1d(active_indices, violated)
                if len(new_active) < size:
                    b2 = lp_best_response_fast(a, M, new_active)
                    if b2 is not None:
                        b = b2
                else:
                    b2 = lp_best_response_fast(a, M, None)
                    if b2 is not None:
                        b = b2
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        # Nesterov momentum
        if t >= 2:
            beta = (t - 1.0) / (t + 2.0)
            a_candidate = a_new + beta * (a_new - a)
            a_candidate = np.maximum(a_candidate, 0.0)
            s_cand = a_candidate.sum()
            if s_cand > 1e-15:
                a_candidate = a_candidate * norm / s_cand
                sc_cand = fast_conv_self(a_candidate).max()
                if sc_cand < fast_conv_self(a_new).max():
                    a_new = a_candidate
        
        a = a_new
        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M:.6f}")
    
    return best_a, best_score

def vegas_optimize(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    n_stages = 30
    etas = np.geomspace(0.2, 0.0005, n_stages)
    
    for si, eta_s in enumerate(etas):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        n_steps = max(30, min(100, int(remaining * 5)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = fast_conv_self(a)
            M = conv.max()
            if M < 1e-15:
                break
            
            # Weight by excess over threshold
            w = np.maximum(conv - M * 0.85, 0.0)
            ww = w / (w.sum() + 1e-15)
            
            W = rfft(ww, fft_size)
            A = rfft(a, fft_size)
            
            ca = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)[:n]
            pressure = ca + aw
            
            a = a * np.exp(-eta_s * pressure)
            a = np.maximum(a, 1e-12)
            a = a * norm / a.sum()
            
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose and si % 6 == 0:
            print(f"    Vegas stage {si} eta={eta_s:.5f}: best={best_score:.8f}")
    
    return best_a, best_score

def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue
            
            # Find lags where conv is near peak
            peak_lags = np.where(conv >= cur_max * 0.8)[0]
            
            # Candidate j's: random + gradient-informed
            j_candidates = set()
            for _ in range(6):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:15]:
                j_candidates.add(int((kl - i + n) % n))
                j_candidates.add(int((kl + i) % n))
            j_candidates.discard(i)
            
            best_delta = 0.0
            best_v = cur_max
            best_j = -1
            
            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]
                
                # Compute the change in conv when moving delta from i to j
                # delta_conv[k] = delta * (a contribution from j) - delta * (a contribution from i)
                g = np.zeros(size)
                
                # Adding delta at position j: conv[k] += delta * a[k-j] for valid k-j
                lo = j
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                
                # Also the "other side": conv[k] += delta * a[j-(k-n+1)] ... 
                # Actually for self-conv: conv[k] = sum_i a[i]*a[k-i]
                # If we add delta to a[j], the change is: delta * sum over valid i of (contributing to conv[k])
                # = delta * a[k-j] (when 0<=k-j<n) + delta * a[j-(k-n+1)] (when...)
                # More carefully: conv_new[k] - conv_old[k] = delta*a[k-j] (if 0<=k-j<n) + delta*a[j-k+n-1] (if 0<=j-k+n-1<n)
                # Wait, let me think more carefully.
                # conv[k] = sum_i a[i] * a[k-i] (with 0<=i<n, 0<=k-i<n)
                # If a[j] increases by delta:
                #   - When i=j: contribution delta * a[k-j] (valid if 0<=k-j<n, i.e., j<=k<j+n)
                #   - When k-i=j, i.e., i=k-j: contribution a[k-j] * delta (valid if 0<=k-j<n, i.e., j<=k<j+n)
                # Wait, these are the same term when i=j and when k-i=j. They're the same!
                # Actually no: conv[k] = sum_i a[i]*a[k-i]. If a[j] changes by delta:
                #   - Term where i=j: a[j]*a[k-j] changes by delta*a[k-j] (if 0<=k-j<n)
                #   - Term where k-i=j, i.e., i=k-j: a[k-j]*a[j] changes by a[k-j]*delta (if 0<=k-j<n)
                # These are the SAME term (i=j and k-i=j implies i=k-j=j, so k=2j).
                # Wait no. k-i=j means i=k-j. So the term is a[k-j]*a[j]. This is the same as the term where i=j only if k-j=j, i.e., k=2j.
                # In general, there are TWO contributions:
                # 1. i=j: delta * a[k-j] (valid when 0 <= k-j < n, i.e., j <= k < j+n)
                # 2. k-i=j => i=k-j: a[k-j] * delta (valid when 0 <= k-j < n, i.e., j <= k < j+n)
                # But wait, these are the same sum! When i runs over all valid values, the term a[i]*a[k-i] includes:
                #   - a[j]*a[k-j] (when i=j)
                #   - a[k-j]*a[j] (when i=k-j)
                # If j != k-j (i.e., k != 2j), these are two different terms.
                # If j == k-j (i.e., k == 2j), it's one term: a[j]*a[j].
                # 
                # So the total change at lag k from adding delta to a[j] is:
                #   delta * a[k-j] (if j <= k < j+n)  [from i=j term]
                #   + delta * a[k-j] (if 0 <= k-j < n, i.e., j <= k < j+n) [from i=k-j term]
                # But wait, when k=2j, both conditions give the same term, so we only count it once with delta^2 correction.
                # 
                # Actually, I think the correct way is:
                # delta_conv[k] = delta * (a[k-j] if 0<=k-j<n else 0) + delta * (a[j-(k-n+1)] if 0<=j-(k-n+1)<n else 0)
                # Hmm this is getting confusing. Let me just use the direct formula.
                # 
                # For the "other side": when i = k-j wraps around... no, there's no wrapping in standard convolution.
                # 
                # Let me just be explicit:
                # conv_new[k] = conv_old[k] + delta*a[k-j] (if 0<=k-j<n) + delta*a[j-(k-j)] 
                # No wait. Let me restart.
                # 
                # conv[k] = sum_{i: max(0,k-n+1)<=i<=min(k,n-1)} a[i] * a[k-i]
                # 
                # If we change a[j] to a[j]+delta:
                # New conv[k] = old conv[k] + delta * a[k-j] (if j is in the valid range for i, i.e., max(0,k-n+1)<=j<=min(k,n-1))
                #             + delta * a[k-j'] where j' is such that k-j'=j, i.e., j'=k-j, and j' is in valid range
                # 
                # Wait, I'm overcomplicating this. Let me just note:
                # conv[k] = sum_i a[i]*a[k-i]
                # = ... + a[j]*a[k-j] + ... + a[k-j]*a[j] + ...
                # (the two terms are the same if j = k-j, i.e., k=2j)
                # 
                # So if a[j] -> a[j] + delta:
                # delta_conv[k] = delta*a[k-j] (if 0<=k-j<n) + delta*a[k-j] (if 0<=k-j<n)
                # Wait, the second term: when i = k-j, the term is a[k-j]*a[j]. If a[j] changes by delta, this becomes a[k-j]*(a[j]+delta) = a[k-j]*a[j] + delta*a[k-j].
                # So delta_conv[k] = delta*a[k-j] (from i=j) + delta*a[k-j] (from i=k-j) = 2*delta*a[k-j] (if 0<=k-j<n and j!=k-j)
                # And if k=2j: delta_conv[2j] = delta*a[j] (only one term, and it becomes delta*a[j] + delta^2)
                # 
                # So: delta_conv[k] = 2*delta*a[k-j] for k != 2j, and delta*a[j]+delta^2 for k=2j.
                # With the condition 0<=k-j<n (i.e., j<=k<j+n).
                # 
                # Hmm wait, I need to also check that i=k-j is in range [0,n-1]:
                # 0 <= k-j < n  =>  j <= k < j+n
                # And i=j is always in range [0, n-1] since j is an index of a.
                # But we also need k-j >= 0 and k-j < n for the second term.
                # And for the first term (i=j), we need k-j to be a valid index: 0 <= k-j < n.
                # 
                # So the condition for BOTH terms is: 0 <= k-j < n, i.e., j <= k < j+n.
                # Under this condition: delta_conv[k] = 2*delta*a[k-j] (for k != 2j) or delta*a[j]+delta^2 (for k=2j).
                
                lo = j
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += 2.0 * a[0:hi - lo]
                # Special correction for k = 2j
                if 0 <= 2 * j < size:
                    g[2 * j] -= delta_placeholder = 0  # We'll handle the delta^2 term separately
                    # Actually: g[2*j] should be delta*a[j] not 2*delta*a[j]
                    # So we need g[2*j] = delta*a[j], not 2*delta*a[j]
                    # Correction: g[2*j] -= delta*a[j]
                    # But we don't know delta yet, so let's just use the 2*delta approximation
                    # The error is O(delta^2) which is small.
                
                # Removing delta from position i:
                lo2 = i
                hi2 = min(size, i + n)
                if hi2 > lo2:
                    g[lo2:hi2] -= 2.0 * a[0:hi2 - lo2]
                
                # Sample some delta values
                d_max = min(ai, norm * 0.5)
                if d_max < 1e-10:
                    continue
                n_samples = 20
                deltas = np.linspace(0.005 * d_max, d_max, n_samples)
                
                for d in deltas:
                    new_conv = conv + d * g
                    # Correct for the diagonal terms
                    if 0 <= 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 0 <= 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_delta = d
                        best_j = j
            
            if best_j >= 0 and best_delta > 0 and best_v < best_score - 1e-12:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 8:
                n_perturb = max(10, n // 50)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.1 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                if no_improve >= 15:
                    break
        else:
            no_improve = 0
        
        if verbose and t0 is not None:
            elapsed = time.monotonic() - t0
            if int(elapsed) % 30 < 1:
                print(f"    PCD ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def structured_init(n, rng):
    """Generate diverse initial patterns for optimization."""
    choice = rng.integers(0, 16)
    x = np.linspace(0.0, 1.0, n)
    
    if choice == 0:
        # Smooth bump
        a = np.exp(-20.0 * (x - 0.5) ** 2)
    elif choice == 1:
        # Two bumps
        c1, c2 = 0.3, 0.7
        a = np.exp(-30.0 * (x - c1) ** 2) + np.exp(-30.0 * (x - c2) ** 2)
    elif choice == 2:
        # Multi-frequency
        f1, f2 = rng.integers(2, 8), rng.integers(8, 20)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * f1 * x) + 0.2 * np.cos(2 * np.pi * f2 * x)
    elif choice == 3:
        # Ramp up
        a = x ** 2
    elif choice == 4:
        # Ramp down
        a = (1 - x) ** 2
    elif choice == 5:
        # Triangle
        a = 1.0 - np.abs(2.0 * x - 1.0)
    elif choice == 6:
        # Alternating (high-low pattern)
        a = np.where(np.arange(n) % 2 == 0, 2.0, 1.0)
    elif choice == 7:
        # Sine window
        a = np.sin(np.pi * x) ** 2
    elif choice == 8:
        # Chirp-like (frequency increasing)
        freq = 3.0
        a = 0.5 + 0.4 * np.cos(2 * np.pi * (freq * x ** 2 + 0.3 * x))
    elif choice == 9:
        # Gaussian mixture
        n_peaks = rng.integers(3, 8)
        centers = rng.uniform(0.1, 0.9, n_peaks)
        a = np.zeros(n)
        for c in centers:
            width = rng.uniform(0.03, 0.1)
            a += np.exp(-(x - c) ** 2 / (2 * width ** 2))
    elif choice == 10:
        # Smoothed random
        a = np.ones(n)
        n_changes = rng.integers(n // 4, n // 2)
        positions = rng.choice(n, size=n_changes, replace=False)
        a[positions] += rng.uniform(0.5, 2.0, n_changes)
        # Smooth
        from scipy.ndimage import uniform_filter1d
        a = uniform_filter1d(a, size=max(3, n // 50))
    elif choice == 11:
        # Anti-concentrated: high at edges, low in middle
        a = (x - 0.5) ** 2 + 0.1
    elif choice == 12:
        # Periodic
        period = max(3, n // rng.integers(5, 15))
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.arange(n) / period)
    elif choice == 13:
        # Power law
        a = x ** rng.uniform(0.3, 2.0)
    elif choice == 14:
        # Smooth random via sine mix
        n_harm = rng.integers(3, 8)
        freqs = rng.integers(1, 20, n_harm)
        phases = rng.uniform(0, 2 * np.pi, n_harm)
        amps = rng.uniform(0.1, 0.5, n_harm)
        a = 1.0
        for f, p, amp in zip(freqs, phases, amps):
            a += amp * np.cos(2 * np.pi * f * x + p)
    else:
        # Near-uniform with perturbation
        a = 1.0 + 0.3 * np.sin(np.pi * x) + 0.1 * np.sin(3 * np.pi * x)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    # Add small noise
    a = a + 0.003 * np.abs(rng.standard_normal(n))
    a = a / a.sum()
    return a

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def symmetrize(a):
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    s = a_sym.sum()
    if s < 1e-15:
        return a
    return a_sym / s * (a.sum() / s)

def anti_symmetrize(a):
    """Make sequence anti-palindromic-ish: first half high, second half low."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return a
    a_norm = a / s
    # Create anti-symmetric version
    a_anti = np.zeros(n)
    for i in range(n):
        a_anti[i] = max(a_norm[i] - a_norm[n-1-i], 0)
    a_anti[n-1-i] = 0  # avoid double counting
    # Actually, let's just do: a_anti[i] = a_norm[i] * (i+1)/(n)
    a_anti = a_norm * (np.arange(n) + 1.0) / n
    ss = a_anti.sum()
    if ss < 1e-15:
        return a
    return a_anti / ss

def diverse_lp_optimize(a0, time_limit, verbose=True):
    """LP optimization with periodic restarts and diverse constraint thresholds."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    
    # Cycle through different threshold strategies
    strategies = [0.7, 0.8, 0.85, 0.9, 0.95]
    strat_idx = 0
    
    for t in range(1, 30000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_a = fast_conv_self(a)
        M = conv_a.max()
        if M < 1e-12:
            break
        
        # Adaptive threshold
        threshold = M * strategies[strat_idx % len(strategies)]
        if t % 100 == 0:
            strat_idx += 1
        
        active_indices = np.where(conv_a > threshold)[0]
        if len(active_indices) < 8:
            active_indices = np.where(conv_a > M * 0.5)[0]
        if len(active_indices) > 1000:
            active_indices = np.where(conv_a > M * 0.97)[0]
            if len(active_indices) < 5:
                active_indices = None
        
        b = lp_best_response_fast(a, M, active_indices)
        
        if b is None:
            b = lp_best_response_fast(a, M, None)
        
        if b is None:
            continue
        
        if active_indices is not None and len(active_indices) < size:
            cross = fast_conv_cross(a, b)
            if cross.max() > M * 1.005:
                violated = np.where(cross > M)[0]
                new_active = np.union1d(active_indices, violated)
                if len(new_active) < size:
                    b2 = lp_best_response_fast(a, M, new_active)
                    if b2 is not None:
                        b = b2
                else:
                    b2 = lp_best_response_fast(a, M, None)
                    if b2 is not None:
                        b = b2
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        
        # Adaptive step size
        alpha = 2.0 / (t + 2.0)
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        a = a_new
        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPD t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score
