Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-start structured initialization
2. LP-based Nesterov accelerated mirror descent with adaptive constraint screening
3. Log-sum-exp smoothed subgradient descent with momentum
4. Pair-transfer coordinate descent targeting high-product pairs
5. Soft best-response fallback using multiplicative weight updates
6. Multi-phase time allocation with periodic restarts
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print("  Using solver (fallback): ECOS")
    return _solver

def fft_convolve(a, b=None):
    if b is None:
        b = a
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def compute_score(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    n = len(a)
    max_b = fft_convolve(a).max()
    return 2.0 * n * max_b / (S * S)

def norm_to_target(a, target_sum):
    s = a.sum()
    if s < 1e-15:
        return np.zeros_like(a)
    return a * target_sum / s

def build_active_matrix(a, active_set):
    n = len(a)
    m = len(active_set)
    T = np.zeros((m, n))
    for r, k in enumerate(active_set):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if hi >= lo:
            T[r, lo:hi+1] = a[k - hi:k - lo + 1][::-1]
    return T

def lp_best_response(a, M, solver, active_set=None):
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    constraints = []
    
    if active_set is not None and 0 < len(active_set) < size:
        T = build_active_matrix(a, active_set)
        constraints.append(T @ b_var <= M)
    elif active_set is None or len(active_set) >= size:
        # Full constraint set - only feasible for moderate n
        if n <= 2500:
            T = build_active_matrix(a, np.arange(size))
            constraints.append(T @ b_var <= M)
        else:
            return None, None
    
    if not constraints:
        return None, None
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    
    duals = None
    try:
        if len(constraints[0].dual_value) is not None:
            if active_set is not None and len(active_set) < size:
                full_duals = np.zeros(size)
                full_duals[active_set] = np.asarray(constraints[0].dual_value, dtype=np.float64)
                duals = full_duals
            else:
                duals = np.asarray(constraints[0].dual_value, dtype=np.float64)
    except Exception:
        pass
    
    return val, duals

def soft_best_response(a, M, n_iter=30):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    b = np.ones(n) / n
    A = rfft(a, fft_size)
    eta = 2.0 / np.sqrt(n)
    
    for _ in range(n_iter):
        conv = irfft(A * rfft(b, fft_size), fft_size)[:size]
        penalty = np.maximum(conv - M, 0.0)
        if penalty.max() < 1e-15:
            break
        P = rfft(penalty, fft_size)
        penalty_grad = irfft(A * np.conj(P), fft_size)[:n]
        scale = eta * penalty_grad / max(M, 1e-10)
        scale = np.clip(scale, -5.0, 5.0)
        b = b * np.exp(-scale)
        b = np.maximum(b, 1e-15)
        b = b / b.sum()
    return b

    n = len(a0)
    target = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = norm_to_target(a, target)
    if a.sum() < 1e-15:
        a = np.ones(n) * target / n
    
    best_a = a.copy()
    best_score = fft_convolve(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    n_stages = 16
    betas = np.geomspace(300.0, 0.5, n_stages)
    v = np.zeros(n)
    
    for si, beta in enumerate(betas):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(20, min(250, int(remaining * 3.5)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = fft_convolve(a)
            cur_max = conv.max()
            if cur_max < 1e-15:
                break
            
            shifted = (conv - cur_max) / beta
            shifted = np.clip(shifted, -40.0, 0.0)
            exp_val = np.exp(shifted)
            w = exp_val / exp_val.sum()
            
            A = rfft(a, fft_size)
            W = rfft(w, fft_size)
            grad = 2.0 * irfft(A * np.conj(W), fft_size)[:n]
            
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            
            lr = target / (beta * np.sqrt(n) * (1.0 + 0.02 * it))
            momentum = 0.85 + 0.1 * min(1.0, it / 50.0)
            
            v = momentum * v - lr * grad
            a_new = a + v
            a_new = np.maximum(a_new, 0.0)
            a_new = norm_to_target(a_new, target)
            
            if a_new.sum() < 1e-15:
                v = np.zeros(n)
                a_new = a.copy()
            else:
                a = a_new
            
            sc = fft_convolve(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            
            if it % 50 == 0 and sc > cur_max * 1.5:
                v = np.zeros(n)
        
        if verbose:
            print(f"    LSE stage {si} beta={beta:.1f}: best={best_score:.8f}")
    
    return best_a, best_score

def lp_nesterov_optimize(a0, time_limit, solver, verbose=True, use_mult=False):
    n = len(a0)
    target = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = norm_to_target(a, target)
    if a.sum() < 1e-15:
        a = np.ones(n) * target / n
    
    best_a = a.copy()
    best_score = fft_convolve(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    avg_a = np.zeros(n)
    n_avg = 0
    no_improve = 0
    prev_obj = best_score
    
    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fft_convolve(a)
        M = conv.max()
        if M < 1e-12:
            break
        
        # Adaptive constraint screening
        if t < 30:
            thresh = 0.4
        elif t < 100:
            thresh = 0.6
        elif t < 300:
            thresh = 0.75
        else:
            thresh = 0.88
        
        active = np.where(conv > M * thresh)[0]
        if len(active) < 20:
            active = np.where(conv > M * 0.3)[0]
        
        b, duals = lp_best_response(a, M, solver, active)
        
        if b is None or b.sum() < 1e-15:
            # Fall back to soft best response
            b = soft_best_response(a, M, n_iter=20)
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * target / b_sum
        
        # Verify dropped constraints
        if len(active) < size:
            check = fft_convolve(a, g)
            if check.max() > M * (1 + 1e-2):
                b2, _ = lp_best_response(a, M, solver, None)
                if b2 is not None and b2.sum() > 1e-15:
                    g = b2 * target / b2.sum()
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        
        if use_mult and t > 10:
            ratio = g / np.maximum(a, 1e-30)
            log_r = np.log(np.clip(ratio, 1e-10, 1e10))
            a_new = a * np.exp(alpha * log_r)
        else:
            a_new = (1.0 - alpha) * a + alpha * g
        
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = norm_to_target(a_new, target)
        
        # Nesterov momentum
        if t == 1:
            a_prev = a.copy()
            a = a_new
        else:
            mu = (t - 1.0) / (t + 2.0)
            a_momentum = a_new + mu * (a_new - a)
            a_momentum = np.maximum(a_momentum, 0.0)
            s_m = a_momentum.sum()
            if s_m < 1e-15:
                a_momentum = a_new.copy()
            else:
                a_momentum = norm_to_target(a_momentum, target)
            a_prev = a
            a = a_momentum
        
        sc_a = fft_convolve(a).max()
        improved = False
        if sc_a < best_score - 1e-12:
            best_score = sc_a
            best_a = a.copy()
            improved = True
        
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
        
        # Restart on increase
        if no_improve == 50 and t > 100:
            n_pert = max(10, n // 25)
            rng_p = np.random.default_rng(t)
            idx = rng_p.choice(n, size=n_pert, replace=False)
            a[idx] *= (1.0 + 0.15 * rng_p.standard_normal(n_pert))
            a = np.maximum(a, 0.0)
            a = norm_to_target(a, target)
            no_improve = 0
        
        avg_a += a
        n_avg += 1
        if n_avg % 200 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = norm_to_target(avg_check, target)
                sc_avg = fft_convolve(avg_check).max()
                if sc_avg < best_score:
                    best_score = sc_avg
                    best_a = avg_check.copy()
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPNest t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    n = len(a0)
    target = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = norm_to_target(a, target)
    if a.sum() < 1e-15:
        a = np.ones(n) * target / n
    
    best_a = a.copy()
    best_score = fft_convolve(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_imp = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False
        
        peak_lags = np.where(conv >= cur_max * 0.92)[0]
        
        for idx_i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx_i]
            ai = a[i]
            if ai < target * 5e-8:
                continue
            
            # Find best j to transfer to
            best_delta = 0.0
            best_val = cur_max
            best_j = -1
            
            # Candidate j's: from peak lags
            candidates = set()
            for kl in peak_lags[:10]:
                candidates.add((kl - i) % n)
                candidates.add((kl + i) % n)
            for _ in range(5):
                candidates.add(int(rng.integers(n)))
            candidates.discard(i)
            
            for j in list(candidates)[:12]:
                if j == i:
                    continue
                aj = a[j]
                
                # Compute change in convolution if we transfer delta from i to j
                # new_a[i] = ai - d, new_a[j] = aj + d
                # delta_conv[k] = d * (a[k-i] - a[k-j] + ...) ... complex
                # Just try a few delta values
                n_samples = 10
                deltas = np.linspace(0.05, 0.8, n_samples) * ai
                
                for d in deltas:
                    # Approximate new conv
                    new_conv = conv.copy()
                    # Contribution change at each lag
                    # Lags affected by i: k = i, i+1, ..., i+n-1 (a[i] was at position i)
                    # Lags affected by j: k = j, j+1, ..., j+n-1
                    # Change = -d * (a shifted by i) + d * (a shifted by j)
                    # + diagonal terms for k=2i and k=2j
                    lo_i = max(0, i)
                    hi_i = min(size, i + n)
                    if hi_i > lo_i:
                        new_conv[lo_i:hi_i] += d * (-a[0:hi_i - lo_i])
                    lo_j = max(0, j)
                    hi_j = min(size, j + n)
                    if hi_j > lo_j:
                        new_conv[lo_j:hi_j] += d * a[0:hi_j - lo_j]
                    # Diagonal correction
                    if 2 * i < size:
                        new_conv[2*i] = conv[2*i] - 2*ai*d + d*d
                    if 2 * j < size:
                        new_conv[2*j] = conv[2*j] + 2*aj*d + d*d
                    # Cross term at i+j
                    if i + j < size and lo_i <= i + j < hi_i and lo_j <= i + j < hi_j:
                        pass  # already handled above
                    
                    v = new_conv.max()
                    if v < best_val:
                        best_val = v
                        best_delta = d
                        best_j = j
            
            if best_j >= 0 and best_delta > 0 and best_val < cur_max - 1e-12:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = norm_to_target(a, target)
                    new_max = fft_convolve(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_imp += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = norm_to_target(a, target)
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 10:
                break
            if no_improve == 5:
                n_pert = max(10, n // 30)
                idx_p = rng.choice(n, size=n_pert, replace=False)
                a[idx_p] *= (1.0 + 0.12 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0.0)
                a = norm_to_target(a, target)
        else:
            no_improve = 0
        
        if verbose and total_imp % 300 == 0 and total_imp > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD imp={total_imp} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def dual_refine(a0, time_limit, solver, verbose=True):
    n = len(a0)
    target = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = norm_to_target(a, target)
    if a.sum() < 1e-15:
        a = np.ones(n) * target / n
    
    best_a = a.copy()
    best_score = fft_convolve(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    for it in range(300):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fft_convolve(a)
        M = conv.max()
        if M < 1e-12:
            break
        
        active = np.where(conv > M * 0.7)[0]
        b, duals = lp_best_response(a, M, solver, active)
        
        if duals is None:
            b2, duals2 = lp_best_response(a, M, solver, None)
            if duals2 is not None:
                duals = duals2
            else:
                duals = np.zeros(size)
                duals[np.argmax(conv)] = 1.0
        
        A = rfft(a, fft_size)
        D = rfft(duals, fft_size)
        grad = 2.0 * irfft(A * np.conj(D), fft_size)[:n]
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        step = 0.05
        for trial in range(15):
            a_new = a - step * grad / grad_norm
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                step *= 0.5
                continue
            a_new = norm_to_target(a_new, target)
            sc = fft_convolve(a_new).max()
            if sc < M - 1e-12:
                a = a_new
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                break
            step *= 0.5
        
        if verbose and it % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DualRef it={it} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def structured_init(n, rng):
    choice = int(rng.integers(0, 10))
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.15 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0.1)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-6.0 * (x - 0.75) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-2.0 * x) + 0.5 * np.exp(-2.0 * (1 - x))
    elif choice == 7:
        a = 0.5 * np.exp(-4.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-4.0 * (x - 0.7) ** 2)
        a = a * (1.0 + 0.3 * np.cos(20 * np.pi * x))
        a = np.maximum(a, 0.01)
    elif choice == 8:
        a = np.abs(np.sin(3 * np.pi * x)) ** 0.8
    else:
        a = 0.4 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(11 * np.pi * x)
        a = np.maximum(a, 0.05)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss

def eval_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent if available
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_safe(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Initialize
    if best_seq is None:
        n_base = 1600
        best_init = None
        best_init_score = float('inf')
        for trial in range(8):
            cand = structured_init(n_base, rng)
            sc = eval_safe(cand)
            if sc < best_init_score:
                best_init_score = sc
                best_init = cand
        best_seq = best_init
        best_score = best_init_score
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)
    
    # Phase 0: Quick LSE warm-up
    remaining = time_budget - (time.monotonic() - t_global)
    t0_phase = min(remaining * 0.06, 90.0)
    if t0_phase > 20:
        print(f"\n=== Phase 0: LSE warm-up (budget={t0_phase:.0f}s) ===")
        a_out, _ = lse_optimize(best_seq, t0_phase, verbose=False)
        try_update(a_out, "lse_init")
    
    # Phase 1: LP Nesterov (main)
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = len(best_seq)
    t1 = min(remaining * 0.30, 480.0)
    print(f"\n=== Phase 1: LP Nesterov n={n1} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = lp_nesterov_optimize(best_seq, t1, solver, verbose=True, use_mult=False)
        try_update(a_out, f"lp1_n{n1}")
    
    # Phase 2: LSE refinement (lower temps)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 2: LSE refine n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = lse_optimize(best_seq, t2, verbose=True)
        try_update(a_out, "lse_refine")
    
    # Phase 3: LP Nesterov round 2 (multiplicative)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 3: LP Nesterov mult n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = lp_nesterov_optimize(best_seq, t3, solver, verbose=True, use_mult=True)
        try_update(a_out, "lp2_mult")
    
    # Phase 4: Dual refinement
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 4: Dual refine (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = dual_refine(best_seq, t4, solver, verbose=True)
        try_update(a_out, "dual_ref")
    
    # Phase 5: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 5: Pair CD (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = pair_cd_optimize(best_seq, t5, rng, verbose=True)
        try_update(a_out, "pcd")
    
    # Phase 6: Final combined polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Final polish (remaining={remaining:.0f}s) ===")
    if remaining > 60:
        t6a = min(remaining * 0.4, 100.0)
        a_out, _ = lse_optimize(best_seq, t6a, verbose=False)
        try_update(a_out, "lse_final")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t6b = min(remaining * 0.4, 100.0)
            a_out, _ = lp_nesterov_optimize(best_seq, t6b, solver, verbose=False, use_mult=False)
            try_update(a_out, "lp_final")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t6c = min(remaining * 0.5, 60.0)
            a_out, _ = pair_cd_optimize(best_seq, t6c, rng, verbose=False)
            try_update(a_out, "pcd_final")
    
    # Phase 7: Try longer sequence
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 400
    print(f"\n=== Phase 7: Length ext n={n_long} (remaining={remaining:.0f}s) ===")
    if remaining > 120:
        init_long = interpolate_seq(best_seq, n_long)
        t7 = min(remaining * 0.5, 150.0)
        a_out, _ = lp_nesterov_optimize(init_long, t7, solver, verbose=False, use_mult=False)
        try_update(a_out, f"lp_long_{n_long}")
    
    # Final evaluation
    if best_seq is not None:
        final_score = eval_safe(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    
    return [float(x) for x in best_seq]
