def fast_conv(a):
    """FFT-based convolution of a with itself."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_crossconv(a, b):
    """FFT-based convolution of a and b."""
    n1, n2 = len(a), len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = np.fft.rfft(np.asarray(b, dtype=np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_soft_subgradient(a, ac, top_k=50, beta=5.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def solve_lp_direction(a, M, solver, top_k_constraints=None):
    """Solve LP to find direction: max sum(b) s.t. b>=0, conv(a,b) <= M.
    Uses only top_k_constraints most restrictive constraints for speed."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)

    # Determine which constraints to use
    if top_k_constraints is None:
        top_k_constraints = n  # use all

    # Build constraint matrix for conv(a, b)
    # conv(a,b)[k] = sum_j a[j] * b[k-j] for valid j
    # = sum_j T[k,j] * b[j] where T[k,j] = a[j] if 0<=k-j<n else 0
    # For speed, only build rows for top_k lags

    ac = fast_conv(a)
    if top_k_constraints >= len(ac):
        k_indices = np.arange(len(ac))
    else:
        k_indices = np.argsort(ac)[::-1][:top_k_constraints]

    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for k in k_indices:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        for j in range(j_start, j_end + 1):
            rows.append(k_int)
            cols.append(j)
            vals.append(a[j])

    if len(vals) == 0:
        return a.copy()

    T = cp.MatMul(cp.diag(np.ones(len(k_indices))),
                  cp.sparse((len(k_indices), n), rows, cols, vals))
    # Actually let's just use the sparse matrix directly
    from scipy.sparse import csr_matrix
    T_sp = csr_matrix((vals, (rows, cols)), shape=(len(k_indices), n))

    constraints = [cp.Mul(T_sp @ b_var) <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return a.copy()

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy()

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return a.copy()
    return b_val / b_sum


def line_search(a, direction, use_best, time_per_search=0.5):
    """Find best step size alpha in [0,1] for updating a toward direction."""
    n = len(a)
    t0 = time.monotonic()
    best_alpha = 0.2
    best_val = use_best

    # Try a set of alphas
    alphas = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
    for alpha in alphas:
        if time.monotonic() - t0 > time_per_search:
            break
        a_trial = (1 - alpha) * a + alpha * direction
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-15:
            continue
        a_trial = a_trial / s
        ac = fast_conv(a_trial)
        val = ac.max()
        if val < best_val:
            best_val = val
            best_alpha = alpha

    return best_alpha, best_val

    stall = 0
    max_stall = 30

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov momentum
        if t > 1:
            beta_t = (t - 0.5) / (t + 1.5)
            y = x + beta_t * (x - y)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy > 1e-15:
                y = y / sy
            else:
                y = x.copy()

        # Solve LP for direction
        ac_y = fast_conv(y)
        M_curr = ac_y.max()
        if M_curr < 1e-14:
            break

        direction = solve_lp_direction(y, M_curr, solver, top_k_constraints)
        if direction is None or np.array_equal(direction, y):
            direction = a.copy()

        # Line search for step size
        alpha_try = min(2.0 / (t + 2.0), 0.5)
        a_trial = (1.0 - alpha_try) * y + alpha_try * direction
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            a_trial = y.copy()
            s_t = 1.0
        a_trial = a_trial / s_t
        M_trial = fast_conv(a_trial).max()

        # Try a few more step sizes if time permits
        if time.monotonic() - t0 < time_limit - 1 and M_trial < best_m:
            for alpha2 in [alpha_try * 1.5, alpha_try * 2.0, 0.5, 0.3]:
                if time.monotonic() - t0 > time_limit:
                    break
                alpha2 = min(max(alpha2, 0.01), 1.0)
                a_t2 = (1.0 - alpha2) * y + alpha2 * direction
                a_t2 = np.maximum(a_t2, 0.0)
                s2 = a_t2.sum()
                if s2 < 1e-15:
                    continue
                a_t2 = a_t2 / s2
                M_t2 = fast_conv(a_t2).max()
                if M_t2 < M_trial:
                    a_trial = a_t2
                    M_trial = M_t2
                    alpha_try = alpha2

        x = a_trial
        y = x.copy()
        iter_count += 1

        if M_trial < best_m - 1e-15:
            best_m = M_trial
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            # Restart from best with perturbation
            rng = np.random.default_rng(t * 7 + 13)
            pert = rng.normal(0, 0.005, n)
            x_restarted = best_a + pert * best_a
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            y = x_restarted
            stall = 0
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    LP restart at iter={t} ({elapsed:.1f}s)")

        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.8f}, stall={stall}")

    return best_a, best_m


def mirror_descent_polish(a0, time_limit, top_k=50, beta=5.0, c0=3.0, verbose=False):
    """Nesterov-accelerated mirror descent for polishing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    x = a.copy()
    x_prev = a.copy()
    gamma = c0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_conv(y)
        g = compute_soft_subgradient(y, ac, top_k=top_k, beta=beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_conv(x).max()
        if M_new < best_m:
            best_m = M_new
            best_a = x.copy()

        # Periodic average check
        if t % 500 == 0:
            # Running average of last 500 iterates would be ideal
            # But for simplicity, just check current
            pass

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.8f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False, rng_seed=42):
    """Coordinate descent: move mass between pairs of coordinates."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        M_curr = ac.max()
        # Focus on top lags
        top_k = np.argsort(ac)[::-1][:min(20, len(ac))]

        # Compute gradient for each coordinate
        grad = np.zeros(n)
        for k in top_k[:10]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += a[j_idx]

        # Find the best coordinate to reduce (highest grad) and best to increase (lowest grad)
        # But also try random pairs for exploration
        n_tries = min(100, max(10, n // 20))
        improved = False

        for _ in range(n_tries):
            if time.monotonic() - t0 > time_limit:
                break

            # Choose source: coordinate with high grad
            if rng.random() < 0.7:
                # Pick from top grad coordinates
                top_grad = np.argsort(grad)[::-1][:max(5, n // 50)]
                src = int(top_grad[rng.integers(len(top_grad))])
            else:
                src = int(rng.integers(n))

            if a[src] < 1e-12:
                continue

            # Choose destination: coordinate with low grad
            if rng.random() < 0.7:
                bot_grad = np.argsort(grad)[:max(5, n // 50)]
                dst = int(bot_grad[rng.integers(len(bot_grad))])
            else:
                dst = int(rng.integers(n))

            if dst == src:
                continue

            # Try different fractions
            for frac in [0.01, 0.03, 0.05, 0.1, 0.15, 0.2]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    # Recompute gradient
                    grad = np.zeros(n)
                    for k in top_k[:10]:
                        k_int = int(k)
                        i_start = max(0, k_int - n + 1)
                        i_end = min(n, k_int + 1)
                        idx = np.arange(i_start, i_end)
                        j_idx = k_int - idx
                        grad[idx] += a[j_idx]
                    break

        if not improved:
            stall += 1
            if stall % 50 == 0:
                # Try a random perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_conv(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            a = a / a.sum()

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord polish ({elapsed:.1f}s): J={J:.8f}")
    return best_a, best_m


def multi_start_optimize(a0, time_limit, solver, verbose=True):
    """Try multiple perturbed starts and return the best."""
    n = len(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_m = fast_conv(a0).max()
    rng = np.random.default_rng(12345)

    n_starts = 3
    time_per_start = time_limit / n_starts - 5

    for s in range(n_starts):
        if time.monotonic() - t0 > time_limit - 10:
            break

        if s == 0:
            a_start = a0.copy()
        else:
            # Perturb the best found so far
            noise = rng.normal(0, 0.02, n)
            a_start = best_a + noise * best_a
            a_start = np.maximum(a_start, 0.0)
            s_s = a_start.sum()
            if s_s > 1e-15:
                a_start = a_start / s_s

        remaining = min(time_per_start, time_limit - 10 - (time.monotonic() - t0))
        if remaining < 30:
            break

        a_out, m_out = lp_frank_wolfe(a_start, remaining, solver,
                                       verbose=verbose and s == 0,
                                       top_k_constraints=max(100, n // 10))

        if m_out < best_m:
            best_m = m_out
            best_a = a_out
            if verbose:
                print(f"    Multi-start {s}: improved to M={m_out:.10f}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent / parent.sum()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 2000
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    n0 = len(best_seq)

    # === Phase 1: LP Frank-Wolfe (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Frank-Wolfe n={n0} ===")
    t1 = min(remaining * 0.45, 700.0)
    if t1 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t1, solver, verbose=True,
                                   top_k_constraints=max(150, n0 // 10))
        try_update(a_out, f"lp_n{n0}")

    # === Phase 2: Mirror descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: MD polish n={len(best_seq)} ===")
    t2 = min(remaining * 0.20, 250.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent_polish(a_in, t2, top_k=60, beta=4.0, c0=4.0, verbose=True)
        try_update(a_out, f"md_n{len(best_seq)}")

    # === Phase 3: LP Frank-Wolfe again (after MD moved us) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP again n={n_best} ===")
    t3 = min(remaining * 0.20, 250.0)
    if t3 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t3, solver, verbose=True,
                                   top_k_constraints=max(100, n_best // 15))
        try_update(a_out, f"lp2_n{n_best}")

    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coord n={n_best} ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t4, verbose=True, rng_seed=42)
        try_update(a_out, f"coord_n{n_best}")

    # === Phase 5: Multi-start from best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 120 and n_best <= 3000:
        print(f"\n=== Phase 5: Multi-start n={n_best} ===")
        t5 = min(remaining * 0.30, 300.0)
        if t5 > 60:
            a_in = norm_sum1(best_seq)
            a_out, _ = multi_start_optimize(a_in, t5, solver, verbose=True)
            try_update(a_out, f"multi_n{n_best}")

    # === Phase 6: Try different length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 100:
        for n_try in [n_best + 400, n_best - 400]:
            if n_try < 100:
                continue
            if time.monotonic() - t_global > time_budget - 80:
                break
            print(f"\n=== Phase 6: Scale to n={n_try} ===")
            a_long = interpolate_sequence(best_seq, n_try)
            t6 = min(remaining * 0.3, 150.0)
            if t6 > 30:
                a_in = norm_sum1(a_long)
                a_out, _ = lp_frank_wolfe(a_in, t6, solver, verbose=False,
                                           top_k_constraints=max(100, n_try // 12))
                try_update(a_out, f"scale_n{n_try}")
                if best_seq is not None and len(best_seq) == n_try:
                    # Polish at this length
                    remaining2 = time_budget - (time.monotonic() - t_global)
                    if remaining2 > 30:
                        a_in2 = norm_sum1(best_seq)
                        a_out2, _ = mirror_descent_polish(a_in2, min(remaining2 * 0.5, 80.0),
                                                          top_k=50, beta=4.0, c0=3.0, verbose=False)
                        try_update(a_out2, f"md_scale_n{n_try}")

    # === Phase 7: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final coord n={n_best} ===")
    if remaining > 20:
        t7 = min(remaining * 0.7, 100.0)
        if t7 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t7, verbose=True, rng_seed=777)
            try_update(a_out, f"coord_final_n{n_best}")

    # === Phase 8: Final LP squeeze ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if n_best <= 2500 and remaining > 30:
        print(f"\n=== Phase 8: Final LP n={n_best} ===")
        t8 = min(remaining * 0.8, 120.0)
        if t8 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t8, solver, verbose=True,
                                       top_k_constraints=200)
            try_update(a_out, f"lp_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
