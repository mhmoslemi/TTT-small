with adaptive constraint screening, LSE gradient refinement, and multi-start.

import time
import cvxpy as cp
from scipy.fft import rfft, irfft

# Solver management
_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(8, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    return _solver

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def lse_gradient_descent(a0, time_limit, rng, verbose=False):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # More stages, more aggressive cooling
    n_stages = 16
    temps = np.geomspace(50.0, 0.1, n_stages)
    
    momentum = np.zeros(n)
    beta = 0.7
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        
        n_steps = max(20, min(200, int(remaining * 8)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -80.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)
            
            # Adaptive step size
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            step = norm / (T_val * np.sqrt(n) * (1.0 + 0.1 * it))
            
            momentum = beta * momentum - step * grad
            a_new = a + momentum
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new
            
            sc = fast_conv_self(a).max()

                best_a = a.copy()
    
    return best_a, best_score

def lp_solve_fast(a, M, n_active, solver):
    """Solve LP with only the top n_active tightest constraints."""
    n = len(a)
    size = 2 * n - 1
    
    # Compute current convolution to find active constraints
    conv_a = fast_conv_self(a)
    
    # Select active constraints (top n_active by value)
    if n_active < size:
        # Use top n_active peaks
        active = np.argpartition(conv_a, -n_active)[-n_active:]
    else:
        active = np.arange(size)
    
    # Build constraint matrix for active constraints
    # T[k, i] = a[k-i] if 0 <= k-i < n, else 0
    # conv(a, b)[k] = sum_i a[k-i] * b[i] (with a[j]=0 for j outside [0,n-1])
    # Wait, I need to be careful about the convolution direction.
    # np.convolve(a, b)[k] = sum_i a[i] * b[k-i]
    # So conv(a, b)[k] = sum_i a[i] * b[k-i]
    # For the constraint conv(a, b)[k] <= M, we need:
    # sum_i a[i] * b[k-i] <= M
    # Let j = k-i, so i = k-j:
    # sum_j a[k-j] * b[j] <= M
    # So the constraint row for lag k is: [a[k-0], a[k-1], ..., a[k-(n-1)]] (padded with 0s)
    # i.e., T[k, j] = a[k-j] if 0 <= k-j < n, else 0
    
    T_active = np.zeros((len(active), n))
    for row, k in enumerate(active):
        # T[row, j] = a[k - j] for j where 0 <= k-j < n
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_max >= j_min:
            T_active[row, j_min:j_max+1] = a[k - np.arange(j_min, j_max+1)]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_active @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    prob.solve(solver=solver, verbose=False)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)

def lp_nesterov_optimize(a0, time_limit, solver, verbose=False, n_active=100):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    y = a.copy()
    
    iteration = 0
    stall_count = 0
    prev_best = best_score
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        if iteration > 100000:
            break
        
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break
        
        # Adaptive active constraint selection
        # Start with more constraints, reduce as we converge
        if iteration < 10:
            n_act = size  # Use all constraints initially
        elif iteration < 50:
            n_act = max(50, int(size * 0.1))
        else:
            n_act = max(30, int(size * 0.05))
        
        b = lp_solve_fast(y, M, n_act, solver)
        if b is None:

            b = lp_solve_fast(y, M, size, solver)
            if b is None:

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        
        # Check if dropped constraints are violated

            check_conv = fast_conv_cross(y, g)
            if check_conv.max() > M * 1.05:

                    b_sum = b.sum()
                    if b_sum > 1e-15:
                        g = b * norm / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:

        else:

        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        # Track best
        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            stall_count = 0
        
        # If stalled for too long, perturb

            # Perturb y slightly

            idx = np.random.default_rng(iteration).choice(n, size=n_pert, replace=False)
            y[idx] *= (1.0 + 0.1 * np.random.default_rng(iteration + 1).standard_normal(n_pert))
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            y = y * norm / s_y
            stall_count = 0
        
        if verbose and iteration % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={iteration} ({elapsed:.0f}s): best={best_score:.8f}")

    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
