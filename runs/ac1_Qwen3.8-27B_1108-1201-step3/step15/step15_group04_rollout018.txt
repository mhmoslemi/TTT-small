def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T

def lp_solve(a, M, solver, active=None):
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    if active is not None and 0 < len(active) < size:
        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    duals = None
    try:
        con = constraints[0]
        if con.dual_value is not None:

                full_duals = np.zeros(size)

                duals = full_duals
            else:
                duals = np.asarray(con.dual_value, dtype=np.float64)
    except Exception:
        pass
    return val, duals

            sc = fast_conv_self(a).max()

                best_a = a.copy()

        if verbose:

def peak_targeted_cd(a0, time_limit, rng, verbose=True):
    """Coordinate descent that specifically targets peak positions."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_score(a)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        
        # Focus on peak positions
        peak_thresh = cur_max * 0.95
        peak_lags = np.where(conv >= peak_thresh)[0]
        
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            # Candidate j's focused on reducing peaks
            j_candidates = set()
            for _ in range(5):
                j_candidates.add(int(rng.integers(n)))
            
            # Structured: positions that appear in peak lags with i
            for kl in peak_lags[:12]:
                j1 = int(kl - i)
                j2 = int(kl + i)
                if 0 <= j1 < n:
                    j_candidates.add(j1)
                if 0 <= j2 < n:
                    j_candidates.add(j2)
            
            # Also try neighbors
            for delta in [-2, -1, 1, 2]:
                j = (i + delta) % n
                j_candidates.add(j)
            
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]

                # Compute gain vector
                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # Binary search for optimal delta
                lo_d, hi_d = 0.0, 0.99 * ai
                best_d = 0.0
                best_v = cur_max
                for _ in range(10):
                    mid = (lo_d + hi_d) / 2
                    new_conv = conv + mid * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * mid + mid * mid
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * mid + mid * mid
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_d = mid
                    if v < cur_max:
                        lo_d = mid
                    else:
                        hi_d = mid

                if best_d > 0 and best_v < best_val - 1e-14:
                    best_val = best_v
                    best_delta = best_d
                    best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-14:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

            no_improve += 1
            if no_improve >= 10:
                break

                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

max sum(b) s.t. T@b <= M, b >= 0

where T is the (2n-1) × n matrix where the k-th row has a at positions [k, k+1, ..., k+n-1] (shifted).

The constraint T@b <= M means: for each lag k, sum_i a[i] * b[k-i] <= M (where the sum is over valid indices). This is the constraint that the cross-correlation of a and b at lag k doesn't exceed M.

At the fixed point, a* = b* (up to normalization), meaning the sequence is its own best response. This is a Nash equilibrium of the game where each player wants to maximize their "mass" while keeping the cross-correlation bounded.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None


def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    return _solver


def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T


def lp_solve(a, M, solver, active=None):
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    if active is not None and 0 < len(active) < size:
        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    duals = None
    try:
        con = constraints[0]
        if con.dual_value is not None:
            if active is not None and 0 < len(active) < size:
                full_duals = np.zeros(size)
                full_duals[active] = np.asarray(con.dual_value, dtype=np.float64)
                duals = full_duals
            else:
                duals = np.asarray(con.dual_value, dtype=np.float64)
    except Exception:
        pass
    return val, duals


def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]

def compute_score(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    n = len(a)
    max_b = fast_conv_self(a).max()
    return 2.0 * n * max_b / (S * S)


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 14
    temps = np.geomspace(150.0, 0.05, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(40, min(350, int(remaining * 5)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -200.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)
            g2 = aw[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                step = 0.0
            else:
                step = norm / (T_val * np.sqrt(n) * (1.0 + 0.03 * it))
                for trial in range(12):
                    a_new = a - step * grad
                    a_new = np.maximum(a_new, 0.0)
                    s_new = a_new.sum()
                    if s_new < 1e-15:
                        step *= 0.5
                        continue
                    a_new = a_new * norm / s_new
                    sc = fast_conv_self(a_new).max()
                    if sc <= max_c + 0.08 * T_val:
                        break
                    step *= 0.5

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.1f}: best={best_score:.8f}")

    return best_a, best_score


def enhanced_lp_nesterov(a0, time_limit, solver, verbose=True, seed=42):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1
    rng = np.random.default_rng(seed)
    no_improve_count = 0
    last_improve_iter = 0
    fft_size = 1 << (size - 1).bit_length()

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Progressive constraint screening
        if t < 20:
            thresh = 0.25
        elif t < 60:
            thresh = 0.5
        elif t < 150:
            thresh = 0.7
        elif t < 400:
            thresh = 0.8
        elif t < 1000:
            thresh = 0.88
        else:
            thresh = 0.92
        active = np.where(conv_y > M * thresh)[0]
        if len(active) < 15:
            active = np.where(conv_y > M * 0.2)[0]

        b, duals = lp_solve(y, M, solver, active)
        if b is None or b.sum() < 1e-15:
            b, duals = lp_solve(y, M, solver, None)
        if b is None or b.sum() < 1e-15:
            continue

        g = b * norm / b.sum()

        # Verify dropped constraints
        if active is not None and len(active) < size:
            check_conv = fast_conv_cross(y, g)
            violated = check_conv > M * (1 + 2e-3)
            if np.any(violated):
                n_viol = int(np.sum(violated))
                if n_viol < 200:
                    all_active = np.unique(np.concatenate([active, np.where(violated)[0]]))
                    b_v, duals_v = lp_solve(y, M, solver, all_active)
                    if b_v is not None and b_v.sum() > 1e-15:
                        b = b_v
                        g = b * norm / b.sum()
                        duals = duals_v
                else:
                    b_full, duals_full = lp_solve(y, M, solver, None)
                    if b_full is not None and b_full.sum() > 1e-15:
                        b = b_full
                        g = b * norm / b.sum()
                        duals = duals_full

        # Adaptive Nesterov step
        base_alpha = 2.0 / (t + 2.0)
        stagnation = t - last_improve_iter
        if stagnation > 200:
            alpha = min(base_alpha * 3.0, 0.3)
        elif stagnation > 100:
            alpha = min(base_alpha * 2.0, 0.2)
        else:
            alpha = base_alpha
        alpha = max(alpha, 0.005)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min(1.0, (t - 1.0) / (t + 2.0))
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        improved = False
        if sc_y < best_score - 1e-13:
            best_score = sc_y
            best_a = y.copy()
            improved = True
            last_improve_iter = t

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score - 1e-13:
            best_score = sc_a
            best_a = a.copy()
            improved = True
            last_improve_iter = t

        if not improved:
            no_improve_count += 1
        else:
            no_improve_count = 0

        # Dual-guided perturbation to escape local optima
        if no_improve_count == 100 and t > 50:
            if duals is not None and np.any(duals > 0):
                D = rfft(duals, fft_size)
                A_y = rfft(y, fft_size)
                # Perturbation direction from dual
                pert = irfft(np.conj(A_y) * D, fft_size)[:n]
                pert = pert - pert.mean()
                rn = np.linalg.norm(pert)
                scale = 0.08 * norm
                if rn > 1e-15:
                    y = y - scale * pert / rn
                    y = np.maximum(y, 0.0)
                    s_y = y.sum()
                    if s_y > 1e-15:
                        y = y * norm / s_y
            else:
                n_perturb = max(15, n // 12)
                idx = rng.choice(n, size=n_perturb, replace=False)
                y[idx] *= (1.0 + 0.2 * rng.standard_normal(n_perturb))
                y = np.maximum(y, 0.0)
                s_y = y.sum()
                if s_y > 1e-15:
                    y = y * norm / s_y
            no_improve_count = 0

        # Second perturbation strategy: peak-targeted
        if no_improve_count == 200 and t > 100:
            conv_check = fast_conv_self(y)
            peaks = np.where(conv_check >= conv_check.max() * 0.98)[0]
            n_pert = max(10, n // 20)
            idx = rng.choice(n, size=n_pert, replace=False)
            y[idx] *= (1.0 + 0.35 * rng.standard_normal(n_pert))
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y * norm / s_y
            no_improve_count = 0

        if verbose and t % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ELPN t={t} ({elapsed:.0f}s): best={best_score:.8f}, imp={improved}, noimp={no_improve_count}")

    return best_a, best_score


def peak_targeted_cd(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_score(a)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        peak_thresh = cur_max * 0.93
        peak_lags = np.where(conv >= peak_thresh)[0]

        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 5e-11:
                continue

            # Candidate j's: random + peak-related + local
            j_candidates = set()
            for _ in range(5):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:10]:
                j1 = int(kl - i)
                j2 = int(kl + i)
                if 0 <= j1 < n:
                    j_candidates.add(j1)
                if 0 <= j2 < n:
                    j_candidates.add(j2)
            for delta in [-3, -2, -1, 1, 2, 3]:
                j_candidates.add((i + delta) % n)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]

                # Compute gain vector for moving mass from i to j
                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # Binary search for optimal delta
                lo_d = 0.01 * ai
                hi_d = 0.98 * ai
                best_d = 0.0
                best_v = cur_max
                for _ in range(12):
                    mid = (lo_d + hi_d) / 2.0
                    new_conv = conv + mid * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2.0 * ai * mid + mid * mid
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2.0 * aj * mid + mid * mid
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_d = mid
                    if v < cur_max:
                        lo_d = mid
                    else:
                        hi_d = mid

                if best_d > 0 and best_v < best_val - 1e-14:
                    best_val = best_v
                    best_delta = best_d
                    best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-14:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 12:
                break
            if no_improve == 5 or no_improve == 9:
                n_perturb = max(10, n // 30)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.15 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and total_improvements % 500 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD imp={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def multi_start_lp(a0, time_limit, solver, rng, n_restarts, verbose=True):
    n = len(a0)
    best_a = a0.copy()
    best_score = fast_conv_self(a0).max()
    t0 = time.monotonic()

    for r in range(n_restarts):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 40:
            break
        t_alloc = remaining / (n_restarts - r)

        if r == 0:
            start = a0.copy()
        else:
            start = best_a.copy()
            n_pert = max(15, n // 12)
            idx = rng.choice(n, size=n_pert, replace=False)
            start[idx] *= (1.0 + 0.25 * rng.standard_normal(n_pert))
            start = np.maximum(start, 0.0)
            s = start.sum()
            if s > 1e-15:
                start = start * np.sqrt(2.0 * n) / s

        a_out, sc = enhanced_lp_nesterov(start, t_alloc, solver, verbose=False, seed=int(time.monotonic() * 1000) + r)
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()
            if verbose:
                print(f"    MultiStart {r}: score={sc:.8f}")

    return best_a, best_score

        best_seq = best_init
        best_score = compute_score(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 0: Quick LSE warm-up
    remaining = time_budget - (time.monotonic() - t_global)
    t0_phase = min(remaining * 0.05, 80.0)
    if t0_phase > 15:
        print(f"\n=== Phase 0: LSE warm-up n={len(best_seq)} (budget={t0_phase:.0f}s) ===")
        a_out, _ = lse_gradient_descent(best_seq, t0_phase, verbose=False)
        do_try_update(a_out, "lse_init")

    # Phase 1: Main enhanced LP Nesterov
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = len(best_seq)
    t1 = min(remaining * 0.30, 500.0)
    print(f"\n=== Phase 1: Enhanced LP Nesterov n={n1} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = enhanced_lp_nesterov(best_seq, t1, solver, verbose=True, seed=seed)
        do_try_update(a_out, f"lp1_n{n1}")

    # Phase 2: Peak-targeted CD
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq)
    t2 = min(remaining * 0.14, 180.0)
    print(f"\n=== Phase 2: Peak CD n={n2} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = peak_targeted_cd(best_seq, t2, rng, verbose=True)
        do_try_update(a_out, f"pcd1_n{n2}")

    # Phase 3: LSE mid refinement
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = len(best_seq)
    t3 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 3: LSE mid n={n3} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_out, _ = lse_gradient_descent(best_seq, t3, verbose=False)
        do_try_update(a_out, "lse_mid")

    # Phase 4: LP Nesterov round 2
    remaining = time_budget - (time.monotonic() - t_global)
    n4 = len(best_seq)
    t4 = min(remaining * 0.22, 320.0)
    print(f"\n=== Phase 4: LP Nesterov r2 n={n4} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = enhanced_lp_nesterov(best_seq, t4, solver, verbose=True, seed=seed + 100)
        do_try_update(a_out, f"lp2_n{n4}")

    # Phase 5: Multi-start LP with perturbations
    remaining = time_budget - (time.monotonic() - t_global)
    n5 = len(best_seq)
    t5 = min(remaining * 0.14, 180.0)
    print(f"\n=== Phase 5: Multi-start LP n={n5} (budget={t5:.0f}s) ===")
    if t5 > 60:
        a_out, _ = multi_start_lp(best_seq, t5, solver, rng, n_restarts=3, verbose=True)
        do_try_update(a_out, f"multilp_n{n5}")

    # Phase 6: Final peak CD
    remaining = time_budget - (time.monotonic() - t_global)
    n6 = len(best_seq)
    t6 = min(remaining * 0.10, 100.0)
    print(f"\n=== Phase 6: Final PCD n={n6} (budget={t6:.0f}s) ===")
    if t6 > 15:
        a_out, _ = peak_targeted_cd(best_seq, t6, rng, verbose=False)
        do_try_update(a_out, f"pcd2_n{n6}")

    # Phase 7: Length extension
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 300
    print(f"\n=== Phase 7: Length extension n={n_long} (remaining={remaining:.0f}s) ===")
    if remaining > 100:
        init_long = interpolate_seq(best_seq, n_long)
        t_long = min(remaining * 0.5, 130.0)
        print(f"  Long LP budget: {t_long:.0f}s")
        a_out, _ = enhanced_lp_nesterov(init_long, t_long, solver, verbose=True, seed=seed + 200)
        do_try_update(a_out, f"lp_long_{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and len(best_seq) == n_long:
            t_cd = min(remaining * 0.4, 50.0)
            a_out2, _ = peak_targeted_cd(best_seq, t_cd, rng, verbose=False)
            do_try_update(a_out2, f"pcd_long_{n_long}")

    # Phase 8: Final combined polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 8: Final polish n={n_final} (remaining={remaining:.0f}s) ===")
    if remaining > 40:
        t_f1 = min(remaining * 0.35, 50.0)
        a_out, _ = lse_gradient_descent(best_seq, t_f1, verbose=False)
        do_try_update(a_out, "lse_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 25:
            t_f2 = min(remaining * 0.5, 50.0)
            a_out, _ = enhanced_lp_nesterov(best_seq, t_f2, solver, verbose=False, seed=seed + 300)
            do_try_update(a_out, "lp_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t_f3 = min(remaining * 0.6, 30.0)
            a_out, _ = peak_targeted_cd(best_seq, t_f3, rng, verbose=False)
            do_try_update(a_out, "pcd_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
