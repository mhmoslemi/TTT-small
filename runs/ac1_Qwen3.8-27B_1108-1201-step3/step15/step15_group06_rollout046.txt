        prev_y = y
        y = y_new
        a_prev = a_new
        a = a_new
        
        # Track best
        M_y = fast_autocorr(y_new).max()
        if M_y < best_M:
            best_M = M_y
            best_a = y_new.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        M_a = fast_autocorr(a_new).max()
        if M_a < best_M:
            best_M = M_a
            best_a = a_new.copy()
            stall_count = 0
        
        # Ergodic average
        avg_a = (n_avg * avg_a + a_new) / (n_avg + 1)
        n_avg += 1
        if n_avg % 50 == 0:
            M_avg = fast_autocorr(avg_a).max()
            if M_avg < best_M:
                best_M = M_avg
                best_a = avg_a.copy()
                stall_count = 0
        
        # Stall detection and restart
        if stall_count > stall_threshold:
            y = best_a.copy()
            a_new = best_a.copy()
            a_prev = best_a.copy()
            avg_a = best_a.copy()
            n_avg = 1
            active_set = None
            stall_count = 0
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best_M={best_M:.8f}")
    
    return best_a, best_M
