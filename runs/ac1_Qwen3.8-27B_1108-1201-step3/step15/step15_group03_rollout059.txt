import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None

def get_solver():

    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:

            prob.solve(solver=s)

        except Exception:

    _solver = cp.ECOS

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T


def lp_solve(a, M, solver, active=None, eps=1e-9):
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    if active is not None and 0 < len(active) < size:
        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)


def multiplicative_weights(a0, time_limit, verbose=True):
    """Mirror descent with negative entropy on the simplex."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * 1e-10
        s = a.sum()
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    eta = 1.0

    for it in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        max_c = conv.max()

        if max_c < 1e-15:
            break

        # Subgradient: gradient of the active max
        k_star = int(np.argmax(conv))
        # grad_i = 2 * a_{k_star - i} for valid indices
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]

        # Multiplicative update: a_i *= exp(-eta * grad_i)
        scale = -eta * grad
        scale -= scale.max()  # for numerical stability
        a_new = a * np.exp(scale)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        a = a_new

        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Adaptive eta: decrease slowly
        eta = max(0.01, 1.0 / (1.0 + 0.005 * it))

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MW it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lse_gradient_descent(a0, time_limit, verbose=True):
    """LSE smooth gradient descent with multiple temperature stages."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 15
    temps = np.geomspace(100.0, 0.1, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        n_steps = max(50, min(500, int(remaining * 8)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)

            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            step = norm / (T_val * np.sqrt(n)) * (1.0 / (1.0 + 0.03 * it))

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best={best_score:.8f}")

    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration and adaptive constraint screening."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1

    # Track progress for adaptive restarts
    last_best = best_score
    no_improve_iters = 0
    restart_count = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Adaptive constraint screening - be less aggressive early on
        if t < 50:
            active = None  # Use all constraints initially
        elif t < 200:
            threshold = M * 0.85
            active = np.where(conv_y >= threshold)[0]
            if len(active) < 20:
                active = None
        else:
            threshold = M * 0.92
            active = np.where(conv_y >= threshold)[0]
            if len(active) < 15:
                threshold = M * 0.75
                active = np.where(conv_y >= threshold)[0]
                if len(active) < 10:
                    active = None

        b = lp_solve(y, M, solver, active)
        if b is None:
            b = lp_solve(y, M, solver, None)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Verify dropped constraints periodically
        if active is not None and len(active) < size and t % 10 == 0:
            check_conv = fast_conv_cross(y, g)
            violated = np.where(check_conv > M * 1.005)[0]
            if len(violated) > 5:
                b_full = lp_solve(y, M, solver, None)
                if b_full is not None:
                    b = b_full
                    b_sum = b.sum()
                    if b_sum < 1e-15:
                        continue
                    g = b * norm / b_sum

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            if best_score < last_best - 1e-6:
                last_best = best_score
                no_improve_iters = 0
            else:
                no_improve_iters += 1

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            if best_score < last_best - 1e-6:
                last_best = best_score
                no_improve_iters = 0
            else:
                no_improve_iters += 1

        # Adaptive restart if stuck
        if no_improve_iters > 500 and restart_count < 5:
            restart_count += 1
            no_improve_iters = 0
            last_best = best_score
            # Perturb to escape
            rng_local = np.random.default_rng(t)
            perturb = 0.05 * rng_local.standard_normal(n)
            a_pert = a + perturb * norm / np.sqrt(n)
            a_pert = np.maximum(a_pert, 0.0)
            s_p = a_pert.sum()
            if s_p > 1e-15:
                a = a_pert * norm / s_p
                y = a.copy()
            if verbose:
                print(f"    LP Restart {restart_count} at t={t}: best={best_score:.8f}")

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def direct_subgradient(a0, time_limit, verbose=True):
    """Direct projected subgradient method with backtracking line search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()

    eta = norm / np.sqrt(n)
    momentum = np.zeros(n)
    beta = 0.7
    no_improve = 0

    for it in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        max_c = conv.max()
        k_star = int(np.argmax(conv))

        # Subgradient of c_{k*}
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]

        # Also consider near-active constraints for a smoother gradient
        threshold = max_c * 0.85
        active_lags = np.where(conv >= threshold)[0]
        if len(active_lags) > 1:
            grad_smooth = np.zeros(n)
            for k in active_lags[:20]:
                for i in range(n):
                    j = int(k) - i
                    if 0 <= j < n:
                        grad_smooth[i] += a[j]
            grad = grad_smooth / len(active_lags[:20])

        # Momentum update
        momentum = beta * momentum - eta * grad
        a_candidate = a + momentum
        a_candidate = np.maximum(a_candidate, 0.0)
        s_c = a_candidate.sum()
        if s_c < 1e-15:
            a_candidate = a.copy()
        else:
            a_candidate = a_candidate * norm / s_c

        # Backtracking line search
        sc = fast_conv_self(a_candidate).max()
        attempts = 0
        while sc > max_c and attempts < 10:
            eta *= 0.7
            momentum = beta * momentum - eta * grad
            a_candidate = a + momentum
            a_candidate = np.maximum(a_candidate, 0.0)
            s_c = a_candidate.sum()
            if s_c < 1e-15:
                a_candidate = a.copy()
                break
            a_candidate = a_candidate * norm / s_c
            sc = fast_conv_self(a_candidate).max()
            attempts += 1

        if sc < max_c:
            a = a_candidate
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            eta *= 0.8

        # Occasionally increase eta if we've been conservative
        if it % 200 == 0 and no_improve > 50:
            eta *= 1.05

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent with exact 1D minimization over transfer amounts."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv_self(a).max()
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]

            # Find good transfer targets based on peak structure
            peak_threshold = cur_max * 0.93
            peak_lags = np.where(conv >= peak_threshold)[0]

            j_candidates = set()
            for _ in range(8):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:5]:
                j_struct = (int(kl) - i) % n
                j_candidates.add(j_struct)
                j_candidates.add((int(kl) + i) % n)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in list(j_candidates)[:15]:
                if j == i:
                    continue
                aj = a[j]

                # Compute gain vector for transferring from i to j
                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # Sample deltas and find best
                n_samples = 30
                max_d = min(ai, norm * 0.1)
                if max_d < 1e-12:
                    continue
                deltas = np.linspace(0.0, max_d, n_samples)
                for d in deltas:
                    if d < 1e-12:
                        continue
                    new_conv = conv.copy()
                    new_conv += d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_val:
                        best_val = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                a = a * norm / s_new
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
                    total_improvements += 1

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
            if no_improve >= 2:
                n_perturb = max(10, n // 25)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.12 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and total_improvements % 500 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD impr={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, rng, verbose=True):
    """Targeted peak shaving: identify coordinates contributing most to peaks and reduce them."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv_self(a).max()
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        k_star = int(np.argmax(conv))

        # Find coordinates that contribute most to the peak
        # Contribution of a_i to c_{k*} is proportional to a_{k*-i}
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[j]

        # Also consider near-peak lags
        sorted_lags = np.argsort(conv)[::-1]
        for k in sorted_lags[1:min(10, size)]:
            if conv[k] >= cur_max * 0.90:
                for i in range(n):
                    j = int(k) - i
                    if 0 <= j < n:
                        contrib[i] += 0.5 * a[j]

        # Reduce high-contributing coordinates, increase low-contributing ones
        top_contrib = np.argsort(contrib)[-n // 5:]
        bottom_contrib = np.argsort(contrib)[:n // 5]

        # Transfer from top to bottom
        total_transfer = 0.0
        n_pairs = min(len(top_contrib), len(bottom_contrib))
        for p in range(n_pairs):
            src = top_contrib[p]
            dst = bottom_contrib[p]
            transfer = 0.02 * a[src]
            if a[src] - transfer > 0:
                a[src] -= transfer
                a[dst] += transfer
                total_transfer += transfer

        if total_transfer > 0:
            s_new = a.sum()
            a = a * norm / s_new
            new_max = fast_conv_self(a).max()
            if new_max < best_score:
                best_score = new_max
                best_a = a.copy()

        # Add small random perturbation for exploration
        if int(time.monotonic() - t0) % 30 < 1:
            pert_idx = rng.choice(n, size=n // 20, replace=False)
            a[pert_idx] += 0.01 * norm * rng.standard_normal(len(pert_idx))
            a = np.maximum(a, 0.0)
            s_new = a.sum()
            if s_new > 1e-15:
                a = a * norm / s_new

        if verbose and int((time.monotonic() - t0) // 30) > 0:
            elapsed = time.monotonic() - t0
            if int(elapsed) % 30 == 0:
                print(f"    PeakShave ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    """Diverse structured initializations."""
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.ones(n)
    elif choice == 1:
        a = 1.0 + 0.3 * np.cos(2 * np.pi * x)
    elif choice == 2:
        a = 1.0 + 0.2 * np.sin(2 * np.pi * 2 * x)
    elif choice == 3:
        a = np.exp(-0.5 * ((x - 0.5) / 0.3) ** 2)
    elif choice == 4:
        a = 0.7 + 0.3 * (1.0 - np.abs(2.0 * x - 1.0))
    elif choice == 5:
        a = 1.0 / (1.0 + 0.5 * np.cos(2 * np.pi * 3 * x))
    elif choice == 6:
        # Ramp (from small-n experiments)
        a = 0.5 + 0.5 * x
    elif choice == 7:
        a = 0.5 + 0.5 * (1.0 - x)
    elif choice == 8:
        # Two-bump
        a = np.exp(-8 * (x - 0.3) ** 2) + np.exp(-8 * (x - 0.7) ** 2)
    elif choice == 9:
        # Step function
        a = np.where(x < 0.5, 1.3, 0.7)
    elif choice == 10:
        a = 1.0 + 0.15 * np.sin(2 * np.pi * 5 * x + rng.random() * 2 * np.pi)
    else:
        a = np.ones(n) + 0.1 * np.sin(2 * np.pi * 7 * x)

    a = np.maximum(a, 0.01)
    a += 0.02 * rng.random(n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        if len(a_list) == 0:
            return float('inf')
        for val in a_list:
            if val < 0 or not np.isfinite(val):
                return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def nesterov_lp_optimize_v2(a0, time_limit, solver, verbose=True):
    """Improved LP Nesterov with adaptive parameters and periodic full-constraint solves."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1

    # Adaptive Nesterov with restarts
    momentum_param = 0.0
    last_improve_score = best_score
    stall_count = 0

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Constraint screening: use all constraints every 50th iteration for accuracy
        if t % 50 == 0:
            active = None
        elif t < 100:
            active = None
        else:
            # Use top constraints
            n_constraints = min(size, max(50, size // 4))
            top_idx = np.argsort(conv_y)[::-1][:n_constraints]
            active = top_idx

        b = lp_solve(y, M, solver, active)
        if b is None:
            b = lp_solve(y, M, solver, None)
        if b is None:
            if t % 1000 == 0:
                print(f"    LP failed at t={t}")
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Check constraint violations for screened solve
        if active is not None and len(active) < size:
            check_conv = fast_conv_cross(y, g)
            if check_conv.max() > M * 1.01:
                # Re-solve with all constraints
                b_full = lp_solve(y, M, solver, None)
                if b_full is not None:
                    b = b_full
                    b_sum = b.sum()
                    if b_sum > 1e-15:
                        g = b * norm / b_sum

        # Nesterov with adaptive momentum
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Adaptive momentum: increase when converging, decrease when oscillating
        progress = last_improve_score - conv_y.max()
        if progress > 1e-6:
            momentum_param = min(0.95, momentum_param + 0.01)
        elif progress < -1e-4:
            momentum_param = max(0.0, momentum_param - 0.05)

        a_new = y_new + momentum_param * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            if best_score < last_improve_score - 1e-7:
                last_improve_score = best_score
                stall_count = 0
            else:
                stall_count += 1

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            if best_score < last_improve_score - 1e-7:
                last_improve_score = best_score
                stall_count = 0

        # Restart if stuck for too long
        if stall_count > 300:
            stall_count = 0
            momentum_param = 0.0
            rng_restart = np.random.default_rng(t)
            a_pert = a * (1.0 + 0.03 * rng_restart.standard_normal(n))
            a_pert = np.maximum(a_pert, 0.0)
            s_p = a_pert.sum()
            if s_p > 1e-15:
                a = a_pert * norm / s_p
                y = a.copy()

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP2 t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Fast LSE warmup (quick shaping)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 100.0)
    print(f"\n=== Phase 1: LSE Warmup n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 15:
        a_work = best_seq.copy()
        a_out, _ = lse_gradient_descent(a_work, t1, verbose=True)
        try_update(a_out, "lse1")

    # Phase 2: Multiplicative weights for additional shaping
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.06, 80.0)
    print(f"\n=== Phase 2: Multiplicative Weights n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 10:
        a_out, _ = multiplicative_weights(best_seq, t2, verbose=True)
        try_update(a_out, "mw1")

    # Phase 3: Main LP Nesterov (improved version)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.30, 500.0)
    print(f"\n=== Phase 3: LP Nesterov v2 n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = nesterov_lp_optimize_v2(best_seq, t3, solver, verbose=True)
        try_update(a_out, "lp3")

    # Phase 4: Direct subgradient (no LP overhead, faster iterations)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 4: Direct SubGrad n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 15:
        a_out, _ = direct_subgradient(best_seq, t4, verbose=True)
        try_update(a_out, "subgrad4")

    # Phase 5: LP Nesterov round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.20, 320.0)
    print(f"\n=== Phase 5: LP Nesterov round 2 n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t5, solver, verbose=True)
        try_update(a_out, "lp5")

    # Phase 6: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.08, 120.0)
    print(f"\n=== Phase 6: Peak Shaving n={len(best_seq)} (budget={t6:.0f}s) ===")
    if t6 > 10:
        a_out, _ = peak_shaving(best_seq, t6, rng, verbose=True)
        try_update(a_out, "peaks")

    # Phase 7: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 7: Pair CD n={len(best_seq)} (budget={t7:.0f}s) ===")
    if t7 > 10:
        a_out, _ = pair_cd_optimize(best_seq, t7, rng, verbose=True)
        try_update(a_out, "pcd7")

    # Phase 8: Final LP polish
    remaining = time_budget - (time.monotonic() - t_global)
    t8 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 8: Final LP polish n={len(best_seq)} (budget={t8:.0f}s) ===")
    if t8 > 15:
        a_out, _ = nesterov_lp_optimize_v2(best_seq, t8, solver, verbose=False)
        try_update(a_out, "lp8")

    # Phase 9: Final pair CD
    remaining = time_budget - (time.monotonic() - t_global)
    t9 = min(remaining * 0.10, 100.0)
    print(f"\n=== Phase 9: Final PCD n={len(best_seq)} (budget={t9:.0f}s) ===")
    if t9 > 10:
        a_out, _ = pair_cd_optimize(best_seq, t9, rng, verbose=True)
        try_update(a_out, "pcd9")

    # Phase 10: Try slightly different lengths
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        n_trial_list = [len(best_seq) - 100, len(best_seq) + 100]
        for n_trial in n_trial_list:
            if n_trial <= 100:
                continue
            if remaining < 60:
                break
            print(f"\n=== Phase 10: Trial length n={n_trial} (budget={remaining:.0f}s) ===")
            init_trial = interpolate_seq(best_seq, n_trial)
            t_trial = min(remaining * 0.45, 180.0)
            if t_trial > 30:
                a_out, _ = nesterov_lp_optimize_v2(init_trial, t_trial, solver, verbose=False)
                try_update(a_out, f"trial_{n_trial}")
            remaining = time_budget - (time.monotonic() - t_global)

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
