def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)

def normalize(a, target_sum=None):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    n = len(a)
    if target_sum is None:
        target_sum = np.sqrt(2.0 * n)
    if s < 1e-15:

    return a * target_sum / s

def lp_solve(a, M, solver, active=None):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for k in active (or all), b >= 0."""
    if not HAS_CVXPY or solver is None:
        return None, None
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    b_var = cp.Variable(n, nonneg=True)
    if active is not None and 0 < len(active) < size:
        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    duals = None
    try:
        con = constraints[0]
        if con.dual_value is not None:
            if active is not None and len(active) < size:
                full_duals = np.zeros(size)
                full_duals[active] = np.asarray(con.dual_value, dtype=np.float64)
                duals = full_duals
            else:
                duals = np.asarray(con.dual_value, dtype=np.float64)
    except Exception:
        pass
    return val, duals

def sharp_norm_descent(a0, time_limit, p=8, verbose=False):
    """
    Gradient descent on the Lp norm of the autocorrelation (sharpened max).
    As p increases, this approaches the true max.
    """
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = normalize(a0, norm_val)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Anneal p from small to large
    n_stages = 8
    p_vals = np.geomspace(2, 50, n_stages)
    # ... (implementation)
    
    for si, p_cur in enumerate(p_vals):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(20, min(200, int(remaining * 3)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            c = fast_conv_self(a)
            max_c = c.max()
            if max_c < 1e-15:
                break
            
            # Lp norm and its gradient
            c_pos = np.maximum(c, 1e-30)
            lp_val = (np.sum(c_pos ** p_cur)) ** (1.0 / p_cur)
            
            # Gradient of Lp: dLp/da[i] = (1/p) * Lp^(1-p) * sum_k c[k]^(p-1) * dc[k]/da[i]
            weights = c_pos ** (p_cur - 1)
            W = rfft(weights, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)
            
            # Adaptive step size
            eta = 0.5 / (1.0 + 0.01 * it)
            for trial in range(10):
                a_new = a - eta * grad
                a_new = np.maximum(a_new, 0.0)
                a_new = normalize(a_new, norm_val)
                sc = compute_score(a_new)
                if sc <= best_score + 0.01:
                    break
                eta *= 0.5
            
            a = a_new
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose:
            print(f"    SharpNorm stage {si} p={p_cur:.1f}: best={best_score:.8f}")
    
    return best_a, best_score

def lp_nesterov(a0, time_limit, solver, verbose=False):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = normalize(a0, norm_val)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    y = a.copy()
    no_improve = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break
        
        # Adaptive screening
        if t < 100:
            thresh = 0.4
        elif t < 500:
            thresh = 0.6
        else:
            thresh = 0.75
        active = np.where(conv_y > M * thresh)[0]
        if len(active) < 20:
            active = None
        
        b, _ = lp_solve(y, M, solver, active)
        if b is None:
            b, _ = lp_solve(y, M, solver, None)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm_val / b_sum
        
        # Verify
        if active is not None:
            check = fast_conv_cross(y, g)
            if check.max() > M * 1.001:
                b_full, _ = lp_solve(y, M, solver, None)
                if b_full is not None and b_full.sum() > 1e-15:
                    g = b_full * norm_val / b_full.sum()
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize(y_new, norm_val)
        
        # Momentum extrapolation for evaluation
        momentum = min(1.0, (t - 1.0) / (t + 2.0))
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, norm_val)
        
        y = y_new
        a = a_new
        
        sc_y = compute_score(y)
        sc_a = compute_score(a)
        improved = False
        if sc_y < best_score - 1e-10:
            best_score = sc_y
            best_a = y.copy()
            improved = True
        if sc_a < best_score - 1e-10:
            best_score = sc_a
            best_a = a.copy()
            improved = True

            no_improve = 0

            no_improve = 0
        
        if verbose and t % 500 == 0:
            print(f"    LP-Nest t={t}: best={best_score:.8f}")
    
    return best_a, best_score

def peak_transfer_optimize(a0, time_limit, rng, verbose=False):
    """
    Targeted peak reduction: identify peak lags and perform optimal transfers.
    """
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = normalize(a0, norm_val)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fast_conv_self(a)
        max_c = c.max()
        if max_c < 1e-15:
            break
        
        # Find peak lags (top 10)
        peak_lags = np.argsort(c)[-10:][::-1]
        peak_threshold = max_c * 0.95
        active_peaks = peak_lags[c[peak_lags] > peak_threshold]
        
        pass_improved = False
        
        for k_star in active_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Find the most contributing pairs for this peak
            # c[k_star] = sum_i a[i] * a[k_star - i]
            i_min = max(0, k_star - n + 1)
            i_max = min(n - 1, k_star)
            
            if i_max < i_min:
                continue
            
            i_range = np.arange(i_min, i_max + 1)
            products = a[i_range] * a[k_star - i_range]
            
            # Find the pair with largest product
            if len(products) == 0:
                continue
            idx_max = np.argmax(products)
            i_src = i_range[idx_max]
            i_dst = k_star - i_range[idx_max]
            
            # We want to move mass from the larger of a[i_src], a[i_dst]
            # to a position that helps reduce other peaks
            if a[i_src] >= a[i_dst]:
                src = i_src
                other = i_dst
            else:
                src = i_dst
                other = i_src
            
            a_src = a[src]
            if a_src < norm_val * 1e-8:
                continue
            
            # Try several target positions
            best_target = -1
            best_gain = 0.0
            best_frac = 0.0
            
            # Candidate targets: positions that are in "valleys" of the autocorrelation
            # and are far from src and other
            candidates = []
            
            # Random candidates
            for _ in range(20):
                j = int(rng.integers(n))
                if j != src and j != other:
                    candidates.append(j)
            
            # Structured candidates: positions that contribute to low autocorrelation lags
            low_lags = np.argsort(c)[:5]
            for k_low in low_lags:
                # If we move mass to position j, it will affect c[k] for k near j
                # We want to increase c[k_low], so j should be such that
                # it interacts with existing mass to increase c[k_low]
                j1 = k_low  # a[j1] * a[0] contributes to c[k_low]
                j2 = n - 1 - k_low  # symmetric
                if 0 <= j1 < n and j1 != src:
                    candidates.append(j1)
                if 0 <= j2 < n and j2 != src:
                    candidates.append(j2)
            
            # Try fractions of mass to move
            fracs = [0.1, 0.2, 0.3, 0.5, 0.7]
            
            for j in set(candidates):
                if j == src:
                    continue
                for frac in fracs:
                    delta = frac * a_src
                    a_test = a.copy()
                    a_test[src] -= delta
                    a_test[j] += delta
                    a_test = np.maximum(a_test, 0.0)
                    a_test = normalize(a_test, norm_val)
                    
                    new_max = fast_conv_self(a_test).max()
                    gain = max_c - new_max
                    if gain > best_gain:
                        best_gain = gain
                        best_target = j
                        best_frac = frac
            
            if best_target >= 0 and best_gain > 1e-10:
                delta = best_frac * a[src]
                a[src] -= delta
                a[best_target] += delta
                a = np.maximum(a, 0.0)
                a = normalize(a, norm_val)
                
                new_score = compute_score(a)
                if new_score < best_score - 1e-12:
                    best_score = new_score
                    best_a = a.copy()
                    pass_improved = True
                else:
                    # Revert
                    a[best_target] -= delta
                    a[src] += delta
                    a = normalize(a, norm_val)
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
            if no_improve == 2:
                # Perturb
                n_pert = max(10, n // 20)
                idx = rng.choice(n, size=n_pert, replace=False)
                a[idx] *= (1.0 + 0.1 * rng.standard_normal(n_pert))
                a = normalize(a, norm_val)
                no_improve = 0
        else:
            no_improve = 0
    
    return best_a, best_score

def population_search(a_best, time_limit, rng, verbose=False):
    """
    Population-based evolutionary search.
    """
    n = len(a_best)
    norm_val = np.sqrt(2.0 * n)
    pop_size = 15
    t0 = time.monotonic()
    
    # Initialize population
    population = [a_best.copy()]
    scores = [compute_score(a_best)]
    
    for _ in range(pop_size - 1):
        # Create diverse candidates
        base = a_best.copy()
        n_pert = rng.integers(n // 10, n // 3)
        idx = rng.choice(n, size=n_pert, replace=False)
        base[idx] *= (1.0 + 0.3 * rng.standard_normal(n_pert))
        base = normalize(base, norm_val)
        population.append(base)
        scores.append(compute_score(base))
    
    # Evolutionary loop
    gen = 0
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        gen += 1
        
        # Sort by fitness
        order = np.argsort(scores)
        # Keep top half
        elite = [population[i].copy() for i in order[:pop_size // 2]]
        elite_scores = [scores[i] for i in order[:pop_size // 2]]
        
        # Generate offspring
        offspring = []
        for _ in range(pop_size - len(elite)):
            if time.monotonic() - t0 > time_limit:
                break
            if len(elite) >= 2:
                # Crossover
                i = rng.integers(len(elite))
                j = rng.integers(len(elite))
                alpha = rng.uniform(0.2, 0.8)
                child = alpha * elite[i] + (1 - alpha) * elite[j]
            else:
                child = elite[0].copy()
            
            # Mutation
            n_mut = rng.integers(n // 20, n // 5)
            idx = rng.choice(n, size=n_mut, replace=False)
            child[idx] *= (1.0 + 0.15 * rng.standard_normal(n_mut))
            child = normalize(child, norm_val)
            
            offspring.append(child)
        
        # Evaluate offspring
        for child in offspring:
            sc = compute_score(child)
            if sc < min(scores):
                # Replace worst
                worst_idx = np.argmax(scores)
                population[worst_idx] = child
                scores[worst_idx] = sc
        
        # Update best
        best_idx = np.argmin(scores)
        
        if verbose and gen % 10 == 0:
            print(f"    PopSearch gen={gen}: best={min(scores):.8f}")
        
        # If no improvement for a while, increase mutation rate
        if gen % 20 == 0:
            for i in range(len(population)):
                n_mut = rng.integers(n // 10, n // 4)
                idx = rng.choice(n, size=n_mut, replace=False)
                population[i][idx] *= (1.0 + 0.2 * rng.standard_normal(n_mut))
                population[i] = normalize(population[i], norm_val)
                scores[i] = compute_score(population[i])
    
    best_idx = np.argmin(scores)
    return population[best_idx], scores[best_idx]

def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 8)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 5 * x) + 0.2 * np.cos(2 * np.pi * 13 * x)
        a = np.maximum(a, 0.05)
    elif choice == 3:
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-4.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        a = 0.4 + 0.5 * np.sin(np.pi * x) ** 1.2
    elif choice == 6:
        # Split into two groups
        a = np.ones(n)
        a[n//3:2*n//3] *= 0.5
    else:
        a = 0.3 + 0.4 * np.cos(3 * np.pi * x) ** 2 + 0.2 * np.cos(7 * np.pi * x) ** 2
        a = np.maximum(a, 0.05)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
