Now let me also think about the PCD (pair coordinate descent) more carefully.

The PCD works by:
1. For each pair (i, j), try moving mass from i to j (or vice versa)
2. Find the move that most reduces the max convolution
3. Accept if it improves

The trick is: moving mass d from i to j changes the convolution by:
delta_c[k] = d * (a[k-j] - a[k-i]) (with appropriate bounds)

Plus there are second-order terms at k = 2i (decreases by d^2) and k = 2j (increases by d^2), and cross terms.

For efficiency, we only need to check the max of the new convolution, which can be done by:
new_max = max(constraint on the "affected" region)

The affected region is the union of the support of a shifted by i and j, which is a contiguous block of length at most 2n.

For the "precision polish" phase, we want to be very careful:
- Use exact convolution (not FFT approximation) for the final check
- Try many candidate j's for each i
- Use a fine grid of d values

def lse_adam_optimize(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize(a, norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    t_adam = 0

    n_stages = 24
    temps = np.geomspace(100.0, 0.03, n_stages)
    lr_base = 0.8 / np.sqrt(n)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(20, min(300, int(remaining * 8)))
        lr = lr_base * (1.0 + 0.5 * si / n_stages)

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            t_adam += 1

            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -60.0)
            exp_c = np.exp(shifted)
            w = exp_c / (exp_c.sum() + 1e-30)

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)

            ca = irfft(np.conj(A) * W, fft_size)
            g1 = ca[:n]

            aw = irfft(A * W, fft_size)
            g2 = aw[:n]

            grad = 2.0 * (g1 + g2)

            # Subtract mean for stability
            grad = grad - grad.sum() / n

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_adam)
            v_hat = v / (1.0 - beta2 ** t_adam)

            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new, norm)

            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose and (si % 6 == 0 or si == n_stages - 1):
            print(f"    LSE-Adam stage {si} T={T_val:.3f}: best={best_score:.8f}")

    return best_a, best_score

def build_active_T(a, active_indices):
    n = len(a)
    T = np.zeros((len(active_indices), n))
    for idx, k in enumerate(active_indices):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                T[idx, i] = a[j]
    return T

def lp_best_response(a, M, solver, active_indices=None):
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)

    if active_indices is not None and 0 < len(active_indices) < size:
        T = build_active_T(a, active_indices)
        constraints = [T @ b_var <= M]
    else:
        T = build_active_T(a, np.arange(size))
        constraints = [T @ b_var <= M]

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize(a, norm)

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    # Persistent active set
    conv0 = fast_conv_self(a)
    active_set = set(np.where(conv0 > best_score * 0.7)[0].tolist())
    if len(active_set) < 20:
        active_set = set(np.where(conv0 > best_score * 0.5)[0].tolist())
    if len(active_set) > 2000:
        top_idx = np.argsort(conv0)[-2000:]
        active_set = set(top_idx.tolist())

    y = a.copy()
    polyak_sum = np.zeros(n)
    max_active = 2500

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute current M
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Solve LP with current active set
        active_list = list(active_set)
        if len(active_list) > max_active:
            # Prune: keep only the most binding constraints
            conv_vals = conv_y[active_list]
            keep = np.argsort(conv_vals)[-max_active:]
            active_set = set(active_list[i] for i in keep)
            active_list = list(active_set)

        b = lp_best_response(y, M, solver, active_list)

        if b is None:
            # Try with fewer constraints
            if len(active_list) > 100:
                conv_vals = conv_y[active_list]
                keep = np.argsort(conv_vals)[-200:]
                active_set = set(active_list[i] for i in keep)
                b = lp_best_response(y, M, solver, list(active_set))
            if b is None:
                b = lp_best_response(y, M, solver, None)
            if b is None:
                continue

        # Check for violated constraints (constraint generation)
        if len(active_set) < size:
            check = fast_conv_cross(y, b)
            violated = np.where(check > M * 1.005)[0]
            violated = violated[~np.isin(violated, active_set)]
            if len(violated) > 0:
                # Add top violators
                viol_vals = check[violated]
                top_viol = violated[np.argsort(viol_vals)[-min(len(violated), 200):]]
                active_set.update(top_viol.tolist())
                # Re-solve with expanded active set
                active_list = list(active_set)
                if len(active_list) > max_active:
                    conv_vals = conv_y[active_list]
                    keep = np.argsort(conv_vals)[-max_active:]
                    active_set = set(active_list[i] for i in keep)
                    active_list = list(active_set)
                b2 = lp_best_response(y, M, solver, active_list)
                if b2 is not None:
                    b = b2

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize(y_new, norm)

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new, norm)

        y = y_new
        a = a_new
        polyak_sum += a

        # Track best (check both y and polyak average)
        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if t >= 10:
            polyak_avg = polyak_sum / t
            sc_polyak = fast_conv_self(polyak_avg).max()
            if sc_polyak < best_score:
                best_score = sc_polyak
                best_a = polyak_avg.copy()

        # Update active set periodically
        if t % 20 == 0:
            conv_a = fast_conv_self(a)
            current_max = conv_a.max()
            # Add new near-max positions
            new_candidates = np.where(conv_a > current_max * 0.85)[0]
            active_set.update(new_candidates.tolist())
            # Prune if too large
            if len(active_set) > max_active:
                all_conv = concatt = conv_a
                active_list = list(active_set)
                conv_vals = all_conv[active_list]
                keep = np.argsort(conv_vals)[-max_active:]
                active_set = set(active_list[i] for i in keep)

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M:.6f}, |A|={len(active_set)}")

    return best_a, best_score

def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize(a, norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            # Find peak lags
            peak_threshold = cur_max * 0.80
            peak_lags = np.where(conv >= peak_threshold)[0]

            # Generate j candidates
            j_candidates = set()
            for _ in range(12):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:15]:
                j_candidates.add(int((kl - i) % n))
                j_candidates.add(int((kl + i) % n))
            j_candidates.discard(i)

            best_delta = 0.0
            best_v = cur_max
            best_j = -1

            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]

                # Compute the change in convolution
                g = np.zeros(size)

                # Contribution from adding d at position j
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]

                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]

                # Subtraction from position i
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]

                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                d_max = ai * 0.9
                n_samples = 30
                deltas = np.linspace(0.02, d_max, n_samples)

                for d in deltas:
                    new_conv = conv + d * g
                    # Second order corrections
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    # Cross term at i+j
                    if i + j < size:
                        new_conv[i + j] += 2 * d * (a[max(0, i - (j - (i + j - i)))] if False else 0)  # approximate
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = normalize(a, norm)
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        # Revert
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = normalize(a, norm)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 8:
                n_perturb = max(5, n // 60)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.2 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                a = normalize(a, norm)
                if no_improve >= 15:
                    break
        else:
            no_improve = 0

        if verbose and total_improvements > 0 and total_improvements % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD improv={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def vegas_optimize(a0, time_limit, verbose=True):

            w = np.maximum(conv - M * 0.85, 0.0)
            ww = w / (w.sum() + 1e-15)

def structured_init(n, rng):
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)

    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        f1 = rng.integers(1, 5)
        f2 = rng.integers(5, 15)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * f1 * x) + 0.2 * np.cos(2 * np.pi * f2 * x)
    elif choice == 3:
        c1 = rng.uniform(0.2, 0.5)
        c2 = rng.uniform(0.5, 0.8)
        a = np.exp(-40.0 * (x - c1) ** 2) + np.exp(-40.0 * (x - c2) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x)
    elif choice == 6:
        n_peaks = rng.integers(3, 8)
        centers = rng.uniform(0.05, 0.95, n_peaks)
        a = np.zeros(n)
        for c in centers:
            width = rng.uniform(0.03, 0.08)
            a += np.exp(-(x - c) ** 2 / (2 * width ** 2))
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(2 * np.pi * 2 * x) + 0.2 * np.sin(2 * np.pi * 5 * x)
    elif choice == 8:
        a = np.sinc(3.0 * (x - 0.5)) ** 2 + 0.1
    else:
        a = 1.0 + 0.2 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def symmetrize(a):
    n = len(a)
    s_orig = a.sum()
    a_sym = (a + a[::-1]) / 2.0
    s_sym = a_sym.sum()
    if s_sym < 1e-15:
        return a.copy()
    return a_sym * (s_orig / s_sym) / s_orig * s_orig
    # Simpler: just normalize
    return a_sym / s_sym * s_orig / s_orig

def symmetrize_simple(a):
    a_sym = (a + a[::-1]) / 2.0
    s = a_sym.sum()
    if s < 1e-15:
        return a
    s_orig = a.sum()
    return a_sym * (s_orig / s)

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            old_best = best_score
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (was {old_best:.6f}, n={len(a)}, t={elapsed:.0f}s)")

    # Phase 1: LSE-Adam (fast initial progress)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 1: LSE-Adam n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = lse_adam_optimize(best_seq, t1, verbose=True)
        check_and_update(a_out, "lse1")

    # Phase 2: Vegas (multiplicative updates)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 2: Vegas n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = vegas_optimize(best_seq, t2, verbose=True)
        check_and_update(a_out, "vegas1")

    # Phase 3: LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.35, 550.0)
    print(f"\n=== Phase 3: LP Nesterov n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 50:
        a_out, _ = nesterov_lp_optimize(best_seq, t3, solver, verbose=True)
        check_and_update(a_out, "lp1")

    # Phase 4: PCD polish
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 350.0)
    print(f"\n=== Phase 4: PCD n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = pair_cd_optimize(best_seq, t4, rng, verbose=True)
        check_and_update(a_out, "pcd1")

    # Phase 5: Second LP round (from improved point)
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 5: LP round 2 n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t5, solver, verbose=False)
        check_and_update(a_out, "lp2")

    # Phase 6: Symmetry + sparsification attempts
    a_sym = symmetrize_simple(best_seq)
    check_and_update(a_sym, "sym")

    # Symmetrized PCD
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 6: Sym + PCD (budget={t6:.0f}s) ===")
    if t6 > 20:
        a_out, _ = pair_cd_optimize(a_sym, t6, rng, verbose=False)
        check_and_update(a_out, "sym_pcd")

    # Phase 7: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        for n_try in [n_base + 200, n_base - 200, n_base + 400]:
            if remaining < 50:
                break
            if n_try < 200 or n_try > 5000:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            t_try = min(remaining * 0.3, 60.0)
            print(f"\n=== Multi-length: n={n_try} (budget={t_try:.0f}s) ===")
            init_try = interpolate_seq(best_seq, n_try)
            a_out, _ = lse_adam_optimize(init_try, min(t_try * 0.5, 30.0), verbose=False)
            check_and_update(a_out, f"ml_lse_n{n_try}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                a_out2, _ = pair_cd_optimize(a_out, min(remaining * 0.4, 25.0), rng, verbose=False)
                check_and_update(a_out2, f"ml_pcd_n{n_try}")

    # Phase 8: Fresh starts
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        for attempt in range(3):
            if remaining < 20:
                break
            n_fresh = max(400, n_base + int(rng.integers(-200, 300)))
            a_fresh = structured_init(n_fresh, rng)
            t_fresh = min(remaining * 0.3, 25.0)
            a_out, _ = lse_adam_optimize(a_fresh, t_fresh, verbose=False)
            check_and_update(a_out, f"fresh{attempt}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 15:
                a_out2, _ = pair_cd_optimize(a_out, min(remaining * 0.4, 15.0), rng, verbose=False)
                check_and_update(a_out2, f"fresh{attempt}_pcd")

    # Final: one more PCD on best
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30 and best_seq is not None:
        print(f"\n=== Final PCD polish (remaining={remaining:.0f}s) ===")
        a_out, _ = pair_cd_optimize(best_seq, min(remaining * 0.5, 60.0), rng, verbose=True)
        check_and_update(a_out, "final_pcd")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
