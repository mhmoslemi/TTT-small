Better: For a given delta, the max is max_k (c_k + delta * h_k). As delta varies, the max changes at "break points" where the argmax changes. We can find the optimal delta by:
1. Start with delta = 0, current argmax = k0
2. Find the delta where k0 is no longer the max (i.e., where some other k catches up)
3. Update argmax, repeat

def lp_direct_maxmin(a, solver, active=None, M_hint=None):

def anderson_step(x_hist, g_hist, m):
    """Anderson acceleration step. Returns extrapolated point."""
    m_eff = min(m, len(x_hist))
    if m_eff <= 1:
        return g_hist[-1].copy()
    X = x_hist[-m_eff:]
    G = g_hist[-m_eff:]
    S = np.array([X[i] - G[i] for i in range(m_eff)])
    G_mat = S @ S.T
    ones = np.ones(m_eff)
    aug = np.zeros((m_eff + 1, m_eff + 1))
    aug[:m_eff, :m_eff] = 2.0 * G_mat
    aug[:m_eff, m_eff] = 1.0
    aug[m_eff, :m_eff] = 1.0
    rhs = np.zeros(m_eff + 1)
    rhs[m_eff] = 1.0
    try:
        sol = np.linalg.solve(aug, rhs)
        alpha = sol[:m_eff]
    except np.linalg.LinAlgError:
        alpha = np.ones(m_eff) / m_eff
    alpha = np.clip(alpha, -1.0, 3.0)
    s = alpha.sum()
    if abs(s) < 1e-10:
        alpha = np.ones(m_eff) / m_eff
    else:
        alpha = alpha / s
    x_new = G[-1].copy() + S.T @ alpha
    return x_new

    return a * target_sum / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def subgradient_polish(a0, time_limit, solver, verbose=True):
    """Polish using LP dual variables as subgradient with backtracking."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = norm_seq(np.asarray(a0, dtype=np.float64), norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    for it in range(500):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        M_curr = conv.max()

        # Get duals from LP
        active = np.where(conv > M_curr * 0.85)[0]
        b_sol, M_opt, duals = lp_direct_maxmin(a, solver, active)

        if duals is None:
            # Fallback: use active lags uniformly
            duals = np.zeros(size)
            duals[active] = 1.0 / max(len(active), 1)
        else:
            d_sum = duals.sum()
            if d_sum > 1e-15:
                duals = duals / d_sum

        # Compute gradient: grad_i = 2 * sum_k duals[k] * (a[k-i] + a[i-k])
        # This is 2 * (cross-correlation of duals and a)
        fft_size = 1 << (size - 1).bit_length()
        A = rfft(a, fft_size)
        D = rfft(duals, fft_size)
        g1 = irfft(np.conj(A) * D, fft_size)[:n]
        g2 = irfft(A * D, fft_size)[:n]
        grad = 2.0 * (g1 + g2)
        grad[norm // n:] = 0  # just to be safe

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Backtracking line search
        step = 0.01
        improved = False
        for trial in range(15):
            a_cand = a - step * grad
            a_cand = norm_seq(a_cand, norm)
            sc_cand = fast_conv_self(a_cand).max()
            if sc_cand < M_curr - 1e-12:
                a = a_cand
                improved = True
                break
            step *= 0.5

        if improved:
            sc = fast_conv_self(a).max()
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def coord_cd_phase(a0, time_limit, verbose=True):
    """Single-coordinate optimization using analytical 1D update."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = norm_seq(np.asarray(a0, dtype=np.float64), norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    no_improve_passes = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve_passes >= 5:
            break

        conv = fast_conv_self(a)
        cur_max = conv.max()
        pass_improved = False
        n_improvements = 0
        S = a.sum()

        # Process coordinates in order of contribution to peak lags
        peak_lags = np.where(conv >= cur_max * 0.95)[0]
        # Prioritize coordinates that appear in many peak lag computations
        priority = np.zeros(n)
        for kl in peak_lags[:10]:
            for i in range(n):
                j = kl - i
                if 0 <= j < n:
                    priority[i] += conv[kl]

        order = np.argsort(-priority)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = order[idx]
            ai = a[i]
            if ai < norm * 1e-8:
                continue

            # Influence vector: h_k = a[k-i] for valid k
            # c'_k = (S/(S+delta))^2 * (c_k + 2*delta*a[k-i] + delta^2*[k=2i])
            h = np.zeros(size)
            # a[k-i] is valid when 0 <= k-i < n, i.e., i <= k <= i+n-1
            lo = i
            hi = min(size, i + n)
            if hi > lo:
                h[lo:hi] = a[0:hi - lo]

            # Also need to handle the case where a[i-k] contributes
            # Actually the formula is c'_k = (S/(S+d))^2 * (c_k + d*(a[k-i] + a[i-k]) + d^2*[k=2i])
            # a[i-k] is valid when 0 <= i-k < n, i.e., i-n+1 <= k <= i
            lo2 = max(0, i - n + 1)
            hi2 = min(size, i + 1)
            if hi2 > lo2:
                h[lo2:hi2] += a[i - np.arange(lo2, hi2)]

            # Try several delta values
            deltas = np.array([-0.5, -0.3, -0.1, -0.05, 0.05, 0.1, 0.3, 0.5]) * ai
            best_delta = 0.0
            best_val = cur_max

            for d in deltas:
                if d == 0:
                    continue
                S_new = S + d
                if S_new <= 0:
                    continue
                scale = (S / S_new) ** 2
                c_new = scale * (conv + d * h)
                if 2 * i < size:
                    c_new[2 * i] += scale * d * d
                # Actually the d^2 term is already included in the formula
                # c_k + d*(a[k-i]+a[i-k]) + d^2*[k=2i]
                # The d^2*[k=2i] term:
                # We already have d*h which includes d*a[k-i] and d*a[i-k]
                # At k=2i: a[k-i] = a[i], a[i-k] = a[-i] = 0 (if i>0) or a[0] (if i=0)
                # So we need to add d^2 * [k=2i] separately
                # But wait, h[2i] = a[2i-i] + a[i-2i] = a[i] + a[-i] = a[i] (if i>0)
                # So the term is d*a[i], and the d^2 term is additional
                if 2 * i < size:
                    c_new[2 * i] += scale * d * d
                v = c_new.max()
                if v < best_val:
                    best_val = v
                    best_delta = d

            if best_delta != 0.0 and best_val < cur_max - 1e-12:
                a[i] += best_delta
                # Renormalize
                S_new = a.sum()
                if S_new < 1e-15:
                    a[i] -= best_delta
                    continue
                a = a * norm / S_new
                cur_max = fast_conv_self(a).max()
                if cur_max < best_score - 1e-12:
                    best_score = cur_max
                    best_a = a.copy()
                    pass_improved = True
                    n_improvements += 1

        if not pass_improved:
            no_improve_passes += 1
        else:
            no_improve_passes = 0

        if verbose and n_improvements > 0 and n_improvements % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordCD ({elapsed:.0f}s): best={best_score:.8f}, improvements={n_improvements}")

    return best_a, best_score

def pair_cd_phase(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent - move mass from one coordinate to another."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = norm_seq(np.asarray(a0, dtype=np.float64), norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False
        total_imp = 0

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-8:
                continue

            # Find candidate j's from peak lags
            peak_lags = np.where(conv >= cur_max * 0.9)[0]
            j_candidates = set()
            for kl in peak_lags[:6]:
                j_candidates.add((kl - i) % n)
                j_candidates.add((kl + i) % n)
            for _ in range(5):
                j_candidates.add(int(rng.integers(n)))
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]
                if aj < 0:
                    continue

                # Gain vector for moving delta from i to j
                g = np.zeros(size)
                # Remove contribution of a[i]
                lo = max(0, i)
                hi = min(size, i + n)
                if hi > lo:
                    g[lo:hi] -= a[0:hi - lo]
                lo2 = max(0, i - n + 1)
                hi2 = min(size, i + 1)
                if hi2 > lo2:
                    g[lo2:hi2] -= a[i - np.arange(lo2, hi2)]
                # Add contribution of a[j]
                lo3 = max(0, j)
                hi3 = min(size, j + n)
                if hi3 > lo3:
                    g[lo3:hi3] += a[0:hi3 - lo3]
                lo4 = max(0, j - n + 1)
                hi4 = min(size, j + 1)
                if hi4 > lo4:
                    g[lo4:hi4] += a[j - np.arange(lo4, hi4)]

                # Try several deltas
                for d in [0.1 * ai, 0.3 * ai, 0.5 * ai, 0.7 * ai]:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_val:
                        best_val = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                S_new = a.sum()
                if S_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / S_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_imp += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 6:
                break
            if no_improve == 3:
                n_pert = max(5, n // 30)
                idx_p = rng.choice(n, size=n_pert, replace=False)
                a[idx_p] *= (1.0 + 0.1 * rng.standard_normal(n_pert))
                a = norm_seq(a, norm)
        else:
            no_improve = 0

        if verbose and total_imp > 0 and total_imp % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PairCD ({elapsed:.0f}s): best={best_score:.8f}, imp={total_imp}")

    return best_a, best_score

def lse_quick(a0, time_limit, verbose=False):
    """Quick LSE gradient descent for initialization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = norm_seq(np.asarray(a0, dtype=np.float64), norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for T_val in [50.0, 20.0, 8.0, 3.0]:
        if time.monotonic() - t0 > time_limit:
            break
        n_steps = max(20, min(100, int((time_limit - (time.monotonic() - t0)) * 3)))
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            shifted = np.maximum((conv - max_c) / T_val, -200.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)
            step = norm / (T_val * np.sqrt(n)) / (1.0 + 0.03 * it)
            a = norm_seq(a - step * grad, norm)
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        if verbose:
            print(f"    LSE T={T_val}: best={best_score:.8f}")

    return best_a, best_score

def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function combining multiple optimization phases."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Solver: {solver}, Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_init = None
        best_init_score = float('inf')
        for trial in range(8):
            cand = structured_init(n_base, rng)
            sc = eval_fn(cand)
            if sc < best_init_score:
                best_init_score = sc
                best_init = cand
        best_seq = best_init
        best_score = best_init_score
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 0: Quick LSE warmup
    remaining = time_budget - (time.monotonic() - t_global)
    t0_budget = min(remaining * 0.05, 90.0)
    if t0_budget > 15:
        a_out, _ = lse_quick(best_seq, t0_budget, verbose=False)
        try_update(a_out, "lse_warmup")

    # Phase 1: Anderson-accelerated LP (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = len(best_seq)
    t1 = min(remaining * 0.35, 550.0)
    print(f"\n=== Phase 1: Anderson LP n={n1} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = anderson_lp_phase(best_seq, t1, solver, verbose=True)
        try_update(a_out, f"and_lp1_n{n1}")

    # Phase 2: Subgradient polish
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq)
    t2 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 2: Subgradient n={n2} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = subgradient_polish(best_seq, t2, solver, verbose=True)
        try_update(a_out, f"subgrad_n{n2}")

    # Phase 3: Anderson LP round 2 (with multiplicative variant)
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = len(best_seq)
    t3 = min(remaining * 0.22, 380.0)
    print(f"\n=== Phase 3: Anderson LP round 2 n={n3} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = anderson_lp_phase(best_seq, t3, solver, verbose=True)
        try_update(a_out, f"and_lp2_n{n3}")

    # Phase 4: Coordinate CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    n4 = len(best_seq)
    t4 = min(remaining * 0.15, 220.0)
    print(f"\n=== Phase 4: CoordCD n={n4} (budget={t4:.0f}s) ===")
    if t4 > 15:
        a_out, _ = coord_cd_phase(best_seq, t4, verbose=True)
        try_update(a_out, f"coordcd_n{n4}")

    # Phase 5: Pair CD
    remaining = time_budget - (time.monotonic() - t_global)
    n5 = len(best_seq)
    t5 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 5: PairCD n={n5} (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_out, _ = pair_cd_phase(best_seq, t5, rng, verbose=True)
        try_update(a_out, f"paircd_n{n5}")

    # Phase 6: Length extension
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 400
    print(f"\n=== Phase 6: Length extension n={n_long} (remaining={remaining:.0f}s) ===")
    if remaining > 120:
        init_long = interpolate_seq(best_seq, n_long)
        t_long = min(remaining * 0.4, 150.0)
        a_out, _ = anderson_lp_phase(init_long, t_long, solver, verbose=False)
        try_update(a_out, f"long_{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and len(best_seq) == n_long:
            t_cd = min(remaining * 0.4, 80.0)
            a_out2, _ = coord_cd_phase(best_seq, t_cd, verbose=False)
            try_update(a_out2, f"coordcd_long_{n_long}")

    # Phase 7: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 7: Final polish n={n_final} (remaining={remaining:.0f}s) ===")
    if remaining > 30:
        t_f1 = min(remaining * 0.3, 50.0)
        a_out, _ = anderson_lp_phase(best_seq, t_f1, solver, verbose=False)
        try_update(a_out, "and_lp_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t_f2 = min(remaining * 0.4, 40.0)
            a_out, _ = coord_cd_phase(best_seq, t_f2, verbose=False)
            try_update(a_out, "coordcd_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t_f3 = min(remaining * 0.5, 30.0)
            a_out, _ = subgradient_polish(best_seq, t_f3, solver, verbose=False)
            try_update(a_out, "subgrad_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
