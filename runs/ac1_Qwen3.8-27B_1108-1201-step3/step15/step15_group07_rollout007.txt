    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

    """Build matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

                break

            if i == j or a[i] < 1e-10:
                continue

                a[i] -= d

                if new_max < best_score - 1e-13:
                    best_score = new_max
                    best_a = a.copy()
                    improved = True
                    no_improve = 0
                    break
                else:
                    a[i] += d
                    a[j] -= d

            no_improve += 1
            # Perturb to escape
            if no_improve % 10 == 0:
                noise = rng.normal(0, norm * 0.01, n)

            elapsed = time.monotonic() - t0
