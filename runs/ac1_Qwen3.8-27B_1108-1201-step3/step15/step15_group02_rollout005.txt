import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def fast_conv_self(a):
    """Compute a*a (full convolution) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    """Compute a*b (full convolution) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def lp_best_response_constraints(a, M, constraints, solver):
    """Solve LP: max sum(g) s.t. rows of T @ g <= M, g >= 0
    where T[k, i] = a[k - i] (0 if out of bounds)
    """

        for i in range(max(0, k - n + 1), min(n, k + 1)):
            T[idx, i] = a[k - i]
    
    constraints_cp = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints_cp)
    try:

def kkt_polish(a0, time_limit, verbose=True):
    """Polish using KKT conditions: identify active set and support, then refine."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()

    best_a = a.copy()
    best_score = fast_conv_self(a).max()

        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_conv_self(a)
        M = conv.max()
        
        # Identify active set (constraints at or near M)
        active_tol = max(M * 0.01, M - best_score * 0.005)
        active_set = np.where(conv >= active_tol)[0]
        
        # Identify support
        support_thresh = norm / n * 0.05
        support = np.where(a > support_thresh)[0]
        
        if len(active_set) == 0 or len(support) == 0:
            break
        
        # Build the gradient matrix for KKT system
        # G[i] = 2 * sum_{k in active} y[k] * a[k-i] for i in support
        # At optimum: G[i] = lambda for all i in support
        # So: A_kkt @ y = lambda * 1 (where A_kkt[i,j] = 2*a[active[j] - support[i]])
        
        # This is a linear system. Solve for y such that A_kkt @ y = 1 (normalized)
        # Then check y >= 0
        
        m_act = len(active_set)
        m_sup = len(support)
        
        if m_act > 200 or m_sup > 200:
            # Too large for direct solve, skip
            continue
        
        A_kkt = np.zeros((m_sup, m_act))
        for i_idx, i in enumerate(support):
            for k_idx, k in enumerate(active_set):
                diff = k - i
                if 0 <= diff < n:
                    A_kkt[i_idx, k_idx] = 2.0 * a[diff]
        
        # Solve A_kkt @ y = 1 with y >= 0 (minimum norm solution)
        try:
            # Use least squares first, then project to non-negative
            y_ls, _, _, _ = np.linalg.lstsq(A_kkt, np.ones(m_sup), rcond=None)
            y_pos = np.maximum(y_ls, 0.0)
            y_sum = y_pos.sum()
            if y_sum < 1e-15:
                y_pos = np.ones(m_act) / m_act
            else:
                y_pos = y_pos / y_sum
        except Exception:
            continue
        
        # Check KKT: compute G for all i and see if it's approximately constant on support
        # G[i] = 2 * sum_{k in active} y_pos[k_idx] * a[k - i]
        # This is just A_kkt @ y_pos
        G = A_kkt @ y_pos
        
        # The KKT condition is G should be constant (all equal to lambda)
        # If not, we can use this to determine the direction of improvement
        G_mean = G.mean()
        G_var = G.std()
        
        if G_var < 1e-6 * G_mean:
            # KKT approximately satisfied - we're at a local optimum
            if verbose:
                print(f"    KKT satisfied at iter {outer_it}: G_var={G_var:.2e}")
            break
        
        # Use the KKT violation to determine improvement direction
        # Increase a[i] where G[i] < lambda, decrease where G[i] > lambda
        adjustment = np.zeros(n)
        for i_idx, i in enumerate(support):
            adjustment[i] = G_mean - G[i_idx]
        
        # Damped update
        step = 0.01 * norm
        a_new = a + step * adjustment
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        sc_new = fast_conv_self(a_new).max()
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            step *= 0.3
            a_new = a + step * adjustment
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new * norm / s_new
                sc_new = fast_conv_self(a_new).max()
                if sc_new < best_score:
                    best_score = sc_new
                    best_a = a_new.copy()
                    a = a_new
        
        if verbose and outer_it % 10 == 0:
            print(f"    KKT polish iter {outer_it}: best={best_score:.8f}, G_var={G_var:.2e}")
    
    return best_a, best_score

                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]
                
                d_max = min(ai, norm * 0.5 / n)
                if d_max < 1e-10:
                    continue
                n_samples = 20
                deltas = np.linspace(0.005, d_max, n_samples)
                
                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_delta = d
                        best_j = j
            
            if best_j >= 0 and best_delta > 0 and best_v < best_score - 1e-12:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 8:
                n_perturb = max(10, n // 50)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.1 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                if no_improve >= 15:
                    break
        else:
            no_improve = 0
        
        if verbose and total_improvements % 200 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD improv={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def structured_init(n, rng):
    choice = rng.integers(0, 14)
    x = np.linspace(0.0, 1.0, n)
    
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        f1 = rng.integers(1, 8)
        f2 = rng.integers(5, 25)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * f1 * x) + 0.2 * np.cos(2 * np.pi * f2 * x)
    elif choice == 3:
        c1 = rng.uniform(0.2, 0.5)
        c2 = rng.uniform(0.5, 0.8)
        a = np.exp(-40.0 * (x - c1) ** 2) + np.exp(-40.0 * (x - c2) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.5
    elif choice == 6:
        a = np.exp(-4.0 * x) + 0.3
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        n_peaks = rng.integers(3, 12)
        centers = rng.uniform(0.05, 0.95, n_peaks)
        a = np.zeros(n)
        for c in centers:
            width = rng.uniform(0.02, 0.08)
            a += np.exp(-(x - c) ** 2 / (2 * width ** 2))
    elif choice == 9:
        a = np.sinc(x - 0.5) ** 2 + 0.1
    elif choice == 10:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * x) + 0.2 * np.sin(2 * np.pi * 7 * x)
    elif choice == 11:
        # Logarithmic ramp
        a = np.log(1 + 100 * x)
    elif choice == 12:
        # Two-scale structure
        a = 0.6 + 0.4 * np.cos(2 * np.pi * rng.integers(2, 10) * x) * np.sin(2 * np.pi * rng.integers(10, 30) * x)
    else:
        # Power law
        alpha = rng.uniform(0.3, 1.5)
        a = x ** alpha + (1 - x) ** alpha
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a = a + 0.003 * np.abs(rng.standard_normal(n))
    a = a / a.sum()
    return a

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def symmetrize(a):
    n = len(a)
    total = a.sum()
    a_sym = (a + a[::-1]) / 2.0
    s = a_sym.sum()
    if s < 1e-15:
        return a
    return a_sym / s * total

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1680.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)
    
    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            old_best = best_score
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (was {old_best:.6f}, n={len(a)}, t={elapsed:.0f}s)")
    
    # Phase 1: Quick LSE-Adam for initial improvement
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 1: LSE-Adam n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_out, _ = lse_adam_optimize(best_seq, t1, verbose=True)
        check_and_update(a_out, f"lse1_n{n_base}")
    
    # Phase 2: Vegas for diversity
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.08, 100.0)
    print(f"\n=== Phase 2: Vegas n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = vegas_optimize(best_seq, t2, verbose=True)
        check_and_update(a_out, f"vegas1_n{len(best_seq)}")
    
    # Phase 3: Main LP Nesterov
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.30, 500.0)
    print(f"\n=== Phase 3: LP Nesterov n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t3, solver, verbose=True)
        check_and_update(a_out, f"lp1_n{len(best_seq)}")
    
    # Phase 4: KKT Polish
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.08, 80.0)
    print(f"\n=== Phase 4: KKT Polish n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 10:
        a_out, _ = kkt_polish(best_seq, t4, verbose=True)
        check_and_update(a_out, f"kkt1_n{len(best_seq)}")
    
    # Phase 5: Pair CD
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 5: Pair CD n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_out, _ = pair_cd_optimize(best_seq, t5, rng, verbose=True)
        check_and_update(a_out, f"pcd1_n{len(best_seq)}")
    
    # Phase 6: LP round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 6: LP round 2 n={len(best_seq)} (budget={t6:.0f}s) ===")
    if t6 > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t6, solver, verbose=False)
        check_and_update(a_out, f"lp2_n{len(best_seq)}")
    
    # Phase 7: KKT polish round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.06, 50.0)
    print(f"\n=== Phase 7: KKT Polish 2 n={len(best_seq)} (budget={t7:.0f}s) ===")
    if t7 > 10:
        a_out, _ = kkt_polish(best_seq, t7, verbose=True)
        check_and_update(a_out, f"kkt2_n{len(best_seq)}")
    
    # Phase 8: Pair CD final
    remaining = time_budget - (time.monotonic() - t_global)
    t8 = min(remaining * 0.10, 100.0)
    print(f"\n=== Phase 8: Pair CD final n={len(best_seq)} (budget={t8:.0f}s) ===")
    if t8 > 10:
        a_out, _ = pair_cd_optimize(best_seq, t8, rng, verbose=True)
        check_and_update(a_out, f"pcd2_n{len(best_seq)}")
    
    # Phase 9: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 9: Multi-length exploration (remaining={remaining:.0f}s) ===")
        lengths_to_try = sorted(set([n_base - 200, n_base - 100, n_base + 100, n_base + 200, n_base + 400]))
        for n_try in lengths_to_try:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 40:
                break
            if n_try < 200 or n_try > 4000:
                continue
            t_try = min(remaining * 0.30, 40.0)
            print(f"  Trying n={n_try} (budget={t_try:.0f}s)")
            init_try = interpolate_seq(best_seq, n_try)
            a_out, _ = lse_adam_optimize(init_try, min(t_try * 0.5, 20.0), verbose=False)
            check_and_update(a_out, f"lse_ml_n{n_try}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20 and a_out is not None:
                a_out2, _ = pair_cd_optimize(a_out, min(remaining * 0.4, 15.0), rng, verbose=False)
                check_and_update(a_out2, f"pcd_ml_n{n_try}")
    
    # Phase 10: Fresh starts
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 50:
        print(f"\n=== Phase 10: Fresh starts (remaining={remaining:.0f}s) ===")
        for attempt in range(4):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 20:
                break
            n_fresh = n_base + int(rng.integers(-300, 400))
            if n_fresh < 400:
                n_fresh = 800
            a_fresh = structured_init(n_fresh, rng)
            t_fresh = min(remaining * 0.25, 20.0)
            a_out, _ = lse_adam_optimize(a_fresh, t_fresh, verbose=False)
            check_and_update(a_out, f"fresh{attempt}_n{n_fresh}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 15 and a_out is not None:
                a_out2, _ = nesterov_lp_optimize(a_out, min(remaining * 0.3, 15.0), solver, verbose=False)
                check_and_update(a_out2, f"fresh_lp{attempt}_n{n_fresh}")
    
    # Final: try symmetrization and sparsification
    if best_seq is not None:
        a_sym = symmetrize(best_seq)
        check_and_update(a_sym, "sym_final")
        
        # Try zeroing small elements
        a_sparse = best_seq.copy()
        thresh = np.sort(a_sparse)[:int(0.02 * len(a_sparse))]
        a_sparse[a_sparse <= thresh.max()] = 0
        check_and_update(a_sparse, "sparse_final")
    
    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    
    return [float(x) for x in best_seq]
