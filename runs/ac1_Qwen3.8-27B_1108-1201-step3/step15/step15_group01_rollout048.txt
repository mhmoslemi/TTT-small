def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n + 1) > (css - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    cross = irfft(W * np.conj(A), fft_size)[:n]
    return 2.0 * cross

def anderson_accelerate(history, m=3):
    """Anderson acceleration: given history of (x_k, F(x_k)), return extrapolated point."""
    n_hist = len(history)
    n_points = min(m, n_hist)
    xs = [history[n_hist - 1 - i][0] for i in range(n_points)]
    Fs = [history[n_hist - 1 - i][1] for i in range(n_points)]
    # g_k = F(x_k) - x_k
    gs = [F - x for x, F in zip(xs, Fs)]
    if n_points == 1:
        return xs[0]
    # Build matrix G[i,j] = <g_i, g_j>, and b_i = <g_i, g_0>
    G = np.zeros((n_points, n_points))
    b = np.zeros(n_points)
    for i in range(n_points):
        for j in range(n_points):
            G[i, j] = np.dot(gs[i], gs[j])
        b[i] = np.dot(gs[i], gs[0])
    # Solve G * gamma = b, with constraint sum(gamma) = 1
    # Use: minimize ||sum gamma_i g_i||^2 s.t. sum gamma_i = 1
    # Lagrangian: G*gamma - mu*1 = 0, 1'*gamma = 1
    # [G  1; 1'  0] [gamma; mu] = [b; 1]
    A_aug = np.zeros((n_points + 1, n_points + 1))
    A_aug[:n_points, :n_points] = G
    A_aug[:n_points, n_points] = 1.0
    A_aug[n_points, :n_points] = 1.0
    rhs = np.zeros(n_points + 1)
    rhs[:n_points] = b
    rhs[n_points] = 1.0
    try:
        sol = np.linalg.solve(A_aug, rhs)
    except np.linalg.LinAlgError:
        return xs[0]
    gamma = sol[:n_points]
    # Clip negative gammas and renormalize
    gamma = np.maximum(gamma, 0)
    gs = gamma.sum()
    if gs < 1e-15:
        return xs[0]
    gamma = gamma / gs
    x_new = np.zeros_like(xs[0])
    for i, x in enumerate(xs):
        x_new += gamma[i] * x
    return x_new


def lp_anderson(a0, time_limit, solver, verbose=True):
    """LP best-response with Anderson acceleration and constraint generation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    history = [(a.copy(), a.copy())]
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    stuck = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        # Anderson extrapolation
        if len(history) >= 3:
            x_aa = anderson_accelerate(history, m=3)
            x_aa = np.maximum(x_aa, 0)
            s_aa = x_aa.sum()
            if s_aa > 1e-10:
                x_aa = x_aa * norm / s_aa
            else:
                x_aa = a.copy()
        else:
            x_aa = a.copy()

        conv_x = fast_autocorr(x_aa)
        M_curr = conv_x.max()
        if M_curr < 1e-12:
            break

        # Constraint generation: adaptive threshold
        threshold = 0.80 * M_curr
        active = np.where(conv_x >= threshold)[0]
        if len(active) < 10:
            threshold = 0.5 * M_curr
            active = np.where(conv_x >= threshold)[0]
        if len(active) < 5:
            active = np.where(conv_x >= 0.3 * M_curr)[0]
        # Always include the center and a few neighbors
        center = n - 1
        for c in range(max(0, center - 3), min(len(conv_x), center + 4)):
            if c not in active:
                active = np.append(active, c)
        active = np.unique(active)

        b = lp_best_response(x_aa, M_curr, solver, active_idx=active)
        if b is None:
            b = lp_best_response(x_aa, M_curr, solver, active_idx=None)
            if b is None:
                stuck += 1
                if stuck > 50:
                    break
                # Perturb and continue
                a = a + 0.01 * norm * np.random.randn(n)
                a = np.maximum(a, 0)
                a = a * norm / a.sum()
                continue

        stuck = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Damped update
        alpha = 2.0 / (t + 2.0)
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = np.maximum(a_new, 0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_a

        # Update history
        history.append((a.copy(), g.copy()))
        if len(history) > 10:
            history = history[-10:]
        a = a_new

        # Track best
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        # Cesàro average
        avg_a += a
        n_avg += 1
        if n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and n_iters % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Ander iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    # Also check the average at the end
    if n_avg > 0:
        avg_check = avg_a / n_avg
        s_avg = avg_check.sum()
        if s_avg > 1e-15:
            avg_check = avg_check * norm / s_avg
            score_avg = fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

    return best_a, best_score


def fista_smooth(a0, time_limit, verbose=True):
    """FISTA smooth optimization of log-sum-exp approximation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    temps = [30.0, 15.0, 8.0, 4.0, 2.0, 1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.008, 0.004]

    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        n_steps = max(50, min(800, int(remaining * 2.5)))

        y = a.copy()
        k = 0
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            k += 1

            c = fast_autocorr(y)
            c_max = c.max()
            exp_c = np.exp((c - c_max) / T)
            w = exp_c / exp_c.sum()

            grad = compute_smooth_grad(y, w)

            # FISTA step size
            L = 4.0 * n  # Lipschitz constant estimate
            step = 1.0 / L
            a_new = y - step * grad
            a_new = project_simplex(a_new, norm)

            # FISTA momentum
            k_prev = k - 1
            t_k = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * k_prev * k_prev))
            y_new = a_new + ((t_k - 1.0) / t_k) * (a_new - y)
            y_new = project_simplex(y_new, norm)

            y = y_new
            a = a_new

            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

            score_y = fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()

            if verbose and int(T * 100) % 5 == 0:
                elapsed = time.monotonic() - t0
                print(f"    FISTA T={T:.3f} step={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_cd(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent with peak-aware greedy selection."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        if no_improve >= 4:
            break
        pass_improved = False
        perm = rng.permutation(n)

        # Identify peak positions for prioritization
        peak_val = conv.max()
        peak_positions = set(np.where(conv >= 0.9 * peak_val)[0].tolist())

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                    break
            i = perm[idx]
            if a[i] < norm * 1e-9:
                continue

            # Try moving mass from i to nearby positions
            offsets = range(1, 31)
            if i in peak_positions:
                offsets = range(1, 61)

            for offset in offsets:
                if a[i] < norm * 1e-9:
                    break
                for j in [(i + offset) % n, (i - offset) % n]:
                    if j == i:
                        continue
                    # Quick test
                    d = a[i] * 0.2
                    a_t = a.copy()
                    a_t[i] -= d
                    a_t[j] += d
                    a_t = a_t * norm / a_t.sum()
                    conv_t = fast_autocorr(a_t)
                    if conv_t.max() < best_score * 0.9999:
                        a = a_t
                        conv = conv_t
                        best_score = conv.max()
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0
                        break
                    else:
                        # Try smaller step
                        d2 = a[i] * 0.05
                        a_t2 = a.copy()
                        a_t2[i] -= d2
                        a_t2[j] += d2
                        a_t2 = a_t2 * norm / a_t2.sum()
                        conv_t2 = fast_autocorr(a_t2)
                        if conv_t2.max() < conv.max() * 0.99999:
                            a = a_t2
                            conv = conv_t2
                            if conv.max() < best_score:
                                best_score = conv.max()
                                best_a = a.copy()
                            pass_improved = True
                            no_improve = 0

        if not pass_improved:
            no_improve += 1

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def make_parametric_init(n, kind, rng):
    """Generate various parametric initializations."""
    if kind == 0:  # uniform
        return np.ones(n) / n
    elif kind == 1:  # gaussian
        x = np.linspace(0, 1, n)
        f = np.exp(-20 * (x - 0.5) ** 2)
        return f / f.sum()
    elif kind == 2:  # triangular
        x = np.linspace(0, 1, n)
        f = np.maximum(0, 1 - 2 * np.abs(x - 0.5))
        return f / f.sum()
    elif kind == 3:  # raised cosine
        x = np.linspace(0, 1, n)
        f = 0.5 * (1 + np.cos(2 * np.pi * x))
        return f / f.sum()
    elif kind == 4:  # two bumps
        x = np.linspace(0, 1, n)
        f = np.exp(-50 * (x - 0.3) ** 2) + np.exp(-50 * (x - 0.7) ** 2)
        return f / f.sum()
    elif kind == 5:  # comb
        f = np.zeros(n)
        f[::2] = 1.0
        if f.sum() < 1e-10:
            f = np.ones(n)
        return f / f.sum()
    elif kind == 6:  # smooth random
        f = np.abs(rng.standard_normal(n))
        # Smooth it
        kernel = np.ones(5) / 5
        f = np.convolve(f, kernel, mode='same')
        f = np.maximum(f, 0)
        return f / f.sum()
    else:
        return np.ones(n) / n


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    return f_interp * np.sqrt(2.0 * new_n) / ss


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search: LP+Anderson, FISTA smooth, pair CD, multi-start, length exploration."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def do_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-10:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 0: Initialization (5% time) ===
    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    t_init = time_budget * 0.05
    t_init_end = t_global + t_init
    print(f"\n=== Phase 0: Initialization (budget={t_init:.0f}s) ===")

    # Try several initializations, keep the sum of parent and best params as candidates
    init_candidates = [best_seq.copy()]
    n_try = n_base
    for kind in range(7):
        if time.monotonic() > t_init_end:
            break
        init = make_parametric_init(n_try, kind, rng)
        # Blend with parent: try 50-50, 70-30
        for frac in [0.5, 0.7, 0.3]:
            blended = frac * best_seq + (1 - frac) * init
            blended = blended / blended.sum()
            sc = evaluate_safe(blended)
            if sc < best_score:
                best_score = sc
                best_seq = blended.copy()
                print(f"  Init blend kind={kind} frac={frac}: {sc:.6f}")
            init_candidates.append(blended)

    # Quick LP on the best init
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.45, time_budget * 0.40)
    if t1 < 30:
        t1 = max(t1, 30)

    # === Phase 1: LP + Anderson (40% time) ===
    n_c = len(best_seq)
    print(f"\n=== Phase 1: LP+Anderson n={n_c} (budget={t1:.0f}s) ===")
    a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
    a_out, sc_out = lp_anderson(a_norm, t1, solver, verbose=True)
    do_update(a_out, f"lp_and_n{n_c}")

    # === Phase 2: FISTA smooth (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t2 = min(remaining * 0.25, time_budget * 0.20)
    print(f"\n=== Phase 2: FISTA n={n_c} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, sc_out = fista_smooth(a_norm, t2, verbose=True)
        do_update(a_out, f"fista_n{n_c}")

    # === Phase 3: Pair CD (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t3 = min(remaining * 0.20, time_budget * 0.15)
    print(f"\n=== Phase 3: Pair CD n={n_c} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, sc_out = pair_cd(a_norm, t3, rng, verbose=True)
        do_update(a_out, f"pcd_n{n_c}")

    # === Phase 4: LP round 2 on refined (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t4 = min(remaining * 0.15, time_budget * 0.10)
    print(f"\n=== Phase 4: LP round 2 n={n_c} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, sc_out = lp_anderson(a_norm, t4, solver, verbose=True)
        do_update(a_out, f"lp2_n{n_c}")

    # === Phase 5: FISTA + PCD polish (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t5 = min(remaining * 0.15, time_budget * 0.10)
    print(f"\n=== Phase 5: Polish n={n_c} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        t5a = t5 * 0.5
        a_out, _ = fista_smooth(a_norm, t5a, verbose=False)
        do_update(a_out, "polish_fista")
        remaining5 = time_budget - (time.monotonic() - t_global)
        if remaining5 > 10:
            n_c2 = len(best_seq)
            a_norm2 = best_seq * np.sqrt(2.0 * n_c2) / best_seq.sum()
            a_out2, _ = pair_cd(a_norm2, min(remaining5 * 0.4, 60.0), rng, verbose=False)
            do_update(a_out2, "polish_pcd")

    # === Phase 6: Diversified restarts (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60 and best_seq is not None:
        n_c = len(best_seq)
        print(f"\n=== Phase 6: Diversified restarts n={n_c} (budget={remaining:.0f}s) ===")

        for restart in range(3):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 40:
                break
            t_r = min(remaining * 0.3, 120.0)

            # Perturb
            a_pert = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
            noise_scale = 0.15 + 0.1 * restart
            noise = rng.normal(0, noise_scale * a_pert.std(), n_c)
            a_pert = a_pert + noise
            a_pert = np.maximum(a_pert, 0)
            s_p = a_pert.sum()
            if s_p > 1e-10:
                a_pert = a_pert * np.sqrt(2.0 * n_c) / s_p

            # Quick LP
            a_out_r, _ = lp_anderson(a_pert, t_r * 0.6, solver, verbose=False)
            do_update(a_out_r, f"restart_lp_{restart}")

            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 15:
                a_out_r2, _ = fista_smooth(a_pert, min(remaining * 0.3, 40.0), verbose=False)
                do_update(a_out_r2, f"restart_fista_{restart}")

    # === Phase 7: Length exploration (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 90 and best_seq is not None:
        n_cur = len(best_seq)
        print(f"\n=== Phase 7: Length exploration (budget={remaining:.0f}s) ===")
        for n_alt in [n_cur + 200, n_cur - 200, n_cur + 400, n_cur - 400]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 40:
                break
            if n_alt < 200:
                continue
            t_alt = min(remaining * 0.3, 80.0)
            init_alt = interpolate_sequence(best_seq, n_alt)
            a_out_alt, _ = lp_anderson(init_alt, t_alt, solver, verbose=False)
            do_update(a_out_alt, f"alt_n{n_alt}")

    # === Final verification ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
