# Global solver cache

def get_solver():

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_T(a):
    """Build the convolution matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T


def lp_solve_constr_gen(a, M, solver, max_iter=20, verbose=False):
    """
    LP with constraint generation: iteratively find violated constraints.
    Returns the optimal b for: max sum(b) s.t. (a*b)[k] <= M, b >= 0.
    """
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    
    # Start with all constraints
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b = np.asarray(b_var.value, dtype=np.float64)
    b = np.maximum(b, 0.0)
    return b

    n = len(a)
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T[active_indices, :] @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b = np.asarray(b_var.value, dtype=np.float64)
    b = np.maximum(b, 0.0)
    return b


def lse_gradient_descent(a0, time_limit, verbose=True):
    """LSE-smoothed gradient descent on the max convolution."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Multiple temperature stages with more refined schedule
    n_stages = 16
    temps = np.geomspace(60.0, 0.1, n_stages)
    
    for si, T_val in enumerate(temps):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        remaining = time_limit - elapsed
        if remaining < 2:
            break
        
        n_steps = max(30, min(400, int(remaining * 8)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            
            # Soft-max weighting (LSE gradient)
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            exp_sum = exp_c.sum()
            if exp_sum < 1e-30:
                w = np.zeros(size)
                w[np.argmax(conv)] = 1.0
            else:
                w = exp_c / exp_sum
            
            # Gradient via FFT
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)
            
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            
            # Adaptive step size
            step = norm / (T_val * np.sqrt(n)) * (1.0 / (1.0 + 0.03 * it))
            
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new
            
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration and constraint generation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1
    
    # For constraint generation: keep a growing set of active constraints
    active_set = None  # None means use all
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break
        
        # Adaptive active set: top constraints by value
        # Use a tiered approach
        if t < 50:
            # First few iterations: use all constraints for accuracy
            active_set = None
        else:
            # After warmup, use active set
            threshold = M * 0.85
            active_idx = np.where(conv_y >= threshold)[0]
            if len(active_idx) < 20:
                threshold = M * 0.5
                active_idx = np.where(conv_y >= threshold)[0]
            if len(active_idx) < 10:
                active_idx = np.where(conv_y >= M * 0.3)[0]
            active_set = active_idx
        
        if active_set is None:
            b = lp_solve_constr_gen(y, M, solver)
        else:
            b = lp_solve_active(y, M, solver, active_set)
        
        if b is None:
            b = lp_solve_constr_gen(y, M, solver)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        
        g = b * norm / b_sum
        
        # Verify dropped constraints (if using active set)
        if active_set is not None:
            check_conv = fast_conv_cross(y, g)
            violated = np.where(check_conv > M * 1.005)[0]
            if len(violated) > 5:
                # Re-solve with violated constraints added
                new_active = np.union1d(active_set, violated)
                b2 = lp_solve_active(y, M, solver, new_active)
                if b2 is not None:
                    b = b2
                    b_sum = b.sum()
                    if b_sum < 1e-15:
                        continue
                    g = b * norm / b_sum
                    active_set = new_active
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    """
    Pair coordinate descent with structured neighbor sampling and 
    vectorized 1D minimization.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv_self(a).max()
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False
        
        # Identify peak lags for structured candidate generation
        peak_threshold = cur_max * 0.93
        peak_lags = np.where(conv >= peak_threshold)[0]
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue
            
            # Generate candidate j's: random + structured from peak lags
            j_candidates = set()
            for _ in range(15):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:6]:
                j1 = int((kl - i) % n)
                j2 = int((kl + i) % n)
                if 0 <= kl - i < n:
                    j_candidates.add(kl - i)
                if 0 <= kl + i < n:
                    j_candidates.add(kl + i)
                j_candidates.add(j1)
                j_candidates.add(j2)
            # Also add neighbors
            for d in range(-3, 4):
                jn = (i + d) % n
                if jn != i:
                    j_candidates.add(jn)
            j_candidates.discard(i)
            
            best_delta = 0.0
            best_val = cur_max
            best_j = -1
            
            for j in list(j_candidates)[:25]:
                if j == i:
                    continue
                aj = a[j]
                if aj < 1e-15 and ai < 1e-15:
                    continue
                
                # Compute the change vector for transferring mass from i to j
                # conv_new = conv + delta * (a shifted to j - a shifted from i)
                # The change in conv at position k is:
                # delta * (sum of a[m] where the j-shift covers k) - delta * (sum of a[m] where the i-shift covers k)
                # Plus the diagonal terms at 2j and 2i
                
                # Precompute the gain vector g where conv_new = conv + delta * g + diagonal corrections
                g = np.zeros(size)
                # Adding to j: conv contribution is a convolved with e_j (unit at j)
                # (a * e_j)[k] = a[k-j] if 0 <= k-j < n
                lo_j = max(0, j)
                hi_j = min(size, j + n)
                if hi_j > lo_j:
                    g[lo_j:hi_j] += a[0:hi_j - lo_j]
                
                # Also the "other direction": (e_j * a)[k] = a[j-k] if 0 <= j-k < n
                # Wait, np.convolve(a, b)[k] = sum_i a[i]*b[k-i]
                # So (a * e_j)[k] = sum_i a[i] * delta_{k-i,j} = a[k-j] if valid
                # And (e_j * a)[k] = sum_i delta_{i,j} * a[k-i] = a[k-j] if valid
                # They're the same! So the total change from a[i] -> a[i]-delta, a[j] -> a[j]+delta:
                # The change in conv is: -delta * (a shifted from i) + delta * (a shifted to j)
                # where "a shifted from i" means the contribution that a[i] makes to conv
                # a[i] contributes to conv at positions 0 to 2i: conv[k] includes a[i]*a[k-i] for valid k-i
                # So removing delta from a[i] changes conv[k] by -delta * a[k-i] for k in [i, i+n-1] (from the a[k-i] term)
                # AND -delta * a[i-k] for k in [0, i] (from the a[i-k] term, but wait...)
                
                # Actually let me be more careful.
                # conv[k] = sum_m a[m] * a[k-m]
                # If we change a[i] -> a[i] - delta and a[j] -> a[j] + delta:
                # New conv[k] = conv[k] - delta * (a[k-i] + a[i-k] - (delta if k==2i else 0))
                #              + delta * (a[k-j] + a[j-k] + (delta if k==2j else 0))
                # Wait, let me redo this.
                # conv_new[k] = sum_m a'[m] * a'[k-m]
                # where a' differs from a only at i and j.
                # a'[i] = a[i] - delta, a'[j] = a[j] + delta
                # 
                # conv_new[k] = conv[k] 
                #   + (a'[i]-a[i]) * (a[k-i] if k!=i else a'[i]+a[i]-a[i])  ... this gets complicated
                # Let me use linearity:
                # conv_new = (a - delta*e_i + delta*e_j) * (a - delta*e_i + delta*e_j)
                # = a*a - delta*(e_i*a) - delta*(a*e_i) + delta*(e_j*a) + delta*(a*e_j)
                #   + delta^2*(e_i*e_i) - delta^2*(e_i*e_j) - delta^2*(e_j*e_i) + delta^2*(e_j*e_j)
                
                # (e_i * a)[k] = a[k-i] (valid when 0 <= k-i < n, i.e., i <= k <= i+n-1)
                # (a * e_i)[k] = a[k-i] (same thing by commutativity)
                # So (e_i * a + a * e_i)[k] = 2*a[k-i] for i <= k <= i+n-1, but with care at k=i where it's a[0]+a[0]=2a[0]... 
                # Actually no. (e_i * a)[k] = sum_m e_i[m]*a[k-m] = a[k-i] (when 0<=k-i<n)
                # (a * e_i)[k] = sum_m a[m]*e_i[k-m] = a[k-i] (when 0<=k-i<n)
                # So yes, they're the same, and the total is 2*a[k-i] for i <= k <= i+n-1.
                
                # But wait, at k = 2i: (e_i * e_i)[2i] = 1, so the delta^2 term adds delta^2 at k=2i.
                # Similarly at k=2j: delta^2 at k=2j.
                # At k=i+j: (e_i * e_j)[i+j] = 1 and (e_j * e_i)[i+j] = 1, so -2*delta^2 at k=i+j (if i!=j).
                
                # So: change in conv[k] = delta * (2*a[k-j] - 2*a[k-i]) + delta^2 * (delta_{k,2j} - delta_{k,2i} - 2*delta_{k,i+j})
                # (where a[k-i] = 0 if k-i not in [0,n-1], etc.)
                
                # Let me recompute g properly:
                g[:] = 0.0
                # Term: 2*a[k-j] for j <= k <= j+n-1
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += 2.0 * a[0:hi - lo]
                # Term: -2*a[k-i] for i <= k <= i+n-1
                lo2 = max(0, i)
                hi2 = min(size, i + n)
                if hi2 > lo2:
                    g[lo2:hi2] -= 2.0 * a[0:hi2 - lo2]
                
                # Diagonal corrections: +1 at k=2j, -1 at k=2i, -2 at k=i+j
                if 2 * j < size:
                    g[2 * j] += 1.0
                if 2 * i < size:
                    g[2 * i] -= 1.0
                if i + j < size:
                    g[i + j] -= 2.0
                
                # Now conv_new[k] = conv[k] + delta * g[k]
                # Wait, I have the delta^2 terms in g, but those should be separate.
                # Let me redo: conv_new = conv + delta * h + delta^2 * q
                # where h[k] = 2*(a[k-j] - a[k-i]) (valid indices)
                # and q[k] = delta_{k,2j} - delta_{k,2i} - 2*delta_{k,i+j}
                
                # For the search, let's compute new_conv = conv + delta * h + delta^2 * q
                # and find delta that minimizes max(new_conv)
                
                h = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    h[lo:hi] += 2.0 * a[0:hi - lo]
                lo2 = max(0, i)
                hi2 = min(size, i + n)
                if hi2 > lo2:
                    h[lo2:hi2] -= 2.0 * a[0:hi2 - lo2]
                
                q = np.zeros(size)
                if 2 * j < size:
                    q[2 * j] += 1.0
                if 2 * i < size:
                    q[2 * i] -= 1.0
                if i + j < size:
                    q[i + j] -= 2.0
                
                # Search for best delta in [0, ai]
                # new_conv[k] = conv[k] + delta * h[k] + delta^2 * q[k]
                # We want to minimize max_k new_conv[k]
                # This is a 1D problem. Use bisection or golden section.
                
                # Quick approach: sample several deltas and pick the best
                n_samples = 30
                deltas = np.linspace(0.0, ai * 0.99, n_samples)
                best_d = 0.0
                best_mv = cur_max
                
                for d in deltas:
                    if d < 1e-12:
                        continue
                    new_conv = conv + d * h + d * d * q
                    v = new_conv.max()
                    if v < best_mv:
                        best_mv = v
                        best_d = d
                
                if best_j < 0 or best_mv < best_val:
                    best_delta = best_d
                    best_val = best_mv
                    best_j = j
            
            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / s_new
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
                    total_improvements += 1
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
            # Perturb to escape local minimum
            n_perturb = max(10, n // 20)
            idx_perturb = rng.choice(n, size=n_perturb, replace=False)
            a[idx_perturb] *= (1.0 + 0.1 * rng.standard_normal(n_perturb))
            a = np.maximum(a, 0.0)
            s_new = a.sum()
            if s_new > 1e-15:
                a = a * norm / s_new
        else:
            no_improve = 0
    
    return best_a, best_score


def peak_targeted_perturbation(a0, time_limit, rng, verbose=True):
    """
    Specifically targets the peak lags by redistributing mass away from 
    coordinates that contribute most to the peaks.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv_self(a).max()
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        
        # Find top peak lags
        top_k = min(20, size)
        peak_lags = np.argsort(conv)[::-1][:top_k]
        
        # Compute pressure: for each coordinate i, how much does it contribute to peaks?
        pressure = np.zeros(n)
        for kl in peak_lags[:10]:
            for i in range(n):
                # a[i] contributes to conv[kl] via a[i]*a[kl-i] and a[kl-i]*a[i]
                # (they're the same term in the sum)
                # conv[kl] = sum_m a[m]*a[kl-m]
                # a[i] appears when m=i (giving a[i]*a[kl-i]) or when kl-m=i i.e. m=kl-i (giving a[kl-i]*a[i])
                # Same term! So a[i] contributes a[i]*a[kl-i] to conv[kl] (if kl-i is valid)
                # Plus a[i]*a[kl-i] again... no wait, it's just one term.
                # conv[kl] = sum_{m: 0<=m<n, 0<=kl-m<n} a[m]*a[kl-m]
                # For a fixed i, a[i] appears in this sum when m=i (if kl-i valid) giving a[i]*a[kl-i]
                # and when kl-m=i i.e. m=kl-i (if kl-i valid) giving a[kl-i]*a[i]
                # These are the same term (just relabeled), so the total contribution of a[i] to conv[kl]
                # is 2*a[i]*a[kl-i] if both i and kl-i are valid, or a[i]^2 if kl=2i.
                
                idx = kl - i
                if 0 <= idx < n:
                    if idx == i:
                        pressure[i] += a[i]  # self-contribution (a[i]^2 term)
                    else:
                        pressure[i] += 2.0 * a[idx]
        
        # Normalize pressure
        p_max = pressure.max()
        if p_max < 1e-15:
            break
        pressure = pressure / p_max
        
        # For each high-pressure coordinate, try to move mass to a low-pressure coordinate
        perm = rng.permutation(n)
        improved = False
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if pressure[i] < 0.3:
                continue
            ai = a[i]
            if ai < norm * 1e-8:
                continue
            
            # Find best target: low pressure, high value
            # Score candidates by (pressure[j] - value[j]/norm)
            # We want low pressure and the target should be able to "absorb" mass
            candidates = []
            for _ in range(30):
                c = int(rng.integers(n))
                if c != i:
                    candidates.append(c)
            
            best_delta = 0.0
            best_val = cur_max
            best_j = -1
            
            for j in candidates[:20]:
                if j == i or pressure[j] > 0.5:
                    continue
                aj = a[j]
                
                h = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    h[lo:hi] += 2.0 * a[0:hi - lo]
                lo2 = max(0, i)
                hi2 = min(size, i + n)
                if hi2 > lo2:
                    h[lo2:hi2] -= 2.0 * a[0:hi2 - lo2]
                
                q = np.zeros(size)
                if 2 * j < size:
                    q[2 * j] += 1.0
                if 2 * i < size:
                    q[2 * i] -= 1.0
                if i + j < size:
                    q[i + j] -= 2.0
                
                n_samples = 15
                deltas = np.linspace(0.0, ai * 0.9, n_samples)
                for d in deltas:
                    if d < 1e-12:
                        continue
                    new_conv = conv + d * h + d * d * q
                    v = new_conv.max()
                    if v < best_val:
                        best_val = v
                        best_delta = d
                        best_j = j
            
            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            # Add perturbation
            n_perturb = max(5, n // 30)
            idx_perturb = rng.choice(n, size=n_perturb, replace=False)
            a[idx_perturb] *= (1.0 + 0.15 * rng.standard_normal(n_perturb))
            a = np.maximum(a, 0.0)
            s_new = a.sum()
            if s_new > 1e-15:
                a = a * norm / s_new
    
    return best_a, best_score


def structured_init(n, rng):
    """Generate a structured initial sequence."""
    x = np.linspace(0.0, 1.0, n)
    choice = rng.integers(0, 12)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.6
    elif choice == 6:
        a = np.exp(-4.0 * x) + 0.4 + 0.2 * np.sin(3 * np.pi * x)
    elif choice == 7:

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length, preserving the sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def eval_fn(a):
    """Evaluate a sequence using the provided evaluate_sequence function."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        if len(a_list) == 0:
            return float('inf')
        for val in a_list:
            if val < 0 or not np.isfinite(val):
                return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')
