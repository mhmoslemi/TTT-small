def build_constraint_matrix(f, active_idx):
    """Build constraint matrix T where T[i,j] = f[active_idx[i] - j] if valid."""
    n = len(f)
    m = len(active_idx)
    T = np.zeros((m, n))
    for i, k in enumerate(active_idx):
        # (f*g)(k) = sum_j f_{k-j} * g_j where valid indices
        # For a fixed k: g_j appears with coefficient f[k-j]
        # j ranges from max(0, k-n+1) to min(n-1, k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            fi = k - j
            if 0 <= fi < n:
                T[i, j] = f[fi]
    return T

def lp_min_max(f, solver, active_idx, norm_val):
    """Solve min t s.t. (f*b)(k) <= t for k in active_idx, sum(b) = norm_val, b >= 0"""
    n = len(f)
    b = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    T = build_constraint_matrix(f, active_idx)
    cons = [T @ b <= t, cp.sum(b) == norm_val]
    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    t_val = float(t.value) if t.value is not None else float('inf')
    if val.sum() < 1e-15:
        return None, None
    return val, t_val

def get_active_set(conv, M, tol):

        active = np.where(conv >= threshold)[0]
    if len(active) > len(conv) * 0.8:
        threshold = M * 0.95
        active = np.where(conv >= threshold)[0]

def subgradient_step(a, conv, active_idx, step_size):
    """Take a subgradient step on the max of autocorrelation."""
    n = len(a)
    # Subgradient of max_k (a*a)(k) at a is:

    grad = np.zeros(n)
    for k in active_idx:

            ki = k - i
            if 0 <= ki < n:
                grad[i] += 2.0 * a[ki]

    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * a.sum() / s

def subgradient_polish(f0, time_limit, verbose=True, seed=0):
    """Subgradient-based polishing."""
    rng = np.random.default_rng(seed)
    n = len(f0)
    norm = np.sqrt(2.0 * n)
    f = np.maximum(np.asarray(f0, dtype=np.float64), 0.0)
    f = f * norm / f.sum()
    
    best_f = f.copy()
    conv = fast_autocorr(f)
    best_score = conv.max()
    t0 = time.monotonic()
    
    # Average subgradient over a window
    window = 50
    grad_avg = np.zeros(n)
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        active = get_active_set(conv, best_score, 0.03)
        grad = np.zeros(n)
        for k in active:
            for i in range(n):
                ki = k - i
                if 0 <= ki < n:
                    grad[i] += 2.0 * f[ki]
        grad /= len(active)
        
        grad_avg += grad
        if t >= window:
            grad_avg -= 0  # We'll use exponential averaging instead
            grad_avg = (1.0 - 0.01) * grad_avg + 0.01 * grad
        
        # Step size: use diminishing step
        step = 0.1 / np.sqrt(t + 1)
        
        f_new = f - step * grad_avg
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-15:
            continue
        f_new = f_new * norm / s
        
        conv_new = fast_autocorr(f_new)
        score_new = conv_new.max()
        
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_f = f_new.copy()
            f = f_new
            conv = conv_new
        else:
            # Revert and reduce step
            f = f  # keep old f
            # Keep conv from old f
    
    return best_f, best_score

def coord_descent(f0, time_limit, rng, verbose=True):
    """Coordinate descent with golden section search."""
    n = len(f0)
    norm = np.sqrt(2.0 * n)
    f = np.maximum(np.asarray(f0, dtype=np.float64), 0.0)
    f = f * norm / f.sum()
    
    conv = fast_autocorr(f)
    best_score = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    
    for iteration in range(500):
        if time.monotonic() - t0 > time_limit:
            break
        
        pass_improved = False
        perm = rng.permutation(n)
        
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            
            fi = f[i]
            if fi < 1e-10 * norm:
                continue
            
            # Golden section search
            lo = 0.0
            hi = fi * 2.5 + norm * 0.05
            
            c1 = lo + (1.0 - 1.0/phi) * (hi - lo)
            c2 = lo + (1.0/phi) * (hi - lo)
            
            f1 = f.copy()
            f1[i] = c1
            f1 = np.maximum(f1, 0)
            f1 = f1 * norm / f1.sum()
            val1 = fast_autocorr(f1).max()
            
            f2 = f.copy()
            f2[i] = c2
            f2 = np.maximum(f2, 0)
            f2 = f2 * norm / f2.sum()
            val2 = fast_autocorr(f2).max()
            
            for _ in range(20):
                if val1 < val2:
                    hi = c2
                    c2 = c1
                    val2 = val1
                    c1 = lo + (1.0 - 1.0/phi) * (hi - lo)
                    f1 = f.copy()
                    f1[i] = c1
                    f1 = np.maximum(f1, 0)
                    f1 = f1 * norm / f1.sum()
                    val1 = fast_autocorr(f1).max()
                else:
                    lo = c1
                    c1 = c2
                    val1 = val2
                    c2 = lo + (1.0/phi) * (hi - lo)
                    f2 = f.copy()
                    f2[i] = c2
                    f2 = np.maximum(f2, 0)
                    f2 = f2 * norm / f2.sum()
                    val2 = fast_autocorr(f2).max()
            
            opt_c = c1 if val1 < val2 else c2
            opt_val = min(val1, val2)
            
            if opt_val < best_score - 1e-12:
                f[i] = opt_c
                f = np.maximum(f, 0)
                f = f * norm / f.sum()
                conv = fast_autocorr(f)
                best_score = conv.max()
                best_f = f.copy()
                pass_improved = True
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        
        if verbose and iteration % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD iter={iteration} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_f, best_score

def structured_init(n, style=0):
    """Generate structured initial sequences."""
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif style == 1:
        a = np.sin(np.pi * x) ** 1.2
    elif style == 2:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(8 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif style == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif style == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif style == 5:
        a = 0.4 + 0.6 * np.sin(np.pi * x) ** 0.8
    elif style == 6:
        a = 1.0 + 0.25 * np.cos(6 * np.pi * x) + 0.1 * np.cos(12 * np.pi * x)
    elif style == 7:
        a = np.sin(np.pi * x) ** 0.6 * (1.0 + 0.15 * np.cos(10 * np.pi * x))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss

def perturb_seq(f, rng, scale=0.03):
    """Perturb a sequence with multiplicative noise."""
    n = len(f)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.05)
    fp = f * noise
    s = fp.sum()
    if s < 1e-15:
        return f.copy()
    return fp / s

        fis = k - js
        T[i, js] = f[fis]

        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver_cache = s
                print(f"  Using LP solver: {s}")
                return s
        except Exception:
            continue
    _solver_cache = cp.SCS
    print("  Fallback LP solver: SCS")
    return _solver_cache


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_autocorr(a)
    return float(2.0 * len(a) * conv.max() / (s * s))


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.maximum(np.asarray(a, dtype=np.float64), 0.0)]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_constraint_matrix(f, active_idx):
    """Build T where T[i,j] = f[active_idx[i] - j] if valid, else 0."""
    n = len(f)
    m = len(active_idx)
    T = np.zeros((m, n))
    for i in range(m):
        k = active_idx[i]
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            js = np.arange(lo, hi)
            fis = k - js
            T[i, js] = f[fis]
    return T


def get_active_set(conv, M, tol):
    """Get indices where conv is within tol fraction of M."""
    if M <= 0:
        return np.arange(len(conv))
    threshold = M * (1.0 - tol)
    active = np.where(conv >= threshold)[0]
    if len(active) < 6:
        threshold = M * 0.5
        active = np.where(conv >= threshold)[0]
    if len(active) > int(len(conv) * 0.85):
        threshold = M * 0.95
        active = np.where(conv >= threshold)[0]
    return active


def lp_min_max(f, solver, active_idx, norm_val):
    """Solve min t s.t. (f*b)(k) <= t for k in active_idx, sum(b) = norm_val, b >= 0."""
    n = len(f)
    b = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    T = build_constraint_matrix(f, active_idx)
    cons = [T @ b <= t, cp.sum(b) == norm_val]
    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    t_val = float(t.value) if t.value is not None else float('inf')
    if val.sum() < 1e-15:
        return None, None
    return val, t_val


def lp_max_sum(f, solver, active_idx, M):
    """Solve max sum(b) s.t. (f*b)(k) <= M for k in active_idx, b >= 0."""
    n = len(f)
    b = cp.Variable(n, nonneg=True)
    T = build_constraint_matrix(f, active_idx)
    cons = [T @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if val.sum() < 1e-15:
        return None
    return val


def nesterov_lp(f0, time_limit, solver, verbose=True, seed=0):
    """Nesterov-accelerated LP with min-max formulation and violation detection."""
    rng = np.random.default_rng(seed)
    n = len(f0)
    size = 2 * n - 1
    norm = np.sqrt(2.0 * n)
    f = np.maximum(np.asarray(f0, dtype=np.float64), 0.0)
    s0 = f.sum()
    if s0 < 1e-15:
        f = np.ones(n) * norm / n
    else:
        f = f * norm / s0

    y = f.copy()
    a = f.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()

    conv_y = fast_autocorr(y)
    M_curr = conv_y.max()
    best_score = M_curr
    best_f = f.copy()
    stall_count = 0
    last_active = None
    last_T = None
    last_f_hash = None

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive active set
        tol = 0.03 + 0.03 * np.exp(-t / 400.0)
        active_set = get_active_set(conv_y, M_curr, tol)

        # Solve min-max LP
        b, t_val = lp_min_max(y, solver, active_set, norm)

        # Fallback to max-sum LP if min-max fails
        if b is None:
            b = lp_max_sum(y, solver, active_set, M_curr)
            if b is None:
                # Try with all constraints
                b = lp_min_max(y, solver, np.arange(size), norm)
                if b is None:
                    b = np.ones(n) * norm / n

        b_sum = b.sum()
        g = b * norm / b_sum

        # Violation detection: check if g violates inactive constraints
        if len(active_set) < size:
            conv_g = fast_autocorr(g)
            viol_mask = conv_g > M_curr * (1.0 + 1e-8)
            if viol_mask.any():
                viol_idx = np.where(viol_mask)[0]
                # Add violated constraints and re-solve (limit to 2 extra iterations)
                for _ in range(2):
                    active_set = np.union1d(active_set, viol_idx)
                    b2, t2 = lp_min_max(y, solver, active_set, norm)
                    if b2 is not None:
                        b = b2
                        b_sum = b.sum()
                        g = b * norm / b_sum
                        conv_g = fast_autocorr(g)
                        viol_mask = conv_g > M_curr * (1.0 + 1e-8)
                        if not viol_mask.any():
                            break
                        viol_idx = np.where(viol_mask)[0]

        # Nesterov update
        alpha = max(2.0 / (t + 2.0), 0.08)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.88)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y
        y = y_new
        a = a_new

        # Track best
        conv_a = fast_autocorr(a)
        score_a = conv_a.max()
        if score_a < best_score - 1e-11:
            best_score = score_a
            best_f = a.copy()
            stall_count = 0
        else:
            conv_y = fast_autocorr(y)
            M_curr = conv_y.max()
            if M_curr < best_score - 1e-11:
                best_score = M_curr
                best_f = y.copy()
                stall_count = 0
            else:
                stall_count += 1

        # Ergodic average
        avg_a += a
        n_avg += 1
        if n_avg % 30 == 0:
            avg_f = avg_a / n_avg
            s_avg = avg_f.sum()
            if s_avg > 1e-15:
                avg_f = avg_f * norm / s_avg
                sc = fast_autocorr(avg_f).max()
                if sc < best_score - 1e-11:
                    best_score = sc
                    best_f = avg_f.copy()
                    stall_count = 0

        # Restart if stalled
        if stall_count > 120:
            y = best_f.copy()
            a = best_f.copy()
            avg_a = np.zeros(n)
            n_avg = 0
            stall_count = 0
            conv_y = fast_autocorr(y)
            M_curr = conv_y.max()

        if verbose and t % 300 == 0:
            elapsed = time.monotonic() - t0
            cv_y = fast_autocorr(y)
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.10f}, M_y={cv_y.max():.10f}")

    return best_f, best_score


def subgradient_polish(f0, time_limit, verbose=True, seed=0):
    """Polish using subgradient of max autocorrelation with projection."""
    rng = np.random.default_rng(seed)
    n = len(f0)
    norm = np.sqrt(2.0 * n)
    f = np.maximum(np.asarray(f0, dtype=np.float64), 0.0)
    f = f * norm / f.sum()

    conv = fast_autocorr(f)
    best_score = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()

    # Exponential moving average of gradients
    grad_ema = np.zeros(n)
    beta = 0.95
    base_step = 0.05

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        active = get_active_set(conv, best_score, 0.02)
        # Compute subgradient: grad_i = (2/|A|) * sum_{k in A} a[k-i] for valid k-i
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            js = np.arange(lo, hi)
            fis = k - js
            grad[js] += 2.0 * f[fis]
        grad /= len(active)

        grad_ema = beta * grad_ema + (1.0 - beta) * grad

        # Adaptive step size
        step = base_step / (1.0 + 0.01 * t)

        f_new = f - step * grad_ema
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-15:
            continue
        f_new = f_new * norm / s

        conv_new = fast_autocorr(f_new)
        score_new = conv_new.max()

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_f = f_new.copy()
            f = f_new
            conv = conv_new
        else:
            # Try smaller step
            f_new2 = f - step * 0.1 * grad_ema
            f_new2 = np.maximum(f_new2, 0.0)
            s2 = f_new2.sum()
            if s2 > 1e-15:
                f_new2 = f_new2 * norm / s2
                sc2 = fast_autocorr(f_new2).max()
                if sc2 < best_score - 1e-13:
                    best_score = sc2
                    best_f = f_new2.copy()
                    f = f_new2
                    conv = fast_autocorr(f)

    return best_f, best_score


def coord_descent(f0, time_limit, rng, verbose=True):
    """Coordinate descent with golden section search per coordinate."""
    n = len(f0)
    norm = np.sqrt(2.0 * n)
    f = np.maximum(np.asarray(f0, dtype=np.float64), 0.0)
    f = f * norm / f.sum()

    conv = fast_autocorr(f)
    best_score = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    phi_inv = 0.6180339887498949

    for iteration in range(200):
        if time.monotonic() - t0 > time_limit:
            break

        pass_improved = False
        perm = rng.permutation(n)

        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break

            fi = f[i]
            if fi < 1e-10 * norm:
                continue

            lo = 0.0
            hi = fi * 2.5 + norm * 0.05

            c1 = lo + phi_inv * (hi - lo)
            c2 = lo + (1.0 - phi_inv) * (hi - lo)

            ft1 = f.copy()
            ft1[i] = c1
            ft1 = np.maximum(ft1, 0)
            ft1 = ft1 * norm / ft1.sum()
            v1 = fast_autocorr(ft1).max()

            ft2 = f.copy()
            ft2[i] = c2
            ft2 = np.maximum(ft2, 0)
            ft2 = ft2 * norm / ft2.sum()
            v2 = fast_autocorr(ft2).max()

            for _ in range(18):
                if v1 < v2:
                    hi = c2
                    c2 = c1
                    v2 = v1
                    c1 = lo + phi_inv * (hi - lo)
                    ft1 = f.copy()
                    ft1[i] = c1
                    ft1 = np.maximum(ft1, 0)
                    ft1 = ft1 * norm / ft1.sum()
                    v1 = fast_autocorr(ft1).max()
                else:
                    lo = c1
                    c1 = c2
                    v1 = v2
                    c2 = lo + (1.0 - phi_inv) * (hi - lo)
                    ft2 = f.copy()
                    ft2[i] = c2
                    ft2 = np.maximum(ft2, 0)
                    ft2 = ft2 * norm / ft2.sum()
                    v2 = fast_autocorr(ft2).max()

            opt_c = c1 if v1 < v2 else c2
            opt_v = min(v1, v2)

            if opt_v < best_score - 1e-12:
                f[i] = opt_c
                f = np.maximum(f, 0)
                f = f * norm / f.sum()
                conv = fast_autocorr(f)
                best_score = conv.max()
                best_f = f.copy()
                pass_improved = True

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break
        else:
            no_improve = 0

        if verbose and iteration % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD iter={iteration} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_f, best_score


def pair_cd(f0, time_limit, rng, verbose=True):
    """Pair coordinate descent: swap mass between pairs of coordinates."""
    n = len(f0)
    norm = np.sqrt(2.0 * n)
    f = np.maximum(np.asarray(f0, dtype=np.float64), 0.0)
    f = f * norm / f.sum()

    conv = fast_autocorr(f)
    best_score = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0

    for iteration in range(100):
        if time.monotonic() - t0 > time_limit:
            break

        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(0, n - 1, 2):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            j = perm[idx + 1]
            fi = f[i]
            fj = f[j]
            if fi < 1e-10 * norm and fj < 1e-10 * norm:
                continue

            best_delta = 0.0
            best_v = conv.max()

            # Try multiple fractions
            max_d = max(fi, fj)
            for frac in [0.1, 0.25, 0.4, 0.6, 0.75, 0.9]:
                d = max_d * frac
                # Direction: decrease i, increase j
                if d <= fi:
                    ft = f.copy()
                    ft[i] = fi - d
                    ft[j] = fj + d
                    ft = np.maximum(ft, 0)
                    s = ft.sum()
                    if s > 1e-15:
                        ft = ft * norm / s
                        v = fast_autocorr(ft).max()
                        if v < best_v - 1e-13:
                            best_v = v
                            best_delta = d
                # Direction: increase i, decrease j
                if d <= fj:
                    ft = f.copy()
                    ft[i] = fi + d
                    ft[j] = fj - d
                    ft = np.maximum(ft, 0)
                    s = ft.sum()
                    if s > 1e-15:
                        ft = ft * norm / s
                        v = fast_autocorr(ft).max()
                        if v < best_v - 1e-13:
                            best_v = v
                            best_delta = -d

            if abs(best_delta) > 1e-14:
                if best_delta > 0:
                    f[i] -= best_delta
                    f[j] += best_delta
                else:
                    f[i] += abs(best_delta)
                    f[j] -= abs(best_delta)
                f = np.maximum(f, 0)
                f = f * norm / f.sum()
                conv = fast_autocorr(f)
                if conv.max() < best_score - 1e-13:
                    best_score = conv.max()
                    best_f = f.copy()
                    pass_improved = True

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break
        else:
            no_improve = 0

        if verbose and iteration % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD iter={iteration} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_f, best_score


def structured_init(n, style=0):
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif style == 1:
        a = np.sin(np.pi * x) ** 1.2
    elif style == 2:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(8 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif style == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif style == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif style == 5:
        a = 0.4 + 0.6 * np.sin(np.pi * x) ** 0.8
    elif style == 6:
        a = 1.0 + 0.25 * np.cos(6 * np.pi * x) + 0.1 * np.cos(12 * np.pi * x)
    elif style == 7:
        a = np.sin(np.pi * x) ** 0.6 * (1.0 + 0.15 * np.cos(10 * np.pi * x))
    elif style == 8:
        a = 0.3 + 0.7 * np.cos(np.pi * x / 2.0) ** 1.5
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def perturb_seq(f, rng, scale=0.03):
    n = len(f)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.05)
    fp = f * noise
    s = fp.sum()
    if s < 1e-15:
        return f.copy()
    return fp / s

        a_out, _ = subgradient_polish(a_norm, t4, verbose=False, seed=55)
        try_update(a_out, f"sg_n{n_cur}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 5: Try longer length (10% time) ===
    n_long = n_base + 300
    print(f"\n=== Phase 5: LP n={n_long} ===")
    t5 = min(remaining * 0.10, 100.0)
    if t5 > 30:
        init_long = interpolate_seq(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = nesterov_lp(a_norm, t5, solver, verbose=True, seed=42)
        try_update(a_out, f"lp_long_n{n_long}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 6: Try shorter length (8% time) ===
    n_short = max(1200, n_base - 200)
    if n_short != n_base and remaining > 80:
        print(f"\n=== Phase 6: LP n={n_short} ===")
        t6 = min(remaining * 0.08, 80.0)
        if t6 > 20:
            init_short = interpolate_seq(best_seq, n_short)
            a_norm = init_short * np.sqrt(2.0 * n_short) / init_short.sum()
            a_out, _ = nesterov_lp(a_norm, t6, solver, verbose=False, seed=77)
            try_update(a_out, f"lp_short_n{n_short}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 7: Pair CD (7% time) ===
    n_best = len(best_seq)
    if remaining > 50:
        print(f"\n=== Phase 7: PCD n={n_best} ===")
        t7 = min(remaining * 0.07, 70.0)
        if t7 > 15:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = pair_cd(a_norm, t7, rng, verbose=False)
            try_update(a_out, f"pcd_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 8: Final LP on best (use remaining time) ===
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP n={n_best} ===")
    t8 = min(remaining * 0.55, remaining - 30)
    t8 = max(t8, 60.0)
    if t8 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t8, solver, verbose=True, seed=999)
        try_update(a_out, f"lp_final_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 9: Final CD polish ===
    n_best = len(best_seq)
    if remaining > 50:
        print(f"\n=== Phase 9: Final CD n={n_best} ===")
        t9 = min(remaining * 0.6, 100.0)
        if t9 > 20:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = coord_descent(a_norm, t9, rng, verbose=False)
            try_update(a_out, f"cd_final_n{n_best}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
