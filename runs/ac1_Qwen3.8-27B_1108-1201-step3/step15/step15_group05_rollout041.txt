def spectral_shape(a, gamma=0.7, n_iters=100):
    """
    Spectral shaping: compress the dynamic range of |FFT(a)| to flatten

    n = len(a)

        a_new = np.fft.irfft(A_new, fft_size)[:n]

        if score < best_score:

            best_a = a_new.copy()
        else:

def spectral_shaping(a0, time_limit, verbose=False):
    """Spectral shaping: compress |FFT(a)| to flatten autocorrelation."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = normalize_to(a0, S)
    
    best_a = a.copy()
    best_max = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    fft_size = 2 * n  # sufficient padding
    
    for gamma in [0.5, 0.6, 0.7, 0.8, 0.9, 0.3, 0.4]:
        if time.monotonic() - t0 > time_limit:
            break
        
        a_cur = a.copy()
        for it in range(200):
            if time.monotonic() - t0 > time_limit:
                break
            
            A = np.fft.rfft(a_cur, fft_size)
            mag = np.abs(A)
            phase = np.angle(A)
            
            # Compress
            mag_new = mag ** gamma
            A_new = mag_new * np.exp(1j * phase)
            a_new = np.fft.irfft(A_new, fft_size)[:n]
            
            # Project
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new * S / s_new
            
            M = fast_autocorr(a_new).max()
            if M < best_max:
                best_max = M
                best_a = a_new.copy()
                a_cur = a_new.copy()
            else:
                # Blend
                a_blend = 0.7 * a_cur + 0.3 * a_new
                a_blend = np.maximum(a_blend, 0.0)
                a_blend = normalize_to(a_blend, S)
                M_blend = fast_autocorr(a_blend).max()
                if M_blend < best_max:
                    best_max = M_blend
                    best_a = a_blend.copy()
                    a_cur = a_blend.copy()
                else:
                    break
        
        if verbose:
            print(f"    Spectral gamma={gamma}: best_max={best_max:.10f}")
    
    return best_a, best_max

def parametric_search(a0, time_limit, verbose=False, seed=0):
    """Search over parametric families: a[j] = max(0, sum of cosine terms)."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    S = np.sqrt(2.0 * n)
    t0 = time.monotonic()
    
    best_a = a0.copy()
    best_score = compute_score(a0)
    
    # Try different parametric forms
    # Form 1: a[j] = max(0, 1 + A*cos(2*pi*f1*j/n + p1) + B*cos(2*pi*f2*j/n + p2))
    # Random search over parameters

        if time.monotonic() - t0 > time_limit:
            break
        
        # Random parameters
        n_terms = rng.integers(2, 6)
        a_param = np.ones(n)
        for m in range(n_terms):
            freq = rng.integers(1, n // 2 + 1)
            amp = rng.uniform(0.1, 0.8)
            phase = rng.uniform(0, 2 * np.pi)
            j = np.arange(n)
            a_param += amp * np.cos(2 * np.pi * freq * j / n + phase)
        
        a_param = np.maximum(a_param, 0.0)
        s = a_param.sum()
        if s < 0.01:
            continue
        a_param = a_param * S / s
        
        sc = compute_score(a_param)
        if sc < best_score:
            best_score = sc
            best_a = a_param.copy()
            if verbose:
                print(f"    Parametric trial {trial}: score={sc:.8f}")
    
    # Also try interpolating between parametric and current best
    if time.monotonic() - t0 < time_limit * 0.5:
        for alpha in np.linspace(0.1, 0.9, 9):
            a_blend = (1 - alpha) * a0 + alpha * best_a
            a_blend = np.maximum(a_blend, 0.0)
            a_blend = normalize_to(a_blend, S)
            sc = compute_score(a_blend)
            if sc < best_score:
                best_score = sc
                best_a = a_blend.copy()
    
    return best_a, best_score

import numpy as np

def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate the score using the provided evaluate_sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def normalize_to(a, S):
    """Clamp to non-negative and normalize sum to S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * S / s


def compute_score(a):
    """Compute the score 2*n*max(conv)/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_autocorr(a)
    return 2.0 * n * conv.max() / (s * s)


def project_simplex_s(x, S):
    """Project x onto {a >= 0, sum(a) = S} using the standard algorithm."""
    n = len(x)
    if n == 0:
        return x
    x_sorted = np.sort(x)[::-1]
    cumsum = np.cumsum(x_sorted)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = x_sorted - (cumsum - S) / rho > 0
    if not np.any(cond):
        return np.ones(n) * S / n
    k = int(np.max(np.where(cond)[0])) + 1
    tau = (cumsum[k - 1] - S) / k
    return np.maximum(x - tau, 0.0)


def compute_multi_gradient(a, peaks, weights):
    """Compute weighted sum of gradients at multiple peaks."""
    n = len(a)
    grad = np.zeros(n)
    for w, k in zip(weights, peaks):

        hi = min(n - 1, k)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            grad[idx] += w * 2.0 * a[k - idx]
    return grad

def spectral_shaping(a0, time_limit, verbose=False):
    """Spectral shaping: compress |FFT(a)| magnitude to flatten autocorrelation."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = normalize_to(a0, S)

    best_a = a.copy()
    best_max = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 2 * n

    # Try multiple compression levels
    for gamma in [0.85, 0.75, 0.65, 0.9, 0.8, 0.55, 0.95, 0.7]:
        if time.monotonic() - t0 > time_limit:
            break

        a_cur = best_a.copy()
        no_improve = 0
        for it in range(300):
            if time.monotonic() - t0 > time_limit or no_improve > 50:
                break

            A = np.fft.rfft(a_cur, fft_size)
            mag = np.abs(A)
            phase = np.angle(A)

            # Compress magnitude
            mag_new = mag ** gamma
            A_new = mag_new * np.exp(1j * phase)
            a_new = np.fft.irfft(A_new, fft_size)[:n]

            # Project to non-negative and normalize
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new * S / s_new

            M = fast_autocorr(a_new).max()
            if M < best_max:
                best_max = M
                best_a = a_new.copy()
                a_cur = a_new.copy()
                no_improve = 0
            else:
                # Try blending
                blend = 0.5
                a_blend = (1 - blend) * a_cur + blend * a_new
                a_blend = np.maximum(a_blend, 0.0)
                sb = a_blend.sum()
                if sb > 1e-15:
                    a_blend = a_blend * S / sb
                    Mb = fast_autocorr(a_blend).max()
                    if Mb < best_max:
                        best_max = Mb
                        best_a = a_blend.copy()
                        a_cur = a_blend.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                else:
                    no_improve += 1

        if verbose:
            print(f"    Spectral gamma={gamma:.2f}: best_max={best_max:.10f}")

    return best_a, best_max

def random_coordinate_polish(a0, time_limit, verbose=False, seed=0):
    """Random coordinate descent with multi-step trial."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = normalize_to(a0, S)

    best_a = a.copy()
    best_max = fast_autocorr(a).max()
    t0 = time.monotonic()

    no_improve = 0
    max_no_improve = 2000

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        # Pick random coordinates
        n_perturb = max(10, n // 100)
        indices = rng.choice(n, size=n_perturb, replace=False)

        improved_this_round = False

            if time.monotonic() - t0 > time_limit:
                break

            a_old = a[idx]

                    M = fast_autocorr(a).max()

                        best_max = M
                        best_a = a.copy()

                        break

                        a[idx] = a_old

                    a[idx] = a_old

                    best_a = a.copy()
                    improved_this_round = True

                else:
                    a[idx] = a_old
                    a = a * S / s

        if improved_this_round:
            no_improve = 0
        else:
            no_improve += 1

        if verbose and no_improve % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordPoly ({elapsed:.0f}s): best_max={best_max:.10f}, no_improve={no_improve}")

    return best_a, best_max


def targeted_peak_reduction(a0, time_limit, verbose=False, seed=0):
    """
    Directly target the peak: identify coordinates that contribute most
    to the peak convolution value and reduce them selectively.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = normalize_to(a0, S)

    best_a = a.copy()
    best_max = fast_autocorr(a).max()
    t0 = time.monotonic()

    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(a)
        M = conv.max()
        k_star = int(np.argmax(conv))

        # Find coordinates that contribute most to C(k_star)
        # C(k_star) = sum_j a[j]*a[k_star-j]
        # The "contribution" of a[i] is 2*a[k_star-i] (the gradient)
        contrib = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        idx = np.arange(lo, hi + 1)
        contrib[idx] = 2.0 * a[k_star - idx]

        # Find top contributors
        top_contributors = np.argsort(contrib)[-max(5, n // 50):]

        # Try reducing each top contributor
        improved = False
        for ci in top_contributors:
            if time.monotonic() - t0 > time_limit:
                break
            a_old = a[ci]
            if a_old < 1e-10:
                continue
            for frac in [0.9, 0.8, 0.7, 0.5, 0.3]:
                a[ci] = a_old * frac
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s < 1e-15:
                    a[ci] = a_old
                    continue
                a = a * S / s
                M_new = fast_autocorr(a).max()
                if M_new < best_max:
                    best_max = M_new
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[ci] = a_old
                    a = a * S / s
            if improved:
                break

        if not improved:
            # Try increasing a low contributor to redistribute
            low_contributors = np.argsort(contrib)[:max(5, n // 50)]
            ci = low_contributors[rng.integers(len(low_contributors))]
            a_old = a[ci]
            for frac in [1.2, 1.5, 2.0, 3.0]:
                a[ci] = a_old * frac
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s < 1e-15:
                    a[ci] = a_old
                    continue
                a = a * S / s
                M_new = fast_autocorr(a).max()
                if M_new < best_max:
                    best_max = M_new
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[ci] = a_old
                    a = a * S / s

        if not improved and it % 1000 == 500:
            # Perturb to escape
            noise = rng.standard_normal(n) * 0.01
            a_pert = a + noise * a
            a = normalize_to(a_pert, S)

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakRed it={it} ({elapsed:.0f}s): best_max={best_max:.10f}")

    return best_a, best_max


def symmetrize(a):
    """Symmetrize: a = (a + reverse(a)) / 2, then normalize."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1].copy())
    s = a_sym.sum()
    if s < 1e-15:
        return a
    return a_sym / s


def interpolate_sequence(a, new_n):
    """Interpolate sequence to new length, preserving normalized shape."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n)
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return a_interp / ss


def propose_candidate():
    """
    Main search function. Uses Nesterov subgradient, spectral shaping,
    targeted peak reduction, and coordinate polish to minimize the score.
    """
    rng = np.random.default_rng(12345)
    t_global = time.monotonic()
    time_budget = 1750.0

    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x)
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    n_base = len(best_seq)
    print(f"Starting: n={n_base}, score={best_score:.8f}")

    # Quick symmetrization check
    a_sym = symmetrize(best_seq)
    sc_sym = evaluate_safe(a_sym)
    if np.isfinite(sc_sym) and sc_sym < best_score:
        best_score = sc_sym
        best_seq = a_sym.copy()
        print(f"  Symmetrization improved: {sc_sym:.8f}")

    # === Phase 1: Spectral shaping (new, fast) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 120.0)
    if t1 > 20:
        print(f"\n=== Phase 1: Spectral shaping, budget={t1:.0f}s ===")
        a_out, _ = spectral_shaping(best_seq, t1, verbose=True)
        try_update(a_out, "spectral1")

        if time.monotonic() - t_global < time_budget - 60:
            a_out2, _ = spectral_shaping_v2(best_seq, min(60.0, time_budget - (time.monotonic() - t_global) - 30), verbose=False)
            try_update(a_out2, "spectral2")

    # === Phase 2: Nesterov subgradient (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.30, 450.0)
    if t2 > 30:
        print(f"\n=== Phase 2: Nesterov subgrad, budget={t2:.0f}s ===")
        a_out, _ = nesterov_subgrad(best_seq, t2, verbose=True, seed=0)
        try_update(a_out, "nesterov1")

    # === Phase 3: Targeted peak reduction ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 250.0)
    if t3 > 30:
        print(f"\n=== Phase 3: Targeted peak reduction, budget={t3:.0f}s ===")
        a_out, _ = targeted_peak_reduction(best_seq, t3, verbose=True, seed=1)
        try_update(a_out, "peakred1")

    # === Phase 4: Nesterov subgradient round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 30:
        print(f"\n=== Phase 4: Nesterov subgrad round 2, budget={t4:.0f}s ===")
        a_out, _ = nesterov_subgrad(best_seq, t4, verbose=True, seed=42)
        try_update(a_out, "nesterov2")

    # === Phase 5: Random coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 180.0)
    if t5 > 20:
        print(f"\n=== Phase 5: Coordinate polish, budget={t5:.0f}s ===")
        a_out, _ = random_coordinate_polish(best_seq, t5, verbose=True, seed=7)
        try_update(a_out, "coordpoly1")

    # === Phase 6: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        print(f"\n=== Phase 6: Length exploration, remaining={remaining:.0f}s ===")
        n_cur = len(best_seq)
        # Try a few nearby lengths
        for new_n in [n_cur - 100, n_cur + 100, n_cur + 200]:
            if time.monotonic() - t_global > time_budget - 60:
                break
            if new_n < 50:
                continue
            a_interp = interpolate_sequence(best_seq, new_n)
            a_norm = normalize_to(a_interp, np.sqrt(2.0 * new_n))
            sc = evaluate_safe(a_norm)
            print(f"    n={new_n}: score={sc:.8f}")
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_norm.copy()
                print(f"  *** Length change improved: {sc:.8f} at n={new_n}")
                # Quick polish at new length
                t_short = min(60.0, time_budget - (time.monotonic() - t_global) - 30)
                if t_short > 20:
                    a_out, _ = nesterov_subgrad(best_seq, t_short, verbose=False, seed=99)
                    try_update(a_out, f"nesterov_n{new_n}")

    # === Phase 7: Final spectral shaping ===
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.15, 100.0)
    if t7 > 15:
        print(f"\n=== Phase 7: Final spectral shaping, budget={t7:.0f}s ===")
        a_out, _ = spectral_shaping(best_seq, t7, verbose=False)
        try_update(a_out, "spectral_final")

    # === Phase 8: Final Nesterov ===
    remaining = time_budget - (time.monotonic() - t_global)
    t8 = min(remaining * 0.30, 150.0)
    if t8 > 15:
        print(f"\n=== Phase 8: Final Nesterov, budget={t8:.0f}s ===")
        a_out, _ = nesterov_subgrad(best_seq, t8, verbose=True, seed=2024)
        try_update(a_out, "nesterov_final")

    # === Phase 9: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t9 = min(remaining * 0.40, 100.0)
    if t9 > 10:
        print(f"\n=== Phase 9: Final coord polish, budget={t9:.0f}s ===")
        a_out, _ = random_coordinate_polish(best_seq, t9, verbose=False, seed=314)
        try_update(a_out, "coordpoly_final")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
