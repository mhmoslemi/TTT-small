1. Direct LP best-response (min M s.t. T@b<=M, sum(b)=c, b>=0) with Nesterov acceleration
2. Multi-dimensional search (n=1600 from parent, n=2000 from scratch)
3. FFT-based coordinate descent polish
4. Adaptive restarts and diverse initializations
Key insight: direct LP formulation gives tighter best-response per iteration,

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_T_sparse(a):
    n = len(a)
    size = 2 * n - 1
    cols = np.repeat(np.arange(n), n)
    rows = np.arange(size) * 0
    all_rows = np.concatenate([np.arange(j, min(j + n, size)) for j in range(n)])
    all_vals = np.concatenate([a[:min(n, size - j)] for j in range(n)])
    all_cols = np.repeat(np.arange(n), [min(n, size - j) for j in range(n)])
    T = csr_matrix((all_vals, (all_rows, all_cols)), shape=(size, n))
    return T


def lp_best_response(a, solver):
    n = len(a)
    norm_val = np.sqrt(2.0 * n)
    size = 2 * n - 1
    T = build_T_sparse(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == norm_val]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None, None
    b_val = np.maximum(b_val, 0.0)
    m_val = float(M_var.value)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val * norm_val / s
    if np.isinf(m_val) or np.isnan(m_val):
        m_val = fast_autocorr(b_val).max()
    return b_val, m_val

        g = g * norm_val / s_g

        y_new = np.maximum(y_new, 0.0)

        if s_y < 1e-15:

        a = a_new

def cd_polish(a0, time_limit, verbose=True):
    """Coordinate descent polish using FFT-based convolution updates."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm_val / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    rng = np.random.default_rng(12345)
    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        if n_passes % 200 == 0 and no_improve > 100:
            break

        conv = fast_autocorr(a)
        cur_max = conv.max()
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_a = a.copy()
            no_improve = 0

        # Find active lags
        top_k = min(20, size)
        active_idx = np.argsort(conv)[-top_k:]

        # Try coordinate moves
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm_val * 1e-10:
                continue

            # Try moving mass to nearby positions
            for off in [1, -1, 2, -2, 3, -3, 5, -5]:
                j = (i + off) % n
                if j == i:
                    continue

                # Compute delta in convolution when we move mass from i to j
                # conv_new = conv + 2*d*(a shifted to j - a shifted to i)
                # where d is the amount moved
                delta_j = np.zeros(size)
                delta_i = np.zeros(size)

                # Contribution of a[j] to convolution positions
                lo_j = max(0, j - n + 1)
                hi_j = min(size - 1, j + n - 1)
                for pos in range(lo_j, hi_j + 1):
                    src = pos - j
                    if 0 <= src < n:
                        delta_j[pos] += a[src]

                lo_i = max(0, i - n + 1)
                hi_i = min(size - 1, i + n - 1)
                for pos in range(lo_i, hi_i + 1):
                    src = pos - i
                    if 0 <= src < n:
                        delta_i[pos] += a[src]

                delta = 2.0 * (delta_j - delta_i)

                # Also need diagonal corrections for positions 2i and 2j
                # At pos 2i: old contribution is a[i]^2, new is (a[i]-d)^2
                # At pos 2j: old is a[j]^2, new is (a[j]+d)^2
                pos_2i = 2 * i
                pos_2j = 2 * j

                # Try several fractions
                best_d = 0.0
                best_new_max = cur_max
                for frac in [0.2, 0.4, 0.6, 0.8]:
                    d = a[i] * frac
                    new_conv = conv + d * delta
                    if 0 <= pos_2i < size:
                        new_conv[pos_2i] = conv[pos_2i] - 2 * a[i] * d + d * d
                    if 0 <= pos_2j < size:
                        new_conv[pos_2j] = conv[pos_2j] + 2 * a[j] * d + d * d
                    new_max = new_conv.max()
                    if new_max < best_new_max - 1e-13:
                        best_new_max = new_max
                        best_d = d

                if best_d > 0:
                    a[i] -= best_d
                    a[j] += best_d
                    s = a.sum()
                    if s < 1e-15:
                        a[i] += best_d
                        a[j] -= best_d
                        continue
                    a = a * norm_val / s
                    conv = fast_autocorr(a)
                    if conv.max() < best_score - 1e-12:
                        best_score = conv.max()
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 50:
                # Perturb
                noise = rng.normal(0, norm_val * 0.003, n)
                a_pert = a + noise
                a_pert = np.maximum(a_pert, 0)
                s_pert = a_pert.sum()
                if s_pert > 1e-15:
                    a = a_pert * norm_val / s_pert
                no_improve = 0

        if verbose and n_passes % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def fw_optimize(a0, time_limit, solver, verbose=True):
    """Frank-Wolfe without Nesterov (more stable for non-convex problems)."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:

    else:
        a = a * norm_val / s0

    best_a = a.copy()

    t0 = time.monotonic()
    n_iters = 0
    patience = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        b, m_lp = lp_best_response(a, solver)
        if b is None:
            if n_iters % 100 == 0:
                print(f"    FW: LP failed at iter {n_iters}")

            if n_iters > 200 and n_iters % 50 == 0:
                break
            continue
        n_iters += 1

        s_g = b.sum()
        if s_g < 1e-15:
            continue
        g = b * norm_val / s_g

        gamma = 2.0 / (t + 2.0)
        a_new = (1.0 - gamma) * a + gamma * g
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:

        else:
            a_new = a_new * norm_val / s_a

        a = a_new
        score = fast_autocorr(a).max()

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0

        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
