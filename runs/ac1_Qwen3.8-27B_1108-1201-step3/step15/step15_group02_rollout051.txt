def build_T_sparse(a, active_rows):
    """Build the constraint matrix T for given active rows."""
    n = len(a)
    T = np.zeros((len(active_rows), n))
    for idx, k in enumerate(active_rows):

        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):

    return T

            v = beta2 * v + (1.0 - beta2) * grad * grad

            v_hat = v / (1.0 - beta2 ** t_adam)
            
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            
            a = a_new
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score

def spectral_shape(a0, time_limit, verbose=False):
    """Try to reduce peak by shaping in frequency domain."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)

    # In frequency domain, we want to flatten |A|^2
    # But we must maintain a >= 0
    # Approach: iteratively reduce the high-frequency components that contribute to the peak
    
    for it in range(200):

            break
        
        A = rfft(a, fft_size)

        # The peak of IFFT(|A|^2) is what we want to reduce
        # Gradient of peak w.r.t. |A|^2 is concentrated at the peak location

        M = conv.max()
        peak_idx = int(np.argmax(conv))

        # To reduce the peak, we want to reduce |A[k]|^2 where the phase aligns with the peak

        A_mag = np.abs(A)

        w = np.zeros(size)

        A_ft = rfft(a, fft_size)
        
        ca = irfft(np.conj(A_ft) * W, fft_size)[:n]
        aw = irfft(A_ft * W, fft_size)[:n]
        grad = 2.0 * (ca + aw)

        lr = 0.1 / np.sqrt(n) * (1.0 / (1.0 + it * 0.1))

                a = a_new
            else:
                # Halve learning rate
                lr *= 0.5
        
        if verbose and it % 50 == 0:
            print(f"    Spectral it={it}: best={best_score:.8f}")

def structured_init(n, rng):

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
