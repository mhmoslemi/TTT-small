2. **Projected subgradient with Nesterov acceleration** (alternative direction)
3. **Coordinate descent polishing** (local refinement)
4. **Multi-length exploration** (find optimal length)
```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. No-regret multiplicative weights game approach (avoids LP, allows more iterations)
2. Projected subgradient descent with Nesterov acceleration
3. Adaptive active-set subgradient computation
4. Coordinate descent with golden section search polishing
5. Multi-length exploration with interpolation
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Compute full autocorrelation (a * a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Quick local evaluation of score (scale-invariant)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_autocorr(a)
    return float(2.0 * len(a) * conv.max() / (s * s))


def evaluate_safe(a):
    """Safe evaluation using the actual evaluate_sequence function."""
    try:
        a_list = [float(x) for x in np.maximum(np.asarray(a, dtype=np.float64), 0.0)]
        a_list = [min(1000.0, x) for x in a_list]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def simplex_project(v, S):
    """Project v onto {x : sum(x)=S, x>=0}."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.full(len(v), S / len(v))
    return v * S / s


def compute_subgradient(f, q, n):
    """Compute g_i = 2 * sum_k q[k] * f[k-i] for i=0..n-1."""
    g = np.zeros(n)
    size = 2 * n - 1
    # Use FFT approach for speed when q has many significant values
    q_sum = q.sum()
    if q_sum < 1e-15:
        return g
    q_norm = q / q_sum
    # Direct computation for sparse q
    threshold = q_norm.max() * 1e-5
    active = np.where(q_norm > threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(q_norm)])
    for k in active:
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            idx = k - np.arange(i_lo, i_hi + 1)
            g[i_lo:i_hi + 1] += q_norm[k] * f[idx]
    return 2.0 * g


def compute_subgradient_fft(f, q, n):
    """Compute subgradient using FFT (faster for dense q)."""
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    # g_i = 2 * sum_k q[k] * f[k-i]
    # This is 2 * (q cross_correlate f)[i] = 2 * ifft(FFT(q) * conj(FFT(f))) shifted
    # Actually: g_i = sum_k q[k] * f[k-i]
    # = sum_k q[k] * f[k-i]
    # If we define the "cross-correlation" R[i] = sum_k q[k] * f[k-i],
    # then R = ifft(FFT(q) * conj(FFT(f))) for the circular case
    # For linear: pad both to fft_size
    Q = rfft(q.astype(np.float64), fft_size)
    F = rfft(f.astype(np.float64), fft_size)
    # R[i] = sum_k q[k] f[k-i] = (q * f_reversed)[i + n - 1] ...
    # Let me just use the direct formula with FFT:
    # g_i = sum_k q[k] * f[k-i]
    # = sum_k q[k] * f[k-i]
    # Define: h[m] = f[m] for 0<=m<n, else 0
    # g_i = sum_k q[k] * h[k-i] = (q * h)[i] where * is correlation
    # Using FFT: correlation = ifft(FFT(q) * conj(FFT(h)))
    # But we need the "linear" version
    R = irfft(Q * np.conj(F), fft_size)
    # R[m] = sum_k q[k] * h[k+m] (circular)
    # We want g_i = sum_k q[k] * h[k-i]
    # = R[-i] = R[fft_size - i] (circular)
    # For linear: g_i = sum_k q[k] * f[k-i] for valid k-i in [0,n-1]
    # = R_circular[i] if we set up the problem correctly
    # Actually it's easier to just do:
    # g_i = sum_k q[k] * f[k-i]
    # = sum_k q_flipped[n-1-k] * f[k-i+n-1] ... 
    
    # Let me just use the simple approach: flip q, convolve with f
    q_flip = q[::-1]
    # (q_flip * f)[m] = sum_j q_flip[j] * f[m-j] = sum_j q[n-1-j] * f[m-j]
    # g_i = sum_k q[k] * f[k-i]
    # Let k' = n-1-k: g_i = sum_k' q[n-1-k'] * f[n-1-k'-i]
    # = sum_k' q_flip[k'] * f[n-1-i-k']
    # = (q_flip * f)[n-1-i]
    
    conv_qf = irfft(rfft(q_flip.astype(np.float64), fft_size) * 
                      rfft(f.astype(np.float64), fft_size), fft_size)[:2*n-1]
    g = 2.0 * conv_qf[n - 1 - np.arange(n)]
    g = np.maximum(g, 0.0)  # Should be non-negative for non-negative f, q
    return g


def noregret_optimize(f0, time_limit, verbose=False, seed=0):
    """
    No-regret (multiplicative weights) game-based optimization.
    Minimizes max_k (f*f)(k) subject to sum(f)=sqrt(2n), f>=0.
    """
    rng = np.random.default_rng(seed)
    n = len(f0)
    size = 2 * n - 1
    S = np.sqrt(2.0 * n)
    
    f = simplex_project(f0.copy(), S)
    q = np.ones(size) / size
    
    best_f = f.copy()
    best_conv = fast_autocorr(f)
    best_score = best_conv.max()
    
    t0 = time.monotonic()
    
    # Parameters
    beta_base = 4.0
    q_momentum = 0.6
    use_fft_subgrad = n > 500
    
    iter_count = 0
    stall = 0
    
    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        iter_count += 1
        
        conv = fast_autocorr(f)
        curr_max = conv.max()
        
        if curr_max < best_score - 1e-10:
            best_score = curr_max
            best_f = f.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart if stalled too long
        if stall > 2000:
            f = best_f.copy() * (1.0 + 0.02 * rng.standard_normal(n))
            f = simplex_project(f, S)
            q = np.ones(size) / size
            stall = 0
            continue
        
        # Update q (Player 2: choose which lags to focus on)
        # Adaptive temperature - start broad, narrow down
        progress = min(t / 5000.0, 1.0)
        beta = beta_base + 6.0 * progress
        # Softmax with numerical stability
        shifted = conv / (curr_max + 1e-10) - 1.0  # max is 0
        logits = beta * shifted
        q_s = np.exp(logits)
        q_s_sum = q_s.sum()
        if q_s_sum < 1e-15:
            q_s = np.ones(size) / size
        else:
            q_s /= q_s_sum
        q = q_momentum * q + (1.0 - q_momentum) * q_s
        
        # Compute subgradient
        if use_fft_subgrad:
            g = compute_subgradient_fft(f, q, n)
        else:
            g = compute_subgradient(f, q, n)
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        
        # Mirror descent update with Nesterov extrapolation
        # f_new_i proportional to f_i * exp(-eta * g_i)
        eta = 0.4 * S / (g_norm * np.sqrt(n))
        eta = min(eta, 2.0)
        
        # Nesterov: extrapolate
        if t > 1:
            f_extrap = f + 0.3 * (f - getattr(noregret_optimize, '_prev_f', f))
            f_extrap = np.maximum(f_extrap, 1e-10)
        else:
            f_extrap = f.copy()
        noregret_optimize._prev_f = f.copy()
        
        f_new = f_extrap * np.exp(-eta * g)
        f_new = simplex_project(f_new, S)
        
        # Verify improvement (accept/reject)
        conv_new = fast_autocorr(f_new)
        if conv_new.max() < curr_max * 1.001:
            f = f_new
        else:
            # Try smaller step
            f_new2 = f * np.exp(-eta * 0.3 * g)
            f_new2 = simplex_project(f_new2, S)
            f = f_new2
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NR t={t} ({elapsed:.0f}s): best={best_score:.8f}, iter={iter_count}")
    
    return best_f, best_score


def subgrad_nesterov(f0, time_limit, verbose=False, seed=0):
    """
    Projected subgradient descent with Nesterov acceleration.
    Minimizes max_k (f*f)(k) on the simplex.
    """
    rng = np.random.default_rng(seed)
    n = len(f0)
    size = 2 * n - 1
    S = np.sqrt(2.0 * n)
    
    f = simplex_project(f0.copy(), S)
    f_prev = f.copy()
    
    best_f = f.copy()
    best_score = fast_autocorr(f).max()
    
    t0 = time.monotonic()
    eta = 0.5
    momentum = 0.0
    stall = 0
    
    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        momentum = min(momentum + 0.001, 0.7)
        f_y = f + momentum * (f - f_prev)
        f_y = simplex_project(f_y, S)
        
        conv = fast_autocorr(f_y)
        curr_max = conv.max()
        
        if curr_max < best_score - 1e-10:
            best_score = curr_max
            best_f = f.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 1500:
            f = best_f.copy()
            f_prev = f.copy()
            momentum = 0.0
            eta = 0.3
            stall = 0
            continue
        
        # Find active lags (top few)
        threshold = curr_max * 0.995
        active = np.where(conv >= threshold)[0]
        if len(active) > 50:
            # Use top 50
            top_idx = np.argsort(conv)[-50:]
            active = top_idx
        
        # Subgradient: average gradient of active lags
        g = np.zeros(n)
        for k in active:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                idx = k - np.arange(i_lo, i_hi + 1)
                g[i_lo:i_hi + 1] += f_y[idx]
        g *= 2.0 / len(active)
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        
        # Adaptive step size
        eta_t = eta * S / (g_norm * np.sqrt(n))
        eta_t = min(eta_t, 1.5)
        
        f_new = f_y - eta_t * g
        f_new = simplex_project(f_new, S)
        
        # Accept/reject
        conv_new = fast_autocorr(f_new)
        if conv_new.max() < curr_max * 1.01:
            f_prev = f.copy()
            f = f_new
            eta = min(eta * 1.01, 2.0)
        else:
            f_prev = f.copy()
            eta = max(eta * 0.8, 0.05)
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_f, best_score


def coord_descent_polish(f0, time_limit, verbose=False):
    """
    Coordinate descent with golden section search on individual coordinates.
    Works on the normalized simplex.
    """
    n = len(f0)
    S = np.sqrt(2.0 * n)
    f = simplex_project(f0.copy(), S)
    
    best_f = f.copy()
    best_score = fast_autocorr(f).max()
    
    t0 = time.monotonic()
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_improved = False
        perm = np.random.permutation(n)
        
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            
            ai = f[i]
            if ai < 1e-6 * S:
                continue
            
            # Golden section search on f[i]
            lo = 0.0
            hi = ai * 2.5 + 0.05 * S
            
            # Initial evaluations
            a1 = lo + (1.0 - phi) * (hi - lo)
            a2 = lo + phi * (hi - lo)
            
            f_t1 = f.copy()
            f_t1[i] = a1
            f_t1 = simplex_project(f_t1, S)
            v1 = fast_autocorr(f_t1).max()
            
            f_t2 = f.copy()
            f_t2[i] = a2
            f_t2 = simplex_project(f_t2, S)
            v2 = fast_autocorr(f_t2).max()
            
            for _ in range(20):
                if v1 < v2:
                    hi = a2
                    a2 = a1
                    v2 = v1
                    a1 = lo + (1.0 - phi) * (hi - lo)
                    f_t1 = f.copy()
                    f_t1[i] = a1
                    f_t1 = simplex_project(f_t1, S)
                    v1 = fast_autocorr(f_t1).max()
                else:
                    lo = a1
                    a1 = a2
                    v1 = v2
                    a2 = lo + phi * (hi - lo)
                    f_t2 = f.copy()
                    f_t2[i] = a2
                    f_t2 = simplex_project(f_t2, S)
                    v2 = fast_autocorr(f_t2).max()
            
            best_val = min(v1, v2)
            best_pos = a1 if v1 < v2 else a2
            
            if best_val < best_score - 1e-10:
                f[i] = best_pos
                f = simplex_project(f, S)
                best_score = fast_autocorr(f).max()
                best_f = f.copy()
                pass_improved = True
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    CD ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_f, best_score


def pair_cd_polish(f0, time_limit, verbose=False):
    """Pair swap coordinate descent for faster local optimization."""
    n = len(f0)
    S = np.sqrt(2.0 * n)
    f = simplex_project(f0.copy(), S)
    
    best_f = f.copy()
    best_score = fast_autocorr(f).max()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_improved = False
        perm = np.random.permutation(n)
        
        for idx in range(0, n - 1, 2):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            j = perm[idx + 1]
            ai, aj = f[i], f[j]
            if ai < 1e-7 * S and aj < 1e-7 * S:
                continue
            
            curr_score = best_score
            # Try several transfer amounts
            for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
                d = min(ai * frac, 0.3 * S)
                if d < 1e-7:
                    continue
                f_try = f.copy()
                f_try[i] = ai - d
                f_try[j] = aj + d
                f_try = simplex_project(f_try, S)
                sc = fast_autocorr(f_try).max()
                if sc < curr_score - 1e-11:
                    f = f_try
                    curr_score = sc
                    if sc < best_score - 1e-11:
                        best_score = sc
                        best_f = f.copy()
                    pass_improved = True
            
            # Reverse direction
            curr_score = best_score
            for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
                d = min(aj * frac, 0.3 * S)
                if d < 1e-7:
                    continue
                f_try = f.copy()
                f_try[i] = ai + d
                f_try[j] = aj - d
                f_try = simplex_project(f_try, S)
                sc = fast_autocorr(f_try).max()
                if sc < curr_score - 1e-11:
                    f = f_try
                    curr_score = sc
                    if sc < best_score - 1e-11:
                        best_score = sc
                        best_f = f.copy()
                    pass_improved = True
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0
    
    return best_f, best_score


def structured_init(n, style=0):
    """Generate structured initial sequences."""
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif style == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif style == 2:
        a = 0.4 + 0.35 * np.cos(4 * np.pi * x) + 0.15 * np.cos(8 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif style == 3:
        a = np.exp(-10.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-10.0 * (x - 0.7) ** 2)
    elif style == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif style == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 1.0
    elif style == 6:
        a = 1.0 + 0.2 * np.cos(6 * np.pi * x) + 0.1 * np.cos(12 * np.pi * x) + 0.05 * np.cos(18 * np.pi * x)
    elif style == 7:
        a = (1.0 + 0.3 * np.cos(2 * np.pi * x)) ** 2
    elif style == 8:
        a = np.exp(-3.0 * (2.0 * x - 1.0) ** 2)
    else:
        a = 0.5 + 0.5 * np.sin(np.pi * x) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def perturb_seq(a, rng, scale=0.03):
    """Perturb a sequence with multiplicative noise."""
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.1)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s

            best_seq = parent.copy()

    except Exception:

                best_seq = cand
        print(f"Best init: n={len(best_seq)}, score={best_score:.8f}")
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)

            best_score = sc
            best_seq = a.copy()

    remaining = time_budget - (time.monotonic() - t_global)

    remaining = time_budget - (time.monotonic() - t_global)

        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = pair_cd_polish(a_norm, t4, verbose=False)
        try_update(a_out, f"pcd_n{n_cur}")
    
    remaining = time_budget - (time.monotonic() - t_global)
    
    # === Phase 5: Perturbation + no-regret restart (15% time) ===
    print(f"\n=== Phase 5: Perturbation restarts ===")
    t5_total = min(remaining * 0.15, 150.0)
    n_restarts = 3
    t5_each = t5_total / n_restarts
    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        t5 = min(t5_each, remaining - 40)
        if t5 < 20:
            break
        n_cur = len(best_seq)
        scale = 0.02 + 0.015 * r
        pert = perturb_seq(best_seq, rng, scale)
        a_norm = pert * np.sqrt(2.0 * n_cur) / pert.sum()
        a_out, _ = noregret_optimize(a_norm, t5, verbose=False, seed=100 + r * 13)
        try_update(a_out, f"pert{r}_n{n_cur}")
    
    remaining = time_budget - (time.monotonic() - t_global)
    
    # === Phase 6: Try longer length (10% time) ===
    n_long = len(best_seq) + 300
    if remaining > 80:
        print(f"\n=== Phase 6: No-regret n={n_long} ===")
        t6 = min(remaining * 0.10, 120.0)
        if t6 > 30:
            init_long = interpolate_seq(best_seq, n_long)
            a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
            a_out, _ = noregret_optimize(a_norm, t6, verbose=True, seed=42)
            try_update(a_out, f"noregret_long_n{n_long}")
    
    remaining = time_budget - (time.monotonic() - t_global)
    
    # === Phase 7: Try shorter length (8% time) ===
    n_short = max(800, len(best_seq) - 400)
    if n_short != len(best_seq) and remaining > 60:
        print(f"\n=== Phase 7: No-regret n={n_short} ===")
        t7 = min(remaining * 0.08, 80.0)
        if t7 > 20:
            init_short = interpolate_seq(best_seq, n_short)
            a_norm = init_short * np.sqrt(2.0 * n_short) / init_short.sum()
            a_out, _ = noregret_optimize(a_norm, t7, verbose=False, seed=77)
            try_update(a_out, f"noregret_short_n{n_short}")
    
    remaining = time_budget - (time.monotonic() - t_global)
    
    # === Phase 8: Final no-regret polish (15% of remaining) ===
    n_best = len(best_seq)
    if remaining > 60:
        print(f"\n=== Phase 8: Final no-regret n={n_best} ===")
        t8 = min(remaining * 0.4, 150.0)
        if t8 > 20:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = noregret_optimize(a_norm, t8, verbose=True, seed=999)
            try_update(a_out, f"noregret_final_n{n_best}")
    
    remaining = time_budget - (time.monotonic() - t_global)
    
    # === Phase 9: Final CD polish ===
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 9: Final CD n={n_best} ===")
        t9 = min(remaining * 0.5, 60.0)
        if t9 > 10:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = coord_descent_polish(a_norm, t9, verbose=False)
            try_update(a_out, f"cd_final_n{n_best}")
    
    remaining = time_budget - (time.monotonic() - t_global)
    
    # === Phase 10: One more no-regret from different init ===
    if remaining > 40:
        print(f"\n=== Phase 10: Different init exploration ===")
        t10 = min(remaining * 0.5, 60.0)
        for sty in [0, 4, 7]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 20:
                break
            t10 = remaining * 0.3
            if t10 < 15:
                break
            n_cur = len(best_seq)
            init = structured_init(n_cur, sty)
            # Blend with current best
            blend = 0.5 * init + 0.5 * (best_seq / best_seq.sum())
            a_norm = blend * np.sqrt(2.0 * n_cur) / blend.sum()
            a_out, _ = noregret_optimize(a_norm, t10, verbose=False, seed=200 + sty)
            try_update(a_out, f"blend{sty}_n{n_cur}")
    
    # === Final: CD polish on absolute best ===
    n_best = len(best_seq)
    if time_budget - (time.monotonic() - t_global) > 20:
        t_final = min(time_budget - (time.monotonic() - t_global) - 5, 30.0)
        print(f"\n=== Final CD polish n={n_best} ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coord_descent_polish(a_norm, t_final, verbose=False)
        try_update(a_out, "final_cd")
    
    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    
    # Ensure all values are in valid range
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
```
