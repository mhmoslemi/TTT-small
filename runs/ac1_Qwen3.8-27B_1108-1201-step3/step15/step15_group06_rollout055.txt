"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated mirror descent
with Nesterov acceleration, adaptive step size, multi-phase search, and
length exploration. Replaces the previous LP-based approach with a direct
first-order method that is O(n log n) per iteration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolution(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_subgradient(a, active_set):
    """Compute averaged subgradient of max_k conv_k(a) over active_set."""
    n = len(a)
    m = len(active_set)
    if m == 0:
        return np.zeros(n)
    grad = np.zeros(n)
    for k in active_set:
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            indices = np.arange(j_lo, j_hi + 1)
            grad[j_lo:j_hi+1] += a[k - indices]
            if k % 2 == 0:
                j_mid = k // 2
                if j_lo <= j_mid <= j_hi:
                    grad[j_mid] -= a[j_mid]
    grad /= m
    return grad


def score_of(a):
    """Compute the objective 2*n*max(conv)/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolution(a)
    return float(2.0 * n * conv.max() / (s * s))

    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

    else:
        a = a / s
    
    conv = fast_convolution(a)
    best_max = conv.max()
    best_a = a.copy()
    
    y = a.copy()
    x_prev = a.copy()
    avg = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    eta = eta_init
    stall = 0
    best_stall = 0
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute gradient at y
        conv_y = fast_convolution(y)
        max_y = conv_y.max()
        
        # Adaptive active set tolerance

        threshold = max_y * (1.0 - tol)
        active_set = np.where(conv_y >= threshold)[0]
        if len(active_set) < 2:

        grad = compute_subgradient(y, active_set)

        # Clip to avoid underflow

        else:

def perturb(a, rng, scale=0.1):
    """Multiplicative perturbation."""
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.1)
    ap = a * noise
    return ap / ap.sum()


def interpolate_seq(a, new_n):
    """Interpolate sequence to new length, preserving sum."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new < 1e-15:
        a_new = np.ones(new_n) / new_n
    else:
        a_new = a_new / s_new
    return a_new
