from scipy.fft import rfft, irfft
import cvxpy as cp

_solver = None

def get_solver():

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        T[i:i + n, i] = a
    return T

def lp_solve_with_constraints(y, M, T, solver, active_set):
    n = len(y)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T[active_set, :] @ b_var <= M, b_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = b_var.value
    if val is None:
        return None
    return np.maximum(np.asarray(val, dtype=np.float64), 0.0)

def lse_gradient_descent(a0, time_limit, verbose=False):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n, dtype=np.float64) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1 - 1).bit_length()

    n_stages = 15
    temps = np.geomspace(100.0, 0.05, n_stages)

    for si, T_val in enumerate(temps):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        remaining = time_limit - elapsed
        if remaining < 1:
            break

        n_steps = max(20, min(500, int(remaining * 8)))

        for it in range(n_steps):
            elapsed = time.monotonic() - t0
            if elapsed > time_limit:
                break

            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -200.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            grad = 2.0 * irfft(np.conj(A) * W, fft_size)[:n]

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break

            lr = norm / (T_val * np.sqrt(n) * (1.0 + 0.02 * it))
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.3f}: best={best_score:.8f} ({time.monotonic()-t0:.0f}s)")

    return best_a, best_score

    if s < 1e-15:
        a = np.ones(n, dtype=np.float64) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1

    prev_b = None

    for t in range(1, 1000000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Constraint generation: start with top constraints, add violated ones
        sorted_indices = np.argsort(conv_y)[::-1]
        active_init = sorted_indices[:max(50, size // 3)]

        b = lp_solve_with_constraints(y, M, build_T(y), solver, active_init)
        if b is None:
            continue

        # Check all constraints and add violated ones
        cross = fast_conv_cross(y, b)
        violated = np.where(cross > M * 1.001)[0]
        if len(violated) > 0:
            active_expanded = np.unique(np.concatenate([active_init, violated]))
            if len(active_expanded) < size:
                b2 = lp_solve_with_constraints(y, M, build_T(y), solver, active_expanded)
                if b2 is not None:
                    b = b2
            else:
                b2 = lp_solve_with_constraints(y, M, build_T(y), solver, np.arange(size))
                if b2 is not None:
                    b = b2

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if verbose and t % 100 == 0:
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def pair_cd_optimize(a0, time_limit, rng, verbose=False):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n, dtype=np.float64) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    improvements = 0

    phi_g = (1.0 + np.sqrt(5.0)) / 2.0  # golden ratio

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        peak_threshold = cur_max * 0.93
        peak_lags = np.where(conv >= peak_threshold)[0]

        for idx in range(n):
            elapsed = time.monotonic() - t0
            if elapsed > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-9:
                continue

            # Candidate j's: random + structured
            j_candidates = set()
            for _ in range(8):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:5]:
                j_struct = (kl - i) % n
                j_candidates.add(j_struct)
                j_struct2 = (kl + i) % n
                j_candidates.add(j_struct2)
                j_struct3 = (2 * kl - i) % n
                j_candidates.add(j_struct3)
            j_candidates.discard(i)

            for j in list(j_candidates)[:12]:
                if j == i:
                    continue
                aj = a[j]

                # Compute gain vector for transfer from i to j
                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # Golden section search for optimal delta
                d_lo = 0.0
                d_hi = ai * 0.95

                def obj_at(d):
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2.0 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2.0 * aj * d + d * d
                    return new_conv.max()

                # Golden section
                c1 = d_hi / phi_g
                c2 = d_hi - c1
                f1 = obj_at(c1)
                f2 = obj_at(c2)

                for _ in range(10):
                    if f1 < f2:
                        d_hi = c2
                        c2 = c1
                        f2 = f1
                        c1 = d_hi / phi_g
                        f1 = obj_at(c1)
                    else:
                        d_lo = c1
                        c1 = c2
                        f1 = f2
                        c2 = d_lo + (d_hi - d_lo) / phi_g
                        f2 = obj_at(c2)

                best_d = (d_lo + d_hi) / 2.0
                best_val = obj_at(best_d)

                if best_val < cur_max - 1e-10 and best_d > 1e-12:
                    a[i] -= best_d
                    a[j] += best_d
                    s_new = a.sum()
                    if s_new < 1e-15:
                        a[i] += best_d
                        a[j] -= best_d
                        continue
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        improvements += 1
                    conv = fast_conv_self(a)
                    cur_max = conv.max()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 6:
                break
            if no_improve >= 2:
                n_perturb = max(5, n // 50)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.08 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and improvements % 50 == 0 and improvements > 0:
            print(f"    PCD imp={improvements} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def spectral_refine(a0, time_limit, verbose=False):
    """Use frequency-domain insights to flatten the autocorrelation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv_self(a).max()
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()

    # Target: make autocorrelation as flat as possible
    target = (norm ** 2) / (2 * n - 1)  # ideal flat level

    for it in range(200000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        conv = fast_conv_self(a)
        cur_max = conv.max()

        # Weight lags that are above the target more heavily
        excess = np.maximum(conv - target, 0.0)
        w = excess ** 2
        w_sum = w.sum()
        if w_sum < 1e-15:
            break
        w = w / w_sum

        # Gradient of sum w[k] * conv[k]
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        grad = 2.0 * irfft(np.conj(A) * W, fft_size)[:n]

        # Step with projection
        step = 0.5 / (np.linalg.norm(grad) + 1e-10) * norm
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new * norm / s_new
        a = a_new

        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Check convergence
        if it > 100 and it % 500 == 0:
            if abs(sc - best_score) < 1e-12:
                break

        if verbose and it % 1000 == 0:
            print(f"    Spectral it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def multi_temperature_lse(a0, time_limit, verbose=False):
    """Multi-round LSE with decreasing temperatures for precise optimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n, dtype=np.float64) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()

    # Multiple rounds with decreasing temperature
    rounds = [(50, 100), (20, 30), (5, 10), (1, 5), (0.3, 1)]

    for round_idx, (t_start, t_end) in enumerate(rounds):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        remaining = time_limit - elapsed
        if remaining < 5:
            break

        temps = np.geomspace(t_start, t_end, 8)
        for T_val in temps:
            elapsed = time.monotonic() - t0
            if elapsed > time_limit:
                break
            remaining = time_limit - elapsed
            if remaining < 2:
                break

            n_steps = max(10, min(200, int(remaining * 5)))
            for it in range(n_steps):
                elapsed = time.monotonic() - t0
                if elapsed > time_limit:
                    break
                conv = fast_conv_self(a)
                max_c = conv.max()
                if max_c < 1e-15:
                    break
                shifted = (conv - max_c) / T_val
                shifted = np.maximum(shifted, -200.0)
                exp_c = np.exp(shifted)
                w = exp_c / exp_c.sum()
                W = rfft(w, fft_size)
                A = rfft(a, fft_size)
                grad = 2.0 * irfft(np.conj(A) * W, fft_size)[:n]
                grad_norm = np.linalg.norm(grad)
                if grad_norm < 1e-15:
                    break
                lr = norm / (T_val * np.sqrt(n) * (1.0 + 0.01 * it))
                a_new = a - lr * grad
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    a_new = a.copy()
                else:
                    a_new = a_new * norm / s_new
                a = a_new
                sc = fast_conv_self(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        if verbose:
            print(f"    MultiLSE round {round_idx} (T: {t_start}->{t_end}) ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def evaluate_seq(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        if len(a_list) == 0:
            return float('inf')
        for val in a_list:
            if val < 0 or not np.isfinite(val):
                return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_seq(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_seq(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.2
        best_seq = np.maximum(best_seq, 0) + 0.005
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_seq(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # === Round 1: Main optimization at base length ===
    # Phase 1: LSE warmup
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 100.0)
    print(f"\n=== Phase 1: LSE GD n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 10:
        a_out, _ = lse_gradient_descent(best_seq, t1, verbose=True)
        try_update(a_out, "lse1")

    # Phase 2: LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.30, 480.0)
    print(f"\n=== Phase 2: LP Nesterov n={n_base} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = lp_nesterov_optimize(best_seq, t2, solver, verbose=True)
        try_update(a_out, "lp1")

    # Phase 3: Multi-temperature LSE for precise max reduction
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 3: Multi-LSE n={n_base} (budget={t3:.0f}s) ===")
    if t3 > 10:
        a_out, _ = multi_temperature_lse(best_seq, t3, verbose=True)
        try_update(a_out, "multilse1")

    # Phase 4: Pair CD
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 4: Pair CD n={n_base} (budget={t4:.0f}s) ===")
    if t4 > 15:
        a_out, _ = pair_cd_optimize(best_seq, t4, rng, verbose=True)
        try_update(a_out, "pcd1")

    # === Round 2: Refine ===
    # Phase 5: LP Nesterov round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 5: LP Nesterov r2 n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 30:
        a_out, _ = lp_nesterov_optimize(best_seq, t5, solver, verbose=True)
        try_update(a_out, "lp2")

    # Phase 6: Spectral refinement
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.08, 100.0)
    print(f"\n=== Phase 6: Spectral refine n={len(best_seq)} (budget={t6:.0f}s) ===")
    if t6 > 10:
        a_out, _ = spectral_refine(best_seq, t6, verbose=True)
        try_update(a_out, "spectral1")

    # Phase 7: Multi-LSE round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.06, 80.0)
    print(f"\n=== Phase 7: Multi-LSE r2 n={len(best_seq)} (budget={t7:.0f}s) ===")
    if t7 > 10:
        a_out, _ = multi_temperature_lse(best_seq, t7, verbose=True)
        try_update(a_out, "multilse2")

    # Phase 8: Final CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    t8 = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 8: Final CD n={len(best_seq)} (budget={t8:.0f}s) ===")
    if t8 > 10:
        a_out, _ = pair_cd_optimize(best_seq, t8, rng, verbose=True)
        try_update(a_out, "pcd_final")

    # === Try longer sequence ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        n_long = n_base + 200
        print(f"\n=== Phase 9: Longer n={n_long} (budget={remaining:.0f}s) ===")
        # Interpolate
        old_x = np.linspace(0.0, 1.0, len(best_seq))
        new_x = np.linspace(0.0, 1.0, n_long)
        a_long = np.interp(new_x, old_x, best_seq)
        a_long = np.maximum(a_long, 0.0)
        a_long = a_long / a_long.sum()

        t9 = remaining * 0.5
        a_out, _ = lse_gradient_descent(a_long, min(t9 * 0.4, 60.0), verbose=False)
        try_update(a_out, f"lse_long{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t9b = min(remaining * 0.6, 180.0)
            a_out, _ = lp_nesterov_optimize(a_long, t9b, solver, verbose=False)
            try_update(a_out, f"lp_long{n_long}")

    # Final evaluation
    if best_seq is not None:
        final_score = evaluate_seq(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
