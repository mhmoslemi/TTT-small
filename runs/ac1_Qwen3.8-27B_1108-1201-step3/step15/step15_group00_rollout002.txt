    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def logsumexp_beta(conv, beta):
    """Compute (1/beta)*log(sum(exp(beta*conv))) with numerical stability."""
    c_max = conv.max()
    shifted = beta * (conv - c_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    log_sum = np.log(np.sum(np.exp(shifted))) + beta * c_max
    return log_sum / beta

def smooth_objective_and_grad(z, beta, n):
    """
    Objective: (1/beta)*log(sum(exp(beta*conv(a,a)))) where a = softmax(z)
    Returns (value, gradient_wrt_z)
    """
    # Compute a from z using softmax
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    a_sum = a.sum()
    a = a / a_sum  # a is on the simplex
    
    # Compute conv
    c = fast_autocorr(a)
    
    # Compute (1/beta)*log(sum(exp(beta*c)))
    c_max = c.max()
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    log_sum_exp = np.log(exp_shifted.sum()) / beta + c_max
    
    # Gradient with respect to a:
    # w[k] = exp(beta*(c[k]-c_max)) / sum(exp(beta*(c[j]-c_max)))
    w = exp_shifted / exp_shifted.sum()  # probability weights
    
    # grad_c[i] = sum_k w[k] * dc[k]/da[i] = sum_k w[k] * a[k-i] (if valid)
    # This is a cross-correlation: grad[i] = sum_m w[m+i] * a[m]
    # Can be computed as: grad = convolve(w_padded, a) ... let me think
    # c[k] = sum_j a[j]*a[k-j], dc[k]/da[i] = a[k-i] (if 0<=k-i<n)
    # grad[i] = sum_k w[k]*a[k-i] where 0<=k-i<n
    # Let m = k-i, then grad[i] = sum_m w[m+i]*a[m] where 0<=m<n and 0<=m+i<2n-1
    # This is like a cross-correlation
    # Use FFT: grad = ifft(fft(w_full, size) * conj(fft(a_padded, size)))? 
    # Actually, let's just do it directly for moderate n, or use FFT
    
    # Using FFT:
    # grad[i] = sum_{m=0}^{n-1} w[m+i] * a[m]
    # This is the cross-correlation of w (length 2n-1) with a (length n), evaluated at lags 0 to n-1
    # cross_corr[i] = sum_m w[m+i]*a[m] = (a * rev_w)[i + (2n-2-n+1)] ... hmm
    # Let me just use a direct approach or a clean FFT formulation
    
    # Actually: define w_full of length 2n-1 (pad to next power of 2 for FFT)
    # grad = ifft(fft(a, size) * conj(fft(w_full, size)))[:n]
    # Because (a * rev(w))[k] = sum_j a[j] * rev(w)[k-j] = sum_j a[j] * w[2n-2-(k-j)] = sum_j a[j]*w[2n-2-k+j]
    # Hmm this is getting complicated. Let me just do it with a direct sum for now if n is moderate.
    
    # For n=2400, a direct sum is O(n^2) = 5.76M which is fine.
    # But let's use FFT for speed.
    
    # The relation is: grad[i] = sum_{m=0}^{n-1} a[m] * w[m+i]
    # This is the cross-correlation between a and w at lag i.
    # cross_corr(a, w)[i] = sum_m a[m] * w[m+i]
    # In FFT terms: cross_corr = ifft(fft(a, N) * conj(fft(w, N)))[:n]
    # But w has length 2n-1, so we need to be careful.
    
    # Let's define: extend a to length 2n-1 (zero-padded), call it a_ext
    # Then grad[i] = sum_m a_ext[m] * w[m+i] for m where both are defined
    # = cross_corr(a_ext, w)[i]
    
    # Using FFT:
    size_fft = 1 << (4 * n - 2).bit_length()
    A_fft = np.fft.rfft(a, size_fft)
    W_fft = np.fft.rfft(w, size_fft)
    cross = np.fft.irfft(A_fft * np.conj(W_fft), size_fft)[:n]
    
    # Now gradient with respect to z (for softmax parameterization):
    # da/dz[i] = a[i] * (delta[i] - a[i])
    # dz/da chain rule: dF/dz = 1/a * (dF/da - <dF/da, a> * a)
    # Wait: a = softmax(z), so da_i/dz_j = a_i*(delta_{ij} - a_j)
    # dF/dz_j = sum_i (dF/da_i) * da_i/dz_j = sum_i grad_a[i] * a_i * (delta_{ij} - a_j)
    # = grad_a[j] * a_j - a_j * sum_i grad_a[i] * a_i
    # = a_j * (grad_a[j] - sum_i grad_a[i] * a_i)
    
    grad_a = cross
    mean_grad = np.dot(grad_a, a)
    grad_z = a * (grad_a - mean_grad)
    
    return log_sum_exp, grad_z

def smooth_lbfgsb(a0, n, start_beta=20, end_beta=200, n_stages=5, time_limit=300, verbose=True):
    """
    Continuation method: start with small beta, gradually increase.
    Use L-BFGS-B on the softmax parameterized problem.
    """
    t0 = time.monotonic()
    a = norm_sum1(a0)
    z = np.log(a + 1e-30)  # inverse softmax
    
    betas = np.geomspace(start_beta, end_beta, n_stages)
    
    for stage, beta in enumerate(betas):
        if time.monotonic() - t0 > time_limit:
            break
        
        if verbose:
            print(f"    L-BFGS-B stage {stage+1}/{n_stages}, beta={beta:.1f}")
        
        try:
            result = minimize(
                smooth_objective_and_grad,
                z,
                args=(beta, n),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-8, 'maxfun': 500}
            )
            z = result.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
        except Exception as e:
            if verbose:
                print(f"    L-BFGS-B error: {e}")
            continue
        
        M_new = fast_autocorr(a_new).max()
        M_old = fast_autocorr(a).max()
        if M_new < M_old:
            a = a_new
            if verbose:
                print(f"      improved: M {M_old:.8f} -> {M_new:.8f}")
    
    return a, fast_autocorr(a).max()

def multiplicative_weights(a0, time_limit, top_k=50, beta_smooth=5.0, verbose=True):
    """
    Mirror descent on the simplex with FFT-based gradients.
    Uses a soft max (top-k weighted) for the subgradient.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    gamma = 2.0
    stall = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        # Compute autocorrelation
        c = fast_autocorr(y)
        
        # Compute soft subgradient (top-k weighted)
        k_max = min(top_k, len(c))
        top_idx = np.argsort(c)[::-1][:k_max]
        top_vals = np.maximum(c[top_idx], 1e-30)
        w = top_vals ** beta_smooth
        w_sum = w.sum()
        w = w / w_sum
        
        # Gradient: grad[i] = sum_k w[k]*c_coeff... 
        # Actually, the subgradient of max(c) at the peak includes contributions from
        # all peaks. The "soft" version weights them.
        # dc[k]/da[i] = a[k-i] (if valid)
        # grad[i] = sum_{k in top} w[k] * a[k-i]
        
        # Use FFT for efficiency
        w_full = np.zeros(2 * n - 1)
        w_full[top_idx] = w
        size_fft = 1 << (4 * n - 2).bit_length()
        A_fft = np.fft.rfft(y, size_fft)
        W_fft = np.fft.rfft(w_full, size_fft)
        grad = np.fft.irfft(A_fft * np.conj(W_fft), size_fft)[:n]
        
        # Step size
        eta = gamma / (np.sqrt(t) + 1.0)
        
        # Mirror descent update (multiplicative weights)
        log_a = np.log(y) - eta * grad
        log_a = log_a - log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 5000:
            # Restart
            rng = np.random.default_rng(t)
            pert = rng.normal(0, 0.02, n)
            x_restarted = best_a * (1 + pert)
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted
            stall = 0
            gamma = max(gamma * 0.8, 0.5)
            top_k = min(top_k + 10, 100)
            beta_smooth = max(beta_smooth - 0.5, 2.0)
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MW t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def coordinate_polish(a0, time_limit, verbose=False, rng_seed=42):
    """Coordinate descent: move mass from peak-contributing positions to low-sensitivity positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 100
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_autocorr(a)
        top_k = np.argsort(c)[::-1][:min(20, len(c))]
        improved = False
        
        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find best destination
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:8]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n - 1, int(k2))
                in_peak[s2:e2 + 1] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.argwhere(a < np.median(a)).flatten()
            
            # Score each candidate by sensitivity
            best_dst = -1
            best_score = float('inf')
            sample = candidates[rng.permutation(len(candidates))[:min(500, len(candidates))]]
            
            for dst in sample:
                if dst == src:
                    continue
                # Sensitivity: how much would adding to dst increase the peaks?
                sens = 0.0
                for k2 in top_k[:5]:
                    k2i = int(k2)
                    d1 = k2i - dst
                    d2 = dst - k2i
                    add = 0.0
                    if 0 <= d1 < n:
                        add += a[d1]
                    if 0 <= d2 < n:
                        add += a[d2]
                    sens += add
                
                if sens < best_score:
                    best_score = sens
                    best_dst = dst
            
            if best_dst < 0:
                continue
            
            dst = best_dst
            for frac in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
                amt = src_val * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
    
    return best_a, best_m

def propose_candidate(budget_s=1800.0):
    """Main search: smooth L-BFGS-B + multiplicative weights + coordinate polish + multi-length."""
    rng = np.random.default_rng(2024)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent / parent.sum()
            best_score = p_score
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    if best_seq is None:
        n_base = 2000
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    
    # === Phase 1: Smooth L-BFGS-B with continuation ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: L-BFGS-B smooth n={n0} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.20, 250.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_lbfgsb(a_in, n0, start_beta=10, end_beta=100, n_stages=4, time_limit=t1, verbose=True)
        try_update(a_out, f"lbfgsb_n{n0}")
    
    # === Phase 2: Multiplicative weights refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: MW n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.35, 400.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = multiplicative_weights(a_in, t2, top_k=50, beta_smooth=4.0, verbose=True)
        try_update(a_out, f"mw_n{len(best_seq)}")
    
    # === Phase 3: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Coord n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 150.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t3, verbose=True, rng_seed=42)
        try_update(a_out, f"coord_n{len(best_seq)}")
    
    # === Phase 4: Try at slightly different length ===
    remaining = time_budget - (time.monotonic() - t_global)
    for n_try in [n0 + 200, n0 + 500, n0 - 200]:
        if n_try < 500 or n_try > 5000:
            continue
        if remaining <
