def fast_autocorr(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n + 1) > (css - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = int(rho[-1])
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_grad(a, w):
    """Gradient of T*log(sum exp(c/T)): grad_i = 2 * (w cross_correlate a)[i]."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # (w * a reversed)[i] = sum_j w[j] * a[i-j] (cross correlation)
    # In FFT: irfft(W * conj(A)) gives (w * a_reversed) truncated
    cross = irfft(W * np.conj(A), fft_size)[:n]
    return 2.0 * cross

def lp_best_response(a, M, solver, active_indices=None):
    """Solve max sum(b) s.t. conv(a,b)[k] <= M for k in active_indices, b >= 0."""
    n = len(a)
    size = 2 * n - 1

    if active_indices is None:
        active_indices = np.arange(size)

    # Build constraint matrix: for each k in active_indices,
    # (conv(a,b))[k] = sum_i a[max(0,k-n+1) <= i <= min(k,n-1)] * b[k-i]
    # = sum_j b[j] * a[k-j] (for valid j)
    # So the constraint matrix T has T[k_idx, j] = a[k - j] if valid, else 0
    
    T = np.zeros((len(active_indices), n))
    for idx, k in enumerate(active_indices):
        for j in range(max(0, k - n + 1), min(k, n - 1) + 1):
            T[idx, j] = a[k - j]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def nesterov_lp(a0, time_limit, solver, verbose=False, avg_window=50):
    """LP best-response with Nesterov acceleration and Polyak averaging."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stuck_count = 0
    avg_a = np.zeros(n)
    n_avg = 0

    while time.monotonic() - t0 < time_limit:
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-14:
            break

        # Adaptive constraint generation
        conv_y = fast_autocorr(y)
        # Use constraints that are close to the max
        frac = 0.9 if n_iters < 100 else 0.85
        active = np.where(conv_y >= frac * M_curr)[0]
        if len(active) < 5:
            active = np.where(conv_y >= 0.5 * M_curr)[0]
        if len(active) < 3:
            active = np.where(conv_y >= 0.3 * M_curr)[0]
        if len(active) < 3:
            active = np.arange(len(conv_y))
        # Cap the number of constraints for speed
        if len(active) > 300:
            # Keep the most active ones
            sorted_active = active[np.argsort(conv_y[active])[::-1]]

        stuck_count = 0
        b_sum = b.sum()

        n_iters += 1

        alpha = 2.0 / (n_iters + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g

        if s_y < 1e-15:

            a_new = g.copy()
        else:

            # Actually the Nesterov momentum is (t-1)/(t+2) for the extrapolation
            beta = (n_iters - 0.0) / (n_iters + 2.0)

            a_new = np.maximum(a_new, 0.0)

                a_new = y_new.copy()

                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        # Polyak averaging

        if n_avg >= avg_window and n_avg % (avg_window // 10) == 0:

            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:

        if verbose and n_iters % 100 == 0:

    # Final Polyak average

        avg_final = avg_a / n_avg
        s_avg_final = avg_final.sum()
        if s_avg_final > 1e-15:

            score_avg_final = fast_autocorr(avg_final).max()
