def smooth_optimize(a0, time_limit, verbose=True):
    """Minimize smooth approximation of max(conv(a,a)) using accelerated projected gradient."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    t0 = time.monotonic()

    # Adaptive temperature schedule
    temps = [30.0, 15.0, 8.0, 4.0, 2.0, 1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]

    y = a.copy()
    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        n_steps = max(50, min(800, int(remaining * 4)))

        y = a.copy()
        momentum = np.zeros(n)
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            value, grad = smooth_obj_and_grad(y, T)

            # Adaptive step size with backtracking
            step = 1.0 / (np.linalg.norm(grad) / max(norm, 1e-10) + 1e-10)
            step = min(step, norm / (n * 0.3))

            a_new = y - step * grad
            a_new = project_simplex(a_new, norm)

            # Nesterov momentum
            alpha = 2.0 / (it + 2.0)
            y_new = (1.0 - alpha) * y + alpha * a_new
            y_new = project_simplex(y_new, norm)

            if it == 0:
                a_curr = a_new.copy()
            else:
                momentum = alpha * (y_new - y)
                a_curr = y_new + momentum
                a_curr = project_simplex(a_curr, norm)

            y = y_new
            a = a_curr

            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

            score_y = fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()

            if verbose and it % 100 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Smooth T={T:.4f} step={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def lp_best_response(a, M, active_indices=None, solver='ECOS'):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    import cvxpy as cp
    n = len(a)
    size = 2 * n - 1
    
    if active_indices is not None and len(active_indices) < size:
        T = np.zeros((len(active_indices), n))
        for idx, j in enumerate(active_indices):
            # T[idx, i] = a[j-i] if 0 <= j-i < n
            lo = max(0, j - n + 1)
            hi = min(n - 1, j)
            if lo <= hi:
                T[idx, lo:hi+1] = a[j-hi:j-lo+1][::-1]
    else:
        T = np.zeros((size, n))
        for i in range(n):
            T[i:i+n, i] = a
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def nesterov_lp(a0, time_limit, solver='ECOS', verbose=True):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive constraint generation
        threshold = 0.8 * M_curr
        active = np.where(conv_y >= threshold)[0]
        if len(active) < 10:
            threshold = 0.5 * M_curr
            active = np.where(conv_y >= threshold)[0]
        if len(active) < 5:
            active = np.argsort(conv_y)[-50:]

        b = lp_best_response(y, M_curr, active_indices=active, solver=solver)
        if b is None:
            b = lp_best_response(y, M_curr, active_indices=None, solver=solver)
            if b is None:
                break

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if verbose and n_iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def peak_shaving_cd(a0, time_limit, rng, verbose=True):
    """Peak-shaving coordinate descent: identify peaks and redistribute mass."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()

    no_improve = 0
    while time.monotonic() - t0 < time_limit:
        if no_improve > 200:
            break
        
        conv = fast_autocorr(a)
        peak_val = conv.max()
        # Find positions contributing to the peak
        peak_pos = np.argmax(conv)
        
        # The peak at position p gets contributions from pairs (i, p-i) for valid i
        # Try to reduce the peak by moving mass from high-contributing positions
        improved = False
        
        # Find the top contributing pairs to the peak
        p = peak_pos
        contributors = []
        for i in range(n):
            j = p - i
            if 0 <= j < n:
                contributors.append((i, j, a[i] * a[j]))
        
        contributors.sort(key=lambda x: -x[2])
        
        # Try moving mass from the highest contributor to a less critical position
        tried = 0
        for i, j, contrib in contributors[:20]:
            if tried > 50:
                break
            tried += 1
            
            # Try moving some mass from position i to a random less-critical position
            for _ in range(5):
                target = rng.integers(0, n)
                if target == i or target == j:
                    continue
                
                # Try different fractions
                for frac in [0.1, 0.3, 0.5]:
                    d = a[i] * frac
                    if d < norm * 1e-10:
                        continue
                    a_test = a.copy()
                    a_test[i] -= d
                    a_test[target] += d
                    a_test = a_test * norm / a_test.sum()
                    conv_test = fast_autocorr(a_test)
                    if conv_test.max() < conv.max() - 1e-12:
                        a = a_test
                        conv = conv_test
                        if conv.max() < best_score:
                            best_score = conv.max()
                            best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    PeakCD ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def freq_flatness_optimize(a0, time_limit, verbose=True):
    """Optimize for flat power spectrum to minimize time-domain peaks."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

    else:
        a = a * norm / s0

    t0 = time.monotonic()

    fft_size = 1 << (2 * n - 1).bit_length()
    
    # We want to minimize the max of the inverse FFT of |A(f)|^2
    # Equivalently, make |A(f)|^2 as flat as possible
    # Use gradient descent on the variance of |A(f)|^2
    lr = 0.01
    decay = 0.999
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = rfft(a, fft_size)
        power = np.abs(A)**2
        
        # Objective: minimize max(a*a) which correlates with peaks in power spectrum
        # Use smooth approximation
        T = 0.5
        p_max = power.max()
        w = np.exp((power - p_max) / T)
        w = w / w.sum()
        
        # Gradient of log-sum-exp of w.r.t. a
        # d/d(a[i]) of sum_j w_j * |A(f_j)|^2
        # = 2 * Re(ifft(2 * w * conj(A)))... this is getting complicated
        # Let's just use the smooth gradient from before
        value, grad = smooth_obj_and_grad(a, T)
        
        step = norm / (np.linalg.norm(grad) + 1.0)
        step = min(step, 0.1)
        
        a_new = a - step * grad
        a_new = project_simplex(a_new, norm)
        
        score = fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score

        lr *= decay
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FreqFlat iter={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def coordinate_descent(a0, time_limit, rng, verbose=True):
    """Fast coordinate descent: try single coordinate updates and pair updates."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if no_improve > 50:
            break
        
        improved = False
        perm = rng.permutation(n)
        
        for idx in perm[:min(n, 400)]:  # Limit work per pass
            if time.monotonic() - t0 > time_limit:
                break
            
            i = perm[idx]
            if a[i] < norm * 1e-12:
                continue
            
            # Try moving mass from i to nearby positions
            offsets = [1, 2, 3, 5, 8, 13, 21, 34, 55]
            for offset in offsets:
                for sign in [1, -1]:
                    j = (i + sign * offset) % n
                    if j == i:
                        continue
                    
                    # Try a few fractions
                    for frac in [0.2, 0.5, 0.8]:
                        d = a[i] * frac
                        if d < norm * 1e-12:
                            continue
                        a_test = a.copy()
                        a_test[i] -= d
                        a_test[j] += d
                        a_test = a_test * norm / a_test.sum()
                        conv_test = fast_autocorr(a_test)
                        if conv_test.max() < conv.max() - 1e-12:
                            a = a_test
                            conv = conv_test
                            if conv.max() < best_score:
                                best_score = conv.max()
                                best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            
            if not improved:
                no_improve += 1
            else:
                no_improve = 0
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    CD ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def get_solver():
    """Try to find the best available LP solver."""
    import cvxpy as cp
    for s in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    print(f"  LP solver (fallback): ECOS")
    return cp.ECOS

def make_gaussian_init(n, sigma_frac=0.3):
    """Create a Gaussian-shaped initial sequence."""
    x = np.arange(n)
    center = n / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    return a

def make_cosine_init(n):
    """Create a cosine window initial sequence."""
    x = np.arange(n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    return np.maximum(a, 0)

def make_triangular_init(n):
    """Create a triangular initial sequence."""
    x = np.arange(n)
    a = 1.0 - np.abs(x - (n-1)/2.0) / ((n-1)/2.0)
    return np.maximum(a, 0)

def make_broader_init(n, width=0.6):
    """Create a broad top-hat initial sequence."""
    a = np.zeros(n)
    start = int(n * (1 - width) / 2)
    end = int(n * (1 + width) / 2)
    a[start:end] = 1.0
    # Smooth edges
    edge = max(1, n // 20)
    for i in range(edge):
        if start - i >= 0:
            a[start - i] = (i + 1) / (edge + 1)
        if end + i < n:
            a[end + i] = (edge - i) / (edge + 1)
    return a

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) * np.sqrt(2.0 * new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) * np.sqrt(2.0 * new_n) / new_n
    return f_interp * np.sqrt(2.0 * new_n) / ss
