        if remaining < 0.5:
            break
        n_steps = max(50, min(2000, int(remaining * 2)))
        
        y = a.copy()  # mirror variable (log scale)
        # Initialize y from a: y_i = log(a_i) approximately
        # Actually, for mirror descent on the simplex, we work with the dual variable
        # Primal: a_i = exp(y_i) / sum exp(y_j)
        # But we can also work directly: a_i <- a_i * exp(-eta * grad_i)
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            c = fast_autocorr(y)
            c_max = c.max()
            exp_c = np.exp((c - c_max) / T)
            w = exp_c / exp_c.sum()
            
            grad = compute_smooth_grad(y, w)
            
            # Adaptive step size
            # The gradient magnitude gives us a natural scale
            grad_norm = np.abs(grad).max()
            if grad_norm < 1e-15:
                break
            eta = 0.5 / grad_norm  # scale so that max relative change is ~0.5
            
            # Multiplicative update
            # a_i <- a_i * exp(-eta * grad_i)
            log_update = -eta * grad
            # Clip to prevent overflow/underflow
            log_update = np.clip(log_update, -5.0, 5.0)
            a_new = y * np.exp(log_update)
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new * norm / a_new.sum()
            
            y = a_new
            
            score = fast_autocorr(y).max()
            if score < best_score:
                best_score = score
                best_a = y.copy()
            
            if verbose and it % 100 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Mirror T={T:.3f} step={it} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def accelerated_mirror_descent(a0, time_limit, verbose=True):
    """Accelerated mirror descent with Nesterov extrapolation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    temps = np.array([30.0, 15.0, 8.0, 4.0, 2.0, 1.0, 0.5, 0.25, 0.125, 0.06, 0.03, 0.015, 0.008, 0.004])
    
    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        n_steps = max(50, min(3000, int(remaining * 1.5)))
        
        y = a.copy()
        avg_a = np.zeros(n)
        n_avg = 0
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            c = fast_autocorr(y)
            c_max = c.max()
            exp_c = np.exp((c - c_max) / T)
            w = exp_c / exp_c.sum()
            
            grad = compute_smooth_grad(y, w)
            
            grad_norm = np.abs(grad).max()
            if grad_norm < 1e-15:
                break
            
            # Adaptive step with Nesterov
            alpha = 2.0 / (it + 2.0)
            eta = 0.3 / (grad_norm * (1.0 + 0.1 * it / n_steps))
            
            log_update = np.clip(-eta * grad, -3.0, 3.0)
            a_new = y * np.exp(log_update)
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new * norm / a_new.sum()
            
            # Nesterov extrapolation
            y_new = (1.0 - alpha) * y + alpha * a_new
            y_new = np.maximum(y_new, 1e-15)
            y_new = y_new * norm / y_new.sum()
            
            # Averaging
            avg_a += a_new
            n_avg += 1
            
            # Check improving solutions
            for cand in [a_new, y_new]:
                sc = fast_autocorr(cand).max()
                if sc < best_score:
                    best_score = sc
                    best_a = cand.copy()
            
            if it % 50 == 0:
                avg_cand = avg_a / n_avg
                avg_cand = avg_cand * norm / avg_cand.sum()
                sc_avg = fast_autocorr(avg_cand).max()
                if sc_avg < best_score:
                    best_score = sc_avg
                    best_a = avg_cand.copy()
            
            y = y_new
            
            if verbose and it % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"    AccMirror T={T:.3f} step={it} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        T[i:i+n, i] = a
    return T

def lp_best_response(a, M, solver, active_indices=None, all_constraints=False):
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    
    if active_indices is not None and not all_constraints:
        T_use = T[active_indices, :]
    else:
        T_use = T
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_use @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def lp_nesterov_v2(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov and constraint generation (improved)."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    stuck_count = 0
    
    # Adaptive constraint threshold
    threshold_start = 0.9
    threshold_end = 0.5
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break
        
        # Adaptive threshold (start tight, loosen over time)
        progress = min(t / 500.0, 1.0)
        threshold = threshold_start - progress * (threshold_start - threshold_end)
        active = np.where(conv_y >= threshold * M_curr)[0]
        if len(active) < 3:
            active = np.arange(len(conv_y))
        
        b = lp_best_response(y, M_curr, solver, active_indices=active)
        if b is None:
            b = lp_best_response(y, M_curr, solver, all_constraints=True)
            if b is None:
                stuck_count += 1
                if stuck_count > 20:
                    break
                continue
        
        stuck_count = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        for cand in [y, a]:
            sc = fast_autocorr(cand).max()
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()
        
        avg_a += a
        n_avg += 1
        if n_avg % 30 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
        
        if verbose and n_iters % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPv2 iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def peak_directed_cd(a0, time_limit, rng, verbose=True):
    """Coordinate descent that specifically targets peak-contributing pairs."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        
        # Find all near-peak positions
        peak_val = conv.max()
        peak_positions = np.where(conv >= 0.98 * peak_val)[0]
        
        # For each peak position, find the top contributing pairs
        for k in peak_positions:
            if time.monotonic() - t0 > time_limit:
                break
            
            # conv[k] = sum_i a[i] * a[k-i] for valid i
            # Find pairs (i, k-i) with large product
            valid_i = np.arange(max(0, k - n + 1), min(k + 1, n))
            products = a[valid_i] * a[k - valid_i]
            
            # Sort by product (descending)
            sorted_idx = np.argsort(products)[::-1]
            
            for si in sorted_idx[:10]:  # Top 10 contributing pairs
                i = valid_i[si]
                j = k - i
                if i == j:
                    # Can't split a pair with itself, try moving mass away
                    continue
                if a[i] < norm * 1e-10 or a[j] < norm * 1e-10:
                    continue
                
                # Try reducing both i and j and adding to a position that doesn't contribute to this peak
                # Find a position m where a[m] * a[k-m] is small (or m or k-m is out of range)
                # Simple heuristic: move mass to a position far from the peak
                for target in [i, j]:  # Try moving from i or j
                    # Find a good target position
                    best_target = -1
                    best_gain = 0
                    for m in range(0, n, max(1, n // 50)):
                        if m == target or m == (k - target) if k - target < n else False:
                            continue
                        # Check if m contributes to this peak
                        contrib = 0
                        if 0 <= k - m < n:
                            contrib = a[m] * a[k - m]
                        if contrib < 0.1 * products[si]:
                            # Try moving small amount from target to m
                            delta = a[target] * 0.05
                            if delta < norm * 1e-10:
                                continue
                            a_test = a.copy()
                            a_test[target] -= delta
                            a_test[m] += delta
                            a_test = a_test * norm / a_test.sum()
                            conv_test = fast_autocorr(a_test)
                            if conv_test.max() < best_score - 1e-12:
                                a = a_test
                                conv = conv_test
                                best_score = conv.max()
                                best_a = a.copy()
                                pass_improved = True
                                no_improve = 0
                                break
        
        if not pass_improve:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def coordinate_descent_targeted(a0, time_limit, rng, verbose=True):
    """Targeted coordinate descent: try moving mass from high-value to low-value positions."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        
        perm = rng.permutation(n)
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-8:
                continue
            
            # Try moving from i to several target positions
            # Prioritize targets that don't increase the peak
            targets = []
            # Positions that are "far" from i in terms of peak contribution
            for offset in [n//4, n//3, n//2, 2*n//3, 3*n//4, n//8, 7*n//8]:
                for sign in [1, -1]:
                    j = (i + sign * offset) % n
                    if j != i:
                        targets.append(j)
            
            # Also try some random targets
            targets.extend(rng.integers(0, n, size=5).tolist())
            
            for j in targets:
                if j >= n or j < 0 or j == i:
                    continue
                if a[i] < norm * 1e-10:
                    break
                
                # Try several move amounts
                for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                    delta = a[i] * frac
                    if delta < norm * 1e-10:
                        continue
                    a_test = a.copy()
                    a_test[i] -= delta
                    a_test[j] += delta
                    a_test = np.maximum(a_test, 0)
                    s_test = a_test.sum()
                    if s_test < 1e-15:
                        continue
                    a_test = a_test * norm / s_test
                    conv_test = fast_autocorr(a_test)
                    if conv_test.max() < conv.max() - 1e-13:
                        a = a_test
                        conv = conv_test
                        if conv.max() < best_score - 1e-12:
                            best_score = conv.max()
                            best_a = a.copy()
                        pass_improved = True
                        no_improve = 0
                        break  # Move done, next i
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    TargetedCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def smooth_optimize_v2(a0, time_limit, verbose=True):
    """Improved smooth optimization with backtracking line search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.array([20.0, 10.0, 5.0, 3.0, 1.5, 0.8, 0.4, 0.2, 0.1, 0.05, 0.03, 0.015, 0.008])

    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        n_steps = max(50, min(1000, int(remaining * 2)))

        y = a.copy()
        step = norm / (n * 0.1)  # initial step size
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            c = fast_autocorr(y)
            c_max = c.max()
            # Protect against overflow
            exp_c = np.exp((c - c_max) / T)
            w = exp_c / exp_c.sum()

            grad = compute_smooth_grad(y, w)
            
            # Backtracking line search
            best_f = c_max + T * np.log(exp_c.sum())  # approximate f(y)
            shrunk_step = step
            found = False
            for ls in range(10):
                a_new = y - shrunk_step * grad
                a_new = project_simplex(a_new, norm)
                
                c_new = fast_autocorr(a_new)
                c_new_max = c_new.max()
                exp_new = np.exp((c_new - c_new_max) / T)
                f_new = c_new_max + T * np.log(exp_new.sum())
                
                if f_new < best_f - 1e-12:
                    found = True
                    break
                shrunk_step *= 0.5
            
            if found:
                y = a_new
            else:
                y = project_simplex(y - step * 0.01 * grad, norm)
            
            step = shrunk_step * 1.5  # grow step back (capped)
            step = min(step, norm / (n * 0.05))

            score = fast_autocorr(y).max()
            if score < best_score:
                best_score = score
                best_a = y.copy()

            if verbose and it % 100 == 0:
                elapsed = time.monotonic() - t0
                print(f"    SmoothV2 T={T:.3f} step={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    return f_interp * np.sqrt(2.0 * new_n) / ss

def multi_start_search(a0, time_limit, rng, solver, verbose=True):
    """Try multiple perturbed starting points."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    n_starts = 5
    time_per_start = time_limit / n_starts
    
    for start_idx in range(n_starts):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Create perturbed version
        noise_scale = 0.1 + 0.2 * start_idx / max(1, n_starts - 1)
        noise = rng.normal(0, noise_scale * a.std(), n)
        a_pert = a + noise
        a_pert = np.maximum(a_pert, 0)
        s_pert = a_pert.sum()
        if s_pert < 1e-15:
            a_pert = a.copy()
        else:
            a_pert = a_pert * norm / s_pert
        
        remaining = time_limit - (time.monotonic() - t0)
        t_sub = min(time_per_start, remaining)
        if t_sub < 10:
            break
        
        # Quick smooth optimization
        a_out, _ = smooth_optimize_v2(a_pert, t_sub * 0.6, verbose=False)
        sc = fast_autocorr(a_out).max()
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()
            if verbose:
                print(f"    MultiStart {start_idx}: new best={best_score:.8f}")
        
        # Quick coordinate descent
        remaining = time_limit - (time.monotonic() - t0)
        if remaining > 10:
            a_out2, _ = coordinate_descent_targeted(a_out, remaining * 0.4, rng, verbose=False)
            sc2 = fast_autocorr(a_out2).max()
            if sc2 < best_score:
                best_score = sc2
                best_a = a_out2.copy()
                if verbose:
                    print(f"    MultiStart {start_idx} CD: new best={best_score:.8f}")
    
    return best_a, best_score
