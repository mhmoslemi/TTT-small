def compute_grad_multi_lag(a, lags, weights):
    """Compute weighted sum of gradients for given lags.
    grad[k][j] = 2*a[k-j] if 0 <= k-j < n."""
    n = len(a)

def lp_best_response(a_fixed, M, active_set, solver):
    n = len(a_fixed)
    k_indices = np.asarray(active_set)
    size = len(k_indices)
    mat = np.zeros((size, n))
    for idx, k in enumerate(k_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            mat[idx, j] = a_fixed[k - j]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

            best_M = M_new
            best_score = 2.0 * n * M_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            a = a_new
            stall += 1
            if stall > 100:
                break

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={t} ({elapsed:.0f}s): best_score={best_score:.10f}")

    return best_a, best_score


def coordinate_descent_fast(a0, time_limit, verbose=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    best_score = 2.0 * n * best_M
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > 3:
            break
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            aj = a[j]
            s_others = a.sum() - aj
            best_x = aj
            best_obj = 2.0 * n * fast_autocorr(a).max()

            for x in [0.0, aj * 0.3, aj * 0.7, aj * 1.3, aj * 1.7, min(s_others * 0.3, 0.5)]:
                x = max(0.0, x)
                new_sum = s_others + x
                if new_sum < 0.01:
                    continue
                a[j] = x
                a_new = a / a.sum()
                obj = 2.0 * n * fast_autocorr(a_new).max()
                if obj < best_obj:
                    best_obj = obj
                    best_x = x
            a[j] = best_x
            a = a / a.sum()
            M_now = fast_autocorr(a).max()
            if M_now < best_M - 1e-14:
                best_M = M_now
                best_score = 2.0 * n * best_M
                best_a = a.copy()
                improved = True

        if improved:
            no_improve = 0
        else:
            no_improve += 1

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    CD ({elapsed:.0f}s): best_score={best_score:.10f}")

    return best_a, best_score


def propose_candidate():
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = 1750.0

    solver = get_solver()
    print(f"Solver: {solver}, Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x)
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 1: Mirror descent on n=1600 (main workhorse)
    n0 = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 450.0)
    if t1 > 30:
        print(f"\n=== Phase 1: Mirror Descent n={n0}, budget={t1:.0f}s ===")
        a_out, _ = mirror_descent_opt(best_seq, t1, n_lags=10, temp_factor=0.05, verbose=True, seed=0)
        try_update(a_out, f"md1_n{n0}")

    # Phase 2: Projected subgradient with different hyperparams
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 20:
        print(f"\n=== Phase 2: Projected Subgrad n={n_cur}, budget={t2:.0f}s ===")
        a_out, _ = projected_subgrad_opt(best_seq, t2, n_lags=15, temp_factor=0.03, verbose=True, seed=1)
        try_update(a_out, f"psg1_n{n_cur}")

    # Phase 3: LP refinement
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 30:
        print(f"\n=== Phase 3: LP Refine n={n_cur}, budget={t3:.0f}s ===")
        a_out, _ = lp_refine(best_seq, t3, solver, verbose=True, seed=0)
        try_update(a_out, f"lp1_n{n_cur}")

    # Phase 4: Mirror descent with different params (more lags, lower temp)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        print(f"\n=== Phase 4: Mirror Descent (more lags) n={n_cur}, budget={t4:.0f}s ===")
        a_out, _ = mirror_descent_opt(best_seq, t4, n_lags=20, temp_factor=0.02, verbose=True, seed=7)
        try_update(a_out, f"md2_n{n_cur}")

    # Phase 5: Try different n (2000)
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.12, 180.0)
    if t5 > 30:
        n_new = 2000
        print(f"\n=== Phase 5: Mirror Descent n={n_new}, budget={t5:.0f}s ===")
        a_init = interpolate_seq(best_seq, n_new)
        a_out, _ = mirror_descent_opt(a_init, t5, n_lags=12, temp_factor=0.04, verbose=True, seed=3)
        try_update(a_out, f"md_n{n_new}")

    # Phase 6: LP refinement again
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t6 = min(remaining * 0.12, 150.0)
    if t6 > 30:
        print(f"\n=== Phase 6: LP Refine round 2 n={n_cur}, budget={t6:.0f}s ===")
        a_out, _ = lp_refine(best_seq, t6, solver, verbose=True, seed=100)
        try_update(a_out, f"lp2_n{n_cur}")

    # Phase 7: Coordinate descent for final polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t7 = min(remaining * 0.10, 120.0)
    if t7 > 15:
        print(f"\n=== Phase 7: Coordinate Descent n={n_cur}, budget={t7:.0f}s ===")
        a_out, _ = coordinate_descent_fast(best_seq, t7, verbose=True)
        try_update(a_out, f"cd_n{n_cur}")

    # Phase 8: Final mirror descent with very fine tuning
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t8 = min(remaining * 0.11, 100.0)
    if t8 > 10:
        print(f"\n=== Phase 8: Fine MD n={n_cur}, budget={t8:.0f}s ===")
        a_out, _ = mirror_descent_opt(best_seq, t8, n_lags=30, temp_factor=0.01, verbose=True, seed=99)
        try_update(a_out, f"md_final_n{n_cur}")

    # Phase 9: Perturbation + quick MD
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 30:
        for scale in [0.005, 0.01, 0.02]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 15:
                break
            t9 = min(remaining * 0.4, min(remaining - 5, 30.0))
            if t9 < 5:
                break
            print(f"\n=== Phase 9: Perturb scale={scale}, budget={t9:.0f}s ===")
            pert = best_seq * (1.0 + scale * rng.standard_normal(n_cur))
            pert = np.maximum(pert, 0.0)
            pert = pert / pert.sum()
            a_out, _ = mirror_descent_opt(pert, t9, n_lags=10, temp_factor=0.03, verbose=False, seed=200 + int(scale * 1000))
            try_update(a_out, f"pert{scale}_n{n_cur}")

    # Final
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
