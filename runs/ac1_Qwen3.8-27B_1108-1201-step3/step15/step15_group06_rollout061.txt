def fast_autocorr(a):

    return irfft(A * A, fft_size)[:size]

def build_T(a, indices):
    """Build the constraint matrix T for conv(y, b)_k = sum_j a[k-j] * b[j]."""
    n = len(a)
    m = len(indices)
    T = np.zeros((m, n))
    for row, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                T[row, j] = a[idx]
    return T

def get_best_solver():

def solve_minmax_lp(a, active_idx, sum_target, solver):
    """Solve min t s.t. conv(a, b)_k <= t for k in active_idx, sum(b) = sum_target, b >= 0."""
    n = len(a)
    T = build_T(a, active_idx)
    b = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    cons = [T @ b <= t, cp.sum(b) == sum_target]
    prob = cp.Problem(cp.Minimize(t), cons)
    prob.solve(solver=solver, verbose=False)
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if val.sum() < 1e-15:
        return None, None
    return val, float(t.value) if t.value is not None else float('inf')

def solve_maxsum_lp(a, M, active_idx, solver):
    """Solve max sum(b) s.t. conv(a, b)_k <= M for k in active_idx, b >= 0."""
    n = len(a)
    T = build_T(a, active_idx)
    b = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(b)), [T @ b <= M])
    prob.solve(solver=solver, verbose=False)
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if val.sum() < 1e-15:
        return None
    return val

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss

def perturb_seq(a, rng, scale=0.05):
    """Perturb a sequence with multiplicative noise."""
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.01)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s

def structured_init(n, rng, style=0):
    """Generate structured initial sequences."""
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        # Cosine squared - smooth, peaked in middle
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif style == 1:
        # Sinusoidal with modulation
        a = (1.0 + 0.3 * np.cos(4.0 * np.pi * x)) * np.sin(np.pi * x) ** 0.8
        a = np.maximum(a, 0.01)
    elif style == 2:
        # Exponential bump
        a = np.exp(-4.0 * (x - 0.5) ** 2) + 0.3 * np.exp(-16.0 * (x - 0.3) ** 2) + 0.3 * np.exp(-16.0 * (x - 0.7) ** 2)
    elif style == 3:
        # Flat with slight modulation
        a = 1.0 + 0.15 * np.cos(6.0 * np.pi * x) + 0.1 * np.cos(10.0 * np.pi * x)
        a = np.maximum(a, 0.1)
    elif style == 4:
        # Super-Gaussian
        a = np.exp(-((x - 0.5) * 4) ** 4)
    elif style == 5:
        # Two peaks
        a = np.exp(-8.0 * (x - 0.25) ** 2) + np.exp(-8.0 * (x - 0.75) ** 2) + 0.2
    elif style == 6:
        # Near-uniform with boundary decay
        a = np.sin(np.pi * x) ** 0.3
    elif style == 7:
        # Random smooth
        freqs = rng.integers(1, 8, size=5)
        phases = rng.uniform(0, 2*np.pi, size=5)
        amps = rng.uniform(0.1, 0.5, size=5)
        a = 1.0
        for f, p, amp in zip(freqs, phases, amps):
            a += amp * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.05)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()

def fast_autocorr(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

    n = len(a)

        hi = min(k, n - 1)
        if lo <= hi:

            mu = min((t - 1.0) / (t + 2.0), 0.94)
            a_new = y_new + mu * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a

        y = y_new

        # Track best
        sc_y = fast_autocorr(y).max()
        if sc_y < best_score - 1e-14:
            best_score = sc_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        sc_a = fast_autocorr(a_new).max()
        if sc_a < best_score - 1e-14:
            best_score = sc_a
            best_a = a_new.copy()
            stall_count = 0

        # Restart if stalled
        if stall_count > 50:
            y = best_a.copy()
            stall_count = 0

        if verbose and t % 50 == 0:
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.12f}")

    return best_a, evaluate_local(best_a)


def gradient_coord_descent(a0, time_limit, rng, verbose=True):
    """
    Gradient-guided coordinate descent focusing on peak-contributing coordinates.
    Uses the subgradient of max_k (conv(a,a))_k to determine update direction.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    size = 2 * n - 1

    for iteration in range(2000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(a)
        M = conv.max()

        # Get top peak indices
        n_peaks = min(8, size)
        peak_indices = np.argsort(conv)[::-1][:n_peaks]

        # Compute subgradient: average of gradients at active peaks
        grad = np.zeros(n)
        for k in peak_indices:
            # d(conv[k])/d(a_i) = 2*a[k-i] if 0 <= k-i < n
            ki = k - np.arange(n)
            mask = (ki >= 0) & (ki < n)
            grad[mask] += 2.0 * a[ki[mask]]
        grad /= len(peak_indices)

        # Try moving in -grad direction (reduce peak)
        for i in rng.permutation(n):
            if time.monotonic() - t0 > time_limit:
                break

            ai = a[i]
            if ai < 1e-12:
                continue

            # Determine direction: if grad[i] > 0, decrease; if < 0, increase
            if grad[i] > 0:
                fracs = [0.01, 0.03, 0.05, 0.1, 0.15]
            else:
                fracs = [0.01, 0.03, 0.05, 0.1]

            improved = False
            for frac in fracs:
                if grad[i] > 0:
                    new_val = ai * (1.0 - frac)
                else:
                    new_val = ai + frac * M * 0.1

                if new_val < 0:
                    continue

                a_try = a.copy()
                a_try[i] = new_val
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                cv_try = fast_autocorr(a_try)
                if cv_try.max() < M - 1e-14:
                    a = a_try
                    conv = cv_try
                    improved = True
                    no_improve = 0
                    break

            if not improved:
                no_improve += 1
                if no_improve >= 15:
                    break

        if no_improve > 30:
            break

        sc = fast_autocorr(a).max()
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
            no_improve = 0

        if verbose and iteration % 50 == 0:
            print(f"      GCD iter={iteration} ({time.monotonic()-t0:.0f}s): best={best_score:.12f}")

    return best_a, evaluate_local(best_a)


def pair_descent(a0, time_limit, rng, verbose=True):
    """Pair swap coordinate descent for final polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    for iteration in range(500):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(a)
        M = conv.max()
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(0, n - 1, 2):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            j = perm[idx + 1]
            ai, aj = a[i], a[j]
            if ai < 1e-12 and aj < 1e-12:
                continue

            best_val = M
            best_frac = 0.0
            best_dir = 0

            # Try moving mass i -> j
            max_d = ai * 0.9
            for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                d = max_d * frac
                if d <= 0:
                    continue
                a_try = a.copy()
                a_try[i] = ai - d
                a_try[j] = aj + d
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                cv_try = fast_autocorr(a_try)
                if cv_try.max() < best_val - 1e-14:
                    best_val = cv_try.max()
                    best_frac = d
                    best_dir = 1

            # Try moving mass j -> i
            max_d = aj * 0.9
            for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                d = max_d * frac
                if d <= 0:
                    continue
                a_try = a.copy()
                a_try[i] = ai + d
                a_try[j] = aj - d
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                cv_try = fast_autocorr(a_try)
                if cv_try.max() < best_val - 1e-14:
                    best_val = cv_try.max()
                    best_frac = d
                    best_dir = -1

            if best_dir != 0:
                a_try = a.copy()
                if best_dir == 1:
                    a_try[i] -= best_frac
                    a_try[j] += best_frac
                else:
                    a_try[i] += best_frac
                    a_try[j] -= best_frac
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try / s_try
                    a = a_try
                    conv = cv_try
                    pass_improved = True
                    no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break

        sc = conv.max()
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()

        if verbose and iteration % 20 == 0:
            print(f"      PD iter={iteration} ({time.monotonic()-t0:.0f}s): best={best_score:.12f}")

    return best_a, evaluate_local(best_a)


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def perturb_seq(a, rng, scale=0.03):
    """Perturb a sequence with multiplicative noise."""
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.05)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s


def structured_init(n, rng, style=0):
    """Generate structured initial sequences."""
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif style == 1:
        a = (1.0 + 0.3 * np.cos(4.0 * np.pi * x)) * np.sin(np.pi * x) ** 0.8
        a = np.maximum(a, 0.01)
    elif style == 2:
        a = np.exp(-4.0 * (x - 0.5) ** 2) + 0.3 * np.exp(-16.0 * (x - 0.3) ** 2) + 0.3 * np.exp(-16.0 * (x - 0.7) ** 2)
    elif style == 3:
        a = 1.0 + 0.15 * np.cos(6.0 * np.pi * x) + 0.1 * np.cos(10.0 * np.pi * x)
        a = np.maximum(a, 0.1)
    elif style == 4:
        a = np.exp(-((x - 0.5) * 5) ** 4)
    elif style == 5:
        a = np.exp(-8.0 * (x - 0.25) ** 2) + np.exp(-8.0 * (x - 0.75) ** 2) + 0.2
    elif style == 6:
        a = np.sin(np.pi * x) ** 0.4
    elif style == 7:
        freqs = [2, 3, 5, 7, 11]
        phases = [0.0, 0.5, 1.0, 1.5, 2.0]
        amps = [0.2, 0.15, 0.1, 0.08, 0.05]
        a = 1.0
        for f, p, amp in zip(freqs, phases, amps):
            a += amp * np.cos(2.0 * np.pi * f * x + p)
        a = np.maximum(a, 0.05)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
