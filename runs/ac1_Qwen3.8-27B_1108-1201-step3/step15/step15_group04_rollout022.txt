import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print(f"  Using solver (fallback): ECOS")
    return _solver

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T

def lp_solve(a, M, solver, active=None):
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    if active is not None and 0 < len(active) < size:
        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    duals = None
    try:
        con = constraints[0]
        if con.dual_value is not None:
            if active is not None and 0 < len(active) < size:
                full_duals = np.zeros(size)
                full_duals[active] = np.asarray(con.dual_value, dtype=np.float64)
                duals = full_duals
            else:
                duals = np.asarray(con.dual_value, dtype=np.float64)
    except Exception:
        pass
    return val, duals

def lse_get_gradient(a, T_val):
    """Compute LSE gradient of the smooth max objective."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    conv = fast_conv_self(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n), conv
    shifted = (conv - max_c) / T_val
    shifted = np.maximum(shifted, -200.0)
    exp_c = np.exp(shifted)
    w = exp_c / exp_c.sum()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    g1 = irfft(np.conj(A) * W, fft_size)[:n]
    aw = irfft(A * W, fft_size)
    g2 = aw[:n]
    grad = 2.0 * (g1 + g2)
    return grad, conv

def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 14
    temps = np.geomspace(80.0, 0.05, n_stages)
    v = np.zeros(n)  # Nesterov velocity

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(20, min(400, int(remaining * 5)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            grad, conv = lse_get_gradient(a, T_val)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            # Nesterov momentum
            alpha_t = 2.0 / (it + 2.0)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            base_step = norm / (T_val * np.sqrt(n)) * 0.5
            step = base_step / (1.0 + 0.02 * it)

            # Line search
            for trial in range(12):
                v = alpha_t * v + (1.0 - alpha_t) * grad
                a_new = a - step * v
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    step *= 0.5
                    continue
                a_new = a_new * norm / s_new
                sc = fast_conv_self(a_new).max()
                if sc <= max_c + 0.1 * T_val:
                    break
                step *= 0.5

            v = alpha_t * v + (1.0 - alpha_t) * grad
            a_new = a - step * v
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.1f}: best={best_score:.8f}")

    return best_a, best_score

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, use_multiplicative=False, perturb_interval=80):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    y_prev = a.copy()
    size = 2 * n - 1
    no_improve_count = 0
    best_threshold = 0.6

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Adaptive constraint screening with gradual tightening
        if t < 20:
            thresh = 0.4
        elif t < 100:
            thresh = 0.55
        elif t < 500:
            thresh = 0.7
        else:
            thresh = min(0.85, 0.6 + 0.0002 * t)

        active = np.where(conv_y > M * thresh)[0]
        if len(active) < 20:
            active = np.where(conv_y > M * 0.3)[0]
        if len(active) > 2000:
            top_active = np.argsort(conv_y)[-1500:]
            active = top_active

        b, duals = lp_solve(y, M, solver, active)
        if b is None or b.sum() < 1e-15:
            b, duals = lp_solve(y, M, solver, None)
        if b is None or b.sum() < 1e-15:
            continue

        b_sum = b.sum()
        g = b * norm / b_sum

        # Verify dropped constraints
        if len(active) < size:
            check_conv = fast_conv_cross(y, g)
            violated = np.where(check_conv > M * (1 + 5e-4))[0]
            if len(violated) > 0:
                b_full, _ = lp_solve(y, M, solver, None)
                if b_full is not None and b_full.sum() > 1e-15:
                    b = b_full
                    g = b * norm / b.sum()

        # Nesterov / momentum update
        alpha = 2.0 / (t + 2.0)
        if use_multiplicative and t > 30:
            ratio = g / np.maximum(y, 1e-30)
            log_ratio = np.log(np.clip(ratio, 1e-10, 1e10))
            y_new = y * np.exp(alpha * log_ratio)
        else:
            y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min(1.0, (t - 1.0) / (t + 3.0))
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y.copy()
        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        improved = False
        if sc_y < best_score - 1e-10:
            best_score = sc_y
            best_a = y.copy()
            improved = True

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score - 1e-10:
            best_score = sc_a
            best_a = a.copy()
            improved = True

        if not improved:
            no_improve_count += 1
        else:
            no_improve_count = 0

        if no_improve_count == perturb_interval:
            n_perturb = max(20, n // 15)
            rng_p = np.random.default_rng(t * 7919)
            idx_perturb = rng_p.choice(n, size=n_perturb, replace=False)
            y[idx_perturb] *= (1.0 + 0.25 * rng_p.standard_normal(n_perturb))
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y * norm / s_y
            no_improve_count = 0

        if verbose and t % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def dual_gradient_refine(a0, time_limit, solver, verbose=True):
    n = len(a0)

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

        return a, fast_conv_self(a).max() if len(a) > 0 else float('inf')
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()

    for it in range(300):

        M = conv.max()
        if M < 1e-12:
            break

            else:

        if verbose and it % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DualGrad it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def smooth_and_reoptimize(a0, time_limit, solver, verbose=True):
    """Apply Gaussian smoothing then re-optimize."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv_self(a).max()
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()

    # Try different smoothing widths
    for sigma in [1.0, 1.5, 2.0, 3.0]:
        if time.monotonic() - t0 > time_limit:
            break
        # Gaussian smoothing via FFT
        fft_size = 1 << (n - 1).bit_length()
        x = np.arange(-fft_size // 2, fft_size // 2)
        gauss = np.exp(-x**2 / (2.0 * sigma**2))
        G = rfft(gauss, fft_size)
        A = rfft(a, fft_size)
        smoothed = irfft(A * G, fft_size)
        smoothed = np.maximum(smoothed, 0.0)
        s_sm = smoothed.sum()
        if s_sm < 1e-15:
            continue
        smoothed = smoothed * norm / s_sm
        sc = fast_conv_self(smoothed).max()
        if sc < best_score:
            best_score = sc
            best_a = smoothed.copy()

    # Re-optimize from smoothed point
    if best_score < fast_conv_self(a).max():
        remaining = time_limit - (time.monotonic() - t0)
        if remaining > 30:
            a_out, sc_out = nesterov_lp_optimize(best_a, remaining * 0.7, solver, verbose=False)
            if sc_out < best_score:
                best_a, best_score = a_out, sc_out

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    Smooth+Reopt ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            j_candidates = set()
            for _ in range(8):
                j_candidates.add(int(rng.integers(n)))
            peak_threshold = cur_max * 0.85
            peak_lags = np.where(conv >= peak_threshold)[0]
            for kl in peak_lags[:6]:
                j_candidates.add(int(kl % n))
                j_candidates.add(int((kl + i) % n))
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in list(j_candidates)[:12]:
                if j == i:
                    continue
                aj = a[j]
                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                n_samples = 15
                deltas = np.linspace(0.02, 0.9, n_samples) * ai
                best_d = 0.0
                best_v = cur_max
                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_d = d

                if best_d > 0 and best_v < best_val - 1e-12:
                    best_val = best_v
                    best_delta = best_d
                    best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 6:
                break
            if no_improve == 3:
                n_perturb = max(10, n // 25)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.12 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and total_improvements % 100 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD improvements={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def coordinated_prune(a0, time_limit, rng, verbose=True):
    """Iteratively try setting pairs of elements to zero or merging them."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    # Find elements that contribute to the peak convolution
    for round_num in range(50):
        if time.monotonic() - t0 > time_limit:
            break
        conv = fast_conv_self(a)
        cur_max = conv.max()
        peak_pos = np.argmax(conv)

        # Elements contributing to the peak at position peak_pos
        # conv[peak_pos] = sum_i a[i]*a[peak_pos-i]
        contrib = np.zeros(n)
        for i in range(n):
            j = peak_pos - i
            if 0 <= j < n:
                contrib[i] += a[j]
                contrib[j] += a[i]

        # Try reducing the top contributors
        top_contrib = np.argsort(contrib)[-10:][::-1]
        improved = False
        for i in top_contrib:
            if a[i] < 1e-10:
                continue
            # Try moving mass from i to a random other position
            for _ in range(5):
                j = int(rng.integers(n))
                if j == i:
                    continue
                for frac in [0.1, 0.25, 0.4, 0.6, 0.8]:
                    d = frac * a[i]
                    a[i] -= d
                    a[j] += d
                    a = np.maximum(a, 0.0)
                    s_new = a.sum()
                    if s_new < 1e-15:
                        break
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < cur_max - 1e-12:
                        cur_max = new_max
                        improved = True
                        break
                    else:
                        a[i] += d
                        a[j] -= d
                        a = np.maximum(a, 0.0)
                        a = a * norm / a.sum()
                        break

            if cur_max < best_score - 1e-12:
                best_score = cur_max
                best_a = a.copy()

        if not improved and round_num > 5:
            break

    return best_a, best_score

def compute_score(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    n = len(a)
    max_b = fast_conv_self(a).max()
    return 2.0 * n * max_b / (S * S)

def eval_fn(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def structured_init(n, rng):
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.15 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0.1)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-6.0 * (x - 0.75) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-2.0 * x) + 0.5 * np.exp(-2.0 * (1 - x))
    elif choice == 7:
        a = 0.5 * np.exp(-4.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-4.0 * (x - 0.7) ** 2)
        a = a * (1.0 + 0.3 * np.cos(20 * np.pi * x))
        a = np.maximum(a, 0.01)
    elif choice == 8:
        a = np.abs(np.sin(3 * np.pi * x)) ** 0.8
    elif choice == 9:
        a = 0.4 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(11 * np.pi * x)
        a = np.maximum(a, 0.05)
    elif choice == 10:
        # Asymmetric ramp with oscillation
        a = x ** 0.5 * (1.0 + 0.2 * np.cos(30 * np.pi * x))
        a = np.maximum(a, 0.01)
    else:
        # Multi-scale wavelet-like
        a = np.ones(n)
        for k in range(1, 5):
            a += (0.5 ** k) * np.cos(2 * np.pi * 2**k * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.05)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def coarse_fine_optimize(a0, time_limit, solver, verbose=True):
    """Coarse-to-fine: downsample, optimize, upsample, optimize."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv_self(a).max()
    a = a / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()

    # Downsample by factor of 2
    n_coarse = n // 2
    if n_coarse >= 100:
        a_coarse = np.zeros(n_coarse)
        for i in range(n_coarse):
            a_coarse[i] = a[2*i] + a[2*i+1]
        s_c = a_coarse.sum()
        if s_c > 1e-15:
            a_coarse = a_coarse / s_c
            # Optimize at coarse level
            t_coarse = min((time_limit - (time.monotonic() - t0)) * 0.3, 100.0)
            a_c_out, sc_c = nesterov_lp_optimize(a_coarse, t_coarse, solver, verbose=False)
            if sc_c < fast_conv_self(a_coarse).max():
                # Upsample
                a_up = np.zeros(n)
                for i in range(n_coarse):
                    a_up[2*i] = a_c_out[i] * 0.5
                    a_up[2*i+1] = a_c_out[i] * 0.5
                s_u = a_up.sum()
                if s_u > 1e-15:
                    a_up = a_up / s_u
                    sc_up = fast_conv_self(a_up).max()
                    if sc_up < best_score:
                        # Now refine at fine level
                        remaining = time_limit - (time.monotonic() - t0)
                        if remaining > 30:
                            a_fine, sc_fine = nesterov_lp_optimize(a_up, remaining * 0.5, solver, verbose=False)
                            if sc_fine < best_score:
                                best_score = sc_fine
                                best_a = a_fine.copy()
                    else:
                        # Try interpolating instead
                        a_interp = interpolate_seq(a_c_out, n)
                        sc_int = fast_conv_self(a_interp).max()
                        if sc_int < best_score:
                            remaining = time_limit - (time.monotonic() - t0)
                            if remaining > 30:
                                a_fin2, sc_fin2 = nesterov_lp_optimize(a_interp, remaining * 0.5, solver, verbose=False)
                                if sc_fin2 < best_score:
                                    best_score = sc_fin2
                                    best_a = a_fin2.copy()

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    CoarseFine ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_init = None
        best_init_score = float('inf')
        for trial in range(8):
            cand = structured_init(n_base, rng)
            sc = eval_fn(cand)
            if sc < best_init_score:
                best_init_score = sc
                best_init = cand
        best_seq = best_init
        best_score = best_init_score
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 0: Quick LSE to improve init
    remaining = time_budget - (time.monotonic() - t_global)
    t_init = min(remaining * 0.06, 100.0)
    if t_init > 15:
        a_out, _ = lse_gradient_descent(best_seq, t_init, verbose=False)
        try_update(a_out, "lse_init")

    # Phase 1: LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = len(best_seq)
    t1 = min(remaining * 0.30, 500.0)
    print(f"\n=== Phase 1: LP Nesterov n={n1} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t1, solver, verbose=True, use_multiplicative=False)
        try_update(a_out, f"lp1_n{n1}")

    # Phase 2: LSE refinement from LP result
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq)
    t2 = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 2: LSE Refine n={n2} (budget={t2:.0f}s) ===")
    if t2 > 15:
        a_out, _ = lse_gradient_descent(best_seq, t2, verbose=True)
        try_update(a_out, f"lse_n{n2}")

    # Phase 3: LP Nesterov round 2 (multiplicative)
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = len(best_seq)
    t3 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 3: LP Nesterov mult n={n3} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t3, solver, verbose=True, use_multiplicative=True)
        try_update(a_out, f"lp2_n{n3}")

    # Phase 4: Smoothing + re-optimization
    remaining = time_budget - (time.monotonic() - t_global)
    n4 = len(best_seq)
    t4 = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 4: Smooth+Reopt n={n4} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = smooth_and_reoptimize(best_seq, t4, solver, verbose=True)
        try_update(a_out, f"smooth_n{n4}")

    # Phase 5: Coarse-fine optimization
    remaining = time_budget - (time.monotonic() - t_global)
    n5 = len(best_seq)
    t5 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 5: CoarseFine n={n5} (budget={t5:.0f}s) ===")
    if t5 > 30:
        a_out, _ = coarse_fine_optimize(best_seq, t5, solver, verbose=True)
        try_update(a_out, f"coarse_n{n5}")

    # Phase 6: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    n6 = len(best_seq)
    t6 = min(remaining * 0.18, 250.0)
    print(f"\n=== Phase 6: Pair CD n={n6} (budget={t6:.0f}s) ===")
    if t6 > 20:
        a_out, _ = pair_cd_optimize(best_seq, t6, rng, verbose=True)
        try_update(a_out, f"pcd_n{n6}")

    # Phase 7: Final LP polish
    remaining = time_budget - (time.monotonic() - t_global)
    n7 = len(best_seq)
    t7 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 7: Final LP n={n7} (budget={t7:.0f}s) ===")
    if t7 > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t7, solver, verbose=True, use_multiplicative=False, perturb_interval=50)
        try_update(a_out, f"lp_final_n{n7}")

    # Phase 8: Dual gradient + final CD
    remaining = time_budget - (time.monotonic() - t_global)
    n8 = len(best_seq)
    print(f"\n=== Phase 8: Final Dual+CD n={n8} (remaining={remaining:.0f}s) ===")
    if remaining > 40:
        t_d = min(remaining * 0.4, 80.0)
        a_out, _ = dual_gradient_refine(best_seq, t_d, solver, verbose=False)
        try_update(a_out, "dual_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t_cd = min(remaining * 0.7, 80.0)
            a_out, _ = pair_cd_optimize(best_seq, t_cd, rng, verbose=False)
            try_update(a_out, "pcd_final2")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
