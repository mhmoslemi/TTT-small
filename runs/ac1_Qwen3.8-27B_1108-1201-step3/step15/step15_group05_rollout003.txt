from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the objective value directly."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    max_conv = float(np.max(fast_autocorr(a)))
    return 2.0 * n * max_conv / (s * s)


def evaluate_safe(a):
    """Safely evaluate using the actual evaluate_sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def projected_gradient_uniform(a, n_steps=5000, time_limit=30.0):
    """
    For non-negative sequences with the score 2n*sum(a^2)/(sum(a))^2,
    the gradient pushes toward uniformity. We use a simple iterative
    averaging that provably converges to the uniform vector.
    """
    t0 = time.monotonic()
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s  # Normalize to sum = 1

    best_a = a.copy()
    best_sc = compute_score(a)

    # Step size schedule: start larger, decay
    step = 0.1
    for it in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        # Move toward uniform: a_new = (1-step)*a + step/n
        # This is the gradient step for minimizing sum(a^2) subject to sum(a)=1, a>=0
        a = (1.0 - step) * a + step / n
        a = np.maximum(a, 0.0)
        s_new = a.sum()
        if s_new < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s_new

        sc = compute_score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
        elif sc > best_sc + 1e-12:
            step = max(step * 0.9, 1e-6)
        else:
            step = max(step * 0.95, 1e-6)

        # Converged if close to uniform
        if np.max(np.abs(a - 1.0 / n)) < 1e-12:
            break

    return best_a


def random_restart_optimize(n, time_limit, seed):
    """Try random initializations and simple optimization."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    best_a = None
    best_sc = float('inf')

    for trial in range(20):
        if time.monotonic() - t0 > time_limit:
            break

        # Different random initializations
        if trial == 0:
            a = np.ones(n) / n
        elif trial == 1:
            a = rng.dirichlet(np.ones(n))
        elif trial == 2:
            a = np.abs(rng.standard_normal(n))
            a = a / a.sum()
        elif trial == 3:
            # Low-frequency sinusoidal
            x = np.linspace(0, 1, n)
            a = 1.0 + 0.5 * np.cos(2 * np.pi * x + rng.uniform(0, 2 * np.pi))
            a = np.maximum(a, 0.0)
            a = a / a.sum()
        elif trial == 4:
            # Block structure
            block = n // 4
            a = np.concatenate([np.ones(block), np.zeros(block),
                                np.ones(block), np.zeros(block)])
            if n % 4 != 0:
                a = np.concatenate([a, np.ones(n - 4 * block)])
            a = a / a.sum()
        else:
            # Exponential decay from center
            x = np.linspace(-1, 1, n)
            a = np.exp(-0.5 * (x ** 2) / (0.1 + 0.5 * rng.random()))
            a = a / a.sum()

        # Quick gradient polish
        a = projected_gradient_uniform(a, n_steps=200, time_limit=2.0)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def lp_refine(a_fixed, M, n, solver=None):
    """
    LP: maximize sum(b) subject to (a_fixed * b)[k] <= M for all k, b >= 0.
    Returns optimal b or None.
    """
    try:
        import cvxpy as cp
    except ImportError:
        return None

    k_indices = np.arange(2 * n - 1)
    b_var = cp.Variable(n, nonneg=True)

    # Build constraint matrix efficiently
    # (a*b)[k] = sum_j a[k-j]*b[j] for valid j
    constraints = []
    for k in range(2 * n - 1):
        # Row: for each j, coefficient is a[k-j] if 0 <= k-j < n, else 0
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        row = np.zeros(n)
        if hi > lo:
            row[lo:hi] = a[k - (hi - 1):k - lo][::-1]
        # Actually let me redo this
        # (a*b)[k] = sum_{j=0}^{n-1} a[k-j] * b[j] where a is defined on [0,n-1]
        # So coefficient of b[j] is a[k-j] if 0 <= k-j <= n-1, else 0
        row = np.zeros(n)
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                row[j] = a_fixed[idx]
        constraints.append(row @ b_var <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver or cp.ECOS, verbose=False)
    except Exception:
        return None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp(a0, time_limit, seed=0):
    """Nesterov-accelerated LP best response method."""
    try:
        import cvxpy as cp
    except ImportError:
        return a0, compute_score(a0)

    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()
    a_curr = a.copy()

    # Precompute: for the active set, use top peaks
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Select active set: peaks above threshold
        tol = max(0.02, 0.05 * np.exp(-t / 200.0))
        active = np.where(conv_y >= M_curr * (1.0 - tol))[0]
        # Always include lag 0
        if 0 not in active:
            active = np.concatenate([active, [0]])

        # Solve LP with active constraints only (for speed)
        b_var = None
        try:
            b_var_lp = cp.Variable(n, nonneg=True)
            constraints = []
            for k in active:
                row = np.zeros(n)
                for j in range(n):
                    idx = k - j
                    if 0 <= idx < n:
                        row[j] = y[idx]
                constraints.append(row @ b_var_lp <= M_curr)

            prob = cp.Problem(cp.Maximize(cp.sum(b_var_lp)), constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate') and b_var_lp.value is not None:
                b_val = np.maximum(np.asarray(b_var_lp.value, dtype=np.float64), 0.0)
                if b_val.sum() > 1e-15:
                    # Verify no violation on non-active constraints
                    conv_check = y @ b_val  # This is cross-correlation, not quite right
                    # Actually verify properly
                    full_conv = np.convolve(y, b_val)
                    if full_conv.max() <= M_curr * (1 + 1e-4):
                        b_var = b_val
                    else:
                        # Add violated constraints and re-solve
                        viol = np.where(full_conv > M_curr * (1 + 1e-4))[0]
                        new_active = np.union1d(active, viol)
                        b_var_lp2 = cp.Variable(n, nonneg=True)
                        constraints2 = []
                        for k in new_active:
                            row2 = np.zeros(n)
                            for j in range(n):
                                idx = k - j
                                if 0 <= idx < n:
                                    row2[j] = y[idx]
                            constraints2.append(row2 @ b_var_lp2 <= M_curr)
                        prob2 = cp.Problem(cp.Maximize(cp.sum(b_var_lp2)), constraints2)
                        prob2.solve(solver=cp.ECOS, verbose=False)
                        if prob2.status in ('optimal', 'optimal_inaccurate') and b_var_lp2.value is not None:
                            b_var = np.maximum(np.asarray(b_var_lp2.value, dtype=np.float64), 0.0)
                            if b_var.sum() < 1e-15:
                                b_var = None
                else:
                    b_var = None
        except Exception:
            b_var = None

        if b_var is None:
            # Fallback: use y itself as b (trivial best response)
            b_var = y.copy()

        b_sum = b_var.sum()
        if b_sum < 1e-15:
            g = np.ones(n) / n * norm
        else:
            g = b_var * norm / b_sum

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_curr = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.95)
            a_curr = y_new + momentum * (y_new - y_prev)
            a_curr = np.maximum(a_curr, 0.0)
            s_a = a_curr.sum()
            if s_a < 1e-15:
                a_curr = y_new.copy()
            else:
                a_curr = a_curr * norm / s_a

        y_prev = y
        y = y_new

        # Track best
        for candidate in [y, a_curr]:
            sc = compute_score(candidate)
            if sc < best_score:
                best_score = sc
                best_a = candidate.copy()

        if t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def propose_candidate():
    """
    Main search function. Key insight: for non-negative sequences, the score
    2n*max(conv)/(sum)^2 >= 2 by Cauchy-Schwarz, with equality for uniform sequences.
    We verify this and run optimization as a safety net.
    """
    t_global = time.monotonic()
    time_budget = 1740.0  # Leave 60s margin

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        a = np.minimum(a, 1000.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)}, t={elapsed:.1f}s)")

    # === Phase 1: Uniform sequences (provably optimal for non-negative inputs) ===
    print("Phase 1: Testing uniform sequences (theoretical optimum = 2.0)")
    for n in [100, 200, 500, 1000, 1600, 2000, 4000, 8000, 10000, 20000]:
        a = np.ones(n) / n
        try_update(a, f"uniform_n{n}")

    # Also try uniform with raw values (no normalization change since score is homogeneous)
    for n in [1600, 2000]:
        a = np.ones(n) * 1000.0  # Max allowed value
        try_update(a, f"uniform_max_n{n}")

    # === Phase 2: Try parent sequence ===
    print(f"\nPhase 2: Checking parent (score so far: {best_score:.10f})")
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            try_update(p, "parent_raw")
            p_norm = p / p.sum()
            try_update(p_norm, "parent_norm")
            p_ones = p / p.max() if p.max() > 0 else p
            try_update(p_ones, "parent_scaled")
    except Exception as e:
        print(f"  Parent not available: {e}")

    # === Phase 3: Try structured sequences ===
    print(f"\nPhase 3: Structured sequences (score so far: {best_score:.10f})")
    for n in [800, 1600, 3200]:
        x = np.linspace(0, 1, n)
        # Various smooth non-negative patterns
        patterns = {
            "cos": 1.0 + 0.3 * np.cos(2 * np.pi * x),
            "cos2": 1.0 + 0.2 * np.cos(4 * np.pi * x),
            "bell": np.exp(-2.0 * (x - 0.5) ** 2),
            "tri": 1.0 - np.abs(2 * x - 1),
            "bell2": 0.5 + 0.5 * np.exp(-4.0 * (x - 0.5) ** 2),
        }
        for name, pat in patterns.items():
            a = np.maximum(pat, 0.0)
            s = a.sum()
            if s > 1e-10:
                try_update(a / s, f"{name}_n{n}")

    # === Phase 4: Gradient optimization from best ===
    print(f"\nPhase 4: Gradient optimization (score so far: {best_score:.10f})")
    remaining = time_budget - (time.monotonic() - t_global)
    if best_seq is not None and remaining > 100:
        n_cur = len(best_seq)
        a_start = best_seq / best_seq.sum()
        t4 = min(remaining * 0.2, 200.0)

        a_out = projected_gradient_uniform(a_start, n_steps=20000, time_limit=t4)
        try_update(a_out, "grad_polish")

    # === Phase 5: LP-based optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    if best_seq is not None and remaining > 150:
        print(f"\nPhase 5: LP optimization (remaining={remaining:.0f}s)")
        n_cur = len(best_seq)
        # Use a moderate length for LP (too large is too slow)
        n_lp = min(n_cur, 2000)

        # Interpolate to n_lp if needed
        a_lp = best_seq.copy()
        if n_cur != n_lp:
            old_x = np.linspace(0, 1, n_cur)
            new_x = np.linspace(0, 1, n_lp)
            a_lp = np.interp(new_x, old_x, best_seq / best_seq.sum())
            a_lp = np.maximum(a_lp, 0.0)
            a_lp = a_lp / a_lp.sum()

        t5 = min(remaining * 0.3, 400.0)
        a_norm = a_lp * np.sqrt(2.0 * n_lp) / a_lp.sum()
        a_out, sc_out = nesterov_lp(a_norm, t5, seed=42)
        try_update(a_out, f"lp_n{n_lp}")

    # === Phase 6: Random restarts ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\nPhase 6: Random restarts (remaining={remaining:.0f}s)")
        for n_try in [1600, 2000, 3000]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 20:
                break
            t6 = min(remaining * 0.3, 60.0)
            a_out, sc_out = random_restart_optimize(n_try, t6, seed=n_try)
            if a_out is not None:
                try_update(a_out, f"random_n{n_try}")

    # === Phase 7: Final verification and best selection ===
    print(f"\nPhase 7: Final verification")
    if best_seq is not None:
        final_score = evaluate_safe(best_seq)
        if np.isfinite(final_score):
            if final_score < best_score:
                best_score = final_score

        # One more check: pure uniform at the same length
        n_final = len(best_seq)
        a_uniform = np.ones(n_final) / n_final
        sc_uniform = evaluate_safe(a_uniform)
        if sc_uniform < best_score:
            best_score = sc_uniform
            best_seq = a_uniform.copy()
            print(f"  Uniform at n={n_final} is better: {sc_uniform:.10f}")

        print(f"\nFINAL: score={best_score:.10f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.1f}s")
    else:
        print("WARNING: Fallback to uniform n=1600")
        best_seq = np.ones(1600) / 1600
        best_score = evaluate_safe(best_seq)
        print(f"Fallback score: {best_score:.10f}")

    elapsed_total = time.monotonic() - t_global
    if elapsed_total > time_budget:
        print(f"WARNING: Exceeded time budget by {elapsed_total - time_budget:.1f}s")

    return [float(x) for x in best_seq]
