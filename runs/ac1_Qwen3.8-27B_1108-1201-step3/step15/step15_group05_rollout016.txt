        weights = weights / weights.sum()
        grad = np.zeros(n)
        for w, k in zip(weights, candidates):
            grad += w * compute_subgradient(y, int(k), n)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        # Nesterov step
        momentum = min(0.95, 0.5 / (1.0 + 0.01 * t))
        y_new = y + momentum * (y - y_prev) - step * grad
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y

        y_prev = y
        y = y_new

        # Check improvement
        score_y = fast_autocorr(y).max()
        if score_y < best_score - 1e-14:
            best_score = score_y
            best_a = y.copy()
            patience = 0
            step = min(step * 1.02, 5.0)
        else:
            patience += 1
            # Decay step more slowly
            if patience % 50 == 0:
                step *= 0.95
            if step < 1e-10:
                break

        # Random walk injection to escape local minima
        if t % 2000 == 0:
            noise_scale = 0.01 * max(1.0, best_score * n)
            a_noisy = y + noise_scale * rng.standard_normal(n)
            a_noisy = normalize_sum1(a_noisy)
            s_noisy = fast_autocorr(a_noisy).max()
            if s_noisy < best_score:
                best_score = s_noisy
                best_a = a_noisy.copy()
                y = a_noisy.copy()
                y_prev = a_noisy.copy()
                patience = 0

        if verbose and (t + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NS iter={t+1} ({elapsed:.0f}s): best={best_score:.10f}, step={step:.6f}")

    # Final: also check the ergodic average style
    return best_a, best_score


def build_constraint_matrix(a, k_indices, n):
    """Build constraint matrix for (a*b)_k <= M rows."""
    size = len(k_indices)
    mat = np.zeros((size, n))
    for idx, k in enumerate(k_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            mat[idx, j] = a[k - j]
    return mat


def lp_solve(a_fixed, M, k_indices, solver, n):
    """Solve LP: max sum(b) s.t. (a*b)_k <= M for k in k_indices, b >= 0."""
    T = build_constraint_matrix(a_fixed, k_indices, n)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals


def verify_expand(a_fixed, b, M, active_set, solver, n):
    """Check for violations outside active set and expand if needed."""
    size = 2 * n - 1
    all_k = np.arange(size)
    complement = np.setdiff1d(all_k, active_set)
    if len(complement) == 0:
        return active_set, b
    T_comp = build_constraint_matrix(a_fixed, complement, n)
    violations = T_comp @ b
    viol_idx = np.where(violations > M * (1 + 1e-5))[0]
    if len(viol_idx) > 0:
        new_active = np.union1d(active_set, complement[viol_idx])
        if len(new_active) > 0.8 * size:
            new_active = all_k
        b_new, _ = lp_solve(a_fixed, M, new_active, solver, n)
        if b_new is not None:
            return new_active, b_new
    return active_set, b


def nesterov_lp(a0, time_limit, solver, seed=0, verbose=False):
    """Nesterov-accelerated LP best-response iteration."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = normalize_sum1(a0)
    t0 = time.monotonic()

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    y_prev = a.copy()
    active_set = None
    stall_count = 0
    stall_threshold = 60
    n_iters = 0
    momentum = 0.5

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-15:
            break

        # Active set selection: top peaks with tight tolerance
        if active_set is None or t % 25 == 0:
            # Use adaptive tolerance
            tol_frac = max(0.02, 0.08 * np.exp(-t / 200.0))
            active_set = np.where(conv_y >= M_curr * (1.0 - tol_frac))[0]
            # Always include top 3
            top_k = np.argsort(conv_y)[-3:]
            active_set = np.unique(np.concatenate([active_set, top_k]))
            # Cap active set size for efficiency
            if len(active_set) > min(50, n // 5):
                top_n = min(50, n // 5)
                active_set = np.argsort(conv_y[active_set])[-top_n:]
                active_set = active_set[active_set] if False else np.sort(active_set)
                active_set = conv_y.argsort()[::-1][:top_n]

        b, duals = lp_solve(y, M_curr, active_set, solver, n)
        if b is None:
            # Fallback: try with larger active set
            active_set = np.argsort(conv_y)[::-1][:min(20, 2 * n - 1)]
            b, duals = lp_solve(y, M_curr, active_set, solver, n)
            if b is None:
                continue

        # Verify and expand
        active_set, b = verify_expand(y, b, M_curr, active_set, solver, n)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b / b_sum
        n_iters += 1

        # Nesterov update with adaptive momentum
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y

        # Momentum
        if t <= 2:
            a_new = y_new.copy()
        else:
            momentum = min(0.95, (t - 1.0) / (t + 2.0))
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a

        y_prev = y
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score - 1e-14:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score - 1e-14:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Restart if stalled
        if stall_count > stall_threshold:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall_count = 0
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    LP restart at iter {t} ({elapsed:.0f}s), score={best_score:.10f}")

        if verbose and (n_iters + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def coordinate_descent_fast(a0, time_limit, verbose=False):
    """Fast coordinate descent using incremental autocorrelation updates.
    When changing a[j] from old to new, the autocorrelation changes as:
    new_conv[k] = old_conv[k] + (new-old)*(a[k-j] + a[j-k]) + (new^2 - old^2)*delta_{k,2j}
    (with appropriate bounds)
    """
    n = len(a0)
    a = normalize_sum1(a0)
    t0 = time.monotonic()
    size = 2 * n - 1

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()

    n_passes = 0
    no_improve = 0
    max_no_improve = 5

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False

        # Random order for diversity
        order = np.random.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]
            if aj < 1e-15 and conv.max() > 1e-10:
                # Try setting this coordinate to a small positive value
                candidates = [aj + 0.01 / n, 0.01 / n, 0.02 / n]
            else:
                candidates = [aj * 0.5, aj * 0.75, aj * 1.25, aj * 1.5,
                              0.0, min(aj * 2.0, 0.1), max(0.0, aj * 0.25)]

            best_x = aj
            best_local = conv.max()

            for x in candidates:
                x = max(0.0, x)
                dx = x - aj
                if abs(dx) < 1e-15:
                    continue
                # Incremental update
                new_conv = conv.copy()
                for k in range(size):
                    val = 0.0
                    i1 = k - j
                    if 0 <= i1 < n:
                        val += a[i1]
                    i2 = j - k
                    if 0 <= i2 < n and i2 != i1:
                        val += a[i2]
                    elif i2 == i1 and 0 <= i1 < n:
                        val += a[i2]
                    new_conv[k] += dx * val
                if 2 * j < size:
                    new_conv[2 * j] += dx * x  # extra term for a[j]*new_a[j] - a[j]^2
                    # Actually: the change in a[j]^2 term is x^2 - aj^2 = (x-aj)(x+aj)
                    # We already added dx * a[j] (from i1 = j or i2 = j), need to fix
                    # Let me redo: new_conv[k] = conv[k] + dx * (a[k-j] + a[j-k]) + (x^2 - aj^2) * delta
                    # For k = 2j: a[k-j] = a[j] = aj, a[j-k] = a[-j] = 0 (if j>0) or a[0]
                    # This is getting complicated. Let me just recompute for the candidates.
                    pass

            # Simpler approach: just recompute for a few candidates
            # But that's O(n) per candidate per coordinate = O(n^2) per pass
            # For n=1600, one pass is ~2.5M ops, which is fast
            # Let's just do full recompute for the best candidate
            best_x = aj
            best_obj = 2.0 * n * conv.max()
            s_others = 1.0 - aj  # since sum = 1

            for x in candidates:
                x = max(0.0, x)
                if x == aj:
                    continue
                a[j] = x
                s_new = a.sum()
                if s_new < 1e-15:
                    a[j] = aj
                    continue
                a_normed = a / s_new
                new_conv = fast_autocorr(a_normed)
                obj = 2.0 * n * new_conv.max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_x = x
                a[j] = aj  # restore

            if best_x != aj:
                a[j] = best_x
                a = a / a.sum()
                conv = fast_autocorr(a)
                cur_score = 2.0 * n * conv.max()
                if cur_score < best_score - 1e-13:
                    best_score = cur_score
                    best_a = a.copy()
                    improved = True
                else:
                    a[j] = aj
                    a = a / a.sum()
                    conv = fast_autocorr(a)

        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best_score={best_score:.10f}")

    return best_a, best_score


def perturb_sequence(a, rng, scale=0.03):
    """Add multiplicative noise."""
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.1)
    a_pert = a * noise
    return normalize_sum1(a_pert)


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a = normalize_sum1(a)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, a)
    f_interp = np.maximum(f_interp, 0.0)
    s = f_interp.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / s


def cosine_init(n, rng=None):
    """Generate a cosine-based initial sequence."""
    x = np.linspace(0.0, 1.0, n)
    a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x) + 0.05 * np.cos(6 * np.pi * x)
    a = np.maximum(a, 0.0)
    return normalize_sum1(a)


def random_init(n, rng):
    """Random non-negative sequence."""
    a = rng.random(n)
    return normalize_sum1(a)


def propose_candidate():
    """
    Main search function. Uses Nesterov subgradient, LP best-response,
    multi-length exploration, and coordinate descent to minimize the score.
    """
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = 1750.0

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = cosine_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    # === Phase 1: Quick multi-length exploration with Nesterov subgradient ===
    print(f"\n=== Phase 1: Multi-length exploration (quick NS) ===")
    candidate_lengths = [1200, 1600, 2000, 2500, 3000]
    if len(parent) != 1600:
        candidate_lengths = sorted(set(candidate_lengths + [len(parent)]))
    candidate_lengths = [n for n in candidate_lengths if 500 <= n <= 5000]

    ns_budget_each = 15.0  # seconds per length
    length_results = {}

    for n_try in candidate_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        t_sub = min(ns_budget_each, remaining - 25)
        if t_sub < 5:
            break

        # Initialize: interpolate parent to this length
        if parent is not None:
            a_init = interpolate_to_length(parent, n_try)
        else:
            a_init = cosine_init(n_try, rng)

        a_out, _ = nesterov_subgrad(a_init, t_sub, seed=n_try, verbose=False)
        sc = evaluate_safe(a_out)
        length_results[n_try] = sc
        print(f"  NS n={n_try}: score={sc:.8f}")
        try_update(a_out, f"ns_n{n_try}")

    # Pick the best length for deeper optimization
    if length_results:
        best_length = min(length_results, key=length_results.get)
    else:
        best_length = len(best_seq)
    print(f"Best length from exploration: {best_length} (score={length_results.get(best_length, 'N/A')})")

    # === Phase 2: LP Best Response on best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_lp = best_length
    print(f"\n=== Phase 2: LP Nesterov n={n_lp}, remaining={remaining:.0f}s ===")
    t_lp = min(remaining * 0.40, 600.0)
    if t_lp > 60 and n_lp <= 3500:
        a_init = best_seq if len(best_seq) == n_lp else interpolate_to_length(best_seq, n_lp)
        a_out, _ = nesterov_lp(a_init, t_lp, solver, seed=0, verbose=True)
        try_update(a_out, f"lp1_n{n_lp}")

    # === Phase 3: Nesterov Subgradient (more iterations, longer) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: NS polish n={n_cur}, remaining={remaining:.0f}s ===")
    t_ns = min(remaining * 0.15, 200.0)
    if t_ns > 20:
        a_out, _ = nesterov_subgrad(best_seq, t_ns, seed=777, verbose=True)
        try_update(a_out, f"ns2_n{n_cur}")

    # === Phase 4: Second LP round (from improved point) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: LP round 2 n={n_cur}, remaining={remaining:.0f}s ===")
    t_lp2 = min(remaining * 0.20, 300.0)
    if t_lp2 > 30 and n_cur <= 3500:
        a_out, _ = nesterov_lp(best_seq, t_lp2, solver, seed=123, verbose=True)
        try_update(a_out, f"lp2_n{n_cur}")

    # === Phase 5: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: CD n={n_cur}, remaining={remaining:.0f}s ===")
    t_cd = min(remaining * 0.15, 150.0)
    if t_cd > 10:
        a_out = coordinate_descent_fast(best_seq, t_cd, verbose=True)
        try_update(a_out, f"cd1_n{n_cur}")

    # === Phase 6: Perturbation restarts ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: Perturbation restarts, remaining={remaining:.0f}s ===")
    n_restarts = 3
    t6_total = min(remaining * 0.25, 250.0)
    t6_each = t6_total / n_restarts
    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        t6 = min(t6_each, remaining - 25)
        if t6 < 15:
            break
        scale = 0.01 + 0.015 * r
        pert = perturb_sequence(best_seq, rng, scale=scale)
        # Quick NS then LP if time allows
        t_ns_part = min(t6 * 0.3, 30.0)
        t_lp_part = t6 - t_ns_part - 5
        a_out, _ = nesterov_subgrad(pert, t_ns_part, seed=500 + r, verbose=False)
        if t_lp_part > 15 and n_cur <= 3000:
            a_out, _ = nesterov_lp(a_out, t_lp_part, solver, seed=600 + r, verbose=False)
        try_update(a_out, f"pert{r}_n{n_cur}")

    # === Phase 7: Try a different length from scratch ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Alternative length exploration, remaining={remaining:.0f}s ===")
    alt_lengths = [l for l in [800, 2000, 2400, 3200] if l != len(best_seq) and 500 <= l <= 4000]
    for n_alt in alt_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        t_alt = min(remaining * 0.3, 120.0)
        if t_alt < 20:
            break
        # Initialize differently
        a_alt = cosine_init(n_alt, rng)
        # Quick NS
        t_ns = min(t_alt * 0.4, 40.0)
        a_out, _ = nesterov_subgrad(a_alt, t_ns, seed=n_alt, verbose=False)
        # LP if time and length allow
        t_lp_rem = t_alt - t_ns - 5
        if t_lp_rem > 15 and n_alt <= 2500:
            a_out, _ = nesterov_lp(a_out, t_lp_rem, solver, seed=n_alt + 100, verbose=False)
        try_update(a_out, f"alt_n{n_alt}")

    # === Phase 8: Final polishing ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 8: Final polish n={n_cur}, remaining={remaining:.0f}s ===")
        t_final = min(remaining * 0.5, 60.0)
        if t_final > 5:
            a_out = coordinate_descent_fast(best_seq, t_final, verbose=False)
            try_update(a_out, "cd_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_out, _ = nesterov_subgrad(best_seq, min(remaining * 0.4, 30.0), seed=9999, verbose=False)
            try_update(a_out, "ns_final")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
