def build_T_rows(a, lags):
    n = len(a)
    T = np.zeros((len(lags), n))

        return None

        return None
    val = np.asarray(b_var.value, dtype=np.float64)

        return None

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    return a_sym / s * (a.sum() / s)

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")
    
    best_seq = None
    best_score = float('inf')
    
    # Try to load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)
    
    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            old_best = best_score
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (was {old_best:.6f}, n={len(a)}, t={elapsed:.0f}s)")
    
    # Phase 1: Quick LSE-Adam for initial improvement
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 1: LSE-Adam n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_out, _ = lse_adam_optimize(best_seq, t1, verbose=True)
        check_and_update(a_out, f"lse1")
    
    # Phase 2: LP Nesterov (main workhorse - most time)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.45, 700.0)
    print(f"\n=== Phase 2: LP Nesterov n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, solver, verbose=True)
        check_and_update(a_out, f"lp1")
    
    # Phase 3: LSE-Adam round 2 (refine)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 3: LSE-Adam round 2 (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_out, _ = lse_adam_optimize(best_seq, t3, verbose=False)
        check_and_update(a_out, f"lse2")
    
    # Phase 4: LP Nesterov round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 4: LP Nesterov round 2 (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t4, solver, verbose=True)
        check_and_update(a_out, f"lp2")
    
    # Phase 5: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 5: Coord Descent (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_out, _ = coord_descent_polish(best_seq, t5, rng, verbose=True)
        check_and_update(a_out, f"cd1")
    
    # Phase 6: Try symmetrization
    a_sym = try_symmetrize(best_seq)
    check_and_update(a_sym, "sym")
    
    # Phase 7: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        print(f"\n=== Phase 7: Multi-length (remaining={remaining:.0f}s) ===")
        lengths_to_try = [n_base - 200, n_base + 200, n_base + 400, n_base - 400]
        for n_try in lengths_to_try:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 50:
                break
            if n_try < 500 or n_try > 4000:
                continue
            t_try = min(remaining * 0.4, 60.0)
            print(f"  Trying n={n_try} (budget={t_try:.0f}s)")
            init_try = interpolate_seq(best_seq, n_try)
            a_out, _ = lse_adam_optimize(init_try, min(t_try, 25.0), verbose=False)
            check_and_update(a_out, f"ml_lse_n{n_try}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 25:
                a_out, _ = coord_descent_polish(a_out, min(remaining * 0.5, 20.0), rng, verbose=False)
                check_and_update(a_out, f"ml_cd_n{n_try}")
    
    # Phase 8: Fresh starts with quick optimization
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 8: Fresh starts (remaining={remaining:.0f}s) ===")
        for attempt in range(5):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 20:
                break
            n_fresh = n_base + int(rng.integers(-300, 500))
            if n_fresh < 800:
                n_fresh = 1200
            a_fresh = structured_init(n_fresh, rng)
            t_fresh = min(remaining * 0.35, 25.0)
            a_out, _ = lse_adam_optimize(a_fresh, min(t_fresh, 20.0), verbose=False)
            check_and_update(a_out, f"fresh_{attempt}_n{n_fresh}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 15:
                a_out, _ = coord_descent_polish(a_out, min(remaining * 0.4, 15.0), rng, verbose=False)
                check_and_update(a_out, f"fresh_cd_{attempt}_n{n_fresh}")
    
    # Final: one more CD pass on the best
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print(f"\n=== Final CD pass (remaining={remaining:.0f}s) ===")
        a_out, _ = coord_descent_polish(best_seq, remaining * 0.8, rng, verbose=True)
        check_and_update(a_out, "final_cd")
    
    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
            best_seq = best_seq / best_seq.sum()
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    
    return [float(x) for x in best_seq]
