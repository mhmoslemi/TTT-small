def nesterov_lp_optimize(a0, time_limit, solver, seed=0, verbose=False):
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = 2*n*fast_autocorr(a).max() / norm**2  # = max(fast_autocorr(a)) since sum=norm
    
    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break
        
        # Select active set: top peaks
        n_active = max(20, min(100, n // 20))
        active_set = np.argsort(conv_y)[-n_active:]
        
        # Solve LP: max sum(b) s.t. (y*b)[k] <= M_curr for k in active_set, b >= 0
        T = build_constraint_matrix(y, active_set)
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        
        # Check all constraints
        conv_b = fast_autocorr_y_b(y, b_val)  # Need this function
        viol = np.where(conv_b > M_curr * (1 + 1e-6))[0]
        if len(viol) > 0 and len(viol) < 2 * n_active:
            # Add violators and re-solve
            new_active = np.unique(np.concatenate([active_set, viol]))
            T = build_constraint_matrix(y, new_active)
            b_var = cp.Variable(n, nonneg=True)
            constraints = [T @ b_var <= M_curr]
            prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
            try:
                prob.solve(solver=solver, verbose=False)
            except Exception:
                pass
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
                b_sum = b_val.sum()
                if b_sum < 1e-15:
                    continue
        
        g = b_val * norm / b_sum
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.95)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y_prev = y
        y = y_new
        a = a_new
        
        # Track best
        score_y = 2*n*fast_autocorr(y).max() / norm**2
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        score_a = 2*n*fast_autocorr(a).max() / norm**2
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
    
    return best_a, best_score

The step is then projected onto {a >= 0, sum(a) = S}.

**This is very efficient to compute!** The gradient is just a shifted version of a, so computing it is O(n).

Let me now write a more efficient subgradient method:

```python
def subgradient_fast(a0, time_limit, verbose=False):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    
    best_a = a.copy()
    best_score = 2*n*fast_autocorr(a).max() / norm**2
    t0 = time.monotonic()
    
    step = 0.1  # Initial step size
    patience = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > 5000:
            break
        
        conv = fast_autocorr(a)
        M = conv.max()
        
        # Get active peaks (top few)
        n_peaks = max(3, min(20, n // 50))
        top_idx = np.argsort(conv)[-n_peaks:]
        
        # Weighted gradient
        # Weight by how close to max
        weights = np.exp(-10.0 * (M - conv[top_idx]) / max(M * 0.01, 1e-10))
        weights = weights / weights.sum()
        
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            # grad_k[j] = 2*a[k-j] if 0 <= k-j < n
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            grad[lo:hi] += w * 2.0 * a[k-hi+1 : k-lo+1][::-1]
        
        # Step
        a_trial = a - step * grad
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial > 1e-15:
            a_trial = a_trial * norm / s_trial
        
        score_trial = 2*n*fast_autocorr(a_trial).max() / norm**2
        if score_trial < best_score - 1e-12:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
        else:
            patience += 1
            step *= 0.99
            if step < 1e-10:
                break
        
        # periodically increase step to escape local minima
        if t % 1000 == 0:
            step = max(step, 0.01)
    
    return best_a, best_score
```
