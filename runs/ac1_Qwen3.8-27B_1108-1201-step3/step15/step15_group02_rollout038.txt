def topk_lse_gradient(a, k_top, T):
    """Compute gradient of Top-k LSE approximation of max(conv(a,a))."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    conv = fast_conv_self(a)
    if conv.max() < 1e-15:
        return conv, a.copy(), 0.0

    # Get top-k indices
    if k_top >= size:
        top_idx = np.arange(size)
    else:
        top_idx = np.argsort(conv)[-k_top:]

    # Softmax weights over top-k
    vals = conv[top_idx] - conv[top_idx].max()
    weights = np.exp(vals / T)
    weights /= weights.sum()

    # Build weighted "effective" vector
    w = np.zeros(size)
    w[top_idx] = weights

    A = rfft(a, fft_size)
    W = rfft(w, fft_size)

    # Gradient: d(sum_j w_j * conv[j])/d(a[i])
    ca = irfft(np.conj(A) * W, fft_size)[:n]
    aw = irfft(A * W, fft_size)[:n]
    grad = 2.0 * (ca + aw)

    max_c = conv.max()
    return conv, grad, max_c

    n_stages = 25
    temps = np.geomspace(60.0, 0.03, n_stages)
    k_fracs = np.clip(np.geomspace(0.02, 1.0, n_stages), 1, size).astype(int)

        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        n_steps = max(10, min(100, int(remaining * 5)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_conv_self(a)
            M = conv.max()
            if M < 1e-15:
                break

            # Find top peaking lags
            k_star = int(np.argmax(conv))
            # Also consider the symmetric counterpart
            k_star_sym = size - 1 - k_star
            if k_star_sym != k_star:
                k_star = int(np.argmax([k_star, k_star_sym], key=lambda x: conv[x]))

            # Compute frequency response for the peak
            freqs = np.arange(fft_size // 2 + 1)
            # h(f) = cos(2*pi*f*k/N) tells us which frequencies contribute to the peak
            h = np.cos(2.0 * np.pi * freqs * k_star / fft_size)

            # Also add contribution from second-highest peak
            conv_sorted = np.sort(conv)[::-1]
            if len(conv_sorted) > 1 and conv_sorted[1] > 0.9 * M:
                k2 = np.where(np.abs(conv - conv_sorted[1]) < 1e-10)[0]
                k2 = k2[0]
                if k2 != k_star:
                    h2 = np.cos(2.0 * np.pi * freqs * k2 / fft_size)
                    h = 0.7 * h + 0.3 * h2

            A = rfft(a, fft_size)
            mag = np.abs(A)
            phase = np.angle(A)

            # Damp frequencies that contribute positively to the peak
            damp = np.exp(-eta * np.maximum(h, 0.0))
            mag_new = mag * damp

            # Reconstruct
            A_new = mag_new * np.exp(1j * phase)
            a_new = irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new * norm / s_new

            sc = compute_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new

        if verbose and (si % 4 == 0 or si == n_stages - 1):
            elapsed = time.monotonic() - t0
            print(f"    Spectral stage {si} eta={eta:.4f}: best={best_score:.8f} ({elapsed:.0f}s)")

    return best_a, best_score

def lp_best_response(a, M, active_set=None):
    """Solve LP: max sum(b) s.t. T(b) <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)

    if active_set is not None and 0 < len(active_set) < size:
        T = np.zeros((len(active_set), n))
        for idx, k in enumerate(active_set):
            row_start = max(0, k - n + 1)
            row_end = min(n, k + 1)
            for i in range(row_start, row_end):
                T[idx, i] = a[k - i]
        constraints = [T @ b_var <= M]
    else:
        # Full constraint matrix (only feasible for small n)
        if size <= 4000:
            T = np.zeros((size, n))
            for m in range(n):
                T[m:m + n, m] = a
            constraints = [T @ b_var <= M]
        else:
            # Use subset of most constraining rows
            T = np.zeros((200, n))
            for idx, k in enumerate(range(200)):
                row_start = max(0, k - n + 1)
                row_end = min(n, k + 1)
                for i in range(row_start, row_end):
                    T[idx, i] = a[k - i]
            constraints = [T @ b_var <= M]

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=get_solver(), verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)

def lp_nesterov_optimize(a0, time_limit, verbose=True):
    """LP-based Nesterov accelerated optimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1

    max_iters = 30000
    for t in range(1, max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Active set: top peaking constraints
        threshold = M * 0.8
        active_set = np.where(conv_y > threshold)[0]
        if len(active_set) < 5:
            active_set = np.where(conv_y > M * 0.5)[0]
        if len(active_set) > 300:
            active_set = np.where(conv_y > M * 0.92)[0]
            if len(active_set) < 3:
                active_set = None

        b = lp_best_response(y, M, active_set)
        if b is None:
            b = lp_best_response(y, M, None)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = compute_score(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    """Enhanced pair coordinate descent with gradient-informed candidates."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        peak_idx = int(np.argmax(conv))
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            # Gradient-informed candidate selection
            # The "pressure" on index i from the peak is a[peak_idx - i] (if valid)
            j_candidates = set()

            # Add peak-related candidates
            for k in range(max(0, peak_idx - n + 1), min(n, peak_idx + 1)):
                j_candidates.add(int((peak_idx - i + n) % n))
                j_candidates.add(int(i + peak_idx - k + n) % n)

            # Random candidates
            for _ in range(6):
                j_candidates.add(int(rng.integers(n)))

            # Top-peak candidates
            top_peaks = np.argsort(conv)[-5:]
            for kl in top_peaks:
                j_candidates.add(int((kl - i) % n))
                j_candidates.add(int((kl + i) % n))

            j_candidates.discard(i)
            j_candidates = list(j_candidates)

            best_delta = 0.0
            best_v = cur_max
            best_j = -1

            for j in j_candidates[:15]:
                if j == i:
                    continue
                aj = a[j]

                # Compute the change in convolution from moving mass from i to j
                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                d_max = ai * 0.9
                n_samples = 20
                deltas = np.linspace(0.01, d_max, n_samples)

                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 8:
                n_perturb = max(5, n // 100)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.1 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                if no_improve >= 20:
                    break
        else:
            no_improve = 0

    return best_a, best_score

def asymmetric_init(n, rng):
    """Generate diverse asymmetric initial sequences."""
    choice = rng.integers(0, 16)
    x = np.linspace(0.0, 1.0, n)
    tilt = rng.uniform(0.3, 1.5)

    if choice == 0:
        a = np.exp(-tilt * x)
    elif choice == 1:
        a = (1.0 - x) ** rng.uniform(1.5, 4.0)
    elif choice == 2:
        a = np.exp(-tilt * x) * (1.0 + 0.3 * np.cos(2 * np.pi * rng.integers(2, 8) * x))
    elif choice == 3:
        a = (1.0 + tilt * np.sin(np.pi * x * rng.uniform(0.5, 1.5)))
        a = np.maximum(a, 0.0)
    elif choice == 4:
        # Two-tone: high then low
        c = rng.uniform(0.3, 0.7)
        a = np.where(x < c, 1.0, 0.3 + 0.3 * np.sin(np.pi * (x - c) / (1 - c)))
    elif choice == 5:
        # Power law decay
        a = (1.0 + x * rng.uniform(2, 10)) ** (-rng.uniform(0.5, 2.0))
    elif choice == 6:
        # Gaussian with linear tail
        a = np.exp(-3.0 * (x - 0.3) ** 2) + 0.3 * (1.0 - x)
    elif choice == 7:
        # Cosine taper (asymmetric)
        c = rng.uniform(0.3, 0.6)
        a = np.cos(np.pi * (x - c) / (2 * max(c, 1 - c))) ** 2
        a = np.maximum(a, 1e-6)
    elif choice == 8:
        # Sigmoid transition
        center = rng.uniform(0.3, 0.6)
        width = rng.uniform(0.05, 0.2)
        a = 1.0 / (1.0 + np.exp((x - center) / width))
    elif choice == 9:
        # Product of two cosines
        f1 = rng.uniform(0.5, 3.0)
        f2 = rng.uniform(5.0, 15.0)
        a = (1.0 + 0.4 * np.cos(2 * np.pi * f1 * x + rng.uniform(0, np.pi))) * \
            (1.0 + 0.2 * np.cos(2 * np.pi * f2 * x + rng.uniform(0, np.pi)))
        a = np.maximum(a, 0.0)
    elif choice == 10:
        # Piecewise linear
        breaks = np.sort(rng.uniform(0.1, 0.9, 3))
        heights = np.sort(rng.uniform(0.5, 2.0, 4))
        a = np.interp(x, [0] + list(breaks) + [1], list(heights))
    elif choice == 11:
        # Exponential with bump
        a = np.exp(-2.0 * x) + 0.5 * np.exp(-20.0 * (x - 0.4) ** 2)
    elif choice == 12:
        # Smooth step
        c = rng.uniform(0.4, 0.6)
        w = rng.uniform(0.1, 0.3)
        a = 1.0 / (1.0 + np.exp(-(x - c) * 20.0))
        a = 0.3 + 0.7 * a
    elif choice == 13:
        # Power with oscillation
        a = (1.0 - 0.5 * x) * (1.0 + 0.3 * np.sin(6 * np.pi * x))
        a = np.maximum(a, 0.0)
    elif choice == 14:
        # Bimodal asymmetric
        c1, c2 = 0.25, 0.7
        w1, w2 = 0.08, 0.12
        a = np.exp(-(x - c1) ** 2 / (2 * w1 ** 2)) + 0.6 * np.exp(-(x - c2) ** 2 / (2 * w2 ** 2))
    else:
        a = np.ones(n) * (1.0 + 0.2 * rng.standard_normal(n))
        a = np.maximum(a, 0.1)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def vegas_optimize(a0, time_limit, verbose=True):
    """Vegas multiplicative updates with refined pressure."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 20
    etas = np.geomspace(0.08, 0.002, n_stages)

    for si, eta_s in enumerate(etas):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        n_steps = max(15, min(60, int(remaining * 4)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_conv_self(a)
            M = conv.max()
            if M < 1e-15:
                break

            # Pressure: focus on top peaking values
            w = np.maximum(conv - M * 0.85, 0.0) ** 2
            ww = w / (w.sum() + 1e-15)

            W = rfft(ww, fft_size)
            A = rfft(a, fft_size)
            ca = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)[:n]
            pressure = ca + aw

            a = a * np.exp(-eta_s * pressure)
            a = np.maximum(a, 1e-12)
            a = a * norm / a.sum()

            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose and si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Vegas stage {si} eta={eta_s:.5f}: best={best_score:.8f} ({elapsed:.0f}s)")

    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    else:
        print("\nERROR: No valid sequence found!")

    return [float(x) for x in best_seq]
