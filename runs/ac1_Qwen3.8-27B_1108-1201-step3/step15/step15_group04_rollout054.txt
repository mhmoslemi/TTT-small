def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T

def lp_solve_with_constraints(a, M, solver, active_set):
    """Solve LP: max sum(b) s.t. T[active] @ b <= M, b >= 0"""
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    if active_set is not None and 0 < len(active_set) < size:
        T_sub = T[active_set, :]
    else:
        T_sub = T
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_sub @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    duals = None
    try:
        con = constraints[0]
        if con.dual_value is not None:
            if active_set is not None and 0 < len(active_set) < size:
                full_duals = np.zeros(size)
                full_duals[active_set] = np.asarray(con.dual_value, dtype=np.float64)
                duals = full_duals
            else:
                duals = np.asarray(con.dual_value, dtype=np.float64)
    except Exception:
        pass
    return val, duals

def lp_best_response_with_cutting_plane(a, M, solver, max_rounds=10, initial_frac=0.5):
    """LP best response with constraint generation (cutting plane method).
    Start with a subset of constraints, add violated ones iteratively."""
    n = len(a)
    size = 2 * n - 1
    conv = fast_conv_self(a)
    
    # Start with constraints where conv is above initial_frac * M
    if initial_frac < 1.0:
        active = np.where(conv >= initial_frac * M)[0]
    else:
        active = np.arange(size)
    
    if len(active) < 10:
        active = np.where(conv >= 0.3 * M)[0]
    if len(active) < 5:
        active = np.arange(size)
    
    b, duals = lp_solve_with_constraints(a, M, solver, active)
    if b is None:
        b, duals = lp_solve_with_constraints(a, M, solver, None)
        if b is None:
            return None
    
    # Check for violated constraints and add them (cutting plane)
    for _ in range(max_rounds):
        check_conv = fast_conv_cross(a, b)
        violated = np.where(check_conv > M * (1 + 1e-4))[0]
        if len(violated) == 0:
            break
        # Add violated constraints to active set
        active = np.unique(np.concatenate([active, violated]))
        b, duals = lp_solve_with_constraints(a, M, solver, active)
        if b is None:
            b, duals = lp_solve_with_constraints(a, M, solver, None)
            if b is None:
                break
    
    return b

def lse_gradient_descent(a0, time_limit, verbose=True):
    """Log-sum-exp smoothed gradient descent for the max convolution."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 14
    temps = np.geomspace(200.0, 0.05, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(20, min(500, int(remaining * 5)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -300.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)
            g2 = aw[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                step = 0.0
            else:
                step = norm / (T_val * np.sqrt(n)) / (1.0 + 0.03 * it)
                for trial in range(10):
                    a_new = a - step * grad
                    a_new = np.maximum(a_new, 0.0)
                    s_new = a_new.sum()
                    if s_new < 1e-15:
                        step *= 0.5
                        continue
                    a_new = a_new * norm / s_new
                    sc = fast_conv_self(a_new).max()
                    if sc <= max_c + 0.1 * T_val:
                        break
                    step *= 0.5

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best={best_score:.8f}")

    return best_a, best_score

def nestorov_lp_optimize(a0, time_limit, solver, verbose=True, use_multiplicative=False, cutting_plane=True):
    """Nesterov-accelerated LP best response with optional cutting plane."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1
    avg_a = np.zeros(n)
    n_avg = 0
    no_improve_count = 0

    max_iters = 200000
    for t in range(1, max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        if cutting_plane:
            b = lp_best_response_with_cutting_plane(y, M, solver, max_rounds=5, initial_frac=0.6)
            if b is None:
                b = lp_best_response_with_cutting_plane(y, M, solver, max_rounds=3, initial_frac=1.0)
        else:
            # Adaptive screening
            if t < 50:
                frac = 0.4
            elif t < 200:
                frac = 0.55
            else:
                frac = 0.7
            active = np.where(conv_y >= frac * M)[0]
            if len(active) < 20:
                active = np.where(conv_y >= 0.3 * M)[0]
            b, _ = lp_solve_with_constraints(y, M, solver, active)
            if b is None:
                b, _ = lp_solve_with_constraints(y, M, solver, None)
        if b is None or b.sum() < 1e-15:
            continue

        g = b * norm / b.sum()

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        if use_multiplicative and t > 30:
            ratio = g / np.maximum(y, 1e-30)
            log_ratio = np.log(np.clip(ratio, 1e-10, 1e10))
            y_new = y * np.exp(0.5 * alpha * log_ratio)
        else:
            y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min(1.0, (t - 1.0) / (t + 2.0))
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        improved = False
        if sc_y < best_score - 1e-10:
            best_score = sc_y
            best_a = y.copy()
            improved = True

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score - 1e-10:
            best_score = sc_a
            best_a = a.copy()
            improved = True

        if not improved:
            no_improve_count += 1
        else:
            no_improve_count = 0

        # Periodic perturbation
        if no_improve_count in (80, 200, 400):
            n_perturb = max(5, n // 25)
            rng_p = np.random.default_rng(t)
            idx_perturb = rng_p.choice(n, size=n_perturb, replace=False)
            y[idx_perturb] *= (1.0 + 0.15 * rng_p.standard_normal(n_perturb))
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y * norm / s_y
            no_improve_count = 0

        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc_avg = fast_conv_self(avg_check).max()
                if sc_avg < best_score:
                    best_score = sc_avg
                    best_a = avg_check.copy()

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def exact_subgradient_descent(a0, time_limit, verbose=True, top_k=30):
    """Gradient descent using exact subgradient of max(conv) at top peaks."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for it in range(5000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        M = conv.max()

        # Find top_k peaks
        peak_threshold = M * 0.95
        peak_idx = np.where(conv >= peak_threshold)[0]
        if len(peak_idx) == 0:
            peak_idx = np.array([np.argmax(conv)])
        peak_idx = peak_idx[:top_k]

        # Compute exact subgradient: for each peak position p,
        # d(conv[p])/da[i] = a[p-i] + a[i-p] (where indices valid)
        # Sum over all peak positions
        grad = np.zeros(n)
        A = rfft(a, fft_size)
        for p in peak_idx:
            # Contribution from peak at position p
            # conv[p] = sum_i a[i] * a[p-i]
            # d(conv[p])/da[i] = a[p-i] (if 0<=p-i<n) + a[i-p] (if 0<=i-p<n)
            # First term: a[p-i] for i in [max(0,p-n+1), min(p, n-1)]
            lo1 = max(0, p - n + 1)
            hi1 = min(p + 1, n)
            if hi1 > lo1:
                grad[lo1:hi1] += a[p - np.arange(lo1, hi1)]
            # Second term: a[i-p] for i in [max(p,0), min(n, p+n)]
            lo2 = max(p, 0)
            hi2 = min(n, p + n)
            if hi2 > lo2:
                grad[lo2:hi2] += a[np.arange(lo2, hi2) - p]

        # Normalize gradient by number of peaks
        grad /= len(peak_idx)

        # Project gradient to be orthogonal to the normalization direction
        # We want to move in direction that reduces M while keeping S fixed
        # Use: grad_perp = grad - (sum(grad)/n) * ones (tangent to simplex)
        grad_mean = grad.mean()
        grad_perp = grad - grad_mean

        grad_norm = np.linalg.norm(grad_perp)
        if grad_norm < 1e-15:
            break

        # Adaptive step size with backtracking
        step = 0.5 / np.sqrt(n)
        for trial in range(20):
            a_new = a - step * grad_perp / grad_norm
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                step *= 0.5
                continue
            a_new = a_new * norm / s_new
            sc = fast_conv_self(a_new).max()
            if sc < M - 1e-12:
                break
            step *= 0.5

        a = a_new
        sc = fast_conv_self(a).max()
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def spectral_smoothing(a, n_smooth=5):
    """Apply spectral smoothing: flatten the magnitude spectrum."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    mag = np.abs(A)
    # Blend with uniform magnitude
    uniform_mag = mag.sum() / len(mag)
    blend = 0.3
    new_mag = (1 - blend) * mag + blend * uniform_mag
    # Preserve phase
    phase = np.angle(A)
    A_new = new_mag * np.exp(1j * phase)
    a_new = irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent: transfer mass from peak-contributing positions to others."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()

        return a, float('inf')

    best_a = a.copy()

    while time.monotonic() - t0 < time_limit:

        pass_improved = False

        # Identify peak lags

            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-12:
                continue

            # Build candidate j set
            j_candidates = set()
            for _ in range(8):

            for kl in peak_lags[:10]:
                j1 = int(kl - i)
                j2 = int(kl + i)
                if 0 <= j1 < n:
                    j_candidates.add(j1)
                if 0 <= j2 < n:
                    j_candidates.add(j2)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]
                if aj < 0 and aj > -1:
                    continue

                # Compute the gain vector g such that new_conv = conv + delta*g + quadratic terms
                g = np.zeros(size)
                # Adding delta to a[j]: contributes a shifted by delta
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                # Removing delta from a[i]: subtracts
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # The quadratic terms: -delta^2 at 2i, +delta^2 at 2j, +2*aj*delta at 2j, -2*ai*delta at 2i
                n_samples = 15
                deltas = np.linspace(0.05, 0.8, n_samples) * ai
                best_d = 0.0
                best_v = cur_max
                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_d = d

                if best_d > 0 and best_v < best_val - 1e-12:
                    best_val = best_v
                    best_delta = best_d
                    best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = np.maximum(a, 0.0)
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 6:
                break
            if no_improve == 3:
                n_perturb = max(5, n // 30)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.1 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and no_improve % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD no_improve={no_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.15 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0.1)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-6.0 * (x - 0.75) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-2.0 * x) + 0.5 * np.exp(-2.0 * (1 - x))
    elif choice == 7:
        a = 0.5 * np.exp(-4.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-4.0 * (x - 0.7) ** 2)
        a = a * (1.0 + 0.3 * np.cos(20 * np.pi * x))
        a = np.maximum(a, 0.01)
    elif choice == 8:
        a = np.abs(np.sin(3 * np.pi * x)) ** 0.8
    elif choice == 9:
        a = 0.4 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(11 * np.pi * x)
        a = np.maximum(a, 0.05)
    elif choice == 10:
        # Hanning-like window
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    else:
        # Multi-frequency
        a = 0.3 + 0.25 * np.cos(2*np.pi*2*x) + 0.2 * np.cos(2*np.pi*5*x + 0.5) + 0.15 * np.cos(2*np.pi*9*x + 1.0)
        a = np.maximum(a, 0.02)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def optimize_single_length(a0, n, time_budget, solver, rng):
    """Run full optimization pipeline for a single sequence length."""
    best_a = a0.copy()
    best_score = eval_fn(best_a)
    t_start = time.monotonic()
    
    def elapsed():
        return time.monotonic() - t_start
    
    def remaining():
        return time_budget - elapsed()
    
    def try_update(a, label):
        nonlocal best_a, best_score
        if a is None:
            return False
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return False
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"    *** NEW BEST [{label}]: {sc:.8f} (n={n}, t={elapsed():.0f}s)")
            return True
        return False

    # Phase A: LSE gradient descent
    tA = min(remaining() * 0.12, 120.0)
    if tA > 15:
        print(f"    Phase A: LSE (budget={tA:.0f}s)")
        a_out, _ = lse_gradient_descent(best_a, tA, verbose=False)
        try_update(a_out, "lse")

    # Phase B: LP Nesterov with cutting plane
    tB = min(remaining() * 0.35, 500.0)
    if tB > 30:
        print(f"    Phase B: LP+CP Nesterov (budget={tB:.0f}s)")
        a_out, _ = nesterov_lp_optimize(best_a, tB, solver, verbose=True, use_multiplicative=False, cutting_plane=True)
        try_update(a_out, "lp_cp")

    # Phase C: Exact subgradient descent
    tC = min(remaining() * 0.18, 200.0)
    if tC > 20:
        print(f"    Phase C: SubGrad (budget={tC:.0f}s)")
        a_out, _ = exact_subgradient_descent(best_a, tC, verbose=True, top_k=20)
        try_update(a_out, "subgrad")

    # Phase D: LP Nesterov round 2 (multiplicative)
    tD = min(remaining() * 0.2, 250.0)
    if tD > 30:
        print(f"    Phase D: LP+CP Nesterov r2 (budget={tD:.0f}s)")
        a_out, _ = nesterov_lp_optimize(best_a, tD, solver, verbose=True, use_multiplicative=True, cutting_plane=True)
        try_update(a_out, "lp_cp2")

    # Phase E: Spectral smoothing + re-optimize
    if remaining() > 30:
        print(f"    Phase E: Spectral smoothing")
        a_smooth = spectral_smoothing(best_a)
        try_update(a_smooth, "spectral")
        tE = min(remaining() * 0.15, 100.0)
        if tE > 15:
            a_out, _ = nesterov_lp_optimize(best_a, tE, solver, verbose=False, cutting_plane=True)
            try_update(a_out, "post_spectral_lp")

    # Phase F: Pair CD polish
    tF = min(remaining() * 0.15, 150.0)
    if tF > 15:
        print(f"    Phase F: Pair CD (budget={tF:.0f}s)")
        a_out, _ = pair_cd_optimize(best_a, tF, rng, verbose=True)
        try_update(a_out, "pcd")

    # Phase G: Final LSE + LP
    if remaining() > 30:
        tG1 = min(remaining() * 0.3, 60.0)
        a_out, _ = lse_gradient_descent(best_a, tG1, verbose=False)
        try_update(a_out, "lse_final")
        if remaining() > 20:
            tG2 = min(remaining() * 0.5, 60.0)
            a_out, _ = nesterov_lp_optimize(best_a, tG2, solver, verbose=False, cutting_plane=True)
            try_update(a_out, "lp_final")

    return best_a, best_score

def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function: finds the best sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1740.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass

    # If no parent, create good initializations
    if best_seq is None:
        for n_try in [800, 1000, 1200]:
            best_init = None
            best_init_score = float('inf')
            for trial in range(8):
                cand = structured_init(n_try, rng)
                sc = eval_fn(cand)
                if sc < best_init_score:
                    best_init_score = sc
                    best_init = cand
            if best_init is not None and best_init_score < best_score:
                best_seq = best_init
                best_score = best_init_score
        print(f"Initial: n={len(best_seq)}, score={best_score:.8f}")

    n_base = len(best_seq)

    # Main optimization at base length
    print(f"\n=== Main optimization at n={n_base} ===")
    t_main = time_budget * 0.7
    a_out, sc_out = optimize_single_length(best_seq, n_base, t_main, solver, rng)
    if np.isfinite(sc_out) and sc_out < best_score:
        best_score = sc_out
        best_seq = a_out.copy()

    # Try a longer length
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 400
    print(f"\n=== Trying longer length n={n_long} (remaining={remaining:.0f}s) ===")
    if remaining > 120:
        init_long = interpolate_seq(best_seq, n_long)
        t_long = min(remaining * 0.6, 400.0)
        a_long, sc_long = optimize_single_length(init_long, n_long, t_long, solver, rng)
        if np.isfinite(sc_long) and sc_long < best_score:
            best_score = sc_long
            best_seq = a_long.copy()
            print(f"  Long length improved! score={best_score:.8f}")

    # Try a shorter length (sometimes shorter is better due to the 2n factor)
    remaining = time_budget - (time.monotonic() - t_global)
    n_short = max(600, n_base - 400)
    if n_short != n_base and remaining > 100:
        print(f"\n=== Trying shorter length n={n_short} (remaining={remaining:.0f}s) ===")
        init_short = interpolate_seq(best_seq, n_short)
        t_short = min(remaining * 0.5, 250.0)
        a_short, sc_short = optimize_single_length(init_short, n_short, t_short, solver, rng)
        if np.isfinite(sc_short) and sc_short < best_score:
            best_score = sc_short
            best_seq = a_short.copy()
            print(f"  Short length improved! score={best_score:.8f}")

    # Final multi-restart at best length
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    if remaining > 60:
        print(f"\n=== Final multi-restart at n={n_final} (remaining={remaining:.0f}s) ===")
        n_restarts = max(2, min(4, int(remaining / 80)))
        for r in range(n_restarts):
            rem_r = time_budget - (time.monotonic() - t_global)
            if rem_r < 40:
                break
            # Perturb the best sequence
            perturbation_scale = 0.05 + 0.03 * r
            a_perturbed = best_seq * (1.0 + perturbation_scale * rng.standard_normal(n_final))
            a_perturbed = np.maximum(a_perturbed, 0.0)
            s_p = a_perturbed.sum()
            if s_p < 1e-15:
                continue
            a_perturbed = a_perturbed / s_p
            t_r = min(rem_r * 0.7, 200.0)
            a_r, sc_r = optimize_single_length(a_perturbed, n_final, t_r, solver, rng)
            if np.isfinite(sc_r) and sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  Restart {r} improved! score={best_score:.8f}")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = min(best_score, final_score)
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
