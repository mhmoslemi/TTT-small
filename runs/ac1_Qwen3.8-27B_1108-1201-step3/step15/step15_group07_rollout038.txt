def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver):
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

    else:
        a = a * norm / s0

def subgradient_polish(a0, time_limit, verbose=True):
    """Subgradient method with backtracking on the peak objective."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    eta = 1.0 / np.sqrt(n)
    n_iters = 0
    patience = 0
    rng = np.random.default_rng(7777)

    while time.monotonic() - t0 < time_limit:
        conv = fast_autocorr(a)
        max_val = conv.max()

        # Get active set (top 15 lags)
        k_actual = min(15, size)
        active_idx = np.argsort(conv)[-k_actual:]

        # Compute subgradient: average of gradients of active c_k
        # grad c_k / grad a_j = 2*a_{k-j} if 0 <= k-j <= n-1
        grad = np.zeros(n)
        for k in active_idx:
            k = int(k)
            for j in range(n):
                m = k - j
                if 0 <= m < n:
                    grad[j] += 2.0 * a[m]
        grad /= len(active_idx)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Backtracking step size
        eta_try = eta
        found = False
        for _ in range(20):
            a_temp = a - eta_try * grad
            # Project onto {x >= 0, sum(x) = norm}
            a_proj = project_simplex(a_temp, norm)
            conv_try = fast_autocorr(a_proj).max()
            if conv_try < max_val - 1e-12:
                a = a_proj
                found = True
                break
            eta_try *= 0.5

        if not found:
            # Accept anyway with very small step
            a_temp = a - eta_try * grad
            a = project_simplex(a_temp, norm)

        n_iters += 1
        cur_score = fast_autocorr(a).max()

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            patience = 0
            eta = max(eta * 1.1, 1.0)
        else:
            patience += 1
            if patience % 500 == 0:
                eta *= 0.8

        if patience > 5000:
            # Perturb to escape
            noise_scale = norm * 0.005
            noise = rng.normal(0, noise_scale, n)
            a_pert = a + noise
            a_pert = np.maximum(a_pert, 0)
            s_pert = a_pert.sum()
            if s_pert > 1e-15:
                a = a_pert * norm / s_pert
            patience = 0
            eta = 1.0 / np.sqrt(n)

        if verbose and n_iters % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, eta={eta:.4f}")

    return best_a, best_score


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    # Binary search on threshold
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    rho = np.searchsorted(cumsum - v_sorted * np.arange(1, n + 1), S)
    if rho == 0:
        rho = 1
    if rho >= n:
        rho = n - 1
    theta = (cumsum[rho] - S) / (rho + 1)
    return np.maximum(v - theta, 0.0)

    print(f"\n=== Phase 3: Mirror Descent n={n3} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n3) / best_seq.sum()
        a_out, _ = mirror_descent_subgrad(a_norm, t3, top_k=15, verbose=True)
        do_try_update(a_out, f"md_n{n3}")

    # Phase 4: LP Nesterov round 2 (12% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n4 = len(best_seq)
    t4 = remaining * 0.12
    print(f"\n=== Phase 4: LP Nesterov R2 n={n4} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n4) / best_seq.sum()
        a_out, _ = lp_nesterov_improved(a_norm, t4, solver, verbose=True)
        do_try_update(a_out, f"lp2_n{n4}")

    # Phase 5: CD polish (7% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n5 = len(best_seq)
    t5 = remaining * 0.07
    print(f"\n=== Phase 5: CD polish n={n5} (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n5) / best_seq.sum()
        a_out, _ = cd_polish(a_norm, t5, verbose=True)
        do_try_update(a_out, f"cd_n{n5}")

    # Phase 6: Trial with different n (6% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        t6 = min(remaining * 0.6, 300.0)
        print(f"\n=== Phase 6: Try n=2000 (budget={t6:.0f}s) ===")
        n_trial = 2000
        # Interpolate best_seq to new length
        a_interp = np.interp(
            np.linspace(0, 1, n_trial),
            np.linspace(0, 1, len(best_seq)),
            best_seq
        )
        a_interp = np.maximum(a_interp, 0)
        s_int = a_interp.sum()
        if s_int > 1e-10:
            a_interp = a_interp / s_int
            a_norm = a_interp * np.sqrt(2.0 * n_trial)
            a_out, _ = lp_nesterov_improved(a_norm, t6 * 0.6, solver, verbose=False)
            do_try_update(a_out, f"lp_n2000")
            remaining2 = time_budget - (time.monotonic() - t_global)
            if remaining2 > 30:
                a_out2, _ = subgradient_polish(a_out, min(remaining2 * 0.5, 120.0), verbose=False)
                do_try_update(a_out2, f"sg_n2000")

    # Phase 7: Final LP + CD polish (remaining time)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_final = len(best_seq)
        t7a = min(remaining * 0.5, 180.0)
        t7b = remaining - t7a - 5
        if t7a > 20:
            print(f"\n=== Phase 7a: Final LP n={n_final} (budget={t7a:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_final) / best_seq.sum()
            a_out, _ = lp_nesterov_improved(a_norm, t7a, solver, verbose=False)
            do_try_update(a_out, f"lp3_n{n_final}")
        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 20:
            print(f"\n=== Phase 7b: Final CD n={n_final} (budget={remaining2*0.8:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_final) / best_seq.sum()
            a_out, _ = cd_polish(a_norm, remaining2 * 0.8, verbose=False)
            do_try_update(a_out, f"cd3_n{n_final}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
