def get_solver():

        try:

            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_constraint_matrix(a, k_indices, n):
    size = len(k_indices)
    mat = np.zeros((size, n))
    for idx, k in enumerate(k_indices):
        lo = max(0, int(k) - n + 1)
        hi = min(n, int(k) + 1)
        for j in range(lo, hi):
            mat[idx, j] = a[int(k) - j]
    return mat


def lp_solve(a_fixed, M, active_set, solver):
    n = len(a_fixed)
    k_indices = np.asarray(active_set, dtype=int)
    T = build_constraint_matrix(a_fixed, k_indices, n)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals


def verify_expand(a_fixed, b, M, active_set, solver, n):
    size = 2 * n - 1
    all_k = np.arange(size)
    complement = np.setdiff1d(all_k, active_set)
    if len(complement) == 0:
        return active_set, b
    T_comp = build_constraint_matrix(a_fixed, complement, n)
    violations = T_comp @ b
    viol_idx = np.where(violations > M * (1 + 1e-6))[0]
    if len(viol_idx) > 0:
        new_active = np.union1d(active_set, complement[viol_idx])
        if len(new_active) > 0.85 * size:
            new_active = all_k
        b_new, _ = lp_solve(a_fixed, M, new_active, solver)
        if b_new is not None:
            return new_active, b_new
    return active_set, b


def nesterov_lp(a0, time_limit, solver, verbose=False, seed=0):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    stall_threshold = 50

    y = a.copy()
    y_prev = a.copy()
    t = 1

    erg_a = np.zeros(n)
    erg_count = 0

    while time.monotonic() - t0 < time_limit:
        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        if active_set is None or t % 30 == 0:
            tol = max(0.005, 0.04 * np.exp(-t / 200.0))
            active_set = np.where(conv_y >= M_curr * (1.0 - tol))[0]
            top_k = np.argsort(conv_y)[-max(5, n // 100):]
            active_set = np.unique(np.concatenate([active_set, top_k]))

        b, duals = lp_solve(y, M_curr, active_set, solver)
        if b is None:
            if t % 10 == 0:
                active_set = np.arange(2 * n - 1)
                b, duals = lp_solve(y, M_curr, active_set, solver)
            if b is None:
                t += 1
                continue

        active_set, b = verify_expand(y, b, M_curr, active_set, solver, n)
        if b is None:
            t += 1
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            t += 1
            continue
        g = b * norm / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            momentum = min(momentum, 0.95)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y
        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        erg_a += a
        erg_count += 1
        if erg_count % 30 == 0:
            erg_check = erg_a / erg_count
            s_erg = erg_check.sum()
            if s_erg > 1e-15:
                erg_check = erg_check * norm / s_erg

                if score_erg < best_score:

            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall_count = 0
            erg_a = np.zeros(n)
            erg_count = 0
            t = 1
            if verbose:

                print(f"    LP restart at iter {t} ({elapsed:.0f}s), score={best_score:.10f}")

def exact_cd(a0, time_limit, verbose=False):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_changes = 0
        conv = fast_autocorr(a).copy()
        a_sum = a.sum()

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]

            # Compute c vector
            c = np.zeros(size)
            k_lo = j
            k_hi = min(size, j + n)
            if k_hi > k_lo:
                c[k_lo:k_hi] = a[k_lo - j:k_hi - j]
            d_idx = 2 * j
            if d_idx < size:
                c[d_idx] = 0.0

            # Compute bg
            bg = conv - 2.0 * aj * c
            if d_idx < size:
                bg[d_idx] -= aj * aj

            s_others = a_sum - aj

            # Golden section search on h(x) in [0, x_max]
            x_max = max(aj * 2.5 + norm / n * 2.0, norm / n * 5.0, 0.1)

            # Quick check: if aj is very small, try a wider range
            if aj < norm / n * 0.1:
                x_max = max(x_max, norm / n * 10.0)

            # Evaluate at endpoints and current
            def h_func(x):
                num = bg + 2.0 * x * c
                if d_idx < size:
                    num[d_idx] += x * x
                den = (s_others + x) ** 2
                if den < 1e-30:
                    return float('inf')
                return num.max() / den

            h_0 = h_func(0.0)
            h_aj = h_func(aj)

            # Golden section
            lo = 0.0
            hi = x_max
            b_pt = hi - gr * (hi - lo)
            c_pt = lo + gr * (hi - lo)
            f_b = h_func(b_pt)
            f_c = h_func(c_pt)

            for _ in range(35):
                if f_b < f_c:
                    hi = c_pt
                    c_pt = b_pt
                    f_c = f_b
                    b_pt = hi - gr * (hi - lo)
                    f_b = h_func(b_pt)
                else:
                    lo = b_pt
                    b_pt = c_pt
                    f_b = f_c
                    c_pt = lo + gr * (hi - lo)
                    f_c = h_func(c_pt)

            x_opt = (lo + hi) / 2.0
            f_opt = h_func(x_opt)

            # Compare with endpoints
            if h_0 < f_opt:
                x_opt = 0.0
                f_opt = h_0
            if h_aj < f_opt:
                x_opt = aj
                f_opt = h_aj

            if abs(x_opt - aj) > 1e-10:
                # Update a[j] and conv incrementally
                a[j] = x_opt
                conv -= 2.0 * (x_opt - aj) * c
                if d_idx < size:
                    conv[d_idx] += x_opt * x_opt - aj * aj
                a_sum += (x_opt - aj)
                s_new = a.sum()
                if s_new > 1e-15:
                    scale = norm / s_new
                    a = a * scale
                    conv = conv * (scale * scale)
                    a_sum = a.sum()
                    pass_changes += 1

        # After full pass, check improvement
        new_conv_max = fast_autocorr(a).max()
        if new_conv_max < best_score - 1e-12:
            best_score = new_conv_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 4:
                break

        if verbose and n_passes % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}, changes={pass_changes}")

    return best_a.copy(), best_score


def subgradient_polish(a0, time_limit, verbose=False):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    step = 0.5 / np.sqrt(n)
    patience = 0
    max_patience = 300
    t = 1

    while time.monotonic() - t0 < time_limit:
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()
        k_star = int(np.argmax(conv))

        # Subgradient at k_star
        grad = np.zeros(n)
        k_lo = max(0, k_star - n + 1)
        k_hi = min(n, k_star + 1)
        for j in range(k_lo, k_hi):
            idx = k_star - j
            if 0 <= idx < n:
                grad[j] += a[idx]
                grad[idx] += a[j]
        # Fix double counting
        for j in range(k_lo, k_hi):
            idx = k_star - j
            if 0 <= idx < n and idx == j:
                grad[j] -= a[j]

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Polyak-like step: use (M_curr - best_target) / ||grad||^2
        # But we don't know optimum, so use heuristic
        target = M_curr * 0.99  # optimistic target
        polyak_step = max(0.0, (M_curr - target) / (grad_norm * grad_norm / n))
        effective_step = max(step, min(polyak_step, 5.0 / grad_norm * n))

        a_trial = a - effective_step * grad / (n ** 0.5)
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-12:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.03, 3.0)
        else:
            patience += 1
            step *= 0.96
            if step < 1e-10:
                break

        # Multi-peak subgradient every 50 iterations
        if t % 50 == 0:
            conv2 = fast_autocorr(a)
            n_peaks = max(3, min(8, n // 200))
            top_idx = np.argsort(conv2)[-n_peaks:]
            M2 = conv2.max()
            weights = np.exp(-10.0 * (M2 - conv2[top_idx]) / max(M2 * 0.01, 1e-10))
            weights = weights / weights.sum()

            grad_multi = np.zeros(n)
            for w, k in zip(weights, top_idx):
                k2 = int(k)
                kk_lo = max(0, k2 - n + 1)
                kk_hi = min(n, k2 + 1)
                for j in range(kk_lo, kk_hi):
                    idx2 = k2 - j
                    if 0 <= idx2 < n:
                        grad_multi[j] += a[idx2]
                        grad_multi[idx2] += a[j]
                for j in range(kk_lo, kk_hi):
                    idx2 = k2 - j
                    if 0 <= idx2 < n and idx2 == j:
                        grad_multi[j] -= a[j]
                grad_multi *= 1.0  # normalize per peak
                grad_multi = grad_multi / (np.linalg.norm(grad_multi) + 1e-30)
                grad_multi = grad_multi * w

            grad_multi_norm = np.linalg.norm(grad_multi)
            if grad_multi_norm > 1e-15:
                a_trial2 = a - step * 0.5 * grad_multi / grad_multi_norm
                a_trial2 = np.maximum(a_trial2, 0.0)
                s2 = a_trial2.sum()
                if s2 > 1e-15:
                    a_trial2 = a_trial2 * norm / s2
                    sc2 = fast_autocorr(a_trial2).max()
                    if sc2 < best_score - 1e-12:
                        a = a_trial2
                        best_score = sc2
                        best_a = a_trial2.copy()
                        patience = 0

        t += 1

    return best_a.copy(), best_score

    t3 = min(remaining * 0.10, 100.0)
    if t3 > 20:
        print(f"\n=== Phase 3: Subgradient n={n_cur}, budget={t3:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = subgradient_polish(a_norm, t3, verbose=True)
        try_update(a_out, "subg1")

    # === Phase 4: LP Nesterov round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 30:
        print(f"\n=== Phase 4: LP round 2 n={n_cur}, budget={t4:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t4, solver, verbose=False, seed=123)
        try_update(a_out, "lp2")

    # === Phase 5: Exact CD round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = min(remaining * 0.20, 300.0)
    if t5 > 20:
        print(f"\n=== Phase 5: Exact CD round 2 n={n_cur}, budget={t5:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = exact_cd(a_norm, t5, verbose=True)
        try_update(a_out, "cd2")

    # === Phase 6: Perturbation + CD restarts ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: Perturbation restarts, remaining={remaining:.0f}s ===")
    n_restarts = max(1, int(remaining // 80))
    n_restarts = min(n_restarts, 3)
    t6_each = min(remaining / (n_restarts + 0.5), 150.0)
    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        t6 = min(t6_each, remaining - 20)
        if t6 < 20:
            break
        scale = 0.015 + 0.01 * r
        pert = perturb(best_seq, rng, scale)
        a_norm = pert * np.sqrt(2.0 * n_cur) / pert.sum()
        a_out, _ = exact_cd(a_norm, t6, verbose=False)
        try_update(a_out, f"pert_cd_{r}")

    # === Phase 7: Final subgradient + CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 40:
        print(f"\n=== Phase 7: Final polish, remaining={remaining:.0f}s ===")
        t7a = min(remaining * 0.4, 60.0)
        t7b = min(remaining - t7a - 5, 100.0)
        if t7a > 15:
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = subgradient_polish(a_norm, t7a, verbose=False)
            try_update(a_out, "subg_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = exact_cd(a_norm, remaining - 5, verbose=False)
            try_update(a_out, "cd_final")

    # Final
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("ERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
