import numpy as np

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

def compute_gradient(a, w):
    """Compute gradient of sum_j w_j * (a*a)(j) with respect to a.
    g[i] = 2 * sum_k w[i+k] * a[k] - w[2i]*a[i] (if 2i < 2n-1)
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # Cross-correlation: result[i] = sum_k w[i+k] * a[k]
    cross = irfft(W * np.conj(A), fft_size)[:n]
    # Diagonal correction
    g = 2.0 * cross.copy()
    valid = 2 * np.arange(n) < size
    g[valid] -= w[2 * np.arange(n)[valid]] * a[valid]
    return g

                f_peak = f[peak_idx_local]
                if f_peak >= 0:
                    # Moving from i to j would increase the peak (or not change it)
                    # Try the reverse: move from j to i
                    continue
                
                # Find the constraint that limits d
                # For k where f[k] > f_peak: c[k] + 2d*f[k] <= c[peak] + 2d*f_peak
                # => d <= (c[peak] - c[k]) / (2*(f[k] - f_peak))
                
                mask = f > f_peak
                if not np.any(mask):
                    # No constraint; use full amount
                    d_opt = a[i] * 0.99
                else:
                    num = c[peak_idx_local] - c[mask]
                    den = 2.0 * (f[mask] - f_peak)
                    # Only consider where num > 0 and den > 0
                    valid = (num > 0) & (den > 0)
                    if not np.any(valid):
                        d_opt = a[i] * 0.01
                    else:
                        d_limits = num[valid] / den[valid]
                        d_opt = min(d_limits.min(), a[i] * 0.8)
                
                d_opt = max(d_opt, 0.0)
                if d_opt < norm * 1e-8:
                    continue
                
                # Apply the move
                a[i] -= d_opt
                a[j] += d_opt
                a = np.maximum(a, 0.0)
                c = fast_autocorr(a)
                
                if c.max() < best_score - 1e-12:
                    best_score = c.max()
                    best_a = a.copy()
                    pass_improved = True
                    no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 2:
                break

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    QuadRefine pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score
