def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v.copy()
    v = np.asarray(v, dtype=np.float64)
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n + 1) > (css - S))[0]
    if len(rho) == 0:
        return np.full(n, S / n)
    rho = rho[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_grad(w, a, fft_size):
    """Compute grad[i] = 2 * sum_k w[k] * a[k-i] for i=0..n-1.
    Uses FFT: grad = 2 * reverse(convolve(reverse(w), a))[0:n]
    """
    n = len(a)
    w_r = w[::-1]
    W_r = rfft(w_r, fft_size)
    A = rfft(a, fft_size)
    conv_full = irfft(W_r * A, fft_size)
    # r[i] = conv_full[n-1-i], so grad[i] = 2*conv_full[n-1-i]
    # grad = 2 * conv_full[n-1, n-2, ..., 0]
    grad = 2.0 * conv_full[n - 1::-1]
    return grad[:n]


def compute_subgrad_single(a, k_star):
    """Compute subgradient of c_{k_star} = conv(a,a)[k_star] w.r.t. a.
    g[i] = 2*a[k_star - i] if 0 <= k_star-i < n, else 0.
    """
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k_star - (n - 1))
    hi = min(n - 1, k_star)
    idx = np.arange(lo, hi + 1)
    g[idx] = 2.0 * a[k_star - idx]
    return g


def nesterov_subgradient(a0, time_limit, verbose=True, T_init=5.0, T_min=0.05):
    """Nesterov-accelerated smoothed subgradient method.
    Minimizes max(conv(a,a)) subject to a>=0, sum(a)=sqrt(2n).
    """
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s0

    best_a = a.copy()
    best_c = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    it = 0
    T = T_init
    fft_size = 1 << (2 * n - 1 - 1).bit_length()

    # Running average for ergodic convergence
    avg_a = np.zeros(n)
    n_avg = 0

    # Adaptive step: start large, diminish
    G_bound = 2.0 * S  # upper bound on subgradient norm

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute convolution at Nesterov point
        c = fast_autocorr(y)
        c_max = c.max()

        # Compute smoothed weights
        exp_c = np.exp((c - c_max) / T)
        w = exp_c / exp_c.sum()

        # Compute gradient via FFT cross-correlation
        grad = compute_smoothed_grad(w, y, fft_size)

        # Nesterov step with adaptive step size
        # For smoothed function with Lipschitz constant ~ 2*S/T:
        lip_const = 2.0 * S / T
        step = lip_const / (lip_const + it)

        a_new = y - step * grad
        a_new = project_simplex(a_new, S)

        # Nesterov momentum update
        alpha = 2.0 / (it + 2.0)
        y_new = (1.0 - alpha) * y + alpha * a_new

        if it == 1:
            a_curr = a_new.copy()
        else:
            mom = (it - 1.0) / (it + 2.0)
            a_curr = y_new + mom * (y_new - y)
            a_curr = project_simplex(a_curr, S)

        y = y_new
        a = a_curr

        # Update running average
        avg_a += a
        n_avg += 1

        # Track best
        c_a = fast_autocorr(a)
        if c_a.max() < best_c:
            best_c = c_a.max()
            best_a = a.copy()

        c_y = fast_autocorr(y)
        if c_y.max() < best_c:
            best_c = c_y.max()
            best_a = y.copy()

        # Check running average periodically
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * S / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_c:
                    best_c = score_avg
                    best_a = avg_check.copy()

        # Decrease temperature
        if it % 500 == 0:
            T = max(T_min, T * 0.92)

        if verbose and it % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad iter={it} T={T:.3f} ({elapsed:.0f}s): best={best_c:.8f}")

    # Final check of average
    if n_avg > 0:
        avg_check = avg_a / n_avg
        s_avg = avg_check.sum()
        if s_avg > 1e-15:
            avg_check = avg_check * S / s_avg
            score_avg = fast_autocorr(avg_check).max()
            if score_avg < best_c:
                best_c = score_avg
                best_a = avg_check.copy()

    return best_a, best_c


def nesterov_subgradient_multipeak(a0, time_limit, verbose=True, n_peaks=5):
    """Nesterov subgradient using average of top-n_peaks subgradients."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * S / n

    best_c = fast_autocorr(a).max()
    y = a.copy()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_autocorr(y)
        c_max = c.max()

        # Find top peaks
        top_idx = np.argsort(c)[::-1][:n_peaks]
        # Weighted subgradient
        weights = np.zeros(len(c))
        for idx in top_idx:
            weights[idx] = (c[idx] - c_max + 1.0) / n_peaks  # normalize
        weights = weights / weights.sum()

        # Compute combined subgradient
        grad = np.zeros(n)
        for k in top_idx:
            if weights[k] > 1e-10:
                grad += weights[k] * compute_subgrad_single(y, k)

        # Step
        step = 2.0 * S / (2.0 * S + it)
        a_new = y - step * grad
        a_new = project_simplex(a_new, S)

        alpha = 2.0 / (it + 2.0)
        y_new = (1.0 - alpha) * y + alpha * a_new
        if it == 1:
            a_curr = a_new.copy()
        else:
            mom = (it - 1.0) / (it + 2.0)
            a_curr = y_new + mom * (y_new - y)
            a_curr = project_simplex(a_curr, S)

        y = y_new
        a = a_curr

        c_a = fast_autocorr(a)
        if c_a.max() < best_c:
            best_c = c_a.max()
            best_a = a.copy()

        c_y = fast_autocorr(y)
        if c_y.max() < best_c:
            best_c = c_y.max()
            best_a = y.copy()

        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MultiPeak iter={it} ({elapsed:.0f}s): best={best_c:.8f}")

    return best_a, best_c


def build_conv_matrix(a):
    """Build convolution matrix T where (T @ b)[j] = conv(a,b)[j]."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        T[i:i + n, i] = a
    return T


def lp_best_response(a, M, solver, active_only=None):
    """Solve max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    if active_only is not None:
        T_use = T[active_only, :]
    else:
        T_use = T
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_use @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

            # Find which coordinates contribute most to conv[p]
            # conv[p] = sum_j a[j]*a[p-j]
            contrib = np.zeros(n)
            lo = max(0, p - (n - 1))
            hi = min(n - 1, p)
            for j in range(lo, hi + 1):
                contrib[j] = a[j] * a[p - j]

            # Try reducing high contributors
            top_contrib = np.argsort(contrib)[::-1][:5]
            for i in top_contrib:
                if a[i] < S * 1e-10:
                    continue
                # Move some mass from i to a random low position
                for _ in range(3):
                    j = rng.integers(0, n)
                    if j == i:
                        continue
                    frac = rng.uniform(0.1, 0.5)
                    delta = a[i] * frac
                    a_test = a.copy()
                    a_test[i] -= delta
                    a_test[j] += delta
                    a_test = a_test * S / a_test.sum()
                    c_test = fast_autocorr(a_test).max()
                    if c_test < best_score - 1e-13:
                        a = a_test
                        best_score = c_test
                        best_a = a.copy()
                        round_improved = True
                        break

        if not round_improved:
            no_improve_rounds += 1
            if no_improve_rounds >= 5:
                break
        else:
            no_improve_rounds = 0

        if verbose and no_improve_rounds % 10 == 10:
            elapsed = time.monotonic() - t0
            print(f"    PairDesc ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    return f_interp * np.sqrt(2.0 * new_n) / ss


def spectral_init(n, rng):
    """Initialize using spectral factorization idea.
    Target: |A(e^{iθ})|^2 ≈ S^2/(2n-1) * |sin(nθ)/sin(θ)|
    Approximate by taking the minimum-phase factor.
    """
    S = np.sqrt(2.0 * n)
    # Compute target spectrum on a fine grid
    n_fft = 4 * n
    theta = np.linspace(0, 2 * np.pi, n_fft, endpoint=False)
    # |sin(nθ)/sin(θ)|
    sin_n = np.abs(np.sin(n * theta))
    sin_1 = np.abs(np.sin(theta))
    ratio = np.where(sin_1 > 1e-10, sin_n / sin_1, 0.0)
    target = np.sqrt(ratio)

    # We want a polynomial A(z) of degree n-1 with |A(e^{iθ})| ≈ target
    # Use minimum phase: phase = Hilbert transform of log(target)
    from scipy.signal import hilbert
    # log spectrum
    log_target = np.log(target + 1e-10)
    # Hilbert transform for minimum phase
    # Actually, let's just do a simpler approach:
    # Use the fact that (1+z+...+z^{n-1}) has the desired magnitude profile
    # and modify it
    a = np.ones(n)
    # Add some structure to reduce peaks
    # Try: a[i] = 1 + alpha * cos(2*pi*i/n) + beta * cos(4*pi*i/n)
    alpha = rng.uniform(-0.5, 0.5)
    beta = rng.uniform(-0.3, 0.3)
    for i in range(n):
        a[i] = 1.0 + alpha * np.cos(2 * np.pi * i / n) + beta * np.cos(4 * np.pi * i / n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n)
    a = a * S / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function combining multiple optimization strategies."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1620.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-10:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: Fast Nesterov subgradient (50% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.50
    n1 = n_base
    print(f"\n=== Phase 1: Nesterov SubGrad n={n1} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n1) / best_seq.sum()
        a_out, _ = nesterov_subgradient(a_norm, t1, verbose=True, T_init=3.0, T_min=0.03)
        do_try_update(a_out, f"subgrad_n{n1}")

    # === Phase 2: LP Nesterov refinement (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t2 = remaining * 0.20
    print(f"\n=== Phase 2: LP Nesterov n={n_c} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t2, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n_c}")

    # === Phase 3: Second subgradient pass with different params (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t3 = remaining * 0.15
    print(f"\n=== Phase 3: SubGrad round 2 n={n_c} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        # Perturb slightly
        noise = rng.normal(0, 0.05 * a_norm.std(), n_c)
        a_pert = a_norm + noise
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert * np.sqrt(2.0 * n_c) / a_pert.sum()
        a_out, _ = nesterov_subgradient(a_pert, t3, verbose=True, T_init=1.0, T_min=0.01)
        do_try_update(a_out, f"subgrad2_n{n_c}")

    # === Phase 4: Coordinate descent (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t4 = remaining * 0.10
    print(f"\n=== Phase 4: Coordinate Descent n={n_c} (budget={t4:.0f}s) ===")
    if t4 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = coordinate_descent(a_norm, t4, verbose=True)
        do_try_update(a_out, f"cd_n{n_c}")

    # === Phase 5: Pair descent (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t5 = remaining * 0.05
    print(f"\n=== Phase 5: Pair Descent n={n_c} (budget={t5:.0f}s) ===")
    if t5 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = pair_descent_fast(a_norm, t5, rng, verbose=True)
        do_try_update(a_out, f"pair_n{n_c}")

    # === Phase 6: Multi-start with different initializations (remaining time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_c = len(best_seq)
        print(f"\n=== Phase 6: Multi-start (budget={remaining:.0f}s) ===")
        t_per_start = remaining / 4

        # Start 1: Spectral init
        if t_per_start > 20:
            a_spec = spectral_init(n_c, rng)
            a_out = nesterov_subgradient(a_spec, min(t_per_start, remaining * 0.25), verbose=False, T_init=5.0, T_min=0.1)[0]
            do_try_update(a_out, "spectral_init")
            remaining = time_budget - (time.monotonic() - t_global)

        # Start 2: Perturbed best
        if remaining > 30:
            a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
            noise = rng.normal(0, 0.3 * a_norm.std(), n_c)
            a_pert = a_norm + noise
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert * np.sqrt(2.0 * n_c) / a_pert.sum()
            a_out, _ = nesterov_subgradient(a_pert, min(remaining * 0.3, 120.0), verbose=False, T_init=2.0, T_min=0.05)
            do_try_update(a_out, "perturbed_start")
            remaining = time_budget - (time.monotonic() - t_global)

        # Start 3: Different length
        if remaining > 40:
            n_alt = n_c + 200
            a_alt = interpolate_sequence(best_seq, n_alt)
            a_out, _ = nesterov_subgradient(a_alt, min(remaining * 0.3, 120.0), verbose=False, T_init=3.0, T_min=0.05)
