def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = np.asarray(f, dtype=np.float64)
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_soft_subgradient(a, ac, top_k=40, beta=6.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def lp_direction(a, M_val, top_k, solver):
    """Solve LP to find Frank-Wolfe direction using top_k constraints."""
    n = len(a)
    ac = fast_autocorr(a)
    top_idx = np.argsort(ac)[::-1][:top_k]
    b_var = cp.Variable(n, nonneg=True)
    constraints = []
    for k in top_idx:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:

        i_range = np.arange(i_start, i_end)
        a_vals = a[k_int - i_range]
        constraints.append(a_vals @ b_var <= M_val + 1e-12)
    if not constraints:
        return None
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_k_start=30, top_k_max=80):
    """LP-based Frank-Wolfe with adaptive constraints."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    iter_count = 0
    current_top_k = top_k_start

    for t in range(1, 10000000):

        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        # Periodically increase top_k for more precise constraints

            current_top_k += 5

        b_dir = lp_direction(a, M_curr, current_top_k, solver)
        if b_dir is None:

        # Standard FW step: alpha = 2/(t+2)
        alpha = 2.0 / (t + 2.0)
        a_new = (1.0 - alpha) * a + alpha * b_dir
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:

        else:
            a_new = a_new / s_new

        a = a_new
        iter_count += 1

        M_new = fast_autocorr(a).max()

            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}, top_k={current_top_k}")

    return best_a, best_m

            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-Nest iter={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def amd_optimize(a0, time_limit, top_k=40, beta=5.0, c0=3.0, verbose=False):
    """Accelerated mirror descent with adaptive parameters."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(54321)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    max_stall = 4000
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_soft_subgradient(y, ac, top_k=cur_top_k, beta=cur_beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > max_stall:
            stall_count = 0
            pert = rng.normal(0, 0.008, n)
            x_r = best_a + pert * best_a
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            cur_top_k = min(cur_top_k + 10, 100)
            cur_beta = max(cur_beta - 0.8, 2.0)
            gamma = c0 * (1.0 + 0.1 * (cur_beta - 2.0))

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.1f}s): J={J:.6f}, tk={cur_top_k}, b={cur_beta:.1f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=42):
    """Aggressive coordinate descent with sensitivity analysis."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        num_peaks = min(25, len(ac))
        top_k = np.argsort(ac)[::-1][:num_peaks]
        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]

            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]

            # Compute sensitivity for moving mass away from src
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n - 1, int(k2))
                in_peak[s2:e2 + 1] = True

            # Find candidate destinations (outside peak regions preferred)
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 50:
                candidates = np.where(rng.random(n) < 0.5)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            chosen = candidates[rng.permutation(len(candidates))[:min(len(candidates), 200)]]

            best_dst = -1
            best_score = float('inf')

            for dst in chosen:
                if dst == src:
                    continue
                # Sensitivity: how much does moving mass to dst increase peaks?
                sens_add = 0.0
                for k2 in top_k[:10]:
                    k2i = int(k2)
                    d1 = k2i - dst
                    d2 = dst - k2i
                    add = 0.0
                    if 0 <= d1 < n:
                        add += a[d1]
                    if 0 <= d2 < n:
                        add += a[d2]
                    sens_add += add * 0.5

                # Reduction at source
                sens_rem = 0.0
                for k2 in top_k[:10]:
                    k2i = int(k2)
                    d1 = k2i - src
                    d2 = src - k2i
                    rem = 0.0
                    if 0 <= d1 < n:
                        rem += a[d1]
                    if 0 <= d2 < n:
                        rem += a[d2]
                    sens_rem += rem * 0.5

                net = sens_add - sens_rem
                if net < best_score:
                    best_score = net
                    best_dst = dst

            if best_dst < 0:
                continue

            dst = best_dst
            found = False
            # Try multiple fraction sizes adaptively
            fracs = [0.002, 0.005, 0.01, 0.02, 0.04, 0.06, 0.1]
            for frac in fracs:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found:
                continue

        if not improved:
            stall += 1
            # Random perturbation to escape
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.03)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            break
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            a = a / a.sum()

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search: LP FW + AMD + coordinate polish with multi-length exploration."""
    rng = np.random.default_rng(7)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1680.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent / parent.sum()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 2000
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: LP Frank-Wolfe (primary workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP FW n={n0} ===")
    t1 = min(remaining * 0.38, 600.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t1, solver, verbose=True, top_k_start=30, top_k_max=70)
        try_update(a_out, f"lpfw_n{n0}")

    # === Phase 2: LP Frank-Wolfe Nesterov ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP NW n={len(best_seq)} ===")
    t2 = min(remaining * 0.18, 280.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_nesterov(a_in, t2, solver, verbose=True, top_k=40)
        try_update(a_out, f"lpnw_n{len(best_seq)}")

    # === Phase 3: AMD with diverse params ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: AMD diverse n={len(best_seq)} ===")
    t3_total = min(remaining * 0.18, 280.0)
    if t3_total > 40:
        n_best = len(best_seq)
        a_in = norm_sum1(best_seq)
        configs = [
            (50, 4.0, 4.0),
            (30, 6.0, 3.0),
            (70, 3.0, 5.0),
        ]
        t_per = t3_total / len(configs)
        for idx, (tk, bt, c0v) in enumerate(configs):
            if time.monotonic() - t_global > time_budget - 60:
                break
            t3 = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t3 > 15:
                a_out, _ = amd_optimize(a_in, t3, top_k=tk, beta=bt, c0=c0v, verbose=False)
                try_update(a_out, f"amd{idx}_n{n_best}")
                a_in = norm_sum1(best_seq)  # Always continue from best

    # === Phase 4: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Coord n={len(best_seq)} ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t4, verbose=True, rng_seed=42)
        try_update(a_out, f"coord_n{len(best_seq)}")

    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Multi-length n={n_best} ===")
    if remaining > 200:
        lengths_to_try = []
        if n_best > 500:
            lengths_to_try.append(max(500, n_best - 300))
        lengths_to_try.append(n_best + 300)
        if n_best < 3000:
            lengths_to_try.append(n_best + 600)

        t_per = min(remaining * 0.12, 120.0)
        seen_lengths = set()
        for nl in lengths_to_try:
            if nl in seen_lengths or nl == n_best or time.monotonic() - t_global > time_budget - 80:
                continue
            seen_lengths.add(nl)
            t5 = min(t_per, time_budget - 80 - (time.monotonic() - t_global))
            if t5 < 15:
                continue
            print(f"  Trying n={nl}...")
            a_long = interpolate_sequence(best_seq, nl)
            a_out, _ = amd_optimize(norm_sum1(a_long), t5, top_k=40, beta=5.0, c0=3.0, verbose=False)
            try_update(a_out, f"amd_n{nl}")
            # Quick LP polish on the best of this length
            if time.monotonic() - t_global < time_budget - 50:
                a_out2, _ = lp_frank_wolfe(norm_sum1(best_seq), min(60.0, time_budget - 50 - (time.monotonic() - t_global)), solver, verbose=False)
                try_update(a_out2, f"lp_n{len(best_seq)}")

    # === Phase 6: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final coord n={n_best} ===")
    if remaining > 40:
        t6 = min(remaining * 0.5, 200.0)
        if t6 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t6, verbose=True, rng_seed=999)
            try_update(a_out, f"coordf_n{n_best}")

    # === Phase 7: Final LP if time permits ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP n={n_best} ===")
    if remaining > 30:
        t7 = min(remaining * 0.7, 150.0)
        if t7 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t7, solver, verbose=True, top_k_start=40, top_k_max=60)
            try_update(a_out, f"lpf_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
