def nesterov_lp(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best-response."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    t0 = time.monotonic()
    n_iters = 0
    stuck_count = 0
    patience = 0
    restart_count = 0
    momentum_t = 1.0
    
    # Build T matrix once (it changes as a changes, so we need to rebuild)
    # Actually T depends on a, so we need to rebuild each iteration
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        # Solve LP: max sum(b) s.t. conv(y, b) <= M_curr, b >= 0
        size = 2 * n - 1
        T = np.zeros((size, n))
        for j in range(n):
            T[j:j+n, j] = y
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            stuck_count += 1
            if stuck_count > 50:
                break
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            stuck_count += 1
            if stuck_count > 50:
                break
            continue

        stuck_count = 0
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g = b_val * norm / b_sum
        n_iters += 1

        # Nesterov
        alpha = 2.0 / (momentum_t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if momentum_t < 2.0:
            a_new = g.copy()
        else:
            momentum = (momentum_t - 1.0) / (momentum_t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new
        momentum_t += 1.0

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            patience = 0
        else:
            patience += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            patience = 0

        if patience > 50 and n_iters % 25 == 0:
            y = best_a.copy()
            a = best_a.copy()
            momentum_t = 1.0
            patience = 0
            restart_count += 1
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    NLP restart {restart_count} at iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

        if verbose and n_iters % 50 == 0:

def gerchberg_saxton(a0, time_limit, verbose=True):
    """Gerchberg-Saxton style spectral flattening with non-negativity constraint."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0
    patience = 0
    rng = np.random.default_rng(777)
    damping = 1.0  # How aggressively to flatten

    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        # FFT
        A = fft(a, n=2*n)  # Use zero-padded FFT
        power = np.abs(A)**2
        
        # Target: flat power spectrum
        mean_power = power.mean()
        
        # New amplitude: interpolate between current and flat
        target_amp = np.sqrt(mean_power)
        new_amp = damping * target_amp + (1 - damping) * np.abs(A)
        
        # Preserve phase
        phase = np.angle(A)
        new_A = new_amp * np.exp(1j * phase)
        
        # IFFT and take first n
        new_a = np.real(ifft(new_A, n=2*n))[:n]
        
        # Only consider non-negative part (or blend)
        new_a = np.maximum(new_a, 0)
        s_new = new_a.sum()
        if s_new < 1e-15:
            break
        new_a = new_a * norm / s_new
        
        score_new = fast_autocorr(new_a).max()
        if score_new < best_score:
            best_score = score_new
            best_a = new_a.copy()
            patience = 0
            a = new_a
        else:
            patience += 1
            # Blend with current
            a = 0.5 * a + 0.5 * new_a
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:
                a = a * norm / s
        
        n_iters += 1
        
        if patience > 100:
            damping = max(0.1, damping * 0.9)
            # Also add some noise
            noise = rng.normal(0, norm * 0.005, n)
            a = a + noise
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:
                a = a * norm / s
            patience = 0
        
        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GS iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, damping={damping:.3f}")

    return best_a, best_score

def compute_subgrad(a, active_idx):
    """Compute subgradient of max_k conv(a,a)[k] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    for k in active_idx:
        k = int(k)
        # d/d a[i] of conv(a,a)[k] = 2*a[k-i] if 0 <= k-i < n
        diff = k - np.arange(n)
        valid = (diff >= 0) & (diff < n)
        if valid.any():
            grad[valid] += 2.0 * a[diff[valid]]
    return grad

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    NLP restart {restart_count} iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, restarts={restart_count}")

    return best_a, best_score


def gerchberg_saxton(a0, time_limit, verbose=True):
    """Gerchberg-Saxton style: alternate between spectral flattening and non-negativity."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0
    patience = 0
    rng = np.random.default_rng(2024)
    damping = 0.5
    blend_factor = 0.3

    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        # Pad to 2n for FFT
        a_pad = np.zeros(2 * n)
        a_pad[:n] = a
        A = fft(a_pad)
        power = np.abs(A) ** 2

        # Target: flat power spectrum (proportional to mean)
        mean_power = power.mean()
        target_amp = np.sqrt(mean_power)

        # Interpolate amplitude towards flat
        current_amp = np.abs(A)
        new_amp = damping * target_amp + (1.0 - damping) * current_amp
        phase = np.angle(A)
        new_A = new_amp * np.exp(1j * phase)

        # IFFT and extract first n entries
        new_a_full = np.real(ifft(new_A))[:n]

        # Enforce non-negativity
        new_a = np.maximum(new_a_full, 0.0)
        s_new = new_a.sum()
        if s_new < 1e-15:
            break
        new_a = new_a * norm / s_new

        # Blend with current to avoid oscillation
        a_candidate = (1.0 - blend_factor) * a + blend_factor * new_a
        a_candidate = np.maximum(a_candidate, 0.0)
        s_cand = a_candidate.sum()
        if s_cand < 1e-15:
            a_candidate = new_a
        else:
            a_candidate = a_candidate * norm / s_cand

        score_cand = fast_autocorr(a_candidate).max()
        if score_cand < best_score:
            best_score = score_cand
            best_a = a_candidate.copy()
            patience = 0
            a = a_candidate
        else:
            # Try the pure new_a too
            if fast_autocorr(new_a).max() < best_score:
                best_score = fast_autocorr(new_a).max()
                best_a = new_a.copy()
                patience = 0
                a = new_a
            else:
                patience += 1
                a = a_candidate

        n_iters += 1

        # Adaptive: reduce damping and blend when stuck
        if patience > 50:
            damping = max(0.1, damping * 0.95)
            blend_factor = min(0.8, blend_factor * 1.05)
            if patience > 200:
                noise = rng.normal(0, norm * 0.01, n)
                a = a + noise
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 1e-15:
                    a = a * norm / s
                patience = 0

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GS iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, damp={damping:.3f}")

    return best_a, best_score


def compute_subgrad(a, active_idx):
    """Compute subgradient of max_k conv(a,a)[k] w.r.t. a at active indices."""
    n = len(a)
    grad = np.zeros(n)
    for k in active_idx:
        k = int(k)
        diff = k - np.arange(n)
        valid = (diff >= 0) & (diff < n)
        if valid.any():
            grad[valid] += 2.0 * a[diff[valid]]
    return grad


def mirror_descent(a0, time_limit, top_k=35, verbose=True):
    """Mirror descent (multiplicative weights) with top-k subgradient."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    eta = 1.0 / np.sqrt(n)
    n_iters = 0
    patience = 0
    rng = np.random.default_rng(31337)

    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(a)
        max_val = conv.max()
        k_actual = min(top_k, 2 * n - 1)
        active_idx = np.argsort(conv)[-k_actual:]
        grad = compute_subgrad(a, active_idx)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        log_a = np.log(a + 1e-30)
        best_a_try = a.copy()
        best_conv_max = max_val
        for factor in [0.2, 0.4, 0.6, 1.0, 1.5, 2.0]:
            eta_try = eta * factor
            log_a_try = log_a - eta_try * grad
            a_try = np.exp(log_a_try - log_a_try.max())
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try * norm / s_try
            conv_try = fast_autocorr(a_try).max()
            if conv_try < best_conv_max:
                best_conv_max = conv_try
                best_a_try = a_try
                eta = eta_try

        a = best_a_try
        n_iters += 1

        if best_conv_max < best_score - 1e-12:
            best_score = best_conv_max
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience % 40 == 0:
                eta *= 0.9

        if patience > 150:
            noise = rng.laplace(0, norm * 0.008, n)
            a_pert = a + noise
            a_pert = np.maximum(a_pert, 0)
            s_pert = a_pert.sum()
            if s_pert > 1e-15:
                a = a_pert * norm / s_pert
            patience = 0
            eta = 1.0 / np.sqrt(n)

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def coord_descent_polish(a0, time_limit, verbose=True):
    """Pair coordinate descent for fine-tuning."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(4242)
    no_improve = 0
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        if no_improve > 8:
            break
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-10:
                continue
            for offset in range(1, 40):
                found = False
                for j in [(i + offset) % n, (i - offset) % n]:
                    if j == i or a[i] < norm * 1e-10:
                        continue
                    for frac in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7]:
                        d = a[i] * frac
                        a[i] -= d
                        a[j] += d
                        s = a.sum()
                        if s < 1e-15:
                            a[i] += d
                            a[j] -= d
                            break
                        a_tmp = a * norm / s
                        new_max = fast_autocorr(a_tmp).max()
                        if new_max < best_score - 1e-14:
                            a = a_tmp
                            conv = fast_autocorr(a)
                            best_score = new_max
                            best_a = a.copy()
                            pass_improved = True
                            no_improve = 0
                            found = True
                            break
                        else:
                            a[i] += d
                            a[j] -= d
                    if found:
                        break
                if found:
                    break

        if not pass_improved:
            no_improve += 1

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1600
        a_init = np.ones(n_base) / n_base
        best_seq = a_init.copy()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LP Nesterov on parent (35%)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.35
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp1_n{n_base}")

    # Phase 2: Gerchberg-Saxton novel approach (15%)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = remaining * 0.15
    print(f"\n=== Phase 2: Gerchberg-Saxton n={n_best} (budget={t2:.0f}s) ===")
    if t2 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = gerchberg_saxton(a_norm, t2, verbose=True)
        do_try_update(a_out, f"gs_n{n_best}")

    # Phase 3: Mirror Descent (12%)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = remaining * 0.12
    print(f"\n=== Phase 3: Mirror Descent n={n_best} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = mirror_descent(a_norm, t3, top_k=40, verbose=True)
        do_try_update(a_out, f"md_n{n_best}")

    # Phase 4: LP Nesterov round 2 (22%)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = remaining * 0.22
    print(f"\n=== Phase 4: LP Nesterov R2 n={n_best} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t4, solver, verbose=True,
                               patience_thresh=30, restart_interval=15)
        do_try_update(a_out, f"lp2_n{n_best}")

    # Phase 5: Try larger n (8%)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        n_try = int(round(len(best_seq) * 1.5 / 100.0) * 100)
        t5 = remaining * 0.5
        print(f"\n=== Phase 5: LP Larger n={n_try} (budget={t5:.0f}s) ===")
        if t5 > 40:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a_interp = np.interp(x_new, x_old, best_seq)
            a_interp = np.maximum(a_interp, 0)
            s = a_interp.sum()
            if s > 1e-10:
                a_interp = a_interp / s
                a_norm = a_interp * np.sqrt(2.0 * n_try) / a_interp.sum()
                a_out, _ = nesterov_lp(a_norm, t5, solver, verbose=True)
                do_try_update(a_out, f"lp_n{n_try}")

    # Phase 6: GS on current best (5%)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        t6 = remaining * 0.5
        print(f"\n=== Phase 6: GS Polish n={n_best} (budget={t6:.0f}s) ===")
        if t6 > 15:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = gerchberg_saxton(a_norm, t6, verbose=True)
            do_try_update(a_out, f"gs2_n{n_best}")

    # Phase 7: Final CD polish (remaining)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        n_best = len(best_seq)
        t7 = remaining * 0.85
        print(f"\n=== Phase 7: CD Polish n={n_best} (budget={t7:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coord_descent_polish(a_norm, t7, verbose=True)
        do_try_update(a_out, f"cd_n{n_best}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
