"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed mirror descent (log-sum-exp smoothing of max autocorrelation)
2. LP best-response with full constraint set via scipy HiGHS
3. Focused coordinate descent on active autocorrelation peaks
4. Multi-restart exploration with diverse initializations
5. Palindromic symmetry exploitation for dimension reduction
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import cvxpy as cp

_global_solver = None

def _get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    S = a.sum()
    if S < 0.01:
        return float('inf')
    n = len(a)
    conv = fast_autocorr(a)
    return 2.0 * n * conv.max() / (S * S)

def smooth_max_grad(a, tau):
    """Compute gradient of (1/tau)*log(sum_k exp(tau*conv_k)) w.r.t. a."""
    n = len(a)
    size = 2 * n - 1
    conv = fast_autocorr(a)
    max_c = conv.max()
    exp_c = np.exp(tau * (conv - max_c))
    W = exp_c / exp_c.sum()
    # grad[m] = 2 * sum_j a[j] * W[j+m] for j+m < 2n-1
    # Use FFT: g[m] = conv(a, W_rev)[2n-2-m]
    W_rev = W[::-1]
    L = 1 << (n + size - 1 - 1).bit_length()
    A = rfft(a, L)
    WR = rfft(W_rev, L)
    G = irfft(A * WR, L)
    # g[m] = G[2n-2-m]
    grad = 2.0 * G[size - 1 - np.arange(n)]
    return grad, conv

def smooth_mirror_descent(a0, time_limit, tau_init=5.0, tau_final=500.0,
                          eta_init=0.5, verbose=True, seed=0):
    """Smoothed mirror descent on the simplex."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    n_iters = 0
    patience = 0
    max_patience = 3000

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        # Adaptive temperature: geometric from tau_init to tau_final
        frac = min(1.0, t / 5000.0)
        tau = tau_init * (tau_final / tau_init) ** frac

        # Adaptive step size
        eta = eta_init / (1.0 + 0.001 * t)

        grad, conv = smooth_max_grad(a, tau)
        G = conv.max() / S ** 2  # current max (proxy)

        # Mirror descent update: a_new = a * exp(-eta * grad / G) / sum(...)
        # Scale gradient by current objective for invariance
        grad_scaled = grad / max(G, 1e-10)
        log_step = -eta * grad_scaled
        # Clamp for numerical stability
        log_step = np.clip(log_step, -50.0, 50.0)
        a_new = a * np.exp(log_step)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        score = compute_score(a_new)
        if score < best_score - 1e-12:
            a = a_new
            best_score = score
            best_a = a_new.copy()
            patience = 0
            # Slightly increase step on success
            eta = min(eta * 1.005, 2.0)
        else:
            patience += 1
            eta *= 0.995
            if eta < 1e-6:
                break

        n_iters += 1
        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SMD iter={n_iters} ({elapsed:.0f}s): tau={tau:.1f}, eta={eta:.6f}, score={best_score:.8f}")

    return best_a, best_score

def lp_best_response_full(a_fixed, M, solver=None):
    """Solve LP: max sum(b) s.t. conv(a_fixed, b)[k] <= M for all k, b >= 0.
    Returns b or None."""
    n = len(a_fixed)
    size = 2 * n - 1
    # Build constraint matrix C: C[k, j] = a_fixed[k-j] if 0 <= k-j < n else 0
    C = np.zeros((size, n))
    for k in range(size):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        C[k, lo:hi] = a_fixed[k + 1 - hi:k + 1 - lo]

    c_obj = -np.ones(n)  # minimize -sum(b)
    A_ub = C
    b_ub = np.full(size, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            b = np.maximum(res.x, 0.0)
            if b.sum() > 1e-15:
                return b
    except Exception:
        pass
    return None

def lp_nesterov(a0, time_limit, solver=None, verbose=True, seed=0):
    """LP best-response with Nesterov acceleration using full constraint set."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / S

    best_a = a.copy()
    best_M = fast_autocorr(a).max()

    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()

        # Solve LP with full constraints
        b = lp_best_response_full(y, M_curr, solver)
        if b is None:
            # Try with slightly relaxed M
            b = lp_best_response_full(y, M_curr * 1.001, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.95)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y
        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_M - 1e-12:
            best_M = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_M - 1e-12:
            best_M = score_a
            best_a = a.copy()
            stall = 0

        if stall > 100:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            stall = 0

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): M={best_M:.8f}")

    return best_a, best_M

def active_peak_cd(a0, time_limit, verbose=True):
    """Coordinate descent focusing on the top active peaks."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    S = a.sum()
    if S < 1e-15:
        return a, float('inf')
    a = a * norm / S

    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        conv = fast_autocorr(a)
        M_curr = conv.max()
        improved = False

        # Focus on coordinates that contribute most to the top peaks
        top_k = np.argsort(conv)[-min(10, n // 20):]

        # Compute importance of each coordinate
        importance = np.zeros(n)
        for k in top_k:
            for j in range(n):
                i = k - j
                if 0 <= i < n:
                    importance[j] += a[i]

        # Process coordinates in order of importance
        order = np.argsort(-importance)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]
            if aj < 1e-12:
                continue

            # Try several candidate values
            s_others = a.sum() - aj
            candidates = [0.0, aj * 0.3, aj * 0.6, aj, aj * 1.4, aj * 1.8,
                          s_others * 0.05, s_others * 0.2]
            candidates = [max(0.0, min(x, norm)) for x in candidates]
            candidates = list(set(candidates))

            best_x = aj
            best_obj = 2.0 * n * M_curr / (a.sum() ** 2)

            for x in candidates:
                new_sum = s_others + x
                if new_sum < 1e-10:
                    continue
                # Quick estimate: only update the affected lags
                # Full conv is expensive, so use a partial check
                # Update conv at affected lags
                new_conv = conv.copy()
                for k in range(2 * n - 1):
                    i = k - j
                    if 0 <= i < n:
                        # old contribution: 2*aj*a[i] (or aj^2 if i==j)
                        old_c = 2.0 * aj * a[i] if i != j else aj * aj
                        # new contribution
                        new_c = 2.0 * x * a[i] if i != j else x * x
                        new_conv[k] += new_c - old_c
                obj = 2.0 * n * new_conv.max() / (new_sum * new_sum)
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_x = x

            if best_x != aj and abs(best_x - aj) > 1e-10:
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-10:
                    a = a * norm / s_new
                    new_M = fast_autocorr(a).max()
                    new_score = 2.0 * n * new_M / (a.sum() ** 2)
                    if new_score < best_M - 1e-12:
                        best_M = new_score
                        best_a = a.copy()
                        improved = True
                    else:
                        a[j] = aj
                        a = a * norm / a.sum()

        if n_passes % 5 == 0:
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    CD pass={n_passes} ({elapsed:.0f}s): M={best_M:.8f}")

        if not improved and n_passes > 3:
            break

    return best_a, best_M

def symmetrize_palindromic(a):
    """Make sequence palindromic: a[i] = a[n-1-i]."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    S = a_sym.sum()
    if S > 1e-15:
        a_sym = a_sym / S
    return a_sym

def perturb(a, rng, scale=0.02):
    noise = 1.0 + scale * rng.standard_normal(len(a))
    noise = np.maximum(noise, 0.0)
    a_p = a * noise
    S = a_p.sum()
    if S < 1e-15:
        return a.copy()
    return a_p / S

def sinusoidal_init(n, rng, seed=0):
    """Generate a sinusoidal-initialized sequence."""
    rrng = np.random.default_rng(seed)
    n_comp = rng.integers(2, 6)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for _ in range(n_comp):
        freq = rrng.uniform(0.5, 5.0)
        phase = rrng.uniform(0, 2 * np.pi)
        amp = rrng.uniform(0.1, 0.4)
        a += amp * np.cos(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return np.ones(n) / n
    return a / S

def cosh_window_init(n, seed=0):
    """Initialize with a cosh-shaped window modulated by cosine."""
    rrng = np.random.default_rng(seed)
    x = np.linspace(-1, 1, n)
    width = rrng.uniform(0.3, 1.0)
    a = np.cosh(x / width)
    freq = rrng.uniform(1, 4)
    a *= 0.5 + 0.5 * np.cos(2 * np.pi * freq * x)
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return np.ones(n) / n
    return a / S
