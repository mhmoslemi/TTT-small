import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver_name = None

def get_solver():
    global _solver_name
    if _solver_name is not None:
        return _solver_name
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver_name = s

        except Exception:

def fft_size_for(n):
    size = 2 * n - 1
    return 1 << (size - 1).bit_length()

def fast_conv_self(a):
    n = len(a)
    L = fft_size_for(n)
    A = rfft(a, L)
    return irfft(A * A, L)[:2*n-1]

def fast_conv_cross(a, b):
    n = len(a)
    L = fft_size_for(n)
    A = rfft(a, L)
    B = rfft(b, L)
    return irfft(A * B, L)[:2*n-1]

    # Find tau such that sum(max(0, y - tau)) = S
    # y - tau is decreasing in tau, so sum(max(0, y-tau)) is decreasing in tau
    # At tau = min(y): sum = sum(max(0, y - min(y))) >= sum(y) - n*min(y) >= S (hopefully)
    # At tau = max(y): sum = 0
    
    # Actually, we want sum(max(0, y_i - tau)) = S
    # This is equivalent to: find tau such that the "water level" tau gives total S
    
    y_min = y.min()
    y_max = y.max()
    
    # If sum(y) > S, we need to subtract some from all elements
    # If sum(y) < S, we need to add to some elements (but max(0, ...) can only reduce)
    # Wait, max(0, y - tau) for tau < min(y) gives y - tau, sum = sum(y) - n*tau
    # So if sum(y) > S, we set tau = (sum(y) - S) / n
    # If sum(y) <= S, we set tau = -infinity (i.e., a = y + alpha where alpha = (S - sum(y))/n)
    # But that's not a projection...
    
    # The correct projection onto {a >= 0, sum(a) = S} is:
    # a_i = max(0, y_i - tau) where tau is chosen so sum(a) = S
    # If sum(y) >= S: tau >= some value, a_i = max(0, y_i - tau)
    # If sum(y) < S: tau < min(y) (possibly negative), a_i = y_i - tau = y_i + |tau|
    #   Then sum(a) = sum(y) + n*|tau| = S, so |tau| = (S - sum(y))/n
    #   a = y + (S - sum(y))/n * ones
    #   This is valid as long as y_i + (S - sum(y))/n >= 0 for all i
    #   i.e., min(y) + (S - sum(y))/n >= 0
    
    if y.sum() <= S:
        # All of y is "below" the target sum, so we just add a constant
        a = y + (S - y.sum()) / len(y)
        if a.min() >= 0:
            return a
        # Otherwise, fall through to bisection
    
    # Bisection on tau
    lo = y.min() - S / len(y)  # At this tau, sum(max(0, y - tau)) >= S
    hi = y.max() + 1.0  # At this tau, sum = 0 < S
    
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.maximum(0.0, y - mid).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    
    tau = (lo + hi) / 2
    return np.maximum(0.0, y - tau)

            W = rfft(w, L)

def build_T(a):
    """Build the Toeplitz matrix T where (T @ b)[k] = (a * b)_k."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m+n, m] = a
    return T

def lp_solve_fast(a, M, solver, active=None):
    """Solve the LP: max sum(b) s.t. T(a)@b <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    if active is not None and 0 < len(active) < size:
        T_sub = T[active, :]
        constraints = [T_sub @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    
    # Get duals
    duals = None
    try:
        con = constraints[0]
        if con.dual_value is not None:
            if active is not None and len(active) < size:
                duals_full = np.zeros(size)
                duals_full[active] = np.asarray(con.dual_value, dtype=np.float64)
                duals = duals_full
            else:
                duals = np.asarray(con.dual_value, dtype=np.float64)
                duals = np.maximum(duals, 0.0)
    except Exception:
        pass
    
    return val, duals

def project_cn(y, SVal):
    """Euclidean projection of y onto {a >= 0, sum(a) = SVal}."""
    n = y.size

        return a
    lo = y.min() - SVal / n - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s_val = np.maximum(0.0, y - mid).sum()
        if s_val > SVal:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    out = np.maximum(0.0, y - tau)
    # Fix sum exactly
    diff = SVal - out.sum()
    if abs(diff) < 1e-12:
        return out
    mask = out > 0
