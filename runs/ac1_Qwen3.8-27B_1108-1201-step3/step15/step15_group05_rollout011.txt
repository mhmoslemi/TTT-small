def spectral_shaping(a, n_iters=100):
    n = len(a)
    fft_size = 1 << (2*n - 1).bit_length()
    S = np.sqrt(2.0 * n)
    a = a * S / a.sum()
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    for it in range(n_iters):
        A = np.fft.fft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        # Target: flat magnitude
        target_mag = S / np.sqrt(fft_size)  # normalization
        # Blend: new_mag = (1-alpha) * mag + alpha * target_mag
        alpha = 0.5
        new_mag = (1 - alpha) * mag + alpha * target_mag
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new * S / s
        score = fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_score

                    new_conv[2 * j] += mid * mid
                obj = 2.0 * n * new_conv.max() / (new_sum * new_sum)
                if obj < best_obj:
                    best_obj = obj
                    best_x = mid
                    lo = mid
                else:
                    hi = mid

            if best_x != aj and abs(best_x - aj) > 1e-12:
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * S / s_new
                    new_score = fast_autocorr(a).max()
                    if new_score < best_score - 1e-14:
                        best_score = new_score
                        best_a = a.copy()
                        improved = True
                    else:
                        a[j] = aj
                        a = a * S / a.sum()

        if not improved:
            break

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={no_improve_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def perturb_sequence(a, rng, scale=0.03):
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.1)
    a_pert = a * noise
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def smooth_sequence(a, sigma=2.0):
    """Apply Gaussian smoothing in the time domain."""
    n = len(a)
    kernel_size = int(2 * sigma * 3) | 1
    x = np.arange(kernel_size) - kernel_size // 2
    kernel = np.exp(-x ** 2 / (2 * sigma ** 2))
    kernel = kernel / kernel.sum()
    a_smooth = np.convolve(a, kernel, mode='same')
    a_smooth = np.maximum(a_smooth, 0.0)
    s = a_smooth.sum()
    if s < 1e-15:
        return a.copy()
    return a_smooth / s


def propose_candidate():
    """
    Main search: multi-phase optimization combining spectral shaping,
    LP best-response with Nesterov, multiplicative updates, and coordinate descent.
    """
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = 1750.0

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x)
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_base = len(best_seq)

    # === Phase 0: Spectral shaping for quick improvement ===
    remaining = time_budget - (time.monotonic() - t_global)
    t0_time = min(remaining * 0.05, 60.0)
    if t0_time > 5:
        print(f"\n=== Phase 0: Spectral shaping, budget={t0_time:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = spectral_shaping(a_norm, t0_time, verbose=False)
        try_update(a_out, "spectral")

    # === Phase 1: LP Nesterov (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t1 = min(remaining * 0.40, 600.0)
    if t1 > 60:
        print(f"\n=== Phase 1: LP Nesterov n={n_cur}, budget={t1:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, seed=0)
        try_update(a_out, "lp1")

    # === Phase 2: Spectral shaping again (from improved point) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.08, 80.0)
    if t2 > 10:
        print(f"\n=== Phase 2: Spectral shaping round 2, budget={t2:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = spectral_shaping(a_norm, t2, verbose=False)
        try_update(a_out, "spectral2")

    # === Phase 3: Subgradient polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 15:
        print(f"\n=== Phase 3: Subgradient polish n={n_cur}, budget={t3:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = subgradient_polish(a_norm, t3, verbose=True)
        try_update(a_out, "subg1")

    # === Phase 4: LP Nesterov round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.25, 300.0)
    if t4 > 30:
        print(f"\n=== Phase 4: LP Nesterov round 2 n={n_cur}, budget={t4:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True, seed=123)
        try_update(a_out, "lp2")

    # === Phase 5: Perturbation + LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5_total = min(remaining * 0.20, 200.0)
    if t5_total > 40:
        print(f"\n=== Phase 5: Perturbation restarts, budget={t5_total:.0f}s ===")
        n_restarts = 3
        t5_each = t5_total / n_restarts
        for r in range(n_restarts):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 40:
                break
            t5 = min(t5_each, remaining - 20)
            if t5 < 15:
                break
            if r == 0:
                pert = perturb_sequence(best_seq, rng, scale=0.015)
            elif r == 1:
                pert = smooth_sequence(best_seq, sigma=1.5)
            else:
                pert = perturb_sequence(best_seq, rng, scale=0.025)
            a_norm = pert * np.sqrt(2.0 * n_cur) / pert.sum()
            a_out, _ = nesterov_lp_optimize(a_norm, t5, solver, verbose=False, seed=300 + r * 7)
            try_update(a_out, f"pert{r}")

    # === Phase 6: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t6 = min(remaining * 0.25, 200.0)
    if t6 > 15:
        print(f"\n=== Phase 6: Coordinate descent n={n_cur}, budget={t6:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coordinate_descent(a_norm, t6, verbose=True)
        try_update(a_out, "cd1")

    # === Phase 7: Final subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 7: Final subgradient n={n_cur} ===")
        t7 = min(remaining * 0.5, 100.0)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = subgradient_polish(a_norm, t7, verbose=False)
        try_update(a_out, "subg_final")

    # === Phase 8: Final LP if time remains ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 8: Final LP n={len(best_seq)} ===")
        t8 = min(remaining * 0.6, 180.0)
        a_norm = best_seq * np.sqrt(2.0 * len(best_seq)) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t8, solver, verbose=True, seed=999)
        try_update(a_out, "lp_final")

    # === Phase 9: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print(f"\n=== Phase 9: Final CD ===")
        t9 = min(remaining * 0.7, 80.0)
        a_norm = best_seq * np.sqrt(2.0 * len(best_seq)) / best_seq.sum()
        a_out, _ = coordinate_descent(a_norm, t9, verbose=False)
        try_update(a_out, "cd_final")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
