```python
def amplify_subgradient(a, ac, threshold_frac=0.99):
    """Compute subgradient using lags near the maximum."""
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)
    
    # Active set: lags where ac[k] >= threshold_frac * M
    active = np.where(ac >= threshold_frac * M)[0]
    
    g = np.zeros(n)
    for k in active:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * a[j_idx]
    
    return g
```

```python
def project_simplex(v):
    """Project v onto {a >= 0, sum(a) = 1}."""
    n = len(v)
    # Binary search for threshold
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n+1) > (cssv[-1] - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)
```

```python
def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]

    except Exception:

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def exact_subgradient(a, ac, threshold_frac=0.995):
    """Compute subgradient using only lags near the maximum (active set)."""
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n), 1
    
    active = np.where(ac >= threshold_frac * M)[0]
    if len(active) == 0:
        active = np.array([int(ac.argmax())])
    
    g = np.zeros(n)
    for k in active:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * a[j_idx]
    
    # Normalize to unit l2 norm for consistent step size interpretation
    gn = np.linalg.norm(g)
    if gn > 1e-15:
        g = g / gn
    return g, len(active)


def soft_subgradient(a, ac, top_k=30, beta=4.0):
    """Soft subgradient using top-k lags with power weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad_contrib = 2.0 * wi * a[j_idx]
        g[idx] += grad_contrib
    return g


def mirror_update(y, g, eta):
    """Mirror descent update in log-space."""
    log_a = np.log(np.maximum(y, 1e-15)) - eta * g
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return y.copy()
    return a_new / s


def amd_exact(a0, time_limit, c0=3.0, threshold_frac=0.995, verbose=True, rng_seed=42):
    """
    Accelerated Mirror Descent with EXACT active-set subgradient.
    Key improvement: uses exact subgradient from lags within threshold_frac of max.
    """
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall_count = 0
    max_stall = 4000
    current_threshold = threshold_frac
    c_current = c0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute autocorrelation and exact subgradient
        ac = fast_autocorr(y)
        g, n_active = exact_subgradient(y, ac, threshold_frac=current_threshold)

        # Step size: adaptive with scale
        eta = c_current / (np.sqrt(t) + 1.0) * np.sqrt(n)

        # Mirror update
        a_new = mirror_update(y, g, eta)

        x_prev = x
        x = a_new

        # Evaluate
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Adaptive: adjust threshold and restart if stuck
        if stall_count > max_stall:
            stall_count = 0
            # Perturb from best
            noise_scale = 0.005 + 0.01 * np.sqrt(stall_count / max_stall)
            perturbation = rng.normal(0, noise_scale, n) * best_a
            x_restarted = best_a + perturbation
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted
            # Adapt parameters
            if current_threshold > 0.98:
                current_threshold = max(0.98, current_threshold - 0.005)
            c_current = c_current * 1.2
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    AMD restart t={t} ({elapsed:.1f}s): thresh={current_threshold:.4f}, c={c_current:.2f}")

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.1f}s): J={J:.6f}, active={n_active}, stall={stall_count}")

    return best_a, best_m

def lp_frank_wolfe(a0, time_limit, solver, verbose=True, max_iters=50):
    """LP-based Frank-Wolfe direction (simplified, fewer iterations)."""
    n = len(a0)
    if n > 2000:
        # Too large for LP, fall back to AMD
        return amd_exact(a0, time_limit, c0=3.0, verbose=verbose, rng_seed=99)

    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        # Build constraint matrix
        b_var = cp.Variable(n, nonneg=True)
        # conv(y, b) <= M_curr for all lags
        # conv(y,b)[k] = sum_i y[i] * b[k-i] (valid indices)
        # This is (T_k @ b) <= M_curr for each k
        # T_k[i,j] = y[i] if j = k - i (and valid)
        
        constraints = []
        # Only use top lags to keep LP small
        top_lags = np.argsort(ac)[::-1][:min(50, len(ac))]
        
        for k in top_lags:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            # conv[k] = sum_{i in idx} y[i] * b[j_idx[i-i_start]]
            # This is a linear form in b: sum coeff_j * b[j]
            coeff = np.zeros(n)
            for ii, j in zip(idx, j_idx):
                coeff[j] += y[ii]
            constraints.append(coeff @ b_var <= M_curr)

        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    Strategy:
    1. Load parent, quick AMD to get baseline
    2. Extended AMD with exact subgradient (main workhorse)
    3. Coordinate descent polish
    4. Multi-length exploration
    5. Final polish
    """
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        # Use a structured initialization
        best_seq = np.sin(np.pi * x) ** 2 + 0.1 * np.cos(2 * np.pi * x) ** 2
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Quick AMD to establish baseline ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: AMD exact n={n0} ===")
    t1 = min(remaining * 0.25, 300.0)
    if t1 > 20:
        a_out, _ = amd_exact(norm_sum1(best_seq), t1, c0=3.0, threshold_frac=0.99, verbose=True, rng_seed=1)
        try_update(a_out, f"amd_exact_n{n0}")

    # === Phase 2: Extended AMD with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: AMD exact (longer) n={n_best} ===")
    t2 = min(remaining * 0.25, 350.0)
    if t2 > 30:
        a_out, _ = amd_exact(norm_sum1(best_seq), t2, c0=4.0, threshold_frac=0.995, verbose=True, rng_seed=2)
        try_update(a_out, f"amd_exact2_n{n_best}")

    # === Phase 3: AMD with soft subgradient (different landscape view) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: AMD soft n={n_best} ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        a_out, _ = amd_soft(norm_sum1(best_seq), t3, c0=2.5, top_k=50, beta=4.0, verbose=True, rng_seed=3)
        try_update(a_out, f"amd_soft_n{n_best}")

    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coord descent n={n_best} ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 15:
        a_out, _ = coordinate_descent(norm_sum1(best_seq), t4, verbose=True, rng_seed=4)
        try_update(a_out, f"coord_n{n_best}")

    # === Phase 5: Pairwise transfer ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Pairwise n={n_best} ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 15:
        a_out, _ = pairwise_transfer_descent(norm_sum1(best_seq), t5, verbose=True, rng_seed=5)
        try_update(a_out, f"pairwise_n{n_best}")

    # === Phase 6: AMD again (from improved point) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: AMD polish n={n_best} ===")
    t6 = min(remaining * 0.15, 180.0)
    if t6 > 15:
        a_out, _ = amd_exact(norm_sum1(best_seq), t6, c0=2.0, threshold_frac=0.999, verbose=True, rng_seed=6)
        try_update(a_out, f"amd_polish_n{n_best}")

    # === Phase 7: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 120:
        print(f"\n=== Phase 7: Multi-length ===")
        # Try n+400
        n_try = min(n_best + 400, 3000)
        if n_try > n_best:
            print(f"  Trying n={n_try}")
            a_long = interpolate_sequence(best_seq, n_try)
            t7a = min(remaining * 0.25, 150.0)
            if t7a > 30:
                a_out, _ = amd_exact(norm_sum1(a_long), t7a, c0=3.0, threshold_frac=0.99, verbose=True, rng_seed=7)
                try_update(a_out, f"amd_long_n{n_try}")

        remaining = time_budget - (time.monotonic() - t_global)
        # Try n-400 (sometimes shorter is better)
        n_best_now = len(best_seq)
        n_short = max(n_best_now - 400, 800)
        if remaining > 60 and n_short > 800:
            print(f"  Trying n={n_short}")
            a_short = interpolate_sequence(best_seq, n_short)
            t7b = min(remaining * 0.20, 100.0)
            if t7b > 20:
                a_out, _ = amd_exact(norm_sum1(a_short), t7b, c0=3.0, threshold_frac=0.99, verbose=True, rng_seed=8)
                try_update(a_out, f"amd_short_n{n_short}")

    # === Phase 8: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final coord n={n_best} ===")
    if remaining > 20:
        t8 = min(remaining * 0.30, 120.0)
        if t8 > 10:
            a_out, _ = coordinate_descent(norm_sum1(best_seq), t8, verbose=True, rng_seed=9)
            try_update(a_out, f"coord_final_n{n_best}")

    # === Phase 9: Final AMD sub-polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        t9 = min(remaining * 0.40, 100.0)
        if t9 > 10:
            a_out, _ = amd_exact(norm_sum1(best_seq), t9, c0=1.5, threshold_frac=0.999, verbose=False, rng_seed=10)
            try_update(a_out, f"amd_subpolish_n{n_best}")

    # === Phase 10: LP if small enough ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if n_best <= 1800 and remaining > 30:
        print(f"\n=== Phase 10: LP n={n_best} ===")
        t10 = min(remaining * 0.50, 100.0)
        if t10 > 15:
            a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t10, solver, verbose=True, max_iters=30)
            try_update(a_out, f"lp_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
```
