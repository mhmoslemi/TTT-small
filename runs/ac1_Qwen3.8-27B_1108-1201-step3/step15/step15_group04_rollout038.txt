def fast_conv_self(a):
    """Compute a * a (full linear convolution) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    """Compute a * b (full linear convolution) using FFT."""

        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):

    n = len(a)
    S = a.sum()
    best_a = a.copy()
    best_peak = fast_conv_self(a).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        improved = False
        perm = rng.permutation(n)
        
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            # Find best delta for coordinate i
            # delta_c[k] = 2*delta*a[k-i] + delta^2*(k==2i)
            # We want to minimize max_k (conv[k] + delta_c[k])
            # subject to a[i] + delta >= 0 and keeping sum fixed (transfer to some j)
            
            # For simplicity, transfer to a random j
            j = int(rng.integers(n))
            if j == i:
                continue
            
            # Change: a[i] -> a[i] - delta, a[j] -> a[j] + delta
            # Delta c[k]:
            # -2*delta*a[k-i] for k in [i, i+n-1] (removal from i)
            # +2*delta*a[k-j] for k in [j, j+n-1] (addition to j)
            # +delta^2 at k=2j (self-term for j)
            # -delta^2 at k=2i (self-term for i, but this is -delta*(-delta) = ... )
            # Actually: (a[i]-delta)^2 vs a[i]^2: change is -2*delta*a[i] + delta^2
            # And: (a[j]+delta)^2 vs a[j]^2: change is +2*delta*a[j] + delta^2
            
            # This is getting complicated. Let me just use a simple grid search.
            
            ai = a[i]
            if ai < 1e-10:
                continue
            
            best_delta = 0
            best_val = cur_max
            
            for frac in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                delta = frac * ai
                # Try transferring from i to j
                a[i] -= delta
                a[j] += delta
                new_conv = fast_conv_self(a)
                new_max = new_conv.max()
                if new_max < best_val:
                    best_val = new_max
                    best_delta = delta
                # Revert
                a[i] += delta
                a[j] -= delta

                a[i] -= best_delta
                a[j] += best_delta
                a = a * S / a.sum()  # renormalize (should be same sum)
                if best_val < best_peak:
                    best_peak = best_val
                    best_a = a.copy()
                    improved = True

def fast_conv_self(a):

    size = 2 * n - 1

    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):

    A = rfft(a, fft_size)
    B = rfft(b, fft_size)

        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm_val * 1e-8:
                continue

            # Select candidate j's
            candidates = set()
            peak_lags = np.where(conv >= cur_max * 0.88)[0]
            for _ in range(4):
                candidates.add(int(rng.integers(n)))
            for pl in peak_lags[:6]:
                j1 = pl - i
                j2 = i + n - 1 - pl
                if 0 <= j1 < n:
                    candidates.add(j1)
                if 0 <= j2 < n:
                    candidates.add(j2)
            candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in list(candidates)[:6]:
                if j == i:
                    continue
                # Compute gain vector incrementally
                gain = np.zeros(size)
                lo_j = j
                hi_j = min(size, j + n)
                if hi_j > lo_j:
                    gain[lo_j:hi_j] += 2.0 * a[:hi_j - lo_j]
                lo_i = i
                hi_i = min(size, i + n)
                if hi_i > lo_i:
                    gain[lo_i:hi_i] -= 2.0 * a[:hi_i - lo_i]

                self_term = np.zeros(size)
                if 2 * i < size:
                    self_term[2 * i] = -1.0
                if 2 * j < size:
                    self_term[2 * j] = 1.0

                # Try several delta values
                for frac in [0.05, 0.1, 0.15, 0.25, 0.4, 0.6, 0.8]:
                    d = frac * ai
                    new_conv = conv + d * gain + d * d * self_term
                    new_max = new_conv.max()
                    if new_max < best_val - 1e-12:
                        best_val = new_max
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = np.maximum(a, 0)
                a = a * norm_val / a.sum()
                conv = fast_conv_self(a)
                if conv.max() < best_peak:
                    best_peak = conv.max()
                    best_a = a.copy()
                pass_improved = True

        if pass_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 6:
                break
            if no_improve == 3:
                n_pert = max(15, n // 20)
                idx_p = rng.choice(n, size=n_pert, replace=False)
                a[idx_p] *= (1.0 + 0.12 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0)
                a = a * norm_val / a.sum()
                conv = fast_conv_self(a)
                no_improve = 0

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"      CD ({elapsed:.0f}s): peak={best_peak:.8f}")

    return best_a, best_peak

def structured_init(n, rng):
    choice = rng.integers(0, 8)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = 0.5 + 0.3 * (x - 0.5) + 0.1 * np.cos(6 * np.pi * x)
        a = np.maximum(a, 0.05)
    elif choice == 1:
        a = np.sin(np.pi * x * 0.8 + 0.2) ** 1.2 + 0.1
    elif choice == 2:
        a = 0.4 + 0.4 * x + 0.1 * np.cos(10 * np.pi * x)
        a = np.maximum(a, 0.05)
    elif choice == 3:
        a = np.exp(-0.5 * ((x - 0.3) / 0.2) ** 2) + 0.7 * np.exp(-0.5 * ((x - 0.7) / 0.2) ** 2) + 0.1
    elif choice == 4:
        a = 0.3 + 0.5 * np.sin(np.pi * x) * (0.5 + 0.5 * x)
    elif choice == 5:
        a = (0.3 + 0.7 * x) * (1.0 + 0.15 * np.cos(8 * np.pi * x))
        a = np.maximum(a, 0.05)
    elif choice == 6:
        a = 0.5 + 0.3 * np.sin(2 * np.pi * x * 2 + 0.5) + 0.2 * x
        a = np.maximum(a, 0.05)
    else:
        a = 0.4 + 0.3 * x + 0.2 * np.cos(np.pi * x * 5)
        a = np.maximum(a, 0.05)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_best_solver()
    print(f"Budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_local(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    # Initialize if needed
    if best_seq is None:
        n_base = 1600
        best_init = None
        best_init_sc = float('inf')
        for tr in range(8):
            cand = structured_init(n_base, rng)
            sc = eval_local(cand)
            if sc < best_init_sc:
                best_init_sc = sc
                best_init = cand
        best_seq = best_init
        best_score = best_init_sc
        print(f"Init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # Phase 1: LP best response (main optimization)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 550.0)
    print(f"\n=== Phase 1: LP Best Response n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = lp_best_response_iteration(best_seq, 1.0, solver, t1, verbose=True)
        try_update(a_out, f"lp1_n{n_base}")

    # Phase 2: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 2: Coord Descent n={n_base} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = coordinate_descent(best_seq, 1.0, t2, rng, verbose=False)
        try_update(a_out, f"cd1_n{n_base}")

    # Phase 3: Second LP round from improved point
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.2, 300.0)
    print(f"\n=== Phase 3: LP Round 2 n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = lp_best_response_iteration(best_seq, 1.0, solver, t3, verbose=True)
        try_update(a_out, f"lp2_n{len(best_seq)}")

    # Phase 4: CD polish again
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 180.0)
    print(f"\n=== Phase 4: CD Round 2 (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = coordinate_descent(best_seq, 1.0, t4, rng, verbose=False)
        try_update(a_out, f"cd2_n{len(best_seq)}")

    # Phase 5: Try longer sequence
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = min(n_base + 400, 2400)
    print(f"\n=== Phase 5: Long sequence n={n_long} (remaining={remaining:.0f}s) ===")
    if remaining > 150:
        init_long = interpolate_to_length(best_seq, n_long)
        t5a = min(remaining * 0.5, 200.0)
        if t5a > 30:
            a_out, _ = lp_best_response_iteration(init_long, 1.0, solver, t5a, verbose=False)
            try_update(a_out, f"lp_long_{n_long}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and len(best_seq) == n_long:
            t5b = min(remaining * 0.5, 100.0)
            a_out, _ = coordinate_descent(best_seq, 1.0, t5b, rng, verbose=False)
            try_update(a_out, f"cd_long_{n_long}")

    # Phase 6: Final LP + CD
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Final polish (remaining={remaining:.0f}s) ===")
    if remaining > 60:
        t6a = min(remaining * 0.4, 100.0)
        if t6a > 20:
            a_out, _ = lp_best_response_iteration(best_seq, 1.0, solver, t6a, verbose=False)
            try_update(a_out, "lp_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t6b = min(remaining * 0.6, 80.0)
            a_out, _ = coordinate_descent(best_seq, 1.0, t6b, rng, verbose=False)
            try_update(a_out, "cd_final")

    # Phase 7: Alternative initialization (if time allows)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120 and best_score > 1.5035:
        print(f"\n=== Phase 7: Alt init (remaining={remaining:.0f}s) ===")
        for tr in range(3):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            alt = structured_init(n_base, rng)
            t7 = min(remaining * 0.3, 100.0)
            if t7 > 20:
                a_out, _ = lp_best_response_iteration(alt, 1.0, solver, t7, verbose=False)
                try_update(a_out, f"alt{tr}_n{n_base}")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_local(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0, 1.0, 1.0]

    return [float(x) for x in best_seq]
