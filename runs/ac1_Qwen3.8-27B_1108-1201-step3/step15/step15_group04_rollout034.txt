If we parameterize A(f) = |A(f)| * exp(i*phi(f)), the autocorrelation depends only on |A(f)|, not on phi(f). So phase doesn't directly help.

But the constraint that a_i >= 0 is in the time domain. In the frequency domain, it's:
a_i = (1/N) sum_f |A(f)| * exp(i*phi(f)) * exp(2πi f i / N) >= 0

This is a complicated set of linear inequalities in the frequency domain (linear in A(f) = real + i*imag parts).

So the problem in the frequency domain is:
minimize max(ifft(|A|^2))
subject to ifft(A) >= 0, sum(ifft(A)) = S

This is still non-convex (minimize max of a convex function of |A|^2), but it's a different formulation that might be easier to optimize.

    A = rfft(a, fft_size)

    """Compute cross-convolution a*b using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def lse_grad(a, T):
    """Compute LSE-smoothed max of a*a and its gradient.
    Returns (lse_value, gradient).
    """
    n = len(a)
    conv = fast_conv_self(a)
    m = conv.max()
    shifted = (conv - m) / T
    shifted = np.clip(shifted, -50, 0)
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()
    
    size = 2 * n - 1
    # N must be >= (2n-1) + n - 1 = 3n - 2 for linear convolution
    N = 1 << (3 * n - 2).bit_length()

    A = rfft(a, N)

    full_corr = irfft(W * np.conj(A), N)[:n]
    
    # Convolution: conv_aw[i] = sum_j a[j] * w[i-j]
    conv_aw = irfft(A * W, N)[:n]
    
    grad = full_corr + conv_aw
    lse_val = m + T * np.log(exp_vals.sum())
    
    return lse_val, grad

    """Project a onto {x >= 0, sum(x) = S}."""
    a = np.maximum(a, 0)
    s = a.sum()

            best_conv_max = sc_a
            best_a = a.copy()
            improved = True
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
        if no_improve == 150:
            n_p = max(10, n // 25)
            idx_p = np.random.default_rng(t).choice(n, size=n_p, replace=False)
            y[idx_p] *= (1.0 + 0.15 * np.random.default_rng(t + 7).standard_normal(n_p))
            y = np.maximum(y, 0)
            sy2 = y.sum()
            if sy2 > 1e-15:
                y = y * S / sy2
            no_improve = 0
        avg += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg / n_avg
            savg = avg_check.sum()
            if savg > 1e-15:
                avg_check = avg_check * S / savg
                sc_avg = fast_conv_self(avg_check).max()
                if sc_avg < best_conv_max:
                    best_conv_max = sc_avg
                    best_a = avg_check.copy()
        if verbose and t % 100 == 0:
            print(f"    LP t={t} ({time.monotonic()-t_g:.0f}s): obj={2*n*best_conv_max/(S*S):.8f}")
    return best_a, 2 * n * best_conv_max / (S * S)
