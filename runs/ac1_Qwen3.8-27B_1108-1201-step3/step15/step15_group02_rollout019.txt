                    # a_{k+i} for i in [0, min(n, k+1))
                    # Actually, the gradient of b_k wrt a_i is:
                    # b_k = sum_j a_j * a_{k-j}
                    # db_k/d(a_i) = a_{k-i} (if 0<=k-i<n) + a_{k+i} (if 0<=k+i<n and i != k-i)
                    # Wait, for self-convolution: b_k = sum_j a_j * a_{k-j}
                    # db_k/d(a_i) = a_{k-i} + a_{i-k} (if both in range)
                    # But a_{i-k} = a_{k-i} only if the sequence is symmetric...
                    # No, a_{i-k} is just a at index (i-k) if 0 <= i-k < n
                    # So db_k/d(a_i) = [a_{k-i} if 0<=k-i<n] + [a_{i-k} if 0<=i-k<n]
                    # = [a_{k-i} if 0<=k-i<n] + [a_{i-k} if 0<=i-k<n]
                    
                    # For the first term: k-i in [0,n) means i in (k-n, k], i.e., [max(0,k-n+1), min(n-1,k)]
                    # For the second term: i-k in [0,n) means i in [k, k+n-1], i.e., [max(0,k), min(n-1,k+n-1)]
                    
                    # Let me just compute it properly
                    pass
            
            # Simpler approach: use FFT to compute the gradient
            # The gradient of sum_k w_k * b_k wrt a_i is:
            # sum_k w_k * (a_{k-i} + a_{i-k})
            # = sum_k w_k * a_{k-i} + sum_k w_k * a_{i-k}
            # The first sum is (w * a)[i] (cross-correlation)
            # The second sum is (w * a)[i] reversed... 
            # Actually, let's just use the FFT approach from before.
            
            # Build the weight vector for the full convolution
            w_full = np.zeros(2 * n - 1)
            for k in top_K:
                w_full[k] = weights[top_K.tolist().index(k)]
            
            # Gradient of sum_k w_full[k] * b_k wrt a:
            # = 2 * (w_full circularly convolved with a, truncated)
            # More precisely, using the same FFT trick:
            fft_size = 1 << (2 * n - 2).bit_length()
            W = rfft(w_full, fft_size)
            A = rfft(a, fft_size)
            ca = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)[:n]
            pressure = ca + aw  # This is grad / 2
            
            # Multiplicative update
            a = a * np.exp(-eta * pressure)
            a = np.maximum(a, 1e-15)
            a = a * norm / a.sum()
            
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if si % 5 == 0:
            print(f"    PeakShave stage {si} eta={eta:.5f} K={K}: best={best_score:.8f}")
    
    return best_a, best_score

def nesterov_subgradient(a0, time_limit, K=3):
    """Nesterov-accelerated subgradient on top K lags."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    fft_size = 1 << (2 * n - 2).bit_length()
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        
        # Top K lags with weights proportional to excess
        top_idx = np.argsort(conv_y)[-K:]
        excess = conv_y[top_idx] - conv_y.mean()
        excess = np.maximum(excess, 0.1 * M)  # ensure positive
        w_top = excess / excess.sum()
        
        # Build weight vector
        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_idx):
            w_full[k] = w_top[i]
        
        # Compute gradient
        W = rfft(w_full, fft_size)
        A = rfft(y, fft_size)
        ca = irfft(np.conj(A) * W, fft_size)[:n]
        aw = irfft(A * W, fft_size)[:n]
        grad = ca + aw
        
        # Normalize gradient
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        grad = grad / g_norm
        
        # Step size
        eta = 2.0 / (t + 2.0)  # Nesterov step size
        # Scale by current "radius"
        eta = eta * norm / np.sqrt(n)
        
        y_new = y - eta * grad
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            x_new = y_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            x_new = y_new + momentum * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new * norm / s_x
        
        y = y_new
        
        # Check both y and x
        for candidate in [y, x_new]:
            sc = fast_conv_self(candidate).max()
            if sc < best_score:
                best_score = sc
                best_a = candidate.copy()
        
        if t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NFSUB t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def pair_cd_optimize(a0, time_limit, rng, n_trials=5):
    """Pair coordinate descent with subgradient-informed candidate selection."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue
            
            # Candidate j's: random + gradient-informed
            peak_lags = np.where(conv >= cur_max * 0.8)[0]
            
            j_candidates = set()
            for _ in range(n_trials):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:15]:
                j_candidates.add(int((kl - i + n) % n))
                j_candidates.add(int((kl + i) % n))
                # Also try neighbors
                j_candidates.add(int(np.clip(kl - i - 1, 0, n-1)))
                j_candidates.add(int(np.clip(kl - i + 1, 0, n-1)))
                j_candidates.add(int(np.clip(kl + i - 1, 0, n-1)))
                j_candidates.add(int(np.clip(kl + i + 1, 0, n-1)))
            j_candidates.discard(i)
            
            best_delta = 0.0
            best_v = cur_max
            best_j = -1
            
            for j in list(j_candidates)[:30]:  # limit candidates
                if j == i:
                    continue
                aj = a[j]
                
                # Compute delta_conv for moving d from i to j
                # new_conv = conv - d*(a shifted by i) - d*(a shifted by i, reversed)
                #                    + d*(a shifted by j) + d*(a shifted by j, reversed)
                # + correction at 2i and 2j
                g = np.zeros(size)
                
                # -d * a shifted right by i (contribution of a_i to lags i, i+1, ..., i+n-1)
                lo = i
                hi = min(size, i + n)
                if hi > lo:
                    g[lo:hi] -= a[0:hi-lo]
                
                # -d * a shifted left by i (contribution of a_i to lags 0, 1, ..., i)
                # a_{k-i} for k in [0, i]: a_{k-i} = a[n-(i-k)] ... 
                # Actually, the contribution of a_i to b_k (for k < i) is a_{k-i} which doesn't exist
                # The contribution of a_i to b_k is: a_i * a_{k-i} if 0<=k-i<n, plus a_i * a_{i-k} if 0<=i-k<n
                # Wait, I need to be more careful.
                
                # b_k = sum_j a_j * a_{k-j}
                # The terms involving a_i are: a_i * a_{k-i} (j=i) + a_{k-i} * a_i (j=k-i, if k-i != i)
                # So db_k/d(a_i) = a_{k-i} + a_{k+i} (where the second term is a_{i-k} = a_{k+i-n}... no)
                
                # Let me just use the direct formula:
                # If we change a_i by -d and a_j by +d:
                # new b_k = old b_k - d*(a_{k-i} + a_{i-k}) + d*(a_{k-j} + a_{j-k})
                # where a_x = 0 if x not in [0,n)
                
                lo2 = max(0, i - n + 1)
                hi2 = min(size, i + 1)
                if hi2 > lo2:
                    for k in range(lo2, hi2):
                        idx1 = k - i
                        idx2 = i - k
                        val = 0.0
                        if 0 <= idx1 < n:
                            val += a[idx1]
                        if 0 <= idx2 < n and idx2 != idx1:
                            val += a[idx2]
                        g[k] -= val
                
                lo3 = j
                hi3 = min(size, j + n)
                if hi3 > lo3:
                    g[lo3:hi3] += a[0:hi3-lo3]
                
                lo4 = max(0, j - n + 1)
                hi4 = min(size, j + 1)
                if hi4 > lo4:
                    for k in range(lo4, hi4):
                        idx1 = k - j
                        idx2 = j - k
                        val = 0.0
                        if 0 <= idx1 < n:
                            val += a[idx1]
                        if 0 <= idx2 < n and idx2 != idx1:
                            val += a[idx2]
                        g[k] += val
                
                # Quadratic corrections
                if 2 * i < size:
                    g[2*i] += ai  # d^2 term coefficient (but we're doing linear approx + correction)
                if 2 * j < size:
                    g[2*j] += aj
                
                # Search for best d in [0, 0.9*ai]
                d_max = ai * 0.9
                n_samples = 15
                deltas = np.linspace(0.02 * d_max, d_max, n_samples)
                
                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2*i] = conv[2*i] - 2*ai*d + d*d
                    if 2 * j < size:
                        new_conv[2*j] = conv[2*j] + 2*aj*d + d*d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_delta = d
                        best_j = j
            
            if best_j >= 0 and best_delta > 1e-10 and best_v < best_score - 1e-12:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                n_perturb = max(10, n // 50)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.1 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                if no_improve >= 12:
                    break
        else:
            no_improve = 0
    
    return best_a, best_score

def sos_optimize(a0, time_limit):
    """
    Sum-of-Squares inspired optimization.
    Key insight: if a = sum of shifted copies of a "base" pattern,
    the autocorrelation has structure we can exploit.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    
    # Idea: parameterize a as a smooth function and optimize parameters
    # a_i = max(0, c0 + c1*cos(2*pi*f1*i/n + p1) + c2*cos(2*pi*f2*i/n + p2) + ...)
    # This gives a low-dimensional optimization problem.
    
    from scipy.optimize import minimize
    
    def param_to_seq(params, n):
        c0 = params[0]
        n_harm = len(params) // 5
        a = np.full(n, c0)
        for h in range(n_harm):
            amp = abs(params[1 + h*4])
            freq = params[2 + h*4]
            phase = params[3 + h*4]
            # flip = params[4 + h*4]  # not used
            a += amp * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
        return np.maximum(a, 0.0)
    
    # Try different numbers of harmonics
    for n_harm in [2, 3, 4, 5]:
        if time.monotonic() - t0 > time_limit:
            break
        
        n_params = 1 + n_harm * 4
        # Initialize from current a
        from numpy.fft import rfft as np_rfft
        A = np_rfft(a)
        # n_harm harmonics
        init_params = [a.mean()]
        for h in range(n_harm):
            k = h + 1
            if k < len(A):
                amp = 2.0 * abs(A[k]) / n
                phase = np.angle(A[k])
            else:
                amp = 0.1
                phase = 0.0
            freq = k
            init_params.extend([amp, freq, phase, 0.0])
        
        def obj(params):
            a_p = param_to_seq(params, n)
            sp = a_p.sum()
            if sp < 1e-15:
                return 1e10
            a_p = a_p * norm / sp
            conv = fast_conv_self(a_p)
            return conv.max()
        
        res = minimize(obj, init_params, method='Nelder-Mead',
                      options={'maxiter': 500, 'xatol': 1e-6, 'fatol': 1e-8})
        
        a_try = param_to_seq(res.x, n)
        sp = a_try.sum()
        if sp > 1e-15:
            a_try = a_try * norm / sp
            sc = fast_conv_self(a_try).max()
            if sc < best_score:
                best_score = sc
                best_a = a_try.copy()
                print(f"    SOS n_harm={n_harm}: score={sc:.8f}")
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    # Cubic interpolation for smoother results
    from scipy.interpolate import CubicSpline
    try:
        cs = CubicSpline(old_x, f_norm, bc_type='not-a-knot')
        f_interp = cs(new_x)
        f_interp = np.maximum(f_interp, 0.0)
    except Exception:
        f_interp = np.interp(new_x, old_x, f_norm)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def symmetrize(a):
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    s = a_sym.sum()
    if s < 1e-15:
        return a
    return a_sym / s * (a.sum() / s)
