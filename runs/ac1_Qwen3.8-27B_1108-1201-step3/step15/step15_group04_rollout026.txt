import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

_global_best = {'score': float('inf'), 'seq': None}

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def compute_score(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    n = len(a)
    max_b = fast_conv_self(a).max()
    return 2.0 * n * max_b / (S * S)

def build_conv_matrix(a):
    """Build the matrix C where conv(a, b) = C @ b."""
    n = len(a)
    size = 2 * n - 1
    C = np.zeros((size, n))
    for m in range(n):
        lo = max(0, m)
        hi = min(size, m + n)
        C[lo:hi, m] = a[0:hi-lo]
    return C

def lp_best_response(a, M, solver, active_mask=None):
    """Solve the LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    
    if active_mask is not None and active_mask.sum() < size:
        C_active = build_conv_matrix(a)[active_mask, :]
        constraints = [C_active @ b_var <= M]
    else:
        C_full = build_conv_matrix(a)
        constraints = [C_full @ b_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    
    # Get duals
    duals = None
    try:
        con = constraints[0]
        if con.dual_value is not None:
            if active_mask is not None and active_mask.sum() < size:
                full_duals = np.zeros(size)
                full_duals[np.where(active_mask)[0]] = np.asarray(con.dual_value, dtype=np.float64)
                duals = full_duals
            else:
                duals = np.asarray(con.dual_value, dtype=np.float64)
    except Exception:
        pass
    
    return val, duals

def lp_best_response_fast(a, M, solver, active_indices):
    """Solve LP with pre-selected active constraints."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    
    # Build only the active rows
    C_rows = np.zeros((len(active_indices), n))
    for idx, k in enumerate(active_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for i in range(lo, hi):
            C_rows[idx, i] = a[k - i]
    
    constraints = [C_rows @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    
    duals = None
    try:
        con = constraints[0]
        if con.dual_value is not None:
            full_duals = np.zeros(size)
            full_duals[active_indices] = np.asarray(con.dual_value, dtype=np.float64)
            duals = full_duals
    except Exception:
        pass
    
    return val, duals

def lse_smoothed_descent(a0, time_limit, verbose=False):
    """LSE (log-sum-exp) smoothed approximation to the max, with gradient descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Multi-stage temperature
    n_stages = 15
    temps = np.geomspace(50.0, 0.05, n_stages)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(50, min(500, int(remaining * 3)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            
            # LSE approximation
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -300.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            
            # Gradient: grad of sum_k w[k] * conv(a,a)[k]
            # = 2 * sum_k w[k] * (C_k^T a) where C_k is the k-th row structure
            # More efficiently: grad_i = 2 * sum_k w[k] * (a[k-i] + a[i-k])
            # = 2 * cross_corr(w, a) (circular-like)
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)
            
            # Step size with backtracking
            step = norm / (T_val * np.sqrt(n)) / (1.0 + 0.02 * it)
            for trial in range(10):
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    step *= 0.5
                    continue
                a_new = a_new * norm / s_new
                sc_new = compute_score(a_new)
                if sc_new <= compute_score(a) + 0.1:
                    break
                step *= 0.5
            
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new
            
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose:
            print(f"    LSE stage {si} T={T_val:.3f}: best={best_score:.8f}")
    
    return best_a, best_score

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1
    no_improve = 0
    max_no_improve = 200
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break
        
        # Adaptive constraint screening
        if t < 30:
            threshold = 0.3
        elif t < 100:
            threshold = 0.5
        elif t < 500:
            threshold = 0.7
        elif t < 2000:
            threshold = 0.85
        else:
            threshold = 0.95
        
        active = np.where(conv_y >= M * threshold)[0]
        if len(active) < 20:
            active = np.where(conv_y >= M * 0.3)[0]
        
        b, duals = lp_best_response_fast(y, M, solver, active)
        if b is None:
            b, duals = lp_best_response(y, M, solver, None)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        
        # Verify dropped constraints
        if len(active) < size:
            check = fast_conv_cross(y, g)
            if check.max() > M * (1 + 1e-2):
                # Re-solve with more constraints
                active2 = np.where(conv_y >= M * 0.5)[0]
                b2, duals2 = lp_best_response_fast(y, M, solver, active2)
                if b2 is not None and b2.sum() > 1e-15:
                    b = b2
                    g = b2 * norm / b2.sum()
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        # Nesterov lookahead for a
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        sc_y = compute_score(y)
        improved = False
        if sc_y < best_score - 1e-10:
            best_score = sc_y
            best_a = y.copy()
            improved = True
        
        sc_a = compute_score(a)
        if sc_a < best_score - 1e-10:
            best_score = sc_a
            best_a = a.copy()
            improved = True
        
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
        
        # Perturbation to escape local optima
        if no_improve == max_no_improve:
            n_pert = max(5, n // 50)
            idx_p = np.random.default_rng(t + 12345).choice(n, size=n_pert, replace=False)
            y[idx_p] *= (1.0 + 0.3 * np.random.default_rng(t + 67890).standard_normal(n_pert))
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y * norm / s_y
            no_improve = 0
            max_no_improve = max(50, max_no_improve // 2)
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M:.6f}")
    
    return best_a, best_score

def subgradient_polish(a0, time_limit, verbose=False):
    """Subgradient method on the non-smooth max objective."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_score(a)
    a = a * norm / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    
    # Averaging
    avg_a = np.zeros(n)
    n_avg = 0
    step_scale = 1.0 / np.sqrt(n)
    
    for it in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_conv_self(a)
        M = conv.max()
        
        # Subgradient: positions of the max
        peak_positions = np.where(conv >= M * (1 - 1e-6))[0]
        
        # Gradient of max at a: average of gradients at peak positions
        # grad_i = 2 * avg over peaks of (a[k-i] + a[i-k])
        fft_size = 1 << (size - 1).bit_length()
        A = rfft(a, fft_size)
        
        grad = np.zeros(n)
        for k in peak_positions:
            # Contribution from position k
            lo1 = max(0, k - n + 1)
            hi1 = min(n, k + 1)
            for i in range(lo1, hi1):
                grad[i] += a[k - i]
            # (the other term a[i-k] is handled by the symmetric contribution)
        
        # Actually, let me use FFT for efficiency
        # grad_i = 2 * sum_k w[k] * (a[k-i] + a[i-k]) where w is uniform on peaks
        w = np.zeros(size)
        w[peak_positions] = 1.0 / len(peak_positions)
        W = rfft(w, fft_size)
        g1 = irfft(np.conj(A) * W, fft_size)[:n]
        g2 = irfft(A * W, fft_size)[:n]
        grad = 2.0 * (g1 + g2)
        
        # Step with diminishing step size
        step = step_scale / np.sqrt(it)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new * norm / s_new
        a = a_new
        
        avg_a += a
        n_avg += 1
        
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        # Also check averaged
        if n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc_avg = compute_score(avg_check)
                if sc_avg < best_score:
                    best_score = sc_avg
                    best_a = avg_check.copy()
        
        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad it={it} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def newton_polish(a0, time_limit, verbose=False):

    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()

        return a, compute_score(a)

    best_a = a.copy()
    best_score = compute_score(a * norm)

        # sum_i a[i] = 1
        
        # This is complex. Let's use a simpler approach:

        # Using scipy minimize

        # The gradient of f(a) = 2n*max(conv(a,a))/sum(a)^2 at points where
        # the max is attained at specific positions.

def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length, preserving the normalized shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

**Alternative: Direct QP formulation**

The problem is: minimize max_k (a^T C_k a) / (sum(a))^2 subject to a >= 0.

This is a ratio of a max of quadratics to a quadratic. We can reformulate as:
minimize t
subject to: a^T C_k a <= t * (sum(a))^2 for all k
            a >= 0
            sum(a) = 1 (normalization)

This is a non-convex problem (the constraint a^T C_k a <= t is non-convex in (a,t) since a^T C_k a can be indefinite... wait, C_k is symmetric and PSD? Let me check.

C_k is the matrix such that a^T C_k a = conv(a,a)[k] = sum_i a[i]*a[k-i].

For k = n-1: a^T C_{n-1} a = sum_i a[i]*a[n-1-i]. The matrix C_{n-1} has C_{n-1}[i,j] = 1 if i+j = n-1, else 0. This is the "anti-diagonal" matrix, which is symmetric but not PSD (it has eigenvalues ±1).

So the problem is non-convex. The LP approach works around this by fixing one factor.

**Alternative: Alternating minimization with scaling**

minimize max_k conv(a,a)[k]
subject to: sum(a) = 1, a >= 0

This is equivalent to our problem (since the score is scale-invariant).

The LP approach solves: for fixed a, minimize max_k conv(a,b)[k] over b >= 0, sum(b) = 1.
This is: minimize M subject to conv(a,b)[k] <= M for all k, b >= 0, sum(b) = 1.
Which is a linear program!

And the fixed point gives a local minimum of the original problem.

So the LP approach is actually solving a sequence of LPs, each of which is the "best response" to the current a. This is the same as what we had before.

def build_active_matrix(a, active_idx):
    """Build C[active_idx, :] where C @ b = conv(a, b)."""
    n = len(a)
    rows = np.zeros((len(active_idx), n))
    for r, k in enumerate(active_idx):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        idx = np.arange(lo, hi)
        rows[r, idx] = a[k - idx]
    return rows

def lp_solve(a, M, solver, active_idx):
    """Solve: max sum(b) s.t. C[active] @ b <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    if active_idx is not None and len(active_idx) > 0:
        C_act = build_active_matrix(a, active_idx)
        constraints = [C_act @ b_var <= M]
    else:
        size = 2 * n - 1
        C_full = build_active_matrix(a, np.arange(size))
        constraints = [C_full @ b_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    
    duals = None
    try:
        con = constraints[0]
        if con.dual_value is not None:

                full_duals[active_idx] = dv
                duals = full_duals

                duals = dv

            res = minimize(
                objective, a, method='L-BFGS-B',
                options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-10}
            )
            
            if res.success or res.fun < objective(a):
                a_new = np.maximum(res.x, 0.0)
                s_new = a_new.sum()
                if s_new > 1e-10:
                    a_new = a_new / s_new
                    sc = compute_score(a_new * norm)
                    if sc < best_score:
                        best_score = sc
                        best_a = (a_new * norm).copy()
                        a = a_new.copy()
    
    return best_a, best_score

def subgradient_polish(a0, time_limit, verbose=False):
    """Subgradient method with averaging for the non-smooth max objective."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_score(a)
    a = a / s
    best_a = (a * norm).copy()
    best_score = compute_score(best_a)
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    avg_a = np.zeros(n)
    n_avg = 0
    step_scale = 2.0 / np.sqrt(n)
    
    for it in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_conv_self(a)
        M = conv.max()
        peaks = np.where(conv >= M * (1 - 1e-5))[0]
        
        w = np.zeros(size)
        w[peaks] = 1.0 / len(peaks)
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        g1 = irfft(np.conj(A) * W, fft_size)[:n]
        g2 = irfft(A * W, fft_size)[:n]
        grad = 2.0 * (g1 + g2)
        grad = grad - grad.mean()
        
        step = step_scale / np.sqrt(it)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        a = a_new
        
        avg_a += a
        n_avg += 1
        
        sc = compute_score(a * norm)
        if sc < best_score:
            best_score = sc
            best_a = (a * norm).copy()
        
        if n_avg % 500 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-10:
                avg_check = avg_check / s_avg
                sc_avg = compute_score(avg_check * norm)
                if sc_avg < best_score:
                    best_score = sc_avg
                    best_a = (avg_check * norm).copy()
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad it={it} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def pair_cd(a0, time_limit, rng, verbose=False):
    """Pair coordinate descent: move mass from peak-contributing positions."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_score(a)
    a = a * norm / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False
        peak_pos = np.where(conv >= cur_max * 0.97)[0]
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-9:
                continue
            
            best_delta = 0.0
            best_j = -1
            best_new_max = cur_max
            
            # Try several candidate j's
            candidates = set()
            for _ in range(5):
                candidates.add(int(rng.integers(n)))
            # Structured: move away from peak-contributing positions
            for pk in peak_pos[:5]:
                # i contributes to conv[pk] via a[i]*a[pk-i] and a[pk-i]*a[i]
                # To reduce, move mass to positions that don't contribute to peaks
                for offset in [1, 2, 3, n//4, n//3]:
                    j1 = (i + offset) % n
                    j2 = (i - offset) % n
                    candidates.add(j1)
                    candidates.add(j2)
            candidates.discard(i)
            
            for j in list(candidates)[:8]:
                for frac in [0.3, 0.6, 0.9]:
                    delta = frac * ai
                    old_ai = a[i]
                    old_aj = a[j]
                    a[i] = old_ai - delta
                    a[j] = old_aj + delta
                    s_new = a.sum()
                    if s_new < 1e-10:
                        a[i] = old_ai
                        a[j] = old_aj
                        continue
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_new_max:
                        best_new_max = new_max
                        best_delta = delta
                        best_j = j
                        best_a_state = a.copy()
                    # Revert
                    a[:] = 0
                    a[i] = old_ai
                    a[j] = old_aj
                    a = a * norm / a.sum()
            
            if best_j >= 0 and best_delta > 0:
                a = best_a_state.copy()
                sc = compute_score(a)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a.copy()
                    pass_improved = True
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 6:
                break
        else:
            no_improve = 0
    
    return best_a, best_score

def diverse_init(n, rng, style):
    """Generate diverse initial sequences."""
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        a = (x + 0.005) ** 1.5
    elif style == 1:
        a = (1.0 - x + 0.005) ** 1.5
    elif style == 2:
        a = np.exp(2.5 * x)
    elif style == 3:
        a = np.exp(-2.5 * x)
    elif style == 4:
        a = np.sin(np.pi * x) ** 2
    elif style == 5:
        a = np.exp(-6.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif style == 6:
        r = 0.997
        a = r ** np.arange(n, dtype=np.float64)
    elif style == 7:
        a = (x + 0.01) ** 0.5
    elif style == 8:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 2 * x) + 0.1 * np.cos(2 * np.pi * 5 * x)
        a = np.maximum(a, 0.05)
    else:
        a = np.abs(np.sin(np.pi * x * 1.5)) ** 1.2 + 0.05
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_len(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = eval_fn(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")
    
    # Load parent if available
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass
    
    # Determine base length
    n_base = 1600
    if best_seq is not None:
        n_base = len(best_seq)
    
    # If no parent, create a good initial sequence
    if best_seq is None:
        best_init = None
        best_init_score = float('inf')
        for trial in range(12):
            style = trial % 9
            cand = diverse_init(n_base, rng, style)
            sc = compute_score(cand)
            if sc < best_init_score:
                best_init_score = sc
                best_init = cand
        best_seq = best_init
        best_score = best_init_score
        print(f"Initial: n={n_base}, score={best_score:.6f}")
    
    # Phase 1: Quick LSE to improve starting point
    remaining = time_budget - (time.monotonic() - t_start)
    t_lse = min(remaining * 0.06, 90.0)
    if t_lse > 15:
        print(f"\n=== Phase 1: LSE init ({t_lse:.0f}s) ===")
        a_out, _ = lse_descent(best_seq, t_lse, verbose=False)
        try_update(a_out, "lse_init")
    
    # Phase 2: Main LP Nesterov
    remaining = time_budget - (time.monotonic() - t_start)
    t_lp1 = min(remaining * 0.30, 480.0)
    print(f"\n=== Phase 2: LP Nesterov n={len(best_seq)} ({t_lp1:.0f}s) ===")
    if t_lp1 > 30:
        a_out, _ = nesterov_lp(best_seq, t_lp1, solver, verbose=True)
        try_update(a_out, "lp1")
    
    # Phase 3: Subgradient polish
    remaining = time_budget - (time.monotonic() - t_start)
    t_sg = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 3: Subgradient polish ({t_sg:.0f}s) ===")
    if t_sg > 20:
        a_out, _ = subgradient_polish(best_seq, t_sg, verbose=False)
        try_update(a_out, "subgrad")
    
    # Phase 4: LP Nesterov round 2
    remaining = time_budget - (time.monotonic() - t_start)
    t_lp2 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 4: LP Nesterov round 2 ({t_lp2:.0f}s) ===")
    if t_lp2 > 30:
        a_out, _ = nesterov_lp(best_seq, t_lp2, solver, verbose=True)
        try_update(a_out, "lp2")
    
    # Phase 5: Scipy polish
    remaining = time_budget - (time.monotonic() - t_start)
    t_sp = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 5: Scipy polish ({t_sp:.0f}s) ===")
    if t_sp > 15:
        a_out, _ = scipy_polish(best_seq, t_sp, verbose=True)
        try_update(a_out, "scipy")
    
    # Phase 6: Pair CD
    remaining = time_budget - (time.monotonic() - t_start)
    t_pcd = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 6: Pair CD ({t_pcd:.0f}s) ===")
    if t_pcd > 20:
        a_out, _ = pair_cd(best_seq, t_pcd, rng, verbose=False)
        try_update(a_out, "pcd")
    
    # Phase 7: Try longer length
    remaining = time_budget - (time.monotonic() - t_start)
    n_long = n_base + 600
    print(f"\n=== Phase 7: Length extension to n={n_long} ===")
    if remaining > 180:
        init_long = interpolate_len(best_seq, n_long)
        t_long = min(remaining * 0.35, 250.0)
        a_out, _ = nesterov_lp(init_long, t_long, solver, verbose=False)
        try_update(a_out, f"lp_long_{n_long}")
        
        remaining = time_budget - (time.monotonic() - t_start)
        if remaining > 60 and len(best_seq) == n_long:
            t_long_sg = min(remaining * 0.4, 80.0)
            a_out2, _ = subgradient_polish(best_seq, t_long_sg, verbose=False)
            try_update(a_out2, f"sg_long_{n_long}")
    
    # Phase 8: Final polish
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 8: Final polish (remaining={remaining:.0f}s) ===")
    if remaining > 50:
        t_f = min(remaining * 0.4, 80.0)
        a_out, _ = lse_descent(best_seq, t_f, verbose=False)
        try_update(a_out, "lse_final")
        
        remaining = time_budget - (time.monotonic() - t_start)
        if remaining > 30:
            t_f2 = min(remaining * 0.5, 60.0)
            a_out, _ = subgradient_polish(best_seq, t_f2, verbose=False)
            try_update(a_out, "sg_final")
        
        remaining = time_budget - (time.monotonic() - t_start)
        if remaining > 30:
            t_f3 = min(remaining * 0.6, 50.0)
            a_out, _ = scipy_polish(best_seq, t_f3, verbose=False)
            try_update(a_out, "scipy_final")
    
    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_start:.0f}s")
    else:
        print("ERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
