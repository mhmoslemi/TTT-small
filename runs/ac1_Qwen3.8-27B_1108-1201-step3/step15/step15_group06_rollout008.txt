"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth gradient descent in log-space with Adam optimizer (main method)
2. Direct peak-attacking subgradient refinement
3. LP-based Nesterov iteration (secondary)
4. Coordinate descent polishing
5. Multi-length exploration
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.maximum(np.asarray(a, dtype=np.float64), 0.0)]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def smooth_gradient_step(a, beta):
    """Compute gradient of soft-max smoothed autocorrelation peak."""
    n = len(a)
    c = fast_autocorr(a)
    # Softmax with numerical stability
    c_max = c.max()
    exp_c = np.exp(beta * (c - c_max))
    p = exp_c / exp_c.sum()
    # Gradient: grad_i = 2 * sum_m a[m] * p[i+m]
    # This is np.correlate(p, a, mode='valid') when p is longer
    grad = 2.0 * np.correlate(p, a, mode='valid')
    return grad, p, c


def smooth_optimize(a_init, time_limit, beta_init=2.0, beta_final=30.0,
                    lr_init=0.01, verbose=True, seed=0):
    """Smooth gradient descent in log-space with Adam optimizer."""
    rng = np.random.default_rng(seed)
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 1e-15)
    a = a / a.sum()  # normalize to sum=1

    # Log parameterization
    b = np.log(a)

    # Adam state
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_conv_max = fast_autocorr(a).max()
    best_score = 2.0 * n * best_conv_max
    t0 = time.monotonic()
    no_improve = 0
    stall_limit = 500

    for it in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive beta
        frac = min(1.0, it / 2000.0)
        beta = beta_init + (beta_final - beta_init) * frac

        # Adaptive learning rate (decreasing)
        lr = lr_init / (1.0 + 0.001 * it)

        a = np.exp(b)
        # Renormalize in log-space
        log_sum = np.log(a.sum())
        target_log = 0.0  # sum(a) = 1
        b = b - (log_sum - target_log)
        a = np.exp(b)

        # Compute gradient
        grad_a, p, c = smooth_gradient_step(a, beta)

        # Gradient in log-space: grad_b_i = a_i * grad_a_i
        grad_b = a * grad_a

        # Adam update
        t_adam += 1
        m_adam = beta1 * m_adam + (1 - beta1) * grad_b
        v_adam = beta2 * v_adam + (1 - beta2) * grad_b ** 2
        m_hat = m_adam / (1 - beta1 ** t_adam)
        v_hat = v_adam / (1 - beta2 ** t_adam)
        b = b - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Keep b in reasonable range to avoid overflow
        b = np.clip(b, -20.0, 5.0)

        # Renormalize to sum(a) = 1
        a = np.exp(b)
        log_sum = np.log(a.sum() + 1e-30)
        b = b - log_sum
        a = np.exp(b)

        # Track best
        conv_max = c.max()
        score = 2.0 * n * conv_max
        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > stall_limit:
            # Restart with perturbation
            b = np.log(best_a + 1e-15)
            b += 0.05 * rng.standard_normal(n)
            log_sum = np.log(np.exp(b).sum())
            b -= log_sum
            m_adam = np.zeros(n)
            v_adam = np.zeros(n)
            t_adam = 0
            no_improve = 0

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SmoothOpt it={it} ({elapsed:.0f}s): score={best_score:.8f}, beta={beta:.1f}")

    return best_a, best_score


def peak_attack(a_init, time_limit, n_iter=5000, verbose=True):
    """Directly attack the maximum peak using subgradient descent."""
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Log parameterization
    b = np.log(a)
    lr = 0.02

    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break

        lr = 0.02 / (1.0 + 0.002 * it)

        a = np.exp(b)
        log_sum = np.log(a.sum() + 1e-30)
        b -= log_sum
        a = np.exp(b)

        c = fast_autocorr(a)
        k_max = c.argmax()

        # Subgradient of c[k_max] w.r.t. a[i] = 2*a[k_max - i] if valid
        grad = np.zeros(n)
        for i in range(n):
            idx = k_max - i
            if 0 <= idx < n:
                grad[i] = 2.0 * a[idx]

        # Update in log-space
        grad_b = a * grad
        b = b - lr * grad_b
        b = np.clip(b, -20.0, 5.0)

        # Renormalize
        a = np.exp(b)
        log_sum = np.log(a.sum() + 1e-30)
        b -= log_sum
        a = np.exp(b)

        score = 2.0 * n * c.max()
        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakAttack it={it} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def multi_peak_attack(a_init, time_limit, n_iter=5000, verbose=True):
    """Attack top few peaks simultaneously."""
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    b = np.log(a)
    lr = 0.01

    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break

        lr = 0.01 / (1.0 + 0.002 * it)

        a = np.exp(b)
        log_sum = np.log(a.sum() + 1e-30)
        b -= log_sum
        a = np.exp(b)

        c = fast_autocorr(a)
        # Get top 5 peaks
        top_k = np.argsort(c)[-5:][::-1]
        weights = c[top_k]
        weights = weights / weights.sum()

        # Combined subgradient
        grad = np.zeros(n)
        for w, k in zip(weights, top_k):
            for i in range(n):
                idx = k - i
                if 0 <= idx < n:
                    grad[i] += w * 2.0 * a[idx]

        grad_b = a * grad
        b = b - lr * grad_b
        b = np.clip(b, -20.0, 5.0)

        a = np.exp(b)
        log_sum = np.log(a.sum() + 1e-30)
        b -= log_sum
        a = np.exp(b)

        score = 2.0 * n * c.max()
        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MultiPeak it={it} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score

def structured_init(n, style=0):
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif style == 1:
        a = np.sin(np.pi * x) ** 1.2
    elif style == 2:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(8 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif style == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif style == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif style == 5:
        a = 0.4 + 0.6 * np.sin(np.pi * x) ** 0.8
    elif style == 6:
        a = 1.0 + 0.25 * np.cos(6 * np.pi * x) + 0.1 * np.cos(12 * np.pi * x)
    else:
        a = np.sin(np.pi * x) ** 0.6 * (1.0 + 0.15 * np.cos(10 * np.pi * x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss
