def freq_waterfilling(a, time_limit, verbose=False):
    n = len(a)
    S = a.sum()
    a = a / S  # normalize to sum 1 for now (score is scale-invariant)
    S = 1.0  # now sum is 1
    
    L = next_pow2(2 * n - 1)
    
    # Precompute index mapping
    freq_idx = np.arange(L)
    
    best_a = a.copy()
    best_score = 2 * n * fft_autocorr(a).max()  # with sum=1, score = 2n*max_conv
    
    t0 = time.time()
    
    # Annealing schedule
    n_iters = 0
    for it in range(1000000):
        if time.time() - t0 > time_limit:
            break
        
        n_iters += 1
        
        # Progressive equalization
        progress = min(1.0, n_iters / 500.0)  # full equalization after 500 iters
        beta = progress  # 0 = no change, 1 = full equalization
        
        # FFT
        a_padded = np.zeros(L)
        a_padded[:n] = a
        A = np.fft.fft(a_padded)
        
        # Compute target magnitude
        # A[0] = sum(a) = 1 (DC)
        # Target for k != 0: equalize
        A_abs = np.abs(A)
        # Mean magnitude of AC components
        ac_mags = A_abs[1:]
        target_mag = np.mean(ac_mags)  # or could use median or sqrt(mean_sq)
        
        # Actually, for a flat autocorrelation, we want |A[k]|^2 to be constant.
        # The "target" should be such that the energy is preserved.
        # Total energy: sum(|A|^2) = L * sum(a^2)
        # A[0]^2 + sum_{k=1}^{L-1} |A[k]|^2 = L * sum(a^2)
        # If we set |A[k]| = C for k != 0:
        #   A[0]^2 + (L-1)*C^2 = L * sum(a^2)
        #   C^2 = (L * sum(a^2) - A[0]^2) / (L-1)
        
        energy = np.sum(a**2)
        C_sq = (L * energy - A[0]**2) / (L - 1)
        C = np.sqrt(max(C_sq, 0))
        
        # Update magnitudes with annealing
        # A_new[k] = A[k] * (C / |A[k]|)^beta for k != 0
        for k in range(1, L):
            mag = A_abs[k]
            if mag > 1e-15:
                scale = (C / mag) ** beta
                A[k] *= scale
        
        # Maintain Hermitian symmetry (should be automatic if input is real)
        # But due to numerical errors, let's enforce it
        # A[k] = conj(A[L-k])
        # Actually, numpy's FFT of a real signal already has this property,
        # and we're only modifying magnitudes (not phases), so it should be fine.
        # But to be safe:
        # for k in range(1, L//2+1):
        #     A[L-k] = np.conj(A[k])
        # This is important!
        
        # Inverse FFT
        a_new = np.fft.ifft(A).real[:n]
        
        # Non-negativity projection
        a_new = np.maximum(a_new, 0)
        
        # Renormalize
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        
        # Check improvement
        score = 2 * n * fft_autocorr(a_new).max()
        if score < best_score - 1e-10:
            best_score = score
            best_a = a_new.copy()
        
        # Always use a_new for next iteration (even if not best)
        a = a_new

def freq_waterfilling(a, time_limit, verbose=False, seed=0, relax_gamma=None):
    """
    Frequency-domain water-filling: iteratively equalize |DFT(a)| magnitudes
    to flatten the autocorrelation, projecting onto non-negative cone each step.
    """
    n = len(a)
    a = normalize(a, 1.0)
    L = 1 << (2 * n - 2).bit_length()

    rng = np.random.default_rng(seed)

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    if relax_gamma is None:
        relax_gamma = 1.0

    n_iters = 0
    stall = 0
    stall_limit = 200

    for it in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break

        n_iters += 1

        # Progressive annealing: beta goes from 0 to 1 over first 300 iterations
        progress = min(1.0, n_iters / 300.0)
        beta = progress ** 0.5  # ease-in

        # Relaxation parameter (blend with previous)
        gamma = min(relax_gamma, 0.3 + 0.7 * progress)

        # FFT of zero-padded sequence
        a_pad = np.zeros(L)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)

        # Compute target magnitude (water level)
        # A[0] = sum(a) = 1 (DC, fixed)
        # For k != 0: target |A[k]| = C where C^2 = (L*energy - A[0]^2)/(L-1)
        energy = np.dot(a, a)
        n_ac = L // 2  # number of AC bins in rfft (excluding DC)
        C_sq = (L * energy - A[0].real ** 2) / (L - 1)
        C = np.sqrt(max(C_sq, 0.0))

        # Update frequencies (keep DC fixed)
        A_new = A.copy()
        for k in range(1, len(A)):
            mag = np.abs(A[k])
            if mag > 1e-15:
                # Geometric interpolation in magnitude
                new_mag = mag * (C / mag) ** beta
                A_new[k] = A[k] / mag * new_mag
            elif beta > 0.5:
                # Small magnitude: set to target with random phase
                angle = rng.uniform(0, 2 * np.pi)
                A_new[k] = C ** beta * np.exp(1j * angle)

        # Inverse FFT
        a_fft = np.fft.irfft(A_new, L)[:n]

        # Non-negativity projection with relaxation
        a_clipped = np.maximum(a_fft, 0.0)
        s_clip = a_clipped.sum()
        if s_clip < 1e-15:
            a_clipped = a.copy()
        else:
            a_clipped = a_clipped / s_clip

        # Relaxation step
        a_new = (1.0 - gamma) * a + gamma * a_clipped
        a_new = np.maximum(a_new, 0.0)
        a = normalize(a_new, 1.0)

        # Track best
        score = compute_score(a)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive: if stalled, try a different approach
        if stall > stall_limit:
            # Add small perturbation to escape local minimum
            a = a * (1.0 + 0.05 * rng.standard_normal(n))
            a = normalize(a, 1.0)
            stall = 0

        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FreqWF iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a

def build_conv_matrix_row(a, k, n):
    """Build row k of the convolution constraint matrix (a fixed, b variable)."""
    row = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    for j in range(lo, hi):
        row[j] = a[k - j]
    return row


def lp_best_response(a_fixed, M, active_set, solver):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for k in active_set, b >= 0."""
    if solver is None:
        return None
    n = len(a_fixed)
    k_indices = np.asarray(active_set, dtype=int)
    size = len(k_indices)
    if size == 0:
        return np.ones(n) / n

    # Build constraint matrix
    T = np.zeros((size, n))
    for idx, k in enumerate(k_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            T[idx, j] = a_fixed[k - j]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def subgradient_polish(a0, time_limit, verbose=False):

    Subgradient method: minimize max_k (a*a)[k] by moving away from active peaks.
    Vectorized for speed.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(a0, norm)
    best_a = a.copy()
    best_score = compute_score(a)

    step = 2.0 / np.sqrt(n)
    patience = 0
    max_patience = 300

        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fft_conv(a)
        M_curr = conv.max()

        # Get top peaks
        n_peaks = max(3, min(15, n // 50))
        top_idx = np.argsort(conv)[-n_peaks:]
        weights = np.exp(-3.0 * (M_curr - conv[top_idx]) / max(M_curr * 0.01, 1e-10))
        weights = weights / weights.sum()

        # Compute combined gradient (vectorized)
        # grad[j] = sum over active k of w_k * (a[k-j] + a[j-k]) with bounds
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            # Contribution from a[k-j] where 0 <= k-j < n
            # i.e., j in [k-n+1, k] intersect [0, n-1]
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                idx = np.arange(j_lo, j_hi + 1)
                src = k - idx
                grad[idx] += w * a[src]
            # Contribution from a[j-k] where 0 <= j-k < n
            # i.e., j in [k, k+n-1] intersect [0, n-1]
            j_lo2 = max(0, k)
            j_hi2 = min(n - 1, k + n - 1)
            if j_lo2 <= j_hi2:
                idx2 = np.arange(j_lo2, j_hi2 + 1)
                src2 = idx2 - k
                grad[idx2] += w * a[src2]

        # Normalized gradient step
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break

        a_trial = a - step * grad / g_norm * np.sqrt(n)
        a_trial = np.maximum(a_trial, 0.0)
        a_trial = normalize(a_trial, norm)

        score = compute_score(a_trial)
        if score < best_score - 1e-12:
            a = a_trial
            best_score = score
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.01, 10.0)
        else:
            patience += 1
            step *= 0.96
            if step < 1e-8:
                break

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a


def coordinate_descent(a0, time_limit, verbose=False):
    """
    Coordinate descent with line search.
    For each coordinate, try several values and pick the best.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(a0, norm)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    size = 2 * n - 1

    no_improve_passes = 0
    max_no_improve = 4

    while time.monotonic() - t0 < time_limit:
        no_improve_passes += 1
        if no_improve_passes > max_no_improve:
            break

        improved = False
        conv = fft_conv(a)

        # Process coordinates in order of influence (largest a first)
        order = np.argsort(-a)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]
            if aj < 1e-10:
                continue

            # Try several values around aj
            s_others = a.sum() - aj
            candidates = [aj * 0.3, aj * 0.5, aj * 0.7, aj * 1.3, aj * 1.5, aj * 2.0, 0.0]
            candidates = [x for x in candidates if 0 <= x <= norm]

            best_x = aj
            best_obj = best_score

            for x in candidates:
                if abs(x - aj) < 1e-10:
                    continue
                a[j] = x
                s_new = a.sum()
                if s_new < 1e-15:
                    a[j] = aj
                    continue
                # Quick score check
                conv_new = conv + (x - aj) * (
                    np.take(a, np.minimum(np.arange(size), n - 1))  # approximation
                )
                # Actually, let's just compute properly but sparsely
                # The change in conv when a[j] goes from aj to x:
                # conv_new[k] = conv[k] + (x-aj)*(a[k-j] if valid + a[j-k] if valid) + (x^2 - aj^2)*delta_{k,2j}
                delta = x - aj
                contrib = np.zeros(size)
                # a[j] * a[k-j]: j fixed, varies with k
                for k in range(size):
                    v = 0.0
                    i1 = k - j
                    if 0 <= i1 < n:
                        v += a[i1] if i1 != j else 0
                    i2 = j - k
                    if 0 <= i2 < n:
                        v += a[i2] if i2 != j else 0
                    contrib[k] = v

                conv_test = conv + delta * contrib + (x * x - aj * aj) * 0  # x^2 term at k=2j
                if 2 * j < size:
                    conv_test[2 * j] += x * x - aj * aj

                obj = 2.0 * n * conv_test.max() / (s_new * s_new)
                if obj < best_obj:
                    best_obj = obj
                    best_x = x

            if abs(best_x - aj) > 1e-10:
                a[j] = best_x
                a = normalize(a, norm)
                new_score = compute_score(a)
                if new_score < best_score - 1e-13:
                    best_score = new_score
                    best_a = a.copy()
                    improved = True
                    conv = fft_conv(a)
                else:
                    a[j] = aj
                    a = normalize(a, norm)

        if not improved:
            no_improve_passes += max_no_improve  # force exit

    return best_a


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
