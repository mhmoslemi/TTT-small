    # = conv_aa - 2t*conv_aa + t^2*conv_aa + 2t*conv_ab - 2t^2*conv_ab + t^2*conv_bb
    # = conv_aa + t*(-2*conv_aa + 2*conv_ab) + t^2*(conv_aa - 2*conv_ab + conv_bb)
    
    c0 = conv_aa
    c1 = -2*conv_aa + 2*conv_ab
    c2 = conv_aa - 2*conv_ab + conv_bb
    
    # Minimize max_k [c0[k] + t*c1[k] + t^2*c2[k]] over t in [0, 1]
    # Use bisection on the derivative
    
    def max_val(t):
        return np.max(c0 + t*c1 + t*t*c2)
    
    def max_deriv(t):
        # Derivative of c0 + t*c1 + t^2*c2 is c1 + 2t*c2
        # The max is achieved where the derivative is 0 or at a kink
        # Approximate: find k that maximizes, then check derivative there
        vals = c0 + t*c1 + t*t*c2
        k = np.argmax(vals)
        return c1[k] + 2*t*c2[k]
    
    # Bisection on [0, 1]
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        d = max_deriv(mid)
        if d > 0:
            hi = mid
        else:
            lo = mid
    
    t_opt = (lo + hi) / 2
    return t_opt

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)

def fast_conv_cross(a, b):
    n = len(a)

    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def build_T_sparse(a):
    n = len(a)
    size = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for m in range(size):
        lo = max(0, m - n + 1)
        hi = min(n, m + 1)
        for j in range(lo, hi):
            k = m - j
            if 0 <= k < n:
                rows.append(m)
                cols.append(j)
                vals.append(a[k])
    from scipy.sparse import csc_matrix
    T = csc_matrix((vals, (rows, cols)), shape=(size, n))
    return T

def lse_gradient_descent(a0, time_limit, verbose=False):
    n = len(a0)
    norm_target = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

    t0 = time.monotonic()
    size = 2 * n - 1

    temps = np.geomspace(50.0, 0.5, 10)
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(20, int(remaining * 8 / (10 - si)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:

            conv = fast_conv_self(a)

            if max_c < 1e-15:
                break
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -80.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            W = np.fft.rfft(w, fft_size)
            A = np.fft.rfft(a, fft_size)
            g1 = np.fft.irfft(np.conj(A) * W, fft_size)[:n]
            g2 = np.fft.irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            step = norm_target / (T_val * np.sqrt(n)) * 1.0 / (1.0 + 0.03 * it)
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm_target / s_new
            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_score:

def weighted_subgradient_optimize(a0, time_limit, verbose=False):
    """Subgradient method with top-K peak weighting and adaptive step size."""
    n = len(a0)

    s = a.sum()

    best_a = a.copy()

    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    lr = 0.5 / np.sqrt(n)
    best_lr = lr
    patience = 0

    for it in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        max_c = conv.max()

        top_k = max(10, n // 20)
        top_idx = np.argsort(conv)[-top_k:]
        threshold = conv[top_idx[0]] * 0.95
        w = np.zeros(size)
        w[conv >= threshold] = 1.0
        w = w / w.sum()

        W = np.fft.rfft(w, fft_size)
        A = np.fft.rfft(a, fft_size)
        g1 = np.fft.irfft(np.conj(A) * W, fft_size)[:n]
        g2 = np.fft.irfft(A * W, fft_size)[:n]
        grad = 2.0 * (g1 + g2)

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:

        else:
            a_new = a_new * norm_target / s_new
        a = a_new

        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            best_lr = lr
            patience = 0
            lr = min(lr * 1.05, 2.0 / np.sqrt(n))
        else:
            patience += 1
            if patience > 500:
                lr = best_lr * 0.8
                patience = 0

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad it={it} ({elapsed:.0f}s): best={best_score:.8f}, lr={lr:.6f}")

    return best_a, best_score

            for _ in range(8):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:5]:
                j1 = (kl - i + n) % n
                j2 = abs(kl - i)
                if j1 != i:
                    j_candidates.add(j1)
                if j2 != i:
                    j_candidates.add(j2)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in list(j_candidates)[:12]:
                if j == i:
                    continue
                aj = a[j]

                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                n_samples = 15
                deltas = np.linspace(0.0, ai * 0.95, n_samples)
                for d in deltas:
                    if d < 1e-12:
                        continue
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_val:
                        best_val = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm_target / s_new
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-10:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
                cur_max = new_max

        if not pass_improved:
            no_improve_passes += 1
            if no_improve_passes >= 4:
                break
            if no_improve_passes >= 2:
                n_perturb = max(5, n // 50)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.12 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm_target / s_new
        else:
            no_improve_passes = 0

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        if len(a_list) == 0:
            return float('inf')
        for val in a_list:
            if val < 0 or not np.isfinite(val):
                return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def structured_init(n, rng):
    """Generate diverse initial structures."""
    choice = rng.integers(0, 8)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.75) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.2
    elif choice == 5:
        a = np.exp(-3.0 * x) + 0.5 + 0.2 * np.sin(2.0 * np.pi * 3 * x)
    elif choice == 6:
        a = 0.4 + 0.6 * np.sin(np.pi * x) ** 0.8
    else:
        a = np.exp(-2.0 * (x - 0.5) ** 2 / 0.1)
    a = np.maximum(a, 0.0)
    a = a + 0.005 * rng.random(n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Quick LSE warmup
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 90.0)
    print(f"\n=== Phase 1: LSE warmup n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 10:
        a_out, _ = lse_gradient_descent(best_seq, t1, verbose=False)
        try_update(a_out, "lse1")

    # Phase 2: LP with exact line search (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.30, 500.0)
    print(f"\n=== Phase 2: LP LineSearch n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = lp_line_search_optimize(best_seq, t2, solver, verbose=True)
        try_update(a_out, "lp_ls1")

    # Phase 3: Weighted subgradient (different angle)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 3: Weighted SubGrad n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_out, _ = weighted_subgradient_optimize(best_seq, t3, verbose=False)
        try_update(a_out, "wsubgrad")

    # Phase 4: LP line search round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 4: LP LineSearch R2 n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = lp_line_search_optimize(best_seq, t4, solver, verbose=True)
        try_update(a_out, "lp_ls2")

    # Phase 5: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 5: Pair CD n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_out, _ = pair_cd_optimize(best_seq, t5, rng, verbose=False)
        try_update(a_out, "pcd1")

    # Phase 6: Try longer sequence
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 300
    print(f"\n=== Phase 6: Longer n={n_long} (remaining={remaining:.0f}s) ===")
    if remaining > 120:
        init_long = interpolate_seq(best_seq, n_long)
        t6a = min(remaining * 0.5, 150.0)
        a_out, _ = lp_line_search_optimize(init_long, t6a, solver, verbose=False)
        try_update(a_out, f"lp_long{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t6b = min(remaining * 0.5, 80.0)
            a_out2, _ = pair_cd_optimize(best_seq, t6b, rng, verbose=False)
            try_update(a_out2, "pcd_long")

    # Phase 7: Final polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 7: Final polish n={n_final} (remaining={remaining:.0f}s) ===")
    if remaining > 30:
        t7a = min(remaining * 0.5, 60.0)
        a_out, _ = lp_line_search_optimize(best_seq, t7a, solver, verbose=False)
        try_update(a_out, "lp_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t7b = min(remaining * 0.6, 40.0)
            a_out2, _ = pair_cd_optimize(best_seq, t7b, rng, verbose=False)
            try_update(a_out2, "pcd_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
