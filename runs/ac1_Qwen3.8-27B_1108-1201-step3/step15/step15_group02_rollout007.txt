def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)

def fast_conv_cross(a, b):

    B = rfft(b, fft_size)

def lse_adam_optimize(a0, time_limit, verbose=True):
    """LSE-Adam with better adaptive temperature and line search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_adam = 0
    
    # Adaptive temperature: start high, decrease geometrically
    n_stages = 30
    temps = np.geomspace(100.0, 0.02, n_stages)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(20, min(150, int(remaining * 8)))
        lr = 0.8 / np.sqrt(n)
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            t_adam += 1
            
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            
            # Smooth max approximation
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -50.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            
            # Gradient computation
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            
            ca = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (ca + aw)
            
            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_adam)
            v_hat = v / (1.0 - beta2 ** t_adam)
            
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            
            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose and (si % 5 == 0 or si == n_stages - 1):
            print(f"    LSE-Adam stage {si} T={T_val:.4f}: best={best_score:.8f}")
    
    return best_a, best_score

def lp_best_response(a, M, solver, active_set=None):
    """Solve LP: max sum(b) s.t. T*b <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    
    if active_set is not None and 0 < len(active_set) < size:
        # Build only the active constraints
        rows = []
        cols = []
        vals = []
        for k in active_set:
            row_idx = len(rows)
            row_start = max(0, k - n + 1)
            row_end = min(n, k + 1)
            for i in range(row_start, row_end):
                rows.append(row_idx)
                cols.append(i)
                vals.append(a[k - i])
        
        if len(rows) == 0:
            return None
            
        from scipy.sparse import coo_matrix
        T = coo_matrix((vals, (rows, cols)), shape=(len(active_set), n)).tocsr()
        constraints = [T @ b_var <= M]
    else:
        # Full constraint matrix
        rows = []
        cols = []
        vals = []
        for k in range(size):
            row_idx = k
            row_start = max(0, k - n + 1)
            row_end = min(n, k + 1)
            for i in range(row_start, row_end):
                rows.append(row_idx)
                cols.append(i)
                vals.append(a[k - i])
        
        from scipy.sparse import coo_matrix
        T = coo_matrix((vals, (rows, cols)), shape=(size, n)).tocsr()
        constraints = [T @ b_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best response with adaptive active set."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

    size = 2 * n - 1
    y = a.copy()
    
    # Adaptive number of active constraints
    n_active_start = 50
    n_active_end = 300
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break
        
        # Adaptive active set: include constraints above a threshold
        frac = 0.5 + 0.5 * min(1.0, t / 200.0)  # gradually increase
        threshold = M * (1.0 - 0.5 * frac)
        active_set = np.where(conv_y > threshold)[0]
        
        # Cap the active set size
        max_active = int(n_active_start + frac * (n_active_end - n_active_start))
        if len(active_set) > max_active:
            # Keep the top max_active constraints
            idx = np.argsort(conv_y[active_set])[::-1][:max_active]
            active_set = active_set[idx]
        
        b = lp_best_response(y, M, solver, active_set)
        
        if b is None:
            # Fallback: try with slightly relaxed M
            b = lp_best_response(y, M * 1.01, solver, active_set)
        
        if b is None:
            b = lp_best_response(y, M, solver, None)
        
        if b is None:
            continue
        
        # Check if the LP solution violates any non-active constraints
        if active_set is not None and len(active_set) < size:
            check = fast_conv_cross(y, b)
            violated = np.where(check > M * 1.005)[0]
            if len(violated) > 0:
                # Add violated constraints and re-solve
                new_active = np.union1d(active_set, violated)
                if len(new_active) <= max_active * 2:
                    b2 = lp_best_response(y, M, solver, new_active)
                    if b2 is not None:
                        b = b2
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        # Track best
        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M:.6f}, active={len(active_set)}")
    
    return best_a, best_score

def spectral_flatten(a0, time_limit, verbose=True):
    """Optimize in the frequency domain to flatten the spectrum."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    
    # Work with log-amplitude in frequency domain
    # a corresponds to A = rfft(a). We want to minimize peak of |A|^2 in time domain
    # which is equivalent to making |A|^2 as flat as possible in frequency domain
    # (by Parseval's theorem, peak in time domain relates to concentration in freq domain)
    
    # Actually, let's use a different approach:
    # The self-convolution peak is related to the L1 norm of the Fourier transform
    # min peak <=> min ||A||_1 (by duality)
    # So we can minimize ||A||_1 = sum|A_k|
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_adam = 0
    
    # Use L1 norm of FFT as proxy for peak
    n_steps = max(50, int(time_limit * 10))
    lr = 0.3 / np.sqrt(n)
    
    # Actually, let's use a softer approach: minimize sum(exp(T*conv))
    # which is equivalent to the LSE approach but let's try a different angle
    
    # Better: use the "pressure" approach from Vegas
    # Pressure at position i = sum of conv values that involve a[i]
    # weighted by how much they contribute to the max
    
    for it in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        t_adam += 1
        
        conv = fast_conv_self(a)
        M = conv.max()
        if M < 1e-15:
            break
        
        # Compute "pressure" on each element
        # For each lag k, the contribution to conv[k] involves pairs (i, k-i)
        # The gradient of max(conv) w.r.t. a[i] is related to which lags are at max
        # that involve index i
        
        # Use soft max for smoothness
        T = 0.5
        w = np.exp((conv - M) / T)
        w = w / w.sum()
        
        # Gradient via FFT
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        ca = irfft(np.conj(A) * W, fft_size)[:n]
        aw = irfft(A * W, fft_size)[:n]
        grad = ca + aw
        
        # Adam
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t_adam)
        v_hat = v / (1.0 - beta2 ** t_adam)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        a = a_new
        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if verbose and it % 50 == 0:
            print(f"    Spectral it={it}: best={best_score:.8f}")
    
    return best_a, best_score

                # So we want j NOT in [k-n+1, k]
                # But that's hard to check for all k. Instead, pick j that minimizes overlap.
                j_candidates.add(int((kl - i + n) % n))
                
            j_candidates.discard(i)
            
            best_delta = 0.0
            best_v = cur_max
            best_j = -1
            
            for j in list(j_candidates)[:20]:
                if j == i:
                    continue
                aj = a[j]
                
                # Compute change in convolution
                g = np.zeros(size)
                
                # Remove contribution of a[i]: conv[k] -= a[i] * a[k-i] for valid k
                # conv[k] = sum_t a[t] * a[k-t]

                # and a[i]*a[t] contributes to conv[i+t] (same thing since self-conv)

                # and -a[k-i]*a[i] for the t=k-i term
                # But if i = k-i (i.e., k=2i), it's the same term
                # So total change: -2*a[i]*a[k-i] for k != 2i, -a[i]^2 for k = 2i

                # This is getting complicated. Let's just do it directly for the candidates.
                
                # Direct computation:
                # Change in conv[k] = -a[i]*a[k-i] - a[k-i]*a[i] + a[j]*a[k-j] + a[k-j]*a[j] + delta terms
                # Wait, that's not right either. Let me think more carefully.

def vegas_optimize(a0, time_limit, verbose=True):
    """Vegas multiplicative update for non-negative optimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    
    n_stages = 30
    etas = np.geomspace(0.05, 0.0005, n_stages)
    
    for si, eta_s in enumerate(etas):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(15, min(60, int(remaining * 5)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = fast_conv_self(a)
            M = conv.max()
            if M < 1e-15:
                break
            
            # Pressure: weighted by how much each lag is above threshold
            threshold = M * 0.85
            w = np.maximum(conv - threshold, 0.0)
            ww = w / (w.sum() + 1e-15)
            
            W = rfft(ww, fft_size)
            A = rfft(a, fft_size)
            ca = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)[:n]
            pressure = ca + aw
            
            a = a * np.exp(-eta_s * pressure)
            a = np.maximum(a, 1e-10)
            a = a * norm / a.sum()
            
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose and si % 5 == 0:
            print(f"    Vegas stage {si} eta={eta_s:.6f}: best={best_score:.8f}")
    
    return best_a, best_score

def symmetrize(a):
    """Try symmetrizing the sequence."""
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    s = a_sym.sum()
    if s < 1e-15:
        return a.copy()
    return a_sym * (a.sum() / s)

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LSE-Adam gradient descent with adaptive temperature annealing
2. LP best-response with full sparse constraint matrix and Nesterov acceleration
3. Peak-directed multiplicative updates (Vegas-style)
4. Pair coordinate descent with convolution-difference computation
5. Multi-length exploration and symmetrization
"""

import numpy as np
import cvxpy as cp
from scipy.sparse import coo_matrix
from scipy.fft import rfft, irfft
import time

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS, cp.SCS]:
        try:
            x = cp.Variable(6, nonneg=True)
            constraints = [cp.sum(x) == 1.0, x <= 0.3]
            prob = cp.Problem(cp.Minimize(cp.sum(x)), constraints)
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print(f"  Using solver (fallback): ECOS")
    return _solver


def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_sparse_T(a):
    """Build sparse convolution matrix T where (T @ b)[k] = sum_i a[k-i]*b[i]."""
    n = len(a)
    size = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(size):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for i in range(lo, hi):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    T = coo_matrix((vals, (rows, cols)), shape=(size, n)).tocsr()
    return T


def lse_adam_optimize(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_adam = 0

    n_stages = 25
    temps = np.geomspace(60.0, 0.03, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(20, min(120, int(remaining * 8)))
        lr = 0.6 / np.sqrt(n)

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            t_adam += 1

            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -60.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            ca = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (ca + aw)

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_adam)
            v_hat = v / (1.0 - beta2 ** t_adam)

            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose and (si % 5 == 0 or si == n_stages - 1):
            print(f"    LSE-Adam stage {si} T={T_val:.4f}: best={best_score:.8f}")

    return best_a, best_score


def lp_best_response(a, M, solver, T_cached=None, active_set=None):
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)

    if T_cached is not None and active_set is None:
        T = T_cached
    elif active_set is not None and 0 < len(active_set) < size:
        rows = []
        cols = []
        vals = []
        for idx, k in enumerate(active_set):
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for i in range(lo, hi):
                rows.append(idx)
                cols.append(i)
                vals.append(a[k - i])
        if not rows:
            return None
        T = coo_matrix((vals, (rows, cols)), shape=(len(active_set), n)).tocsr()
    else:
        T = build_sparse_T(a)

    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    y = a.copy()

    # Pre-build the full sparse T matrix (rebuild if a changes significantly)
    T_cached = build_sparse_T(a)
    T_valid = True
    T_rebuild_interval = 30

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Adaptive active set: start with top constraints, expand over time
        phase = min(1.0, t / 50.0)
        n_active_target = int(100 + 2500 * phase)
        n_active_target = min(n_active_target, size)

        threshold_frac = 0.95 - 0.4 * phase
        threshold = M * threshold_frac
        active_set = np.where(conv_y > threshold)[0]

        if len(active_set) > n_active_target:
            idx = np.argsort(conv_y[active_set])[::-1][:n_active_target]
            active_set = active_set[idx]
        elif len(active_set) < 20:
            active_set = None  # Use full set

        # Rebuild T if using full set and it may be stale
        if active_set is None:
            if t % T_rebuild_interval == 1:
                T_cached = build_sparse_T(y)
                T_valid = True
            b = lp_best_response(y, M, solver, T_cached, None)
        else:
            b = lp_best_response(y, M, solver, None, active_set)

        if b is None:
            b = lp_best_response(y, M * 1.001, solver, None, active_set)
        if b is None:
            b = lp_best_response(y, M, solver, None, None)
        if b is None:
            continue

        # Verify solution satisfies all constraints if using active set
        if active_set is not None and len(active_set) < size:
            check = fast_conv_cross(y, b)
            violated = np.where(check > M * 1.01)[0]
            if len(violated) > 0:
                new_active = np.union1d(active_set, violated)
                b2 = lp_best_response(y, M, solver, None, new_active)
                if b2 is not None:
                    b = b2

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Nesterov acceleration
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if verbose and t % 15 == 0:
            elapsed = time.monotonic() - t0
            ns = len(active_set) if active_set is not None else size
            print(f"    LP t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M:.6f}, active={ns}")

    return best_a, best_score


def vegas_optimize(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    n_stages = 30
    etas = np.geomspace(0.08, 0.0003, n_stages)

    for si, eta_s in enumerate(etas):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(10, min(50, int(remaining * 6)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_conv_self(a)
            M = conv.max()
            if M < 1e-15:
                break

            threshold = M * 0.80
            w = np.maximum(conv - threshold, 0.0)
            ww = w / (w.sum() + 1e-15)

            W = rfft(ww, fft_size)
            A = rfft(a, fft_size)
            ca = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)[:n]
            pressure = ca + aw

            a = a * np.exp(-eta_s * pressure)
            a = np.maximum(a, 1e-10)
            a = a * norm / a.sum()

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose and si % 5 == 0:
            print(f"    Vegas stage {si} eta={eta_s:.6f}: best={best_score:.8f}")

    return best_a, best_score


def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-8:
                continue

            peak_lags = np.where(conv >= cur_max * 0.92)[0]

            j_candidates = set()
            for _ in range(6):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:12]:
                j_candidates.add(int((kl - i + n) % n))
                j_candidates.add(int((kl + i) % n))
            j_candidates.discard(i)

            best_delta = 0.0
            best_v = cur_max
            best_j = -1

            for j in list(j_candidates)[:15]:
                if j == i:
                    continue
                aj = a[j]

                # Compute change in convolution: move delta from i to j
                # delta_conv[k] = -2*d*a[k-i] + 2*d*a[k-j] (approximately)
                g = np.zeros(size)
                lo1 = i
                hi1 = min(size, i + n)
                g[lo1:hi1] -= a[0:hi1 - lo1]
                lo2 = j
                hi2 = min(size, j + n)
                g[lo2:hi2] += a[0:hi2 - lo2]

                # Self-term corrections
                if 2 * i < size:
                    g[2 * i] += a[i]  # was -2a[i], should be -a[i]
                if 2 * j < size:
                    g[2 * j] -= a[j]  # was +2a[j], should be +a[j]

                d_max = min(ai * 0.7, norm * 0.05)
                n_samples = 12
                deltas = np.linspace(0.02, d_max, n_samples)

                for d in deltas:
                    new_conv = conv + d * g
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0 and best_v < best_score - 1e-10:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-10:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                n_perturb = max(8, n // 60)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.12 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                if no_improve >= 10:
                    break
        else:
            no_improve = 0

        if verbose and total_improvements > 0 and total_improvements % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD improv={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def symmetrize(a):
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    s = a_sym.sum()
    if s < 1e-15:
        return a.copy()
    return a_sym * (a.sum() / s)


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = np.maximum(best_seq, 0.01)
        best_seq = best_seq + 0.01 * np.abs(rng.standard_normal(n_base))
        best_seq = best_seq / best_seq.sum()
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            old_best = best_score
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (was {old_best:.6f}, n={len(a)}, t={elapsed:.0f}s)")

    # Phase 1: LSE-Adam
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.14, 200.0)
    print(f"\n=== Phase 1: LSE-Adam n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_out, _ = lse_adam_optimize(best_seq, t1, verbose=True)
        check_and_update(a_out, f"lse1_n{len(a_out)}")

    # Phase 2: Vegas
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.09, 130.0)
    print(f"\n=== Phase 2: Vegas n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = vegas_optimize(best_seq, t2, verbose=True)
        check_and_update(a_out, f"vegas_n{len(a_out)}")

    # Phase 3: LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.33, 480.0)
    print(f"\n=== Phase 3: LP Nesterov n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 60:
        a_out, _ = nesterov_lp_optimize(best_seq, t3, solver, verbose=True)
        check_and_update(a_out, f"lp1_n{len(a_out)}")

    # Phase 4: LSE-Adam round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.08, 100.0)
    print(f"\n=== Phase 4: LSE-Adam r2 n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = lse_adam_optimize(best_seq, t4, verbose=False)
        check_and_update(a_out, f"lse2_n{len(a_out)}")

    # Phase 5: LP round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.20, 250.0)
    print(f"\n=== Phase 5: LP r2 n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 40:
        a_out, _ = nesterov_lp_optimize(best_seq, t5, solver, verbose=False)
        check_and_update(a_out, f"lp2_n{len(a_out)}")

    # Phase 6: Pair CD
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 6: Pair CD n={len(best_seq)} (budget={t6:.0f}s) ===")
    if t6 > 20:
        a_out, _ = pair_cd_optimize(best_seq, t6, rng, verbose=True)
        check_and_update(a_out, f"pcd1_n{len(a_out)}")

    # Phase 7: Symmetrize + short LP
    a_sym = symmetrize(best_seq)
    check_and_update(a_sym, f"sym_n{len(a_sym)}")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        t7 = min(remaining * 0.25, 80.0)
        print(f"\n=== Phase 7: Short LP n={len(best_seq)} ===")
        a_out, _ = nesterov_lp_optimize(best_seq, t7, solver, verbose=False)
        check_and_update(a_out, f"lp3_n{len(a_out)}")

    # Phase 8: Multi-length
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        lengths = [n_base + 200, n_base + 400, n_base - 200]
        for n_try in lengths:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 50:
                break
            if n_try < 500 or n_try > 4000:
                continue
            t_try = min(remaining * 0.25, 35.0)
            print(f"\n=== Multi-length: n={n_try} (budget={t_try:.0f}s) ===")
            init_try = interpolate_seq(best_seq, n_try)
            a_out, _ = lse_adam_optimize(init_try, min(t_try, 20.0), verbose=False)
            check_and_update(a_out, f"ml_n{n_try}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 25:
                a_out, _ = pair_cd_optimize(a_out, min(remaining * 0.4, 15.0), rng, verbose=False)
                check_and_update(a_out, f"ml_pcd_n{n_try}")

    # Final
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0] * 10

    return [float(x) for x in best_seq]
