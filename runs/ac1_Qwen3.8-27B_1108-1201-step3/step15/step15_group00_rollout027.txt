    # Convolution
    c = fast_conv(a)
    
    # LSE
    cs = c / tau
    cs_max = cs.max()
    ecs = np.exp(cs - cs_max)
    Z = ecs.sum()
    LSE = cs_max + np.log(Z)
    obj = 2.0 * n * LSE / tau
    
    # Weights
    w = ecs / Z  # shape (2n-1,)
    
    # s_j = sum_k w_k * a_{k-j} for j=0,...,n-1
    # s = cross_correlation of w with a (w lagged)
    # s_j = sum_{i=0}^{n-1} a_i * w_{i+j}
    # = conv(w, a_rev)[n-1-j] where a_rev = a[::-1]
    a_rev = a[::-1]
    conv_w_ar = fast_conv(w, a_rev)  # length (2n-1) + n - 1 = 3n-2
    s = conv_w_ar[n-1::-1][:n]
    
    # w_c = sum_k w_k * c_k
    w_c = np.dot(w, c)
    
    # Gradient
    grad = (4.0 * n / tau) * a * (s - w_c)
    
    return obj, a, grad

def lse_lbfgs(a0, n, time_limit, verbose=True):
    """Run LSE-smoothed L-BFGS with continuation."""
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    a = np.maximum(a0, 1e-15)

    best_wm = fast_conv(a).max()

    c = fast_conv(a)

    tau_factor = 0.75
    min_tau = c_max * 0.005
    
    phase = 0
    while tau > min_tau:
        phase += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        phase_t0 = time.monotonic()
        
        def func_and_grad(theta_vec):
            obj, a_curr, grad = compute_lse_grad(theta_vec, n, tau)
            return obj, grad
        
        # Run L-BFGS
        try:
            res = minimize(
                fun=lambda th: compute_lse_grad(th, n, tau)[0],
                x0=theta,
                jac=lambda th: compute_lse_grad(th, n, tau)[2],
                method='L-BFGS-B',
                options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-10, 'maxfun': 2000}
            )
            theta = res.x
        except Exception as e:
            if verbose:
                print(f"    L-BFGS error in phase {phase}: {e}")
            break
        
        # Update a and check
        tmax = theta.max()
        et = np.exp(theta - tmax)
        a = et / et.sum()
        a = np.maximum(a, 0)

def lp_frank_wolfe(a0, time_limit, solver, verbose=True):
    """LP-based Frank-Wolfe for the exact max objective."""
    n = len(a0)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_wm = fast_conv(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    for t in range(1, 10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_conv(y)
        M_curr = ac.max()
        
        if M_curr < 1e-14:
            break
        
        # Solve LP: max sum(b) s.t. b >= 0, (y*b)(k) <= M_curr for active k
        # Active k: those with ac(k) > 0.99 * M_curr
        active = np.where(ac > 0.995 * M_curr)[0]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = []
        for k in active:
            k_int = int(k)
            # (y*b)(k) = sum_i b_i * y_{k-i} (valid indices)
            # = sum_i b_i * y[k-i] where 0 <= k-i < n
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            coeffs = y[k_int - i_start:k_int + 1 - (i_end - 1)]  # y[k - (i_start)...i_end-1]
            # Actually: (y*b)(k) = sum_{i=0}^{n-1} b_i * y_{k-i}
            # Valid when 0 <= k-i <= n-1, i.e., k-n+1 <= i <= k
            # i ranges from max(0, k-n+1) to min(n-1, k)
            i_lo = max(0, k_int - n + 1)
            i_hi = min(n - 1, k_int)
            if i_lo <= i_hi:
                coeff = y[k_int - np.arange(i_lo, i_hi + 1)]
                constraints.append(cp.sum(coeff * b_var[i_lo:i_hi+1]) <= M_curr)
        
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        my = fast_conv(y).max()
        if my < best_wm:
            best_wm = my
            best_a = y.copy()
        
        mx = fast_conv(x).max()
        if mx < best_wm:
            best_wm = mx
            best_a = x.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_wm
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.8f}")
    
    return best_a

def coordinate_polish(a0, time_limit, verbose=False, rng_seed=999):
    """Coordinate descent polish."""
    n = len(a0)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_wm = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 100
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        # Find the peak lag
        k_peak = int(np.argmax(c))
        
        # For the peak lag, the contribution is sum_i a_i * a_{k_peak - i}
        # Try to reduce mass at positions that contribute most to the peak
        i_lo = max(0, k_peak - n + 1)
        i_hi = min(n - 1, k_peak)
        idx = np.arange(i_lo, i_hi + 1)
        contrib = a[idx] * a[k_peak - idx]
        
        if contrib.max() < 1e-20:
            stall += 1
            break
        
        # Find the position with maximum contribution
        src = idx[np.argmax(contrib)]
        
        # Try moving mass from src to a position that doesn't contribute to the peak
        best_dst = -1
        best_improvement = 0
        
        # Candidate destinations: positions not in the peak's support
        in_peak = set(idx.tolist())
        candidates = [i for i in range(n) if i not in in_peak]
        rng.shuffle(candidates)
        candidates = candidates[:min(len(candidates), 200)]
        
        for dst in candidates:
            if a[src] < 1e-15:
                break
            for frac in [0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_wm - 1e-15:
                    improvement = best_wm - M_trial
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_dst = dst
                        best_frac = frac
                        best_trial = a_trial.copy()
        
        if best_dst >= 0:
            a = best_trial
            best_wm = fast_conv(a).max()
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                st = a.sum()
                if st > 1e-15:
                    a = a / st
                    M_new = fast_conv(a).max()
                    if M_new < best_wm:
                        best_wm = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = a / a.sum()
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_wm
        print(f"    Coord polish ({elapsed:.1f}s): J={J:.8f}")
    
    return best_a

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    try:

        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')
