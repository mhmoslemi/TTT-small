def project_simplex(v, s):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if s <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_T(a, indices):
    """Build constraint matrix for given convolution positions."""
    n = len(a)
    m = len(indices)
    T = np.zeros((m, n))
    for row, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            T[row, j] = a[k - j]
    return T

def lp_solve(a, M, solver, active_idx):

def subgradient_descent(a0, time_limit, n_iter=50000, verbose=False):
    """Subgradient-based directional descent with line search."""
    n = len(a0)
    size = 2 * n - 1
    norm = np.sqrt(2.0 * n)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    eta = norm * 0.1  # initial step size
    n_iter_actual = 0
    
    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        n_iter_actual = it
        
        conv = fast_autocorr(a)
        M = conv.max()
        peaks = np.where(conv >= M * 0.999)[0]
        
        # Compute subgradient
        g = np.zeros(n)
        for k in peaks:
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for j in range(lo, hi):
                g[j] += 2.0 * a[k - j]
        g /= len(peaks)
        
        # Center the gradient (to maintain sum constraint)
        g_centered = g - g.mean()
        
        # Line search
        best_step = eta
        best_val = M
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            step = eta * frac
            a_try = a - step * g_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try * norm / s_try
            val = fast_autocorr(a_try).max()
            if val < best_val:
                best_val = val
                best_step = step
        
        a = np.maximum(a - best_step * g_centered, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            a = best_a.copy()
            break
        a = a * norm / s_a
        
        if best_val < best_score:
            best_score = best_val
            best_a = a.copy()
            eta = min(eta * 1.05, norm * 0.5)
        else:
            eta *= 0.95
        
        if verbose and it % 5000 == 0:
            print(f"    SD it={it} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}, eta={eta:.4f}")
    
    return best_a, best_score

    if s < 1e-15:
        return a, float('inf')
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)
        
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = fast_autocorr(a)
            M = conv.max()
            peaks = np.where(conv >= M * 0.999)[0]
            
            # Compute "pressure" on coordinate i from peaks
            pressure = 0.0
            for k in peaks:
                idx = k - i
                if 0 <= idx < n:
                    pressure += 2.0 * a[idx]
            pressure /= len(peaks)
            
            # Try to decrease high-pressure coordinates, increase low-pressure
            # Optimal value: try a few candidates
            ai = a[i]
            candidates = [ai * 0.5, ai * 0.75, ai * 1.25, ai * 1.5, 0.0, ai]
            best_val = M
            best_candidate = ai
            
            for cand in candidates:
                if cand < 0:
                    continue
                a_try = a.copy()
                a_try[i] = cand
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try * norm / s_try
                val = fast_autocorr(a_try).max()
                if val < best_val:
                    best_val = val
                    best_candidate = cand
            
            if best_val < best_score - 1e-13:
                a[i] = best_candidate
                a = a * norm / a.sum()
                best_score = best_val
                best_a = a.copy()
                pass_improved = True
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 8:
                break
        else:
            no_improve = 0
    
    return best_a, best_score

def jitter_search(a0, time_limit, rng, verbose=False):
    """Random jitter local search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    jitter_scale = 0.05 * norm / n
    n_accepted = 0
    n_total = 0
    
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        n_total += 1
        
        # Randomly select a few coordinates to perturb
        n_perturb = rng.integers(1, min(10, n // 10) + 1)
        idx = rng.choice(n, size=n_perturb, replace=False)
        deltas = rng.standard_normal(n_perturb) * jitter_scale
        
        a_try = a.copy()
        a_try[idx] += deltas
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            continue
        a_try = a_try * norm / s_try
        
        val = fast_autocorr(a_try).max()
        if val < best_score:
            a = a_try
            best_score = val
            best_a = a.copy()
            n_accepted += 1
            jitter_scale = min(jitter_scale * 1.01, 0.2 * norm / n)
        else:
            jitter_scale *= 0.999
        
        if n_total % 10000 == 0 and n_total > 0:
            rate = n_accepted / n_total
            if rate < 0.001 and it > 10000:
                break
            if verbose:
                print(f"    JS it={it} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}, rate={rate:.4f}")
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss
