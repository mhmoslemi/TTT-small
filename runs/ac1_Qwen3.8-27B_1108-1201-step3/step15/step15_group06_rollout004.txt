def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    if s < 0.01:

    """Build constraint matrix for given autocorrelation indices."""
    n = len(a)
    m = len(indices)
    T = np.zeros((m, n))
    for row, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                T[row, j] = a[idx]
    return T

    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if val.sum() < 1e-15:
        return None
    if normalize_sum is not None:
        val = val * normalize_sum / val.sum()

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_autocorr(a)
    return float(2.0 * len(a) * conv.max() / (s * s))


def build_T(a, indices):
    n = len(a)
    m = len(indices)
    T = np.zeros((m, n))
    for row in range(m):
        k = indices[row]
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                T[row, j] = a[idx]
    return T


def solve_lp_max_sum(a, M, active_idx, solver):
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    T = build_T(a, active_idx)
    cons = [T @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if val.sum() < 1e-15:
        return None
    return val


def solve_lp_min_max(a, active_idx, solver):
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    T = build_T(a, active_idx)
    cons = [T @ b <= t, cp.sum(b) == 1.0, b >= 0]
    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if val.sum() < 1e-15:
        return None, None
    t_val = float(t.value) if t.value is not None else float('inf')
    return val, t_val


def nesterov_lp(a0, time_limit, solver, seed=0, verbose=True):
    """Improved LP-based Nesterov with hysteresis active sets and adaptive blending."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    size = 2 * n - 1
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_conv = fast_autocorr(a)
    best_score = best_conv.max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    stall_reset = 100

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-14:
            break

        # Adaptive active set with hysteresis
        if active_set is None or t % 30 == 0:
            if active_set is None:
                # Initial: use top constraints
                tol_init = 0.05
                threshold = M_curr * (1.0 - tol_init)
                active_set = np.where(conv_y >= threshold)[0]
                if len(active_set) < 10:
                    active_set = np.where(conv_y >= M_curr * 0.6)[0]
            else:
                # Hysteresis: add if clearly violated, remove if far below
                add_threshold = M_curr * 0.85
                rem_threshold = M_curr * 0.5
                current_conv_check = conv_y
                to_add = np.where((current_conv_check >= add_threshold) &
                                  ~np.isin(np.arange(size), active_set))[0]
                to_remove = np.where((active_set >= 0) &
                                     (current_conv_check[active_set] < rem_threshold))[0]
                if len(to_add) > 0:
                    active_set = np.union1d(active_set, to_add)
                if len(to_remove) > 0:
                    active_set = np.delete(active_set, to_remove)

                if len(active_set) < 8:
                    active_set = np.where(conv_y >= M_curr * 0.6)[0]
                if len(active_set) > 0.85 * size:
                    active_set = np.where(conv_y >= M_curr * 0.92)[0]

        # Solve max-sum LP
        b_max = solve_lp_max_sum(y, M_curr, active_set, solver)

        # Adaptive blending: more min-max early, more max-sum late
        use_min_max = (t < 200) or (t % 4 == 0)
        b_min = None
        if use_min_max:
            b_min, _ = solve_lp_min_max(y, active_set, solver)

        if b_max is None:
            if t % 30 == 0:
                active_set = None
                b_max = solve_lp_max_sum(y, M_curr, np.arange(size), solver)
            if b_max is None:
                continue

        b_sum = b_max.sum()
        if b_sum < 1e-15:
            continue
        g = b_max / b_sum  # normalized to sum=1

        if b_min is not None and b_min.sum() > 1e-10:
            # Time-varying blend: start with 50% min-max, shift to 20%
            blend = 0.2 + 0.3 * np.exp(-t / 150.0)
            g = (1.0 - blend) * g + blend * (b_min / b_min.sum())

        # Check for constraint violations on inactive set
        if len(active_set) < size:
            inactive = np.setdiff1d(np.arange(size), active_set)
            if len(inactive) > 0:
                # Sample check for efficiency
                if len(inactive) > 200:
                    sample_idx = rng.choice(inactive, 200, replace=False)
                else:
                    sample_idx = inactive
                T_check = build_T(y, sample_idx)
                viol = T_check @ g
                viol_positions = np.where(viol > M_curr * (1.0 + 1e-5))[0]
                if len(viol_positions) > 0:
                    new_active = inactive[viol_positions]
                    active_set = np.union1d(active_set, new_active)

        # Nesterov update with adaptive momentum
        alpha_t = max(2.0 / (t + 2.0), 0.05)
        y_new = (1.0 - alpha_t) * y + alpha_t * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y  # normalize to sum=1
            y_new = y_new * norm  # rescale to norm

        if t == 1:
            a_new = g * norm
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.90)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y
        y = y_new
        a = a_new

        # Track best (using sum=1 normalized version for comparison)
        score_y = fast_autocorr(y / y.sum() if y.sum() > 0 else y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a / a.sum() if a.sum() > 0 else a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Ergodic average
        avg_a += a
        n_avg += 1
        if n_avg % 30 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc = fast_autocorr(avg_check).max()
                if sc < best_score:
                    best_score = sc
                    best_a = avg_check.copy()
                    stall_count = 0

        # Restart if stalled
        if stall_count > stall_reset:
            y = best_a.copy()
            a = best_a.copy()
            avg_a = np.zeros(n)
            n_avg = 0
            active_set = None
            stall_count = 0

        if verbose and t % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best_max_corr={best_score:.10f}")

    return best_a, best_score


def subgradient_polish(a0, time_limit, verbose=True, seed=42):
    """Subgradient method with exact subgradients for the peak autocorrelation."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    size = 2 * n - 1
    norm = np.sqrt(2.0 * n)

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    f_star = best_score
    duality_gap = best_score * 0.01  # initial gap estimate

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()
        score = M_curr / (a.sum() ** 2) * 2 * n  # actual objective
        # For tracking: use max_corr / norm^2 as proxy
        max_corr = M_curr

        # Find subgradient: weighted average of gradients at peak indices
        # Peak indices: where conv is within 1% of max
        peak_tol = 0.02
        peak_mask = conv >= M_curr * (1.0 - peak_tol)
        peak_indices = np.where(peak_mask)[0]

        # Subgradient components: g[j] = sum_{k in peaks} w_k * a[k-j]
        weights = conv[peak_indices] / conv[peak_indices].sum()

        g = np.zeros(n)
        for i, k in enumerate(peak_indices):
            w = weights[i]
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for j in range(lo, hi):
                idx = k - j
                if 0 <= idx < n:
                    g[j] += w * a[idx]

        # Project gradient: remove mean to maintain sum
        g_centered = g - g.mean()
        g_norm = np.linalg.norm(g_centered)
        if g_norm < 1e-15:
            break

        # Polyak-like step size
        # f_current - f_lower_bound over ||g||^2
        # Lower bound: 1/(2n-1) * norm^2 (theoretical minimum of max autocorr)
        f_lower = norm * norm / (2 * n - 1)
        gap = max_corr / (a.sum() ** 2) * 2 * n - 2 * n / (a.sum() ** 2) * f_lower
        # Simpler: use gap = current_score - best_score_history + epsilon
        eta = max(0.001, min(0.1, (best_score - max_corr / norm * norm + 0.01) / (g_norm ** 2 + 1e-10)))
        # Actually use a simpler adaptive step
        eta = 0.1 / (1.0 + t * 0.01)

        a_new = a - eta * g_centered
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        a = a_new

        sc = fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def coord_descent_fast(a0, time_limit, rng, verbose=True):
    """Fast coordinate descent using analytic structure of autocorrelation updates."""
    n = len(a0)
    size = 2 * n - 1
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break

            ai = a[i]
            if ai < 1e-10:
                # Try increasing from zero
                for trial in [norm * 0.001, norm * 0.005, norm * 0.01]:
                    a_try = a.copy()
                    a_try[i] = trial
                    s_try = a_try.sum()
                    a_try = a_try * norm / s_try
                    sc = fast_autocorr(a_try).max()
                    if sc < best_score - 1e-12:
                        a = a_try
                        best_score = sc
                        best_a = a.copy()
                        pass_improved = True
                        ai = trial
                        break
                continue

            # Try several fractions of current value
            best_val = best_score
            best_frac = 1.0
            for frac in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 0.3]:
                a_try = a.copy()
                a_try[i] = ai * frac
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try * norm / s_try
                sc = fast_autocorr(a_try).max()
                if sc < best_val - 1e-13:
                    best_val = sc
                    best_frac = frac

            if best_frac != 1.0:
                a[i] = ai * best_frac
                s = a.sum()
                a = a * norm / s
                conv = fast_autocorr(a)
                new_score = conv.max()
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
                    pass_improved = True

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def perturb_seq(a, rng, scale=0.05):
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.02)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def structured_init(n, style=0):

    if style == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif style == 1:
        a = np.sin(np.pi * x) ** 1.2
    elif style == 2:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(8 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif style == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif style == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif style == 5:
        a = 0.4 + 0.6 * np.sin(np.pi * x) ** 0.8
    elif style == 6:
        a = 1.0 + 0.25 * np.cos(6 * np.pi * x) + 0.1 * np.cos(12 * np.pi * x)
    else:
        a = np.sin(np.pi * x) ** 0.6 * (1.0 + 0.15 * np.cos(10 * np.pi * x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

    n_base = len(best_seq)

    # === Phase 1: LP Nesterov on parent (40% time) ===
    print(f"\n=== Phase 1: LP Nesterov n={n_base} ===")
    t1 = min(remaining_time() * 0.40, 550.0)
    if t1 > 80:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t1, solver, verbose=True, seed=0)
        try_update(a_out, f"lp1_n{n_base}")

    # === Phase 2: Subgradient polish (10% time) ===
    print(f"\n=== Phase 2: Subgradient n={len(best_seq)} ===")
    t2 = min(remaining_time() * 0.10, 150.0)
    if t2 > 30:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = subgradient_polish(a_norm, t2, verbose=True, seed=42)
        try_update(a_out, f"subgrad_n{n_cur}")

    # === Phase 3: Coordinate descent (15% time) ===
    print(f"\n=== Phase 3: CD n={len(best_seq)} ===")
    t3 = min(remaining_time() * 0.15, 200.0)
    if t3 > 30:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coord_descent_fast(a_norm, t3, rng, verbose=True)
        try_update(a_out, f"cd_n{n_cur}")

    # === Phase 4: Perturbation + LP restarts (15% time) ===
    print(f"\n=== Phase 4: Perturbation + LP ===")
    t4_total = min(remaining_time() * 0.15, 200.0)
    n_restarts = 3
    t4_each = t4_total / n_restarts
    for r in range(n_restarts):
        if remaining_time() < 80:
            break
        t4 = min(t4_each, remaining_time() - 50)
        if t4 < 30:
            break
        n_cur = len(best_seq)
        scale = 0.02 + 0.02 * r
        pert = perturb_seq(best_seq, rng, scale)
        a_norm = pert * np.sqrt(2.0 * n_cur) / pert.sum()
        a_out, _ = nesterov_lp(a_norm, t4, solver, verbose=False, seed=100 + r * 13)
        try_update(a_out, f"pert{r}_n{n_cur}")

    # === Phase 5: Longer length (10% time) ===
    n_long = len(best_seq) + 400
    if remaining_time() > 150:
        print(f"\n=== Phase 5: LP Nesterov n={n_long} ===")
        t5 = min(remaining_time() * 0.10, 170.0)
        if t5 > 60:
            init_long = interpolate_seq(best_seq, n_long)
            a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
            a_out, _ = nesterov_lp(a_norm, t5, solver, verbose=True, seed=7)
            try_update(a_out, f"lp_long_n{n_long}")

    # === Phase 6: Even longer (5% time) ===
    n_long2 = len(best_seq) + 800
    if remaining_time() > 100:
        print(f"\n=== Phase 6: LP Nesterov n={n_long2} ===")
        t6 = min(remaining_time() * 0.05, 100.0)
        if t6 > 40:
            init_long2 = interpolate_seq(best_seq, n_long2)
            a_norm = init_long2 * np.sqrt(2.0 * n_long2) / init_long2.sum()
            a_out, _ = nesterov_lp(a_norm, t6, solver, verbose=False, seed=13)
            try_update(a_out, f"lp_long2_n{n_long2}")

    # === Phase 7: Final LP polish (all remaining) ===
    n_best = len(best_seq)
    if remaining_time() > 60:
        print(f"\n=== Phase 7: Final LP n={n_best} ===")
        t7 = remaining_time() - 40
        if t7 > 50:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = nesterov_lp(a_norm, t7, solver, verbose=True, seed=999)
            try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 8: Final CD ===
    n_best = len(best_seq)
    if remaining_time() > 40:
        print(f"\n=== Phase 8: Final CD n={n_best} ===")
        t8 = remaining_time() - 20
        if t8 > 20:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = coord_descent_fast(a_norm, t8, rng, verbose=False)
            try_update(a_out, f"cd_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_sequence([float(x) for x in np.maximum(best_seq, 0.0)])
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in np.maximum(best_seq, 0.0)]
