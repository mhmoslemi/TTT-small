_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:

            prob.solve(solver=s)

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T

def lp_solve(a, M, solver, active=None):
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    if active is not None and 0 < len(active) < size:
        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    duals = None
    try:
        con = constraints[0]
        if con.dual_value is not None:
            if active is not None and 0 < len(active) < size:
                full_duals = np.zeros(size)
                full_duals[active] = np.asarray(con.dual_value, dtype=np.float64)
                duals = full_duals
            else:
                duals = np.asarray(con.dual_value, dtype=np.float64)
    except Exception:
        pass
    return val, duals

def lse_gradient_descent(a0, time_limit, verbose=True):

    for si, T_val in enumerate(temps):

            break
        remaining = time_limit - (time.monotonic() - t0)

            break

        for it in range(n_steps):

            if max_c < 1e-15:

            grad = 2.0 * (g1 + g2)

            step = 2.0 / (it + 10.0)
            a_new = a - step * grad / T_val
            a_new = np.maximum(a_new, 0.0)

            if s_new < 1e-15:

            else:

            sc = fast_conv_self(a).max()
            if sc < best_score:

                best_a = a.copy()
        
        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best={best_score:.8f}, score={2*n*best_score/norm**2:.6f}")
    
    return best_a, best_score
