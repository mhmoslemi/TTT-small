                # So (e_i * a)[k] = (a * e_i)[k] = a[k-i] * 1{0 <= k-i < n}.
                
                # Therefore:
                # conv_new[k] = conv[k] - d*(a[k-i] + a[k-i])*1{0<=k-i<n} + d*(a[k-j] + a[k-j])*1{0<=k-j<n}
                #              + d²*(1{0<=k-2i<... } - 1{...}) ... 
                # Actually the d² terms:
                # (e_i * e_i)[k] = 1 iff k-i=i iff k=2i (and 0<=i<n, which is true)
                # (e_i * e_j)[k] = 1 iff k-i=j iff k=i+j
                # (e_j * e_i)[k] = 1 iff k-j=i iff k=i+j
                # (e_j * e_j)[k] = 1 iff k=2j
                
                # So: conv_new[k] = conv[k] - 2d*a[k-i]*1{0<=k-i<n} + 2d*a[k-j]*1{0<=k-j<n}
                #                    + d²*(1{k=2i} - 1{k=i+j} - 1{k=i+j} + 1{k=2j})
                #                  = conv[k] + 2d*(a[k-j]*1{0<=k-j<n} - a[k-i]*1{0<=k-i<n})
                #                    + d²*(1{k=2i} + 1{k=2j} - 2*1{k=i+j})
                
                # Let me just compute this numerically for each candidate:
                gain = np.zeros(size)
                # 2d * (a[k-j] - a[k-i]) terms
                for k in range(size):
                    val = 0.0
                    if 0 <= k - j < n:
                        val += a[k-j]
                    if 0 <= k - i < n:
                        val -= a[k-i]
                    gain[k] = 2.0 * val
                
                # d² correction
                if 2*i < size:
                    gain[2*i] += 1.0  # will multiply by d²
                if 2*j < size:
                    gain[2*j] += 1.0
                if i + j < size:
                    gain[i+j] -= 2.0
                    
                # Evaluate: new_conv[k] = conv[k] + d*gain1[k] + d²*gain2[k]
                # where gain1[k] = 2*(a[k-j]*1{valid} - a[k-i]*1{valid})
                # and gain2[k] = 1{k=2i} + 1{k=2j} - 2*1{k=i+j}
                
                # Actually let me just do it the simple way: compute the two gain vectors
                gain1 = np.zeros(size)
                lo = j
                hi = min(size, j + n)
                gain1[lo:hi] += a[0:hi-lo]
                lo2 = j - (n-1)
                hi2 = min(size, j + 1)
                if lo2 < 0:
                    lo2 = 0
                # Hmm, a[k-j] is valid when 0 <= k-j < n, i.e., j <= k <= j+n-1
                # That's already covered above. But we also need the "reflected" part?
                # No, for convolution (not correlation), a[k-j] is just a[k-j].
                
                # Wait, I think I'm overcomplicating this. Let me just use the formula:
                # conv[a'][k] where a' = a - d*e_i + d*e_j
                # = conv_a[k] + d * (conv_{e_j, a}[k] + conv_{a, e_j}[k] - conv_{e_i, a}[k] - conv_{a, e_i}[k])
                #   + d² * (conv_{e_j, e_j}[k] - conv_{e_j, e_i}[k] - conv_{e_i, e_j}[k] + conv_{e_i, e_i}[k])
                # conv_{e_c, a}[k] = a[k-c] (if 0 <= k-c < n)
                # conv_{a, e_c}[k] = a[k-c] (same thing by commutativity of convolution)
                # conv_{e_c, e_d}[k] = 1 if k = c+d, else 0
                
                # So: linear part = 2d * (a[k-j] * 1{0<=k-j<n} - a[k-i] * 1{0<=k-i<n})
                # quadratic part = d² * (1{k=2j} + 1{k=2i} - 2*1{k=i+j})
                
                # For a given delta d, new_conv[k] = conv[k] + 2d*lin[k] + d²*quad[k]
                # where lin[k] = a[k-j]*1{0<=k-j<n} - a[k-i]*1{0<=k-i<n}
                # and quad[k] = 1{k=2j} + 1{k=2i} - 2*1{k=i+j}
                
                # We want to find d that minimizes max_k new_conv[k]
                # This is a 1D minimization!
                
                lin = np.zeros(size)
                # a[k-j] for 0 <= k-j < n => j <= k <= j+n-1
                llo = j
                lhi = min(size, j + n)
                lin[llo:lhi] += a[0:lhi-llo]
                # -a[k-i] for 0 <= k-i < n => i <= k <= i+n-1
                mlo = i
                mhi = min(size, i + n)
                lin[mlo:mhi] -= a[0:mhi-mlo]
                
                quad = np.zeros(size)
                if 2*j < size:
                    quad[2*j] += 1
                if 2*i < size:
                    quad[2*i] += 1
                if i+j < size:
                    quad[i+j] -= 2
                
                # Minimize max_k (conv[k] + 2d*lin[k] + d²*quad[k]) over d in [0, a[i]]
                # This is a convex 1D problem (max of convex functions)
                # Use ternary search or golden section
                
                d_lo = 0.0
                d_hi = ai  # can't shift more than a[i]
                
                # Golden section search
                gr = (np.sqrt(5) - 1) / 2
                for _ in range(30):
                    d1 = d_hi - gr * (d_hi - d_lo)
                    d2 = d_lo + gr * (d_hi - d_lo)
                    v1 = np.max(conv + 2*d1*lin + d1*d1*quad)
                    v2 = np.max(conv + 2*d2*lin + d2*d2*quad)
                    if v1 < v2:
                        d_hi = d2
                    else:
                        d_lo = d1
                
                d_opt = (d_lo + d_hi) / 2
                val_opt = np.max(conv + 2*d_opt*lin + d_opt*d_opt*quad)
                
                if val_opt < best_val - 1e-10:
                    best_val = val_opt
                    best_delta = d_opt
                    best_j = j
            
            if best_j >= 0 and best_delta > 1e-10:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                a = a / s_new
                new_max = compute_score(a)
                if new_max < best_score - 1e-10:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
        
        if not pass_improve if False else not pass_improved:
            no_improve += 1
            if no_improve >= 6:
                break
            if no_improve >= 2:
                # Perturb
                n_perturb = max(10, n // 30)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.2 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a / s_new
                no_improve = 0
        else:
            no_improve = 0
        
        if verbose and (idx + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakCD ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def compute_score(a):
    """Compute the evaluation score for a normalized sequence."""
    n = len(a)
    conv = fast_conv_self(a)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)


def shape_init(n, rng, shape_type=None):
    """Generate initial shapes."""
    if shape_type is None:
        shape_type = int(rng.integers(0, 8))
    
    x = np.linspace(0.0, 1.0, n)
    
    if shape_type == 0:
        # Exponential decay
        alpha = rng.uniform(1.0, 4.0)
        a = np.exp(-alpha * x)
    elif shape_type == 1:
        # Power law decay
        beta = rng.uniform(1.0, 3.0)
        a = (1.0 - x) ** beta
    elif shape_type == 2:
        # Linear ramp (increasing)
        a = 0.1 + 0.9 * x
    elif shape_type == 3:
        # Cosine bump (asymmetric)
        center = rng.uniform(0.3, 0.7)
        width = rng.uniform(0.3, 0.6)
        a = np.exp(-((x - center) / width) ** 2)
    elif shape_type == 4:
        # Skewed cosine
        a = np.cos(np.pi * x * rng.uniform(0.5, 1.0)) ** 2
    elif shape_type == 5:
        # Two-bump asymmetric
        c1, c2 = 0.3, 0.7
        a = np.exp(-((x-c1)/0.15)**2) + 0.6*np.exp(-((x-c2)/0.2)**2)
    elif shape_type == 6:
        # S-curve
        t = rng.uniform(2.0, 5.0)
        a = 1.0 / (1.0 + np.exp(-t*(x - 0.5)))
    else:
        # Uniform with noise
        a = np.ones(n) + 0.1 * rng.standard_normal(n)
    
    a = np.maximum(a, 0.0)
    a += 0.01 * rng.random(n)  # Small noise to avoid symmetries
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
