"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LSE gradient descent with diagonal Newton correction
2. LP best-response with Nesterov acceleration and aggressive constraint screening
3. Targeted coordinate descent on peak lags

5. Final fine-grained polish

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print("  Using solver (fallback): ECOS")
    return _solver


def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_T_sparse(a, active):
    """Build the convolution matrix T restricted to active rows."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((len(active), n))
    for idx, m in enumerate(active):
        # conv[m] = sum_i a[i] * a[m-i] (valid indices)
        # = sum_i T[m,i] * a[i] where T[m,i] = a[m-i] if valid
        lo = max(0, m - n + 1)
        hi = min(n, m + 1)
        for i in range(lo, hi):
            T[idx, i] = a[m - i]
    return T


def build_T_full(a):
    """Build the full convolution matrix T."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(size):
        lo = max(0, m - n + 1)
        hi = min(n, m + 1)
        for i in range(lo, hi):
            T[m, i] = a[m - i]
    return T


def lp_solve_fast(a, M, solver, active=None, n_active_max=80):
    """Solve the LP: max sum(b) s.t. T@b <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    
    if active is None or len(active) == 0:
        active = np.arange(size)
    
    if len(active) > n_active_max:
        # Sort by how close they are to M and take the most constraining
        # (We don't have conv here, so just take evenly spaced ones from the top)
        # Actually, the caller should pass the most important active set
        pass
    
    T = build_T_sparse(a, active)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)


def lse_gradient_descent(a0, time_limit, verbose=True):
    """LSE gradient descent with diagonal Newton correction."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 15
    temps = np.geomspace(100.0, 0.1, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        n_steps = max(50, min(500, int(remaining * 8)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)

            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            # Diagonal Newton: approximate Hessian diagonal
            # H_ii ≈ 2 * sum_k w_k * (contribution of a[i] to conv[k] via a[i]*a[i])
            # For simplicity, use grad^2 + epsilon as diagonal approximation
            diag_h = grad * grad + 1e-8
            step = grad / diag_h  # Newton step
            
            # Scale the step
            step_norm = np.linalg.norm(step)
            if step_norm > 0:
                max_step = norm / np.sqrt(n)
                if step_norm > max_step:
                    step = step * max_step / step_norm
            
            a_new = a - step
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best={best_score:.8f}")

    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, n_active_max=60):
    """LP Nesterov with aggressive constraint screening and plateau shaking."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1
    no_improve = 0
    rng = np.random.default_rng(12345)

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Aggressive constraint screening: top n_active_max constraints
        if n_active_max < size:
            top_idx = np.argpartition(conv_y, -n_active_max)[-n_active_max:]
            active = top_idx
        else:
            active = np.arange(size)

        b = lp_solve_fast(y, M, solver, active, n_active_max)
        if b is None:
            b = lp_solve_fast(y, M, solver, None, size)
        if b is None:
            no_improve += 1
            if no_improve > 10:
                break
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            no_improve += 1
            if no_improve > 10:
                break
            continue
        g = b * norm / b_sum

        # Verify dropped constraints (only if we used a reduced set)
        if len(active) < size:
            check_conv = fast_conv_cross(y, g)
            violated = np.where(check_conv > M * 1.005)[0]
            if len(violated) > 5:
                # Add violated constraints and re-solve
                active2 = np.unique(np.concatenate([active, violated]))
                b2 = lp_solve_fast(y, M, solver, active2[:n_active_max + 20], n_active_max + 20)
                if b2 is not None:
                    b = b2
                    b_sum = b.sum()
                    if b_sum < 1e-15:
                        continue
                    g = b * norm / b_sum

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        improved = False
        if sc_y < best_score - 1e-12:
            best_score = sc_y
            best_a = y.copy()
            improved = True

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score - 1e-12:
            best_score = sc_a
            best_a = a.copy()
            improved = True

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            # Plateau shake
            if no_improve > 50 and no_improve % 25 == 0:
                n_perturb = max(5, n // 50)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                y[idx_perturb] *= (1.0 + 0.05 * rng.standard_normal(n_perturb))
                y = np.maximum(y, 0.0)
                s_y = y.sum()
                if s_y > 1e-15:
                    y = y * norm / s_y
                no_improve = 0

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
