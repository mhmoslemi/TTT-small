        if b is None:
            b = lp_best_response(y, M, solver, None)
        if b is None:
            continue

        # Check constraint violations and add them
        if active_set is not None and len(active_set) < size:
            check = fast_conv_cross(y, b)
            violated = np.where(check > M * 1.001)[0]
            if len(violated) > 0:
                active_set = np.union1d(active_set, violated)
                b2 = lp_best_response(y, M, solver, active_set)
                if b2 is not None:
                    b = b2
                else:
                    b2 = lp_best_response(y, M, solver, None)
                    if b2 is not None:
                        b = b2

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M:.6f}, |act|={len(active_set) if active_set is not None else size}")

        # Check convergence
        if t > 50:
            prev_sc = fast_conv_self(y).max()
            if abs(prev_sc - M) < 1e-10 * M:
                break

    return best_a, best_score

def vegas_optimize(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 30
    etas = np.geomspace(0.15, 0.0005, n_stages)

    for si, eta_s in enumerate(etas):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        n_steps = max(30, min(100, int(remaining * 5)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_conv_self(a)
            M = conv.max()
            if M < 1e-15:
                break

            # Focus on the hottest peaks
            w = np.maximum(conv - M * 0.85, 0.0)
            ww = w / (w.sum() + 1e-15)

            W = rfft(ww, fft_size)
            A = rfft(a, fft_size)

            ca = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)[:n]
            pressure = ca + aw

            a = a * np.exp(-eta_s * pressure)
            a = np.maximum(a, 1e-12)
            a = a * norm / a.sum()

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose and si % 5 == 0:
            print(f"    Vegas stage {si} eta={eta_s:.5f}: best={best_score:.8f}")

    return best_a, best_score

def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            peak_lags = np.where(conv >= cur_max * 0.8)[0]

            j_candidates = set()
            for _ in range(10):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:15]:
                j_candidates.add(int((kl - i + n) % n))
                j_candidates.add(int((kl + i) % n))
            j_candidates.discard(i)

            best_delta = 0.0
            best_v = cur_max
            best_j = -1

            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]

                # Compute gradient of convolution change
                g = np.zeros(size)
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]

                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]

                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]

                d_max = min(ai, norm * 0.3)
                if d_max <= 0:
                    continue
                n_samples = 20
                deltas = np.linspace(0.01, d_max, n_samples)

                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0 and best_v < best_score - 1e-10:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-10:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 6:
                n_perturb = max(5, n // 60)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.2 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                if no_improve >= 12:
                    break
        else:
            no_improve = 0

        if verbose and total_improvements % 300 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD improv={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def structured_init(n, rng):
    choice = rng.integers(0, 14)
    x = np.linspace(0.0, 1.0, n)

    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        f1 = rng.integers(1, 6)
        f2 = rng.integers(6, 20)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * f1 * x) + 0.2 * np.cos(2 * np.pi * f2 * x)
    elif choice == 3:
        c1 = rng.uniform(0.2, 0.5)
        c2 = rng.uniform(0.5, 0.8)
        a = np.exp(-30.0 * (x - c1) ** 2) + np.exp(-30.0 * (x - c2) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.5
    elif choice == 6:
        a = np.exp(-5.0 * x) + 0.4
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        n_peaks = rng.integers(3, 10)
        centers = rng.uniform(0.05, 0.95, n_peaks)
        a = np.zeros(n)
        for c in centers:
            width = rng.uniform(0.02, 0.1)
            a += np.exp(-(x - c) ** 2 / (2 * width ** 2))
    elif choice == 9:
        a = np.sinc(x - 0.5) ** 2 + 0.1
    elif choice == 10:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * x) + 0.2 * np.sin(2 * np.pi * 7 * x)
    elif choice == 11:
        # Log-normal-like
        mu = rng.uniform(-1, 1)
        sigma = rng.uniform(0.3, 1.0)
        a = np.exp(-0.5 * ((x - 0.5 + mu * 0.3) / sigma) ** 2)
    elif choice == 12:
        # Piecewise linear with random breakpoints
        pts = np.sort(rng.uniform(0, 1, 5))
        vals = rng.uniform(0.1, 1.0, 5)
        a = np.interp(x, pts, vals)
    else:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a = a + 0.005 * np.abs(rng.standard_normal(n))
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1680.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            old_best = best_score
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (was {old_best:.6f}, n={len(a)}, t={elapsed:.0f}s)")

    # Phase 1: LSE-Adam initial optimization
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.15, 240.0)
    print(f"\n=== Phase 1: LSE-Adam n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = lse_adam_optimize(best_seq, t1, verbose=True)
        check_and_update(a_out, f"lse1_n{n_base}")

    # Phase 2: Vegas multiplicative
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 2: Vegas n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = vegas_optimize(best_seq, t2, verbose=True)
        check_and_update(a_out, f"vegas_n{len(best_seq)}")

    # Symmetrize before LP (can help convergence)
    a_sym = symmetrize(best_seq)
    check_and_update(a_sym, f"sym_n{len(best_seq)}")

    # Phase 3: LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.35, 600.0)
    print(f"\n=== Phase 3: LP Nesterov n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t3, solver, verbose=True)
        check_and_update(a_out, f"lp1_n{len(best_seq)}")

    # Phase 4: LSE-Adam round 2 from LP result
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 4: LSE-Adam round 2 n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = lse_adam_optimize(best_seq, t4, verbose=False)
        check_and_update(a_out, f"lse2_n{len(best_seq)}")

    # Phase 5: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 5: Pair CD n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = pair_cd_optimize(best_seq, t5, rng, verbose=True)
        check_and_update(a_out, f"pcd1_n{len(best_seq)}")

    # Phase 6: LP round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 6: LP round 2 n={len(best_seq)} (budget={t6:.0f}s) ===")
    if t6 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t6, solver, verbose=False)
        check_and_update(a_out, f"lp2_n{len(best_seq)}")

    # Phase 7: Sparsification + symmetrization
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Sparsify + Sym ===")
    a_spar = sparsify(best_seq, threshold_frac=0.002)
    check_and_update(a_spar, f"sparse_n{len(best_seq)}")
    a_sym2 = symmetrize(best_seq)
    check_and_update(a_sym2, f"sym2_n{len(best_seq)}")
    a_spar_sym = symmetrize(sparsify(best_seq, threshold_frac=0.005))
    check_and_update(a_spar_sym, f"sparse_sym_n{len(best_seq)}")

    # Phase 8: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        lengths_to_try = [n_base + 200, n_base + 400, n_base - 200, n_base + 600]
        for n_try in lengths_to_try:
            if remaining < 50:
                break
            if n_try < 100 or n_try > 5000:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            t_try = min(remaining * 0.35, 60.0)
            print(f"\n=== Multi-length: n={n_try} (budget={t_try:.0f}s) ===")
            init_try = interpolate_seq(best_seq, n_try)
            a_out, _ = lse_adam_optimize(init_try, min(t_try, 30.0), verbose=False)
            check_and_update(a_out, f"lsg_ml_n{n_try}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 25:
                a_out, _ = pair_cd_optimize(a_out, min(remaining * 0.5, 25.0), rng, verbose=False)
                check_and_update(a_out, f"pcd_ml_n{n_try}")

    # Phase 9: Final PCD + Vegas polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t9 = min(remaining * 0.5, 100.0)
        print(f"\n=== Phase 9: Final PCD (budget={t9:.0f}s) ===")
        a_out, _ = pair_cd_optimize(best_seq, t9, rng, verbose=True)
        check_and_update(a_out, f"pcd_final_n{len(best_seq)}")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t10 = min(remaining * 0.5, 50.0)
        print(f"\n=== Phase 10: Final Vegas (budget={t10:.0f}s) ===")
        a_out, _ = vegas_optimize(best_seq, t10, verbose=False)
        check_and_update(a_out, f"vegas_final_n{len(best_seq)}")

    # Fresh start exploration if time remains
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Fresh start exploration (remaining={remaining:.0f}s) ===")
        for attempt in range(3):
            if remaining < 25:
                break
            n_fresh = n_base + int(rng.integers(-200, 400))
            if n_fresh < 300:
                n_fresh = 800
            a_fresh = structured_init(n_fresh, rng)
            t_fresh = min(remaining * 0.3, 20.0)
            a_out, _ = lse_adam_optimize(a_fresh, t_fresh, verbose=False)
            check_and_update(a_out, f"fresh_{attempt}_n{n_fresh}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 15:
                a_out, _ = pair_cd_optimize(a_out, min(remaining * 0.4, 15.0), rng, verbose=False)
                check_and_update(a_out, f"fresh_pcd_{attempt}_n{n_fresh}")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
