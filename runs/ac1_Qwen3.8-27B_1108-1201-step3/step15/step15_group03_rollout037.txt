def lse_gradient_descent(a0, time_limit, verbose=True):
    """LSE-smoothed gradient descent with Polyak-inspired step size."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Lower bound on max_conv: average value = norm^2 / size = 2n/(2n-1)
    lower_bound = 2.0 * n / (2 * n - 1)

    n_stages = 15
    temps = np.geomspace(100.0, 0.1, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < lower_bound * 1.001:

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -80.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break

            # Polyak-inspired step: (f - f_lower) / ||g||^2 * safety
            polyak_step = (max_c - lower_bound) / (grad_norm * grad_norm + 1e-10)
            base_step = norm / (T_val * np.sqrt(n) * 2)
            step = min(polyak_step, base_step) * (1.0 / (1.0 + 0.02 * it))

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best={best_score:.8f}")

    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best-response with constraint generation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1

    # Initialize active constraint set
    conv_init = fast_conv_self(y)
    active_indices = list(np.argsort(conv_init)[::-1][:min(200, size)])

    warm_b = None
    lp_fail_count = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-10:
            break

        # Adaptive constraint screening
        thresh = M * 0.85
        candidate_active = np.where(conv_y >= thresh)[0]
        if len(candidate_active) > 0:
            active_indices = list(candidate_active)
        elif len(candidate_active) < 20:
            thresh2 = M * 0.5
            candidate_active = np.where(conv_y >= thresh2)[0]
            if len(candidate_active) > 0:
                active_indices = list(candidate_active)

        # Cap active set size
        if len(active_indices) > 400:
            conv_y_arr = conv_y
            active_indices = list(np.argsort(conv_y_arr)[::-1][:400])

        b, active_indices = lp_best_response_warm(y, M, solver, active_indices, warm_b)

        if b is None:
            lp_fail_count += 1
            if lp_fail_count > 20:
                # Reset to LP with all top constraints
                active_indices = list(np.argsort(conv_y)[::-1][:min(300, size)])
                b, active_indices = lp_best_response_warm(y, M, solver, active_indices, None)
                if b is None:
                    continue
                lp_fail_count = 0
            continue

        lp_fail_count = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        warm_b = b

        # Nesterov update with adaptive momentum
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nest t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M:.6f}")

    return best_a, best_score


def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent with FFT-accelerated gain computation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv_self(a).max()
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        # Pre-compute peak lags
        peak_threshold = cur_max * 0.93
        peak_lags = np.where(conv >= peak_threshold)[0]

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-8:
                continue

            # Candidate j's: random + structured
            j_candidates = set()
            for _ in range(8):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:5]:
                j1 = int(kl - i)
                j2 = int(kl + i)
                if 0 <= j1 < n:
                    j_candidates.add(j1)
                if 0 <= j2 < n:
                    j_candidates.add(j2)
            j_candidates.discard(i)
            j_list = list(j_candidates)[:20]

            # Compute gain vector for transferring from i to j
            # delta > 0: a[i] -= delta, a[j] += delta
            # change in conv = delta * (gain_j - gain_i) + quadratic terms

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in j_list:
                if j == i:
                    continue
                aj = a[j]

                # Gain vector: how conv changes when we add delta to a[j] and remove from a[i]
                # conv_new[k] = conv[k] + delta*(cross[j] - cross[i]) + delta^2*(a[i]-a[j]) adjustments
                # where cross[j] is the contribution of adding a[j] to conv

                # Compute gain = conv(a + e_j * d, a + e_j * d) - conv(a - e_i * d, a - e_i * d) - conv(a,a)
                # = d * (crosscorr(a, e_j) + crosscorr(a, e_j)) - d * (crosscorr(a, e_i) + crosscorr(a, e_i))
                #   + d^2 * (a[j] - a[i]) at specific positions
                # More precisely:
                # Adding d to position j: conv_new[k] += d * (a[k-j] + a[j-k]) for valid k
                # Removing d from position i: conv_new[k] -= d * (a[k-i] + a[i-k]) for valid k
                # Plus quadratic terms at k=2j: +d^2, at k=2i: -d^2, cross terms at k=i+j and k=j+i

                # Use FFT for the linear part
                # Actually for speed, let's just sample deltas and compute directly

                n_samples = 15
                max_d = min(ai * 0.95, norm * 0.05)
                if max_d < 1e-10:
                    continue
                deltas = np.linspace(0, max_d, n_samples)

                # Precompute the linear contribution vectors
                # gain_j[k] = a[k-j] (if valid) + a[j-k] (if valid)
                # gain_i[k] = a[k-i] (if valid) + a[i-k] (if valid)
                gain = np.zeros(size)
                # From j (adding)
                for k in range(max(0, j), min(size, j + n)):
                    gain[k] += a[k - j]
                for k in range(max(0, j - n + 1), min(size, j + 1)):
                    gain[k] += a[j - k]
                # From i (removing)
                for k in range(max(0, i), min(size, i + n)):
                    gain[k] -= a[k - i]
                for k in range(max(0, i - n + 1), min(size, i + 1)):
                    gain[k] -= a[i - k]

                for d in deltas:
                    if d < 1e-12:
                        continue
                    new_conv = conv + d * gain
                    # Quadratic corrections
                    if 2 * i < size:
                        new_conv[2 * i] -= 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] += 2 * aj * d + d * d
                    # Cross terms at i+j
                    if i + j < size and i != j:
                        new_conv[i + j] += 2 * d * (aj - ai)
                    v = new_conv.max()
                    if v < best_val - 1e-12:
                        best_val = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 1e-12:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-10:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
                    total_improvements += 1
                # Update conv for next iterations
                conv = fast_conv_self(a)
                cur_max = conv.max()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break
            if no_improve >= 2:
                n_perturb = max(5, n // 50)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.12 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and total_improvements > 0 and total_improvements % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD imp={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_equalization(a0, time_limit, rng, verbose=True):
    """Target top peaks by redistributing mass."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv_self(a).max()
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)

        # Identify peak lags
        peak_threshold = cur_max * 0.90
        peak_lags = np.where(conv >= peak_threshold)[0]

        # Compute pressure: how much each coordinate contributes to peaks
        pressure = np.zeros(n)
        for kl in peak_lags[:min(30, len(peak_lags))]:
            for i in range(n):
                idx1 = kl - i
                idx2 = i - kl
                if 0 <= idx1 < n:
                    pressure[i] += a[idx1]
                if 0 <= idx2 < n:
                    pressure[i] += a[idx2]

        pass_improved = False
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-8:
                continue

            # Transfer to low-pressure targets
            j_candidates = set()
            low_pressure = np.argsort(pressure)[:min(10, n)]
            for j in low_pressure:
                if j != i:
                    j_candidates.add(int(j))
            for _ in range(5):
                j_candidates.add(int(rng.integers(n)))
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in list(j_candidates)[:12]:
                if j == i:
                    continue
                aj = a[j]
                max_d = min(ai * 0.9, norm * 0.04)
                if max_d < 1e-10:
                    continue

                # Compute gain vector
                gain = np.zeros(size)
                for k in range(max(0, j), min(size, j + n)):
                    gain[k] += a[k - j]
                for k in range(max(0, j - n + 1), min(size, j + 1)):
                    gain[k] += a[j - k]
                for k in range(max(0, i), min(size, i + n)):
                    gain[k] -= a[k - i]
                for k in range(max(0, i - n + 1), min(size, i + 1)):
                    gain[k] -= a[i - k]

                n_samples = 12
                deltas = np.linspace(0, max_d, n_samples)
                for d in deltas:
                    if d < 1e-12:
                        continue
                    new_conv = conv + d * gain
                    if 2 * i < size:
                        new_conv[2 * i] -= 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] += 2 * aj * d + d * d
                    if i + j < size and i != j:
                        new_conv[i + j] += 2 * d * (aj - ai)
                    v = new_conv.max()
                    if v < best_val - 1e-12:
                        best_val = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 1e-12:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-10:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break
            if no_improve == 2:
                n_perturb = max(5, n // 40)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.15 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and (idx + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakEq ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.25) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.75) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2.0
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.8
    elif choice == 6:
        a = np.exp(-3.0 * x) + 0.4 + 0.2 * np.sin(4 * np.pi * x)
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(6 * np.pi * x) * np.cos(3 * np.pi * x)
    elif choice == 8:
        a = np.sinc(4.0 * (x - 0.5)) ** 2 + 0.1
    elif choice == 9:
        # U-shaped: high at ends, low in middle
        a = 0.3 + 0.7 * (x ** 2 + (1 - x) ** 2) / (x ** 2 + (1 - x) ** 2 + 0.1)
    elif choice == 10:
        # W-shaped
        a = 0.2 + 0.8 * (1 - np.cos(4 * np.pi * x)) / 2
    else:
        a = np.exp(-3.0 * (x - 0.5) ** 2 / 0.1)
    a = np.maximum(a, 0.0)
    a = a + 0.01 * rng.random(n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        if len(a_list) == 0:
            return float('inf')
        for val in a_list:
            if val < 0 or not np.isfinite(val):
                return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function with multi-phase optimization."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}, solver={solver}")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LSE gradient descent (fast warmup)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 1: LSE GD n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_out, _ = lse_gradient_descent(best_seq.copy(), t1, verbose=True)
        try_update(a_out, "lse1")

    # Phase 2: LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq)
    t2 = min(remaining * 0.35, 500.0)
    print(f"\n=== Phase 2: LP Nesterov n={n2} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq.copy(), t2, solver, verbose=True)
        try_update(a_out, "lp1")

    # Phase 3: Peak equalization
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 3: Peak Equalization (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = peak_equalization(best_seq.copy(), t3, rng, verbose=True)
        try_update(a_out, "peak_eq")

    # Phase 4: LP Nesterov round 2 (benefits from Phase 3 improvements)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 4: LP Nesterov R2 (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq.copy(), t4, solver, verbose=True)
        try_update(a_out, "lp2")

    # Phase 5: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 5: Pair CD (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = pair_cd_optimize(best_seq.copy(), t5, rng, verbose=True)
        try_update(a_out, "pcd1")

    # Phase 6: LSE polish
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.08, 80.0)
    print(f"\n=== Phase 6: LSE Polish (budget={t6:.0f}s) ===")
    if t6 > 15:
        a_out, _ = lse_gradient_descent(best_seq.copy(), t6, verbose=False)
        try_update(a_out, "lse2")

    # Phase 7: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_base_now = len(best_seq)
    print(f"\n=== Phase 7: Multi-length exploration (budget={remaining:.0f}s) ===")
    if remaining > 60:
        lengths_to_try = []
        for dn in [-200, -100, 100, 200, 400]:
            nl = n_base_now + dn
            if 600 <= nl <= 4000 and nl != n_base_now:
                lengths_to_try.append(nl)

        for nl in lengths_to_try[:3]:  # Try at most 3 new lengths
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 40:
                break
            t_long = min(remaining * 0.4, 100.0)
            if t_long < 20:
                continue
            print(f"  Trying n={nl} (budget={t_long:.0f}s)")
            init_long = interpolate_seq(best_seq, nl)
            # Quick LP optimization
            a_out, _ = nesterov_lp_optimize(init_long, t_long * 0.6, solver, verbose=False)
            try_update(a_out, f"lp_n{nl}")
            # Quick CD polish
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20:
                a_out2, _ = pair_cd_optimize(a_out if a_out is not None else init_long,
                                             min(remaining * 0.3, 30.0), rng, verbose=False)
                try_update(a_out2, f"pcd_n{nl}")

    # Phase 8: Final intensive polish on best sequence
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 8: Final polish n={n_final} (budget={remaining:.0f}s) ===")
    if remaining > 30:
        # Quick LP
        t_lp = min(remaining * 0.4, 60.0)
        if t_lp > 15:
            a_out, _ = nesterov_lp_optimize(best_seq.copy(), t_lp, solver, verbose=False)
            try_update(a_out, "lp_final")
        # Quick CD
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_out, _ = pair_cd_optimize(best_seq.copy(), min(remaining * 0.5, 40.0), rng, verbose=False)
            try_update(a_out, "pcd_final")
        # Final LSE
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            a_out, _ = lse_gradient_descent(best_seq.copy(), min(remaining * 0.5, 20.0), verbose=False)
            try_update(a_out, "lse_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
