    # These are the SAME term! So d(conv_k)/da_i = a_{k-i} + a_{k-i} = 2*a_{k-i} when 0<=k-i<n
    # But wait, when k-i = i (i.e., k=2i), the term is a_i * a_i = a_i^2, and the derivative is 2*a_i.
    # Which is consistent with 2*a_{k-i} = 2*a_i.
    # So d(conv_k)/da_i = 2*a_{k-i} if 0 <= k-i < n, else 0.
    # But this double-counts when k-i = i. Let me think again...
    # Actually no. conv_k = sum_j a_j * a_{k-j}. When we differentiate w.r.t. a_i:
    # - If j = i: the term is a_i * a_{k-i}, derivative is a_{k-i}
    # - If k-j = i, i.e., j = k-i: the term is a_{k-i} * a_i, derivative is a_{k-i}
    # - If j = i AND k-j = i (i.e., k = 2i): these are the same term a_i^2, derivative is 2*a_i
    # So in general: d(conv_k)/da_i = a_{k-i} * [0<=k-i<n] + a_{k-i} * [0<=k-i<n]
    # But when k=2i, both conditions are the same, giving 2*a_i. ✓
    # When k≠2i, the two terms are different (j=i and j=k-i are different), so we get:
    # d(conv_k)/da_i = a_{k-i} (from j=i) + a_{k-i} (from j=k-i) = 2*a_{k-i}
    # Wait, that's the same formula. Let me verify with a small example.
    # a = [a0, a1, a2], n=3
    # conv_1 = a0*a1 + a1*a0 = 2*a0*a1
    # d(conv_1)/da_0 = 2*a1. Formula: 2*a_{1-0} = 2*a1. ✓
    # conv_2 = a0*a2 + a1*a1 + a2*a0 = 2*a0*a2 + a1^2
    # d(conv_2)/da_1 = 2*a1. Formula: 2*a_{2-1} = 2*a1. ✓
    # conv_2: d(conv_2)/da_0 = 2*a2. Formula: 2*a_{2-0} = 2*a2. ✓
    
    # So gradient of smooth obj = sum_k w_k * 2*a_{k-i} (for valid k-i)
    # = 2 * sum_{k: 0<=k-i<2n-1} w_k * a_{k-i} (with a_out_of_range = 0)
    # Let m = k-i, then k = m+i, m ranges from -i to 2n-2-i, and we need 0<=m<n
    # So m ranges from max(0, -i) to min(n-1, 2n-2-i)
    # grad_i = 2 * sum_{m=max(0,-i)}^{min(n-1, 2n-2-i)} w_{m+i} * a_m
    # But w is defined on indices 0 to 2n-2, so we need 0 <= m+i <= 2n-2
    # i.e., -i <= m <= 2n-2-i
    # Combined with 0 <= m < n: m ranges from 0 to min(n-1, 2n-2-i)
    # And we need m+i >= 0, which is always true for m >= 0, i >= 0.
    # And m+i <= 2n-2, i.e., m <= 2n-2-i. For i < n, 2n-2-i > n-2 >= n-1 (for n>=2), so the upper bound is n-1.
    # So grad_i = 2 * sum_{m=0}^{n-1} w_{m+i} * a_m (where w_j = 0 for j >= 2n-1)
    # This is a cross-correlation! grad_i = 2 * cross_corr(w, a)[i]
    
    # Actually, this is just the convolution of w (padded) with a, evaluated at specific positions.
    # Let's use FFT for efficiency.
    
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # We need sum_m w_{m+i} * a_m for each i
    # This is like a "reversed cross-correlation"
    # Let's define v = w (length 2n-1), and we want (v * a)[i + n - 1] (convolution evaluated at i+n-1)
    # Actually: sum_m w_{m+i} * a_m = (reverse(w) * a)[len(w)-1-i] ... this is getting complicated
    # Let me just use the direct formula with FFT
    # sum_m w_{m+i} * a_m = cross-correlation of w and a at lag i
    # = (w * reverse(a))[i + n - 1] (convolution of w with reversed a)
    # Hmm, let me think differently.
    # Define b = a (length n), c = w (length 2n-1)
    # We want for each i in [0, n-1]: sum_{m=0}^{n-1} c[m+i] * b[m]
    # This is the cross-correlation (c ⊛ b)[i] = sum_m c[m+i] * b[m]
    # In terms of convolution: c ⊛ b = reverse(c) * b (linear convolution)
    # Let d = reverse(c), then (d * b)[k] = sum_m d[m] * b[k-m] = sum_m c[2n-2-m] * b[k-m]
    # We want sum_m c[m+i] * b[m]. Let j = m+i-m = i... 
    # OK this is getting too complicated analytically. Let me just use the FFT approach directly.
    
    # Alternative: use the fact that grad_i = 2 * sum_k w_k * a_{k-i} (for valid k-i in [0,n-1])
    # Let's define the "shifted" version: for each i, we sum w_k * a_{k-i} over valid k.
    # This is the same as the cross-correlation of w and a.
    # Using FFT: cross_corr = ifft(conj(FFT(a)) * FFT(w)) ... but the lengths are different.
    
    # Simplest approach: just compute it as a convolution
    # grad_i = 2 * sum_{k=i}^{i+n-1} w_k * a_{k-i} (where w_k = 0 outside [0, 2n-2])
    # = 2 * (w * a)[i] (where * denotes cross-correlation, or equivalently convolution of w with reversed a)
    
    # Let me just do it with a single FFT computation
    # sum_k w_k * a_{k-i} for i = 0, ..., n-1
    # = cross-correlation of w and a at lags 0, 1, ..., n-1
    # Using FFT: ifft(FFT(w) * conj(FFT(a))) gives cross-correlation
    # But w has length 2n-1 and a has length n, so we pad a to length 2n-1
    
    a_padded = np.zeros(2*n-1)
    a_padded[:n] = a
    W_full = rfft(w, fft_size)
    A_full = rfft(a_padded, fft_size)
    cross = irfft(W_full * np.conj(A_full), fft_size)
    # cross[i] = sum_k w[k] * a_padded[k-i] (with appropriate wrapping for circular conv)
    # For linear cross-correlation, we need the non-wrapping part
    # cross[i] = sum_{k=max(0,i)}^{2n-2} w[k] * a_padded[k-i] for i >= 0
    # = sum_{k=i}^{2n-2} w[k] * a[k-i] (since a_padded[k-i] = a[k-i] for 0<=k-i<n)
    # = sum_{m=0}^{2n-2-i} w[m+i] * a[m] (substituting m=k-i)
    # For m >= n, a[m] = 0, so: sum_{m=0}^{n-1} w[m+i] * a[m] ✓
    
    grad = 2.0 * cross[:n]
    
    return f_val, grad

def smooth_optimize_v2(a0, time_limit, verbose=True):
    """Minimize smooth approximation with adaptive step size."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    t0 = time.monotonic()
    
    # Temperature schedule
    temps = [10.0, 5.0, 2.5, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05, 0.03, 0.02, 0.012, 0.008, 0.005, 0.003, 0.002, 0.001, 0.0007, 0.0005]
    
    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        n_steps = max(50, min(2000, int(remaining * 5)))
        
        y = a.copy()  # Nesterov momentum variable
        momentum = 0.0
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            f_val, grad = smooth_objective_and_grad(y, T)
            
            # Backtracking line search
            step = norm / (n * T + 1e-10)
            f_curr = f_val
            a_trial = y - step * grad
            a_trial = project_nonneg_sum(a_trial, norm)
            
            for ls in range(20):
                f_trial, _ = smooth_objective_and_grad(a_trial, T)
                if f_trial < f_curr - 1e-4 * step * np.sum(grad * (y - a_trial)):
                    break
                step *= 0.5
                a_trial = y - step * grad
                a_trial = project_nonneg_sum(a_trial, norm)
            
            # Nesterov update
            momentum = (1 + np.sqrt(1 + 4 * momentum**2)) / 2
            y_new = (1 - 1/momentum) * y + (1/momentum) * a_trial
            y_new = project_nonneg_sum(y_new, norm)
            
            if it == 0:
                a = a_trial.copy()
            else:
                a = y_new + (momentum - 1) / momentum * (y_new - y)
                a = project_nonneg_sum(a, norm)
            
            y = y_new
            
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
            
            score_y = fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()
            
            if verbose and it % 100 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Smooth T={T:.4f} step={it} ({elapsed:.0f}s): obj={f_val:.8f}, best={best_score:.8f}")
    
    return best_a, best_score

def get_solver():
    for s in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    print("  LP solver (fallback): ECOS")
    return cp.ECOS

def lp_best_response(a, M, solver, active=None):
    """Solve max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    # T is size x n, T[k, i] = a[k-i] if 0 <= k-i < n
    # For efficiency, use sparse representation
    # T[k, i] = a[k-i] for max(0, k-n+1) <= i <= min(n-1, k)
    
    # Build constraint matrix as sparse
    from scipy.sparse import lil_matrix
    T = lil_matrix((size, n))
    for i in range(n):
        T[i:i+n, i] = a
    
    if active is not None:
        T_use = T.tocsr()[active, :]
    else:
        T_use = T.tocsr()
    
    b_var = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T_use @ b_var <= M])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def nesterov_lp_v2(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration and adaptive constraint selection."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stuck_count = 0
    momentum = 0.0
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break
        
        # Adaptive constraint selection: use more constraints as we converge
        # Early on: use only top constraints for speed
        # Later: use more for accuracy
        if n_iters < 100:
            threshold_frac = 0.8
        elif n_iters < 500:
            threshold_frac = 0.7
        else:
            threshold_frac = 0.5
        
        active = np.where(conv_y >= threshold_frac * M_curr)[0]
        if len(active) < 10:
            active = np.where(conv_y >= 0.3 * M_curr)[0]
        if len(active) < 5:
            active = np.arange(len(conv_y))
        
        b = lp_best_response(y, M_curr, solver, active=active)
        if b is None:
            b = lp_best_response(y, M_curr, solver, active=None)
            if b is None:
                stuck_count += 1
                if stuck_count > 50:
                    break
                continue
        
        stuck_count = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        # Nesterov update
        momentum = (1 + np.sqrt(1 + 4 * momentum**2)) / 2
        alpha = 1.0 / momentum
        y_new = (1 - alpha) * y + alpha * g
        y_new = project_nonneg_sum(y_new, norm)
        
        if t == 1:
            a_new = g.copy()
        else:
            a_new = y_new + (momentum - 1) / momentum * (y_new - y)
            a_new = project_nonneg_sum(a_new, norm)
        
        y = y_new
        a = a_new
        
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        # Periodic average check
        if n_iters % 50 == 0:
            # Simple moving average (exponential)
            pass
        
        if verbose and n_iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def block_coordinate_descent(a0, time_limit, rng, block_size=10, verbose=True):
    """Block coordinate descent: update blocks of adjacent elements."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    
    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if no_improve > 10:
            break
        
        n_blocks = n // block_size
        perm = rng.permutation(n_blocks)
        
        improved = False
        for idx in perm:
            if time.monotonic() - t0 > time_limit:
                break
            
            start = idx * block_size
            end = min(start + block_size, n)
            block = a[start:end].copy()
            block_sum = block.sum()
            if block_sum < 1e-15:
                continue
            
            # Try to optimize the block: find the best distribution within this block
            # that minimizes the peak of the convolution
            # Simple approach: try a few random redistributions within the block
            best_block = block.copy()
            best_block_score = conv.max()
            
            for trial in range(5):
                # Random redistribution preserving sum
                weights = rng.dirichlet(np.ones(end - start))
                new_block = weights * block_sum
                a_trial = a.copy()
                a_trial[start:end] = new_block
                conv_trial = fast_autocorr(a_trial)
                if conv_trial.max() < best_block_score - 1e-12:
                    best_block = new_block.copy()
                    best_block_score = conv_trial.max()
            
            if best_block_score < conv.max() - 1e-12:
                a[start:end] = best_block
                conv = fast_autocorr(a)
                if conv.max() < best_score - 1e-12:
                    best_score = conv.max()
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    Block CD ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def spectral_projection(a, strength=0.3):
    """Apply a spectral flattening projection to reduce peak of convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    A = rfft(a, fft_size)
    B = A * A  # This is FFT of conv(a,a)
    
    # Flatten B: reduce peaks
    B_mag = np.abs(B)
    B_angle = np.angle(B)
    
    # Blend with uniform magnitude
    target_mag = B_mag.mean()
    B_mag_new = (1 - strength) * B_mag + strength * target_mag
    B_new = B_mag_new * np.exp(1j * B_angle)
    
    # Recover A: A_new = sqrt(B_new)
    # But we need A to be conjugate symmetric (for real a)
    A_new = np.sqrt(np.abs(B_new)) * np.exp(1j * B_angle / 2)
    
    # Enforce conjugate symmetry
    A_new = 0.5 * (A_new + np.conj(A_new[np.fft.fftshift(np.arange(fft_size)[::-1]) % fft_size]))
    
    # Actually, for real sequences, A[k] = conj(A[fft_size-k])
    # Let me just take the real part of the IFFT
    a_new = np.real(irfft(A_new, fft_size)[:n])
    
    return np.maximum(a_new, 0.0)

def chirp_init(n, k=1, phase=0):
    """Generate a chirp-like initialization."""
    x = np.arange(n)
    a = np.cos(2 * np.pi * k * x**2 / (4 * n) + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def golay_like_init(n):
    """Generate a Golay-like initialization (product of smaller Golay sequences)."""
    # Start with the base Golay sequence [1, 1, 1, -1, 1, -1, -1, -1]
    # Take absolute values and upsample
    base = np.array([1, 1, 1, 1, 1, 1, 1, 1], dtype=float)  # abs of Golay
    # For a more interesting pattern, use a different base
    # Actually, let's use a random binary sequence
    rng = np.random.default_rng(12345)
    a = rng.integers(0, 2, n).astype(float)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
