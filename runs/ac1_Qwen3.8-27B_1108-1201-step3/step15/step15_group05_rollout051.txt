```python
def project_simplex(v, S):
    # Project v onto {x >= 0, sum(x) = S}
    n = len(v)
    # Binary search for threshold
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cv - S))[0][-1]
    theta = (cv[rho] - S) / (rho + 1)
    return np.maximum(v - theta, 0)
```

def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cv - S))
    if len(rho[0]) == 0:
        return np.full(n, S / n)
    rho = rho[0][-1]
    theta = (cv[rho] - S) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def compute_smooth_gradient(a, weights):
    """
    Compute gradient of sum_k weights[k] * (a*a)[k] w.r.t. a.
    Uses FFT for O(n log n) computation.
    grad[j] = 2 * sum_m a[m] * weights[j+m] (for valid j+m < 2n-1)
    """
    n = len(a)
    size = 2 * n - 1
    L = 1 << (n + size - 1).bit_length()
    
    a_pad = np.zeros(L)
    a_pad[:n] = a
    w_pad = np.zeros(L)
    w_pad[:size] = weights
    
    A = rfft(a_pad)
    W = rfft(w_pad)
    
    corr = irfft(np.conj(A) * W, L)
    return 2.0 * corr[:n]


def logsumexp_fista(a0, t_schedule, time_limit, verbose=False):
    """
    FISTA with log-sum-exp smoothed objective.
    Minimizes max_k (a*a)[k] subject to sum(a)=1, a>=0.
    
    t_schedule: list of temperatures to use (decreasing)
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s  # normalize to sum=1
    
    best_a = a.copy()
    conv = fast_autocorr(a)
    best_score = conv.max()
    
    t0 = time.monotonic()
    
    for temp_idx, temp in enumerate(t_schedule):
        t_phase = time.monotonic()
        phase_time_limit = time_limit - (time.monotonic() - t0)
        if phase_time_limit < 5:
            break
        
        # FISTA state
        x = a.copy()
        y = a.copy()
        x_prev = a.copy()
        tk = 1.0
        lr = None  # will be set adaptively
        
        for it in range(1, 1000000):
            if time.monotonic() - t0 > time_limit:
                break
            if time.monotonic() - t_phase > phase_time_limit * 0.95:
                break
            
            # Compute weights
            conv_y = fast_autocorr(y)
            max_conv = conv_y.max()
            if max_conv < 1e-15:
                break
            
            # Log-sum-exp weights
            shifted = (conv_y - max_conv) / temp
            # Clip for numerical stability
            shifted = np.clip(shifted, -100, 100)
            exp_shifted = np.exp(shifted)
            weights = exp_shifted / exp_shifted.sum()
            
            # Compute gradient
            grad = compute_smooth_gradient(y, weights)
            
            # Lipschitz estimate (rough)
            if lr is None:
                # Estimate L = ||grad||_inf / ||y||_inf as a rough proxy
                # Better: use the formula L <= 4 * max(weights) * n
                lr = 1.0 / (4.0 * max(weights.max(), 1e-10) * n)
            
            # Gradient step
            x_new = y - lr * grad
            x_new = project_simplex(x_new, 1.0)
            
            # Check if step is valid (decrease in smooth objective)
            # For now, just accept and use backtracking if needed
            tk_new = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
            y_new = x_new + ((tk - 1.0) / tk_new) * (x_new - x_prev)
            y_new = project_simplex(y_new, 1.0)
            
            x_prev = x
            x = x_new
            y = y_new
            tk = tk_new
            
            # Track best
            conv_x = fast_autocorr(x)
            score_x = conv_x.max()
            if score_x < best_score:
                best_score = score_x
                best_a = x.copy()
            
            # Adaptive learning rate: if oscillating, reduce lr
            if it % 100 == 0:
                # Check for convergence
                if np.abs(score_x - conv_y.max()) < 1e-10:
                    break
        
        # Update a for next temperature phase
        a = best_a.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    FISTA temp={temp:.4f} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score


def build_constraint_matrix(a, k_indices, n):
    """Build constraint matrix for LP."""
    size = len(k_indices)
    mat = np.zeros((size, n))
    for idx, k in enumerate(k_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            mat[idx, j] = a[k - j]
    return mat

    n = len(a_fixed)
    k_indices = np.asarray(active_set)
    if len(k_indices) == 0:
        return np.ones(n) / n, None
    T = build_constraint_matrix(a_fixed, k_indices, n)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    return b_val, None


def nesterov_lp(a0, time_limit, solver, verbose=False, seed=0):
    """Nesterov-accelerated LP best-response."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s  # normalize to sum=1
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    stall_threshold = 50
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break
        
        # Adaptive active set
        if active_set is None or t % 30 == 0:
            tol = max(0.005, 0.03 * np.exp(-t / 200.0))
            active_set = np.where(conv_y >= M_curr * (1.0 - tol))[0]
            top_k = np.argsort(conv_y)[-max(5, n // 100):]
            active_set = np.unique(np.concatenate([active_set, top_k]))
        
        b, _ = lp_best_response(y, M_curr, active_set, solver)
        
        if b is None:
            if t % 10 == 0:
                active_set = np.arange(2 * n - 1)
                b, _ = lp_best_response(y, M_curr, active_set, solver)
            if b is None:
                continue
        
        # Verify and expand if needed
        if t % 20 == 0:
            all_k = np.arange(2 * n - 1)
            complement = np.setdiff1d(all_k, active_set, assume_unique=True)
            if len(complement) > 0:
                T_comp = build_constraint_matrix(y, complement, n)
                violations = T_comp @ b
                viol_idx = np.where(violations > M_curr * (1 + 1e-5))[0]
                if len(viol_idx) > 0:
                    new_active = np.union1d(active_set, complement[viol_idx])
                    if len(new_active) > 0.85 * (2 * n - 1):
                        new_active = all_k
                    b, _ = lp_best_response(y, M_curr, new_active, solver)
                    if b is None:
                        continue
                    active_set = new_active
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b / b_sum  # normalize to sum=1
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            momentum = min(momentum, 0.97)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a
        
        y_prev = y
        y = y_new
        a = a_new
        
        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        
        # Restart if stalled
        if stall_count > stall_threshold:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall_count = 0
            
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    LP restart at iter {t} ({elapsed:.0f}s), score={best_score:.10f}")
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={t} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score


def subgradient_mirror(a0, time_limit, verbose=False):
    """
    Subgradient method with mirror descent (entropy regularization).
    Uses exponential weighting for the non-negativity constraint.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-10)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Entropy mirror: use log-domain updates
    # Update rule: a_new[j] = a[j] * exp(-eta * grad[j]) / sum(a * exp(-eta * grad))
    eta = 1.0 / n ** 0.5
    patience = 0
    max_patience = 300
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break
        
        conv = fast_autocorr(a)
        M_curr = conv.max()
        k_star = int(np.argmax(conv))
        
        # Subgradient at k_star: grad[j] = 2*a[k_star - j] if valid
        grad = np.zeros(n)
        for j in range(n):
            k = k_star - j
            if 0 <= k < 2 * n - 1:
                # grad[j] = 2 * a[k_star - j] if 0 <= k_star - j < n
                pass
        # Using the formula: grad[j] = 2*a[k_star - j] if 0 <= k_star - j < n
        lo = max(0, k_star - n + 1)
        hi = min(n, k_star + 1)
        for j in range(lo, hi):
            grad[j] = 2.0 * a[k_star - j]
        
        # Also try a few other peaks
        top_k = min(5, n)
        peaks = np.argsort(conv)[-top_k:]
        for k_p in peaks:
            grad_p = np.zeros(n)
            lo_p = max(0, k_p - n + 1)
            hi_p = min(n, k_p + 1)
            for j in range(lo_p, hi_p):
                grad_p[j] = 2.0 * a[k_p - j]
            # Weight by proximity to max
            w_p = np.exp(-10.0 * (M_curr - conv[k_p]) / max(M_curr * 0.05, 1e-10))
            grad += w_p * grad_p
        grad /= (1.0 + (top_k - 1))  # rough normalization
        
        # Mirror descent update (exponential)
        # a_new[j] = a[j] * exp(-eta * grad[j])
        log_update = -eta * grad
        log_update -= log_update.max()  # for numerical stability
        update = np.exp(log_update)
        a_new = a * update
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        score_new = fast_autocorr(a_new).max()
        if score_new < best_score - 1e-12:
            a = a_new
            best_score = score_new
            best_a = a_new.copy()
            patience = 0
            eta = min(eta * 1.01, 5.0)
        else:
            patience += 1
            eta *= 0.97
            if eta < 1e-8:
                break
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Mirror t={t} ({elapsed:.0f}s): best={best_score:.10f}, eta={eta:.6f}")
    
    return best_a, best_score


def perturb_asymmetric(a, rng, scale=0.02):
    """Asymmetric perturbation: more noise on one side."""
    n = len(a)
    # Create a linearly varying noise scale
    scale_vec = 1.0 + scale * rng.standard_normal(n)
    # Make it asymmetric: more perturbation on the left side
    asym = 1.0 + 0.5 * scale * np.linspace(-1, 1, n)
    scale_vec = np.maximum(scale_vec * asym, 0.1)
    a_pert = a * scale_vec
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s
