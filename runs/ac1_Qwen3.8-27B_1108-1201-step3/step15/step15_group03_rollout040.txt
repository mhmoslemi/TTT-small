"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-structure initialization
2. Adaptive LSE smooth gradient descent with restarts
3. Peak-targeted local optimization (block-wise)
4. Stochastic perturbation for escaping local optima
5. Progressive refinement at increasing resolution

import numpy as np
import time

def _fast_conv_self(a):
    """Compute self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def _fast_grad_lse(a, T):
    """Compute gradient of LSE smooth approximation of max(autoconv)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    conv = _fast_conv_self(a)
    max_c = conv.max()
    shifted = (conv - max_c) / T
    shifted = np.maximum(shifted, -50.0)
    w = np.exp(shifted)
    w = w / w.sum()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    # Gradient of sum_k w[k] * conv[k] w.r.t. a
    # conv[k] = sum_i a[i]*a[k-i]
    # d/d a[j] sum_k w[k]*conv[k] = sum_k w[k] * (a[k-j] + a[j-k]) (with bounds)
    # = (w * a)[j] + (flip(w) * a)[j] approximately
    # Using FFT: grad[j] = sum_k w[k]*a[k-j] + sum_k w[k]*a[j-k]
    # First term: cross-correlation of w and a
    # Second term: cross-correlation of a and w (same by symmetry of the formula)
    g1 = np.fft.irfft(np.conj(A) * W, fft_size)[:n]
    g2 = np.fft.irfft(A * W, fft_size)[:n]
    return 2.0 * (g1 + g2), conv, max_c

def _normalize(a, target_sum):
    """Normalize a to have target_sum and be non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a) * target_sum
    return a * target_sum / s

def _eval_local(a):
    """Local evaluation (same as evaluate_sequence but faster)."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0 or a.sum() < 0.01:
        return float('inf')
    conv = _fast_conv_self(a)
    max_b = conv.max()
    sum_a = a.sum()
    return float(2.0 * n * max_b / (sum_a ** 2))

def _init_uniform(n):
    return np.ones(n)

def _init_u_shape(n, rng):
    """U-shaped: high at edges, low in middle."""
    x = np.linspace(0, 1, n)
    a = (2 * x - 1) ** 2 + 0.1
    a = np.maximum(a, 0) + 0.01 * rng.random(n)
    s = a.sum()
    return a / s

def _init_bimodal(n, rng):
    """Bimodal: two peaks."""
    x = np.linspace(0, 1, n)
    a = np.exp(-50 * (x - 0.2) ** 2) + np.exp(-50 * (x - 0.8) ** 2) + 0.05
    a = np.maximum(a, 0) + 0.01 * rng.random(n)
    s = a.sum()
    return a / s

def _init_cosine(n, rng):
    """Cosine taper."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x) ** 2 + 0.05
    a = np.maximum(a, 0) + 0.01 * rng.random(n)
    s = a.sum()
    return a / s

def _init_random_smooth(n, rng):
    """Random smooth function via Fourier series."""
    n_harm = rng.integers(3, 10)
    coeffs = rng.random(n_harm) * np.exp(-np.arange(n_harm) * 0.5)
    phases = rng.random(n_harm) * 2 * np.pi
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for k in range(1, n_harm):
        a += coeffs[k] * np.cos(2 * np.pi * k * x + phases[k])
    a = np.maximum(a, 0.01)
    s = a.sum()
    return a / s

def _lse_gd(a0, time_limit, verbose=False):
    """Adaptive LSE smooth gradient descent."""
    n = len(a0)
    target_sum = np.sqrt(2.0 * n)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = _init_uniform(n)
    a = _normalize(a, target_sum)
    
    best_a = a.copy()
    best_max = _fast_conv_self(a).max()
    t0 = time.monotonic()
    
    # Adaptive temperature: start high, decrease
    T_start = 50.0
    T_end = 0.5
    
    # Use multiple restart phases
    n_phases = 5
    for phase in range(n_phases):
        T_cur = T_start * (T_end / T_start) ** (phase / max(1, n_phases - 1))
        phase_time = time_limit / n_phases
        phase_start = time.monotonic()
        
        lr = 1.0 / (n ** 0.5)
        momentum = np.zeros(n)
        beta = 0.85
        
        it = 0
        while time.monotonic() - phase_start < phase_time:
            grad, conv, max_c = _fast_grad_lse(a, T_cur)
            
            # Polyak-style step size
            step = lr / (1.0 + 0.005 * it)
            momentum = beta * momentum - step * grad
            a_new = a + momentum
            a_new = _normalize(a_new, target_sum)
            a = a_new
            it += 1
            
            cur_max = _fast_conv_self(a).max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
            
            if it % 200 == 0 and verbose:
                print(f"    LSE phase {phase} T={T_cur:.1f} it={it}: max={cur_max:.6f} best={best_max:.6f}")
    
    return best_a, best_max

def _peak_targeting(a0, time_limit, rng, verbose=False):
    """Identify top peaks and reduce them by adjusting contributing elements."""
    n = len(a0)
    target_sum = np.sqrt(2.0 * n)
    a = _normalize(a0.copy(), target_sum)
    best_a = a.copy()
    best_max = _fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    
    no_improve = 0
    while time.monotonic() - t0 < time_limit:
        conv = _fast_conv_self(a)
        cur_max = conv.max()
        peak_idx = conv.argmax()
        
        # Find elements contributing most to the peak
        # conv[peak_idx] = sum_i a[i]*a[peak_idx-i]
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        contributions = np.zeros(n)
        for i in range(lo, hi + 1):
            j = peak_idx - i
            if 0 <= j < n:
                contributions[i] = a[i] * a[j]
                if i != j:
                    contributions[j] = a[i] * a[j]
        
        # Reduce the top contributors and add to low contributors
        top_k = min(10, n // 5)
        top_contrib = np.argsort(contributions)[-top_k:]
        low_contrib = np.argsort(contributions)[:min(10, n // 5)]
        
        improved = False
        for i in top_contrib:
            if a[i] < target_sum * 1e-8:
                continue
            # Try reducing a[i] and adding to a random low-contribution element
            for j in low_contrib:
                if j == i:
                    continue
                # Try a small transfer
                delta = a[i] * 0.1
                a[i] -= delta
                a[j] += delta
                a = _normalize(a, target_sum)
                new_max = _fast_conv_self(a).max()
                if new_max < best_max - 1e-12:
                    best_max = new_max
                    best_a = a.copy()
                    improved = True
                else:
                    # Undo
                    a[i] += delta
                    a[j] -= delta
                    a = _normalize(a, target_sum)
        
        if not improved:
            no_improve += 1
            if no_improve >= 5:
                break
            # Perturb
            n_pert = max(10, n // 20)
            idx = rng.choice(n, size=n_pert, replace=False)
            a[idx] *= (1.0 + 0.2 * rng.standard_normal(n_pert))
            a = _normalize(a, target_sum)
        else:
            no_improve = 0
        
        if verbose and no_improve % 500 == 0:
            print(f"    PeakTarget: max={cur_max:.6f} best={best_max:.6f}")
    
    return best_a, best_max

def _block_cd(a0, time_limit, rng, verbose=False):
    """Block coordinate descent: optimize blocks of consecutive elements."""
    n = len(a0)
    target_sum = np.sqrt(2.0 * n)
    a = _normalize(a0.copy(), target_sum)
    best_a = a.copy()
    best_max = _fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    
    block_size = max(3, min(20, n // 100))
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        n_blocks = n // block_size
        
        for _ in range(min(100, n_blocks)):
            if time.monotonic() - t0 > time_limit:
                break
            
            start = int(rng.integers(0, max(1, n - block_size)))
            end = start + block_size
            block = a[start:end].copy()
            
            # Try to optimize this block via random perturbations
            best_block = block.copy()
            best_block_max = _fast_conv_self(a).max()
            
            for trial in range(30):
                perturbed = block * (1.0 + 0.3 * rng.standard_normal(block_size))
                perturbed = np.maximum(perturbed, 0.0)
                a_trial = a.copy()
                a_trial[start:end] = perturbed
                a_trial = _normalize(a_trial, target_sum)
                trial_max = _fast_conv_self(a_trial).max()
                if trial_max < best_block_max:
                    best_block_max = trial_max
                    best_block = perturbed.copy()
            
            if best_block_max < _fast_conv_self(a).max() - 1e-12:
                a[start:end] = best_block
                a = _normalize(a, target_sum)
                cur_max = _fast_conv_self(a).max()
                if cur_max < best_max:
                    best_max = cur_max
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0
    
    return best_a, best_max

def _stochastic_hill_climb(a0, time_limit, rng, verbose=False):
    """Stochastic hill climbing with adaptive perturbation size."""
    n = len(a0)
    target_sum = np.sqrt(2.0 * n)
    a = _normalize(a0.copy(), target_sum)
    best_a = a.copy()
    best_max = _fast_conv_self(a).max()
    t0 = time.monotonic()
    
    pert_size = 0.1
    accept_temp = 1e-4
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        # Perturb a random subset
        n_pert = max(5, min(50, n // 20))
        idx = rng.choice(n, size=n_pert, replace=False)
        a_trial = a.copy()
        a_trial[idx] *= (1.0 + pert_size * rng.standard_normal(n_pert))
        a_trial = _normalize(a_trial, target_sum)
        
        trial_max = _fast_conv_self(a_trial).max()
        cur_max = _fast_conv_self(a).max()
        
        # Accept if better, or with probability based on difference
        if trial_max < cur_max:
            a = a_trial
            it = 0
            if trial_max < best_max:
                best_max = trial_max
                best_a = a.copy()
                if verbose and it % 100 == 0:
                    print(f"    SHC: best={best_max:.6f}")
        else:
            diff = trial_max - cur_max
            if rng.random() < np.exp(-diff / accept_temp):
                a = a_trial
            it += 1
        
        # Adaptive perturbation
        if it % 50 == 0:
            pert_size *= 0.95
            accept_temp *= 0.9
    
    return best_a, best_max

import time

def _fft_len(n):
    size = 2 * n - 1
    return 1 << (size - 1).bit_length()

def _conv_self(a):
    n = len(a)
    fs = _fft_len(n)
    A = np.fft.rfft(a, fs)
    return np.fft.irfft(A * A, fs)[:2*n-1]

def _norm(a, s_target):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.full(len(a), s_target / len(a))
    return a * (s_target / s)

def _obj(a):
    """The objective: 2n*max(conv)/sum^2. Scale invariant."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * _conv_self(a).max() / (s * s)

def _lse_grad(a, T):
    n = len(a)
    fs = _fft_len(n)
    c = _conv_self(a)
    mc = c.max()
    if mc < 1e-15:
        return np.zeros(n), c, mc
    sh = np.clip((c - mc) / T, -60.0, 0.0)
    w = np.exp(sh)
    w /= w.sum()
    W = np.fft.rfft(w, fs)
    A = np.fft.rfft(a, fs)
    g = 2.0 * (np.fft.irfft(np.conj(A) * W, fs)[:n] + np.fft.irfft(A * W, fs)[:n])
    return g, c, mc

    Returns (g, q) vectors such that after transfer d from i to j:
    c_new[k] = c[k] + d*g[k] + d^2*q[k]

    size = 2 * n - 1
    g = np.zeros(size)
    q = np.zeros(size)

    q[2*j] += 1.0

def _lse_gd_phase(a0, time_limit, T_start, T_end, verbose=False):
    """One phase of LSE gradient descent with Adam optimizer."""
    n = len(a0)
    s_target = np.sqrt(2.0 * n)
    a = _norm(a0.copy(), s_target)
    best_a = a.copy()
    best_mc = _conv_self(a).max()
    t0 = time.monotonic()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    lr = 1.0 / (5.0 * np.sqrt(n))
    
    n_steps_target = max(100, int(time_limit * 50))
    
    while t < n_steps_target:
        if time.monotonic() - t0 > time_limit:
            break
        t += 1
        
        # Interpolate temperature
        frac = min(1.0, t / n_steps_target)
        T = T_start * (T_end / T_start) ** frac
        
        g, c, mc = _lse_grad(a, T)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr / (1.0 + 0.001 * t)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a = _norm(a_new, s_target)
        
        cur_mc = _conv_self(a).max()
        if cur_mc < best_mc:
            best_mc = cur_mc
            best_a = a.copy()
        
        if verbose and t % 500 == 0:
            el = time.monotonic() - t0
            print(f"    LSE GD t={t} T={T:.2f} ({el:.0f}s): mc={cur_mc:.6f} best={best_mc:.6f}")
    
    return best_a, best_mc

def _pair_transfer_search(a0, time_limit, rng, verbose=False):
    """Search for improvements via pair transfers targeting peaks."""
    n = len(a0)
    s_target = np.sqrt(2.0 * n)
    a = _norm(a0.copy(), s_target)
    best_a = a.copy()
    best_mc = _conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        c = _conv_self(a)
        cur_mc = c.max()
        peak_threshold = cur_mc * 0.95
        peak_lags = np.where(c >= peak_threshold)[0]
        
        improved = False
        n_trials = min(200, max(20, n // 10))
        
        for _ in range(n_trials):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Random source and target
            i = int(rng.integers(n))
            ai = a[i]
            if ai < s_target * 1e-10:
                continue
            
            # Target: prefer elements that are "far" from peak contributors
            j = int(rng.integers(n))
            if j == i:
                continue
            
            g, q = _pair_gain(c, a, i, j, n)
            d, val = _optimal_d(c, g, q, ai, n_samples=30)
            
            if d > 1e-12 and val < best_mc - 1e-10:
                a[i] -= d
                a[j] += d
                a = _norm(a, s_target)
                new_mc = _conv_self(a).max()
                if new_mc < best_mc:
                    best_mc = new_mc
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            no_improve += 1
            if no_improve >= 8:
                break
            # Perturb to escape
            n_pert = max(10, n // 30)
            idx = rng.choice(n, size=n_pert, replace=False)
            a[idx] *= np.exp(0.15 * rng.standard_normal(n_pert))
            a = _norm(a, s_target)
        else:
            no_improve = 0
        
        if verbose and no_improve % 1000 == 0:
            el = time.monotonic() - t0
            print(f"    PairTransfer ({el:.0f}s): mc={cur_mc:.6f} best={best_mc:.6f} no_improve={no_improve}")
    
    return best_a, best_mc
