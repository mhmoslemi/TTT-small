def interpolate_sequence(seq, new_length):
    # Piecewise linear interpolation of the cumulative distribution
    old_length = len(seq)
    if new_length <= 0:
        return np.zeros((new_length,))
    if new_length == old_length:
        return seq
    # Parametrize new positions in [0,1]
    new_space = np.linspace(0.0,1.0,new_length)
    interp = np.interp(new_space, np.linspace(0.0,1.0,old_length), seq)
    # Capping at minimum valid sum
    s = np.sum(interp)
    if s < 0.01:
        return np.ones(new_length) / len(new_space)
    return interp/s
