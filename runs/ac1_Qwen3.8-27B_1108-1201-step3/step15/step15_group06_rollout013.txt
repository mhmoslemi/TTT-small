def compute_subgradient(a, w_active, fft_size):
    """Compute s_i = 2 * sum_m w[m] * a[m-i] for i = 0..n-1"""
    n = len(a)
    # w_active is a vector of length 2n-1 (mostly zeros)
    # s = 2 * cross_correlation(w_active, a) evaluated at lags 0..n-1
    # Using FFT:
    W = rfft(w_active, fft_size)
    A = rfft(a, fft_size)
    # Cross-correlation: corr[i] = sum_j w[j+i] * a[j]
    # = IFFT(W * conj(A)) with appropriate shift
    # Actually, let me verify:
    # IFFT(FFT(w) * conj(FFT(a)))[n] = sum_k w[k] * a[(k-n) mod N]
    # For n=0: sum_k w[k] * a[k] (circular)
    # For n=1: sum_k w[k] * a[k-1] (circular) = sum_k w[k+1] * a[k]
    # So IFFT(W * conj(A))[i] = sum_k w[k+i] * a[k] (circular, with wrap-around)
    # For linear (non-circular) correlation, we need N >= len(w) + len(a) - 1
    corr = irfft(W * np.conj(A), fft_size)
    # corr[i] = sum_k w[k+i] * a[k] (linear, for i = 0..n-1)
    # (with the convention that out-of-range indices are 0)
    return 2.0 * corr[:n]

Let me verify the formula:
- W = FFT(w, N) where w is zero-padded to length N
- A = FFT(a, N) where a is zero-padded to length N
- corr = IFFT(W * conj(A), N)

For i = 0: corr[0] = sum_k w[k] * a[k] (circular, but since we zero-padded, this equals the linear sum for k where both are non-zero)

For i = 1: corr[1] = sum_k w[k] * a[(k-1) mod N]
For k >= 1: a[k-1] is the actual value
For k = 0: a[-1 mod N] = a[N-1] = 0 (since we zero-padded)

    best_score = evaluate_local(a)
    no_improve = 0

    base_eta = 2.0  # base step size
    tau_start = 3.0  # initial temperature for soft max
    k_start = 10  # initial number of top constraints
    
    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute autocorrelation
        A = rfft(a, fft_size_ac)
        conv = irfft(A * A, fft_size_ac)[:size]
        
        # Anneal parameters
        tau = max(1.0, tau_start * np.exp(-t / 500.0))
        k_eff = min(size, k_start + t // 50)
        
        # Find active set
        if k_eff >= size:
            active = np.arange(size)
            vals = conv
        else:
            top_idx = np.argpartition(conv, -k_eff)[-k_eff:]
            active = top_idx
            vals = conv[active]
        
        # Compute weights (softmax)
        v_max = vals.max()
        weights = np.exp(tau * (vals - v_max))
        weights_sum = weights.sum()
        if weights_sum < 1e-30:
            weights = np.ones(len(active)) / len(active)
        else:
            weights = weights / weights_sum
        
        # Build w vector
        w = np.zeros(size)
        w[active] = weights
        
        # Compute subgradient using FFT
        W = rfft(w, fft_size_sg)
        A_sg = rfft(a, fft_size_sg)
        corr = irfft(W * np.conj(A_sg), fft_size_sg)
        s_grad = 2.0 * corr[:n]
        
        # Mirror descent update
        eta = base_eta / np.sqrt(t + 1.0)
        
        # Damped update for stability
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - eta * s_grad
        # Normalize in log space for numerical stability
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new_sum = a_new.sum()
        a_new = a_new / a_new_sum
        
        # Check improvement
        score = evaluate_local(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  MD2 t={t} ({elapsed:.0f}s): best={best_score:.10f}, k={k_eff}, tau={tau:.1f}")
        
        if no_improve > 500:
            # Restart with current best
            a = best_a.copy()
            no_improve = 0
    
    return best_a, best_score

def fast_autocorr(a):

    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_autocorr(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

def compute_subgradient_fft(a, w):

def nesterov_mirror_descent(a0, time_limit, base_eta=2.0, tau=5.0, k_active=50,

        if time.monotonic() - t0 > time_limit:
            break

        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 1e-30)
        y_sum = y.sum()
        y = y / y_sum

        A_fft = rfft(y, fft_size_ac)

def fourier_kkt_solver(a0, time_limit, n_iter=200, verbose=True):
    """Solve KKT conditions in Fourier domain: iterate on active set, solve for a."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    t0 = time.monotonic()
    best_a = a.copy()
    best_score = evaluate_local(a)

    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(a)
        M = conv.max()

        # Active set: lags where conv is within 5% of max
        threshold = M * 0.95
        active = np.where(conv >= threshold)[0]
        if len(active) < 3:
            active = np.where(conv >= M * 0.8)[0]
        if len(active) < 3:
            active = np.argsort(conv)[-10:]

        # Solve: (w * a)_i = constant for all i
        # In Fourier: A(k) = C / W(k) for k != 0
        # w is uniform over active set
        w = np.zeros(size)
        w[active] = 1.0 / len(active)

        W = rfft(w, fft_size)
        # A(k) = 1/W(k) for k != 0, A(0) determined by sum constraint
        A_sol = np.zeros(fft_size // 2 + 1, dtype=np.complex128)
        for k in range(1, fft_size // 2 + 1):
            if abs(W[k]) > 1e-15:
                A_sol[k] = 1.0 / W[k]
            else:
                A_sol[k] = 0.0

        a_sol = irfft(A_sol, fft_size)[:n].real
        # Project onto simplex (a >= 0, sum = 1)
        a_sol = np.maximum(a_sol, 0.0)
        a_sol_sum = a_sol.sum()
        if a_sol_sum < 1e-15:
            continue
        a_sol = a_sol / a_sol_sum

        # Blend with current a
        alpha = 0.5
        a_new = (1 - alpha) * a + alpha * a_sol
        a_new = np.maximum(a_new, 0.0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            continue
        a_new = a_new / a_new_sum

        score = evaluate_local(a_new)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            a = a_new

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FKKT it={it} ({elapsed:.0f}s): best={best_score:.10f}, |S|={len(active)}")

    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss
