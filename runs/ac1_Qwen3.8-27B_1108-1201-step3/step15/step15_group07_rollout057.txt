```python
def compute_smooth_grad(a, tau):
    n = len(a)
    conv = fast_autocorr(a)  # length 2n-1
    # Softmax
    conv_max = conv.max()
    w = np.exp((conv - conv_max) / tau)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(conv)
        w[conv.argmax()] = 1.0
    else:
        w /= w_sum
    # Reversed w
    w_rev = w[::-1]
    # FFT size
    N = 1 << (3*n).bit_length()  # next power of 2 >= 3n
    A_f = rfft(a, N)
    W_f = rfft(w_rev, N)
    conv_aw = irfft(A_f * W_f, N)[:3*n-1]
    grad = 2.0 * conv_aw[n-1:2*n-1][::-1]
    return grad, conv.max(), conv
```

This is O(n log n) per iteration!

```python
def smooth_mirror_descent(a0, time_limit, tau0=1.0, tau_min=0.01, tau_decay=0.999):
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * S / a.sum()
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()  # Nesterov extrapolated point
    t = 1.0
    tau = tau0
    eta = 1.0  # step size
    
    t0 = time.monotonic()
    n_iters = 0
    patience = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute smooth gradient at y
        grad, conv_max, conv = compute_smooth_grad(y, tau)
        
        # Adaptive step size via line search
        best_eta = eta
        best_f = None
        for factor in [0.5, 0.7, 1.0, 1.5, 2.0]:
            eta_try = eta * factor
            # Mirror step
            log_a = np.log(y + 1e-30) - eta_try * grad
            a_try = np.exp(log_a - log_a.max())
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try * S / s_try
            f_try = log_sum_exp_approx(a_try, tau)  # or just use conv max
            if best_f is None or f_try < best_f:
                best_f = f_try
                best_eta = eta_try
        
        eta = best_eta
        
        # Mirror step
        log_a = np.log(y + 1e-30) - eta * grad
        a_new = np.exp(log_a - log_a.max())
        s_new = a_new.sum()
        a_new = a_new * S / s_new
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * a_new
        y_new = np.maximum(y_new, 1e-30)
        y_new = y_new * S / y_new.sum()
        
        if t < 2.0:
            a = a_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a = y_new + momentum * (y_new - y)
            a = np.maximum(a, 1e-30)
            a = a * S / a.sum()
        
        y = y_new
        t += 1.0
        n_iters += 1
        
        # Check improvement
        cur_max = fast_autocorr(a).max()
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        # Decay tau
        if n_iters % 10 == 0:
            tau = max(tau_min, tau * tau_decay)
        
        # Reset if stuck
        if patience > 200:
            # Reset Nesterov
            y = best_a.copy()
            a = best_a.copy()
            t = 1.0
            patience = 0
            tau = max(tau_min, tau * 2)  # increase tau (smoother)
            eta = 1.0 / np.sqrt(n)
        
        if n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SMD iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, tau={tau:.4f}, eta={eta:.4f}")
    
    return best_a, best_score
```

```python
def compute_smooth_grad(a, tau):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    conv = irfft(A * A, fft_size)[:size]
    conv_max = conv.max()
    shifted = (conv - conv_max) / tau
    shifted = np.clip(shifted, -700.0, 700.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(size)
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= w_sum
    w_rev = w[::-1]

    A2 = rfft(a, fft_size2)
    W2 = rfft(w_rev, fft_size2)
    conv_aw = irfft(A2 * W2, fft_size2)[:conv_len]
    grad = 2.0 * conv_aw[:n][::-1]
    return grad, conv_max, conv


def smooth_mirror_descent(a0, time_limit, tau0=0.5, tau_min=0.005, tau_decay=0.9995, verbose=True):

            a_try = np.exp(log_a - log_a.max())
            s_try = a_try.sum()

def build_conv_matrix_rows(a, indices):
    n = len(a)
    T = np.zeros((len(indices), n))
    for idx, k in enumerate(indices):
        i_lo = max(0, int(k) - n + 1)
        i_hi = min(int(k), n - 1)
        for j in range(i_lo, i_hi + 1):
            T[idx, j] = a[int(k) - j]
    return T


def lp_best_response_active(a, M, solver, top_k=150, verbose=False):
    n = len(a)
    size = 2 * n - 1
    conv = fast_autocorr(a)
    active = np.where(conv >= 0.7 * M)[0]
    if len(active) > top_k:
        order = np.argsort(conv[active])[::-1]
        active = active[order[:top_k]]
    elif len(active) < 10:
        order = np.argsort(conv)[::-1]
        active = order[:min(top_k, size)]
    T = build_conv_matrix_rows(a, active)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def nesterov_lp_optimize(a0, time_limit, solver, use_active=True, top_k=150, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    t_k = 1.0
    t0 = time.monotonic()
    n_iters = 0
    patience = 0
    stuck_count = 0
    while time.monotonic() - t0 < time_limit:
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        if use_active:
            b = lp_best_response_active(y, M_curr, solver, top_k=top_k)
        else:
            b = lp_best_response_full(y, M_curr, solver)
        if b is None:
            stuck_count += 1
            if stuck_count > 20:
                break
            continue
        stuck_count = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        alpha = 2.0 / (t_k + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 1e-30)
        y_new = y_new * norm / y_new.sum()
        if t_k < 2.0:
            a = g.copy()
        else:
            momentum = (t_k - 1.0) / (t_k + 2.0)
            a = y_new + momentum * (y_new - y)
            a = np.maximum(a, 1e-30)
            a = a * norm / a.sum()
        y = y_new
        t_k += 1.0
        score_y = fast_autocorr(y).max()
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            patience = 0
        else:
            patience += 1
        score_a = fast_autocorr(a).max()
        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
            patience = 0
        if patience > 60 and n_iters >= 30:
            y = best_a.copy()
            a = best_a.copy()
            t_k = 1.0
            patience = 0
        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, active={'Y' if use_active else 'N'}")
    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, n_offsets=8, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

    n_pass = 0
    while time.monotonic() - t0 < time_limit:
        n_pass += 1
        if no_improve > 100:
            break
        conv = fast_autocorr(a)
        size = 2 * n - 1
        top_lags = np.argsort(conv)[-min(20, size):]
        active = set(int(k) for k in top_lags)
        perm = rng.permutation(n)
        improved = False
        for idx in perm:
            if time.monotonic() - t0 > time_limit:
                break
            i = int(idx)
            if a[i] < norm * 1e-12:
                continue
            for off in range(1, n_offsets + 1):

                    j = int(j_candidate)
                    if j == i or a[j] >= a[i] * 0.8:
                        continue
                    for frac in [0.2, 0.4, 0.6, 0.8]:

                            break

                            a[i] += d
                            a[j] -= d

            a_out, _ = nesterov_lp_optimize(a_norm, min(remaining * 0.6, 60.0), solver, use_active=True, top_k=80, verbose=False)
            do_try_update(a_out, f"lp_final_n{n_best}")

    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    return [float(x) for x in best_seq]
```
