def fast_conv_self(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)

def build_T(a):

    size = 2 * n - 1
    T = np.zeros((size, n))

    return T

def lp_solve(a, M, solver, active=None, warm_start=None):

        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    duals = None

def fista_smoothed_max(a0, time_limit, verbose=True):
    """FISTA on the smoothed max of self-convolution."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Adaptive temperature schedule
    n_stages = 15
    temps = np.geomspace(50.0, 0.05, n_stages)

    for si, beta in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        # FISTA state
        y = a.copy()
        t_fista = 1.0
        L = 4.0 * n / (beta * beta)  # initial Lipschitz estimate (conservative)

        n_steps = max(50, min(500, int(remaining * 5)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            # Compute convolution of y
            conv_y = fast_conv_self(y)
            max_c = conv_y.max()
            if max_c < 1e-15:
                break

            # Smoothed max and its gradient
            shifted = (conv_y - max_c) / (1.0 / beta)
            shifted = np.maximum(shifted, -400.0)
            exp_c = np.exp(shifted)
            Z = exp_c.sum()
            w = exp_c / Z

            # Gradient: grad[i] = 2 * sum_k w[k] * a_y[k-i] (valid indices)
            # This is 2 * cross_correlation(w, y)
            W = rfft(w, fft_size)
            Y = rfft(y, fft_size)
            g1 = irfft(np.conj(Y) * W, fft_size)[:n]
            g2 = irfft(Y * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            # Backtracking line search
            step = 1.0 / L
            for _bt in range(20):
                a_new = y - step * grad
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    step *= 0.5
                    continue
                a_new = a_new * norm / s_new
                conv_new = fast_conv_self(a_new).max()
                # Check descent condition
                f_y = (1.0 / beta) * np.log(Z) + max_c
                f_new_max = conv_new.max()
                if f_new_max <= f_y + 1e-10:
                    break
                step *= 0.5

            a_candidate = y - step * grad
            a_candidate = np.maximum(a_candidate, 0.0)
            s_cand = a_candidate.sum()
            if s_cand < 1e-15:
                a_candidate = y.copy()
            else:
                a_candidate = a_candidate * norm / s_cand

            # FISTA update
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_fista * t_fista))
            y_new = a_candidate + ((t_fista - 1.0) / t_new) * (a_candidate - y)
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = a_candidate.copy()
            else:
                y_new = y_new * norm / s_y

            y = y_new
            t_fista = t_new
            L = max(L * 0.9, 1e-10)  # slowly decrease L estimate

            sc = fast_conv_self(a_candidate).max()
            if sc < best_score:
                best_score = sc
                best_a = a_candidate.copy()

            # Also check y
            sc_y = fast_conv_self(y).max()
            if sc_y < best_score:
                best_score = sc_y
                best_a = y.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    FISTA stage {si} beta={beta:.2f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

            a_new = y - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new * norm / s_new

            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_f * t_f))
            y_new = a_new + ((t_f - 1.0) / t_new) * (a_new - y)
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = a_new.copy()
            else:
                y_new = y_new * norm / s_y
            y = y_new
            t_f = t_new

            sc = fast_conv_self(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()

        a = y

        # LP correction every 30 FISTA steps
        if time.monotonic() - t0 < time_limit:
            conv_a = fast_conv_self(a)
            M = conv_a.max()
            active = np.where(conv_a > M * 0.7)[0]
            if len(active) < 30:
                active = None
            b, _ = lp_solve(a, M, solver, active, warm_start=prev_lp)
            if b is not None and b.sum() > 1e-15:
                g = b * norm / b.sum()
                # Blend with current
                blend = 0.3
                a = (1 - blend) * a + blend * g
                a = np.maximum(a, 0.0)
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a * norm / s_a
                    sc = fast_conv_self(a).max()
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                prev_lp = b.copy()

        it_count += 1
        # Decrease beta over time
        beta = max(0.5, beta * 0.95)

        if verbose and it_count % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Hybrid it={it_count} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent for final polishing."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_score(a)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            j_candidates = set()
            for _ in range(8):
                j_candidates.add(int(rng.integers(n)))

            peak_lags = np.where(conv >= cur_max * 0.85)[0]
            for kl in peak_lags[:6]:
                j1 = (kl - i) % n
                j2 = (kl + i) % n
                j_candidates.add(j1)
                j_candidates.add(j2)
            j_candidates.discard(i)

            if not j_candidates:
                continue

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in list(j_candidates)[:12]:
                if j == i:
                    continue
                aj = a[j]

                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                n_samples = 15
                deltas = np.linspace(0.01, 0.9, n_samples) * ai
                best_d = 0.0
                best_v = cur_max
                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_d = d

                if best_d > 0 and best_v < best_val - 1e-12:
                    best_val = best_v
                    best_delta = best_d
                    best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 6:
                break
            if no_improve == 3:
                n_perturb = max(5, n // 40)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.1 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and total_improvements % 100 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD imp={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def structured_init(n, rng):
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.15 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0.1)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-6.0 * (x - 0.75) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-2.0 * x) + 0.5 * np.exp(-2.0 * (1 - x))
    elif choice == 7:
        a = 0.5 * np.exp(-4.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-4.0 * (x - 0.7) ** 2)
        a = a * (1.0 + 0.3 * np.cos(20 * np.pi * x))
        a = np.maximum(a, 0.01)
    elif choice == 8:
        a = np.abs(np.sin(3 * np.pi * x)) ** 0.8
    else:
        a = 0.4 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(11 * np.pi * x)
        a = np.maximum(a, 0.05)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_init = None
        best_init_score = float('inf')
        for trial in range(8):
            cand = structured_init(n_base, rng)
            sc = eval_fn(cand)
            if sc < best_init_score:
                best_init_score = sc
                best_init = cand
        best_seq = best_init
        best_score = best_init_score
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 0: Quick FISTA warm-up
    remaining = time_budget - (time.monotonic() - t_global)
    t0_fista = min(remaining * 0.06, 80.0)
    if t0_fista > 15:
        print(f"\n=== Phase 0: FISTA warm-up n={n_base} (budget={t0_fista:.0f}s) ===")
        a_out, _ = fista_smoothed_max(best_seq, t0_fista, verbose=False)
        try_update(a_out, "fista_warmup")

    # Phase 1: LP Mirror Descent (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = len(best_seq)
    t1 = min(remaining * 0.30, 480.0)
    print(f"\n=== Phase 1: LP Mirror Descent n={n1} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = lp_mirror_descent(best_seq, t1, solver, verbose=True, use_warm_start=True)
        try_update(a_out, f"lp1_n{n1}")

    # Phase 2: Hybrid FISTA+LP
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq)
    t2 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 2: Hybrid FISTA+LP n={n2} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = hybrid_fista_lp(best_seq, t2, solver, verbose=True)
        try_update(a_out, f"hybrid_n{n2}")

    # Phase 3: LP Mirror Descent round 2 (refined)
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = len(best_seq)
    t3 = min(remaining * 0.22, 350.0)
    print(f"\n=== Phase 3: LP MD round 2 n={n3} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = lp_mirror_descent(best_seq, t3, solver, verbose=True, use_warm_start=True)
        try_update(a_out, f"lp2_n{n3}")

    # Phase 4: Try longer sequence
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 400
    print(f"\n=== Phase 4: Length extension n={n_long} (remaining={remaining:.0f}s) ===")
    if remaining > 120:
        init_long = interpolate_seq(best_seq, n_long)
        # Quick FISTA on longer sequence
        t_long_f = min(remaining * 0.15, 80.0)
        if t_long_f > 20:
            a_out, _ = fista_smoothed_max(init_long, t_long_f, verbose=False)
            try_update(a_out, f"fista_long_{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80 and len(best_seq) == n_long:
            t_long_lp = min(remaining * 0.5, 200.0)
            a_out, _ = lp_mirror_descent(best_seq, t_long_lp, solver, verbose=True, use_warm_start=True)
            try_update(a_out, f"lp_long_{n_long}")

    # Phase 5: Pair CD polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    n5 = len(best_seq)
    t5 = min(remaining * 0.20, 200.0)
    print(f"\n=== Phase 5: Pair CD n={n5} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = pair_cd_optimize(best_seq, t5, rng, verbose=True)
        try_update(a_out, f"pcd_n{n5}")

    # Phase 6: Final FISTA + LP polish
    remaining = time_budget - (time.monotonic() - t_global)
    n6 = len(best_seq)
    t6 = min(remaining * 0.35, 150.0)
    print(f"\n=== Phase 6: Final FISTA n={n6} (budget={t6:.0f}s) ===")
    if t6 > 20:
        a_out, _ = fista_smoothed_max(best_seq, t6, verbose=False)
        try_update(a_out, "fista_final")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        t6b = min(remaining * 0.5, 100.0)
        print(f"\n=== Phase 7: Final LP n={len(best_seq)} (budget={t6b:.0f}s) ===")
        a_out, _ = lp_mirror_descent(best_seq, t6b, solver, verbose=False, use_warm_start=True)
        try_update(a_out, "lp_final")

    # Phase 8: Last pair CD
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t8 = min(remaining * 0.6, 60.0)
        a_out, _ = pair_cd_optimize(best_seq, t8, rng, verbose=False)
        try_update(a_out, "pcd_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
