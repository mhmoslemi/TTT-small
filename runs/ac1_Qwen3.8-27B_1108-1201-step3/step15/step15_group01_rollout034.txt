    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)  # strict positive for log
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    t0 = time.monotonic()
    eta0 = 1.0  # initial step size
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_autocorr(a)
        peak = conv.max()
        
        # Find peak lags
        peak_threshold = peak * (1 - 1e-6)
        peak_lags = np.where(conv >= peak_threshold)[0]
        
        # Compute subgradient
        g = np.zeros(n)
        for k in peak_lags:
            for j in range(n):
                # d(a*a)[k]/da[j] = a[k-j] (if valid) + a[j-k] (if valid)
                if 0 <= k - j < n:
                    g[j] += a[k - j]
                if 0 <= j - k < n:
                    g[j] += a[j - k]
        
        # Adaptive step size
        eta = eta0 / (it + 1)
        
        # Mirror descent update
        logits = np.log(a) - eta * g
        logits -= logits.max()  # for numerical stability
        a_new = np.exp(logits)
        a_new = a_new / a_new.sum()
        
        a = a_new
        
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Also check averaged iterate
        if (it + 1) % 100 == 0:
            pass  # Could add averaging here
    
    return best_a, best_score

def lp_best_response(a, M, solver):

        logits -= logits.max()
        a_new = np.exp(logits)
        a_new = a_new / a_new.sum()

        a = a_new

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Ergodic average
        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            avg_check = avg_check / avg_check.sum()
            score_avg = fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

        if verbose and (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MG iter={it+1} ({elapsed:.0f}s): best={best_score:.8f}")

    # Rescale back to norm
    best_a = best_a * norm / best_a.sum()
    return best_a, best_score

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for i in range(n):
        T[i:i + n, i] = a
    return T


def lp_best_response(a, M, solver):
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stuck = 0
    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-10:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            b = lp_best_response(y, M_curr * 1.001, solver)
            if b is None:
                stuck += 1
                if stuck > 30:
                    break
                continue

        stuck = 0
        b_sum = b.sum()
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_curr = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            a_curr = y_new + beta * (y_new - y)
            a_curr = np.maximum(a_curr, 0.0)
            s_a = a_curr.sum()
            if s_a < 1e-15:
                a_curr = y_new.copy()
            else:
                a_curr = a_curr * norm / s_a

        y = y_new
        a = a_curr

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 30 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and n_iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def smooth_grad_step(a, T_temp):
    n = len(a)
    conv = fast_autocorr(a)
    c_max = conv.max()
    exp_c = np.exp((conv - c_max) / T_temp)
    w = exp_c / exp_c.sum()

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    cross = irfft(W * np.conj(A), fft_size)[:n]
    grad = 2.0 * cross
    return grad, conv.max()


def smooth_optimize(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = [5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]

    def project_simplex(v, S):
        m = len(v)
        if m == 0:
            return v
        v_sorted = np.sort(v)[::-1]
        css = np.cumsum(v_sorted)
        thrust = (css - S) / np.arange(1, m + 1)
        rho = np.where(v_sorted > thrust)[0]
        if len(rho) == 0:
            return np.zeros(m)
        rho = rho[-1]
        theta = thrust[rho]
        return np.maximum(v - theta, 0.0)

    for T_val in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        n_steps = max(20, min(300, int(remaining * 4)))

        y = a.copy()
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            grad, _ = smooth_grad_step(y, T_val)
            step = norm / (n * T_val * (1.0 + 0.02 * it))
            step = min(step, norm / (n * 0.3))

            a_new = y - step * grad
            a_new = project_simplex(a_new, norm)

            alpha = 2.0 / (it + 2.0)
            y_new = (1.0 - alpha) * y + alpha * a_new
            y_new = project_simplex(y_new, norm)

            if it == 0:
                a_curr = a_new.copy()
            else:
                momentum = (it - 1.0 + 1.0) / (it + 2.0)
                a_curr = y_new + momentum * (y_new - y)
                a_curr = project_simplex(a_curr, norm)

            y = y_new
            a = a_curr

            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

            score_y = fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    Smooth T={T_val:.3f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def subgradient_mirror(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-10)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    avg_a = np.zeros(n)
    n_avg = 0

    for it in range(300000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(a)
        peak = conv.max()
        threshold = peak * (1.0 - 1e-3)
        u = (conv >= threshold).astype(np.float64)

        g = 2.0 * np.correlate(u, a, mode='valid')
        g = g[:n]

        eta_t = 1.0 / np.sqrt(it + 1)
        logits = np.log(a) - eta_t * g
        logits -= logits.max()
        a_new = np.exp(logits)
        a_new = a_new / a_new.sum()
        a = a_new

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 200 == 0:
            avg_check = avg_a / n_avg
            avg_check = avg_check / avg_check.sum()
            score_avg = fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

        if verbose and (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MG iter={it+1} ({elapsed:.0f}s): best={best_score:.8f}")

    best_a = best_a * norm / best_a.sum()
    return best_a, best_score


def pair_cd_polish(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    n_passes = 0
    no_improve = 0
    window = max(20, n // 40)

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-10:
                continue

            for offset in range(1, window + 1, max(1, window // 20)):
                j_candidates = [(i + offset) % n, (i - offset) % n]
                for j in j_candidates:
                    if j == i or a[i] < norm * 1e-10:
                        continue

                    for frac in [0.5, 0.25, 0.1]:
                        d = a[i] * frac
                        if d < 1e-15:
                            continue
                        a_test = a.copy()
                        a_test[i] -= d
                        a_test[j] += d
                        a_test = a_test * norm / a_test.sum()
                        conv_test = fast_autocorr(a_test)
                        if conv_test.max() < best_score - 1e-12:
                            a = a_test
                            conv = conv_test
                            best_score = conv.max()
                            best_a = a.copy()
                            pass_improved = True
                            no_improve = 0
                            break

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def spectral_polish(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

    for it in range(500):
        if time.monotonic() - t0 > time_limit:
            break

        A = rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()

        # Try to flatten the power spectrum
        for lam in [0.1, 0.3, 0.5, 0.8]:
            P_new = P * (1.0 + lam * (P_mean / (P + 1e-15) - 1.0))
            P_new = np.maximum(P_new, 1e-15)

            A_new[0] = np.abs(A[0])  # DC is real and non-negative

            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new * norm / s_new
            score = fast_autocorr(a_new).max()
            if score < best_score - 1e-12:
                best_score = score
                best_a = a_new.copy()
                a = a_new.copy()

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    Spectral polish ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-10:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: LP Nesterov (50% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.50
    n1 = n_base
    print(f"\n=== Phase 1: LP Nesterov n={n1} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n1) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp1_n{n1}")

    # === Phase 2: Smooth gradient (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t2 = remaining * 0.20
    print(f"\n=== Phase 2: Smooth grad n={n_c} (budget={t2:.0f}s) ===")
    if t2 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = smooth_optimize(a_norm, t2, verbose=True)
        do_try_update(a_out, f"smooth_n{n_c}")

    # === Phase 3: Pair CD (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t3 = remaining * 0.15
    print(f"\n=== Phase 3: Pair CD n={n_c} (budget={t3:.0f}s) ===")
    if t3 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = pair_cd_polish(a_norm, t3, rng, verbose=True)
        do_try_update(a_out, f"pcd_n{n_c}")

    # === Phase 4: Spectral polish (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t4 = remaining * 0.10
    print(f"\n=== Phase 4: Spectral polish n={n_c} (budget={t4:.0f}s) ===")
    if t4 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = spectral_polish(a_norm, t4, verbose=True)
        do_try_update(a_out, f"spectral_n{n_c}")

    # === Phase 5: LP Nesterov round 2 (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t5 = remaining * 0.05
    print(f"\n=== Phase 5: LP Nesterov r2 n={n_c} (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t5, solver, verbose=False)
        do_try_update(a_out, f"lp2_n{n_c}")

    # === Phase 6: Diversified restart ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t6 = remaining * 0.10
    print(f"\n=== Phase 6: Diversified restart n={n_c} (budget={t6:.0f}s) ===")
    if t6 > 30:
        a_pert = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        noise = rng.normal(0, 0.05 * a_pert.std(), n_c)
        a_pert = a_pert + noise
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert * np.sqrt(2.0 * n_c) / a_pert.sum()
        t6a = t6 * 0.5
        a_out, _ = smooth_optimize(a_pert, t6a, verbose=False)
        do_try_update(a_out, "restart_smooth")
        remaining6 = time_budget - (time.monotonic() - t_global)
        if remaining6 > 10 and best_seq is not None:
            n_r = len(best_seq)
            a_norm_r = best_seq * np.sqrt(2.0 * n_r) / best_seq.sum()
            a_out_r, _ = pair_cd_polish(a_norm_r, min(remaining6 * 0.5, 30.0), rng, verbose=False)
            do_try_update(a_out_r, "restart_pcd")

    # === Final: try alternate lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60 and best_seq is not None:
        n_base_final = len(best_seq)
        for n_alt in [n_base_final + 200, n_base_final - 200]:
            if n_alt < 100:
                continue
            remaining_sub = time_budget - (time.monotonic() - t_global)
            if remaining_sub < 30:
                break
            print(f"\n=== Alt length n={n_alt} (budget={min(remaining_sub*0.3, 90.0):.0f}s) ===")
            t_alt = min(remaining_sub * 0.3, 90.0)
            # Interpolate
            old_x = np.linspace(0.0, 1.0, n_base_final, endpoint=False)
            new_x = np.linspace(0.0, 1.0, n_alt, endpoint=False)
            f_norm = best_seq / best_seq.sum()
            f_interp = np.interp(new_x, old_x, f_norm)
            f_interp = np.maximum(f_interp, 0.0)
            ss = f_interp.sum()
            if ss > 1e-15:
                f_interp = f_interp * np.sqrt(2.0 * n_alt) / ss
            else:
                f_interp = np.ones(n_alt) * np.sqrt(2.0 * n_alt) / n_alt
            a_out_alt, _ = nesterov_lp(f_interp, t_alt, solver, verbose=False)
            do_try_update(a_out_alt, f"alt_n{n_alt}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
