        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    duals = None
    try:
        if len(active_rows) < (2 * n - 1):
            full_duals = np.zeros(2 * n - 1)
            d = np.asarray(constraints[0].dual_value, dtype=np.float64)
            if d is not None and np.all(np.isfinite(d)):
                full_duals[active_rows] = d
                duals = full_duals
    except Exception:
        pass
    return val, duals

def evaluate_sequence_local(sequence):
    try:
        sc = evaluate_sequence(sequence)
        return float(sc)
    except Exception:
        return float('inf')

def compute_local_score(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    n = len(a)
    max_b = fast_conv_self(a).max()
    return 2.0 * n * max_b / (S * S)

def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    n_stages = 14
    temps = np.geomspace(200.0, 0.05, n_stages)
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(50, min(500, int(remaining * 5)))
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -300.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            base_step = norm / (T_val * np.sqrt(n) * (1.0 + 0.02 * it))
            step = base_step
            accepted = False
            for trial in range(20):
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    step *= 0.5
                    continue
                a_new = a_new * norm / s_new
                sc = fast_conv_self(a_new).max()
                if sc <= max_c + 0.1 * T_val:
                    a = a_new
                    accepted = True
                    break
                step *= 0.5
            sc = fast_conv_self(a).max()
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best={best_score:.8f}")
    return best_a, best_score

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, use_restarts=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    y = a.copy()
    no_improve = 0
    total_improvements = 0
    last_best_score = best_score
    for t in range(1, 800000):
        if time.monotonic() - t0 > time_limit:
            break
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break
        if t < 30:
            active = np.where(conv_y > M * 0.6)[0]
        elif t < 100:
            active = np.where(conv_y > M * 0.75)[0]
        elif t < 300:
            active = np.where(conv_y > M * 0.85)[0]
        else:
            active = np.where(conv_y > M * 0.9)[0]
        max_active = min(len(active), 600)
        if max_active < 50:
            active = np.where(conv_y > M * 0.5)[0]
            max_active = min(len(active), 600)
        if max_active > 50:
            sorted_idx = np.argsort(conv_y[active])[::-1][:max_active]
            active_use = active[sorted_idx]
        else:
            active_use = np.arange(size)
        b, duals = lp_solve_fast(y, M, active_use, solver)
        if b is None:
            b, duals = lp_solve_fast(y, M, np.arange(size), solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        if len(active_use) < size:
            check_conv = fast_conv_cross(y, g)
            if check_conv.max() > M * (1 + 1e-2):
                b_full, _ = lp_solve_fast(y, M, np.arange(size), solver)
                if b_full is not None and b_full.sum() > 1e-15:
                    g = b_full * norm / b_full.sum()
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        if t == 1:
            a_new = g.copy()
        else:
            momentum = min(1.0, (t - 1.0) / (t + 2.0))
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        y = y_new
        a = a_new
        sc_y = fast_conv_self(y).max()
        improved = False
        if sc_y < best_score - 1e-10:
            best_score = sc_y
            best_a = y.copy()
            improved = True
            total_improvements += 1
        sc_a = fast_conv_self(a).max()
        if sc_a < best_score - 1e-10:
            best_score = sc_a
            best_a = a.copy()
            improved = True
            total_improvements += 1
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
        if use_restarts:
            if no_improve == 80:
                n_pert = max(20, n // 15)
                idx_p = np.random.default_rng(t * 7 + 3).choice(n, size=n_p, replace=False)
                y[idx_p] *= (1.0 + 0.25 * np.random.default_rng(t * 11 + 5).standard_normal(n_p))
                y = np.maximum(y, 0.0)
                s_y2 = y.sum()
                if s_y2 > 1e-15:
                    y = y * norm / s_y2
                no_improve = 0
            if no_improve == 200:
                y = a.copy()
                no_improve = 0
                if verbose:
                    print(f"    full restart at t={t}")
        if verbose and t % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}, impr={total_improvements}")
    return best_a, best_score

def peak_shaving_cd(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_local_score(a)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        peak_lags = np.where(conv >= cur_max * 0.95)[0]
        pass_improved = False
        peak_lags_sorted = np.argsort(conv[peak_lags])[::-1]
        for pl_idx in peak_lags_sorted[:20]:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(peak_lags[pl_idx])
            for i in range(n):
                if time.monotonic() - t0 > time_limit:
                    break
                j = k - i
                if j < 0 or j >= n:
                    continue
                if i == j:
                    continue
                if i > j:
                    continue
                ai = a[i]
                aj = a[j]
                if ai < norm * 1e-8 or aj < norm * 1e-8:
                    continue
                delta = 0.1 * min(ai, aj)
                a[i] -= delta
                a[j] -= delta
                a_new = np.maximum(a, 0.0)
                new_conv = fast_conv_self(a_new)
                new_max = new_conv.max()
                if new_max < cur_max - 1e-10:
                    a = a_new
                    cur_max = new_max
                    pass_improved = True
                    if cur_max < best_score - 1e-12:
                        best_score = cur_max
                        best_a = a.copy()
                else:
                    a[i] += delta
                    a[j] += delta
                    continue
                redistribute = delta * 2.0
                target_idx = int(rng.integers(n))
                a[target_idx] += redistribute
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        if not pass_improve if False else not pass_improved:
            no_improve += 1
            if no_improve >= 10:
                break
            if no_improve == 5:
                n_pert = max(10, n // 20)
                idx_p = rng.choice(n, size=n_pert, replace=False)
                a[idx_p] *= (1.0 + 0.1 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0
        if verbose and no_improve % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave ({elapsed:.0f}s): best={best_score:.8f}, impr={not pass_improved}")
    return best_a, best_score

def structured_init(n, rng):
    choice = int(rng.integers(0, 12))
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.2
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.15 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0.1)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-6.0 * (x - 0.75) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.5
    elif choice == 6:
        a = np.exp(-2.0 * x) + 0.5 * np.exp(-2.0 * (1 - x))
    elif choice == 7:
        a = 0.5 * np.exp(-4.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-4.0 * (x - 0.7) ** 2)
        a = a * (1.0 + 0.3 * np.cos(20 * np.pi * x))
        a = np.maximum(a, 0.01)
    elif choice == 8:
        a = np.abs(np.sin(3 * np.pi * x)) ** 0.8
    elif choice == 9:
        a = 0.4 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(11 * np.pi * x)
        a = np.maximum(a, 0.05)
    elif choice == 10:
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 5 * x) + 0.1 * np.cos(2 * np.pi * 13 * x)
        a = np.maximum(a, 0.05)
    else:
        a = np.sin(np.pi * x * 0.5) ** 0.7 + 0.2 * np.sin(np.pi * x * 2.0)
        a = np.maximum(a, 0.01)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def symmetric_refine(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = 0.5 * (a + a[::-1])
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    y = a.copy()
    no_improve = 0
    for t in range(1, 400000):
        if time.monotonic() - t0 > time_limit:
            break
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break
        if t < 50:
            active = np.where(conv_y > M * 0.7)[0]
        else:
            active = np.where(conv_y > M * 0.85)[0]
        max_act = min(len(active), 500)
        if max_act < 30:
            active = np.arange(size)
            max_act = size
        else:
            order = np.argsort(conv_y[active])[::-1][:max_act]
            active_use = active[order]
        b, _ = lp_solve_fast(y, M, active_use, solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        g = 0.5 * (g + g[::-1])
        g = np.maximum(g, 0.0)
        sg = g.sum()
        if sg < 1e-15:
            continue
        g = g * norm / sg
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = 0.5 * (y_new + y_new[::-1])
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / sy
        y = y_new
        sc = fast_conv_self(y).max()
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve == 100:
                n_p = max(10, n // 25)
                half = n // 2
                idx_p = np.random.default_rng(t).choice(half, size=min(n_p, half), replace=False)
                y[idx_p] *= (1.0 + 0.2 * np.random.default_rng(t+1).standard_normal(len(idx_p)))
                y[half:] = y[:half][::-1] if n % 2 == 0 else y[:half][::-1]
                y = np.maximum(y, 0.0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y * norm / sy2
                no_improve = 0
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SymLP t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    return best_a, best_score

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")
    best_seq = None
    best_score = float('inf')
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_sequence_local([float(x) for x in best_seq])
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_sequence_local([float(x) for x in a])
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
    if best_seq is None:
        n_base = 1600
        best_init = None
        best_init_score = float('inf')
        for trial in range(8):
            cand = structured_init(n_base, rng)
            sc = evaluate_sequence_local([float(x) for x in cand])
            if sc < best_init_score:
                best_init_score = sc
                best_init = cand
        best_seq = best_init
        best_score = best_init_score
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t_init = min(remaining * 0.06, 90.0)
    if t_init > 15:
        print(f"\n=== Init LSE (budget={t_init:.0f}s) ===")
        a_out, _ = lse_gradient_descent(best_seq, t_init, verbose=False)
        try_update(a_out, "lse_init")
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = len(best_seq)
    t1 = min(remaining * 0.30, 480.0)
    print(f"\n=== Phase 1: LP Nesterov n={n1} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t1, solver, verbose=True, use_restarts=True)
        try_update(a_out, f"lp1_n{n1}")
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq)
    t2 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 2: Symmetric LP n={n2} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = symmetric_refine(best_seq, t2, solver, verbose=True)
        try_update(a_out, f"sym_n{n2}")
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = len(best_seq)
    t3 = min(remaining * 0.08, 100.0)
    print(f"\n=== Phase 3: LSE refine n={n3} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_out, _ = lse_gradient_descent(best_seq, t3, verbose=False)
        try_update(a_out, "lse_refine")
    remaining = time_budget - (time.monotonic() - t_global)
    n4 = len(best_seq)
    t4 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 4: LP Nesterov round2 n={n4} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t4, solver, verbose=True, use_restarts=True)
        try_update(a_out, f"lp2_n{n4}")
    remaining = time_budget - (time.monotonic() - t_global)
    n5 = len(best_seq)
    t5 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 5: Peak shaving n={n5} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = peak_shaving_cd(best_seq, t5, rng, verbose=True)
        try_update(a_out, f"peak_n{n5}")
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 400
    print(f"\n=== Phase 6: Length extension n={n_long} (remaining={remaining:.0f}s) ===")
    if remaining > 120:
        init_long = interpolate_seq(best_seq, n_long)
        t_long = min(remaining * 0.5, 180.0)
        a_out, _ = nesterov_lp_optimize(init_long, t_long, solver, verbose=False, use_restarts=True)
        try_update(a_out, f"lp_long_{n_long}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t_long2 = min(remaining * 0.5, 100.0)
            a_out2, _ = lse_gradient_descent(best_seq, t_long2, verbose=False)
            try_update(a_out2, "lse_long")
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 7: Final polish n={n_final} (remaining={remaining:.0f}s) ===")
    if remaining > 30:
        t_f1 = min(remaining * 0.4, 60.0)
        a_out, _ = lse_gradient_descent(best_seq, t_f1, verbose=False)
        try_update(a_out, "lse_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t_f2 = min(remaining * 0.5, 50.0)
            a_out, _ = nesterov_lp_optimize(best_seq, t_f2, solver, verbose=False, use_restarts=True)
            try_update(a_out, "lp_final")
    if best_seq is not None:
        final_score = evaluate_sequence_local([float(x) for x in best_seq])
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    return [float(x) for x in best_seq]
