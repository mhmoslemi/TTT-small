        if b is None:
            patience += 1
            if patience > restart_patience:
                y = best_a.copy()
                t_mom = 1.0
                patience = 0
                consecutive_no_improve += 1
            if consecutive_no_improve > 5:
                break
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            patience += 1
            if patience > restart_patience:
                y = best_a.copy()
                t_mom = 1.0
                patience = 0
            continue

        patience = 0
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t_mom + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t_mom < 3.0:
            a_new = g.copy()
        else:
            momentum = (t_mom - 1.0) / (t_mom + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new
        t_mom += 1.0

        score_y = fast_autocorr(y).max()
        improved = False
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            improved = True

        score_a = fast_autocorr(a).max()
        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
            improved = True

        if improved:
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > restart_patience and n_iters % 10 == 0:
            y = best_a.copy()
            a = best_a.copy()
            t_mom = 1.0
            consecutive_no_improve = 0
            patience = 0

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP iter={n_iters} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def cd_polish(a0, time_limit, window=15, verbose=False, rng=None):
    if rng is None:
        rng = _global_rng
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, fast_autocorr(a).max()
    a = a * norm / s0

    best_a = a.copy()
    conv = fast_autocorr(a)
    best_score = conv.max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_autocorr(a)
        max_val = conv.max()
        top_lags = np.argsort(conv)[-min(10, size):]
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-11:
                continue
            for off in [1, -1, 2, -2, 3, -3, 5, -5, 10, -10]:
                j = (i + off) % n
                if j == i:
                    continue
                for frac in [0.2, 0.4, 0.6]:
                    d = a[i] * frac
                    if d < norm * 1e-12:
                        continue
                    a[i] -= d
                    a[j] += d
                    new_conv = fast_autocorr(a)
                    new_max = new_conv.max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        conv = new_conv
                        pass_improved = True
                        no_improve = 0
                        break
                    else:
                        a[i] += d
                        a[j] -= d
                else:
                    continue
                break

        if not pass_improved:
            no_improve += 1
            if no_improve >= 8:
                noise = rng.normal(0, norm * 0.003, n)
                a_pert = a + noise
                a_pert = np.maximum(a_pert, 0)
                s_pert = a_pert.sum()
                if s_pert > 1e-15:
                    a = a_pert * norm / s_pert
                no_improve = 0
            if no_improve >= 30:
                break

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"      CD: best={best_score:.8f} ({elapsed:.0f}s)")

    return best_a, best_score


def resample_sequence(a, n_target, rng=None):
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    if rng is None:
        rng = _global_rng
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    a_interp = np.interp(x_tgt, x_src, a)
    a_interp = np.maximum(a_interp, 0)
    s = a_interp.sum()
    if s < 1e-15:
        a_interp = np.ones(n_target)
    else:
        a_interp = a_interp / s
    return a_interp


def get_parent_sequence():
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p / p.sum()
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1800.0):
    global _global_rng
    _global_rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1720.0)

    best_seq = None
    best_score = float('inf')

    parent = get_parent_sequence()

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is not None:
        best_seq = parent.copy()
        best_score = evaluate_safe(best_seq)
        print(f"Parent: n={len(parent)}, score={best_score:.6f}")
    else:
        n_init = 1600
        best_seq = spectral_init(n_init, mode='raised_cosine', rng=_global_rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_parent = len(best_seq)

    # Phase 1: Multi-start exploration (45% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = remaining * 0.45
    print(f"\n=== Phase 1: Multi-start exploration (budget={t_per:.0f}s) ===".format(t_per=t_phase1))

    configs = []
    # Parent-based with different hyperparams
    configs.append({'n': n_parent, 'init': 'parent', 'active_k': 30, 'patience': 50, 'm_pert': 0.0})
    configs.append({'n': n_parent, 'init': 'parent', 'active_k': 50, 'patience': 80, 'm_pert': 0.02})
    configs.append({'n': n_parent, 'init': 'parent', 'active_k': 40, 'patience': 60, 'm_pert': 0.05})
    # Different n values
    for n_try in [1200, 2000]:
        if n_try != n_parent:
            configs.append({'n': n_try, 'init': 'resample', 'active_k': 30, 'patience': 50, 'm_pert': 0.0})
    # Spectral init
    configs.append({'n': n_parent, 'init': 'spectral', 'active_k': 30, 'patience': 50, 'm_pert': 0.0})
    # Random
    configs.append({'n': n_parent, 'init': 'random', 'active_k': 25, 'patience': 40, 'm_pert': 0.0})

    t_per_config = t_phase1 / len(configs)
    for ci, cfg in enumerate(configs):
        if time.monotonic() - t_global > time_budget - 60:
            break
        t_sub = min(t_per_config, time_budget - (time.monotonic() - t_global) - 30)
        if t_sub < 10:
            break

        n_cfg = cfg['n']
        init_mode = cfg['init']

        if init_mode == 'parent' and parent is not None:
            a_init = resample_sequence(parent, n_cfg)
        elif init_mode == 'resample' and parent is not None:
            a_init = resample_sequence(parent, n_cfg)
        elif init_mode == 'spectral':
            a_init = spectral_init(n_cfg, mode='raised_cosine', rng=_global_rng)
        elif init_mode == 'random':
            a_init = spectral_init(n_cfg, mode='random_exp', rng=_global_rng)
        else:
            a_init = resample_sequence(best_seq, n_cfg)

        elapsed_cfg = time.monotonic() - t_global
        print(f"  Config {ci}: n={n_cfg}, init={init_mode}, k={cfg['active_k']} (t={elapsed_cfg:.0f}s)")

        a_out, sc = nesterov_lp_optimize(
            a_init, t_sub,
            active_k=cfg['active_k'],
            restart_patience=cfg['patience'],
            m_perturb=cfg['m_pert'],
            verbose=False,
            rng=_global_rng
        )
        try_update(a_out, f"cfg{ci}_n{n_cfg}")

    # Phase 2: Refined optimization on best (35% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase2 = remaining * 0.35
    print(f"\n=== Phase 2: Refined optimization (budget={t_phase2:.0f}s) ===")

    if t_phase2 > 30:
        n_best = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()

        # Run multiple variants
        t_per_variant = t_phase2 / 3.0
        variants = [
            {'active_k': 40, 'patience': 70, 'm_pert': 0.0},
            {'active_k': 60, 'patience': 100, 'm_pert': 0.03},
            {'active_k': 25, 'patience': 50, 'm_pert': 0.0},
        ]
        for vi, var in enumerate(variants):
            t_sub = min(t_per_variant, time_budget - (time.monotonic() - t_global) - 20)
            if t_sub < 10:
                break
            print(f"  Variant {vi}: k={var['active_k']}, patience={var['patience']} (t={time.monotonic()-t_global:.0f}s)")
            a_out, sc = nesterov_lp_optimize(
                best_seq, t_sub,
                active_k=var['active_k'],
                restart_patience=var['patience'],
                m_perturb=var['m_pert'],
                verbose=True,
                rng=_global_rng
            )
            try_update(a_out, f"refine{vi}")

    # Phase 3: CD polish + random restarts (20% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase3 = remaining * 0.20
    print(f"\n=== Phase 3: Polish (budget={t_phase3:.0f}s) ===")

    if t_phase3 > 15:
        n_best = len(best_seq)
        t_cd = t_phase3 * 0.5
        t_rr = t_phase3 * 0.5

        # CD polish
        if t_cd > 10:
            print(f"  CD polish (budget={t_cd:.0f}s)")
            a_out, sc = cd_polish(best_seq, t_cd, window=20, verbose=False, rng=_global_rng)
            try_update(a_out, "cd")

        # Random restarts
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            n_best = len(best_seq)
            t_rr = remaining * 0.6
            n_restarts = max(2, int(t_rr / 30))
            t_per_restart = t_rr / n_restarts
            for ri in range(n_restarts):
                if time.monotonic() - t_global > time_budget - 20:
                    break
                t_sub = min(t_per_restart * 0.7, time_budget - (time.monotonic() - t_global) - 10)
                if t_sub < 10:
                    break
                scale = np.sqrt(2.0 * n_best) * 0.03 * (ri + 1)
                noise = _global_rng.laplace(0, scale, n_best)
                a_pert = best_seq + noise
                a_pert = np.maximum(a_pert, 0)
                s_pert = a_pert.sum()
                if s_pert < 1e-15:
                    continue
                a_pert = a_pert / s_pert
                a_out, sc = nesterov_lp_optimize(
                    a_pert, t_sub,
                    active_k=35, patience=40, m_perturb=0.02,
                    verbose=False, rng=_global_rng
                )
                try_update(a_out, f"rr{ri}")

    # Phase 4: Final LP + verification
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_best = len(best_seq)
        t_final = min(remaining * 0.7, 120.0)
        print(f"\n=== Phase 4: Final optimization (budget={t_final:.0f}s) ===")
        a_out, sc = nesterov_lp_optimize(
            best_seq, t_final,
            active_k=50, patience=80, m_perturb=0.0,
            verbose=True, rng=_global_rng
        )
        try_update(a_out, "final_lp")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            a_out, sc = cd_polish(best_seq, remaining * 0.5, window=10, verbose=False, rng=_global_rng)
            try_update(a_out, "final_cd")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
