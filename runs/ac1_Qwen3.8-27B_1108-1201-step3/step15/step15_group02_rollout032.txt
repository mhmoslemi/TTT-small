def lse_adam_optimize(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    t_adam = 0

    n_stages = 25
    temps = np.geomspace(100.0, 0.03, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(20, min(150, int(remaining * 8)))
        lr = 0.4 / np.sqrt(n)

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            t_adam += 1

            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / (exp_c.sum() + 1e-30)

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)

            ca = irfft(np.conj(A) * W, fft_size)
            g1 = ca[:n]

            aw = irfft(A * W, fft_size)
            g2 = aw[:n]

            grad = 2.0 * (g1 + g2)

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_adam)
            v_hat = v / (1.0 - beta2 ** t_adam)

            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new

            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose and (si % 5 == 0 or si == n_stages - 1):
            print(f"    LSE-Adam stage {si} T={T_val:.4f}: best={best_score:.8f}")

    return best_a, best_score

    else:

                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def subgradient_nesterov_optimize(a0, time_limit, verbose=True):
    """Exact subgradient method with heavy-ball momentum for max(conv(a,a))."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    # Heavy ball momentum
    v = np.zeros(n)
    mu = 0.9  # momentum
    t_iter = 0
    lr_base = 0.1
    target_M = norm * norm / (2 * n)  # theoretical minimum ~ 1

    while time.monotonic() - t0 < time_limit:
        t_iter += 1

        conv = fast_conv_self(a)
        M = conv.max()
        if M < 1e-12:
            break

        # Find active set (entries close to max)
        tol = max(1e-6 * M, (M - norm * norm / size) * 0.3)
        active = np.where(conv > M - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv)])

        # Compute subgradient: average of gradients of active constraints
        # dc_k/da_j = 2 * a_{k-j} if 0 <= k-j < n, else 0
        grad = np.zeros(n)
        for k in active:
            # grad contribution: for each j, add 2*a_{k-j} if valid
            for j in range(n):
                ki = k - j
                if 0 <= ki < n:
                    grad[j] += 2.0 * a[ki]
        grad /= len(active)

        # Adaptive step size (Polyak-like)
        gap = M - 1.0  # target is ~1
        if gap < 1e-6:
            break
        g_norm_sq = np.dot(grad, grad)
        if g_norm_sq < 1e-30:
            break
        lr = min(lr_base / np.sqrt(t_iter + 1), gap / (g_norm_sq + 1e-15))
        lr = max(lr, 1e-8)

        # Heavy ball update
        v = mu * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        a = a_new
        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose and t_iter % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad t={t_iter} ({elapsed:.0f}s): best={best_score:.8f}")

        # Decay momentum slightly over time
        if t_iter % 500 == 0:
            mu = max(0.5, mu * 0.95)

    return best_a, best_score

def spectral_refine(a0, time_limit, verbose=True):
    """Iterative spectral domain refinement to flatten autocorrelation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()

    n_rounds = 500
    for it in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute circular autocorrelation and its spectrum
        A = rfft(a)
        power = np.abs(A) ** 2

        # The DC component (k=0) gives the total power
        # We want to reduce non-DC power to flatten the autocorrelation
        # Use a "target spectrum" that's mostly DC
        target_power = np.zeros(len(A))
        target_power[0] = power[0]

        # Exponential decay of non-DC components
        decay = 0.02 + 0.05 * (it / max(n_rounds - 1, 1))
        A_new = A.copy()
        for k in range(1, len(A)):
            A_new[k] = A[k] * np.sqrt(1 - decay)

        # Reconstruct with the original phase but reduced magnitude for non-DC
        A_mag = np.abs(A_new)
        A_phase = np.angle(A)
        A_combined = A_mag * np.exp(1j * A_phase)

        a_new = irfft(A_combined, n)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new * norm / s_new

        sc = fast_conv_self(a_new).max()
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        # Accept even non-improving moves early on (annealing)
        elif it < n_rounds * 0.3 and sc < best_score * 1.001:
            if np.random.random() < 0.3:
                a = a_new

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0
    temperature = 0.01

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            peak_lags = np.where(conv >= cur_max * 0.80)[0]

            # Build candidate set: random + peak-lag related
            j_candidates = set()
            for _ in range(12):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:15]:
                j1 = int((kl - i + n) % n)
                j2 = int((kl + i) % n)
                if j1 < n:
                    j_candidates.add(j1)
                if j2 < n:
                    j_candidates.add(j2)
            j_candidates.discard(i)

            best_delta = 0.0
            best_v = cur_max
            best_j = -1

            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]

                # Compute gradient of convolution when moving mass from i to j
                g = np.zeros(size)

                # Contribution of a[j]: a[0..n-1] shifted by j (a_j * a_{k-j})
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]

                # Contribution of a[j] from the other side: a_{j-k} * a_k shifted
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]

                # Remove contribution of a[i]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]

                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                d_max = ai * 0.9
                n_samples = 30
                deltas = np.linspace(0.005, d_max, n_samples)

                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_delta = d
                        best_j = j

            # Accept move (with occasional annealing)
            accept = False
            if best_j >= 0 and best_delta > 0:
                if best_v < best_score - 1e-10:
                    accept = True
                elif temperature > 0.001 and rng.random() < np.exp(-(best_v - best_score) / temperature):
                    accept = True

            if accept:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-10:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        temperature *= 0.98

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                n_perturb = max(3, n // 100)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.2 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                if no_improve >= 12:
                    break
        else:
            no_improve = 0

        if verbose and total_improvements > 0 and total_improvements % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD improv={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def sparsify(a, threshold_frac=0.002):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return a
    a_norm = a / s
    threshold = threshold_frac * np.max(a_norm)
    mask = a_norm > threshold
    if np.sum(mask) < n // 5:
        mask = a_norm > threshold * 0.1
    a_s = np.where(mask, a, 0.0)
    s_new = a_s.sum()
    if s_new < 1e-15:
        return a
    return a_s / s_new

def symmetrize(a):
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    s = a_sym.sum()
    if s < 1e-15:
        return a
    return a_sym / s * (a.sum() / s)

def structured_init(n, rng):
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)

    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        f1 = rng.integers(1, 5)
        f2 = rng.integers(5, 15)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * f1 * x) + 0.2 * np.cos(2 * np.pi * f2 * x)
    elif choice == 3:
        c1 = rng.uniform(0.2, 0.5)
        c2 = rng.uniform(0.5, 0.8)
        a = np.exp(-40.0 * (x - c1) ** 2) + np.exp(-40.0 * (x - c2) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 4
    elif choice == 5:
        n_peaks = rng.integers(3, 8)
        centers = rng.uniform(0.05, 0.95, n_peaks)
        a = np.zeros(n)
        for c in centers:
            width = rng.uniform(0.03, 0.12)
            a += np.exp(-(x - c) ** 2 / (2 * width ** 2))
    elif choice == 6:
        a = 0.5 + 0.4 * np.sin(2 * np.pi * 2 * x) ** 2
    elif choice == 7:
        a = np.exp(-8.0 * x * (1 - x))
    elif choice == 8:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * x) + 0.2 * np.cos(2 * np.pi * 7 * x)
    else:
        a = 0.6 + 0.4 * np.cos(2 * np.pi * x) ** 2

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a = a + 0.003 * np.abs(rng.standard_normal(n))
    a = a / a.sum()
    return a

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1650.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            old_best = best_score
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (was {old_best:.6f}, n={len(a)}, t={elapsed:.0f}s)")

    # Phase 1: LSE-Adam (fast initial progress)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.15, 240.0)
    print(f"\n=== Phase 1: LSE-Adam n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_out, _ = lse_adam_optimize(best_seq, t1, verbose=True)
        check_and_update(a_out, f"lse1_n{len(a_out)}")

    # Phase 2: LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.28, 450.0)
    print(f"\n=== Phase 2: LP Nesterov n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, solver, verbose=True)
        check_and_update(a_out, f"lp1_n{len(a_out)}")

    # Phase 3: Subgradient Nesterov (exact gradients, different landscape exploration)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 3: Subgradient Nesterov n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = subgradient_nesterov_optimize(best_seq, t3, verbose=True)
        check_and_update(a_out, f"subgrad_n{len(a_out)}")

    # Phase 4: LSE-Adam round 2 from improved point
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 4: LSE-Adam round 2 n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = lse_adam_optimize(best_seq, t4, verbose=True)
        check_and_update(a_out, f"lse2_n{len(a_out)}")

    # Phase 5: LP Nesterov round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.18, 280.0)
    print(f"\n=== Phase 5: LP Nesterov round 2 n={len(best_seq)} (budget={t5:.0f}s) ===")

    # Phase 6: Spectral refinement (new approach)

    # Phase 8: Symmetrization + sparsification

    # Phase 9: Multi-length exploration

        n_current = len(best_seq)

                break
            if n_try < 400 or n_try > 4000:

            t_try = min(remaining * 0.30, 50.0)
            print(f"  Trying n={n_try} (budget={t_try:.0f}s)")
            init_try = interpolate_seq(best_seq, n_try)
            a_out, _ = lse_adam_optimize(init_try, min(t_try, 25.0), verbose=False)
            check_and_update(a_out, f"ml_lse_n{n_try}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20:
                a_out, _ = pair_cd_optimize(a_out, min(remaining * 0.5, 20.0), rng, verbose=False)
                check_and_update(a_out, f"ml_pcd_n{n_try}")

    # Phase 10: Fresh random starts
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        print(f"\n=== Phase 10: Fresh starts (remaining={remaining:.0f}s) ===")

            if n_fresh < 400:
                n_fresh = 800
            if n_fresh > 4000:
                n_fresh = 2400
            a_fresh = structured_init(n_fresh, rng)
            t_fresh = min(remaining * 0.25, 25.0)
            a_out, _ = lse_adam_optimize(a_fresh, t_fresh, verbose=False)
            check_and_update(a_out, f"fresh_{attempt}_n{n_fresh}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 12:
                a_out, _ = pair_cd_optimize(a_out, min(remaining * 0.4, 12.0), rng, verbose=False)
                check_and_update(a_out, f"fresh_pcd_{attempt}_n{n_fresh}")

    # Final: one more short LP + PCD on best
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Final polish (remaining={remaining:.0f}s) ===")
        t_final_lp = min(remaining * 0.4, 60.0)
        if t_final_lp > 10:
            a_out, _ = nesterov_lp_optimize(best_seq, t_final_lp, solver, verbose=False)

        remaining = time_budget - (time.monotonic() - t_global)
        t_final_pcd = min(remaining * 0.5, 40.0)
        if t_final_pcd > 5:
            a_out, _ = pair_cd_optimize(best_seq, t_final_pcd, rng, verbose=False)
            check_and_update(a_out, "final_pcd")

    # Ultimate final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
