def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. (a*b)_k <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    
    # Build convolution matrix T where (T*b)_k = (a*b)_k
    # T is size x n, T[k, j] = a[k-j] if 0 <= k-j < n
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if momentum_t < 2.0:
            a_new = g.copy()
        else:
            momentum = (momentum_t - 1.0) / (momentum_t + 2.0)
            a_new = np.maximum(y_new + momentum * (y_new - y), 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        momentum_t += 1.0
        
        score_y = fast_conv(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            patience = 0
        else:
            patience += 1
        
        score_a = fast_conv(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            patience = 0
        
        if patience > 100 and n_iters % 50 == 0:
            y = best_a.copy()
            a = best_a.copy()
            momentum_t = 1.0
            patience = 0
        
        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score

def subgradient_optimize(a0, time_limit, top_k=50, verbose=True):
    """Subgradient method with averaging and adaptive step size."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(a0, 1e-15)
    a = a * norm / a.sum()
    
    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    
    # Running average for Polyak step size
    a_avg = a.copy()
    avg_count = 0
    eta = 1.0 / np.sqrt(n)
    rng = np.random.default_rng(42)
    
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_conv(a)
        max_val = conv.max()
        
        # Active set: top k lags
        k_actual = min(top_k, size)
        active_idx = np.argsort(conv, kind='stable')[-k_actual:]
        
        # Subgradient
        grad = np.zeros(n)
        for k in active_idx:
            k_int = int(k)
            # d(conv_k)/d(a_j) = 2 * a_{k-j} if 0 <= k-j < n
            lo = max(0, k_int - n + 1)
            hi = min(k_int, n - 1)
            if hi >= lo:
                # For j in [lo, hi]: k-j is in [0, n-1]
                # grad[j] += 2 * a[k-j]
                indices = np.arange(lo, hi + 1)
                grad[lo:hi+1] += 2.0 * a[k_int - indices]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm
        
        # Mirror descent (exponential weights)
        log_a = np.log(a + 1e-30)
        
        # Try a few step sizes
        best_a_try = a.copy()
        best_conv_try = max_val
        for factor in [0.5, 1.0, 1.5, 2.0]:
            eta_try = eta * factor / np.sqrt(it + 1)
            log_a_try = log_a - eta_try * grad
            a_try = np.exp(log_a_try - log_a_try.max())
            a_try = a_try * norm / a_try.sum()
            conv_try = fast_conv(a_try).max()
            if conv_try < best_conv_try:
                best_conv_try = conv_try
                best_a_try = a_try
                eta = eta_try * np.sqrt(it + 1)  # Store base eta
        
        a = best_a_try
        
        # Running average
        a_avg = (avg_count * a_avg + a) / (avg_count + 1)
        a_avg = a_avg * norm / a_avg.sum()
        avg_count += 1
        
        score = fast_conv(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        score_avg = fast_conv(a_avg).max()
        if score_avg < best_score:
            best_score = score_avg
            best_a = a_avg.copy()
        
        if verbose and (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad iter={it+1} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score
