import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(8, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:

    _solver = cp.ECOS
    print("  Using solver (fallback): ECOS")

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_T(a):
    """Build the convolution matrix T such that (T @ b) = conv(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        T[i:i + n, i] = a
    return T


def lp_solve_active(a, M, solver, active_idx):
    """Solve the LP with only active constraints."""
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T[active_idx, :] @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)


def lp_solve_full(a, M, solver):
    """Solve the LP with all constraints."""
    n = len(a)
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)


def line_search_step(y, g, A, B, C, n_grid=30):
    """
    Find best step s in [0,1] minimizing max_k [((1-s)^2*A_k + 2s(1-s)*B_k + s^2*C_k)]
    where A=conv(y,y), B=conv(y,g), C=conv(g,g).
    """
    best_s = 0.0
    best_val = np.inf
    for s in np.linspace(0.0, 1.0, n_grid):
        val = ((1.0 - s) ** 2) * A + 2.0 * s * (1.0 - s) * B + (s * s) * C
        mx = val.max()
        if mx < best_val:
            best_val = mx
            best_s = s
    return best_s


def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 15
    temps = np.geomspace(50.0, 0.1, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        n_steps = max(50, min(500, int(remaining * 8)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            step = norm / (T_val * np.sqrt(n)) * (1.0 / (1.0 + 0.03 * it))

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best={best_score:.8f}")

    return best_a, best_score


def nesterov_fw_optimize(a0, time_limit, solver, verbose=True):
    """
    Nesterov-accelerated Frank-Wolfe with line search and adaptive active set.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    x = a.copy()       # actual iterate
    x_prev = a.copy()  # previous iterate
    y = a.copy()       # Nesterov extrapolated point

    # Track active constraints
    active_history = set()
    full_count = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation (for x)
        if t > 1:
            beta = (t - 1.0) / (t + 2.0)
            y = x + beta * (x - x_prev)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = x.copy()
            else:
                y = y * norm / s_y
        else:
            y = x.copy()

        # Compute M and autocorrelation at y
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Decide active set
        use_full = (t <= 30) or (t % 50 == 0) or (len(active_history) < 80)

        if use_full:
            b = lp_solve_full(y, M, solver)
            full_count += 1
        else:
            # Use active set: constraints near max + recent history
            threshold = M * 0.95
            near_active = set(np.where(conv_y >= threshold)[0].tolist())
            active_idx = sorted(near_active | active_history)
            if len(active_idx) > 600:
                # Keep only the top ones
                conv_vals = conv_y[active_idx]
                order = np.argsort(conv_vals)[::-1][:600]
                active_idx = sorted(active_idx[i] for i in order)

            b = lp_solve_active(y, M, solver, np.array(active_idx))
            if b is None:
                b = lp_solve_full(y, M, solver)
                full_count += 1

        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Update active history based on binding constraints
        cross_conv = fast_conv_cross(y, g)
        binding = set(np.where(cross_conv >= M * 0.9999)[0].tolist())
        active_history = binding | (active_history & set(np.where(cross_conv >= M * 0.95)[0].tolist()))
        # Cap history size
        if len(active_history) > 400:
            # Keep only the most binding
            hist_list = list(active_history)
            conv_vals = conv_y[hist_list]
            order = np.argsort(conv_vals)[::-1][:400]
            active_history = set(hist_list[i] for i in order)

        # Line search for best step
        A = conv_y
        B = fast_conv_cross(y, g)
        C = fast_conv_self(g)
        s_best = line_search_step(y, g, A, B, C, n_grid=25)

        # Apply step
        x_new = (1.0 - s_best) * y + s_best * g
        x_new = np.maximum(x_new, 0.0)
        s_new = x_new.sum()
        if s_new < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new * norm / s_new

        x_prev = x
        x = x_new

        # Track best
        sc = fast_conv_self(x).max()
        if sc < best_score:
            best_score = sc
            best_a = x.copy()

        # Also check y
        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M:.6f}, s={s_best:.3f}")

    return best_a, best_score


def peak_shaving_cd(a0, time_limit, rng, verbose=True):
    """
    Coordinate descent that specifically targets the top peaks in autocorrelation.
    Uses structured candidate selection and finer delta search.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv_self(a).max()
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_iter = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()

        # Find top peaks
        peak_thresh = cur_max * 0.96
        peak_lags = np.where(conv >= peak_thresh)[0]

        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            total_iter += 1
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-8:
                continue

            # Build candidate set for j
            j_candidates = set()
            # Random candidates
            for _ in range(8):
                j_candidates.add(int(rng.integers(n)))
            # Structured: based on peak lags
            for kl in peak_lags[:6]:
                # Contribution of a[i] to conv[kl] involves a[kl-i] and a[i-kl]
                if 0 <= kl - i < n:
                    j_candidates.add(kl - i)
                if 0 <= i - kl < n:
                    j_candidates.add(i - kl)
            # Neighbors
            for d in [-3, -2, -1, 1, 2, 3]:
                j = (i + d) % n
                j_candidates.add(j)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in list(j_candidates)[:25]:
                if j == i:
                    continue
                aj = a[j]

                # Compute gain vector for transferring delta from i to j
                # conv_new[k] = conv[k] + delta * (contribution from adding to j - contribution from removing from i)
                g = np.zeros(size)
                # Add delta at position j: adds delta * a shifted by j (forward) and a reversed shifted by j (backward)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                # Remove delta from position i
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # Search over delta values
                n_samples = 30
                deltas = np.geomspace(ai * 0.01, ai * 0.99, n_samples)
                # Also try small deltas
                if ai > 0:
                    deltas = np.concatenate([[ai * 0.001], deltas, [ai * 0.999]])
                for d in deltas:
                    if d < 1e-12:
                        continue
                    new_conv = conv + d * g
                    # Correct the diagonal terms (2i and 2j)
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_val - 1e-12:
                        best_val = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-10:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break
            # Perturb to escape local minimum
            if no_improve >= 2:
                n_perturb = max(10, n // 25)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.12 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and total_iter % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakCD iter={total_iter} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # Finer delta search
                n_samples = 25
                deltas = np.linspace(0.02 * ai, 0.98 * ai, n_samples)
                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_val - 1e-12:
                        best_val = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-10:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True

        if not pass_improved:
            no_pass_improve += 1
            if no_pass_improve >= 3:
                break
        else:
            no_pass_improve = 0

        if verbose and total_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SysCD pass={total_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.2
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.1 * np.cos(2 * np.pi * 7 * x)
    elif choice == 3:
        a = np.exp(-5.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-5.0 * (x - 0.75) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.8
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.5
    elif choice == 6:
        a = np.exp(-3.0 * x) + 0.5 + 0.15 * np.sin(4 * np.pi * x)
    elif choice == 7:
        a = 0.5 + 0.25 * np.sin(5 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        a = np.sinc(2.5 * (x - 0.5)) ** 2 + 0.15
    elif choice == 9:
        a = np.exp(-3.0 * (x - 0.4) ** 2 / 0.05) + 0.6 * np.exp(-3.0 * (x - 0.6) ** 2 / 0.03)
    elif choice == 10:
        a = 0.4 + 0.4 * np.cos(2 * np.pi * np.sqrt(x)) + 0.2 * np.sin(6 * np.pi * x)
    else:
        a = (1 + np.cos(2 * np.pi * x)) / 2 * (1 + 0.3 * np.cos(4 * np.pi * x))
    a = np.maximum(a, 0.0)
    a = a + 0.005 * rng.random(n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        if len(a_list) == 0:
            return float('inf')
        for val in a_list:
            if val < 0 or not np.isfinite(val):
                return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LSE gradient descent warmup
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 100.0)
    print(f"\n=== Phase 1: LSE GD n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 15:
        a_work = best_seq.copy()
        a_out, _ = lse_gradient_descent(a_work, t1, verbose=True)
        try_update(a_out, "lse1")

    # Phase 2: Nesterov FW with line search (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq)
    t2 = min(remaining * 0.40, 600.0)
    print(f"\n=== Phase 2: Nesterov FW n={n2} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_fw_optimize(best_seq, t2, solver, verbose=True)
        try_update(a_out, "fw1")

    # Phase 3: Peak shaving coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = len(best_seq)
    t3 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 3: Peak Shaving CD n={n3} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_out, _ = peak_shaving_cd(best_seq, t3, rng, verbose=True)
        try_update(a_out, "peak_cd1")

    # Phase 4: Systematic pair CD
    remaining = time_budget - (time.monotonic() - t_global)
    n4 = len(best_seq)
    t4 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 4: Systematic CD n={n4} (budget={t4:.0f}s) ===")
    if t4 > 15:
        a_out, _ = systematic_pair_cd(best_seq, t4, verbose=True)
        try_update(a_out, "sys_cd1")

    # Phase 5: Second Nesterov FW round (re-engage LP)
    remaining = time_budget - (time.monotonic() - t_global)
    n5 = len(best_seq)
    t5 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 5: Nesterov FW round 2 n={n5} (budget={t5:.0f}s) ===")
    if t5 > 30:
        a_out, _ = nesterov_fw_optimize(best_seq, t5, solver, verbose=True)
        try_update(a_out, "fw2")

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Multi-length exploration (budget={remaining:.0f}s) ===")
    if remaining > 100:
        # Try n + 200
        n_long = n_base + 200
        t6a = min(remaining * 0.30, 150.0)
        print(f"  Trying n={n_long} (budget={t6a:.0f}s)")
        if t6a > 30:
            init_long = interpolate_seq(best_seq, n_long)
            a_out, _ = nesterov_fw_optimize(init_long, t6a, solver, verbose=False)
            try_update(a_out, f"fw_n{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        # Try n - 200
        n_short = max(800, n_base - 200)
        t6b = min(remaining * 0.25, 120.0)
        print(f"  Trying n={n_short} (budget={t6b:.0f}s)")
        if t6b > 20:
            init_short = interpolate_seq(best_seq, n_short)
            a_out, _ = nesterov_fw_optimize(init_short, t6b, solver, verbose=False)
            try_update(a_out, f"fw_n{n_short}")

    # Phase 7: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 7: Final polish n={n_final} (budget={remaining:.0f}s) ===")
    if remaining > 30:
        # Quick LSE
        t7a = min(remaining * 0.25, 40.0)
        if t7a > 10:
            a_out, _ = lse_gradient_descent(best_seq, t7a, verbose=False)
            try_update(a_out, "lse_final")

        remaining = time_budget - (time.monotonic() - t_global)
        # Peak shaving
        t7b = min(remaining * 0.40, 80.0)
        if t7b > 10:
            a_out, _ = peak_shaving_cd(best_seq, t7b, rng, verbose=True)
            try_update(a_out, "cd_final")

        remaining = time_budget - (time.monotonic() - t_global)
        # Final FW
        t7c = min(remaining * 0.50, 60.0)
        if t7c > 15:
            a_out, _ = nesterov_fw_optimize(best_seq, t7c, solver, verbose=False)
            try_update(a_out, "fw_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
