def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver_cached = None

def get_lp_solver():
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solved = False
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            solved = True
    except Exception:
        pass
    if not solved:
        for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue
    if not solved:
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=500000):
    """Nesterov dual averaging with LP best responses."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score
    rng = np.random.default_rng(42)
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        
        # Nesterov averaging
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / ((it + 1) + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        
        if (it + 1) % 2000 == 0:
            if best_score >= prev_best - 1e-10:
                stall_count += 1
            else:
                stall_count = 0
            prev_best = best_score
            
            if stall_count > 5 and (it + 1) % 10000 == 0:
                # Add noise to escape potential stagnation
                noise = rng.standard_normal(n) * 0.01
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0
            
            if (it + 1) % 20000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def ftrl_refine(a0, time_limit, solver=None, n_max_iters=500000):
    """FTRL with quadratic regularizer."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    z = np.zeros(n)
    eta = 1.0
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            continue
        
        z += b_new
        
        # FTRL: a_new = argmin_{x in simplex} <z, x> + (eta/2)||x - a0||^2
        # For simplicity, use projection of -z/eta onto simplex
        v = -z / eta
        # Project onto simplex
        u = np.sort(v)[::-1]
        cv = np.cumsum(u)
        rho = np.nonzero(u * np.arange(1, n + 1) > (cv - 1))[0]
        if len(rho) == 0:
            a_new = a.copy()
        else:
            tau = (cv[rho[-1]] - 1) / (rho[-1] + 1.0)
            a_new = np.maximum(v - tau, 0)
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        a = a_new
        
        # Adaptive: if stalling, increase eta (stronger regularization)
        if stall_count % 5000 == 0 and stall_count >= 5000:
            eta *= 2.0
            z /= 2.0  # keep z/eta invariant
            stall_count = 0
        
        if (it + 1) % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FTRL iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} eta={eta:.1f}")
    
    return best_a, best_score

        # = 2*n*(max(b) + T*log(sum(exp((b-max(b))/T))))

        # But we need to be careful: the gradient of T*logsumexp(b/T) w.r.t. a
        # is (T/T) * (1/sum(exp)) * sum(exp * db/da) = weighted average of db/da

            # db[k]/da[m] = a[k-m] + a[m-k] if both in range...

            # b[k] = sum_j a_j * a_{k-j}
            # db[k]/da[m] = a_{k-m} (from the j=m term) + ...

            # So db[k]/da[m] = a_{k-m} if 0 <= k-m < n, else 0.

            S = np.dot(grad_a, a)

def subgradient_refine(a0, time_limit, n_steps=100000):
    """Exact subgradient with Polyak-like step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.05
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Find active set (within 0.1% of max)
        threshold = max_b * 0.9995
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # Subgradient: for each active k, grad_k[j] = a[k-j] (if valid)
        # Average over active set
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        
        # Projected step
        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st
        
        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 20:
            lr *= 0.85
            no_improve = 0
        lr = max(1e-8, lr)
        
        if step % 20000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad ({elapsed:.0f}s): score={best_score:.8f} lr={lr:.2e}")
    
    return best_a, best_score


def peak_coordinate_descent(a0, time_limit, n_steps=50000):
    """Coordinate descent targeting peak contributors."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Find all peaks within 1% of max
        threshold = max_b * 0.995
        peak_positions = np.where(b >= threshold)[0]
        
        improved = False
        for pk in peak_positions[:3]:  # Try top 3 peaks
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            
            # Top 5 contributors
            top_m = min(5, len(idx))
            top_pos = idx[np.argsort(contribs)[::-1][:top_m]]
            
            for pos in top_pos:
                if a[pos] < 1e-12:
                    continue
                for factor in [0.999, 0.998, 0.995, 0.99, 0.98]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-16:
                        break
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others_sum = a_trial[mask].sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] += diff * a_trial[mask] / others_sum
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 500:
                break
        else:
            stall = 0
        
        if step % 10000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakCD ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    else:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

- FTRL (Follow The Regularized Leader) with simplex projection
- LogSumExp L-BFGS with multi-temperature annealing

- Peak-directed coordinate descent for final polishing
Key insight: exploit the zero-sum game structure where both players share

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver_cached = None

def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solved = False
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            solved = True
    except Exception:
        pass
    if not solved:
        for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue
    if not solved:
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=1000000):
    """Nesterov dual averaging with LP best responses for game solving."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score
    rng = np.random.default_rng(42)
    iters_done = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        iters_done += 1

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / ((it + 1) + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        improved = False
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            improved = True
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            improved = True

        if improved:
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > 20000 and (it + 1) % 5000 == 0:
            noise_scale = 0.015 / np.sqrt(stall_count / 20000)
            noise = rng.standard_normal(n) * noise_scale
            y = y + noise
            y = np.maximum(y, 0)
            ys = y.sum()
            if ys > 1e-15:
                y = y / ys
            stall_count = 0

        if (it + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def ftrl_refine(a0, time_limit, solver=None, n_max_iters=500000):
    """FTRL with quadratic regularizer for the zero-sum game."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    z = np.zeros(n)
    eta = 0.8
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            continue

        z += b_new

        # FTRL: project(-z / eta) onto simplex
        v = -z / eta
        u = np.sort(v)[::-1]
        cv = np.cumsum(u)
        rho = np.nonzero(u * np.arange(1, n + 1) > (cv - 1))[0]
        if len(rho) == 0:
            a_new = a.copy()
        else:
            tau = (cv[rho[-1]] - 1) / (rho[-1] + 1.0)
            a_new = np.maximum(v - tau, 0)
            sn = a_new.sum()
            if sn < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / sn

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        a = a_new

        if stall_count % 5000 == 0 and stall_count >= 5000:
            eta *= 1.8
            z = z * (0.8 / eta) * 0.8
            stall_count = 0

        if (it + 1) % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FTRL iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} eta={eta:.2f}")

    return best_a, best_score


def _lse_obj_grad(z, n, T, time_limit, t0):
    """Compute objective and gradient of LogSumExp-smoothed function in log-domain."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    b = fast_convolve(a)
    b_shift = b - b.max()
    exp_b = np.exp(b_shift / T)
    w = exp_b / exp_b.sum()

    # Gradient: 4n * cross(w, a)[0:n]
    # cross(w, a)[m] = sum_l w[l+m] * a[l] = conv(w, a_rev)[m + n - 1]
    a_rev = a[::-1]
    len_w = len(w)
    len_total = len_w + n - 1
    fft_size = 1 << (len_total - 1).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv_result = irfft(W * AR, fft_size)[:len_total]
    c = conv_result[n - 1:n - 1 + n]

    grad_a = 4.0 * n * c
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    obj = 2.0 * n * (b.max() + T * np.log(exp_b.sum()))
    return obj, grad_z


def logsumexp_lbgfs(a0, time_limit, T_stages=None, maxiter=3000):
    """L-BFGS on LogSumExp-smoothed objective in log-domain."""
    n = len(a0)
    if T_stages is None:
        T_stages = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]

    a0_safe = np.maximum(a0, 1e-30)
    a0_norm = a0_safe / a0_safe.sum()
    z0 = np.log(a0_norm)

    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            return _lse_obj_grad(z, n, T, time_limit, t0)[0]

        def jac(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            return _lse_obj_grad(z, n, T, time_limit, t0)[1]

        try:
            res = minimize(
                fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxfun': maxiter * 200, 'disp': False}
            )
            z0 = res.x
            a_res = np.exp(z0 - z0.max())
            a_res = a_res / a_res.sum()
            score = 2.0 * n * fast_convolve(a_res).max()
            if score < best_score:
                best_score = score
                best_a = a_res.copy()
                print(f"    LSE T={T}: score={best_score:.8f}")
        except Exception as e:
            print(f"    LSE error T={T}: {e}")

    return best_a, best_score


def subgradient_refine(a0, time_limit, n_steps=200000):
    """Exact subgradient with adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.04
    no_improve = 0
    # Nesterov momentum
    y = a.copy()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(y)
        max_b = b.max()
        threshold = max_b * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        # Nesterov step
        momentum = (np.sqrt(1.0 + 4.0 * (step + 1) ** 2) - 1.0) / 2.0
        y_new = y + momentum * (y - a) - lr * g
        y_new = np.maximum(y_new, 0)
        sny = y_new.sum()
        if sny < 1e-15:
            y_new = a.copy()
        else:
            y_new = y_new / sny

        a = y_new
        y = a

        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 30:
            lr *= 0.88
            no_improve = 0
        lr = max(1e-8, lr)

        if step % 30000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad ({elapsed:.0f}s): score={best_score:.8f} lr={lr:.2e}")

    return best_a, best_score


def peak_coordinate_descent(a0, time_limit, n_steps=50000):
    """Targeted coordinate descent on peak-contributing elements."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.995
        peak_positions = np.where(b >= threshold)[0]
        if len(peak_positions) == 0:
            peak_positions = np.array([int(np.argmax(b))])

        improved = False
        for pk in peak_positions[:3]:
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            top_m = min(6, len(idx))
            top_pos = idx[np.argsort(contribs)[::-1][:top_m]]

            for pos in top_pos:
                if a[pos] < 1e-12:
                    continue
                for factor in [0.999, 0.998, 0.995, 0.99, 0.985, 0.98]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-16:
                        break
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others_sum = a_trial[mask].sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] += diff * a_trial[mask] / others_sum
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 300:
                break
        else:
            stall = 0

        if step % 10000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakCD ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 7)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    else:
        a = 1.0 + 0.2 * np.cos(2 * np.pi * 5 * x) + 0.1 * np.sin(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence using local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

        n_best,
        int(n_best * 1.1),
        int(n_best * 1.25),
        int(n_best * 1.5),
        int(n_best * 1.75),
        int(n_best * 2.0),
    ]))
    # Filter to reasonable range
    n_candidates = [n for n in n_candidates if 400 <= n <= 3500]

    # Allocate time per sweep based on remaining budget
    total_sweep_budget = min(rem * 0.20, 300.0)
    per_sweep = max(20.0, total_sweep_budget / len(n_candidates))

    for n_trial in n_candidates:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        if n_trial == n_best and rem < 60:
            continue
        t_alloc = min(per_sweep, rem * 0.3)
        if t_alloc < 15:
            continue
        print(f"  Sweeping n={n_trial}, budget={t_alloc:.0f}s")
        if abs(n_trial - n_best) < 10:
            init = best_seq.copy()
        else:
            init = interpolate_sequence(best_seq, n_trial)
        a_sweep, _ = nesterov_lp_refine(init, t_alloc, solver=solver, n_max_iters=200000)
        try_update(a_sweep, f"sweep_n{n_trial}")

    # === Phase 2: Extended Nesterov LP on best n ===
    print(f"\n=== Phase 2: Extended Nesterov LP (score={best_score:.8f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 450.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=2000000)
        try_update(a_lp, "nesterov_ext")

    # === Phase 3: FTRL ===
    print(f"\n=== Phase 3: FTRL (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 30:
        a_ftrl, _ = ftrl_refine(best_seq, t_alloc, solver=solver, n_max_iters=300000)
        try_update(a_ftrl, "ftrl")

    # === Phase 4: LogSumExp L-BFGS ===
    print(f"\n=== Phase 4: L-BFGS LogSumExp (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 30:
        T_stages = [15.0, 8.0, 4.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]
        a_lb = logsumexp_lbgfs(best_seq, t_alloc, T_stages=T_stages, maxiter=4000)
        try_update(a_lb, "lbgfs")

    # === Phase 5: Subgradient ===
    print(f"\n=== Phase 5: Subgradient (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 100.0)
    if t_alloc > 20:
        a_sg, _ = subgradient_refine(best_seq, t_alloc, n_steps=200000)
        try_update(a_sg, "subgrad")

    # === Phase 6: Peak coordinate descent ===
    print(f"\n=== Phase 6: Peak CD (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 80.0)
    if t_alloc > 15:
        a_cd, _ = peak_coordinate_descent(best_seq, t_alloc, n_steps=50000)
        try_update(a_cd, "peak_cd")

    # === Phase 7: Final Polish - another round of LP + L-BFGS ===
    print(f"\n=== Phase 7: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t_fin = min(rem * 0.5, 120.0)
        a_fin, _ = nesterov_lp_refine(best_seq, t_fin, solver=solver, n_max_iters=500000)
        try_update(a_fin, "final_lp")

    rem = deadline - time.monotonic()
    if rem > 25:
        T_final = [2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
        a_fin2 = logsumexp_lbgfs(best_seq, min(rem * 0.7, 50.0),
                                  T_stages=T_final, maxiter=3000)
        try_update(a_fin2, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 15:
        a_fin3, _ = peak_coordinate_descent(best_seq, min(rem * 0.6, 25.0), n_steps=20000)
        try_update(a_fin3, "final_cd")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
