```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based coordinate descent
with multiple restarts, adaptive n selection, and peak shaving polish.
Key: alternate between LP best responses for a and b (bilinear coordinate descent),
exploiting the fact that fixing one variable makes the problem linear.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    # T[k,j] = a[k-j] if 0 <= k-j < n, else 0
    # For efficiency, use the fact that T is block-Toeplitz
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (sum=1, nonneg), find b minimizing max conv(a,b)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable(nonneg=True)
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    if solver:
        prob.solve(solver=solver)
    else:
        prob.solve(solver=cp.ECOS, max_iter=10000)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        prob.solve(solver=cp.SCS, max_iters=10000)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lp_coordinate_descent(a0, time_limit, solver=None, max_rounds=50000):
    """
    Coordinate descent on the bilinear objective:
    alternate between LP best response for a (given b) and for b (given a).
    """
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0).astype(np.float64))
    b = a.copy()  # Initialize b = a
    
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for it in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Step 1: Given b, find optimal a
        a_new, M_a = solve_lp_best_response(b, solver)
        if a_new is None:
            continue
        a = a_new
        
        # Step 2: Given a, find optimal b
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            continue
        b = b_new
        
        # Check score
        score_a = score_normalized(a)
        score_b = score_normalized(b)
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_b < best_score:
            best_score = score_b
            best_a = b.copy()
        
        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP CD iter={it} ({elapsed:.0f}s): score={best_score:.6f} M_a={M_a:.6f} M_b={M_b:.6f}")
        
        # Convergence check
        if it > 10 and abs(score_a - best_score) < 1e-12 and abs(score_b - best_score) < 1e-12:
            # Try perturbation
            rng = np.random.default_rng(it)
            a = a + 0.01 * rng.standard_normal(n)
            a = project_simplex(a)
            b = b + 0.01 * rng.standard_normal(n)
            b = project_simplex(b)
    
    return best_a, best_score


def hybrid_optimize(a0, time_limit, solver=None):
    """
    Hybrid: LP coordinate descent with periodic gradient steps and perturbations.
    """
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0).astype(np.float64))
    b = a.copy()
    
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    stall_count = 0
    prev_best = best_score
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # LP best response for a given b
        a_new, M_a = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new
        
        # LP best response for b given a
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        
        score_a = score_normalized(a)
        score_b = score_normalized(b)
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_b < best_score:
            best_score = score_b
            best_a = b.copy()
            stall_count = 0
            
        if it % 200 == 0:
            if best_score < prev_best - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            
            if stall_count >= 5:
                # Perturb to escape local optimum
                rng = np.random.default_rng(42 + it)
                perturbation_scale = 0.005 + 0.01 * stall_count / 10
                a = a + perturbation_scale * rng.standard_normal(n)
                a = project_simplex(a)
                b = b + perturbation_scale * rng.standard_normal(n)
                b = project_simplex(b)
                stall_count = 0
                print(f"    Perturbed at iter {it}, scale={perturbation_scale:.4f}")
        
        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score


def peak_shave(a0, time_limit, n_steps=50000):
    """Reduce peaks in the convolution by adjusting coordinates."""
    n = len(a0)
    a = project_simplex(a0.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        
        # Find which pairs contribute most to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_idx = np.argsort(contribs)[::-1][:5]
        
        improved = False
        for ci in top_idx:
            i = idx[ci]
            j = peak_k - i
            if a[i] < 1e-14:
                continue
            for factor in [0.98, 0.95, 0.90, 0.85]:
                a_trial = a.copy()
                a_trial[i] *= factor
                # Redistribute to j (or split)
                if j != i and 0 <= j < n:
                    a_trial[j] += (a_trial[i] / factor - a_trial[i]) / 2
                a_trial = project_simplex(a_trial)
                score = score_normalized(a_trial)
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try reducing multiple peaks simultaneously
            top_peaks = np.argsort(b)[::-1][:3]
            a_trial = a.copy()
            for pk in top_peaks:
                if pk == 0 or pk == 2*n-2:
                    continue
                js2 = max(0, pk - n + 1)
                je2 = min(n - 1, pk)
                idx2 = np.arange(js2, je2 + 1)
                contribs2 = a[idx2] * a[pk - idx2]
                top_i = idx2[np.argmax(contribs2)]
                a_trial[top_i] *= 0.97
            a_trial = project_simplex(a_trial)
            score = score_normalized(a_trial)
            if score < best_score - 1e-15:
                a = a_trial
                best_score = score
                best_a = a.copy()
            else:
                break
    
    return best_a, best_score


def make_init(n, mode, rng):

    return a / a.sum()
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based coordinate descent
(bilinear alternation), sparse convolution matrices, adaptive perturbations,
multi-length exploration, and peak shaving. The key insight is that fixing
one variable makes the maximin problem a linear program, enabling fast
alternating optimization with global restarts.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix_sparse(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    rows = np.repeat(np.arange(n), n)
    cols = np.concatenate([np.arange(n) + i for i in range(n)])
    vals = np.tile(a, n)
    return csr_matrix((vals, (rows, cols)), shape=(size, n))

            x = cp.Variable(50, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            t0 = time.monotonic()
            prob.solve(solver=solver_name)
            dt = time.monotonic() - t0
            if prob.status in ('optimal', 'optimal_inaccurate'):
                results.append((dt, solver_name))
                print(f"  Solver {solver_name}: OK ({dt*1000:.1f}ms)")
        except Exception as e:
            print(f"  Solver {solver_name}: failed")
    if not results:
        print("  No solver found, defaulting to ECOS")
        return cp.ECOS
    results.sort()
    print(f"  Best solver: {results[0][1]}")
    return results[0][1]


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (sum=1, nonneg), find b minimizing max conv(a,b)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable(nonneg=True)
    T = build_conv_matrix_sparse(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        try:
            prob.solve(solver=cp.ECOS, max_iter=20000)
        except Exception:
            pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        try:
            prob.solve(solver=cp.SCS, max_iters=20000)
        except Exception:
            pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None
    b_val = b_val / s
    M_val = float(M_var.value)
    
    # Get dual variables if available (for perturbation guidance)
    duals = None
    try:
        for con in constraints:
            if con.dual_value is not None:
                duals = np.asarray(con.dual_value, dtype=np.float64)
                break
    except Exception:
        pass
    
    return b_val, M_val, duals


def lp_coordinate_descent(a0, time_limit, solver, max_iters=200000):
    """
    Alternating LP coordinate descent for the bilinear objective.
    Given a, solve LP for optimal b. Given b, solve LP for optimal a.
    With adaptive perturbations to escape local optima.
    """
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0).astype(np.float64))
    b = a.copy()
    
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    stall_count = 0
    prev_best = best_score
    perturbation_count = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Step 1: Given b, find optimal a (minimize max conv(b, a))
        a_new, M_a, _ = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new
        
        # Step 2: Given a, find optimal b (minimize max conv(a, b))
        b_new, M_b, _ = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        
        # Evaluate
        score_a = score_normalized(a)
        score_b = score_normalized(b)
        current_best = min(score_a, score_b)
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_b < best_score:
            best_score = score_b
            best_a = b.copy()
            stall_count = 0
        
        # Periodic convergence check and perturbation
        if it % 100 == 0:
            if best_score < prev_best - 1e-9:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            
            if stall_count >= 3:
                perturbation_count += 1
                scale = min(0.003 * perturbation_count, 0.05)
                rng = np.random.default_rng(1000 + it)
                
                # Try different perturbation strategies
                strategy = perturbation_count % 4
                if strategy == 0:
                    # Random noise
                    a = a + scale * rng.standard_normal(n)
                    a = project_simplex(a)
                    b = b + scale * rng.standard_normal(n)
                    b = project_simplex(b)
                elif strategy == 1:
                    # Peak-aware: reduce high-coord contributions
                    conv_b = fast_convolve(b)
                    peak_idx = np.argsort(conv_b)[::-1][:10]
                    for pk in peak_idx:
                        js = max(0, int(pk) - n + 1)
                        je = min(n - 1, int(pk))
                        if js <= je:
                            a[js:je+1] *= (1 - scale)
                    a = project_simplex(a)
                    conv_a = fast_convolve(a)
                    peak_idx2 = np.argsort(conv_a)[::-1][:10]
                    for pk in peak_idx2:
                        js = max(0, int(pk) - n + 1)
                        je = min(n - 1, int(pk))
                        if js <= je:
                            b[js:je+1] *= (1 - scale)
                    b = project_simplex(b)
                elif strategy == 2:
                    # Mix with random initialization
                    rng2 = np.random.default_rng(2000 + it)
                    a_rand = rng2.dirichlet(np.ones(n))
                    a = (1 - 2*scale) * a + 2*scale * a_rand
                    a = project_simplex(a)
                    b_rand = rng2.dirichlet(np.ones(n))
                    b = (1 - 2*scale) * b + 2*scale * b_rand
                    b = project_simplex(b)
                else:
                    # Swap-based perturbation
                    perm = rng.permutation(n)
                    a_swap = a[perm]
                    a = (1 - 2*scale) * a + 2*scale * a_swap
                    a = project_simplex(a)
                    b = (1 - scale) * b + scale * b[perm]
                    b = project_simplex(b)
                
                stall_count = 0
                if perturbation_count % 5 == 1:
                    elapsed = time.monotonic() - t0
                    print(f"    Perturbed #{perturbation_count} at iter {it} ({elapsed:.0f}s), scale={scale:.4f}, best={best_score:.6f}")
        
        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}, M_a={M_a:.6f}, M_b={M_b:.6f}")
    
    return best_a, best_score


def peak_shave(a0, time_limit, n_steps=100000):
    """Targeted peak reduction in the auto-convolution."""
    n = len(a0)
    a = project_simplex(a0.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        # Find top peaks
        top_peaks = np.argsort(b)[::-1][:5]
        peak_k = int(top_peaks[0])
        
        # Find top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        if len(idx) == 0:
            break
        contribs = a[idx] * a[peak_k - idx]
        top_idx_list = np.argsort(contribs)[::-1][:8]
        
        improved = False
        for ci in top_idx_list:
            i = idx[ci]
            j = peak_k - i
            if a[i] < 1e-14:
                continue
            for factor in [0.99, 0.97, 0.94, 0.90, 0.85]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = project_simplex(a_trial)
                score = score_normalized(a_trial)
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try pair reduction on the peak
            for ci1 in top_idx_list[:4]:
                i = idx[ci1]
                j = peak_k - i
                if i == j or i >= n or j >= n:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.97, 0.94]:
                    for f2 in [0.97, 0.94]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = project_simplex(a_trial)
                        score = score_normalized(a_trial)
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
        
        if not improved:
            # Try reducing the second and third peaks
            if len(top_peaks) >= 2:
                a_trial = a.copy()
                for pk in top_peaks[1:min(4, len(top_peaks))]:
                    pk = int(pk)
                    js2 = max(0, pk - n + 1)
                    je2 = min(n - 1, pk)
                    if js2 > je2:
                        continue
                    idx2 = np.arange(js2, je2 + 1)
                    contribs2 = a[idx2] * a[pk - idx2]
                    top_i2 = idx2[np.argmax(contribs2)]
                    if a[top_i2] > 1e-14:
                        a_trial[top_i2] *= 0.98
                a_trial = project_simplex(a_trial)
                score = score_normalized(a_trial)
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                else:
                    break
    
    return best_a, best_score


def make_init(n, mode, rng):
    """Create various initializations."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = 1.0 + 0.5 * np.sin(2 * np.pi * x)
    elif mode == 1:
        a = 1.0 + 0.7 * (x - 0.3)
    elif mode == 2:
        a = np.exp(-3.0 * (x - 0.25)**2) + 0.6 * np.exp(-3.0 * (x - 0.75)**2)
    elif mode == 3:
        a = 1.0 + 0.4 * np.sin(4 * np.pi * x) + 0.15 * np.cos(8 * np.pi * x)
    elif mode == 4:
        a = rng.random(n) + 0.05
    elif mode == 5:
        a = 1.0 / (1.0 + np.exp(-10 * (x - 0.5)))
        a = a + 0.1 * np.sin(6 * np.pi * x)
    elif mode == 6:
        # Triangular
        a = 1.0 - np.abs(2*x - 1)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. LP coordinate descent (primary method)
    2. Peak shaving
    3. Multi-length exploration
    4. Multi-start with diverse initializations
    5. Final polished LP + peak shave
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    print("=== Finding best LP solver ===")
    solver = get_lp_solver()
    
    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except (NameError, Exception) as e:
        parent = None
        print(f"No parent available: {e}")

    n_parent = len(best_seq) if best_seq is not None else 960

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}) ***")

    # === Phase 1: LP Coordinate Descent (main workhorse) ===
    print(f"\n=== Phase 1: LP CD (n={n_parent}, best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 540.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a1, _ = lp_coordinate_descent(best_seq, t_alloc, solver)
        try_update(a1, "lp_cd_1")

    # === Phase 2: Peak Shaving ===
    print(f"\n=== Phase 2: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 90.0)
    if t_alloc > 10:
        a2, _ = peak_shave(best_seq, t_alloc)
        try_update(a2, "peak_shave_1")

    # === Phase 3: LP CD round 2 ===
    print(f"\n=== Phase 3: LP CD Round 2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        a3, _ = lp_coordinate_descent(best_seq, t_alloc, solver)
        try_update(a3, "lp_cd_2")

    # === Phase 4: Peak Shaving round 2 ===
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 60.0)
    if t_alloc > 10:
        a3b, _ = peak_shave(best_seq, t_alloc)
        try_update(a3b, "peak_shave_2")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-Length (best={best_score:.6f}) ===")
    alt_ns = []
    for delta in [-100, -50, -20, 20, 50, 100, 200, 300, 500]:
        alt_n = n_parent + delta
        if 256 <= alt_n <= 4000:
            alt_ns.append(alt_n)
    for fixed_n in [384, 512, 640, 768, 1024, 1280, 1536, 2048, 2560]:
        if fixed_n not in alt_ns:
            alt_ns.append(fixed_n)
    alt_ns = sorted(set(alt_ns))
    
    n_tried = 0
    for alt_n in alt_ns:
        rem = deadline - time.monotonic()
        if rem < 50 or n_tried >= 5:
            break
        t_total = min(rem * 0.18, 240.0)
        if t_total < 20:
            break
        n_tried += 1
        print(f"  n={alt_n}, budget={t_total:.0f}s")
        
        # Interpolate to new length
        init_alt = interpolate_sequence(best_seq, alt_n)
        
        a_alt, _ = lp_coordinate_descent(init_alt, t_total * 0.75, solver)
        try_update(a_alt, f"lp_n{alt_n}")
        
        rem = deadline - time.monotonic()
        if rem > 15:
            a_alt2, _ = peak_shave(a_alt, min(rem * 0.3, 25.0))
            try_update(a_alt2, f"ps_n{alt_n}")

    # === Phase 6: Multi-start with diverse inits ===
    print(f"\n=== Phase 6: Multi-Start (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    n_try = len(best_seq)
    for mode in range(7):
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t_sub = min(rem * 0.25, 180.0)
        if t_sub < 20:
            break
        a_init = make_init(n_try, mode, rng)
        print(f"  mode={mode}, n={n_try}, budget={t_sub:.0f}s")
        a_sub, _ = lp_coordinate_descent(a_init, t_sub, solver)
        try_update(a_sub, f"start_m{mode}")

    # === Phase 7: Final LP polish on best ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.5, 180.0)
        a_fin, _ = lp_coordinate_descent(best_seq, t_fin, solver)
        try_update(a_fin, "final_lp")
        
        rem = deadline - time.monotonic()
        if rem > 15:
            a_ps, _ = peak_shave(best_seq, min(rem * 0.6, 45.0))
            try_update(a_ps, "final_ps")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\n=== FINAL: score={best_score:.6f} (n={len(best_seq)}) ===")
    else:
        best_seq = np.ones(960) / 960
        print(f"\nFALLBACK: uniform n=960, score=2.0")

    return [float(x) for x in best_seq]
```
