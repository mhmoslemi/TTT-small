def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact-line-search coordinate descent (primary method)
2. Multi-start with diverse initializations
3. LP fictitious play for global exploration
4. Simulated annealing restarts

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None):
    import cvxpy as cp
    if solver is None:
        solver = _get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK, cp.GLOP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def build_conv_matrix(n):

def exact_cd(p, time_limit, rng=None, verbose=False, label=""):
    """
    Coordinate descent with exact 1D line search (ternary search).
    Minimizes max(conv(p,p)) subject to p >= 0, sum(p) = 1.
    Works on the simplex; after each coord update, renormalizes.
    """
    n = len(p)
    p = np.maximum(np.asarray(p, dtype=np.float64), 0.0).copy()
    s = p.sum()
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / s

    conv = fast_convolve(p)
    M = float(conv.max())
    best_p = p.copy()
    best_M = M
    t0 = time.monotonic()
    sweeps = 0
    stall = 0

    if rng is None:
        rng = np.random.default_rng(0)

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved_this_sweep = False
        order = rng.permutation(n)

        for i in order:
            v = p[i]
            if v < 1e-15:
                continue

            k_start = i
            k_end = min(i + n, 2 * n - 1)
            length = k_end - k_start

            # Unaffected max
            if k_start > 0 and k_end < 2 * n - 1:
                unaff_max = max(float(conv[:k_start].max()), float(conv[k_end:].max()))
            elif k_start > 0:
                unaff_max = float(conv[:k_start].max())
            elif k_end < 2 * n - 1:
                unaff_max = float(conv[k_end:].max())
            else:
                unaff_max = -1e30

            # Search range for d: [-v, v] keeps p[i] in [0, 2v]
            lo = -v * 0.999
            hi = v * 0.999

            # Quick check: evaluate at a few points first
            # d = 0
            f0 = max(unaff_max, float(conv[k_start:k_end].max()))

            # Ternary search
            for _ in range(28):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0

                # Evaluate f at m1
                new_affected_1 = conv[k_start:k_end] + 2.0 * m1 * p[:length]
                if 2 * i < 2 * n - 1:
                    idx_2i = 2 * i - k_start
                    new_affected_1 = new_affected_1.copy()
                    new_affected_1[idx_2i] = conv[2 * i] + 2.0 * v * m1 + m1 * m1
                f1 = max(unaff_max, float(new_affected_1.max()))

                # Evaluate f at m2
                new_affected_2 = conv[k_start:k_end] + 2.0 * m2 * p[:length]
                if 2 * i < 2 * n - 1:
                    new_affected_2 = new_affected_2.copy()
                    new_affected_2[idx_2i] = conv[2 * i] + 2.0 * v * m2 + m2 * m2
                f2 = max(unaff_max, float(new_affected_2.max()))

                if f1 < f2:
                    hi = m2
                else:
                    lo = m1

            d_opt = (lo + hi) / 2.0

            # Evaluate f at d_opt
            new_affected = conv[k_start:k_end] + 2.0 * d_opt * p[:length]
            if 2 * i < 2 * n - 1:
                idx_2i = 2 * i - k_start
                new_affected = new_affected.copy()
                new_affected[idx_2i] = conv[2 * i] + 2.0 * v * d_opt + d_opt * d_opt
            f_opt = max(unaff_max, float(new_affected.max()))

            # Also check d=0
            if f0 < f_opt:
                d_opt = 0.0
                f_opt = f0

            # Apply if improvement
            if abs(d_opt) > 1e-14 and f_opt < M - 1e-15:
                # Update conv in-place
                conv[k_start:k_end] += 2.0 * d_opt * p[:length]
                if 2 * i < 2 * n - 1:
                    conv[2 * i] += d_opt * d_opt
                p[i] = v + d_opt
                M = f_opt
                improved_this_sweep = True

                if M < best_M - 1e-15:
                    best_M = M
                    best_p = p.copy()

        if not improved_this_sweep:
            stall += 1
            if stall > 3:
                break
        else:
            stall = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sweep={sweeps} ({elapsed:.1f}s): M={M:.12f} best={best_M:.12f}")

    return best_p, best_M, sweeps

def peak_targeted_perturb(a0, rng, strength=0.03):
    """Perturb by reducing values that contribute to convolution peaks."""
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a0)
    M = conv.max()
    threshold = M * 0.90
    peak_positions = np.where(conv >= threshold)[0]

    pressure = np.zeros(n)
    for k in peak_positions:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            w = a0[i] * a0[j]
            pressure[i] += w
            if j != i:
                pressure[j] += w * 0.5

    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max

    noise = rng.standard_normal(n)
    a_new = a * (1.0 - strength * pressure + 0.2 * strength * noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a0.copy()
    return a_new / s


def random_perturb(a0, rng, frac=0.1):
    """Random perturbation: zero out some coords, add noise to others."""
    n = len(a0)
    a = a0.copy()
    mask = rng.random(n) < frac
    a[mask] = 0
    a[~mask] *= (1.0 + rng.standard_normal((~mask).sum()) * 0.1)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    return a / s


def init_symmetric(n, rng):
    half = n // 2
    t = np.linspace(0, 1, half)

        freq = rng.uniform(0.5, 4.0)
        amp = rng.uniform(0.1, 0.4)
        phase = rng.uniform(0, np.pi)
        profile += amp * np.cos(np.pi * freq * t + phase)
    profile = np.maximum(profile, 0)
    if n % 2 == 0:
        a = np.concatenate([profile, profile[::-1]])
    else:
        mid = max(profile[-1], profile[0]) * rng.uniform(0.5, 1.5)
        a = np.concatenate([profile, [mid], profile[::-1]])
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
