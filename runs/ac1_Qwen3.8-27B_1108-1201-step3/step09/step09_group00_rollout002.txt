    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return a / s


def init_quadratic_phase(n, rng=None):
    """Initialize with quadratic phase pattern (CAZAC-inspired)."""
    j = np.arange(n, dtype=np.float64)
    # Try different quadratic phase patterns
    patterns = [
        (1.0 + np.cos(2.0 * np.pi * j * j / n)) / 2.0,
        (1.0 + np.cos(2.0 * np.pi * j * (j + 1) / (2.0 * n))) / 2.0,
        (1.0 + 0.5 * np.cos(2.0 * np.pi * j * j / (n / 2.0))) / 1.5,
        np.exp(-0.5 * (1.0 - np.cos(2.0 * np.pi * j * j / n))),
    ]
    if rng is not None:
        idx = rng.integers(0, len(patterns))
    else:
        idx = 0
    a = patterns[idx].copy()
    # Add small random perturbation
    if rng is not None:
        a += rng.standard_normal(n) * 0.01
    return normalize(a)


def init_anti_peak(n, rng=None):
    """Initialize with reduced center mass (anti-peak shape)."""
    j = np.arange(n, dtype=np.float64)
    x = j / (n - 1) if n > 1 else np.array([0.5])
    # Shape that's higher at edges, lower in center
    center = 0.5
    a = 1.0 + 0.4 * np.cos(2.0 * np.pi * (x - center) * 2)
    a = np.maximum(a, 0.05)
    if rng is not None:
        a += rng.standard_normal(n) * 0.02
    return normalize(a)


def init_gaussian_window(n, rng=None):
    """Gaussian window centered."""
    j = np.arange(n, dtype=np.float64)
    x = (j - n / 2.0) / (n / 2.0)
    a = np.exp(-2.0 * x * x)
    if rng is not None:
        a += rng.standard_normal(n) * 0.01
    return normalize(a)


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=200000, verbose=True):
    """Nesterov dual averaging LP refinement."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = normalize(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_rounds = 0
    rng = np.random.default_rng(99999)
    lp_count = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        lp_count += 1

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
       .score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-10:
                stall_rounds = 0
            else:
                stall_rounds += 1
            prev_best = best_score

            if stall_rounds >= 5:
                noise_scale = 0.05 / np.sqrt(stall_rounds)
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_rounds = 0

            if verbose and (it + 1) % 1000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP-Newton iter={it+1} ({elapsed:.0f}s, {lp_count} LPs): score={best_score:.8f}")

    return best_a, best_score


def logsumexp_smoothed_obj_and_grad(a, T, n):
    """Compute LogSumExp-smoothed objective and its gradient w.r.t. a (with sum=1 constraint)."""
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp((b - max_b) / T)
    Z = exp_b.sum()
    w = exp_b / Z
    obj = T * np.log(Z) + max_b  # This is (1/T)*log(sum(exp(b/T))) = logsumexp(b/T)
    # Actually let's be more careful:
    # Smoothed max = (1/T) * log(sum(exp(T*b)))
    # For large T, this approximates max(b)
    # Let's use: smoothed = T * log(sum(exp((b - max_b)/T))) + max_b
    # Gradient of smoothed max w.r.t. a_j:
    # d/d a_j [T * log(sum_k exp((b_k - max_b)/T))] = sum_k w_k * db_k/da_j
    # where w_k = exp((b_k - max_b)/T) / sum_l exp((b_l - max_b)/T)
    
    # db_k / da_j = a[k-j] + a[j-k] (for valid indices, since b = conv(a,a))
    # Actually db_k/da_j = a[k-j] if 0<=k-j<n, else 0, plus a[j-k] if 0<=j-k<n, else 0
    # But since b_k = sum_i a[i]*a[k-i], db_k/da_j = a[k-j] (when valid) + a[j-k] (when valid, for j != k-j case)
    # Wait: b_k = sum_i a_i * a_{k-i}. db_k/da_j = a_{k-j} (from the i=j term) + sum over i where k-i=j, i.e., i=k-j
    # So db_k/da_j = a_{k-j} + a_{k-j} = 2*a_{k-j} if k != 2j, else a_{k-j} (when k=2j, both terms are the same)
    # Actually no: b_k = sum_i a_i * a_{k-i}. Taking derivative w.r.t. a_j:
    # term where i=j: a_{k-j}
    # term where k-i=j, i.e., i=k-j: a_{k-j} (this is a different term unless j = k-j, i.e., k=2j)
    # So db_k/da_j = a_{k-j} + a_{k-j} = 2*a_{k-j} (when k != 2j)
    # and db_k/da_j = a_{k-j} (when k = 2j, since it's a^2)
    # Wait, let me be more careful:
    # b_k = sum_i a[i] * a[k-i]
    # If j != k-j (i.e., k != 2j): db_k/d a_j = a[k-j] (from i=j) + a[k-j] (from i=k-j) = 2*a[k-j]
    # If j == k-j (i.e., k == 2j): b_k contains term a[j]*a[j] = a[j]^2, so db_k/d a_j = 2*a[j]
    # So in all cases: db_k/da_j = 2*a[k-j] (when 0 <= k-j < n), else 0
    # Wait, that's not right either. Let me redo:
    # b_k = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i] * a[k-i]
    # db_k/d a_j: we need j to be in the range AND k-j to be in the range
    # If 0 <= j < n AND 0 <= k-j < n: then a[j] appears in b_k
    # As a factor of a[k-j] (when i=j): contributes a[k-j]
    # As a factor of a[j] (when k-i=j, i=k-j): contributes a[k-j]
    # Unless j = k-j, in which case it's a[j]^2 and the derivative is 2*a[j]
    # So: db_k/d a_j = 2 * a[k-j] if (0<=j<n and 0<=k-j<n), else 0
    
    grad = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        wk = w[k]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            grad[j] += 2.0 * wk * a[k - j]
    
    # For the constrained problem (sum a = 1), the effective gradient is:
    # grad_eff = grad - mean(grad * a) * ... no, for the log-barrier or projection
    # Actually for theLogSumExp approach in the simplex, we use:
    # The objective on the simplex is f(a) = T*log(sum(exp((conv(a,a)-max)/T))) + max
    # We want to minimize this subject to sum(a)=1, a>=0
    # Using the fact that on the simplex, the gradient of f w.r.t. a (unconstrained) projected
    # to the tangent space is: g - a * (g @ a) / (a @ a) ... no
    # Simpler: just return the unconstrained gradient and let L-BFGS-B handle it
    # But we need to parameterize on the simplex. Let's use the log-parameterization.
    
    return obj, grad


def lbgfs_simplex_refine(a0, time_limit, T_stages=None, maxiter=3000):
    """L-BFGS on the log-parameterization for simplicity and efficiency."""
    n = len(a0)
    # Parameterize: a_j = exp(z_j) / sum(exp(z))
    # This automatically satisfies sum(a)=1, a>0
    
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [50.0, 25.0, 15.0, 8.0, 4.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]
    
    remaining_iters = maxiter
    iters_per_stage = max(20, min(remaining_iters // len(T_stages), 500))
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        def func(z_vec):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            z_shifted = z_vec - z_vec.max()
            exp_z = np.exp(z_shifted)
            a = exp_z / exp_z.sum()
            b = fast_convolve(a)
            max_b = b.max()
            exp_tb = np.exp((b - max_b) / T)
            return T * np.log(exp_tb.sum()) + max_b
        
        def jac(z_vec):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_vec)
            z_shifted = z_vec - z_vec.max()
            exp_z = np.exp(z_shifted)
            a = exp_z / exp_z.sum()
            b = fast_convolve(a)
            max_b = b.max()
            exp_tb = np.exp((b - max_b) / T)
            w = exp_tb / exp_tb.sum()
            
            # Compute grad of smoothed max w.r.t. a
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                wk = w[k]
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * wk * a[k - idx]
            
            # Chain rule: grad_z = a * (grad_a - sum(grad_a * a))
            # because a = softmax(z), da_i/dz_j = a_i*(delta_ij - a_j)
            # So grad_z_j = sum_i grad_a_i * da_i/dz_j = a_j * (grad_a_j - sum_i a_i * grad_a_i)
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-15,
                         'gtol': 1e-10, 'maxfun': iters_per_stage * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    # Final a
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def peak_pair_descent(a0, time_limit, n_steps=50000, verbose=True):
    """
    Novel peak-pair swap coordinate descent.
    For the peak position, find pairs (i, j) where moving mass from i to j
    most reduces the peak while not increasing other positions too much.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Find the top few peak positions
        top_k = min(3, len(b))
        peak_positions = np.argsort(b)[::-1][:top_k]
        
        improved = False
        for pk in peak_positions:
            pk = int(pk)
            # Find contributing pairs
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            
            # Top contributors
            top_m = min(10, len(idx))
            sorted_idx = idx[np.argsort(contribs)[::-1]]
            top_positions = sorted_idx[:top_m]
            
            for src in top_positions:
                if a[src] < 1e-10:
                    continue
                # Find best target: position j where a[pk-j] is small (reduces peak)
                # and where moving mass doesn't create a new peak elsewhere
                # Candidates: positions where the "cross contribution" at pk is small
                cross = a[(pk - src + np.arange(n)) % n]  # not quite right for linear conv
                
                # Better: for target position t, the change in b[pk] is:
                # delta_b_pk = a[pk-t]*delta - a[pk-src]*delta = delta * (a[pk-t] - a[pk-src])
                # We want a[pk-t] < a[pk-src] for reduction
                # But we also need t and pk-t to be valid (0 <= t < n, 0 <= pk-t < n)
                
                t_lo = max(0, pk - n + 1)
                t_hi = min(n - 1, pk)
                valid_targets = np.arange(t_lo, t_hi + 1)
                valid_targets = valid_targets[valid_targets != src]
                
                if len(valid_targets) == 0:
                    continue
                
                # Score each target by the reduction in peak
                cross_vals = a[pk - valid_targets] - a[pk - src]
                # We want cross_vals to be negative (reduction)
                # Also check impact on other positions
                best_target = None
                best_reduction = 0
                
                # Only check top candidates by cross reduction
                neg_mask = cross_vals < -1e-10
                if neg_mask.any():
                    candidates = valid_targets[neg_mask]
                    reductions = -cross_vals[neg_mask]
                    # Take top few
                    top_cands = candidates[np.argsort(reductions)[::-1][:5]]
                    
                    for tgt in top_cands:
                        # Try moving a small fraction of mass
                        for frac in [0.05, 0.1, 0.2, 0.3]:
                            delta = a[src] * frac
                            if delta < 1e-15:
                                continue
                            a_trial = a.copy()
                            a_trial[src] -= delta
                            a_trial[tgt] += delta
                            a_trial = np.maximum(a_trial, 0)
                            st = a_trial.sum()
                            if st < 1e-15:
                                continue
                            a_trial = a_trial / st
                            
                            score = 2.0 * n * fast_convolve(a_trial).max()
                            if score < best_score - 1e-12:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break
                if improved:
                    break
            if improved:
                stall = 0
                break
        
        if not improved:
            stall += 1
            if stall > 200:
                # Try a random perturbation
                a_trial = a + np.random.default_rng(step).standard_normal(n) * 0.01
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score + 0.01:  # Allow small degradation
                    a = a_trial
                stall = 0
            if stall > 1000:
                break
        
        if verbose and step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakPair {elapsed:.0f}s: score={best_score:.8f} stall={stall}")
    
    return best_a, best_score


def subgradient_refine(a0, time_limit, n_steps=50000, verbose=True):
    """Exact subgradient method targeting active peak positions."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.03
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Active set: positions within 0.1% of max
        threshold = max_b * 0.9999
        active = np.where(b >= threshold)[0]
        
        # Compute subgradient (average over active positions)
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        
        # Projected step
        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st
        
        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 20:
            lr *= 0.85
            no_improve = 0
        lr = max(1e-7, lr)
        
        if verbose and step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.8f} lr={lr:.6f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Combines LP Nesterov, L-BFGS LogSumExp, peak-pair descent, and subgradient.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        n_init = 1200
        best_seq = init_quadratic_phase(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
    
    # === Phase 1: Quick LP refinement on parent ===
    print(f"\n=== Phase 1: LP Refinement n={len(best_seq)} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=100000, verbose=True)
        try_update(a_lp, "lp_1")
    
    # === Phase 2: L-BFGS LogSumExp ===
    print(f"\n=== Phase 2: L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 300.0)
    if t_alloc > 30:
        T_stages = [40.0, 20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02]
        a_lb = lbgfs_simplex_refine(best_seq, t_alloc, T_stages=T_stages, maxiter=10000)
        try_update(a_lb, "lbgfs_1")
    
    # === Phase 3: Peak-pair descent ===
    print(f"\n=== Phase 3: Peak-Pair Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 250.0)
    if t_alloc > 20:
        a_pp, _ = peak_pair_descent(best_seq, t_alloc, n_steps=80000, verbose=True)
        try_update(a_pp, "peakpair_1")
    
    # === Phase 4: Alternative initializations ===
    print(f"\n=== Phase 4: Alternative inits (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_cur = len(best_seq)
        alts = []
        alts.append(init_quadratic_phase(n_cur, rng))
        alts.append(init_anti_peak(n_cur, rng))
        
        # Also try different length
        n_alt = int(n_cur * 1.3)
        alts.append(interpolate_sequence(best_seq, n_alt))
        
        for i, alt in enumerate(alts[:3]):
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            t_alt = min(rem * 0.2, 60.0)
            if t_alt < 10:
                continue
            print(f"  Alt {i}: n={len(alt)}, initial score={evaluate_local(alt):.8f}")
            # Quick LP
            a_alt, _ = nesterov_lp_refine(alt, t_alt, solver=solver, n_max_iters=20000, verbose=False)
            try_update(a_alt, f"alt_lp_{i}")
    
    # === Phase 5: Extended LP on best ===
    print(f"\n=== Phase 5: Extended LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 300.0)
    if t_alloc > 30:
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=100000, verbose=True)
        try_update(a_lp2, "lp_2")
    
    # === Phase 6: Very low T L-BFGS ===
    print(f"\n=== Phase 6: Low-T L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 150.0)
    if t_alloc > 15:
        T_stages2 = [5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]
        a_lb2 = lbgfs_simplex_refine(best_seq, t_alloc, T_stages=T_stages2, maxiter=8000)
        try_update(a_lb2, "lbgfs_lowT")
    
    # === Phase 7: Subgradient polish ===
    print(f"\n=== Phase 7: Subgradient (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 100.0)
    if t_alloc > 10:
        a_sg, _ = subgradient_refine(best_seq, t_alloc, n_steps=50000, verbose=True)
        try_update(a_sg, "subgrad")
    
    # === Phase 8: Final peak-pair + L-BFGS polish ===
    print(f"\n=== Phase 8: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_pp = min(rem * 0.4, 60.0)
        if t_pp > 10:
            a_pp2, _ = peak_pair_descent(best_seq, t_pp, n_steps=30000, verbose=False)
            try_update(a_pp2, "final_peakpair")
        
        rem = deadline - time.monotonic()
        if rem > 15:
            t_lb = min(rem * 0.5, 30.0)
            if t_lb > 5:
                a_lb3 = lbgfs_simplex_refine(best_seq, t_lb,
                                             T_stages=[1.0, 0.5, 0.2, 0.1, 0.05, 0.01],
                                             maxiter=3000)
                try_update(a_lb3, "final_lbgfs")
    
    # Final
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
