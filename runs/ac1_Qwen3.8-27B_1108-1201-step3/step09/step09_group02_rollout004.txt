import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def init_flat_spectrum(n, rng):
    """Generate non-negative sequence with approximately flat power spectrum."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = np.ones(n_freq)  # flat amplitude
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_bell(n, rng, center_frac=0.5, width_frac=0.3):
    """Gaussian bell initialization."""
    t = np.arange(n, dtype=np.float64)
    center = n * center_frac
    width = n * width_frac
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_ramp(n, rng):
    """Linear ramp (asymmetric)."""
    t = np.arange(n, dtype=np.float64) / n
    direction = rng.random()
    if direction < 0.5:
        a = 1.0 - t
    else:
        a = t
    a = np.maximum(a, 1e-10)
    s = a.sum()
    return a / s

def init_fourier_modes(n, rng, n_modes=5):
    """Sum of a few cosine modes."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n) * 0.5
    for _ in range(n_modes):
        freq = rng.integers(1, n // 4 + 1)
        amp = rng.uniform(0.1, 0.4)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    """
    Exact coordinate descent: for each coordinate, use golden section search
    to find the optimal value.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            # Unaffected max
            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]

            if has_quad:
                quad_idx = 2 * i - k_lo

            # Define score function
            def score_fn(x):
                ns = current_sum + (x - v)
                if ns < 0.01:
                    return 1e10
                nv = c_vals + 2.0 * x * a_partners
                if has_quad:
                    nv[quad_idx] = c_vals[quad_idx] + x * x
                nm = max(unaff_max, float(nv.max()))
                return 2.0 * n * nm / (ns * ns)

            # Golden section search with tighter bounds
            # The optimal x should be near v; use a small range
            lower = 0.0
            upper = max(2.0 * v + 1.0/n, 3.0 / n)
            
            # Quick check: if current value is already good, use a tighter range
            cur_score = score_fn(v)
            if cur_score < best_score:
                lower = max(0.0, v - 0.5 * v)
                upper = v + 0.5 * v + 1.0/n
            
            best_x = golden_section_search(score_fn, lower, upper, max_iter=30)

            if abs(best_x - v) > 1e-12:
                delta = best_x - v

                # Update convolution
                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                a[i] = best_x
                current_sum += delta

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, sweeps

def golden_section_search(f, lo, hi, max_iter=35):
    """Golden section search for minimum of f on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-14:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

def fine_coord_descent(a0, time_limit, verbose=False, label=""):
    """Fine coordinate descent with small multiplicative steps for final polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    sweeps = 0

    # Try progressively smaller step sizes
    step_sizes = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]

    for step_idx, base_step in enumerate(step_sizes):
        step = base_step
        while time.monotonic() - t0 < time_limit:
            sweeps += 1
            improved = False

            for i in range(n):
                if time.monotonic() - t0 > time_limit:
                    break
                if a[i] < 1e-14:
                    continue

                v = a[i]
                k_lo = i
                k_hi = min(i + n - 1, 2 * n - 2)

                unaff_max = -1e30
                if k_lo > 0:
                    unaff_max = max(unaff_max, float(conv[:k_lo].max()))
                if k_hi < 2 * n - 2:
                    unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

                partners = np.arange(k_lo, k_hi + 1) - i
                a_partners = a[partners]
                has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

                c_vals = np.empty(k_hi - k_lo + 1)
                for idx in range(k_hi - k_lo + 1):
                    k = k_lo + idx
                    if k == 2 * i:
                        c_vals[idx] = conv[k] - v * v
                    else:
                        c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]

                if has_quad:
                    quad_idx = 2 * i - k_lo

                # Try shrinking (most useful direction)
                for factor in [1.0 - step, 1.0 - 2*step, 1.0 + step]:
                    x = v * factor
                    if x < 0:
                        x = 0
                    ns = current_sum + (x - v)
                    if ns < 0.01:
                        continue
                    nv = c_vals + 2.0 * x * a_partners
                    if has_quad:
                        nv[quad_idx] = c_vals[quad_idx] + x * x
                    nm = max(unaff_max, float(nv.max()))
                    sc = 2.0 * n * nm / (ns * ns)
                    if sc < best_score - 1e-14:
                        # Apply the change
                        delta = x - v
                        changes = 2.0 * delta * a_partners
                        if has_quad:
                            changes = changes.copy()
                            changes[quad_idx] = delta * (2.0 * v + delta)
                        conv[k_lo:k_hi + 1] += changes
                        a[i] = x
                        current_sum = ns
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        v = x  # Update for next factor
                        # Recompute c_vals for the new v
                        break

        if not improved and sweeps > 1:
            break

    return best_a, best_score, sweeps

def subgradient_step(a, conv, lr, n_peaks=20):
    """One subgradient step using active peak weighting with soft assignment."""
    n = len(a)
    k_act = min(n_peaks, 2 * n - 1)
    peaks = np.argpartition(conv, -k_act)[-k_act:]
    pv = conv[peaks]
    # Soft weighting
    temp = 0.003 * max(pv.max(), 1e-10)
    w = np.exp((pv - pv.max()) / temp)
    ws = w.sum()
    if ws < 1e-15:
        return a
    w = w / ws

    # Compute subgradient: grad_i = 2 * sum_k w_k * a[(k-i) mod something]
    # Actually: d(conv_k)/d(a_i) = a[k-i] if 0 <= k-i < n, else 0
    # So grad_i = 2 * sum_k w_k * a[k-i] (where valid)
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-14:
            continue
        k = int(peaks[idx])
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        j = np.arange(i_start, i_end + 1)
        grad[j] += 2.0 * w[idx] * a[k - j]

    # Subgradient descent with simplex projection
    a_new = simplex_project(a - lr * grad)
    return a_new

def basin_hopping(a0, time_limit, cd_func, rng, n_perturbs=10, verbose=False):
    """Basin hopping with autocorrelation-aware perturbations."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()

    for p in range(n_perturbs):
        if time.monotonic() - t0 > time_limit:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 15:
            break
        t_alloc = min(rem * 0.5, 45.0)

        # Type of perturbation
        ptype = rng.integers(0, 4)
        if ptype == 0:
            # Gaussian noise
            sigma = 0.01 + 0.04 * rng.random()
            perturbed = a + rng.standard_normal(n) * sigma
        elif ptype == 1:
            # Peak breaking: reduce values at autocorrelation peaks
            conv_p = fast_convolve(a)
            top_k = min(8, 2 * n - 1)
            peaks = np.argpartition(conv_p, -top_k)[-top_k:]
            perturbed = a.copy()
            for k in peaks:
                k_int = int(k)
                i_start = max(0, k_int - n + 1)
                i_end = min(n - 1, k_int)
                idx = np.arange(i_start, i_end + 1)
                n_pick = max(1, len(idx) // 4)
                picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
                for ii in picks:
                    perturbed[ii] *= (1.0 - 0.1 * rng.random())
        elif ptype == 2:
            # Spectral perturbation: modify in frequency domain
            n_fft = 1 << (n - 1).bit_length()
            A = rfft(a, n_fft)
            # Add small random phase noise
            phase_noise = rng.standard_normal(len(A)) * 0.05
            phase_noise[0] = 0  # Keep DC
            A_pert = A * np.exp(1j * phase_noise)
            perturbed = irfft(A_pert, n_fft)[:n]
        else:
            # Coordinate swap perturbation
            perturbed = a.copy()
            n_swaps = max(5, n // 20)
            for _ in range(n_swaps):
                i1 = rng.integers(0, n)
                i2 = rng.integers(0, n)
                swap_scale = rng.uniform(0.7, 1.3)
                perturbed[i1] = perturbed[i1] * swap_scale
                perturbed[i2] = perturbed[i2] / swap_scale

        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        a_opt, s_opt, _ = cd_func(perturbed, t_alloc)
        if s_opt < best_score:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    BH p={p} type={ptype}: improved to {best_score:.8f}")

    return best_a, best_score

def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    for solver_name in [cp.GLPK, cp.ECOS, cp.PDLP, cp.SCS, cp.CBC]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.PDLP, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def fictitious_play(a0, time_limit, solver=None, verbose=False):
    """LP fictitious play with line search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 45:
            break

        b, M_val = solve_lp_best_response(a, solver)
        if b is None:
            continue

        # Line search
        best_t = 0.0
        best_ls_score = best_score
        for trial_t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            cand = (1.0 - trial_t) * a + trial_t * b
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = 2.0 * n * float(fast_convolve(cand).max())
            if sc < best_ls_score:
                best_ls_score = sc
                best_t = trial_t

        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * b
            a = np.maximum(a, 0)
            cs = a.sum()
            if cs < 1e-10:
                continue
            a = a / cs
            sc = 2.0 * n * float(fast_convolve(a).max())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        else:
            # Simple averaging
            a = 0.7 * a + 0.3 * b
            a = np.maximum(a, 0)
            cs = a.sum()
            if cs < 1e-10:
                continue
            a = a / cs
            sc = 2.0 * n * float(fast_convolve(a).max())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose and iters % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score
