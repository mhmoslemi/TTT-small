def projected_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    # Use bisection on the water level
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    rho = np.where(v_sorted > (cumsum - target_sum) / np.arange(1, n + 1))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[0] + 1
    theta = (cumsum[rho - 1] - target_sum) / rho
    result = np.maximum(v - theta, 0)
    # Normalize to ensure exact sum
    s = result.sum()
    if s > 1e-15:

    return result

1. Gerchberg-Saxton spectral shaping for diverse initial structure
2. Nesterov-accelerated mirror descent (exponentiated gradient) with FFT-based smoothed subgradients
3. Iterated Local Search with autocorrelation-guided structured perturbations
4. Multi-stage L-BFGS with log-sum-exp annealing for precise local refinement
5. Multi-length exploration with warm-start interpolation
Key differences from previous: no LP solvers, O(n log n) per subgradient step,
ILS for escaping local minima, and more aggressive multi-n search.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Evaluate sequence using the provided evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sequence(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def lse_gradient(a, T):
    """Compute gradient of T*log(sum(exp(b/T))) where b=convolve(a,a)."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    grad = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * w[k] * a[k - idx]
    S = np.dot(grad, a)
    return a * (grad - S)


def lse_value(a, T):
    """Compute T*log(sum(exp(b/T))) where b=convolve(a,a)."""
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    return (2.0 * len(a) / T) * (max_Tb + np.log(exp_Tb.sum()))


def lbgfs_refine(a0, time_limit, T_start=100.0, T_end=0.05, n_stages=12):
    """L-BFGS with LogSumExp smoothing and geometric temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()

    def make_func(T_val):
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, 8000 // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
            a_cur = softmax_from_z(z)
            sc = 2.0 * n * fast_convolve(a_cur).max()
            if sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
        except Exception:
            pass

    return best_a, best_score


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def mirror_descent_nesterov(a0, time_limit, T=1.0, n_max_iters=2000000):
    """Nesterov-accelerated mirror descent (exponentiated gradient) in log-space."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).astype(np.float64)
    z_prev = z.copy()
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    t0 = time.monotonic()

    momentum = 0.0
    best_momentum = 0.75

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        a = softmax_from_z(z)
        g = lse_gradient(a, T)

        # Backtracking line search in log-space
        step = 1.0
        f_cur = lse_value(a, T)
        improved = False
        for _ in range(8):
            z_trial = z - step * g
            a_trial = softmax_from_z(z_trial)
            f_trial = lse_value(a_trial, T)
            if f_trial < f_cur - 1e-10:
                z_prev = z.copy()
                z = z_trial
                momentum = best_momentum
                improved = True
                break
            step *= 0.5

        if not improved:
            # No improvement possible, try small random nudge
            step = 0.01
            z_prev = z.copy()
            z = z - step * g
            momentum = 0.0

        if it > 0:
            z = z + momentum * (z - z_prev)

        # Periodic check
        if it % 200 == 0:
            a_check = softmax_from_z(z)
            sc = 2.0 * n * fast_convolve(a_check).max()
            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()

    a_final = softmax_from_z(z)
    sc = 2.0 * n * fast_convolve(a_final).max()
    if sc < best_score:
        best_score = sc
        best_a = a_final.copy()

    return best_a, best_score


def ild_search(a0, time_limit, n_restarts=50):
    """
    Iterated Local Search with autocorrelation-guided perturbations.
    Each restart: perturb current best, do quick local optimization, accept if improved.
    """
    n = len(a0)
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for restart in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break

        # Create perturbation
        a = best_a.copy()
        b = fast_convolve(a)
        k_star = int(np.argmax(b))

        # Find top contributors to peak
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[k_star - idx]

        # Also consider near-peaks
        b_work = b.copy()
        b_work[0] = -np.inf
        b_work[-1] = -np.inf
        top_peaks = np.argsort(b_work)[::-1][:5]

        # Build contribution array over all peaks
        total_contrib = np.zeros(n)
        for k in top_peaks:
            js_k = max(0, k - n + 1)
            je_k = min(n - 1, k)
            idx_k = np.arange(js_k, je_k + 1)
            total_contrib[idx_k] += a[idx_k] * a[k - idx_k] * (b[k] / b[k_star])

        # Perturbation scale decreases over restarts
        base_scale = 0.05 / (1.0 + restart * 0.05)
        scale = base_scale * (1.0 + rng.random() * 0.5)

        # Move mass from high-contrib to low-contrib positions
        n_high = max(5, n // 20)
        n_low = max(5, n // 10)

        # Get positions sorted by contribution
        sorted_idx = np.argsort(total_contrib)
        high_positions = sorted_idx[-n_high:]
        low_positions = sorted_idx[:n_low]

        # Compute how much mass to move
        mass_to_move = scale * a[high_positions].sum()

        # Reduce high positions
        reduction = a[high_positions] * scale
        a[high_positions] -= reduction
        a = np.maximum(a, 0)

        # Distribute to low positions proportionally
        low_weights = a[low_positions] + 1e-10
        low_weights = low_weights / low_weights.sum()
        a[low_positions] += mass_to_move * low_weights

        # Also add small random noise
        noise = rng.standard_normal(n) * scale * 0.3
        a = a + noise
        a = np.maximum(a, 0)
        a = a / a.sum()

        # Quick local optimization (mirror descent, short)
        t_sub = min(5.0, (time_limit - (time.monotonic() - t0)) * 0.3)
        if t_sub > 1.0:
            a, sc = mirror_descent_nesterov(a, t_sub, T=5.0, n_max_iters=50000)
        else:
            sc = 2.0 * n * fast_convolve(a).max()

        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def gs_phase(a0, n_iterations=2000):
    """Gerchberg-Saxton spectral shaping."""
    n = len(a0)
    n_fft = 1 << (2 * n - 1).bit_length()
    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    for it in range(n_iterations):
        a_padded = np.zeros(n_fft)
        a_padded[:n] = a
        A = np.fft.rfft(a_padded)
        phase = np.angle(A)
        mag = np.abs(A)
        # Soft flattening
        target_mag = np.sqrt(np.mean(mag**2))
        alpha = 0.5
        mag_new = mag ** (1.0 - alpha) * np.maximum(target_mag, 1e-10) ** alpha
        mag_new = np.maximum(mag_new, 1e-10)
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, n_fft)
        a_new = a_new[:n]
        a = np.maximum(a_new, 0)
        s = a.sum()
        if s > 1e-15:
            a = a / s
        else:
            a = a0.copy()
            break

        if it % 100 == 0:
            sc = 2.0 * n * fast_convolve(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc = 2.0 * n * fast_convolve(a).max()
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


def peak_shave_coordinate(a0, time_limit, n_steps=50000):
    """Coordinate-wise peak shaving with redistribution."""
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_work = b.copy()
        if len(b_work) > 2:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        # Try reducing top contributor
        top_pi = np.argsort(contribs)[::-1][:10]
        improved = False
        for pi in top_pi:
            i = idx[pi]
            if a[i] < 1e-12:
                continue
            for factor in [0.95, 0.90, 0.85, 0.80]:
                a_trial = a.copy()
                diff = a_trial[i] * (1.0 - factor)
                a_trial[i] *= factor
                # Redistribute to least-contributing positions
                low_idx = idx[np.argsort(contribs)[:n // 4]]
                low_idx = low_idx[low_idx != i]
                if len(low_idx) > 0:
                    w = a_trial[low_idx] + 1e-12
                    w = w / w.sum()
                    a_trial[low_idx] += diff * w
                    a_trial = a_trial / a_trial.sum()
                    sc = 2.0 * n * fast_convolve(a_trial).max()
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
        if not improved:
            no_improve += 1
            if no_improve > 500:
                break
        else:
            no_improve = 0

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. GS spectral shaping for diversity
    2. Nesterov mirror descent (fast subgradient)
    3. ILS with structured perturbations
    4. L-BFGS multi-stage polish
    5. Multi-n exploration
    6. Final coordinate-wise polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        x = np.linspace(0, np.pi, n_init)
        best_seq = np.cos(x / 2) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = norm_sequence(a)
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: GS spectral shaping (fast, for diversity) ===
    print(f"\n=== Phase 1: GS Spectral Shaping (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        a_gs, _ = gs_phase(best_seq, n_iterations=3000)
        try_update(a_gs, "gs")

    # === Phase 2: Nesterov Mirror Descent ===
    print(f"\n=== Phase 2: Mirror Descent Nesterov (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 350.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s, T=1.0")
        a_md, _ = mirror_descent_nesterov(best_seq, t_alloc, T=1.0, n_max_iters=5000000)
        try_update(a_md, "mirror_desc")

    # === Phase 3: ILS ===
    print(f"\n=== Phase 3: ILS (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 450.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ils, _ = ild_search(best_seq, t_alloc, n_restarts=200)
        try_update(a_ils, "ils")

    # === Phase 4: L-BFGS multi-stage polish ===
    print(f"\n=== Phase 4: L-BFGS Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s, T: 50->0.05, 14 stages")
        a_lb, _ = lbgfs_refine(best_seq, t_alloc, T_start=50.0, T_end=0.05, n_stages=14)
        try_update(a_lb, "lbgfs_main")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration (best={best_score:.6f}) ===")
    alt_lengths = sorted(set([
        int(n_best * 0.85), int(n_best * 0.9), int(n_best * 1.05),
        int(n_best * 1.1), int(n_best * 1.15), int(n_best * 1.25),
        896, 928, 1024, 1088, 1152, 1280
    ]))
    alt_lengths = [x for x in alt_lengths if 512 <= x <= 2560 and x != n_best]

    for i, alt_n in enumerate(alt_lengths[:5]):
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t_total = min(rem * 0.20, 150.0)
        if t_total < 20:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)

        # Quick mirror descent
        t_md = t_total * 0.4
        a_alt, _ = mirror_descent_nesterov(init_alt, t_md, T=5.0, n_max_iters=2000000)
        try_update(a_alt, f"md_n{alt_n}")

        # L-BFGS polish
        rem = deadline - time.monotonic()
        if rem > 15:
            t_lb = min(rem * 0.3, 60.0)
            if t_lb > 10:
                a_alt2, _ = lbgfs_refine(best_seq, t_lb, T_start=20.0, T_end=0.1, n_stages=8)
                try_update(a_alt2, f"lbgfs_n{alt_n}")

    # === Phase 6: Second ILS round (finer) ===
    print(f"\n=== Phase 6: ILS Round 2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_alloc = min(rem * 0.25, 200.0)
        a_ils2, _ = ild_search(best_seq, t_alloc, n_restarts=100)
        try_update(a_ils2, "ils2")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t_fin = min(rem * 0.3, 60.0)
        a_fin, _ = lbgfs_refine(best_seq, t_fin, T_start=10.0, T_end=0.03, n_stages=10)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_ps = min(rem * 0.2, 40.0)
        a_ps, _ = peak_shave_coordinate(best_seq, t_ps, n_steps=30000)
        try_update(a_ps, "final_ps")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_md_fin = min(rem * 0.2, 30.0)
        a_md_fin, _ = mirror_descent_nesterov(best_seq, t_md_fin, T=0.5, n_max_iters=1000000)
        try_update(a_md_fin, "final_md")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
