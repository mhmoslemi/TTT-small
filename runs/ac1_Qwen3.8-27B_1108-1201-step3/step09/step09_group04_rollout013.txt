    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def score_of(a):

    c = fast_convolve(a)
    return 2.0 * len(a) * c.max()

def subgradient_topk(a, K=5):
    """Compute subgradient using top-K peaks of the auto-convolution."""
    n = len(a)
    c = fast_convolve(a)

    c_work = c.copy()
    c_work[0] = -np.inf
    c_work[-1] = -np.inf
    top_k_idx = np.argsort(c_work)[::-1][:K]
    
    # Weight by how much above the mean
    c_mean = c.sum() / len(c)
    weights = np.maximum(c[top_k_idx] - c_mean, 0)

        je = min(n - 1, k)
        for i in range(js, je + 1):

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def obj_and_grad_logspace(z, K=5, var_weight=0.0):

    obj = 2.0 * n * (smooth_max + var_weight * variance)

    grad_z = 2.0 * n * a * (grad_a - np.dot(grad_a, a))

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Fast subgradient with momentum in log-space (top-K peak smoothing)
2. IPF (Iterative Proportional Fitting) for auto-convolution flattening
3. Variance-regularized L-BFGS for smooth landscape exploitation
4. Direct peak shaving (coordinate descent)
5. Simulated annealing for local minimum escape
6. Multi-length and multi-shape exploration

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def normalize_positive(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def score_fast(a):
    return 2.0 * len(a) * fast_convolve(a).max()


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def compute_grad_a(a, K=5, var_weight=0.0):
    """Compute gradient of 2n*(smooth_max + var_weight*var) w.r.t. a.
    smooth_max uses top-K peak weighting."""
    n = len(a)
    c = fast_convolve(a)
    c_mean = c.sum() / len(c)
    
    c_work = c.copy()
    if len(c_work) > 2:
        c_work[0] = -np.inf
        c_work[-1] = -np.inf
    top_k = np.argsort(c_work)[::-1][:K]
    peak_vals = c[top_k]
    
    weights = np.maximum(peak_vals - c_mean, 0)
    if weights.sum() < 1e-15:
        weights = np.ones(K) / K
    else:
        weights = weights / weights.sum()
    
    grad_a = np.zeros(n)
    for w_k, k in zip(weights, top_k):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for i in range(js, je + 1):
            grad_a[i] += 2.0 * w_k *
