def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def golden_section_min(f, lo, hi, tol=1e-15, max_iter=200):
    """Golden section search for minimum."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):

        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)

                a[i] = best_x
                current_sum += delta

def active_set_lp(a_fixed, active_lags, solver=None):
    """
    Solve LP: minimize M subject to:
    - conv(a, b)[k] <= M for k in active_lags
    - b >= 0, sum(b) = 1
    """
    import cvxpy as cp
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    # Build constraints for active lags
    # conv(a_fixed, b)[k] = sum_j a_fixed[j] * b[k-j] for valid j
    constraints = [cp.sum(b_var) == 1]

    for k in active_lags:
        # conv[k] = sum_{j: 0<=j<n, 0<=k-j<n} a_fixed[j] * b[k-j]
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min > j_max:
            continue
        # contribution = sum_{j=j_min}^{j_max} a_fixed[j] * b[k-j]
        expr = 0
        for j in range(j_min, j_max + 1):
            expr += a_fixed[j] * b_var[k - j]
        constraints.append(expr <= M_var)

    prob = cp.Problem(cp.Minimize(M_var), constraints)

    if solver is None:
        solvers_to_try = [cp.PDLP, cp.ECOS, cp.GLPK, cp.SCS, cp.CBC]
    else:
        solvers_to_try = [solver, cp.PDLP, cp.ECOS, cp.GLPK, cp.SCS]

    for sol in solvers_to_try:
        try:
            prob.solve(solver=sol)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break
        except Exception:
            continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def full_conv_lp(a_fixed, solver=None):
    """Solve the full LP: min M s.t. conv(a,b) <= M, b>=0, sum(b)=1."""
    import cvxpy as cp
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    # Build full convolution matrix (sparse for efficiency)
    # conv(a, b)[k] = sum_j a[j] * b[k-j]
    # We'll build it as a dense matrix for simplicity (n up to ~2048)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed

    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    if solver is None:
        solvers_to_try = [cp.PDLP, cp.ECOS, cp.GLPK, cp.SCS, cp.CBC]
    else:
        solvers_to_try = [solver, cp.PDLP, cp.ECOS, cp.GLPK, cp.SCS]

    for sol in solvers_to_try:
        try:
            prob.solve(solver=sol)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break
        except Exception:
            continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def mirror_descent(a0, time_limit, verbose=False, label=""):
    """
    Mirror descent (multiplicative weight) with adaptive learning rate.
    a_new[i] = a[i] * exp(-lr * grad[i]), then normalize.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    lr = 0.05
    iters = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1

        # Find active peaks (top K lags)
        K = min(20, 2 * n - 1)
        peaks = np.argpartition(conv, -K)[-K:]
        peak_val = conv.max()

        # Compute subgradient: grad[i] = sum_{k in peaks} w_k * a[k-i] (for valid k-i)
        # Weight by how close each peak is to the max
        pv = conv[peaks]
        temp = 0.002 * peak_val
        w = np.exp((pv - peak_val) / temp)
        ws = w.sum()
        w = w / ws

        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                grad[j] += w[idx] * a[k - j]

        # Mirror descent update
        a_new = a * np.exp(-lr * grad)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        conv_new = fast_convolve(a_new)
        score_new = 2.0 * n * float(conv_new.max())

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            conv = conv_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                lr *= 0.8
            no_improve = min(no_improve, 100)

        # Adaptive lr
        lr *= 0.999

        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD {label} it={iters} ({elapsed:.0f}s): best={best_score:.8f}, lr={lr:.6f}")

    return best_a, best_score

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def score_normalized(a):
    """Compute the objective for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max())

def golden_section_min(f, lo, hi, tol=1e-15, max_iter=120):
    """Golden section search for minimum."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5


def precise_coord_descent(a0, time_limit, verbose=False, label="", random_order=False, rng=None):
    """
    High-precision coordinate descent exploiting piecewise structure.
    For coordinate i, affected conv values c[k] for k in [i, i+n-1]:
    - c[2i] is quadratic (x^2), rest are linear in x.
    Objective: 2*n*max(all c) / (sum)^2, minimized over x >= 0.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    order = np.arange(n)

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        if random_order and rng is not None:
            rng.shuffle(order)
        improved = False

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-16:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]

            if has_quad:
                quad_idx = 2 * i - k_lo

            def obj_func(x, _c=c_vals, _ap=a_partners, _qm=unaff_max,
                         _hq=has_quad, _qi=quad_idx, _cs=current_sum,
                         _v=v, _n=n):
                ns = _cs + (x - _v)
                if ns < 0.01:
                    return 1e10
                nv = _c + 2.0 * x * _ap
                if _hq:
                    nv[_qi] = _c[_qi] + x * x
                nm = max(_qm, float(nv.max()))
                return 2.0 * _n * nm / (ns * ns)

            upper = max(3.0 * v, 8.0 / n)
            best_x = golden_section_min(obj_func, 0.0, upper, tol=1e-15, max_iter=100)

            if abs(best_x - v) > 1e-14:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta

                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-15:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, sweeps


def build_conv_matrix_sparse(a_fixed):
    """Build convolution (toeplitz-like) action: T @ b = conv(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def active_set_lp(a_fixed, active_lags, solver=None):
    """
    Solve LP: minimize M s.t. conv(a_fixed, b)[k] <= M for k in active_lags,
    b >= 0, sum(b) = 1.
    """
    import cvxpy as cp
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    constraints = [cp.sum(b_var) == 1]
    for k in active_lags:
        k = int(k)
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min > j_max:
            continue
        expr = 0.0
        for j in range(j_min, j_max + 1):
            expr = expr + a_fixed[j] * b_var[k - j]
        constraints.append(expr <= M_var)

    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [cp.PDLP, cp.ECOS, cp.GLPK, cp.SCS, cp.CBC]
    if solver is not None:
        solvers_to_try = [solver] + solvers_to_try

    status = None
    for sol in solvers_to_try:
        try:
            prob.solve(solver=sol)
            status = prob.status
            if status in ('optimal', 'optimal_inaccurate'):
                break
        except Exception:
            continue

    if status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def full_lp_refine(a_fixed, solver=None):
    """Solve the full LP: min M s.t. T@b <= M, b>=0, sum(b)=1."""
    import cvxpy as cp
    n = len(a_fixed)

def mirror_descent(a0, time_limit, verbose=False, label=""):
    """
    Mirror descent (multiplicative weight) with adaptive learning rate.
    a_new[i] = a[i] * exp(-lr * grad[i]), then normalize.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    lr = 0.08
    iters = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1

        K = min(25, 2 * n - 1)
        peaks = np.argpartition(conv, -K)[-K:]
        peak_val = conv.max()

        pv = conv[peaks]
        temp = 0.003 * max(peak_val, 1e-10)
        w = np.exp((pv - peak_val) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws

        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            js = np.arange(j_min, j_max + 1)
            grad[js] += w[idx] * a[k - js]

        a_new = a * np.exp(-lr * grad)
        a_new = np.maximum(a_new, 1e-16)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        conv_new = fast_convolve(a_new)
        score_new = 2.0 * n * float(conv_new.max())

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            conv = conv_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.75

        lr *= 0.9995

        if verbose and iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD {label} it={iters} ({elapsed:.0f}s): best={best_score:.8f}, lr={lr:.5f}")

    return best_a, best_score

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_flat_spectrum(n, rng):
    """Generate non-negative sequence with approximately flat power spectrum."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.85, 1.15, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

                        best_t = t_trial

                if best_t > 0:
                    a_work = (1.0 - best_t) * a_work + best_t * b_lp
                    a_work = np.maximum(a_work, 0)
                    a_work = a_work / a_work.sum()
                    best_seq, best_score = check_and_update(a_work, best_seq, best_score, f"lp_{lp_iter}")

            # Quick CD after LP
            rem = deadline - time.monotonic() - 450
            if rem > 20:
                t_cd = min(40.0, rem * 0.3)
                a_cd, s_cd, _ = precise_coord_descent(a_work, t_cd)
                best_seq, best_score = check_and_update(a_cd, best_seq, best_score, f"cd_after_lp{lp_iter}")
                if s_cd < best_score + 1e-10:
                    a_work = best_seq.copy()
                    n_work = len(a_work)

        print(f"  Done ({time.monotonic()-t_lp:.0f}s): best={best_score:.8f}")
    except Exception as e:
        print(f"  LP phase error: {e}")
        a_work = best_seq.copy()
        n_work = len(a_work)

    # === Phase 3: Intensive coordinate descent ===
    print(f"\n=== Phase 3: Intensive CD (best={best_score:.8f}) ===")
    a_work = best_seq.copy()
    n_work = len(a_work)
    t_cd_start = time.monotonic()

    for rnd in range(6):
        rem = deadline - time.monotonic() - 400
        if rem < 30:
            break
        t_alloc = min(rem * 0.28, 250.0)
        use_random = (rnd % 2 == 1)
        a_opt, s_opt, sweeps = precise_coord_descent(a_work, t_alloc,
                                                      verbose=True, label=f"cd_r{rnd}",
                                                      random_order=use_random, rng=rng)
        best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"cd_r{rnd}")
        a_work = best_seq.copy()
        n_work = len(a_work)

    print(f"  Done ({time.monotonic()-t_cd_start:.0f}s): best={best_score:.8f}")

    # === Phase 4: Mirror descent ===
    print(f"\n=== Phase 4: Mirror Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 350
    if rem > 30:
        t_alloc_md = min(rem * 0.3, 150.0)
        a_md, s_md = mirror_descent(best_seq, t_alloc_md, verbose=True, label="p4")
        best_seq, best_score = check_and_update(a_md, best_seq, best_score, "mirror_descent")
        # Follow up with CD
        rem = deadline - time.monotonic() - 300
        if rem > 20:
            t_cd = min(rem * 0.4, 60.0)
            a_opt, s_opt, _ = precise_coord_descent(a_md, t_cd, label="cd_after_md")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, "cd_after_md")
        print(f"  Done: best={best_score:.8f}")

    # === Phase 5: Structure-aware basin hopping ===
    print(f"\n=== Phase 5: Basin Hopping (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 250
    if rem > 40:
        t_alloc_bh = min(rem * 0.5, 200.0)
        a_bh, s_bh = basin_hopping_structured(best_seq, t_alloc_bh, rng,
                                               precise_coord_descent,
                                               n_perturbs=12, verbose=True)
        best_seq, best_score = check_and_update(a_bh, best_seq, best_score, "basin_hop")
        print(f"  Done: best={best_score:.8f}")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-length (best={best_score:.8f}) ===")
    n_best = len(best_seq)
    alt_lengths = [n_best - 100, n_best - 50, n_best + 50, n_best + 100, n_best + 200]
    alt_lengths = sorted(set(x for x in alt_lengths if 256 <= x <= 2048))

    for n_alt in alt_lengths[:4]:
        rem = deadline - time.monotonic() - 150
        if rem < 30:
            break
        t_alloc = min(rem * 0.25, 60.0)
        a_interp = interpolate_seq(best_seq, n_alt)
        a_opt, s_opt, _ = precise_coord_descent(a_interp, t_alloc, label=f"len{n_alt}")
        best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"len_{n_alt}")

    print(f"  Done: best={best_score:.8f}")

    # === Phase 7: Final intensive polishing ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    a_work = best_seq.copy()

    # CD round 1
    rem = deadline - time.monotonic() - 80
    if rem > 20:
        t_alloc = min(rem * 0.5, 100.0)
        a_opt, s_opt, _ = precise_coord_descent(a_work, t_alloc, verbose=True, label="polish1")
        best_seq, best_score = check_and_update(a_opt, best_seq, best_score, "polish_cd1")
        a_work = best_seq.copy()

    # Mirror descent (short)
    rem = deadline - time.monotonic() - 60
    if rem > 15:
        t_alloc = min(rem * 0.4, 50.0)
        a_md, s_md = mirror_descent(a_work, t_alloc, label="polish_md")
        best_seq, best_score = check_and_update(a_md, best_seq, best_score, "polish_md")
        a_work = best_seq.copy()

    # CD round 2 (final)
    rem = deadline - time.monotonic() - 30
    if rem > 15:
        t_alloc = min(rem * 0.6, 40.0)
        a_opt, s_opt, _ = precise_coord_descent(a_work, t_alloc, label="polish_final")
        best_seq, best_score = check_and_update(a_opt, best_seq, best_score, "polish_cd2")

    # One more mirror descent pass
    rem = deadline - time.monotonic() - 15
    if rem > 10:
        t_alloc = min(rem * 0.6, 20.0)
        a_md, s_md = mirror_descent(a_work, t_alloc, label="last_md")
        best_seq, best_score = check_and_update(a_md, best_seq, best_score, "last_md")

    # Final CD
    rem = deadline - time.monotonic() - 5
    if rem > 8:
        t_alloc = min(rem * 0.7, 15.0)
        a_opt, s_opt, _ = precise_coord_descent(a_work, t_alloc, label="very_last")
        best_seq, best_score = check_and_update(a_opt, best_seq, best_score, "very_last")

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            if fs < best_score:
                best_score = fs
            print(f"  Verified: {best_score:.8f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
